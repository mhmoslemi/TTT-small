"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses MWUA (Multiplicative Weights Update Algorithm) for the zero-sum game interpretation,
combined with exact subgradient descent, LP best-response, and coordinate descent refinement.
Employs FFT-based convolution and proper simplex projection.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def compute_conv(a):
    """Compute autocorrelation conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return float(2.0 * n * max_b / (s * s))


def proj_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([1.0]) if v[0] > 0 else np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def subgradient_of_conv_k(a, k):
    """Compute the gradient of (conv(a,a))[k] w.r.t. a.
    Result: g[i] = 2*a[k-i] if 0 <= k-i < n, else 0."""
    n = len(a)
    g = np.zeros(n)
    i_min = max(0, k - n + 1)
    i_max = min(n - 1, k)
    for i in range(i_min, i_max + 1):
        g[i] = 2.0 * a[k - i]
    return g


def exact_subgradient_step(a, conv, step_size, top_m=3):
    """One step of exact subgradient descent on the simplex.
    Uses a weighted combination of top-m subgradients."""
    n = len(a)
    m = min(top_m, 2 * n - 1)
    # Get top m indices
    top_idx = np.argsort(conv)[::-1][:m]
    top_vals = conv[top_idx]
    
    # Softmax weights (temperature based on spread)
    spread = top_vals[0] - top_vals[-1]
    if spread > 1e-12:
        temp = spread * 0.5
        weights = np.exp((top_vals - top_vals[0]) / temp)
        weights /= weights.sum()
    else:
        weights = np.ones(m) / m
    
    # Weighted subgradient
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        grad += w * subgradient_of_conv_k(a, int(k))
    
    # Center for sum constraint
    grad_centered = grad - np.mean(grad)
    
    # Step and project
    a_new = a - step_size * grad_centered
    a_new = proj_simplex(a_new)
    return a_new


def smooth_subgradient_step(a, conv, step_size, tau=10.0):
    """Smoothed subgradient step using log-sum-exp weights."""
    n = len(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        # All mass on argmax
        k_star = int(np.argmax(conv))
        return exact_subgradient_step(a, conv, step_size, top_m=1)
    weights /= w_sum
    
    # Weighted subgradient (using FFT for efficiency)
    # grad[i] = sum_k weights[k] * 2 * a[k-i] (if valid)
    # This is 2 * correlate(weights, a, 'full')[i] ... let me just do it directly
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-15:
            continue
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        grad[i_min:i_max + 1] += weights[k] * 2.0 * a[i_max:i_min - 1:-1]
    
    grad_centered = grad - np.mean(grad)
    a_new = a - step_size * grad_centered
    a_new = proj_simplex(a_new)
    return a_new


def build_A_matrix(a):
    """Build the constraint matrix for (a*g)_k <= M for all k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def solve_lp_for_g(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns (g, sum_g) or (None, None) on failure."""
    n = len(a)
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = res.x
            sum_g = float(-res.fun)
            return g, sum_g
    except Exception:
        pass
    return None, None


def lp_optimize(a, max_iter=50, t=0.3, time_limit=60.0, verbose=False):
    """Alternating LP optimization."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iter):
        if time.monotonic() - start > time_limit:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g, sum_g = solve_lp_for_g(a, M)
        if g is None:
            break
        
        if sum_g <= 1.0 + 1e-4:
            stall += 1
            if stall > 30:
                break
            continue
        stall = 0
        
        g = g / sum_g
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose:
                print(f"      LP it={it}: obj={best_obj:.6f}")
        a = a_new
    
    return best_a, best_obj


def mwua_optimize(a, T=500, eta=0.5, qpw_iters=200, verbose=False):
    """MWUA optimization for the zero-sum game.
    min_a max_k (conv(a,a))_k  <->  game between (a, p) where p is over lags.
    
    For the minimizer (a): solve QP min a^T S a s.t. sum(a)=1, a>=0 where S_{ij}=p_{i+j}
    For the maximizer (p): multiplicative weights update based on conv(a,a).
    """
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    a = a / s
    
    # Initialize p uniformly
    num_lags = 2 * n - 1
    p = np.ones(num_lags) / num_lags
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for t in range(T):
        # --- Minimizer step: solve QP min a^T S a s.t. 1^T a = 1, a >= 0 ---
        # Compute S @ a where (S@a)_i = sum_j p_{i+j} * a_j
        Sa = np.zeros(n)
        for i in range(n):
            # Sa[i] = sum_{j=0}^{n-1} p[i+j] * a[j]
            # p is defined for indices 0 to 2n-2
            Sa[i] = np.dot(p[i:i+n], a)
        
        # Projected gradient on the QP
        # Gradient of a^T S a is 2 * S @ a (but S is not symmetric in general... wait)
        # S_{i,j} = p_{i+j}. Is S symmetric? S_{i,j} = p_{i+j} = p_{j+i} = S_{j,i}. Yes!
        # So the gradient is 2 * S @ a = 2 * Sa.
        
        for qpw_it in range(qpw_iters):
            grad = 2.0 * Sa  # gradient of a^T S a
            # Projected gradient step
            lr = 1.0 / (np.max(np.abs(grad)) + 1e-10)
            a_new = a - lr * (grad - np.mean(grad))
            a_new = proj_simplex(a_new)
            
            if np.max(np.abs(a_new - a)) < 1e-12:
                a = a_new
                break
            a = a_new
            # Recompute Sa
            for i in range(n):
                Sa[i] = np.dot(p[i:i+n], a)
        
        # --- Compute objective ---
        conv = compute_conv(a)
        obj = 2.0 * n * np.max(conv) / (np.sum(a) ** 2)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose and (t % 50 == 0):
                print(f"      MWUA t={t}: obj={best_obj:.6f}")
        
        # --- Maximizer step: multiplicative weights ---
        max_conv = np.max(conv)
        if max_conv < 1e-30:
            break
        # p_k *= exp(eta * conv_k / max_conv)
        p *= np.exp(eta * conv / max_conv)
        p /= np.sum(p)
        # Add small uniform to prevent p from becoming too sparse
        p = 0.95 * p + 0.05 / num_lags
    
    return best_a, best_obj


def coordinate_descent(a, max_rounds=100, time_limit=30.0, verbose=False):
    """Coordinate descent refinement: try changing pairs of coordinates."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    start = time.monotonic()
    
    for rnd in range(max_rounds):
        if time.monotonic() - start > time_limit:
            break
        improved = False
        conv = compute_conv(a)
        max_conv = np.max(conv)
        
        for i in range(n):
            for delta in [-0.1, -0.05, -0.02, 0.02, 0.05, 0.1]:
                if time.monotonic() - start > time_limit:
                    break
                # Move delta from a[i] to a[(i+1)%n] (or spread)
                a_try = a.copy()
                a_try[i] -= delta
                a_try[(i + 1) % n] += delta
                if np.min(a_try) < 0:
                    continue
                s_try = np.sum(a_try)
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                
                conv_try = compute_conv(a_try)
                if np.max(conv_try) < max_conv - 1e-10:
                    a = a_try
                    max_conv = np.max(conv_try)
                    improved = True
                    obj_try = compute_obj(a)
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a.copy()
        
        if not improved and rnd > 20:
            break
    
    return best_a, best_obj

                a_sg = exact_subgradient_step(a_sg, conv, step, top_m=3)
            else:
                a_sg = smooth_subgradient_step(a_sg, conv, step, tau=20.0)
            
            step *= 0.999
        
        # Phase 2: MWUA
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5 and n <= 300:
            T = int(remaining * 2)
            a_mw, _ = mwua_optimize(a_sg, T=T, eta=0.3, qpw_iters=100, verbose=False)
            obj_mw = compute_obj(a_mw)
            if obj_mw < best_obj:
                best_obj = obj_mw
                best_a = a_mw.copy()
                a_sg = a_mw.copy()
        
        # Phase 3: LP
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 350:
            a_lp, obj_lp = lp_optimize(a_sg, max_iter=30, t=0.3,
                                       time_limit=min(remaining * 0.5, 30.0))
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
                a_sg = a_lp.copy()
        
        # Phase 4: Coordinate descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 3 and n <= 200:
            a_cd, obj_cd = coordinate_descent(a_sg, max_rounds=30,
                                              time_limit=min(remaining, 10.0))
            if obj_cd < best_obj:
                best_obj = obj_cd
                best_a = a_cd.copy()
    
    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)
    
    return best_a, best_obj
