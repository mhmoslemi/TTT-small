```python
def project_simplex(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
"""
Optimized step function coefficient search combining LP-based alternating,
L-BFGS-B with augmented Lagrangian, simulated annealing with block moves,
and coordinate refinement. Targets objective below 1.5027.
Uses multiple initialization strategies and adaptive time management.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

# Global state for L-BFGS callback functions (to avoid lambdas/closures)
_g_T = 0.01
_g_mu = 100.0
_g_n = 100


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    """Normalize to sum=1, ensuring non-negativity."""
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def solve_lp_direction(a):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M, b >= 0. Returns b or None."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    if M < 1e-15:
        return None

    # Build constraint matrix: A[k][j] contributes a[k-j] * b[j] to (a*b)[k]
    # (a*b)[k] = sum_j a[k-j] * b[j] for valid k-j
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, 10.0)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, M
    return None, M


def lp_alternate(a, max_iters, time_limit, start_time=None):
    """LP alternating method with line search on the actual objective."""
    if start_time is None:
        start_time = time.monotonic()
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() - start_time > time_limit:
            break

        b, M = solve_lp_direction(a)
        if b is None:
            break

        b_sum = float(np.sum(b))
        if b_sum < 1.0 + 1e-6:
            stall_count += 1
            if stall_count > 12:
                break
            continue
        stall_count = 0

        b = b / b_sum

        # Line search: try multiple step sizes
        best_t_obj = best_obj
        best_t_a = None
        for t in [0.7, 0.5, 0.35, 0.2, 0.1, 0.8, 0.9, 0.15, 0.25, 0.4, 0.6]:
            a_new = (1.0 - t) * a + t * b
            a_new = np.clip(a_new, 0.0, None)
            s = np.sum(a_new)
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_t_obj:
                best_t_obj = obj
                best_t_a = a_new.copy()

        if best_t_a is not None and best_t_obj < best_obj:
            a = best_t_a
            best_a = a.copy()
            best_obj = best_t_obj
        else:
            # Small step in b direction anyway
            a = 0.95 * a + 0.05 * b
            a = normalize(a)
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    return best_a, best_obj


def _lbfgs_obj(x):
    """Objective for L-BFGS: soft-max surrogate + penalty for sum constraint."""
    n = _g_n
    x = np.clip(x, 0.0, None)
    conv = np.convolve(x, x)
    max_c = float(np.max(conv))
    shifted = np.clip((conv - max_c) / _g_T, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        sm_val = max_c
    else:
        sm_val = max_c + _g_T * np.log(sum_exp)
    obj = 2.0 * n * sm_val
    penalty = _g_mu * (np.sum(x) - 1.0) ** 2
    return obj + penalty


def _lbfgs_jac(x):
    """Gradient for L-BFGS."""
    n = _g_n
    x = np.clip(x, 0.0, None)
    conv = np.convolve(x, x)
    max_c = float(np.max(conv))
    shifted = np.clip((conv - max_c) / _g_T, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))

    if sum_exp < 1e-300:
        grad = 2.0 * n * x.copy()
    else:
        p = exp_vals / sum_exp
        # grad_sm[i] = (2/T) * sum_j p[i+j] * x[j]
        M = np.zeros((n, n))
        for i in range(n):
            M[i, :] = p[i:i + n]
        grad = 2.0 * n * (2.0 / _g_T) * (M @ x)

    grad += 2.0 * _g_mu * (np.sum(x) - 1.0) * np.ones(n)
    return grad


def lbfgs_refine(a, time_limit, T_start=0.01, T_end=0.002):
    """L-BFGS-B refinement with augmented Lagrangian."""
    global _g_T, _g_mu, _g_n
    n = len(a)
    _g_n = n
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T = T_start
    mu = 50.0

    for outer in range(30):
        if time.monotonic() - start > time_limit:
            break
        if T < T_end:
            T = T_end

        _g_T = T
        _g_mu = mu

        result = minimize(
            _lbfgs_obj, a, jac=_lbfgs_jac,
            method='L-BFGS-B',
            bounds=[(0.0, None)] * n,
            options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 40}
        )

        a_new = np.clip(result.x, 0.0, None)
        s = np.sum(a_new)
        if s > 1e-10:
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
            a = a_new
        else:
            break

        mu *= 2.5
        T *= 0.7

    return best_a, best_obj


def soft_max_gd(a, time_limit, T_init=0.005):
    """Projected gradient descent on soft-max with adaptive step size."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    T = T_init
    lr = 0.5
    stagnant = 0

    while time.monotonic() - start < time_limit:
        conv = np.convolve(a, a)
        max_c = float(np.max(conv))
        shifted = np.clip((conv - max_c) / T, -500.0, 500.0)
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        p = exp_vals / sum_exp

        # Gradient
        M = np.zeros((n, n))
        for i in range(n):
            M[i, :] = p[i:i + n]
        grad = (2.0 / T) * (M @ a)

        # Project gradient to tangent space of simplex
        grad = grad - np.mean(grad)

        a_new = a - lr * grad
        # Project to simplex
        a_new = np.clip(a_new, 0.0, None)
        s = np.sum(a_new)
        if s > 1e-10:
            a_new = a_new / s
        else:
            lr *= 0.5
            continue

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
            T = max(0.001, T * 0.995)
        else:
            lr *= 0.85
            stagnant += 1
            if stagnant > 100:
                break
            if lr < 1e-8:
                break

    return best_a, best_obj


def block_perturb_refine(a, time_limit, rng):

    n = len(a)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_limit:

        # Progressive narrowing

        candidate = best_a.copy()

        s = np.sum(candidate)
        if s < 1e-10:
            continue
        obj = compute_obj(candidate)
        if obj < best_obj:

            best_obj = obj

def water_filling(a, n_iters=200, beta=5.0):
    """Multiplicative water-filling update: reduce a[i] where it contributes to hot spots."""
    n = len(a)
    a = normalize(a.copy())

    for _ in range(n_iters):
        conv = np.convolve(a, a)
        # For each position i, compute its "energy contribution"
        # energy[i] = sum_j a[j] * conv[i+j]
        # Use FFT for efficiency
        L = 1
        while L < 3 * n:
            L *= 2
        A = np.fft.rfft(a, L)
        C = np.fft.rfft(conv, L)
        # energy[i] = sum_j a[j] * conv[i+j] = cross-correlation
        # = IFFT(conj(FFT(a)) * FFT(conv)) ... let's verify
        # Actually: sum_j a[j] * conv[i+j]
        # Let's define d[m] = conv[m] (length 2n-1)
        # energy[i] = sum_j a[j] * d[i+j]
        # This is the cross-correlation of a and d at lag i
        # = IFFT(conj(FFT(a_padded)) * FFT(d_padded))[i]
        pad_a = np.zeros(L)
        pad_a[:n] = a
        pad_d = np.zeros(L)
        pad_d[:2*n-1] = conv
        corr = np.fft.irfft(np.conj(np.fft.rfft(pad_a, L)) * np.fft.rfft(pad_d, L))
        energy = corr[:n]

        # Multiplicative update
        a *= np.exp(-beta * energy / (np.max(energy) + 1e-10))
        a = np.clip(a, 1e-12, None)
        a = a / np.sum(a)

    return a


def generate_init(n, strategy, rng):
    """Generate initial sequence using various strategies."""
    if strategy == 0:
        # Uniform
        a = np.ones(n)
    elif strategy == 1:
        # Gaussian center
        center = n / 2.0
        width = n * rng.uniform(0.2, 0.5)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        a += 0.1
    elif strategy == 2:
        # Edge-weighted
        a = np.ones(n)
        a[0] = 2.0
        a[-1] = 2.0
        a[1] = 1.5
        a[-2] = 1.5
        if n > 4:
            a[2] = 1.2
            a[-3] = 1.2
    elif strategy == 3:
        # Periodic
        period = max(2, int(rng.integers(3, max(4, n // 20))))
        pattern = np.abs(np.sin(np.pi * np.arange(period) / period)) + 0.3
        a = np.tile(pattern, n // period + 1)[:n]
        a += 0.05 * rng.random(n)
    elif strategy == 4:
        # Two-block
        a = np.zeros(n)
        block_len = max(3, n // 4)
        a[:block_len] = 1.0
        a[-block_len:] = 1.0
        a += 0.1 * rng.random(n)
    elif strategy == 5:
        # Random positive
        a = np.abs(rng.normal(1.0, 0.5, n))
    elif strategy == 6:
        # Cosine with random phase
        phase = rng.uniform(0, 2 * np.pi)
        freq = rng.uniform(0.1, 0.8)
        a = np.maximum(0, np.cos(2 * np.pi * freq * np.arange(n) / n + phase)) + 0.1
    elif strategy == 7:
        # Log-uniform random
        a = 10.0 ** rng.uniform(-1, 1, n)
    else:
        a = np.abs(rng.standard_normal(n)) + 0.1

    return normalize(a)


def optimize_from_start(a_init, time_budget, rng):

    n = len(a_init)

        a2, obj2 = lbfgs_refine(a, lbfgs_time)
        if obj2 < best_obj:

            best_a = a2.copy()
            a = a2

    # Phase 3: LP again

        a3, obj3 = lp_alternate(a, max(5, int(lp_time2 / 0.1)), lp_time2)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            a = a3

    # Phase 4: Soft-max GD
    if time.monotonic() - start < time_budget * 0.9:

        a4, obj4 = soft_max_gd(a, gd_time)
        if obj4 < best_obj:

            best_a = a4.copy()
            a = a4

    # Phase 5: Block perturbation

        if bp_time > 2:
            a5, obj5 = block_perturb_refine(a, bp_time, rng)
            if obj5 < best_obj:
                best_obj = obj5
                best_a = a5.copy()

def simulated_annealing(a_init, time_limit, rng):
    """Simulated annealing with block scaling moves."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    T_sa = 0.05  # annealing temperature
    cooling = 0.9995

    while time.monotonic() - start < time_limit:
        width = int(rng.integers(1, max(2, n // 5)))
        pos = int(rng.integers(0, n - width + 1))

        candidate = a.copy()
        sigma = 0.05 + 0.1 * T_sa / 0.05
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate = candidate / s

        obj = compute_obj(candidate)
        delta = obj - compute_obj(a)

        if delta < 0 or rng.random() < np.exp(-delta / max(T_sa, 1e-6)):
            a = candidate
            if obj < best_obj:
                best_obj = obj
                best_a = candidate.copy()

        T_sa *= cooling
        if T_sa < 0.001:
            T_sa = 0.001

    return best_a, best_obj

            else:
                # Mirror扰动 - mix with reversed version
                t = rng.uniform(0.2, 0.8)
                candidate = t * candidate + (1 - t) * candidate[::-1]

            candidate = np.clip(candidate, 0.0, None)
            s = np.sum(candidate)
            if s < 1e-10:
                continue
            candidate = candidate / s

            a_opt, obj_opt = optimize_from_start(candidate, time_per_perturb, rng)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Perturb {pi}: obj={obj_opt:.6f} ** NEW BEST **", flush=True)
            else:
                print(f"  Perturb {pi}: obj={obj_opt:.6f}", flush=True)

    # Phase C: Fresh starts at different n values
    print(f"\nPhase C: Fresh starts (remaining={deadline - time.monotonic():.0f}s)", flush=True)
    n_values = [50, 80, 120, 150, 200, 250, 300, 350, 400, 500]
    # Filter to those with enough time
    remaining = deadline - time.monotonic()

    # Allocate time proportional to number of n values remaining
    for ni, n in enumerate(n_values):
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        num_remaining = len(n_values[ni:])
        if num_remaining == 0:
            break
        time_per_n = min(remaining / num_remaining, 120.0)
        if time_per_n < 10:
            break

        # Number of restarts for this n
        num_restarts = max(1, min(4, int(time_per_n / 35)))
        time_per_restart = time_per_n / num_restarts

        print(f"  n={n}: {num_restarts} restarts, ~{time_per_restart:.0f}s each", flush=True)

        for r in range(num_restarts):
            if time.monotonic() > deadline - 30:
                break
            strategy = int(rng.integers(0, 8))
            a_init = generate_init(n, strategy, rng)
            a_opt, obj_opt = optimize_from_start(a_init, time_per_restart, rng)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"    n={n} r={r} strat={strategy}: obj={obj_opt:.6f} ** NEW BEST **", flush=True)
            else:
                print(f"    n={n} r={r} strat={strategy}: obj={obj_opt:.6f}", flush=True)

    # Phase D: Final intensive refinement on best
    print(f"\nPhase D: Final refinement (remaining={deadline - time.monotonic():.0f}s)", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 5.0

        # LP + L-BFGS cycle
        if remaining > 20:
            t1 = min(remaining * 0.3, 60.0)
            a_f, obj_f = lp_alternate(best_seq.copy(), max(20, int(t1 / 0.05)), t1)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final LP: obj={best_score:.6f}", flush=True)

        if time.monotonic() < deadline - 10:
            t2 = min((deadline - time.monotonic() - 8) * 0.4, 40.0)
            a_f2, obj_f2 = lbfgs_refine(best_seq.copy(), t2, T_start=0.003, T_end=0.0005)
            if obj_f2 < best_score:
                best_score = obj_f2
                best_seq = a_f2.copy()
                print(f"  Final L-BFGS: obj={best_score:.6f}", flush=True)

        # Block refinement
        if time.monotonic() < deadline - 5:
            t3 = deadline - time.monotonic() - 4.0
            a_f3, obj_f3 = block_perturb_refine(best_seq.copy(), t3, rng)
            if obj_f3 < best_score:
                best_score = obj_f3
                best_seq = a_f3.copy()
                print(f"  Final blocks: obj={best_score:.6f}", flush=True)

        # Coordinate refinement
        if time.monotonic() < deadline - 3:
            t4 = deadline - time.monotonic() - 3.0
            a_f4, obj_f4 = coordinate_refine(best_seq.copy(), t4, rng)
            if obj_f4 < best_score:
                best_score = obj_f4
                best_seq = a_f4.copy()
                print(f"  Final coords: obj={best_score:.6f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s", flush=True)

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
```
