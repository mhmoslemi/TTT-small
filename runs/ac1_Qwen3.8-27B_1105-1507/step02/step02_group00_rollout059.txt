"""
Optimized step-function search combining LP alternating, subgradient polishing,
iterated local search with perturbations, and coordinate refinement.
Targets score below 1.5027 by exploiting structured initializations and
aggressive multi-phase local optimization.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build convolution constraint matrix A[k,j] = a[k-j] if valid else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_g(a, M_relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M*M_relax, g >= 0."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv)) * M_relax
    if M < 1e-15:
        return None
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_alternate(a_init, max_iter=200, relax=1.0, blend=0.5, ema_decay=0.6):
    """LP alternating optimization with EMA tracking."""
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    ema = a.copy()
    n = len(a)

    for it in range(max_iter):
        g = solve_lp_g(a, M_relax=relax)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum

        ema = ema_decay * ema + (1 - ema_decay) * g
        obj_g = compute_obj(g)
        obj_ema = compute_obj(ema)

        if obj_g < best_obj:
            best_a = g.copy()
            best_obj = obj_g
        if obj_ema < best_obj:
            best_a = ema.copy()
            best_obj = obj_ema

        t = blend
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            a = g.copy()
        else:
            a = a_new / s

    return best_a, best_obj


def subgradient_polish(a, n_steps=3000):
    """Subgradient method with active set averaging."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for step in range(n_steps):
        h = np.convolve(a, a)
        max_h = np.max(h)
        if max_h < 1e-15:
            break

        # Adaptive tolerance
        tol = max_h * (0.0005 + 0.02 * (1 - step / n_steps))
        active = h >= max_h - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            active = h >= max_h - 1e-12
            n_active = max(1, int(np.sum(active)))

        # Compute subgradient
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            vals = a[k - j_end:k - j_start + 1][::-1]
            grad[j_start:j_end + 1] += 2.0 * vals
            if k % 2 == 0 and 0 <= k // 2 < n:
                grad[k // 2] -= a[k // 2]
        grad /= n_active

        # Projected gradient step
        step_size = 1.0 / (1.0 + 0.005 * step)
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_obj


def coord_refine(a, n_rounds=5):
    """Coordinate descent with fine grid search per coordinate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for rnd in range(n_rounds):
        improved = False
        for j in range(n):
            old_val = a[j]
            best_x = old_val
            best_local = best_obj
            # Try multipliers around current value
            multipliers = [0.0, 0.1, 0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5]
            for mult in multipliers:
                x = old_val * mult
                if x < 0:
                    x = 0.0
                a[j] = x
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_x = x
            a[j] = best_x
            if best_x != old_val:
                improved = True

        current_obj = compute_obj(a)
        if current_obj < best_obj:
            best_obj = current_obj
            best_a = a.copy()

    return best_a, best_obj


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 8))
    if choice == 0:
        # Two-level step: [p]*k + [q]*(n-k)
        k = int(rng.integers(1, n))
        p = 1.0
        q = rng.uniform(0.1, 0.8)
        a = np.zeros(n)
        a[:k] = p
        a[k:] = q
    elif choice == 1:
        # Three-level step
        k1 = int(rng.integers(1, n // 2))
        k2 = int(rng.integers(1, n - k1))
        p = 1.0
        r = rng.uniform(0.2, 0.9)
        q = rng.uniform(0.05, 0.5)
        a = np.zeros(n)
        a[:k1] = p
        a[k1:k1 + k2] = r
        a[k1 + k2:] = q
    elif choice == 2:
        # Smooth ramp (high to low)
        alpha = rng.uniform(0.5, 2.0)
        a = np.exp(-alpha * np.arange(n) / max(1, n - 1))
    elif choice == 3:
        # Cosine window variant
        freq = rng.uniform(0.3, 1.5)
        phase = rng.uniform(0, np.pi)
        a = np.maximum(0, np.cos(np.pi * freq * np.arange(n) / n + phase))
    elif choice == 4:
        # Periodic pattern
        period = int(rng.integers(max(2, n // 20), max(3, n // 3)))
        pattern = rng.random(period)
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 5:
        # Power law decay
        gamma = rng.uniform(0.5, 2.0)
        a = np.power(np.arange(1, n + 1), -gamma)
    elif choice == 6:
        # Bimodal (high at both ends)
        center = n // 2
        width = int(rng.integers(n // 4, n // 2))
        a = np.ones(n)
        a[center - width:center + width] = rng.uniform(0.1, 0.5)
    else:
        # Pure random
        a = rng.random(n)

    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s


def perturb(a, rng, scale=0.1):
    """Randomly perturb a sequence."""
    a_pert = a + rng.normal(0, scale, size=len(a))
    a_pert = np.maximum(a_pert, 0)
    s = np.sum(a_pert)
    if s < 1e-10:
        return a.copy()
    return a_pert / s


def slsqp_polish(a, max_iter=500):

    a0 = a.copy().astype(np.float64)
    s = np.sum(a0)
    a0 = a0 / s

    def objective(x):
        x = np.maximum(x, 0)
        sx = np.sum(x)
        if sx < 0.01:
            return 1e10
        conv = np.convolve(x, x)
        return 2.0 * n * np.max(conv) / (sx * sx)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 1.0}]
    bounds = [(0.0, None)] * n

    try:
        res = minimize(objective, a0, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': max_iter,
                                                         'ftol': 1e-12})
        if res.success or res.fun < compute_obj(a0):
            a_res = np.maximum(res.x, 0)
            s_res = np.sum(a_res)
            if s_res > 0.01:
                return a_res / s_res
    except Exception:
        pass
    return a0


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: LP alternation + subgradient + iterated local search + coordinate refine.
    Tries multiple n values with structured initializations.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # N values to try - focus on ranges likely to give good results
    n_values = [10, 15, 20, 30, 40, 50, 60, 75, 100, 120, 150, 200, 250, 300, 350, 400, 500]

    # Phase 1: LP optimization with multiple restarts per n
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        # Estimate time per LP iteration for this n
        if n <= 100:
            n_restarts = 4
            lp_iters = 300
        elif n <= 200:
            n_restarts = 3
            lp_iters = 200
        elif n <= 300:
            n_restarts = 3
            lp_iters = 150
        else:
            n_restarts = 2
            lp_iters = 100

        print(f"N={n}: {remaining:.0f}s remaining, best={best_score:.6f}")

        for restart in range(n_restarts):
            if time.monotonic() > deadline:
                break

            if restart == 0 and best_seq is not None and len(best_seq) == n:
                a_init = best_seq.copy()
            else:
                a_init = structured_init(n, rng)

            a, obj = lp_alternate(a_init, max_iter=lp_iters, relax=1.0,
                                  blend=0.5, ema_decay=0.6)

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  LP n={n} r={restart}: obj={obj:.6f}")

            # Subgradient polish from LP result
            if time.monotonic() < deadline - 30:
                ns = int(min(2000, (deadline - time.monotonic()) / 0.01))
                if ns > 200:
                    a2, obj2 = subgradient_polish(a, n_steps=ns)
                    if obj2 < best_score:
                        best_score = obj2
                        best_seq = a2.copy()
                        print(f"  SG n={n} r={restart}: obj={obj2:.6f}")

    # Phase 2: Iterated local search on the best solution
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\nPhase 2: ILS on n={n_best}, obj={best_score:.6f}, {remaining:.0f}s left")

        # Multiple perturbation scales
        scales = [0.01, 0.02, 0.05, 0.1, 0.2]
        n_perturbations = int(min(50, remaining / 2.0))

        for p in range(n_perturbations):
            if time.monotonic() > deadline:
                break
            scale = scales[p % len(scales)]
            a_pert = perturb(best_seq, rng, scale=scale)
            # Short LP from perturbed point
            a_p, obj_p = lp_alternate(a_pert, max_iter=50, relax=1.0, blend=0.4)
            if obj_p < best_score:
                best_score = obj_p
                best_seq = a_p.copy()
                print(f"  ILS p={p} (scale={scale}): obj={obj:.6f}")

    # Phase 3: Final polishing
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\nPhase 3: Final polish n={n_best}, obj={best_score:.6f}, {remaining:.0f}s left")

        # Extended subgradient
        ns = int(min(5000, remaining / 0.01))
        if ns > 500:
            a3, obj3 = subgradient_polish(best_seq, n_steps=ns)
            if obj3 < best_score:
                best_score = obj3
                best_seq = a3.copy()
                print(f"  Final SG: obj={obj3:.6f}")

        # SLSQP polish (only for smaller n where it's tractable)
        if n_best <= 150 and time.monotonic() < deadline - 30:
            a4 = slsqp_polish(best_seq, max_iter=300)
            obj4 = compute_obj(a4)
            if obj4 < best_score:
                best_score = obj4
                best_seq = a4.copy()
                print(f"  SLSQP: obj={obj4:.6f}")

        # Coordinate refinement
        if n_best <= 200 and time.monotonic() < deadline - 15:
            rounds = int(min(5, (deadline - time.monotonic()) / 3.0))
            if rounds > 0:
                a5, obj5 = coord_refine(best_seq, n_rounds=rounds)
                if obj5 < best_score:
                    best_score = obj5
                    best_seq = a5.copy()
                    print(f"  Final Coord: obj={obj5:.6f}")

        # One more LP pass
        if n_best <= 300 and time.monotonic() < deadline - 20:
            iter_left = int(min(150, (deadline - time.monotonic()) / 0.5))
            if iter_left > 30:
                a6, obj6 = lp_alternate(best_seq, max_iter=iter_left, relax=1.0)
                if obj6 < best_score:
                    best_score = obj6
                    best_seq = a6.copy()
                    print(f"  Final LP: obj={obj6:.6f}")

                # Final subgradient
                if time.monotonic() < deadline - 5:
                    a7, obj7 = subgradient_polish(best_seq, n_steps=2000)
                    if obj7 < best_score:
                        best_score = obj7
                        best_seq = a7.copy()
                        print(f"  Very Final SG: obj={obj7:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
