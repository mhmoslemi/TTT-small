"""
Improved multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: adaptive LP blending, Nesterov-accelerated subgradient,
signal-processing-inspired initializations, and multi-scale refinement.

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A

def solve_lp_g(a, M):
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_alternate(a_init, deadline, max_iter=500):
    """LP alternation with adaptive blending."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        g = solve_lp_g(a, M)
        if g is None:
            break
        
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Adaptive blending
        t = 0.4 * (1.0 - 0.5 * it / max_iter) + 0.1
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            a = g
        else:
            a = a_new / s
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj

    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    # Momentum
    v = np.zeros(n)
    mu = 0.9  # momentum coefficient
    
    for step in range(n_steps):

        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_maxsum(a, M):
    """Standard LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def solve_lp_flat(a):
    """LP: minimize (t_max - t_min) s.t. (a*g)_k <= t_max, (a*g)_k >= t_min, sum(g)=1, g>=0."""
    n = len(a)
    A = build_A(a)
    n_vars = n + 2
    c = np.zeros(n_vars)
    c[-2] = 1.0
    c[-1] = -1.0

    A1 = np.zeros((2 * n - 1, n_vars))
    A1[:, :n] = A
    A1[:, -2] = -1.0
    b1 = np.zeros(2 * n - 1)

    A2 = np.zeros((2 * n - 1, n_vars))
    A2[:, :n] = -A
    A2[:, -1] = 1.0
    b2 = np.zeros(2 * n - 1)

    A_ub = np.vstack([A1, A2])
    b_ub = np.concatenate([b1, b2])

    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(-np.inf, np.inf)] * 2

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n]
    return None


def solve_lp_hybrid(a, M, alpha=0.5):
    """Hybrid LP: minimize (-sum(g) + alpha*(t_max - t_min)) s.t. (a*g)_k <= t_max, (a*g)_k >= t_min, (a*g)_k <= M, sum(g)=1, g>=0."""
    n = len(a)
    A = build_A(a)
    n_vars = n + 2
    c = np.zeros(n_vars)
    c[:n] = -1.0
    c[-2] = alpha
    c[-1] = -alpha

    A1 = np.zeros((2 * n - 1, n_vars))
    A1[:, :n] = A
    A1[:, -2] = -1.0
    b1 = np.zeros(2 * n - 1)

    A2 = np.zeros((2 * n - 1, n_vars))
    A2[:, :n] = -A
    A2[:, -1] = 1.0
    b2 = np.zeros(2 * n - 1)

    A3 = A.copy()
    b3 = np.full(2 * n - 1, M)

    A_ub = np.vstack([A1, A2, A3])
    b_ub = np.concatenate([b1, b2, b3])

    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(-np.inf, np.inf)] * 2

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n]
    return None


def lp_alternate_improved(a_init, deadline, use_flat=True, use_hybrid=True):
    """LP alternation with flatness-targeted LP and adaptive blending."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    no_improve_count = 0
    max_no_improve = 50

    it = 0
    total_iters = 2000

    while it < total_iters:
        if time.monotonic() > deadline:
            break

        conv = np.convolve(a, a)
        M = float(np.max(conv))

        # Choose which LP to use
        if use_flat and (it % 3 == 0 or not use_hybrid):
            g = solve_lp_flat(a)
        elif use_hybrid:
            g = solve_lp_hybrid(a, M, alpha=0.3)
        else:
            g = solve_lp_maxsum(a, M)

        if g is None:
            # Fallback to max-sum
            g = solve_lp_maxsum(a, M)
            if g is None:
                break

        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive blending
        progress = min(1.0, it / 200.0)
        t = 0.5 - 0.3 * progress
        t = max(0.1, t)

        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            a = g.copy()
        else:
            a = a_new / s

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count > max_no_improve:
            # Perturb and continue
            noise = np.random.default_rng(it).normal(0, 0.01, n)
            a = a + noise
            a = np.maximum(a, 0.0)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            no_improve_count = 0

        it += 1

    return best_a, best_obj


def subgradient_momentum(a_init, deadline, n_steps=10000):
    """Subgradient method with Nesterov momentum for polishing."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a_init, compute_obj(a_init)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    v = np.zeros(n)
    mu = 0.85
    step = 0

    # Yaushev-type step size
    initial_step = 1.0
    decay = 0.0005

    while step < n_steps:
        if time.monotonic() > deadline:
            break

        h = np.convolve(a, a)
        max_h = float(np.max(h))

        if step % 500 == 0:
            curr_obj = compute_obj(a)
            if curr_obj < best_obj:
                best_obj = curr_obj
                best_a = a.copy()

        # Active set: top positions
        tol = max_h * (0.02 + 0.03 * step / n_steps)
        active = h >= max_h - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            active = np.zeros(len(h), dtype=bool)
            active[int(np.argmax(h))] = True
            n_active = 1

        # Compute subgradient using convolution
        # grad_i = (2/n_active) * sum_{k in active} a[k-i] for valid k-i
        # This is (2/n_active) * (active_indicator convolve a_reflected)_i
        active_arr = active.astype(np.float64)
        # Cross-correlation: sum_k active_k * a[k-i] = (active * a_reflected) via convolution
        # a_reflected[j] = a[n-1-j]
        a_ref = a[::-1]
        # conv(active_arr, a_ref)[j] = sum_k active_arr[k] * a_ref[j-k]
        # We want sum_k active_arr[k] * a[k-i]
        # a[k-i] = a_ref[n-1-(k-i)] = a_ref[n-1-k+i]
        # So sum_k active_arr[k] * a_ref[n-1-k+i]
        # Let m = n-1-i, then this is sum_k active_arr[k] * a_ref[m+k-i+1-1+1]... 
        # This is getting confusing. Let me just do it directly.
        
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    grad[j] += 2.0 * a[idx]
        grad /= n_active

        # Nesterov step
        step_size = initial_step / (1.0 + decay * step)
        y = a - mu * v
        v = y - a + step_size * grad
        a = y - step_size * grad

        # Project onto simplex
        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s < 1e-10:
            break
        a = a / s

        step += 1

    # Final check
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a, best_obj


def coord_descent(a_init, deadline, n_rounds=5):
    """1D coordinate descent: optimize each coordinate via ternary search."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a_init, compute_obj(a_init)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        for j in range(n):
            if time.monotonic() > deadline:
                break

            old_val = a[j]
            # Search in [0, 3*old_val] or [0, 3/max(a)] 
            lo = 0.0
            hi = max(3.0 * old_val, 3.0 / n * 1.5)
            if old_val < 1e-10:
                hi = 3.0 / n

            # Ternary search
            for _ in range(20):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                a[j] = m1
                a[j] = max(0.0, m1)
                s_tmp = np.sum(a)
                if s_tmp < 1e-10:
                    f1 = np.inf
                else:
                    f1 = compute_obj(a / s_tmp)
                a[j] = m2
                a[j] = max(0.0, m2)
                s_tmp = np.sum(a)
                if s_tmp < 1e-10:
                    f2 = np.inf
                else:
                    f2 = compute_obj(a / s_tmp)

                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

            a[j] = (lo + hi) / 2.0
            a[j] = max(0.0, a[j])
            s = np.sum(a)
            if s > 1e-10:
                a = a / s

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def random_init(n, rng):
    """Generate a random non-negative sequence normalized to sum 1."""
    choice = int(rng.integers(0, 5))
    if choice == 0:
        a = rng.random(n)
    elif choice == 1:
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0, np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
    elif choice == 2:
        center = n // 2
        width = rng.uniform(n * 0.2, n * 0.5)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
    elif choice == 3:
        period = int(rng.integers(max(2, n // 20), max(3, n // 3)))
        pattern = rng.random(period)
        a = np.tile(pattern, n // period + 1)[:n]
    else:
        a = np.exp(-rng.uniform(0.5, 3.0) * np.abs(np.arange(n) / n - 0.5) * 2)

    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: flatness LP alternation + momentum subgradient + coordinate descent.
    Tries multiple sequence lengths and initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf
    rng = np.random.default_rng(seed)

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            s = np.sum(parent)
            if s > 0.01:
                obj = compute_obj(parent)
                if obj < best_score:
                    best_score = obj
                    best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Parent error: {e}")

    # If no parent, create initial
    if best_seq is None:
        best_seq = random_init(300, rng)
        best_score = compute_obj(best_seq)
        print(f"Initial (n=300): obj={best_score:.6f}")

    # Phase 1: LP alternation on parent length and nearby
    n_values = [len(best_seq)]
    # Add a few other lengths to explore
    parent_n = len(best_seq)
    for candidate_n in [200, 250, 300, 350, 400]:
        if candidate_n != parent_n and candidate_n not in n_values:
            n_values.append(candidate_n)
    n_values = sorted(n_values)

    time_per_n = (deadline - time.monotonic()) / len(n_values)
    print(f"Will try n values: {n_values}, ~{time_per_n:.0f}s each")

    for n in n_values:
        if time.monotonic() > deadline:
            break

        # Initialize for this n
        if n == len(best_seq) and best_seq is not None:
            a_init = best_seq.copy()
        else:
            a_init = random_init(n, rng)

        # LP alternation
        a, obj = lp_alternate_improved(a_init, deadline)
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  LP n={n}: obj={obj:.6f} (improved!)")
        else:
            print(f"  LP n={n}: obj={obj:.6f} (best so far: {best_score:.6f})")

        # Subgradient polish
        if time.monotonic() < deadline - 30:
            a2, obj2 = subgradient_momentum(a, deadline, n_steps=8000)
            if obj2 < best_score:
                best_score = obj2
                best_seq = a2.copy()
                print(f"  SG n={n}: obj={obj2:.6f} (improved!)")

    # Phase 2: Final refinement on best solution
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\nPhase 2: Final refinement, n={len(best_seq)}, score={best_score:.6f}")

        # More subgradient
        a3, obj3 = subgradient_momentum(best_seq, deadline, n_steps=10000)
        if obj3 < best_score:
            best_score = obj3
            best_seq = a3.copy()
            print(f"  Final SG: obj={obj3:.6f}")

        # Coordinate descent
        if time.monotonic() < deadline - 20 and len(best_seq) <= 400:
            a4, obj4 = coord_descent(best_seq, deadline, n_rounds=3)
            if obj4 < best_score:
                best_score = obj4
                best_seq = a4.copy()
                print(f"  Coord: obj={obj4:.6f}")

        # One more LP pass
        if time.monotonic() < deadline - 30:
            a5, obj5 = lp_alternate_improved(best_seq, deadline, use_flat=True, use_hybrid=True)
            if obj5 < best_score:
                best_score = obj5
                best_seq = a5.copy()
                print(f"  Final LP: obj={obj5:.6f}")

        # Final subgradient
        if time.monotonic() < deadline - 5:
            a6, obj6 = subgradient_momentum(best_seq, deadline, n_steps=5000)
            if obj6 < best_score:
                best_score = obj6
                best_seq = a6.copy()
                print(f"  Final SG2: obj={obj6:.6f}")

    # Ensure valid output
    if best_seq is None:
        best_seq = np.ones(300) / 300.0
        best_score = compute_obj(best_seq)

    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
