def projective_subgradient(f, deadline, max_iters=10000, eta_init=0.1):

    """
    n = len(f)
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

            no_improve += 1

                eta *= 0.7
                no_improve = 0

f[k_star - lo : k_star - hi + 1 : -1] gives f[k_star-lo], f[k_star-lo-1], ..., f[k_star-hi]
And we assign this to grad[lo], grad[lo+1], ..., grad[hi].

grad[lo:hi+1] = 2*f[k_star-lo : k_star-hi-1 : -1]  # This has (hi-lo+1) elements? 

k_star-lo, k_star-lo-1, ..., k_star-hi: that's (k_star-lo) - (k_star-hi) + 1 = hi - lo + 1 elements. ✓

So: grad[lo:hi+1] = 2.0 * f[k_star - lo : k_star - hi - 1 : -1]

Let me verify with an example: f = [1, 2, 3, 4], n=4, k_star = 2.
C[2] = f[0]*f[2] + f[1]*f[1] + f[2]*f[0] = 3 + 4 + 3 = 10.
dC[2]/df[0] = f[2] = 3
dC[2]/df[1] = f[1] = 2
dC[2]/df[2] = f[0] = 1
dC[2]/df[3] = f[-1] = 0 (invalid)

So grad = [6, 4, 2, 0].

Using my formula: lo = max(0, 2-3) = 0, hi = min(2, 3) = 2.
grad[0:3] = 2 * f[2 : 2-2-1 : -1] = 2 * f[2 : -1 : -1] = 2 * [f[2], f[1], f[0]] = 2 * [3, 2, 1] = [6, 4, 2]. ✓

def projective_subgradient_toppk(f, deadline, max_iters=10000, eta_init=0.1, k_top=5):
    n = len(f)
    f = f / np.sum(f)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    eta = eta_init
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        C = np.convolve(f, f)
        M = C.max()
        # Find top-k positions
        threshold = M - 1e-6 * M  # positions within 0.1% of max
        top_positions = np.where(C >= threshold)[0]
        if len(top_positions) > k_top:
            top_positions = np.argsort(C)[::-1][:k_top]
        
        # Average subgradient
        grad = np.zeros(n)
        for k_star in top_positions:
            lo = max(0, k_star - n + 1)
            hi = min(k_star, n - 1)
            grad[lo:hi+1] += f[k_star - lo : k_star - hi - 1 : -1]
        grad *= 2.0 / len(top_positions)
        
        f_new = f - eta * grad
        f_new = np.clip(f_new, 0.0, None)
        S_new = float(np.sum(f_new))
        if S_new < 1e-10:
            break
        f_new = f_new / S_new
        
        sc = evaluate_sequence(f_new.tolist())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                eta *= 0.8
                no_improve = 0
                if eta < 1e-10:
                    break
    
    return best_f, best_score

```python
"""
Multi-method optimization combining Gerchberg-Saxton spectral design,
projective subgradient, and soft-max gradient descent for minimizing
2n*max(conv(f,f))/sum(f)^2 over non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def next_pow2(n):
    return 1 << (n - 1).bit_length()


def build_conv_matrix_sparse(f):
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((np.array(vals), (np.array(rows, dtype=np.int32), 
                          np.array(cols, dtype=np.int32))), shape=(2*n-1, n))


def solve_lp_direction(f, M, A_cache=None):
    n = len(f)
    if A_cache is None:
        A = build_conv_matrix_sparse(f)
    else:
        A = A_cache
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def gs_algorithm(f, deadline, n_fft=None, target_mags=None, max_iters=5000):
    """
    Gerchberg-Saxton algorithm for flat autocorrelation design.
    """
    n = len(f)
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, np.inf
    f = f / S
    
    if n_fft is None:
        n_fft = next_pow2(2 * n - 1)
    
    if target_mags is None:
        target_mags = [1.0 / np.sqrt(n_fft - 1)]
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    for target_mag in target_mags:
        f_curr = best_f.copy()
        for it in range(max_iters):
            if time.monotonic() > deadline:
                break
            
            F = np.fft.fft(f_curr, n_fft)
            
            # Enforce target magnitude
            phases = np.angle(F)
            abs_F = np.abs(F)
            F_new = np.zeros(n_fft, dtype=np.complex128)
            F_new[0] = F[0].real  # DC is real and positive
            mask = np.abs(F[1:]) > 1e-12
            F_new[1:][mask] = target_mag * np.exp(1j * phases[1:][mask])
            # For zero-magnitude bins, keep a random phase
            if not mask.all():
                rand_phases = np.random.uniform(0, 2 * np.pi, size=mask.sum() if not mask.all() else 0)
                # Actually, let's just use the existing phases (which are arbitrary for zeros)
                F_new[1:][~mask] = target_mag * np.exp(1j * phases[1:][~mask])
            
            f_new = np.fft.ifft(F_new).real[:n]
            f_new = np.clip(f_new, 0.0, None)
            S_new = float(np.sum(f_new))
            if S_new < 1e-10:
                break
            f_new = f_new / S_new
            
            sc = evaluate_sequence(f_new.tolist())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = f_new.copy()
            f_curr = f_new
    
    return best_f, best_score


def projective_subgradient(f, deadline, max_iters=20000, eta_init=0.05, k_top=3):
    """
    Projective subgradient with top-k max positions.
    """
    n = len(f)
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, np.inf
    f = f / S
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    eta = eta_init
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        C = np.convolve(f, f)
        M = C.max()
        
        # Find top-k positions
        sorted_idx = np.argsort(C)[::-1]
        top_k = sorted_idx[:k_top]
        
        # Check if they're all close to max
        if C[top_k[-1]] < M * 0.95:
            top_k = sorted_idx[:1]  # Only the actual max
        
        grad = np.zeros(n)
        for k_star in top_k:
            lo = max(0, k_star - n + 1)
            hi = min(k_star, n - 1)
            grad[lo:hi+1] += f[k_star - lo : k_star - hi - 1 : -1]
        grad *= 2.0 / len(top_k)
        
        f_new = f - eta * grad
        f_new = np.clip(f_new, 0.0, None)
        S_new = float(np.sum(f_new))
        if S_new < 1e-10:
            eta *= 0.5
            continue
        f_new = f_new / S_new
        
        it += 1
        sc = evaluate_sequence(f_new.tolist())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                eta *= 0.7
                no_improve = 0
                if eta < 1e-10:
                    break
    
    return best_f, best_score


def softmax_gradient_optimize(f, deadline, beta_start=50.0, beta_end=500.0,
                               lr_start=0.02, max_iters=5000):
    """Soft-max gradient descent with Adam."""
    n = len(f)
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, np.inf
    f = f / S
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    lr = lr_start
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        C = np.convolve(f, f)
        max_C = C.max()
        exp_C = np.exp(beta := beta_start * (beta_end/beta_start) ** min(1.0, it/(max_iters*0.7)))
        # Actually let me compute beta properly
        it_frac = min(1.0, it / max(1, max_iters * 0.7))
        beta = beta_start * (beta_end / beta_start) ** it_frac
        exp_C = np.exp(beta * (C - max_C))
        total = exp_C.sum()
        if total < 1e-300:
            break
        p = exp_C / total
        
        corr = np.correlate(p, f, mode='valid')
        grad = 2.0 * corr
        
        it += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad ** 2)
        m_hat = m / (1 - b1 ** it)
        v_hat = v / (1 - b2 ** it)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        f_new = f - lr * update
        f_new = np.clip(f_new, 0.0, None)
        S_new = float(np.sum(f_new))
        if S_new < 1e-10:
            f_new = f
            S_new = 1.0
        f_new = f_new / S_new
        
        sc = evaluate_sequence(f_new.tolist())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            if no_improve > 300:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-7:
                    break
    
    return best_f, best_score


def lp_iterate(f, deadline, step_size_init=0.1, max_iters=100):
    """LP-based direction finding."""
    n = len(f)
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, np.inf
    f = f / S
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    step_size = step_size_init
    no_improve = 0
    A_cache = None
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break
        
        if A_cache is None:
            A_cache = build_conv_matrix_sparse(f)
        
        g = solve_lp_direction(f, M, A_cache)
        it += 1
        
        if g is None:
            no_improve += 1
            if no_improve > 15:
                break
            continue
        
        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue
        
        g = g / 1.0  # normalize to sum = 1 (same as f)
        
        best_t = step_size
        best_s = best_score
        for trial_t in [step_size, step_size * 1.5, step_size * 0.5]:
            new_f = (1.0 - trial_t) * f + trial_t * g
            new_f = np.clip(new_f, 0.0, 1000.0)
            ns = float(np.sum(new_f))
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = trial_t
        
        if best_s < best_score:
            f = (1.0 - best_t) * f + best_t * g
            f = np.clip(f, 0.0, 1000.0)
            f = f / float(np.sum(f))
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            A_cache = None  # Need to rebuild
        else:
            no_improve += 1
            step_size *= 0.6
            if step_size < 0.003 or no_improve > 30:
                break
    
    return best_f, best_score
