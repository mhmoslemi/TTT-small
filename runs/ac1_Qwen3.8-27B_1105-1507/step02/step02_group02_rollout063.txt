"""
Minimize J(a) = 2 n max(convolve(a,a)) / (sum(a))^2.

Strategy used here:
- Normalize to sum 1; then J = 2 n max_k (a*aconv)_k.
- Use a smooth log-sum-exp + variance surrogate with analytic gradients.
- Run a natural-gradient / mirror-descent optimizer in simplex coordinates.
- Polish with L-BFGS-B on a softmax reparameterization.
- Use multi-scale, multi-start search from the parent, spectral-projection
  starts, random smooth starts, and perturbations.
- Include a short LP best-response polish for smaller lengths.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog

_PO2_CACHE = {}


def next_pow2(m):
    m = int(m)
    if m in _PO2_CACHE:
        return _PO2_CACHE[m]
    p = 1
    while p < m:
        p <<= 1
    _PO2_CACHE[m] = p
    return p


def normalize_sequence(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-15:
        return None
    return a / s


def resample_sequence(a, n):
    a = normalize_sequence(a)
    if a is None:
        return None
    n = int(n)
    m = a.size
    if m == n:
        return a.copy()
    if m == 1:
        y = np.full(n, float(a[0]), dtype=np.float64)
    else:
        x_old = np.linspace(0.0, 1.0, m, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n, endpoint=False)
        y = np.interp(x_new, x_old, a)
    y = np.clip(y, 0.0, None)
    return normalize_sequence(y)


def fast_conv_n(a, n):
    N = next_pow2(2 * n - 1)
    af = np.fft.rfft(a, N)
    out = np.fft.irfft(af * af, N)[:2 * n - 1]
    return np.maximum(out, 0.0)


def compute_score(a):
    a = normalize_sequence(a)
    if a is None:
        return float("inf")
    n = int(a.size)
    if n <= 0:
        return float("inf")
    b = fast_conv_n(a, n)
    mx = float(b.max())
    if not np.isfinite(mx):
        return float("inf")
    return float(2.0 * n * mx)


def smooth_val_only(p, beta, rho, n):
    N = next_pow2(2 * n - 1)
    P = np.fft.rfft(p, N)
    b = np.fft.irfft(P * P, N)[:2 * n - 1]
    b = np.maximum(b, 0.0)
    maxb = float(b.max())
    shifted = beta * (b - maxb)
    e = np.exp(shifted)
    sume = float(e.sum())
    if sume <= 0.0 or not np.isfinite(sume):
        lse = maxb
    else:
        lse = maxb + float(np.log(sume)) / beta
    val = 2.0 * n * lse
    if rho > 0.0:
        val += n * rho * float(np.dot(b, b))
    if not np.isfinite(val):
        return float("inf")
    return float(val)


def smooth_components(p, beta, rho, n):
    N = next_pow2(2 * n - 1)
    P = np.fft.rfft(p, N)
    b = np.fft.irfft(P * P, N)[:2 * n - 1]
    b = np.maximum(b, 0.0)

    maxb = float(b.max())
    shifted = beta * (b - maxb)
    e = np.exp(shifted)
    sume = float(e.sum())
    if sume <= 0.0 or not np.isfinite(sume):
        q = np.full(b.size, 1.0 / b.size, dtype=np.float64)
        lse = maxb
    else:
        q = e / sume
        lse = maxb + float(np.log(sume)) / beta

    val = 2.0 * n * lse

    Pc = np.conj(P)
    Q = np.fft.rfft(q, N)
    h = np.fft.irfft(Q * Pc, N)[:n]
    h = np.maximum(np.asarray(h, dtype=np.float64), 0.0)

    if rho > 1e-6:
        B = np.fft.rfft(b, N)
        hb = np.fft.irfft(B * Pc, N)[:n]
        hb = np.maximum(np.asarray(hb, dtype=np.float64), 0.0)
        h = h + rho * hb
        val += n * rho * float(np.dot(b, b))

    grad_p = 4.0 * n * h
    if not np.isfinite(val):
        val = float("inf")
    return float(val), grad_p, b, q


def smooth_lbfgs_objective(u, beta, rho, n):
    try:
        n = int(n)
        z = np.empty(n, dtype=np.float64)
        if n == 1:
            z[0] = 0.0
        else:
            z[:n - 1] = np.asarray(u, dtype=np.float64).ravel()
            z[n - 1] = 0.0

        z -= float(np.max(z))
        ez = np.exp(z)
        s = float(ez.sum())
        if not np.isfinite(s) or s <= 1e-300:
            return 1e100, np.zeros(max(0, n - 1), dtype=np.float64)

        p = ez / s
        val, grad_p, b, q = smooth_components(p, beta, rho, n)

        if not np.isfinite(val):
            return 1e100, np.zeros(max(0, n - 1), dtype=np.float64)

        gbar = float(np.dot(p, grad_p))
        grad_z = p * (grad_p - gbar)
        if n == 1:
            grad_u = np.zeros(0, dtype=np.float64)
        else:
            grad_u = grad_z[:n - 1]
        grad_u = np.where(np.isfinite(grad_u), grad_u, 0.0)
        return float(val), grad_u
    except Exception:
        return 1e100, np.zeros(max(0, int(n) - 1), dtype=np.float64)

    if p is None:

    p = np.maximum(p, 1e-12).astype(np.float64)
    p = p / float(p.sum())

    best_p = p.copy()
    best_score = compute_score(best_p)

    if n <= 256:

    elif n <= 1024:

    elif n <= 4096:
        betas = [10.0, 30.0, 100.0, 300.0, 1000.0, 3000.0]
    else:
        betas = [10.0, 40.0, 150.0, 500.0, 1500.0]

    base_step = 0.35 if n <= 1024 else 0.25
    max_step = base_step * 2.0
    check_interval = 50 if n <= 256 else 100 if n <= 1024 else 300

    it = 0
    no_improve = 0
    step = base_step

    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break

        rho = 0.25 * (0.62 ** bi)
        if rho < 0.012:

            if not np.isfinite(val):

            it += 1
            if it % check_interval == 0:
                sc = compute_score(p)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best_score = sc
                    best_p = p.copy()
                    no_improve = 0
                else:
                    no_improve += check_interval
                    if np.isfinite(sc) and sc > best_score * 1.05 + 1e-7:
                        p = best_p.copy()
                        step = max(0.03, 0.5 * step)

            A = grad_p - float(np.dot(p, grad_p))
            maxA = float(np.max(np.abs(A))) if A.size else 0.0
            if not np.isfinite(maxA) or maxA < 1e-12:
                scale = 0.03 + 0.02 * (float(bi) / float(len(betas)))
                p = p * np.exp(scale * rng.normal(0.0, 1.0, n))
                p = np.maximum(p, 1e-12)
                p = p / float(p.sum())
                step = max(0.03, 0.5 * step)
                continue

            step_local = step
            accepted = False
            for _ in range(5):
                logf = -step_local * A / maxA
                p_new = p * np.exp(logf)
                s = float(p_new.sum())
                if np.isfinite(s) and s > 1e-14:
                    p_new = p_new / s
                    val_new = smooth_val_only(p_new, beta, rho, n)
                    if np.isfinite(val_new) and val_new <= val + 1e-8:
                        p = p_new
                        accepted = True
                        if val_new < val * (1.0 - 1e-4):
                            step = min(max_step, step_local * 1.07)
                        else:
                            step = step_local
                        break
                step_local *= 0.5

            if not accepted:
                step = max(1e-3, 0.5 * step_local)
                if step < 1e-3:
                    break

            if no_improve > 4000:
                scale = 0.02 + 0.03 * (float(bi) / float(len(betas)))
                p = p * np.exp(scale * rng.normal(0.0, 1.0, n))
                p = np.maximum(p, 1e-12)
                p = p / float(p.sum())
                no_improve = 0
                step = max(0.03, 0.7 * step)

    sc = compute_score(p)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_p = p.copy()
    return best_p, best_score

    best_score = compute_score(best_p)

    if n <= 512:
        betas = [50.0, 200.0, 800.0, 3000.0, 10000.0]
        maxiter = 350
        maxfun = 1400
    elif n <= 1024:
        betas = [50.0, 200.0, 800.0, 3000.0]
        maxiter = 250
        maxfun = 1000
    elif n <= 2048:
        betas = [30.0, 150.0, 600.0, 2000.0]
        maxiter = 150
        maxfun = 600
    else:
        betas = [20.0, 100.0, 400.0]
        maxiter = 80
        maxfun = 320

    rho_list = [0.12, 0.06, 0.02, 0.0]

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        rho = rho_list[min(i, len(rho_list) - 1)]

        p_work = np.maximum(best_p, 1e-12).astype(np.float64)
        p_work = p_work / float(p_work.sum())
        z = np.log(p_work)
        z -= z[-1]
        u = z[:-1].copy()

        try:
            res = minimize(
                smooth_lbfgs_objective,
                u,
                args=(beta, rho, n),
                method="L-BFGS-B",
                jac=True,
                options={
                    "maxiter": maxiter,
                    "maxfun": maxfun,
                    "ftol": 1e-13,
                    "gtol": 1e-7,
                    "iprint": 0,
                    "maxls": 50,
                },
            )
            if res.fun < 1e50:
                z_r = np.empty(n, dtype=np.float64)
                z_r[:n - 1] = res.x
                z_r[-1] = 0.0
                z_r -= float(np.max(z_r))
                ez = np.exp(z_r)
                p_res = ez / float(ez.sum())
                p_res = np.maximum(p_res, 0.0)
                s = float(p_res.sum())
                if s > 1e-12:
                    p_res = p_res / s
                    sc = compute_score(p_res)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_p = p_res.copy()
        except Exception:
            pass

    return best_p, best_score


def build_conv_matrix(a):
    n = int(a.size)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)

    for k in range(n):
        A[k, 0:k + 1] = a[k::-1]

    for k in range(n, m):
        lo = k - n + 1
        A[k, lo:n] = a[k - (n - 1):n][::-1]

    return A


def lp_best_response(a, M):
    n = int(a.size)
    m = 2 * n - 1
    if M <= 1e-15:
        return None
    A = build_conv_matrix(a)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=np.full(m, float(M), dtype=np.float64),
            bounds=bounds,
            method="highs",
        )
    except Exception:
        return None

    if not res.success:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0.0)
    if not np.isfinite(g.sum()) or g.sum() <= 1e-12:
        return None
    return g


def lp_polish(p, n, deadline):
    a = normalize_sequence(p)
    if a is None:
        return None, float("inf")
    if a.size != n:
        a = resample_sequence(a, n)
    if a is None or n > 1024:
        return a, compute_score(a)

    best_a = a.copy()
    best_score = compute_score(best_a)
    t = 0.42

    for _ in range(5000):
        if time.monotonic() > deadline:
            break

        b = fast_conv_n(a, n)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        sg = float(g.sum())
        if sg <= 1e-12:
            break
        g = g / sg

        tt_list = [t, 0.5 * t, 0.2 * t, 0.06]
        improved = False
        for tt in tt_list:
            if tt <= 0.0:
                continue
            a_new = (1.0 - tt) * a + tt * g
            a_new = np.maximum(a_new, 1e-12)
            a_new = a_new / float(a_new.sum())
            sc = compute_score(a_new)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                improved = True
                break

        if improved:
            t = min(0.55, t * 1.02)
        else:
            t *= 0.45
            if t < 0.008:
                break

    return best_a, best_score


def local_search(p, n, budget_s, deadline, rng, allow_heavy=True):
    p = normalize_sequence(p)
    if p is None:
        p = np.ones(n, dtype=np.float64)
        p /= n
    if p.size != n:
        p = resample_sequence(p, n)
    if p is None:
        p = np.ones(n, dtype=np.float64)
        p /= n

    start = time.monotonic()
    dl = min(deadline, start + max(0.0, float(budget_s)))
    best_p = p.copy()
    best_score = compute_score(best_p)
    if not np.isfinite(best_score):
        best_score = compute_score(np.ones(n) / n)
        best_p = np.ones(n, dtype=np.float64) / n

    if dl - start < 5.0:
        return best_p, best_score

    heavy = allow_heavy and float(budget_s) >= 90.0
    if heavy and n <= 4096:
        mirror_dl = min(dl, start + max(20.0, 0.55 * float(budget_s)))
    else:
        mirror_dl = dl

    p_m, sc_m = mirror_optimize(best_p, n, mirror_dl, rng)
    if sc_m < best_score:
        best_score = sc_m
        best_p = p_m

    if heavy and n <= 4096 and time.monotonic() < dl:
        if n <= 1024:
            lb_dl = min(dl, start + max(mirror_dl - start + 10.0, 0.82 * float(budget_s)))
        else:
            lb_dl = min(dl, start + float(budget_s))
        p_l, sc_l = lbfgs_polish(best_p, n, lb_dl, rng)
        if sc_l < best_score:
            best_score = sc_l
            best_p = p_l

    if heavy and n <= 1024 and time.monotonic() < dl:
        p_lp, sc_lp = lp_polish(best_p, n, dl)
        if sc_lp < best_score:
            best_score = sc_lp
            best_p = p_lp

    sc = compute_score(best_p)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
    return best_p, best_score


def random_hermitian(magt, rng):
    N = int(len(magt))
    P = np.zeros(N, dtype=np.complex128)
    if N == 0:
        return P
    P[0] = float(magt[0])
    half = N // 2

    if half > 1:
        ang = rng.uniform(0.0, 2.0 * np.pi, half - 1)
        coeff = np.exp(1j * ang)
        P[1:half] = magt[1:half] * coeff
        P[N - half + 1:] = magt[half - 1:0:-1] * np.conj(coeff)[::-1]

    if half >= 1:
        P[half] = float(magt[half]) * (1.0 if rng.random() < 0.5 else -1.0)

    return P


def spectral_start(n, rng, iters, mode):
    n = int(n)
    for attempt in range(3):
        N = next_pow2(2 * n - 1)
        if mode == 1:
            magt = np.full(N, 1.0 / np.sqrt(float(N)), dtype=np.float64)
        else:
            target = np.zeros(N, dtype=np.float64)
            target[:2 * n - 1] = 1.0 / float(2 * n - 1)
            Bt = np.fft.fft(target)
            magt = np.sqrt(np.abs(Bt) + 1e-300)

        P = random_hermitian(magt, rng)
        p = None

        for it in range(iters + 1):
            p = np.real(np.fft.ifft(P))
            p = np.asarray(p, dtype=np.float64)
            p[:n] = np.maximum(p[:n], 0.0)
            p[n:] = 0.0
            s = float(p.sum())
            if s < 1e-3:
                break
            p = p / s
            if it == iters:
                break
            Pcur = np.fft.fft(p)
            phase = Pcur / (np.abs(Pcur) + 1e-300)
            phase[0] = 1.0
            P = magt * phase

        if p is not None:
            p = np.maximum(p, 0.0)
            s = float(p.sum())
            if s > 1e-6:
                return p / s
    return None


def random_smooth_start(n, rng):
    n = int(n)
    t = np.linspace(0.0, 1.0, n, endpoint=False)
    mode = int(rng.integers(0, 4))
    f = np.zeros(n, dtype=np.float64)

    if mode == 0:
        k = int(rng.integers(2, 5))
        for _ in range(k):
            mu = float(rng.uniform(0.0, 1.0))
            sig = float(rng.uniform(0.03, 0.25))
            amp = float(rng.uniform(0.5, 2.0))
            f += amp * np.exp(-0.5 * ((t - mu) / sig) ** 2)
        f += 0.05

    elif mode == 1:
        a = float(rng.uniform(0.15, 4.0))
        b = float(rng.uniform(0.15, 4.0))
        tt = np.clip(t, 1e-7, 1.0 - 1e-7)
        f = np.power(tt, a) * np.power(1.0 - tt, b)
        f += 1e-3

    elif mode == 2:
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        freq = float(rng.uniform(1.0, 4.0))
        f = 1.0 + 0.45 * np.sin(2.0 * np.pi * freq * t + ph)
        mu = float(rng.uniform(0.0, 1.0))
        sig = float(rng.uniform(0.05, 0.20))
        f += 0.6 * np.exp(-0.5 * ((t - mu) / sig) ** 2)
        f += 0.25

    else:
        z = rng.normal(0.0, 1.0, n)
        kern = np.ones(5, dtype=np.float64) / 5.0
        z = np.convolve(z, kern, mode="same")
        f = 0.5 + 0.5 * np.exp(z)
        f += 0.05

    f = np.convolve(f, np.ones(min(5, max(3, n // 64 + 1)), dtype=np.float64) / 5.0, mode="same")
    f = np.maximum(f, 1e-6)
    out = normalize_sequence(f)
    if out is None:
        out = np.ones(n, dtype=np.float64) / n
    return out


def perturb_sequence(a, n, rng, scale):
    p = resample_sequence(a, n)
    if p is None:
        return None
    p = np.maximum(p, 1e-12)
    p = p * np.exp(float(scale) * rng.normal(0.0, 1.0, int(n)))
    return normalize_sequence(p)


def build_candidates(n, rng, parent, best_arr, num_spec, include_best, include_parent):
    n = int(n)
    cands = []

    if include_best and best_arr is not None:
        cands.append(resample_sequence(best_arr, n))
        cands.append(resample_sequence(best_arr[::-1], n))

    if include_parent and parent is not None:
        cands.append(resample_sequence(parent, n))

    if n <= 1024:
        iters = 6
    elif n <= 4096:
        iters = 3
    else:
        iters = 2

    for r in range(int(num_spec)):
        sp = spectral_start(n, rng, iters, r % 2)
        if sp is not None:
            cands.append(sp)

    cands.append(random_smooth_start(n, rng))
    if include_best and best_arr is not None:
        cands.append(perturb_sequence(best_arr, n, rng, 0.12))
        cands.append(perturb_sequence(best_arr, n, rng, 0.30))

    return cands


def process_candidates(candidates, n, phase_budget, deadline, rng, best_arr, best_score):
    n = int(n)
    scored = []
    seen = []
    idx = 0

    for c in candidates:
        if c is None:
            continue
        cn = normalize_sequence(c)
        if cn is None:
            continue
        if cn.size != n:
            cn = resample_sequence(cn, n)
        if cn is None:
            continue

        dup = False
        for prev in seen:
            if np.linalg.norm(cn - prev) < 0.12:
                dup = True
                break
        if dup:
            continue

        sc = compute_score(cn)
        if np.isfinite(sc):
            scored.append((sc, idx, cn))
            seen.append(cn)
            idx += 1

    if not scored:
        return best_arr, best_score

    scored.sort()
    m = len(scored)

    for i, item in enumerate(scored):
        if time.monotonic() > deadline:
            break
        remaining_phase = deadline - time.monotonic()
        if remaining_phase < 35.0:
            break

        base_alloc = max(35.0, float(phase_budget) / float(max(1, m - i)))
        alloc = min(remaining_phase, base_alloc)
        if i == 0 and m > 1:
            alloc = min(remaining_phase, max(alloc, 0.45 * float(phase_budget)))

        do_heavy = float(phase_budget) >= 90.0
        p_res, sc_res = local_search(item[2], n, alloc, deadline, rng, allow_heavy=do_heavy)
        if np.isfinite(sc_res) and sc_res < best_score:
            best_arr = p_res
            best_score = sc_res
            print(f"  n={n} cand[{i}]: {sc_res:.8f}", flush=True)

    return best_arr, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    total = float(budget_s)
    if not np.isfinite(total) or total <= 0.0:
        total = 1500.0
    deadline = time.monotonic() + total - 20.0
    rng = np.random.default_rng(int(seed))

    best_arr = None
    best_score = float("inf")

    parent = None
    try:
        parr = np.asarray(height_sequence_1, dtype=np.float64)
        if parr.ndim == 1 and parr.size > 0 and np.all(np.isfinite(parr)):
            parr = np.clip(parr, 0.0, 1000.0)
            s = float(parr.sum())
            if s > 0.01:
                parent = parr / s
                best_arr = parent.copy()
                best_score = compute_score(parent)
                print(f"parent: n={parent.size}, J={best_score:.8f}", flush=True)
    except Exception:
        parent = None

    if parent is None or best_arr is None:
        n0 = 512
        parent = np.ones(n0, dtype=np.float64) / n0
        best_arr = parent.copy()
        best_score = compute_score(parent)
        print(f"fallback uniform: n={n0}, J={best_score:.8f}", flush=True)

    if total < 120.0:
        n = int(min(512, max(128, parent.size)))
        p_res, sc_res = local_search(best_arr, n, max(5.0, total - 20.0), deadline, rng, allow_heavy=False)
        if np.isfinite(sc_res) and sc_res < best_score:
            best_arr = p_res
            best_score = sc_res
        final = np.clip(np.asarray(best_arr, dtype=np.float64), 0.0, 1e3)
        ss = float(final.sum())
        if ss < 0.01:
            final = np.ones(512, dtype=np.float64) / 512
        else:
            final = final / ss
        print(f"FINAL: J={best_score:.8f}, n={final.size}", flush=True)
        return [float(x) for x in final]

    n_par = int(parent.size)
    if n_par < 128 or n_par > 2048:
        n_par = 512

    n2 = min(8192, max(256, n_par * 2))
    n4 = min(8192, max(512, n_par * 4))
    n8 = min(8192, max(1024, n_par * 8))
    n16 = min(8192, max(2048, n_par * 16))
    n_half = max(128, n_par // 2)

    phase_specs = []
    phase_specs.append((n_par, 0.18, 2, True, True, True))
    phase_specs.append((n2, 0.24, 2, True, False, True))
    phase_specs.append((n4, 0.20, 1, True, False, False))
    phase_specs.append((n8, 0.14, 1, True, False, False))
    phase_specs.append((n16, 0.08, 0, True, False, False))
    phase_specs.append((n_half, 0.04, 1, False, True, True))

    # Remove duplicate n values, preserving order.
    used_n = set()
    clean_specs = []
    for spec in phase_specs:
        nn = int(spec[0])
        if 64 <= nn <= 8192 and nn not in used_n and spec[1] > 0.0:
            used_n.add(nn)
            clean_specs.append(spec)

    final_reserve = 0.10 * total

    for spec in clean_specs:
        nn = int(spec[0])
        frac = float(spec[1])
        num_spec = int(spec[2])
        include_best = bool(spec[3])
        include_parent = bool(spec[4])

        if time.monotonic() > deadline - 60.0:
            break

        available = deadline - time.monotonic()
        if available < 70.0:
            break

        phase_budget = min(frac * total, max(60.0, available - final_reserve))
        if phase_budget < 45.0:
            continue

        t0 = time.monotonic()
        print(f"phase n={nn} budget={phase_budget:.1f}s", flush=True)

        cands = build_candidates(nn, rng, parent, best_arr, num_spec, include_best, include_parent)
        best_arr, best_score = process_candidates(
            cands, nn, phase_budget, deadline, rng, best_arr, best_score
        )
        print(f"  phase done in {time.monotonic()-t0:.1f}s, best={best_score:.8f}", flush=True)

    # Final polish.
    if best_arr is not None and time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem > 40.0:
            nn = int(best_arr.size)
            b1 = 0.42 * rem
            p_res, sc_res = local_search(best_arr, nn, b1, deadline, rng, allow_heavy=True)
            if np.isfinite(sc_res) and sc_res < best_score:
                best_score = sc_res
                best_arr = p_res
                print(f"final polish same length: {best_score:.8f}", flush=True)

        if best_arr is not None and time.monotonic() < deadline - 60.0:
            nn2 = min(8192, max(256, int(best_arr.size) * 2))
            if nn2 != int(best_arr.size):
                rem = deadline - time.monotonic()
                b2 = 0.50 * rem
                cands = []
                cands.append(resample_sequence(best_arr, nn2))
                cands.append(spectral_start(nn2, rng, 2, 0))
                cands.append(random_smooth_start(nn2, rng))
                best_arr, best_score = process_candidates(
                    cands, nn2, b2, deadline, rng, best_arr, best_score
                )
                print(f"final up-sample to {nn2}: best={best_score:.8f}", flush=True)

    # Final sanity check.
    if best_arr is None:
        best_arr = np.ones(512, dtype=np.float64) / 512

    final_arr = np.clip(np.asarray(best_arr, dtype=np.float64), 0.0, 1000.0)
    ss = float(final_arr.sum())
    if not np.isfinite(ss) or ss < 0.01:
        final_arr = np.ones(512, dtype=np.float64) / 512
    else:
        final_arr = final_arr / ss

    final_score = best_score
    try:
        fs = evaluate_sequence([float(x) for x in final_arr])
        if np.isfinite(fs):
            final_score = min(final_score, float(fs))
    except Exception:
        pass

    print(f"FINAL: J={final_score:.8f}, n={final_arr.size}", flush=True)
    return [float(x) for x in final_arr]
