"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            segment = a[k - jh:k - jl + 1][::-1]
            vals.extend(segment.tolist())
    A = lil_matrix((2 * n - 1, n))
    if len(rows) > 0:
        A[rows, cols] = vals
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_objective_and_grad(a, T):
    """
    Compute log-sum-exp approximation to max(b) and its gradient.
    Returns (value, gradient) where value = (1/T)*log(sum(exp(T*b)))
    and gradient is wrt a (before normalization).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = T * (b - b_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    value = b_max + log_sum / T

    w = exp_shifted / exp_shifted.sum()  # softmax weights, length 2n-1

    # Gradient: g[i] = 2 * sum_j w[i+j] * a[j]
    # This is cross-correlation of w (length 2n-1) and a (length n)
    size = 1
    while size < 3 * n:
        size <<= 1
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(W * np.conj(A), size)[:n]
    grad = 2.0 * corr
    return value, grad


def lse_normalized_objective(x, T):
    """Objective for L-BFGS-B: normalizes x to sum=1, then computes LSE."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    a = a / s
    val, _ = lse_objective_and_grad(a, T)
    return val


def lse_normalized_gradient(x, T):
    """Gradient for L-BFGS-B with normalization chain rule."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones_like(x)
    a_norm = a / s
    _, grad_a = lse_objective_and_grad(a_norm, T)
    # Chain rule for a_norm = a/s: d(a_norm)/d(x) involves (e_i - a_i/s)/s
    # grad_x[i] = (grad_a[i] - a_norm[i] * sum(grad_a)) / s
    grad_x = (grad_a - a_norm * grad_a.sum()) / s
    # Where x[i] = 0, the subgradient of max(0,x) is 0, so grad should be 0
    grad_x = np.where(x > 0, grad_x, 0.0)
    return grad_x


def fourier_init(n, rng):
    """Generate initialization via Fourier synthesis targeting flat spectrum."""
    # Target: flat |F(a)|^2 means low PAPR
    # Create a random flat-magnitude spectrum with random phases
    mag = np.ones(n)
    # Add some randomness to magnitude
    mag *= 1.0 + 0.3 * rng.standard_normal(n)
    mag = np.maximum(mag, 0.1)
    phase = 2 * np.pi * rng.random(n)
    spectrum = mag * np.exp(1j * phase)
    # Inverse FFT
    sig = np.fft.ifft(spectrum).real
    # Make non-negative
    sig = np.abs(sig) + 0.1
    sig = sig / sig.sum()
    return sig


def cosine_init(n, rng):
    """Cosine-based initialization with random frequencies."""
    t = np.linspace(0, 1, n, endpoint=False)
    n_freq = rng.integers(2, 6)
    sig = np.ones(n)
    for _ in range(n_freq):
        freq = rng.uniform(0.5, 3.0)
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        sig += amp * np.cos(2 * np.pi * freq * t + phase)
    sig = np.maximum(sig, 0.01)
    return sig / sig.sum()


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def run_lp_momentum(a, max_iters, deadline, tracker, beta=0.3):
    """LP mirror descent with Nesterov-like momentum and line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    v = np.zeros_like(a)
    current_J = compute_J(a)
    tracker.update(a)
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Momentum update
        v = beta * v + (1.0 - beta) * (g - a)

        # Line search: try multiple step sizes
        step_candidates = [0.3, 0.5, 0.7, 1.0, 0.15]
        # Scale by decay factor
        decay = max(0.1, 1.0 - 0.3 * it / max_iters)
        step_candidates = [s * decay for s in step_candidates]

        best_a = None
        best_J = current_J

        for step in step_candidates:
            a_trial = a + step * v
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial.copy()

        if best_a is None:
            # Fallback: small step toward g
            a_trial = (1.0 - 0.05) * a + 0.05 * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-15:
                a = a_trial
                current_J = J_trial
                v = 0.5 * (a - (a - 0.05 * v))  # dampen velocity
            else:
                # Try even smaller
                a_trial = (1.0 - 0.01) * a + 0.01 * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                a_trial = a_trial / max(s, 1e-10)
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-15:
                    a = a_trial
                    current_J = J_trial
                else:
                    break
        else:
            v = 0.8 * v  # dampen
            a = best_a
            current_J = best_J

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_lse_lbfgs(a, deadline, tracker):
    """LSE-smoothed L-BFGS-B optimization."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    T_schedule = [5.0, 10.0, 20.0, 40.0]

    for T in T_schedule:
        if time.monotonic() > deadline:
            break

        def obj(x):
            return lse_normalized_objective(x, T)

        def grad(x):
            return lse_normalized_gradient(x, T)

        bounds_list = [(0.0, None)] * n
        x0 = a.copy()

        res = minimize(obj, x0, jac=grad, method='L-BFGS-B',
                       bounds=bounds_list,
                       options={'maxiter': 200, 'max_fun': 5000, 'ftol': 1e-18, 'gtol': 1e-12})
        if res.fun < 1e9:
            a_new = np.maximum(res.x, 0.0)
            sn = a_new.sum()
            if sn > 1e-10:
                a_new = a_new / sn
                J_new = compute_J(a_new)
                J_old = compute_J(a)
                if J_new < J_old:
                    a = a_new
                    tracker.update(a)

    return a


def peak_targeted_refine(a, deadline, tracker, n_iters=50):

        grad_b = np.zeros(n)

            if 0 <= idx < n:

        if J_trial < current_J - 1e-15:
            a = a_trial
            current_J = J_trial
            if it % 10 == 0:

        else:
            # Try half step
            a_trial = a - 0.5 * step * grad_b
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            a_trial = a_trial / max(st, 1e-10)
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-15:
                a = a_trial
                current_J = J_trial
            else:
                break

    tracker.update(a)
    return a


def generate_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []

    if parent is not None:
        inits.append(resample_seq(parent, n))

    # Uniform
    inits.append(np.ones(n) / n)

    # Random
    inits.append(rng.uniform(0.1, 1.0, n))

    # Fourier synthesis (multiple)
    for _ in range(3):
        inits.append(fourier_init(n, rng))

    # Cosine
    for _ in range(3):
        inits.append(cosine_init(n, rng))

    # Triangle
    t = np.linspace(0, 1, n)
    inits.append(np.abs(1.0 - 2.0 * t) + 0.1)

    # Gaussian bump
    inits.append(np.exp(-4.0 * (t - 0.5) ** 2) + 0.05)

    # Wiggly
    inits.append(1.0 + 0.3 * np.sin(4.0 * np.pi * t) + 0.2 * np.sin(7.0 * np.pi * t))
    inits.append(1.0 + 0.25 * np.sin(2.0 * np.pi * t) * np.cos(5.0 * np.pi * t))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: LP momentum descent across multiple lengths
    Phase 2: LSE + L-BFGS-B polishing
    Phase 3: Peak-targeted refinement
    Phase 4: Final LP polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            parent = parent / parent.sum()
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128) / 128
        tracker.update(parent)
        print(f"No parent, using uniform n=128, J={tracker.best_score:.8f}", flush=True)

    # === Phase 1: LP Momentum Descent ===
    lengths = [128, 256, 512]
    # If parent length matches, prioritize that
    parent_n = parent.size

    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 1: LP Momentum Descent", flush=True)
    print(f"{'='*60}", flush=True)

    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n  --- LP n={n} ---", flush=True)

        inits = generate_inits(n, rng, parent)
        # Determine iterations based on length
        iters_per_init = max(50, min(200, int(60000 // n)))
        n_inits_to_try = min(len(inits), max(3, int(120.0 / max(1, (time.monotonic() - tracker.t_start) / 60))))

        for idx, a0 in enumerate(inits[:n_inits_to_try]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 = a0 / a0.sum()
            # Run LP with momentum
            a = run_lp_momentum(a0, iters_per_init, deadline, tracker, beta=0.3)
            # Quick LSE polish
            if time.monotonic() < deadline - 30:
                a = run_lse_lbfgs(a, deadline, tracker)

        elapsed = time.monotonic() - t0
        tracker.report(f"Phase1 n={n}: {elapsed:.1f}s")

    # === Phase 2: LSE + L-BFGS-B Polishing on best ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 2: LSE + L-BFGS-B Polishing", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"  n={n}", flush=True)
        a = run_lse_lbfgs(tracker.best, deadline, tracker)
        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Peak-Targeted Refinement ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 3: Peak-Targeted Refinement", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        a = peak_targeted_refine(tracker.best, deadline, tracker, n_iters=100)
        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Final LP Polish ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 4: Final LP Polish", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 512:
            a = run_lp_momentum(tracker.best, max(30, int(30000 // n)), deadline, tracker, beta=0.2)
        elif n <= 1024:
            a = run_lp_momentum(tracker.best, max(20, int(20000 // n)), deadline, tracker, beta=0.2)
        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Try resampling best to different lengths and re-optimizing ===
    print(f"\n{'='*60}", flush=True)
    print(f"PHASE 5: Multi-length Resampling", flush=True)
    print(f"{'='*60}", flush=True)

    if tracker.best is not None and time.monotonic() < deadline - 30:
        best_n = tracker.best.size
        target_lengths = []
        for ln in [128, 256, 512, 1024]:
            if ln != best_n:
                target_lengths.append(ln)

        for ln in target_lengths[:2]:
            if time.monotonic() > deadline - 20:
                break
            t0 = time.monotonic()
            a_resampled = resample_seq(tracker.best, ln)
            a_resampled = a_resampled / a_resampled.sum()
            tracker.update(a_resampled)

            # Quick LP + LSE
            iters = max(20, min(80, int(20000 // ln)))
            a = run_lp_momentum(a_resampled, iters, deadline, tracker, beta=0.3)
            if time.monotonic() < deadline - 10:
                a = run_lse_lbfgs(a, deadline, tracker)
            tracker.report(f"Resample to n={ln}: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
