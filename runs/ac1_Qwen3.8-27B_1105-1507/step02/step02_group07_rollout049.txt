import numpy as np

def _next_pow2(n):
    """Next power of 2 >= n."""
    p = 1
    while p < n:
        p <<= 1
    return p


def _convolve_fft(f, N):
    """Compute linear convolution f*f via FFT. Returns array of length len(f)*2-1."""
    F = np.fft.rfft(f, N)
    C = np.fft.irfft(F * F, N)
    return C[:2 * len(f) - 1]


def _softmax_gradient(f, C, beta, N):
    """
    Compute softmax-smoothed gradient of max(C) where C = conv(f,f).
    Returns (gradient_normalized, soft_max_value).
    grad[j] = 2 * sum_i f[i] * p[j+i] where p = softmax(beta * C)
    """
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros_like(f), max_C
    p = exp_C / total
    # grad[j] = 2 * sum_i f[i] * p[j+i]
    # This is 2 * correlate(p, f, 'valid')
    grad = 2.0 * np.correlate(p, f, mode='valid')
    soft_val = max_C + np.log(total) / beta
    return grad, soft_val


def _adam_update(m, v, m_hat, v_hat, grad, beta1, beta2, t, lr):
    """Adam update step. Returns updated (m, v, new_f direction)."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    direction = m_hat / (np.sqrt(v_hat) + 1e-8)
    return m, v, direction

    return f / s

def _freq_domain_flatten(f, n_iters=5, verbose=""):
    """
    Frequency-domain flattening: try to make |FFT(f)|^2 closer to the target
    (which would give a flat convolution).
    """
    n = len(f)
    N = _next_pow2(2 * n - 1)
    f = f.copy()
    
    # Target: we want conv(f,f) to be as flat as possible.
    # Target C_target[k] = (sum(f))^2 / (2n-1) = 1/(2n-1) for all k (with sum=1)
    # In frequency domain: |F_target[k]|^2 = |FFT(C_target, N)|^2
    # But this target might have zeros, so use a softened version.
    C_target_flat = np.ones(2 * n - 1) / (2 * n - 1)
    C_target_flat_np = np.zeros(N)
    C_target_flat_np[:2 * n - 1] = C_target_flat
    H_target = np.fft.rfft(C_target_flat_np)
    target_mag = np.maximum(np.abs(H_target), 1e-10)
    
    for it in range(n_iters):
        F = np.fft.rfft(f, N)
        mag = np.abs(F)
        phase = np.angle(F)
        # Blend current magnitude with target
        F_new_mag = 0.5 * mag + 0.5 * target_mag
        F_new = F_new_mag * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, N)[:n]
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-12:
            break
        f = f_new / s
    
    return f


def _multi_peak_gradient(f, N, top_k=3):
    """
    Compute gradient that attacks the top-k peaks simultaneously.
    """
    C = _convolve_fft(f, N)
    C = np.maximum(C, 0)
    # Get top-k peak indices
    peak_indices = np.argsort(C)[-top_k:]
    peak_weights = np.zeros(len(C))
    peak_values = C[peak_indices]
    # Weight by how much above the mean each peak is
    mean_C = C.mean()
    for k in peak_indices:
        peak_weights[k] = max(0, C[k] - mean_C)
    if peak_weights.sum() < 1e-15:
        peak_weights[peak_indices] = 1.0 / top_k
    peak_weights /= peak_weights.sum()
    
    # Gradient = sum_k peak_weights[k] * 2*f[k-j] (shifted)
    # = 2 * correlate(peak_weights, f, 'valid')
    grad = 2.0 * np.correlate(peak_weights, f, mode='valid')
    return grad


def _tukey_window(n, r=0.3):
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def _hann_window(n):
    n = max(n, 2)
    return 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))


def _psf_window(n, alpha=2.0):
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    return np.exp(-((idx - center) / (n / 2.0)) ** (2 * alpha))


def _random_smooth(n, seed):
    rng = np.random.default_rng(seed)
    x = rng.standard_normal(n)
    kernel_size = max(3, n // 20)
    if kernel_size % 2 == 0:
        kernel_size += 1
    kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (kernel_size / 4.0)) ** 2)
    kernel /= kernel.sum()
    smooth_x = np.convolve(x, kernel, mode='same')
    smooth_x = np.abs(smooth_x) + 0.05
    return smooth_x


def _gs_initialize(n, n_fft, n_iters=20, seed=42):
    """
    Gerchberg-Saxton style initialization with target flat convolution magnitude.
    """
    rng = np.random.default_rng(seed)
    N = n_fft
    
    # Target magnitude in frequency domain
    # We want conv(f,f) to be flat => |F|^2 = |H|^2 where H = FFT(flat_conv)
    C_target = np.ones(2 * n - 1) / (2 * n - 1)
    C_target_np = np.zeros(N)
    C_target_np[:2 * n - 1] = C_target
    H_mag = np.abs(np.fft.rfft(C_target_np))
    H_mag = np.maximum(H_mag, 1e-10)  # avoid zeros
    
    # Random initial phase
    phase = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    
    f = np.zeros(n)
    for it in range(n_iters):
        # Frequency domain: set magnitude to target, keep phase
        F = H_mag * np.exp(1j * phase)
        f_complex = np.fft.irfft(F, N)
        f = np.real(f_complex)[:n]
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-12:
            # Fallback
            f = tukey_fallback = np.ones(n) / n
            break
        f = f / s
        # Time domain: get new phase
        F_current = np.fft.rfft(f, N)

    return f


def _peak_shave(f, N, strength=0.01):
    """
    Directly reduce the peak by lowering elements that contribute most to it.
    """
    n = len(f)
    C = _convolve_fft(f, N)
    C = np.maximum(C, 0)
    k_star = int(np.argmax(C))
    
    # Elements contributing to C[k_star]: f[j] * f[k_star - j]
    # Gradient at peak: 2*f[k_star - j] for valid j
    k_min = max(0, k_star - n + 1)
    k_max = min(n - 1, k_star)
    
    contribution = np.zeros(n)
    for j in range(k_min, k_max + 1):
        contribution[j] = 2.0 * f[k_star - j]
    
    # Lower the contributing elements proportionally
    max_contrib = contribution.max()
    if max_contrib > 1e-12:
        reduction = strength * contribution / max_contrib
        f = f - reduction
        f = np.maximum(f, 0.0)
        f = f / f.sum()
    
    return f

    beta1 = 0.85
    beta2 = 0.999
    no_improve = 0
    
    while time.monotonic() < deadline:
        # Determine mode: gradient or freq flatten
        mode = "grad" if (t % 20 < 16) else "freq"
        
        if mode == "grad":
            C = _convolve_fft(f, N := n_fft)
            C = np.maximum(C, 0)
            
            # Adaptive beta
            frac = min(1.0, t / 1000)
            beta = 10.0 * (400.0 / 10.0) ** frac
            grad, soft_val = _softmax_gradient(f, C, beta, n_fft)
            
            gnorm = np.sqrt(np.sum(grad ** 2)) + 1e-12
            grad_norm = grad / gnorm
            
            t += 1
            m = beta1 * m + (1 - beta1) * grad_norm
            v = beta2 * v + (1 - beta2) * (grad_norm ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            direction = m_hat / (np.sqrt(v_hat) + 1e-8)
            
            new_f = f - lr * direction
            new_f = np.maximum(new_f, 0.0)
            ns = new_f.sum()
            if ns > 1e-12:
                new_f = new_f / ns
                sc = evaluate_sequence(new_f.tolist())
                if np.isfinite(sc) and sc < best_score:
                    f = new_f
                    best_f = f.copy()
                    best_score = sc
                    no_improve = 0
                else:
                    no_improve += 1
                    if no_improve > 80:
                        lr *= 0.75
                        no_improve = 0
                        if lr < 1e-7:
                            break
        else:
            # Frequency domain flattening (1-2 iterations)
            f_flattened = _freq_domain_flatten(f, n_iters=1)
            sc = evaluate_sequence(f_flattened.tolist())
            if np.isfinite(sc) and sc < best_score:
                f = f_flattened
                best_f = f.copy()
                best_score = sc
                no_improve = 0
            else:
                # Peak shaving
                f_shaved = _peak_shave(f, n_fft, strength=0.005)
                sc2 = evaluate_sequence(f_shaved.tolist())
                if np.isfinite(sc2) and sc2 < best_score:
                    f = f_shaved
                    best_f = f.copy()
                    best_score = sc2
        
        # Occasional peak shave
        if t % 100 == 50:
            f_shaved = _peak_shave(f, n_fft, strength=0.01)
            sc = evaluate_sequence(f_shaved.tolist())
            if np.isfinite(sc) and sc < best_score:
                f = f_shaved
                best_f = f.copy()
                best_score = sc
        
        if t % 500 == 0:
            print(f"    {verbose}t={t}: score={best_score:.6f} lr={lr:.5f}")
    
    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    """
    Multi-strategy search for optimal non-negative sequence minimizing
    2n*max(conv(f,f))/sum(f)^2.
    Uses FFT-accelerated gradient descent, frequency-domain flattening,
    peak shaving, and diverse multi-start with restarts.
    """
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)
    
    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(2024)
    
    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                score = evaluate_sequence(parent.tolist())
                if np.isfinite(score):
                    best_score = score
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")
    
    # Lengths to try
    lengths_to_try = [512, 768, 384, 1024]
    
    # Time budget allocation
    total_budget = deadline - t0
    time_budgets = {512: 0.30, 768: 0.22, 384: 0.15, 1024: 0.20}
    
    for n in lengths_to_try:
        if time.monotonic() > deadline - 40:
            print("Time running out, stopping length exploration")
            break
        
        N = _next_pow2(2 * n - 1)
        n_budget = total_budget * time_budgets.get(n, 0.1)
        n_deadline = time.monotonic() + n_budget
        print(f"\n=== n={n}, N_fft={N} (budget {n_budget:.0f}s) ===")
        
        # Generate initializations
        inits = []
        
        # GS initialization (diverse seeds)
        for seed in [42, 123, 7, 999]:
            inits.append((f'gs_s{seed}', _gs_initialize(n, N, n_iters=15, seed=seed)))
        
        # Windows
        for r in [0.2, 0.35, 0.5]:
            inits.append((f'tukey_r{r}', _tukey_window(n, r)))
        inits.append(('hann', _hann_window(n)))
        for alpha in [1.5, 2.5, 4.0]:
            inits.append((f'psf_a{alpha}', _psf_window(n, alpha)))
        
        # Random smooth
        for seed in [1, 2, 3]:
            inits.append((f'rs_{seed}', _random_smooth(n, seed * 100 + n)))
        
        # Interpolated parent
        if best_sequence is not None:
            if len(best_sequence) == n:
                inits.insert(0, ('parent', best_sequence.copy()))
            else:
                parent_interp = np.interp(
                    np.linspace(0, 1, n),
                    np.linspace(0, 1, len(best_sequence)),
                    best_sequence
                )
                parent_interp = np.maximum(parent_interp, 0)
                inits.insert(0, ('parent_interp', parent_interp))
        
        # Optimize top initializations (by initial score)
        scored_inits = []
        for label, f_init in inits:
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            S = float(np.sum(f_init))
            if S < 0.01:
                continue
            f_init = f_init / S
            sc = evaluate_sequence(f_init.tolist())
            if np.isfinite(sc):
                scored_inits.append((sc, label, f_init))
        
        scored_inits.sort(key=lambda x: x[0])
        
        # Budget per init
        n_good = min(len(scored_inits), 6)  # optimize top 6
        per_init_time = n_budget / max(n_good, 1) * 0.85
        
        for rank in range(n_good):
            sc, label, f_init = scored_inits[rank]
            init_deadline = min(n_deadline, time.monotonic() + per_init_time)
            
            if time.monotonic() > init_deadline - 3:
                break
            
            if sc < best_score:
                best_score = sc
                best_sequence = f_init.copy()
                print(f"  [n={n} {label}] init score={sc:.6f}")
            
            # Run combined optimizer
            if time.monotonic() < init_deadline - 5:
                try:
                    bf, bs = _combined_optimize(
                        f_init, init_deadline, N,
                        verbose=f"[n={n} {label}] "
                    )
                    if bs < best_score:
                        best_score = bs
                        best_sequence = bf.copy()
                        print(f"  [n={n} {label}] optimized to {bs:.6f}")
                except Exception as e:
                    print(f"  [n={n} {label}] error: {e}")
    
    print(f"\nAfter main search: best n={len(best_sequence) if best_sequence is not None else '?'}, score={best_score:.6f}")
    
    # Perturbation restarts
    if best_sequence is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Perturbation restarts (best: {best_score:.6f}) ===")
        n_best = len(best_sequence)
        N_best = _next_pow2(2 * n_best - 1)
        restart_budget = deadline - time.monotonic() - 15
        n_restarts = min(8, max(3, int(restart_budget / 25)))
        per_restart = restart_budget / n_restarts
        
        for r in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break
            
            # Different perturbation strategies
            strategy = r % 4
            if strategy == 0:
                # Gaussian multiplicative noise
                noise = 1.0 + 0.05 * rng.standard_normal(n_best)
                f_pert = best_sequence * np.clip(noise, 0.5, 1.5)
            elif strategy == 1:
                # Shift mass from peak contributors to others
                C = _convolve_fft(best_sequence, N_best)
                C = np.maximum(C, 0)
                k_star = int(np.argmax(C))
                f_pert = best_sequence.copy()
                # Lower elements near the peak contribution
                k_min = max(0, k_star - n_best + 1)
                k_max = min(n_best - 1, k_star)
                for j in range(k_min, k_max + 1):
                    f_pert[j] *= 0.95
                # Raise others
                mask = np.ones(n_best, dtype=bool)
                mask[k_min:k_max + 1] = False
                f_pert[mask] *= 1.02
                f_pert = np.maximum(f_pert, 0)
            elif strategy == 2:
                # Mirror partial flip
                f_pert = best_sequence.copy()
                split = n_best // 2 + rng.integers(-n_best//4, n_best//4)
                f_pert[:split] = best_sequence[:split][::-1]
            else:
                # Add a bump at random location
                f_pert = best_sequence.copy()
                center = rng.integers(n_best // 4, 3 * n_best // 4)
                width = n_best // 10
                for i in range(n_best):
                    f_pert[i] += 0.02 * np.exp(-0.5 * ((i - center) / width) ** 2)
            
            f_pert = np.maximum(f_pert, 0)
            s = f_pert.sum()
            if s < 0.01:
                continue
            f_pert = f_pert / s
            
            pert_deadline = time.monotonic() + per_restart * 0.8
            try:
                bf, bs = _combined_optimize(f_pert, pert_deadline, N_best,
                                            verbose=f"[restart {r}] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  Restart {r} (strat={strategy}): {bs:.6f}")
                else:
                    print(f"  Restart {r} (strat={strategy}): no improvement ({bs:.6f})")
            except Exception as e:
                print(f"  Restart {r} error: {e}")
    
    # Final polish on the absolute best
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print(f"\n=== Final polish (n={len(best_sequence)}, score={best_score:.6f}) ===")
        n_final = len(best_sequence)
        N_final = _next_pow2(2 * n_final - 1)
        try:
            bf, bs = _fast_optimize(
                best_sequence, deadline - 5, N_final,
                beta_start=200.0, beta_end=800.0, lr=0.005,
                max_iters=3000, verbose="[final] "
            )
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()
                print(f"  After final gradient: {best_score:.6f}")
        except Exception as e:
            print(f"  Final error: {e}")
    
    if best_sequence is None:
        best_sequence = _tukey_window(256, 0.3)
        best_score = evaluate_sequence(best_sequence.tolist())
    
    # Ensure valid
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = evaluate_sequence(best_sequence.tolist())
    
    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
