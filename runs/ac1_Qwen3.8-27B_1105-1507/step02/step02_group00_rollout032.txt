import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_obj(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k-n+1)
        je = min(n-1, k)
        A[k, js:je+1] = a[k-je:k-js+1][::-1]
    return A

        return res.x
    return None

        if g is None:

        gs = np.sum(g)
        if gs < 1e-10:
            break
        g = g / gs
        # Blend
        t = 0.5 + 0.3 * (it / max_iter)  # Start with 0.5, increase to 0.8
        a_new = (1-t)*a + t*g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a = a_new / s
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    return best_a, best_obj

def smooth_max_obj_grad(z, T, n):
    """Objective and gradient for smooth max optimization.
    z is the log-parameter, a = softmax(z)."""
    # Compute a = softmax(z)
    z_shift = z - np.max(z)
    ez = np.exp(z_shift)
    a = ez / np.sum(ez)
    
    # Compute convolution
    conv = np.convolve(a, a)
    
    # Smooth max
    c = np.max(conv)
    scaled = (conv - c) / T
    # Clip for numerical stability
    scaled = np.clip(scaled, -50, 50)
    log_sum = T * np.log(np.sum(np.exp(scaled)))
    sm = c + log_sum
    
    # Weights
    p = np.exp(scaled) / np.sum(np.exp(scaled))
    
    # Gradient of smooth max w.r.t. a
    # d(sm)/da[m] = sum_k p[k] * d(conv[k])/da[m]
    # d(conv[k])/da[m] = a[k-m]*1{valid} + a[m-k]*1{valid}
    grad_a = np.zeros(n)
    for m in range(n):
        val = 0.0
        for k in range(2*n-1):
            pk = p[k]
            if pk < 1e-15:
                continue
            km = k - m
            mk = m - k
            if 0 <= km < n:
                val += pk * a[km]
            if 0 <= mk < n:
                val += pk * a[mk]
        grad_a[m] = val
    
    objective = 2.0 * n * sm
    grad_obj_a = 2.0 * n * grad_a
    
    # Chain rule for softmax: da[m]/dz[l] = a[m]*(delta_ml - a[l])
    # d_obj/dz[l] = sum_m grad_obj_a[m] * a[m] * (delta_ml - a[l])
    #             = grad_obj_a[l] * a[l] - a[l] * sum_m grad_obj_a[m] * a[m]
    sum_ga_a = np.dot(grad_obj_a, a)
    grad_z = a * (grad_obj_a - sum_ga_a)
    
    return objective, grad_z

```python
def smooth_obj_grad(z, T, n):
    """Compute smooth max objective and gradient w.r.t. z (log-parameters)."""
    # a = softmax(z)
    z_shift = z - np.max(z)
    ez = np.exp(z_shift)
    a = ez / np.sum(ez)
    
    # conv(a, a)
    conv = np.convolve(a, a)
    
    # Smooth max with temperature T
    c = np.max(conv)
    scaled = (conv - c) / T
    scaled = np.clip(scaled, -100, 100)
    exp_scaled = np.exp(scaled)
    sum_exp = np.sum(exp_scaled)
    p = exp_scaled / sum_exp
    sm = c + T * np.log(sum_exp)
    
    # Gradient w.r.t. a
    grad_a = 2.0 * np.correlate(p, a, mode='full')[n-1 : 2*n-1]
    
    # Objective and gradient w.r.t. z
    obj = 2.0 * n * sm
    grad_obj_a = 2.0 * n * grad_a
    
    # Chain rule: ∂obj/∂z = diag(a) @ (grad_obj_a - sum(grad_obj_a * a))
    # Because ∂a[m]/∂z[l] = a[m]*(δ_ml - a[l])
    sum_ga = np.dot(grad_obj_a, a)
    grad_z = a * (grad_obj_a - sum_ga)
    
    return obj, grad_z
```

"""
Multi-phase optimization: LP alternating + smooth-max L-BFGS-B polishing.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    """Build constraint matrix for LP: A[k,j] = a[k-j] if valid else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je + 1] = a[k - je:k - js + 1][::-1]
    return A

def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. A@b_ub constraints, g>=0."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0.0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_alternate(n, a_init, max_iter=200, time_deadline=None):
    """LP alternating optimization."""
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        if time_deadline is not None and time.monotonic() > time_deadline:
            break
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        g = solve_lp(a, M)
        if g is None:
            break
        gs = np.sum(g)
        if gs < 1e-10:
            break
        g = g / gs
        # Adaptive mixing: start more aggressive, become more conservative
        progress = it / max(max_iter - 1, 1)
        t = 0.3 + 0.5 * progress
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a = a_new / s
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    return best_a, best_obj

def smooth_obj_grad(z, T, n):
    """Smooth max objective and gradient for L-BFGS-B."""
    z_shift = z - np.max(z)
    ez = np.exp(z_shift)
    a = ez / np.sum(ez)
    
    conv = np.convolve(a, a)
    c = np.max(conv)
    scaled = (conv - c) / T
    scaled = np.clip(scaled, -100, 100)
    exp_scaled = np.exp(scaled)
    sum_exp = np.sum(exp_scaled)
    p = exp_scaled / sum_exp
    sm = c + T * np.log(sum_exp)
    
    grad_a = 2.0 * np.correlate(p, a, mode='full')[n - 1:2 * n - 1]
    
    obj = 2.0 * n * sm
    grad_obj_a = 2.0 * n * grad_a
    sum_ga = np.dot(grad_obj_a, a)
    grad_z = a * (grad_obj_a - sum_ga)
    
    return obj, grad_z

def smooth_max_optimize(a_init, n, T_start, T_end, max_iter=100, time_deadline=None):
    """Optimize using smooth max with L-BFGS-B and temperature annealing."""
    # Start from z where softmax(z) = a_init
    # a_init = exp(z) / sum(exp(z))
    # We can set z = log(a_init) (with a small offset for zeros)
    z0 = np.log(np.maximum(a_init, 1e-10))
    
    best_a = a_init.copy()
    best_obj = compute_obj(best_a)
    
    # Anneal temperature
    n_phases = 3
    for phase in range(n_phases):
        if time_deadline is not None and time.monotonic() > time_deadline:
            break
        T = T_start * (T_end / T_start) ** (phase / max(n_phases - 1, 1))
        iters_per_phase = max_iter // n_phases
        
        def fun(z):
            obj, _ = smooth_obj_grad(z, T, n)
            return obj
        
        def fun_grad(z):
            _, grad = smooth_obj_grad(z, T, n)
            return grad
        
        res = minimize(fun, z0, jac=fun_grad, method='L-BFGS-B',
                       options={'maxiter': iters_per_phase, 'ftol': 1e-12})
        
        if res.success or res.fun < best_obj:
            z0 = res.x
            z_shift = z0 - np.max(z0)
            ez = np.exp(z_shift)
            a_new = ez / np.sum(ez)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
    
    return best_a, best_obj

def subgradient_polish(a, n_steps=2000, time_deadline=None):
    """Subgradient method for final polishing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for step in range(n_steps):
        if time_deadline is not None and time.monotonic() > time_deadline:
            break
        h = np.convolve(a, a)
        max_h = np.max(h)
        if max_h < 1e-15:
            break
        
        # Active set
        tol = max_h * (0.001 + 0.01 * (step / n_steps))
        active = h >= max_h - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            active = np.zeros(len(h), dtype=bool)
            active[np.argmax(h)] = True
            n_active = 1
        
        # Subgradient
        p = np.zeros(len(h))
        p[active] = 1.0 / n_active
        grad_a = 2.0 * np.correlate(p, a, mode='full')[n - 1:2 * n - 1]
        
        # Adaptive step size
        step_size = 1.0 / (1.0 + 0.001 * step)
        a_new = a - step_size * grad_a
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_obj

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 8))
    if choice == 0:
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0, np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
    elif choice == 1:
        mid = n // 2
        a = np.ones(n)
        a[:mid] = rng.uniform(0.3, 1.5)
        a[mid:] = rng.uniform(0.3, 1.5)
    elif choice == 2:
        period = int(rng.integers(max(2, n // 20), max(3, n // 3)))
        pattern = rng.random(period)
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 3:
        width = rng.uniform(0.1, 0.5)
        a = np.exp(-((np.arange(n) / n - 0.5) / width) ** 2)
    elif choice == 4:
        a = np.abs(np.sin(np.arange(n) * rng.uniform(0.1, 2.0)))
    elif choice == 5:
        # Power law
        a = np.power(np.arange(n) + 1, rng.uniform(-1, 1))
    elif choice == 6:
        # Two peaks
        p1, p2 = int(rng.integers(0, n//2)), int(rng.integers(n//2, n))
        w = max(1, n // 10)
        a = np.exp(-((np.arange(n) - p1) / w)**2) + np.exp(-((np.arange(n) - p2) / w)**2)
    else:
        a = rng.random(n)
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s
