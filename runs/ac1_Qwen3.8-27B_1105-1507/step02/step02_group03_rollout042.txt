import numpy as np
from scipy.optimize import linprog
import time

def compute_score(f):
    """Compute score for non-negative sequence f."""
    n = len(f)
    s = np.sum(f)
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * np.max(conv) / (s * s)

def normalize(f):
    """Normalize to sum=1."""
    s = np.sum(f)
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s

def build_conv_matrix(f):
    """Build matrix A where A @ g = np.convolve(f, g)."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[k, j] = f[k - j]
    return A

def spectral_flat(f, n_iter=200, rng=None):
    """Spectral flattening via alternating projections."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = np.array(f, dtype=np.float64).copy()
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()
    
    best_f = f.copy()
    best_score = compute_score(f)
    
    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Target: equal energy across all non-DC bins
        # Total signal energy = sum(f^2) = (1/n_fft) * sum(|F|^2)
        total_energy = np.sum(f ** 2)
        n_bins = len(mag)
        # If all non-DC bins had equal magnitude:
        # n_fft * total_energy = 1^2 + (n_bins-1) * target_mag^2
        # target_mag^2 = (n_fft * total_energy - 1) / (n_bins - 1)
        target_mag_sq = (n_fft * total_energy - 1.0) / (n_bins - 1)
        if target_mag_sq < 0:
            target_mag_sq = total_energy * n_fft / n_bins
        target_mag = np.sqrt(max(0, target_mag_sq))
        
        mag_new = np.full(n_bins, target_mag)
        mag_new[0] = 1.0  # DC = sum(f) = 1
        
        # Add small randomization to escape local minima

            best_f = f_new.copy()
            f = f_new
        else:

            f = f_new  # Still update to continue the iteration
    
    return best_f, best_score

def lp_refine(f, time_budget=300, t0=0.5, rng=None, verbose=True):

    LP-based refinement: repeatedly find g maximizing sum(g)
    subject to conv(f,g) <= M, g >= 0, then blend.
    """
    if rng is None:

    n = len(f)

    iter_count = 0
    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = np.max(conv)
        
        # Build constraint matrix
        A = build_conv_matrix(f)
        
        # LP: maximize sum(g) s.t. A@g <= M, g >= 0

        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,

        except Exception:
            break
        
        if not res.success or res.x is None:
            # Try with tighter constraint

            try:

            except Exception:
                break
            if not res.success or res.x is None:
                break

def coordinate_descent(f, n_sweeps=5, rng=None):

    if rng is None:

    f = np.array(f, dtype=np.float64).copy()
    f = normalize(f)

        order = rng.permutation(n)
        for i in order:
            orig = f[i]
            if orig < 1e-12:
                continue
            best_val = orig
            best_local = best_score
            # Try various multipliers
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                f[i] = max(0.0, orig * mult)

                    f[i] = orig
                    continue
                score = compute_score(f)
                if score < best_local:
                    best_local = score
                    best_val = f[i]
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
    
    f = normalize(f)
    return f, best_score

def pressure_update(f, n_iter=200, eta=0.1, rng=None):
    """Pressure-based update to reduce convolution peaks."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = np.array(f, dtype=np.float64).copy()
    f = normalize(f)
    n = len(f)

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = np.max(conv)
        
        # Compute pressure from all peaks (not just max)
        # Weight each position by how much it exceeds the mean
        mean_conv = np.mean(conv)
        excess = np.maximum(conv - mean_conv, 0)
        excess = excess / (excess.max() + 1e-15)
        
        # Pressure on f[i] = sum over k of excess[k] * f[k-i] (where valid)
        pressure = np.zeros(n)
        for i in range(n):
            for k in range(max(0, i), min(2*n-1, i+n)):
                j = k - i
                pressure[i] += excess[k] * f[j]
        
        if pressure.max() > 0:
            pressure = pressure / pressure.max()
        
        eta_eff = eta / (1 + 0.05 * it)
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)
        
        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
        else:
            f = f_new  # Still update to continue
    
    return best_f, best_score

**Idea: Use the dual of the LP to identify critical constraints**

The dual LP is:
minimize M * sum_k y_k
subject to: sum_k A[k,j] * y_k >= 1 for all j
             y_k >= 0

The optimal dual solution y* tells us which convolution positions are "tight" (y*_k > 0). These are the positions we need to worry about.

**Idea: Use a "shrinkage" approach**

Instead of the LP, we could directly optimize:
minimize max_k (sum_i f[i]*f[k-i])
subject to: sum(f) = 1, f >= 0

This is a non-convex problem, but we can use a "bisection + feasibility" approach:
- For a given target T, check if there exists f with sum=1, f>=0, and conv(f,f)[k] <= T for all k.
- The feasibility check is: find f with sum=1, f>=0, and for all k, sum_i f[i]*f[k-i] <= T.

The constraint sum_i f[i]*f[k-i] <= T is non-linear (quadratic) in f. But if we fix an upper bound B on each f[i], then sum_i f[i]*f[k-i] <= B * sum_i f[i] = B (for the overlapping part). So if B <= T, the constraint is automatically satisfied.

This gives us a simple sufficient condition: if max(f) <= T, then conv(f,f) <= T (for positions with full overlap). But this is very conservative.

**Idea: Use a "convex relaxation" of the problem**

Introduce variables y[i,j] = f[i]*f[j]. Then:
- conv(f,f)[k] = sum_{i+j=k} y[i,j] (linear in y)
- y[i,j] = f[i]*f[j] (non-linear, but we can relax to y[i,j] <= (f[i]+f[j])/2)
- The relaxation y[i,j] <= (f[i]+f[j])/2 is valid by AM-GM: f[i]*f[j] <= (f[i]+f[j])/2

With this relaxation, the problem becomes:
minimize T
subject to: sum_j y[i,j] + sum_j y[j,i] <= T for all k (not quite right)

Actually this is getting complicated. Let me stick with the LP approach but make it more effective.

**Key improvement: Use "increasing target" LP**

Instead of conv(f,g) <= M (the current max), use:
conv(f,g) <= M * (1 - epsilon)

This forces g to be strictly better than f. If no feasible g exists (which can happen if f is already very flat), reduce epsilon.

Actually, the LP as stated (maximize sum(g) s.t. conv(f,g) <= M, g >= 0) will always have a feasible solution (g=0 is feasible, and g=f is feasible since conv(f,f) <= M). The optimal solution will have sum(g) >= sum(f) = 1 (since f is feasible and has sum 1). So the LP is always feasible and the optimal g will have sum >= 1.

After normalizing g to sum 1, we get a new sequence that, when convolved with f, has max <= M * (1/sum(g_original)) < M (if sum(g_original) > 1). Then the blend (1-t)f + tg will have even smaller max convolution.

    """LP refinement using only top-k convolution positions as constraints."""
    n = len(f)
    f = np.array(f, dtype=np.float64).copy()
    f = normalize(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    t = t0
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = np.max(conv)
        
        # Get top-k positions
        top_positions = np.argsort(conv)[::-1][:top_k]
        
        # Build sparse constraint matrix
        n_constraints = len(top_positions)
        A_sparse = np.zeros((n_constraints, n))
        for idx, k in enumerate(top_positions):
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                A_sparse[idx, j] = f[k - j]
        
        c_obj = -np.ones(n)
        b_ub = np.full(n_constraints, M)
        bounds = [(0, None)] * n
        
        res = linprog(c_obj, A_ub=A_sparse, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        
        if not res.success or res.x is None:
            break
        
        g = np.clip(res.x, 0, None)
        g_sum = np.sum(g)
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Line search
        best_alpha = 0
        best_trial = best_score
        for alpha in [t, t*0.5, t*0.25, 0.1, 0.05]:
            f_try = (1-alpha)*f + alpha*g
            f_try = np.clip(f_try, 0, None)
            s = np.sum(f_try)
            if s < 1e-10:
                continue
            f_try = f_try / s
            sc = compute_score(f_try)
            if sc < best_trial:
                best_trial = sc
                best_alpha = alpha
        
        if best_alpha > 0:
            f = f_try
            best_f = f.copy()
            best_score = best_trial
            t = min(1.0, t * 1.05)
        else:
            t *= 0.5
            if t < 0.01:
                break
    
    return best_f, best_score

"""
Optimized step function search using LP-based refinement with sparse constraints,
spectral initialization, pressure updates, and adaptive coordinate descent.
Focuses on aggressive LP iteration with line search and multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute score for non-negative sequence f: 2*n*max(conv(f,f))/sum(f)^2."""
    n = len(f)
    s = np.sum(f)
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * np.max(conv) / (s * s)


def normalize(f):
    """Normalize to sum=1."""
    s = np.sum(f)
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s


def build_conv_matrix(f):
    """Build matrix A where A @ g = np.convolve(f, g) (full mode)."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_start < j_end:
            A[k, j_start:j_end] = f[k - j_start: k - j_start + (j_end - j_start)][::-1]
    return A


def build_sparse_conv_matrix(f, top_positions):
    """Build constraint matrix using only specified convolution positions."""
    n = len(f)
    A = np.zeros((len(top_positions), n))
    for idx, k in enumerate(top_positions):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        if j_start < j_end:
            A[idx, j_start:j_end] = f[k - j_start: k - j_start + (j_end - j_start)][::-1]
    return A


def spectral_flat(f, n_iter=200, rng=None):
    """Spectral flattening via alternating projections onto flat spectrum + non-negativity."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = np.array(f, dtype=np.float64).copy()
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()

    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)

        # Target: equal energy in non-DC bins
        total_energy = np.sum(f ** 2)
        n_bins = len(mag)
        target_mag_sq = (n_fft * total_energy - 1.0) / max(1, n_bins - 1)
        if target_mag_sq < 0:
            target_mag_sq = total_energy * n_fft / n_bins
        target_mag = np.sqrt(max(0.0, target_mag_sq))

        mag_new = np.full(n_bins, target_mag)
        mag_new[0] = 1.0

        # Periodic phase perturbation to escape local minima
        if it % 25 == 24:
            phase = phase + rng.normal(0, 0.02, n_bins)

        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        f_new = normalize(f_new)

        score = compute_score(f_new)
        if score < best_score - 1e-10:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            f = f_new
            stall += 1
            if stall > 150:
                break

    return best_f, best_score


def pressure_update(f, n_iter=200, eta=0.1, rng=None):
    """Pressure-based update: reduce elements contributing to convolution peaks."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = np.array(f, dtype=np.float64).copy()
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = np.max(conv)

        # Use top-5 peaks for pressure computation
        top_k = min(5, len(conv))
        top_positions = np.argsort(conv)[::-1][:top_k]

        pressure = np.zeros(n)
        for k in top_positions:
            weight = (conv[k] - M * 0.8) / (M + 1e-15)
            weight = max(0, weight)
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            for j in range(j_start, j_end):
                pressure[j] += weight * f[k - j]

        if pressure.max() > 0:
            pressure = pressure / pressure.max()

        eta_eff = eta / (1.0 + 0.03 * it)
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
        else:
            f = f_new

    return best_f, best_score


def lp_refine(f, time_budget=300, t0=0.4, rng=None, top_k=80, verbose=True):
    """
    LP-based refinement with sparse constraints.
    Repeatedly solve: max sum(g) s.t. conv(f,g)[top_k positions] <= M, g >= 0.
    Use line search for step size.
    """
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(f)
    f = np.array(f, dtype=np.float64).copy()
    f = normalize(f)

    best_f = f.copy()
    best_score = compute_score(f)
    t = t0
    start = time.monotonic()
    iter_count = 0
    no_improve = 0

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = np.max(conv)

        # Use top-k positions for constraints
        k_use = min(top_k, 2 * n - 1)
        top_positions = np.argsort(conv)[::-1][:k_use]

        # Build sparse constraint matrix
        A = build_sparse_conv_matrix(f, top_positions)
        c_obj = -np.ones(n)
        b_ub = np.full(k_use, M)
        bounds = [(0, None)] * n

        remaining_time = time_budget - (time.monotonic() - start)
        lp_time = max(0.5, min(5.0, remaining_time * 0.3))

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': lp_time})
        except Exception:
            break

        if not res.success or res.x is None:
            # Try with slightly tighter bound
            try:
                b_ub_tight = np.full(k_use, M * 0.995)
                res = linprog(c_obj, A_ub=A, b_ub=b_ub_tight, bounds=bounds,
                              method='highs', options={'presolve': True, 'time_limit': lp_time})
            except Exception:
                break
            if not res.success or res.x is None:
                no_improve += 1
                if no_improve > 5:
                    break
                continue

        g = np.clip(res.x, 0, None)
        g_sum = np.sum(g)
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Verify that g actually helps (check full convolution bound)
        conv_fg = np.convolve(f, g)
        if np.max(conv_fg) > M * 1.01:
            # g violates some non-top-k constraint, scale it down
            scale = M / np.max(conv_fg) * 0.99
            g = g * scale
            g_sum = np.sum(g)
            if g_sum < 1e-15:
                break
            g = g / g_sum

        # Line search for optimal step size
        best_alpha = 0.0
        best_trial_score = best_score
        alphas = [min(1.0, t), t * 0.6, t * 0.3, t * 0.15, 0.08, 0.04]
        alphas = sorted(set([a for a in alphas if 0 < a <= 1.0]), reverse=True)

        f_best_trial = None
        for alpha in alphas:
            f_trial = (1.0 - alpha) * f + alpha * g
            f_trial = np.clip(f_trial, 0, None)
            s = np.sum(f_trial)
            if s < 1e-10:
                continue
            f_trial = f_trial / s
            trial_score = compute_score(f_trial)
            if trial_score < best_trial_score - 1e-14:
                best_trial_score = trial_score
                best_alpha = alpha
                f_best_trial = f_trial

        if best_alpha > 0 and f_best_trial is not None:
            f = f_best_trial
            best_f = f.copy()
            best_score = best_trial_score
            t = min(1.0, t * 1.08)
            no_improve = 0
            iter_count += 1
            if verbose and iter_count % 5 == 0:
                elapsed = time.monotonic() - start
                print(f"    LP iter {iter_count}: score={best_score:.8f}, t={t:.3f}, elapsed={elapsed:.1f}s")
        else:
            t *= 0.5
            no_improve += 1
            if t < 0.005:
                # Try perturbation to escape
                f_pert = f * (1.0 + 0.05 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                pert_score = compute_score(f_pert)
                if pert_score < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = pert_score
                    t = 0.15
                    no_improve = 0
                    if verbose:
                        print(f"    LP perturb: score={best_score:.8f}")
                else:
                    break

    return best_f, best_score


def coordinate_descent(f, n_sweeps=5, rng=None):
    """Coordinate descent for fine-tuning individual elements."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = np.array(f, dtype=np.float64).copy()
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        improved_this_sweep = 0
        for i in order:
            orig = f[i]
            if orig < 1e-12:
                continue
            best_val = orig
            best_local = best_score
            for mult in [0.4, 0.6, 0.8, 0.9, 1.1, 1.25, 1.5]:
                f[i] = max(0.0, orig * mult)
                s = np.sum(f)
                if s < 1e-10:
                    f[i] = orig
                    continue
                score = compute_score(f)
                if score < best_local:
                    best_local = score
                    best_val = f[i]
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
                improved_this_sweep += 1

        f = normalize(f)
        if improved_this_sweep == 0:
            break

    return f, best_score


def diverse_init(n, rng):
    """Generate a diverse initial sequence."""
    r = rng.random()
    if r < 0.2:
        f = np.ones(n) * (1 + 0.08 * rng.standard_normal(n))
    elif r < 0.4:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.2, 0.4)
        f = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    elif r < 0.6:
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n)) + 0.05
    elif r < 0.8:
        f = np.ones(n)
        for _ in range(4):
            c = int(rng.integers(0, n))
            w = int(max(3, rng.integers(3, n // 3)))
            kernel = np.exp(-0.5 * (np.arange(n) - c) ** 2 / max(1, w ** 2))
            f = f * (0.4 + 0.6 * kernel) + 0.05
    else:
        f = np.abs(rng.standard_normal(64)) + 0.1
        f = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, 64), f)

    f = np.clip(f, 0.01, None)
    f = normalize(f)
    f = f * (1 + 0.03 * rng.standard_normal(n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f


def multi_scale_init(n, rng):
    """Create initial sequence by optimizing a small one and upscaling."""
    n_small = 32
    f_small = np.ones(n_small) * (1 + 0.15 * rng.standard_normal(n_small))

    f_small = normalize(f_small)

    f_large = np.clip(f_large, 0.005, None)

            remaining = deadline - time.monotonic()
            if remaining > 5:
                f_cd2, s_cd2 = coordinate_descent(best_overall, n_sweeps=3, rng=rng)
                if s_cd2 < best_score_overall:
                    best_score_overall = s_cd2
                    best_overall = f_cd2.copy()
                    print(f"  Final CD: {s_cd2:.8f}")

    # Verify with actual evaluate_sequence
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.8f}, n={len(final_list)}")
        print(f"{'='*50}")
        return final_list

    # Fallback
    print("Using fallback uniform sequence")
    return [1.0] * 512
