def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)

def build_conv_matrix(a):
    """Build matrix A where (A @ x) gives full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))

        jh = min(n - 1, k)
        if jl <= jh:

    return A

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using a multi-phase approach:
Phase 1: LP best-response with backtracking line search (multiple starts)
Phase 2: Gradient descent on log-sum-exp smoothed objective (FFT-accelerated)
Phase 3: Perturbation + re-optimization to escape local optima
Phase 4: Final polish with aggressive smooth gradient descent
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build matrix A where (A @ x) gives full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None and res.x.sum() > 1e-10:
        return res.x
    return None


def lp_iteration(a, max_iters, deadline):
    """LP best-response with backtracking line search."""
    best_local = a.copy()
    best_score = compute_score(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 1e-6, None)
    a = a / a.sum()

    stall = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M)
        if g is None:
            break
        g = g / g.sum()

        # Check convergence
        if np.max(np.abs(g - a)) < 1e-7:
            break

        # Backtracking line search
        t = 0.8
        improved = False
        for _ in range(15):
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0, None)
            s = a_new.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_new = a_new / s
            score = compute_score(a_new)
            if score < best_score - 1e-12:
                best_score = score
                best_local = a_new.copy()
                a = a_new
                improved = True
                break
            t *= 0.5

        if not improved:
            stall += 1
            if stall > 30:
                break
        else:
            stall = 0

    return best_local, best_score


def smooth_gradient_step(a, beta):
    """One gradient step on the log-sum-exp smoothed objective. Returns (new_a, score_estimate)."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()

    # Softmax weights with numerical stability
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -100, 100)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return a, 0.0
    w = w / w_sum

    # Gradient: 2n * (A^T @ w)
    # (A^T @ w)_i = sum_{j=0}^{n-1} w[i+j] * a[j]
    # Vectorized using index matrix
    idx = np.arange(n)
    W_mat = w[idx[:, None] + idx[None, :]]
    Aw = W_mat @ a

    grad = 2.0 * n * Aw
    return grad, w


def smooth_grad_descent(a, deadline, betas=(50, 200, 800, 3000, 10000), max_iters_per_beta=5000):
    """Gradient descent on log-sum-exp smoothed objective with momentum."""
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)

    for beta in betas:
        if time.monotonic() > deadline:
            break

        a_work = best.copy()
        a_work = np.clip(a_work, 1e-6, None)
        a_work = a_work / a_work.sum()

        # Adaptive learning rate
        lr = 4.0 / (beta + 1)
        momentum = np.zeros(n)
        mu = 0.9  # momentum coefficient

        stalled = 0
        local_best_score = compute_score(a_work)
        local_best = a_work.copy()

        for it in range(max_iters_per_beta):
            if time.monotonic() > deadline:
                break

            grad, w = smooth_gradient_step(a_work, beta)
            if not np.all(np.isfinite(grad)):
                break

            # Momentum update
            momentum = mu * momentum - lr * grad
            a_new = a_work + momentum
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-10:
                break
            a_new = a_new / s

            a_work = a_new

            # Periodically check score
            if it % 5 == 0:
                score = compute_score(a_work)
                if score < local_best_score:
                    local_best_score = score
                    local_best = a_work.copy()
                    stalled = 0
                    if score < best_score:
                        best_score = score
                        best = a_work.copy()
                else:
                    stalled += 1
                    if stalled > 200:
                        break

            # Decay learning rate
            lr *= 0.9998

        # End of beta phase: keep best
        if local_best_score < best_score:
            best_score = local_best_score
            best = local_best.copy()

    return best, best_score


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    t = np.linspace(0, 1, n)

    # Parent resampled
    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        resampled = np.maximum(np.interp(t, t_old, parent), 0)
        if resampled.sum() > 0.01:
            inits.append(resampled / resampled.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine harmonics
    for k in [1, 2, 3, 4, 6, 8]:
        c = 1.0 + 0.4 * np.cos(2 * np.pi * k * t)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    # Sine
    c = 1.0 + 0.3 * np.sin(2 * np.pi * t)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())

    # Bimodal
    c = 1.0 + 0.5 * np.cos(4 * np.pi * t) ** 2
    inits.append(c / c.sum())

    # Triangular
    c = 1.0 - np.abs(t - 0.5) * 1.5
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())

    # Random
    inits.append(rng.uniform(0.2, 1.0, n) / n)

    # Sparse-ish
    c = (rng.uniform(0, 1, n) > 0.3).astype(float) + 0.1
    inits.append(c / c.sum())

    return inits
