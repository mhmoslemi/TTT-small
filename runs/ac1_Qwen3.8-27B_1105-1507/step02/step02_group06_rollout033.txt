"""
LP-based and spectral optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses alternating LP direction finding with line search, spectral flattening
iterations, and coordinate descent polish across multiple sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, csc_matrix


def compute_score(f):
    """Compute the objective value directly."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def build_conv_matrix(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    # A has shape (2n-1, n)
    # A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    # Total nonzeros: sum_{k=0}^{2n-2} min(k+1, n, 2n-1-k) = n^2
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def solve_lp_direction(f, S):
    """Solve LP: minimize t s.t. conv(f,g)_k <= t, sum(g) = S, g >= 0.
    Returns optimal g and t."""
    n = len(f)
    A_conv = build_conv_matrix(f)
    
    # Variables: g_0, ..., g_{n-1}, t (total n+1 variables)
    # Objective: minimize t -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: conv(f,g)_k <= t -> conv(f,g)_k - t <= 0
    # A_ub has shape (2n-1, n+1)
    A_ub = np.hstack([A_conv.toarray(), -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    # Bounds: g_i >= 0, t >= 0
    bounds = [(0, None)] * (n + 1)
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                        bounds=bounds, method='highs')
        if result.success:
            return result.x[:n], result.x[-1]
    except Exception:
        pass
    return None, None

Let me try a different formulation that might be more amenable to efficient solving:

**Formulation: Proximal gradient**
- Objective: F(g) = max_k (f*g)_k (convex in g since f >= 0)
- Constraint: sum(g) = S, g >= 0
- We want to minimize F(g) subject to the constraints.

This is a convex problem! (f*g)_k = sum_j f_j * g_{k-j} is linear in g (since f is fixed), so max of linear functions is convex.

So we can solve this with proximal gradient or projected subgradient:
1. Compute g_k* = argmax_k (f*g)_k
2. Compute subgradient: s = A[k*, :] (the k* row of the convolution matrix)
3. Take step: g_new = g - lr * s
4. Project onto {g >= 0, sum(g) = S}

The projection onto {g >= 0, sum(g) = S} is:
- Find threshold τ such that sum(max(g - lr*s - τ, 0)) = S
- g_projected = max(g - lr*s - τ, 0)

This bisection on τ takes O(log(1/ε) * n) time.

This is much cheaper than solving an LP! And it's a standard convex optimization algorithm.

Let me use this as the primary method:

```python
def projected_subgradient(f, S, max_iter=1000, lr_init=0.1):
    """Minimize max_k (f*g)_k s.t. sum(g)=S, g>=0 using projected subgradient."""
    n = len(f)
    g = np.ones(n) * S / n  # Initialize
    lr = lr_init
    best_g = g.copy()
    best_val = float(np.max(np.convolve(f, g)))
    
    for it in range(max_iter):
        conv = np.convolve(f, g)
        M = float(np.max(conv))
        if M < best_val:
            best_val = M
            best_g = g.copy()
        
        k_star = int(np.argmax(conv))
        # Subgradient: f[k_star - j] for j where 0 <= k_star - j < n
        s = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                s[j] = f[idx]
        
        # Step
        g_new = g - lr * s
        # Project onto {g >= 0, sum(g) = S}
        g_new = project_sum(g_new, S)
        g = g_new
        lr *= 0.99  # Decay learning rate
    
    return best_g

```python
def project_sum(x, S):
    """Project x onto {g >= 0, sum(g) = S}."""
    # Find τ such that sum(max(x - τ, 0)) = S
    # Use bisection
    lo = np.min(x) - S  # If τ < min(x), all components are positive, sum = sum(x) - n*τ
    hi = np.max(x)
    
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.maximum(x - mid, 0)
        s = np.sum(proj)
        if s > S:
            lo = mid
        else:
            hi = mid
    
    τ = (lo + hi) / 2
    return np.maximum(x - τ, 0)
```

For our problem:
- a_k = row k of the convolution matrix (as a vector in R^n)
- ||a_k|| = sqrt(sum_j f_{k-j}^2) (over valid j)
- L = max_k ||a_k||

The acceleration factor gives us O(1/T^2) convergence, which is much better than O(1/T) for vanilla subgradient.

def nesterov_projected_gradient(f, S, max_iter=5000):
    """Nesterov accelerated projected gradient for min max_k (f*g)_k s.t. sum(g)=S, g>=0."""
    n = len(f)
    g = np.ones(n) * S / n
    y = g.copy()
    
    # Compute L (Lipschitz constant)
    # L = max_k ||a_k|| where a_k is the k-th row of conv matrix
    # a_k[j] = f[k-j] if 0 <= k-j < n
    # ||a_k||^2 = sum_j f[k-j]^2 (over valid j) = sum of f squared in a window
    # The max is achieved at k where the window covers the most "energy" of f
    f_sq = f ** 2
    # For each k, ||a_k||^2 = sum of f_sq over the valid range
    # This is a sliding window sum
    cumsum = np.cumsum(f_sq)
    L_sq = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        L_sq[k] = cumsum[j_max] - (cumsum[j_min - 1] if j_min > 0 else 0)
    L = np.sqrt(np.max(L_sq))
    
    lr = 1.0 / L
    
    best_g = g.copy()
    best_val = float(np.max(np.convolve(f, g)))
    
    for t in range(max_iter):
        conv = np.convolve(f, y)
        M = float(np.max(conv))
        k_star = int(np.argmax(conv))
        
        if M < best_val:
            best_val = M
            best_g = y.copy()
        
        # Subgradient at y
        s = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                s[j] = f[idx]
        
        # Nesterov update
        beta = (1 + np.sqrt(1 + 4 * (t - 1) ** 2)) / 2 if t > 0 else 1
        g_new = y - lr * s
        g_new = project_sum(g_new, S)
        y = g_new + ((t - 1) / beta) * (g_new - g) if t > 0 else g_new
        g = g_new
    
    return best_g

"""
Multi-method optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses projected subgradient with adaptive steps, LP-based direction finding,
spectral flattening, and coordinate descent across multiple sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def project_simplex(x, S):
    """Project x onto {g >= 0, sum(g) = S}."""
    n = len(x)
    # Use the standard algorithm
    if np.sum(np.maximum(x, 0)) < S:
        # All would be zero, distribute evenly
        return np.full(n, S / n)
    
    mu_hi = np.max(x)
    mu_lo = np.min(x)
    
    for _ in range(100):
        mu = (mu_hi + mu_lo) / 2
        proj = np.maximum(x - mu, 0)
        s = np.sum(proj)
        if abs(s - S) < 1e-12:
            break
        if s > S:
            mu_lo = mu
        else:
            mu_hi = mu
    
    return np.maximum(x - (mu_hi + mu_lo) / 2, 0)


def subgradient_of_max_conv(f, g):
    """Compute a subgradient of max_k (f*g)_k with respect to g."""
    conv = np.convolve(f, g)
    k_star = int(np.argmax(conv))
    n = len(f)
    s = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            s[j] = f[idx]
    return s, conv, k_star


def projected_subgradient_optimize(f, S, deadline, verbose_prefix=""):
    """Optimize g to minimize max_k (f*g)_k using projected subgradient."""
    n = len(f)
    g = np.ones(n) * S / n
    best_g = g.copy()
    best_score = compute_score(g)
    
    # Compute Lipschitz constant
    f_sq = f ** 2
    cumsum_f_sq = np.concatenate([[0], np.cumsum(f_sq)])
    L_sq = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        L_sq[k] = cumsum_f_sq[j_max + 1] - cumsum_f_sq[j_min]
    L = np.sqrt(np.max(L_sq)) + 1e-10
    
    lr = S / L  # Initial step size
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        s, conv, k_star = subgradient_of_max_conv(f, g)
        M = float(np.max(conv))
        current_score = compute_score(g)

            best_g = g.copy()
            no_improve = 0

            no_improve += 1

        g_new = project_simplex(g_new, S)

def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int64)
    cols = np.empty(n * n, dtype=np.int64)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def lp_optimize(f, S, deadline, verbose_prefix=""):

def solve_lp_min_peak(f, S):
    """Solve LP: min t s.t. conv(f,g)_k <= t, sum(g)=S, g>=0."""
    n = len(f)
    A_conv = build_conv_matrix_sparse(f)
    
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub: shape (2n-1, n+1)
    # conv(f,g)_k - t <= 0
    A_ub = A_conv.toarray()
    A_ub = np.hstack([A_ub, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # A_eq: sum(g) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs',
                        options={'presolve': True, 'time_limit': 60})
        if result.success:
            return result.x[:n].copy(), result.x[-1]
    except Exception:
        pass
    return None, None


def spectral_flatten(f, S, deadline, verbose_prefix=""):
    """Spectral method to flatten autocorrelation."""
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a * (S / np.sum(a))
    best_a = a.copy()
    best_score = compute_score(a)
    
    it = 0
    while time.monotonic() < deadline:
        # Compute spectrum
        # Use FFT of size 2n for linear convolution
        N = 2 * n
        A = np.fft.rfft(a, N)
        power = np.abs(A) ** 2
        
        # Compute autocorrelation from power spectrum
        R = np.fft.irfft(power, N)
        # R[0] is the autocorrelation at lag 0, R[1:n] at lags 1 to n-1, R[N-1:N-n] at negative lags
        # For real signals, R is symmetric
        
        # Target: make R[1:n] as flat as possible (equal to some target value)
        # Strategy: clip the peaks in R
        R_target = R.copy()
        max_lag = np.max(np.abs(R_target[1:n]))
        # Reduce peaks by moving energy from peak lags to flat lags
        threshold = np.mean(R_target[1:n]) + 0.5 * (max_lag - np.mean(R_target[1:n]))
        R_target[1:n] = np.minimum(R_target[1:n], threshold)
        # Also handle negative lags (by symmetry for real signals)
        R_target[N-n:N] = R_target[1:n][::-1]
        
        # Compute new spectrum
        P_target = np.fft.rfft(R_target, N)
        # Ensure non-negative (should be if R_target is a valid autocorrelation)
        # Take magnitude from target, phase from current
        A_new = np.sqrt(np.maximum(P_target, 0).real) * np.exp(1j * np.angle(A))
        
        # Inverse transform
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        a_new_sum = np.sum(a_new)
        if a_new_sum < 1e-15:
            break
        a_new = a_new * (S / a_new_sum)
        
        score = compute_score(a_new)
        a = a_new
        it += 1
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if it % 100 == 0:
                print(f"  {verbose_prefix}Spectral iter {it}: score={best_score:.6f}")
        
        if it % 1000 == 0:
            print(f"  {verbose_prefix}Spectral iter {it}: current={score:.6f}, best={best_score:.6f}")
    
    return best_a, best_score

def make_initializations(n, rng, parent=None):
    """Create diverse initializations."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    
    inits.append(('uniform', np.ones(n)))
    
    cos_t = (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0
    inits.append(('cosine', cos_t))
    
    inits.append(('ramp', np.linspace(1.0, 0.1, n)))
    inits.append(('ramp_inc', np.linspace(0.1, 1.0, n)))
    
    inits.append(('exp_decay', np.exp(-2.0 * idx / n)))
    
    center = (n - 1) / 2.0
    bell = np.exp(-0.5 * ((idx - center) / (n / 4.0)) ** 2)
    inits.append(('bell', bell))
    
    r = rng.uniform(0.1, 2.0, size=n)
    inits.append(('random', r))
    
    # W-shaped
    w = np.minimum(idx, n - 1 - idx).astype(float) / (n / 4)
    w = 1.0 + 0.5 * np.cos(np.pi * np.clip(w, 0, 1))
    inits.append(('w_shape', w))
    
    if parent is not None:
        from numpy import interp as np_interp
        if len(parent) != n:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            p = np_interp(x_new, x_old, parent)
        else:
            p = parent.copy()
        inits.append(('parent', p))
    
    return inits

For a real sequence, this IS sum_i a[i]^2 if we reindex... no, it's sum_j a[j] * a[n-1-j], which is the "inner product" of a with its reverse. By Cauchy-Schwarz, this is <= sqrt(sum a[j]^2 * sum a[n-1-j]^2) = sum a[j]^2. And by the rearrangement inequality, it's >= the inner product with any other permutation.
