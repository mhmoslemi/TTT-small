import numpy as np
from scipy.optimize import linprog, minimize, Project
import time

def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def build_conv_matrix(a):
    """Build matrix A where A @ x = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # (a * x)_k = sum_j a[k-j] * x_j = sum_j A[k,j] * x_j
    A = np.zeros((m, n))
    for k in range(m):
        # j ranges over max(0, k-n+1) to min(n-1, k)
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            # A[k, j] = a[k-j]
            idx = k - np.arange(jl, jh + 1)
            A[k, jl:jh+1] = a[idx]
    return A

def build_conv_matrix_sparse(a):
    """Build sparse convolve matrix for efficiency."""
    from scipy.sparse import diags, csr_matrix
    n = len(a)
    m = 2 * n - 1
    # Each column j has a_{k-j} for k = j, j+1, ..., j+n-1 (clamped to [0, 2n-2])
    # So column j has nonzeros at rows j to min(j+n-1, 2n-2)
    # values are a[0], a[1], ..., a[min(n-1, 2n-2-j)]
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        for k in range(k_start, k_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(m, n))
    return A

def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))

def lp_best_response(a, M, use_sparse=False):
    """Solve LP: max 1^T g s.t. conv(a,g) <= M, g >= 0. Returns (g, dual_vars)."""
    n = len(a)
    m = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(m, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        # res.ineqlin.marginals gives the dual variables (shadow prices)
        # For scipy linprog with highs, the dual variables are in res.ineqlin.marginals
        dual = np.zeros(m)
        if hasattr(res, 'ineqlin') and res.ineqlin is not None:
            if hasattr(res.ineqlin, 'marginals'):
                dual = np.abs(res.ineqlin.marginals)
        return res.x, dual
    return None, None

def lp_iteration_v2(a, max_iters, deadline, verbose=False):
    """LP best-response with backtracking line search."""
    n = len(a)
    best_local = a.copy()
    best_score = compute_score(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 1e-8, None)
    a = a / a.sum()

    prev_score = best_score
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g, dual = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()

        # Backtracking line search
        scores = []
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            score = compute_score(a_new)
            scores.append((score, t, a_new.copy()))
        
        if not scores:
            break
        
        # Pick the best score
        scores.sort(key=lambda x: x[0])
        best_score_try, best_t, best_a = scores[0]
        
        if best_score_try < best_score:
            best_score = best_score_try
            best_local = best_a.copy()
            a = best_a
            stall_count = 0
        else:
            # No improvement with any step size
            stall_count += 1
            # Try the smallest step anyway to move

            a = a / a.sum()
            if stall_count > 20:
                break

        if verbose and it % 100 == 0:
            print(f"    iter {it}: J={best_score:.6f}, t={best_t:.3f}", flush=True)

def smooth_objective(x, beta, n):
    """Log-sum-exp smoothed objective for normalized a."""
    x_pos = np.maximum(x, 1e-12)
    s = x_pos.sum()
    if s < 0.01:
        return 1e10
    x_norm = x_pos / s
    b = fast_conv(x_norm)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    log_sum = b_max + np.log(np.exp(shifted).sum()) / beta
    return float(2.0 * n * log_sum)

def smooth_refine_v2(a, deadline, verbose=False):
    """Refine using smooth log-sum-exp with L-BFGS-B, beta annealing."""
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)

    betas = [10, 50, 200, 1000, 5000, 20000]
    for beta in betas:
        if time.monotonic() > deadline:
            break
        x0 = best.copy()
        x0 = np.clip(x0, 1e-6, None)
        x0 = x0 / x0.sum()

        try:
            res = minimize(smooth_objective, x0, args=(beta, n),
                           method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': 200, 'maxfun': 5000,
                                    'ftol': 1e-18, 'gtol': 1e-18,
                                    'maxls': 50})
            if res.fun < 1e10:
                res_x = np.clip(res.x, 0, None)
                if res_x.sum() > 0.01:
                    score = compute_score(res_x)
                    if score < best_score:
                        best = res_x.copy()
                        best_score = score
                        if verbose:
                            print(f"    smooth beta={beta}: J={score:.6f}", flush=True)
        except Exception as e:
            pass

    return best, best_score

    c = (1 - t_arr) + 0.1
    inits.append(c / c.sum())

    inits.append(c / c.sum())

    c = 1.0 - 2.0*(t_arr - 0.5)**2
    inits.append(np.maximum(c, 0.01) / np.maximum(c, 0.01).sum())
