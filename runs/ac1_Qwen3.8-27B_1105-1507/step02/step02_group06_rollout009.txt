"""
Optimization of step function to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with analytical gradients on a smoothed objective,
combined with LP-based refinement and multi-scale initialization.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import math


def compute_score(f):
    """Fast score computation without type checks."""
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 1e-10:
        return float('inf')
    return float(2.0 * n * M / (S * S))


def smooth_obj_grad(x, t):
    """
    Compute smooth objective and gradient.
    x: log-parameterization (a = exp(x))
    t: temperature for log-sum-exp
    Returns (objective, gradient).
    """
    n = len(x)
    a = np.exp(x)
    S = np.sum(a)
    
    # Compute convolution
    conv = np.convolve(a, a)
    conv_max = np.max(conv)
    
    # Stable log-sum-exp
    shifted = t * (conv - conv_max)
    # Clip to prevent underflow issues
    shifted = np.clip(shifted, -700, 700)
    L = np.sum(np.exp(shifted))
    p = np.exp(shifted) / L
    
    M_smooth = conv_max + np.log(L) / t
    
    # G_i = 2 * sum_{m=0}^{n-1} a_m * p_{m+i}
    G = 2.0 * np.correlate(p, a, mode='valid')
    
    obj = 2.0 * n * M_smooth / (S * S)
    
    # Gradient: dF/dx_i = 2n * a_i * (G_i * S - 2 * M_smooth) / S^3
    grad = 2.0 * n * a * (G * S - 2.0 * M_smooth) / (S ** 3)
    
    return obj, grad


def optimize_lbfgs(f_init, deadline, t_start=5.0, t_end=500.0, 
                   n_phases=8, seed=42, max_iter_per_phase=200):
    """
    L-BFGS optimization with progressive sharpening.
    """
    n = len(f_init)
    best_f = f_init.copy()
    best_score = compute_score(f_init)
    rng = np.random.default_rng(seed)
    
    epsilon = 1e-10
    x = np.log(np.maximum(f_init, epsilon))
    
    for phase in range(n_phases):
        if time.monotonic() > deadline:
            break
        
        t = t_start + (t_end - t_start) * (phase / max(n_phases - 1, 1))
        
        def obj_fun(x_eval):
            obj, _ = smooth_obj_grad(x_eval, t)
            return obj
        
        def grad_fun(x_eval):
            _, grad = smooth_obj_grad(x_eval, t)
            return grad
        
        t0 = time.monotonic()
        result = minimize(obj_fun, x, jac=grad_fun, method='L-BFGS-B',
                         options={'maxiter': max_iter_per_phase, 'ftol': 1e-14,
                                  'gtol': 1e-10, 'maxls': 30})
        t_taken = time.monotonic() - t0
        
        a_opt = np.exp(result.x)
        S_opt = np.sum(a_opt)
        if S_opt > 1e-10:
            a_norm = a_opt / S_opt
            score = compute_score(a_norm)
            if score < best_score:
                best_score = score
                best_f = a_norm.copy()
        
        x = result.x.copy()
        
        # Small perturbation for diversity in next phase
        if phase < n_phases - 1 and t_taken < 0.5:
            x += rng.normal(0, 0.02, n)
        
        print(f"    L-BFGS phase {phase+1}/{n_phases} t={t:.1f}: "
              f"obj={result.fun:.6f} best={best_score:.6f} "
              f"time={t_taken:.2f}s iters={result.nit}")
    
    return best_f, best_score


def build_conv_matrix(f):
    """Build dense matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = f[k - j]
    return A


def lp_refine(f, deadline, max_iter=50, verbose=False):
    """
    LP-based refinement.
    Solves: min M s.t. conv(f,g)_k <= M, sum(g)=sum(f), g>=0
    Then updates f <- g and repeats.
    """
    n = len(f)
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = compute_score(f)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # s.t. [A, -1] @ [g; M] <= 0
        #      [1...1, 0] @ [g; M] = S
        #      g >= 0, M >= 0
        # Variables: [g_0, ..., g_{n-1}, M]
        
        c = np.zeros(n + 1)
        c[n] = 1.0  # minimize M
        
        A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
        b_ub = np.zeros(2 * n - 1)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([S])
        
        bounds = [(0, None)] * (n + 1)
        
        t0 = time.monotonic()
        try:
            result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

            break

            break

        f_new_sum = np.sum(f_new)
        if f_new_sum < 1e-10:
            break
        f_new = f_new * (S / f_new_sum)
        
        score_new = compute_score(f_new)
        f = f_new

def make_initializations(n, rng, parent=None):
    """Create diverse initializations."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(('uniform', np.ones(n)))
    
    # Cosine
    inits.append(('cosine', (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0))
    
    # Random smooth
    r = rng.uniform(0.3, 1.7, size=n)
    # Smooth it
    from scipy.ndimage import gaussian_filter1d
    r_smooth = gaussian_filter1d(r, sigma=n/20.0)
    inits.append(('random_smooth', r_smooth))
    
    # Bimodal
    inits.append(('bimodal', np.exp(-((idx-n*0.3)/(n/10))**2) + np.exp(-((idx-n*0.7)/(n/10))**2)))
    
    # Alternating (high-low pattern)
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits.append(('alternating', alt))
    
    # Parent interpolation
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            inits.append(('parent', parent.copy()))
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(('parent_interp', np.interp(x_new, x_old, parent)))
    
    return inits

F(a) = 2n * M_smooth / S^2

dF/dx_i = dF/da_i * da_i/dx_i = dF/da_i * a_i

But wait, I need to double-check G_i.

G_i = sum_k p_k * 2 * a_{k-i} * I(0 <= k-i <= n-1)

Let me substitute m = k - i:
G_i = 2 * sum_m p_{m+i} * a_m * I(0 <= m <= n-1)
= 2 * sum_{m=0}^{n-1} a_m * p_{m+i}

Now, np.correlate(p, a, mode='valid'):
The 'valid' mode returns the correlation that only includes locations where the signals fully overlap.
p has length 2n-1, a has length n.
The valid output has length (2n-1) - n + 1 = n.

np.correlate(p, a, 'valid')[i] = sum_j p[i+j] * a[j] for j = 0, ..., n-1

So np.correlate(p, a, 'valid')[i] = sum_{j=0}^{n-1} a_j * p[i+j]

And G_i = 2 * sum_{j=0}^{n-1} a_j * p[i+j] = 2 * np.correlate(p, a, 'valid')[i]

Yes! So G = 2 * np.correlate(p, a, mode='valid') is correct.
