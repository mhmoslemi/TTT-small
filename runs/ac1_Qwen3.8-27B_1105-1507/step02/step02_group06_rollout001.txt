Optimization of non-negative step function to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses active-set LP with dynamic constraint selection, subgradient with momentum,

Primary focus on n=2048 with fallback to n=1024 parent.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    """Compute the objective value directly (fast)."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def build_active_set_matrix(f, top_indices):
    """Build sparse matrix for active set constraints.
    A[i,j] = f[top_indices[i] - j] if valid, else 0.
    Shape: (len(top_indices), n)
    """
    n = len(f)
    num_con = len(top_indices)
    # Pre-allocate: each constraint has at most n nonzeros, typically min(n, k+1)
    max_nnz = num_con * n
    rows = np.empty(max_nnz, dtype=np.int32)
    cols = np.empty(max_nnz, dtype=np.int32)
    vals = np.empty(max_nnz, dtype=np.float64)
    idx = 0
    for i, k in enumerate(top_indices):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = i
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(num_con, n))


def solve_active_set_lp(f, M, top_indices, time_limit=3.0):
    """Solve: max sum(g) s.t. g >= 0, A@g <= M where A encodes top constraints."""
    n = len(f)
    A = build_active_set_matrix(f, top_indices)
    c = -np.ones(n)
    b_ub = np.full(len(top_indices), M)
    bounds = [(0, None)] * n
    try:

                         options={'presolve': True, 'time_limit': time_limit})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def subgradient_direction(f, top_k=5):

        j_max = min(k, n - 1)

        # Random perturbation
        eps = rng.uniform(0.02, 0.08)
        perturbation = rng.uniform(-1, 1, size=n) * eps
        candidate = f + perturbation
        candidate = project_to_feasible(candidate, S)
        sc = compute_score(candidate)

        if sc < best_s:
            best_f = candidate.copy()
            best_s = sc

        # Quick re-optimization
        f_tmp, s_tmp = optimize_active_set(candidate, time_per_trial, top_k_init=10, top_k_max=20)
        if s_tmp < best_s:
            best_f = f_tmp.copy()
            best_s = s_tmp

    return best_f, best_s


def make_structured_inits(n, rng):
    """Create diverse structured initializations."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(('uniform', np.ones(n)))

    # Cosine
    inits.append(('cosine', (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0 + 0.01))

    # Sigmoid / S-curve
    inits.append(('sigmoid', 1.0 / (1.0 + np.exp(-6.0 * (idx / (n - 1) - 0.5)))))

    # Bimodal
    inits.append(('bimodal', np.exp(-8.0 * ((idx - n * 0.25) / (n * 0.5)) ** 2) +
                         np.exp(-8.0 * ((idx - n * 0.75) / (n * 0.5)) ** 2) + 0.01))

    # Triangular
    tri = 1.0 - 2.0 * np.abs(idx / (n - 1) - 0.5)
    inits.append(('triangular', tri + 0.01))

    # Smooth random (low-pass filtered)
    raw = rng.uniform(0.3, 1.0, size=n)
    # Simple moving average smoothing
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    smoothed = np.convolve(raw, kernel, mode='same')
    inits.append(('smooth_random', smoothed))

    # Exponential ramp down then up (U-shape)
    u_shape = np.exp(-3.0 * (1.0 - 2.0 * np.abs(idx / (n - 1) - 0.5)))
    inits.append(('u_shape', u_shape + 0.05))

    # Plateau with roll-off
    center_width = 0.6
    plateau = np.ones(n)
    roll = int((1.0 - center_width) * n / 2)
    for i in range(roll):
        t = i / roll
        plateau[i] = 0.3 * t + 0.7 * (1 - t)
        plateau[n - 1 - i] = 0.3 * t + 0.7 * (1 - t)
    inits.append(('plateau', plateau))

    return inits


def propose_candidate(budget_s=1500, **kwargs):
    """Main entry point: search for the best sequence minimizing 2n*max(conv)/sum^2."""
    t0 = time.monotonic()
    total_budget = float(budget_s)
    deadline = t0 + total_budget - 10.0  # Leave 10s safety margin

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)

    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            pscore = evaluate_sequence(parent.tolist())
            if np.isfinite(pscore):
                best_score = pscore
                best_sequence = parent.copy()
                print(f"Parent n={parent.size}: score={pscore:.6f}")
    except (NameError, Exception) as e:
        print(f"No parent available: {e}")

    # Phase 1: Quick exploration (first 150 seconds)
    print(f"\n=== Phase 1: Quick Exploration (budget: {total_budget:.0f}s) ===")
    phase1_deadline = t0 + 150.0

    candidate_list = []  # (score, array, n, label)

    # n values to try
    n_values = [2048, 1024]
    if parent is not None and len(parent) not in n_values:
        n_values.append(len(parent))
    n_values = sorted(set(n_values), reverse=True)

    for n in n_values:
        if time.monotonic() > phase1_deadline - 10:
            break
        print(f"\n--- Exploring n={n} ---")

        inits = make_structured_inits(n, rng)

        # Add interpolated parent if applicable
        if parent is not None and len(parent) != n:
            from numpy import interp
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            parent_interp = np.interp(x_new, x_old, parent)
            parent_interp = np.clip(parent_interp, 0.01, None)
            inits.append(('parent_interp', parent_interp))
        elif parent is not None and len(parent) == n:
            inits.append(('parent', parent.copy()))

        for label, f_init in inits:
            if time.monotonic() > phase1_deadline - 5:
                break
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            S_init = float(np.sum(f_init))
            if S_init < 0.01:
                continue
            f_init = f_init * (S_init / S_init)  # keep as is, score is scale invariant

            init_score = compute_score(f_init)
            if not np.isfinite(init_score):
                continue

            if init_score < best_score:
                best_score = init_score
                best_sequence = f_init.copy()

            # Quick optimization (10-15 seconds)
            quick_time = min(12.0, max(5.0, phase1_deadline - time.monotonic() - 3))
            f_opt, s_opt = optimize_subgradient(f_init, quick_time, lr_init=0.015, top_k=5)
            if s_opt < best_score:
                best_score = s_opt
                best_sequence = f_opt.copy()

            candidate_list.append((s_opt, f_opt.copy(), n, label))
            print(f"  {label}: init={init_score:.6f} -> opt={s_opt:.6f}")

    # Sort and show top candidates
    candidate_list.sort(key=lambda x: x[0])
    print(f"\nTop candidates after exploration (best={best_score:.6f}):")
    for sc, seq, n, lbl in candidate_list[:5]:
        print(f"  n={n} {lbl}: {sc:.6f}")

    # Phase 2: Focused optimization (next ~1100 seconds)
    print(f"\n=== Phase 2: Focused Optimization ===")
    phase2_deadline = deadline - 150.0  # Leave 150s for final polish

    # Select top candidates (prefer distinct n values and diverse shapes)
    seen_keys = set()
    top_candidates = []
    for sc, seq, n, lbl in candidate_list:
        key = (n, lbl[:3])
        if key not in seen_keys:
            top_candidates.append((sc, seq, n, lbl))
            seen_keys.add(key)
        if len(top_candidates) >= 4:
            break

    # If parent is available and not in top candidates, add it
    if parent is not None:
        pkey = (len(parent), 'parent')
        if pkey not in seen_keys:
            top_candidates.append((best_score, parent.copy(), len(parent), 'parent_direct'))

    for sc, f_init, n, lbl in top_candidates:
        if time.monotonic() > phase2_deadline - 30:
            break
        remaining = phase2_deadline - time.monotonic()
        # Allocate time proportionally, minimum 60s each
        alloc_time = min(remaining - 30, max(60, remaining / len(top_candidates)))
        print(f"\n--- Focused: n={n} {lbl} (start={sc:.6f}, time={alloc_time:.0f}s) ---")

        # Active set optimization
        f_opt, s_opt = optimize_active_set(f_init, alloc_time, top_k_init=10, top_k_max=25)
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()
            print(f"  Improved to: {best_score:.6f}")

        # Perturbation attempts
        if time.monotonic() < deadline - 160 and best_sequence is not None:
            perturb_time = min(60.0, deadline - time.monotonic() - 160)
            if perturb_time > 20:
                f_pert, s_pert = perturb_and_reoptimize(
                    best_sequence, best_score,
                    time_per_trial=min(8.0, perturb_time / 4),
                    num_trials=max(2, int(perturb_time / 8)),
                    rng=rng)
                if s_pert < best_score:
                    best_score = s_pert
                    best_sequence = f_pert.copy()
                    print(f"  After perturbation: {best_score:.6f}")

    # Phase 3: Final polish (last 150 seconds)
    print(f"\n=== Phase 3: Final Polish (best={best_score:.6f}) ===")
    if best_sequence is not None and time.monotonic() < deadline - 5:
        polish_time = deadline - time.monotonic() - 5
        print(f"Polishing for {polish_time:.0f}s...")

        # Active set polish
        f_pol, s_pol = optimize_active_set(best_sequence, polish_time * 0.5,
                                           top_k_init=15, top_k_max=30)
        if s_pol < best_score:
            best_score = s_pol
            best_sequence = f_pol.copy()

        # One more perturbation round
        remaining_polish = deadline - time.monotonic() - 5
        if remaining_polish > 15:
            f_pert, s_pert = perturb_and_reoptimize(
                best_sequence, best_score,
                time_per_trial=min(5.0, remaining_polish / 3),
                num_trials=max(2, int(remaining_polish / 6)),
                rng=rng)
            if s_pert < best_score:
                best_score = s_pert
                best_sequence = f_pert.copy()

    # Fallback
    if best_sequence is None:
        print("WARNING: No valid sequence found. Using fallback.")
        best_sequence = np.ones(512)
        best_score = evaluate_sequence(best_sequence.tolist())

    # Final verification
    final_score = evaluate_sequence([float(x) for x in best_sequence])
    if not np.isfinite(final_score) or final_score > best_score + 0.01:
        print(f"WARNING: Verification mismatch! internal={best_score:.6f}, official={final_score}")
        # Try parent as fallback
        if parent is not None:
            pf = evaluate_sequence([float(x) for x in parent])
            if np.isfinite(pf) and pf < final_score:
                best_sequence = parent.copy()
                final_score = pf
                best_score = pf

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, "
          f"internal={best_score:.6f}, elapsed={elapsed:.1f}s")

    return [float(x) for x in best_sequence]
