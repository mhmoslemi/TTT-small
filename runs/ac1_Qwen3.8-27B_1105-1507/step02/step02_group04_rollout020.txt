def compute_conv(a):

    n = len(a)
    size = 1

    conv = np.fft.irfft(fa * fa, size)

def compute_obj(a):

    n = len(a)
    if n == 0:
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (sum_a * sum_a)


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    return e / np.sum(e)

    Smooth objective and gradient using softmax parameterization.
    x is the logit vector; a = softmax(x) so sum(a)=1, a>=0.

    conv = compute_conv(a)

    """
    Optimize using L-BFGS-B with softmax parameterization.
    tau_schedule: list of (tau, fraction_of_time) tuples.
    """
    n = len(a_init)
    a_init = a_init.copy()
    sum_a = np.sum(a_init)
    if sum_a < 1e-10:
        return a_init
    a_init = a_init / sum_a
    
    best_a = a_init.copy()
    best_obj = compute_obj(best_a)
    start = time.monotonic()
    
    # Initialize x from a: x = log(a) (plus small offset to avoid -inf)
    x = np.log(np.maximum(a_init, 1e-30))
    
    for tau, frac in tau_schedule:
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 2:
            break
        
        def fun(xv):
            val, _ = smooth_obj_and_grad(xv, tau)
            return val
        
        def fgrad(xv):
            _, g = smooth_obj_and_grad(xv, tau)
            return g
        
        try:
            res = minimize(fun, x, jac=fgrad, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'maxfun': max_iter,
                                   'ftol': 1e-12, 'gtol': 1e-8,
                                   'maxls': 50})
            a_new = softmax(res.x)
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                x = res.x.copy()
        except Exception:
            pass
    
    return best_a, best_obj


def multiplicative_optimize(a_init, max_time, alpha_init=0.1, tau=5.0):
    """
    Multiplicative (exponentiated gradient) updates to reduce peak autocorrelation.
    Works on the simplex: a >= 0, sum(a) = 1.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    start = time.monotonic()
    alpha = alpha_init
    no_improve = 0
    
    while time.monotonic() - start < max_time:
        conv = compute_conv(a)
        max_c = float(np.max(conv))
        
        if max_c <= 0:
            break
        
        # Softmax weights on convolution
        shifted = (conv - max_c) / tau
        shifted = np.clip(shifted, -50, 50)
        p = np.exp(shifted)
        p_sum = np.sum(p)
        if p_sum < 1e-300:
            p = np.zeros(len(conv))
            p[int(np.argmax(conv))] = 1.0
        else:
            p = p / p_sum
        
        # Gradient of smooth max w.r.t. a
        size = 1
        while size < 4 * n:
            size <<= 1
        P = np.fft.rfft(p, size)
        A = np.fft.rfft(a, size)
        grad_a = 2.0 * np.fft.irfft(P * A, size)[:n]
        
        # Multiplicative update: a_i *-> a_i * exp(-alpha * grad_a_i / max_c)
        # This is mirror descent with entropy on the simplex
        update = np.exp(-alpha * grad_a / max_c)
        a_new = a * update
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            alpha *= 0.5
            continue
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new  # still accept for exploration
            no_improve += 1
            if no_improve > 200:
                alpha *= 0.7
                no_improve = 0
                if alpha < 1e-6:
                    break
        
        # Slowly decrease alpha
        alpha *= 0.999
        
        # Occasionally increase tau to explore more
        if no_improve > 500:
            tau *= 1.5
            alpha = alpha_init * 0.5
    
    return best_a, best_obj


def coordinate_descent(a_init, max_time, max_rounds=5000):
    """
    Greedy coordinate descent: at each step, try to reduce the peak by
    adjusting one coordinate.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    conv = compute_conv(a)
    start = time.monotonic()
    
    for rnd in range(max_rounds):
        if time.monotonic() - start > max_time:
            break
        
        peak_idx = int(np.argmax(conv))
        
        # For each coordinate i, compute how much conv[peak_idx] would change
        # if we transfer a small amount from i to the best "receiver"
        # d conv[peak_idx] / d a[i] = 2 * a[peak_idx - i] (approximately)
        
        best_reduction = 0
        best_i = -1
        best_receiver = -1
        
        # Try decreasing each coordinate
        earn = a.copy()
        for i in range(n):
            if earn[i] < 1e-10:
                continue
            # Reduction in peak if we decrease a[i]
            # The derivative: d conv[k]/d a[i] = a[k-i] + a[i-k] (for valid indices)
            # For the peak at k = peak_idx:
            k = peak_idx
            val1 = a[k - i] if 0 <= k - i < n else 0.0
            val2 = a[i - k] if 0 <= i - k < n else 0.0
            deriv = val1 + val2  # simplified (factor of 2 absorbed)
            # We want positive deriv (decreasing a[i] reduces the peak)
            if deriv > best_reduction:
                best_reduction = deriv
                best_i = i
        
        if best_i < 0 or best_reduction < 1e-15:
            # Try increasing a coordinate that doesn't hurt
            # Find the coordinate that minimally increases the peak
            min_increase = np.inf
            for i in range(n):
                val1 = a[peak_idx - i] if 0 <= peak_idx - i < n else 0.0
                val2 = a[i - peak_idx] if 0 <= i - peak_idx < n else 0.0
                inc = val1 + val2
                if inc < min_increase:
                    min_increase = inc
                    best_receiver = i
            
            if best_receiver < 0 or best_i < 0:
                break
            
            # Transfer from best_i to best_receiver
            delta = a[best_i] * 0.1
            a[best_i] -= delta
            a[best_receiver] += delta
        else:
            # Find best receiver (minimally increases peak)
            min_increase = np.inf
            for j in range(n):
                val1 = a[peak_idx - j] if 0 <= peak_idx - j < n else 0.0
                val2 = a[j - peak_idx] if 0 <= j - peak_idx < n else 0.0
                inc = val1 + val2
                if j != best_i and inc < min_increase:
                    min_increase = inc
                    best_receiver = j
            
            if best_receiver < 0:
                best_receiver = (best_i + 1) % n
            
            delta = a[best_i] * 0.1
            a[best_i] -= delta
            a[best_receiver] += delta
        
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s < 1e-10:
            break
        a = a / s
        
        conv = compute_conv(a)
        obj = 2.0 * n * float(np.max(conv))  # sum(a)=1
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj


def generate_initialization(n, rng, init_type=None):
    """Generate a random initialization of the given length."""
    if init_type is None:
        init_type = rng.integers(0, 8)
    
    if init_type == 0:
        # Uniform
        a = np.ones(n)
    elif init_type == 1:
        # Cosine bump
        phase = rng.uniform(0, 2 * np.pi)
        freq = rng.uniform(0.5, 3.0)
        a = 0.5 * (1 + np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
    elif init_type == 2:
        # Exponential decay (front-loaded)
        rate = rng.uniform(0.5, 3.0)
        a = np.exp(-rate * np.arange(n) / n)
    elif init_type == 3:
        # Exponential decay (back-loaded)
        rate = rng.uniform(0.5, 3.0)
        a = np.exp(-rate * (n - 1 - np.arange(n)) / n)
    elif init_type == 4:
        # Random with smoothing
        a = rng.random(n)
        # Apply a few smoothing steps
        for _ in range(3):
            a = 0.5 * a + 0.25 * np.roll(a, 1) + 0.25 * np.roll(a, -1)
    elif init_type == 5:
        # Two bumps
        c1 = rng.uniform(0, n * 0.4)
        c2 = rng.uniform(n * 0.6, n)
        w1 = rng.uniform(0.05, 0.3) * n
        w2 = rng.uniform(0.05, 0.3) * n
        a = np.exp(-((np.arange(n) - c1) / w1) ** 2) + np.exp(-((np.arange(n) - c2) / w2) ** 2)
    elif init_type == 6:
        # Linear ramp
        t = rng.uniform(0.2, 0.8)
        a = np.where(np.arange(n) < n * t, 
                    1.0 - np.arange(n)[np.arange(n) < n * t] / (n * t),
                    np.arange(n)[np.arange(n) >= n * t] / (n - n * t))
    else:
        # Pure random
        a = rng.random(n)
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def try_parent_sequence():
    """Try to use the parent sequence as a starting point."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_with_all_methods(a_init, max_time):
    """Run multiple optimization methods and return the best result."""
    n = len(a_init)
    a_init = a_init.copy()
    s = np.sum(a_init)
    if s < 1e-10:
        return a_init, np.inf
    a_init = a_init / s
    
    best_a = a_init.copy()
    best_obj = compute_obj(best_a)
    start = time.monotonic()
    
    # Method 1: L-BFGS-B with smooth max (primary)
    if max_time > 5:
        # Use multiple tau values: start smooth, get sharper
        tau_schedule = [(20.0, 0.3), (10.0, 0.25), (5.0, 0.2), (2.0, 0.15), (1.0, 0.1)]
        t1 = max_time * 0.4
        a1, obj1 = lbfgs_optimize(a_init, tau_schedule, t1, max_iter=500)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
            print(f"    L-BFGS: obj={best_obj:.6f}")
    
    # Method 2: Multiplicative updates
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        t2 = min(remaining * 0.3, 30.0)
        a2, obj2 = multiplicative_optimize(best_a, t2, alpha_init=0.05, tau=3.0)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            print(f"    Multiplicative: obj={best_obj:.6f}")
    
    # Method 3: L-BFGS-B again with sharper max (polish)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        tau_schedule2 = [(3.0, 0.4), (1.0, 0.3), (0.5, 0.3)]
        t3 = min(remaining * 0.3, 20.0)
        a3, obj3 = lbfgs_optimize(best_a, tau_schedule2, t3, max_iter=300)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            print(f"    L-BFGS polish: obj={best_obj:.6f}")
    
    # Method 4: Coordinate descent (fast local search)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 2:
        t4 = min(remaining * 0.5, 15.0)
        a4, obj4 = coordinate_descent(best_a, t4, max_rounds=2000)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()
            print(f"    Coord descent: obj={best_obj:.6f}")
    
    # Final L-BFGS polish
    remaining = max_time - (time.monotonic() - start)
    if remaining > 2 and best_obj < np.inf:
        tau_schedule3 = [(0.5, 0.5), (0.2, 0.5)]
        a5, obj5 = lbfgs_optimize(best_a, tau_schedule3, min(remaining - 1, 10.0), max_iter=200)
        if obj5 < best_obj:
            best_obj = obj5
            best_a = a5.copy()
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Search for the best sequence of coefficients minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    - Start from parent sequence and diverse initializations
    - Use L-BFGS-B with softmax + log-sum-exp, multiplicative updates, and coordinate descent
    - Explore sequence lengths from 20 to 600
    - Adaptive time allocation based on current best
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    
    best_seq = None
    best_score = np.inf
    
    # Try parent sequence
    parent, parent_obj = try_parent_sequence()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()
    
    # Define sequence lengths to explore
    n_values = [20, 30, 40, 50, 60, 80, 100, 120, 150, 200, 250, 300, 400, 500, 600]
    
    # If parent exists, add its neighborhood
    if parent is not None:
        n_parent = len(parent)
        extra_ns = [max(10, n_parent - 50), max(10, n_parent - 20),
                    min(800, n_parent + 20), min(800, n_parent + 50)]
        n_values = sorted(set(n_values + extra_ns))
    
    print(f"Total time budget: {budget_s}s, exploring n values: {n_values}")
    
    # Time allocation: give more time to promising n values
    # First pass: quick evaluation to rank n values
    # Second pass: intensive optimization on top candidates
    
    # Phase 1: Quick screening (short time per n)
    screen_results = []
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        
        alloc = min(15.0, remaining * 0.1)
        alloc = max(3.0, alloc)
        
        # If n matches parent length, use parent as init
        if parent is not None and n == len(parent):
            a_init = parent.copy()
            a_init = a_init / np.sum(a_init)
        else:
            a_init = generate_initialization(n, rng, init_type=1)  # cosine bump
        
        a_opt, obj = optimize_with_all_methods(a_init, alloc)
        screen_results.append((n, a_opt, obj))
        
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Screen n={n}: obj={best_score:.6f} (UPDATED)")
        else:
            print(f"  Screen n={n}: obj={obj:.6f}")
    
    # Phase 2: Intensive optimization on top candidates
    screen_results.sort(key=lambda x: x[2])
    top_ns = [r[0] for r in screen_results[:6]]  # Top 6
    
    print(f"\nPhase 2: Intensive on top n values: {top_ns}")
    
    for idx, n in enumerate(top_ns):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        
        # Allocate more time to earlier (promising) candidates
        if idx == 0:
            alloc = min(remaining * 0.35, 200.0)
        elif idx == 1:
            alloc = min(remaining * 0.25, 150.0)
        elif idx == 2:
            alloc = min(remaining * 0.2, 100.0)
        else:
            alloc = min(remaining * 0.1, 60.0)
        alloc = max(10.0, alloc)
        
        print(f"\n  Intensive n={n}, alloc={alloc:.0f}s")
        
        # Multiple restarts
        n_inits = max(2, int(alloc / 25))
        
        for init_idx in range(n_inits):
            if time.monotonic() > deadline:
                break
            init_remaining = deadline - time.monotonic()
            if init_remaining < 10:
                break
            
            # Choose initialization
            if init_idx == 0:
                # Best from screening
                for r in screen_results:
                    if r[0] == n:
                        a_init = r[1].copy()
                        break
                else:
                    a_init = generate_initialization(n, rng)
            else:
                a_init = generate_initialization(n, rng)
            
            init_alloc = min(init_remaining * 0.8 / max(1, n_inits - init_idx), alloc / max(1, n_inits))
            init_alloc = max(5.0, init_alloc)
            
            a_opt, obj = optimize_with_all_methods(a_init, init_alloc)
            
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"    n={n} init{init_idx}: obj={best_score:.6f} (UPDATED)")
            
            # Also try perturbing the current best for this n
            if obj < best_score + 0.01 and time.monotonic() < deadline - 10:
                noise = rng.normal(0, 0.1, n)
                a_perturb = a_init * np.exp(noise)
                a_perturb = np.maximum(a_perturb, 0)
                s_p = np.sum(a_perturb)
                if s_p > 1e-10:
                    a_perturb = a_perturb / s_p
                    a_opt2, obj2 = optimize_with_all_methods(a_perturb, min(init_alloc, 10.0))
                    if obj2 < best_score:
                        best_score = obj2
                        best_seq = a_opt2.copy()
                        print(f"    n={n} init{init_idx} perturb: obj={best_score:.6f} (UPDATED)")
    
    # Phase 3: Final polishing on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\nPhase 3: Final polishing (n={n_best}), {remaining:.0f}s remaining")
        
        a_cur = best_seq.copy()
        
        # Multiple polish rounds
        polish_round = 0
        while time.monotonic() < deadline - 5:
            polish_round += 1
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            
            # L-BFGS with very sharp max
            alloc = min(remaining * 0.4, 30.0)
            tau_sched = [(0.3, 0.3), (0.1, 0.4), (0.05, 0.3)]
            a_pol, obj_pol = lbfgs_optimize(a_cur, tau_sched, alloc, max_iter=300)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  Polish {polish_round} L-BFGS: obj={best_score:.6f}")
            
            # Multiplicative polish
            remaining = deadline - time.monotonic()
            if remaining > 5:
                alloc2 = min(remaining * 0.3, 15.0)
                a_mul, obj_mul = multiplicative_optimize(a_cur, alloc2, alpha_init=0.01, tau=1.0)
                if obj_mul < best_score:
                    best_score = obj_mul
                    best_seq = a_mul.copy()
                    a_cur = a_mul.copy()
                    print(f"  Polish {polish_round} mult: obj={best_score:.6f}")
            
            # Coordinate descent
            remaining = time.monotonic() and deadline - time.monotonic()
            if remaining > 3:
                alloc3 = min(remaining * 0.2, 10.0)
                a_cd, obj_cd = coordinate_descent(a_cur, alloc3, max_rounds=3000)
                if obj_cd < best_score:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    a_cur = a_cd.copy()
                    print(f"  Polish {polish_round} CD: obj={best_score:.6f}")
            
            # Small random perturbation + re-optimize
            remaining = deadline - time.monotonic()
            if remaining > 8:
                for pert_scale in [0.05, 0.02]:
                    noise = rng.normal(0, pert_scale, n_best)
                    a_pert = a_cur * np.exp(noise)
                    a_pert = np.maximum(a_pert, 0)
                    s_p = np.sum(a_pert)
                    if s_p > 1e-10:
                        a_pert = a_pert / s_p
                        a_pp, obj_pp = lbfgs_optimize(a_pert, [(1.0, 0.5), (0.3, 0.5)], 
                                                       min(remaining * 0.15, 8.0), max_iter=100)
                        if obj_pp < best_score:
                            best_score = obj_pp
                            best_seq = a_pp.copy()
                            a_cur = a_pp.copy()
                            print(f"  Polish {polish_round} pert({pert_scale}): obj={best_score:.6f}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)
    
    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    
    # Final validation
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf and validation < best_score:
        print(f"  Validation: {validation:.6f} (better than computed {best_score:.6f})")
        best_score = validation
    elif validation < np.inf:
        print(f"  Validation: {validation:.6f} (computed was {best_score:.6f})")
        # Use the validated value
        best_score = validation
    
    return result
