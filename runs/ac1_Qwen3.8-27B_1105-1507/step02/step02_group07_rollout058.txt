Actually, for the QP approach: we can linearize the problem around the current point. At the current f, the peaks are at specific indices. We can set up a QP:
minimize sum_k (c[k] - target)^2 subject to f >= 0, sum(f) = 1.

This is a QP (quadratic objective, linear constraints) which can be solved exactly with scipy!

Wait, but c[k] = sum_j f[j]*f[k-j] is quadratic in f, so sum (c[k]-target)^2 is quartic in f. That's not a QP.

However, if we linearize c[k] around the current point f0:
c[k] ≈ c0[k] + 2 * sum_j f0[j] * (f[j] - f0[j]) * ... 

Actually, the linearization of c[k] = sum_j f[j]*f[k-j] at f0 is:
c[k] ≈ c0[k] + sum_j (f0[k-j]*(f[j]-f0[j]) + f0[j]*(f[k-j]-f0[k-j]))
= c0[k] + 2 * sum_j f0[k-j]*(f[j]-f0[j])  (by symmetry)

So c[k] ≈ c0[k] + 2 * (f0 * delta_f)[k] where * is convolution.

Then min sum_k (c0[k] + 2*(f0*delta_f)[k] - target)^2 subject to f0 + delta_f >= 0, sum(f0 + delta_f) = 1.

This IS a QP in delta_f! Let me set this up.

Let d = delta_f. Then:
minimize sum_k (c0[k] - target + 2*(f0*d)[k])^2
subject to d >= -f0, sum(d) = 0 (since sum(f) = sum(f0) = 1).

The objective is: sum_k (a[k] + 2*(f0*d)[k])^2 where a[k] = c0[k] - target.
= sum_k a[k]^2 + 4 * sum_k a[k]*(f0*d)[k] + 4 * sum_k (f0*d)[k]^2

The first term is constant. The second is linear in d. The third is quadratic in d.

Let A be the matrix such that (A*d)[k] = (f0*d)[k] = sum_j f0[j]*d[k-j] (with appropriate bounds).
Then the quadratic term is 4 * d^T A^T A d.
The linear term is 4 * a^T A d.

So the QP is: minimize d^T (4 A^T A) d + (4 A^T a)^T d subject to d >= -f0, sum(d) = 0.

This can be solved with scipy.optimize.minimize or cvxpy!

This is a great approach because it gives an exact step in the direction of reducing the variance (locally). It's like a Newton step for the variance minimization.

Let me code this up. The matrix A has size (2n-1) x n:
A[k, j] = f0[k-j] if 0 <= k-j <= n-1, else 0.

For n=512, this is a 1023 x 512 matrix. Forming A^T A gives a 512x512 matrix, which is manageable.

Actually, for larger n (like 2048), A would be 4095 x 2048, and A^T A would be 2048 x 2048. That's still manageable.

Let me write the complete program now.

"""
Multi-strategy optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses variance-gradient descent, QP-based Newton steps, peak suppression,
flat-spectrum initialization, and Adam optimizer.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def evaluate_score(f):
    """Quick score computation without the overhead of evaluate_sequence."""
    f = np.clip(f, 0.0, 1000.0)
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    c = np.convolve(f, f)
    max_c = float(np.max(c))
    return 2.0 * n * max_c / (S * S)


def build_conv_matrix(n, f=None):
    """Build matrix A where (A @ d)[k] = convolve(f, d)[k] for given f."""
    if f is None:
        f = np.ones(n)
    L = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(L):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(L, n))


def qp_newton_step(f, max_iters_qp=50):
    """
    Take a QP-based Newton step to reduce variance of conv(f,f).
    Linearizes conv around f, solves QP, returns new f.
    """
    n = len(f)
    S = float(np.sum(f))
    f = f / S  # normalize to sum=1
    
    c0 = np.convolve(f, f)
    target = 1.0 / (2 * n - 1)  # since sum(f)=1, sum(c) = 1
    a = c0 - target  # residual
    
    # Build A matrix: (A @ d)[k] = sum_j f[k-j] * d[j]
    A = build_conv_matrix(n, f)
    
    # QP: min d^T (A^T A) d + (A^T a) d + const
    # subject to d >= -f, sum(d) = 0
    
    # For the QP, we use scipy minimize with trust-constr
    # Or we can use a simpler projected gradient approach
    
    # Actually, let's just do a few steps of projected gradient on the QP
    # The gradient of the QP objective at d is: 2*A^T A d + A^T a
    # For step size, use backtracking
    
    d = np.zeros(n)
    At_A = A.T @ A  # n x n
    At_a = A.T @ a  # n x 1
    
    # Solve the unconstrained QP: (A^T A) d = -A^T a / 2
    # Then project onto the constraint set
    
    try:
        # Solve (A^T A + lambda*I) d = -A^T a for regularization
        lambda_reg = 1e-6
        d_unconstrained = np.linalg.solve(At_A.toarray() + lambda_reg * np.eye(n), -0.5 * At_a.toarray())
    except np.linalg.LinAlgError:
        d_unconstrained = -0.1 * At_a.toarray()
    
    # Project d onto: d >= -f, sum(d) = 0
    # This is a projection onto an intersection of a half-space and a hyperplane
    # Use iterative projection (POCS)
    for _ in range(max_iters_qp):
        # Project onto sum(d) = 0
        d = d - np.mean(d)
        # Project onto d >= -f
        d = np.maximum(d, -f)
    
    # Try the step
    new_f = f + d
    new_f = np.clip(new_f, 0, None)
    new_S = float(np.sum(new_f))
    if new_S < 0.01:
        return f
    new_f = new_f / new_S
    
    # Line search
    best_f = f.copy()
    best_score = evaluate_score(f)
    for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
        trial = (1 - alpha) * f + alpha * new_f
        trial = np.clip(trial, 0, None)
        ts = float(np.sum(trial))
        if ts < 0.01:
            continue
        trial = trial / ts
        sc = evaluate_score(trial)
        if sc < best_score:
            best_score = sc
            best_f = trial
    
    return best_f


def variance_gradient(f):
    """Compute gradient of sum_k (c[k] - mean)^2 where c = conv(f,f)."""
    n = len(f)
    S = float(np.sum(f))
    f_norm = f / S
    c = np.convolve(f_norm, f_norm)
    target = 1.0 / (2 * n - 1)
    c_shifted = c - target
    # grad[j] = 2 * sum_{j'} c_shifted[j+j'] * f_norm[j']
    grad = 2.0 * np.correlate(c_shifted, f_norm, mode='valid')
    return grad


def peak_suppression_gradient(f, top_k=10, alpha=1.0):
    """Gradient that focuses on suppressing the top_k peaks of conv(f,f)."""
    n = len(f)
    S = float(np.sum(f))
    f_norm = f / S
    c = np.convolve(f_norm, f_norm)
    target = 1.0 / (2 * n - 1)
    
    # Find top_k indices where c exceeds target most
    excess = c - target
    top_indices = np.argsort(excess)[-top_k:]
    
    # Weight: alpha for top peaks, 1.0 for others
    weights = np.ones(len(c))
    weights[top_indices] = alpha
    
    c_weighted = c_shifted * weights  # wait, need to recompute
    c_shifted = c - target
    weighted_excess = c_shifted * weights
    
    # Same gradient computation but with weighted excess
    grad = 2.0 * np.correlate(weighted_excess, f_norm, mode='valid')
    return grad


def flat_spectrum_init(n, rng):
    """Initialize using random phases (flat spectrum in frequency domain)."""
    L = 2 * n - 1
    # Generate random phases with conjugate symmetry
    phases = rng.uniform(0, 2 * np.pi, L)
    phases[0] = 0  # DC is real
    for k in range(1, (L + 1) // 2):
        phases[L - k] = -phases[k]
    if L % 2 == 0:
        phases[L // 2] = 0  # Nyquist is real
    
    F = np.exp(1j * phases) / np.sqrt(L)
    f_full = np.real(np.fft.ifft(F))
    f = f_full[:n].copy()
    f = np.clip(f, 0, None)
    S = float(np.sum(f))
    if S < 1e-6:
        # Fallback: try again or use uniform
        f = np.ones(n) / n
    else:
        f = f / S
    return f


def random_smooth_init(n, rng):
    """Generate a random smooth non-negative sequence."""
    # Generate low-pass filtered noise
    F = rng.standard_normal(2 * n) + 1j * rng.standard_normal(2 * n)
    # Apply low-pass filter
    freq = np.fft.fftfreq(2 * n)
    bandwidth = 0.3
    mask = np.exp(-0.5 * (freq / bandwidth) ** 2)
    F *= mask
    f = np.real(np.fft.ifft(F))
    f = np.abs(f[:n]) + 0.01
    f = f / float(np.sum(f))
    return f


def tukey_window(n, r=0.3):
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def hann_window(n):
    n = max(n, 2)
    return 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))


def gaussian_window(n, sigma_frac=0.3):
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((idx - center) / sigma) ** 2)


def super_gaussian_window(n, p=2.0):
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    return np.exp(-((idx - center) / (n / 2.0)) ** (2.0 * p))


def optimize_variance_gradient(f, deadline, lr_base=0.01, beta_sweep=None, 
                                peak_alpha=5.0, verbose_prefix=""):
    """
    Optimize using variance + peak suppression gradient with Adam.
    """
    n = len(f)
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_score(f)
    f = f / S
    
    best_f = f.copy()
    best_score = evaluate_score(f)
    
    if beta_sweep is None:
        beta_sweep = [1.0, 3.0, 10.0, 30.0, 100.0]
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    no_improve = 0
    lr = lr_base
    beta_idx = 0  # index into beta_sweep
    
    while time.monotonic() < deadline:
        # Compute gradient: variance + peak suppression
        grad_var = variance_gradient(f)
        if peak_alpha > 1:
            grad_peak = peak_suppression_gradient(f, top_k=max(5, n // 50), alpha=peak_alpha)
            grad = grad_var + 0.5 * (grad_peak - grad_var)  # blend
        else:
            grad = grad_var
        
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        new_f = f - lr * update
        new_f = np.clip(new_f, 0, None)
        ns = float(np.sum(new_f))
        if ns < 1e-10:
            new_f = f
        new_f = new_f / ns
        
        sc = evaluate_score(new_f)
        if np.isfinite(sc) and sc < best_score:
            f = new_f
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            if t % 200 == 0:
                print(f"    {verbose_prefix}iter {t}: score={best_score:.6f} (lr={lr:.6f}, beta_idx={beta_idx})")
        else:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.7
                no_improve = 0
                if lr < 1e-8:
                    # Try next beta
                    beta_idx = min(beta_idx + 1, len(beta_sweep) - 1)
                    lr = lr_base * (0.5 ** beta_idx)
                    no_improve = 0
                    if beta_idx >= len(beta_sweep) - 1 and lr < 1e-10:
                        break
    
    return best_f, best_score


def optimize_qp(f, deadline, verbose_prefix=""):
    """
    Optimize using QP-based Newton steps.
    """
    n = len(f)
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_score(f)
    f = f / S
    
    best_f = f.copy()
    best_score = evaluate_score(f)
    it = 0
    
    while time.monotonic() < deadline:
        new_f = qp_newton_step(f)
        it += 1
        sc = evaluate_score(new_f)
        if sc < best_score - 1e-10:
            f = new_f
            best_f = f.copy()
            best_score = sc
            if it % 10 == 0:
                print(f"    {verbose_prefix}QP iter {it}: score={best_score:.6f}")
        else:
            if it > 5:
                break
    
    return best_f, best_score


def hybrid_optimize(f_init, deadline, verbose_prefix=""):
    """
    Multi-phase optimization: QP steps -> variance gradient -> peak refinement.
    """
    f_init = f_init.copy().astype(np.float64)
    S = float(np.sum(f_init))
    if S < 1e-10:
        return f_init, evaluate_score(f_init)
    f_init = f_init / S
    
    best_f = f_init.copy()
    best_score = evaluate_score(f_init)
    t0 = time.monotonic()
    total = deadline - t0
    
    # Phase 1: QP Newton steps (fast convergence initially)
    p1_dead = t0 + total * 0.25
    if time.monotonic() < p1_dead:
        f1, s1 = optimize_qp(f_init, p1_dead, verbose_prefix=verbose_prefix + "QP")
        if s1 < best_score:
            best_f, best_score = f1, s1
    
    # Phase 2: Variance + peak gradient descent
    p2_dead = t0 + total * 0.75
    if time.monotonic() < p2_dead:
        f2, s2 = optimize_variance_gradient(best_f, p2_dead, lr_base=0.005,
                                             peak_alpha=10.0, verbose_prefix=verbose_prefix + "VG")
        if s2 < best_score:
            best_f, best_score = f2, s2
    
    # Phase 3: QP refinement
    if time.monotonic() < deadline - 5:
        f3, s3 = optimize_qp(best_f, deadline - 5, verbose_prefix=verbose_prefix + "QP2")
        if s3 < best_score:
            best_f, best_score = f3, s1 if s1 < best_score else s3
    
    # Phase 4: Final peak suppression
    if time.monotonic() < deadline - 2:
        f4, s4 = optimize_variance_gradient(best_f, deadline - 2, lr_base=0.001,
                                             peak_alpha=50.0, verbose_prefix=verbose_prefix + "PK")
        if s4 < best_score:
            best_f, best_score = f4, s4
    
    return best_f, best_score


def perturb(f, rng, strength=0.05):
    """Add multiplicative noise and renormalize."""
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.1, 2.0)
    new_f = np.clip(new_f, 0.0, 1000.0)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S

1. Start with f >= 0, support [0, n-1].

5. Compute f_new = IDFT(F_new), take real part.
6. Project onto support [0, n-1]: zero out other elements.
7. Project onto f >= 0: clip negative values.
8. Repeat.

def gs_flat_spectrum(f, n_iters=100):
    """Gerchber-Saxton algorithm to make DFT magnitude flat."""
    n = len(f)
    L = 2 * n - 1
    # Pad f to length L
    f_padded = np.zeros(L)
    f_padded[:n] = f
    
    target_mag = np.sum(f) / np.sqrt(L)
    
    for _ in range(n_iters):
        F = np.fft.fft(f_padded)
        # Normalize magnitude
        phases = F / np.abs(F + 1e-30)  # unit phasor (handle zeros)
        F_new = target_mag * phases
        # Enforce conjugate symmetry for real output
        F_new[0] = abs(F_new[0])
        for k in range(1, (L+1)//2):
            F_new[L-k] = np.conj(F_new[k])
        if L % 2 == 0:
            F_new[L//2] = abs(F_new[L//2])
        
        f_full = np.real(np.fft.ifft(F_new))
        f_padded = np.zeros(L)
        f_padded[:n] = np.maximum(f_full[:n], 0)
    
    f = f_padded[:n]
    S = float(np.sum(f))
    if S > 0.01:
        f = f / S
    return f

def evaluate_score(f):

    f = np.clip(f, 0.0, 1000.0)

def gs_flat_spectrum(f, n_iters=50, target='uniform'):
    """Gerchber-Saxton algorithm to flatten DFT magnitude."""
    n = len(f)
    L = 2 * n - 1
    f_padded = np.zeros(L)
    f_padded[:n] = f.copy()
    
    S = float(np.sum(f_padded))
    target_mag = S / np.sqrt(L)
    
    for _ in range(n_iters):

        mag = np.abs(F)

        nonzero = mag > 1e-30

def topk_gradient(f, k_frac=0.05, alpha=5.0):
    """Gradient focusing on top-k peaks."""

    weights[top_idx] = alpha
    
    weighted = c_shifted = excess * weights
    return 2.0 * np.correlate(weighted, f_n, mode='valid')


def softmax_gradient(f, beta):
