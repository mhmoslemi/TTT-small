import numpy as np

def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_cross_matrix(a):
    """Build the matrix A where (A @ b)_k = (a*b)[k] (cross-correlation/convolution)."""
    n = len(a)
    # (a*b)_k = sum_i a_i * b_{k-i} for valid indices
    # So A_{k,i} = a_{k-i} if 0 <= k-i <= n-1, else 0
    # Equivalently, A_{k,j} = a[k-j] if 0 <= k-j <= n-1 (i.e., j is in [k-n+1, k] ∩ [0, n-1])
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        # A[k, j] = a[k-j] for j in [j_lo, j_hi]
        # k-j ranges from k-j_hi to k-j_lo
        A[k, j_lo:j_hi+1] = a[k - j_hi : k - j_lo + 1][::-1]
    return A

def lp_max_sum(a):
    """Max-sum LP: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_cross_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    if res.status == 0:
        return res.x, M
    return None, M

def lp_minimax(a):
    """Minimax LP: min M' s.t. (a*b)_k <= M' for all k, sum(b) = 1, b >= 0."""
    n = len(a)
    A = build_cross_matrix(a)
    # Variables: b_0, ..., b_{n-1}, M' (total n+1 variables)
    # Minimize M' → c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    # Constraints: (a*b)_k - M' <= 0 for all k
    # A_ub has shape (2n-1, n+1)
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]  # b >= 0, M' free
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n], res.x[-1]
    return None, None

def soft_max_obj_and_grad(a, T):

    # Actually, grad_i = sum_k p_k * (a[k-i] if valid)

    # (A^T p)_i = sum_k A_{k,i} * p_k = sum_k a[k-i] * p_k (for valid k-i)
    # This is the cross-correlation of p and a, evaluated at each shift i
    # = np.convolve(p, a[::-1]) ... let me verify

    # For large n this is expensive, so let me use a faster method
    # grad_i = sum_{k: 0<=k-i<=n-1} p_k * a[k-i] = sum_{j: 0<=j<=n-1} p_{j+i} * a[j]
    # = (p * a)_i where * is cross-correlation
    # In numpy: this is np.correlate(p, a, mode='valid') ... no
    # Let me think again.
    # grad_i = sum_j a_j * p_{i+j} (where p is defined for indices 0..2n-2)
    # This is a "cross-correlation" of a and p.
    # np.correlate(a, p, mode=?) ... 
    # Actually, the simplest is to just use the matrix form for moderate n
    if n <= 500:
        A = build_cross_matrix(a)  # (2n-1, n)
        grad = A.T @ p  # (n,)
    else:
        # For large n, compute directly
        grad = np.zeros(n)
        for i in range(n):
            # grad_i = sum_j a_j * p[i+j] for j=0..n-1 (with p[k] = 0 for k < 0 or k >= 2n-1)

def subgradient_with_momentum(a, T=0.1, lr=0.01, momentum=0.9, max_iters=5000, max_time=30):
    """Projected subgradient descent with momentum on the simplex."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    v = np.zeros(n)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break
        
        conv = np.convolve(a, a)
        k_star = int(np.argmax(conv))
        
        # Subgradient of (a*a)_{k_star} w.r.t. a
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j <= n-1:
                grad[i] = a[j]
                if k_star == 2 * i:
                    grad[i] = 2 * a[i]
        
        # Update with momentum
        v = momentum * v - lr * grad
        a_new = a + v
        
        # Project onto simplex {a >= 0, sum(a) = 1}
        a_new = project_simplex(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        
        # Adaptive learning rate
        if it % 100 == 0:
            lr = max(0.001, lr * 0.8)
    
    return best_a, best_obj

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def lp_alternating(a, method='minimax', max_iters=200, max_time=30.0):
    """LP alternating method."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break
        
        if method == 'minimax':
            b, M_prime = lp_minimax(a)
        else:
            b, M = lp_max_sum(a)
        
        if b is None:
            break
        
        b = np.clip(b, 0, None)
        sb = b.sum()
        if sb < 1e-10:
            break
        b = b / sb
        
        # Try different step sizes
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
            a_new = (1 - t) * a + t * b
            a_new = np.clip(a_new, 0, None)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-10:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        
        if not improved:
            stall += 1
            if stall > 10:
                break
        else:
            stall = 0
    
    return best_a, best_obj

def softmax_gd(a, max_time=20.0, T_init=0.05):

Optimized step function coefficient search using:
1. Minimax LP alternating (find simplex direction minimizing peak cross-correlation)
2. Max-sum LP alternating (classic approach)
3. Soft-max gradient descent with FFT-based gradient computation

import numpy as np
from scipy.optimize import linprog

def compute_obj(a):
    """Compute objective: 2n * max(conv(a,a)) / (sum(a))^2"""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_cross_matrix(a):
    """Build matrix A where (A @ b)_k = conv(a,b)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        A[k, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
    return A


def lp_minimax(a):
    """Minimax LP: min M' s.t. (a*b)_k <= M' for all k, sum(b)=1, b>=0."""

    c[-1] = 1.0

    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n], float(res.x[-1])
    return None, None


def lp_max_sum(a):
    """Max-sum LP: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_cross_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x, M
    return None, M


def softmax_gradient(a, T):
    """Compute soft-max value and gradient using FFT-based convolution."""
    n = len(a)
    conv = np.convolve(a, a)
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    sm_val = max_c + T * np.log(sum_exp)
    # grad_i = 2 * sum_j a_j * p_{i+j}
    q = p[::-1]
    conv_aq = np.convolve(a, q)
    grad = 2.0 * conv_aq[n - 1:2 * n - 1][::-1]
    return sm_val, grad


def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def init_hann(n):
    a = np.sin(np.pi * np.arange(n) / n) ** 2
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n


def init_flat(n):
    return np.ones(n) / n


def init_triangle(n):
    mid = (n - 1) / 2.0
    a = 1.0 - np.abs(np.arange(n) - mid) / (mid + 1.0)
    a = np.maximum(a, 0.05)
    s = a.sum()
    return a / s


def init_edge(n):
    a = np.ones(n) * 0.3
    a[0] = 1.0
    a[-1] = 1.0
    if n > 2:
        a[1] = 0.7
        a[-2] = 0.7
    s = a.sum()
    return a / s


def init_random(n, rng):
    a = np.abs(rng.normal(1.0, 0.5, n))
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n


def init_gaussian(n, rng):
    center = n * rng.uniform(0.3, 0.7)
    width = n * rng.uniform(0.15, 0.4)
    a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
    a += 0.05
    s = a.sum()
    return a / s


def init_periodic(n, rng):
    period = int(rng.integers(max(3, n // 20), max(4, n // 5)))
    phase = rng.uniform(0, 2 * np.pi)
    a = np.maximum(0, np.cos(2 * np.pi * np.arange(n) / period + phase)) ** 2
    a += 0.1
    s = a.sum()
    return a / s


def lp_alt_minimax(a, max_time=30.0, max_iters=500):
    """LP alternating using minimax LP."""
    n = len(a)
    a = np.clip(a.copy(), 0, None).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break

        b, M_prime = lp_minimax(a)
        if b is None:
            break

        b = np.clip(b, 0, None)
        sb = b.sum()
        if sb < 1e-10:
            break
        b = b / sb

        improved = False
        for t in [0.5, 0.4, 0.3, 0.2, 0.1, 0.05]:
            a_new = (1.0 - t) * a + t * b
            a_new = np.clip(a_new, 0, None)
            s_new = a_new.sum()
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 15:
                break

    return best_a, best_obj

def softmax_gd(a, max_time=20.0, T_init=0.05, lr_init=0.1):
    """Soft-max gradient descent with adaptive temperature."""
    n = len(a)
    a = np.clip(a.copy(), 0, None).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    T = T_init
    lr = lr_init
    start = time.monotonic()
    last_improve = 0
    patience = 3000

    for it in range(100000):
        if time.monotonic() - start > max_time:
            break

        sm_val, grad = softmax_gradient(a, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        a_new = a - lr * grad / (gnorm + 1e-15)
        a_new = project_simplex(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            last_improve = it
            T = max(0.002, T * 0.998)
        else:
            lr *= 0.95
            if it - last_improve > patience:
                T = max(0.002, T * 0.7)
                lr = lr_init * 0.5
                patience = int(patience * 0.7)
                if patience < 200:
                    break
            if lr < 1e-8:
                break

    return best_a, best_obj


def subgradient_momentum(a, max_time=15.0):
    """Projected subgradient descent with heavy-ball momentum."""
    n = len(a)
    a = np.clip(a.copy(), 0, None).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    v = np.zeros(n)
    momentum = 0.9
    lr = 0.05
    start = time.monotonic()
    last_improve = 0

    for it in range(200000):
        if time.monotonic() - start > max_time:
            break

        conv = np.convolve(a, a)
        k_star = int(np.argmax(conv))

        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j <= n - 1:
                grad[i] = 2.0 * a[j]

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            last_improve = it

        if it % 50 == 0:
            lr *= 0.95
            if lr < 1e-6:
                if it - last_improve > 5000:
                    break
                lr = 0.01
                v = np.zeros(n)
                last_improve = it

    return best_a, best_obj


def block_refine(a, rng, max_time=20.0):
    """Adaptive block perturbation."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < max_time:
        progress = (time.monotonic() - start) / max(max_time, 1e-10)
        width = max(1, int(n * rng.uniform(0.02, 0.15 * (1 - 0.5 * progress))))
        pos = int(rng.integers(0, max(1, n - width + 1)))
        candidate = best_a.copy()
        sigma = 0.02 + 0.12 * (1.0 - progress)
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.clip(candidate, 0, None)
        s = candidate.sum()
        if s < 1e-10:
            continue
        candidate = candidate / s
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj

    return best_a, best_obj


def full_optimize(a_init, time_budget, seed=0):
    """Complete optimization pipeline for one initial sequence."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = np.clip(a_init.copy(), 0, None).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: Minimax LP (primary)
    t1 = min(time_budget * 0.35, 40.0)
    if t1 > 3.0:
        a_opt, obj_opt = lp_alt_minimax(a, max_time=t1)
        if obj_opt < best_obj:
            best_a, best_obj = a_opt.copy(), obj_opt
            a = a_opt

    # Phase 2: Max-sum LP
    elapsed = time.monotonic() - start
    if elapsed < time_budget * 0.65:
        t2 = min(time_budget * 0.25, 25.0)
        a_opt, obj_opt = lp_alt_max_sum(a, max_time=t2)
        if obj_opt < best_obj:
            best_a, best_obj = a_opt.copy(), obj_opt
            a = a_opt

    # Phase 3: Soft-max GD
    elapsed = time.monotonic() - start
    if elapsed < time_budget * 0.85:
        t3 = min(time_budget * 0.15, 20.0)
        a_opt, obj_opt = softmax_gd(a, max_time=t3, T_init=0.03)
        if obj_opt < best_obj:
            best_a, best_obj = a_opt.copy(), obj_opt
            a = a_opt

    # Phase 4: Subgradient momentum
    elapsed = time.monotonic() - start
    if elapsed < time_budget * 0.95:
        t4 = min(time_budget * 0.1, 12.0)
        a_opt, obj_opt = subgradient_momentum(a, max_time=t4)
        if obj_opt < best_obj:
            best_a, best_obj = a_opt.copy(), obj_opt

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization searching for optimal step function coefficients.
    Combines minimax LP, max-sum LP, soft-max GD, subgradient, and block refinement.
    Explores diverse initializations and sequence lengths up to 1000.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 8.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Optimize parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all() and parent.size <= 600:
            p_opt, p_obj = full_optimize(parent.copy(), min(60.0, budget_s * 0.1), seed=seed)
            if p_opt is not None and p_obj < best_score:
                best_score = p_obj
                best_seq = p_opt.copy()
                print(f"Parent optimized: obj={p_obj:.6f}")
    except Exception:
        pass

    # Sequence lengths and initializations
    n_values = [80, 120, 160, 200, 250, 300, 400, 500, 650, 800, 1000]
    init_fns = [init_hann, init_flat, init_triangle, init_edge, init_gaussian, init_periodic]

    # Time budget allocation: weight larger n more heavily
    total_weight = sum(n ** 0.7 for n in n_values)
    available = deadline - time.monotonic() - 60.0  # reserve 60s for final refinement
    available = max(60.0, available)

    print(f"\nAvailable search time: {available:.0f}s")

    for n in n_values:
        if time.monotonic() > deadline - 60:
            break

        remaining = deadline - time.monotonic() - 50
        if remaining < 20:
            break

        # Allocate time proportional to n^0.7
        w = n ** 0.7 / total_weight
        time_per_n = available * w
        time_per_n = max(10.0, min(time_per_n, remaining * 0.3))

        # Number of restarts based on time available
        num_restarts = max(1, min(3, int(time_per_n / 25)))
        time_per_restart = time_per_n / num_restarts

        print(f"\nN={n}: time_per_n={time_per_n:.0f}s, restarts={num_restarts}", flush=True)

        for r in range(num_restarts):
            if time.monotonic() > deadline - 50:
                break

            # Choose initialization
            if r < len(init_fns):
                init_fn = init_fns[r]
                a_init = init_fn(n) if not hasattr(init_fn, '__wrapped__') else init_fn(n, rng)
            else:
                a_init = init_random(n, rng)

            # For random/periodic/gaussian inits, pass rng
            if init_fn in (init_random, init_gaussian, init_periodic):
                a_init = init_fn(n, rng)

            a_opt, obj_opt = full_optimize(a_init, time_per_restart, seed=seed + n * 100 + r)

            if a_opt is not None and obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} r={r} init={init_fn.__name__}: obj={obj_opt:.6f} ** NEW BEST **", flush=True)
            else:
                print(f"  n={n} r={r} init={init_fn.__name__}: obj={obj_opt:.6f}", flush=True)

    # Final refinement on best found
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 5
        print(f"\nFinal refinement: {remaining:.0f}s", flush=True)

        # Minimax LP polish
        a_ref, obj_ref = lp_alt_minimax(best_seq, max_time=min(remaining * 0.3, 30))
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  After LP-minimax: obj={best_score:.6f}", flush=True)

        # Soft-max GD polish
        if time.monotonic() < deadline - 8:
            t_gd = min((deadline - time.monotonic() - 5) * 0.4, 20.0)
            a_ref2, obj_ref2 = softmax_gd(best_seq, max_time=t_gd, T_init=0.01, lr_init=0.05)
            if obj_ref2 < best_score:
                best_score = obj_ref2
                best_seq = a_ref2.copy()
                print(f"  After GD: obj={best_score:.6f}", flush=True)

        # Block refinement
        if time.monotonic() < deadline - 4:
            bp_time = max(1.0, deadline - time.monotonic() - 3)
            a_bp, obj_bp = block_refine(best_seq, rng, max_time=bp_time)
            if obj_bp < best_score:
                best_score = obj_bp
                best_seq = a_bp.copy()
                print(f"  After block: obj={best_score:.6f}", flush=True)

    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s", flush=True)
    return [float(x) for x in best_seq]
