import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_conv(a):
    """Compute autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smoothed_max_and_grad(a, tau):
    """Compute smoothed max of conv(a,a) and its gradient."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum
    # Smoothed max value
    smooth_max = max_conv + np.log(w_sum) / tau
    # Gradient: grad[i] = sum_k weights[k] * d(conv[k])/d(a[i])
    # d(conv[k])/d(a[i]) = a[k-i] + a[i-k] (with appropriate bounds)
    # This is equivalent to 2 * correlate(weights, a)[:n] (approximately)
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[:n]
    return smooth_max, grad

def adam_optimize(a, max_iter=5000, tau_start=20.0, tau_end=500.0,
                  lr=0.01, beta1=0.9, beta2=0.999, epsilon=1e-8):
    """Adam optimizer for the smoothed max of convolution."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    
    for t in range(1, max_iter + 1):
        # Animate tau
        frac = t / max_iter
        tau = tau_start * (tau_end / tau_start) ** frac
        
        smooth_max, grad = smoothed_max_and_grad(a, tau)
        
        # Center gradient (for sum constraint)
        grad_centered = grad - np.mean(grad)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_centered
        v = beta2 * v + (1 - beta2) * grad_centered ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a
            s_new = s
        else:
            a_new = a_new / s_new
        
        # Check improvement
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj

def nesterov_subgradient(a, max_iter=20000, tau_start=10.0, tau_end=200.0,
                          step_init=0.05, verbose=False):
    """Nesterov accelerated subgradient method."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    
    y = a.copy()  # Nesterov lookahead
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for t in range(1, max_iter + 1):
        frac = t / max_iter
        tau = tau_start * (tau_end / tau_start) ** frac
        
        smooth_max, grad = smoothed_max_and_grad(y, tau)
        grad_centered = grad - np.mean(grad)
        
        # Step size: O(1/sqrt(t)) or fixed
        step = step_init / (1 + 0.01 * t)
        
        # Update
        a_new = y - step * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = y
            s_new = s
        else:
            a_new = a_new / s_new
        
        # Nesterov momentum
        beta_t = (1 + np.sqrt(1 + 4 * t * t)) / 2
        y_new = a_new + (beta_t - 1) / beta_t * (a_new - a)
        y_new = np.maximum(y_new, 0)
        s_y = np.sum(y_new)
        if s_y > 1e-10:
            y_new = y_new / s_y
        else:
            y_new = a_new
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        y = y_new
    
    return best_a, best_obj

def window_initializations(n, rng):

    inits.append(np.maximum(w, 0))

    inits.append(w)
    
    # Exponential decay

    w = np.exp(-rate * j / n)
    inits.append(w)
    
    # Left-heavy: high at start, low at end
    w = np.exp(-rate * j / n) + 0.1
    inits.append(w)
    
    # Cosine tapered (only one side)

    w = 0.5 + 0.5 * np.sin(np.pi * freq * j / n)
    inits.append(np.maximum(w, 0.01))

Multi-strategy optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses high-tau smoothed subgradient with Polyak step size, LP-based global moves,
coordinate descent, and Adam optimization. Explores n=150-1000 with diverse
initializations including window functions and the parent sequence.

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_max_grad(a, tau):
    """Compute smoothed max of conv(a,a) and gradient w.r.t. a."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -700, 700)
    weights = np.exp(shifted)
    w_sum = float(np.sum(weights))
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum
    smooth_val = max_conv + np.log(max(w_sum, 1e-300)) / tau
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[:n]
    return smooth_val, grad


def subgrad_polyak(a, max_iter=30000, tau_start=50.0, tau_end=2000.0,
                   verbose=False):
    """Subgradient method with Polyak step size and animated tau."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    # Lower bound on max(conv) is (sum a)^2/(2n-1) = 1/(2n-1) when sum=1
    f_lower = 1.0 / (2.0 * n - 1.0)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    for t in range(1, max_iter + 1):
        frac = min(1.0, t / max_iter)
        # Logarithmic tau schedule
        log_tau = np.log(tau_start) + frac * (np.log(tau_end) - np.log(tau_start))
        tau = np.exp(log_tau)

        smooth_val, grad = smooth_max_grad(a, tau)
        grad_centered = grad - np.mean(grad)
        grad_norm_sq = float(np.dot(grad_centered, grad_centered))

        if grad_norm_sq < 1e-30:
            break

        # Polyak step size
        gap = smooth_val - f_lower
        if gap < 1e-12:
            gap = 1e-6
        step = min(gap / grad_norm_sq, 0.5 / np.sqrt(t + 1))
        step = max(step, 1e-6)

        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        current_max = float(np.max(compute_conv(a_new)))
        if current_max < best_max:
            best_max = current_max
            best_a = a_new.copy()

        a = a_new

    return best_a, best_max


def subgrad_backtrack(a, max_iter=20000, tau=200.0, verbose=False):
    """Subgradient with backtracking line search on the true max."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    step = 0.1
    for t in range(1, max_iter + 1):
        smooth_val, grad = smooth_max_grad(a, tau)
        grad_centered = grad - np.mean(grad)
        grad_norm = float(np.sqrt(np.dot(grad_centered, grad_centered)))
        if grad_norm < 1e-14:
            break

        # Normalize gradient for step size interpretation
        grad_dir = -grad_centered / grad_norm

        # Backtracking line search
        current_max = float(np.max(compute_conv(a)))
        alpha = step
        improved = False
        for _ in range(20):
            a_trial = a + alpha * grad_dir
            a_trial = np.maximum(a_trial, 0)
            s_trial = np.sum(a_trial)
            if s_trial < 1e-10:
                alpha *= 0.5
                continue
            a_trial = a_trial / s_trial
            trial_max = float(np.max(compute_conv(a_trial)))
            if trial_max < current_max - 1e-14:
                a = a_trial
                improved = True
                break
            alpha *= 0.5

        if not improved:
            step *= 0.8
            if step < 1e-10:
                break
            # Try small random perturbation
            rng = np.random.default_rng(t)
            noise = rng.normal(0, 0.01, n)
            a = a + step * noise
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s < 1e-10:
                break
            a = a / s
            continue

        a = np.maximum(a, 0)
        s = np.sum(a)
        if s < 1e-10:
            break
        a = a / s

        current_max = float(np.max(compute_conv(a)))
        if current_max < best_max:
            best_max = current_max
            best_a = a.copy()
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.9

    return best_a, best_max


def adam_smooth(a, max_iter=20000, tau_start=100.0, tau_end=5000.0,
                lr=0.05, beta1=0.9, beta2=0.999, epsilon=1e-8):
    """Adam optimizer on smoothed max objective."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    m = np.zeros(n)
    v = np.zeros(n)

    for t in range(1, max_iter + 1):
        frac = min(1.0, t / max_iter)
        log_tau = np.log(tau_start) + frac * (np.log(tau_end) - np.log(tau_start))
        tau = np.exp(log_tau)

        smooth_val, grad = smooth_max_grad(a, tau)
        grad_centered = grad - np.mean(grad)

        m = beta1 * m + (1.0 - beta1) * grad_centered
        v = beta2 * v + (1.0 - beta2) * grad_centered ** 2
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a
            s_new = s
        else:
            a_new = a_new / s_new

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A @ g)[k] = (a*g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def lp_global_move(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0. Return (g, sum_g)."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None


def lp_iteration(a, max_iter=50, t_blend=0.3, time_per_lp=15):
    """Alternating LP optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    stall = 0

    for it in range(max_iter):
        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = lp_global_move(a, M, time_limit=time_per_lp)
        if g is None:
            break

        if sum_g <= 1.0 + 1e-3:
            stall += 1
            if stall > 15:
                break
            continue
        stall = 0

        g = g / sum_g
        a_new = (1.0 - t_blend) * a + t_blend * g
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_obj


def coordinate_descent(a, max_sweeps=20, tau=200.0):
    """Cyclic coordinate descent on smoothed max."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    for sweep in range(max_sweeps):
        for i in range(n):
            # Try a range of values for a[i]
            old_val = a[i]
            # Compute base convolution (without a[i]'s contribution)
            # conv[k] includes a[i]*a[k-i] + a[i]*a[i-k] (when valid)
            # We just do a 1D search: try several multipliers
            best_trial = old_val
            best_trial_val = np.inf

            smooth_cur, _ = smooth_max_grad(a, tau)
            best_trial_val = smooth_cur

            for mult in [0.0, 0.25, 0.5, 0.75, 1.5, 2.0, 3.0]:
                a[i] = old_val * mult
                s_tmp = np.sum(a)
                if s_tmp < 1e-10:
                    a[i] = old_val
                    continue
                a_norm = a / s_tmp
                smooth_tr, _ = smooth_max_grad(a_norm, tau)
                if smooth_tr < best_trial_val:
                    best_trial_val = smooth_tr
                    best_trial = old_val * mult
            a[i] = best_trial

        a = np.maximum(a, 0)
        s = np.sum(a)
        if s < 1e-10:
            break
        a = a / s
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def window_inits(n, rng):
    """Generate diverse window function initializations."""
    inits = []
    j = np.arange(n, dtype=np.float64)

    # Hann
    inits.append(0.5 * (1.0 - np.cos(2.0 * np.pi * j / max(1, n - 1))))
    # Hamming
    inits.append(0.54 - 0.46 * np.cos(2.0 * np.pi * j / max(1, n - 1)))
    # Blackman
    w = 0.42 - 0.5 * np.cos(2.0 * np.pi * j / max(1, n - 1)) + \
        0.08 * np.cos(4.0 * np.pi * j / max(1, n - 1))
    inits.append(np.maximum(w, 0.0))
    # Flat top (approximate)
    inits.append(np.ones(n))
    # Triangular
    mid = (n - 1) / 2.0
    inits.append(np.maximum(1.0 - np.abs(j - mid) / mid, 0.01))
    # Exponential decay (left-heavy)
    rate = rng.uniform(1.0, 4.0)
    inits.append(np.exp(-rate * j / n) + 0.05)
    # Exponential decay (right-heavy)
    inits.append(np.exp(-rate * (n - 1 - j) / n) + 0.05)
    # Sine bump
    freq = rng.uniform(0.5, 2.0)
    inits.append(np.maximum(np.sin(np.pi * freq * j / (n - 1 + 1e-10)), 0.01))
    # Two-peaked
    c1, c2 = n * 0.3, n * 0.7
    w = np.exp(-4.0 * (j - c1) ** 2 / n) + np.exp(-4.0 * (j - c2) ** 2 / n)
    inits.append(w + 0.05)
    # Random
    inits.append(rng.random(n) + 0.01)

    return inits


def optimize_for_n(n, seed, time_budget=30.0, verbose=False):
    """Full optimization pipeline for a given sequence length n."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    best_a = None
    best_obj = np.inf

    inits = window_inits(n, rng)

    # Include parent if matching length
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size == n and np.isfinite(parent).all():
            inits.append(parent.copy())
    except Exception:
        pass

    num_inits = min(len(inits), max(2, int(time_budget / 8)))

    for idx in range(num_inits):
        if time.monotonic() - start > time_budget:
            break

        a = inits[idx].copy().astype(np.float64)
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s < 1e-10:
            a = rng.random(n) + 0.01
            s = np.sum(a)
        a = a / s

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # Phase 1: Subgradient with Polyak step
        t1 = int(min(15000, remaining * 4))
        a1, _ = subgrad_polyak(a, max_iter=t1, tau_start=30.0, tau_end=500.0)

        # Phase 2: Adam refinement
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            t2 = int(min(15000, remaining * 3))
            a2, _ = adam_smooth(a1, max_iter=t2, tau_start=100.0, tau_end=3000.0, lr=0.03)
            if compute_obj(a2) < compute_obj(a1):
                a1 = a2

        # Phase 3: LP if n is small enough
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 500:
            a3, obj3 = lp_iteration(a1, max_iter=int(min(40, remaining // 3)),
                                     t_blend=0.3, time_per_lp=max(5, remaining // 10))
            if obj3 < compute_obj(a1):
                a1 = a3

        # Phase 4: Coordinate descent polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5 and n <= 400:
            a4, _ = coordinate_descent(a1, max_sweeps=max(3, int(remaining // 3)), tau=300.0)
            if compute_obj(a4) < compute_obj(a1):
                a1 = a4

        obj = compute_obj(a1)
        if obj < best_obj:
            best_obj = obj
            best_a = a1.copy()
            if verbose:
                print(f"    n={n} init{idx}: obj={best_obj:.7f} "
                      f"(elapsed={time.monotonic()-start:.1f}s)")

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def try_parent():
    """Try to load the parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent loaded: n={parent.size}, obj={obj:.7f}")
            return parent.copy(), obj
    except Exception as e:
        print(f"Parent load failed: {e}")
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-strategy search for minimum 2*n*max(conv(a,a))/(sum(a))^2.
    Combines Polyak subgradient, Adam, LP moves, and coordinate descent.
    Explores n=150-800 with window-based and parent initializations.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None:
        best_score = parent_obj
        best_seq = parent.copy()

    print(f"Starting: best_score={best_score:.7f}")
    print(f"Time budget: {budget_s}s, effective deadline in "
          f"{deadline - start_time:.0f}s")

    # Phase 1: Explore various n values
    n_values = [150, 200, 250, 300, 350, 400, 500, 600, 800]
    if parent is not None:
        np_ = len(parent)
        n_values = sorted(set(n_values + [np_, max(50, np_ - 50), min(1000, np_ + 100)]))

    print(f"Exploring n values: {n_values}")

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        # Time allocation
        if n <= 200:
            alloc = min(50.0, remaining * 0.1)
        elif n <= 350:
            alloc = min(70.0, remaining * 0.12)
        elif n <= 500:
            alloc = min(100.0, remaining * 0.15)
        else:
            alloc = min(120.0, remaining * 0.15)
        alloc = max(20.0, alloc)

        print(f"  N={n}: allocating {alloc:.0f}s of {remaining:.0f}s remaining")

        a_opt, obj = optimize_for_n(n, seed=seed + n * 104729,
                                     time_budget=alloc, verbose=(n == 250))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  *** UPDATE: n={n}, obj={best_score:.7f}")

    # Phase 2: Intensive polishing on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        print(f"\n=== Polishing best (n={n_best}), "
              f"{deadline - time.monotonic():.0f}s remaining ===")

        a_cur = best_seq.copy()
        round_num = 0

        while time.monotonic() < deadline - 15:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 25:
                break

            # High-tau Adam
            a_tmp, _ = adam_smooth(a_cur, max_iter=int(remaining * 2),
                                    tau_start=200.0, tau_end=10000.0, lr=0.02)
            obj_tmp = compute_obj(a_tmp)
            if obj_tmp < best_score:
                best_score = obj_tmp
                best_seq = a_tmp.copy()
                a_cur = a_tmp.copy()
                print(f"  Round {round_num} Adam: obj={best_score:.7f}")

            # Polyak subgradient
            remaining = deadline - time.monotonic()
            if remaining > 30:
                a_tmp, _ = subgrad_polyak(a_cur, max_iter=int(remaining * 2),
                                           tau_start=200.0, tau_end=5000.0)
                obj_tmp = compute_obj(a_tmp)
                if obj_tmp < best_score:
                    best_score = obj_tmp
                    best_seq = a_tmp.copy()
                    a_cur = a_tmp.copy()
                    print(f"  Round {round_num} Polyak: obj={best_score:.7f}")

            # LP move
            remaining = deadline - time.monotonic()
            if remaining > 30 and n_best <= 500:
                a_tmp, obj_lp = lp_iteration(a_cur, max_iter=int(min(30, remaining // 5)),
                                              t_blend=0.2, time_per_lp=max(5, remaining // 8))
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_tmp.copy()
                    a_cur = a_tmp.copy()
                    print(f"  Round {round_num} LP: obj={best_score:.7f}")

            # Coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 30 and n_best <= 350:
                a_tmp, _ = coordinate_descent(a_cur,
                                               max_sweeps=max(3, int(remaining // 5)),
                                               tau=500.0)
                obj_tmp = compute_obj(a_tmp)
                if obj_tmp < best_score:
                    best_score = obj_tmp
                    best_seq = a_tmp.copy()
                    a_cur = a_tmp.copy()
                    print(f"  Round {round_num} CoordDescent: obj={best_score:.7f}")

            # Perturbation + quick optimize
            remaining = deadline - time.monotonic()
            if remaining > 40:
                rng_p = np.random.default_rng(round_num * 7919 + seed)
                noise = rng_p.normal(0, 0.02, n_best)
                a_pert = a_cur * np.exp(noise)
                a_pert = np.maximum(a_pert, 0)
                s_p = np.sum(a_pert)
                if s_p > 1e-10:
                    a_pert = a_pert / s_p
                    a_pert, _ = adam_smooth(a_pert, max_iter=int(remaining // 2),
                                             tau_start=100.0, tau_end=2000.0, lr=0.03)
                    obj_p = compute_obj(a_pert)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_pert.copy()
                        a_cur = a_pert.copy()
                        print(f"  Round {round_num} Perturb: obj={best_score:.7f}")

    # Phase 3: Try a few more n values near the best if time permits
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        for dn in [-25, -10, 10, 25, 50]:
            n_try = n_best + dn
            if n_try < 30 or n_try > 1500:
                continue
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            # Initialize from best_seq by interpolation
            a_init = np.interp(np.linspace(0, n_best - 1, n_try),
                               np.arange(n_best), best_seq)
            a_init = np.maximum(a_init, 0)
            s_i = np.sum(a_init)
            if s_i > 1e-10:
                a_init = a_init / s_i
            # Quick optimize
            a_try, obj_try = optimize_for_n(n_try, seed=seed + n_try * 31337,
                                             time_budget=min(30.0, remaining * 0.3),
                                             verbose=False)
            # Also try from interpolated init
            a_inter, _ = adam_smooth(a_init, max_iter=min(5000, int(remaining)),
                                      tau_start=50.0, tau_end=2000.0, lr=0.03)
            obj_inter = compute_obj(a_inter)
            if obj_inter < obj_try:
                a_try, obj_try = a_inter, obj_inter

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  Phase3 n={n_try}: obj={best_score:.7f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(250)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\n=== DONE: n={len(best_seq)}, score={best_score:.7f}, time={elapsed:.1f}s ===")

    # Validate with the actual evaluate_sequence
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf and abs(validation - best_score) > 1e-6:
        print(f"  Validation: computed={best_score:.7f}, actual={validation:.7f}")
        best_score = validation

    return result
