"""
Improved optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated convolution, smoothed subgradient descent,
coordinate descent, and LP refinement. Explores sequences up to n=1000
with diverse structured initializations.
"""

import numpy as np
from scipy.optimizeimport linprog, minimize_scalar
import time


def compute_conv(a):
    """Compute autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smoothed_subgradient(conv, a, tau):
    """Compute smoothed subgradient of max(conv) using log-sum-exp."""
    n = len(a)
    max_conv = np.max(conv)
    shifted = np.clip(tau * (conv - max_conv), -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[:n]
    return grad


def subgradient_optimize(a, max_iter=20000, tau_start=5.0, tau_end=50.0,
                         step_init=0.1, patience=300):
    """Smoothed subgradient method to minimize max(conv) with fixed sum."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    step = step_init
    stall_count = 0

    for it in range(max_iter):
        frac = it / max(1, max_iter)
        tau = tau_start + (tau_end - tau_start) * frac

        conv = compute_conv(a)
        current_max = float(np.max(conv))

        if current_max < best_max - 1e-12:
            best_max = current_max
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > patience:
                step *= 0.7
                if step < 1e-10:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                noise = rng.normal(0, 0.02, n)
                a = a + noise * step
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s < 1e-10:
                    break
                a = a / s
                continue

        grad = smoothed_subgradient(conv, a, tau)
        grad_centered = grad - np.mean(grad)

        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new
        step *= 0.9998

    return best_a, best_max


def coordinate_descent(a, max_rounds=100,verbose=False):
    """Coordinate descent: for each coordinate, find optimal value via 1D search."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_obj = compute_obj(a)
    current_obj = best_obj

    for rnd in range(max_rounds):
        improved = False
        # Shuffle order for diversity
        idx_order = np.random.default_rng(rnd * 1000 + 42).permutation(n)
        
        for i in idx_order:
            # Try to find better value for a[i]
            # Parametrize: a[i] = x, rest scaled to maintain sum
            old_val = a[i]
            rest_sum = s - old_val
            if rest_sum < 1e-10:
                continue
            
            def obj_1d(x):
                a_test = a.copy()
                a_test[i] = x
                s_test = np.sum(a_test)
                if s_test < 1e-10:
                    return np.inf
                a_test = a_test / s_test
                conv = compute_conv(a_test)
                return np.max(conv)
            
            # Bounded minimization
            # Upper bound: old_val * 2 or so
            ub = max(old_val * 3, 0.01)
            res = minimize_scalar(obj_1d, bounds=(0, ub), method='bounded',
                                  options={'maxiter': 50})
            if res.success and res.fun < current_obj - 1e-10:
                a[i] = res.x
                s_new = np.sum(a)
                a = a / s_new
                current_obj = res.fun
                improved = True
        
        if not improved:
            break
    
    return a


def build_A_matrix(a):
    """Build the constraint matrix for LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def lp_optimize(a, max_iter=30, t=0.3):
    """Alternating LP optimization."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    stall = 0
    for it in range(max_iter):
        conv = compute_conv(a)
        M = float(np.max(conv))

        A = build_A_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 20})
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break

        sum_g = float(-res.fun)
        if sum_g <= 1.0 + 1e-4:
            stall += 1
            if stall > 10:
                break
            continue
        stall = 0

        g = g / sum_g
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_obj


def generate_initializations(n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n)
    mid = (n - 1) / 2.0

    # Uniform
    inits.append(np.ones(n))

    # Raised cosine (Hann)
    inits.append(0.5 * (1 + np.cos(2 * np.pi * x / max(1, n - 1))))

    # Gaussian
    for sigma in [0.3, 0.5, 0.7]:
        inits.append(np.exp(-0.5 * ((x - mid) / (sigma * n / 2))**2))

    # Tent
    inits.append(1 - np.abs(x - mid) / (n / 2))

    # Power-law decay from center
    for p in [0.5, 1.0, 2.0]:
        inits.append(np.maximum(1 - np.abs(x - mid) / (n / 2), 0) ** p)

    # Super-ellipse
    for p in [0.5, 2, 4]:
        inits.append(np.maximum(1 - (np.abs(x - mid) / (n / 2)) ** p, 0))

    # Cosine with random phase
    phase = rng.uniform(0, 2 * np.pi)
    inits.append(0.5 * (1 + np.cos(2 * np.pi * x / n + phase)))

    # Random smooth
    freq = rng.uniform(1, 3)
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n))

    # Bimodal
    gap = rng.uniform(0.2, 0.4)
    c1 = mid * (1 - gap)
    c2 = mid * (1 + gap)
    inits.append(np.maximum(
        np.exp(-0.5 * ((x - c1) / (n * 0.15))**2) +
        np.exp(-0.5 * ((x - c2) / (n * 0.15))**2), 0))

    # Nearly uniform with small perturbation
    inits.append(np.ones(n) + 0.1 * rng.normal(0, 1, n))

    # Triangular with flat top
    frac = rng.uniform(0.2, 0.5)
    inits.append(np.clip(1 - np.abs(x - mid) / (n / 2 * (1 - frac)), 0, 1))

    return [np.maximum(a, 0) for a in inits]


def run_optimization_for_n(n, seed, max_time, verbose=False):
    """Run full optimization pipeline for a given sequence length."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    best_a = None
    best_obj = np.inf

    inits = generate_initializations(n, rng)

    for idx, a_init in enumerate(inits):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break

        a = np.maximum(a_init, 0).astype(np.float64)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n)
            s = n
        a = a / s

        # Time allocation per init
        t_remaining = max_time - (time.monotonic() - start)
        t_per_init = t_remaining / (len(inits) - idx)

        # Phase 1: Subgradient
        t_sg = min(t_per_init * 0.5, t_remaining * 0.4)
        sg_iters = int(t_sg * 2000)  # rough estimate of iterations per second
        a_sg, _ = subgradient_optimize(a, max_iter=sg_iters,
                                        tau_start=5.0, tau_end=40.0,
                                        step_init=0.08)
        obj_sg = compute_obj(a_sg)

        # Phase 2: Coordinate descent
        t_cd = min(t_per_init * 0.25, t_remaining * 0.2)
        cd_rounds = int(t_cd * 5)  # rough estimate
        a_cd = coordinate_descent(a_sg, max_rounds=max(5, cd_rounds))
        obj_cd = compute_obj(a_cd)

        # Phase 3: LP (only for smaller n)
        a_best = a_cd if obj_cd < obj_sg else a_sg
        obj_best = min(obj_cd, obj_sg)
        t_remaining = max_time - (time.monotonic() - start)
        if t_remaining > 10 and n <= 500:
            a_lp, obj_lp = lp_optimize(a_best, max_iter=20, t=0.3)
            if obj_lp < obj_best:
                a_best = a_lp
                obj_best = obj_lp

        # Phase 4: Final subgradient polish
        t_remaining = max_time - (time.monotonic() - start)
        if t_remaining > 5:
            a_pol, _ = subgradient_optimize(a_best, max_iter=int(t_remaining * 3000),
                                             tau_start=20.0, tau_end=100.0,
                                             step_init=0.03)
            obj_pol = compute_obj(a_pol)
            if obj_pol < obj_best:
                a_best = a_pol
                obj_best = obj_pol

        if obj_best < best_obj:
            best_obj = obj_best
            best_a = a_best.copy()
            if verbose:
                print(f"    n={n} init{idx}: obj={best_obj:.6f}")

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj

"""
Improved optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated convolution, smoothed subgradient descent with adaptive steps,
coordinate descent, LP refinement, and structured initializations (windows, Gaussians,

Key improvements over previous: better structured inits, periodic-AC subgradient for

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear autocorrelation conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_periodic_conv(a):
    """Compute periodic autocorrelation using FFT. Returns array of length n."""
    n = len(a)
    A = np.fft.fft(a)
    r = np.fft.ifft(A * A).real
    return r


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def subgradient_step(a, tau, use_periodic=False):
    """Compute smoothed subgradient of max autocorrelation. Returns (grad, current_max)."""
    n = len(a)
    if use_periodic:
        r = compute_periodic_conv(a)
        r_max = float(np.max(r))
        shifted = np.clip(tau * (r - r_max), -500, 500)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            weights = np.zeros(n)
            weights[int(np.argmax(r))] = 1.0
        else:
            weights /= w_sum
        # Gradient: 2 * circular_correlation(weights, a)
        W = np.fft.fft(weights)
        A = np.fft.fft(a)
        grad = 2.0 * np.real(np.fft.ifft(W * np.conj(A)))
    else:
        conv = compute_conv(a)
        r_max = float(np.max(conv))

        weights = np.exp(shifted)

            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum
        c = np.correlate(weights, a, mode='full')
        grad = 2.0 * c[:n]

    return grad, r_max

                a = a / s_t
                current_max = best_mx
                improved = True

        if not improved:
            break

    return a


def build_A_matrix(a):
    """Build constraint matrix for LP: rows = lags, cols = g variables."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def lp_optimize(a, max_iter=25, t=0.3, time_limit_per_lp=15):
    """Alternating LP optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    stall = 0
    for it in range(max_iter):
        conv = compute_conv(a)
        M = float(np.max(conv))

        A_mat = build_A_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True,
                                                     'time_limit': time_limit_per_lp})
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break

        sum_g = float(-res.fun)
        if sum_g <= 1.0 + 1e-3:
            stall += 1
            if stall > 8:
                break
            continue
        stall = 0

        g = g / sum_g
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_obj


def generate_initializations(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    # Uniform
    inits.append(np.ones(n))

    # Raised cosine (Hann)
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1))))

    # Blackman
    if n > 2:
        a_bm = 0.42 + 0.5 * np.cos(2.0 * np.pi * x / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * x / (n - 1))
        inits.append(np.maximum(a_bm, 0))

    # Kaiser windows (different beta)
    for beta in [2.0, 5.0, 8.0]:
        from scipy.special import i0
        I0_beta = i0(beta)
        arg = np.sqrt(1.0 - (2.0 * x / (n - 1) - 1.0) ** 2)
        arg = np.maximum(arg, 0)
        inits.append(i0(beta * arg) / I0_beta)

    # Gaussian (different widths)
    for sigma_frac in [0.2, 0.35, 0.5, 0.7]:
        inits.append(np.exp(-0.5 * ((x - mid) / (sigma_frac * n / 2.0)) ** 2))

    # Tent (triangular)
    inits.append(np.maximum(1.0 - np.abs(x - mid) / (n / 2.0), 0))

    # Power-law tapers
    for p in [0.3, 0.5, 1.0, 2.0, 4.0]:
        inits.append(np.maximum(1.0 - np.abs(x - mid) / (n / 2.0), 0) ** p)

    # Super-ellipse
    for p in [0.5, 2.0, 6.0]:
        inits.append(np.maximum(1.0 - (np.abs(x - mid) / (n / 2.0)) ** p, 0))

    # Bimodal
    gap = 0.3
    c1 = mid * (1.0 - gap)
    c2 = mid * (1.0 + gap)
    inits.append(np.exp(-0.5 * ((x - c1) / (n * 0.12)) ** 2) +
                 np.exp(-0.5 * ((x - c2) / (n * 0.12)) ** 2))

    # Smooth random
    freq = rng.uniform(1.0, 3.0)
    inits.append(0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi)))

    # Nearly uniform with smooth perturbation
    inits.append(np.ones(n) * (1.0 + 0.15 * np.sin(2.0 * np.pi * x / n)))

    # Flat top with tapered edges
    frac = 0.4
    t = np.abs(x - mid) / (n / 2.0)
    inits.append(np.where(t < frac, 1.0, np.maximum(1.0 - (t - frac) / (1.0 - frac), 0)))

    return [np.maximum(arr, 0.0) for arr in inits]


def run_optimization_for_n(n, seed, max_time, verbose=False):
    """Run full optimization pipeline for a given sequence length."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    best_a = None
    best_obj = np.inf

    inits = generate_initializations(n, rng)
    n_inits = len(inits)

    for idx, a_init in enumerate(inits):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break

        a = np.maximum(a_init, 0.0).astype(np.float64)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n)
            s = n
        a = a / s

        t_per_init = (max_time - (time.monotonic() - start)) / (n_inits - idx)

        # Phase 1: Fast subgradient (periodic AC for speed)
        t_sg1 = min(t_per_init * 0.4, 30.0)
        iters1 = max(2000, int(t_sg1 * 5000))
        a1, _ = subgradient_optimize(a, max_iter=iters1,
                                       tau_start=5.0, tau_end=50.0,
                                       step_init=0.1, use_periodic=True)
        obj1 = compute_obj(a1)

        # Phase 2: Subgradient with linear AC (more accurate)
        t_sg2 = min(t_per_init * 0.25, 20.0)
        iters2 = max(1000, int(t_sg2 * 3000))
        a2, _ = subgradient_optimize(a1, max_iter=iters2,
                                       tau_start=20.0, tau_end=80.0,
                                       step_init=0.05, use_periodic=False)
        obj2 = compute_obj(a2)

        # Phase 3: Coordinate descent
        t_cd = min(t_per_init * 0.2, 15.0)
        cd_rounds = max(5, int(t_cd * 5))
        a3 = coordinate_descent(a2 if obj2 < obj1 else a1, max_rounds=cd_rounds)
        obj3 = compute_obj(a3)

        # Phase 4: LP (only for manageable n)
        a_best = a3 if obj3 <= min(obj1, obj2) else (a2 if obj2 < obj1 else a1)
        obj_best = min(obj3, obj1, obj2)

        remaining = max_time - (time.monotonic() - start)
        if remaining > 8 and n <= 450:
            a_lp, obj_lp = lp_optimize(a_best, max_iter=20, t=0.3)
            if obj_lp < obj_best:
                a_best = a_lp
                obj_best = obj_lp

        # Phase 5: Final subgradient polish
        remaining = max_time - (time.monotonic() - start)
        if remaining > 3:
            iters5 = max(2000, int(remaining * 3000))
            a_pol, _ = subgradient_optimize(a_best, max_iter=iters5,
                                             tau_start=30.0, tau_end=150.0,
                                             step_init=0.03, use_periodic=True)
            obj_pol = compute_obj(a_pol)
            if obj_pol < obj_best:
                a_best = a_pol
                obj_best = obj_pol

            # Quick coordinate descent after polish
            a_cd2 = coordinate_descent(a_best, max_rounds=10)
            obj_cd2 = compute_obj(a_cd2)
            if obj_cd2 < obj_best:
                a_best = a_cd2
                obj_best = obj_cd2

        if obj_best < best_obj:
            best_obj = obj_best
            best_a = a_best.copy()
            if verbose:
                print(f"    n={n} init{idx} [{len(inits)} total]: obj={best_obj:.6f}")

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Full search for optimal step function sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    Uses structured initializations, subgradient (periodic + linear AC),
    coordinate descent, and LP refinement. Explores n from 50 to 800.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.all(np.isfinite(parent)):
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
    except Exception:
        pass

    # Sequence lengths: focus on medium-to-large where improvement is possible
    n_values = [50, 80, 100, 150, 200, 250, 300, 350, 400, 500, 600, 700]

    if best_seq is not None:
        n_par = len(best_seq)
        n_values = sorted(set(n_values + [n_par,
                                           max(30, n_par - 50),
                                           min(900, n_par + 50),
                                           max(30, n_par - 100)]))

    print(f"Exploring n values: {n_values}")
    print(f"Total time budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Allocate more time to promising (larger) n values
        if n <= 100:
            alloc = min(30.0, remaining * 0.06)
        elif n <= 200:
            alloc = min(50.0, remaining * 0.09)
        elif n <= 350:
            alloc = min(80.0, remaining * 0.12)
        elif n <= 500:
            alloc = min(120.0, remaining * 0.14)
        else:
            alloc = min(150.0, remaining * 0.16)

        alloc = max(20.0, alloc)
        print(f"\nN={n}: {remaining:.0f}s remaining, allocating {alloc:.0f}s")

        a_opt, obj = run_optimization_for_n(n, seed=seed + n * 7919,
                                             max_time=alloc,
                                             verbose=(n in [100, 200, 300, 500]))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f}")
        else:
            print(f"  n={n} best: {obj:.6f} (no improvement over {best_score:.6f})")

    # === Final intensive polishing ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== Intensive polishing (n={n_best}), {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()
        polish_round = 0

        while time.monotonic() < deadline - 15:
            polish_round += 1
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # 1. High-tau subgradient (focus on active constraints)
            iters = max(5000, int(remaining * 2500))
            a_cur, _ = subgradient_optimize(a_cur, max_iter=iters,
                                             tau_start=40.0, tau_end=200.0,
                                             step_init=0.04, use_periodic=True)
            obj = compute_obj(a_cur)
            if obj < best_score:
                best_score = obj
                best_seq = a_cur.copy()
                print(f"  P{polish_round} sg: obj={best_score:.6f}")

            # 2. Coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 15:
                cd_rounds = max(10, int(remaining * 3))
                a_cd = coordinate_descent(a_cur, max_rounds=cd_rounds)
                obj_cd = compute_obj(a_cd)
                if obj_cd < best_score:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    a_cur = a_cd.copy()
                    print(f"  P{polish_round} cd: obj={best_score:.6f}")

            # 3. LP (if n is manageable)
            remaining = deadline - time.monotonic()
            if remaining > 25 and n_best <= 400:
                a_lp, obj_lp = lp_optimize(a_cur, max_iter=30, t=0.25,
                                            time_limit_per_lp=10)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    print(f"  P{polish_round} LP: obj={best_score:.6f}")

            # 4. Subgradient with linear AC after LP/CD
            remaining = deadline - time.monotonic()
            if remaining > 10:
                iters2 = max(3000, int(remaining * 2000))
                a_cur, _ = subgradient_optimize(a_cur, max_iter=iters2,
                                                 tau_start=50.0, tau_end=150.0,
                                                 step_init=0.03, use_periodic=False)
                obj2 = compute_obj(a_cur)
                if obj2 < best_score:
                    best_score = obj2
                    best_seq = a_cur.copy()
                    print(f"  P{polish_round} sg2: obj={best_score:.6f}")

            # 5. Small perturbation to escape local optima
            remaining = deadline - time.monotonic()
            if remaining > 20:
                for noise_scale in [0.02, 0.05, 0.1]:
                    a_pert = a_cur * np.exp(rng.normal(0, noise_scale, n_best))
                    a_pert = np.maximum(a_pert, 0)
                    s_p = np.sum(a_pert)
                    if s_p > 1e-10:
                        a_pert = a_pert / s_p
                        obj_p = compute_obj(a_pert)
                        if obj_p < best_score:
                            best_score = obj_p
                            best_seq = a_pert.copy()
                            a_cur = a_pert.copy()
                            print(f"  P{polish_round} pert({noise_scale}): obj={best_score:.6f}")
                            break

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\n=== DONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    result = [float(x) for x in best_seq]

    # Validate with the actual evaluator
    validation = evaluate_sequence(result)
    if validation == np.inf:
        print("  WARNING: validation returned inf! Using fallback.")
        result = [float(x) for x in best_seq]
        validation = evaluate_sequence(result)
    if validation < np.inf:
        print(f"  Final validated score: {validation:.6f}")
        if validation > best_score:
            best_score = validation

    return result
