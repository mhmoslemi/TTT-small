- Find tau by bisection

- We can try to reduce conv[k*] by moving a in the direction -d(conv[k*])/da
- But we need to keep other conv[k] from exceeding the new (lower) peak
- This is a subgradient step on the exact max

1. Flat autocorrelation recursive construction (new initialization)

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full convolution conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_candidates = np.where(cond)[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(w, a):
    """
    Compute gradient of sum_k w[k]*conv(a,a)[k] with respect to a.
    w has length 2n-1, a has length n.
    Returns gradient of length n.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    if n == 1:
        return np.array([2.0 * w[0] * a[0]])
    
    # Determine FFT size
    size = 1
    while size < 3 * n:
        size <<= 1
    
    # Zero-pad
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    
    # The gradient formula:
    # grad[i] = sum_k w[k]*a[k-i] (for valid k-i) + sum_k w[k]*a[i-k] (for valid i-k)
    # 
    # First term = cross-correlation of w and a: sum_j w[i+j]*a[j]
    # This equals ifft(conj(FFT(w)) * FFT(a)) at appropriate indices
    # 
    # Actually, let's use a cleaner approach:
    # conv(k) = sum_j a[j]*a[k-j]
    # d(conv[k])/da[i] = a[k-i] (+a[i-k] if k-i != i-k)
    # 
    # grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) where out-of-range = 0
    #
    # Define f1[i] = sum_k w[k]*a[k-i]*[0<=k-i<n] = sum_j a[j]*w[i+j] (cross-corr)
    # Define f2[i] = sum_k w[k]*a[i-k]*[0<=i-k<n] = sum_j a[j]*w[i-j] (cross-corr reversed)
    
    # f1[i] = sum_{j=0}^{n-1} a[j] * w[i+j]  (for i+j < 2n-1)
    # This is the cross-correlation: (a ⊛ w)[i] where (a⊛w)[i] = sum_j a[j]*w[j+i]
    # Using FFT: ifft(FFT(w) * conj(FFT(a))) gives sum_j w[j-i]*a[j]... 
    # Let me use the direct relation:
    # sum_j a[j]*w[j+i] = cross_correlation(w, a) at lag i
    # In FFT: this is ifft(conj(FFT(a)) * FFT(w)) ... 
    
    # Let me just use the reliable approach:
    # f1 = ifft(FFT(w_zero_padded) * conj(FFT(a_zero_padded)))
    # This gives circular cross-correlation. With sufficient padding, it equals linear.
    
    cross1 = np.fft.irfft(W * np.conj(A), size)
    # cross1[i] = sum_j w[j] * a[j-i] (circular) = sum_j w[j+i] * a[j] (shifted)
    # Actually: ifft(FFT(w)*conj(FFT(a)))[_n] = sum_j w[j] * a[j-n mod size]
    # For our purposes with sufficient padding:
    # cross1[i] for i in range gives us the cross-correlation
    
    # Let me verify with a different approach. 
    # The standard result: ifft(FFT(f) * conj(FFT(g)))[k] = sum_j f[j] * g[j-k]
    # So: ifft(FFT(w) * conj(FFT(a)))[k] = sum_j w[j] * a[j-k]
    # We want: f1[i] = sum_j a[j] * w[j+i] = sum_j w[j+i] * a[j]
    # Let m = j+i: f1[i] = sum_m w[m] * a[m-i] = ifft(FFT(w)*conj(FFT(a)))[-i] = cross1[size-i]
    
    # And: f2[i] = sum_j a[j] * w[i-j] = sum_k w[k] * a[i-k] (sub k=i-j)
    # = ifft(FFT(w)*conj(FFT(a)))[i] = cross1[i]
    
    # So grad[i] = cross1[i] + cross1[size-i] (for i > 0), and grad[0] = 2*cross1[0]
    
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = cross1[i] + cross1[(-i) % size]
    
    return grad


def compute_smoothed_gradient_direct(w, a):
    """
    Direct O(n^2) computation of the smoothed gradient.
    Falls back to this for verification or when FFT has issues.
    """
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        # Term 1: sum_j a[j]*w[i+j] for j=0..n-1 where i+j < 2n-1
        j_max = min(n, 2 * n - 1 - i)
        if j_max > 0:
            grad[i] += np.dot(a[:j_max], w[i:i + j_max])
        
        # Term 2: sum_j a[j]*w[i-j] for j=0..n-1 where i-j >= 0, i.e., j <= i
        j_max = min(n, i + 1)
        if j_max > 0:
            grad[i] += np.dot(a[:j_max], w[i - j_max + 1:i + 1][::-1])
    
    return grad


def flat_autocorrelation_construction(n):
    """
    Construct a sequence that approximately equalizes the autocorrelation.
    Uses recursive formula: a[k] = (C - cross_sum) / (2*a[0])
    where C is the target autocorrelation value.
    """
    a = np.zeros(n)
    a[0] = 1.0
    C = 1.0  # Target conv value (will normalize later)
    
    for k in range(1, n):
        # conv[k] = 2*a[0]*a[k] + sum_{j=1}^{k-1} a[j]*a[k-j]
        # Set conv[k] = C:
        cross_sum = 0.0
        for j in range(1, k):
            cross_sum += a[j] * a[k - j]
        
        a[k] = (C - cross_sum) / (2.0 * a[0])
        if a[k] < 0:
            a[k] = 0.0
    
    # Normalize
    s = np.sum(a)
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    
    return a


def geometric_init(n, r):
    """Geometric sequence a[i] = r^i, normalized."""
    a = np.power(r, np.arange(n))
    s = np.sum(a)
    return a / s


def exponential_init(n, alpha):
    """Exponential decay a[i] = exp(-alpha*i), normalized."""
    a = np.exp(-alpha * np.arange(n))
    s = np.sum(a)
    return a / s


def quadratic_init(n, c):
    """Quadratic: a[i] = 1 - c*(i/(n-1)-0.5)^2, clipped at 0, normalized."""
    x = np.arange(n) / max(n - 1, 1)
    a = 1.0 - c * (x - 0.5) ** 2
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def random_init(n, rng):
    """Random non-negative sequence, normalized."""
    a = rng.random(n)
    # Add some structure
    shape = rng.choice(['geometric', 'quadratic', 'uniform', 'bilinear'])
    if shape == 'geometric':
        r = rng.uniform(0.3, 0.95)
        a = a * np.power(r, np.arange(n))
    elif shape == 'quadratic':
        c = rng.uniform(0, 3)
        x = np.arange(n) / max(n - 1, 1)
        a = a * (1.0 - c * (x - 0.5) ** 2)
    elif shape == 'bilinear':
        peak = rng.integers(0, n)
        a = a * (1.0 - np.abs(np.arange(n) - peak) / n)
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def smoothed_gradient_descent(a, max_iters=5000, tau_start=10.0, tau_end=200.0,
                               lr_start=0.1, lr_end=0.001, verbose=False):
    """
    Smoothed gradient descent on the normalized problem:
    minimize max_k conv[k] subject to a >= 0, sum(a) = 1.
    Uses log-sum-exp smoothing with annealing temperature.
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    
    stall_count = 0
    patience = max(100, max_iters // 10)
    
    for it in range(max_iters):
        # Anneal temperature
        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac * tau
        lr = lr_start * (lr_end / lr_start) ** frac
        
        conv = compute_conv(a)
        current_max = float(np.max(conv))
        
        if current_max < best_max - 1e-12:
            best_max = current_max
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > patience:
                # Reduce learning rate and add small perturbation
                lr *= 0.5
                if lr < 1e-10:
                    break
                # Small random perturbation
                rng = np.random.default_rng(it * 7 + n)
                a = a + rng.normal(0, 0.01, n) * lr
                a = project_simplex(a)
                stall_count = 0
                continue
        
        # Compute smoothed weights
        shifted = tau * (conv - current_max)
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights /= w_sum
        
        # Compute gradient
        grad = compute_smoothed_gradient(weights, a)
        
        # Gradient step with projection
        a_new = a - lr * grad
        a_new = project_simplex(a_new)
        
        # Verify improvement (accept-reject)
        conv_new = compute_conv(a_new)
        max_new = float(np.max(conv_new))
        
        if max_new <= current_max + 1e-10:
            a = a_new
        else:
            # Reject, try smaller step
            lr *= 0.7
            a_new = a - lr * grad
            a_new = project_simplex(a_new)
            a = a_new
    
    return best_a, best_max


def peak_deflation(a, max_iters=1000, lr=0.01, verbose=False):
    """
    Subgradient method on the exact max:
    minimize max_k conv[k] subject to a >= 0, sum(a) = 1.
    Focuses specifically on reducing the current peak.
    """
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    
    stall = 0
    for it in range(max_iters):
        conv = compute_conv(a)
        current_max = float(np.max(conv))
        
        if current_max < best_max - 1e-12:
            best_max = current_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.8
                if lr < 1e-10:
                    break
                continue
        
        # Exact subgradient: gradient of conv[k*] where k* = argmax
        k_star = int(np.argmax(conv))
        # d(conv[k*])/da[i] = a[k*-i] + a[i-k*] (for valid indices)
        grad = np.zeros(n)
        for i in range(n):
            val = 0.0
            if 0 <= k_star - i < n:
                val += a[k_star - i]
            if 0 <= i - k_star < n:
                val += a[i - k_star]
            grad[i] = val
        
        # Center gradient for sum constraint
        grad_centered = grad - np.mean(grad)
        
        a_new = a - lr * grad_centered
        a_new = project_simplex(a_new)
        a = a_new
    
    return best_a, best_max


def build_A_matrix(a):
    """Build the Toeplitz constraint matrix for the LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def solve_lp_for_g(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0, sum(g) <= 1."""
    n = len(a)
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    # Add constraint sum(g) <= 1 to normalize
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None


def lp_optimize(a, max_iter=30, t=0.3, verbose=False):
    """Alternating LP optimization."""
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    stall = 0
    for it in range(max_iter):
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g, sum_g = solve_lp_for_g(a, M)
        if g is None:
            break
        
        g = project_simplex(g)
        a_new = (1.0 - t) * a + t * g
        a_new = project_simplex(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-8:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
            if verbose:
                print(f"        LP it={it}: obj={best_obj:.6f}")
        else:
            stall += 1
            if stall > 15:
                break
        
        a = a_new
    
    return best_a, best_obj

    best_a = a.copy()
    best_obj = compute_obj(a)
    
    # Phase 1: Smoothed gradient descent (main workhorse)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        max_iters = int(min(10000, remaining * 5))
        a_sg, _ = smoothed_gradient_descent(a, max_iters=max_iters,
                                             tau_start=5.0, tau_end=100.0,
                                             lr_start=0.05, lr_end=0.0005)
        obj = compute_obj(a_sg)
        if obj < best_obj:
            best_obj = obj
            best_a = a_sg.copy()
            a = a_sg.copy()
            if verbose:
                print(f"      SG: obj={best_obj:.6f}")
    
    # Phase 2: Second round of smoothed GD with different params
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        max_iters = int(min(8000, remaining * 4))
        a_sg2, _ = smoothed_gradient_descent(a, max_iters=max_iters,
                                              tau_start=20.0, tau_end=300.0,
                                              lr_start=0.02, lr_end=0.000

1)
        obj = compute_obj(a_sg2)
        if obj < best_obj:
            best_obj = obj
            best_a = a_sg2.copy()
            a = a_sg2.copy()
            if verbose:
                print(f"      SG2: obj={best_obj:.6f}")
    
    # Phase 3: Peak deflation
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        max_iters = int(min(3000, remaining * 3))
        a_pd, _ = peak_deflation(a, max_iters=max_iters, lr=0.005)
        obj = compute_obj(a_pd)
        if obj < best_obj:
            best_obj = obj
            best_a = a_pd.copy()
            a = a_pd.copy()
            if verbose:
                print(f"      PD: obj={best_obj:.6f}")
    
    # Phase 4: LP alternation (if n is reasonable and time allows)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10 and n <= 300:
        a_lp, obj_lp = lp_optimize(a, max_iter=20, t=0.25)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            a = a_lp.copy()
            if verbose:
                print(f"      LP: obj={best_obj:.6f}")
    
    # Phase 5: Final polish with aggressive peak deflation
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        max_iters = int(min(5000, remaining * 5))
        a_final, _ = peak_deflation(a, max_iters=max_iters, lr=0.003)
        obj = compute_obj(a_final)
        if obj < best_obj:
            best_obj = obj
            best_a = a_final.copy()
            if verbose:
                print(f"      Final: obj={best_obj:.6f}")
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Search for the best sequence of coefficients minimizing evaluate_sequence.
    
    Strategy:
    1. Try parent sequence
    2. Use flat autocorrelation construction + geometric init for multiple n
    3. Run multi-phase optimization (SGD -> peak deflation -> LP -> polish)
    4. Random restarts if time remains
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    
    best_seq = None
    best_score = np.inf
    
    # Try parent sequence
    parent, parent_obj = try_parent_sequence()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        print(f"Starting with parent: obj={best_score:.6f}, n={len(parent)}")
    
    # Sequence lengths to explore
    n_values = [30, 50, 75, 100, 125, 150, 200, 250, 300, 400]
    
    # If we have a parent, include its neighborhood
    if parent is not None:
        n_parent = len(parent)
        extra = [max(10, n_parent - 30), n_parent, min(600, n_parent + 30)]
        n_values = sorted(set(n_values + extra))
    
    print(f"Exploring n values: {n_values}")
    
    # First pass: quick evaluation of initializations to find good n values
    print("\n--- Phase 1: Quick initialization scan ---")
    init_results = []
    for n in n_values:
        if time.monotonic() > deadline:
            break
        
        candidates = []
        
        # Flat autocorrelation
        a = flat_autocorrelation_construction(n)
        candidates.append(('flat_ac', a, compute_obj(a)))
        
        # Geometric (several ratios)
        for r in [0.7, 0.8, 0.85, 0.9, 0.95]:
            a = geometric_init(n, r)
            candidates.append((f'geo_{r}', a, compute_obj(a)))
        
        # Exponential
        for alpha in [0.5, 1.0, 2.0]:
            a = exponential_init(n, alpha)
            candidates.append((f'exp_{alpha}', a, compute_obj(a)))
        
        # Quadratic
        for c in [0.5, 1.0, 2.0]:
            a = quadratic_init(n, c)
            candidates.append((f'quad_{c}', a, compute_obj(a)))
        
        # Random
        for _ in range(2):
            a = random_init(n, rng)
            candidates.append(('rand', a, compute_obj(a)))
        
        # Find best initialization for this n
        best_init = min(candidates, key=lambda x: x[2])
        init_results.append((n, best_init[2], best_init[1]))
        
        if best_init[2] < best_score:
            best_score = best_init[2]
            best_seq = best_init[1].copy()
            print(f"  n={n}: init '{best_init[0]}' obj={best_score:.6f}")
    
    # Sort by best initialization objective
    init_results.sort(key=lambda x: x[1])
    print(f"\nBest inits: {[(n, f'{obj:.4f}') for n, obj, _ in init_results[:5]]}")
    
    # Second pass: full optimization on the most promising n values
    print("\n--- Phase 2: Full optimization ---")
    # Allocate more time to better initializations
    n_explored = set()
    for n, init_obj, init_a in init_results:
        if time.monotonic() > deadline - 60:
            break
        
        # Time allocation based on initial quality and n
        if init_obj < 1.6:
            alloc = 40.0
        elif init_obj < 1.8:
            alloc = 30.0
        else:
            alloc = 15.0
        
        # Scale by n (larger n needs more time)
        alloc = alloc * max(1.0, n / 100.0)
        alloc = min(alloc, 80.0)
        
        remaining = deadline - time.monotonic()
        if remaining < alloc + 30:
            alloc = min(alloc, remaining - 30)
        if alloc < 5:
            continue
        
        verbose = (n in [100, 150, 200, 250])
        a_opt, obj = optimize_sequence(init_a, alloc, verbose=verbose)
        
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  n={n}: obj={best_score:.6f} (init_obj={init_obj:.4f})")
        
        n_explored.add(n)
    
    # Third pass: try multiple random restarts on the best n
    print("\n--- Phase 3: Random restarts ---")
    if best_seq is not None:
        n_best = len(best_seq)
    else:
        n_best = 150
    
    restart_configs = [
        (n_best, 'restart_flat'),
        (n_best, 'restart_geo'),
        (min(600, n_best + 50), 'restart_flat'),
        (max(30, n_best - 50), 'restart_geo'),
    ]
    
    for n_r, method in restart_configs:
        if time.monotonic() > deadline - 30:
            break
        
        alloc = min(30.0, deadline - time.monotonic() - 30)
        if alloc < 5:
            break
        
        if method == 'restart_flat':
            a_init = flat_autocorrelation_construction(n_r)
            # Perturb it
            a_init = a_init * np.exp(rng.normal(0, 0.1, n_r))
            a_init = project_simplex(a_init)
        else:
            r = rng.uniform(0.6, 0.95)
            a_init = geometric_init(n_r, r)
        
        a_opt, obj = optimize_sequence(a_init, alloc, verbose=False)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Restart n={n_r} ({method}): obj={best_score:.6f}")
    
    # Final intensive polishing
    print("\n--- Phase 4: Final polishing ---")
    if best_seq is not None:
        n_fin = len(best_seq)
        remaining = deadline - time.monotonic()
        if remaining > 10:
            print(f"  Polishing n={n_fin}, {remaining:.0f}s remaining")
            
            a_cur = best_seq.copy()
            
            # Multiple rounds of peak deflation + SGD
            round_num = 0
            while time.monotonic() < deadline - 5:
                round_num += 1
                rem = deadline - time.monotonic()
                if rem < 5:
                    break
                
                # Peak deflation
                pd_time = min(rem * 0.4, 20.0)
                max_iters = int(pd_time * 5)
                a_cur, _ = peak_deflation(a_cur, max_iters=max_iters,
                                           lr=0.002 * (0.9 ** round_num))
                obj = compute_obj(a_cur)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_cur.copy()
                    print(f"    Round {round_num} PD: obj={best_score:.6f}")
                
                # SGD polish
                rem = deadline - time.monotonic()
                if rem > 5:
                    sg_time = min(rem * 0.4, 15.0)
                    max_iters = int(sg_time * 4)
                    a_cur, _ = smoothed_gradient_descent(a_cur, max_iters=max_iters,
                                                          tau_start=50.0, tau_end=500.0,
                                                          lr_start=0.01, lr_end=0.0001)
                    obj = compute_obj(a_cur)
                    if obj < best_score:
                        best_score = obj
                        best_seq = a_cur.copy()
                        print(f"    Round {round_num} SG: obj={best_score:.6f}")
                
                # LP step if reasonable
                rem = deadline - time.monotonic()
                if rem > 15 and n_fin <= 250:
                    a_lp, obj_lp = lp_optimize(a_cur, max_iter=15, t=0.2)
                    if obj_lp < best_score:
                        best_score = obj_lp
                        best_seq = a_lp.copy()
                        a_cur = a_lp.copy()
                        print(f"    Round {round_num} LP: obj={best_score:.6f}")
                    a_cur = project_simplex(a_cur)
                
                # Perturbation to escape local minima
                rem = deadline - time.monotonic()
                if rem > 10 and round_num % 3 == 0:
                    noise = np.exp(rng.normal(0, 0.05, n_fin))
                    a_pert = a_cur * noise
                    a_pert = project_simplex(a_pert)
                    obj_p = compute_obj(a_pert)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_pert.copy()
                        a_cur = a_pert.copy()
                        print(f"    Round {round_num} perturb: obj={best_score:.6f}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(50) / 50
        best_score = compute_obj(best_seq)
    
    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    
    # Validate
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if np.isfinite(validation) and validation != best_score:
        print(f"  Validation: computed={best_score:.6f}, validated={validation:.6f}")
        if validation < best_score:
            best_score = validation
    
    return result
