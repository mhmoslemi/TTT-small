"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Diverse initializations (chirp, asymmetric, random, parent)
2. LP best-response iteration with backtracking
3. Smoothed projected gradient descent (log-sum-exp) with Adam
4. Multi-start perturbation and re-optimization
5. Adaptive time allocation
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build matrix A where (A @ x) = convolve(a, x) (full)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None


def lp_iteration(a, max_iters, deadline, base_t=0.5):
    """LP best-response with adaptive step size and backtracking."""
    best_local = a.copy()
    best_score = compute_score(a)
    a = a.copy()
    a = np.clip(a, 0.001, None)
    a = a / a.sum()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()

        # Backtracking line search
        t = base_t
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        score = compute_score(a_new)

        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            improved = False
            for _ in range(5):
                t *= 0.5
                if t < 0.005:
                    break
                a_try = (1.0 - t) * a + t * g
                a_try = np.clip(a_try, 0.0, None)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    break
                a_try = a_try / s_try
                score_try = compute_score(a_try)
                if score_try < best_score:
                    best_score = score_try
                    best_local = a_try.copy()
                    a = a_try
                    improved = True
                    break
            if not improved and t < 0.01:
                break

    return best_local, best_score


def smooth_grad_descent(a0, beta_schedule, lr_schedule, deadline, n_max_iter_per_beta=2000):
    """Projected gradient descent on smoothed log-sum-exp objective."""
    n = len(a0)
    a = a0.copy()
    a = np.clip(a, 0.001, None)
    a = a / a.sum()
    
    best = a.copy()
    best_score = compute_score(a)
    
    # Adam state
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t_step = 0
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        lr_base = lr_schedule.get(beta, 0.01)
        
        for it in range(n_max_iter_per_beta):
            if time.monotonic() > deadline:
                break
            
            # Compute b = conv(a, a)
            b = fast_conv(a)
            b_max = b.max()
            
            # Softmax weights for smooth max
            shifted = beta * (b - b_max)
            shifted = np.clip(shifted, -50, 50)
            w = np.exp(shifted)
            w = w / w.sum()
            
            # Gradient: d(2n*max_smooth)/da_j = 4n * sum_{m=0}^{n-1} a_m * w_{m+j}
            # (derived earlier)
            # g[j] = sum_{m=0}^{n-1} a[m] * w[m+j]
            # This is the "valid" cross-correlation of a and w
            g = np.zeros(n)
            for j in range(n):
                g[j] = np.dot(a, w[j:j+n])
            
            grad = 4.0 * n * g
            
            # Adam update
            t_step += 1
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t_step)
            v_hat = v / (1 - beta2**t_step)
            a = a - lr_base * m_hat / (np.sqrt(v_hat) + eps)
            
            # Project onto simplex {x >= 0, sum(x) = 1}
            a = np.clip(a, 0, None)
            s = a.sum()
            if s < 1e-10:
                a = np.ones(n) / n
            else:
                a = a / s
            
            # Check improvement periodically
            if it % 50 == 0:
                score = compute_score(a)
                if score < best_score:
                    best_score = score
                    best = a.copy()
    
    # Final score check
    score = compute_score(a)
    if score < best_score:
        best_score = score
        best = a.copy()
    
    return best, best_score
