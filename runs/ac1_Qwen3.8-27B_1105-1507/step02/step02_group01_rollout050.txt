"""

on soft-max surrogate with temperature annealing, LP direction finding,
coordinate descent, and multi-scale refinement. Targets objective below 1.5027.

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def softmax_value_and_grad(a, T):
    """Compute soft-max value and its gradient wrt a."""
    n = len(a)
    conv = np.convolve(a, a)
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    sm_val = max_c + T * np.log(sum_exp)

    # grad[i] = (2/T) * sum_k p[k] * a[k-i] where 0<=k-i<n
    # = (2/T) * np.convolve(p, a[::-1])[n-1 : n-1+n]
    a_rev = a[::-1]
    conv_pa_rev = np.convolve(p, a_rev)
    grad = (2.0 / T) * conv_pa_rev[n - 1:n - 1 + n]
    return sm_val, p, grad


def true_subgrad_direction(a):
    """Compute the true subgradient direction (pointing to reduce the max lag)."""
    n = len(a)
    conv = np.convolve(a, a)
    max_c = float(np.max(conv))
    # Find active lags
    active_mask = (conv >= max_c - 1e-10 * max_c)
    p = np.zeros(len(conv))
    p[active_mask] = 1.0 / active_mask.sum()

    a_rev = a[::-1]
    conv_pa_rev = np.convolve(p, a_rev)
    grad = 2.0 * conv_pa_rev[n - 1:n - 1 + n]
    return grad, max_c, active_mask


def adam_softmax(a, max_time, T_initial=0.05, T_final=0.001, lr_initial=0.05):
    """Adam optimizer on soft-max surrogate with temperature annealing."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Normalize
    s = np.sum(a)
    if s < 1e-10:
        return best_a, best_obj
    a = a / s

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_initial

    conv = np.convolve(a, a)
    T = T_initial * float(np.max(conv))

    while time.monotonic() - start < max_time:
        t += 1
        # Anneal temperature
        frac = min(1.0, t / max(1, int(max_time / 0.05)))
        T = T_initial * float(np.max(conv)) * (T_final / T_initial) ** (0.3 * frac)
        T = max(T, 1e-4 * float(np.max(conv)))

        sm_val, p, grad = softmax_value_and_grad(a, T)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        # Normalize gradient to have unit norm for stable updates
        grad = grad / (grad_norm + 1e-12)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        update = m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - lr * update
        a_new = np.clip(a_new, 0.0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            # Slightly increase lr on improvement
            lr = min(0.1, lr * 1.001)
        else:
            lr *= 0.995
            if lr < 1e-5:
                # Try to escape with a random perturbation
                a = a + 0.01 * np.random.randn(n)
                a = np.clip(a, 0.0, None)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                lr = 0.01

    return best_a, best_obj

    while time.monotonic() - start < max_time:

        grad = grad / grad_norm

        momentum = beta * momentum + (1 - beta) * grad

def lp_direction(a, upper_bound=100.0):
    """Solve LP to find a good direction."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))

    # Build A matrix: A[k, i] = a[k-i] if 0 <= k-i < n else 0
    k_range = np.arange(2 * n - 1)
    i_range = np.arange(n)
    diff = k_range[:, None] - i_range[None, :]
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[mask] = a[diff[mask]]

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 10})
    if res.status == 0:
        return res.x
    return None


def lp_phase(a, max_time, t_start=0.5, t_min=0.1):
    """LP-based alternating optimization."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        g = lp_direction(a)
        if g is None:
            break
        g_sum = float(np.sum(g))
        if g_sum <= 1.0 + 1e-5:
            stall += 1
            if stall > 10:
                break
            continue
        stall = 0
        g = g / g_sum

        # Try multiple step sizes
        t = max(t_min, t_start * (1.0 - it / max(100, it)))
        for tt in [t, t * 0.5, t * 0.2, t * 2.0]:
            a_new = (1.0 - tt) * a + tt * g
            a_new = np.clip(a_new, 0.0, None)
            s = np.sum(a_new)
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_a = a_new.copy()
                best_obj = obj
                a = a_new
                break
        else:
            # No improvement, take small step anyway
            a = (1.0 - t * 0.1) * a + t * 0.1 * g
            a = np.clip(a, 0.0, None)
            s = np.sum(a)
            if s > 1e-10:
                a = a / s

    return best_a, best_obj


def block_refine(a, max_time, rng_seed=None):

    else:
        rng = np.random.default_rng()

        candidate = candidate / s
        obj = compute_obj(candidate)

            best_obj = obj
            consec_fail = 0
        else:

def structured_init(n, rng):

    elif choice == 1:

        a += 0.1

        return a / s
    elif choice == 2:

        period = int(rng.integers(max(2, n // 50), max(3, n // 10)))

        a += 0.05
        s = np.sum(a)
        return a / s
    elif choice == 3:
        # Gaussian
        center = n / 2.0
        width = n * rng.uniform(0.15, 0.5)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        a += 0.1
        s = np.sum(a)
        return a / s
    elif choice == 4:
        # Two peaks

    elif choice == 5:

    else:

def full_optimize(a_init, time_budget, rng_seed=None):
    """Multi-phase optimization pipeline."""
    if rng_seed is not None:
        rng = np.random.default_rng(rng_seed)
    else:
        rng = np.random.default_rng()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    def remaining():
        return time_budget - (time.monotonic() - start)

    # Phase 1: Adam soft-max (largest time allocation)
    if remaining() > 5:
        adam_time = min(remaining() * 0.4, 60.0)
        a1, obj1 = adam_softmax(a, adam_time, T_initial=0.1, T_final=0.005, lr_initial=0.03)
        if obj1 < best_obj:
            best_a = a1.copy()
            best_obj = obj1

    # Phase 2: Subgradient descent
    if remaining() > 5:
        sg_time = min(remaining() * 0.2, 20.0)
        a2, obj2 = subgrad_descent(best_a, sg_time, lr_initial=0.005)
        if obj2 < best_obj:
            best_a = a2.copy()
            best_obj = obj2

    # Phase 3: LP direction finding
    if remaining() > 5:
        lp_time = min(remaining() * 0.25, 30.0)
        a3, obj3 = lp_phase(best_a, lp_time, t_start=0.3, t_min=0.05)
        if obj3 < best_obj:
            best_a = a3.copy()
            best_obj = obj3

    # Phase 4: Adam again (finer)
    if remaining() > 3:
        adam_time2 = min(remaining() * 0.25, 30.0)
        a4, obj4 = adam_softmax(best_a, adam_time2, T_initial=0.02, T_final=0.001, lr_initial=0.01)
        if obj4 < best_obj:
            best_a = a4.copy()
            best_obj = obj4

    # Phase 5: Block refinement
    if remaining() > 3:
        bp_time = min(remaining() * 0.4, 40.0)
        a5, obj5 = block_refine(best_a, bp_time, rng_seed=int(rng.integers(0, 2**31)))
        if obj5 < best_obj:
            best_a = a5.copy()
            best_obj = obj5

    # Phase 6: Coordinate descent
    if remaining() > 2:
        cd_time = remaining() - 1.0
        a6, obj6 = coordinate_descent(best_a, cd_time, rng_seed=int(rng.integers(0, 2**31)))
        if obj6 < best_obj:
            best_a = a6.copy()
            best_obj = obj6

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization: diverse initializations -> Adam soft-max ->
    subgradient descent -> LP -> fine Adam -> block refine -> coordinate descent.
    Tries multiple sequence lengths with time-adaptive allocation.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 8.0

    best_seq = None
    best_score = np.inf

    # Try parent if available
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Optimize the parent thoroughly
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all() and parent.size <= 500:
            p_time = min(120.0, (deadline - time.monotonic()) * 0.3)
            p_opt, p_obj = full_optimize(parent.copy(), p_time, rng_seed=123)
            if p_obj < best_score:
                best_score = p_obj
                best_seq = p_opt.copy()
            print(f"Parent optimized (n={parent.size}): obj={p_obj:.6f}")
    except Exception as e:
        print(f"Parent optimization error: {e}")

    # Explore different n values
    n_values = [200, 250, 300, 350, 400, 450, 500]

    for n in n_values:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        num_remaining_n = len([x for x in n_values if x >= n])
        time_per_n = remaining / max(1, num_remaining_n)
        time_per_n = min(time_per_n, 200.0)

        num_restarts = max(1, min(3, int(time_per_n / 40)))
        time_per_restart = time_per_n / num_restarts

        print(f"\nN={n}: remaining={remaining:.0f}s, {num_restarts} restarts, "
              f"{time_per_restart:.0f}s each")

        for r in range(num_restarts):
            if time.monotonic() > deadline - 30:
                break
            a_init = structured_init(n, rng)
            a_opt, obj_opt = full_optimize(a_init, time_per_restart,
                                           rng_seed=int(rng.integers(0, 2**31)))
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} r={r}: obj={obj_opt:.6f} ** NEW BEST **")
            else:
                print(f"  n={n} r={r}: obj={obj_opt:.6f}")

    # Final intensive refinement on the best found
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        print(f"\nFinal intensive refinement: {remaining:.0f}s budget")

        # Multiple rounds of refinement with different strategies
        for round_i in range(3):
            if time.monotonic() >= deadline - 5:
                break
            round_budget = (deadline - time.monotonic() - 5) / (3 - round_i)
            round_budget = min(round_budget, 60.0)

            # Perturb slightly then re-optimize
            a_pert = best_seq.copy() + 0.005 * np.random.randn(len(best_seq))
            a_pert = np.clip(a_pert, 0.0, None)
            s = np.sum(a_pert)
            if s > 1e-10:
                a_pert = a_pert / s
            a_ref, obj_ref = full_optimize(a_pert, round_budget,
                                           rng_seed=int(rng.integers(0, 2**31)))
            if obj_ref < best_score:
                best_score = obj_ref
                best_seq = a_ref.copy()
                print(f"  Round {round_i}: obj={best_score:.6f} ** NEW BEST **")
            else:
                print(f"  Round {round_i}: obj={obj_ref:.6f} (no improvement)")

    # Very last: coordinate descent + block refine with remaining time
    if best_seq is not None and time.monotonic() < deadline - 2:
        rem = deadline - time.monotonic() - 2
        if rem > 2:
            a_cd, obj_cd = coordinate_descent(best_seq, rem / 2,
                                              rng_seed=int(rng.integers(0, 2**31)))
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  After CD: obj={best_score:.6f}")
            if time.monotonic() < deadline - 2:
                rem2 = deadline - time.monotonic() - 2
                if rem2 > 2:
                    a_bp, obj_bp = block_refine(best_seq, rem2,
                                                rng_seed=int(rng.integers(0, 2**31)))
                    if obj_bp < best_score:
                        best_score = obj_bp
                        best_seq = a_bp.copy()
                        print(f"  After BP: obj={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    # Ensure all values are in valid range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
