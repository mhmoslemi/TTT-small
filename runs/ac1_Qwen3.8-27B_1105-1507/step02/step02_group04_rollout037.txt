def compute_conv(a):
    """Compute autocorrelation conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:

    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smoothed_grad_of_max(conv, tau):
    """
    Compute gradient of smooth max approximation.
    Returns (smooth_max_value, gradient_wrt_conv) where gradient has same length as conv.
    """
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum
    return max_conv + np.log(w_sum) / tau, weights

def build_toeplitz(a):
    """Build the matrix A such that (A @ g)[k] = conv(a, g)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        # conv(a, g)[k] = sum_j a[j] * g[k-j] for valid j
        # So A[k, j] = a[j] if k-j is in [0, n-1], i.e., j in [k-(n-1), k]
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[j]
    return A


def lp_direction(a, M, time_limit=20):
    """
    Solve LP: max sum(g) s.t. conv(a, g)[k] <= M for all k, g >= 0.
    Returns g (normalized to sum=1) or None.
    """
    n = len(a)
    if n > 500:
        return None  # Too large for LP
    
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = res.x
            s = np.sum(g)
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

Combined optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated smoothed gradient, active-set LP,
spectral shaping initialization, and intensive polishing.

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute conv(a,a) via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def logsumexp_max_and_weights(conv, tau):
    """Return (smooth_max, weights) for log-sum-exp approximation of max."""
    m = np.max(conv)
    shifted = tau * (conv - m)
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    ws = np.sum(w)
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    return m + np.log(ws) / tau, w


def grad_from_weights(weights, a):
    """Gradient of sum_k weights[k]*conv(a,a)[k] w.r.t. a."""
    n = len(a)
    c = np.correlate(weights, a, mode='full')
    return 2.0 * c[:n]


def nesterov_smoothed_opt(a, max_iter, tau_start, tau_end, step_init, momentum_decay=0.9995):
    """Nesterov-accelerated projected gradient with LSE smoothing."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    conv = compute_conv(a)
    best_max = float(np.max(conv))

    y = a.copy()  # Nesterov lookahead
    t_k = 1.0
    step = step_init
    stall = 0
    patience = 300

    for it in range(max_iter):
        frac = (it + 1) / max_iter
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.5)  # slow start, fast end

        conv_y = compute_conv(y)
        _, weights = logsumexp_max_and_weights(conv_y, tau)
        grad = grad_from_weights(weights, y)
        grad -= np.mean(grad)  # center for sum constraint

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta = (t_k - 1.0) / t_new

        y_new = y - step * grad
        y_new = np.maximum(y_new, 0)
        sy = np.sum(y_new)
        if sy < 1e-10:
            y_new = np.ones(n) / n
            sy = 1.0
        y_new = y_new / sy

        a_new = y_new + beta * (y_new - a)
        a_new = np.maximum(a_new, 0)
        sa = np.sum(a_new)
        if sa < 1e-10:
            a_new = y_new
            sa = sy
        a_new = a_new / sa

        a, y = a_new, y_new
        t_k = t_new

        conv_a = compute_conv(a)
        cur_max = float(np.max(conv_a))
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > patience:
            step *= 0.5
            if step < 1e-10:
                break
            # Inject noise
            rng = np.random.default_rng(it * 7919)
            noise = rng.normal(0, 0.02, n)
            a = a + step * noise
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s < 1e-10:
                break
            a = a / s
            y = a.copy()
            t_k = 1.0
            stall = 0

        if it > 50:
            step *= momentum_decay

    return best_a, best_max


def build_toeplitz(a):
    """Build matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[j_min:j_max + 1]
    return A


def lp_step(a, M, time_limit=15):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    if n > 400:
        return None
    try:
        A = build_toeplitz(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = res.x
            s = np.sum(g)
            if s > 1e-8:
                return g / s
    except Exception:
        pass
    return None


def lp_polish(a, max_iters=30, t=0.3, time_limit=15):
    """Repeated LP-based direction updates."""
    n = len(a)
    a = a.astype(np.float64).copy()

    a = a / s

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    stall = 0

    for it in range(max_iters):
        conv = compute_conv(a)
        M = float(np.max(conv))
        g = lp_step(a, M, time_limit=time_limit)
        if g is None:
            break

        sum_g = 1.0  # already normalized
        # Check if LP found something better than current
        conv_g = np.zeros(2 * n - 1)
        # Actually g is normalized to sum 1, so let's just blend
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        new_max = float(np.max(compute_conv(a_new)))
        if new_max < best_max:
            best_max = new_max
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            t *= 0.7
            if t < 0.01:
                break
        a = a_new

        if stall > 15:
            break

    return best_a, best_max


def spectral_init(n, seed, n_iter=200):
    """Generate initialization via iterative spectral flattening."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
    a = a / s

    N = 1
    while N < 2 * n:
        N <<= 1

    for _ in range(n_iter):
        A = np.fft.fft(a, N)
        mags = np.abs(A).copy()
        mags[0] = 0
        nonzero = mags[1:]
        # Target: reduce peak-to-mean ratio
        mean_mag = np.mean(nonzero[nonzero > 1e-10]) if np.any(nonzero > 1e-10) else 1.0
        # Soft clipping: reduce magnitudes above 2*mean
        threshold = 2.0 * mean_mag
        excess = np.maximum(nonzero - threshold, 0)
        nonzero -= 0.5 * excess
        nonzero = np.maximum(nonzero, 0.01 * mean_mag)
        mags[1:] = nonzero

        A_new = mags * np.exp(1j * np.angle(A))
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a = a_new / s_new

    return a


def generate_init(n, seed, method='random'):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)

    if method == 'random':
        a = rng.random(n)
    elif method == 'cosine':
        freq = rng.uniform(1.0, 8.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 * (1 + np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
        a += 0.2 * rng.random(n)
    elif method == 'two_bump':
        c1 = rng.uniform(0.1, 0.35) * n
        c2 = rng.uniform(0.65, 0.9) * n
        w1 = rng.uniform(0.03, 0.12) * n
        w2 = rng.uniform(0.03, 0.12) * n
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - c1) / w1) ** 2) + np.exp(-0.5 * ((x - c2) / w2) ** 2)
    elif method == 'multi_bump':
        a = np.zeros(n)
        x = np.arange(n)
        for _ in range(rng.integers(3, 7)):
            c = rng.uniform(0, n)
            w = rng.uniform(n * 0.02, n * 0.08)
            h = rng.uniform(0.3, 1.0)
            a += h * np.exp(-0.5 * ((x - c) / w) ** 2)
    elif method == 'alternating':
        a = np.ones(n)
        a[::2] = rng.uniform(0.1, 0.4)
        a[1::2] = rng.uniform(0.6, 1.0)
    elif method == 'spectral':
        a = spectral_init(n, seed, n_iter=100)
        return a  # Already normalized
    elif method == 'ramp':
        a = np.linspace(0.1, 1.0, n)
        a += rng.normal(0, 0.1, n)
    else:
        a = rng.random(n)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s
