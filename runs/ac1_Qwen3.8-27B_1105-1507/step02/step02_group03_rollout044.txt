"""
Step function coefficient optimization using iterative LP refinement with
adaptive target tightening, spectral phase retrieval initialization,
pressure-based gradient steps, and coordinate descent polishing.
Multi-start across sequence lengths with careful time management.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute 2*n*max(conv)/sum^2 for a sequence."""
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)


def normalize(f):
    """Normalize to sum=1."""
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def build_A(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n, k + 1)
        for j in range(j_lo, j_hi):
            A[k, j] = f[k - j]
    return A


def solve_lp(f, M_target, time_limit=2):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M_target, b >= 0."""
    n = len(f)
    A = build_A(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None


def line_search(f, direction, best_score, n_steps=8):
    """Line search: find best step size along (1-t)*f + t*direction."""
    n = len(f)
    best_t = 0.0
    best_sc = best_score
    for i in range(1, n_steps + 1):
        t = 2.0 ** (-i)  # 0.5, 0.25, 0.125, ...
        f_new = (1.0 - t) * f + t * direction
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        sc = compute_score(f_new)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    return f_new if best_t > 0 else f, best_sc, best_t


def lp_refine(f, time_budget, rng, lp_time=2, max_iter=200):
    """Iterative LP refinement with adaptive target and line search."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    best_f = f.copy()
    best_score = compute_score(f)
    start = time.monotonic()
    stall = 0

    for it in range(max_iter):
        if time.monotonic() - start > time_budget:
            break

        conv = np.convolve(f, f)
        M_curr = conv.max()

        # Try progressively tighter targets
        improved = False
        for target_frac in [0.92, 0.85, 0.75, 0.65, 0.5]:
            M_target = M_curr * target_frac
            b = solve_lp(f, M_target, time_limit=lp_time)
            if b is None or b.sum() < 1e-10:
                continue
            direction = b / b.sum()

            # Line search
            f_cand, sc, t_used = line_search(f, direction, best_score)
            if sc < best_score - 1e-13:
                best_f = f_cand.copy()
                best_score = sc
                f = f_cand
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall > 15:
                # Try perturbation to escape
                f_pert = f * (1.0 + 0.03 * rng.standard_normal(n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                sc_pert = compute_score(f_pert)
                if sc_pert < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = sc_pert
                    stall = 0
                elif stall > 40:
                    break
            elif stall % 10 == 0:
                # Small perturbation
                f = f * (1.0 + 0.01 * rng.standard_normal(n))
                f = np.clip(f, 0, None)
                f = normalize(f)

    return best_f, best_score


def spectral_phase_retrieval(n, n_iter=150, rng=None):
    """Phase retrieval: alternate flat-spectrum and non-negative constraints."""
    if rng is None:
        rng = np.random.default_rng(42)
    f = np.ones(n) / n
    n_fft = 1 << (2 * n - 1).bit_length()
    best_f = f.copy()
    best_score = compute_score(f)

    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)

        # Target: flatten non-DC magnitudes
        non_dc = mag[1:]
        target_mag = non_dc.mean() if len(non_dc) > 0 else 1.0

        new_mag = np.full(len(mag), target_mag)
        new_mag[0] = 1.0

        # Occasionally randomize phase to escape
        if it % 30 == 29:
            phase = phase + rng.normal(0, 0.1, len(phase))

        F_new = new_mag * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s

        sc = compute_score(f_new)
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
        f = f_new

    return best_f, best_score


def pressure_step(f, eta=0.08):
    """Reduce elements that contribute most to convolution peaks."""
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()
    threshold = 0.85 * M
    peak_pos = np.where(conv >= threshold)[0]

    pressure = np.zeros(n)
    for p in peak_pos:
        j_lo = max(0, p - n + 1)
        j_hi = min(n, p + 1)
        for j in range(j_lo, j_hi):
            pressure[j] += f[p - j]

    if pressure.max() > 0:
        pressure /= pressure.max()

    f_new = f * np.exp(-eta * pressure)
    f_new = np.clip(f_new, 1e-15, None)
    return normalize(f_new)


def coordinate_descent(f, n_sweeps=5, rng=None):
    """Coordinate descent with multiplicative factor search."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)
    factors = [0.3, 0.5, 0.7, 0.85, 1.17, 1.4, 2.0, 3.0]

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            orig = f[i]
            if orig < 1e-12:
                # Try turning on
                for fv in [0.01, 0.005, 0.002]:
                    f[i] = fv
                    sc = compute_score(f)
                    if sc < best_score:
                        best_score = sc
                f[i] = orig
                continue
            best_val = orig
            best_local = best_score
            for factor in factors:
                f[i] = orig * factor
                s = f.sum()
                if s < 1e-10:
                    f[i] = orig
                    continue
                conv = np.convolve(f, f)
                sc = 2.0 * n * conv.max() / (s * s)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_val = orig * factor
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
        f = np.clip(f, 0, None)
        f = normalize(f)

    return f, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    r = rng.random()
    if r < 0.2:
        f = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / n))
    elif r < 0.4:
        beta = rng.uniform(0.5, 2.5)
        f = np.abs(np.cos(np.pi * np.arange(n) / n * beta)) + 0.05
    elif r < 0.6:
        c1 = n * rng.uniform(0.1, 0.35)
        c2 = n * rng.uniform(0.65, 0.9)
        w = n * rng.uniform(0.05, 0.2)
        f = (np.exp(-0.5 * (np.arange(n) - c1) ** 2 / w ** 2) +
             np.exp(-0.5 * (np.arange(n) - c2) ** 2 / w ** 2))
    elif r < 0.8:
        f = np.ones(n) * 0.5
        for _ in range(4):
            c = int(rng.integers(0, n))
            w = int(max(3, rng.integers(3, n // 4)))
            kernel = np.exp(-0.5 * (np.arange(n) - c) ** 2 / w ** 2)
            f *= (1.0 + rng.uniform(-0.6, 0.6) * kernel)
        f = np.abs(f) + 0.05
    else:
        f = np.ones(n)
        f = f * (1.0 + 0.3 * rng.standard_normal(n))
        f = np.abs(f) + 0.1

    f = np.clip(f, 0.01, None)
    f = normalize(f)
    f = f * (1.0 + 0.02 * rng.standard_normal(n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f


def multi_length_pipeline(n, parent_f, rng, time_budget):
    """Full pipeline for a given length: init -> spectral -> pressure -> LP -> CD."""
    start = time.monotonic()

    # Try multiple initializations
    best_f = None
    best_score = np.inf

    n_inits = min(3, max(1, int(time_budget // 60)))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget * 0.85:
            break

        if init_idx == 0 and parent_f is not None and len(parent_f) == n:
            f = parent_f * (1.0 + 0.08 * rng.standard_normal(n))
            f = np.clip(f, 0, None)
            f = normalize(f)
        elif init_idx == 1:
            f, s_spec = spectral_phase_retrieval(n, n_iter=100, rng=rng)
        else:
            f = diverse_init(n, rng)
            f, s_spec = spectral_phase_retrieval(n, n_iter=50, rng=rng)

        sc = compute_score(f)

        # Short pressure refinement

            if time.monotonic() - start > time_budget * 0.85:
                break
            f_new = pressure_step(f, eta=0.06)
            sc_new = compute_score(f_new)
            if sc_new < sc:
                f = f_new
                sc = sc_new

        if sc < best_score:
            best_score = sc
            best_f = f.copy()

    if best_f is None:
        return None, np.inf

    # LP refinement (main optimization)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        t_lp = min(remaining * 0.7, time_budget * 0.6)

        n_sweeps = min(8, max(2, int(remaining // 8)))
        best_f, best_score = coordinate_descent(best_f, n_sweeps=n_sweeps, rng=rng)

    return best_f, best_score
