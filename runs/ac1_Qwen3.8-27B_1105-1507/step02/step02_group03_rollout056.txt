def lp_step(f, M, time_limit=2.0):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.clip(res.x, 0, None)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

def coordinate_descent(f, n_sweeps=10, rng=None):

        for i in order:
            orig = f[i]
            best_local = best_score
            best_val = orig
            
            for factor in [0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.5]:

                    best_val = orig * factor
            
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
        
        f = normalize(f)
    
    return f, best_score

def compute_score(f):

    for j in range(n):

        target_mag = np.sqrt(total_energy / n_bins)
        flat_mag = np.full(n_bins, target_mag)
        flat_mag[0] = mag[0]  # DC = sum(f) = 1

        F_new = flat_mag * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s

        score = compute_score(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
        f = f_new

        # Periodically add phase noise to escape local minima
        if it % 100 == 99:
            phase_noisy = phase + rng.normal(0, 0.05, n_bins)
            F_new2 = flat_mag * np.exp(1j * phase_noisy)
            f_pert = np.fft.irfft(F_new2, n_fft)[:n]
            f_pert = np.clip(f_pert, 0, None)
            sp = f_pert.sum()
            if sp > 1e-10:
                f_pert = f_pert / sp
                if compute_score(f_pert) < best_score:
                    best_score = compute_score(f_pert)
                    best_f = f_pert.copy()
                    f = f_pert

    return best_f


def diverse_init(n, rng):
    """Create a diverse random initial sequence."""
    r = rng.random()
    x = np.arange(n)
    if r < 0.25:
        c1, c2 = rng.uniform(0.2, 0.5), rng.uniform(0.5, 0.8)
        f = np.exp(-((x - n * c1) ** 2) / (n * 0.1)) + np.exp(-((x - n * c2) ** 2) / (n * 0.08))
    elif r < 0.5:
        f = 0.5 + 0.3 * np.sin(2 * np.pi * x / n * rng.uniform(1, 4)) + 0.2 * np.cos(4 * np.pi * x / n)
    elif r < 0.75:
        f = np.abs(np.sin(np.pi * x / n * rng.uniform(1, 3))) + 0.3
    else:
        f = np.ones(n)
        for _ in range(4):
            c = int(rng.integers(0, n))
            w = int(rng.integers(5, n // 3))
            f *= (1 + 0.5 * rng.standard_normal() * np.exp(-0.5 * ((x - c) / w) ** 2))
    f = np.clip(f, 0.01, None)
    f = f * (1 + 0.1 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    return normalize(f)


def solve_lp(f, M, time_limit=3.0):
    """Solve: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit,
                                                'dual_feasibility_tolerance': 1e-8,
                                                'primal_feasibility_tolerance': 1e-8})
        if res.success and res.x is not None:
            g = np.clip(res.x, 0, None)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def solve_lp_tight(f, M, time_limit=3.0):
    """Try LP with progressively tighter bounds for better direction."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    for factor in [1.0, 0.999, 0.997, 0.995, 0.99, 0.98]:
        b_ub = np.full(2 * n - 1, M * factor)
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': time_limit,
                                                    'dual_feasibility_tolerance': 1e-8,
                                                    'primal_feasibility_tolerance': 1e-8})
            if res.success and res.x is not None:
                g = np.clip(res.x, 0, None)
                s = g.sum()
                if s > 1e-10:
                    return g / s, factor
        except Exception:
            pass
    return None, 1.0


def lp_optimize(f, time_budget, rng, verbose=True):
    """Main LP optimization loop with adaptive step size and momentum."""
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    cur_score = best_score
    t = 0.3
    start = time.monotonic()
    stall = 0
    momentum = np.zeros(n)
    beta = 0.7
    iteration = 0

    if verbose:
        print(f"  LP start: n={n}, score={best_score:.6f}, budget={time_budget:.0f}s")

    while time.monotonic() - start < time_budget:
        iteration += 1
        conv = np.convolve(f, f)
        M = conv.max()

        # Solve LP (try tight first, fall back to loose)
        g, factor = solve_lp_tight(f, M, time_limit=2.0)
        if g is None:
            g = solve_lp(f, M, time_limit=2.0)
            if g is None:
                break

        # Compute direction with momentum
        direction = g - f
        momentum = beta * momentum + (1 - beta) * direction

        # Line search over multiple step sizes
        improved = False
        candidates = [t, t * 0.5, t * 0.75, t * 1.5, 0.1, 0.2, 0.5, 0.8]
        candidates = [min(1.0, max(0.01, c)) for c in candidates]
        candidates = list(dict.fromkeys(candidates))  # Remove duplicates, preserve order

        for step in candidates:
            # Use momentum-enhanced direction
            f_new = f + step * momentum
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_score = compute_score(f_new)
            if new_score < cur_score - 1e-13:
                f = f_new
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_f = f.copy()
                    if verbose and (iteration % 20 == 0 or best_score < 1.505):
                        print(f"    iter {iteration}: score={best_score:.6f}, t={step:.3f}, factor={factor:.4f}")
                improved = True
                break

        if improved:
            stall = 0
            t = min(1.0, t * 1.03)
        else:
            # Try direct interpolation (without momentum) as fallback
            for step in [t * 0.5, t * 0.25, 0.05, 0.1]:
                f_new = (1 - step) * f + step * g
                f_new = np.clip(f_new, 0, None)
                s = f_new.sum()
                if s < 1e-10:
                    continue
                f_new = f_new / s
                new_score = compute_score(f_new)
                if new_score < cur_score - 1e-13:
                    f = f_new
                    cur_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_f = f.copy()
                    improved = True
                    break

            if not improved:
                stall += 1
                t *= 0.6
                beta = max(0.3, beta * 0.95)

                if t < 0.01 or stall > 30:
                    # Perturb to escape
                    strength = 0.03 + 0.02 * min(stall / 10, 2.0)
                    f_pert = f * np.exp(rng.normal(0, strength, n))
                    f_pert = np.clip(f_pert, 0, None)
                    sp = f_pert.sum()
                    if sp > 1e-10:
                        f_pert = f_pert / sp
                    pert_score = compute_score(f_pert)
                    if pert_score < best_score:
                        f = f_pert
                        best_f = f.copy()
                        best_score = pert_score
                        cur_score = pert_score
                        t = 0.3
                        beta = 0.7
                        stall = 0
                    elif stall > 60:
                        break

    return best_f, best_score


def pressure_descent(f, n_iter=200, eta=0.1, rng=None):
    """Fast pressure-based update to reduce convolution peaks."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        conv = np.convolve(f, f)
        # Weighted pressure: emphasize positions near the max
        M = conv.max()
        weight = np.exp(-0.5 * (conv - M) ** 2 / max(M * 0.1, 1e-10) ** 2)

        # Pressure on f[i]: sum over k of weight[k] * f[k-i]
        pressure = np.zeros(n)
        for i in range(n):
            k_vals = np.arange(i, min(i + n, 2 * n - 1))
            f_vals = f[0:min(n, 2 * n - 1 - i)]
            w_vals = weight[k_vals[:len(f_vals)]]
            pressure[i] = np.sum(w_vals * f_vals)

        if pressure.max() > 0:
            pressure = pressure / pressure.max()

        eta_eff = eta / (1.0 + 0.05 * it)
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break

    return best_f, best_score


def coordinate_descent(f, n_sweeps=10, rng=None):
    """Coordinate descent polishing."""
    if rng is None:
        rng = np.random.default_rng(42)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            orig = f[i]
            if orig < 1e-12:
                continue
            best_local = best_score
            best_val = orig
            for factor in [0.4, 0.6, 0.8, 0.9, 1.1, 1.2, 1.4, 1.7]:
                f[i] = orig * factor
                new_score = compute_score(f)
                if new_score < best_local:
                    best_local = new_score
                    best_val = orig * factor
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
        f = normalize(f)

    return f, best_score


def randomized_search(f, rng, n_iter=2000, T0=0.02, T_end=0.0001):
    """Simulated annealing with diverse moves."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    cur_score = best_score

    for it in range(n_iter):
        T = T0 * (T_end / T0) ** (it / n_iter)
        f_new = f.copy()
        r = rng.random()

        if r < 0.25:
            # Block scaling
            width = max(1, int(rng.integers(1, max(2, n // 12))))
            start = int(rng.integers(0, n - width + 1))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.5))
        elif r < 0.5:
            # Gaussian perturbation
            center = int(rng.integers(0, n))
            width = max(2, int(n // 20))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
            f_new += rng.normal(0, 0.15) * kernel
        elif r < 0.7:
            # Single element
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] + rng.normal(0, 0.05))
        elif r < 0.85:
            # Two-element exchange
            i, j = rng.integers(0, n, 2)
            if i != j:
                diff = f_new[i] - f_new[j]
                f_new[i] = f_new[j]
                f_new[j] = f_new[i] + diff  # Actually just swap
        else:
            # Global perturbation
            f_new *= np.exp(rng.normal(0, 0.2, n))

        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score(f_new)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f.copy()

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-strategy LP-based optimization.
    Combines LP refinement, spectral initialization, pressure descent,
    coordinate descent, and simulated annealing across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0

    best_overall = None
    best_score_overall = np.inf

    # Load parent
    parent = None
    try:
        parent_arr = np.array(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim == 1 and len(parent_arr) > 0 and np.isfinite(parent_arr).all():
            parent_arr = np.clip(parent_arr, 0, None)
            if parent_arr.sum() > 0.01:
                parent = normalize(parent_arr)
    except Exception:
        parent = None

    if parent is not None:
        p_score = compute_score(parent)
        print(f"Parent: n={len(parent)}, normalized score={p_score:.6f}")
        best_overall = parent.copy()
        best_score_overall = p_score

    # Lengths to try (prioritize parent length)
    lengths = [512, 768, 384, 1024]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths:
            lengths.insert(0, pn)
        else:
            lengths.remove(pn)
            lengths.insert(0, pn)

    print(f"Search lengths: {lengths}, total budget: {budget_s}s")

    # Phase 1: For each length, run the optimization pipeline
    for n in lengths:
        remaining = deadline - time.monotonic()
        if remaining < 180:
            break

        print(f"\n{'='*50}")
        print(f"n={n}, remaining={remaining:.0f}s")
        print(f"{'='*50}")

        # Allocate time
        t_init = min(20, remaining * 0.05)
        t_pressure = min(30, remaining * 0.08)
        t_lp = min(remaining * 0.55, 500)
        t_cd = min(30, remaining * 0.08)
        t_sa = min(remaining * 0.2, 200)

        # Initialization
        if parent is not None and len(parent) == n:
            f0 = parent * (1 + 0.15 * rng.normal(0, 1, n))
            f0 = np.clip(f0, 0, None)
            f0 = normalize(f0)
        else:
            f0 = spectral_init(n, rng, n_iter=200)

        s0 = compute_score(f0)
        print(f"  Initial: {s0:.6f}")
        if s0 < best_score_overall:
            best_score_overall = s0
            best_overall = f0.copy()

        # Pressure descent (fast, good for quick improvement)
        rem = deadline - time.monotonic()
        if rem > t_pressure + t_lp:
            f1, s1 = pressure_descent(f0, n_iter=200, eta=0.12, rng=rng)
            print(f"  After pressure: {s1:.6f}")
            if s1 < best_score_overall:
                best_score_overall = s1
                best_overall = f1.copy()
            f_current = f1
        else:
            f_current = f0

        # LP optimization (main workhorse)
        rem = deadline - time.monotonic()
        t_lp_actual = min(t_lp, rem * 0.5)
        if t_lp_actual > 30:
            f2, s2 = lp_optimize(f_current, t_lp_actual, rng, verbose=True)
            print(f"  After LP pass 1: {s2:.6f}")
            if s2 < best_score_overall:
                best_score_overall = s2
                best_overall = f2.copy()

        # Second LP pass with perturbed start
        rem = deadline - time.monotonic()
        if rem > 120 and best_overall is not None and len(best_overall) == n:
            f_pert = best_overall * np.exp(rng.normal(0, 0.1, n))
            f_pert = np.clip(f_pert, 0, None)
            f_pert = normalize(f_pert)
            t_lp2 = min(rem * 0.4, 300)
            if t_lp2 > 30:
                f3, s3 = lp_optimize(f_pert, t_lp2, rng, verbose=False)
                if s3 < best_score_overall:
                    best_score_overall = s3
                    best_overall = f3.copy()
                    print(f"  After LP pass 2: {s3:.6f}")

        # Coordinate descent polishing
        rem = deadline - time.monotonic()
        if rem > 20 and best_overall is not None and len(best_overall) == n:
            n_sweeps = max(3, int(min(15, rem * 0.15)))
            f_cd, s_cd = coordinate_descent(best_overall, n_sweeps=n_sweeps, rng=rng)
            if s_cd < best_score_overall:
                best_score_overall = s_cd
                best_overall = f_cd.copy()
                print(f"  After CD: {s_cd:.6f}")

        # Simulated annealing
        rem = deadline - time.monotonic()
        if rem > 60 and best_overall is not None and len(best_overall) == n:
            n_iter_sa = min(3000, int(rem * 50))
            f_sa, s_sa = randomized_search(best_overall, rng, n_iter=n_iter_sa,
                                            T0=0.015, T_end=0.00005)
            if s_sa < best_score_overall:
                best_score_overall = s_sa
                best_overall = f_sa.copy()
                print(f"  After SA: {s_sa:.6f}")

    # Phase 2: Use remaining time for intensive LP on the best solution
    if best_overall is not None:
        rem = deadline - time.monotonic()
        if rem > 60:
            n_best = len(best_overall)
            print(f"\n{'='*50}")
            print(f"Deep refinement: n={n_best}, remaining={rem:.0f}s")
            print(f"{'='*50}")

            # Multiple LP passes with perturbations
            for pass_idx in range(5):
                rem = deadline - time.monotonic()
                if rem < 30:
                    break
                t_pass = min(rem * 0.3, 120)

                if pass_idx == 0:
                    f_start = best_overall.copy()
                else:
                    f_start = best_overall * np.exp(rng.normal(0, 0.05, n_best))
                    f_start = np.clip(f_start, 0, None)
                    f_start = normalize(f_start)

                f_res, s_res = lp_optimize(f_start, t_pass, rng, verbose=False)
                if s_res < best_score_overall:
                    best_score_overall = s_res
                    best_overall = f_res.copy()
                    print(f"  Deep pass {pass_idx}: {s_res:.6f}")

            # Final CD
            rem = deadline - time.monotonic()
            if rem > 15:
                f_final, s_final = coordinate_descent(best_overall, n_sweeps=5, rng=rng)
                if s_final < best_score_overall:
                    best_score_overall = s_final
                    best_overall = f_final.copy()
                    print(f"  Final CD: {s_final:.6f}")

    # Final verification with evaluate_sequence
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"Target: 1.5027")
        print(f"{'='*50}")
        return final_list

    # Fallback
    print("Fallback: using uniform sequence of length 256")
    return [1.0] * 256
