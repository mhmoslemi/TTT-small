import numpy as np
import time

def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s

def compute_score(f):
    """Compute 2*n*max(conv)/sum^2 for given f."""
    n = len(f)
    conv = np.convolve(f, f)
    max_b = conv.max()
    sum_a = f.sum()
    if sum_a < 1e-15:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_simplex(f):
    """Project f onto {x >= 0, sum(x) = 1}."""
    f = np.clip(f, 0, None)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s

def subgradient_descent(f, time_budget, rng, n_inner_iter=None):

"""
Optimized sequence search using multi-peak subgradient descent, Gerchberg-Saxton
spectral flattening, peak shaving, and coordinate descent across multiple lengths.
Targets score < 1.5027.

import numpy as np
import time


def normalize(f):
    """Normalize f to sum=1, preserving non-negativity."""
    f = np.clip(f, 0.0, None)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s


def compute_score(f):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(f)
    conv = np.convolve(f, f)
    max_b = conv.max()
    sum_a = f.sum()
    if sum_a < 1e-15:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def multi_peak_subgradient(f, time_budget, rng, top_k=5):
    """Subgradient descent targeting the top-k peaks of the convolution."""
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    eta = 5.0 / n
    start = time.monotonic()
    no_improve = 0
    iter_count = 0

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()

        # Get top-k peak positions
        top_idx = np.argsort(conv)[::-1][:top_k]
        weights = np.zeros(len(top_idx))
        for i, idx in enumerate(top_idx):
            weights[i] = conv[idx] / M  # Relative weight

        # Compute weighted subgradient
        subgrad = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            for j in range(n):
                k = idx - j
                if 0 <= k < n:
                    subgrad[j] += w * 2.0 * f[k]

        # Update with backtracking
        improved = False
        for attempt in range(8):
            f_try = f - eta * subgrad
            f_try = np.maximum(f_try, 0.0)
            s = f_try.sum()
            if s < 1e-10:
                break
            f_try = f_try / s
            score_try = compute_score(f_try)
            if score_try < best_score - 1e-12:
                best_score = score_try
                best_f = f_try.copy()
                f = f_try
                eta = min(eta * 1.08, 0.5)
                no_improve = 0
                improved = True
                break
            else:
                eta *= 0.85

        iter_count += 1
        if not improved:
            no_improve += 1
            if no_improve > 200:
                # Perturb to escape local minimum
                strength = 0.02 + 0.03 * min(1.0, no_improve / 500.0)
                f = f * (1.0 + strength * rng.normal(0, 1, n))
                f = normalize(f)
                eta = max(eta * 2, 5.0 / n)
                no_improve = 0

        # Adaptive top_k: increase over time for more global view
        if iter_count % 500 == 0 and top_k < min(20, n // 4):
            top_k += 1

    return best_f, best_score

def peak_shaving(f, n_iter, rng):
    """Reduce convolution peaks by scaling down high-contributing elements."""
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)

    for it in range(n_iter):
        conv = np.convolve(f, f)
        idx_max = np.argmax(conv)
        M = conv[idx_max]

        # Compute contribution of each element to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = idx_max - i
            if 0 <= j < n:
                contrib[i] = f[i] * f[j]

        max_contrib = contrib.max()
        if max_contrib < 1e-15:
            break

        # Exponentially reduce high-contribution elements
        rate = 0.3 * (1.0 + 0.5 * it / n_iter)
        f_try = f * np.exp(-rate * contrib / max_contrib)
        f_try = normalize(f_try)

        score = compute_score(f_try)
        if score < best_score - 1e-10:
            best_score = score
            best_f = f_try.copy()
            f = f_try
        else:
            # Try gentler
            f_try2 = f * np.exp(-0.1 * contrib / max_contrib)
            f_try2 = normalize(f_try2)
            score2 = compute_score(f_try2)
            if score2 < best_score - 1e-10:
                best_score = score2
                best_f = f_try2.copy()
                f = f_try2
            else:
                # Perturb and retry
                f = f * (1.0 + 0.05 * rng.normal(0, 1, n))
                f = normalize(f)
                if it > n_iter // 2:
                    break

    return best_f, best_score


def coordinate_descent(f, n_sweeps, rng):
    """Coordinate descent with multiplicative search."""
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        any_improved = False
        for i in order:
            orig = f[i]
            if orig < 1e-12:
                continue
            best_local = best_score
            best_val = orig

            factors = [0.5, 0.7, 0.85, 0.92, 1.0/0.92, 1.0/0.85, 1.0/0.7, 1.0/0.5]
            for fac in factors:
                f[i] = max(0.0, orig * fac)
                score = compute_score(f)
                if score < best_local:
                    best_local = score
                    best_val = f[i]

            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
                any_improved = True

        f = normalize(f)
        if not any_improved:
            break

    return f.copy(), best_score

    elif method == 4:

def upsample(f_small, n_target):
    """Linearly upsample a sequence."""
    x_small = np.linspace(0, 1, len(f_small))

    f_large = np.interp(x_large, x_small, f_small)
    f_large = np.clip(f_large, 0.005, None)
    return normalize(f_large)

        remaining = deadline - time.monotonic()
        t4 = min(t_alloc * 0.15, remaining * 0.25)
        if t4 > 5:
            f4, s4 = multi_peak_subgradient(cur_f, t4, rng, top_k=10)
            print(f"  Subgrad(k=10): {s4:.6f}")
            if s4 < cur_score:
                cur_f, cur_score = f4, s4

        # Phase 5: Smoothing
        remaining = deadline - time.monotonic()
        t5 = min(t_alloc * 0.08, remaining * 0.15)
        if t5 > 2:
            f5, s5 = smooth_optimize(cur_f, max(20, int(t5 * 50)), rng)
            print(f"  Smooth: {s5:.6f}")
            if s5 < cur_score:
                cur_f, cur_score = f5, s5

        # Phase 6: Coordinate descent polish
        remaining = deadline - time.monotonic()
        t6 = min(t_alloc * 0.12, remaining * 0.2)
        if t6 > 3:
            n_sweeps = max(3, int(min(15, t6)))
            f6, s6 = coordinate_descent(cur_f, n_sweeps, rng)
            print(f"  CD: {s6:.6f}")
            if s6 < cur_score:
                cur_f, cur_score = f6, s6

        # Update global
        if cur_score < best_score_overall:
            best_score_overall = cur_score
            best_overall = cur_f.copy()
            print(f"  *** Global best: {best_score_overall:.6f} ***")

    # Final polishing phase
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 15:
        print(f"\n=== Final polishing ({remaining:.0f}s) ===")

        # One more subgradient pass
        if remaining > 20:
            f_pol, s_pol = multi_peak_subgradient(best_overall, min(30, remaining * 0.3), rng, top_k=7)
            if s_pol < best_score_overall:
                best_score_overall = s_pol
                best_overall = f_pol
                print(f"  Polish subgrad: {best_score_overall:.6f}")

        # Final CD
        remaining = deadline - time.monotonic()
        if remaining > 8:
            f_cd, s_cd = coordinate_descent(best_overall, min(10, max(3, int(remaining // 3))), rng)
            if s_cd < best_score_overall:
                best_score_overall = s_cd
                best_overall = f_cd
                print(f"  Polish CD: {best_score_overall:.6f}")

    # Verify and return
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0.0, 1000.0)
        # Scale so sum is in a good range (not too small)
        s = best_overall.sum()
        if s < 1.0:
            best_overall = best_overall * (1.0 / s)
        elif s > 100.0:
            best_overall = best_overall * (100.0 / s)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal: score={final_score:.6f}, n={len(final_list)}")
        return final_list

    # Fallback
    print("Using fallback")
    fallback = list(np.ones(512))
    return fallback
