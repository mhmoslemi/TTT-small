            mask1 = (idx1 >= 0) & (idx1 < n)
            sg[mask1] += f[idx1[mask1]]
            # f[i - k_star] if 0 <= i - k_star < n
            idx2 = np.arange(n) - k_star
            mask2 = (idx2 >= 0) & (idx2 < n)
            sg[mask2] += f[idx2[mask2]]

        sg /= len(peak_positions)

        if sg.max() > 0:
            sg = sg / sg.max()

        # Adaptive learning rate
        eta_eff = eta * (1.0 / (1.0 + 0.005 * it))
        f_new = f * np.exp(-eta_eff * sg)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 80:
                # Perturb to escape
                f_pert = f * (1.0 + 0.08 * rng.standard_normal(n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    stall = 0
                elif stall > 150:
                    break

    return best_f, best_score


def lp_refine(f, time_budget, rng=None, lp_factors=None):
    """LP refinement with multiple M factors and piecewise line search."""
    if rng is None:
        rng = np.random.default_rng(0)
    if lp_factors is None:
        lp_factors = [1.0, 0.98, 0.95, 0.92, 0.90]

    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)

    best_f = f.copy()
    best_score = compute_score(f)
    start = time.monotonic()
    no_improve_count = 0

    while time.monotonic() - start < time_budget:
        conv = fast_conv(f)
        M = conv.max()
        A = build_conv_matrix(f)

        g = None
        # Try different M factors (tighter first for more aggressive direction)
        for factor in lp_factors:
            M_bound = M * factor
            if M_bound < 1e-15:
                continue

            c_obj = -np.ones(n)
            b_ub = np.full(2 * n - 1, M_bound)
            bounds = [(0, None)] * n

            try:
                res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                              method='highs', options={'presolve': True,
                                                       'time_limit': max(1, int(time_budget * 0.05))})
            except Exception:
                continue

            if res.success and res.x is not None:
                candidate = np.clip(res.x, 0, None)
                if candidate.sum() > 1e-10:
                    # Check it's meaningfully different from f
                    cos_sim = np.dot(candidate, f) / (np.linalg.norm(candidate) * np.linalg.norm(f) + 1e-15)
                    if cos_sim < 0.999 or factor < 1.0:
                        g = candidate
                        break

        if g is None:
            # Fallback: use the least aggressive LP
            c_obj = -np.ones(n)
            b_ub = np.full(2 * n - 1, M)
            bounds = [(0, None)] * n
            try:
                res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                              method='highs', options={'presize': True, 'time_limit': 2})
                if res.success and res.x is not None:
                    g = np.clip(res.x, 0, None)
            except Exception:
                pass

        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()

        # Piecewise line search: try decreasing step sizes
        improved = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
            f_new = (1.0 - t) * f + t * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_score = compute_score(f_new)
            if new_score < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                improved = True
                no_improve_count = 0
                break

        if not improved:
            no_improve_count += 1
            if no_improve_count > 10:
                # Try perturbation
                f_pert = f + rng.standard_normal(n) * 0.01
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    no_improve_count = 0
                elif no_improve_count > 20:
                    break

    return best_f, best_score


def coordinate_descent(f, n_sweeps=10, rng=None, deadline=None):
    """Coordinate descent with geometric multipliers."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)
    factors = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.6]

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if f[i] < 1e-12:
                continue
            orig = f[i]
            best_val = orig
            best_local = best_score

            for fac in factors:
                f[i] = orig * fac
                s = f.sum()
                if s < 1e-10:
                    f[i] = orig
                    continue
                conv = fast_conv(f)
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]

            f[i] = best_val
            if best_local < best_score:
                best_score = best_local

    f = np.clip(f, 0, None)
    f = normalize(f)
    return f, best_score


def simulated_annealing(f, rng, n_iter=5000, T0=0.04, T_end=0.00005, deadline=None):
    """Simulated annealing with diverse move types."""
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    cur_score = best_score
    no_improve = 0

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        T = T0 * (T_end / T0) ** (it / max(1, n_iter))
        f_new = f.copy()
        r = rng.random()

        if r < 0.25:
            # Block multiplicative scaling
            width = max(1, int(rng.integers(1, max(2, n // 12))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.standard_normal() * 0.3)
        elif r < 0.45:
            # Single element
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] * np.exp(rng.standard_normal() * 0.4))
        elif r < 0.65:
            # Smooth kernel perturbation
            center = int(rng.integers(0, n))
            width = max(3, int(n / 20))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (width ** 2))
            f_new += rng.standard_normal() * 0.08 * kernel
        elif r < 0.8:
            # Two-element exchange
            i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
            if i != j:
                f_new[i], f_new[j] = f_new[j], f_new[i]
        else:
            # Global reweighting
            f_new *= np.exp(rng.standard_normal(n) * 0.2)

        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score(f_new)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > min(3000, n_iter // 2) and it > n_iter // 3:
            break

    return best_f, best_score


def lse_pressure_update(f, n_iter=300, eta=0.08, rng=None, deadline=None):
    """Log-sum-exp based pressure: uses all peaks weighted by softness."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0
    softness = 0.1

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        conv = fast_conv(f)
        M = conv.max()
        # Soft weights: exp((conv - M) / (softness * M))
        weights = np.exp((conv - M) / (softness * max(M, 1e-15)))
        weights /= weights.sum()

        # Weighted subgradient
        sg = np.zeros(n)
        for k_star in np.where(weights > weights.max() * 0.01)[0]:
            w_k = weights[k_star]
            idx1 = k_star - np.arange(n)
            mask1 = (idx1 >= 0) & (idx1 < n)
            sg[mask1] += w_k * f[idx1[mask1]]
            idx2 = np.arange(n) - k_star
            mask2 = (idx2 >= 0) & (idx2 < n)
            sg[mask2] += w_k * f[idx2[mask2]]

        if sg.max() > 0:
            sg = sg / sg.max()

        eta_eff = eta * (1.0 / (1.0 + 0.003 * it))
        f_new = f * np.exp(-eta_eff * sg)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                f_pert = f * (1.0 + 0.06 * rng.standard_normal(n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    stall = 0
                elif stall > 200:
                    break

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500):
    """Main search: multi-strategy optimization across diverse inits and lengths."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 25.0

    best_f = None
    best_score = np.inf

    # Try parent sequence
    parent_n = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and len(parent) > 0 and np.isfinite(parent).all():
            parent = np.clip(parent, 0, None)
            if parent.sum() > 0.01:
                p = normalize(parent)
                ps = compute_score(p)
                print(f"Parent: n={len(p)}, score={ps:.6f}")
                best_f = p.copy()
                best_score = ps
                parent_n = len(p)
    except (NameError, Exception):
        pass

    # Determine which lengths to try
    if parent_n is not None:
        lengths = [parent_n]
    else:
        lengths = [512]
    for nl in [256, 1024]:
        if nl not in lengths:
            lengths.append(nl)

    print(f"Target: 1.5027 | Current best: {best_score:.6f} | Lengths: {lengths}")

    for n in lengths:
        remaining = deadline - time.monotonic()
        if remaining < 120:
            break
        print(f"\n{'='*50}\n  n={n}, remaining={remaining:.0f}s\n{'='*50}")

        # Build diverse initializations
        inits = []
        if best_f is not None and n == len(best_f):
            inits.append(('parent', best_f.copy()))
        inits.append(('cos', normalize(cosine_window(n))))
        inits.append(('gauss', normalize(gaussian_window(n, 0.25))))
        inits.append(('gauss2', normalize(gaussian_window(n, 0.4))))
        inits.append(('smooth_rand', smooth_random_init(n, rng)))

        # Upsampled init
        if n >= 64:
            n_small = max(16, n // 16)
            f_s = np.clip(np.ones(n_small) * (1 + 0.3 * rng.standard_normal(n_small)), 0.1, None)
            f_s = normalize(f_s)
            f_s, _ = lse_pressure_update(f_s, n_iter=150, eta=0.15, rng=rng)
            x_s = np.linspace(0, 1, n_small)
            x_l = np.linspace(0, 1, n)
            f_up = np.interp(x_l, x_s, f_s)
            inits.append(('upsampled', normalize(np.clip(f_up, 0.01, None))))

        best_n_score = best_score if (parent_n is not None and n == parent_n) else np.inf
        best_n_f = best_f.copy() if (parent_n is not None and n == parent_n) else None

        for init_name, f0 in inits:
            remaining = deadline - time.monotonic()
            if remaining < 60:
                break

            print(f"  [{init_name}] ", end="", flush=True)

            # Phase 1: Quick LSE pressure (fast, uses all peaks)
            t1 = min(15, remaining * 0.12)
            f1, s1 = lse_pressure_update(f0, n_iter=int(t1 * 40), eta=0.1, rng=rng,
                                         deadline=deadline)
            if s1 < best_n_score:
                best_n_score = s1
                best_n_f = f1.copy()

            # Phase 2: Standard pressure (focused on main peak)
            remaining = deadline - time.monotonic()
            t2 = min(15, remaining * 0.12)
            f2, s2 = pressure_update(f1, n_iter=int(t2 * 30), eta=0.1, rng=rng,
                                     deadline=deadline)
            if s2 < best_n_score:
                best_n_score = s2
                best_n_f = f2.copy()

            # Phase 3: LP refinement (main workhorse)
            remaining = deadline - time.monotonic()
            t_lp = min(remaining * 0.45, 150)
            if t_lp > 15:
                f3, s3 = lp_refine(f2, t_lp, rng=rng)
                if s3 < best_n_score:
                    best_n_score = s3
                    best_n_f = f3.copy()
                print(f"LP: {s3:.6f}", end="  ")
            else:
                f3, s3 = f2, s2
                print(f"(skip LP)", end="  ")

            # Phase 4: Coordinate descent
            remaining = deadline - time.monotonic()
            t_cd = min(remaining * 0.1, 20)
            if t_cd > 5:
                n_sweeps = max(3, int(t_cd * 0.4))
                f4, s4 = coordinate_descent(f3, n_sweeps=n_sweeps, rng=rng,
                                            deadline=deadline)
                if s4 < best_n_score:
                    best_n_score = s4
                    best_n_f = f4.copy()
                print(f"CD: {s4:.6f}")
            else:
                print(f"best: {best_n_score:.6f}")

        # Update global best
        if best_n_f is not None and best_n_score < best_score:
            best_score = best_n_score
            best_f = best_n_f.copy()
            print(f"  >>> New global best: {best_score:.6f} (n={n})")

    # Final polishing phase on the best sequence found
    if best_f is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n{'='*50}\n  Final polishing ({remaining:.0f}s)\n{'='*50}")

            # Extended LP
            t_lp_final = min(remaining * 0.4, 120)
            if t_lp_final > 10:
                f, s = lp_refine(best_f, t_lp_final, rng=rng)
                if s < best_score:
                    best_score = s
                    best_f = f.copy()
                print(f"  Final LP: {best_score:.6f}")

            # Coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 10:
                n_sweeps = max(5, int(remaining * 0.25))
                f, s = coordinate_descent(best_f, n_sweeps=n_sweeps, rng=rng,
                                          deadline=deadline)
                if s < best_score:
                    best_score = s
                    best_f = f.copy()
                print(f"  Final CD: {best_score:.6f}")

            # Simulated annealing
            remaining = deadline - time.monotonic()
            if remaining > 10:
                n_iter = min(15000, int(remaining * 250))
                f, s = simulated_annealing(best_f, rng, n_iter=n_iter,
                                           T0=0.02, T_end=0.00001, deadline=deadline)
                if s < best_score:
                    best_score = s
                    best_f = f.copy()
                print(f"  Final SA: {best_score:.6f}")

            # One more short LP burst
            remaining = deadline - time.monotonic()
            if remaining > 10:
                f, s = lp_refine(best_f, min(remaining - 5, 30), rng=rng,
                                 lp_factors=[1.0, 0.97, 0.94, 0.91, 0.88])
                if s < best_score:
                    best_score = s
                    best_f = f.copy()
                print(f"  Final LP2: {best_score:.6f}")

    # Verify with evaluate_sequence
    if best_f is not None:
        best_f = np.clip(best_f, 0, 1000.0)
        final_list = [float(x) for x in best_f]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}\n  FINAL: score={final_score:.6f}, n={len(final_list)}\n{'='*50}")
        return final_list

    print("Fallback: uniform 512")
    return [1.0] * 512
