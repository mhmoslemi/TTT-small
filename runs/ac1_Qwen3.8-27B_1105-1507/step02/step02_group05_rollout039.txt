The convolution (a*a)_k can be interpreted as: if we think of a as a probability distribution on {0,...,n-1}, then (a*a)_k is the probability that the sum of two independent draws from this distribution equals k. We want to minimize the maximum of this "sum distribution."

This is related to the problem of finding a distribution on a finite group whose sum distribution is as flat as possible. This is related to "difference sets" and combinatorial design theory!

A difference set in a group G is a subset D such that every non-identity element is represented the same number of times as a difference of two elements of D. If we could find a "difference set" in {0,...,2n-2} (or a cyclic group), we'd get a perfectly flat convolution.

For cyclic groups, difference sets exist for certain parameters (e.g., (v,k,lambda) difference sets). This could give us a good starting point!

For example, a (7,3,1) difference set in Z_7: {0,1,3}. The autocorrelation is perfectly flat (all non-zero lags have the same value).

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using:
1. Softmax parameterization (ensures a >= 0, sum(a) = 1)
2. Log-sum-exp smoothing of the max function
3. L-BFGS-B with beta-annealing
4. Multi-length search with diverse initializations
5. Parent-seeding and cross-length transfer

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = rfft(a, s)
    return irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    b = fast_conv(a)
    return float(2 * len(a) * b.max())

def softmax_obj_and_grad(x, beta):
    """
    Objective: (1/beta)*log(sum_k exp(beta*(a*a)_k)) where a = softmax(x)
    Returns (value, gradient).
    """
    n = len(x)
    # Softmax
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a_sum = a.sum()
    a = a / a_sum
    
    # Convolution
    b = fast_conv(a)
    
    # Log-sum-exp
    b_max = b.max()
    log_sum = b_max + np.log(np.sum(np.exp(beta * (b - b_max)))) / beta
    w = np.exp(beta * (b - b_max))
    w /= w.sum()
    
    # Gradient of smooth max w.r.t. a
    W_flip = w[::-1]
    conv_res = np.convolve(W_flip, a, mode='full')
    grad_a = 2.0 * conv_res[n-1:2*n-1][::-1]
    
    # Gradient w.r.t. x via softmax
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    
    return log_sum, grad_x

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    src = np.maximum(src, 0.0)
    if src.sum() < 1e-10:
        return np.ones(dest_n)
    src = src / src.sum()
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    result = np.interp(t_new, t_old, src)
    result = np.maximum(result, 1e-10)
    return result / result.sum()

def to_logspace(a):
    """Convert a (on simplex) to log-space x where a = softmax(x)."""
    a = np.maximum(a, 1e-30)
    return np.log(a)

def run_lbb(x0, n, betas, maxiter_per_beta=200, deadline=None):
    """Run L-BFGS-B with beta annealing. Returns best (x, a, J)."""
    best_J = float('inf')
    best_a = None
    x = x0.copy()
    
    for beta in betas:
        if deadline is not None and time.monotonic() > deadline:
            break
        res = minimize(
            fun=lambda xx: softmax_obj_and_grad(xx, beta)[0],
            x0=x,
            jac=lambda xx: softmax_obj_and_grad(xx, beta)[1],
            method='L-BFGS-B',
            options={'maxiter': maxiter_per_beta, 'max_fun': maxiter_per_beta * 10}
        )
        x = res.x
        a = np.exp(x - x.max())
        a /= a.sum()
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
    
    if best_a is None:
        a = np.exp(x - x.max())
        a /= a.sum()
        best_J = compute_J(a)
        best_a = a.copy()
    
    return best_a, best_J

        a = self._get_a(x)
        return self._grad_smooth_max(x, a)

        a = np.exp(x_shifted)
        return a / a.sum()
    
    def _smooth_max(self, a):
        b = fast_conv(a)
        b_max = b.max()

        b_max = b.max()

        w /= w.sum()

        return a * (grad_a - dot)
