"""
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))^2 using:
- Min-peak LP best-response with backtracking line search
- Smoothed log-sum-exp L-BFGS-B with multi-beta annealing
- Diverse multi-start from parent + structured/random inits
- Perturbation restarts with LP recovery
- Adaptive time allocation across 4 phases
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_conv_matrix(a):
    """Matrix A where (A @ x)[k] = (a * x)[k] for linear convolution."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def compute_score(a):
    """J(a) = 2n * max(conv(a,a)) / (sum(a))^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def lp_min_peak(a):
    """Solve min t s.t. (a*g)_k <= t for all k, sum(g)=1, g>=0. Returns g."""
    n = len(a)
    A = build_conv_matrix(a)
    m = 2 * n - 1
    c = np.zeros(n + 1)
    c[0] = 1.0
    A_ub = np.zeros((m, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = A
    b_ub = np.zeros(m)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])
    bounds = [(None, None)] + [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        g = np.clip(res.x[1:], 0, None)
        gs = g.sum()
        if gs > 1e-10:
            return g / gs
    return None


def smooth_obj(x, beta, n):
    """Log-sum-exp smoothed objective (scale-invariant, internally normalizes)."""
    x_pos = np.maximum(x, 1e-15)
    s = x_pos.sum()
    if s < 1e-10:
        return 1e10
    a = x_pos / s
    b = fast_conv(a)
    b_max = float(b.max())
    diff = beta * (b - b_max)
    diff = np.clip(diff, -50.0, 50.0)
    logsumexp_val = beta * b_max + np.log(np.exp(diff).sum())
    return float(2.0 * n * logsumexp_val / beta)


def lp_refine(a, deadline, max_iters=5000):
    """LP best-response iteration with backtracking. Returns (best_a, best_score)."""
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    a = np.clip(a, 1e-8, None)
    a = a / a.sum()

    stagnant = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        g = lp_min_peak(a)
        if g is None:
            break

        current_score = compute_score(a)
        found_improve = False
        accepted = False

        for t in [1.0, 0.8, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0, None)
            if a_new.sum() < 1e-10:
                continue
            a_new = a_new / a_new.sum()
            score = compute_score(a_new)
            if score < best_score - 1e-14:
                best = a_new.copy()
                best_score = score
                a = a_new
                found_improve = True
                accepted = True
                break
            elif score < current_score + 1e-12:
                a = a_new
                accepted = True
                break

        if found_improve:
            stagnant = 0
        elif accepted:
            stagnant += 1
        else:
            stagnant += 3

        if stagnant > 80:
            break

    return best, best_score


def smooth_refine(a, deadline):
    """L-BFGS-B refinement with multi-beta annealing schedule."""
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)

    betas = [10, 30, 60, 120, 250, 500, 1000, 2000, 4000, 8000]
    for beta in betas:
        if time.monotonic() > deadline:
            break
        x0 = np.clip(best, 1e-8, None)
        x0 = x0 / x0.sum()
        try:
            res = minimize(smooth_obj, x0, args=(beta, n),
                           method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': 400, 'maxfun': 8000,
                                    'ftol': 1e-20, 'gtol': 1e-14,
                                    'iprint': -1})
            if res.fun < 1e10:
                x = np.clip(res.x, 0, None)
                if x.sum() > 0.01:
                    sc = compute_score(x)
                    if sc < best_score:
                        best = x.copy()
                        best_score = sc
                        print(f"    smooth beta={beta}: J={sc:.6f}", flush=True)
        except Exception:
            pass

    return best, best_score
