"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating, Nesterov-accelerated mirror descent, and coordinate refinement.
Explores multiple sequence lengths and initialization strategies.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix A where (A*g)_k = conv(a,g)_k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # A[k, j] = a[k-j] for j in [j_start, j_end]
        # k-j ranges from k-j_end to k-j_start
        A[k, j_start:j_end+1] = a[k-j_end:k-j_start+1][::-1]
    return A

def solve_lp_step(a, M=None, active_only=True, active_frac=0.3):
    """
    Solve LP: max sum(g) s.t. conv(a,g)_k <= M for active k, g >= 0.
    Returns g or None.
    """
    n = len(a)
    if M is None:
        conv_aa = np.convolve(a, a)
        M = float(np.max(conv_aa))
    
    if M <= 0:
        return None
    
    if active_only:
        conv_aa = np.convolve(a, a)
        threshold = M - active_frac * (M - np.min(conv_aa))
        active = conv_aa >= threshold
        # Always include the peak
        active[int(np.argmax(conv_aa))] = True
    else:
        active = np.ones(2*n-1, dtype=bool)
    
    A_full = build_conv_matrix(a)
    A = A_full[active, :]
    
    c = -np.ones(n)
    b_ub = np.full(int(np.sum(active)), M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_alternate(a_init, max_iter=200, active_frac=0.3, blend=0.5, verbose=False):
    """LP alternating optimization."""
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    prev_obj = best_obj
    
    for it in range(max_iter):
        g = solve_lp_step(a, active_frac=active_frac)
        if g is None:
            break
        
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Blend
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            a = g
        else:
            a = a_new / s
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if verbose and it % 20 == 0:
            print(f"    LP iter {it}: obj={obj:.6f}, best={best_obj:.6f}")
        
        # Convergence check
        if abs(obj - prev_obj) < 1e-9:
            if it > 10:
                break
        prev_obj = obj
    
    return best_a, best_obj

def mirror_descent(a_init, n_steps=5000, eta0=1.0, decay=0.999, verbose=False):
    """Mirror descent (multiplicative weights) with Nesterov acceleration."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s
    
    # Nesterov momentum
    y = a.copy()  # ghost point
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for step in range(n_steps):
        # Compute gradient at y
        conv_y = np.convolve(y, y)
        max_conv = np.max(conv_y)
        
        if max_conv < 1e-15:
            break
        
        # Active set
        tol = 0.01 * max_conv
        active = conv_y >= max_conv - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            active = np.zeros(2*n-1, dtype=bool)
            active[int(np.argmax(conv_y))] = True
            n_active = 1
        
        # Compute subgradient (average over active set)
        grad = np.zeros(n)
        for k in np.where(active)[0]:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            grad[j_start:j_end+1] += 2.0 * y[k-j_end:k-j_start+1][::-1]
        grad /= n_active
        
        # Step size with decay
        eta = eta0 / (1.0 + 0.01 * step)
        
        # Mirror descent step
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)
        
        # Nesterov update
        t = 2.0 / (step + 2.0)
        y = (1.0 - t) * y + t * a_new
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and step % 500 == 0:
            print(f"    MD step {step}: obj={obj:.6f}, best={best_obj:.6f}")
    
    return best_a, best_obj

def coord_refine(a_init, n_rounds=5, n_evals_per_coord=7, verbose=False):

    n = len(a_init)
    a = a_init.copy().astype(np.float64)

    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)

            if old_val < 1e-10:

                candidates = [0.0, 0.01, 0.05, 0.1, 0.2]
                candidates = [c * s / n for c in candidates]  # Scale relative to uniform
            else:
                # Try multipliers of current value
                multipliers = [0.0, 0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]
                candidates = [old_val * m for m in multipliers]
            
            best_x = old_val
            best_local = best_obj
            
            for x in candidates:
                if x < 0:
                    x = 0.0

                    best_x = x

                a[j] = best_x
                improved = True
            
            a[j] = best_x
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if not improved:

def structured_init(n, rng):

Optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2 using LP alternating
with line search, symmetrization, mirror descent polishing, and coordinate refinement.

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end+1] = a[k-j_end:k-j_start+1][::-1]
    return A

def solve_lp(a, M=None, active_frac=0.3):
    n = len(a)
    conv_aa = np.convolve(a, a)
    if M is None:
        M = float(np.max(conv_aa))
    if M <= 0:
        return None
    
    # Active set
    if active_frac <= 1.0:
        threshold = M * (1.0 - active_frac)
        active = conv_aa >= threshold
        active[int(np.argmax(conv_aa))] = True
    else:
        active = np.ones(2*n-1, dtype=bool)
    
    A_full = build_conv_matrix(a)
    A = A_full[active, :]
    n_active = int(np.sum(active))
    
    c = -np.ones(n)
    b_ub = np.full(n_active, M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def line_search_blend(a, g, n_trials=5):

    best_t = 0.5
