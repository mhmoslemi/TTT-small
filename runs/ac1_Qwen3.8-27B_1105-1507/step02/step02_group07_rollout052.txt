"""
Multi-strategy optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses mirror descent (multiplicative weights) as the primary optimizer,
with adaptive soft-max beta, LP refinement, and multi-start diverse
initializations. Key innovation over previous approach: mirror descent
handles the simplex constraint more naturally than projected gradient.
"""
import numpy as np
import time


def compute_soft_max_grad(f, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*C[k])) where C=conv(f,f)."""
    n = len(f)
    C = np.convolve(f, f)
    max_C = C.max()
    shifted = beta * (C - max_C)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-300:
        return np.zeros(n), max_C
    p = exp_shifted / total
    grad = 2.0 * np.correlate(p, f, mode='valid')
    soft_val = max_C + np.log(total) / beta
    return grad, soft_val


def mirror_descent(f_init, deadline, beta_start=5.0, beta_end=200.0,
                   lr_start=0.08, lr_min=0.001, momentum=0.3,
                   max_iters=10000, verbose_prefix=""):
    """Mirror descent with multiplicative weight updates."""
    f = np.array(f_init, dtype=np.float64).copy()
    S = float(np.sum(f))
    if S < 1e-10:
        return f, np.inf
    f = f / S
    n = len(f)
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    velocity = np.zeros(n)
    it = 0
    no_improve_count = 0
    patience = 300
    
    while time.monotonic() < deadline and it < max_iters:
        frac = min(1.0, float(it) / (max_iters * 0.7))
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = max(lr_min, lr_start * (0.5 ** (float(it) / 2000.0)))
        
        grad, _ = compute_soft_max_grad(f, beta)
        velocity = momentum * velocity + (1.0 - momentum) * grad
        
        # Multiplicative update (mirror descent with negative entropy)
        update = np.exp(-lr * velocity)
        f = f * update
        f = np.maximum(f, 0.0)
        S = float(np.sum(f))

            # Reset to uniform
            f = np.ones(n) / n
            velocity = np.zeros(n)
            continue
        f = f / S

        it += 1

            best_f = f.copy()

        else:

            if no_improve_count > patience:
                break
    
    return best_f, best_score


def build_conv_matrix(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    from scipy.sparse import csr_matrix
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_refine(f_init, deadline, step_size_init=0.1, max_iters=50, verbose_prefix=""):
    """LP-based direction finding refinement."""
    from scipy.optimize import linprog
    f = np.array(f_init, dtype=np.float64).copy()
    S = float(np.sum(f))
    if S < 1e-10:
        return f, np.inf
    f = f / S
    n = len(f)
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    it = 0
    no_improve = 0
    step_size = step_size_init
    A_cache = None
    
    while time.monotonic() < deadline and it < max_iters:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break
        
        if A_cache is None:
            A_cache = build_conv_matrix(f)
        else:
            A_cache = build_conv_matrix(f)  # Rebuild since f changed
        
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        try:
            result = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds, method='highs')
            if not result.success:
                no_improve += 1
                if no_improve > 10:
                    break
                it += 1
                continue
            g = result.x
        except Exception:
            no_improve += 1
            if no_improve > 10:
                break
            it += 1
            continue
        
        it += 1
        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue
        g = g / S  # Normalize g to same sum as f
        
        # Line search
        best_t = step_size
        best_s = best_score
        for trial_t in [step_size, step_size * 1.5, step_size * 0.5, step_size * 2.0]:
            new_f = (1.0 - trial_t) * f + trial_t * g
            new_f = np.clip(new_f, 0.0, 1000.0)
            ns = float(np.sum(new_f))
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = trial_t
        
        if best_s < best_score:
            f = (1.0 - best_t) * f + best_t * g
            f = np.clip(f, 0.0, 1000.0)
            S = float(np.sum(f))
            f = f / S
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            A_cache = None  # Invalidate cache
            if it % 10 == 0:
                print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f}")
        else:
            no_improve += 1
            step_size *= 0.6
            if step_size < 0.005 or no_improve > 20:
                break
    
    return best_f, best_score
