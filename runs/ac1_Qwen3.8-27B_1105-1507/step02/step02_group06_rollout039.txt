```python
def solve_lp2(f, S):
    """Solve: min M' s.t. sum(g) = S, conv(f,g)_k ≤ M', g ≥ 0"""
    n = len(f)
    # Variables: g_0, ..., g_{n-1}, M'  (n+1 variables)
    # Objective: minimize M' → c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0  # M' is the last variable
    
    # Constraints:
    # 1. sum(g) = S → equality
    # 2. conv(f,g)_k ≤ M' for k = 0, ..., 2n-2 → inequality
    #    i.e., sum_j f_{k-j} * g_j - M' ≤ 0
    
    # Build A_ub for inequality constraints (2n-1 rows)
    # And A_eq for equality constraints (1 row)
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A_ub[k, j] = f[k - j]
        A_ub[k, n] = -1.0  # coefficient of M'
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    bounds = [(0, None)] * n + [(None, None)]  # g ≥ 0, M' free
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    if result.success:
        return result.x[:n], result.x[n]
    return None, None
```

```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def build_conv_matrix_sparse(f):
    """Build sparse matrix where A @ g = convolve(f, g)"""
    n = len(f)
    # For efficiency, build the dense-to-sparse conversion
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))

def solve_lp2(f, S):
    """min M' s.t. sum(g) = S, conv(f,g)_k <= M', g >= 0"""
    n = len(f)
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Use sparse matrix for the convolution constraints
    A_conv = build_conv_matrix_sparse(f)  # shape (2n-1, n)
    # Add column for M': each row gets -1
    # A_ub = [A_conv | -1]
    
    # For efficiency with large n, let's use dense if n <= 2048
    if n <= 2048:

        A_ub = np.hstack([A_dense, -np.ones((2 * n - 1, 1))])
        b_ub = np.zeros(2 * n - 1)
    else:

        M_col = csc_matrix(-np.ones((2 * n - 1, 1)))
        A_ub = hstack([A_conv, M_col]).tocsc()
        b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

def coordinate_descent_polish(f, max_sweeps=10, time_limit=None):
    """Polish using coordinate descent."""
    f = f.copy()
    n = len(f)
    S = float(np.sum(f))
    best_score = compute_score(f)
    t0 = time.monotonic()

            # Candidate values

            candidates = [max(0, c) for c in candidates]
            
            best_val = current
            best_local_score = compute_score(f)

                if score < best_local_score:

                    best_val = cv
            f[i] = best_val

        if not improved:
```

d/dx [(a + bx)/(S'+x)²] = 0
→ (S'+x)² * b - (a+bx) * 2(S'+x) = 0
→ b(S'+x) - 2(a+bx) = 0
→ bS' + bx - 2a - 2bx = 0
→ bS' - 2a - bx = 0
→ x = (bS' - 2a) / b = S' - 2a/b

For the quadratic case (k = 2i):
score(x) ≈ 2n * (a_{2i} + x²) / (S' + x)²

d/dx [(a+x²)/(S'+x)²] = 0
→ 2x(S'+x)² - (a+x²)*2(S'+x) = 0
→ 2x(S'+x) - 2(a+x²) = 0
→ x² + xS' - a - x² = 0
→ xS' = a
→ x = a/S'

So for coordinate descent, we can compute the optimal x_i analytically! This is much better than grid search.

But we need to track which k is the bottleneck. In practice, multiple k's might be close to the max, so we should consider the top few.

```python
"""
Optimization of 2n*max(conv(f,f))/sum(f)^2 using:
1. Subgradient descent with momentum (fast)
2. Linearized LP direction finding (accurate)
3. Coordinate descent polish
Multiple lengths explored with random restarts.
"""
import numpy as np
import time
from scipy.optimize import linprog

def compute_score(f):
    """Compute 2n*max(conv(f,f))/sum(f)^2 efficiently."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))

def subgradient_step(f, top_k=3):
    """Compute subgradient direction for max convolution."""
    n = len(f)
    conv = np.convolve(f, f)
    # Get top-k indices
    top_indices = np.argsort(conv)[-top_k:]
    weights = conv[top_indices] - conv.mean()
    weights = np.maximum(weights, 0)
    if weights.sum() < 1e-15:
        weights = np.ones(top_k)
    
    grad = np.zeros(n)
    for k, w in zip(top_indices, weights):
        if w <= 0:
            continue
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += w * 2.0 * f[k - j]
    return grad

def subgradient_descent(f, deadline, lr=0.01, momentum_decay=0.9, top_k=5):
    """Subgradient descent with momentum."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = compute_score(f)
    momentum = np.zeros(n)
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        if M <= 0:
            break
        
        grad = subgradient_step(f, top_k=top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Normalized gradient descent with momentum
        momentum = momentum_decay * momentum - lr * grad / gnorm
        new_f = f + momentum
        new_f = np.clip(new_f, 0.0, None)
        new_S = float(np.sum(new_f))
        if new_S < 1e-10:
            lr *= 0.5
            no_improve += 1
            if no_improve > 100:
                break
            it += 1
            continue
        new_f *= S / new_S
        
        score = compute_score(new_f)
        it += 1
        
        if finite and score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.95
            if lr < 1e-8 or no_improve > 500:
                break
    
    return best_f, best_score

def solve_lp_linearized(f, S):
    """Solve linearized LP: min M' s.t. 2conv(f,g)_k - conv(f,f)_k <= M', sum(g)=S, g>=0"""
    n = len(f)
    conv_f = np.convolve(f, f)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Build constraint matrix
    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = conv_f.copy()
    
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A_ub[k, j] = 2.0 * f[k - j]
        A_ub[k, n] = -1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    if result.success:
        return result.x[:n], result.x[n]
    return None, None

def lp_slp(f, deadline, step_init=0.5):
    """SLP using linearized LP direction finding."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = compute_score(f)
    step = step_init
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        g, M_prime = solve_lp_linearized(f, S)
        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        # Rescale g to match sum S
        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            no_improve += 1
            continue
        g *= S / sum_g
        
        # Try several step sizes
        best_step_score = best_score
        best_step_f = None
        for alpha in [step, step * 0.5, step * 1.5, 0.1, 0.05]:
            alpha = min(alpha, 1.0)
            if alpha < 0.01:
                continue
            candidate = (1.0 - alpha) * f + alpha * g
            candidate = np.clip(candidate, 0.0, None)
            cand_S = float(np.sum(candidate))
            if cand_S < 1e-10:
                continue
            candidate *= S / cand_S
            cand_score = compute_score(candidate)
            if cand_score < best_step_score:
                best_step_score = cand_score
                best_step_f = candidate
        
        it += 1
        
        if best_step_f is not None:
            f = best_step_f
            best_f = f.copy()
            best_score = best_step_score
            no_improve = 0
            if it % 10 == 0:
                print(f"    LP-slp iter {it}: score={best_score:.6f} (step={step:.3f})")
            step = max(0.05, step * 0.98)
        else:
            no_improve += 1
            step *= 0.7
            if step < 0.01 or no_improve > 30:
                break
    
    return best_f, best_score

def coordinate_descent(f, deadline):
    """Coordinate descent polish."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    best_score = compute_score(f)
    best_f = f.copy()
    t0 = time.monotonic()
    
    for sweep in range(200):
```

```python
"""
Optimization of 2n*max(conv(f,f))/sum(f)^2 using:
1. Linearized LP (SLP) for direction finding - minimizes predicted max convolution
2. Subgradient descent with momentum for fast iterations
3. Coordinate descent polish
4. Random mutations for diversity
Explores n=1024 and n=2048 with multiple initializations.
"""
import numpy as np
import time
from scipy.optimize import linprog


def compute_score(f):
    """Compute 2n*max(conv(f,f))/sum(f)^2 efficiently."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))


def build_conv_matrix(f):
    """Build dense matrix A where (A @ g)[k] = convolve(f, g)[k]. Shape: (2n-1, n)."""
    n = len(f)
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    valid = (diff >= 0) & (diff < n)
    diff_clipped = np.clip(diff, 0, n - 1)
    A = np.where(valid, f[diff_clipped], 0.0)
    return A


def solve_lp_linearized(f, S):
    """
    Solve linearized LP:
    min M' subject to:
      2*conv(f,g)_k - (f*f)_k <= M' for all k
      sum(g) = S
      g >= 0
    Returns (g_opt, M_prime_opt) or (None, None) on failure.
    """
    n = len(f)
    conv_f = np.convolve(f, f)

    # Build constraint matrix efficiently
    A_conv = build_conv_matrix(f)  # shape (2n-1, n)
    A_ub = np.hstack([2.0 * A_conv, -np.ones((2 * n - 1, 1))])
    b_ub = conv_f.copy()

    # Equality: sum(g) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])

    # Objective: minimize M' (last variable)
    c = np.zeros(n + 1)
    c[n] = 1.0

    bounds = [(0, None)] * n + [(-np.inf, np.inf)]

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.success:
            return result.x[:n], result.x[n]
    except Exception:
        pass
    return None, None


def solve_lp_standard(f, S, M):
    """
    Solve: max sum(g) subject to conv(f,g)_k <= M, g >= 0.
    Returns g_opt or None.
    """
    n = len(f)
    A_conv = build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=A_conv, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def subgradient_descent(f, deadline, lr=0.02, momentum_decay=0.9, top_k=5):
    """Subgradient descent with momentum for max convolution minimization."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = compute_score(f)
    momentum = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        if M <= 0:
            break

        # Subgradient: weighted combination of top-k lags
        top_indices = np.argsort(conv)[-top_k:]
        threshold = conv.min()
        weights = np.maximum(conv[top_indices] - threshold, 0)
        if weights.sum() < 1e-15:
            weights = np.ones(top_k)
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for k, w in zip(top_indices, weights):
            if w <= 0:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            idx = k - np.arange(j_end, j_start - 1, -1)
            grad[j_start:j_end + 1] += w * 2.0 * f[idx]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        momentum = momentum_decay * momentum - lr * grad / gnorm
        new_f = f + momentum
        new_f = np.clip(new_f, 0.0, None)
        new_S = float(np.sum(new_f))
        if new_S < 1e-10:
            lr *= 0.7
            no_improve += 1
            if no_improve > 200:
                break
            it += 1
            continue
        new_f *= S / new_S

        score = compute_score(new_f)
        it += 1

        if np.isfinite(score) and score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = score
            no_improve = 0
            lr = min(lr * 1.01, 0.1)
        else:
            no_improve += 1
            lr *= 0.97
            momentum *= 0.8
            if lr < 1e-8 or no_improve > 1000:
                break

    return best_f, best_score


def lp_slp_optimize(f, deadline, verbose=False):
    """SLP using linearized LP direction finding with adaptive step size."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = compute_score(f)
    step = 0.6
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        t_start = time.monotonic()

        g, M_prime = solve_lp_linearized(f, S)
        if g is None:
            no_improve += 1
            if no_improve > 15:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            no_improve += 1
            continue
        g = g * (S / sum_g)

        # Check potential improvement: M_prime vs current M
        current_M = float(np.max(np.convolve(f, f)))
        potential_reduction = current_M - M_prime

        # Line search with multiple step sizes
        best_step_score = best_score
        best_step_f = None

        # Choose step sizes based on potential improvement
        if potential_reduction > 1e-6:
            alphas = [step, step * 0.5, step * 1.5, 0.8, 0.3, 0.1]
        else:
            alphas = [0.3, 0.1, 0.05]

        for alpha in alphas:
            alpha = min(max(alpha, 0.01), 1.0)
            candidate = (1.0 - alpha) * f + alpha * g
            candidate = np.clip(candidate, 0.0, None)
            cand_S = float(np.sum(candidate))
            if cand_S < 1e-10:
                continue
            candidate = candidate * (S / cand_S)
            cand_score = compute_score(candidate)
            if np.isfinite(cand_score) and cand_score < best_step_score:
                best_step_score = cand_score
                best_step_f = candidate

        it += 1
        lp_time = time.monotonic() - t_start

        if best_step_f is not None:
            f = best_step_f
            best_f = f.copy()
            best_score = best_step_score
            no_improve = 0
            if verbose and it % 5 == 0:
                print(f"    LP-slp {it}: score={best_score:.6f} step={step:.3f} "
                      f"t={lp_time:.2f}s red={potential_reduction:.2e}")
            # Adapt step size
            step = min(0.8, step * 1.01)
        else:
            no_improve += 1
            step *= 0.7
            if step < 0.02 or no_improve > 25:
                break

        # Safety: if LP is taking too long, reduce frequency
        if lp_time > 10:
            break

    return best_f, best_score


def coordinate_descent(f, deadline, max_sweeps=100):
    """Coordinate descent polish using analytical 1D optimization."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = compute_score(f)

    for sweep in range(max_sweeps):
        if time.monotonic() >= deadline:
            break
        improved = False
        conv = np.convolve(f, f)
        current_M = float(np.max(conv))

        # Process coordinates in random order for diversity
        order = np.random.permutation(n)

        for i in order:
            if time.monotonic() >= deadline:
                break
            current = f[i]
            S_rest = S - current
            if S_rest < 1e-10 and current < 1e-10:
                continue

            best_val = current
            best_local = compute_score(f)

            # Try analytical optimum based on top bottlenecks
            top_ks = np.argsort(conv)[-4:]
            for k in top_ks:
                k_minus_i = k - i
                if k == 2 * i:
                    # Quadratic case: conv_k = base + x^2
                    base = conv[k] - current * current
                    if S_rest > 1e-10:
                        x_opt = max(0.0, base / S_rest)
                    else:
                        continue
                elif 0 <= k_minus_i < n:
                    f_ki = f[k_minus_i]
                    if f_ki < 1e-12:
                        continue
                    # Linear case: conv_k = base + 2*x*f_ki
                    base = conv[k] - 2.0 * current * f_ki
                    x_opt = S_rest - 2.0 * base / f_ki
                    x_opt = max(0.0, x_opt)
                else:
                    continue

                x_opt = min(x_opt, 5.0)  # Cap
                f[i] = x_opt
                new_S = S_rest + x_opt
                if new_S < 0.01:
                    f[i] = current
                    continue
                sc = compute_score(f)
                if sc < best_local:
                    best_local = sc
                    best_val = x_opt

            # Also try a few simple candidates
            for factor in [0.3, 0.7, 1.4, 2.0]:
                candidate = current * factor
                candidate = max(0.0, min(candidate, 5.0))
                f[i] = candidate
                new_S = S_rest + candidate
                if new_S < 0.01:
                    f[i] = current
                    continue
                sc = compute_score(f)
                if sc < best_local:
                    best_local = sc
                    best_val = candidate

            f[i] = best_val
            S = S_rest + best_val
            if abs(best_val - current) > 1e-10:
                improved = True

        sc = compute_score(f)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()
        if not improved:
            break

    return best_f, best_score


def random_mutate(f, n_mut=5, strength=0.3):
    """Apply random mutation to a few coordinates."""
    f = f.copy()
    n = len(f)
    S = float(np.sum(f))
    indices = np.random.choice(n, size=min(n_mut, n), replace=False)
    for i in indices:
        f[i] = max(0.0, f[i] + np.random.normal(0, strength * max(f[i], 0.1)))
    new_S = float(np.sum(f))
    if new_S > 1e-10:
        f *= S / new_S
    return f


def make_initializations(n, rng, parent=None):
    """Create diverse initializations for length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n))

    # Cosine taper
    inits.append((1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0 + 0.01)

    # Exponential decay (both sides)
    inits.append(np.exp(-3.0 * idx / n) + np.exp(-3.0 * (n - 1 - idx) / n) + 0.05)

    # Bell
    center = (n - 1) / 2.0
    inits.append(np.exp(-0.8 * ((idx - center) / (n / 5.0)) ** 2) + 0.01)

    # Random
    inits.append(rng.uniform(0.2, 3.0, size=n))

    # Sigmoid
    inits.append(1.0 / (1.0 + np.exp(-5.0 * (idx / (n - 1) - 0.5))) + 0.01)

    # Flat with bumps
    flat = np.ones(n) * 0.5
    for _ in range(3):
        pos = int(rng.integers(0, n))
        width = max(1, n // 20)
        for d in range(-width, width + 1):
            pos_d = (pos + d) % n
            flat[pos_d] += np.exp(-0.5 * (d / width) ** 2)
    inits.append(flat)

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent))

    return inits


def propose_candidate(budget_s=1500, **kwargs):
    """Main search: find best sequence minimizing 2n*max(conv)/sum^2."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 8.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)

    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            score = evaluate_sequence(parent.tolist())
            if np.isfinite(score):
                best_score = score
                best_sequence = parent.copy()
                print(f"Parent: n={parent.size}, score={score:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # Phase 1: Quick exploration (first 100s)
    print("\n" + "=" * 60)
    print("Phase 1: Quick Exploration")
    print("=" * 60)
    phase1_deadline = t0 + 100.0

    lengths_to_try = [1024, 2048]
    if parent is not None and len(parent) not in lengths_to_try:
        lengths_to_try.insert(0, len(parent))

    candidates = []  # (score, seq, n)

    for n in lengths_to_try:
        if time.monotonic() > phase1_deadline - 15:
            break
        print(f"\n--- n={n} ---")
        inits = make_initializations(n, rng, parent)

        for idx_init, f_init in enumerate(inits):
            if time.monotonic() > phase1_deadline - 5:
                break
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            S_init = float(np.sum(f_init))
            if S_init < 0.01:
                continue
            f_init = f_init / S_init  # Normalize to sum=1

            init_score = compute_score(f_init)

            # Quick subgradient (5-8 seconds each)
            quick_dl = min(phase1_deadline, time.monotonic() + 7.0)
            f_opt, s_opt = subgradient_descent(f_init, quick_dl, lr=0.03, top_k=5)

            if s_opt < best_score:
                best_score = s_opt
                best_sequence = f_opt.copy()

            candidates.append((s_opt, f_opt.copy(), n))
            print(f"  init {idx_init}: {init_score:.6f} -> {s_opt:.6f}")

    candidates.sort(key=lambda x: x[0])
    print(f"\nBest after Phase 1: {best_score:.6f} (n={len(best_sequence) if best_sequence is not None else 0})")
    for sc, seq, n in candidates[:3]:
        print(f"  n={n}: {sc:.6f}")

    # Phase 2: LP-SLP focused optimization (next ~1150s)
    print("\n" + "=" * 60)
    print("Phase 2: LP-SLP Optimization")
    print("=" * 60)
    phase2_deadline = deadline - 200.0

    # Work on top candidates, prioritizing different lengths
    seen_lengths = set()
    top_cands = []
    for sc, seq, n in candidates:
        if n not in seen_lengths:
            top_cands.append((sc, seq, n))
            seen_lengths.add(n)
        if len(top_cands) >= 3:
            break

    for ci, (sc, f_init, n) in enumerate(top_cands):
        if time.monotonic() > phase2_deadline - 30:
            break

        time_per_cand = max(60.0, (phase2_deadline - time.monotonic()) / (len(top_cands) - ci))
        cand_deadline = time.monotonic() + time_per_cand

        print(f"\n--- LP-SLP on n={n} (start={sc:.6f}, budget={time_per_cand:.0f}s) ---")

        S = float(np.sum(f_init))
        f_work = f_init.copy()

        # Main LP-SLP loop
        f_opt, s_opt = lp_slp_optimize(f_work, cand_deadline - 20, verbose=True)

        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()
            print(f"  LP-SLP improved to: {best_score:.6f}")
        else:
            print(f"  LP-SLP: {s_opt:.6f} (no improvement over {best_score:.6f})")

        # Try with mutation if time remains
        if time.monotonic() < cand_deadline - 10:
            f_mut = random_mutate(f_opt, n_mut=10, strength=0.5)
            f_mut = np.clip(f_mut, 0, None)
            S_mut = float(np.sum(f_mut))
            if S_mut > 0.01:
                f_mut /= S_mut
                f_opt2, s_opt2 = lp_slp_optimize(f_mut, cand_deadline, verbose=False)
                if s_opt2 < best_score:
                    best_score = s_opt2
                    best_sequence = f_opt2.copy()
                    print(f"  Mutated improved to: {best_score:.6f}")

    # Also try a fresh start at n=2048 if not already done well
    if best_sequence is not None and len(best_sequence) != 2048 and time.monotonic() < phase2_deadline - 60:
        print(f"\n--- Fresh LP-SLP n=2048 (budget={phase2_deadline - time.monotonic():.0f}s) ---")
        f_fresh = np.ones(2048)
        # Quick subgradient warmup
        f_fresh, s_fresh = subgradient_descent(f_fresh, time.monotonic() + 30, lr=0.03, top_k=5)
        print(f"  After warmup: {s_fresh:.6f}")

        f_opt, s_opt = lp_slp_optimize(f_fresh, phase2_deadline, verbose=True)
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()
            print(f"  n=2048 improved to: {best_score:.6f}")

    print(f"\nBest after Phase 2: {best_score:.6f}")

    # Phase 3: Coordinate descent + final polish (last 200s)
    print("\n" + "=" * 60)
    print(f"Phase 3: Polish (best={best_score:.6f})")
    print("=" * 60)

    if best_sequence is not None and time.monotonic() < deadline - 15:
        S = float(np.sum(best_sequence))
        if S > 0.01:
            f_work = best_sequence / S

            # Coordinate descent
            cd_deadline = deadline - 80
            f_cd, s_cd = coordinate_descent(f_work, cd_deadline, max_sweeps=200)
            if s_cd < best_score:
                best_score = s_cd
                best_sequence = f_cd.copy()
                print(f"  Coord descent: {best_score:.6f}")

            # Final subgradient polish with very small lr
            f_sg, s_sg = subgradient_descent(best_sequence / float(np.sum(best_sequence)),
                                              deadline - 20, lr=0.003, top_k=7,
                                              momentum_decay=0.95)
            if s_sg < best_score:
                best_score = s_sg
                best_sequence = f_sg.copy()
                print(f"  Final subgrad: {best_score:.6f}")

            # Another coordinate descent pass
            if time.monotonic() < deadline - 15:
                f_cd2, s_cd2 = coordinate_descent(best_sequence / float(np.sum(best_sequence)),
                                                  deadline - 10, max_sweeps=50)
                if s_cd2 < best_score:
                    best_score = s_cd2
                    best_sequence = f_cd2.copy()
                    print(f"  Final coord: {best_score:.6f}")

    # Final verification and return
    if best_sequence is None:
        best_sequence = np.ones(256)

    # Ensure valid output
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S_final = float(np.sum(best_sequence))
    if S_final < 0.01:
        best_sequence = np.ones(256)

    final_score = evaluate_sequence([float(x) for x in best_sequence])
    if not np.isfinite(final_score):
        print("WARNING: Final evaluation failed, using fallback")
        best_sequence = np.ones(256)
        final_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\n{'=' * 60}")
    print(f"FINAL: n={len(best_sequence)}, score={final_score:.6f}, "
          f"internal_best={best_score:.6f}, time={elapsed:.1f}s")
    print(f"{'=' * 60}")

    return [float(x) for x in best_sequence]
```
