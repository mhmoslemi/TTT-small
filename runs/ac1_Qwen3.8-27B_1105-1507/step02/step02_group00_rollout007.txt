Optimization of step function sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) FISTA with log-sum-exp smoothing, (2) Multiplicative weight updates,

import numpy as np

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_c / (s * s)


def softmax_weights(c, tau):
    tc = tau * c
    tc = tc - np.max(tc)
    exp_tc = np.exp(tc)
    return exp_tc / np.sum(exp_tc)


def compute_smooth_grad(a, p, n):
    """grad[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m] where j+m < 2n-1."""
    grad = np.zeros(n)
    for j in range(n):
        m_max = min(n, 2*n - 1 - j)
        if m_max > 0:
            grad[j] = 2.0 * np.dot(p[j:j+m_max], a[:m_max])
    return grad

def multiplicative_updates(a_init, tau=5.0, n_steps=5000, verbose=False):

    n = len(a_init)

    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()

        p = softmax_weights(c, tau)

        obj = compute_obj(a)

            best_a = a.copy()

def lp_alternate(a_init, max_iter=200, verbose=False):
    """LP-based alternating optimization (from previous approach)."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        c = np.convolve(a, a)
        M = float(np.max(c))
        
        # Build constraint matrix
        A = np.zeros((2*n - 1, n))
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            A[k, j_start:j_end+1] = a[k - j_end:k - j_start + 1][::-1]
        
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),

        if res.status != 0:
            break
        
        g = res.x / np.sum(res.x)
        
        # Blend
        t = 0.5
        a_new = (1-t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a = g
        else:
            a = a_new / s_new
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

def coord_descent(a, n_rounds=5, verbose=False):
    """Coordinate descent with local search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for rnd in range(n_rounds):
        improved = False
        for j in range(n):
            old = a[j]
            # Try a range of values
            candidates = [0.0, old*0.1, old*0.3, old*0.5, old*0.7, old, old*1.3, old*1.5, old*2.0]
            best_x = old
            best_local = best_obj
            for x in candidates:
                a[j] = x
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_x = x
            if best_x != old:
                improved = True
            a[j] = best_x
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if not improved:
            break
    
    return best_a, best_obj


def differentiable_polish(a, n_steps=3000, tau_start=1.0, tau_end=100.0, verbose=False):
    """Gradient descent with adaptive temperature (annealing)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for k in range(n_steps):
        # Anneal temperature
        progress = k / n_steps
        tau = tau_start * (tau_end / tau_start) ** progress
        
        c = np.convolve(a, a)
        p = softmax_weights(c, tau)
        
        # Compute "force" on each element
        force = np.zeros(n)
        for j in range(n):
            m_max = min(n, 2*n - 1 - j)
            if m_max > 0:
                force[j] = np.dot(p[j:j+m_max], a[:m_max])
        
        # Update: push mass away from high-responsibility positions
        # Use a normalized step
        max_force = np.max(force)
        if max_force < 1e-10:
            break
        force_norm = force / max_force
        
        # Multiplicative update with annealed strength
        strength = 0.1 / (1.0 + 0.1 * progress)
        a_new = a * (1.0 - strength * force_norm)
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        a = a_new
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if verbose and k % 500 == 0:
            print(f"    Polish k={k}: obj={obj:.6f}, tau={tau:.1f}")
    
    return best_a, best_obj
