```python
        v = np.zeros(n)  # second moment
        t = 0
        lr = 1.0 / beta  # Learning rate scale
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        
        # Number of iterations for this beta
        max_iter = 2000
        for it in range(max_iter):
            if time.monotonic() > deadline:

            if it % 500 == 0 and it > 0:
                # Reduce learning rate
                lr *= 0.5
            
            # Compute gradient
            grad, p, f_smooth = gradient_lse_fast(a, beta)
            
            # Adam update
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Project to simplex (non-negative, sum=1)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / s
            
            a = a_new
            
            # Check true objective periodically
            if it % 100 == 0:
                score = compute_score(a)
                if score < best_score:
                    best_score = score
                    best = a.copy()
        
        # Evaluate at end of each beta
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best = a.copy()
        else:
            # Revert to best if this beta didn't help
            a = best.copy()
    
    return best, best_score
```

```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FFT-based gradient of log-sum-exp smoothed autocorrelation peak
2. Adam optimizer with adaptive learning rate
3. Multi-scale beta annealing (smooth to sharp)
4. Structured perturbation targeting autocorrelation hot spots
5. Multi-start with diverse initializations and sequence lengths
"""

import numpy as np
import time

def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))

def lse_gradient(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*conv(a,a)[k])) w.r.t. a.
    Returns (grad, smooth_value).
    """
    n = len(a)
    m = 2 * n - 1
    b = fast_conv(a)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    p = exp_shifted / Z
    
    # Smooth value
    smooth_val = b_max + np.log(Z) / beta
    
    # FFT size (need >= 3n-2 for linear conv of length-n and length-(2n-1) sequences)
    s = 1
    while s < 3 * n - 2:
        s <<= 1
    
    # Pad a and p to size s
    a_pad = np.zeros(s)
    a_pad[:n] = a
    p_pad = np.zeros(s)
    p_pad[:m] = p
    
    fft_a = np.fft.rfft(a_pad)
    fft_p = np.fft.rfft(p_pad)
    
    # First term: cross-correlation (a ⊛ p)[i] = sum_j a[j]*p[j+i]
    first = np.fft.irfft(np.conj(fft_a) * fft_p)[:n]
    
    # Second term: convolution (a * p)[i] = sum_j a[j]*p[i-j]
    second = np.fft.irfft(fft_a * fft_p)[:n]
    
    grad = (first + second) / beta
    return grad, smooth_val

def adam_optimize(a_init, deadline, betas=None, verbose=True):
    """Adam optimization on LSE-smoothed objective with beta annealing."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-8)
    a = a / a.sum()

def structured_perturbation(a, k_perts=10, seed=0):
    """
    Generate perturbations that target reducing autocorrelation hot spots.
    Returns a list of perturbed sequences.
    """
    rng = np.random.default_rng(seed)
    n = len(a)
    b = fast_conv(a)
    
    # Find top hot spots
    top_k = np.argsort(b)[-k_perts:][::-1]
    
    results = []
    for k* in top_k:
        # The autocorrelation at lag k* involves pairs (j, k*-j)
        # Find the j with the largest contribution
        j_lo = max(0, k* - n + 1)
        j_hi = min(n - 1, k*)
        
        if j_lo >= j_hi:
            continue
        
        contributions = np.array([a[j] * a[k* - j] for j in range(j_lo, j_hi + 1)])
        js = np.arange(j_lo, j_hi + 1)
        
        # Find the j with max contribution
        j_max = js[np.argmax(contributions)]
        
        # Create perturbation: move mass from j_max to a random position
        # that doesn't contribute to lag k*
        # Positions that don't contribute: j' where k*-j' is not in [0, n-1]
        # i.e., j' < k*-(n-1) or j' > k*
        # But also j' should be in [0, n-1]
        
        not_contributing = []
        for j_prime in range(n):
            if 0 <= k* - j_prime < n:
                continue
            not_contributing.append(j_prime)
        
        if len(not_contributing) == 0:
            continue
        
        j_dest = not_contributing[rng.integers(len(not_contributing))]
        
        # Move a small fraction of mass
        frac = 0.01 * rng.uniform(0.5, 2.0)
        amount = a[j_max] * frac
        if amount < 1e-12:
            continue
        
        a_new = a.copy()
        a_new[j_max] -= amount
        a_new[j_dest] += amount
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        results.append(a_new)
    
    # Also add some random perturbations
    for _ in range(3):
        a_new = a + rng.normal(0, 0.02, n)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        results.append(a_new)
    
    return results

def resample_sequence(a, new_n):
    """Resample a sequence to a different length."""
    old_n = len(a)
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    a_new = np.interp(t_new, t_old, a)
    a_new = np.maximum(a_new, 0)
    if a_new.sum() < 0.01:
        a_new = np.ones(new_n) / new_n
    else:
        a_new = a_new / a_new.sum()
    return a_new
```

The gradient of the LSE-smoothed peak autocorrelation can be computed in O(n log n) using the cross-correlation and convolution with the "pressure" distribution $p_k \propto e^{\beta \cdot \text{conv}[k]}$.

```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FFT-based gradient of log-sum-exp smoothed autocorrelation peak (O(n log n))
2. Adam optimizer with beta annealing (smooth to sharp)
3. Structured perturbation targeting autocorrelation hot spots
4. Multi-start with diverse initializations and sequence lengths
5. LP best-response as complementary perturbation mechanism
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def lse_gradient(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*conv(a,a)[k])) w.r.t. a.
    Uses FFT for O(n log n) computation.
    Returns (grad, smooth_value).
    """
    n = len(a)
    m = 2 * n - 1
    b = fast_conv(a)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    p = exp_shifted / Z

    smooth_val = b_max + np.log(Z) / beta

    # FFT size: need >= 3n-2 for linear conv of length-n and length-(2n-1)
    s = 1
    while s < 3 * n - 2:
        s <<= 1

    a_pad = np.zeros(s)
    a_pad[:n] = a
    p_pad = np.zeros(s)
    p_pad[:m] = p

    fft_a = np.fft.rfft(a_pad)
    fft_p = np.fft.rfft(p_pad)

    # First term: cross-correlation (a ⊛ p)[i] = sum_j a[j]*p[j+i]
    first = np.fft.irfft(np.conj(fft_a) * fft_p)[:n]

    # Second term: convolution (a * p)[i] = sum_j a[j]*p[i-j]
    second = np.fft.irfft(fft_a * fft_p)[:n]

    grad = (first + second) / beta
    return grad, smooth_val


def build_conv_matrix(a):
    """Build matrix A where (A @ x) = convolve(a, x) of full length 2n-1."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jlo = max(0, k - n + 1)
        jhi = min(n - 1, k)
        if jlo <= jhi:
            A[k, jlo:jhi + 1] = a[k - jhi:k - jlo + 1][::-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None


def adam_optimize(a_init, deadline, betas=None, verbose=False, max_iter_per_beta=3000):
    """Adam optimization on LSE-smoothed objective with beta annealing."""
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-8)
    a = a / a.sum()
    n = len(a)

    if betas is None:
        betas = [5, 20, 100, 500, 2000, 10000]

    best = a.copy()
    best_score = compute_score(a)

    for beta in betas:
        if time.monotonic() > deadline:
            break

        m_adam = np.zeros(n)
        v_adam = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.85, 0.999, 1e-8

        base_lr = 2.0 / n
        iters = 0

        for it in range(max_iter_per_beta):
            if time.monotonic() > deadline:
                break
            iters += 1

            # Decay learning rate
            lr = base_lr * (0.7 ** (it // 800))

            grad, smooth_val = lse_gradient(a, beta)

            # Adam update
            t_adam += 1
            m_adam = beta1 * m_adam + (1 - beta1) * grad
            v_adam = beta2 * v_adam + (1 - beta2) * grad * grad
            m_hat = m_adam / (1 - beta1 ** t_adam)
            v_hat = v_adam / (1 - beta2 ** t_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / s
            a = a_new

            # Check true objective periodically
            if it % 400 == 0:
                score = compute_score(a)
                if score < best_score:
                    best_score = score
                    best = a.copy()
                    if verbose:
                        print(f"      beta={beta}, it={it}: J={best_score:.8f}", flush=True)

        # Evaluate at end of beta phase
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best = a.copy()
            if verbose:
                print(f"      beta={beta} done ({iters} iters): J={best_score:.8f}", flush=True)
        else:
            a = best.copy()

    return best, best_score


def structured_perturbation(a, seed=0, n_perts=8):
    """
    Generate perturbations targeting autocorrelation hot spots.
    Moves mass from positions contributing to high autocorrelation peaks.
    """
    rng = np.random.default_rng(seed)
    n = len(a)
    b = fast_conv(a)
    results = []

    # Find top hot spots (exclude lag 0 and 2n-2 which are single-element squares)
    valid_lags = np.arange(1, 2 * n - 2)
    top_lags = valid_lags[np.argsort(b[valid_lags])[-min(6, len(valid_lags)):]][::-1]

    for k_star in top_lags[:4]:
        # Find the j with largest contribution to b[k_star]
        jlo = max(0, k_star - n + 1)
        jhi = min(n - 1, k_star)
        if jlo >= jhi:
            continue

        js = np.arange(jlo, jhi + 1)
        contributions = a[js] * a[k_star - js]
        j_max = js[np.argmax(contributions)]

        # Find destination: position that does NOT contribute to lag k_star
        # j' contributes to lag k_star iff 0 <= k_star - j' < n
        # j' does NOT contribute iff j' < k_star - (n-1) or j' > k_star
        candidates = []
        for j_prime in range(n):
            if 0 <= k_star - j_prime < n:
                continue
            candidates.append(j_prime)

        if len(candidates) == 0:
            continue

        # Pick a few random destinations
        for _ in range(2):
            j_dest = candidates[rng.integers(len(candidates))]
            frac = 0.005 * rng.uniform(1.0, 3.0)
            amount = a[j_max] * frac
            if amount < 1e-12:
                continue
            a_new = a.copy()
            a_new[j_max] -= amount
            a_new[j_dest] += amount
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            results.append(a_new)

    # Add some random perturbations at different scales
    for scale in [0.01, 0.03, 0.005, 0.05]:
        a_new = a + rng.normal(0, scale, n)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        results.append(a_new)

    return results


def resample_sequence(a, new_n):
    """Resample a sequence to a different length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    a_new = np.interp(t_new, t_old, a)
    a_new = np.maximum(a_new, 0)
    if a_new.sum() < 0.01:
        a_new = np.ones(new_n) / new_n
    else:
        a_new = a_new / a_new.sum()
    return a_new


def lp_perturb_and_refine(a, deadline):
    """Apply one LP best-response step then blend."""
    n = len(a)
    a_norm = a / a.sum()
    b = fast_conv(a_norm)
    M = float(b.max())
    g = lp_best_response(a_norm, M)
    if g is None:
        return a, compute_score(a)
    g = g / g.sum()

    best_local = a.copy()
    best_score = compute_score(a)

    for t in [0.3, 0.5, 0.7, 0.9]:
        if time.check_deadline(deadline):
            break
        a_new = (1 - t) * a_norm + t * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_local = a_new.copy()

    return best_local, best_score
```
