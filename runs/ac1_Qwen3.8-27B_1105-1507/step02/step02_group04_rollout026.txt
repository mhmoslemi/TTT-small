    """Smoothed objective: 2n * LSE(tau*c) / (tau * (sum a)^2)."""
    n = len(a)

    max_c = np.max(c)

"""
Multi-tau L-BFGS optimization with diverse initializations for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses FFT-based convolution, log-sum-exp
smoothing with increasing sharpness, and a coarse-to-fine tau schedule.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute exact objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = compute_conv(a)
    max_c = float(np.max(c))
    return 2.0 * n * max_c / (s * s)


def smoothed_obj_grad(a, tau):
    """
    Compute smoothed objective and its gradient w.r.t. a.
    Objective: 2n * LSE(tau*c) / (tau * (sum a)^2)
    This approximates 2n * max(c) / (sum a)^2 as tau -> inf.
    """
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf, np.zeros(n)
    
    c = compute_conv(a)
    max_c = float(np.max(c))
    # Numerically stable log-sum-exp
    tau_c = tau * (c - max_c)
    tau_c = np.clip(tau_c, -500.0, 500.0)
    exp_tc = np.exp(tau_c)
    sum_exp = float(np.sum(exp_tc))
    if sum_exp < 1e-300:
        # Fallback: just use the max
        lse_val = max_c
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        lse_val = max_c + np.log(sum_exp) / tau
        w = exp_tc / sum_exp
    
    obj = 2.0 * n * lse_val / (s * s)
    
    # Gradient: d(obj)/d(a_i) = 2n/s^2 * sum_k w_k * 2*a_{k-i} - 4n*lse/s^3 * a_i
    grad_a = 2.0 * correlate(w, a, 'valid')
    grad = (2.0 * n * grad_a) / (s * s) - (4.0 * n * lse_val * a) / (s ** 3)
    
    return obj, grad


def opt_fun(x, tau_holder):
    """Objective function for scipy minimize."""
    return smoothed_obj_grad(x, tau_holder[0])[0]


def opt_jac(x, tau_holder):
    """Gradient function for scipy minimize."""
    return smoothed_obj_grad(x, tau_holder[0])[1]


def run_lbfgs(a_init, tau_schedule, max_iter_per_tau=300):
    """
    Run L-BFGS-B optimization with a schedule of tau values.
    Returns the optimized sequence.
    """
    a = np.maximum(a_init, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.maximum(a_init, 0.0) + 0.01
        s = np.sum(a)
    a = a / s  # Normalize
    
    tau_holder = [tau_schedule[0]]
    
    for tau in tau_schedule:
        tau_holder[0] = tau
        try:
            result = minimize(
                fun=opt_fun,
                x0=a,
                jac=opt_jac,
                args=(tau_holder,),
                method='L-BFGS-B',
                bounds=[(0.0, None)] * len(a),
                options={'maxiter': max_iter_per_tau, 'gtol': 1e-7, 'iprint': 0}
            )
            if result.fun < np.inf:
                a = np.maximum(result.x, 0.0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
        except Exception:
            pass
    
    return a

"""
Multi-tau L-BFGS optimization with diverse initializations for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses FFT-based convolution, log-sum-exp
smoothing with a coarse-to-fine tau schedule, multiple sequence lengths,
and intensive final polishing with peak-focused optimization.

import numpy as np
from scipy.optimize import minimize

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute exact objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = compute_conv(a)
    max_c = float(np.max(c))
    return 2.0 * n * max_c / (s * s)


def smoothed_obj(a, tau):
    """Compute smoothed objective value."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    c = compute_conv(a)
    max_c = float(np.max(c))
    tau_c = np.clip(tau * (c - max_c), -500.0, 500.0)
    exp_tc = np.exp(tau_c)
    sum_exp = float(np.sum(exp_tc))
    if sum_exp < 1e-300:
        return 2.0 * n * max_c / (s * s)
    lse_val = max_c + np.log(sum_exp) / tau
    return 2.0 * n * lse_val / (s * s)


def smoothed_grad(a, tau):
    """Compute gradient of smoothed objective w.r.t. a."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.zeros(n)
    c = compute_conv(a)
    max_c = float(np.max(c))
    tau_c = np.clip(tau * (c - max_c), -500.0, 500.0)
    exp_tc = np.exp(tau_c)

            if time.monotonic() - start > time_budget:
                break
            a_pert = best_a * np.exp(rng.normal(0, 0.05, n))
            a_pert = np.maximum(a_pert, 0)
            s = np.sum(a_pert)
            if s > 1e-10:
                a_pert = a_pert / s
                a_opt = run_lbfgs(a_pert, [500, 2000, 5000], max_iter_per_tau=500,
                                  time_limit=10.0)
                obj_opt = compute_obj(a_opt)
                if obj_opt < best_obj:
                    best_obj = obj_opt
                    best_a = a_opt

    return best_a, best_obj


def try_parent_sequence():
    """Try to use the parent sequence as a starting point."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-tau L-BFGS optimization with diverse initializations.
    Strategy:
    1. Explore multiple sequence lengths (100-500)
    2. For each length, try many diverse initializations
    3. Optimize with coarse-to-fine tau schedule using L-BFGS-B
    4. Final intensive polishing with peak-focused methods
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence first
    parent, parent_obj = try_parent_sequence()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # Sequence lengths to explore
    n_values = [120, 150, 200, 250, 300, 350, 400, 500]
    if parent is not None:
        n_parent = len(parent)
        n_values = sorted(set(n_values + [n_parent, max(50, n_parent - 50),
                                          max(50, n_parent - 100),
                                          min(600, n_parent + 50)]))

    print(f"Exploring n values: {n_values}")

    # Tau schedules - coarse to fine
    tau_schedule_short = [5, 10, 20, 50, 100, 200]
    tau_schedule_medium = [5, 10, 20, 50, 100, 200, 500]
    tau_schedule_long = [3, 5, 10, 20, 50, 100, 200, 500]

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Determine how many initializations to try
        n_max_inits = min(20, int(remaining / 8))
        if n_max_inits < 3:
            n_max_inits = 3

        inits = generate_initializations(n, parent, seed + n * 7919)
        n_to_try = min(len(inits), n_max_inits)

        # Select a diverse subset if too many
        if n_to_try < len(inits):
            # Take evenly spaced indices
            indices = np.linspace(0, len(inits) - 1, n_to_try).astype(int)
            inits = [inits[i] for i in indices]

        for idx in range(n_to_try):
            if time.monotonic() > deadline:
                break

            a_init = inits[idx].copy()
            time_per = max(5.0, (deadline - time.monotonic()) / (n_to_try - idx))

            # Use longer schedule for more time
            if time_per > 20:
                schedule = tau_schedule_long
                max_iter = 400
            elif time_per > 10:
                schedule = tau_schedule_medium
                max_iter = 300
            else:
                schedule = tau_schedule_short
                max_iter = 200

            a_opt = run_lbfgs(a_init, schedule, max_iter_per_tau=max_iter,
                              time_limit=time_per * 0.8)
            score = compute_obj(a_opt)

            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  n={n}, init={idx}: score={best_score:.6f}")

    # Phase 2: Intensive polishing on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\nPolishing best (n={n_best}, score={best_score:.6f}), "
              f"{remaining:.0f}s remaining")

        # Multiple rounds of polishing with random restarts
        polish_time = min(remaining * 0.6, 300.0)
        a_pol, obj_pol = peak_chase_polish(best_seq, polish_time)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol
            print(f"  After peak_chase_polish: {best_score:.6f}")

        # Try perturbations of the best and re-optimize
        if time.monotonic() < deadline - 10:
            remaining = deadline - time.monotonic()
            n_attempts = min(10, int(remaining / 15))
            for attempt in range(n_attempts):
                if time.monotonic() > deadline - 5:
                    break
                # Different perturbation scales
                scales = [0.02, 0.05, 0.1, 0.15, 0.2, 0.03, 0.07, 0.12, 0.04, 0.08]
                scale = scales[attempt % len(scales)]
                a_pert = best_seq * np.exp(rng.normal(0, scale, n_best))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s > 1e-10:
                    a_pert = a_pert / s
                    a_opt = run_lbfgs(a_pert, [100, 500, 2000, 5000],
                                      max_iter_per_tau=400, time_limit=15.0)
                    score = compute_obj(a_opt)
                    if score < best_score:
                        best_score = score
                        best_seq = a_opt.copy()
                        print(f"  Perturb {attempt} (scale={scale}): {best_score:.6f}")

        # Final ultra-high-tau polish
        if time.monotonic() < deadline - 5:
            remaining = deadline - time.monotonic()
            try:
                result = minimize(
                    fun=opt_fun_unpacked,
                    x0=best_seq,
                    jac=opt_jac_unpacked,
                    args=(10000.0,),
                    method='L-BFGS-B',
                    bounds=[(0.0, None)] * len(best_seq),
                    options={'maxiter': 2000, 'gtol': 1e-12, 'iprint': 0}
                )
                if result.x is not None and np.isfinite(result.fun):
                    a_final = np.maximum(result.x, 0.0)
                    s = np.sum(a_final)
                    if s > 1e-10:
                        a_final = a_final / s
                        score = compute_obj(a_final)
                        if score < best_score:
                            best_score = score
                            best_seq = a_final
                            print(f"  Final ultra-tau: {best_score:.6f}")
            except Exception:
                pass

    # Fallback
    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    # Validate with actual evaluate_sequence
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    validation = evaluate_sequence(result)
    if np.isfinite(validation):
        if validation < best_score:
            print(f"  Validation improved: {best_score:.6f} -> {validation:.6f}")
        elif validation > best_score + 1e-6:
            print(f"  Validation discrepancy: computed={best_score:.6f}, "
                  f"validated={validation:.6f}")

    return result
