from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A

    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    return None, None

    conv = np.convolve(a, a)
    M = float(np.max(conv))

    if g_sum < 1e-10:
        return a, False
    g = g / g_sum  # normalize to simplex

    best_alpha = 0.0
    best_obj = compute_obj(a)
    
    for alpha in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]:

        a_new = np.maximum(a_new, 0)

        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha
        if obj < best_obj * 0.999:  # good enough
            break
    
    if best_alpha == 0:

    a_new = (1 - best_alpha) * a + best_alpha * g
    a_new = np.maximum(a_new, 0)
    a_new = a_new / np.sum(a_new)

def multiplicative_weight_step(a, eta=0.5):
    """Mirror descent step with KL divergence."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    
    # Subgradient of max at active positions
    tol = M * 0.01
    active = conv >= M - tol
    if not np.any(active):
        active = conv >= M * 0.99
    
    # Compute subgradient (average of active)
    grad = np.zeros(n)
    active_idx = np.where(active)[0]
    for k in active_idx:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        vals = a[k - j_end:k - j_start + 1][::-1]
        grad[j_start:j_end + 1] += 2.0 * vals
    grad /= len(active_idx)
    
    # Multiplicative weight update
    a_new = a * np.exp(-eta * grad)
    a_new = np.maximum(a_new, 0)

Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based Frank-Wolfe, multiplicative weight updates, simulated annealing,
and coordinate descent with diverse initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix for LP: (a*g)_k for all k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Returns (g, dual_values)."""
    n = len(a)
    A = build_A(a)

    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        g = res.x
        duals = -res.ineqlin_margin if hasattr(res, 'ineqlin_margin') else None
        return g, duals
    return None, None

"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix: A[k,j] such that (A@x)_k = (a*x)_k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0. Returns (g, slack)."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        g = res.x
        slack = b_ub - A @ g if hasattr(res, 'x') else None
        return g, slack
    return None, None


def get_subgradient(a, conv, tolerance_frac=0.01):
    """Compute normalized subgradient of max(conv) at a."""
    n = len(a)
    M = float(np.max(conv))
    if M < 1e-15:
        return np.zeros(n)
    tol = M * tolerance_frac
    active = conv >= M - tol
    if not np.any(active):
        active = conv >= M * 0.99
    if not np.any(active):
        active = np.zeros_like(conv, dtype=bool)
        active[np.argmax(conv)] = True

    grad = np.zeros(n)
    active_idx = np.where(active)[0]
    for k in active_idx:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        vals = a[k - j_end:k - j_start + 1][::-1]
        grad[j_start:j_end + 1] += 2.0 * vals
    grad /= len(active_idx)
    return grad


def fw_with_linesearch(a, alphas=None):
    """Frank-Wolfe step with line search. Returns (new_a, improved)."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    if M < 1e-15:
        return a, False

    g, slack = solve_lp(a, M)
    if g is None:
        return a, False

    g_sum = np.sum(g)
    if g_sum < 1e-10:
        return a, False
    g = g / g_sum

    best_obj = compute_obj(a)
    best_alpha = 0.0

    if alphas is None:
        alphas = [1.0, 0.85, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]

    for alpha in alphas:
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha

    if best_alpha == 0.0:
        return a, False

    a_new = (1.0 - best_alpha) * a + best_alpha * g
    a_new = np.maximum(a_new, 0.0)
    a_new = a_new / np.sum(a_new)
    return a_new, True

def nesterov_subgradient(a, n_steps=500, tol_frac_start=0.005, tol_frac_end=0.05):

        conv_y = np.convolve(y, y)

        a_new = y + v
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)

            best_a = a_new.copy()

        momentum = 2.0 / (step + 3.0)
        y = (1.0 - momentum) * a_new + momentum * a_prev

def coord_descent(a, n_rounds=10, verbose=False):
    """Coordinate descent with multi-point 1D search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    improvement = 0.0

    for rnd in range(n_rounds):
        rnd_improve = 0.0
        for j in range(n):
            old_val = a[j]
            a[j] = 0.0
            base_obj = compute_obj(a)

            multipliers = [0.0, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.85, 1.0, 1.15, 1.35, 1.6, 2.0]
            candidates = [old_val * m for m in multipliers]
            # Add some absolute-scale candidates
            med_val = np.median(a[a > 0]) if np.any(a > 0) else 0.01
            candidates += [med_val * m for m in [0.1, 0.3, 0.5, 1.0, 1.5, 2.0]]

            best_x = old_val
            best_local = compute_obj(a) if a[j] == old_val else base_obj
            a[j] = old_val

            for x in candidates:
                if x < 0:
                    x = 0.0
                a[j] = x
                obj = compute_obj(a)
                if obj < best_local - 1e-12:
                    best_local = obj
                    best_x = x

            if best_x != old_val:
                a[j] = best_x
                rnd_improve += (old_val - best_x) ** 2

        a[j] = old_val  # restore
        final_obj = compute_obj(a)
        if final_obj < best_obj - 1e-12:
            best_obj = final_obj
            best_a = a.copy()
        improvement = rnd_improve
        if verbose and rnd % 3 == 0:
            print(f"      CD round {rnd}: obj={best_obj:.8f}, improvement={improvement:.2e}")

    return best_a, best_obj


def simulated_annealing(a, T, n_perturb=None, scale=0.3):
    """Single SA step: perturb and accept/reject."""
    n = len(a)
    if n_perturb is None:
        n_perturb = max(1, n // 15)

    a_new = a.copy()
    indices = np.random.choice(n, n_perturb, replace=False)
    for idx in indices:
        a_new[idx] *= np.exp(np.random.randn() * scale)
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a
    a_new = a_new / s

    obj_orig = compute_obj(a)
    obj_new = compute_obj(a_new)

    if obj_new <= obj_orig:
        return a_new
    delta = obj_new - obj_orig
    if np.random.random() < np.exp(-delta / max(T, 1e-10)):
        return a_new
    return a

        print(f"  Refining n={n_best}, obj={best_score:.8f}, {remaining:.0f}s left")

        # More LP iterations on the best
        if n_best <= 500 and remaining > 30:
            a = best_seq.copy() / np.sum(best_seq)
            n_iter = int(min(300, remaining / 0.1))
            for it in range(n_iter):
                if time.monotonic() > deadline - 10:
                    break
                a_new, improved = fw_with_linesearch(a, alphas=[1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3])
                if improved:
                    obj = compute_obj(a_new)
                    if obj < best_score:
                        best_score = obj
                        best_seq = a_new.copy()
                        if it % 50 == 0:
                            print(f"    Polish FW it={it}: obj={obj:.8f}")
                a = a_new

        # Nesterov subgradient
        if time.monotonic() < deadline - 15:
            n_sg = int(min(1000, (deadline - time.monotonic()) / 0.03))
            if n_sg > 100:
                a_sg, obj_sg = nesterov_subgradient(best_seq, n_steps=n_sg,
                                                    tol_frac_start=0.002, tol_frac_end=0.02)
                if obj_sg < best_score:
                    best_score = obj_sg
                    best_seq = a_sg.copy()
                    print(f"    Final SG: obj={obj_sg:.8f}")

        # Coordinate descent (multiple rounds)
        if time.monotonic() < deadline - 10 and n_best <= 400:
            n_rounds = int(min(20, (deadline - time.monotonic()) / (n_best * 0.002)))
            n_rounds = max(3, n_rounds)
            a_cd, obj_cd = coord_descent(best_seq, n_rounds=n_rounds)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"    Final CD: obj={obj_cd:.8f}")

        # One more round of FW + CD
        if time.monotonic() < deadline - 15 and n_best <= 400:
            a = best_seq.copy() / np.sum(best_seq)
            n_iter = int(min(100, (deadline - time.monotonic()) / 0.2))
            for it in range(max(10, n_iter)):
                if time.monotonic() > deadline - 10:
                    break
                a_new, improved = fw_with_linesearch(a)
                if improved:
                    obj = compute_obj(a_new)
                    if obj < best_score:
                        best_score = obj
                        best_seq = a_new.copy()
                a = a_new

            if time.monotonic() < deadline - 5:
                a_cd, obj_cd = coord_descent(a, n_rounds=5)
                if obj_cd < best_score:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    print(f"    Second CD: obj={obj_cd:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    # Final validation
    best_seq = np.clip(best_seq, 0, 1000.0)
    best_seq = [float(x) for x in best_seq]

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s")
    return best_seq
