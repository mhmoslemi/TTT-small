    if S < 0.01:

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Adam-based smooth optimization with log-sum-exp and beta annealing
2. LP best-response iteration for local refinement
3. Multi-start with diverse initializations (parent, cosines, random)
4. Periodic perturbation to escape local minima
5. Adaptive time management across phases
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build matrix where A @ x = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None and res.x.sum() > 1e-10:
        return res.x
    return None


def adam_smooth_optimize(a0, deadline, max_steps=2000, beta_start=5.0, beta_end=2000.0, lr=0.005):
    """
    Adam optimization of 2n * LSE(beta * conv(a,a)) with sum(a)=1 constraint.
    Beta is annealed from beta_start to beta_end over the course of optimization.
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-10:
        return a0, compute_score(a0)
    a = a / s
    
    m = np.zeros(n)
    v = np.zeros(n)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    # Precompute FFT size
    size = 1
    while size < 3 * n:
        size <<= 1
    
    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        # Beta schedule: exponential annealing
        frac = (step + 1) / max_steps
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Compute convolution
        a_pos = np.maximum(a, 1e-12)
        b = fast_conv(a_pos)
        
        # Softmax weights
        b_max = b.max()
        shifted = beta * (b - b_max)
        shifted = np.clip(shifted, -500, 500)
        exp_shifted = np.exp(shifted)
        sum_exp = exp_shifted.sum()
        if sum_exp < 1e-300:
            break
        p = exp_shifted / sum_exp
        
        # Gradient: 4n * cross_corr(p, a)
        P = np.fft.rfft(p, size)
        A = np.fft.rfft(a_pos, size)
        cross = np.fft.irfft(np.conj(P) * A, size)
        grad = 4.0 * n * cross[:n]
        
        # Adam update
        t = step + 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)
        
        # Learning rate decay
        lr_t = lr * (0.5 ** (step / 500.0))
        
        a_new = a - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = np.clip(a_new, 0, None)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a = a_new
            break
        a_new = a_new / s_new
        a = a_new
        
        # Evaluate every 100 steps
        if (step + 1) % 100 == 0 or step == max_steps - 1:

    inits.append(np.exp(rng.normal(0, 0.5, n)) )

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Multi-stage Adam optimization with log-sum-exp smoothing and beta annealing
2. Randomized subgradient method on the hard max objective
3. LP best-response for final polish
4. Multi-start with diverse initializations (parent, cosines, random)
5. Contractive perturbations (mixing with uniform) to escape local minima
6. Multiple sequence lengths exploration

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None and res.x.sum() > 1e-10:
        return res.x
    return None
