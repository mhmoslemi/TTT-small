    n = len(a)
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            rows_list.append(np.full(len(js), k, dtype=np.int32))
            cols_list.append(js.astype(np.int32))
            vals_list.append(a[k - js])
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    A = lil_matrix((2 * n - 1, n))
    A[rows, cols] = vals
    return A.tocsr()


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def spectral_flatten(a, strength=0.3):
    """Blend a with its flat-spectrum projection to reduce autocorrelation peak."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return a.copy()
    a_norm = a / s
    F = np.fft.fft(a_norm)
    mag = np.abs(F)
    phase = np.angle(F)
    # Compress dynamic range: new_mag = mag^alpha with alpha in (0,1)
    alpha = 1.0 - strength
    safe_mag = np.maximum(mag, 1e-12)
    new_mag = safe_mag ** alpha
    # Preserve DC component
    new_mag[0] = mag[0]
    F_new = new_mag * phase
    a_new = np.fft.ifft(F_new).real
    a_new = np.maximum(a_new, 0.0)
    sn = a_new.sum()
    if sn < 1e-10:
        return a.copy()
    a_new /= sn
    result = (1.0 - strength) * a_norm + strength * a_new
    result = np.maximum(result, 0.0)
    rs = result.sum()
    if rs < 1e-10:
        return a.copy()
    return result / rs


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def tracker_update(a):
    """Update global best if a is better."""
    global _best_seq, _best_score
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.size == 0 or a.sum() < 0.01:
        return
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < _best_score:
        _best_seq = a.copy()
        _best_score = score
        elapsed = time.monotonic() - _t_start
        print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)


def lp_mirror_descent(a, max_iters, deadline):
    """LP mirror descent with backtracking and periodic spectral projection."""
    global _best_seq, _best_score
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    tracker_update(a)
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        # Backtracking line search with Armijo-like condition
        best_a = None
        best_J = current_J
        t_candidates = [0.8, 0.6, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-10:
                continue
            a_trial /= st
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
                if t >= 0.4:
                    break

        if best_a is None:
            no_improve_count += 1
            # Try spectral projection every few failed iterations
            if no_improve_count % 3 == 0:
                a_sp = spectral_flatten(a, strength=0.2)
                J_sp = compute_J(a_sp)
                if J_sp < current_J - 1e-12:
                    a = a_sp
                    current_J = J_sp
                    tracker_update(a)
                    no_improve_count = 0
                elif no_improve_count > 15:
                    break
        else:
            a = best_a
            current_J = best_J
            no_improve_count = 0
            if it % 8 == 0:
                tracker_update(a)

    tracker_update(a)
    return a


def local_descent(a, deadline, max_rounds=50):
    """Coordinate descent: perturb one element at a time."""
    global _best_seq, _best_score
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()
    current_J = compute_J(a)
    n = len(a)
    rng = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        # Random order
        idx_order = rng.permutation(n)
        for i in idx_order:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            for pert in [0.5 * old_val, 2.0 * old_val, 0.0, 0.3 * old_val, 1.5 * old_val]:
                a[i] = pert
                a[i] = max(a[i], 0.0)
                s = a.sum()
                if s < 1e-10:
                    a[i] = old_val
                    continue
                a_norm = a / s
                J_try = compute_J(a_norm)
                if J_try < current_J - 1e-12:
                    a = a_norm
                    current_J = J_try
                    improved = True
                    break
            a[i] = old_val if not improved else a[i]
        if not improved:
            break
        if rnd % 10 == 0:
            tracker_update(a)

    tracker_update(a)
    return a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: LP mirror descent + spectral projection + SA + local descent.
    Multi-length exploration with parent seeding.
    """
    global _best_seq, _best_score, _t_start
    _best_seq = None
    _best_score = float('inf')
    _t_start = time.monotonic()
    deadline = _t_start + float(budget_s) - 25.0

    rng = np.random.default_rng(seed)

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker_update(parent)
                print(f"Parent loaded: n={parent.size}, J={_best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: LP from parent at multiple lengths ===
    if parent is not None:
        parent_n = parent.size
        target_lengths = sorted(set([parent_n, max(64, parent_n // 2),
                                     min(4096, parent_n * 2),
                                     min(4096, parent_n * 4)]))
        for n in target_lengths:
            if time.monotonic() > deadline:
                break
            t0 = time.monotonic()
            print(f"\n[Phase1] LP from parent, n={n}", flush=True)

            # Forward
            a0 = resample_seq(parent, n)
            a0 = np.maximum(a0, 1e-8)
            a0 /= a0.sum()
            iters = max(40, min(400, int(40000 // n)))
            lp_mirror_descent(a0, iters, deadline)

            # Reversed
            if time.monotonic() < deadline:
                a0r = resample_seq(parent[::-1], n)
                a0r = np.maximum(a0r, 1e-8)
                a0r /= a0r.sum()
                lp_mirror_descent(a0r, iters, deadline)

            print(f"  Phase1 n={n} done: {time.monotonic()-t0:.1f}s, best J={_best_score:.6f}", flush=True)

    # === Phase 2: Diversified LP from analytic initializations ===
    if time.monotonic() < deadline:
        lengths = [128, 256, 512, 1024]
        for n in lengths:
            if time.monotonic() > deadline:
                break
            t0 = time.monotonic()
            print(f"\n[Phase2] Diversified LP, n={n}", flush=True)

            t_arr = np.linspace(0, 1, n, endpoint=False)
            inits = []

            # From parent
            if parent is not None:
                inits.append(resample_seq(parent, n))
                inits.append(resample_seq(parent[::-1], n))

            # Asymmetric shapes (key: non-palindromic!)
            inits.append(t_arr ** 0.5)
            inits.append((1.0 - t_arr + 0.01) ** 0.5)
            inits.append(np.exp(-1.5 * (t_arr - 0.3) ** 2) + 0.05)
            inits.append(np.exp(-1.5 * (t_arr - 0.7) ** 2) + 0.05)
            inits.append(1.0 + 0.4 * np.cos(2 * np.pi * 2 * t_arr))
            inits.append(1.0 + 0.3 * np.sin(2 * np.pi * 3 * t_arr + 0.5))
            inits.append(1.0 + 0.4 * t_arr)  # Linear ramp
            inits.append(1.0 + 0.4 * (1.0 - t_arr))  # Linear anti-ramp
            inits.append((t_arr + 0.1) ** 2)
            inits.append(1.0 + 0.3 * np.cos(2 * np.pi * 5 * t_arr) + 0.2 * t_arr)
            inits.append(np.abs(np.sin(np.pi * t_arr)) * (1.0 + 0.5 * t_arr) + 0.05)

            for a0 in inits:
                if time.monotonic() > deadline:
                    break
                a0 = np.maximum(a0.astype(np.float64), 1e-8)
                a0 /= a0.sum()
                iters = max(30, min(300, int(30000 // n)))
                lp_mirror_descent(a0, iters, deadline)

            print(f"  Phase2 n={n} done: {time.monotonic()-t0:.1f}s, best J={_best_score:.6f}", flush=True)

    # === Phase 3: Simulated Annealing restarts ===
    if _best_seq is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = _best_seq.size
        print(f"\n[Phase3] SA restarts, base n={n}", flush=True)

        temp = 0.12
        n_restarts = 50
        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break
            # Perturb best
            a = _best_seq.copy().astype(np.float64)
            noise = rng.normal(0, temp, n)
            a = a + noise
            a = np.maximum(a, 1e-8)
            a /= a.sum()
            iters = max(20, min(80, 100 - r // 2))
            lp_mirror_descent(a, iters, deadline)
            temp *= 0.96

        # Try different lengths for SA
        for n2 in sorted(set([n * 2, n // 2, n * 3])):
            if n2 < 64 or n2 > 4096:
                continue
            if time.monotonic() > deadline:
                break
            for r in range(15):
                if time.monotonic() > deadline:
                    break
                a = resample_seq(_best_seq, n2)
                a = a + rng.normal(0, 0.06, n2)
                a = np.maximum(a, 1e-8)
                a /= a.sum()
                lp_mirror_descent(a, 40, deadline)

        print(f"  Phase3 done: {time.monotonic()-t0:.1f}s, best J={_best_score:.6f}", flush=True)

    # === Phase 4: Local coordinate descent ===
    if _best_seq is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n[Phase4] Local descent, n={_best_seq.size}", flush=True)
        max_rounds = max(20, int((deadline - time.monotonic()) * 5))
        local_descent(_best_seq, deadline, max_rounds=min(100, max_rounds))
        print(f"  Phase4 done: {time.monotonic()-t0:.1f}s, best J={_best_score:.6f}", flush=True)

    # === Phase 5: Final intensive LP polish ===
    if _best_seq is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n[Phase5] Final LP polish, n={_best_seq.size}", flush=True)
        remaining = deadline - time.monotonic()
        max_iters = max(50, min(800, int(remaining * 20)))
        lp_mirror_descent(_best_seq.copy(), max_iters, deadline)

        # Spectral + LP alternation
        for _ in range(10):
            if time.monotonic() > deadline:
                break
            a_sp = spectral_flatten(_best_seq, strength=0.15)
            J_sp = compute_J(a_sp)
            if J_sp < _best_score - 1e-10:
                _best_seq = a_sp
                _best_score = J_sp
                tracker_update(a_sp)
                lp_mirror_descent(a_sp, 30, deadline)
            else:
                break

        print(f"  Phase5 done: {time.monotonic()-t0:.1f}s, best J={_best_score:.6f}", flush=True)

    # Fallback
    if _best_seq is None:
        _best_seq = np.ones(128)
        _best_score = compute_J(_best_seq)
        print(f"  Fallback used: J={_best_score:.6f}", flush=True)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL RESULT: J={_best_score:.8f}, n={_best_seq.size}, time={time.monotonic()-_t_start:.1f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in _best_seq]
