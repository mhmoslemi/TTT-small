            quad = np.zeros(k_end - k_start)

            for idx_op in range(k_end - k_start):
                k = k_start + idx_op
                j = j_indices[idx_op]
                if k == 2 * i:
                    base[idx_op] = conv[k] - v_old * v_old
                    quad[idx_op] = 1.0
                else:
                    base[idx_op] = conv[k] - 2.0 * v_old * f[j]
                    lin[idx_op] = 2.0 * f[j]

            # Objective function for 1D optimization
            S_base_val = S_base
            n_val = n

            def obj_1d(v):
                changed = base + lin * v + quad * v * v
                max_changed = float(np.max(changed))
                max_conv = max(max_changed, max_unchanged)
                S_new = S_base_val + v
                if S_new < 1e-15:
                    return 1e10
                return 2.0 * n_val * max_conv / (S_new * S_new)

            # Bounded minimization
            res = minimize_scalar(obj_1d, bounds=(0.0, upper), method='bounded',
                                  options={'xatol': 1e-12, 'maxiter': 60})
            v_new = float(res.x)

            # Update convolution incrementally
            if v_new != v_old:
                delta_lin = 2.0 * (v_new - v_old)
                for idx_op in range(k_end - k_start):
                    k = k_start + idx_op
                    j = j_indices[idx_op]
                    if k == 2 * i:
                        conv[k] += v_new * v_new - v_old * v_old
                    else:
                        conv[k] += delta_lin * f[j]
                f[i] = v_new
                S = S_base_val + v_new

        pass_count += 1
        if verbose and pass_idx % 10 == 0:
            cur_score = compute_score(f)
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
            print(f"    {tag} CD pass {pass_idx}: score={cur_score:.6f} best={best_score:.6f}")

    cur_score = compute_score(f)
    if cur_score < best_score:
        best_score = cur_score
        best_f = f.copy()

    return best_f, best_score, f, conv, S


def subgradient_step(f, f_prime=None, lr=0.01, softmax_tau=None):
    """Single subgradient step. Returns new f.
    If f_prime is given, use it for gradient computation (Nesterov)."""
    n = len(f)
    if f_prime is None:
        f_prime = f
    conv = np.convolve(f_prime, f_prime)
    M = float(np.max(conv))
    if M < 1e-15:
        return f

    # Softmax-weighted subgradient of the max
    if softmax_tau is None:
        tau = 0.02 * M
    else:
        tau = softmax_tau
    shifted = conv / tau - np.max(conv) / tau
    weights = np.exp(shifted)
    weights /= np.sum(weights)

    # Compute gradient: grad[i] = sum_k weights[k] * d(conv[k])/d(f[i])
    # d(conv[k])/d(f[i]) = 2*f[k-i] if 0 <= k-i < n, else 0
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        i_lo = max(0, k - (n - 1))
        i_hi = min(k, n - 1)
        for i in range(i_lo, i_hi + 1):
            grad[i] += 2.0 * w * f_prime[k - i]

    return grad, M, tau


def nesterov_subgradient(f, deadline, lr_init=0.05, verbose=False, tag=""):
    """Nesterov-accelerated subgradient with projection onto {f>=0, sum(f)=S}."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    f = f / S  # Normalize to sum=1

    best_f = f.copy()
    best_score = compute_score(best_f)

    v = np.zeros(n)  # momentum
    lr = lr_init
    no_improve = 0
    it = 0
    tau_scale = 0.02

    while time.monotonic() < deadline:
        it += 1
        # Nesterov extrapolation
        if it > 1:
            beta = (1.0 + np.sqrt(1.0 + 4.0 * (it - 1) ** 2)) / 2.0
            beta_prev = (1.0 + np.sqrt(1.0 + 4.0 * (it - 2) ** 2)) / 2.0
            f_prime = f + (beta - 1.0) / (beta_prev + 1.0) * (v / max(lr, 1e-15))
            f_prime = np.clip(f_prime, 0, None)
            S_fp = np.sum(f_prime)
            if S_fp > 1e-15:
                f_prime = f_prime / S_fp
        else:
            f_prime = f

        grad, M, tau = subgradient_step(f, f_prime, lr=lr, softmax_tau=tau_scale * M)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm

        # Update
        v = v - lr * grad
        f_new = np.clip(f + v, 0.0, None)
        S_new = np.sum(f_new)
        if S_new < 1e-15:
            lr *= 0.5
            no_improve += 1
            if no_improve > 200:
                break
            continue
        f_new = f_new / S_new

        new_score = compute_score(f_new)
        if new_score < best_score:
            f = f_new
            best_f = f_new.copy()
            best_score = new_score
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    {tag}nest iter {it}: score={best_score:.6f} lr={lr:.5f}")
        else:
            no_improve += 1
            lr *= 0.97
            if lr < 1e-8 or no_improve > 500:
                break

    # Scale back to original S
    return best_f * S, best_score


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_step(f, deadline, max_iters=20, step_init=0.1, verbose=False, tag=""):
    """LP-based direction finding with line search."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = compute_score(f)
    step = step_init
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break

        g = solve_lp_direction(f, M)
        if g is None:
            no_improve += 1
            if no_improve > 5:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-15:
            no_improve += 1
            continue

        g = g * (S / sum_g)

        # Line search
        best_step_score = best_score
        best_step_f = None
        for alpha in [step, step * 0.5, step * 0.25, step * 1.5]:
            alpha = min(alpha, 1.0)
            if alpha < 0.001:
                continue
            candidate = np.clip((1.0 - alpha) * f + alpha * g, 0.0, None)
            cand_sum = float(np.sum(candidate))
            if cand_sum < 1e-15:
                continue
            candidate = candidate * (S / cand_sum)
            cand_score = compute_score(candidate)
            if np.isfinite(cand_score) and cand_score < best_step_score:
                best_step_score = cand_score
                best_step_f = candidate

        if best_step_f is not None:
            f = best_step_f
            best_f = f.copy()
            best_score = best_step_score
            no_improve = 0
            if verbose:
                print(f"    {tag}LP iter {it}: score={best_score:.6f}")
            step = max(step * 0.95, 0.02)
        else:
            no_improve += 1
            step *= 0.7
            if step < 0.005 or no_improve > 10:
                break

    return best_f, best_score


def make_structured_init(n, rng):
    """Create a structured initialization designed to have flat auto-correlation."""
    idx = np.arange(n, dtype=np.float64)
    inits = []

    # Raised cosine
    inits.append((1.0 + np.cos(2 * np.pi * idx / n)) / 2.0)

    # Sinc-like (windowed)
    center = (n - 1) / 2.0
    x = (idx - center) / (n / 6.0)
    sinc = np.sinc(x * 0.5)
    sinc = np.clip(sinc, 0.01, None)
    inits.append(sinc)

    # Tukey window (tapered)
    r = 0.3
    tukey = np.ones(n)
    taper_len = int(r * n / 2)
    if taper_len > 1:
        t = np.linspace(0, taper_len, taper_len)
        taper = 0.5 * (1 + np.cos(np.pi * t / taper_len))
        tukey[:taper_len] = taper
        tukey[-taper_len:] = taper[::-1]
    inits.append(tukey)

    # Gaussian
    inits.append(np.exp(-0.5 * ((idx - center) / (n / 5.0)) ** 2))

    # Two peaks
    inits.append(np.exp(-0.5 * ((idx - n * 0.2) / (n / 10.0)) ** 2) +
                np.exp(-0.5 * ((idx - n * 0.8) / (n / 10.0)) ** 2))

    # Flat with small random perturbation
    flat = np.ones(n) + 0.1 * rng.standard_normal(n)
    flat = np.clip(flat, 0.01, None)
    inits.append(flat)

    # Random
    inits.append(0.5 + rng.random(n))

    return inits


def propose_candidate(budget_s=1500, **kwargs):
    """Main entry point: search for the best sequence of length 1024."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 10.0)
    N = 1024
    rng = np.random.default_rng(42)

    best_sequence = None
    best_score = float('inf')

    # Try to load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            # Interpolate to length N
            if parent.size != N:
                x_old = np.linspace(0, 1, parent.size)
                x_new = np.linspace(0, 1, N)
                parent_N = np.interp(x_new, x_old, parent)
            else:
                parent_N = parent.copy()
            parent_N = np.clip(parent_N, 0.0, None)
            if np.sum(parent_N) > 0.01:
                score = compute_score(parent_N)
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = parent_N.copy()
                print(f"Parent (n={parent.size}): score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")
        parent_N = None

    # If no parent, use structured init
    if best_sequence is None:
        print("No parent found, using structured initialization.")
        inits = make_structured_init(N, rng)
        best_init_score = float('inf')
        for label, init in inits:
            init = np.clip(init, 0.01, None)
            s = compute_score(init)
            if s < best_init_score:
                best_init_score = s
                best_sequence = init.copy()
        best_score = best_init_score
        print(f"Best structured init: score={best_score:.6f}")

    print(f"Starting optimization: n={N}, initial score={best_score:.6f}")

    # Phase 1: Quick coordinate descent to get a good starting point
    print("\n=== Phase 1: Initial Coordinate Descent ===")
    phase1_deadline = t0 + 120.0
    f, s, f_work, conv_work, S_work = coord_desc_passes(
        best_sequence, num_passes=200, deadline=phase1_deadline,
        verbose=True, tag="P1-CD ")
    if s < best_score:
        best_score = s
        best_sequence = f.copy()
    print(f"Phase 1 done: score={best_score:.6f}")

    # Phase 2: Main optimization loop
    print("\n=== Phase 2: Main Optimization ===")
    phase2_deadline = deadline - 200.0  # Leave 200s for final polish

    # Work with normalized sequence (sum=1) for CD
    f_work = best_sequence.copy()
    S_work = float(np.sum(f_work))
    f_norm = f_work / S_work
    conv_norm = np.convolve(f_norm, f_norm).copy()
    n = N

    last_cd_start = time.monotonic()
    no_big_improve = 0
    lp_interval = 300  # seconds between LP calls

    while time.monotonic() < phase2_deadline:
        time_since_lp = time.monotonic() - last_cd_start

        # Coordinate descent block
        cd_deadline = min(phase2_deadline, time.monotonic() + 120.0)
        f_cd, s_cd, f_work, conv_work, S_work = coord_desc_passes(
            f_work, num_passes=200, deadline=cd_deadline, tag="P2-CD ")
        if s_cd < best_score:
            best_score = s_cd
            best_sequence = f_cd.copy()
            no_big_improve = 0
        else:
            no_big_improve += 1

        # Nesterov subgradient (to escape local minima)
        if time.monotonic() < phase2_deadline - 30:
            nest_deadline = min(phase2_deadline, time.monotonic() + 60.0)
            f_nest, s_nest = nesterov_subgradient(
                f_work, nest_deadline, lr_init=0.03, tag="P2-")
            if s_nest < best_score:
                best_score = s_nest
                best_sequence = f_nest.copy()
                f_work = f_nest.copy()
                no_big_improve = 0
            else:
                # Try CD from the Nesterov result too
                f_cd2, s_cd2, f_work2, conv_w2, S_w2 = coord_desc_passes(
                    f_work, num_passes=50,
                    deadline=min(phase2_deadline, time.monotonic() + 30.0),
                    tag="P2-CD2")
                if s_cd2 < best_score:
                    best_score = s_cd2
                    best_sequence = f_cd2.copy()
                    f_work = f_work2.copy()
                    no_big_improve = 0

        # LP step (less frequently, as it's expensive)
        if time_since_lp > lp_interval and time.monotonic() < phase2_deadline - 60:
            lp_deadline = min(phase2_deadline, time.monotonic() + 60.0)
            f_lp, s_lp = lp_step(best_sequence, lp_deadline, max_iters=15,
                                  step_init=0.08, tag="P2-LP")
            if s_lp < best_score:
                best_score = s_lp
                best_sequence = f_lp.copy()
                f_work = f_lp.copy()
                no_big_improve = 0
            last_cd_start = time.monotonic()

        # Random perturbation restart (every ~600s if stuck)
        if no_big_improve > 8 and time.monotonic() < phase2_deadline - 60:
            print(f"  Stuck at {best_score:.6f}, trying perturbation...")
            f_pert = best_sequence * (1.0 + 0.05 * rng.standard_normal(N))
            f_pert = np.clip(f_pert, 0.0, None)
            if np.sum(f_pert) > 0.01:
                f_pert = f_pert / np.sum(f_pert) * np.sum(best_sequence)
                f_pert_cd, s_pert, _, _, _ = coord_desc_passes(
                    f_pert, num_passes=30,
                    deadline=min(phase2_deadline, time.monotonic() + 30.0),
                    tag="P2-PERT")
                if s_pert < best_score:
                    best_score = s_pert
                    best_sequence = f_pert_cd.copy()
                    f_work = f_pert_cd.copy()
                    no_big_improve = 0
                    print(f"  Perturbation helped! score={best_score:.6f}")
                no_big_improve = 0  # Reset to avoid too many perturbations

    print(f"Phase 2 done: best_score={best_score:.6f}")

    # Phase 3: Final coordinate descent polish
    print(f"\n=== Phase 3: Final CD Polish (target < {best_score:.6f}) ===")
    if best_sequence is not None:
        f_final, s_final, _, _, _ = coord_desc_passes(
            best_sequence, num_passes=2000, deadline=deadline,
            verbose=True, tag="P3-CD ")
        if s_final < best_score:
            best_score = s_final
            best_sequence = f_final.copy()

        # One more LP polish if time
        if time.monotonic() < deadline - 10:
            f_lp, s_lp = lp_step(best_sequence, deadline, max_iters=10,
                                  step_init=0.03, tag="P3-LP")
            if s_lp < best_score:
                best_score = s_lp
                best_sequence = f_lp.copy()

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(N)
        best_score = compute_score(best_sequence)

    # Ensure all values are valid for the evaluator
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    final_score = evaluate_sequence([float(x) for x in best_sequence])
    if not np.isfinite(final_score):
        print("WARNING: Final sequence failed evaluation! Using fallback.")
        best_sequence = np.ones(512)
        final_score = evaluate_sequence([float(x) for x in best_sequence])

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, "
          f"internal_best={best_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_sequence]
