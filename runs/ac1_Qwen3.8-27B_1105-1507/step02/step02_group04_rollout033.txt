"""
Optimized search for step function minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses accelerated subgradient with momentum, diverse initializations including
structured sequences, adaptive smoothing, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smoothed_max_and_grad(conv, a, tau):
    """Compute smoothed max and its gradient w.r.t. a."""
    n = len(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum
    # grad[i] = 2 * sum_j weights[i+j] * a[j] (for valid indices)
    # This is 2 * (weights correlated with a)[i]
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[:n]
    smooth_max = max_conv - np.log(w_sum) / tau
    return smooth_max, grad

def accelerated_subgradient(a, max_iter=50000, tau_start=3.0, tau_end=80.0,

def build_A_matrix(a):
    """Build Toeplitz constraint matrix for LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A

def solve_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def lp_step(a, t=0.3, time_limit=20):
    """One LP-based update step."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv))
    g = solve_lp(a, M, time_limit)
    if g is None:
        return None
    sum_g = float(np.sum(g))
    if sum_g < 1.0 + 1e-4:
        return None
    g = g / sum_g
    a_new = (1.0 - t) * a + t * g
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return None
    a_new = a_new / s
    return a_new

def init_quadratic_residue(n, rng):
    """Initialize using quadratic residues for good autocorrelation properties."""
    # Find prime p >= n
    p = n
    while True:
        # Simple primality test
        is_prime = True
        for d in range(2, int(np.sqrt(p)) + 1):
            if p % d == 0:
                is_prime = False
                break
        if is_prime:
            break
        p += 1
    if p > 3 * n:
        return init_random_structured(n, rng)
    
    # Quadratic residue sequence of length p
    # QR: x is QR mod p if x = y^2 mod p for some y
    qr = np.zeros(p)
    for x in range(1, p):
        # Check if x is QR mod p (Euler's criterion)
        if pow(x, (p-1)//2, p) == 1:
            qr[x] = 1.0
    
    # Take first n elements (or a shifted version)
    shift = rng.integers(0, p)
    a = qr[shift:shift+n]
    if len(a) < n:
        a = np.concatenate([qr[shift:], qr[:n-len(a)]])
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return init_random_structured(n, rng)
    a = a / s
    return a
