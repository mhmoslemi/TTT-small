    n = len(a)

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Accelerated projected gradient (FISTA) with log-sum-exp smoothing and beta annealing
2. LP best-response for final polish

4. Adaptive time allocation

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build Toeplitz-like matrix A where (A @ x) = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(1.0 - u * (np.arange(1, n + 1) / cssv) > 0)[0]
    if len(rho) == 0:
        # All components are 0 or negative
        return np.ones(n) / n
    r = rho[-1]
    theta = cssv[r] / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_objective(a, beta, n):
    """(2n/beta) * log(sum(exp(beta * b))) where b = conv(a,a), a normalized."""
    b = fast_conv(a)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    log_sum = b_max + np.log(np.exp(shifted).sum()) / beta
    return 2.0 * n * log_sum

def fISTA_smooth(a_init, n, A, betas, iters_per_beta, deadline):
    """
    Run FISTA with smoothed objective and beta annealing.
    a_init should be on the simplex (sum=1, a>=0).
    Returns (best_a, best_score).
    """
    a = a_init.copy()
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_score = compute_score(a)
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        a = a.copy()
        y = a.copy()
        t = 1.0
        
        # Estimate step size: L ≈ 4n * max_eig(A.T A) / beta or similar
        # For safety, use backtracking
        step = 1.0 / (4.0 * n)  # conservative initial step
        
        for it in range(iters_per_beta):
            if time.monotonic() > deadline:
                break
            
            # Compute gradient at y
            b_y = fast_conv(y)
            b_max = b_y.max()
            shifted = beta * (b_y - b_max)
            shifted = np.clip(shifted, -500.0, 500.0)
            exp_shifted = np.exp(shifted)
            p = exp_shifted / exp_shifted.sum()
            grad = 4.0 * n * (A.T @ p)
            
            # Backtracking line search
            # f(y - step*grad) <= f(y) - 0.5*step*||grad||^2
            # For simplicity, use a fixed step (the Lipschitz constant estimate)
            # L for the smoothed objective: ||∇f(x) - ∇f(y)|| <= L*||x-y||
            # ∇f = 4n * A^T @ p(a) where p depends on a
            # This is complex to bound. Use a safe step.
            
            step = 1.0 / (4.0 * n * (1 + beta))  # more conservative for larger beta
            
            # FISTA update
            z = a - step * grad
            a_new = project_simplex(z)
            
            # Check if we improved (optional, for safety)
            # a_new_score = compute_score(a_new)
            # Only update if improved (conservative)
            
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = a_new + ((t - 1.0) / t_new) * (a_new - a)
            y = np.clip(y, 0.0, None)
            y_sum = y.sum()
            if y_sum < 1e-10:
                y = a_new.copy()
            else:
                y = y / y_sum  # Project y back to simplex for numerical stability
            
            a = a_new
            t = t_new
        
        # After each beta, check score
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Also check a version with slight perturbation
        if it < len(betas) - 1:
            # Smooth transition to next beta
            pass
    
    return best_a, best_score


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None


def lp_iteration(a, max_iters, deadline):
    """Run LP best-response iterations."""
    best_local = a.copy()
    best_score = compute_score(a)
    a = a.copy()
    a = np.clip(a, 0.001, None)
    a = a / a.sum()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()
        
        t = 0.3
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            a = a_new
        else:
            t2 = t * 0.3
            a_new2 = (1.0 - t2) * a + t2 * g
            a_new2 = np.clip(a_new2, 0.0, None)
            s2 = a_new2.sum()
            if s2 > 1e-10:
                a_new2 = a_new2 / s2
                score2 = compute_score(a_new2)
                if score2 < best_score:
                    best_score = score2
                    best_local = a_new2.copy()
                a = a_new2
            if t2 < 0.01:
                break
    
    return best_local, best_score


def generate_initializations(n, rng, parent=None):

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated projected gradient with log-sum-exp smoothing and beta annealing
2. LP best-response iteration for final polish
3. Multi-start with diverse initializations across multiple sequence lengths
4. Perturbation-based local search to escape local optima
5. Adaptive time budget allocation
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build Toeplitz-like matrix A where (A @ x) = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = 1.0 - u * (np.arange(1, n + 1) / cssv) > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    r = idx[-1]
    theta = cssv[r] / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(a, beta):
    """
    Compute gradient of (2n/beta)*log(sum(exp(beta*b))) where b=conv(a,a).
    Returns (gradient, b) tuple. Uses the identity:
    grad_k = 4n * sum_j p_j * a_{j-k+(n-1)} (if valid)
           = 4n * (A_a.T @ p)_k
    where p_j = exp(beta*b_j)/sum(exp(beta*b)).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    A = build_conv_matrix(a)
    grad = 4.0 * n * (A.T @ p)
    return grad, b


def accelerated_gradient(a_init, betas, iters_per_beta, deadline):
    """
    Nesterov-accelerated projected gradient on smoothed objective with beta annealing.
    a_init must be on the simplex.
    """
    n = len(a_init)
    best_a = a_init.copy()
    best_score = compute_score(a_init)
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        a = best_a.copy()
        y = a.copy()
        t_momentum = 1.0
        
        # Adaptive step size with backtracking
        step = 1.0 / (4.0 * n * max(1.0, beta * 0.1))
        
        for it in range(iters_per_beta):
            if time.monotonic() > deadline:
                break
            
            # Compute gradient at y
            grad_y, _ = compute_smoothed_grad(y, beta)
            
            # FISTA update
            z = y - step * grad_y
            a_new = project_simplex(z)
            
            # Check if we should restart (if no progress)
            if it > 0 and it % 50 == 0:
                score_check = compute_score(a_new)
                if score_check < best_score:
                    best_score = score_check
                    best_a = a_new.copy()
                    # Restart momentum
                    y = a_new.copy()
                    t_momentum = 1.0
                
                # Adaptive step: if we're making progress, keep; else reduce
                # Simple heuristic: reduce step every 100 iters
                if it % 100 == 0:
                    step *= 0.8
            
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_momentum * t_momentum))
            y = a_new + ((t_momentum - 1.0) / t_new) * (a_new - a)
            y = np.clip(y, 0.0, None)
            y_sum = y.sum()
            if y_sum < 1e-10:
                y = a_new.copy()
            else:
                y = y / y_sum
            
            a = a_new
            t_momentum = t_new
        
        # Check score after this beta
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        else:
            # Try the y point too
            score_y = compute_score(y)
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
    
    return best_a, best_score


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None


def lp_iteration(a, max_iters, deadline):
    """Run LP best-response iterations with adaptive step size."""
    best_local = a.copy()
    best_score = compute_score(a)
    a = a.copy()
    a = np.clip(a, 0.001, None)
    a = a / a.sum()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()
        
        # Adaptive step: start moderate, decay
        t_step = 0.4 * (1.0 - 0.5 * min(1.0, it / max(1, max_iters * 0.5)))
        t_step = max(0.05, t_step)
        
        a_new = (1.0 - t_step) * a + t_step * g
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            t2 = t_step * 0.3
            a_new2 = (1.0 - t2) * a + t2 * g
            a_new2 = np.clip(a_new2, 0.0, None)
            s2 = a_new2.sum()
            if s2 > 1e-10:
                a_new2 = a_new2 / s2
                score2 = compute_score(a_new2)
                if score2 < best_score:
                    best_score = score2
                    best_local = a_new2.copy()
                a = a_new2
            # If stuck, break
            if t2 < 0.01:
                break
    
    return best_local, best_score


def gradient_descent_polish(a, deadline, max_iters=5000):
    """
    Simple projected gradient descent on the unsmoothed max objective.
    Uses subgradient (direction of the maximizing lag).
    """
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    a = a.copy()
    
    step = 0.01
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        k_star = np.argmax(b)
        
        # Subgradient: direction to reduce b_{k_star}
        # grad b_{k_star} / da_k = 2*a_{k_star - k + (n-1)} if valid
        # This is 2 * (shifted a)
        A = build_conv_matrix(a)
        grad = 2.0 * A[k_star, :]
        
        # Also add a term to reduce other high lags (smoothed subgradient)
        # Use top-5 lags with weighted subgradient
        top_k = min(5, 2 * n - 1)
        top_indices = np.argsort(b)[::-1][:top_k]
        weights = b[top_indices]
        weights = weights / weights.sum()
        grad_multi = np.zeros(n)
        for w, k in zip(weights, top_indices):
            grad_multi += w * A[k, :]
        
        # Use the multi-lag gradient for smoother descent
        v = momentum * v + (1.0 - momentum) * grad_multi
        a_new = a - step * v
        a_new = project_simplex(a_new)
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            # Increase step slightly on success
            step = min(0.05, step * 1.01)
        else:
            # Reduce step on failure
            step *= 0.7
            a = a_new  # Still move (with smaller effective step)
        
        if step < 1e-6:
            break
    
    return best_a, best_score


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    t_arr = np.linspace(0, 1, n)
    
    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        resampled = np.maximum(np.interp(t_arr, t_old, parent), 0)
        if resampled.sum() > 0.01:
            inits.append(resampled / resampled.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Various cosine harmonics
    for h in [1, 2, 3, 4, 5, 6, 8, 12]:
        amp = min(0.4, 0.8 / h)
        c = 1.0 + amp * np.cos(2 * np.pi * h * t_arr)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())
    
    # Sine (odd symmetry)
    for h in [1, 2, 3]:
        c = 1.0 + 0.3 * np.sin(2 * np.pi * h * t_arr)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())
    
    # Quadratic / parabolic
    c = 1.0 + 0.5 * (1.0 - 4.0 * (t_arr - 0.5) ** 2)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())
    
    # Triangular
    c = 1.0 - 0.5 * np.abs(t_arr - 0.5)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())
    
    # Bimodal
    c = 1.0 + 0.3 * np.cos(4 * np.pi * t_arr)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())
    
    # Random smooth
    freqs = rng.integers(1, 5, size=3)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amps = rng.uniform(0.1, 0.3, size=3)
    c = np.ones(n)
    for f, p, a in zip(freqs, phases, amps):

    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())
    
    # Pure random
    c = rng.uniform(0.2, 1.0, n)
    inits.append(c / c.sum())
    
    return inits

        print(f"  Perturb done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)
    
    # === PHASE 5: Final LP + subgradient alternation ===
    if best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n=== Final Polish ===", flush=True)
        remaining = deadline - time.monotonic()
        
        # Alternate LP and subgradient for a few rounds
        for round_i in range(3):
            if time.monotonic() > deadline:
                break
            # LP step
            n = best.size
            est_time = 0.03 if n <= 512 else 0.06
            max_iters = max(50, int((deadline - time.monotonic()) * 0.2 / est_time))
            a_result, score = lp_iteration(best, max_iters, deadline)
            if score < best_score:
                best = a_result.copy()
                best_score = score
            
            # Subgradient step
            if time.monotonic() < deadline:
                max_sg = max(1000, int((deadline - time.monotonic()) * 0.3 / 0.005))
                a_result, score = gradient_descent_polish(best, deadline, max_sg)
                if score < best_score:
                    best = a_result.copy()
                    best_score = score
        
        print(f"  Final polish done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)
    
    # Final validation
    if best is None:
        best = np.ones(256) / 256
        best_score = compute_score(best)
    
    final_score = float(evaluate_sequence([float(x) for x in best]))
    if np.isfinite(final_score):
        if final_score < best_score:
            best_score = final_score
    
    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
