Optimized step function coefficient search.
Implements improved LP-alternating, L-BFGS-B on soft-max convolution objective,
and block perturbation refinement to minimize 2*n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_obj(a):
    """Compute the objective: 2*n*max(conv) / sum^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

    """Build A such that A @ b = convolve(a, b)."""
