```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FFT-based flat spectrum synthesis (alternating projections)
2. LP best-response iteration with line search
3. L-BFGS-B on log-sum-exp smoothed objective
4. Multi-scale (coarse-to-fine) construction
5. Mutational hill climbing for escape from local optima
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_score(a):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))

def build_conv_matrix(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A

def lp_best_response(a, M):
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None

def flat_spectrum_synthesis(a, n_iters=10000, deadline=None, variant='hard'):
    """
    Iterative flat spectrum synthesis with non-negativity constraint.
    Alternates between:
    - Frequency domain: reduce spectral peaks
    - Time domain: enforce non-negativity and normalization
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return a, compute_score(a)
    a = a / s
    
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    # Pre-compute index
    a_pad = np.zeros(N)
    a_pad[:n] = a
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        
        if variant == 'hard':
            # Set all non-DC magnitudes to a common target
            a_energy = np.sum(a**2)
            target_sq = (N * a_energy - A[0].real**2) / (N - 1)
            if target_sq < 0:
                target_sq = 0
            target_mag = np.sqrt(target_sq)
            phase = np.angle(A)
            A_new = np.full(N, target_mag, dtype=np.complex128)
            A_new[0] = A[0]  # Keep DC
            A_new = A_new * np.exp(1j * phase)
            # Ensure Hermitian symmetry for real signal
            A_new = (A_new + np.conj(A_new[np.mod(-np.arange(N), N)])) / 2
            A_new[0] = A[0].real  # DC is real and positive
        elif variant == 'soft':
            # Compress peaks: A_new = A * (target/|A|)^alpha where |A|>target
            a_energy = np.sum(a**2)
            target_sq = (N * a_energy - A[0].real**2) / (N - 1)
            target_mag = np.sqrt(max(0, target_sq))
            mags = np.abs(A)
            mags[0] = 1  # DC
            ratio = np.ones(N)
            mask = mags > target_mag
            ratio[mask] = target_mag / mags[mask]
            # Apply soft compression (alpha=0.5 for geometric mean)
            alpha = 0.5
            ratio = ratio ** alpha
            ratio[0] = 1.0
            A_new = A * ratio
        elif variant == 'cap':
            # Hard cap on magnitudes
            a_energy = np.sum(a**2)
            target_sq = (N * a_energy - A[0].real**2) / (N - 1)
            target_mag = np.sqrt(max(0, target_sq)) * 1.5  # Cap at 1.5x target
            mags = np.abs(A)
            A_new = A.copy()
            mask = mags > target_mag
            A_new[mask] = A_new[mask] * (target_mag / mags[mask])
        
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 0.01:
            break
        a_new = a_new / s
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

def smooth_objective(x, beta, n):
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 0.01:
        return 1e10
    x_norm = x_pos / s
    b = fast_conv(x_norm)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    log_sum = b_max + np.log(np.exp(shifted).sum()) / beta
    return float(2.0 * n * log_sum)

def smooth_refine(a, deadline, betas=None):
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    
    if betas is None:
        betas = [10, 50, 100, 500, 1000, 5000, 10000, 50000]
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        x0 = best.copy()
        x0 = np.clip(x0, 0.0001, None)
        x0 = x0 / x0.sum()
        
        try:
            res = minimize(smooth_objective, x0, args=(beta, n),
                           method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': 200, 'maxfun': 3000,
                                    'ftol': 1e-18, 'gtol': 1e-18})
            if res.fun < 1e10:
                res_x = np.clip(res.x, 0, None)
                if res_x.sum() > 0.01:
                    res_x = res_x / res_x.sum()
                    score = compute_score(res_x)
                    if score < best_score:
                        best = res_x.copy()
                        best_score = score
        except Exception:
            pass
    
    return best, best_score

def lp_iteration(a, max_iters, deadline, base_t=0.5):
    best_local = a.copy()
    best_score = compute_score(a)
    a = a.copy()
    a = np.clip(a, 0.001, None)
    a = a / a.sum()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()
        
        # Adaptive step size with line search
        best_t = 0
        best_s = best_score
        for t in [0.5, 0.3, 0.1, 0.05, 0.8, 0.2]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0, None)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            score = compute_score(a_new)
            if score < best_s:
                best_s = score
                best_t = t
        
        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.clip(a_new, 0, None)
            a_new = a_new / a_new.sum()
            best_score = best_s
            best_local = a_new.copy()
            a = a_new
        else:
            # No improvement, try very small step
            t = 0.01
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0, None)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                score = compute_score(a_new)
                if score < best_score:
                    best_score = score
                    best_local = a_new.copy()
                    a = a_new
            break
    
    return best_local, best_score
```

```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FFT-based flat spectrum synthesis (alternating projections in freq/time domains)
2. Multi-scale coarse-to-fine construction
3. LP best-response iteration with line search
4. L-BFGS-B on log-sum-exp smoothed objective with aggressive beta annealing
5. Mutational hill climbing to escape local optima
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build matrix A where (A @ x) = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None


def flat_spectrum_synthesis(a, n_iters, deadline, variant='hard', target_scale=1.0, alpha=0.5):
    """
    Iterative flat spectrum synthesis with non-negativity constraint.
    Alternates between frequency-domain peak reduction and time-domain projection.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return a, compute_score(a)
    a = a / s

    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    best_a = a.copy()
    best_score = compute_score(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_pad[:n] = a
        a_pad[n:] = 0.0
        A = np.fft.fft(a_pad)

        a_energy = np.sum(a ** 2)
        dc_sq = A[0].real ** 2
        target_sq = (N * a_energy - dc_sq) / (N - 1)
        target_sq = max(0, target_sq)
        target_mag = np.sqrt(target_sq) * target_scale

        if variant == 'hard':
            mags = np.abs(A)
            mags[0] = 0
            phase = np.angle(A)
            new_mags = np.where(mags > target_mag, target_mag, mags)
            new_mags[0] = A[0].real
            A_new = new_mags * np.exp(1j * phase)
            A_new[0] = A[0].real
        elif variant == 'soft':
            mags = np.abs(A)
            ratio = np.ones(N)
            mask = mags > target_mag
            ratio[mask] = (target_mag / mags[mask]) ** alpha
            ratio[0] = 1.0
            A_new = A * ratio
        elif variant == 'gradient':
            # Gradient step: reduce A in direction that reduces max of IFFT(|A|^2)
            b = np.fft.ifft(A * A).real
            b_max = b.max()
            # Subgradient w.r.t. A at peaks
            peak_indices = np.where(b > b_max * 0.95)[0]
            A_new = A.copy()
            for pi in peak_indices:
                # d/dA_conj[pi] of |A[pi]|^2 = A[pi]
                A_new[pi] -= 0.01 * A[pi] * (b[pi] - target_mag ** 2) / max(b_max, 1e-10)
        else:
            # Default: soft with alpha=0.5
            mags = np.abs(A)
            ratio = np.ones(N)
            mask = mags > target_mag
            ratio[mask] = (target_mag / mags[mask]) ** alpha
            ratio[0] = 1.0
            A_new = A * ratio

        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 0.01:
            break
        a_new = a_new / s

        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

            a = np.maximum(np.interp(t_new, t_old, a), 0)
            if a.sum() > 0.01:
                a = a / a.sum()

def smooth_objective(x, beta, n):
    """Log-sum-exp smoothed objective for normalized a."""
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 0.01:
        return 1e10
    x_norm = x_pos / s
    b = fast_conv(x_norm)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    log_sum = b_max + np.log(np.exp(shifted).sum()) / beta
    return float(2.0 * n * log_sum)


def smooth_refine(a, deadline):
    """Refine using log-sum-exp smoothed objective with aggressive beta annealing."""
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)

    betas = [5, 20, 50, 100, 300, 1000, 3000, 10000, 30000, 100000]

    for beta in betas:
        if time.monotonic() > deadline:
            break
        x0 = best.copy()
        x0 = np.clip(x0, 0.0001, None)
        x0 = x0 / x0.sum()

        try:
            maxiter = 300 if beta < 1000 else 200
            maxfun = 5000 if beta < 1000 else 3000
            res = minimize(smooth_objective, x0, args=(beta, n),
                           method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': maxiter, 'maxfun': maxfun,
                                    'ftol': 1e-18, 'gtol': 1e-18})
            if res.fun < 1e10:
                res_x = np.clip(res.x, 0, None)
                if res_x.sum() > 0.01:
                    res_x = res_x / res_x.sum()
                    score = compute_score(res_x)
                    if score < best_score:
                        best = res_x.copy()
                        best_score = score
                        print(f"    smooth beta={beta}: J={score:.6f}", flush=True)
        except Exception:
            pass

    return best, best_score


def lp_iteration(a, max_iters, deadline):
    """LP best-response iteration with line search step sizes."""
    best_local = a.copy()
    best_score = compute_score(a)
    a = a.copy()
    a = np.clip(a, 0.001, None)
    a = a / a.sum()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()

        # Line search over step sizes
        best_t = 0
        best_s = best_score
        step_candidates = [0.5, 0.3, 0.1, 0.7, 0.2, 0.05, 0.9]
        for t in step_candidates:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0, None)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            score = compute_score(a_new)
            if score < best_s:
                best_s = score
                best_t = t

        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.clip(a_new, 0, None)
            a_new = a_new / a_new.sum()
            best_score = best_s
            best_local = a_new.copy()
            a = a_new
        else:
            break

    return best_local, best_score

            best = a_result.copy()
            best_score = score
            print(f"  polished (LP): J={score:.6f}", flush=True)

        # One more smooth pass
        if time.monotonic() < deadline:
            a_result, score = smooth_refine(best, deadline)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  polished (smooth): J={score:.6f}", flush=True)

        print(f"Phase 5 done in {time.monotonic()-t0:.1f}s", flush=True)

    # Final validation
    if best is None:
        best = np.ones(256) / 256
        best_score = compute_score(best)

    # Ensure all values are in valid range
    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = np.ones(256) / 256

    final_score = float(evaluate_sequence([float(x) for x in best]))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
```
