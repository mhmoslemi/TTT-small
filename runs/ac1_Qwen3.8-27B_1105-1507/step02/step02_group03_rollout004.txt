    """Pressure-based multiplicative update to reduce convolution peaks."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = normalize(f.copy().astype(np.float64))
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()

        # Compute gradient of max at the argmax
        j_star = np.argmax(conv)
        grad = np.zeros(n)
        for i in range(n):
            idx = j_star - i
            if 0 <= idx < n:
                grad[i] = f[idx]

        # Also consider near-peak positions for smoother gradient
        conv_sorted = np.sort(conv)[::-1]
        for rank in range(1, min(5, len(conv_sorted))):
            if conv_sorted[rank] < 0.95 * M:
                break
            # Find position of this peak
            mask = np.isclose(conv, conv_sorted[rank], rtol=1e-10)
            positions = np.where(mask)[0]
            for jp in positions:
                if jp == j_star:
                    continue
                for i in range(n):
                    idx = jp - i
                    if 0 <= idx < n:
                        grad[i] += 0.5 * f[idx]

        if grad.max() > 0:
            grad = grad / grad.max()

        eta_eff = eta / (1.0 + 0.005 * it)
        f_new = f * np.exp(-eta_eff * grad)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 100:
                # Perturb
                f_pert = f * (1.0 + 0.1 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    stall = 0
                else:
                    break

    return best_f, best_score


def lp_refine(f, time_budget, t0=0.4, lp_time_limit=5):
    """LP-based iterative refinement with all convolution constraints."""
    n = len(f)
    f = normalize(f.copy().astype(np.float64))
    best_f = f.copy()
    best_score = compute_score(f)
    t = t0
    start = time.monotonic()
    iter_count = 0

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 10:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs',
                          options={'presolve': True, 'time_limit': lp_time_limit})
        except Exception:
            break

        if not res.success or res.x is None:
            # Try slightly relaxed
            try:
                res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M * 1.001),
                              bounds=bounds, method='highs',
                              options={'presolve': True, 'time_limit': lp_time_limit})
            except Exception:
                break
            if not res.success or res.x is None:
                break

        g = np.clip(res.x, 0, None)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Line search over step sizes
        improved = False
        candidates = [min(1.0, t * 2.0), t, t * 0.5, t * 0.25, t * 0.1]
        for step in candidates:
            if step <= 0 or step >= 1:
                continue
            f_new = (1.0 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_score = compute_score(f_new)
            if new_score < best_score - 1e-13:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                t = min(1.0, t * 1.05)
                improved = True
                iter_count += 1
                break

        if not improved:
            t *= 0.65
            if t < 0.002:
                # Try perturbation to escape
                f_pert = f + 0.02 * np.random.default_rng(iter_count).normal(0, 1, n)
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    t = 0.1
                else:
                    break

    return best_f, best_score


def coordinate_descent(f, n_sweeps=20, rng=None):
    """Coordinate descent for final polishing."""
    if rng is None:
        rng = np.random.default_rng(42)
    f = normalize(f.copy().astype(np.float64))
    n = len(f)
    best_score = compute_score(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-12:
                continue
            best_val = original
            best_local = best_score
            for factor in [0.3, 0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.0/0.5, 1.0/0.3, 0.1, 2.0]:
                f[i] = max(0.0, original * factor)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                conv = np.convolve(f, f)
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
        f = normalize(f)

    return f, best_score


def soft_lp_refine(f, time_budget, t0=0.3, alpha=0.02):
    """LP refinement with softly weighted constraints (penalize peaks more)."""
    n = len(f)
    f = normalize(f.copy().astype(np.float64))
    best_f = f.copy()
    best_score = compute_score(f)
    t = t0
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        if time.monotonic() - start > time_budget - 15:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        # Weight constraints: peaks get tighter bounds
        weights = np.ones(2 * n - 1)
        # Make top peaks have tighter bounds
        sorted_idx = np.argsort(conv)[::-1]
        for rank, idx in enumerate(sorted_idx[:min(10, 2*n-1)]):
            weights[idx] = 1.0 - alpha * (1.0 - rank / min(10, 2*n-1))

        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = M * weights
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 5})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        g = np.clip(res.x, 0, None)
        if g.sum() < 1e-15:
            break
        g = g / g.sum()

        improved = False
        for step in [min(1.0, t), t * 0.5, t * 0.25]:
            f_new = (1.0 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_score = compute_score(f_new)
            if new_score < best_score - 1e-13:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                t = min(1.0, t * 1.03)
                improved = True
                break

        if not improved:
            t *= 0.6
            if t < 0.002:
                break

    return best_f, best_score


def upsample_sequence(f_small, n_large):
    """Upsample a sequence to a larger length using interpolation."""
    n_small = len(f_small)
    x_small = np.linspace(0, 1, n_small)
    x_large = np.linspace(0, 1, n_large)
    f_large = np.interp(x_large, x_small, f_small)
    f_large = np.clip(f_large, 0.001, None)
    return normalize(f_large)


def diverse_init(n, rng):
    """Generate a diverse initial sequence."""
    r = rng.random()
    if r < 0.25:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.15, 0.4)
        f = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    elif r < 0.5:
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    elif r < 0.7:
        f = np.abs(np.sin(np.pi * np.arange(n) / n * rng.uniform(1.5, 3.0))) + 0.1
    elif r < 0.85:
        f = np.ones(n)
        for _ in range(8):
            center = int(rng.integers(0, n))
            width = max(3, int(n // 32))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
            f *= (1.0 + 0.3 * rng.standard_normal() * kernel)
        f = np.abs(f) + 0.05
    else:
        # Smooth random via FFT
        coeffs = rng.normal(0, 1, n) + 1j * rng.normal(0, 1, n)
        coeffs[0] = np.abs(coeffs[0])  # ensure positive DC
        f = np.fft.irfft(coeffs, n)
        f = np.abs(f) + 0.05

    f = np.clip(f, 0.01, None)
    f = normalize(f)
    # Add small asymmetry
    f = f * (1.0 + 0.03 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: multi-strategy optimization combining LP refinement,
    spectral flattening, pressure updates, and coordinate descent.
    Explores multiple sequence lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0

    best_overall = None
    best_score_overall = np.inf

    # Try parent sequence
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except (NameError, Exception):
        parent = None

    if parent is not None:
        parent = np.clip(parent, 0, None)
        if parent.sum() > 0.01:
            parent = normalize(parent)
            parent_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}", flush=True)
            best_overall = parent.copy()
            best_score_overall = parent_score

    # Define lengths to explore
    lengths = [512]
    if parent is not None and len(parent) != 512:
        lengths.insert(0, len(parent))
    lengths.extend([1024, 2048])

    # Remove duplicates while preserving order
    seen = set()
    unique_lengths = []
    for l in lengths:
        if l not in seen:
            seen.add(l)
            unique_lengths.append(l)
    lengths = unique_lengths

    print(f"Lengths to try: {lengths}, budget: {budget_s}s, target: 1.5027", flush=True)

    # Phase 1: Quick optimization of parent at its native length
    if parent is not None and len(parent) <= 1024:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            n_p = len(parent)
            print(f"\n=== Phase 1: Quick refine parent (n={n_p}, remaining={remaining:.0f}s) ===", flush=True)

            # Light spectral + pressure
            f1, s1 = spectral_flatten(parent, n_iter=100, rng=rng)
            print(f"  Spectral: {s1:.6f}", flush=True)
            if s1 < best_score_overall:
                best_score_overall = s1
                best_overall = f1.copy()

            f2, s2 = pressure_update(f1, n_iter=300, eta=0.08, rng=rng)
            print(f"  Pressure: {s2:.6f}", flush=True)
            if s2 < best_score_overall:
                best_score_overall = s2
                best_overall = f2.copy()

            # Short LP
            remaining2 = deadline - time.monotonic()
            if remaining2 > 30:
                f3, s3 = lp_refine(f2, min(60, remaining2 * 0.5), t0=0.4, lp_time_limit=3)
                print(f"  LP: {s3:.6f}", flush=True)
                if s3 < best_score_overall:
                    best_score_overall = s3
                    best_overall = f3.copy()

    # Phase 2: Extended LP refinement on best found so far
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            n_best = len(best_overall)
            t_lp = min(remaining * 0.4, 300)
            print(f"\n=== Phase 2: Extended LP (n={n_best}, budget={t_lp:.0f}s) ===", flush=True)
            f_lp, s_lp = lp_refine(best_overall, t_lp, t0=0.3, lp_time_limit=5)
            print(f"  LP result: {s_lp:.6f}", flush=True)
            if s_lp < best_score_overall:
                best_score_overall = s_lp
                best_overall = f_lp.copy()

    # Phase 3: Try longer sequences
    for n_target in [1024, 2048]:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        print(f"\n=== Phase 3: Trying n={n_target} (remaining={remaining:.0f}s) ===", flush=True)

        # Create initial sequence
        if best_overall is not None and len(best_overall) <= n_target:
            f_init = upsample_sequence(best_overall, n_target)
        else:
            f_init = diverse_init(n_target, rng)

        # Spectral + pressure
        t_spec = min(40, remaining * 0.1)
        f_s, s_s = spectral_flatten(f_init, n_iter=int(t_spec * 30), rng=rng)
        print(f"  Spectral (n={n_target}): {s_s:.6f}", flush=True)
        if s_s < best_score_overall:
            best_score_overall = s_s
            best_overall = f_s.copy()

        f_p, s_p = pressure_update(f_s, n_iter=int(t_spec * 30), eta=0.06, rng=rng)
        print(f"  Pressure (n={n_target}): {s_p:.6f}", flush=True)
        if s_p < best_score_overall:
            best_score_overall = s_p
            best_overall = f_p.copy()

        # LP refinement
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            t_lp2 = min(remaining2 * 0.35, 180)
            f_l, s_l = lp_refine(f_p, t_lp2, t0=0.3, lp_time_limit=5)
            print(f"  LP (n={n_target}): {s_l:.6f}", flush=True)
            if s_l < best_score_overall:
                best_score_overall = s_l
                best_overall = f_l.copy()

    # Phase 4: Soft LP refinement (different constraint weighting)
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            n_best = len(best_overall)
            t_soft = min(remaining * 0.2, 120)
            print(f"\n=== Phase 4: Soft LP (n={n_best}, budget={t_soft:.0f}s) ===", flush=True)
            f_soft, s_soft = soft_lp_refine(best_overall, t_soft, t0=0.3, alpha=0.03)
            print(f"  Soft LP: {s_soft:.6f}", flush=True)
            if s_soft < best_score_overall:
                best_score_overall = s_soft
                best_overall = f_soft.copy()

    # Phase 5: Coordinate descent polishing
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15:
            n_cd_sweeps = max(10, int(min(30, remaining * 0.4)))
            print(f"\n=== Phase 5: CD polishing (sweeps={n_cd_sweeps}) ===", flush=True)
            f_cd, s_cd = coordinate_descent(best_overall, n_sweeps=n_cd_sweeps, rng=rng)
            print(f"  CD: {s_cd:.6f}", flush=True)
            if s_cd < best_score_overall:
                best_score_overall = s_cd
                best_overall = f_cd.copy()

    # Phase 6: Final LP pass on best
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            n_best = len(best_overall)
            t_final = min(remaining * 0.7, 200)
            print(f"\n=== Phase 6: Final LP (n={n_best}, budget={t_final:.0f}s) ===", flush=True)
            f_fin, s_fin = lp_refine(best_overall, t_final, t0=0.2, lp_time_limit=5)
            print(f"  Final LP: {s_fin:.6f}", flush=True)
            if s_fin < best_score_overall:
                best_score_overall = s_fin
                best_overall = f_fin.copy()

    # Phase 7: Final CD polish
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            f_fin2, s_fin2 = coordinate_descent(best_overall, n_sweeps=10, rng=rng)
            print(f"\n=== Phase 7: Final CD: {s_fin2:.6f} ===", flush=True)
            if s_fin2 < best_score_overall:
                best_score_overall = s_fin2
                best_overall = f_fin2.copy()

    # Verify with evaluate_sequence
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}", flush=True)
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}", flush=True)
        print(f"Target: 1.5027, Previous best: 1.512006", flush=True)
        print(f"{'='*50}", flush=True)
        return final_list

    # Fallback
    print("Fallback: returning uniform sequence", flush=True)
    return [1.0] * 256
