    n = len(a)
    # Variables: b_0, ..., b_{n-1}, M (total n+1 variables)
    # Objective: minimize M -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    """Compute the objective: 2*n*max(a*a)/sum(a)^2"""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def solve_lp_min_M(a):
    """Solve LP: minimize M s.t. (a*b)_k <= M, sum(b)=1, b>=0, M>=0"""
    n = len(a)
    # Variables: [b_0, ..., b_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # Build constraint matrix A_ub: (2n-1) x (n+1)
    # (a*b)_k - M <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    
    for k in range(2 * n - 1):
        A_ub[k, -1] = -1.0  # -M
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, k - j] = a[j]
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n], float(res.x[-1])
    return None, None

def normalize(a):
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s
