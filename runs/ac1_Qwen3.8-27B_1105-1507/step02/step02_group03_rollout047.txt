from scipy.optimize import linprog
import time

def compute_score(f):
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (s * s)

def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s

def build_conv_matrix(f):
    n = len(f)
    # A @ g = convolve(f, g, mode='full')
    # A has shape (2n-1, n)
    # A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = f[j]
    return A

def lp_direction(f, M):
    """Find g that maximizes sum(g) subject to (f*g)[k] <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2 * n - 1, M)

                  options={'presolve': True, 'time_limit': 5})
    if res.success and res.x is not None:
        return np.clip(res.x, 0, None)
    return None

"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute score for sequence: 2*n*max(conv)/sum^2."""
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (s * s)


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def build_conv_matrix(f):
    """Build matrix A such that A @ g = np.convolve(f, g, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            A[k, i] = f[k - i]
    return A


def lp_direction(f, M, time_limit=3):
    """Find g maximizing sum(g) subject to (f*g)[k] <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None

        f2, s2 = pressure_update(f1, iters_press, 0.12, rng, n_peaks=3)
        print(f"  After pressure ({iters_press} iters): {s2:.6f}")
        if s2 < best_score_overall:
            best_score_overall = s2
            best_overall = f2.copy()

        # Stage 3: LP refinement (main workhorse)
        remaining = deadline - time.monotonic()
        t_lp1 = min(remaining * 0.45, 500)
        if t_lp1 > 30:
            print(f"  LP phase 1: budget={t_lp1:.0f}s")
            f3, s3 = lp_refine_improved(f2, t_lp1, rng, t0=0.5)
            print(f"  After LP1: {s3:.6f}")
            if s3 < best_score_overall:
                best_score_overall = s3
                best_overall = f3.copy()

        # Stage 4: Coordinate descent
        remaining = deadline - time.monotonic()
        if remaining > 30:
            n_sweeps = max(3, min(15, int(remaining * 0.04)))
            f4, s4 = coordinate_descent(best_overall, n_sweeps, rng)
            print(f"  After CD ({n_sweeps} sweeps): {s4:.6f}")
            if s4 < best_score_overall:
                best_score_overall = s4
                best_overall = f4.copy()

        # Stage 5: Second LP round with smaller step
        remaining = deadline - time.monotonic()
        t_lp2 = min(remaining * 0.3, 200)
        if t_lp2 > 20:
            print(f"  LP phase 2: budget={t_lp2:.0f}s")
            f5, s5 = lp_refine_improved(best_overall, t_lp2, rng, t0=0.25)
            print(f"  After LP2: {s5:.6f}")
            if s5 < best_score_overall:
                best_score_overall = s5
                best_overall = f5.copy()

        # Stage 6: Final pressure + CD
        remaining = deadline - time.monotonic()
        if remaining > 20:
            f6, s6 = pressure_update(best_overall, 100, 0.05, rng, n_peaks=5)
            if s6 < best_score_overall:
                best_score_overall = s6
                best_overall = f6.copy()
                print(f"  After final pressure: {s6:.6f}")

            remaining = deadline - time.monotonic()
            if remaining > 15:
                f7, s7 = coordinate_descent(best_overall, 5, rng)
                if s7 < best_score_overall:
                    best_score_overall = s7
                    best_overall = f7.copy()
                    print(f"  After final CD: {s7:.6f}")

        print(f"  Best so far: {best_score_overall:.6f}")

    # --- Alternative: try a different initialization strategy for best length ---
    remaining = deadline - time.monotonic()
    if remaining > 180 and best_overall is not None:
        n_best = len(best_overall)
        print(f"\n--- Alternative init phase (remaining={remaining:.0f}s) ---")

        # Try multi-scale init
        f_alt = multi_scale_init(n_best, n_levels=4, rng=rng)
        f_alt, s_alt = spectral_flatten(f_alt, 100, rng)
        f_alt, s_alt = pressure_update(f_alt, 150, 0.1, rng, n_peaks=3)
        print(f"  Multi-scale init + opt: {s_alt:.6f}")

        if s_alt < best_score_overall:
            best_score_overall = s_alt
            best_overall = f_alt.copy()
            remaining = deadline - time.monotonic()
            if remaining > 60:
                f_alt, s_alt = lp_refine_improved(f_alt, remaining * 0.6, rng, t0=0.4)
                print(f"  After LP on alt: {s_alt:.6f}")
                if s_alt < best_score_overall:
                    best_score_overall = s_alt
                    best_overall = f_alt.copy()

    # --- Final polishing ---
    remaining = deadline - time.monotonic()
    if remaining > 30 and best_overall is not None:
        print(f"\n--- Final polishing (remaining={remaining:.0f}s) ---")

        # Very fine LP
        t_final_lp = min(remaining * 0.5, 120)
        if t_final_lp > 15:
            f_fin, s_fin = lp_refine_improved(best_overall, t_final_lp, rng, t0=0.15)
            print(f"  After final LP: {s_fin:.6f}")
            if s_fin < best_score_overall:
                best_score_overall = s_fin
                best_overall = f_fin.copy()

        # Final CD
        remaining = deadline - time.monotonic()
        if remaining > 15:
            f_fin2, s_fin2 = coordinate_descent(best_overall, 5, rng)
            print(f"  After final CD: {s_fin2:.6f}")
            if s_fin2 < best_score_overall:
                best_score_overall = s_fin2
                best_overall = f_fin2.copy()

    # Verify and return
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*50}")
        return final_list

    # Fallback
    print("Fallback: uniform sequence")
    return [1.0] * 256
