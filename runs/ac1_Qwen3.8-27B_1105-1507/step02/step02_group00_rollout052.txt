```python
def solve_lp_given_a(a, M=None):
    """
    Given fixed a (sum=1), solve:
    minimize t
    subject to: sum_j a[j]*g[k-j] <= t, k=0,...,2n-2
                sum(g) = 1
                g >= 0
    
    Returns optimal g and t.
    """
    n = len(a)
    # Variables: g[0],...,g[n-1], t
    # Objective: minimize t (last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each k: sum_j a[j]*g[k-j] - t <= 0
    # A_ub has 2n-1 rows, n+1 columns
    A_ub = np.zeros((2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    
    for k in range(2*n-1):
        for j in range(n):
            # g index in the sum: we need a[j]*g[k-j]
            # g[k-j] is valid if 0 <= k-j <= n-1
            gj_idx = k - j
            if 0 <= gj_idx < n:
                A_ub[k, gj_idx] = a[j]
        A_ub[k, -1] = -1.0  # -t
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]  # g >= 0, t free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    if res.status == 0:
```

So for a given k, the coefficients of g[j] in this constraint are:
`a[k-j]` for j such that `0 <= k-j < n`, i.e., `max(0, k-n+1) <= j <= min(n-1, k)`.

The matrix A_ub[k, j] = a[k-j] if `0 <= k-j < n`, else 0.

```python
def build_lp_matrix(a):
    n = len(a)
    # A_ub is (2n-1) x (n+1): first n columns for g, last column for -t
    A = np.zeros((2*n-1, n+1))
    A[:, -1] = -1.0  # -t column
    
    # For column j (variable g[j]), the non-zero entries are in rows k where 0 <= k-j < n
    # i.e., j <= k <= j+n-1, and 0 <= k <= 2n-2
    # So k ranges from j to min(j+n-1, 2n-2)
    # A[k, j] = a[k-j]
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2*n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    
    return A
```

```python
"""
Minimizes 2*n*max(autocorr(a))/(sum(a))^2 using:
1. FFT-accelerated multiplicative weights (mirror descent)
2. LP-based alternating optimization  
3. Subgradient polishing with adaptive step size
Tries multiple sequence lengths from 300 to 2000.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import fft, ifft
import time

def _next_pow2(n):
    """Next power of 2 >= n."""
    p = 1
    while p < n:
        p *= 2
    return p

def autocorr(a):
    """Aperiodic autocorrelation via FFT. Input: 1D array. Output: array of length 2n-1."""
    n = len(a)
    size = _next_pow2(2 * n - 1)
    A = fft(a.astype(np.float64), size)
    c = ifft(A * A).real
    return c[:2*n-1]

def obj_value(a):
    """Compute 2*n*max(autocorr(a))/sum(a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = autocorr(a)
    return 2.0 * n * np.max(c) / (s * s)

def mw_optimize(a, max_iter=50000, eta_init=1.0, eta_decay=0.999, tol=1e-8, verbose=False):
    """Multiplicative weights (mirror descent) optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = obj_value(a)
    
    eta = eta_init
    for it in range(max_iter):
        c = autocorr(a)
        max_c = np.max(c)
        
        # Find active lags (within 10% of max)
        tol_active = max_c * 0.9
        active = c >= tol_active
        n_active = np.sum(active)
        if n_active == 0:
            active = c >= max_c - 1e-15
            n_active = max(1, np.sum(active))
        
        # Compute subgradient via FFT
        size = _next_pow2(2*n - 1 + n)  # enough for cross-correlation
        A_fft = fft(a, size)
        I_fft = fft(active.astype(np.float64), size)
        cross = ifft(I_fft * np.conj(A_fft)).real[:n]
        grad = (2.0 / n_active) * cross
        
        # Multiplicative update
        log_a = np.log(a + 1e-30) - eta * grad
        log_a -= np.max(log_a)  # for numerical stability
        a_new = np.exp(log_a)
        a_new /= np.sum(a_new)
        
        obj_new = obj_value(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        eta *= eta_decay
        
        if verbose and (it + 1) % 5000 == 0:
            print(f"    MW it={it+1}: obj={best_obj:.6f}, eta={eta:.4f}")
    
    return best_a, best_obj

def lp_one_step(a):
    """One LP step: given a (sum=1), find g minimizing max_k conv(a,g)[k] s.t. sum(g)=1, g>=0."""
    n = len(a)
    # Build constraint matrix
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, -1] = -1.0
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        A_ub[k, j_start:j_end+1] = a[k-j_end:k-j_start+1][::-1]
    
    # Objective: minimize t (last variable)
    c_obj = np.zeros(n+1)
    c_obj[-1] = 1.0
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g >= 0, t >= 0
    bounds = [(0, None)] * (n+1)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=np.zeros(2*n-1),
                  A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n]
    return None

def lp_alternate_v2(a, max_iter=300, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = obj_value(a)
    
    for it in range(max_iter):
        g = lp_one_step(a)
        if g is None:
            break
        
        # Adaptive step size: alpha = 2/(it+2)
        alpha = 2.0 / (it + 2)
        a_new = (1 - alpha) * a + alpha * g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            a = g
        else:
            a = a_new / s
        
        obj_new = obj_value(a)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()
        
        if verbose and (it + 1) % 50 == 0:
            print(f"    LP it={it+1}: obj={best_obj:.6f}")
    
    return best_a, best_obj

def sg_polish(a, n_steps=10000, verbose=False):
    """Subgradient polishing with decreasing step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = obj_value(a)
    
    for step in range(n_steps):
        c = autocorr(a)
        max_c = np.max(c)
        # Adaptive tolerance for active set
        tol = max_c * (0.01 + 0.09 * step / n_steps)
        active = c >= max_c - tol
        n_active = np.sum(active)
        if n_active == 0:
            active = c >= max_c - 1e-10
            n_active = max(1, np.sum(active))
        
        # Subgradient via FFT
        size = _next_pow2((2*n-1) + n)
        A_fft = fft(a, size)
        I_fft = fft(active.astype(np.float64), size)
        cross = ifft(I_fft * np.conj(A_fft)).real[:n]
        grad = (2.0 / n_active) * cross
        
        # Step size: 1/sqrt(t)
        eta = 1.0 / np.sqrt(step + 1)
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = obj_value(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and (step + 1) % 2000 == 0:
            print(f"    SG step={step+1}: obj={best_obj:.6f}")
    
    return best_a, best_obj

def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-scale convex optimization for minimum-peak autocorrelation sequences.

    Tries sequence lengths 300-2000.
    """
    start_time = time.monotonic()

        a = a / np.sum(a)
```
