import numpy as np
from scipy.optimize import linprog
import time


def compute_score_normalized(f):
    """Compute score for normalized sequence (sum=1): 2*n*max(conv)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def build_conv_matrix(f):
    """Build matrix A such that A @ g = np.convolve(f, g, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def lp_solve_direction(f, M, time_limit=2.0):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0. Returns (g, success, dual_vars)."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None, False, None

    if not res.success or res.x is None:
        return None, False, None

    g = np.clip(res.x, 0, None)
    dual = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else None
    return g, True, dual

def peak_shave(f, time_budget, rng=None):
    """Directly target the maximum convolution entry by reducing contributing elements."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        j_star = np.argmax(conv)

        # Find the top contributing pairs to the peak
        # conv[j_star] = sum_i f[i] * f[j_star - i] (where indices are valid)
        contributions = np.zeros(n)
        for i in range(n):
            idx = j_star - i
            if 0 <= idx < n:
                contributions[i] = f[i] * f[idx]

        # Also find second and third peaks
        conv_copy = conv.copy()
        top_positions = []
        for _ in range(min(5, len(conv))):
            pos = np.argmax(conv_copy)
            top_positions.append(pos)
            conv_copy[pos] = -np.inf

        # For each top peak, add its contribution pattern
        for pos in top_positions[1:]:
            for i in range(n):
                idx = pos - i
                if 0 <= idx < n:
                    contributions[i] += 0.7 * f[i] * f[idx]

        # Multiply elements by exp(-eta * normalized_contribution)
        max_contrib = contributions.max()
        if max_contrib < 1e-15:
            break
        contributions_norm = contributions / max_contrib

        eta = 0.3
        f_new = f * np.exp(-eta * contributions_norm)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        sc_new = compute_score_normalized(f_new)
        if sc_new < best_score - 1e-12:
            best_score = sc_new
            best_f = f_new.copy()
            f = f_new
        else:
            # Try smaller eta
            eta = 0.1
            f_new = f * np.exp(-eta * contributions_norm)
            f_new = np.clip(f_new, 1e-15, None)
            f_new = normalize(f_new)
            sc_new = compute_score_normalized(f_new)
            if sc_new < best_score - 1e-12:
                best_score = sc_new
                best_f = f_new.copy()
                f = f_new
            else:
                # Perturb and try again
                f_pert = f * (1 + 0.05 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score_normalized(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score_normalized(f)
                else:
                    break

    return best_f, best_score


def spectral_equalize(f, time_budget, rng=None):
    """Iterative spectral equalization with relaxed projection."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    start = time.monotonic()
    n_fft = 1 << (2 * n - 1).bit_length()

    iteration = 0
    stall = 0
    while time.monotonic() - start < time_budget:
        iteration += 1
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)

        n_bins = len(mag)
        # Target: equal magnitude for all non-DC bins
        # Current energy: sum(f^2) = (1/n_fft) * sum(|F|^2)
        energy = np.sum(f ** 2)
        target_energy = energy * 0.99  # Slightly reduce energy (spreads mass)
        total_power = target_energy * n_fft
        # DC bin is fixed at 1 (sum = 1)
        dc_power = 1.0
        remaining_power = total_power - dc_power
        n_non_dc = n_bins - 1
        target_mag = np.sqrt(remaining_power / max(1, n_non_dc))

        # Blend: keep some of current magnitude, add target
        alpha = 0.3 + 0.4 * min(1.0, iteration / 100.0)  # Increase aggressiveness
        mag_new = (1 - alpha) * mag + alpha * target_mag
        mag_new[0] = 1.0

        # Random phase perturbation to escape local minima
        if iteration % 100 == 50:
            phase = phase + rng.normal(0, 0.1, len(phase))

        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            f_new = f * (1 + 0.05 * rng.normal(0, 1, n))
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
        f_new = f_new / max(s, 1e-15)

        sc = compute_score_normalized(f_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 300:
                # Big jump

                f = np.clip(f, 0, None)
                f = normalize(f)
                stall = 0

    return best_f, best_score


def dual_guided_update(f, time_budget, rng=None):
    """Use LP dual variables to construct a better update direction."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()

        # Solve LP and get dual variables
        g, success, dual = lp_solve_direction(f, M, time_limit=2.0)
        if not success or g is None:
            break
        g = g / max(g.sum(), 1e-15)

        # Construct direction from dual: elements with high dual pressure should be reduced more
        if dual is not None and len(dual) > 0:
            # dual[k] is the shadow price for constraint k (conv position k)
            # High dual means that constraint is "expensive" to relax
            # We want to reduce f at positions that contribute to high-dual constraints
            pressure = np.zeros(n)
            for k in range(2 * n - 1):
                if dual[k] > 1e-10:
                    for i in range(n):
                        idx = k - i
                        if 0 <= idx < n:
                            pressure[i] += dual[k] * f[idx]
            if pressure.max() > 0:
                pressure = pressure / pressure.max()
                # Modify g to reduce high-pressure elements more
                g_modified = g * np.exp(-0.3 * pressure)
                g_modified = g_modified / max(g_modified.sum(), 1e-15)
                g = g_modified

        # Line search
        best_t = 0
        best_local = best_score
        best_f_local = f.copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            f_new = np.clip((1 - t) * f + t * g, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            sc = compute_score_normalized(f_new)
            if sc < best_local:
                best_local = sc
                best_t = t
                best_f_local = f_new

        if best_t > 0 and best_local < best_score - 1e-12:
            f = best_f_local
            best_f = f.copy()
            best_score = best_local
        else:
            break

    return best_f, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    r = rng.random()
    if r < 0.2:
        # Near-uniform with small modulation
        f = np.ones(n)
        f += 0.1 * np.sin(2 * np.pi * rng.uniform(1, 5) * np.arange(n) / n)
        f += 0.05 * rng.normal(0, 1, n)
    elif r < 0.4:
        # Cosine window variants
        freq = rng.uniform(0.5, 3.0)
        f = 0.7 + 0.3 * np.cos(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
    elif r < 0.6:
        # Product of smooth bumps
        f = np.ones(n)
        for _ in range(rng.integers(2, 6)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(-0.3, 0.3)
            f *= (1 + amp * np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2))
    elif r < 0.8:
        # Low-pass filtered random
        f = rng.random(n)
        # Apply simple blur
        kernel_size = max(3, n // 20)
        kernel = np.ones(kernel_size) / kernel_size
        f = np.convolve(f, kernel, mode='same')
        f = f / f.max()
    else:
        # Ramp + noise
        f = np.linspace(0.5, 1.5, n) * rng.uniform(0.5, 1.0, n)
        f += 0.1 * rng.normal(0, 1, n)

    f = np.clip(f, 0.01, None)
    f = normalize(f)
    return f


def coordinate_descent_fast(f, n_sweeps=5, rng=None):
    """Fast coordinate descent - only try a few values per coordinate."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    best_score = compute_score_normalized(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-10:
                f[i] = 0.0
                continue
            best_val = original
            best_local = best_score
            for factor in [0.5, 0.7, 0.85, 1.0 / 0.85, 1.0 / 0.7, 2.0]:
                f[i] = max(0.0, original * factor)
                conv = np.convolve(f, f)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    break
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
        f = np.clip(f, 0, None)
        f = normalize(f)

    return f, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: aggressive LP refinement + peak shaving + spectral equalization
    + dual-guided updates + coordinate descent, with multi-restart strategy.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0

    best_overall = None
    best_score_overall = np.inf

    # Try parent sequence
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None

    if parent is not None:
        parent = np.clip(parent, 0, None)
        if parent.sum() > 0.01:
            parent = normalize(parent)
            parent_score = compute_score_normalized(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
            best_overall = parent.copy()
            best_score_overall = parent_score

    n_parent = len(parent) if parent is not None else 512

    # === Phase 1: Aggressive LP refinement on parent (or init) ===
    print("\n=== Phase 1: LP refinement on main sequence ===")
    t_start = time.monotonic()
    t_lp1 = min(400, (deadline - t_start) * 0.35)

    if parent is not None:
        f1, s1 = lp_refine_aggressive(parent.copy(), t_lp1, rng=rng)
    else:
        f1 = diverse_init(512, rng)
        f1, s1 = lp_refine_aggressive(f1, t_lp1, rng=rng)

    print(f"  LP refine 1: score={s1:.8f}, time={time.monotonic()-t_start:.1f}s")
    if s1 < best_score_overall:
        best_score_overall = s1
        best_overall = f1.copy()

    # === Phase 2: Peak shaving ===
    print("\n=== Phase 2: Peak shaving ===")
    t_start = time.monotonic()
    t_ps = min(100, (deadline - t_start) * 0.1)
    f2, s2 = peak_shave(f1, t_ps, rng=rng)
    print(f"  Peak shave: score={s2:.8f}, time={time.monotonic()-t_start:.1f}s")
    if s2 < best_score_overall:
        best_score_overall = s2
        best_overall = f2.copy()
        f1 = f2

    # === Phase 3: Spectral equalization ===
    print("\n=== Phase 3: Spectral equalization ===")
    t_start = time.monotonic()
    t_spec = min(120, (deadline - t_start) * 0.15)
    f3, s3 = spectral_equalize(f1, t_spec, rng=rng)
    print(f"  Spectral: score={s3:.8f}, time={time.monotonic()-t_start:.1f}s")
    if s3 < best_score_overall:
        best_score_overall = s3
        best_overall = f3.copy()

    # === Phase 4: More LP refinement from best so far ===
    print("\n=== Phase 4: Second LP refinement ===")
    t_start = time.monotonic()
    t_lp2 = min(300, (deadline - t_start) * 0.25)
    f4, s4 = lp_refine_aggressive(best_overall.copy(), t_lp2, rng=rng)
    print(f"  LP refine 2: score={s4:.8f}, time={time.monotonic()-t_start:.1f}s")
    if s4 < best_score_overall:
        best_score_overall = s4
        best_overall = f4.copy()

    # === Phase 5: Dual-guided update ===
    print("\n=== Phase 5: Dual-guided update ===")
    t_start = time.monotonic()
    t_dual = min(120, (deadline - t_start) * 0.15)
    f5, s5 = dual_guided_update(best_overall.copy(), t_dual, rng=rng)
    print(f"  Dual-guided: score={s5:.8f}, time={time.monotonic()-t_start:.1f}s")
    if s5 < best_score_overall:
        best_score_overall = s5
        best_overall = f5.copy()

    # === Phase 6: Try different length (n_parent * 2 or other) ===
    print("\n=== Phase 6: Different sequence length ===")
    t_start = time.monotonic()
    remaining = deadline - t_start
    if remaining > 60:
        # Try a different length
        n_alt = n_parent * 2 if n_parent < 1024 else n_parent // 2
        n_alt = max(256, min(1024, n_alt))
        if n_alt != n_parent:
            f_alt = diverse_init(n_alt, rng)
            t_lp_alt = min(150, remaining * 0.4)
            f_alt, s_alt = lp_refine_aggressive(f_alt, t_lp_alt, rng=rng)
            print(f"  Alt length n={n_alt}: score={s_alt:.8f}")
            if s_alt < best_score_overall:
                best_score_overall = s_alt
                best_overall = f_alt.copy()

    # === Phase 7: Restart with different init on best length ===
    print("\n=== Phase 7: Multi-restart ===")
    n_best = len(best_overall)
    t_start = time.monotonic()
    remaining = deadline - t_start
    n_restarts = max(1, int(remaining / 80))
    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        f_r = diverse_init(n_best, rng)
        # Quick spectral + LP
        f_r, _ = spectral_equalize(f_r, 20, rng=rng)
        t_lr = min(50, (deadline - time.monotonic()) * 0.7)
        if t_lr > 10:
            f_r, s_r = lp_refine_aggressive(f_r, t_lr, rng=rng)
            if s_r < best_score_overall:
                best_score_overall = s_r
                best_overall = f_r.copy()
                print(f"  Restart {r+1}: score={s_r:.8f}")

    # === Phase 8: Final polishing ===
    print("\n=== Phase 8: Final polishing ===")
    t_start = time.monotonic()
    remaining = deadline - t_start
    if remaining > 15 and best_overall is not None:
        # Coordinate descent
        n_sweeps = max(3, int(min(10, remaining * 0.3)))
        f_cd, s_cd = coordinate_descent_fast(best_overall.copy(), n_sweeps=n_sweeps, rng=rng)
        if s_cd < best_score_overall:
            best_score_overall = s_cd
            best_overall = f_cd.copy()
            print(f"  CD: score={s_cd:.8f}")

        # One more short LP
        remaining = deadline - time.monotonic()
        if remaining > 15:
            f_final, s_final = lp_refine_aggressive(best_overall.copy(), min(60, remaining), rng=rng)
            if s_final < best_score_overall:
                best_score_overall = s_final
                best_overall = f_final.copy()
                print(f"  Final LP: score={s_final:.8f}")

    # Final peak shave
    remaining = deadline - time.monotonic()
    if remaining > 10 and best_overall is not None:
        f_ps, s_ps = peak_shave(best_overall.copy(), min(30, remaining), rng=rng)
        if s_ps < best_score_overall:
            best_score_overall = s_ps
            best_overall = f_ps.copy()
            print(f"  Final peak shave: score={s_ps:.8f}")

    # Verify with evaluate_sequence
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.8f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    print("FALLBACK!")
    f_fb = np.ones(512) / 512
    return [float(x) for x in f_fb]
