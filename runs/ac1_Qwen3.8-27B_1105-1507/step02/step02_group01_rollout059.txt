- Equivalently, we want `|FFT(a)|^2` to be as flat as possible (minimize the peak of its inverse FFT).
- This is like a "flat spectrum" design problem.

For non-negative sequences, the optimal flat spectrum would be `|A(f)|^2 = constant`, which means `A(f)` has constant magnitude. The inverse FFT of a constant is a delta function, not a valid sequence. So we need a compromise.

A practical spectral approach:
1. Start with an initial sequence.
2. Compute its FFT.
3. Try to flatten the power spectrum while maintaining non-negativity of the time-domain signal.
4. This can be done with iterative projection: project onto the set of sequences with flat-ish spectrum, then project onto the non-negative cone.

This is similar to the "Gerchberg-Saxton" algorithm in phase retrieval.

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def normalize(a):
    """Normalize to sum=1, maintaining non-negativity."""
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_A_matrix(a):
    """Build the constraint matrix for the LP."""
    n = len(a)
    # (a * b)[k] = sum_j a[j] * b[k-j]
    # We want (a * b)[k] <= M for all k
    # This is a linear constraint in b: sum_j a[j] * b[k-j] <= M
    # For each k, the coefficients of b are: A[k, j] = a[k-j] if 0 <= k-j < n else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    return A

def lp_alternate(a, M=None, max_iters=50, time_limit=20.0):
    """LP-based alternating optimization."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if M is None:
        conv = np.convolve(a, a)
        M = float(np.max(conv))
    
    A = build_A_matrix(a)
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        # Solve LP: maximize sum(b) subject to A @ b <= M, b >= 0
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), 
                     bounds=[(0, None)]*n, method='highs')
        if res.status != 0:
            break
        
        b = res.x
        b_sum = float(np.sum(b))
        
        if b_sum <= 1.0 + 1e-6:
            # No improvement possible
            break
        
        # Rescale b to sum=1
        b = b / b_sum
        
        # Try different mixing ratios
        improved = False
        for t in [0.8, 0.6, 0.4, 0.2, 0.1]:
            a_new = (1-t)*a + t*b
            a_new = np.maximum(a_new, 0)
            a_new = normalize(a_new)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        
        if not improved:
            # Try with the full step
            a = b
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        
        # Update M
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        A = build_A_matrix(a)
    
    return best_a, best_obj

def subgradient_refine(a, max_iters=5000, time_limit=30.0, lr_init=0.01):
    """Subgradient method on the simplex."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        conv = np.convolve(a, a)
        max_val = float(np.max(conv))
        # Active set: convolution coefficients near the max
        active = np.where(conv >= max_val * 0.95)[0]
        
        # Compute subgradient
        # d/d a[i] of (a*a)[k] = 2*a[k-i] if 0<=k-i<n and k-i!=i, a[i] if k=2i
        # Subgradient of max is average of gradients of active constraints
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                idx = k - i
                if 0 <= idx < n:
                    if idx == i:
                        grad[i] += a[i] / len(active)
                    else:
                        grad[i] += 2.0 * a[idx] / len(active)
        
        # Projected gradient descent on simplex
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            # Decrease learning rate slowly
            lr = max(lr * 0.999, 0.0001)
        else:
            lr *= 0.95
            if lr < 1e-6:
                break
    
    return best_a, best_obj

    sum_a = float(np.sum(a))
    if sum_a < 0.01:

def normalize(a):
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_A(a):
    """Build constraint matrix for LP: row k has A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        # A[k, j] = a[k - j] for j in [j_min, j_max]
        # k-j ranges from k-j_max to k-j_min
        A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def gradient_of_conv_at_k(a, k):
    """Compute gradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    # grad[i] = 2*a[k-i] if 0 <= k-i < n
    i_min = max(0, k - n + 1)
    i_max = min(n - 1, k)
    # For i in [i_min, i_max], k-i is in [0, n-1]
    # grad[i] = 2 * a[k-i]
    # k-i ranges from k-i_max to k-i_min
    grad[i_min:i_max + 1] = 2.0 * a[k - i_max:k - i_min + 1][::-1]
    return grad

def spectral_init(n, rng):
    """Initialize using spectral methods - try to create flat-ish power spectrum."""
    # Start with random phase in frequency domain
    freq = np.fft.rfftfreq(n)
    # Want |A(f)| to be roughly constant
    # A(f) = sqrt(1 + noise) * exp(i * random_phase)
    n_freq = n // 2 + 1
    phases = rng.uniform(0, 2 * np.pi, n_freq)
    magnitudes = np.ones(n_freq) * (1 + 0.3 * rng.standard_normal(n_freq))
    magnitudes = np.maximum(magnitudes, 0.1)
    
    # Construct complex spectrum
    spectrum = magnitudes * np.exp(1j * phases)
    
    # Inverse FFT
    a = np.fft.irfft(spectrum, n=n)
    a = np.maximum(a, 0)
    a = normalize(a)
    return a


def cosine_init(n, rng):
    """Cosine window initialization."""
    x = np.arange(n)
    width = n * rng.uniform(0.3, 0.8)
    center = n / 2.0
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    # Add some structure
    if rng.random() < 0.5:
        freq = rng.uniform(0.5, 3.0)
        a *= (1 + 0.3 * np.cos(2 * np.pi * freq * x / n))
    a = np.maximum(a, 0)
    return normalize(a)


def uniform_init(n, rng):
    """Uniform with noise."""
    a = np.ones(n) + rng.uniform(-0.3, 0.3, n)
    a = np.maximum(a, 0)
    return normalize(a)


def multi_bump_init(n, rng):
    """Multi-bump initialization."""
    n_bumps = int(rng.integers(2, 5))
    a = np.zeros(n)
    centers = rng.integers(0, n, n_bumps)
    widths = rng.uniform(n * 0.05, n * 0.15, n_bumps)
    amps = rng.uniform(0.5, 1.5, n_bumps)
    for c, w, amp in zip(centers, widths, amps):
        a += amp * np.exp(-0.5 * ((np.arange(n) - c) / w) ** 2)
    a = np.maximum(a, 0)
    return normalize(a)
