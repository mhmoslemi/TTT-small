```python
def lp_direction(f, t=1.0):
    """Find optimal perturbation direction d using LP."""
    n = len(f)
    # LP: minimize M_d
    # subject to sum_i f[i]*d[k-i] <= M_d for k=0..2n-2
    #          sum(d) = 0
    #          d >= -f/t
    
    # Variables: d[0..n-1], M_d  (n+1 variables)
    # Constraints:
    #   For each k: sum_i f[i]*d[k-i] - M_d <= 0  (2n-1 constraints)
    #   sum(d) = 0  (1 equality constraint)
    #   d[i] >= -f[i]/t  (n bound constraints)
    
    from scipy.optimize import linprog
    
    # c = [0, ..., 0, 1] (minimize M_d, which is the last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each k: sum_i f[i]*d[k-i] - M_d <= 0
    A_ub = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i] = f[j]
        A_ub[k, -1] = -1.0
    b_ub = np.zeros(2*n-1)
    
    # Equality constraint: sum(d) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: d[i] >= -f[i]/t, M_d unbounded below (or we can set a reasonable lower bound)
    bounds = [(-f[i]/t, None) for i in range(n)] + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return None
```

```python
                best_f = f.copy()
            t = min(1.0, t * 1.2)
```

```python
"""
LP-based steepest descent for minimizing convolution peak of non-negative sequences.
Uses exact LP to find optimal perturbation direction at each step, combined with
line search and multi-start strategy.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score_normalized(f):
    """Compute score for normalized sequence (sum=1): 2*n*max(conv)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def simplex_projection(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lp_direction(f, lower_bound=-5.0):
    """
    Find optimal perturbation direction d using LP.
    Minimizes max_k (f*d)[k] subject to sum(d)=0, d >= lower_bound.
    """
    n = len(f)
    
    # Variables: d[0..n-1], M  (n+1 variables, M is the last)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # Build constraint matrix: for each k, sum_i f[k-i]*d[i] - M <= 0
    n_rows = 2 * n - 1
    A_ub = np.zeros((n_rows, n + 1))
    for k in range(n_rows):
        # conv[k] = sum_i f[i] * d[k-i] where 0 <= k-i < n
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            A_ub[k, i] = f[k - i]
        A_ub[k, -1] = -1.0
    b_ub = np.zeros(n_rows)
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(lower_bound, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
    except Exception as e:
        print(f"  LP error: {e}")
        return None
    
    if res.success and res.x is not None:
        return res.x[:n]
    return None


def lp_steepest_descent(f_init, time_budget, init_step=0.3, verbose=True):
    """
    Main optimization: LP-based steepest descent with line search.
    """
    n = len(f_init)
    f = f_init.copy().astype(np.float64)
    f = normalize(f)
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    if verbose:
        print(f"  Initial score: {best_score:.6f}")
    
    t = init_step
    start = time.monotonic()
    stall_count = 0
    iteration = 0
    
    while time.monotonic() - start < time_budget:
        d = lp_direction(f, lower_bound=-max(5.0, f.max() * 2))
        if d is None:
            break
        
        # Check if direction is meaningful
        if np.abs(d).max() < 1e-12:
            break
        
        # Line search
        improved = False
        for factor in [1.0, 0.7, 0.4, 0.2, 0.1]:
            trial_t = t * factor
            f_new = f + trial_t * d
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            score = compute_score_normalized(f_new)
            if score < best_score - 1e-13:
                f = f_new
                best_f = f_new.copy()
                best_score = score
                improved = True
                break
        
        iteration += 1
        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"  Iter {iteration}, score: {best_score:.6f}, elapsed: {elapsed:.1f}s")
        
        if improved:
            stall_count = 0
            t = min(1.0, t * 1.05)
        else:
            stall_count += 1
            t *= 0.5
            if t < 0.001:
                # Try perturbation to escape
                f_pert = f + 0.05 * np.random.default_rng(iteration).standard_normal(n)
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score_normalized(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score_normalized(f)
                    t = 0.1
                    stall_count = 0
                else:
                    break
    
    if verbose:
        elapsed = time.monotonic() - start
        print(f"  LP SD finished: score={best_score:.6f}, iters={iteration}, time={elapsed:.1f}s")
    
    return best_f, best_score


def pressure_update(f, n_iter=200, eta=0.1, rng=None):
    """Pressure-based update to reduce convolution peaks."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    stall_count = 0
    
    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Subgradient at the peak
        j_star = np.argmax(conv)
        pressure = np.zeros(n)
        for i in range(n):
            idx = j_star - i
            if 0 <= idx < n:
                pressure[i] = f[idx]
        
        # Also consider near-peak positions
        conv_sorted_idx = np.argsort(conv)[::-1]
        for pos in conv_sorted_idx[:3]:
            if conv[pos] > 0.95 * M:
                for i in range(n):
                    idx = pos - i
                    if 0 <= idx < n:
                        pressure[i] += 0.3 * f[idx]
        
        if pressure.max() > 0:
            pressure = pressure / pressure.max()
        
        eta_eff = eta * max(0.1, 1.0 / (1.0 + 0.01 * it))
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)
        
        score = compute_score_normalized(f_new)
        if score < best_score - 1e-13:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 50:
                f_pert = f * (1.0 + 0.1 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score_normalized(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score_normalized(f)
                    stall_count = 0
                else:
                    break
    
    return best_f, best_score


def spectral_flatten(f, n_iter=300, rng=None):
    """Spectral flattening: iteratively equalize the magnitude spectrum."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    stall = 0
    
    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Target: flat magnitude (except DC)
        n_bins = len(mag)
        current_energy = np.sum(f ** 2)
        target_mag = np.sqrt(current_energy * n_fft / n_bins)
        
        mag_new = np.full_like(mag, target_mag)
        mag_new[0] = 1.0  # DC fixed (sum = 1)
        
        # Occasional phase perturbation
        if it % 100 == 99:
            phase = phase + rng.normal(0, 0.03, len(phase))
        
        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        
        score = compute_score_normalized(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break
        f = f_new
    
    return best_f, best_score


def coordinate_descent(f, n_sweeps=10, rng=None):
    """Coordinate descent polishing."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score_normalized(f)
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-12:
                continue
            best_val = original
            best_local = best_score
            
            for factor in [0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.0/0.5]:
                f[i] = max(0.0, original * factor)
                conv = np.convolve(f, f)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]
            
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
        
        f = np.clip(f, 0, None)
        f = normalize(f)
    
    return f, best_score


def simulated_annealing(f, rng, n_iter=5000, T0=0.03, T_end=0.0001):
    """Simulated annealing with diverse moves."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    cur_score = best_score
    no_improve = 0
    
    for it in range(n_iter):
        T = T0 * (T_end / T0) ** (it / max(1, n_iter))
        f_new = f.copy()
        
        r = rng.random()
        if r < 0.3:
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.3))
        elif r < 0.5:
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] + rng.normal(0, 0.03))
        elif r < 0.7:
            i, j = rng.integers(0, n, 2)
            if i != j:
                f_new[i], f_new[j] = f_new[j], f_new[i]
        elif r < 0.85:
            center = int(rng.integers(0, n))
            width = max(2, n // 32)
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (width ** 2))
            f_new += rng.normal(0, 0.05) * kernel
        else:
            f_new *= np.exp(rng.normal(0, 0.2, n))
        
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score_normalized(f_new)
        
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 1500 and it > n_iter // 2:
            break
    
    return best_f, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    r = rng.random()
    if r < 0.25:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.15, 0.4)
        f = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    elif r < 0.5:
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    elif r < 0.75:
        freq = rng.uniform(1, 3)
        f = np.abs(np.sin(np.pi * np.arange(n) / n * freq)) + 0.1
    else:
        f = np.ones(n)
        for _ in range(5):
            center = int(rng.integers(0, n))
            width = int(rng.integers(3, n // 4))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / max(1, width ** 2))
            f *= (1 + 0.5 * rng.standard_normal() * kernel)
        f = np.abs(f) + 0.05
    
    f = np.clip(f, 0.01, None)
    f = normalize(f)
    f = f * (1 + 0.05 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f
```

```python
"""
LP-based steepest descent for minimizing convolution peak of non-negative sequences.
Uses exact LP to find the optimal perturbation direction at each step (minimizing
the first-order increase in convolution peak), combined with line search,
spectral flattening for initialization, and coordinate descent / simulated annealing
for final polishing. Multi-start with diverse initializations across multiple lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score_normalized(f):
    """Compute score for normalized sequence (sum=1): 2*n*max(conv)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def lp_direction(f, lb=-5.0):
    """
    Find optimal perturbation direction d via LP.
    Minimizes M subject to: sum_i f[k-i]*d[i] <= M for all k, sum(d)=0, d >= lb.
    Returns the direction d or None if LP fails.
    """
    n = len(f)
    n_rows = 2 * n - 1

    # Objective: minimize M (last variable)
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Inequality constraints: (f*d)[k] - M <= 0 for k = 0..2n-2
    A_ub = np.zeros((n_rows, n + 1))
    for k in range(n_rows):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            A_ub[k, i] = f[k - i]
        A_ub[k, -1] = -1.0
    b_ub = np.zeros(n_rows)

    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    # Bounds: d[i] >= lb, M free
    bounds = [(lb, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
    except Exception:
        return None

    if res.success and res.x is not None:
        d = res.x[:n]
        if not np.all(np.isfinite(d)):
            return None
        return d
    return None


def lp_steepest_descent(f_init, time_budget, init_step=0.3, verbose=True):
    """
    LP-based steepest descent with adaptive step size and line search.
    At each iteration, solves an LP to find the direction that most reduces
    the first-order increase in convolution peak, then takes a step.
    """
    n = len(f_init)
    f = f_init.copy().astype(np.float64)
    f = normalize(f)

    best_f = f.copy()
    best_score = compute_score_normalized(f)
    if verbose:
        print(f"    LP-SD start: score={best_score:.6f}")

    t = init_step
    start = time.monotonic()
    stall = 0
    it = 0
    rng_local = np.random.default_rng(12345)

    while time.monotonic() - start < time_budget:
        # Compute perturbation lower bound
        lb = -max(5.0, 2.0 * np.max(f))
        d = lp_direction(f, lb=lb)
        if d is None:
            break

        if np.abs(d).max() < 1e-14:
            break

        # Line search with multiple step sizes
        improved = False
        for factor in [1.0, 0.6, 0.35, 0.15, 0.05]:
            trial_t = t * factor
            f_new = f + trial_t * d
            f_new = np.clip(f_new, 0.0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            score = compute_score_normalized(f_new)
            if score < best_score - 1e-14:
                f = f_new
                best_f = f_new.copy()
                best_score = score
                improved = True
                break

        it += 1
        if verbose and (it % 100 == 0 or (it < 20 and it % 5 == 0)):
            elapsed = time.monotonic() - start
            print(f"    it={it}, score={best_score:.6f}, t={t:.4f}, elapsed={elapsed:.1f}s")

        if improved:
            stall = 0
            t = min(1.0, t * 1.08)
        else:
            stall += 1
            t *= 0.4
            if stall > 30:
                # Perturb to escape local optimum
                f_pert = f + 0.02 * rng_local.standard_normal(n)
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                pert_score = compute_score_normalized(f_pert)
                if pert_score < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = pert_score
                    t = 0.15
                    stall = 0
                else:
                    break

    if verbose:
        elapsed = time.monotonic() - start
        print(f"    LP-SD done: score={best_score:.6f}, iters={it}, time={elapsed:.1f}s")

    return best_f, best_score


def spectral_flatten(f, n_iter=200, rng=None):
    """Spectral flattening: iteratively equalize the magnitude spectrum."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()

    best_f = f.copy()
    best_score = compute_score_normalized(f)
    stall = 0

    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)

        n_bins = len(mag)
        current_energy = np.sum(f ** 2)
        target_mag = np.sqrt(current_energy * n_fft / n_bins)

        mag_new = np.full_like(mag, target_mag)
        mag_new[0] = 1.0

        if it % 80 == 79:
            phase = phase + rng.normal(0, 0.03, len(phase))

        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s

        score = compute_score_normalized(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break
        f = f_new

    return best_f, best_score


def pressure_update(f, n_iter=150, eta=0.12, rng=None):
    """Pressure-based update targeting convolution peaks."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score_normalized(f)
    stall = 0

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()

        # Subgradient: pressure from top peaks
        pressure = np.zeros(n)
        top_idx = np.argsort(conv)[::-1][:4]
        for pos in top_idx:
            weight = conv[pos] / M
            for i in range(n):
                idx = pos - i
                if 0 <= idx < n:
                    pressure[i] += weight * f[idx]

        if pressure.max() > 0:
            pressure = pressure / pressure.max()

        eta_eff = eta * max(0.1, 1.0 / (1.0 + 0.005 * it))
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        score = compute_score_normalized(f_new)
        if score < best_score - 1e-13:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 40:
                f_pert = f * (1.0 + 0.08 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score_normalized(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score_normalized(f)
                    stall = 0
                else:
                    break

    return best_f, best_score


def coordinate_descent(f, n_sweeps=10, rng=None):
    """Coordinate descent polishing with geometric step grid."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score_normalized(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-12:
                continue
            best_val = original
            best_local = best_score

            for factor in [0.4, 0.6, 0.8, 0.9, 1.0/0.9, 1.0/0.8, 1.0/0.6, 1.0/0.4]:
                f[i] = max(0.0, original * factor)
                conv = np.convolve(f, f)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]

            f[i] = best_val
            if best_local < best_score:
                best_score = best_local

        f = np.clip(f, 0, None)
        f = normalize(f)

    return f, best_score


def simulated_annealing(f, rng, n_iter=5000, T0=0.02, T_end=0.00005):
    """Simulated annealing with diverse perturbation moves."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    cur_score = best_score
    no_improve = 0

    for it in range(n_iter):
        T = T0 * (T_end / T0) ** (it / max(1, n_iter))
        f_new = f.copy()

        r = rng.random()
        if r < 0.25:
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start_i = int(rng.integers(0, max(1, n - width)))
            f_new[start_i:start_i + width] *= np.exp(rng.normal(0, 0.3))
        elif r < 0.45:
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] + rng.normal(0, 0.02))
        elif r < 0.65:
            i, j = rng.integers(0, n, 2)
            if i != j:
                f_new[i], f_new[j] = f_new[j], f_new[i]
        elif r < 0.85:
            center = int(rng.integers(0, n))
            width = max(2, n // 32)
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (width ** 2))
            f_new += rng.normal(0, 0.04) * kernel
        else:
            f_new *= np.exp(rng.normal(0, 0.15, n))

        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score_normalized(f_new)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 1200 and it > n_iter // 2:
            break

    return best_f, best_score


def diverse_init(n, rng):
    """Generate a diverse initial sequence."""
    r = rng.random()
    if r < 0.25:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.15, 0.4)
        f = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    elif r < 0.5:
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    elif r < 0.75:
        freq = rng.uniform(1.5, 4.0)
        f = np.abs(np.sin(np.pi * np.arange(n) / n * freq)) + 0.1
    else:
        f = np.ones(n)
        for _ in range(4):
            center = int(rng.integers(0, n))
            width = int(rng.integers(3, max(4, n // 4)))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / max(1, width ** 2))
            f *= (1 + 0.4 * rng.standard_normal() * kernel)
        f = np.abs(f) + 0.05

    f = np.clip(f, 0.01, None)
    f = normalize(f)
    f = f * (1 + 0.04 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: LP-based steepest descent as primary optimization,
    combined with spectral flattening, pressure updates, coordinate descent,
    and simulated annealing. Multi-start with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0

    best_overall = None
    best_score_overall = np.inf

    # Try parent sequence
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None

    if parent is not None:
        parent = np.clip(parent, 0, None)
        if parent.sum() > 0.01:
            parent = normalize(parent)
            parent_score = compute_score_normalized(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
            best_overall = parent.copy()
            best_score_overall = parent_score

    n_primary = len(parent) if parent is not None else 512

    # === Phase 1: LP Steepest Descent on primary (give it the most time) ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 1: LP-SD primary (n={n_primary}, budget={remaining:.0f}s) ===")

    if remaining > 90:
        if parent is not None:
            f0 = parent.copy()
        else:
            f0 = diverse_init(n_primary, rng)

        # Quick spectral init to get into a good basin
        f0, s0 = spectral_flatten(f0, n_iter=80, rng=rng)
        print(f"  After spectral init: {s0:.6f}")
        if s0 < best_score_overall:
            best_score_overall = s0
            best_overall = f0.copy()

        # Main LP optimization
        t_lp = min(remaining * 0.45, 400)
        f1, s1 = lp_steepest_descent(f0, t_lp, init_step=0.4, verbose=True)
        if s1 < best_score_overall:
            best_score_overall = s1
            best_overall = f1.copy()

    # === Phase 2: Different init + LP ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 2: Alt init + LP (remaining={remaining:.0f}s) ===")

    if remaining > 90:
        f0b = diverse_init(n_primary, rng)
        f0b, s0b = spectral_flatten(f0b, n_iter=60, rng=rng)
        t_lp2 = min(remaining * 0.35, 250)
        f2, s2 = lp_steepest_descent(f0b, t_lp2, init_step=0.3, verbose=True)
        if s2 < best_score_overall:
            best_score_overall = s2
            best_overall = f2.copy()
            print(f"  New best: {s2:.6f}")

    # === Phase 3: Try n=1024 ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 3: n=1024 (remaining={remaining:.0f}s) ===")

    if remaining > 120:
        n_alt = 1024
        f0c = diverse_init(n_alt, rng)
        f0c, s0c = spectral_flatten(f0c, n_iter=40, rng=rng)
        t_lp3 = min(remaining * 0.25, 150)
        f3, s3 = lp_steepest_descent(f0c, t_lp3, init_step=0.2, verbose=True)
        if s3 < best_score_overall:
            best_score_overall = s3
            best_overall = f3.copy()
            print(f"  New best at n=1024: {s3:.6f}")

    # === Phase 4: Pressure update + LP on best ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 4: Pressure+LP polish (remaining={remaining:.0f}s) ===")

    if remaining > 40 and best_overall is not None:
        fp, sp = pressure_update(best_overall, n_iter=80, eta=0.12, rng=rng)
        if sp < best_score_overall:
            best_score_overall = sp
            best_overall = fp.copy()
            print(f"  After pressure: {sp:.6f}")

        remaining2 = deadline - time.monotonic()
        if remaining2 > 25:
            f4, s4 = lp_steepest_descent(best_overall, min(remaining2 * 0.5, 80),
                                          init_step=0.15, verbose=True)
            if s4 < best_score_overall:
                best_score_overall = s4
                best_overall = f4.copy()

    # === Phase 5: Final polishing ===
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        print(f"\n=== Phase 5: Final Polish (remaining={remaining:.0f}s) ===")

        if remaining > 15:
            f_cd, s_cd = coordinate_descent(best_overall, n_sweeps=6, rng=rng)
            if s_cd < best_score_overall:
                best_score_overall = s_cd
                best_overall = f_cd.copy()
                print(f"  After CD: {s_cd:.6f}")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            n_sa = min(4000, int(remaining * 150))
            f_sa, s_sa = simulated_annealing(best_overall, rng, n_iter=n_sa,
                                             T0=0.015, T_end=0.00003)
            if s_sa < best_score_overall:
                best_score_overall = s_sa
                best_overall = f_sa.copy()
                print(f"  After SA: {s_sa:.6f}")

        # Final short LP pass
        remaining = deadline - time.monotonic()
        if remaining > 15:
            f_fin, s_fin = lp_steepest_descent(best_overall, min(remaining - 5, 40),
                                               init_step=0.05, verbose=False)
            if s_fin < best_score_overall:
                best_score_overall = s_fin
                best_overall = f_fin.copy()
                print(f"  Final LP: {s_fin:.6f}")

    # Validate and return
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal result: score={final_score:.6f}, n={len(final_list)}")
        return final_list

    # Fallback
    print("Fallback: uniform sequence")
    return [1.0] * 256
```
