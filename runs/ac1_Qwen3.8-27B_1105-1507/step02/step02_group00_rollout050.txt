Actually, let me reconsider the problem. The objective is `2*n*max_b/(sum_a)^2`. For a given `n`, we want to minimize `max_b / (sum_a)^2`. Since the objective is scale-invariant, we can fix `sum_a = 1` and minimize `2*n*max_b`.

The key question is: what's the best `n`? Larger `n` gives more "freedom" to spread out the convolution, but the factor of `n` in the numerator penalizes large `n`.

For the uniform sequence: objective = 2 regardless of `n`.
For a "concentrated" sequence (most mass at one point): `max_b ≈ 1`, `sum_a = 1`, objective ≈ `2n`. Bad.
For a "sparse" sequence (mass spread over `m < n` points uniformly): `max_b ≈ 1/m` (roughly), objective ≈ `2n/m`. Better if `m` is large.

The optimal tradeoff is to find a sequence where the convolution is as flat as possible. This is related to the concept of "low peak-to-average power ratio" (PAPR) sequences in communications.

**Key insight**: The problem is very similar to finding a sequence with minimal PAPR. There's a rich literature on this, including the "MMA" (Minimum Magnitude Algorithm) and "conference algorithm" from the OFDM literature.

The **conference algorithm** (also known as the SRM - Selected Mapping approach, or the phase retrieval approach) works like this:
1. Start with an initial sequence.
2. Compute the convolution (analogous to IFFT in OFDM).
3. Replace the peak with the mean (or apply a nonlinear function to reduce peaks).
4. Transform back (analogous to FFT).
5. Repeat.

In our case:
1. Start with `a`.
2. Compute `b = conv(a, a)`.
3. Reduce the peaks: `b_new = min(b, threshold)` or `b_new = b - alpha * relu(b - threshold)`.
4. "Transform back": find `a_new` such that `conv(a_new, a_new) ≈ b_new`. This is the hard part - it's like a "square root" of the convolution.

Step 4 is not straightforward. However, we can use the LP approach as an approximation: find `a_new` that minimizes `||conv(a_new, a_new) - b_new||` subject to `a_new >= 0, sum(a_new) = 1`.

But this is a non-convex quadratic minimization. The LP approach (fixing one variable) avoids this.

def _smoothed_obj_grad(a, T):
    n = len(a)
    c = np.convolve(a, a)
    mx = c.max()
    z = (c - mx) / T
    z -= z.max()
    w = np.exp(z)
    Z = w.sum()
    obj = (mx / T + z.max() + np.log(Z)) / n
    w /= Z
    gf = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        gf[js:je + 1] += 2.0 * w[k] * a[k - je:k - js + 1][::-1]

def _project_simplex(v):

    r = rho[-1]

def subgrad_polish(a_in, steps, stop_obj=None, coeff=1.0):

def _coord_one(j, a_now, C, n):
    """Minimize max_k f_k(x) over x in [0, a_now[j]/0.1] using bisection on
    the optimal-value function. C[k] = conv(a,a)[k] - 2 a[j] a[k-j] - x^2 indicator,
    i.e. C[k] is the part of conv[k] independent of x (with current a[j] removed)."""
    cur = a_now[j]
    if cur < 1e-15:
        return 0.0
    a = a_now.copy()
    # Build C: contribution of coordinate j removed
    # conv[a][k] = sum_{j'} a[j'] a[k-j']; terms involving j: 2 a[j] a[k-j] (k!=2j) or a[j]^2 (k=2j)
    a[j] = 0.0
    base = np.convolve(a, a)          # conv with a[j]=0
    Sother = a.sum()
    a[j] = cur
    # For candidate x: conv_k(x) = base[k] + 2 x a[k-j] (if 0<=k-j<n, k!=2j) + x^2 (if k==2j)
    # We'll just evaluate by writing a[j]=x and taking max of conv (O(n) per eval with precompute).
    # Precompute per-k coefficients: lin_k = 2 a[k-j], quad_k = 1 if k==2j else 0.
    lin = np.zeros(2 * n - 1)
    quad = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        m = k - j
        if 0 <= m < n:
            lin[k] = 2.0 * a[m]
            if k == 2 * j:
                quad[k] = 1.0
                lin[k] = 0.0  # x^2 term only, no linear (the two a[j]*a[j] merge)
    def fmax(x):
        # conv_k = base[k] + lin[k]*x + quad[k]*x^2
        vals = base + lin * x + quad * x * x
        return float(vals.max())
    # minimize F(x) = fmax(x) / (Sother + x)^2  ... but scale invariant so just min fmax(x)
    # subject to x >= 0 ; also keep sum reasonable. We minimize fmax(x) over x in [0, hi].
    lo, hi = 0.0, max(cur * 10.0, 1e-2)
    # shrink hi: if fmax(hi) < fmax(0) keep going else bisect in [0,hi]
    f0, fhi = fmax(0.0), fmax(hi)
    if fhi < f0:
        # try extending
        hi2 = hi * 10
        if fmax(hi2) < fhi:
            lo, hi = hi, hi2
    # ternary-ish / bisection on the convex-in-x max of convex functions: fmax is convex in x,
    # so F(x)=fmax(x)/(Sother+x)^2 ; not necessarily unimodal but fmax convex -> find min of fmax.
    # Use golden-section on fmax over [lo,hi].
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1, c2 = hi - gr * (hi - lo), lo + gr * (hi - lo)
    f1, f2 = fmax(c1), fmax(c2)
    for _ in range(45):
        if f1 < f2:
            hi, c2, f2 = c2, c1, f1
            c1 = hi - gr * (hi - lo)
            f1 = fmax(c1)
        else:
            lo, c1, f1 = c1, c2, f2
            c2 = lo + gr * (hi - lo)
            f2 = fmax(c2)
    xopt = 0.5 * (lo + hi)
    # ensure not blowing sum
    if xopt > 50.0:
        xopt = 50.0
    return float(xopt)

            for _ in range(40):
                if f1 < f2:
                    hi, c2, f2 = c2, c1, f1
                    c1 = hi - gr * (hi - lo); f1 = fmax(c1)
                else:
                    lo, c1, f1 = c1, c2, f2
                    c2 = lo + gr * (hi - lo); f2 = fmax(c2)
            xopt = 0.5 * (lo + hi)
            a[j] = xopt
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss < 1e-12:
            break
        a /= ss
        o = compute_obj(a)
        if o < best_obj - 1e-15:
            best_obj, best_a = o, a.copy()
            if best_obj <= 1.5005:
                break
    return best_a, best_obj


# ----------------------------- initializers ---------------------------------

def inv_poly_init(n, rng, beta_lo=0.2, beta_hi=1.2):
    x = np.linspace(0.0, 1.0, n)
    beta = float(rng.uniform(beta_lo, beta_hi))
    c = float(rng.uniform(0.2, 0.9))
    a = 1.0 / (c + (1.0 - c) * x) ** beta
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n)
    return a / a.sum()


def structured_init(n, rng):
    r = int(rng.integers(0, 6))
    if r == 0:
        f = float(rng.uniform(0.5, 3.0)); ph = float(rng.uniform(0, 2 * np.pi))
        a = np.maximum(0.0, np.cos(2 * np.pi * f * np.arange(n) / n + ph))
    elif r == 1:
        mid = n // 2
        a = np.ones(n); a[:mid] = float(rng.uniform(0.5, 1.5)); a[mid:] = float(rng.uniform(0.5, 1.5))
    elif r == 2:
        per = int(rng.integers(max(2, n // 30), max(3, n // 5)))
        pat = rng.random(per); a = np.tile(pat, n // per + 1)[:n]
    elif r == 3:
        a = np.exp(-float(rng.uniform(0, 2)) * (np.arange(n) / n - 0.5) ** 2)
    elif r == 4:
        a = np.abs(np.sin(np.arange(n) * float(rng.uniform(0.1, 1.0))))
    else:
        a = rng.random(n)
    s = a.sum()
    if s < 1e-12:
        a = rng.random(n); s = a.sum()
    return a / s


def symmetrize(a):
    b = 0.5 * (a + a[::-1])
    s = b.sum()
    if s < 1e-12:
        return a
    return b / s


# ----------------------------- main entry point -----------------------------

def propose_candidate(seed=42, budget_s=1500.0, **kw):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    epic = start + float(budget_s) - 4.0

    best_seq = None
    best_score = np.inf
    parent_obj = None

    # seed from globally-available previous construction if present
    try:
        p = np.asarray(globals().get('height_sequence_1'), dtype=np.float64)
        if p is not None and p.size > 0 and np.all(np.isfinite(p)):
            p = np.clip(p, 0.0, 1000.0)
            po = compute_obj(p)
            parent_obj = po
            best_score, best_seq = po, p.copy()
            print(f"[parent] n={p.size} obj={po:.6f}", flush=True)
    except Exception:
        pass

    n_values = [100, 120, 150, 180, 200, 250, 300, 360, 420, 500, 600]
    n_values = [n for n in n_values if n <= 700]

    # -------- Phase 1: LP alternating over many (n, init) --------
    for n in n_values:
        if time.monotonic() > epic - 200:
            break
        rem = epic - time.monotonic()
        if rem < 25:
            break
        print(f"[P1] n={n} rem={rem:.0f}s best={best_score:.6f}", flush=True)
        n_restarts = 4 if n <= 220 else 2
        for r in range(n_restarts):
            if time.monotonic() > epic - 200:
                break
            if r == 0 and best_seq is not None and len(best_seq) == n:
                a0 = best_seq.copy()
            elif r == 1:
                a0 = inv_poly_init(n, rng)
            else:
                a0 = structured_init(n, rng)
            mx = int(min(220, max(30, rem / 6.0)))
            a, o = lp_alternate(a0, max_iter=mx, stop_obj=min(best_score, 2.0) * 0.999)
            if o < best_score:
                best_score, best_seq = o, a.copy()
                print(f"   LP n={n} r={r} -> {o:.6f}", flush=True)
            # quick subgradient
            ns = int(min(600, max(50, (epic - time.monotonic()) / 0.05)))
            if ns > 30:
                a2, o2 = subgrad_polish(a, ns, stop_obj=best_score * 0.999)
                if o2 < best_score:
                    best_score, best_seq = o2, a2.copy()
                    print(f"   SG n={n} r={r} -> {o2:.6f}", flush=True)

    # ---- Phase 2: smoothed temperature-annealed gradient on best + candidates
    if best_seq is not None and time.monotonic() < epic - 60:
        nB = len(best_seq)
        print(f"[P2] smoothed refine n={nB} obj={best_score:.6f}", flush=True)
        aB = best_seq.copy(); sB = aB.sum()
        if sB < 1e-12:
            aB = np.ones(nB)
        else:
            aB = aB / sB
        # try a couple of n around nB with smoothed phase
        cand_ns = sorted(set([nB] + [max(20, nB - 30), nB + 30]))
        for nc in cand_ns:
            if time.monotonic() > epic - 60:
                break
            if nc == nB:
                ac = aB.copy()
            elif best_seq is not None and abs(len(best_seq) - nc) <= 5:
                # resize by truncation/padding
                ac = np.zeros(nc)
                m = min(nc, len(best_seq))
                ac[:m] = best_seq[:m]
                ac /= ac.sum()
            else:
                ac = inv_poly_init(nc, rng)
            it = int(min(700, max(60, (epic - time.monotonic()) / 0.08)))
            a3, o3 = smoothed_phase(ac, start, epic, iters=it, momentum=0.85)
            if o3 < best_score:
                best_score, best_seq = o3, a3.copy()
                print(f"   smooth n={nc} -> {o3:.6f}", flush=True)
            # symmetrize
            a4 = symmetrize(a3)
            o4 = compute_obj(a4)
            if o4 < best_score:
                best_score, best_seq = o4, a4.copy()
                print(f"   smooth+sym n={nc} -> {o4:.6f}", flush=True)

    # ---- Phase 3: heavy subgradient polish on the global best
    if best_seq is not None and time.monotonic() < epic - 20:
        nB = len(best_seq)
        rem = epic - time.monotonic()
        ns = int(min(15000, max(500, rem / 0.004)))
        print(f"[P3] subgrad n={nB} steps={ns}", flush=True)
        a5, o5 = subgrad_polish(best_seq, ns, stop_obj=best_score * 0.9999)
        if o5 < best_score:
            best_score, best_seq = o5, a5.copy()
            print(f"   -> {o5:.6f}", flush=True)
        a5s = symmetrize(a5)
        o5s = compute_obj(a5s)
        if o5s < best_score:
            best_score, best_seq = o5s, a5s.copy()
            print(f"   sym -> {o5s:.6f}", flush=True)

    # ---- Phase 4: 1-D coordinate refinement (if small enough)
    if best_seq is not None and len(best_seq) <= 400 and time.monotonic() < epic - 8:
        rem = epic - time.monotonic()
        rounds = int(min(4, max(1, rem / 3.0)))
        print(f"[P4] coord n={len(best_seq)} rounds={rounds}", flush=True)
        a6, o6 = coord_refine(best_seq, rounds=rounds, early=epic - 4)
        if o6 < best_score:
            best_score, best_seq = o6, a6.copy()
            print(f"   -> {o6:.6f}", flush=True)
        a6s = symmetrize(a6)
        o6s = compute_obj(a6s)
        if o6s < best_score:
            best_score, best_seq = o6s, a6s.copy()
            print(f"   sym -> {o6s:.6f}", flush=True)

    # ---- Phase 5: final LP + subgrad re-pass
    if best_seq is not None and time.monotonic() < epic - 4:
        a7, o7 = lp_alternate(best_seq, max_iter=40, stop_obj=best_score * 0.9999)
        if o7 < best_score:
            best_score, best_seq = o7, a7.copy()
            print(f"[P5] LP -> {o7:.6f}", flush=True)
        ns = int(min(4000, max(200, (epic - time.monotonic()) / 0.005)))
        a8, o8 = subgrad_polish(a7, ns, stop_obj=best_score * 0.9999)
        if o8 < best_score:
            best_score, best_seq = o8, a8.copy()
            print(f"[P5] SG -> {o8:.6f}", flush=True)

    # final symmetrize check
    if best_seq is not None:
        afs = symmetrize(best_seq)
        ofs = compute_obj(afs)
        if ofs < best_score:
            best_score, best_seq = ofs, afs.copy()
            print(f"[final sym] -> {ofs:.6f}", flush=True)

    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start
    print(f"DONE n={len(best_seq)} score={best_score:.6f} t={elapsed:.1f}s", flush=True)
    return [float(x) for x in np.clip(best_seq, 0.0, 1000.0)]
