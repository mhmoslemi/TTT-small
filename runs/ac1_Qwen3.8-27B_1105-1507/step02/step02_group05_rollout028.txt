def lp_solve(a, M, row_idx, col_idx, active_rows=None):
    n = len(a)
    if active_rows is not None:
        mask = np.isin(row_idx, active_rows)
        rows = row_idx[mask]
        cols = col_idx[mask]
        n_constraints = len(active_rows)
    else:
        rows = row_idx
        cols = col_idx
        n_constraints = 2 * n - 1

    if len(rows) == 0:
        return None

    vals = a[cols]
    A = csr_matrix((vals, (rows, cols)), shape=(n_constraints, n))
    c = -np.ones(n)
    b_ub = np.full(n_constraints, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
    if r.success and r.x is not None:
        return np.maximum(r.x, 0)
    return None


def lp_solve_cp(a, M, row_idx, col_idx, K_init=10, K_max=60):
    """Cutting-plane LP: start with top-K peaks, add constraints iteratively."""
    n = len(a)
    b = fast_conv(a)
    top_k = np.argsort(b)[-min(K_init, 2 * n - 1):]

    for _ in range(5):
        g = lp_solve(a, M, row_idx, col_idx, active_rows=top_k)
        if g is None:
            return None
        # Check if any new peak exceeds M
        ag = np.convolve(a, g, mode='full')
        new_peaks = np.where(ag > M + 1e-10)[0]
        if len(new_peaks) == 0:
            return g
        # Add new peaks to active set
        top_k = np.union1d(top_k, new_peaks)
        if len(top_k) >= K_max:
            top_k = np.argsort(ag)[-K_max:]
            g = lp_solve(a, M, row_idx, col_idx, active_rows=top_k)
            if g is not None:
                return g
    return g


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_inits(n, rng, parent):
    inits = []
    if parent is not None:

        # Shifted versions

            if p.sum() > 0.01:

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.3, 1.0, n))

    t_arr = np.linspace(0, 1, n, endpoint=False)
    for freq in [1, 2, 3, 4, 5]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2.0 * np.pi * freq * t_arr), 0.01))

    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)
    inits.append(np.abs(np.sin(np.pi * t_arr * 3)) + 0.1)
    inits.append(1.0 / (t_arr + 0.5) * (1.0 - t_arr) * 3 + 0.1)

    # Asymmetric "ramp" shapes
    inits.append(np.linspace(0.1, 1.0, n))
    inits.append(np.linspace(1.0, 0.1, n))
    inits.append(np.maximum(np.sin(np.pi * t_arr) ** 2, 0.01))

    # Sparse-ish
    sparse = np.zeros(n)
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_len = max(3, n // 32)
    kernel = np.exp(-0.5 * (np.arange(kernel_len) - kernel_len / 2) ** 2 / (kernel_len / 4))
    kernel /= kernel.sum()
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.02, 0.01))

    # Random smooth
    raw = rng.normal(0, 1, n)
    smooth = np.convolve(raw, np.ones(5) / 5, mode='same')
    inits.append(np.maximum(smooth - smooth.min() + 0.2, 0.01))

    for i in range(len(inits)):
        inits[i] = np.maximum(inits[i].astype(np.float64), 0.001)
        s = inits[i].sum()
        if s > 0.01:

def run_mirror_descent(a, max_iters, deadline, row_idx, col_idx, use_cp=True):
    """Run LP-based mirror descent iterations."""
    a = np.maximum(a.astype(np.float64), 0.001)

    current_J = compute_J(a)
    no_improve = 0

            break

        b = fast_conv(a)
        M = float(b.max())

        if use_cp:

        else:
            g = lp_solve(a, M, row_idx, col_idx)

            for lr in [0.005, 0.01, 0.02, 0.05]:
                a_trial = a - lr * grad
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-13:
                    a = a_trial
                    current_J = J_trial
                    no_improve = 0
                    break
            else:
                break
            continue

        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Multi-step-size evaluation
        best_a = None
        best_J = current_J
        step_candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for t in step_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-13:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            no_improve += 1
            if no_improve > 5:
                break
            # Tiny gradient step
            grad, _, _ = gradient_max_conv(a)
            a_trial = a - 0.005 * grad
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-13:
                a = a_trial
                current_J = J_trial
            else:
                break
        else:
            a = best_a
            current_J = best_J
            no_improve = 0

    return a, current_J


def run_gradient_descent(a, max_iters, deadline):
    """Gradient descent with adaptive learning rate."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    current_J = compute_J(a)
    lr = 0.01
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, k_star, b = gradient_max_conv(a)
        # Also consider the "average" gradient (all peaks weighted)
        top_idx = np.argsort(b)[-5:]
        avg_grad = np.zeros(len(a))
        for k in top_idx:
            for i in range(len(a)):
                j = k - i
                if 0 <= j < len(a):
                    avg_grad[i] += 2.0 * a[j]
        avg_grad /= len(top_idx)

        # Try gradient steps
        best_a = None
        best_J = current_J
        for g in [grad, avg_grad]:
            for lr_try in [lr * 0.1, lr, lr * 5, lr * 20]:
                a_trial = a - lr_try * g
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    best_J = J_trial
                    best_a = a_trial
                    lr = lr_try

        if best_a is None:
            no_improve += 1
            lr *= 0.5
            if no_improve > 20 or lr < 1e-8:
                break
        else:
            a = best_a
            current_J = best_J
            no_improve = 0
            lr = min(lr * 1.2, 0.5)
