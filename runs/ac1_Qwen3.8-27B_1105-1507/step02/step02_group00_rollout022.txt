```python
Optimized step function search using accelerated smoothed gradient descent,
LP alternation, and tau-annealing. Exploits the structure of the peak 
autocorrelation minimization problem with non-negative constraint.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A_matrix(a):
    """Build the convolution constraint matrix A where (A @ g)[k] = (a*g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def smooth_obj_grad(a, tau):
    """Compute smooth approximation to max(conv(a,a)) and its gradient.
    
    Returns (smooth_value, gradient).
    """
    n = len(a)
    conv = np.convolve(a, a)
    max_c = np.max(conv)
    w = np.exp((conv - max_c) / tau)
    w_sum = np.sum(w)
    w = w / w_sum
    
    smooth_val = tau * np.log(w_sum) + max_c
    
    # Gradient: grad_j = sum_m (w[j+m] + w[j-m]) * a_m
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(n)[None, :]
    idx1 = j_idx + m_idx
    idx2 = j_idx - m_idx
    
    # Clip for safe indexing
    idx1_clipped = np.clip(idx1, 0, 2 * n - 2)
    idx2_clipped = np.clip(idx2, 0, 2 * n - 2)
    
    valid1 = (idx1 >= 0) & (idx1 < 2 * n - 1)
    valid2 = (idx2 >= 0) & (idx2 < 2 * n - 1)
    
    vals1 = np.where(valid1, w[idx1_clipped], 0.0)
    vals2 = np.where(valid2, w[idx2_clipped], 0.0)
    
    M = vals1 + vals2
    grad = M @ a
    
    return smooth_val, grad


def accelerated_smooth_descent(a_init, n_iters=10000, tau_start=None, tau_end=None, 
                                verbose=False, deadline=None):
    """Nesterov-accelerated smoothed gradient descent on the normalized simplex."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    if tau_start is None:
        tau_start = 0.05
    if tau_end is None:
        tau_end = 0.0005
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    # Nesterov momentum
    y = a.copy()
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Anneal tau (geometric)
        progress = it / max(1, n_iters - 1)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Compute gradient at y
        _, grad_y = smooth_obj_grad(y, tau)
        
        # Step size: use 1/L where L is Lipschitz constant of gradient
        # L is bounded by 2 * max(w) * 2n / tau (very rough)
        # In practice, use a smaller step
        # The gradient of 2n * smooth_max has Lipschitz constant ~ 2n * 2 / (tau * sum(w)^2)
        # Let's use a safe step
        step = 1.0 / (4.0 * n / tau)
        step = min(step, 0.5)  # cap the step
        
        # Gradient step
        a_new = y - step * grad_y
        a_new = project_simplex(a_new)
        
        # Nesterov update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        t_prev = (1.0 + np.sqrt(1.0 + 4.0 * it ** 2)) / 2.0 if it > 0 else 1.0
        beta = (t_k - 1.0) / t_prev
        
        y = a_new + beta * (a_new - a)
        y = np.maximum(y, 0.0)
        s_y = np.sum(y)
        if s_y < 1e-10:
            y = a_new
        else:
            y = y / s_y
        
        a = a_new
        
        # Check objective
        if it % 10 == 0 or it == n_iters - 1:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        
        if verbose and it % 1000 == 0:
            obj = compute_obj(a)
            print(f"      SmoothGD it={it}, tau={tau:.6f}, obj={obj:.6f}, best={best_obj:.6f}")
    
    # Final check
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    
    return best_a, best_obj


def lp_alternate_fast(a_init, n_iters=100, verbose=False, deadline=None):
    """Fast LP alternation."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        g = solve_lp(a, M)
        if g is None:
            break
        
        g_s = np.sum(g)
        if g_s < 1e-10:
            break
        g = g / g_s
        
        # Check g
        obj_g = compute_obj(g)
        if obj_g < best_obj:
            best_obj = obj_g
            best_a = g.copy()
        
        # Blend
        t = 0.5 + 0.3 * (it / max(1, n_iters))
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a = g
        else:
            a = a_new / s_new
        
        # Also try symmetric update: solve LP with g fixed
        conv_g = np.convolve(g, g)
        M_g = float(np.max(conv_g))
        a_sol = solve_lp(g, M_g)
        if a_sol is not None:
            a_sol_s = np.sum(a_sol)
            if a_sol_s > 1e-10:
                a_sol = a_sol / a_sol_s
                obj_sol = compute_obj(a_sol)
                if obj_sol < best_obj:
                    best_obj = obj_sol
                    best_a = a_sol.copy()
                # Blend
                a_new2 = (1 - t) * a + t * a_sol
                a_new2 = np.maximum(a_new2, 0.0)
                s_new2 = np.sum(a_new2)
                if s_new2 > 1e-10:
                    a = a_new2 / s_new2
    
    return best_a, best_obj


def subgradient_polish(a_init, n_steps=5000, verbose=False, deadline=None):
    """Subgradient method with decreasing step size."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        conv = np.convolve(a, a)
        max_c = np.max(conv)
        
        # Active set
        tol = max_c * 0.005
        active = conv >= max_c - tol
        if not np.any(active):
            active = conv >= max_c - 1e-10
        
        active_idx = np.where(active)[0]
        
        # Subgradient
        grad = np.zeros(n)
        for k in active_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            vals = a[k - j_end:k - j_start + 1][::-1]
            grad[j_start:j_end + 1] += 2.0 * vals
        grad /= len(active_idx)
        
        # Step size (Polyak-like or diminishing)
        step_size = 1.0 / (np.sqrt(step + 1) * np.sqrt(n))
        
        a_new = a - step_size * grad
        a_new = project_simplex(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 8))
    x = np.arange(n)
    if choice == 0:
        freq = rng.uniform(1, 4)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0, np.cos(2 * np.pi * freq * x / n + phase))
    elif choice == 1:
        # Two-lobe
        mid = n // 2
        a = np.ones(n)
        a[:mid] *= rng.uniform(0.5, 1.5)
        a[mid:] *= rng.uniform(0.3, 1.0)
    elif choice == 2:
        # Gaussian
        center = rng.uniform(0.2 * n, 0.8 * n)
        width = rng.uniform(n / 10, n / 3)
        a = np.exp(-0.5 * ((x - center) / width) ** 2)
    elif choice == 3:
        # Periodic
        period = int(rng.integers(max(2, n // 20), max(3, n // 5)))
        pattern = np.abs(np.sin(2 * np.pi * np.arange(period) / period * rng.uniform(0.5, 2.0))) + 0.1
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 4:
        # Triangular
        peak = n // 2
        a = 1.0 - 2.0 * np.abs(x - peak) / n
        a = np.maximum(a, 0.0)
    elif choice == 5:
        # Sinc-like (non-negative)
        bw = rng.uniform(0.5, 2.0)
        a = np.sinc(bw * (x / n - 0.5))
        a = np.maximum(a, 0.0)
    elif choice == 6:
        # Power law decay from center
        center = n // 2
        a = 1.0 / (1.0 + (0.5 * np.abs(x - center)) ** rng.uniform(0.5, 3.0))
    else:
        # Random smooth
        a = np.random.default_rng(rng.integers(0, 2**31)).random(n)
        # Smooth it
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    
    s = np.sum(a)
    if s < 1e-10:
        a = np.random.default_rng(rng.integers(0, 2**31)).random(n)
        s = np.sum(a)
    return a / s
```

```python
    n = len(a)
    conv = np.convolve(a, a)
    max_c = np.max(conv)
    w = np.exp((conv - max_c) / tau)
    w_sum = np.sum(w)

    # and first row [2*w[n-1], 2*w[n], ..., 2*w[2n-2]]
    # Actually, M[j,m] = 2*w[j+m]

    # First row (j=0): [2*w[0], 2*w[1], ..., 2*w[n-1]]

    # Efficient computation: this is a convolution!
    # grad_j = 2 * (w_rev * a)_j where w_rev is... hmm
    # Actually, grad_j = 2 * sum_m w_{j+m} * a_m

        end = min(j + n, 2 * n - 1)
        grad[j] = 2.0 * np.dot(w[j:end], a[:end-j])
    
    return smooth_val, grad
```

```python
def smooth_obj_grad(a, tau):
    n = len(a)
    conv = np.convolve(a, a)
    max_c = np.max(conv)
    w = np.exp((conv - max_c) / tau)
    w_sum = np.sum(w)
    w = w / w_sum
    
    smooth_val = tau * np.log(w_sum) + max_c
    
    # M[j,m] = w[j+m] if j+m < 2n-1, else 0
    # Build M as a Toeplitz-like matrix
    j_idx = np.arange(n)[:, None]  # (n, 1)
    m_idx = np.arange(n)[None, :]  # (1, n)
    idx = j_idx + m_idx  # (n, n)
    valid = idx < (2 * n - 1)
    M = np.where(valid, 2.0 * w[np.clip(idx, 0, 2*n-2)], 0.0)
    
    grad = M @ a
    return smooth_val, grad
```
