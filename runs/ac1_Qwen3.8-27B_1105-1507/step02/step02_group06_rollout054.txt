    rows = np.empty(n * (n + 1), dtype=np.int32)
    cols = np.empty(n * (n + 1), dtype=np.int32)
    vals = np.empty(n * (n + 1), dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            rows[idx] = k
            cols[idx] = i
            vals[idx] = f[k - i]
            idx += 1
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2*n-1, n))


def solve_lp_direction(f, M, verbose=False):
    """
    Solve: max sum(g) s.t. g >= 0, conv(f,g)[k] <= M for all k.
    Returns g or None.
    """
    n = len(f)
    A = build_A_matrix(f)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception as e:
        if verbose:
            print(f"  LP error: {e}")
    return None


def solve_lp_dual(f, M):
    """
    Solve the dual: min M*sum(y) s.t. A^T y >= 1, y >= 0.
    Returns y (dual variables) and the optimal value.
    """
    n = len(f)
    A = build_A_matrix(f)
    AT = A.T.tocsr()
    c = np.full(2 * n - 1, M)  # minimize M * sum(y)
    b_lb = np.ones(n)  # A^T y >= 1
    bounds = [(0, None)] * (2 * n - 1)
    try:
        result = linprog(c, A_ub=-AT, b_ub=-b_lb, bounds=bounds, method='highs')
        # Note: A^T y >= 1 is equivalent to -A^T y <= -1
        if result.success:
            return result.x, result.fun
    except Exception:
        pass
    return None, None


def lp_optimize(f_init, deadline, step_init=0.3, verbose_prefix="", max_no_improve=100):
    """
    LP-based optimization: iteratively find best response direction and move toward it.
    """
    f = f_init.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    step = step_init
    no_improve = 0
    it = 0
    t0 = time.monotonic()
    last_print = t0

    while time.monotonic() < deadline:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 1e-15:
            break

        # Solve LP for best response direction
        g = solve_lp_direction(f, M)
        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            step *= 0.5
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-15:
            no_improve += 1
            break

        # Rescale g to match sum of f
        g = g * (S / sum_g)

        # Try multiple step sizes
        best_cand_score = best_score
        best_cand_f = None
        # Geometric sequence of step sizes around current step
        steps = [step * 0.5, step * 0.75, step, step * 1.25, step * 1.5, step * 2.0]
        for alpha in steps:
            alpha = min(alpha, 1.0)
            if alpha < 0.01:
                continue
            candidate = (1.0 - alpha) * f + alpha * g
            candidate = np.clip(candidate, 0.0, 1000.0)
            cand_sum = float(np.sum(candidate))
            if cand_sum < 1e-10:
                continue
            candidate = candidate * (S / cand_sum)
            cand_score = compute_score(candidate)
            if np.isfinite(cand_score) and cand_score < best_cand_score:
                best_cand_score = cand_score
                best_cand_f = candidate

        it += 1
        if best_cand_f is not None:
            f = best_cand_f
            best_f = f.copy()
            best_score = best_cand_score
            no_improve = 0
            # Adapt step size
            step = min(step * 1.05, 0.8)
        else:
            no_improve += 1
            step *= 0.8
            if step < 0.02 or no_improve > max_no_improve:
                break

        # Periodic reporting
        if time.monotonic() - last_print > 30:
            print(f"    {verbose_prefix}iter {it}: score={best_score:.6f}, step={step:.3f}, M={M:.4f}")
            last_print = time.monotonic()

    return best_f, best_score


def flatten_step(f, top_k=5):
    """
    Direct flattening: identify top-k lags in conv(f,f) and redistribute mass
    from positions that heavily contribute to those lags.
    """
    n = len(f)
    S = float(np.sum(f))
    conv = np.convolve(f, f)
    top_indices = np.argsort(conv)[-top_k:]
    
    # For each top lag k, the contribution to a[j] is 2*f[k-j]
    # We want to reduce f[j] where it contributes to top lags
    penalty = np.zeros(n)
    for k in top_indices:
        for j in range(n):
            if 0 <= k - j < n:
                penalty[j] += conv[k] * f[k - j]
    
    # Multiplicative update: reduce penalized positions
    # f_new[j] = f[j] * exp(-eta * penalty[j] / max(penalty))
    max_pen = np.max(penalty)
    if max_pen < 1e-15:
        return f.copy()
    
    eta = 0.1  # Update strength
    f_new = f * np.exp(-eta * penalty / max_pen)
    f_new = np.clip(f_new, 0.0, 1000.0)
    
    # Rescale
    new_sum = float(np.sum(f_new))
    if new_sum < 1e-10:
        return f.copy()
    f_new = f_new * (S / new_sum)
    
    return f_new


def subgradient_step(f, top_k=3, lr=0.01):
    """Simple subgradient descent step on max conv."""
    n = len(f)
    S = float(np.sum(f))
    conv = np.convolve(f, f)
    top_indices = np.argsort(conv)[-top_k:]
    
    grad = np.zeros(n)
    for k in top_indices:
        for j in range(n):
            if 0 <= k - j < n:
                grad[j] += 2.0 * f[k - j]
    
    # Normalize
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return f.copy()
    grad = grad / grad_norm
    
    f_new = f - lr * grad
    f_new = np.clip(f_new, 0.0, 1000.0)
    new_sum = float(np.sum(f_new))
    if new_sum < 1e-10:
        return f.copy()
    f_new = f_new * (S / new_sum)
    
    return f_new


def mixed_phase(f_init, deadline, verbose_prefix=""):
    """
    Mixed optimization phase: alternate LP steps with occasional flattening
    and subgradient steps to escape local minima.
    """
    f = f_init.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    
    # Phase 1: LP optimization (60% of time)
    t1 = time.monotonic() + (deadline - time.monotonic()) * 0.6
    f, s = lp_optimize(f, t1, step_init=0.3, verbose_prefix=verbose_prefix)

        best_score = s

    # Phase 2: Flattening + LP (20% of time)
    t2 = time.monotonic() + (deadline - time.monotonic()) * 0.2
    while time.monotonic() < t2:
        f_flat = flatten_step(f, top_k=5)

        if s_flat < best_score:
            best_score = s_flat
            best_f = f_flat.copy()

        # Short LP polish

        f, s = lp_optimize(f, t_sub, step_init=0.1, verbose_prefix="")
        if s < best_score:
            best_score = s

    # Phase 3: Fine LP polish (remaining time)
    f, s = lp_optimize(f, deadline, step_init=0.05, verbose_prefix=verbose_prefix + "fine ")
    if s < best_score:
        best_score = s
        best_f = f.copy()
    
    return best_f, best_score


def perturbation_restart(f, rng, scale=0.05):
    """Add random perturbation to escape local minima."""
    n = len(f)
    S = float(np.sum(f))
    noise = rng.uniform(-scale, scale, size=n)
    f_new = f + noise * f  # Multiplicative-like perturbation
    f_new = np.clip(f_new, 0.0, 1000.0)
    new_sum = float(np.sum(f_new))
    if new_sum < 1e-10:
        return f.copy()
    f_new = f_new * (S / new_sum)
    return f_new

"""
Optimization minimizing 2n*max(conv(a,a))/sum(a)^2 using:
- LP-based best response with adaptive constraint tightening
- Symmetry exploitation (restrict to palindromic sequences)
- Adaptive step sizes with multi-point line search
- Structured perturbation restarts to escape local minima
- Flattening heuristic as auxiliary improvement step
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    """Compute objective: 2n * max(conv(f,f)) / sum(f)^2"""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def build_A_matrix(f):
    """Build sparse matrix A (2n-1 x n) such that (A@g)[k] = conv(f,g)[k]."""
    n = len(f)
    total_nz = n * (n + 1)
    rows = np.empty(total_nz, dtype=np.int32)
    cols = np.empty(total_nz, dtype=np.int32)
    vals = np.empty(total_nz, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            rows[idx] = k
            cols[idx] = i
            vals[idx] = f[k - i]
            idx += 1
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def symmetrize(f):
    """Make sequence palindromic: f[j] = f[n-1-j]."""
    n = len(f)
    return (f + f[::-1]) / 2.0


def solve_lp(f, M, eps=0.0):
    """
    Solve: max sum(g) s.t. g >= 0, conv(f,g)[k] <= M*(1-eps) for all k.
    Returns g or None.
    """
    n = len(f)
    A = build_A_matrix(f)
    c = -np.ones(n)
    M_tight = M * (1.0 - eps)
    b_ub = np.full(2 * n - 1, M_tight)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_step(f, step_sizes, eps=0.0):
    """
    Perform one LP-based step. Returns (best_candidate, best_score) or (None, None).
    """
    n = len(f)
    S = float(np.sum(f))
    conv_f = np.convolve(f, f)
    M = float(np.max(conv_f))
    if M < 1e-15:
        return None, None

    g = solve_lp(f, M, eps=eps)
    if g is None:
        return None, None

    sum_g = float(np.sum(g))
    if sum_g < 1e-15:
        return None, None

    # Rescale g to match sum of f
    g = g * (S / sum_g)

    best_cand = None
    best_cand_score = float('inf')

    for alpha in step_sizes:
        alpha = min(alpha, 1.0)
        if alpha < 0.005:
            continue
        candidate = (1.0 - alpha) * f + alpha * g
        candidate = np.clip(candidate, 0.0, 1000.0)
        candidate = symmetrize(candidate)
        cand_sum = float(np.sum(candidate))
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (S / cand_sum)
        cand_score = compute_score(candidate)
        if np.isfinite(cand_score) and cand_score < best_cand_score:
            best_cand_score = cand_score
            best_cand = candidate

    return best_cand, best_cand_score


def flatten_heuristic(f, top_k=5, strength=0.05):
    """
    Reduce values that contribute most to the peak lags of conv(f,f).
    """
    n = len(f)
    S = float(np.sum(f))
    conv = np.convolve(f, f)
    top_indices = np.argsort(conv)[-top_k:]

    penalty = np.zeros(n)
    for k in top_indices:
        for j in range(n):
            if 0 <= k - j < n:
                penalty[j] += conv[k] * f[k - j]

    max_pen = np.max(penalty)
    if max_pen < 1e-15:
        return f.copy()

    f_new = f * np.exp(-strength * penalty / max_pen)
    f_new = np.clip(f_new, 0.0, 1000.0)
    f_new = symmetrize(f_new)
    new_sum = float(np.sum(f_new))
    if new_sum < 1e-10:
        return f.copy()
    f_new = f_new * (S / new_sum)
    return f_new


def subgradient_step(f, top_k=3, lr=0.01):
    """One subgradient descent step on max conv."""
    n = len(f)
    S = float(np.sum(f))
    conv = np.convolve(f, f)
    top_indices = np.argsort(conv)[-top_k:]

    grad = np.zeros(n)
    for k in top_indices:
        for j in range(n):
            if 0 <= k - j < n:
                grad[j] += 2.0 * f[k - j]

    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return f.copy()

    f_new = f - lr * grad / grad_norm
    f_new = np.clip(f_new, 0.0, 1000.0)
    f_new = symmetrize(f_new)
    new_sum = float(np.sum(f_new))
    if new_sum < 1e-10:
        return f.copy()
    f_new = f_new * (S / new_sum)
    return f_new

def make_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    idx = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(('uniform', np.ones(n)))

    # Cosine (raised cosine window)
    inits.append(('cosine', (1.0 + np.cos(2 * np.pi * idx / (n + 1))) / 2.0 + 0.1))

    # Tukey (near-Gaussian)
    center = (n - 1) / 2.0
    inits.append(('tukey', np.exp(-0.5 * ((idx - center) / (n * 0.3)) ** 2)))

    # Flat top with smooth edges
    edge = n // 8
    f_flat = np.ones(n)
    for i in range(edge):
        t = i / edge
        f_flat[i] = 0.5 * (1 - np.cos(np.pi * t))
        f_flat[n - 1 - i] = f_flat[i]
    inits.append(('flat_top', f_flat))

    # Sine taper
    inits.append(('sine_taper', np.sin(np.pi * (idx + 0.5) / n) ** 0.5))

    # Power taper
    inits.append(('power', np.sin(np.pi * (idx + 0.5) / n) ** 0.3))

    # Bilinear (trapezoidal)
    rise = n // 6
    f_trap = np.minimum(np.minimum(idx / rise, (n - 1 - idx) / rise), 1.0)
    inits.append(('trapezoid', f_trap + 0.1))

    # Random (bounded)
    inits.append(('random', rng.uniform(0.3, 1.5, size=n)))

    # Two-scale: base + ripple
    base = np.ones(n)
    ripple = 0.3 * np.sin(2 * np.pi * 5 * idx / n)
    inits.append(('ripple', base + ripple))

    # Raised cosine with higher order
    inits.append(('cos2', (1.0 + np.cos(2 * np.pi * idx / (n + 1))) ** 1.5 + 0.1))

    return inits

    if best_sequence is None:
        best_sequence = np.ones(n)

    # Ensure valid
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    final_list = [float(x) for x in best_sequence]
    final_score = evaluate_sequence(final_list)

    if not np.isfinite(final_score):
        print("WARNING: Invalid final sequence, using uniform fallback.")
        final_list = [1.0] * n
        final_score = evaluate_sequence(final_list)

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(final_list)}, score={final_score:.6f}, time={elapsed:.1f}s")

    return final_list
