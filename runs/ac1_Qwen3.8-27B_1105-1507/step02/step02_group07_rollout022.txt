            ss = np.dot(s, s)
            if ss > 1e-15 and abs(gs) > 1e-15:
                alpha = ss / max(abs(gs), 1e-15)
                alpha = np.clip(alpha, 1e-6, 10.0)

        y = y_next

"""
Multi-strategy optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses: (1) frequency-domain flatness (GS-style) initialization,
(2) FISTA on combined variance+softmax objective,
(3) LP-based direction refinement,
(4) perturbation restarts.
Key insight: optimal sequence has flat (uniform) autocorrelation.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_conv(f):
    """Compute linear autocorrelation of f."""
    return np.convolve(f, f)


def compute_soft_max_gradient(f, beta):
    """Compute gradient of soft_max(conv(f,f)) and the soft-max value."""
    n = len(f)
    R = np.convolve(f, f)
    max_R = R.max()
    exp_R = np.exp(beta * (R - max_R))
    total = exp_R.sum()
    if total < 1e-300:
        return np.zeros(n), 0.0
    p = exp_R / total
    corr = np.correlate(p, f, mode='valid')
    grad = 2.0 * corr
    soft_max_val = max_R + np.log(total) / beta
    return grad, soft_max_val


def compute_variance_gradient(f):
    """Compute gradient of ||conv(f,f)||_2^2 (encourages flat autocorrelation)."""
    n = len(f)
    R = np.convolve(f, f)
    corr = np.correlate(R, f, mode='valid')
    grad = 4.0 * corr
    return grad


def gs_flatness_init(f_init, n_GS_iters=500, n_alpha_start=0.3, n_alpha_end=1.0):
    """
    Gerchberg-Saxton style iterations to flatten the autocorrelation.
    Iterates: DFT -> partial magnitude flattening -> IDFT -> clip -> normalize.
    """
    f = f_init.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f
    f = f / S
    n = len(f)
    N = 2 * n  # Zero-pad to at least 2n-1

    for it in range(n_GS_iters):
        alpha = n_alpha_start + (n_alpha_end - n_alpha_start) * (it / max(n_GS_iters - 1, 1))
        F = np.fft.fft(f, N)
        mag = np.abs(F)
        # Target: flat magnitude
        target_mag = np.sqrt(np.sum(mag ** 2) / N)
        if target_mag < 1e-15:
            break
        # Partial flattening
        new_mag = (1.0 - alpha) * mag + alpha * target_mag
        # Preserve phase
        phase = F / (mag + 1e-30)
        F_new = new_mag * phase
        f_new = np.real(np.fft.ifft(F_new, N))[:n]
        f = np.clip(f_new, 0.0, None)
        S = float(np.sum(f))
        if S < 1e-10:
            f = f_init.copy() / float(np.sum(f_init))
            break
        f = f / S

    return f


def gs_flatness_optimize(f_init, deadline, max_iters=20000, alpha_start=0.1, alpha_end=0.8):
    """
    GS-style optimization with progress tracking and early stopping.
    """
    f = f_init.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)
    N = 2 * n

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, it / (max_iters * 0.7))
        F = np.fft.fft(f, N)
        mag = np.abs(F)
        target_mag = np.sqrt(np.sum(mag ** 2) / N)
        if target_mag < 1e-15:
            break
        new_mag = (1.0 - alpha) * mag + alpha * target_mag
        phase = F / (mag + 1e-30)
        F_new = new_mag * phase
        f_new = np.real(np.fft.ifft(F_new, N))[:n]
        f = np.clip(f_new, 0.0, None)
        S = float(np.sum(f))
        if S < 1e-10:
            f = best_f.copy()
            break
        f = f / S

        if it % 200 == 0:
            sc = evaluate_sequence(f.tolist())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = f.copy()
                no_improve = 0
                if it % 2000 == 0:
                    print(f"    GS iter {it}: score={best_score:.6f}")
            else:
                no_improve += 1
                if no_improve > 50:
                    break

    # Final evaluation
    sc = evaluate_sequence(f.tolist())
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_f = f.copy()

    return best_f, best_score

                   var_weight_start=0.5, var_weight_end=0.1, max_iters=20000):
    """

    f = f_init.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

    # FISTA state
    y = f.copy()
    t_k = 1.0
    alpha = 1.0  # step size

    # BB state
    prev_y = None
    prev_grad = None

    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive parameters
        frac = min(1.0, it / (max_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** frac
        var_w = var_weight_start + (var_weight_end - var_weight_start) * frac

        # Compute combined gradient at y
        grad_sm, soft_val = compute_soft_max_gradient(y, beta)
        grad_var = compute_variance_gradient(y)
        # Normalize variance gradient so both terms are comparable
        sm_scale = np.linalg.norm(grad_sm)
        var_scale = np.linalg.norm(grad_var)
        if sm_scale > 1e-15 and var_scale > 1e-15:
            grad_var = grad_var * (sm_scale / var_scale)
        grad = var_w * grad_var + (1.0 - var_w) * grad_sm

        # Step size: BB if available, else inverse norm
        if prev_y is not None and prev_grad is not None:
            s = y - prev_y
            g_diff = grad - prev_grad
            gs = np.dot(g_diff, s)
            ss = np.dot(s, s)
            if ss > 1e-15:
                bb_alpha = ss / max(gs, 1e-10)
                if bb_alpha > 0:
                    alpha = np.clip(0.5 * bb_alpha, 1e-8, 5.0 / (beta + 1))
                else:
                    alpha = 1.0 / (np.linalg.norm(grad) + 1e-10) * 0.1
            else:
                alpha = 1.0 / (np.linalg.norm(grad) + 1e-10) * 0.1
        else:
            alpha = 0.05 / (beta + 1.0)

        # Gradient step
        new_y = y - alpha * grad
        new_y = np.clip(new_y, 0.0, None)
        S_new = float(np.sum(new_y))
        if S_new < 1e-10:
            new_y = y.copy()
        else:
            new_y = new_y / S_new

        # Backtracking if function increased
        R_new = np.convolve(new_y, new_y)
        R_y = np.convolve(y, y)
        obj_new = var_w * np.sum(R_new ** 2) + (1 - var_w) * soft_val
        obj_y = var_w * np.sum(R_y ** 2) + (1 - var_w) * soft_val
        backtrack_count = 0
        while obj_new > obj_y + 1e-10 and backtrack_count < 10:
            alpha *= 0.5
            new_y = y - alpha * grad
            new_y = np.clip(new_y, 0.0, None)
            S_new = float(np.sum(new_y))
            if S_new < 1e-10:
                new_y = y.copy()
                S_new = 1.0
            else:
                new_y = new_y / S_new
            R_new = np.convolve(new_y, new_y)
            obj_new = var_w * np.sum(R_new ** 2) + (1 - var_w) * soft_val
            backtrack_count += 1

        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k ** 2)) / 2.0
        y_next = new_y + ((t_k - 1.0) / t_new) * (new_y - y)
        y_next = np.clip(y_next, 0.0, None)
        S_next = float(np.sum(y_next))
        if S_next < 1e-10:
            y_next = new_y.copy()
        else:
            y_next = y_next / S_next

        # Evaluate and track best
        if it % 5 == 0:
            sc = evaluate_sequence(new_y.tolist())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = new_y.copy()
                no_improve = 0
                if it % 500 == 0:
                    print(f"    FISTA iter {it}: score={best_score:.6f} (beta={beta:.1f}, vw={var_w:.2f})")
            else:
                no_improve += 1
                if no_improve > 100:
                    alpha *= 0.5
                    no_improve = 0
                    if alpha < 1e-8:
                        break

        prev_y = y.copy()
        prev_grad = grad.copy()
        y = y_next
        t_k = t_new

    return best_f, best_score


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    if A_cache is None:
        A = build_conv_matrix_sparse(f)
    else:
        A = A_cache
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_refine(f_init, deadline, max_iters=100):
    """LP-based direction finding refinement."""
    f = f_init.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    step_size = 0.1
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break

        g = solve_lp_direction(f, M)
        if g is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            continue
        g = g / sum_g  # normalize g to sum=1

        # Line search
        best_t = step_size
        best_s = best_score
        for trial_t in [step_size, step_size * 1.5, step_size * 0.5, step_size * 2, step_size * 0.8]:
            if trial_t <= 0 or trial_t > 1:
                continue
            new_f = (1.0 - trial_t) * f + trial_t * g
            new_f = np.clip(new_f, 0.0, None)
            ns = float(np.sum(new_f))
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = trial_t

        if best_s < best_score:
            f = (1.0 - best_t) * f + best_t * g
            f = np.clip(f, 0.0, None)
            f = f / float(np.sum(f))
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            if it % 20 == 0:
                print(f"    LP iter {it}: score={best_score:.6f}")
        else:
            no_improve += 1
            step_size *= 0.7
            if step_size < 0.01 or no_improve > 30:
                break

    return best_f, best_score


def variance_gradient_descent(f_init, deadline, max_iters=50000, lr_start=0.01):
    """
    Plain gradient descent on ||conv(f,f)||_2^2 with adaptive step size.
    Fast iterations (no soft-max), good for flatness.
    """
    f = f_init.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    lr = lr_start
    no_improve = 0
    prev_f = f.copy()
    prev_grad = None

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad = compute_variance_gradient(f)
        grad_norm = np.linalg.norm(grad)

        # BB step size
        if prev_grad is not None:
            s = f - prev_f
            g_diff = grad - prev_grad
            gs = np.dot(g_diff, s)
            ss = np.dot(s, s)
            if ss > 1e-15 and gs > 1e-15:
                lr_bb = ss / gs
                lr = np.clip(0.5 * lr_bb, 1e-8, 1.0 / (grad_norm + 1e-10))
            else:
                lr = lr * 0.9

        new_f = f - lr * grad
        new_f = np.clip(new_f, 0.0, None)
        S_new = float(np.sum(new_f))
        if S_new < 1e-10:
            new_f = f
        else:
            new_f = new_f / S_new

        if it % 10 == 0:
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = new_f.copy()
                no_improve = 0
                if it % 1000 == 0:
                    print(f"    varGD iter {it}: score={best_score:.6f}")
            else:
                no_improve += 1
                if no_improve > 50:
                    lr *= 0.5
                    no_improve = 0
                    if lr < 1e-10:
                        break

        prev_f = f.copy()
        prev_grad = grad.copy()
        f = new_f

    return best_f, best_score


def tukey_window(n, r=0.3):
    """Generate a Tukey window."""
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def hann_window(n):
    """Generate a Hann window."""
    n = max(n, 2)
    return 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))


def flat_autocorrelation_init(n):
    """
    Initialize using the insight that optimal sequences have flat autocorrelation.
    Use GS iterations from a uniform start to converge toward flat autocorrelation.
    """
    f = np.ones(n)
    # Run GS to flatten
    f = gs_flatness_init(f, n_GS_iters=200, n_alpha_start=0.2, n_alpha_end=0.9)
    return f


def symmetric_random_init(n, rng, smoothness=3.0):
    """Generate a symmetric, smooth, non-negative random sequence."""
    half = n // 2
    x = rng.standard_normal(half)
    # Smooth
    kernel_size = int(smoothness * 3)
    kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / smoothness) ** 2)
    x = np.convolve(x, kernel, mode='same')
    x = np.abs(x) + 0.1
    # Make symmetric
    if n % 2 == 0:
        f = np.concatenate([x, x[::-1]])
    else:
        f = np.concatenate([x, x[-2::-1]])
    return f[:n]


def full_optimize_pipeline(f_init, deadline, use_gs=True, use_fista=True, use_lp=True):
    """Run the full optimization pipeline on a given starting point."""
    f_init = f_init.copy().astype(np.float64)
    S = float(np.sum(f_init))
    if S < 1e-10:
        return f_init, evaluate_sequence(f_init.tolist())
    f_init = f_init / S

    best_f = f_init.copy()
    best_score = evaluate_sequence(f_init.tolist())
    t0 = time.monotonic()
    total_time = deadline - t0

    # Phase 1: GS flatness (very fast)
    if use_gs and total_time > 10:
        gs_deadline = t0 + total_time * 0.2
        f1, s1 = gs_flatness_optimize(f_init, gs_deadline, max_iters=50000,
                                       alpha_start=0.1, alpha_end=0.7)
        if s1 < best_score:
            best_f, best_score = f1, s1

    # Phase 2: Variance gradient descent
    if total_time > 20:
        var_deadline = t0 + total_time * 0.4
        f2, s2 = variance_gradient_descent(best_f, var_deadline, max_iters=100000,
                                            lr_start=0.005)
        if s2 < best_score:
            best_f, best_score = f2, s2

    # Phase 3: FISTA on combined objective
    if use_fista and total_time > 40:
        fista_deadline = t0 + total_time * 0.7
        f3, s3 = fista_combined(best_f, fista_deadline, beta_start=10.0, beta_end=400.0,
                                 var_weight_start=0.3, var_weight_end=0.05, max_iters=50000)
        if s3 < best_score:
            best_f, best_score = f3, s3

    # Phase 4: LP refinement
    if use_lp and time.monotonic() < deadline - 5:
        lp_deadline = deadline - 5
        f4, s4 = lp_refine(best_f, lp_deadline, max_iters=80)
        if s4 < best_score:
            best_f, best_score = f4, s4

    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    """
    Main search function. Tries multiple lengths and initialization strategies,
    using GS flatness, FISTA, variance GD, and LP refinement.
    """
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                score = evaluate_sequence(parent.tolist())
                if np.isfinite(score):
                    best_score = score
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    # Lengths to try, ordered by expected return
    # Key insight: longer n allows flatter autocorrelation, approaching score=1
    lengths_to_try = [512, 768, 1024, 1536, 2048]
    time_fractions = {512: 0.22, 768: 0.18, 1024: 0.20, 1536: 0.18, 2048: 0.15}
    total_budget = deadline - t0

    for n in lengths_to_try:
        if time.monotonic() > deadline - 60:
            print("Time nearly out, stopping.")
            break

        n_budget = total_budget * time_fractions.get(n, 0.1)
        n_deadline = t0 + total_budget * sum(time_fractions.get(l, 0) for l in lengths_to_try[:lengths_to_try.index(n) + 1])
        n_deadline = min(n_deadline, deadline - 30)  # save 30s for final polish
        print(f"\n=== n={n} (budget ~{n_budget:.0f}s) ===")

        # Generate initializations
        inits = []

        # Best parent (interpolated if needed)
        if best_sequence is not None:
            if len(best_sequence) == n:
                inits.append(('parent', best_sequence.copy()))
            else:
                # Interpolate
                x_old = np.linspace(0, 1, len(best_sequence))
                x_new = np.linspace(0, 1, n)
                parent_interp = np.interp(x_new, x_old, best_sequence)
                parent_interp = np.clip(parent_interp, 0, None)
                inits.append(('parent_interp', parent_interp))

        # Flat autocorrelation init (GS from uniform)
        inits.append(('flat_ac', flat_autocorrelation_init(n)))

        # Tukey windows
        for r in [0.2, 0.4]:
            inits.append((f'tukey_{r}', tukey_window(n, r)))

        # Hann
        inits.append(('hann', hann_window(n)))

        # Symmetric random smooth
        inits.append(('rand_sym', symmetric_random_init(n, rng, smoothness=n / 8.0)))

        # Optimize each init with time allocation
        n_inits = len(inits)
        per_init_time = (n_deadline - time.monotonic()) / max(n_inits, 1) * 0.85

        for label, f_init in inits:
            if time.monotonic() > n_deadline:
                break

            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            S = float(np.sum(f_init))
            if S < 0.01:
                continue

            init_score = evaluate_sequence(f_init.tolist())
            if not np.isfinite(init_score):
                continue

            if init_score < best_score:
                best_score = init_score
                best_sequence = f_init.copy()
                print(f"  [n={n} {label}] init: {best_score:.6f}")

            # Run full pipeline
            init_deadline = min(n_deadline, time.monotonic() + per_init_time)
            if time.monotonic() < init_deadline - 3:
                try:
                    bf, bs = full_optimize_pipeline(f_init, init_deadline)
                    if bs < best_score:
                        best_score = bs
                        best_sequence = bf.copy()
                        print(f"  [n={n} {label}] optimized: {bs:.6f}")
                    else:
                        print(f"  [n={n} {label}] no improvement (best={best_score:.6f})")
                except Exception as e:
                    print(f"  [n={n} {label}] error: {e}")

    # Perturbation restarts on best found
    if best_sequence is not None and time.monotonic() < deadline - 40:
        print(f"\n=== Perturbation restarts (best: {best_score:.6f}) ===")
        restart_budget = deadline - time.monotonic() - 30
        n_restarts = min(5, max(2, int(restart_budget / 40)))
        per_restart = restart_budget / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline - 35:
                break

            strength = 0.02 + 0.02 * r
            noise = 1.0 + strength * rng.standard_normal(len(best_sequence))
            f_pert = best_sequence * np.clip(noise, 0.2, 2.0)
            f_pert = np.clip(f_pert, 0.0, 1000.0)
            S_p = float(np.sum(f_pert))
            if S_p < 0.01:
                continue
            f_pert = f_pert / S_p

            # Also try a "flattening" perturbation: push toward flat autocorrelation
            R = np.convolve(f_pert, f_pert)
            R_target = np.full_like(R, np.mean(R))
            # Add small correction toward flat
            f_flat_pert = f_pert + 0.01 * compute_variance_gradient(f_pert)
            f_flat_pert = np.clip(f_flat_pert, 0.0, 1000.0)
            S_fp = float(np.sum(f_flat_pert))
            if S_fp > 0.01:
                f_flat_pert = f_flat_pert / S_fp
                f_pert = f_flat_pert  # Use the flattened version

            restart_deadline = time.monotonic() + per_restart * 0.8
            try:
                bf, bs = full_optimize_pipeline(f_pert, restart_deadline,
                                                 use_gs=True, use_fista=True, use_lp=True)
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  Restart {r} (str={strength:.3f}): {bs:.6f}")
                else:
                    print(f"  Restart {r}: no improvement ({bs:.6f})")
            except Exception as e:
                print(f"  Restart {r} error: {e}")

    # Final intensive refinement on the absolute best
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Final refinement (n={len(best_sequence)}, score={best_score:.6f}) ===")
        final_deadline = deadline - 5

        # GS polish
        if time.monotonic() < final_deadline - 30:
            try:
                bf, bs = gs_flatness_optimize(best_sequence, min(final_deadline - 30, time.monotonic() + 60),
                                               max_iters=100000, alpha_start=0.05, alpha_end=0.5)
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  GS final: {best_score:.6f}")
            except Exception as e:
                print(f"  GS final error: {e}")

        # FISTA polish with high beta
        if time.monotonic() < final_deadline - 10:
            try:
                bf, bs = fista_combined(best_sequence, min(final_deadline - 5, time.monotonic() + 80),
                                         beta_start=100.0, beta_end=800.0,
                                         var_weight_start=0.1, var_weight_end=0.0,
                                         max_iters=50000)
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  FISTA final: {best_score:.6f}")
            except Exception as e:
                print(f"  FISTA final error: {e}")

        # LP polish
        if time.monotonic() < final_deadline - 3:
            try:
                bf, bs = lp_refine(best_sequence, final_deadline - 2, max_iters=50)
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  LP final: {best_score:.6f}")
            except Exception as e:
                print(f"  LP final error: {e}")

    # Fallback
    if best_sequence is None:
        best_sequence = tukey_window(512, 0.3)
        best_score = evaluate_sequence(best_sequence.tolist())

    # Ensure validity
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
