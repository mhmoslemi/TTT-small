def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def compute_score_norm(f):

    conv = np.convolve(f, f)

def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)


def build_conv_matrix(f, top_k=None):
    """Build constraint matrix for LP. Only use top_k constraints if specified."""
    n = len(f)
    conv = np.convolve(f, f)
    if top_k is not None and top_k < len(conv):
        # Only keep top_k largest convolution values as constraints
        top_indices = np.argsort(conv)[::-1][:top_k]
        rows = top_indices
    else:
        rows = range(len(conv))

    A = np.zeros((len(rows), n))
    for idx, k in enumerate(rows):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[idx, j] = f[k - j]
    return A, list(rows)


def lp_step_active(f, M, top_k=20, t=0.5):
    """One LP step using only top-k active constraints."""
    n = len(f)
    A, rows = build_conv_matrix(f, top_k=top_k)
    b_ub = np.full(len(rows), M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 2})
    if not res.success or res.x is None:
        return None
    return np.clip(res.x, 0, None)

        active_indices = np.where(conv >= threshold)[0]

                if 0 <= idx < n:
                    pressure[i] += weight * f[idx]

        if pressure.max() > 0:

        eta_eff = eta * max(0.1, 1.0 - it / n_iter)

                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score_norm(f)

        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)

        # Target: equalize magnitudes (flat spectrum)
        # DC must be 1 (sum=1)
        # For other bins, target magnitude = sqrt(energy * n_fft / n_non_dc_bins)
        n_non_dc = len(mag) - 1
        current_energy = np.sum(f ** 2)
        target_mag = np.sqrt(current_energy * n_fft / n_non_dc)

        mag_new = np.full_like(mag, target_mag)
        mag_new[0] = 1.0

        # Small randomization
        if it % 30 == 29:
            phase = phase + rng.normal(0, 0.03, len(phase))

        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s

        score = compute_score_norm(f_new)
        if score < best_score - 1e-14:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 150:
                break

    return best_f, best_score


def coordinate_descent_fast(f, n_sweeps=20, rng=None):
    """Fast coordinate descent with binary search on optimal value."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score_norm(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-12:
                continue
            best_val = original
            best_local = best_score

            # Try several multiplicative factors
            for factor in [0.2, 0.4, 0.6, 0.8, 1.25, 1.5, 2.0, 3.0]:
                f[i] = max(0.0, original * factor)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                score = compute_score(f)
                if score < best_local:
                    best_local = score
                    best_val = f[i]

            f[i] = best_val

        f = np.clip(f, 0, None)
        f = normalize(f)
        cur = compute_score_norm(f)
        if cur < best_score:
            best_score = cur

    return f, best_score


def multi_start_search(n, time_budget, rng):
    """Try multiple random initializations and optimize each briefly."""
    start = time.monotonic()
    best_f = None
    best_score = np.inf

    n_starts = max(3, int(time_budget / 30))

    for s in range(n_starts):
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 5:
            break
        remaining = time_budget - elapsed
        if remaining < 10:
            break

        # Diverse initializations
        r = rng.random()
        if r < 0.3:
            f0 = np.ones(n) * (1 + 0.1 * rng.normal(0, 1, n))
        elif r < 0.5:
            center = n * rng.uniform(0.3, 0.7)
            width = n * rng.uniform(0.15, 0.4)
            f0 = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
        elif r < 0.7:
            f0 = 0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / n * rng.uniform(0.5, 2))
        elif r < 0.85:
            f0 = np.abs(np.sin(np.pi * np.arange(n) / n * rng.uniform(1, 4))) + 0.1
        else:
            f0 = np.ones(n)
            for _ in range(3):
                c = int(rng.integers(0, n))
                w = int(rng.integers(5, n // 4))
                f0 *= (1 + 0.5 * rng.standard_normal() * np.exp(-0.5 * (np.arange(n) - c) ** 2 / max(1, w ** 2)))
            f0 = np.abs(f0) + 0.05

        f0 = np.clip(f0, 0.01, None)
        f0 = normalize(f0) / normalize(f0).sum() * normalize(f0).sum()  # just normalize
        f0 = normalize(f0)

        # Quick optimization
        t1 = min(15, remaining * 0.25)
        f1, s1 = pressure_update_fast(f0, n_iter=max(50, int(t1 * 30)), eta=0.15, rng=rng)
        if s1 < best_score:
            best_score = s1
            best_f = f1.copy()

        elapsed = time.monotonic() - start
        if elapsed > time_budget - 10:
            break

        t2 = min(20, remaining * 0.3)
        if t2 > 5:
            f2, s2 = lp_refine_active(f1, t2, t0=0.4, top_k=10, rng=rng)
            if s2 < best_score:
                best_score = s2
                best_f = f2.copy()

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: combines LP active-set, pressure updates, spectral methods,
    and coordinate descent to minimize the auto-convolution score.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0

    # Get parent sequence
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None

    best_overall = None
    best_score_overall = np.inf

    # Uniform baseline
    n_default = 512
    uniform = np.ones(n_default) / n_default
    uniform_score = compute_score_norm(uniform)
    print(f"Uniform baseline: n={n_default}, score={uniform_score:.8f}")
    best_overall = uniform.copy()
    best_score_overall = uniform_score

    # Try parent
    if parent is not None:
        parent = np.clip(parent, 0, None)
        if parent.sum() > 0.01:
            parent = normalize(parent)
            p_score = compute_score_norm(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            if p_score < best_score_overall:
                best_score_overall = p_score
                best_overall = parent.copy()

    # Determine lengths to try
    lengths = [512]
    if parent is not None and len(parent) != 512:
        lengths.append(len(parent))

    for n in lengths:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        print(f"\n=== n={n}, remaining={remaining:.0f}s ===")

        # Strategy 1: Start from parent or uniform, do pressure + LP
        if parent is not None and len(parent) == n:
            f_start = parent.copy()
        else:
            f_start = np.ones(n) / n

        # Phase 1: Pressure update (fast, many iterations)
        t_press = min(30, remaining * 0.15)
        f1, s1 = pressure_update_fast(f_start, n_iter=max(100, int(t_press * 40)),
                                      eta=0.12, rng=rng)
        print(f"  After pressure: {s1:.8f}")
        if s1 < best_score_overall:
            best_score_overall = s1
            best_overall = f1.copy()

        # Phase 2: LP refine
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            t_lp = min(remaining2 * 0.4, 120)
            f2, s2 = lp_refine_active(f1, t_lp, t0=0.5, top_k=12, rng=rng)
            print(f"  After LP: {s2:.8f}")
            if s2 < best_score_overall:
                best_score_overall = s2
                best_overall = f2.copy()

        # Phase 3: Spectral
        remaining3 = deadline - time.monotonic()
        if remaining3 > 20:
            f3, s3 = spectral_gradient_step(f2, n_iter=200, rng=rng)
            print(f"  After spectral: {s3:.8f}")
            if s3 < best_score_overall:
                best_score_overall = s3
                best_overall = f3.copy()

            # Phase 4: Another LP pass on spectral result
            remaining4 = deadline - time.monotonic()
            if remaining4 > 15:
                f4, s4 = lp_refine_active(f3, min(remaining4 * 0.5, 60), t0=0.3,
                                          top_k=10, rng=rng)
                print(f"  After LP2: {s4:.8f}")
                if s4 < best_score_overall:
                    best_score_overall = s4
                    best_overall = f4.copy()

        # Strategy 2: Multi-start
        remaining5 = deadline - time.monotonic()
        if remaining5 > 30:
            f_ms, s_ms = multi_start_search(n, remaining5 * 0.4, rng)
            if f_ms is not None and s_ms < best_score_overall:
                print(f"  Multi-start: {s_ms:.8f}")
                best_score_overall = s_ms
                best_overall = f_ms.copy()

    # Final polishing
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            print(f"\n=== Polishing, remaining={remaining:.0f}s ===")
            # Coordinate descent
            f_cd, s_cd = coordinate_descent_fast(best_overall, n_sweeps=
                                                 max(5, int(remaining * 0.3)), rng=rng)
            if s_cd < best_score_overall:
                best_score_overall = s_cd
                best_overall = f_cd.copy()
                print(f"  After CD: {s_cd:.8f}")

            # Final LP
            remaining2 = deadline - time.monotonic()
            if remaining2 > 10:
                f_final, s_final = lp_refine_active(best_overall, min(remaining2 * 0.7, 60),
                                                    t0=0.2, top_k=8, rng=rng)
                if s_final < best_score_overall:
                    best_score_overall = s_final
                    best_overall = f_final.copy()
                    print(f"  After final LP: {s_final:.8f}")

    # Verify with evaluate_sequence
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal verified: score={final_score:.8f}, n={len(final_list)}")
        return final_list

    # Fallback: uniform
    final = [1.0] * 512
    print(f"Fallback uniform, score={evaluate_sequence(final):.8f}")
    return final
