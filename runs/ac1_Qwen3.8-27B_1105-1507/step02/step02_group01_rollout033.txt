```python
Optimized step function coefficient search using Adam soft-max gradient descent,
LP-based alternating minimization, and multi-scale refinement.
Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    """Compute the objective function value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def softmax_obj_and_grad(a, T):
    """Compute soft-max of convolution and its gradient w.r.t. a."""
    n = len(a)
    conv = np.convolve(a, a)
    # Soft max
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    sm_val = max_c + T * np.log(sum_exp)
    # Gradient: d(sm_val)/d(a[j]) = sum_i p[i] * d(conv[i])/d(a[j])
    # d(conv[i])/d(a[j]) = a[i-j] if 0 <= i-j < n, else 0
    # So grad[j] = sum_{i: 0<=i-j<n} p[i] * a[i-j]
    # = sum_{k: 0<=k<n} p[j+k] * a[k] (where k = i-j, so i = j+k)
    # But we need 0 <= i <= 2n-2, so 0 <= j+k <= 2n-2
    # For j in [0, n-1] and k in [0, n-1], j+k is in [0, 2n-2], always valid
    # So grad[j] = dot(p[j:j+n], a)
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = np.dot(p[j:j+n], a)
    return sm_val, grad, conv, p
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def safe_normalize(a):
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s

def softmax_obj_and_grad(a, T):
    n = len(a)
    conv = np.convolve(a, a)
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    sm_val = max_c + T * np.log(sum_exp)
    # Gradient via matrix
    idx = np.arange(n)[:, None] + np.arange(n)[None, :]
    M = p[idx]
    grad = M @ a
    return sm_val, grad, conv, p

    """Adam gradient descent on soft-max of convolution."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if T_init is None:
        conv = np.convolve(a, a)
        T_init = 0.1 * float(np.max(conv) - np.min(conv) + 1e-10)
    
    T = T_init

    v = np.zeros(n)  # Second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    
    while time.monotonic() - start < max_time:
        sm_val, grad, conv, p = softmax_obj_and_grad(a, T)
        t += 1

        v = beta2 * v + (1 - beta2) * grad**2

            if lr < 1e-12:
                break
```

```python
def max_grad_descent(a, max_time):
    """Gradient descent on the max of convolution, using subgradient at the argmax."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.05
    
    while time.monotonic() - start < max_time:
        conv = np.convolve(a, a)
        max_val = float(np.max(conv))
        # Find positions near the max
        threshold = max_val - 0.01 * max_val
        hot_positions = np.where(conv >= threshold)[0]
        
        # Compute subgradient: average of gradients at hot positions
        # d(conv[i])/d(a[j]) = a[i-j] if 0 <= i-j < n
        # For each hot position i, grad_i[j] = a[i-j] (if valid)
        # Average over hot positions
        grad = np.zeros(n)
        for i in hot_positions:
            for j in range(n):
                k = i - j
                if 0 <= k < n:
                    grad[j] += a[k]
        grad /= len(hot_positions)
        
        # Take step
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0.0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.01, 0.1)
        else:
            lr *= 0.9
    
    return best_a, best_obj
```

```python
"""
Optimized step function coefficient search using LP-based alternating
minimization, Adam soft-max gradient descent, and multi-scale block refinement.
Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def build_A(a):
    """Build (2n-1) x n matrix A such that conv(a, g) = A @ g."""
    n = len(a)
    i_idx = np.arange(2 * n - 1)[:, None]
    k_idx = np.arange(n)[None, :]
    diff = i_idx - k_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    d = diff[valid]
    A[valid] = a[d]
    return A


def solve_lp_alternate(a):
    """Solve LP to find best direction g for alternating minimization."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    if M <= 0:
        return None
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    upper = float(np.max(a)) * 2.0
    if upper < 1e-6:
        upper = 1.0
    bounds = [(0.0, upper)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.status == 0:
        return res.x
    return None


def lp_phase(a, max_time):
    """LP-based alternating minimization phase."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    while time.monotonic() - start < max_time:
        g = solve_lp_alternate(a)
        if g is None:
            break
        g_sum = float(np.sum(g))
        if g_sum < 1e-10:
            break
        g = g / g_sum
        a_sum = float(np.sum(a))
        g = g * a_sum  # Match the sum
        
        # Try multiple step sizes
        found = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, None)
            s = np.sum(a_new)
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_a = a_new.copy()
                best_obj = obj
                a = a_new
                found = True
                stall = 0
                break
        if not found:
            stall += 1
            if stall > 15:
                break
    
    return best_a, best_obj


def softmax_gd(a, max_time, T_init=None, lr=0.05):
    """Adam gradient descent on soft-max of convolution."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if T_init is None:
        conv = np.convolve(a, a)
        spread = float(np.max(conv) - np.min(conv))
        T_init = max(0.01 * spread, 1e-4)
    
    T = T_init
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    last_improve = 0
    
    while time.monotonic() - start < max_time:
        conv = np.convolve(a, a)
        max_c = float(np.max(conv))
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500.0, 500.0)
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        p = exp_vals / sum_exp
        
        # Gradient
        idx = np.arange(n)[:, None] + np.arange(n)[None, :]
        M_mat = p[idx]
        grad = M_mat @ a
        
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = np.clip(a_new, 0.0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            lr *= 0.5
            t = 0
            m = np.zeros(n)
            v = np.zeros(n)
            continue
        a_new = a_new / s
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            last_improve = t
            if T > 0.001:
                T = max(0.001, T * 0.998)
        else:
            lr *= 0.97
            if lr < 1e-12:
                break
            if t - last_improve > 200:
                T = max(0.001, T * 0.8)
                lr = max(lr, 0.01)
                t = last_improve
                m = np.zeros(n)
                v = np.zeros(n)
    
    return best_a, best_obj


def block_refine(a, rng, max_time):
    """Block perturbation local search."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

def init_random(n, rng):
    a = np.maximum(0, rng.normal(1.0, 0.5, n))
    return normalize(a)


def init_tapered(n, rng):
    """Tapered (cosine) initialization."""
    x = np.linspace(0, np.pi, n)
    a = 0.5 * (1 + np.cos(x))
    a += 0.1 * rng.random(n)
    return normalize(a)


def init_edge_boost(n, rng):
    """Boost edges, reduce center."""
    center = n / 2.0
    a = np.exp(-((np.arange(n) - center) ** 2) / (2 * (n * 0.3) ** 2))
    a[0] = 1.0
    a[-1] = 1.0
    a += 0.05 * rng.random(n)
    return normalize(a)


def init_cyclic(n, rng):
    """Cyclic frequency initialization."""
    freq = rng.uniform(0.1, 0.5)
    phase = rng.uniform(0, 2 * np.pi)
    a = np.maximum(0, 0.5 + 0.5 * np.cos(2 * np.pi * freq * np.arange(n) + phase))
    a += 0.05 * rng.random(n)
    return normalize(a)


def init_parent(parent, n, rng):
    """Extend or truncate parent to length n."""
    if len(parent) == n:
        a = parent.copy()
    elif len(parent) > n:
        a = parent[:n].copy()
    else:
        # Extend by tiling or padding
        repeats = (n // len(parent)) + 1
        a = np.tile(parent, repeats)[:n].copy()
    a += 0.02 * rng.random(n)
    return normalize(a)
```

```python
"""
Optimized step function coefficient search using LP-based alternating
minimization, Adam soft-max gradient descent, and multi-scale block refinement.
Targets objective below 1.5027 by exploiting larger sequence lengths and
aggressive LP-based max-convolution reduction.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def build_A(a):
    """Build (2n-1) x n matrix A such that convolve(a, g) = A @ g."""
    n = len(a)
    i_idx = np.arange(2 * n - 1)[:, None]
    k_idx = np.arange(n)[None, :]
    diff = i_idx - k_idx
    valid = (diff >= 0) & (diff < n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    d = diff[valid]
    A[valid] = a[d]
    return A


def solve_lp(a):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns (g, ratio)."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    if M <= 0:
        return None, 0.0
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    upper = float(np.max(a)) * 3.0
    if upper < 1e-6:
        upper = 1.0
    bounds = [(0.0, upper)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'primal_feasibility_tolerance': 1e-8})
    if res.status == 0:
        g = res.x
        g_sum = float(np.sum(g))
        a_sum = float(np.sum(a))
        ratio = g_sum / a_sum if a_sum > 1e-10 else 0.0
        return g, ratio
    return None, 0.0


def lp_phase(a, max_time):
    """LP-based alternating minimization. Key phase for reducing max convolution."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        g, ratio = solve_lp(a)
        if g is None:
            break
        if ratio < 1.0 + 1e-4:
            stall += 1
            if stall > 20:
                break
            continue
        stall = 0
        g = g / float(np.sum(g))
        a_sum = float(np.sum(a))
        g = g * a_sum

        found = False
        # Adaptive step sizes: larger when ratio is large
        if ratio > 1.5:
            t_trials = [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]
        elif ratio > 1.1:
            t_trials = [0.8, 0.6, 0.4, 0.3, 0.2, 0.1]
        else:
            t_trials = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]

        for t in t_trials:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, None)
            s = float(np.sum(a_new))
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_a = a_new.copy()
                best_obj = obj
                a = a_new
                found = True
                break
        if not found:
            stall += 1
            if stall > 20:
                break

    return best_a, best_obj


def softmax_gd(a, max_time, T_init=None, lr=0.05):
    """Adam gradient descent on soft-max of convolution."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if T_init is None:
        conv = np.convolve(a, a)
        spread = float(np.max(conv) - np.min(conv))
        T_init = max(0.05 * spread, 1e-4)

    T = T_init
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    last_improve = 0
    idx = np.arange(n)[:, None] + np.arange(n)[None, :]

    while time.monotonic() - start < max_time:
        conv = np.convolve(a, a)
        max_c = float(np.max(conv))
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500.0, 500.0)
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        p = exp_vals / sum_exp

        # Gradient via matrix-vector product
        M_mat = p[idx]
        grad = M_mat @ a

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = np.clip(a_new, 0.0, None)
        s = float(np.sum(a_new))
        if s < 1e-10:
            lr *= 0.5
            t = 0
            m = np.zeros(n)
            v = np.zeros(n)
            continue
        a_new = a_new / s

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            last_improve = t
            if T > 0.0005:
                T = max(0.0005, T * 0.997)
        else:
            lr *= 0.97
            if lr < 1e-12:
                break
            if t - last_improve > 300:
                T = max(0.0005, T * 0.7)
                lr = max(lr, 0.01)
                m = np.zeros(n)
                v = np.zeros(n)
                t = last_improve

    return best_a, best_obj


def block_refine(a, rng, max_time):
    """Block perturbation local search."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < max_time:
        elapsed = time.monotonic() - start
        width = max(1, int(n * rng.uniform(0.02, 0.12)))
        pos = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        sigma = 0.03 + 0.12 * max(0.0, 1.0 - elapsed / max(1.0, max_time))
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.clip(candidate, 0.0, None)
        s = float(np.sum(candidate))
        if s < 1e-10:
            continue
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj

    return best_a, best_obj


def coord_refine(a, max_time):
    """Coordinate-wise refinement: try scaling individual blocks."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while time.monotonic() - start < max_time:
        # Pick a random block
        width = int(np.random.randint(1, max(2, n // 20)))
        pos = int(np.random.randint(0, n - width + 1))
        # Try several multipliers
        best_local = None
        best_local_obj = best_obj
        for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
            candidate = best_a.copy()
            candidate[pos:pos + width] *= mult
            candidate = np.clip(candidate, 0.0, None)
            s = float(np.sum(candidate))
            if s < 1e-10:
                continue
            obj = compute_obj(candidate)
            if obj < best_local_obj:
                best_local_obj = obj
                best_local = candidate
        if best_local is not None:
            best_a = best_local
            best_obj = best_local_obj

    return best_a, best_obj


def init_uniform(n, rng):
    a = np.ones(n) + 0.1 * rng.random(n)
    return normalize(a)


def init_tapered(n, rng):
    x = np.linspace(0, np.pi, n)
    a = 0.5 * (1.0 + np.cos(x))
    a += 0.05 * rng.random(n)
    return normalize(a)


def init_edge(n, rng):
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((np.arange(n) - center) / (n * 0.35)) ** 2)
    a[0] = 1.5
    a[-1] = 1.5
    if n > 4:
        a[1] = 1.2
        a[-2] = 1.2
    a += 0.03 * rng.random(n)
    return normalize(a)


def init_cyclic(n, rng):
    freq = rng.uniform(0.1, 0.4)
    phase = rng.uniform(0, 2 * np.pi)
    a = np.maximum(0, 0.5 + 0.5 * np.cos(2 * np.pi * freq * np.arange(n) + phase))
    a += 0.05 * np.maximum(0, rng.random(n))
    return normalize(a)


def init_random(n, rng):
    a = np.maximum(0.01, rng.normal(1.0, 0.6, n))
    return normalize(a)


def init_parent_ext(parent, n, rng):
    """Extend or truncate parent to length n with slight noise."""
    if parent.size == n:
        a = parent.copy().astype(np.float64)
    elif parent.size > n:
        a = parent[:n].copy().astype(np.float64)
    else:
        reps = (n // parent.size) + 1
        a = np.tile(parent, reps)[:n].astype(np.float64)
    a = np.clip(a, 0.0, None)
    a += 0.02 * rng.random(n)
    return normalize(a)


def full_pipeline(a_init, time_budget, rng, use_lp=True):
    """Full optimization pipeline: GD -> LP -> GD -> LP -> refine."""
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: Initial soft-max GD
    t1 = min(time_budget * 0.2, 25.0)
    remaining1 = time_budget * 0.2 - (time.monotonic() - start)
    if remaining1 > 1.0:
        a, obj = softmax_gd(a, remaining1)
        if obj < best_obj:
            best_a, best_obj = a, obj

    # Phase 2: LP alternating (main workhorse)
    if use_lp:
        t2 = min(time_budget * 0.45, 50.0)
        remaining2 = time_budget * 0.65 - (time.monotonic() - start)
        t2 = min(t2, max(0.0, remaining2))
        if t2 > 1.0:
            a, obj = lp_phase(a, t2)
            if obj < best_obj:
                best_a, best_obj = a, obj

    # Phase 3: Fine GD
    remaining3 = time_budget * 0.80 - (time.monotonic() - start)
    if remaining3 > 1.0:
        a, obj = softmax_gd(a, remaining3, T_init=0.003)
        if obj < best_obj:
            best_a, best_obj = a, obj

    # Phase 4: LP again
    if use_lp:
        remaining4 = time_budget * 0.90 - (time.monotonic() - start)
        if remaining4 > 1.0:
            a, obj = lp_phase(a, remaining4)
            if obj < best_obj:
                best_a, best_obj = a, obj

    # Phase 5: Block refine + coord refine
    remaining5 = time_budget - (time.monotonic() - start)
    if remaining5 > 0.5:
        half = remaining5 * 0.6
        a, obj = block_refine(best_a, rng, half)
        if obj < best_obj:
            best_a, best_obj = a, obj
        remaining5b = time_budget - (time.monotonic() - start)
        if remaining5b > 0.5:
            a, obj = coord_refine(best_a, remaining5b)
            if obj < best_obj:
                best_a, best_obj = a, obj

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization combining LP alternating, Adam soft-max GD,
    and block/coordinate refinement. Tries multiple n values (150-500) with
    diverse initializations to find the best non-negative sequence.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 10.0

    best_seq = None
    best_score = np.inf

    # Try parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        print("No parent available.")

    # Optimize parent
    if parent is not None and 50 <= parent.size <= 500:
        time_left = deadline - time.monotonic()
        if time_left > 80:
            p_time = min(100.0, time_left * 0.10)
            p_opt, p_obj = full_pipeline(parent.copy(), p_time, rng)
            if p_obj < best_score:
                best_score = p_obj
                best_seq = p_opt.copy()
                print(f"Parent optimized (n={parent.size}): obj={p_obj:.6f}")

    # Main search over n values
    n_values = [150, 200, 250, 300, 350, 400, 450, 500]

    # Include parent size if useful
    if parent is not None:
        pn = int(parent.size)
        if pn not in n_values and 100 <= pn <= 600:
            n_values.append(pn)
            n_values.sort()

    for n in n_values:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 35:
            break

        num_remaining = len([x for x in n_values if x >= n])
        time_per_n = remaining / max(1, num_remaining)
        time_per_n = min(time_per_n, 130.0)

        num_restarts = max(2, min(4, int(time_per_n / 28)))
        time_per_restart = time_per_n / num_restarts

        print(f"\nN={n}: remaining={remaining:.0f}s, {num_restarts} restarts, "
              f"~{time_per_restart:.0f}s each")

        for r in range(num_restarts):
            if time.monotonic() > deadline - 60:
                break

            t_start = time.monotonic()

            # Choose initialization strategy
            if r == 0 and parent is not None:
                a_init = init_parent_ext(parent, n, rng)
            elif r == 1:
                a_init = init_tapered(n, rng)
            elif r == 2:
                a_init = init_edge(n, rng)
            elif r == 3:
                ch = int(rng.integers(0, 3))
                if ch == 0:
                    a_init = init_random(n, rng)
                elif ch == 1:
                    a_init = init_cyclic(n, rng)
                else:
                    a_init = init_uniform(n, rng)
            else:
                a_init = init_random(n, rng)

            a_opt, obj_opt = full_pipeline(a_init, time_per_restart, rng)
            elapsed = time.monotonic() - t_start

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} r={r}: obj={obj_opt:.6f} ** NEW BEST ** ({elapsed:.1f}s)")
            else:
                print(f"  n={n} r={r}: obj={obj_opt:.6f} ({elapsed:.1f}s)")

    # Final refinement on best found
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: {remaining:.0f}s remaining, "
              f"best={best_score:.6f}, n={len(best_seq)}")

        # Extended LP
        lp_time = min(remaining * 0.35, 150.0)
        a_ref, obj_ref = lp_phase(best_seq.copy(), lp_time)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  After final LP: obj={best_score:.6f}")

        # GD polish
        if time.monotonic() < deadline - 15:
            gd_time = deadline - time.monotonic() - 15.0
            gd_time = min(gd_time, remaining * 0.25)
            if gd_time > 2.0:
                a_ref, obj_ref = softmax_gd(best_seq, gd_time, T_init=0.002, lr=0.02)
                if obj_ref < best_score:
                    best_score = obj_ref
                    best_seq = a_ref.copy()
                    print(f"  After final GD: obj={best_score:.6f}")

        # Block refine
        if time.monotonic() < deadline - 8:
            bp_time = deadline - time.monotonic() - 8.0
            if bp_time > 1.0:
                a_ref, obj_ref = block_refine(best_seq, rng, bp_time)
                if obj_ref < best_score:
                    best_score = obj_ref
                    best_seq = a_ref.copy()
                    print(f"  After final BP: obj={best_score:.6f}")

        # One last LP
        if time.monotonic() < deadline - 3:
            lp_time2 = deadline - time.monotonic() - 2.0
            if lp_time2 > 1.0:
                a_ref, obj_ref = lp_phase(best_seq, lp_time2)
                if obj_ref < best_score:
                    best_score = obj_ref
                    best_seq = a_ref.copy()
                    print(f"  After final LP2: obj={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
```
