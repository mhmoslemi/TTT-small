Optimized step function search using LP-based refinement, pressure updates,
and spectral shaping to minimize 2*n*max(autocorrelation)/sum^2.
"""

import numpy as np
from scipy.optimize import linprog
import time

def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s

def compute_score(f):
    """Compute 2*n*max(conv)/sum^2 for the given sequence."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (f.sum() ** 2)

def score_normalized(f):
    """Compute score assuming f is normalized to sum=1."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()

def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b)."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            A[k, i] = f[k - i]
    return A

def pressure_update(f, n_iter=100, eta=0.1, rng=None):
    """Pressure-based multiplicative update."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = score_normalized(f)
    stall = 0
    
    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        k_star = np.argmax(conv)
        
        # Compute pressure
        pressure = np.zeros(n)
        for i in range(n):
            idx = k_star - i
            if 0 <= idx < n:
                pressure[i] = f[idx]
        
        if pressure.max() > 0:
            pressure /= pressure.max()
        
        eta_eff = eta / (1.0 + 0.01 * it)
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)
        
        s = score_normalized(f_new)
        if s < best_score - 1e-12:
            best_score = s
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                # Perturb
                f_pert = f * (1 + 0.2 * rng.standard_normal(n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if score_normalized(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = score_normalized(f)
                    stall = 0
                else:
                    break
    
    return best_f, best_score

def lp_step(f, M_target, rng=None):
    """Solve LP to find best direction, return normalized g."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:
        return None
    
    if not res.success or res.x is None:
        return None
    
    b = np.clip(res.x, 0, None)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    
    return b / b_sum

def lp_refine(f, time_budget, rng=None, initial_t=0.5):
    """LP-based refinement loop."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    
    best_f = f.copy()
    best_score = score_normalized(f)
    t = initial_t
    start = time.monotonic()
    stagnant = 0
    
    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Try with slightly tighter constraint
        g = lp_step(f, M * 0.999, rng)
        if g is None:
            g = lp_step(f, M, rng)
        
        if g is None:
            break
        
        # Line search
        improved = False
        for step in [min(1.0, t * 2), t, t * 0.5, t * 0.25, 0.05]:
            f_new = (1 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_score = score_normalized(f_new)
            if new_score < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                stagnant = 0
                t = min(1.0, t * 1.1)
                improved = True
                break
        
        if not improved:
            stagnant += 1
            t *= 0.7
            if t < 0.01:
                # Perturb
                f_pert = f + rng.standard_normal(n) * 0.05
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                sp = score_normalized(f_pert)
                if sp < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = sp
                    t = 0.3
                    stagnant = 0
                elif stagnant > 50:
                    break
    
    return best_f, best_score

def coordinate_descent_polish(f, n_sweeps=5, rng=None):
    """Polish with coordinate descent."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    best_score = score_normalized(f)
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            orig = f[i]
            if orig < 1e-12:
                continue
            best_val = orig
            best_local = best_score
            
            for factor in [0.5, 0.7, 0.85, 1.18, 1.4, 2.0]:
                f[i] = orig * factor
                s = f.sum()
                if s < 1e-10:
                    f[i] = orig
                    continue
                # Quick score computation
                conv = np.convolve(f, f)
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]
            
            f[i] = best_val
        
        f = np.clip(f, 0, None)
        f = normalize(f)
        current_score = score_normalized(f)
        if current_score < best_score:
            best_score = current_score
    
    return f, best_score

    """Generate diverse initial sequence."""
    method = rng.integers(0, 5)
    
    if method == 0:
        # Gaussian
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.15, 0.35)

    elif method == 1:

        x = np.arange(n) / (n - 1)
        f = 1.0 / (0.1 + np.abs(x - 0.5)) ** 0.5
    elif method == 3:
        # Two bumps
        c1 = n * rng.uniform(0.25, 0.4)
        c2 = n * rng.uniform(0.6, 0.75)
        w = n * rng.uniform(0.05, 0.15)

    else:
        # Random smooth
        f = np.ones(n)
        for _ in range(3):
            c = int(rng.integers(0, n))
            w = max(3, int(n * rng.uniform(0.05, 0.2)))
            kernel = np.exp(-0.5 * (np.arange(n) - c) ** 2 / w ** 2)
            f *= (1 + 0.5 * rng.standard_normal() * kernel)
        f = np.abs(f) + 0.1
    
    f = np.clip(f, 0.01, None)
    f = normalize(f)
    # Break symmetry
    f = f * (1 + 0.05 * rng.standard_normal(n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f

"""
Optimized step function search using aggressive LP-based refinement,
pressure updates, spectral shaping, and coordinate descent polishing
to minimize 2*n*max(autocorrelation)/sum^2.

import numpy as np
from scipy.optimize import linprog
import time


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def score_normalized(f):
    """Compute 2*n*max(conv) assuming f is normalized to sum=1."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            A[k, i] = f[k - i]
    return A


def pressure_update(f, n_iter=200, eta=0.15, rng=None):
    """Pressure-based multiplicative update to reduce convolution peaks."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = score_normalized(f)
    stall = 0

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        k_star = int(np.argmax(conv))

        # Compute pressure from top peaks
        pressure = np.zeros(n)
        # Main peak
        for i in range(n):
            idx = k_star - i
            if 0 <= idx < n:
                pressure[i] = f[idx]

        # Secondary peaks (if close to max)
        conv_sorted = np.sort(conv)[::-1]
        n_secondary = 0
        for s_idx in range(1, min(5, len(conv_sorted))):
            if conv_sorted[s_idx] > 0.85 * M:
                # Find position of this peak

                candidates = np.where(mask)[0]
                if len(candidates) > 0:
                    k2 = int(candidates[0])
                    if k2 != k_star:
                        for i in range(n):
                            idx = k2 - i
                            if 0 <= idx < n:
                                pressure[i] += 0.3 * f[idx]
                        n_secondary += 1
            if n_secondary >= 3:
                break

        if pressure.max() > 0:
            pressure /= pressure.max()

        eta_eff = eta / (1.0 + 0.005 * it)
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        s = score_normalized(f_new)
        if s < best_score - 1e-12:
            best_score = s

            stall = 0

                if sp < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = sp
                    stall = 0
                else:

    return best_f, best_score


def solve_lp(f, M_target):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M_target, b >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    b = np.clip(res.x, 0, None)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    return b / b_sum


def lp_refine(f, time_budget, rng=None, initial_t=0.5):
    """Aggressive LP-based refinement loop with adaptive step sizing."""
    if rng is None:
        rng = np.random.default_rng(0)

def coordinate_descent_polish(f, n_sweeps=10, rng=None):
    """Two-phase coordinate descent polishing."""
    if rng is None:
        rng = np.random.default_rng(0)

    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    best_score = score_normalized(f)

    coarse_factors = [0.4, 0.6, 0.8, 1.25, 1.6, 2.5]
    fine_factors = [0.9, 0.95, 1.05, 1.1]

    for sweep in range(n_sweeps):
        factors = coarse_factors if sweep < n_sweeps // 2 else fine_factors
        order = rng.permutation(n)
        for i in order:
            orig = f[i]
            if orig < 1e-12:
                continue
            best_val = orig
            best_local = best_score

            for factor in factors:
                f[i] = max(0.0, orig * factor)
                s = f.sum()
                if s < 1e-10:
                    f[i] = orig
                    continue
                conv = np.convolve(f, f)
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local - 1e-14:
                    best_local = score
                    best_val = f[i]

            f[i] = best_val

        f = np.clip(f, 0, None)
        f = normalize(f)
        current_score = score_normalized(f)
        if current_score < best_score:
            best_score = current_score

    return f, best_score


def spectral_shaping(f, n_iter=100, rng=None):
    """FFT-based spectral flattening with non-negativity projection."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = score_normalized(f)
    stall = 0

    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)

        # Target: equalize magnitude (keep DC fixed)
        n_bins = len(mag)
        current_energy = np.sum(f ** 2)
        target_mag = np.sqrt(current_energy * n_fft / n_bins)

        mag_new = np.full_like(mag, target_mag)
        mag_new[0] = 1.0  # DC = sum(f) = 1

        # Blend with current to avoid oscillation
        blend = 0.3 + 0.4 * (it / max(1, n_iter))
        mag_target = (1 - blend) * mag + blend * mag_new
        mag_target[0] = 1.0

        F_new = mag_target * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new /= s

        score = score_normalized(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                # Phase randomization to escape
                phase += rng.standard_normal(len(phase)) * 0.1
                if stall > 100:
                    break

    return best_f, best_score


def diverse_init(n, rng):

    elif method == 1:

    elif method == 4:
        # Random smooth product
        f = np.ones(n)
        for _ in range(4):
            c = int(rng.integers(0, n))
            w = max(3, int(n * rng.uniform(0.05, 0.2)))
            kernel = np.exp(-0.5 * (np.arange(n) - c) ** 2 / w ** 2)
            f *= (1 + 0.4 * rng.standard_normal() * kernel)

        x = np.arange(n) / max(1, n - 1)

    f = normalize(f)
    # Break symmetry with small noise
    f = f * (1 + 0.05 * rng.standard_normal(n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f
