"""
Optimized subgradient + coordinate descent for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses adaptive top-K subgradient with backtracking line search, simplex projection,
and exact 1D coordinate descent for polish. Multiple initializations and restarts.
"""
import numpy as np
import time

def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x) = target_sum, x >= 0}."""
    n = len(v)
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_topk(f, K=10, threshold_ratio=0.95):
    """Compute subgradient using top-K lags and lags above threshold ratio of max."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    
    # Find active lags: top K or above threshold
    active = np.where(conv >= max(M * threshold_ratio, 0))[0]
    if len(active) < K:
        active = np.argsort(conv)[-K:]
    
    # Weight by how close to max
    weights = np.maximum(conv[active] - conv.min(), 0)
    # Or: weight = 1 for all active (uniform weight on active set)
    # Let's use weight proportional to conv value for active lags
    w = conv[active]
    w = w / (w.sum() + 1e-30)
    
    grad = np.zeros(n)
    for k, wi in zip(active, w):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        # grad[j] += w * 2 * f[k-j]
        grad[j_min:j_max+1] += wi * 2.0 * f[n-1-j_max:n-1-j_min+1][::-1]
    
    return grad, M, active

def line_search_subgrad(f, grad, eta0, max_iter_ls=30):
    """Backtracking line search for subgradient method."""
    n = len(f)
    S = float(np.sum(f))
    f_norm = np.copy(f)
    f_norm = f_norm / S  # normalize to sum=1
    
    score0 = compute_score(f_norm)
    eta = eta0
    
    for _ in range(max_iter_ls):
        f_trial = f_norm - eta * grad / S
        f_trial = project_simplex(f_trial, 1.0)
        
        if np.all(f_trial == 0):
            eta *= 0.5
            continue
        
        score_trial = compute_score(f_trial)
        
        if score_trial < score0:
            return f_trial, score_trial, eta
        else:
            eta *= 0.7
    
    return None, score0, eta

"""
Active-set linear system + subgradient optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key innovation: solve the KKT linear system A*f = c*1 for the current active set (much faster
than LP), combined with smoothed subgradient steps and periodic coordinate descent polish.
Uses FFT-accelerated convolution and adaptive active set management.
"""
import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import spsolve

def compute_score(f):
    """Fast objective computation (assumes f is numpy array, non-negative)."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))

def project_simplex(v, target_sum=1.0):
    """Project v onto {x: sum(x) = target_sum, x >= 0}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_candidates) == 0:
        return np.full(n, target_sum / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_active_set(S_list, n):
    """Solve the KKT linear system A*f = (|S|/n)*1 where A[j,m] = 1 if m+j in S."""
    nS = len(S_list)
    rows_all = []
    cols_all = []
    for k in S_list:
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            m = k - j
            rows_all.append(j)
            cols_all.append(m)
    
    if len(rows_all) == 0:
        return np.ones(n) / n
    
    A = csr_matrix((np.ones(len(rows_all)), (rows_all, cols_all)), shape=(n, n))
    b = np.full(n, nS / n)
    
    try:
        x = spsolve(A + 1e-8 * csr_matrix(np.eye(n)), b)
        if np.all(np.isfinite(x)):
            return x
    except Exception:
        pass
    
    # Fallback: use numpy lstsq on dense matrix (only for small n)
    try:
        A_dense = A.toarray()
        x, _, _, _ = np.linalg.lstsq(A_dense + 1e-8 * np.eye(n), b, rcond=None)
        return x
    except Exception:
        return None

def smoothed_subgradient(f, T=50.0):
    """Compute smoothed subgradient using softmax weighting over convolution lags."""
    n = len(f)
    conv = np.convolve(f, f)
    # Softmax weights
    conv_shifted = conv - np.max(conv)
    weights = np.exp(T * conv_shifted)
    weights = weights / (np.sum(weights) + 1e-30)
    
    grad = np.zeros(n)
    # Only use lags with significant weight for efficiency
    significant = weights > 1e-10
    for k in np.where(significant)[0]:
        wk = weights[k]
        if wk < 1e-12:
            continue
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        lo = k - j_max
        hi = k - j_min
        grad[j_min:j_max + 1] += wk * 2.0 * f[lo:hi + 1][::-1]
    
    return grad, conv

def subgradient_step(f, deadline, T=50.0, eta_init=0.5, verbose_prefix=""):
    """Smoothed subgradient descent with backtracking."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    f = f / S
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    eta = eta_init
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        grad, conv = smoothed_subgradient(f, T)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        improved = False
        for _ in range(20):
            f_trial = f - eta * grad
            f_trial = project_simplex(f_trial, 1.0)
            score_trial = compute_score(f_trial)
            if score_trial < best_score:
                f = f_trial
                best_score = score_trial
                best_f = f.copy()
                no_improve = 0
                it += 1
                improved = True
                if it % 2000 == 0:
                    print(f"    {verbose_prefix}subgrad iter {it}: score={best_score:.6f}")
                break
            else:
                eta *= 0.8
        
        if not improved:
            no_improve += 1
            eta *= 0.5
            T *= 2.0  # Make softmax sharper (more like hard max)
            if no_improve > 200 or eta < 1e-7:
                break
    
    return best_f, best_score

def coordinate_descent_polish(f, deadline, max_rounds=5):
    """Coordinate descent: optimize one coordinate at a time using 1D minimax."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    f = f / S
    best_score = compute_score(f)
    rng = np.random.default_rng(777)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(f, f)
        order = rng.permutation(n)
        
        for j in order:
            if time.monotonic() > deadline:
                break
            
            # Update conv efficiently: change f[j] from old_val to new_val
            old_val = f[j]
            f[j] = 0.0
            base = np.convolve(f, f)
            
            # Try several candidate values
            candidates = [0.0, old_val * 0.5, old_val, old_val * 1.5, old_val * 2.0]
            # Add the value from the active-set solution if available
            best_x = old_val
            best_m = float(np.max(base))  # x=0 gives base
            
            for x in candidates:
                if x < 0:
                    continue
                # conv_new[k] = base[k] + 2*x*f[k-j] for k-j valid, k != 2j
                # conv_new[2j] = base[2j] + x^2 if 2j < 2n-1
                # For efficiency, compute max over affected lags only
                m_new = best_m  # start with max of unaffected (conservative)
                # Actually base already has x=0, so base[k] is the value without f[j]
                # Affected k: j <= k <= j+n-1
                k_lo = max(0, j)
                k_hi = min(2 * n - 2, j + n - 1)
                affected = base[k_lo:k_hi + 1].copy()
                # Add the linear/quadratic terms
                for idx, k in enumerate(range(k_lo, k_hi + 1)):
                    kj = k - j
                    if k == 2 * j:
                        affected[idx] += x * x
                    else:
                        affected[idx] += 2.0 * x * f[kj]
                m_candidate = float(np.max(np.maximum(affected, base)))
                # Actually m_candidate should be max(affected_max, max of unaffected)
                unaffected_mask = np.ones(2 * n - 1, dtype=bool)
                unaffected_mask[k_lo:k_hi + 1] = False
                max_unaffected = float(np.max(base[unaffected_mask])) if np.any(unaffected_mask) else 0
                m_candidate = max(float(np.max(affected)), max_unaffected)
                
                if m_candidate < best_m:
                    best_m = m_candidate
                    best_x = x
            
            f[j] = best_x
        
        # Renormalize
        S_new = float(np.sum(f))
        if S_new > 1e-10:
            f = f / S_new
        new_score = compute_score(f)
        if new_score < best_score:
            best_score = new_score
    
    return f, best_score

def make_initializations(n, rng, parent=None):
    """Create diverse initializations."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    # Uniform
    inits.append(('uniform', np.ones(n)))
    
    # Raised cosine (Hann window)
    hann = 0.5 * (1 + np.cos(2 * np.pi * idx / (n - 1)))
    hann = np.maximum(hann, 0.01)
    inits.append(('hann', hann))
    
    # Gaussian (wide)
    gauss = np.exp(-0.5 * ((idx - center) / (n * 0.3)) ** 2)
    inits.append(('gauss_wide', gauss))
    
    # Gaussian (narrow)
    gauss2 = np.exp(-0.5 * ((idx - center) / (n * 0.15)) ** 2)
    inits.append(('gauss_narrow', gauss2))
    
    # Flat-top (super-gaussian)
    flat_top = np.exp(-((idx - center) / (n * 0.4)) ** 8)
    inits.append(('flat_top', flat_top))
    
    # Bimodal
    bimodal = (np.exp(-0.5 * ((idx - n * 0.25) / (n * 0.15)) ** 2) +
               np.exp(-0.5 * ((idx - n * 0.75) / (n * 0.15)) ** 2))
    inits.append(('bimodal', bimodal))
    
    # Random smooth
    r = rng.uniform(0.5, 1.5, size=n)
    # Apply moving average for smoothness
    kernel_size = max(3, n // 50)
    kernel = np.ones(kernel_size) / kernel_size
    r_smooth = np.convolve(r, kernel, mode='same')
    inits.append(('random_smooth', r_smooth))
    
    # Interpolated parent
    if parent is not None and len(parent) > 0:
        if len(parent) != n:
            from numpy import interp
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            parent_interp = interp(x_new, x_old, parent)
        else:
            parent_interp = parent.copy()
        parent_interp = np.maximum(parent_interp, 0)
        inits.append(('parent', parent_interp))
    
    return inits

    # Focus on 1024 (parent length) and also try 2048
    target_lengths = [1024, 2048, 768]
    
    # Phase 1: Quick exploration with multiple inits (first 300s)
    print("\n=== Phase 1: Exploration ===")
    phase1_deadline = t0 + 300.0
    
    candidates = []  # (score, sequence, n, label)
    
    for n in target_lengths:
        if time.monotonic() > phase1_deadline - 60:
            break
        print(f"\n--- Exploring n={n} ---")
        
        inits = make_initializations(n, rng, parent)
        
        for label, f_init in inits:
            if time.monotonic() > phase1_deadline - 20:
                break
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            S_init = float(np.sum(f_init))
            if S_init < 0.01:
                continue
            f_init = f_init / S_init
            
            score_init = compute_score(f_init)
            
            # Quick active-set optimization (30s budget)
            quick_deadline = min(phase1_deadline, time.monotonic() + 30.0)
            f_opt, s_opt = active_set_optimize(f_init, quick_deadline,
                                                verbose_prefix=f"[n={n} {label}] ")
            
            if s_opt < best_score:
                best_score = s_opt
                best_sequence = f_opt.copy()
            
            candidates.append((s_opt, f_opt.copy(), n, label))
            print(f"  {label}: init={score_init:.6f} -> {s_opt:.6f}")
    
    # Sort by score
    candidates.sort(key=lambda x: x[0])
    print(f"\nTop candidates after phase 1:")
    for sc, seq, n, lbl in candidates[:5]:
        print(f"  n={n} {lbl}: {sc:.6f}")
    print(f"Best so far: {best_score:.6f}")
    
    # Phase 2: Focused optimization (next ~900s)
    print(f"\n=== Phase 2: Focused Optimization (best={best_score:.6f}) ===")
    phase2_deadline = deadline - 150.0
    
    # Take top candidates with unique n values
    seen_n = set()
    focus_candidates = []
    for sc, seq, n, lbl in candidates:
        if n not in seen_n:
            focus_candidates.append((sc, seq, n, lbl))
            seen_n.add(n)
        if len(focus_candidates) >= 4:
            break
    
    for sc, f_init, n, lbl in focus_candidates:
        if time.monotonic() > phase2_deadline - 30:
            break
        print(f"\n--- Focused: n={n} {lbl} (start={sc:.6f}) ---")
        
        # Split time: active-set (50%), subgradient (25%), coordinate descent (25%)
        remaining = phase2_deadline - time.monotonic()
        
        # Active-set optimization
        as_deadline = time.monotonic() + remaining * 0.45
        f_as, s_as = active_set_optimize(f_init, as_deadline,
                                          verbose_prefix=f"[AS n={n}] ")
        
        # Subgradient from active-set result
        sg_deadline = time.monotonic() + (phase2_deadline - time.monotonic()) * 0.3
        f_sg, s_sg = subgradient_step(f_as, sg_deadline, T=100.0, eta_init=0.3,
                                       verbose_prefix=f"[SG n={n}] ")
        
        # Coordinate descent polish
        cd_deadline = min(phase2_deadline, time.monotonic() + (phase2_deadline - time.monotonic()) * 0.3)
        f_cd, s_cd = coordinate_descent_polish(f_sg, cd_deadline, max_rounds=3)
        
        # Also try active-set from subgradient result
        f_as2, s_as2 = active_set_optimize(f_sg, phase2_deadline,
                                            verbose_prefix=f"[AS2 n={n}] ")
        
        local_best = min(s_as, s_sg, s_cd, s_as2)
        if local_best < best_score:
            best_score = local_best
            # Find which sequence gave the best
            best_local = f_as if local_best == s_as else (f_sg if local_best == s_sg else (f_cd if local_best == s_cd else f_as2))
            best_sequence = best_local.copy()
            print(f"  Improved to: {best_score:.6f}")
    
    # Phase 3: Final polish (last 150s)
    print(f"\n=== Phase 3: Final Polish (best={best_score:.6f}) ===")
    if best_sequence is not None and time.monotonic() < deadline - 10:
        n_best = len(best_sequence)
        print(f"Polishing n={n_best}...")
        
        # Active-set polish
        f_p1, s_p1 = active_set_optimize(best_sequence, deadline,
                                          verbose_prefix="[polish-AS] ")
        if s_p1 < best_score:
            best_score = s_p1
            best_sequence = f_p1.copy()
        
        # Subgradient polish (smoothed)
        f_p2, s_p2 = subgradient_step(best_sequence, deadline, T=200.0, eta_init=0.1,
                                       verbose_prefix="[polish-SG] ")
        if s_p2 < best_score:
            best_score = s_p2
            best_sequence = f_p2.copy()
        
        # Coordinate descent
        f_p3, s_p3 = coordinate_descent_polish(best_sequence, deadline, max_rounds=5)
        if s_p3 < best_score:
            best_score = s_p3
            best_sequence = f_p3.copy()
    
    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(1024)
    
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    final_score = evaluate_sequence([float(x) for x in best_sequence])
    if not np.isfinite(final_score):
        print("WARNING: Final sequence failed! Using fallback.")
        best_sequence = np.ones(512)
        final_score = evaluate_sequence(best_sequence.tolist())
    
    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, time={elapsed:.1f}s")
    
    return [float(x) for x in best_sequence]
