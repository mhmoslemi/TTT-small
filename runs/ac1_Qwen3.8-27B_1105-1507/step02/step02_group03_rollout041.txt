"""
Optimized step function search using smooth gradient descent (soft-max
approximation of the max-autocorrelation objective), LP-based refinement,
and coordinate descent polishing. Uses FFT-based gradient computation for
efficiency. Tries multiple sequence lengths and initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def conv_fft(f):
    """Compute full linear convolution of f with itself using FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def score_of(f):
    """Compute 2*n*max(conv)/sum^2 for a sequence."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = conv_fft(f)
    return 2.0 * n * c.max() / (s * s)


def smooth_gradient(f, beta):
    """
    Compute gradient of the soft-max smoothed objective 2n*softmax_beta(conv(f,f)).
    Uses FFT-based correlation for O(n log n) gradient computation.
    """
    n = len(f)
    c = conv_fft(f)
    mc = c.max()
    b = beta * (c - mc)
    bm = b.max()
    e = np.exp(b - bm)
    w = e / e.sum()
    # w has length 2n-1
    # grad[i] = 2n * sum_{j=0}^{n-1} w[i+j] * f[j]
    # Using: (W@f)[i] = (h * f)[2n-2-i] where h = w[::-1]
    h = w[::-1]
    # Full convolution of h (len 2n-1) and f (len n) has length 3n-1
    size = 1 << (3 * n - 1).bit_length()
    H = np.fft.rfft(h, size)
    F = np.fft.rfft(f, size)
    conv_full = np.fft.irfft(H * F, size)[:3 * n - 1]
    result = conv_full[n - 1:2 * n - 1][::-1]
    return 2.0 * n * result


def smooth_score(f, beta):
    """Compute soft-max smoothed score (for monitoring)."""
    n = len(f)
    c = conv_fft(f)
    mc = c.max()
    b = beta * (c - mc)
    bm = b.max()
    e = np.exp(b - bm)
    soft_max = mc + bm / beta + np.log(e.sum()) / beta
    return 2.0 * n * soft_max


def grad_descent(f, time_budget, beta_start=5.0, beta_end=200.0, rng=None):
    """
    Smooth gradient descent with multiplicative updates and adaptive step size.
    Anneals beta from beta_start to beta_end over the time budget.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(f)
    f = np.array(f, dtype=np.float64)
    f = normalize(f)
    best_f = f.copy()
    best_s = 2.0 * n * conv_fft(f).max()

    t0 = time.monotonic()
    eta = 1.0 / n
    step = 0
    no_improve = 0

    while time.monotonic() - t0 < time_budget:
        progress = min(1.0, (time.monotonic() - t0) / time_budget)
        beta = beta_start + (beta_end - beta_start) * progress ** 0.7

        g = smooth_gradient(f, beta)

        # Multiplicative update (maintains non-negativity)
        f_new = f * np.exp(-eta * g)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        s_new = 2.0 * n * conv_fft(f_new).max()

        if s_new < best_s - 1e-10:
            best_s = s_new
            best_f = f_new.copy()
            f = f_new
            eta = min(eta * 1.03, 3.0 / n)
            no_improve = 0
        else:
            eta *= 0.88
            no_improve += 1
            if no_improve > 300:
                # Perturb to escape local minimum
                scale = 0.1 + 0.1 * progress
                f = f * (1.0 + scale * rng.normal(0, 1, n))
                f = np.clip(f, 1e-8, None)
                f = normalize(f)
                eta = 1.0 / n
                no_improve = 0

        step += 1
        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GD[{step}]: beta={beta:.1f}, best={best_s:.6f}, eta={eta:.2e}, elapsed={elapsed:.0f}s")

    return best_f, best_s


def build_conv_matrix(f):
    """Build matrix A such that A @ g = np.convolve(f, g, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def lp_solve(f, M, time_limit=5):
    """
    Solve LP: maximize sum(g) subject to f*g <= M, g >= 0.
    Returns normalized g or None.
    """
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.clip(res.x, 0, None)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_refine(f, time_budget, rng=None):
    """
    LP-based refinement: iteratively solve LP and blend.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(f)
    f = np.array(f, dtype=np.float64)
    f = normalize(f)
    best_f = f.copy()
    best_s = 2.0 * n * conv_fft(f).max()

    t0 = time.monotonic()
    t_blend = 0.4
    stall = 0

    while time.monotonic() - t0 < time_budget:
        c = conv_fft(f)
        M = c.max()

        # Determine LP time limit based on remaining time
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 10:
            break
        lp_tl = min(5, max(1, int(remaining * 0.1)))

        g = lp_solve(f, M, time_limit=lp_tl)
        if g is None:
            break

        # Try multiple blend fractions
        improved = False
        for t in [t_blend, 0.5, 0.3, 0.2, 0.15, 0.1, 0.05]:
            f_new = (1.0 - t) * f + t * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            s_new = 2.0 * n * conv_fft(f_new).max()
            if s_new < best_s - 1e-10:
                best_s = s_new
                best_f = f_new.copy()
                f = f_new
                t_blend = min(1.0, t * 1.2)
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 5:
                # Perturb
                f = f + 0.03 * rng.normal(0, 1, n)
                f = np.clip(f, 0, None)
                f = normalize(f)
                t_blend = 0.2
                stall = 0
            if stall > 20:
                break

    return best_f, best_s


def coordinate_descent(f, n_sweeps=5, rng=None):
    """Coordinate descent with geometric search for optimal value at each coordinate."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(f)
    f = np.array(f, dtype=np.float64)
    f = normalize(f)
    best_s = 2.0 * n * conv_fft(f).max()
    multipliers = [0.4, 0.6, 0.8, 0.9, 1.0 / 0.9, 1.0 / 0.8, 1.0 / 0.6, 1.0 / 0.4]

    for sweep in range(n_sweeps):
        for i in rng.permutation(n):
            orig = f[i]
            if orig < 1e-10:
                continue
            best_val = orig
            best_local = best_s
            for mult in multipliers:
                f[i] = max(0.0, orig * mult)
                s = f.sum()
                if s < 1e-10:
                    f[i] = orig
                    continue
                c = conv_fft(f)
                sc = 2.0 * n * c.max() / (s * s)
                if sc < best_local:
                    best_local = sc
                    best_val = f[i]
            f[i] = best_val
        f = normalize(f)
        best_s = 2.0 * n * conv_fft(f).max()

    return f, best_s


def diverse_init(n, rng):
    """Generate a diverse non-negative initial sequence."""
    r = rng.random()
    if r < 0.2:

    elif r < 0.4:
        # Cosine window
        f = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / n))
    elif r < 0.6:
        # Product of sines
        freq = rng.uniform(1, 3)
        f = np.abs(np.sin(np.pi * freq * np.arange(n) / n))
        f += 0.1
    elif r < 0.8:
        # Smooth random
        f = np.ones(n)
        for _ in range(4):
            center = int(rng.integers(0, n))
            width = max(2, int(n * rng.uniform(0.05, 0.2)))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (width * width))
            f *= (1.0 + 0.5 * rng.standard_normal() * kernel)
        f = np.abs(f) + 0.05
    else:
        # Triangle with noise
        f = 1.0 - np.abs(2.0 * np.arange(n) / n - 1.0)
        f += 0.1 * rng.normal(0, 1, n)

    f = np.clip(f, 0.01, None)
    f = normalize(f)
    f = f * (1.0 + 0.03 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f


def spectral_init(n, rng):
    """Initialize via spectral method: start with flat magnitude, random phase."""
    size = 1 << (2 * n - 1).bit_length()
    # Flat magnitude with DC fixed
    mag = np.full(size // 2 + 1, 1.0 / np.sqrt(n))
    mag[0] = 1.0
    phase = rng.uniform(0, 2 * np.pi, size // 2 + 1)
    phase[0] = 0.0
    F = mag * np.exp(1j * phase)
    f = np.fft.irfft(F, size)[:n]
    f = np.clip(f, 0, None)
    if f.sum() < 1e-10:
        f = np.ones(n)
    f = normalize(f)
    return f
