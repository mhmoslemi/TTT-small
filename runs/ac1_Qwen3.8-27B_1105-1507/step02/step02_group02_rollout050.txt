import time


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_conv_matrix(a):
    """Build Toeplitz-like matrix A where (A @ x) = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2 directly."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None


def lp_iteration(a, deadline, max_iters=2000):
    """LP best-response with adaptive M and step size."""
    best = a.copy()
    best_score = compute_score(a)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a, best_score
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        
        # Adaptive M: increase slightly in early iterations for exploration
        if it < 20:
            M_eff = M * 1.2
        elif it < 50:
            M_eff = M * 1.05
        else:
            M_eff = M
        
        g = lp_best_response(a, M_eff)
        if g is None:
            g = lp_best_response(a, M * 1.5)
            if g is None:
                break
        g = g / g.sum()
        
        # Step size: starts moderate, decays
        t = np.maximum(0.1, 0.4 * (0.97 ** it))
        
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best = a_new.copy()
            a = a_new
            stall_count = 0
        else:
            stall_count += 1
            # Backtrack
            t2 = t * 0.3
            a_new2 = (1.0 - t2) * a + t2 * g
            a_new2 = np.clip(a_new2, 0, None)
            if a_new2.sum() > 1e-10:
                a_new2 = a_new2 / a_new2.sum()
                score2 = compute_score(a_new2)
                if score2 < best_score:
                    best_score = score2
                    best = a_new2.copy()
                    stall_count = 0
                a = a_new2
            
            # If stalled for a while, perturb
            if stall_count > 30:
                a = best + 0.02 * np.random.default_rng(it).standard_normal(len(a))
                a = np.clip(a, 0, None)
                if a.sum() > 1e-10:
                    a = a / a.sum()
                else:
                    a = best.copy()
                stall_count = 0
    
    return best, best_score


def subgradient_refine(a, deadline, max_iters=500):
    """Subgradient descent on the max autocorrelation."""
    best = a.copy()
    best_score = compute_score(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a, best_score
    a = a / s
    
    n = len(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        max_b = b.max()
        # Find all positions achieving the max (for subgradient)
        max_pos = np.where(b >= max_b * (1 - 1e-8))[0]
        
        # Subgradient of max(b) w.r.t. a
        # ∂b_k/∂a_i = a[i] contributes to b[k] and b[k+1] ... 
        # Actually: b[k] = sum_j a[j]*a[k-j]
        # ∂b[k]/∂a[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
        grad = np.zeros(n)
        for k in max_pos:
            # a[i] appears in b[k] as a[k-i]*a[i] + a[i]*a[k-i] when both indices valid
            # More precisely: b[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
            # ∂b[k]/∂a[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
            # But if k-i == i, i.e., k = 2i, then it's 2*a[i]... no.
            # b[k] = sum_j a[j] * a[k-j]
            # ∂b[k]/∂a[i] = a[k-i] (from j=i term) + a[i-k] (from k-j=i, i.e., j=k-i term)
            # But if i == k-i (i.e., k=2i), these are the same term counted twice
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                pass  # This is O(n) per max_pos, too slow for large n
        
        # Better: use the convolution matrix approach
        # If A is the conv matrix, then b = A_sym @ a where (A_sym @ a)[k] = sum_j M[k,j]*a[j]*a[j]
        # Actually, b = conv(a,a), and ∂b[k]/∂a = row k of the "Hessian-like" matrix
        # For efficiency, use FFT: the gradient of b[k] w.r.t. a is:
        # grad_a[i] = a[(k-i) mod n] + a[(i-k) mod n] ... but this is circular conv
        # For linear conv, it's: a[k-i] if 0<=k-i<n else 0, plus a[i-k] if 0<=i-k<n else 0
        
        # Efficient computation using the fact that we only need the sum over max positions
        # grad[i] = sum_{k in max_pos} (a[k-i] if 0<=k-i<n else 0 + a[i-k] if 0<=i-k<n else 0)
        
        # This is equivalent to: grad = conv(indicator_max_pos_reversed, a) + conv(indicator_max_pos, a)
        # where indicator_max_pos is a sequence of 1s at max positions and 0s elsewhere
        
        # Let's use FFT for this
        m = len(b)  # 2n-1
        indicator = np.zeros(m)
        for k in max_pos:
            indicator[k] = 1.0 / len(max_pos)
        
        # grad[i] = sum_k indicator[k] * (a[k-i] * 1[0<=k-i<n] + a[i-k] * 1[0<=i-k<n])
        # First term: sum_k indicator[k] * a[k-i] * 1[0<=k-i<n]
        #   = (indicator * a)_i where * is cross-correlation
        # Second term: sum_k indicator[k] * a[i-k] * 1[0<=i-k<n]
        #   = (indicator * a)_i (same thing, just reversed)
        
        # For linear cross-correlation:
        # (corr1)[i] = sum_k indicator[k] * a[k-i] for valid k
        # This is convolve(indicator, a) shifted... 
        
        # Let me just do it directly for small n
        if n <= 2048:
            for i in range(n):
                gi = 0.0
                for k in max_pos:
                    idx1 = k - i
                    idx2 = i - k
                    if 0 <= idx1 < n:
                        gi += a[idx1] / len(max_pos)
                    if 0 <= idx2 < n:
                        gi += a[idx2] / len(max_pos)
                grad[i] = gi
        
        # Objective: f = 2n * max_b (with sum(a)=1)
        # Gradient: 2n * grad
        # But we also need to account for the normalization sum(a)=1
        # Using Lagrange multiplier: L = 2n*max_b - lambda*(sum(a)-1)
        # ∇_a L = 2n*grad - lambda
        # lambda = sum(2n*grad) / n = 2*sum(grad)
        # So effective gradient = 2n*grad - 2*sum(grad)
        
        lr = 0.01 * (0.99 ** (it // 50))  # Learning rate with decay
        a_new = a - lr * (2 * n * grad - 2 * grad.sum())
        a_new = np.clip(a_new, 0, None)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best = a_new.copy()
            a = a_new
    
    return best, best_score


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n)
    
    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        resampled = np.maximum(np.interp(t, t_old, parent), 0)
        if resampled.sum() > 0.01:
            inits.append(resampled / resampled.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Gaussian
    c = np.exp(-0.5 * ((t - 0.5) / 0.2) ** 2)
    inits.append(c / c.sum())
    
    # Bimodal
    c = np.exp(-0.5 * ((t - 0.25) / 0.15) ** 2) + np.exp(-0.5 * ((t - 0.75) / 0.15) ** 2)
    inits.append(c / c.sum())
    
    # Cosine fundamental
    c = 1.0 + 0.3 * np.cos(2 * np.pi * t)
    inits.append(np.maximum(c, 0.01) / np.maximum(c, 0.01).sum())
    
    # Sinc-like
    x = (t - 0.5) * n
    c = np.sinc(x / (n * 0.3))  # sinc(x) = sin(pi*x)/(pi*x)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())
    
    # Random smooth
    r = rng.uniform(0, 1, n)
    # Smooth with a moving average
    k = max(1, n // 32)
    kernel = np.ones(k) / k
    r_smooth = np.convolve(r, kernel, mode='same')
    r_smooth = np.maximum(r_smooth, 0.01)
    inits.append(r_smooth / r_smooth.sum())
    
    # Triangle
    c = 1.0 - 2.0 * np.abs(t - 0.5)
    c = np.maximum(c, 0.05)
    inits.append(c / c.sum())
    
    # Plateau
    c = np.ones(n)
    half = n // 4
    c[:half] = 0.5
    c[-half:] = 0.5
    inits.append(c / c.sum())
    
    return inits
