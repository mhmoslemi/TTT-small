def compute_obj_fft(a):
    """Compute objective using FFT for speed."""
    n = len(a)
    if n == 0:
        return np.inf
    # Use FFT for convolution
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    conv = np.fft.irfft(fa * fa, fsize)[:size]
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

    sum_a = float(np.sum(a))
    if sum_a < 0.01:

import numpy as np

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

    lags = np.arange(2 * n - 1)
    size = 2 * n - 1

            valid = (kj >= 0) & (kj < n)

            if np.any(valid_off):

            # Build coef and quad

def init_cosine(n, rng):
    """Cosine-based initialization."""
    freq = rng.uniform(0.5, 4.0)
    phase = rng.uniform(0, 2 * np.pi)
    t = np.arange(n) / n
    a = np.maximum(0.0, np.cos(2 * np.pi * freq * t + phase) + rng.uniform(0, 0.5))
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n) + 0.1
        s = np.sum(a)
    return a / s


def init_spectral(n, rng):
    """Spectral initialization: random PSD, IFFT, clip to non-negative."""
    fsize = 1 << (2 * n - 1).bit_length()
    # Random magnitudes with low-pass bias
    freqs = np.arange(fsize // 2 + 1)
    magnitudes = rng.uniform(0.5, 1.0, fsize // 2 + 1)
    # Add some structure
    peak = int(rng.integers(1, fsize // 4))
    width = int(rng.integers(2, fsize // 8))
    magnitudes += np.exp(-0.5 * ((freqs - peak) / width) ** 2) * rng.uniform(0.5, 2.0)
    # Build real FFT
    fft_vals = magnitudes.astype(complex)
    # Reshape to proper format
    n_half = fsize // 2
    full_fft = np.zeros(fsize, dtype=complex)
    full_fft[:n_half + 1] = magnitudes
    full_fft[n_half + 1:] = np.conjugate(full_fft[1:n_half][::-1])
    sig = np.fft.ifft(full_fft).real
    sig = np.maximum(0, sig[:n])
    s = np.sum(sig)
    if s < 1e-10:
        sig = rng.random(n) + 0.1
        s = np.sum(sig)
    return sig / s


def init_periodic(n, rng):

def init_ramp(n, rng):
    """Ramp/trapezoidal initialization."""
    t = np.arange(n) / max(n - 1, 1)

def init_two_lobe(n, rng):

    mid = n // 2

def diversify(a, rng, temperature):
    """Perturb a sequence with simulated annealing style moves."""
    n = len(a)
    a = a.copy()
    n_moves = int(rng.integers(1, max(2, n // 10)))
    for _ in range(n_moves):
        move_type = rng.integers(0, 4)
        if move_type == 0:
            # Add noise to a block
            start = int(rng.integers(0, n))
            length = int(rng.integers(1, max(2, n // 5)))
            end = min(start + length, n)
            a[start:end] += temperature * rng.standard_normal(end - start)
        elif move_type == 1:
            # Swap two random coordinates
            i, j = rng.integers(0, n, 2)
            a[i], a[j] = a[j], a[i]
        elif move_type == 2:
            # Scale a block
            start = int(rng.integers(0, n))
            length = int(rng.integers(1, max(2, n // 4)))
            end = min(start + length, n)
            a[start:end] *= (1.0 + temperature * rng.standard_normal())
        else:
            # Inject a spike
            j = int(rng.integers(0, n))
            a[j] += temperature * rng.uniform(0.5, 2.0)
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s
    return a

        n_best = len(best_seq)
        print(f"\n=== Final refinement: n={n_best}, obj={best_score:.6f}, {remaining:.0f}s ===", flush=True)

        # Multiple CD passes from slightly perturbed versions
        for trial in range(5):
            if time.monotonic() > deadline - 5:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            if trial == 0:
                a_refine = best_seq.copy()
            else:
                a_refine = diversify(best_seq, rng, 0.05 * (1.0 / (trial + 1)))

            max_passes = int(min(500, remaining / 0.2))
            a_final, obj_final = coord_descent(a_refine, max_passes=max_passes)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final CD trial={trial}: obj={obj_final:.6f}", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s", flush=True)
    return [float(x) for x in best_seq]
