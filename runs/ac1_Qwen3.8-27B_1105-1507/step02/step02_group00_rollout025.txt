def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        # j must satisfy 0 <= j < n and 0 <= k-j < n
        # i.e., 0 <= j < n and k-n+1 <= j <= k
        j_start = max(0, k - n + 1)

def subgradient_polish(a_init, max_iters, deadline, verbose=False):
    """Subgradient method with Polyak step size."""
    global GLOBAL_DEADLINE
    n = len(a_init)
    a = a_init.copy()

            best_obj = obj
            best_a = a_new.copy()
            upper_bound = obj / (2.0 * n)

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def coordinate_descent(a_init, n_rounds, deadline, verbose=False):
    """Coordinate descent with golden section search."""

            break

                    break

            a[j] = 0.0

            a[j] = hi
            obj_hi = compute_obj(a)

            d = hi - c
            a[j] = c
            obj_c = compute_obj(a)
            a[j] = d
            obj_d = compute_obj(a)

                    d = c

                    c = hi - (hi - lo) / gr

                else:
                    lo = c
                    c = d
                    obj_c = obj_d
                    d = hi - (hi - lo) / gr
                    a[j] = d

            # Best point

            best_x = a[j]

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating, subgradient polishing, mirror descent, and coordinate descent
with efficient convolution updates. Focuses on n=300 (parent length) with
diverse initializations and adaptive time management.
"""

import numpy as np
from scipy.optimize import linprog
import time

GLOBAL_DEADLINE = 0.0


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build matrix A such that (A @ g)[k] = (a * g)[k] (full convolution)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je + 1] = a[k - je:k - js + 1][::-1]
    return A


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.status == 0:
        return res.x
    return None


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_alternate(a_init, max_iters, deadline, verbose=False):
    """LP alternating optimization with adaptive blending."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        conv_a = np.convolve(a, a)
        M = float(np.max(conv_a))

        g = lp_solve(a, M)
        if g is None:
            break

        g_sum = float(np.sum(g))
        if g_sum <= 1.0 + 1e-12:
            break
        g = g / g_sum

        # Try multiple blend factors
        best_local_obj = best_obj
        best_local_a = best_a

        for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
            candidate = (1.0 - t) * a + t * g
            candidate = np.maximum(candidate, 0.0)
            cs = float(np.sum(candidate))
            if cs < 1e-15:
                continue
            candidate = candidate / cs
            obj = compute_obj(candidate)
            if obj < best_local_obj:
                best_local_obj = obj
                best_local_a = candidate.copy()

        if best_local_obj < best_obj:
            best_obj = best_local_obj
            best_a = best_local_a.copy()

        a = best_local_a.copy()

        if verbose and (it % 25 == 0):
            print(f"    LP it={it}: obj={best_obj:.8f}")

    return best_a, best_obj


def subgradient_polish(a_init, max_iters, deadline, verbose=False):
    """Subgradient method with Polyak step size and best-tracking."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    upper_bound_f = best_obj  # upper bound on 2n*max(conv) for normalized a

    # Lower bound: for any a with sum=1, max(conv) >= max(conv)[n-1] >= 1/n
    # So 2n*max(conv) >= 2. Use a tighter empirical lower bound.
    lower_bound_f = 1.50

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        conv = np.convolve(a, a)
        max_val = float(np.max(conv))
        current_f = 2.0 * n * max_val

        # Active set with adaptive tolerance
        tol = max_val * (0.005 + 0.02 * it / max_iters)
        active = np.where(conv >= max_val - tol)[0]
        if len(active) == 0:
            active = np.where(conv == max_val)[0]

        # Compute subgradient (average of active subgradients)
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] += 2.0 * a[k - je:k - js + 1][::-1]
        grad /= float(len(active))

        grad_norm_sq = float(np.dot(grad, grad))
        if grad_norm_sq < 1e-30:
            break

        # Polyak step size
        gap = current_f - lower_bound_f
        if gap < 1e-10:
            break
        eta = gap / (grad_norm_sq * float(n))
        eta = min(eta, 0.05)

        a_new = a - eta * grad
        a_new = project_simplex(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            upper_bound_f = obj

        a = a_new

        if verbose and (it % 100 == 0):
            print(f"    SG it={it}: obj={best_obj:.8f}")

    return best_a, best_obj


def mirror_descent(a_init, max_iters, deadline, verbose=False):
    """Mirror descent (multiplicative weights) on the simplex."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    a = np.maximum(a, 1e-15)  # avoid log(0)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    lower_bound_f = 1.50

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        conv = np.convolve(a, a)
        max_val = float(np.max(conv))
        current_f = 2.0 * n * max_val

        tol = max_val * 0.01
        active = np.where(conv >= max_val - tol)[0]
        if len(active) == 0:
            active = np.where(conv == max_val)[0]

        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] += 2.0 * a[k - je:k - js + 1][::-1]
        grad /= float(len(active))

        # Multiplicative update with adaptive step
        gap = current_f - lower_bound_f
        if gap < 1e-10:
            break
        grad_norm = float(np.linalg.norm(grad))
        eta = gap / (grad_norm * grad_norm * float(n) + 1e-15)
        eta = min(eta, 0.02)

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()

        a = a_new

        if verbose and (it % 100 == 0):
            print(f"    MD it={it}: obj={best_obj:.8f}")

    return best_a, best_obj


def coord_descent(a_init, n_rounds, deadline, verbose=False):
    """Coordinate descent with golden section search and fast conv update."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    inv_gr = 2.0 / (1.0 + np.sqrt(5.0))

    for rnd in range(n_rounds):
        if time.monotonic() > deadline - 5.0:
            break

        conv = np.convolve(a, a).copy()

        for j in range(n):
            if time.monotonic() > deadline - 5.0:
                break

            old = a[j]
            # Search range
            mean_val = 1.0 / n
            hi = max(4.0 * old, 4.0 * mean_val)
            lo = 0.0

            # Evaluate at bounds
            a[j] = lo
            conv_lo = conv.copy()
            obj_lo = float(np.max(conv_lo))
            a[j] = hi
            conv_hi = conv.copy()
            obj_hi = float(np.max(conv_hi))

            # If neither endpoint is better, try a few discrete values
            if obj_lo >= float(np.max(conv)) and obj_hi >= float(np.max(conv)):
                # Try a few multipliers
                best_x = old
                best_o = float(np.max(conv))
                for mult in [0.3, 0.5, 0.7, 1.3, 1.5, 2.0]:
                    x = old * mult
                    if x < 0:
                        x = 0.0
                    a[j] = x
                    conv_trial = conv.copy()
                    delta = x - old
                    ks = j
                    ke = min(j + n, 2 * n - 1)
                    idx = np.arange(ks, ke)
                    conv_trial[idx] += 2.0 * delta * a[idx - j]
                    if 0 <= 2 * j < 2 * n - 1:
                        conv_trial[2 * j] += delta * delta
                    o = float(np.max(conv_trial))
                    if o < best_o:
                        best_o = o
                        best_x = x
                a[j] = best_x
                # Update conv
                delta = best_x - old
                ks = j
                ke = min(j + n, 2 * n - 1)
                idx = np.arange(ks, ke)
                conv[idx] += 2.0 * delta * a[idx - j]
                if 0 <= 2 * j < 2 * n - 1:
                    conv[2 * j] += delta * delta
            else:
                # Golden section search
                c = hi * inv_gr
                d = hi - c

                a[j] = c
                conv_c = conv.copy()
                delta_c = c - old
                ks = j
                ke = min(j + n, 2 * n - 1)
                idx_c = np.arange(ks, ke)
                conv_c[idx_c] += 2.0 * delta_c * a[idx_c - j]
                if 0 <= 2 * j < 2 * n - 1:
                    conv_c[2 * j] += delta_c * delta_c
                obj_c = float(np.max(conv_c))

                a[j] = d
                conv_d = conv.copy()
                delta_d = d - old
                idx_d = np.arange(ks, ke)
                conv_d[idx_d] += 2.0 * delta_d * a[idx_d - j]
                if 0 <= 2 * j < 2 * n - 1:
                    conv_d[2 * j] += delta_d * delta_d
                obj_d = float(np.max(conv_d))

                for _ in range(25):
                    if obj_c < obj_d:
                        hi = d
                        d = c
                        obj_d = obj_c
                        c = hi - (hi - lo) * inv_gr
                        a[j] = c
                        conv_c = conv.copy()
                        delta_c = c - old
                        conv_c[idx_c] += 2.0 * delta_c * a[idx_c - j]
                        if 0 <= 2 * j < 2 * n - 1:
                            conv_c[2 * j] += delta_c * delta_c
                        obj_c = float(np.max(conv_c))
                    else:
                        lo = c
                        c = d
                        obj_c = obj_d
                        d = hi - (hi - lo) * inv_gr
                        a[j] = d
                        conv_d = conv.copy()
                        delta_d = d - old
                        conv_d[idx_d] += 2.0 * delta_d * a[idx_d - j]
                        if 0 <= 2 * j < 2 * n - 1:
                            conv_d[2 * j] += delta_d * delta_d
                        obj_d = float(np.max(conv_d))

                if obj_c <= obj_d:
                    best_x = c
                    best_o = obj_c
                else:
                    best_x = d
                    best_o = obj_d

                a[j] = best_x
                delta = best_x - old
                conv[idx_c] += 2.0 * delta * a[idx_c - j]
                if 0 <= 2 * j < 2 * n - 1:
                    conv[2 * j] += delta * delta

            # Renormalize
            s = float(np.sum(a))
            if s < 1e-15:
                a = np.ones(n) / n
                conv = np.convolve(a, a).copy()
                continue
            a = a / s

            # Check if improved
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        # Rebuild conv after each round (to avoid drift)
        conv = np.convolve(a, a).copy()
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if verbose:
            print(f"    CD round={rnd}: obj={best_obj:.8f}")

    return best_a, best_obj


def init_uniform(n):
    return np.ones(n) / n


def init_parent(parent_arr, n):
    """Resize parent to length n by interpolation or truncation."""
    p = parent_arr.astype(np.float64)
    if len(p) == n:
        return p / np.sum(p)
    # Interpolate
    x_old = np.linspace(0, 1, len(p))
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, p)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_cosine(n, rng):
    freq = rng.uniform(0.5, 2.5)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.arange(n) / n
    a = np.maximum(0.0, np.cos(2 * np.pi * freq * x + phase))
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_parabola(n, rng):
    x = np.arange(n) / n - 0.5
    width = rng.uniform(0.3, 1.5)
    a = np.exp(-x**2 / (2 * width**2))
    s = np.sum(a)
    return a / s


def init_two_lobe(n, rng):
    mid = n // 2
    a = np.ones(n)
    a[:mid] *= rng.uniform(0.5, 1.5)
    a[mid:] *= rng.uniform(0.5, 1.5)
    s = np.sum(a)
    return a / s


def init_random(n, rng):
    a = rng.random(n)
    a = np.maximum(a, 1e-6)
    s = np.sum(a)
    return a / s
