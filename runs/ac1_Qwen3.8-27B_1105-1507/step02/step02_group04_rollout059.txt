Combines Adam-based smoothed subgradient descent, multi-resolution approach,
spectral initialization, and LP polishing. Key improvements over baseline:
- Adam optimizer for faster convergence
- FFT-based gradient computation for large n

- Adaptive temperature scheduling with restarts
- Variance-aware objective for better flatness

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full convolution conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def get_weights(conv, tau):
    """Get softmax weights over convolution values."""
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -700, 0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
        w_sum = 1.0
    else:
        weights /= w_sum
    return weights, max_conv

    """
    Compute gradient of sum_k weights[k] * b_k(a) w.r.t. a.
    b_k = sum_{i+j=k} a_i * a_j
    grad[m] ≈ 2 * sum_k weights[k] * a[k-m] (for valid indices)

        while size < lw + n:

        A = np.fft.rfft(a, size)

        r = np.fft.irfft(np.conj(A) * W, size)
        grad = 2.0 * r[:n]
    else:
        c = np.correlate(weights, a, mode='full')
        grad = 2.0 * c[n - 1:n - 1 + n]
    return grad

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def adam_optimize(a, max_iter=5000, lr_init=0.1, tau_start=2.0, tau_end=80.0,
                  beta1=0.9, beta2=0.999, eps=1e-8, verbose=False):
    """
    Adam optimizer for minimizing smoothed max(conv(a,a)) with sum(a)=1, a>=0.
    """
    n = len(a)
    a = normalize(a.copy())
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    stall = 0
    patience = 300
    t = 0
    
    for it in range(max_iter):
        t += 1
        frac = t / max_iter
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.7)
        lr = lr_init / (1 + 0.01 * t)
        
        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        cur_obj = 2.0 * n * max_conv  # since sum(a)=1
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Restart with noise
                noise = np.random.default_rng(it * 7 + 13).normal(0, 0.03, n)
                a = normalize(a + noise)
                m = np.zeros(n)
                v = np.zeros(n)
                stall = 0
                continue
        
        weights, _ = get_weights(conv, tau)
        grad = grad_conv_wrt_a(a, weights)
        
        # Center gradient (for sum constraint)
        grad = grad - np.mean(grad)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = normalize(a_new)
    
    return best_a, best_obj


def build_lp_matrix(a):
    """Build constraint matrix for LP: (conv(a, g))_k <= M."""
    n = len(a)
    # (a * g)_k = sum_{i+j=k} a_i * g_j
    # For each k, the constraint is: sum_{i=max(0,k-n+1)}^{min(k,n-1)} a_i * g_{k-i} <= M
    # The coefficient of g_j in constraint k is a_{k-j} (if valid)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = a[i]
    return A


def solve_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if n > 400:
        return None
    A = build_lp_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = res.x
            g_sum = -res.fun
            if g_sum > 0:
                return g / g_sum
    except Exception:
        pass
    return None


def lp_polish(a, max_iter=30, t_mix=0.3, time_limit=15):
    """LP-based polishing."""
    n = len(a)
    if n > 400:
        return a, compute_obj(a)
    
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        conv = compute_conv(a)
        M = float(np.max(conv))
        g = solve_lp(a, M, time_limit=time_limit)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum <= 1.0 + 1e-3:
            break
        a_new = normalize((1 - t_mix) * a + t_mix * g)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_obj


def make_init(n, init_type, rng):
    """Generate a random initialization of type init_type."""
    if init_type == 0:
        # Uniform
        a = np.ones(n)
    elif init_type == 1:
        # Hann window (peaks at edges)
        i = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * i / (n - 1)))
    elif init_type == 2:
        # Random
        a = rng.random(n)
    elif init_type == 3:
        # Cosine with random frequency
        i = np.arange(n)
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 * (1 + np.cos(2 * np.pi * freq * i / n + phase))
    elif init_type == 4:
        # U-shaped
        i = np.arange(n)
        center = (n - 1) / 2.0
        a = 1.0 / (0.1 + 0.9 * (2 * np.abs(i - center) / (n - 1)) ** 2)
    elif init_type == 5:
        # Random with spectral shaping
        a = rng.random(n)
        a = normalize(a)
        # Shape spectrum
        size = 1
        while size < 2 * n:
            size <<= 1
        A = np.fft.rfft(a, size)
        # Boost low frequencies
        freqs = np.arange(len(A))
        shape = 1.0 / (1.0 + 0.5 * freqs / n)
        A *= shape
        a = np.real(np.fft.irfft(A, size))[:n]
        a = np.maximum(a, 0)
    elif init_type == 6:
        # Bimodal
        a = rng.random(n)
        center = n // 2
        for i in range(n):
            if abs(i - center) < n // 4:
                a[i] *= 0.3
    else:
        a = rng.random(n)
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Adam-based smoothed subgradient descent with FFT gradients,
multi-resolution approach, diverse initializations (Hann window, U-shaped,
spectral shaping), adaptive temperature scheduling, and LP polishing.
Key improvements: Adam optimizer, longer sequences (up to 2000),
two-phase search (screening + refinement), and variance-aware objectives.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full convolution conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective value: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def get_weights(conv, tau):
    """Get normalized softmax weights over convolution values."""
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -700, 0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
        w_sum = 1.0
    else:
        weights /= w_sum
    return weights, max_conv


def grad_conv_wrt_a(a, weights):
    """
    Compute gradient of sum_k weights[k] * b_k(a) w.r.t. a.
    Uses FFT for efficiency.
    grad[m] = 2 * sum_k weights[k] * a[k-m] (approximately, ignoring diagonal)
    """
    n = len(a)
    lw = len(weights)
    size = 1
    while size < lw + n:
        size <<= 1
    W = np.fft.rfft(weights, size)
    A = np.fft.rfft(a, size)
    # Cross-correlation: r[m] = sum_j weights[j+m] * a[j]
    r = np.fft.irfft(np.conj(A) * W, size)
    grad = 2.0 * r[:n]
    return grad


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def adam_optimize(a, max_iter=5000, lr_init=0.08, tau_start=2.0, tau_end=80.0,
                  beta1=0.85, beta2=0.999, eps=1e-8, var_weight=0.0,
                  hard_every=60):
    """
    Adam optimizer for minimizing smoothed max(conv(a,a)) with sum(a)=1, a>=0.
    Optionally includes a variance penalty for encouraging flatness.
    Uses periodic 'hard' gradient steps targeting the actual argmax.
    """
    n = len(a)
    a = normalize(a.copy())

    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(best_a)))

    stall = 0
    patience = 400
    t = 0
    rng_local = np.random.default_rng(max_iter + n)

    for it in range(max_iter):
        t += 1
        frac = min(1.0, t / max_iter)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.6)
        lr = lr_init * (1.0 / (1.0 + 0.005 * t))

        conv = compute_conv(a)
        max_conv = float(np.max(conv))
        cur_obj = 2.0 * n * max_conv

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Restart with multiplicative noise
                noise = rng_local.normal(0, 0.04, n)
                a = normalize(a * np.exp(noise))
                m = np.zeros(n)
                v = np.zeros(n)
                stall = 0
                continue

        # Compute gradient
        if hard_every > 0 and it % hard_every == hard_every - 1:
            # Hard step: target only the argmax
            weights = np.zeros(len(conv))
            weights[int(np.argmax(conv))] = 1.0
        else:
            weights, _ = get_weights(conv, tau)

        grad = grad_conv_wrt_a(a, weights)

        # Optional variance penalty gradient
        if var_weight > 0:
            # d/d(a_i) [mean(b^2)] = 4/(2n-1) * conv(b, a)[i] (approximately)
            size_v = 1
            while size_v < 4 * n:
                size_v <<= 1
            B = np.fft.rfft(conv, size_v)
            A_f = np.fft.rfft(a, size_v)
            var_grad = np.fft.irfft(B * A_f, size_v)[:n]
            grad += var_weight * var_grad

        # Center gradient (for sum=1 constraint)
        grad = grad - np.mean(grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = normalize(a_new)

    return best_a, best_obj


def build_lp_matrix(a):
    """Build constraint matrix for LP: (conv(a, g))_k <= M for all k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            for j in range(j_min, j_max + 1):
                A[k, j] = a[k - j]
    return A


def solve_lp(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    if n > 350:
        return None
    A = build_lp_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = res.x
            g_sum = float(-
                         res.fun)
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def lp_polish(a, max_iter=25, t_mix=0.25, time_limit=12):
    """LP-based polishing step."""
    n = len(a)
    if n > 350:
        return a, compute_obj(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    stall = 0

    for it in range(max_iter):
        conv = compute_conv(a)
        M = float(np.max(conv))
        g = solve_lp(a, M, time_limit=time_limit)
        if g is None:
            break
        g_sum = float(np.sum(g))
        if g_sum <= 1.0 + 5e-4:
            stall += 1
            if stall > 10:
                break
            continue
        stall = 0
        g = g / g_sum
        a_new = normalize((1.0 - t_mix) * a + t_mix * g)
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_obj


def make_init(n, init_type, rng):
    """Generate initialization of the specified type."""
    i = np.arange(n)
    if init_type == 0:
        a = np.ones(n)
    elif init_type == 1:
        # Hann window (peaks at edges)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * i / max(1, n - 1)))
    elif init_type == 2:
        # Pure random
        a = rng.random(n)
    elif init_type == 3:
        # Cosine with random frequency and phase
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * freq * i / n + phase))
    elif init_type == 4:
        # U-shaped (peaks at edges, dip in middle)
        center = (n - 1) / 2.0
        a = 1.0 / (0.05 + 0.95 * (2.0 * np.abs(i - center) / max(1, n - 1)) ** 2)
    elif init_type == 5:
        # Random with spectral shaping (boost low frequencies)
        a = rng.random(n)
        a = normalize(a)
        size = 1
        while size < 2 * n:
            size <<= 1
        A = np.fft.rfft(a, size)
        freqs = np.arange(len(A), dtype=np.float64)
        shape = 1.0 / (1.0 + 0.3 * freqs / n)
        A *= shape
        a = np.real(np.fft.irfft(A, size))[:n]
        a = np.maximum(a, 0)
    elif init_type == 6:
        # Bimodal: two peaks
        a = rng.random(n)
        for peak_pos in [n // 4, 3 * n // 4]:
            width = max(1, n // 8)
            for j in range(n):
                a[j] += np.exp(-((j - peak_pos) ** 2) / (2 * width ** 2))
    elif init_type == 7:
        # Linear ramp + noise
        a = np.linspace(0.1, 1.0, n) + 0.1 * rng.random(n)
    else:
        a = rng.random(n)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def optimize_one(n, seed, time_budget, use_lp=True, verbose=False):
    """Run full optimization pipeline for a given sequence length."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    best_a = None
    best_obj = np.inf

    # Number of initializations based on time
    n_inits = max(3, int(time_budget / 12))

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget - 8:
            break

        init_type = init_idx % 8
        a = make_init(n, init_type, rng)

        # Phase 1: Main Adam optimization
        remaining = time_budget - (time.monotonic() - start)
        max_iter = int(min(10000, max(1500, remaining * 2.5)))

        a_opt, obj_opt = adam_optimize(
            a, max_iter=max_iter,
            lr_init=0.06, tau_start=2.0, tau_end=60.0,
            var_weight=0.01
        )

        # Phase 2: LP polish (if applicable and time allows)
        if use_lp and n <= 350:
            remaining = time_budget - (time.monotonic() - start)
            if remaining > 25:
                a_lp, obj_lp = lp_polish(a_opt, max_iter=12, t_mix=0.2, time_limit=8)
                if obj_lp < obj_opt:
                    a_opt = a_lp
                    obj_opt = obj_lp

        # Phase 3: Fine Adam polish with high tau
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            a_pol, obj_pol = adam_optimize(
                a_opt, max_iter=int(min(5000, remaining * 2)),
                lr_init=0.02, tau_start=30.0, tau_end=100.0
            )
            if obj_pol < obj_opt:
                a_opt = a_pol
                obj_opt = obj_pol

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            if verbose:
                elapsed = time.monotonic() - start
                print(f"      n={n} init{init_idx}(type={init_type}): obj={obj_opt:.6f} ({elapsed:.1f}s)")

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def try_parent_sequence():
    """Try to use the parent sequence as a starting point."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def upsample_sequence(a_short, n_target):
    """Upsample a shorter optimized sequence to a longer target length."""
    n_short = len(a_short)
    a_long = np.zeros(n_target)
    if n_short >= n_target:
        return a_short[:n_target]
    # Map short indices to long indices
    ratio = n_target / n_short
    for i in range(n_short):
        center = int(i * ratio + 0.5)
        width = max(1, int(ratio * 0.4))
        for j in range(max(0, center - width), min(n_target, center + width + 1)):
            dist = abs(j - center) / max(1.0, width)
            a_long[j] += a_short[i] * np.exp(-dist ** 2)
    return normalize(a_long)


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Quick screening across multiple sequence lengths.
    Phase 2: Deep optimization on top candidates.
    Phase 3: Final polishing with LP and perturbation.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent, parent_obj = try_parent_sequence()
    parent_n = 0
    if parent is not None:
        parent_n = len(parent)
        best_seq = parent.copy()
        best_score = parent_obj
        print(f"Parent sequence (n={parent_n}): obj={best_score:.6f}")

    # Phase 1: Quick screening
    print("\n=== Phase 1: Screening ===")
    n_screen = [30, 50, 80, 120, 200, 300, 400, 600]
    if parent_n > 0:
        n_screen = sorted(set(n_screen + [max(20, parent_n - 50), parent_n, min(1500, parent_n + 100)]))

    candidates = []  # (n, obj, a)
    screen_time_each = min(30.0, (budget_s * 0.3) / len(n_screen))

    for n in n_screen:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        alloc = min(screen_time_each, remaining * 0.1)
        alloc = max(10.0, alloc)

        a_opt, obj = optimize_one(n, seed=seed + n * 31, time_budget=alloc,
                                  use_lp=(n <= 200), verbose=False)
        candidates.append((n, obj, a_opt))
        print(f"  n={n}: obj={obj:.6f}")

        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()

    candidates.sort(key=lambda x: x[1])
    print(f"\nTop candidates: {[(n, f'{o:.5f}') for n, o, _ in candidates[:5]]}")

    # Phase 2: Deep optimization on top candidates
    print("\n=== Phase 2: Deep optimization ===")
    deep_time_total = min(budget_s * 0.45, deadline - time.monotonic())
    n_deep = min(4, len(candidates))

    for i in range(n_deep):
        if time.monotonic() > deadline:
            break
        n_cand, obj_cand, a_cand = candidates[i]
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        alloc = deep_time_total / (n_deep - i) * (1.0 if i == 0 else 0.7)
        alloc = min(alloc, remaining * 0.6)

        # If this is the parent length, also try from the parent
        seeds_to_try = [seed + n_cand * 104729 + i * 7]
        if n_cand == parent_n and parent is not None:
            seeds_to_try.append(seed + 999999)

        for s_idx, s in enumerate(seeds_to_try):
            remaining = deadline - time.monotonic()
            if remaining < 20:
                break
            a_opt, obj = optimize_one(
                n_cand, seed=s, time_budget=alloc / max(1, len(seeds_to_try)),
                use_lp=(n_cand <= 350), verbose=True
            )
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  ** Phase2 UPDATE: n={n_cand}, obj={best_score:.6f}")

    # Try upsampled versions of best short sequences
    if best_seq is not None:
        n_best = len(best_seq)
        for n_target in [n_best + 100, n_best + 200, n_best + 400]:
            if n_target > 1500 or time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break
            a_up = upsample_sequence(best_seq, n_target)
            alloc = min(60.0, remaining * 0.2)
            a_opt, obj = optimize_one(n_target, seed=seed + n_target * 53,
                                       time_budget=alloc, use_lp=(n_target <= 350))
            # Also try from the upsampled directly
            a_opt2, obj2 = adam_optimize(a_up, max_iter=int(alloc * 1.5),
                                          lr_init=0.04, tau_start=5.0, tau_end=50.0)
            obj2 = compute_obj(a_opt2)
            if obj < obj2:
                a_opt2, obj2 = a_opt, obj
            if obj2 < best_score:
                best_score = obj2
                best_seq = a_opt2.copy()
                print(f"  ** Upsampled UPDATE: n={n_target}, obj={best_score:.6f}")

    # Phase 3: Final polishing
    print(f"\n=== Phase 3: Polishing (n={len(best_seq) if best_seq is not None else '?'}) ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_cur = best_seq.copy()
        n_cur = len(a_cur)
        polish_round = 0

        while time.monotonic() < deadline - 10:
            polish_round += 1
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            # Adam polish (high tau, focus on peak)
            a_pol, obj_pol = adam_optimize(
                a_cur, max_iter=int(min(8000, remaining * 1.2)),
                lr_init=0.015, tau_start=40.0, tau_end=120.0
            )
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  Polish {polish_round} Adam: obj={best_score:.6f}")

            # LP step
            remaining = deadline - time.monotonic()
            if remaining > 25 and n_cur <= 350:
                a_lp, obj_lp = lp_polish(a_cur, max_iter=15, t_mix=0.15, time_limit=8)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    print(f"  Polish {polish_round} LP: obj={best_score:.6f}")

            # Multiplicative perturbation
            remaining = deadline - time.monotonic()
            if remaining > 20:
                improved = False
                for ns in [0.03, 0.06, 0.1]:
                    a_pert = a_cur * np.exp(rng.normal(0, ns, n_cur))
                    a_pert = normalize(a_pert)
                    obj_p = compute_obj(a_pert)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_pert.copy()
                        a_cur = a_pert.copy()
                        print(f"  Polish {polish_round} perturb({ns}): obj={best_score:.6f}")
                        improved = True
                        break
                if not improved and remaining > 30:
                    # Try additive perturbation on a subset
                    n_perturb = max(5, n_cur // 10)
                    indices = rng.choice(n_cur, size=n_perturb, replace=False)
                    a_pert2 = a_cur.copy()
                    a_pert2[indices] += rng.normal(0, 0.02, n_perturb)
                    a_pert2 = normalize(a_pert2)
                    obj_p2 = compute_obj(a_pert2)
                    if obj_p2 < best_score:
                        best_score = obj_p2
                        best_seq = a_pert2.copy()
                        a_cur = a_pert2.copy()
                        print(f"  Polish {polish_round} add_perturb: obj={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    result = [float(x) for x in best_seq]
    return result
