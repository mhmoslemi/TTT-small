"""
Improved optimization using LP-based Frank-Wolfe with free t, 
spectral shaping, and adaptive coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_score(f):
    """Compute score for sequence f: 2*n*max(conv)/(sum)^2"""
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * n * conv.max() / (s * s))

def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s

def build_conv_matrix(f):
    """Build matrix A such that A @ g = np.convolve(f, g, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        # (f*g)[k] = sum_j f[k-j] * g[j] where 0 <= k-j < n and 0 <= j < n
        # So j ranges from max(0, k-n+1) to min(n-1, k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    return A

def lp_optimize_t(f, time_limit=5):
    """
    Solve: min t s.t. (f*g)[k] <= t for all k, sum(g)=1, g>=0
    Using linprog with variable t.
    """
    n = len(f)
    A_conv = build_conv_matrix(f)
    m = 2 * n - 1
    
    # Variables: [g_0, ..., g_{n-1}, t]
    # Objective: minimize t -> c = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Constraints:
    # 1) A_conv @ g - t <= 0 for each k  ->  [A_conv, -1] @ [g, t] <= 0
    A_ub = np.hstack([A_conv, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    # 2) sum(g) = 1 -> handled as equality
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: g_i >= 0, t free (but t >= 0 practically)
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.clip(res.x[:n], 0, None)
            t_val = res.x[n]
            g_sum = g.sum()
            if g_sum > 1e-10:
                g = g / g_sum
            return g, t_val
    except Exception:
        pass
    return None, None

    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)

    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()
    
    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0
    
    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Target: make |F|^2 as flat as possible (excluding DC)
        # sum(|F|^2) = n_fft * sum(f^2) by Parseval
        energy = np.sum(f ** 2)
        n_bins = len(mag)
        # If flat: each non-DC bin has magnitude sqrt(energy * n_fft / (n_bins - 1))

def pressure_descent_v2(f, n_iter=300, eta=0.15, rng=None):
    """Pressure-based descent considering multiple peaks."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0
    
    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Compute pressure from top K peaks
        K = min(5, 2 * n - 1)
        top_k = np.argsort(conv)[::-1][:K]
        weights = np.exp(-np.arange(K) * 0.3)  # Exponential decay for peak importance
        weights /= weights.sum()
        
        pressure = np.zeros(n)
        for w, j_star in zip(weights, top_k):
            for i in range(n):
                idx = j_star - i
                if 0 <= idx < n:
                    pressure[i] += w * f[idx]
        
        if pressure.max() > 0:
            pressure = pressure / pressure.max()
        
        # Adaptive learning rate
        eta_eff = eta / (1.0 + 0.005 * it)
        
        # Exponential update
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)
        
        # Also try a "push up the low" variant
        r = rng.random()
        if r < 0.2:
            # Boost low-pressure positions
            f_alt = f * np.exp(eta_eff * 0.5 * pressure)
            f_alt = np.clip(f_alt, 1e-15, None)
            f_alt = normalize(f_alt)
            if compute_score(f_alt) < compute_score(f_new):
                f_new = f_alt
        
        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 80:
                # Perturbation
                f_pert = f * (1.0 + 0.1 * rng.standard_normal(n))
                f_pert = np.clip(f_pert, 0, None)

                    best_score = compute_score(f)

def lp_frank_wolfe(f, time_budget, rng=None, t0=0.5):
    """Frank-Wolfe style: solve LP for optimal g given f, then convex combination."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    t = t0
    start = time.monotonic()
    
    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Solve LP: min t s.t. (f*g) <= t, sum(g)=1, g>=0
        g, t_opt = lp_optimize_t(f, time_limit=3)
        
        if g is None:
            break
        
        # Try convex combinations
        improved = False
        for step in [min(1.0, t * 1.3), t, t * 0.6, t * 0.3, t * 0.1]:
            if step <= 0.001:
                break
            f_new = (1 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            
            new_score = compute_score(f_new)
            if new_score < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                t = min(1.0, t * 1.15)
                improved = True
                break
        
        if not improved:
            t *= 0.5
            if t < 0.003:
                # Try perturbation
                f_pert = f + rng.normal(0, 0.01, n)
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    t = 0.2

            print(f"  LP-FW: t={t:.4f}, score={best_score:.6f}, time={time.monotonic()-start:.0f}s")
    
    return best_f, best_score

def coordinate_descent_v2(f, n_sweeps=10, rng=None):
    """Coordinate descent with adaptive 1D search."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-12:
                f[i] = original
                continue
            
            best_val = original
            best_local = best_score
            
            # Try a range of multiplicative factors
            for factor in np.logspace(-2, 2, 17):  # 0.01 to 100
                f[i] = original * factor
                s = f.sum()
                if s < 1e-12:
                    f[i] = original
                    continue
                # Quick score (we don't renormalize here to save time, 
                # the score formula handles it)
                conv = np.convolve(f, f)
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]
            
            f[i] = best_val
            if best_local < best_score - 1e-12:
                best_score = best_local
        
        f = np.clip(f, 0, None)
        f = normalize(f)
        # Recompute after normalization
        best_score = compute_score(f)
    
    return f, best_score

def simulated_annealing_v2(f, rng, n_iter=3000, T0=0.03):
    """Enhanced simulated annealing."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    cur_score = best_score
    no_improve = 0
    
    for it in range(n_iter):
        T = T0 * np.exp(-3.0 * it / n_iter)
        f_new = f.copy()
        
        r = rng.random()
        if r < 0.25:
            # Block multiplicative
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.5))
        elif r < 0.5:
            # Single element
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] * np.exp(rng.normal(0, 0.5)))
        elif r < 0.7:
            # Smooth Gaussian perturbation
            center = int(rng.integers(0, n))
            width = max(2, n // 16)
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
            f_new += rng.normal(0, 0.05) * kernel
        elif r < 0.85:
            # Large random reweight
            f_new *= np.exp(rng.normal(0, 0.4, n))
        else:
            # Symmetric flip (try to create anti-symmetric structure)
            half = n // 2
            f_new[:half] = f[half:half*2][::-1] * np.exp(rng.normal(0, 0.2))
        
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score(f_new)
        
        accept = new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12))
        if accept:
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1

"""
Optimized step-function coefficient search using LP-based Frank-Wolfe,
variance gradient descent, spectral shaping, pressure updates, and
adaptive coordinate descent with multi-start strategy.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute score: 2*n*max(conv)/(sum)^2. Scale invariant."""
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def build_conv_matrix(f):
    """Build matrix A such that A @ g = np.convolve(f, g, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    return A


def lp_optimize_t(f, time_limit=5):
    """Solve: min t s.t. (f*g)[k] <= t for all k, sum(g)=1, g>=0."""
    n = len(f)
    A_conv = build_conv_matrix(f)
    m = 2 * n - 1

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub = np.hstack([A_conv, -np.ones((m, 1))])
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.clip(res.x[:n], 0, None)
            t_val = res.x[n]
            g_sum = g.sum()
            if g_sum > 1e-10:
                g = g / g_sum
            return g, t_val
    except Exception:
        pass
    return None, None


def variance_gradient(f):
    """
    Compute gradient of variance of autocorrelation w.r.t. f.
    grad[f[i]] = 4/(2n-1) * sum_j (c[i+j] - mu) * f[j]
    where c = convolve(f,f), mu = 1/(2n-1).
    """
    n = len(f)
    conv = np.convolve(f, f)
    m = 2 * n - 1
    mu = 1.0 / m
    d = conv - mu  # length m

    # Cross-correlation: R[i] = sum_j d[i+j] * f[j]
    # Use FFT for efficiency
    n_fft = 1 << (m + n - 1).bit_length()
    D = np.fft.rfft(d, n_fft)
    F = np.fft.rfft(f, n_fft)
    # R[i] = sum_j d[i+j] f[j] = (d reversed cross-correlate with f)
    # Actually: R[i] = sum_j d[i+j]*f[j]. Let's compute via:
    # If we define d_ext with d[0..m-1], then R[i] = sum_{j=0}^{n-1} d[i+j]*f[j]
    # This is a cross-correlation. Let's use:
    # sum_j d[i+j]*f[j] = (full correlation of d and f)[i] (with appropriate indexing)
    # corr[m-lag] for standard np.correlate... let's just do it via FFT
    # R[i] = sum_j d[i+j] f[j]
    # Let d_padded[j] = d[j] for j in [0,m-1], 0 otherwise
    # R[i] = sum_j d_padded[i+j] f[j]
    # This is the "afl" type correlation. 
    # FFT approach: R = ifft(fft(d_padded, n_fft) * conj(fft(f_rev, n_fft))) won't work directly
    # Let's think differently: R[i] = sum_j f[j] * d[i+j]
    # Define g[k] = d[k] (length m), h[k] = f[k] (length n)
    # R[i] = sum_j h[j] g[i+j] = (h * g_rev)[reversed offset]
    # Actually simplest: R[i] = sum_j f[j] d[i+j]
    # = sum_{j=0}^{n-1} f[j] d[i+j]
    # Let's just compute it directly for clarity (n=512 is small enough)
    R = np.zeros(n)
    for i in range(n):
        # d[i+j] for j=0..n-1, valid when i+j < m i.e. j < m-i
        valid = min(n, m - i)
        if valid > 0:
            R[i] = np.dot(f[:valid], d[i:i + valid])
    
    grad = (4.0 / m) * R
    return grad


def variance_gradient_fft(f):
    """Faster variance gradient using FFT."""
    n = len(f)
    conv = np.convolve(f, f)
    m = 2 * n - 1
    mu = 1.0 / m
    d = conv - mu

    # R[i] = sum_{j=0}^{min(n-1, m-1-i)} f[j] * d[i+j]
    # This is a cross-correlation. We can compute it as:
    # Let's define d_ext of length n_fft with d_ext[k] = d[k] for k < m, 0 else
    # R[i] = sum_j f[j] d_ext[i+j]
    # This equals (cross-correlation of f and d_ext) at lag = -i... 
    # Actually R[i] = sum_j f[j] d_ext[i+j]
    # = sum_j f[j] d_ext_rev[m-1-i-j] (if we reverse)
    # Let me just use the direct formula with a smarter approach:
    # R = ifft( conj(fft(f, n_fft)) * fft(d_ext, n_fft) ) gives circular correlation
    # But we need linear, and the indexing is tricky.
    # For n=512, the direct method takes 512*512 = 262K ops, which is fast.
    # Let's just do it directly.
    R = np.zeros(n)
    for i in range(n):
        valid = min(n, m - i)
        if valid > 0:
            R[i] = np.dot(f[:valid], d[i:i + valid])
    return (4.0 / m) * R

        if s < 1e-10:
            stall += 1
            if stall > 100:
                break
            continue
        f_new = f_new / s

        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 120:
                break

    return best_f, best_score


def pressure_descent_v2(f, n_iter=300, eta=0.15, rng=None):
    """Pressure-based descent considering top peaks."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()

        K = min(5, 2 * n - 1)
        top_k = np.argsort(conv)[::-1][:K]
        weights = np.exp(-np.arange(K) * 0.3)
        weights /= weights.sum()

        pressure = np.zeros(n)
        for w, j_star in zip(weights, top_k):
            for i in range(n):
                idx = j_star - i
                if 0 <= idx < n:
                    pressure[i] += w * f[idx]

        if pressure.max() > 0:
            pressure = pressure / pressure.max()

        eta_eff = eta / (1.0 + 0.005 * it)
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 60:
                f_pert = f * (1.0 + 0.08 * rng.standard_normal(n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    stall = 0
                else:
                    break

    return best_f, best_score


def variance_grad_descent(f, n_iter=500, lr=0.01, rng=None):
    """Gradient descent on variance of autocorrelation."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        grad = variance_gradient_fft(f)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        lr_eff = lr / (1.0 + 0.01 * it)
        f_new = f - lr_eff * grad
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            stall += 1
            if stall > 100:
                break
            continue
        f_new = f_new / s

        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            # Try with smaller lr
            f_new2 = f - lr_eff * 0.1 * grad
            f_new2 = np.clip(f_new2, 0, None)
            s2 = f_new2.sum()
            if s2 > 1e-10:
                f_new2 = f_new2 / s2
                score2 = compute_score(f_new2)
                if score2 < best_score - 1e-12:
                    best_score = score2
                    best_f = f_new2.copy()
                    f = f_new2
                    stall = 0
                    continue
            stall += 1
            if stall > 80:
                f_pert = f * (1 + 0.05 * rng.standard_normal(n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    stall = 0
                else:
                    break

    return best_f, best_score


def lp_frank_wolfe(f, time_budget, rng=None, t0=0.4):
    """Frank-Wolfe with LP oracle: min t s.t. (f*g)<=t, sum(g)=1, g>=0."""
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)

    best_f = f.copy()
    best_score = compute_score(f)
    t = t0
    start = time.monotonic()
    lp_calls = 0

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()

        g, t_opt = lp_optimize_t(f, time_limit=3)

        if g is None:
            break
        lp_calls += 1

        improved = False
        for step in [min(1.0, t * 1.3), t, t * 0.6, t * 0.3, t * 0.1, t * 0.05]:
            if step <= 0.001:
                break
            f_new = (1 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s

            new_score = compute_score(f_new)
            if new_score < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                t = min(1.0, t * 1.12)
                improved = True
                break

        if not improved:
            t *= 0.5
            if t < 0.002:
                f_pert = f + rng.normal(0, 0.008, n)
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    t = 0.15
                else:
                    break

        if lp_calls % 10 == 0:
            print(f"    LP-FW: calls={lp_calls}, t={t:.4f}, score={best_score:.6f}, "
                  f"elapsed={time.monotonic()-start:.0f}s")

    return best_f, best_score


def alternating_lp(f, time_budget, rng=None):
    """Alternate LP optimization on both arguments (bilinear approach)."""
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    g = f.copy()

    best_f = f.copy()
    best_score = compute_score(f)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        # Fix g, optimize f
        A_g = build_conv_matrix(g)
        m = 2 * n - 1
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.hstack([A_g, -np.ones((m, 1))])
        b_ub = np.zeros(m)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])
        bounds = [(0, None)] * n + [(0, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 3})
            if res.success and res.x is not None:
                f_new = np.clip(res.x[:n], 0, None)
                fs = f_new.sum()
                if fs > 1e-10:
                    f_new = f_new / fs
                    f = 0.5 * f + 0.5 * f_new
                    f = np.clip(f, 0, None)
                    f = f / f.sum()
        except Exception:
            pass

        # Fix f, optimize g
        g_new, _ = lp_optimize_t(f, time_limit=3)
        if g_new is not None:
            g = 0.5 * g + 0.5 * g_new
            g = np.clip(g, 0, None)
            gs = g.sum()
            if gs > 1e-10:
                g = g / gs

        # Evaluate combined
        conv_fg = np.convolve(f, g)
        score_fg = 2.0 * n * conv_fg.max() / (f.sum() * g.sum())
        
        # Also check f*f
        score_ff = compute_score(f)
        
        if score_ff < best_score:
            best_score = score_ff
            best_f = f.copy()

        if score_fg < best_score:
            # Use g as our new candidate
            score_g = compute_score(g)
            if score_g < best_score:
                best_score = score_g
                best_f = g.copy()

        if (int((time.monotonic() - start)) % 10 == 0 and time.monotonic() - start > 1):
            print(f"    Alt-LP: score_ff={score_ff:.6f}, best={best_score:.6f}, "
                  f"elapsed={time.monotonic()-start:.0f}s")

    return best_f, best_score


def coordinate_descent_v2(f, n_sweeps=8, rng=None):
    """Coordinate descent with multiplicative factors."""
    if rng is None:
        rng = np.random.default_rng(0)

    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)

    factors = np.array([0.1, 0.25, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 4.0])

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-12:
                continue

            best_val = original
            best_local = best_score

            for factor in factors:
                f[i] = original * factor
                s = f.sum()
                if s < 1e-12:
                    f[i] = original
                    continue
                conv = np.convolve(f, f)
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]

            f[i] = best_val
            if best_local < best_score - 1e-12:
                best_score = best_local

        f = np.clip(f, 0, None)
        f = normalize(f)
        best_score = compute_score(f)

    return f, best_score


def simulated_annealing_v2(f, rng, n_iter=4000, T0=0.02):
    """Simulated annealing with diverse moves."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    cur_score = best_score
    no_improve = 0

    for it in range(n_iter):
        T = T0 * np.exp(-4.0 * it / n_iter)
        f_new = f.copy()

        r = rng.random()
        if r < 0.25:
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.5))
        elif r < 0.45:
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] * np.exp(rng.normal(0, 0.6)))
        elif r < 0.65:
            center = int(rng.integers(0, n))
            width = max(2, n // 16)
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
            f_new += rng.normal(0, 0.04) * kernel
        elif r < 0.85:
            f_new *= np.exp(rng.normal(0, 0.35, n))
        else:
            # Block swap
            block = max(2, n // 16)
            i = int(rng.integers(0, n - block))
            j = int(rng.integers(0, n - block))
            if i != j:
                f_new[i:i+block], f_new[j:j+block] = f_new[j:j+block].copy(), f_new[i:i+block].copy()

        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score(f_new)

        accept = new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12))
        if accept:
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 1200 and it > n_iter // 2:
            break

    return best_f, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    r = rng.random()
    if r < 0.2:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.4)
        f = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    elif r < 0.4:
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    elif r < 0.6:
        a = rng.uniform(1, 3)
        b = rng.uniform(1, 3)
        f = np.abs(np.sin(np.pi * a * np.arange(n) / n)) * np.abs(np.sin(np.pi * b * np.arange(n) / n))
        f += 0.1
    elif r < 0.8:
        f = np.ones(n)
        for _ in range(rng.integers(3, 8)):
            center = int(rng.integers(0, n))
            width = int(rng.integers(3, n // 4))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / max(1, width ** 2))
            f *= (1 + 0.5 * rng.standard_normal() * kernel)
        f = np.abs(f) + 0.05
    else:
        f = np.ones(n)
        for _ in range(rng.integers(2, 6)):
            center = int(rng.integers(0, n))
            width = int(rng.integers(2, n // 8))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / max(1, width ** 2))
            f -= 0.6 * kernel
        f = np.clip(f, 0.01, None)

    f = np.clip(f, 0.001, None)
    f = f * (1 + 0.05 * rng.standard_normal(n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f
