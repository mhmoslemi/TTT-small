def resample_seq(src, dest_n):

    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def coordinate_descent_pass(a, n_passes=1):

    n = len(a)
    a = a.copy()
    S = a.sum()
    
    for _ in range(n_passes):
        b = fast_conv(a)
        S = a.sum()
        improved = False

def subgradient_step(a, eta):
    """One subgradient step."""
    n = len(a)
    S = a.sum()
    b = fast_conv(a)
    M = b.max()
    k_star = np.argmax(b)
    
    # Subgradient of M w.r.t. a_j: a[k_star - j] (if valid)
    # For j in range [k_star - n + 1, k_star]
    d = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for j in range(lo, hi + 1):
        d[j] = a[k_star - j]
    
    # Gradient of J
    grad = 2.0 * n / (S ** 3) * (S * d - 2.0 * M)
    
    a_new = a - eta * grad
    a_new = np.maximum(a_new, 0.0)
    return a_new

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using coordinate descent
