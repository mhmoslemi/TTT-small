def compute_conv(f):
    """Compute linear convolution of f with itself."""
    return np.convolve(f, f)


def get_score(f):
    """Compute the evaluation score directly."""
    n = len(f)
    conv = compute_conv(f)
    S = np.sum(f)
    if S < 0.01:
        return np.inf
    return float(2 * n * np.max(conv) / S**2)


def soft_max_grad(f, beta):
    """Compute gradient and soft-max value of (1/beta)*log(sum(exp(beta*C[k]))) where C=conv(f,f)."""
    n = len(f)
    C = compute_conv(f)
    maxC = C.max()
    eC = np.exp(beta * (C - maxC))
    total = eC.sum()
    if total < 1e-300:
        return np.zeros(n), 0.0
    p = eC / total
    # grad[j] = 2 * sum_i f[i] * p[j+i]
    # p has length 2n-1, we need sum over i of f[i]*p[j+i] for j=0..n-1
    # This is cross-correlation of p and f (valid mode)
    grad = 2.0 * np.correlate(p, f, mode='valid')
    soft_max_val = maxC + np.log(total) / beta
    return grad, soft_max_val


def fista_softmax(f, deadline, beta_start=5.0, beta_end=500.0, lr=None,
                   max_iters=5000, verbose=False, verbose_prefix=""):
    """
    FISTA (Nesterov accelerated gradient descent) with soft-max surrogate.
    Minimize (1/beta)*log(sum(exp(beta*conv(f,f)[k]))) subject to sum(f)=1, f>=0.
    The projection onto {f>=0, sum(f)=1} is: clip to non-negative, then normalize.
    """
    f = f.copy().astype(np.float64)
    S = np.sum(f)
    if S < 1e-10:
        return f, np.inf
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

    # FISTA state
    y = f.copy()
    t_k = 1.0

    # Step size
    if lr is None:
        lr = 0.5 / (2.0 * n * beta_end)
    lr_current = lr

    it = 0
    no_improve = 0
    patience = 800

    while time.monotonic() < deadline and it < max_iters:
        it_frac = min(1.0, it / (max_iters * 0.85))
        beta = beta_start * (beta_end / beta_start) ** it_frac

        # Gradient at y
        grad_y, sm_y = soft_max_grad(y, beta)

        # Scale gradient by beta (since we minimize (1/beta)*log(sum(exp(beta*C))),
        # the gradient is beta * (soft_max_grad as computed) ... actually let me recheck)
        # L_beta(f) = (1/beta) * log(sum_k exp(beta * C_k))
        # dL/df[j] = (1/beta) * sum_k beta * C_k' * exp(beta*C_k) / sum_k exp(beta*C_k)
        #          = sum_k C_k' * p_k
        # where C_k' = dC_k/df[j] = 2*f[k-j] (with appropriate bounds)
        # So the gradient IS what soft_max_grad returns (the beta cancels)
        # But we want a Lipschitz constant for the step size...
        # Actually the Hessian of L_beta involves beta, so larger beta -> less smooth
        # Step size should scale as 1/beta for safety
        
        step = lr_current / max(beta, 1.0)
        
        # Gradient step
        x_new = y - step * grad_y
        # Project onto {f >= 0, sum(f) = 1}
        x_new = np.clip(x_new, 0.0, 1000.0)
        S_new = np.sum(x_new)
        if S_new < 1e-10:
            x_new = y * 1.0
            S_new = 1.0
        x_new = x_new / S_new

        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k**2)) / 2.0
        y_new = x_new + ((t_k - 1.0) / t_new) * (x_new - y)
        y_new = np.clip(y_new, 0.0, None)
        S_y = np.sum(y_new)
        if S_y < 1e-10:
            y_new = x_new * 1.0
            S_y = 1.0
        y_new = y_new / S_y

        # Update
        f = x_new
        y = y_new
        t_k = t_new
        it += 1

        # Periodic evaluation
        if it % 20 == 0 or it < 5:
            sc = get_score(f)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 200 == 0:
                    print(f"    {verbose_prefix}FISTA iter {it}: score={best_score:.6f} (beta={beta:.1f})")
            else:
                no_improve += 1
                if no_improve > patience:
                    # Reset momentum, reduce step size slightly
                    y = f.copy()
                    t_k = 1.0
                    no_improve = 0
                    if verbose:
                        print(f"    {verbose_prefix}FISTA reset momentum at iter {it}")

    # Final evaluation
    final_score = get_score(f)
    if final_score < best_score:
        best_score = final_score
        best_f = f.copy()

    return best_f, best_score


def mirror_descent(f, deadline, beta=50.0, lr=None, max_iters=3000, verbose=False, verbose_prefix=""):
    """Mirror descent with exponential weights - natural for non-negative optimization."""
    f = f.copy().astype(np.float64)
    S = np.sum(f)
    if S < 1e-10:
        return f, np.inf
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

    if lr is None:
        lr = 1.0 / (n * 2.0)

    it = 0
    no_improve = 0

    while time.monotonic() < deadline and it < max_iters:
        it_frac = min(1.0, it / (max_iters * 0.8))
        b = beta * (20.0 ** it_frac)

        grad, _ = soft_max_grad(f, b)

        # Mirror descent update: f[j] * exp(-lr * grad[j])
        # To avoid underflow, subtract the mean
        grad_shifted = grad - np.mean(grad)
        f_new = f * np.exp(-lr * b * grad_shifted)
        f_new = np.clip(f_new, 1e-30, None)
        S_new = np.sum(f_new)
        if S_new < 1e-10:
            f_new = f.copy()
            S_new = 1.0
        f_new = f_new / S_new

        f = f_new
        it += 1

        if it % 50 == 0:
            sc = get_score(f)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 300:
                    break

    final_score = get_score(f)
    if final_score < best_score:
        best_score = final_score
        best_f = f.copy()

    return best_f, best_score


def active_set_lp(f, deadline, top_k=20, max_lps=30, verbose=False, verbose_prefix=""):
    """
    Active-set LP refinement: identify top peaks in autocorrelation,
    then solve LP to find a direction that reduces those peaks while maintaining sum.
    """
    f = f.copy().astype(np.float64)
    S = np.sum(f)
    if S < 1e-10:
        return f, np.inf
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

    for lp_iter in range(max_lps):
        if time.monotonic() > deadline:
            break

        conv_f = compute_conv(f)
        M = conv_f.max()
        S_target = np.sum(f)

        # Get top K active constraints
        top_indices = np.argsort(conv_f)[-top_k:]
        top_indices = np.sort(top_indices)

        # Build constraint matrix: for each active k, conv(f, g)[k] <= M
        # conv(f, g)[k] = sum_j f[j] * g[k-j] (with valid bounds)
        rows = []
        cols = []
        vals = []
        for k in top_indices:
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                rows.append(len(rows) // max(1, len(top_indices)) if False else 0)  # placeholder
                rows[-1] = top_indices.tolist().index(k) if k in top_indices.tolist() else 0
                cols.append(j)
                vals.append(f[k - j])

        # Rebuild properly
        rows = []
        cols = []
        vals = []
        for idx, k in enumerate(top_indices):
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(f[k - j])

        if len(rows) == 0:
            break

        A = csr_matrix((vals, (rows, cols)), shape=(len(top_indices), n))
        c = -np.ones(n)  # maximize sum(g)
        b_ub = np.full(len(top_indices), M)
        bounds = [(0, None)] * n

        try:
            result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                             options={'time_limit': 5})
            if result.success:
                g = result.x
                sum_g = np.sum(g)
                if sum_g > 1e-10:
                    g = g / S_target  # normalize to same sum as f

                    # Try several step sizes
                    best_t = 0.1
                    best_s = best_score
                    for trial_t in [0.05, 0.1, 0.15, 0.2, 0.3]:
                        new_f = (1.0 - trial_t) * f + trial_t * g
                        new_f = np.clip(new_f, 0.0, 1000.0)
                        ns = np.sum(new_f)
                        if ns < 1e-10:
                            continue
                        new_f = new_f / ns
                        sc = get_score(new_f)
                        if np.isfinite(sc) and sc < best_s:
                            best_s = sc
                            best_t = trial_t

                    if best_s < best_score:
                        f = (1.0 - best_t) * f + best_t * g
                        f = np.clip(f, 0.0, 1000.0)
                        S = np.sum(f)
                        f = f / S
                        best_f = f.copy()
                        best_score = best_s
                        if verbose:
                            print(f"    {verbose_prefix}LP iter {lp_iter}: score={best_score:.6f} (t={best_t:.3f}, K={top_k})")
                else:
                    break
        except Exception:
            break

    return best_f, best_score


def coordinate_descent(f, deadline, max_sweeps=10, verbose=False, verbose_prefix=""):
    """Coordinate descent: optimize one element at a time."""
    f = f.copy().astype(np.float64)
    S = np.sum(f)
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old_val = f[j]
            f[j] = 0.0
            conv_base = compute_conv(f)

            # Find optimal f[j] by binary search or golden section
            # conv(f,j)[k] = conv_base[k] + f[j]*f[k-j] for valid k
            # We want to minimize max_k (conv_base[k] + f[j]*f[k-j])
            # where f[k-j] is the "other" element (0 if out of bounds)

            # For each k, the contribution is f[j] * f[k-j]
            # The max is at the k where conv_base[k] + f[j]*f[k-j] is largest
            # This is a 1D optimization: minimize max_k (a_k + f[j]*b_k)

            # b_k = f[k-j] if 0 <= k-j < n else 0
            bs = np.zeros(2 * n - 1)
            j_min_k = 0
            j_max_k = 2 * n - 2
            for k in range(2 * n - 1):
                other = k - j
                if 0 <= other < n:
                    bs[k] = f[other]

            # Maximize f[j] [for good sum] but minimize max(conv_base + f_val * bs)
            # Binary search on f_val: find smallest f_val such that
            # no constraint is violated... actually we want to find f_val that
            # minimizes the max while keeping sum close to 1

            # Simple approach: try a few values
            best_fj = old_val
            best_max = np.max(conv_base + old_val * bs)
            for fj_candidate in [old_val * 0.5, old_val * 0.7, old_val * 0.9,
                                  old_val * 1.1, old_val * 1.3, old_val * 1.5,
                                  0.0, old_val * 2.0]:
                fj_candidate = max(0.0, fj_candidate)
                candidate_max = np.max(conv_base + fj_candidate * bs)
                if candidate_max < best_max:
                    best_max = candidate_max
                    best_fj = fj_candidate

            f[j] = best_fj
            new_S = np.sum(f)
            if new_S > 0.01:
                f = f / new_S

            if sweep == 0 and not improved:
                # Check if any improvement happened
                pass

            if abs(f[j] - old_val / (S / new_S)) > 1e-10 if S > 0 else True:
                improved = True

        sc = get_score(f)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f.copy()
            if verbose and sweep % 2 == 0:
                print(f"    {verbose_prefix}CD sweep {sweep}: score={best_score:.6f}")

    return best_f, best_score


def tukey_window(n, r=0.3):
    """Tukey (tapered cosine) window."""
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def hann_window(n):
    """Hann window."""
    return 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))


def blackman_window(n):
    """Blackman window."""
    return 0.42 - 0.5 * np.cos(2 * np.pi * np.arange(n) / (n - 1)) + \
           0.08 * np.cos(4 * np.pi * np.arange(n) / (n - 1))


def super_gaussian(n, alpha=2.0):
    """Super-Gaussian window."""
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    w = np.exp(-((idx - center) / (n / 2.0)) ** (2 * alpha))
    return w


def raised_sine(n, p=2.0):
    """Raised sine to the p-th power."""
    x = np.linspace(0, np.pi, n)
    return np.sin(x) ** p


def random_smooth(n, seed, sigma=5.0):
    """Random smooth non-negative sequence."""
    rng = np.random.default_rng(seed)
    x = rng.standard_normal(n)
    kernel_len = max(3, int(6 * sigma))
    if kernel_len % 2 == 0:
        kernel_len += 1
    kernel = np.exp(-0.5 * (np.arange(kernel_len) - kernel_len // 2) ** 2 / (sigma ** 2))
    kernel = kernel / kernel.sum()
    smooth = np.convolve(x, kernel, mode='same')
    return np.abs(smooth) + 0.05


def periodic_tapered(n, period, taper_frac=0.1):
    """Periodic pattern with taper."""
    pattern = np.ones(period)
    full = np.tile(pattern, n // period + 1)[:n]
    # Apply taper
    taper_len = int(taper_frac * n)
    if taper_len > 0 and taper_len < n // 2:
        w = np.ones(n)
        for i in range(taper_len):
            frac = i / taper_len
            w[i] = 0.5 * (1 - np.cos(np.pi * frac))
            w[n - 1 - i] = w[i]
        full = full * w
    return full


def generate_initializations(n, rng):
    """Generate diverse initializations for a given length."""
    inits = []
    inits.append(('tukey_02', tukey_window(n, 0.2)))
    inits.append(('tukey_03', tukey_window(n, 0.3)))
    inits.append(('tukey_05', tukey_window(n, 0.5)))
    inits.append(('hann', hann_window(n)))
    inits.append(('blackman', np.abs(blackman_window(n))))
    inits.append(('sg_15', super_gaussian(n, 1.5)))
    inits.append(('sg_20', super_gaussian(n, 2.0)))
    inits.append(('sg_30', super_gaussian(n, 3.0)))
    inits.append(('rsine_2', raised_sine(n, 2.0)))
    inits.append(('rsine_3', raised_sine(n, 3.0)))
    inits.append(('rsine_4', raised_sine(n, 4.0)))
    inits.append(('rand_smooth_1', random_smooth(n, seed=42 + n, sigma=5.0)))
    inits.append(('rand_smooth_2', random_smooth(n, seed=123 + n, sigma=10.0)))
    inits.append(('rand_smooth_3', random_smooth(n, seed=456 + n, sigma=3.0)))
    # Periodic
    for period in [8, 16, 32]:
        if period <= n // 2:
            inits.append((f'periodic_{period}', periodic_tapered(n, period)))
    return inits


def full_optimize_pipeline(f_init, deadline, verbose=False, verbose_prefix=""):
    """Run full optimization pipeline: FISTA -> LP -> CD -> FISTA -> LP."""
    f_init = f_init.copy().astype(np.float64)
    S = np.sum(f_init)
    if S < 1e-10:
        return f_init, np.inf
    f_init = f_init / S

    best_f = f_init.copy()
    best_score = get_score(f_init)
    t0 = time.monotonic()
    total = deadline - t0

    # Phase 1: FISTA (60% of time)
    fista_deadline = t0 + total * 0.55
    if time.monotonic() < fista_deadline:
        f1, s1 = fista_softmax(f_init, fista_deadline, beta_start=5.0, beta_end=300.0,
                               max_iters=5000, verbose=verbose, verbose_prefix=verbose_prefix + "F1")
        if s1 < best_score:
            best_f, best_score = f1, s1

    # Phase 2: Active-set LP (20% of time)
    lp_deadline = t0 + total * 0.75
    if time.monotonic() < lp_deadline:
        f2, s2 = active_set_lp(best_f, lp_deadline, top_k=30, max_lps=20,
                               verbose=verbose, verbose_prefix=verbose_prefix + "LP")
        if s2 < best_score:
            best_f, best_score = f2, s2

    # Phase 3: Coordinate descent (10% of time)
    cd_deadline = t0 + total * 0.88
    if time.monotonic() < cd_deadline:
        f3, s3 = coordinate_descent(best_f, cd_deadline, max_sweeps=5,
                                    verbose=verbose, verbose_prefix=verbose_prefix + "CD")
        if s3 < best_score:
            best_f, best_score = f3, s3

    # Phase 4: Final FISTA with high beta (remaining time)
    if time.monotonic() < deadline - 2:
        f4, s4 = fista_softmax(best_f, deadline - 2, beta_start=100.0, beta_end=1000.0,
                               max_iters=3000, verbose=verbose, verbose_prefix=verbose_prefix + "F2")
        if s4 < best_score:
            best_f, best_score = f4, s2 if s4 < s2 else s4
            best_f = f4

    # Final LP polish
    if time.monotonic() < deadline - 1:
        f5, s5 = active_set_lp(best_f, deadline - 1, top_k=15, max_lps=10,
                               verbose=verbose, verbose_prefix=verbose_prefix + "LP2")
        if s5 < best_score:
            best_f, best_score = f5, s5

    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    """
    Main search function. Tries multiple sequence lengths with diverse initializations,
    using FISTA + active-set LP + coordinate descent pipeline.
    """
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)

    # Try to load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                score = evaluate_sequence(parent.tolist())
                if np.isfinite(score):
                    best_score = score
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    # Lengths to try with time allocation
    # Focus on lengths around 512-2048 where we expect good performance
    lengths_and_budgets = [
        (512, 0.22),
        (768, 0.15),
        (1024, 0.25),
        (1536, 0.15),
        (2048, 0.10),
    ]

    total_budget = deadline - t0
    print(f"Total budget: {total_budget:.0f}s")

    for n, frac in lengths_and_budgets:
        if time.monotonic() > deadline - 60:
            print("Time running out, stopping length exploration")
            break

        n_budget = total_budget * frac
        n_deadline = time.monotonic() + n_budget
        print(f"\n{'='*50}")
        print(f"=== n={n} (budget {n_budget:.0f}s) ===")
        print(f"{'='*50}")

        # Generate initializations
        inits = generate_initializations(n, rng)

        # Add parent-based initialization
        if best_sequence is not None:
            if len(best_sequence) == n:
                inits.insert(0, ('parent_exact', best_sequence.copy()))
            else:
                # Interpolate
                parent_interp = np.interp(np.linspace(0, 1, n),
                                          np.linspace(0, 1, len(best_sequence)),
                                          best_sequence)
                parent_interp = np.clip(parent_interp, 0, None)
                inits.insert(0, ('parent_interp', parent_interp))

        # Evaluate all inits quickly
        init_scores = []
        for label, f_init in inits:
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            S = np.sum(f_init)
            if S < 0.01:
                continue
            f_norm = f_init / S
            sc = get_score(f_norm)
            if np.isfinite(sc):
                init_scores.append((sc, label, f_norm))
                if sc < best_score:
                    best_score = sc
                    best_sequence = f_norm.copy()

        # Sort by score and pick top few for optimization
        init_scores.sort(key=lambda x: x[0])
        top_inits = init_scores[:min(6, len(init_scores))]

        print(f"  Top initializations: " +
              ", ".join([f"{lbl}={sc:.4f}" for sc, lbl, _ in top_inits[:5]]))

        # Allocate time among top inits
        n_top = len(top_inits)
        per_init_time = n_budget * 0.85 / max(n_top, 1)

        for i, (init_sc, init_label, f_init) in enumerate(top_inits):
            if time.monotonic() > n_deadline - 10:
                break

            # Give more time to better initializations
            time_weight = 1.0 / (i + 1)
            weights = np.array([1.0 / (j + 1) for j in range(n_top)])
            weights = weights / weights.sum()
            init_deadline = time.monotonic() + n_budget * 0.85 * weights[i]
            init_deadline = min(init_deadline, n_deadline - 5)

            if time.monotonic() >= init_deadline:
                continue

            print(f"  [{init_label}] init_score={init_sc:.4f}, time={init_deadline - time.monotonic():.0f}s")

            try:
                bf, bs = full_optimize_pipeline(f_init, init_deadline,
                                                verbose=False, verbose_prefix=f"[n={n} {init_label}] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  [{init_label}] IMPROVED to {bs:.6f}")
                else:
                    print(f"  [{init_label}] best={bs:.6f} (no improvement over {best_score:.6f})")
            except Exception as e:
                print(f"  [{init_label}] error: {e}")

        # If we already have a very good result, skip remaining lengths
        if best_score < 1.505:
            print(f"\nAlready below 1.505 ({best_score:.6f}), will still try to improve")

    # === Final phase: perturbation restarts and refinement on best ===
    if best_sequence is not None and time.monotonic() < deadline - 30:
        print(f"\n{'='*50}")
        print(f"=== Final refinement phase (best: {best_score:.6f}, n={len(best_sequence)}) ===")
        print(f"{'='*50}")

        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 10

        # Try different perturbation strengths
        perturbations = [0.02, 0.05, 0.1, 0.15]
        per_perturb = remaining / (len(perturbations) + 2)

        for strength in perturbations:
            if time.monotonic() > deadline - 20:
                break

            f_pert = best_sequence * np.clip(1.0 + strength * rng.standard_normal(n_best), 0.1, 2.0)
            f_pert = np.clip(f_pert, 0.0, 1000.0)
            S_p = np.sum(f_pert)
            if S_p < 0.01:
                continue
            f_pert = f_pert / S_p

            pert_deadline = time.monotonic() + per_pert
            try:
                bf, bs = full_optimize_pipeline(f_pert, pert_deadline,
                                                verbose=False, verbose_prefix="[pert] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  Perturb (str={strength:.2f}): IMPROVED to {bs:.6f}")
                else:
                    print(f"  Perturb (str={strength:.2f}): {bs:.6f} (no improvement)")
            except Exception as e:
                print(f"  Perturb error: {e}")

        # Final FISTA + LP on the absolute best
        if time.monotonic() < deadline - 5:
            print(f"\n  Final polish...")
            try:
                bf, bs = fista_softmax(best_sequence, min(deadline - 3, time.monotonic() + remaining * 0.2),
                                       beta_start=200.0, beta_end=2000.0, max_iters=3000,
                                       verbose=False, verbose_prefix="[final_F] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  Final FISTA: {bs:.6f}")
            except Exception as e:
                print(f"  Final FISTA error: {e}")

            if time.monotonic() < deadline - 2:
                try:
                    bf, bs = active_set_lp(best_sequence, deadline - 2, top_k=20, max_lps=15,
                                           verbose=False, verbose_prefix="[final_LP] ")
                    if bs < best_score:
                        best_score = bs
                        best_sequence = bf.copy()
                        print(f"  Final LP: {bs:.6f}")
                except Exception as e:
                    print(f"  Final LP error: {e}")

    # Validate and return
    if best_sequence is None:
        best_sequence = tukey_window(512, 0.3)
        best_score = evaluate_sequence(best_sequence.tolist())

    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S_final = float(np.sum(best_sequence))
    if S_final < 0.01:
        best_sequence = np.ones(256)
        best_score = evaluate_sequence(best_sequence.tolist())

    # Final verification
    final_check = evaluate_sequence(best_sequence.tolist())
    if np.isfinite(final_check) and final_check < best_score:
        best_score = final_check
    elif not np.isfinite(final_check):
        # Fallback
        best_sequence = np.ones(256)
        best_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    print(f"{'='*50}")

    return [float(x) for x in best_sequence]
