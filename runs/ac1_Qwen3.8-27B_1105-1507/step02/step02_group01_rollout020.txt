Optimized step function coefficient search using Gerchberg-Saxton spectral
flattening, LP-based refinement, and adaptive gradient descent.
Combines multiple optimization strategies to minimize the peak-to-mean ratio
of the autocorrelation.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)

def build_A_matrix(a):
    n = len(a)
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for k in range(L):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def solve_lp(a, upper_bound=100.0):
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M

def gs_iterations(a, max_iters=2000, time_limit=30.0, alpha_0=1.0, alpha_decay=0.999):
    """Gerchberg-Saxton iterations to flatten the convolution spectrum."""
    n = len(a)
    L = 2 * n - 1
    a = np.array(a, dtype=np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    alpha = alpha_0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        # Forward FFT
        A = np.fft.fft(a, L)
        
        # Project magnitudes toward target
        mag = np.abs(A)
        target_mag = 1.0 / np.sqrt(L)
        # Soft projection: move mag toward target
        ratio = target_mag / np.maximum(mag, 1e-15)
        ratio = np.clip(ratio, 0.5, 2.0)  # limit per-step change
        A_new = A * (ratio ** alpha)
        
        # Enforce real constraint
        A_new[0] = np.real(A_new[0])
        for k in range(1, L):
            A_new[L - k] = np.conjugate(A_new[k])
        # Re-average to ensure exact Hermitian symmetry
        A_new = (A_new + np.conjugate(np.roll(A_new, -np.arange(L)))) / 2
        # Actually let me do this more carefully
        A_real = np.zeros(L, dtype=complex)
        A_real[0] = np.real(A_new[0])

                A_real[k] = A_new[k]
                A_real[L - k] = np.conjugate(A_new[k])
        else:
            for k in range(1, L // 2):
                A_real[k] = A_new[k]
                A_real[L - k] = np.conjugate(A_new[k])
            A_real[L // 2] = np.real(A_new[L // 2])
        A_new = A_real
        
        # Inverse FFT
        a_new = np.fft.ifft(A_new)[:n]
        
        # Project to non-negative

def block_refine(a, rng, max_time, start_time=None):
    if start_time is None:
        start_time = time.monotonic()
    n = len(a)
    a = np.array(a, dtype=np.float64).copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    while time.monotonic() - start_time < max_time:
        width = max(1, int(n * rng.uniform(0.005, 0.1)))
        pos = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        sigma = 0.02 + 0.05 * rng.random()
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue

        if obj < best_obj:

            best_obj = obj

def init_bimodal(n, rng):
    """Bimodal initialization: high at edges, low in middle."""
    x = np.arange(n) / (n - 1)
    a = 0.3 + 0.7 * (2 * x - 1)**2
    a += 0.1 * rng.random(n)
    a = np.maximum(a, 0)
    s = np.sum(a)
    return a / s if s > 1e-10 else np.ones(n) / n

def init_unimodal(n, rng):
    """Unimodal: high in middle."""
    center = n / 2.0
    width = n * rng.uniform(0.2, 0.5)
    a = np.exp(-0.5 * ((np.arange(n) - center) / width)**2)
    a += 0.05 * rng.random(n)
    s = np.sum(a)
    return a / s if s > 1e-10 else np.ones(n) / n

def init_random_smooth(n, rng):
    """Random smooth initialization."""
    a = np.maximum(0, rng.normal(1.0, 0.5, n))
    # Smooth it
    kernel = np.ones(5) / 5
    a = np.convolve(a, kernel, mode='same')
    s = np.sum(a)
    return a / s if s > 1e-10 else np.ones(n) / n

def init_fourier_based(n, rng):
    """Initialize based on Fourier constraints."""
    L = 2 * n - 1
    # Random phases
    phases = rng.uniform(0, 2 * np.pi, L)
    A = np.exp(1j * phases) / np.sqrt(L)
    # Enforce real constraint
    A[0] = np.abs(A[0])
    for k in range(1, (L+1)//2):
        A[L-k] = np.conjugate(A[k])
    a = np.fft.ifft(A)[:n]
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_cascade(n, rng):
    """Cascade of different frequency components."""
    a = np.zeros(n)
    num_components = int(rng.integers(3, 10))
    for _ in range(num_components):
        freq = rng.uniform(0.02, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 1.0)
        a += amp * np.maximum(0, np.cos(2 * np.pi * freq * np.arange(n) + phase))
    a += 0.1 * rng.random(n)
    a = np.maximum(a, 0)
    s = np.sum(a)
    return a / s if s > 1e-10 else np.ones(n) / n

"""
Optimized step function coefficient search using Gerchberg-Saxton spectral
flattening combined with LP-based refinement and adaptive gradient descent.
The GS method directly targets flat power spectrum in the frequency domain,
which is the key structural property needed to minimize peak autocorrelation.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def build_A_matrix(a):
    n = len(a)
    L = 2 * n - 1
    A = np.zeros((L, n), dtype=np.float64)
    for k in range(L):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def solve_lp(a, upper_bound=100.0):
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M


def gs_iterations(a, max_iters=5000, time_limit=30.0, alpha_start=1.0, alpha_end=0.3, beta=1.0):
    """
    Gerchberg-Saxton iterations to flatten the linear convolution spectrum.
    Alternates between:
      - Fourier domain: project |A[k]| toward constant target
      - Spatial domain: project to non-negative, normalize sum to 1
    """
    n = len(a)
    L = 2 * n - 1
    a = np.array(a, dtype=np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    target_mag = 1.0 / np.sqrt(float(L))

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        # Progressive alpha: start hard, end soft
        frac = min(1.0, it / float(max_iters))
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # Forward FFT (zero-pad to L)
        A = np.fft.fft(a, L)

        # Project magnitudes toward target
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-15)
        ratio = target_mag / mag_safe
        # Clip ratio to avoid extreme jumps
        ratio = np.clip(ratio, 0.3, 3.0)
        scale = ratio ** (alpha * beta)
        A_new = A * scale

        # Enforce Hermitian symmetry (real condition)
        A_new[0] = np.real(A_new[0])
        if L % 2 == 1:
            for k in range(1, (L + 1) // 2):
                A_new[L - k] = np.conjugate(A_new[k])
        else:
            for k in range(1, L // 2):
                A_new[L - k] = np.conjugate(A_new[k])
            A_new[L // 2] = np.real(A_new[L // 2])

        # Inverse FFT
        a_new = np.fft.ifft(A_new)[:n]

        # Project to non-negative
        a_new = np.maximum(np.real(a_new), 0.0)

        # Normalize
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        # Track best
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj

        a = a_new

    return best_a, best_obj


def lp_phase(a, max_iters=80, max_time=30.0):
    """LP-based alternating: find b maximizing sum subject to conv(a,b) <= M."""
    n = len(a)
    a = np.array(a, dtype=np.float64).copy()
    s = np.sum(a)
    if s > 1e-10:
        a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break
        g, g_sum, M = solve_lp(a)
        if g is None:
            break
        if g_sum <= 1.0 + 1e-5:
            stall += 1
            if stall > 12:
                break
            continue
        stall = 0
        g = g / g_sum
        accepted = False
        for t in [0.5, 0.3, 0.2, 0.1]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, None)
            s2 = np.sum(a_new)
            if s2 < 1e-10:
                continue
            a_new = a_new / s2
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_a = a_new.copy()
                best_obj = obj
                a = a_new
                accepted = True
                break
        if not accepted:
            a = 0.5 * a + 0.5 * g
            a = np.clip(a, 0.0, None)
            s2 = np.sum(a)
            if s2 > 1e-10:
                a = a / s2

    return best_a, best_obj


def soft_max_gd(a, max_time=30.0, T_init=None, lr_init=0.1):
    """
    Gradient descent on the soft-max surrogate of max convolution value.
    Uses exact gradient computation via cross-correlation.
    """
    n = len(a)
    a = np.array(a, dtype=np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if T_init is None:
        conv = np.convolve(a, a)
        T_init = 0.03 * float(np.max(conv))
    T = T_init
    lr = lr_init
    last_improve = 0
    patience = 0

    for it in range(100000):
        if time.monotonic() - start > max_time:
            break

        conv = np.convolve(a, a)
        max_c = float(np.max(conv))
        if max_c < 1e-15:
            break

        # Compute probabilities
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500.0, 500.0)
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        p = exp_vals / sum_exp

        # Compute gradient: grad[j] = 2 * sum_l a[l] * p[l+j]
        # Using cross-correlation via convolution
        a_flip = a[::-1]
        cross = np.convolve(a_flip, p, mode='full')
        # cross has length 3n-2; we want indices n-1 to 2n-2
        grad = 2.0 * cross[n - 1: n - 1 + n]

        # Normalize gradient for stable step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        # Adaptive step: target step size relative to a
        step_size = lr * np.mean(a)
        step = (step_size / grad_norm) * grad

        a_new = a - step
        a_new = np.clip(a_new, 0.0, None)
        s2 = np.sum(a_new)
        if s2 < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s2

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-9:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            last_improve = it
            patience = 0
            # Anneal T
            if T > 0.001 * max_c:
                T = max(0.001 * max_c, T * 0.995)
        else:
            patience += 1
            lr *= 0.93
            if patience > 80:
                # Reheat
                T = max(0.005 * max_c, T * 0.5)
                lr = max(lr, 0.02)
                patience = 0
                last_improve = it
            if lr < 1e-8:
                break

    return best_a, best_obj


def block_refine(a, rng, max_time, start_time=None):
    """Random block multiplicative perturbation refinement."""
    if start_time is None:
        start_time = time.monotonic()
    n = len(a)
    a = np.array(a, dtype=np.float64).copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    count = 0

    while time.monotonic() - start_time < max_time:
        width = max(1, int(n * rng.uniform(0.005, 0.08)))
        pos = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        sigma = 0.015 + 0.04 * rng.random()
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            count += 1
            continue
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj
        count += 1

    return best_a, best_obj


def init_bimodal(n, rng):
    """High at edges, low in middle."""
    x = np.arange(n) / max(1, n - 1)
    a = 0.2 + 0.8 * (2.0 * x - 1.0) ** 2
    a += 0.1 * rng.random(n)
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    return a / s if s > 1e-10 else np.ones(n) / n


def init_unimodal(n, rng):
    """High in middle, low at edges."""
    center = n / 2.0
    width = n * rng.uniform(0.15, 0.5)
    a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
    a += 0.05 * rng.random(n)
    s = np.sum(a)
    return a / s if s > 1e-10 else np.ones(n) / n


def init_random_smooth(n, rng):
    """Random with smoothing."""
    a = np.maximum(0.0, rng.normal(1.0, 0.6, n))
    kernel_size = max(2, n // 30)
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(a, kernel, mode='same')
    s = np.sum(a)
    return a / s if s > 1e-10 else np.ones(n) / n


def init_fourier_flat(n, rng):
    """Initialize with random phases targeting flat spectrum."""
    L = 2 * n - 1
    phases = rng.uniform(0.0, 2.0 * np.pi, L)
    A = np.exp(1j * phases) / np.sqrt(float(L))
    # Enforce Hermitian symmetry
    A[0] = np.abs(A[0])
    if L % 2 == 1:
        for k in range(1, (L + 1) // 2):
            A[L - k] = np.conjugate(A[k])
    else:
        for k in range(1, L // 2):
            A[L - k] = np.conjugate(A[k])
        A[L // 2] = np.abs(A[L // 2])
    a = np.fft.ifft(A)[:n]
    a = np.maximum(np.real(a), 0.0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_cascade(n, rng):
    """Superposition of smooth oscillatory components."""
    a = np.zeros(n)
    num_comp = int(rng.integers(3, 12))
    for _ in range(num_comp):
        freq = rng.uniform(0.01, 0.35)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.uniform(0.2, 1.0)
        a += amp * np.maximum(0.0, np.cos(2.0 * np.pi * freq * np.arange(n) + phase))
    a += 0.05 * rng.random(n)
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    return a / s if s > 1e-10 else np.ones(n) / n


def init_alternating(n, rng):
    """Alternating high-low pattern with noise."""
    a = np.ones(n) * 0.5
    a[::2] = 1.0 + 0.3 * rng.random(n // 2 + 1)[:len(a[::2])]
    a[1::2] = 0.2 + 0.3 * rng.random((n - 1) // 2 + 1)[:len(a[1::2])]
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    return a / s if s > 1e-10 else np.ones(n) / n

                    best_score = obj_r4
                    best_seq = a_r4.copy()
                    print(f"  BP refine: obj={best_score:.6f}")

        # Final GS
        if time.monotonic() < deadline - 2:
            t5 = deadline - time.monotonic() - 1.0
            if t5 > 2:
                a_r5, obj_r5 = gs_iterations(best_seq.copy(), max_iters=10000,
                                             time_limit=t5, alpha_start=0.2,
                                             alpha_end=0.1, beta=0.5)
                if obj_r5 < best_score:
                    best_score = obj_r5
                    best_seq = a_r5.copy()
                    print(f"  Final GS: obj={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(200) / 200.0
        best_score = compute_obj(best_seq)
        print(f"Fallback used: obj={best_score:.6f}")

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
