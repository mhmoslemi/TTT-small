    """Compute gradient of sum_k weights[k]*(a*a)[k] w.r.t. a.
    weights has length 2n-1, a has length n."""

"""
Multi-scale Nesterov-accelerated projected gradient optimization with LP refinement
for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses FFT-based convolution, adaptive smoothed gradients, multi-scale interpolation,
and diverse structured initializations to escape local minima.

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_grad(weights, a):
    """Compute gradient of sum_k weights[k]*(a*a)[k] w.r.t. a.
    weights has length 2n-1, a has length n.
    Returns gradient of length n."""
    n = len(a)
    a_rev = a[::-1]
    size = 1
    while size < 3 * n:
        size <<= 1
    fa = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(weights, size)
    conv = np.fft.irfft(fa * fw, size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad


def nesterov_projected_grad(a, max_time=30.0, tau_start=5.0, tau_end=50.0,
                            step_init=0.1, topk_start=None, topk_end=None):
    """Nesterov-accelerated projected gradient descent.
    Minimizes max(conv(a,a)) with sum(a)=1, a>=0 constraint.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    a_prev = a.copy()
    momentum = 0.9
    step = step_init

    start = time.monotonic()
    it = 0
    stall = 0
    patience = 200

    while time.monotonic() - start < max_time:
        it += 1
        frac = it / max(1, int(max_time * 10))
        frac = min(frac, 1.0)

        # Nesterov extrapolation
        a_extrap = a + momentum * (a - a_prev)
        a_extrap = np.maximum(a_extrap, 0)
        s_ex = np.sum(a_extrap)
        if s_ex < 1e-10:
            a_extrap = a.copy()
            s_ex = 1.0
        a_extrap = a_extrap / s_ex

        # Compute convolution and gradient
        conv = compute_conv(a_extrap)
        current_max = float(np.max(conv))

        # Adaptive switching between log-sum-exp and top-k
        if topk_start is not None and frac > topk_start:
            # Use top-k
            k = max(1, int((topk_end or 1) + (topk_start - frac) / max(topk_start, 1e-10) * (10 - (topk_end or 1))))
            k = min(k, len(conv))
            top_idx = np.argsort(conv)[-k:]
            weights = np.zeros(len(conv))
            weights[top_idx] = 1.0 / k
        else:
            # Log-sum-exp smoothing
            tau = tau_start + (tau_end - tau_start) * frac
            max_conv = np.max(conv)
            shifted = tau * (conv - max_conv)
            shifted = np.clip(shifted, -500, 0)
            weights = np.exp(shifted)
            wsum = np.sum(weights)
            if wsum < 1e-300:
                weights = np.zeros(len(conv))
                weights[int(np.argmax(conv))] = 1.0
            else:
                weights /= wsum

        grad = compute_grad(weights, a_extrap)
        # Center gradient for sum constraint
        grad = grad - np.mean(grad)

        # Take step with backtracking
        a_new = a_extrap - step * grad
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            step *= 0.5
            continue
        a_new = a_new / s_new

        # Evaluate new objective
        new_max = float(np.max(compute_conv(a_new)))
        
        if new_max < current_max + 1e-10:
            # Accept
            step = min(step * 1.005, 0.5)
            a_prev = a
            a = a_new
            if new_max < best_max - 1e-8:
                best_max = new_max
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
        else:
            # Reject, reduce step
            step *= 0.7
            stall += 1
            if step < 1e-8:
                break

        # Update momentum periodically
        if it % 50 == 0:
            momentum = 0.95 - 0.05 * frac

        # Stall handling
        if stall > patience:
            # Add noise to escape
            noise = np.random.default_rng(it * 31 + 7).normal(0, 0.02, n)
            a = a + noise
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s < 1e-10:
                break
            a = a / s
            a_prev = a.copy()
            step = step_init * 0.5
            stall = 0
            patience = int(patience * 1.5)

    return best_a, best_max


def build_A_matrix(a):
    """Build the constraint matrix for LP: (a*g)[k] = sum_j A[k,j]*g[j] <= M."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def solve_lp_for_g(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None


def lp_refine(a, max_iter=30, t=0.3, time_limit_per_lp=15):
    """LP-based refinement step."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    stall = 0
    for it in range(max_iter):
        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_for_g(a, M, time_limit=time_limit_per_lp)
        if g is None:
            break

        if sum_g <= 1.0 + 1e-3:
            stall += 1
            if stall > 10:
                break
            continue
        stall = 0

        g = g / sum_g
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_obj

def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

"""
Multi-scale Nesterov-accelerated projected gradient with LP refinement
for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Key improvements over previous approach:
- Nesterov momentum for faster convergence
- Multi-scale interpolation to explore larger n
- FFT-based gradient computation (exact, not approximate)
- Diverse structured initializations including chirp and multi-harmonic
- Adaptive top-k / log-sum-exp gradient switching
- Exploration up to n=2000
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full autocorrelation conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_grad(weights, a):
    """Compute exact gradient of sum_k weights[k]*(a*a)[k] w.r.t. a.
    weights has length 2n-1, a has length n. Returns gradient of length n.
    Uses the identity: grad[m] = 2 * (a_rev * weights)[m+n-1]
    where a_rev = a[::-1] and * denotes full convolution.
    """
    n = len(a)
    a_rev = a[::-1]
    size = 1
    while size < 3 * n:
        size <<= 1
    fa = np.fft.rfft(a_rev, size)
    fw = np.fft.rfft(weights, size)
    conv = np.fft.irfft(fa * fw, size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad


def nesterov_projected_grad(a, max_time=30.0, tau_start=5.0, tau_end=80.0,
                            step_init=0.1, topk_frac=None, topk_count=3):

    Returns (best_a, best_max_conv).

    a = a / s

    best_max = float(np.max(compute_conv(a)))

    step = step_init
    best_step = step_init

    stall = 0

        it += 1

        a_extrap = a + momentum * (a - a_prev)
        a_extrap = np.maximum(a_extrap, 0)
        s_ex = np.sum(a_extrap)
        if s_ex < 1e-10:
            a_extrap = a.copy()
            s_ex = 1.0
        a_extrap = a_extrap / s_ex

        current_max = float(np.max(conv))

def build_A_matrix(a):
    """Build constraint matrix for LP: (a*g)[k] = sum_j A[k,j]*g[j]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def solve_lp_for_g(a, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None


def lp_refine(a, max_iter=30, t=0.3, time_limit_per_lp=12, stop_time=None):
    """LP-based refinement. Alternates between solving LP and taking a step."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    stall = 0
    for it in range(max_iter):
        if stop_time is not None and time.monotonic() > stop_time:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_for_g(a, M, time_limit=time_limit_per_lp)
        if g is None:
            break

        if sum_g <= 1.0 + 1e-3:
            stall += 1
            if stall > 8:
                break
            continue
        stall = 0

        g = g / sum_g
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_obj


def generate_initialization(n, method, rng):

        k = max(1, n // 15)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
    elif method == 4:
        t = x / max(n - 1, 1)
        a = 0.5 * (t ** 2 + (1 - t) ** 2) + 0.4

    a = np.maximum(a, 0.0)
    s = np.sum(a)

def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n < 2:
        return np.full(new_n, a[0] if n == 1 else 0.5)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 0)

        if remaining_for_lp > 15:
            a3, obj3 = lp_refine(best_a, max_iter=int(remaining_for_lp / 12),
                                  t=0.3, time_limit_per_lp=min(12, remaining_for_lp // 2),
                                  stop_time=stop_time)
            if obj3 < best_obj:
                best_obj = obj3
                best_a = a3.copy()

            # Phase 4: Final gradient polish after LP
            if stop_time is None or time.monotonic() < stop_time - 5:
                a4, _ = nesterov_projected_grad(best_a, max_time=8.0,
                                                tau_start=50.0, tau_end=200.0,
                                                step_init=0.03, topk_frac=0.3, topk_count=2)
                obj4 = compute_obj(a4)
                if obj4 < best_obj:
                    best_obj = obj4
                    best_a = a4.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-scale Nesterov + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Explores diverse initializations and sequence lengths (50-2000),
    uses Nesterov momentum with exact FFT-based gradients,
    multi-scale interpolation, and LP for global improvements at moderate n.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            s_p = np.sum(parent)
            if s_p >= 0.01:
                parent_obj = compute_obj(parent)
                print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
                best_seq = parent.copy()
                best_score = parent_obj
    except Exception:
        pass

    # Sequence lengths to explore
    n_values = [50, 80, 100, 120, 150, 200, 250, 300, 400, 500, 700, 1000, 1500, 2000]

    # Add parent length and neighbors
    if best_seq is not None:
        n_parent = len(best_seq)
        extra_ns = [n_parent, max(30, n_parent // 2), min(2500, n_parent * 2)]
        n_values = sorted(set(n_values + [x for x in extra_ns if x >= 30]))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    # Allocate time proportionally, favoring medium-large n
    # Smaller n get less time (quick to optimize, limited improvement potential)
    # Medium n (200-500) get moderate time
    # Large n (500-2000) get more time (more room for improvement)
    time_weights = {}
    for n in n_values:
        if n <= 100:
            time_weights[n] = 25.0
        elif n <= 200:
            time_weights[n] = 40.0
        elif n <= 400:
            time_weights[n] = 60.0
        elif n <= 700:
            time_weights[n] = 70.0
        elif n <= 1000:
            time_weights[n] = 80.0
        else:
            time_weights[n] = 70.0

    total_weight = sum(time_weights.values())
    total_avail = deadline - time.monotonic()

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        alloc = min(time_weights[n] * total_avail / total_weight, remaining * 0.4)
        alloc = max(15.0, alloc)
        use_lp = n <= 400

        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s, LP={use_lp}")

        # Number of initializations based on time
        n_inits = max(2, int(alloc / 18))
        per_init_time = alloc / n_inits

        for init_idx in range(n_inits):
            if time.monotonic() > deadline:
                break
            method = (init_idx + int(seed % 7)) % 8
            a_init = generate_initialization(n, method, rng)

            a_opt, obj_opt = run_optimization_pipeline(
                n, a_init, time_budget=per_init_time,
                use_lp=use_lp, stop_time=deadline)

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  UPDATE n={n} init{init_idx} (m={method}): obj={best_score:.6f}")

    # Multi-scale refinement: interpolate best to larger n
    if best_seq is not None and time.monotonic() < deadline - 40:
        print(f"\n--- Multi-scale refinement ---")
        print(f"Best: n={len(best_seq)}, obj={best_score:.6f}")

        n_cur = len(best_seq)
        a_cur = best_seq.copy()

        for _ in range(4):
            if time.monotonic() > deadline - 25:
                break
            n_new = min(n_cur * 2, 2500)
            if n_new <= n_cur:
                break

            remaining = deadline - time.monotonic()
            alloc = min(remaining * 0.3, 150.0)
            if alloc < 25:
                break

            print(f"  Interpolating n={n_cur} -> n={n_new}, time={alloc:.0f}s")
            a_interp = interpolate_sequence(a_cur, n_new)
            use_lp = n_new <= 400

            a_opt, obj_opt = run_optimization_pipeline(
                n_new, a_interp, time_budget=alloc,
                use_lp=use_lp, stop_time=deadline)

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                n_cur = n_new
                a_cur = a_opt.copy()
                print(f"  IMPROVED: n={n_new}, obj={best_score:.6f}")
            else:
                print(f"  No improvement at n={n_new} (got {obj_opt:.6f} vs {best_score:.6f})")
                break

    # Final intensive polishing
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n--- Final polishing (n={n_best}, {remaining:.0f}s) ---")

        a_pol = best_seq.copy()
        polish_iter = 0
        consecutive_no_improve = 0

        while time.monotonic() < deadline - 8:
            polish_iter += 1
            remaining = deadline - time.monotonic()
            if remaining < 12:
                break

            # Alternating strategies
            if polish_iter % 2 == 1:
                # High-tau gradient (focus on the peak)
                t_pol = min(remaining * 0.35, 50.0)
                a_pol, _ = nesterov_projected_grad(
                    a_pol, max_time=t_pol, tau_start=80.0, tau_end=300.0,
                    step_init=0.025, topk_frac=0.25, topk_count=2)
            else:
                # Lower tau gradient (broader view)
                t_pol = min(remaining * 0.35, 50.0)
                a_pol, _ = nesterov_projected_grad(
                    a_pol, max_time=t_pol, tau_start=20.0, tau_end=80.0,
                    step_init=0.06, topk_frac=None)

            obj_pol = compute_obj(a_pol)

            if obj_pol < best_score - 1e-9:
                best_score = obj_pol
                best_seq = a_pol.copy()
                consecutive_no_improve = 0
                print(f"  Polish {polish_iter}: obj={best_score:.6f}")
            else:
                consecutive_no_improve += 1

                # Try LP if applicable
                if n_best <= 350 and remaining > 35 and consecutive_no_improve >= 2:
                    a_lp, obj_lp = lp_refine(a_pol, max_iter=int(remaining / 18),
                                             t=0.2, time_limit_per_lp=10,
                                             stop_time=deadline)
                    if obj_lp < best_score:
                        best_score = obj_lp
                        best_seq = a_lp.copy()
                        a_pol = a_lp.copy()
                        consecutive_no_improve = 0
                        print(f"  Polish {polish_iter} LP: obj={best_score:.6f}")
                    else:
                        # Perturbation to escape
                        noise_scale = 0.03 * consecutive_no_improve
                        a_pert = a_pol * np.exp(rng.normal(0, min(noise_scale, 0.15), n_best))
                        a_pert = np.maximum(a_pert, 0)
                        s_p = np.sum(a_pert)
                        if s_p > 1e-10:
                            a_pert = a_pert / s_p
                            obj_p = compute_obj(a_pert)
                            if obj_p < best_score:
                                best_score = obj_p
                                best_seq = a_pert.copy()
                                a_pol = a_pert.copy()
                                consecutive_no_improve = 0
                                print(f"  Polish {polish_iter} perturb: obj={best_score:.6f}")
                            else:
                                print(f"  Polish {polish_iter}: no improvement "
                                      f"(sg={obj_pol:.6f}, lp={obj_lp:.6f}, pert={obj_p:.6f})")
                else:
                    # Just perturb
                    if remaining > 20:
                        a_pert = a_pol * np.exp(rng.normal(0, 0.04, n_best))
                        a_pert = np.maximum(a_pert, 0)
                        s_p = np.sum(a_pert)
                        if s_p > 1e-10:
                            a_pert = a_pert / s_p
                            obj_p = compute_obj(a_pert)
                            if obj_p < best_score:
                                best_score = obj_p
                                best_seq = a_pert.copy()
                                a_pol = a_pert.copy()
                                consecutive_no_improve = 0
                                print(f"  Polish {polish_iter} perturb: obj={best_score:.6f}")
                            else:
                                print(f"  Polish {polish_iter}: no improvement (obj={obj_pol:.6f})")

            if consecutive_no_improve > 5 and remaining > 30:
                # Major restart: random perturbation
                a_pert = a_pol * (1.0 + 0.3 * rng.normal(0, 1, n_best))
                a_pert = np.maximum(a_pert, 0)
                s_p = np.sum(a_pert)
                if s_p > 1e-10:
                    a_pert = a_pert / s_p
                    obj_p = compute_obj(a_pert)
                    t_restart = min(remaining * 0.3, 40.0)
                    a_r, _ = nesterov_projected_grad(
                        a_pert, max_time=t_restart, tau_start=10.0, tau_end=60.0,
                        step_init=0.08, topk_frac=0.5, topk_count=3)
                    obj_r = compute_obj(a_r)
                    if obj_r < best_score:
                        best_score = obj_r
                        best_seq = a_r.copy()
                        a_pol = a_r.copy()
                        consecutive_no_improve = 0
                        print(f"  Restart {polish_iter}: obj={best_score:.6f}")
                    else:
                        print(f"  Restart {polish_iter}: no improvement (obj={obj_r:.6f})")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)
        print(f"  Using fallback: obj={best_score:.6f}")

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    # Prepare result ensuring all values are valid floats
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate
    validation = evaluate_sequence(result)
    if validation != np.inf and abs(validation - best_score) > 1e-4:
        print(f"  Validation discrepancy: computed={best_score:.6f}, validated={validation:.6f}")
        best_score = validation

    return result
