def smooth_max_gradient(a, epsilon):
    """Compute gradient of (1/epsilon)*log(sum exp(epsilon*conv_k))."""
    n = len(a)
    conv = np.convolve(a, a)
    # Numerically stable log-sum-exp
    max_conv = np.max(conv)
    shifted = epsilon * (conv - max_conv)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    # Gradient = 2 * correlate(w, a, 'valid')
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad, float(max_conv + np.log(np.sum(exp_shifted)) / epsilon)

    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()

    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def normalize(a):
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def smooth_max_gradient(a, epsilon):
    n = len(a)
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    shifted = epsilon * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad

def mirror_descent(a, n_steps, eps_start, eps_end, eta_scale=0.5):
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for step in range(n_steps):
        t = step / max(1, n_steps - 1)
        eps = eps_start * (eps_end / eps_start) ** t
        eta = eta_scale * n / (1.0 + 0.1 * t * n)
        
        grad = smooth_max_gradient(a, eps)
        
        # Mirror descent update
        log_ratio = -eta * grad
        log_ratio -= np.max(log_ratio)
        a_new = a * np.exp(log_ratio)
        s = np.sum(a_new)
        if s < 1e-15:
            break
        a_new /= s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        
    return best_a, best_obj

def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A

def solve_lp_g(a):
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_alternate(a_init, max_iter=200, t_blend=0.5):
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        g = solve_lp_g(a)
        if g is None:
            break
        g = normalize(g)
        
        obj_g = compute_obj(g)
        if obj_g < best_obj:
            best_obj = obj_g
            best_a = g.copy()
        
        a_new = (1 - t_blend) * a + t_blend * g
        a = normalize(np.maximum(a_new, 0))
        
        obj_a = compute_obj(a)
        if obj_a < best_obj:
            best_obj = obj_a
            best_a = a.copy()
    
    return best_a, best_obj

"""
Optimization of step function heights using LP alternating, mirror descent with
smooth max, and active-set polishing. Key innovations over previous approach:
- Efficient O(n) gradient computation for smooth max via cross-correlation
- Mirror descent (multiplicative weights) for precise polishing
- Geometric mean blending in LP phase
- Better annealing schedules and step-size adaptation
- Asymmetric initializations exploiting the fact that asymmetry reduces center autocorrelation
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    """Normalize to sum to 1."""
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def smooth_max_gradient(a, epsilon):
    """
    Compute gradient of smooth max: (1/eps)*log(sum_k exp(eps*(a*a)_k)).
    Uses efficient cross-correlation for O(n) gradient after O(n log n) conv.
    """
    n = len(a)
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    shifted = epsilon * (conv - max_conv)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    # Gradient of smooth max = 2 * correlate(w, a, 'valid')
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad


def smooth_max_value(a, epsilon):
    """Compute the smooth max value (for monitoring)."""
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    shifted = epsilon * (conv - max_conv)
    shifted = np.clip(shifted, -500.0, 500.0)
    return float(max_conv + np.log(np.sum(np.exp(shifted))) / epsilon)


def mirror_descent(a_init, n_steps, eps_start=5.0, eps_end=50.0,
                   eta_scale=1.0, use_backtrack=True):
    """
    Mirror descent (multiplicative weights) with smooth max objective.
    Anneals epsilon (temperature) and step size over iterations.
    """
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    current_obj = best_obj

    for step in range(n_steps):
        t = step / max(1, n_steps - 1)
        # Anneal epsilon: start small (smoother), end large (sharper)
        eps = eps_start * (eps_end / eps_start) ** t
        # Step size: start larger, decay over time
        eta = eta_scale * n / (1.0 + 50.0 * t)

        grad = smooth_max_gradient(a, eps)

        # Mirror descent update in log-space
        log_ratio = -eta * grad
        log_ratio -= np.max(log_ratio)
        multiplier = np.exp(log_ratio)
        a_new = a * multiplier
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-15:
            break
        a_new /= s

        if use_backtrack:
            obj_new = compute_obj(a_new)
            if obj_new > current_obj * 1.001 and step > 10:
                # Backtrack: reduce step
                a_new = a * np.exp(0.5 * (log_ratio - np.max(log_ratio)))
                a_new = np.maximum(a_new, 0.0)
                s = np.sum(a_new)
                if s < 1e-15:
                    break
                a_new /= s
                obj_new = compute_obj(a_new)
                if obj_new > current_obj * 1.001:
                    a = a  # skip this step
                    continue

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        current_obj = obj_new
        a = a_new

    return best_a, best_obj


def build_A_matrix(a):
    """Build constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        indices = np.arange(j_end, j_start - 1, -1)
        A[k, j_start:j_end + 1] = a[indices]
    return A


def solve_lp_for_g(a):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_alternate(a_init, max_iter=150, t_blend=0.5, use_geometric=False):
    """LP alternating optimization with optional geometric mean blending."""
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for it in range(max_iter):
        g = solve_lp_for_g(a)
        if g is None:
            break
        g = normalize(g)

        obj_g = compute_obj(g)
        if obj_g < best_obj:
            best_obj = obj_g
            best_a = g.copy()

        # Blend
        if use_geometric:
            # Geometric mean in log-space
            a_safe = np.maximum(a, 1e-30)
            g_safe = np.maximum(g, 1e-30)
            log_blend = (1 - t_blend) * np.log(a_safe) + t_blend * np.log(g_safe)
            a_new = np.exp(log_blend - np.max(log_blend))
        else:
            a_new = (1 - t_blend) * a + t_blend * g
        a_new = np.maximum(a_new, 0.0)
        a = normalize(a_new)

        obj_a = compute_obj(a)
        if obj_a < best_obj:
            best_obj = obj_a
            best_a = a.copy()

    return best_a, best_obj


def structured_init(n, rng):
    """Generate diverse structured initializations with emphasis on asymmetry."""
    choice = int(rng.integers(0, 8))
    if choice == 0:

        center = rng.uniform(0.2, 0.8)

    elif choice == 1:
        # Linear tilt
        slope = rng.uniform(0.5, 3.0)
        a = np.exp(slope * (np.arange(n) / n - 0.5))
    elif choice == 2:
        # Cosine with random frequency
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0.01, np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
    elif choice == 3:
        # Two-lobe asymmetric
        a = np.ones(n)
        split = int(n * rng.uniform(0.3, 0.7))
        a[:split] = rng.uniform(0.3, 1.0)
        a[split:] = rng.uniform(0.3, 1.5)
    elif choice == 4:
        # Periodic
        period = int(rng.integers(max(2, n // 50), max(3, n // 5)))
        pattern = rng.random(period) + 0.1
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 5:
        # Quadratic phase (asymmetric)
        a = np.exp(-rng.uniform(0.5, 2.0) * (np.arange(n) / (n - 1)) ** 2)
    elif choice == 6:
        # Sawtooth (inherently asymmetric)
        a = np.abs(np.sin(2 * np.pi * rng.uniform(1, 10) * np.arange(n) / n)) + 0.01
    else:
        # Random
        a = rng.random(n) + 0.01

    return normalize(a)


def active_set_polish(a, n_steps=2000):
    """
    Polish using only the active set (constraints where conv[k] is close to max).
    This is more focused and can converge faster than full smooth max.
    """
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for step in range(n_steps):
        conv = np.convolve(a, a)
        max_c = np.max(conv)
        if max_c < 1e-15:
            break

        # Active set with adaptive tolerance
        tol = max_c * (0.01 + 0.05 * (step / max(1, n_steps)))
        active = conv >= max_c - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            active = conv >= max_c - 1e-10
            n_active = 1

        # Compute subgradient as average of active gradients
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            # grad += 2 * a[k-m] for valid m
            m_start = max(0, k - n + 1)
            m_end = min(n - 1, k)
            for m in range(m_start, m_end + 1):
                grad[m] += 2.0 * a[k - m]
        grad /= n_active

        # Adaptive step size
        step_size = 0.3 / (1.0 + 0.05 * step)

        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-15:
            break
        a_new /= s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: LP alternating with multiple restarts and n values
    Phase 2: Mirror descent polishing with annealing
    Phase 3: Active set refinement and final LP pass
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Phase 1: LP + Mirror Descent on various n
    n_values = [250, 300, 350, 400, 500]

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        print(f"N={n}: {remaining:.0f}s remaining, best={best_score:.6f}")

        n_restarts = 5 if n <= 300 else (3 if n <= 400 else 2)

        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            # Initialize
            if r == 0 and best_seq is not None and len(best_seq) == n:
                a_init = best_seq.copy()
            elif r == 1 and best_seq is not None and len(best_seq) != n:
                # Resample parent to this n
                a_init = np.interp(np.linspace(0, 1, n),
                                   np.linspace(0, 1, len(best_seq)),
                                   best_seq)
                a_init = normalize(a_init)
            else:
                a_init = structured_init(n, rng)

            # LP alternating
            max_lp = int(min(120, max(20, remaining / 0.8)))
            if max_lp < 15:
                break
            a_lp, obj_lp = lp_alternate(a_init, max_iter=max_lp, t_blend=0.4,
                                        use_geometric=(r == 0))
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  LP n={n} r={r}: {obj_lp:.6f}")

            # Mirror descent polish
            if time.monotonic() < deadline - 20:
                remaining2 = deadline - time.monotonic()
                md_steps = int(min(4000, remaining2 / 0.015))
                if md_steps > 300:
                    a_md, obj_md = mirror_descent(a_lp, md_steps,
                                                  eps_start=3.0, eps_end=40.0,
                                                  eta_scale=0.8)
                    if obj_md < best_score:
                        best_score = obj_md
                        best_seq = a_md.copy()
                        print(f"  MD n={n} r={r}: {obj_md:.6f}")

            # Try from different init with mirror descent only
            if r > 0 and time.monotonic() < deadline - 15:
                remaining3 = deadline - time.monotonic()
                md_steps2 = int(min(3000, remaining3 / 0.015))
                if md_steps2 > 200:
                    a_md2, obj_md2 = mirror_descent(a_init, md_steps2,
                                                    eps_start=1.0, eps_end=15.0,
                                                    eta_scale=1.5)
                    if obj_md2 < best_score:
                        best_score = obj_md2
                        best_seq = a_md2.copy()
                        print(f"  MD-direct n={n} r={r}: {obj_md2:.6f}")

    # Phase 2: Intensive polishing of the best solution
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\nPhase 2: polishing n={n_best}, obj={best_score:.6f}, {remaining:.0f}s left")

        # Extended mirror descent with fine annealing
        md_steps = int(min(8000, remaining * 0.3 / 0.02))
        if md_steps > 1000:
            a_p1, obj_p1 = mirror_descent(best_seq, md_steps,
                                          eps_start=5.0, eps_end=80.0,
                                          eta_scale=0.5)
            if obj_p1 < best_score:
                best_score = obj_p1
                best_seq = a_p1.copy()
                print(f"  Phase2 MD: {obj_p1:.6f}")

        # Active set polish
        if time.monotonic() < deadline - 30:
            remaining2 = deadline - time.monotonic()
            ap_steps = int(min(3000, remaining2 * 0.3 / 0.03))
            if ap_steps > 500:
                a_p2, obj_p2 = active_set_polish(best_seq, n_steps=ap_steps)
                if obj_p2 < best_score:
                    best_score = obj_p2
                    best_seq = a_p2.copy()
                    print(f"  Phase2 AP: {obj_p2:.6f}")

        # LP pass on refined solution
        if time.monotonic() < deadline - 40 and n_best <= 400:
            remaining3 = deadline - time.monotonic()
            lp_iter = int(min(80, remaining3 * 0.3 / 0.6))
            if lp_iter > 20:
                a_p3, obj_p3 = lp_alternate(best_seq, max_iter=lp_iter,
                                            t_blend=0.3, use_geometric=True)
                if obj_p3 < best_score:
                    best_score = obj_p3
                    best_seq = a_p3.copy()
                    print(f"  Phase2 LP: {obj_p3:.6f}")

        # Final mirror descent with very sharp (high epsilon)
        if time.monotonic() < deadline - 15:
            remaining4 = deadline - time.monotonic()
            md_steps_final = int(min(5000, remaining4 / 0.02))
            if md_steps_final > 500:
                a_p4, obj_p4 = mirror_descent(best_seq, md_steps_final,
                                              eps_start=20.0, eps_end=200.0,
                                              eta_scale=0.3)
                if obj_p4 < best_score:
                    best_score = obj_p4
                    best_seq = a_p4.copy()
                    print(f"  Phase2 Final MD: {obj_p4:.6f}")

        # One more round: LP + MD
        if time.monotonic() < deadline - 40 and n_best <= 350:
            remaining5 = deadline - time.monotonic()
            lp_iter2 = int(min(50, remaining5 * 0.4 / 0.6))
            md_steps2 = int(min(3000, remaining5 * 0.5 / 0.02))
            if lp_iter2 > 15 and md_steps2 > 500:
                a_p5, obj_p5 = lp_alternate(best_seq, max_iter=lp_iter2, t_blend=0.3)
                a_p6, obj_p6 = mirror_descent(a_p5, md_steps2,
                                              eps_start=10.0, eps_end=100.0,
                                              eta_scale=0.4)
                if obj_p6 < best_score:
                    best_score = obj_p6
                    best_seq = a_p6.copy()
                    print(f"  Phase2 LP+MD: {obj_p6:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
