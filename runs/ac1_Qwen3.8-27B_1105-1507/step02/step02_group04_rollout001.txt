"""
Adaptive peak-shaving optimization with momentum, line search, and multi-scale exploration.
Uses exact subgradients of the autocorrelation peak with adaptive top-K smoothing,
Barzilai-Borwein step sizing, momentum, and resampling-based length adaptation
to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time


def compute_conv(a):
    """Compute autocorrelation conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    # Remove tiny negative values from FFT numerical error
    conv = np.maximum(conv, 0.0)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def subgradient_max_conv(a, conv, top_k=1, tau=None):
    """Compute subgradient of max(conv(a,a)).
    If tau is given, use softmax weighting over all conv values.
    Otherwise, use uniform weighting over top-k peaks.
    Returns gradient of length n."""
    n = len(a)
    
    if tau is not None and tau > 0:
        # Softmax weighting
        max_conv = float(np.max(conv))
        shifted = tau * (conv - max_conv)
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            idx = int(np.argmax(conv))
            weights = np.zeros(len(conv))
            weights[idx] = 1.0
        else:
            weights /= w_sum
    elif top_k >= len(conv):
        weights = np.ones(len(conv)) / len(conv)
    else:
        # Top-k with equal weighting
        idx = np.argpartition(-conv, top_k)[:top_k]
        weights = np.zeros(len(conv))
        weights[idx] = 1.0 / top_k
    
    # Compute gradient: grad[m] = 2 * sum_k weights[k] * a[k-m] (if 0<=k-m<n)
    # This is equivalent to: grad = 2 * correlate(weights, a, 'full')[n-1:2n-1]
    # But let's do it more efficiently with FFT
    size = 1
    while size < 2 * n:
        size <<= 1
    # We need: grad[m] = 2 * sum_k w[k] * a[k-m] for valid k-m
    # This is the cross-correlation of w and a, shifted
    # correlate(w, a, 'full')[n-1 + m - 0] ... let me just use the direct formula via FFT
    # Actually: grad[m] = 2 * (w * reverse(a))[m + n - 1] ... 
    # Let me use: grad = 2 * np.convolve(w, a[::-1], 'full')[n-1:n-1+n]
    # Wait, let me verify:
    # (w * a_rev)[j] = sum_i w[i] * a_rev[j-i] = sum_i w[i] * a[n-1-(j-i)] = sum_i w[i] * a[n-1-j+i]
    # We want grad[m] = 2 * sum_k w[k] * a[k-m] (if 0<=k-m<n)
    # Let j = n-1-m: n-1-j = m, so a[n-1-j+i] = a[m+i]... that's not right.
    # 
    # Let me just use: grad[m] = 2 * sum_k w[k] * a[k-m]
    # = 2 * sum_k w[k] * a[k-m] where k ranges over [0, 2n-2] and 0<=k-m<n
    # Let u = k-m: grad[m] = 2 * sum_u w[u+m] * a[u] where 0<=u<n and 0<=u+m<=2n-2
    # = 2 * (w_shifted * a)[0] where w_shifted[u] = w[u+m]
    # This is the cross-correlation: grad[m] = 2 * <w[m:m+2n-1], a> (with zero-padding)
    # 
    # Using FFT: Let W = FFT(w, size), A = FFT(a, size)
    # The linear convolution (w * a)[j] = sum_i w[i]*a[j-i]
    # We want sum_k w[k]*a[k-m] = sum_i w[i+m]*a[i] ... 
    # = sum_i w[i+m]*a[i] = cross-correlation of w and a at lag m
    # = (conj(FFT(w)) * FFT(a)) inverse FFT at lag m
    # Actually cross-correlation: R[m] = sum_i w[i+m] * a[i] = IFFT(conj(W) * A)[m]
    # (with appropriate circular shift for linear vs circular)
    
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    # Cross-correlation: sum_i w[i+m] * a[i] = IFFT(conj(fw) * fa)
    # But we need linear cross-correlation, so zero-padding helps
    xc = np.fft.irfft(np.conj(fw) * fa, size)
    # xc[m] = sum_i w[(i+m) % size] * a[i] for circular
    # For linear (with zero padding): xc[m] = sum_i w[i+m]*a[i] for m >= 0 (if size >= 2n-1+n)
    # Actually with size >= 2n-1, and w has length 2n-1, a has length n:
    # The linear cross-correlation at lag m is: sum_{i=0}^{n-1} w[i+m]*a[i] (if i+m < 2n-1)
    # = xc[m] for m in [0, 2n-2] (using the zero-padded FFT)
    
    grad = 2.0 * xc[:n]
    return grad


def optimize_local(a, max_time=60.0, momentum=0.9, use_bb=True, verbose=False, tag=""):
    """Local optimization using projected subgradient with momentum and line search.
    a is normalized to sum=1, non-negative.
    Returns (best_a, best_obj)."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    v = np.zeros(n)  # Momentum
    alpha = 0.1
    prev_grad = None
    prev_a = None
    prev_conv = None
    stall = 0
    patience = 300
    
    it = 0
    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))
        cur_obj = 2.0 * n * cur_max  # since sum=1
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                # Restart with perturbation
                rng = np.random.default_rng(it * 7919)
                noise = rng.normal(0, 0.05, n)
                a = a + noise
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s < 1e-10:
                    a = np.ones(n) / n
                    s = 1.0
                a = a / s
                v = np.zeros(n)
                stall = 0
                patience = max(100, patience // 2)
                continue
        
        # Adaptive top-K: use more peaks when convolution is flat
        sorted_conv = np.sort(conv)[::-1]
        max_val = sorted_conv[0]
        if max_val < 1e-15:
            break
        # Adaptive tau: smaller when peak is sharp, larger when flat
        ratio = sorted_conv[min(1, len(sorted_conv)-1)] / max_val if max_val > 0 else 1.0
        if ratio > 0.95:
            tau = 5.0
        elif ratio > 0.8:
            tau = 10.0
        else:
            tau = 2.0
        
        # Compute subgradient
        grad = subgradient_max_conv(a, conv, top_k=5, tau=tau)
        
        # Center gradient (tangent to sum=1 constraint)
        grad_centered = grad - np.mean(grad)
        
        # Barzilai-Borwein step size
        if use_bb and prev_grad is not None and prev_a is not None:
            s_vec = a - prev_a
            y_vec = grad_centered - prev_grad
            s_norm2 = np.dot(s_vec, s_vec)
            y_norm2 = np.dot(y_vec, y_vec)
            sy = np.dot(s_vec, y_vec)
            if s_norm2 > 1e-15 and abs(sy) > 1e-15:
                if sy > 0:
                    alpha_bb = s_norm2 / sy
                else:
                    alpha_bb = sy / y_norm2
                alpha_bb = np.clip(abs(alpha_bb), 1e-6, 1.0)
                # Damp the BB step
                alpha = 0.5 * alpha + 0.5 * alpha_bb
            prev_grad = grad_centered
            prev_a = a.copy()
        
        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad_centered
        
        # Line search (backtracking)
        best_alpha_ls = alpha
        best_obj_ls = cur_obj
        for ls_iter in range(10):
            a_try = a - best_alpha_ls * v
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                best_alpha_ls *= 0.5
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < cur_obj - 1e-15 * cur_obj:
                best_obj_ls = obj_try
                break
            best_alpha_ls *= 0.5
            if best_alpha_ls < 1e-8:
                break
        
        a = a - best_alpha_ls * v
        a = np.maximum(a, 0)
        s = np.sum(a)
        if s < 1e-10:
            a = np.ones(n) / n
            s = 1.0
        a = a / s
        
        # Slowly decay momentum for stability
        if it % 500 == 0:
            momentum *= 0.99
        
        if verbose and it % 200 == 0:
            cur_obj_check = compute_obj(a)
            print(f"    {tag} it={it}: obj={cur_obj_check:.6f}, best={best_obj:.6f}, stall={stall}")
    
    # Final check
    final_obj = compute_obj(a)
    if final_obj < best_obj:
        best_obj = final_obj
        best_a = a.copy()
    
    return best_a, best_obj


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        a_new = np.ones(new_n) / new_n
    else:
        a_new = a_new / s
    return a_new


def generate_initialization(n, seed, method=0):

def refine_with_lp(a, max_iter=20, verbose=False):
    """Try a simple LP-based refinement: find direction g that stays within peak bound."""
    from scipy.optimize import linprog
    
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    
    conv = compute_conv(a)
    M = float(np.max(conv))
    
    # Build constraint matrix: (a*g)[k] <= M for all k
    # (a*g)[k] = sum_i a[i] * g[k-i] (with a[i]=0 for invalid i)
    # So the constraint is: for each k, sum_i a[i] * g[k-i] <= M
    # The coefficient of g[j] in constraint k is a[k-j] (if valid)
    # So A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    # For n up to ~400, this matrix is (2n-1) x n
    if n > 400:
        return a, compute_obj(a)
    
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            # A[k, j] = a[k-j] for j in [j_min, j_max]
            A[k, j_min:j_max+1] = a[k-j_max:k-j_min+1][::-1]
    
    c = -np.ones(n)  # Maximize sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15})
        if res.status == 0:
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g > 1.0 + 1e-4:
                g = g / sum_g
                # Blend
                t = 0.3
                a_new = (1 - t) * a + t * g
                a_new = np.maximum(a_new, 0)
                s_new = np.sum(a_new)
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    obj_new = compute_obj(a_new)
                    obj_old = compute_obj(a)
                    if obj_new < obj_old and verbose:
                        print(f"    LP improved: {obj_old:.6f} -> {obj_new:.6f}")
                    return a_new, obj_new
    except Exception:
        pass
    
    return a, compute_obj(a)

            if n_best <= 350 and time.monotonic() < deadline - 20:
                a_lp, obj_lp = refine_with_lp(a_cur, max_iter=1, verbose=True)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    print(f"  LP polish: obj={best_score:.6f}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_obj(best_seq)
    
    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    
    # Convert to list of floats for evaluation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    
    # Validate
    try:
        validation = evaluate_sequence(result)
        if validation < np.inf and abs(validation - best_score) > 1e-4:
            print(f"  Validation: computed={best_score:.6f}, validated={validation:.6f}")
            best_score = validation
    except Exception as e:
        print(f"  Validation error: {e}")
    
    return result
