   Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
   1. Symmetric sequence parameterization (exploits reversal symmetry)
   2. Multi-start with diverse parametric initializations
   3. Beta-annealed L-BFGS-B on log-sum-exp smoothed objective
   4. Peak-shaving subgradient steps between optimization levels
   5. LP-based best-response for cross-validation
   6. Coordinate descent polish
   """
   import numpy as np
   from scipy.optimize import minimize, linprog
   import time

   def fast_conv(a):
       """Full linear self-convolution via FFT."""
       n = len(a)
       s = 1
       while s < 2 * n - 1:
           s <<= 1
       fa = np.fft.rfft(a, s)
       return np.fft.irfft(fa * fa, s)[:2 * n - 1]

   def make_symmetric(half):
       """Create full symmetric sequence from half."""
       nh = len(half)
       full = np.zeros(2 * nh)
       full[:nh] = half
       full[nh:] = half[::-1]
       return full

   def compute_score(a):
       """Compute J(a) directly."""
       a = np.clip(a, 0.0, 1000.0)
       s = a.sum()
       if s < 0.01:
           return float('inf')
       a = a / s
       n = len(a)
       b = fast_conv(a)
       return float(2.0 * n * b.max())

   def smooth_obj_sym(half_x, beta):
       """Log-sum-exp smoothed objective for symmetric sequences."""
       half = np.maximum(half_x, 0.0)
       s = half.sum()
       if s < 0.01:
           return 1e10
       full = make_symmetric(half)
       full_s = full.sum()
       if full_s < 0.01:
           return 1e10
       full = full / full_s
       n = len(full)
       b = fast_conv(full)
       b_max = b.max()
       shifted = beta * (b - b_max)
       shifted = np.clip(shifted, -80.0, 80.0)
       lse = b_max * beta + np.log(np.sum(np.exp(shifted)))
       return float(2.0 * n * lse)

       """Log-sum-exp smoothed objective for full (non-symmetric) sequences."""
       a = np.maximum(a_x, 0.0)
       s = a.sum()

       return float(2.0 * n * lse)

   def peak_shave_sym(half, n_steps, eta=0.01):
       """Subgradient peak-shaving for symmetric sequences."""
       half = half.copy()
       for _ in range(n_steps):
           full = make_symmetric(half)
           s = full.sum()

               break

           # Subgradient: g[i] = 2 * sum_k weights[k] * a[k-i]
           # This is a cross-correlation
           # g = 2 * correlate(weights, full)
           # correlate(weights, full)[i] = sum_k weights[k] * full[k-i]
           # Using FFT:
           n_full = len(full)
           size = 1
           while size < 2 * n_full - 1:

           W = np.fft.rfft(weights, size)
           F = np.fft.rfft(full, size)
           # Cross-correlation: ifft(conj(W) * F)
           g_full = np.fft.irfft(np.conj(W) * F, size)[:n_full]
           # Map back to half
           g_half = (g_full[:n_full//2] + g_full[n_full//2:][::-1]) / 2.0
           # Step
           half = half - eta * g_half
           half = np.maximum(half, 0.0)
           if half.sum() < 0.01:

           half = half / half.sum()
       return half

   def build_conv_matrix(a):
       """Build matrix A where (A @ x) = full convolve(a, x)."""
       n = len(a)
       m = 2 * n - 1
       A = np.zeros((m, n))
       for k in range(m):
           jl = max(0, k - n + 1)
           jh = min(n - 1, k)
           if jl <= jh:
               A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
       return A

       """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
