"""
Optimized step function search using LP refinement, subgradient descent,
and adaptive strategies to minimize 2n*max(conv(f,f))/sum(f)^2.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_score(f):
    """Compute the objective: 2n*max(conv(f,f))/sum(f)^2"""
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)

def normalize(f):
    """Normalize to sum=1"""
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s

def build_conv_matrix(f):
    """Build A such that (A @ g)[k] = (conv(f, g))[k] for all k"""
    n = len(f)
    A = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            A[k, i] = f[k - i]
    return A

def lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Return normalized g."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2*n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 10})
    if not res.success or res.x is None:
        return None
    g = np.clip(res.x, 0, None)
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s

def subgradient_step(f, eta, rng, n_near_max=5):
    """One projected subgradient step."""
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()
    
    # Find near-max positions
    threshold = M - 0.01 * M
    max_positions = np.where(conv >= threshold)[0]
    if len(max_positions) == 0:
        max_positions = [np.argmax(conv)]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in max_positions:
        for i in range(n):
            idx = k - i
            if 0 <= idx < n:
                grad[i] += 2.0 * f[idx]
    grad /= len(max_positions)
    
    # Projected step
    f_new = f - eta * grad
    # Project onto simplex {x >= 0, sum(x) = 1}
    f_new = project_simplex(f_new)
    return f_new

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return v

    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.clip(v - theta, 0, None)

LP-refinement + subgradient + coordinate descent optimizer for minimizing
2n*max(conv(f,f))/sum(f)^2 over non-negative sequences.
Uses adaptive step sizes, periodic tightening of LP constraints, and
multiple initialization strategies.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute 2n*max(conv(f,f))/sum(f)^2"""
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)


def normalize(f):
    """Normalize to sum=1."""
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def build_conv_matrix(f):
    """Build A such that (A @ g)[k] = (convolve(f,g))[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_end = min(2 * n - 1, i + n)
        A[i:k_end, i] = f[:k_end - i]
    return A


def solve_lp(f, M):
    """Solve: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 8})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    g = np.clip(res.x, 0, None)
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.clip(v - theta, 0, None)

                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                sp = compute_score(f_pert)
                if sp < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = sp
                    t = 0.2
                    stall = 0
                elif stall > 80:
                    break

        if iter_count % 200 == 0:
            print(f"    LP iter {iter_count}: score={best_score:.8f}, t={t:.4f}")

    return best_f, best_score


def subgradient_descent(f, time_budget, rng):
    """Projected subgradient descent on the max-convolution objective."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    best_f = f.copy()
    best_score = compute_score(f)
    eta = 0.005
    start = time.monotonic()
    stall = 0

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()

        # Find near-max positions (within 5% of max)
        threshold = M * 0.95
        max_pos = np.where(conv >= threshold)[0]
        if len(max_pos) == 0:
            max_pos = np.array([np.argmax(conv)])

        # Compute subgradient
        grad = np.zeros(n)
        for k in max_pos:
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                grad[i] += 2.0 * f[k - i]
        grad /= len(max_pos)

        # Normalized step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        f_new = f - eta * grad / gnorm
        f_new = project_simplex(f_new)

        sc = compute_score(f_new)
        if sc < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = sc
            f = f_new
            eta = min(0.05, eta * 1.03)
            stall = 0
        else:
            stall += 1
            eta *= 0.97
            if stall > 300:
                # Perturb
                f_pert = f + rng.normal(0, 0.01, n)
                f_pert = project_simplex(f_pert)
                sp = compute_score(f_pert)
                if sp < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = sp
                    eta = 0.005
                    stall = 0
                else:
                    break

    return best_f, best_score


def coordinate_descent(f, n_sweeps, rng):
    """Coordinate descent with geometric trial values."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    best_score = compute_score(f)
    factors = [0.5, 0.7, 0.85, 0.92, 1.0/0.92, 1.0/0.85, 1.0/0.7, 1.0/0.5, 0.1, 2.0, 5.0]

    for _ in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            orig = f[i]
            if orig < 1e-14:
                continue
            best_val = orig
            best_local = best_score
            for factor in factors:
                f[i] = orig * factor
                s = f.sum()
                if s < 1e-10:
                    f[i] = orig
                    continue
                sc = 2.0 * n * np.convolve(f, f).max() / (s * s)
                if sc < best_local:
                    best_local = sc
                    best_val = f[i]
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local

    f = normalize(f)
    return f, best_score


def diverse_init(n, rng):
    """Generate diverse non-negative initial sequences."""
    x = np.arange(n, dtype=np.float64) / n
    r = rng.random()
    if r < 0.25:
        # U-shaped
        f = 1.0 / (0.2 + np.abs(x - 0.5) * 3.0)
    elif r < 0.5:
        # Two bumps at 1/4 and 3/4
        f = np.exp(-((x - 0.25) ** 2) / 0.01) + np.exp(-((x - 0.75) ** 2) / 0.01) + 0.05
    elif r < 0.7:
        # Raised cosine
        f = 0.5 * (1 + np.cos(2 * np.pi * x)) + 0.1
    elif r < 0.85:
        # Geometric + mirror
        ratio = rng.uniform(0.95, 0.995)
        f = ratio ** np.arange(n)
        f = f + 0.5 * f[::-1]
    else:
        # Smooth random product
        f = np.ones(n)
        for _ in range(4):
            c = rng.uniform(0.1, 0.9) * n
            w = max(1.0, rng.uniform(0.05, 0.25) * n)
            f *= (1.0 + 0.3 * rng.standard_normal() * np.exp(-0.5 * (x * n - c) ** 2 / w ** 2))
        f = np.abs(f) + 0.05

    f = np.clip(f, 0.01, None)
    f = f * (1 + 0.03 * rng.standard_normal(n))
    f = np.clip(f, 0.005, None)
    return normalize(f)

    remaining = deadline - time.monotonic()
