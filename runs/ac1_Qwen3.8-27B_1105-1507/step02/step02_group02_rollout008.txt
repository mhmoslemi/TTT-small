    """Compute gradient of log-sum-exp smoothing of max(conv(a,a))."""

    b = fast_conv(a)

    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50, 50)

    # Compute grad[i] = 2 * sum_{j=0}^{n-1} a[j] * weights[i+j]
    # Using FFT cross-correlation
    w_rev = weights[::-1]  # length 2n-1
    fft_size = 1
    while fft_size < n + (2*n-1) - 1:
        fft_size <<= 1
    v_fft = np.fft.rfft(a, fft_size)
    wr_fft = np.fft.rfft(w_rev, fft_size)

    # c[i] = conv_full[2n-2-i] for i=0..n-1

    idx = 2*n-2 - np.arange(n)  # [2n-2, 2n-3, ..., n-1]
    grad = 2.0 * conv_full[idx]
    
    lse_val = float(b_max + np.log(weights.sum()) / beta)
    
    return grad, lse_val

def proj_simplex(v):
    """Project vector v onto the simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
