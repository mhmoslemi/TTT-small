import time

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def solve_convex(n, solver=None, verbose=True):
    """Solve the convex optimization problem for given n."""
    a = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    
    constraints = [cp.sum(a) == 1]
    
    # Build constraints conv[k] <= t for all k
    for k in range(2 * n - 1):
        Q = np.zeros((n, n))
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                Q[i, j] = 1
        constraints.append(cp.quad_form(a, Q) <= t)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    
    if solver is None:
        prob.solve()
    else:
        prob.solve(solver=solver)
    
    if a.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
        result = np.maximum(a.value, 0)
        s = np.sum(result)
        if s > 0:
            result = result / s
        score = compute_obj(result)
        if verbose:
            print(f"  n={n}, solver={solver}, status={prob.status}, score={score:.6f}, t={t.value:.6f}")
        return result, score
    else:
        if verbose:
            print(f"  n={n}, solver={solver}, FAILED: status={prob.status}")

"""
Convex optimization approach: minimize max_k conv(a,a) s.t. sum(a)=1, a>=0.
This is a convex QCP solved globally with cvxpy.
For larger n, use constraint generation to reduce problem size.
"""

import numpy as np
import cvxpy as cp
import scipy.sparse as sp
import time
import sys

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_Q_sparse(k, n):
    """Build sparse Q_k matrix where Q_k[i,j] = 1 if i+j=k."""
    rows = []
    cols = []
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            rows.append(i)
            cols.append(j)
    data = np.ones(len(rows))
    return sp.csc_matrix((data, (rows, cols)), shape=(n, n))

def solve_convex_full(n, solver=cp.MOSEK, verbose=True, timeout=300):
    """Solve the full convex problem for given n."""
    a = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    
    constraints = [cp.sum(a) == 1]
    
    for k in range(2 * n - 1):
        Q = build_Q_sparse(k, n)
        constraints.append(cp.quad_form(a, Q) <= t)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    t0 = time.time()
    try:
        prob.solve(solver=solver)
    except Exception as e:
        if verbose:
            print(f"  n={n}: solver error: {e}")
        return None, np.inf
    
    elapsed = time.time() - t0
    
    if a.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
        result = np.maximum(a.value, 0)
        s = np.sum(result)
        if s > 1e-10:
            result = result / s
        score = compute_obj(result)
        if verbose:
            print(f"  n={n}: status={prob.status}, t={t.value:.8f}, score={score:.6f}, time={elapsed:.1f}s")
        return result, score
    else:
        if verbose:
            print(f"  n={n}: FAILED, status={prob.status}, time={elapsed:.1f}s")
        return None, np.inf

def solve_convex_constrained(n, k_indices, solver=cp.MOSEK, verbose=True):
    """Solve convex problem with only constraints for specified k values."""
    a = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    
    constraints = [cp.sum(a) == 1]
    
    for k in k_indices:
        Q = build_Q_sparse(k, n)
        constraints.append(cp.quad_form(a, Q) <= t)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception as e:
        if verbose:
            print(f"  n={n}, |K|={len(k_indices)}: solver error: {e}")
        return None, np.inf
    
    if a.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
        result = np.maximum(a.value, 0)
        s = np.sum(result)
        if s > 1e-10:
            result = result / s
        score = compute_obj(result)
        if verbose:
            print(f"  n={n}, |K|={len(k_indices)}: status={prob.status}, t={t.value:.8f}, score={score:.6f}")
        return result, score
    else:
        if verbose:
            print(f"  n={n}, |K|={len(k_indices)}: FAILED, status={prob.status}")
        return None, np.inf

def constraint_generation(n, solver=cp.MOSEK, max_rounds=10, verbose=True):
    """
    Constraint generation: start with central constraints, add violated ones.
    """
    # Start with central lags
    center = (2 * n - 1) // 2
    window = max(5, n // 5)
    k_indices = list(range(max(0, center - window), min(2*n-1, center + window + 1)))
    
    best_a = None
    best_score = np.inf
    
    for round_idx in range(max_rounds):
        a, score = solve_convex_constrained(n, k_indices, solver=solver, verbose=verbose)
        
        if a is None:
            # Try with fewer constraints
            k_indices = [center]
            a, score = solve_convex_constrained(n, k_indices, solver=solver, verbose=verbose)
            if a is None:
                break
        else:
            if score < best_score:
                best_score = score
                best_a = a.copy()
        
        # Check which constraints are violated
        conv = np.convolve(a, a)
        max_conv = np.max(conv)
        violated = [k for k in range(2*n-1) if conv[k] > max_conv + 1e-10 and k not in k_indices]
        
        if not violated:
            # Also check if the current max_conv lags are in k_indices
            active_k = set(k for k in range(2*n-1) if conv[k] >= max_conv - 1e-10)
            missing = [k for k in active_k if k not in k_indices]
            if not missing:
                if verbose:
                    print(f"  Constraint generation converged in {round_idx+1} rounds")
                break
            k_indices.extend(missing)
        else:
            # Add the most violated constraints (limit to avoid explosion)
            violated_sorted = sorted(violated, key=lambda k: -conv[k])
            k_indices.extend(violated_sorted[:max(10, len(violated)//3)])
        
        k_indices = list(set(k_indices))
    
    return best_a, best_score

"""
Convex optimization approach to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key insight: after normalizing sum(a)=1, the problem min max_k (a*a)[k] s.t. a>=0, sum(a)=1
is a convex QCP (minimizing max of convex quadratic forms over a simplex).
We use cvxpy to find the global optimum, with constraint generation for larger n.
Subgradient and LP-based refinement are used for final polishing.
"""

import numpy as np
import cvxpy as cp
import scipy.sparse as sp
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_Q_sparse(k, n):
    """Build sparse symmetric Q_k where Q_k[i,j]=1 iff i+j=k, 0<=i,j<n."""
    rows = []
    cols = []
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            rows.append(i)
            cols.append(j)
    data = np.ones(len(rows))
    return sp.csc_matrix((data, (rows, cols)), shape=(n, n))


def solve_convex_full(n, solver=None, verbose=True):
    """Solve the full convex QCP for given n."""
    a = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    constraints = [cp.sum(a) == 1]
    for k in range(2 * n - 1):
        Q = build_Q_sparse(k, n)
        constraints.append(cp.quad_form(a, Q) <= t)
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    t0 = time.monotonic()
    try:
        if solver is not None:
            prob.solve(solver=solver)
        else:
            prob.solve()
    except Exception as e:
        if verbose:
            print(f"  n={n}: solver exception: {e}")
        return None, np.inf
    elapsed = time.monotonic() - t0
    if a.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
        result = np.maximum(a.value, 0.0)
        s = np.sum(result)
        if s > 1e-10:
            result = result / s
        score = compute_obj(result)
        if verbose:
            print(f"  n={n}: status={prob.status}, t={t.value:.10f}, score={score:.6f}, time={elapsed:.1f}s")
        return result, score
    else:
        if verbose:
            print(f"  n={n}: FAILED status={prob.status}, time={elapsed:.1f}s")
        return None, np.inf


def solve_convex_subset(n, k_indices, solver=None, verbose=False):
    """Solve convex QCP with only constraints for specified k values."""
    a = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    constraints = [cp.sum(a) == 1]
    for k in k_indices:
        Q = build_Q_sparse(k, n)
        constraints.append(cp.quad_form(a, Q) <= t)
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    try:
        if solver is not None:
            prob.solve(solver=solver)
        else:
            prob.solve()
    except Exception:
        return None, np.inf
    if a.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
        result = np.maximum(a.value, 0.0)
        s = np.sum(result)
        if s > 1e-10:
            result = result / s
        score = compute_obj(result)
        if verbose:
            print(f"  n={n}, |K|={len(k_indices)}: t={t.value:.10f}, score={score:.6f}")
        return result, score
    return None, np.inf


def constraint_generation(n, solver=None, max_rounds=15, verbose=True):
    """Constraint generation: start with central lags, add violated ones iteratively."""
    center = (2 * n - 1) // 2
    window = max(5, n // 4)
    k_indices = list(range(max(0, center - window), min(2 * n - 1, center + window + 1)))
    best_a = None
    best_score = np.inf
    for rnd in range(max_rounds):
        a, score = solve_convex_subset(n, k_indices, solver=solver, verbose=False)
        if a is None:
            k_indices = list(range(max(0, center - 5), min(2 * n - 1, center + 6)))
            a, score = solve_convex_subset(n, k_indices, solver=solver, verbose=False)
            if a is None:
                break
        if score < best_score:
            best_score = score
            best_a = a.copy()
        conv = np.convolve(a, a)
        max_conv = float(np.max(conv))
        tol = 1e-8
        violated = [k for k in range(2 * n - 1) if conv[k] > max_conv - tol + 1e-12 and k not in k_indices]
        if verbose:
            print(f"    CG round {rnd}: |K|={len(k_indices)}, max_conv={max_conv:.8f}, violated={len(violated)}")
        if not violated:
            if verbose:
                print(f"    CG converged in {rnd+1} rounds")
            break
        violated_sorted = sorted(violated, key=lambda k: -conv[k])
        k_indices.extend(violated_sorted[:min(20, max(5, len(violated) // 4))])
        k_indices = list(set(k_indices))
    return best_a, best_score


def vectorized_subgradient_refine(a, n_steps=5000, verbose=False):
    """Vectorized subgradient method using np.correlate for fast gradient computation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    for step in range(n_steps):
        conv = np.convolve(a, a)
        max_conv = float(np.max(conv))
        if max_conv < 1e-15:
            break
        rel_tol = 0.002
        active = conv >= max_conv * (1.0 - rel_tol)
        n_active = int(np.sum(active))
        if n_active == 0:
            active = np.zeros(len(conv), dtype=bool)
            active[int(np.argmax(conv))] = True
            n_active = 1
        active_full = np.zeros(2 * n - 1)
        active_full[active] = 1.0
        # cross_corr[i] = sum_j a[j] * active[j+i] = np.correlate(active_full, a)[i]
        cross_corr = np.correlate(active_full, a, mode='full')[:n]
        grad = (2.0 / n_active) * cross_corr
        # Adaptive step size (Polyak-like)
        step_size = 0.5 / (3.0 + 0.005 * step)
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        if verbose and step % 1000 == 0:
            print(f"    SG step {step}: best={best_obj:.6f}")
    return best_a, best_obj


def lp_refine(a, max_iter=80, verbose=False):
    """LP-based alternating refinement."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    for it in range(max_iter):
        M = float(np.max(np.convolve(a, a)))
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            from scipy.optimize import linprog
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if res.status == 0 and res.x is not None:
                g = np.maximum(res.x, 0)
                g_sum = np.sum(g)
                if g_sum > 1e-10:
                    g = g / g_sum
                    for t_blend in [0.3, 0.5, 0.7, 0.9]:
                        a_try = (1.0 - t_blend) * a + t_blend * g
                        a_try = np.maximum(a_try, 0)
                        s_try = np.sum(a_try)
                        if s_try > 1e-10:
                            a_try = a_try / s_try
                            obj_try = compute_obj(a_try)
                            if obj_try < best_obj:
                                best_obj = obj_try
                                best_a = a_try.copy()
                    a = best_a.copy()
        except Exception:
            pass
    return best_a, best_obj

            n_steps = int(min(8000, max(1000, remaining / 0.03)))
            a_sg, obj_sg = vectorized_subgradient_refine(best_seq, n_steps=n_steps, verbose=True)
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
                print(f"  After SG: {best_score:.6f}")

        # LP refinement
        remaining = deadline - time.monotonic()
        if remaining > 30 and n_best <= 500:
            max_iter = int(min(60, remaining / 2))
            a_lp, obj_lp = lp_refine(best_seq, max_iter=max_iter, verbose=False)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  After LP: {best_score:.6f}")

        # More subgradient after LP
        remaining = deadline - time.monotonic()
        if remaining > 20:
            n_steps = int(min(5000, max(500, remaining / 0.03)))
            a_sg2, obj_sg2 = vectorized_subgradient_refine(best_seq, n_steps=n_steps, verbose=False)
            if obj_sg2 < best_score:
                best_score = obj_sg2
                best_seq = a_sg2.copy()
                print(f"  After SG2: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)
        print(f"Fallback: n=100, score={best_score:.6f}")

    final_score = evaluate_sequence([float(x) for x in best_seq])
    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: n={len(best_seq)}, internal={best_score:.6f}, evaluate={final_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
