def compute_score(f):
    """Fast score computation."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')

def coord_descent_pass(f, max_passes=1):
    """Exact coordinate descent. Multiple passes."""
    n = len(f)
    a = f.copy().astype(np.float64)
    S = float(np.sum(a))
    
    for _pass in range(max_passes):
        conv = np.convolve(a, a)
        improved = False
        for i in range(n):
            # Determine D_k for k in [i, i+n-1]
            # D[k-i] = 2*a[k-i] for k-i != i, a[i] for k-i == i
            # i.e., D[j] = 2*a[j] for j != i, a[i] for j == i
            # where j = k - i
            
            D = 2.0 * a  # D[j] = 2*a[j] for all j
            D[i] = a[i]  # D[i] = a[i] (the k=2i case)
            
            # C[k] = conv[k] - a[i] * D[k-i] for k in [i, i+n-1]
            k_idx = np.arange(i, i + n)
            C = conv[k_idx] - a[i] * D
            
            # C_min: max of conv at positions where D_k = 0
            # D_k = 0 for k < i and k >= i+n

            if i > 0:
