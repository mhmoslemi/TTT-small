Multi-stage optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines exact subgradient descent with multiplicative updates,
LP-based alternating, and adaptive active set management.
"""

import numpy as np
from scipy.optimize import linprog
import time

# === Core computation ===

def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

# === Subgradient computation ===

def compute_subgradient(a, conv, active_tol_frac=0.0):
    """
    Compute (averaged) subgradient of max_k (a*a)_k at point a.
    active_tol_frac: fraction of max to include in active set.
    0.0 = only the exact maximum.
    Returns array of length n.
    """
    n = len(a)
    max_val = float(np.max(conv))
    if max_val <= 0:
        return np.zeros(n)
    
    threshold = max_val * (1.0 - active_tol_frac)
    active = np.where(conv >= threshold)[0]
    
    # grad_i = (1/|active|) * sum_{k in active} 2*a[k-i] (if valid)
    grad = np.zeros(n)
    for k in active:
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        if i_min <= i_max:
            # a[k-i] for i = i_min,...,i_max
            # k-i ranges from k-i_max down to k-i_min
            seg = a[k-i_max:k-i_min+1][::-1]  # length = i_max - i_min + 1
            grad[i_min:i_max+1] += 2.0 * seg
    
    grad /= len(active)
    return grad

    Uses adaptive active set and backtracking line search.

    a = a / s  # normalize to sum=1

def lp_step(a, M, time_limit=10):
    """
    Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns g (normalized to sum=1) and the optimal sum.
    """
    n = len(a)
    # Build constraint matrix A where (A @ g)[k] = (a*g)[k]
    # (a*g)[k] = sum_j a[j] * g[k-j] for valid j, k-j
    # This is a Toeplitz-like matrix
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                A[k, j] = a[j]  # wait, this is wrong
    
    # Let me redo: (a*g)[k] = sum_j a[j] * g[k-j]
    # For fixed k, as j varies: g[k-j] is at position (k-j)
    # So A[k, j] = a[j] if k-j is in [0, n-1], i.e., j in [max(0,k-n+1), min(k,n-1)]
    # And the variable g[k-j] is at index (k-j)
    # So: (a*g)[k] = sum_{j} a[j] * g[k-j] = sum_{m} a[k-m] * g[m] (where m = k-j)
    # So A[k, m] = a[k-m] if 0 <= k-m < n
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        m_min = max(0, k - n + 1)
        m_max = min(k, n - 1)
        for m in range(m_min, m_max + 1):
            A[k, m] = a[k - m]
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g > 1e-10:
                return g / sum_g, sum_g
    except Exception as e:
        pass
    return None, None

def lp_alternating(a, max_iters=50, t=0.3, verbose=False, time_limit=None, start_time=None):
    """
    LP-based alternating optimization.
    """
    n = len(a)
    a = a.copy().astype(np.float64)

def generate_initializations(n, rng):
    """Generate diverse initializations of length n."""
    inits = []
    
    # 1. Uniform
    inits.append(np.ones(n) / n)
    
    # 2. Raised cosine (Hann window)
    j = np.arange(n)
    hann = 0.5 * (1 + np.cos(2*np.pi*j/n))
    hann = np.maximum(hann, 0) / np.sum(hann)
    inits.append(hann)
    
    # 3. Edge-concentrated
    a = np.exp(-2.0 * np.minimum(j, n-1-j) / (n/4.0))
    inits.append(a / np.sum(a))
    
    # 4. Sinusoidal modulation
    freq = rng.uniform(0.8, 2.5)
    phase = rng.uniform(0, 2*np.pi)
    a = np.maximum(0, 1 + np.cos(2*np.pi*freq*j/n + phase))
    inits.append(a / np.sum(a))
    
    # 5. Two-peak
    center1 = n * rng.uniform(0.1, 0.3)
    center2 = n * rng.uniform(0.7, 0.9)
    width = n * rng.uniform(0.05, 0.15)
    a = np.exp(-0.5*((j-center1)/width)**2) + np.exp(-0.5*((j-center2)/width)**2)
    inits.append(a / np.sum(a))
    
    # 6. Random with smoothing
    a = rng.random(n)
    # Apply Gaussian smoothing
    kernel_size = max(3, n // 10)
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size//2)**2 / (kernel_size//4)**2)
    kernel /= kernel.sum()
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    inits.append(a / np.sum(a))
    
    # 7. "Anti-peak" - try to make convolution flat
    # Start with uniform and add a structured perturbation
    a = np.ones(n) / n
    perturbation = np.sin(2*np.pi*rng.uniform(1, 5) * j / n) * 0.3
    a = np.maximum(a + perturbation, 0)
    inits.append(a / np.sum(a))
    
    return inits

import cvxpy as cp

def qp_step(a, t_target, time_limit=30):
    """
    Solve QP: min ||a_new - a||^2 s.t. (a_new * a)_k <= t_target, sum(a_new)=1, a_new >= 0
    """
    n = len(a)
    a_new = cp.Variable(n)
    
    # Build constraint matrix (same as LP)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        m_min = max(0, k - n + 1)
        m_max = min(k, n - 1)
        for m in range(m_min, m_max + 1):
            A[k, m] = a[k - m]
    
    obj = cp.Minimize(cp.sum_squares(a_new - a))
    constraints = [
        A @ a_new <= t_target,
        cp.sum(a_new) == 1,
        a_new >= 0
    ]
    
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, time_limit=time_limit)  # or cp.CLARABEL
        if a_new.value is not None:
            a_new_val = np.asarray(a_new.value).flatten()
            a_new_val = np.maximum(a_new_val, 0)
            a_new_val = a_new_val / np.sum(a_new_val)
            return a_new_val
    except Exception:
        pass
    return None

import numpy as np
from scipy.optimize import linprog
import cvxpy as cp
import time


def compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_conv_matrix(a):
    """Build matrix A where (A @ x)_k = (a*x)_k for all k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        m_min = max(0, k - n + 1)
        m_max = min(k, n - 1)
        if m_min <= m_max:
            A[k, m_min:m_max + 1] = a[k - m_max:k - m_min + 1][::-1]
    return A


def compute_subgradient(a, conv, active_frac=0.0):
    """
    Compute averaged subgradient of max_k (a*a)_k at point a.
    active_frac: fraction of max to include in active set (0 = exact max only).
    """
    n = len(a)
    max_val = float(np.max(conv))
    if max_val <= 0:
        return np.zeros(n)

    threshold = max_val * (1.0 - active_frac)
    active = np.where(conv >= threshold)[0]

    grad = np.zeros(n)
    for k in active:
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        if i_min <= i_max:
            seg = a[k - i_max:k - i_min + 1][::-1]
            grad[i_min:i_max + 1] += 2.0 * seg

    grad /= len(active)
    return grad


def mult_subgrad_descent(a, max_iters=50000, eta0=0.1, verbose=False,
                         time_limit=None, start_time=None):
    """
    Multiplicative subgradient descent to minimize max(conv(a,a)) with sum(a)=1.
    Uses adaptive active set and backtracking line search.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    conv = compute_conv(a)
    best_max = float(np.max(conv))
    best_obj = compute_obj(best_a)

    eta = eta0
    stall = 0
    patience = 800
    prev_max = best_max

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.monotonic() - start_time > time_limit:
                break

        max_val = float(np.max(conv))

        # Adaptive active set: broad early, narrow later
        if it < 200:
            active_frac = 0.15
        elif it < 1000:
            active_frac = 0.08
        elif it < 5000:
            active_frac = 0.03
        else:
            active_frac = 0.0

        grad = compute_subgradient(a, conv, active_frac)
        grad_centered = grad - np.mean(grad)

        # Backtracking line search
        eta_try = eta
        improved = False
        for bt in range(12):
            a_try = a * np.exp(-eta_try * grad_centered)
            a_try = np.maximum(a_try, 1e-30)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                eta_try *= 0.5
                continue
            a_try = a_try / s_try
            conv_try = compute_conv(a_try)
            max_try = float(np.max(conv_try))

            if max_try < max_val - 1e-14:
                a = a_try
                conv = conv_try
                max_val = max_try
                if bt > 0:
                    eta = eta_try / 0.65
                else:
                    eta = min(eta * 1.05, 1.0)
                improved = True
                break
            eta_try *= 0.65

        if not improved:
            stall += 1
            eta *= 0.85
            if stall > patience or eta < 1e-12:
                break
        else:
            stall = 0
            # Occasionally reset eta if we've been improving steadily
            if it % 1000 == 0 and it > 0:
                eta = max(eta * 0.9, eta0 * 0.5)

        if max_val < best_max - 1e-14:
            best_max = max_val
            best_a = a.copy()
            best_obj = compute_obj(best_a)

        if verbose and it % 10000 == 0:
            print(f"    sg it={it}: max={max_val:.10f}, eta={eta:.6f}, stall={stall}")

    return best_a, best_obj


def qp_projection(a, t_target, time_limit=30):
    """
    Solve QP: min ||a_new - a||^2 s.t. (a_new*a)_k <= t_target, sum(a_new)=1, a_new>=0.
    Returns the solution or None.
    """
    n = len(a)
    if n > 600:
        return None

    A = build_conv_matrix(a)
    a_new = cp.Variable(n)

    obj = cp.Minimize(cp.sum_squares(a_new - a))
    constraints = [
        A @ a_new <= t_target,
        cp.sum(a_new) == 1,
        a_new >= 0
    ]

    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, time_limit=time_limit, max_iters=5000)
        if a_new.value is not None:
            a_new_val = np.asarray(a_new.value, dtype=np.float64).flatten()
            a_new_val = np.maximum(a_new_val, 0)
            s = np.sum(a_new_val)
            if s > 1e-10:
                a_new_val /= s
                return a_new_val
    except Exception:
        pass
    return None


def qp_iterative(a, max_iters=200, reduction=0.02, verbose=False,
                 time_limit=None, start_time=None):
    """
    Iteratively tighten the convolution bound using QP projection.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    conv = compute_conv(a)
    best_max = float(np.max(conv))
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    failure_count = 0
    current_reduction = reduction

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.monotonic() - start_time > time_limit:
                break

        max_val = float(np.max(conv))
        t_target = max_val * (1.0 - current_reduction)

        a_new = qp_projection(a, t_target, time_limit=15)
        if a_new is None:
            current_reduction *= 0.5
            failure_count += 1
            if failure_count > 10:
                break
            continue

        failure_count = 0
        conv_new = compute_conv(a_new)
        max_new = float(np.max(conv_new))

        if max_new < max_val - 1e-14:
            a = a_new
            conv = conv_new
            # If we got close to target, try slightly larger reduction
            if max_new <= t_target * 1.01:
                current_reduction = min(current_reduction * 1.1, 0.08)
            else:
                current_reduction = max(current_reduction * 0.9, 0.005)

            if max_new < best_max:
                best_max = max_new
                best_a = a.copy()
                best_obj = compute_obj(best_a)
        else:
            # The QP solution didn't help; reduce the step
            current_reduction *= 0.5
            if current_reduction < 0.001:
                break

        if verbose and it % 50 == 0:
            print(f"    QP it={it}: max={max_val:.10f}, red={current_reduction:.4f}")

    return best_a, best_obj


def lp_step(a, M, time_limit=10):
    """
    Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns (g normalized to sum=1, optimal sum_g) or (None, None).
    """
    n = len(a)
    if n > 500:
        return None, None

    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.asarray(res.x, dtype=np.float64)
            sum_g = float(np.sum(g))
            if sum_g > 1e-10:
                return g / sum_g, sum_g
    except Exception:
        pass
    return None, None


def lp_alternating(a, max_iters=80, t=0.3, verbose=False,
                   time_limit=None, start_time=None):
    """
    LP-based alternating optimization with line search over blend parameter.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    stall = 0

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.monotonic() - start_time > time_limit:
                break

        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = lp_step(a, M, time_limit=min(15, max(3, time_limit or 10)))
        if g is None:
            break

        if sum_g <= 1.0 + 1e-4:
            stall += 1
            if stall > 20:
                break
            continue
        stall = 0

        # Line search over blend parameter
        best_t = 0
        best_obj_t = best_obj
        for t_try in [t, t * 0.5, t * 0.25, t * 1.5, t * 2.0]:
            t_try = min(t_try, 0.8)
            a_try = (1.0 - t_try) * a + t_try * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t_try

        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * g
            a_new = np.maximum(a_new, 0)
            a_new = a_new / np.sum(a_new)

            if best_obj_t < best_obj:
                best_obj = best_obj_t
                best_a = a_new.copy()
                a = a_new
                if verbose:
                    print(f"    LP it={it}: obj={best_obj:.6f}, sum_g={sum_g:.4f}, t={best_t:.3f}")

    return best_a, best_obj


def generate_initializations(n, rng):
    """Generate diverse initializations of length n, each summing to 1."""
    inits = []
    j = np.arange(n, dtype=np.float64)

    # 1. Uniform
    inits.append(np.ones(n) / n)

    # 2. Raised cosine (Hann)
    hann = 0.5 * (1 + np.cos(2 * np.pi * j / n))
    hann = np.maximum(hann, 0)
    if np.sum(hann) > 0:
        inits.append(hann / np.sum(hann))

    # 3. Edge-concentrated (two peaks at edges)
    dist_edge = np.minimum(j, n - 1 - j)
    a = np.exp(-3.0 * dist_edge / (n / 5.0))
    inits.append(a / np.sum(a))

    # 4. Sinusoidal modulation
    freq = rng.uniform(0.8, 3.0)
    phase = rng.uniform(0, 2 * np.pi)
    a = np.maximum(0, 1.0 + 0.8 * np.cos(2 * np.pi * freq * j / n + phase))
    if np.sum(a) > 0:
        inits.append(a / np.sum(a))

    # 5. Two interior peaks
    c1 = n * rng.uniform(0.15, 0.35)
    c2 = n * rng.uniform(0.65, 0.85)
    w = n * rng.uniform(0.05, 0.12)
    a = np.exp(-0.5 * ((j - c1) / w) ** 2) + np.exp(-0.5 * ((j - c2) / w) ** 2)
    inits.append(a / np.sum(a))

    # 6. Smoothed random
    a = rng.random(n)
    ksize = max(3, n // 8)
    kernel = np.exp(-0.5 * (np.arange(ksize) - ksize // 2) ** 2 / max(1, (ksize // 4) ** 2))
    kernel /= kernel.sum()
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    if np.sum(a) > 0:
        inits.append(a / np.sum(a))

    # 7. Anti-symmetric perturbation of uniform
    a = np.ones(n) / n
    pert = 0.25 * np.sin(2 * np.pi * rng.uniform(1.5, 4.0) * j / n)
    a = np.maximum(a + pert, 0)
    inits.append(a / np.sum(a))

    # 8. "Bark-like" - alternating high/low with smooth envelope
    env = 0.5 + 0.5 * np.cos(2 * np.pi * j / n)
    alt = 0.5 + 0.5 * np.sign(np.sin(2 * np.pi * rng.uniform(2, 6) * j / n))
    a = env * (0.5 + 0.5 * alt)
    a = np.maximum(a, 0)
    if np.sum(a) > 0:
        inits.append(a / np.sum(a))

    return inits


def try_parent_sequence():
    """Try to load and evaluate the parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception as e:
        print(f"Could not load parent: {e}")
    return None, np.inf


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    resampled = np.interp(x_new, x_old, a)
    resampled = np.maximum(resampled, 0)
    s = np.sum(resampled)
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return resampled / s


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function: multi-stage optimization combining subgradient descent,
    QP projection, and LP alternating. Returns best sequence found within budget.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + float(budget_s) - 12.0

    best_seq = None
    best_score = np.inf

    def record(a, obj):
        nonlocal best_seq, best_score
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  ** BEST: n={len(a)}, obj={obj:.6f} (t={time.monotonic()-t0:.0f}s)")

    # Load parent
    parent, parent_obj = try_parent_sequence()
    if parent is not None:
        record(parent, parent_obj)

    # Determine n values
    n_candidates = [200, 250, 300, 350, 400, 500]
    if parent is not None:
        np_ = len(parent)
        n_candidates = sorted(set(n_candidates + [np_, max(100, np_ - 50), min(600, np_ + 50)]))

    print(f"\nExploring n values: {n_candidates}")
    print(f"Deadline in {deadline - t0:.0f}s\n")

    # === Phase 1: Multi-start exploration ===
    print("=== Phase 1: Multi-start exploration ===")
    for n in n_candidates:
        if time.monotonic() > deadline - 90:
            break

        remaining = deadline - time.monotonic()
        alloc = min(45.0, max(8.0, remaining * 0.12))

        print(f"\n  n={n} (alloc {alloc:.0f}s)")
        inits = generate_initializations(n, rng)

        # Include resampled parent if close in length
        if parent is not None and abs(len(parent) - n) <= 30:
            inits.append(resample_sequence(parent, n))

        for idx, a_init in enumerate(inits):
            if time.monotonic() > deadline - 60:
                break
            per_init = alloc / len(inits)
            if per_init < 2:
                break

            a_opt, obj_opt = mult_subgrad_descent(
                a_init, max_iters=3000, eta0=0.08,
                time_limit=per_init, start_time=time.monotonic())
            record(a_opt, obj_opt)

    # === Phase 2: Deep subgradient on best ===
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        print(f"\n=== Phase 2: Deep subgradient (n={n_best}) ===")

        a_cur = best_seq.copy()
        s = np.sum(a_cur)
        a_cur = a_cur / s

        remaining = deadline - time.monotonic() - 30
        alloc = remaining * 0.35

        # Multiple rounds with different eta
        for round_idx in range(3):
            if time.monotonic() > deadline - 30:
                break
            t_round = alloc / (3 - round_idx) if round_idx < 2 else remaining - 20
            a_opt, obj_opt = mult_subgrad_descent(
                a_cur, max_iters=100000, eta0=0.05 / (round_idx + 1),
                time_limit=t_round, start_time=time.monotonic())
            record(a_opt, obj_opt)
            if obj_opt < best_score:
                a_cur = a_opt.copy()

    # === Phase 3: QP projection ===
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        print(f"\n=== Phase 3: QP projection (n={n_best}) ===")

        if n_best <= 500:
            a_cur = best_seq.copy()
            s = np.sum(a_cur)
            a_cur = a_cur / s

            remaining = deadline - time.monotonic() - 20
            alloc_qp = remaining * 0.25

            a_opt, obj_opt = qp_iterative(
                a_cur, max_iters=100, reduction=0.015,
                time_limit=alloc_qp, start_time=time.monotonic())
            record(a_opt, obj_opt)

            # Follow up with subgradient
            if time.monotonic() < deadline - 10 and obj_opt < best_score:
                rem = deadline - time.monotonic() - 5
                a_sg, obj_sg = mult_subgrad_descent(
                    a_opt, max_iters=50000, eta0=0.02,
                    time_limit=rem, start_time=time.monotonic())
                record(a_sg, obj_sg)

    # === Phase 4: LP alternating ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = len(best_seq)
        print(f"\n=== Phase 4: LP alternating (n={n_best}) ===")

        if n_best <= 400:
            a_cur = best_seq.copy()
            s = np.sum(a_cur)
            a_cur = a_cur / s

            remaining = deadline - time.monotonic() - 10
            alloc_lp = remaining * 0.7

            a_opt, obj_opt = lp_alternating(
                a_cur, max_iters=150, t=0.25,
                time_limit=alloc_lp, start_time=time.monotonic())
            record(a_opt, obj_opt)

            if obj_opt < best_score and time.monotonic() < deadline - 5:
                rem = deadline - time.monotonic() - 3
                a_sg, obj_sg = mult_subgrad_descent(
                    a_opt, max_iters=30000, eta0=0.01,
                    time_limit=rem, start_time=time.monotonic())
                record(a_sg, obj_sg)

    # === Phase 5: Final polish ===
    if best_seq is not None and time.monotonic() < deadline - 3:
        print(f"\n=== Phase 5: Final polish ===")
        a_cur = best_seq.copy()
        s = np.sum(a_cur)
        a_cur = a_cur / s

        rem = deadline - time.monotonic() - 2
        if rem > 2:
            a_opt, obj_opt = mult_subgrad_descent(
                a_cur, max_iters=200000, eta0=0.003,
                time_limit=rem, start_time=time.monotonic())
            record(a_opt, obj_opt)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - t0
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf and validation != best_score:
        print(f"  Validation: computed={best_score:.6f}, validated={validation:.6f}")

    return result
