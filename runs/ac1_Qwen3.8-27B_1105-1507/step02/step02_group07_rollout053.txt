import numpy as np

from scipy.optimize import minimize
from scipy.sparse import csr_matrix

def compute_score(f):
    """Compute the objective value (scale-invariant)."""
    n = len(f)
    C = np.convolve(f, f)
    max_b = np.max(C)
    S = np.sum(f)
    if S < 0.01:
        return np.inf
    return 2.0 * n * max_b / (S * S)

def softmax_objective_and_grad(f, beta):
    """Compute softmax-smoothed objective and gradient."""
    n = len(f)
    C = np.convolve(f, f)
    max_C = np.max(C)

    soft_val = max_C + np.log(total) / beta
    return soft_val, grad, p

    val, grad, _ = softmax_objective_and_grad(f, beta)

    while time.monotonic() < deadline:
        # Interpolate beta
        frac = min(1.0, n_outer / (total_outer - 1))
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Run L-BFGS-B for a few iterations
        result = minimize(
            fun=objective_wrapper,
            x0=best_f,
            args=(beta,),
            jac=gradient_wrapper,
            method='L-BFGS-B',
            bounds=[(0, 1000)] * n,
            options={'maxiter': max_inner_iters, 'maxfun': max_inner_iters * 3}
        )
        
        n_outer += 1
        if result.success or result.status == 1:
            f_new = np.clip(result.x, 0, 1000)
            S_new = np.sum(f_new)
            if S_new > 0.01:
                f_new = f_new / S_new
                sc = compute_score(f_new)
                if np.isfinite(sc) and sc < best_score:
                    best_f = f_new
                    best_score = sc
                    print(f"  {verbose_prefix}outer {n_outer}: beta={beta:.1f}, score={best_score:.6f}")
        
        if time.monotonic() > deadline - 10:
            break
        
        # Check if we should increase beta further
        if beta >= beta_end * 0.9:
            # Do a few more iterations at high beta
            beta = beta_end
    
    return best_f, best_score

def lp_direction_optimize(f_init, deadline, max_iters=100, verbose_prefix=""):
    """LP-based direction finding (from previous approach, simplified)."""
    f = f_init.copy().astype(np.float64)
    S = np.sum(f)
    f = f / S
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    step_size = 0.1
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        C = np.convolve(f, f)
        M = np.max(C)
        if M <= 0:
            break
        
        # Build constraint matrix: A @ g <= M, where (A@g)[k] = conv(f,g)[k]
        # A is (2n-1) x n, A[k,j] = f[k-j] if 0<=k-j<n else 0
        # For efficiency, only use constraints near the peak
        # Actually, use all constraints for correctness
        
        # For large n, building the full matrix is expensive.
        # Use a subset: only the top 20 constraints.
        # Find the lags with highest values
        top_k = np.argsort(C)[-min(20, 2*n-1):]
        
        # Build sparse matrix for selected constraints
        rows, cols, vals = [], [], []
        for idx, k in enumerate(top_k):
            for j in range(max(0, k - n + 1), min(k, n - 1) + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(f[k - j])
        
        if len(rows) == 0:
            break
        
        A = csr_matrix((vals, (rows, cols)), shape=(len(top_k), n))
        c = -np.ones(n)
        b_ub = np.full(len(top_k), M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True})
            if not res.success:
                no_improve += 1
                if no_improve > 10:
                    break
                continue
        except Exception:
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        g = res.x
        sum_g = np.sum(g)
        if sum_g < 1e-15:
            no_improve += 1
            continue
        
        g = g / sum_g  # normalize to sum=1
        
        # Line search
        best_t = 0
        best_s = best_score
        for t in [step_size, step_size*0.5, step_size*1.5, step_size*0.25]:
            new_f = (1-t)*f + t*g
            new_f = np.clip(new_f, 0, 1000)
            ns = np.sum(new_f)
            if ns < 0.01:
                continue
            new_f = new_f / ns
            sc = compute_score(new_f)
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = t
        
        if best_t > 0:
            f = (1-best_t)*f + best_t*g
            f = np.clip(f, 0, 1000)
            f = f / np.sum(f)
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            step_size = min(step_size * 1.1, 0.3)
        else:
            no_improve += 1
            step_size *= 0.7
            if step_size < 0.01 or no_improve > 20:
                break
    
    return best_f, best_score

def peak_reduction(f, n_steps=50, deadline=None):
    """Reduction of the peak autocorrelation value via targeted adjustments."""
    f = f.copy().astype(np.float64)
    S = np.sum(f)
    f = f / S
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        C = np.convolve(f, f)
        k_star = np.argmax(C)
        M = C[k_star]
        
        # Compute gradient of C[k_star] with respect to f
        # dC[k_star]/df[j] = 2*f[k_star-j] for valid k_star-j
        grad = np.zeros(n)
        j_min = max(0, k_star - n + 1)
        j_max = min(k_star, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] = 2.0 * f[k_star - j]
        
        # Project gradient to maintain sum constraint
        grad = grad - np.mean(grad)
        
        # Normalized gradient direction
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
        
        # Try several step sizes
        best_t = 0
        best_s = best_score
        for t in [0.01, 0.005, 0.02, 0.001, 0.05]:
            new_f = f - t * grad / grad_norm
            new_f = np.clip(new_f, 0, 1000)
            ns = np.sum(new_f)
            if ns < 0.01:
                continue
            new_f = new_f / ns
            sc = compute_score(new_f)
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = t
        
        if best_t > 0:
            f = f - best_t * grad / grad_norm
            f = np.clip(f, 0, 1000)
            f = f / np.sum(f)
            best_f = f.copy()
            best_score = best_s
        else:
            break
    
    return best_f, best_score

def coordinate_descent(f, deadline, max_rounds=100):
    """Simple coordinate descent: try increasing/decreasing each element."""
    f = f.copy().astype(np.float64)
    S = np.sum(f)
    f = f / S
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Random order for diversity
        order = np.random.permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            for delta in [0.01, 0.005, -0.01, -0.005, 0.02, -0.02]:
                new_f = f.copy()
                new_f[j] = max(0, new_f[j] + delta / S)
                ns = np.sum(new_f)
                if ns < 0.01:
                    continue
                new_f = new_f / ns
                sc = compute_score(new_f)
                if np.isfinite(sc) and sc < best_score - 1e-10:
                    f = new_f
                    best_f = f.copy()
                    best_score = sc
                    improved = True
                    break
        
        if not improved:
            break
    
    return best_f, best_score

def tukey_window(n, r=0.3):
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2*i/(2*m) - 1)))
            w[n-1-i] = w[i]
    return w

def hann_window(n):
    n = max(n, 2)
    return 0.5 * (1 + np.cos(2*np.pi*np.arange(n)/(n-1)))

def raised_cosine_p(n, p):
    """Generalized raised cosine: (1 - |2x-1|^p)"""
    x = np.linspace(0, 1, n)
    return np.maximum(0, 1 - np.abs(2*x - 1)**p)

def asymmetric_window(n, shift=0.0, taper=0.3):
    """Window with shifted peak."""
    x = np.linspace(0, 1, n)
    peak = 0.5 + shift
    width = 0.5 - taper/2
    # Gaussian-like
    w = np.exp(-((x - peak)/width)**2)
    # Apply edge tapers
    m = int(taper * n / 2)
    if m > 0:
        for i in range(m):
            w[i] *= 0.5 * (1 + np.cos(np.pi*(2*i/(2*m) - 1)))
            w[n-1-i] *= 0.5 * (1 + np.cos(np.pi*(2*i/(2*m) - 1)))
    return w

def multi_lobe_window(n, n_lobes=3):
    """Window with multiple lobes."""
    x = np.linspace(0, 1, n)
    w = np.ones(n)
    for i in range(n_lobes):
        center = (i + 0.5) / n_lobes
        width = 0.3 / n_lobes
        w += np.exp(-((x - center)/width)**2)
    # Apply edge tapers
    m = int(0.2 * n / 2)
    if m > 0:
        for i in range(m):
            w[i] *= 0.5 * (1 + np.cos(np.pi*(2*i/(2*m) - 1)))
            w[n-1-i] *= 0.5 * (1 + np.cos(np.pi*(2*i/(2*m) - 1)))
    return np.maximum(0, w)
