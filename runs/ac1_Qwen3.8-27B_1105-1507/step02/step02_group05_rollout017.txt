"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP mirror descent with adaptive step sizes
2. Smooth max (log-sum-exp) gradient refinement via L-BFGS-B
3. Spectral initialization and diverse restarts
4. Multi-length search with parent seeding
5. Final polynomial-time LP polish
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            # (a*g)_k = sum_{j=jl}^{jh} a_{k-j} * g_j
            for j in range(jl, jh + 1):
                A[k, j] = a[k - j]
    return A.tocsr()

def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except:
        pass
    return None

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def smooth_max_grad(a, T):
    """Compute gradient of (1/T)*log(sum_k exp(T*(a*a)_k))."""
    n = len(a)
    b = fast_conv(a)
    # Numerical stability
    b_max = b.max()
    exp_b = np.exp(T * (b - b_max))
    softmax = exp_b / exp_b.sum()
    # Gradient: sum_k softmax_k * 2*a_{k-m} for each m
    # grad[m] = 2 * sum_k softmax_k * a_{k-m} (when 0 <= k-m < n)
    # This is 2 * (softmax * a_reversed) convolved appropriately
    # (a*a)_k = sum_j a_j * a_{k-j}
    # d(a*a)_k / da_m = 2 * a_{k-m} if 0 <= k-m < n
    # grad_m = sum_k softmax_k * 2 * a_{k-m} * 1_{0<=k-m<n}
    #        = 2 * (softmax * a)_m where * is "cross-correlation"
    # Actually: grad_m = 2 * sum_k softmax_k * a_{k-m} for k-m in [0,n-1]
    # Let l = k-m, then k = l+m, range: l in [0,n-1], k in [0, 2n-2]
    # So m in [0, n-1], l in [max(0,-m), min(n-1, 2n-2-m)] = [0, min(n-1, 2n-2-m)]
    # For m < n-1: l in [0, n-1], so grad_m = 2 * sum_{l=0}^{n-1} softmax_{l+m} * a_l
    # For m >= n-1: l in [0, 2n-2-m], so grad_m = 2 * sum_{l=0}^{2n-2-m} softmax_{l+m} * a_l
    
    # This is a cross-correlation of softmax with a
    # grad = 2 * correlate(softmax, a) where correlate(x,y)[m] = sum_l x[l+m]*y[l]
    
    # Using FFT for efficiency:
    # correlate(softmax, a) = ifft(fft(softmax) * conj(fft(a)))... 
    # Actually, let me just do it directly for n up to ~1024
    
    grad = np.zeros(n)
    for m in range(n):
        l_max = min(n - 1, 2 * n - 2 - m)
        if l_max >= 0:
            grad[m] = 2.0 * np.dot(softmax[m:m + l_max + 1], a[:l_max + 1])
    return grad

def smooth_max_value(a, T):
    """Compute (1/T)*log(sum_k exp(T*(a*a)_k))."""
    b = fast_conv(a)
    b_max = b.max()
    return b_max + np.log(np.exp(T * (b - b_max)).sum()) / T

def run_smooth_max_opt(a, deadline, tracker, T_start=10.0, T_end=100.0, n_steps=20):
    """Optimize using smooth max with annealing."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a_sum = a.sum()
    if a_sum < 1e-10:
        return a
    a = a / a_sum
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        T = T_start * (T_end / T_start) ** (step / max(1, n_steps - 1))
        
        def obj(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-8:
                return 1e10
            ax = ax / s
            return smooth_max_value(ax, T)
        
        def obj_grad(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-8:
                return 1e10, np.zeros_like(x)
            ax = ax / s
            g = smooth_max_grad(ax, T)
            return smooth_max_value(ax, T), g
        
        res = minimize(obj_grad, a, method='L-BFGS-B', 
                       jac=True,
                       options={'maxiter': 100, 'max_fun': 2000, 'ftol': 1e-14})
        if res.fun < 1e10:
            a = np.maximum(res.x, 0.0)
            a_sum = a.sum()
            if a_sum > 1e-10:
                a = a / a_sum
            tracker.update(a)
    
    return a

def run_lp_iterations(a, max_iters, deadline, tracker, step_decay=0.995):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    base_t = 0.3
    n = len(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        t = base_t * (step_decay ** it)
        t = max(t, 0.01)
        
        # Try a few step sizes
        best_a = None
        best_J = current_J
        
        for t_try in [t, t * 0.5, t * 2.0]:
            a_trial = (1.0 - t_try) * a + t_try * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            # Try very small step
            a_trial = (1.0 - 0.01) * a + 0.01 * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-15:
                a = a_trial
                current_J = J_trial
            else:
                # Stuck, try reducing M slightly
                M_new = M * 0.99
                g = lp_best_response(a, M_new)
                if g is None or g.sum() < 1e-10:
                    break
                g = g / g.sum()
                a_trial = (1.0 - 0.05) * a + 0.05 * g

                if J_trial < current_J - 1e-15:
                    a = a_trial
                    current_J = J_trial
                else:
                    break
        else:
            a = best_a
            current_J = best_J
        
        if it % 10 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

def spectral_init(n, rng):
    """Generate initialization from flat power spectrum with random phases."""
    # Create a sequence with (approximately) flat power spectrum
    # a_k = (1/n) * sum_m exp(i * (2*pi*k*m/n + phi_m))
    # For real non-negative: use cosine with random phases
    m = np.arange(n)
    k = np.arange(n)
    phases = rng.uniform(0, 2 * np.pi, n)
    # a_k = sum_m cos(2*pi*k*m/n + phi_m) / n
    # This might have negative values, so take abs or rectify
    a = np.zeros(n)
    for mm in range(n):
        a += np.cos(2 * np.pi * k * mm / n + phases[mm])
    a = a / n
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a.sum()
    return a

def generate_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random uniform
    a = rng.uniform(0.1, 1.0, n)
    inits.append(a / a.sum())
    
    # Random with structure
    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * t)
        a = np.maximum(a, 0.01)
        inits.append(a / a.sum())
    
    # Gaussian bump
    a = np.exp(-5.0 * (t - 0.5) ** 2) + 0.05
    inits.append(a / a.sum())
    
    # Spectral
    inits.append(spectral_init(n, rng))
    
    # Power law
    a = (t + 0.05) ** rng.uniform(-2, 2)
    a = np.maximum(a, 0.01)
    inits.append(a / a.sum())
    
    # Bimodal
    a = np.exp(-20 * (t - 0.3) ** 2) + np.exp(-20 * (t - 0.7) ** 2) + 0.05
    inits.append(a / a.sum())
    
    return inits
