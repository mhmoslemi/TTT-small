def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def compute_gradient_at_lag(f, k):

"""
Optimization for minimizing 2n*max(conv(f,f))/sum(f)^2 using
LP-based direction finding, active set polishing, and subgradient methods.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def build_conv_matrix_sparse(f):
    n = len(f)
    rows = np.zeros(2*n-1, dtype=np.int32)  # will be filled
    # Actually, let's build it properly
    n_nonzeros = 0
    for k in range(2*n-1):
        j_min = max(0, k-(n-1))
        j_max = min(k, n-1)
        n_nonzeros += (j_max - j_min + 1)
    
    rows = np.zeros(n_nonzeros, dtype=np.int32)
    cols = np.zeros(n_nonzeros, dtype=np.int32)
    vals = np.zeros(n_nonzeros, dtype=np.float64)
    
    idx = 0
    for k in range(2*n-1):
        j_min = max(0, k-(n-1))
        j_max = min(k, n-1)
        for j in range(j_min, j_max+1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k-j]
            idx += 1
    
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def solve_lp(f, M):
    """Solve: max sum(g) s.t. g>=0, (f*g)[k] <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x, result
    except Exception:
        pass
    return None, None

def evaluate(f):
    """Quick evaluation without full security checks."""
    f = np.clip(f, 0, 1000)
    s = np.sum(f)
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * len(f) * np.max(conv) / (s * s)

def tukey_window(n, r=0.3):
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2*i/(2*m) - 1)))
            w[n-1-i] = w[i]
    return w

def gauss_window(n, alpha=2.0):
    idx = np.arange(n, dtype=np.float64)
    center = (n-1)/2.0
    return np.exp(-alpha * ((idx-center)/(n/2.0))**2)

def lp_optimize(f, deadline, max_iters=500, verbose_prefix=""):
    """LP-based optimization with adaptive step size."""
    f = f.copy().astype(np.float64)
    S = np.sum(f)
    f = f / S
    n = len(f)
    
    best_f = f.copy()
    best_score = evaluate(f)
    step_size = 0.1
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break
        
        g, lp_result = solve_lp(f, M)
        it += 1
        
        if g is None or np.sum(g) < 1e-15:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        g = g / np.sum(g) * S  # normalize to same sum
        
        # Backtracking line search
        best_t = None
        best_s = best_score
        for t in [step_size, step_size*0.5, step_size*1.5, step_size*0.25, step_size*2]:
            new_f = np.clip((1-t)*f + t*g, 0, 1000)
            ns = np.sum(new_f)
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = evaluate(new_f)
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = t
        
        if best_t is not None and best_s < best_score:
            f = np.clip((1-best_t)*f + best_t*g, 0, 1000)
            f = f / np.sum(f)

        else:

            step_size *= 0.7

    return best_f, best_score

def subgradient_polish(f, deadline, max_iters=5000, verbose_prefix=""):
    """Polish using exact subgradient of max(conv(f,f))."""
    f = f.copy().astype(np.float64)
    S = np.sum(f)
    f = f / S
    n = len(f)
    
    best_f = f.copy()
    best_score = evaluate(f)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = 0.01
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        conv_f = np.convolve(f, f)
        max_val = float(np.max(conv_f))
        
        # Find all lags at or near the max (within 1% tolerance)
        active_mask = conv_f >= 0.99 * max_val
        active_lags = np.where(active_mask)[0]
        
        # Compute subgradient: average of gradients at active lags
        grad = np.zeros(n)
        for k in active_lags:
            # grad_k[i] = 2 * f[k-i] if 0 <= k-i < n
            low = max(0, k - n + 1)
            high = min(n - 1, k)
            i_vals = np.arange(low, high + 1)
            f_idx = k - i_vals
            grad[i_vals] += 2.0 * f[f_idx]
        grad /= len(active_lags)
        
        # Also add a small regularization towards uniform to prevent concentration
        grad += 0.001 * (f - 1.0/n)
        
        it += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        new_f = np.clip(f - lr*update, 0, 1000)
        ns = np.sum(new_f)
        if ns < 1e-10:
            continue
        new_f = new_f / ns
        
        sc = evaluate(new_f)
        if np.isfinite(sc) and sc < best_score:
            f = new_f
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            if it % 1000 == 0:
                print(f"  {verbose_prefix}SG it={it}: score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 500:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-7:
                    break
    
    return best_f, best_score

def coordinate_polish(f, deadline, max_rounds=100, verbose_prefix=""):
    """Coordinate-wise polish: optimize each coordinate while keeping others fixed."""
    f = f.copy().astype(np.float64)
    S = np.sum(f)
    f = f / S
    n = len(f)
    
    best_f = f.copy()
    best_score = evaluate(f)
    improved_total = 0
    
    for round_idx in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        improved = 0
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            # Try a few values for f[i]
            orig = f[i]
            best_val = orig
            best_sc = best_score
            
            # Try multiplicative changes
            for mult in [0.8, 0.9, 0.95, 1.05, 1.1, 1.2]:
                trial = orig * mult
                if trial < 0 or trial > 1000:
                    continue
                f[i] = trial
                sc = evaluate(f)
                if sc < best_sc:
                    best_sc = sc
                    best_val = trial
            
            if best_val != orig:
                f[i] = best_val
                improved += 1
            
            # Renormalize occasionally
            S = np.sum(f)
            if S > 0:
                f = f / S
                if best_sc < best_score:
                    best_score = best_sc
                    best_f = f.copy()
        
        if improved == 0:
            break
        improved_total += improved
    
    return best_f, best_score

"""
Optimization for minimizing 2n*max(conv(f,f))/sum(f)^2 using
aggressive LP-based direction finding, active-set polishing,
subgradient methods, and perturbation restarts.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    n_nonzeros = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        n_nonzeros += (j_max - j_min + 1)
    rows = np.zeros(n_nonzeros, dtype=np.int32)
    cols = np.zeros(n_nonzeros, dtype=np.int32)
    vals = np.zeros(n_nonzeros, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g>=0, (f*g)[k] <= M for all k. Returns (g, dual_info)."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x, result
    except Exception:
        pass
    return None, None


def fast_eval(f):
    """Fast internal evaluation (assumes f is valid non-negative array)."""
    f = np.clip(f, 0, 1000)
    s = np.sum(f)
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * np.max(conv) / (s * s))


def tukey_window(n, r=0.3):
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def gauss_window(n, alpha=2.0):
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    return np.exp(-alpha * ((idx - center) / (n / 2.0)) ** 2)


def smooth_random(n, rng, smoothness=3.0):

def subgradient_optimize(f, deadline, max_iters=10000, lr_start=0.01, verbose_prefix=""):
    """Subgradient method with Adam for the exact max function."""
    f = f.copy().astype(np.float64)
    S = np.sum(f)
    if S < 1e-10:
        return f, fast_eval(f)
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = fast_eval(f)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = lr_start
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        conv_f = np.convolve(f, f)
        max_val = float(np.max(conv_f))

        # Find active lags (within 2% of max for stability)
        tol = 0.02 * max_val
        active_mask = conv_f >= (max_val - tol)
        active_lags = np.where(active_mask)[0]
        if len(active_lags) == 0:
            active_lags = np.array([np.argmax(conv_f)])

        # Compute subgradient: average of gradients at active lags
        grad = np.zeros(n)
        for k in active_lags:
            low = max(0, k - n + 1)
            high = min(n - 1, k)
            i_vals = np.arange(low, high + 1)
            f_idx = k - i_vals
            grad[i_vals] += 2.0 * f[f_idx]
        grad /= len(active_lags)

        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        update = m_hat / (np.sqrt(v_hat) + eps)

        new_f = np.clip(f - lr * update, 0, 1000)
        ns = np.sum(new_f)
        if ns < 1e-10:
            continue
        new_f = new_f / ns

        sc = fast_eval(new_f)
        if np.isfinite(sc) and sc < best_score:
            f = new_f
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            if it % 2000 == 0:
                print(f"  {verbose_prefix}SG it={it}: score={best_score:.7f} lr={lr:.6f}")
        else:
            no_improve += 1
            if no_improve > 800:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-8:
                    break

    return best_f, best_score


def coordinate_polish(f, deadline, max_rounds=30, verbose_prefix=""):
    """Coordinate-wise polish: try local changes to each coordinate."""
    f = f.copy().astype(np.float64)
    S = np.sum(f)
    if S < 1e-10:
        return f, fast_eval(f)
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = fast_eval(f)

    for round_idx in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = 0
        for i in range(n):
            if time.monotonic() > deadline:
                break
            orig = f[i]
            if orig < 1e-10:
                # Try making it small positive
                for trial in [1e-4, 1e-3, 5e-3]:
                    f[i] = trial
                    sc = fast_eval(f)
                    if sc < best_score:
                        best_score = sc
                        best_f = f.copy()
                        improved += 1
                    f[i] = orig
            else:
                for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                    trial = orig * mult
                    if trial > 1000:
                        continue
                    f[i] = trial
                    sc = fast_eval(f)
                    if sc < best_score:
                        best_score = sc
                        best_f = f.copy()
                        improved += 1
                        # Keep the improved value for next trial
                    else:
                        f[i] = orig
                # Reset if no single trial stuck
                if f[i] != orig and fast_eval(f) >= best_score:
                    f[i] = orig
        if improved == 0:
            break
        # Renormalize
        S = np.sum(f)
        if S > 0:
            f = f / S

    return best_f, best_score


def perturb_sequence(f, rng, strategy=0):
    """Perturb a sequence using different strategies."""
    n = len(f)
    if strategy == 0:
        # Multiplicative Gaussian noise
        noise = 1.0 + 0.04 * rng.standard_normal(n)
        new_f = f * np.clip(noise, 0.3, 2.0)
    elif strategy == 1:
        # Additive noise
        new_f = f + 0.005 * np.abs(rng.standard_normal(n))
    elif strategy == 2:
        # Smooth (blur)
        kernel = np.exp(-0.5 * (np.arange(-5, 6) / 1.5) ** 2)
        new_f = np.convolve(f, kernel, mode='same')
    elif strategy == 3:
        # Random coordinate zeroing and redistribution
        new_f = f.copy()
        n_zero = max(1, n // 50)
        zeros = rng.choice(n, size=n_zero, replace=False)
        mass = new_f[zeros].sum()
        new_f[zeros] = 0
        new_f += mass / n
    else:
        # High-frequency noise
        new_f = f + 0.01 * rng.standard_normal(n)
        new_f = np.convolve(new_f, np.ones(3)/3, mode='same')
    new_f = np.clip(new_f, 0, 1000)
    S = np.sum(new_f)
    if S < 0.01:
        return f.copy()
    return new_f / S

        len_dl = min(p3_deadline, time.monotonic() + per_len)
        print(f"  --- n={n_try} ---")

        # Interpolate best sequence to this length
        if best_sequence is not None:
            from numpy import interp
            f_interp = interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, len(best_sequence)),
                              best_sequence)
            f_interp = np.clip(f_interp, 0, None)
            S = np.sum(f_interp)
            if S > 0.01:
                f_interp = f_interp / S
                sc = fast_eval(f_interp)
                if sc < best_score:
                    best_score, best_sequence = sc, f_interp.copy()
                    print(f"  n={n_try} interp: {sc:.7f}")
                # Optimize
                f2, s2 = lp_optimize(f_interp, min(len_dl, time.monotonic() + per_len * 0.6),
                                      max_iters=80, step_init=0.1)
                f2, s2 = subgradient_optimize(f2, min(len_dl, time.monotonic() + per_len * 0.2),
                                               max_iters=2000, lr_start=0.005)
                if s2 < best_score:
                    best_score, best_sequence = s2, f2.copy()
                    print(f"  n={n_try} optimized: {s2:.7f} (NEW BEST)")
                else:
                    print(f"  n={n_try} optimized: {s2:.7f}")

        # Also try a fresh window
        f_tukey = tukey_window(n_try, 0.35)
        f_tukey = f_tukey / np.sum(f_tukey)
        sc = fast_eval(f_tukey)
        if sc < best_score:
            best_score, best_sequence = sc, f_tukey.copy()
            print(f"  n={n_try} tukey: {sc:.7f} (NEW BEST)")

    # ===== PHASE 4: Perturbation restarts on best (10% time) =====
    p4_start = max(p3_deadline, time.monotonic())
    p4_deadline = min(deadline, p4_start + total_budget * 0.10)
    if best_sequence is not None and p4_deadline > p4_start + 20:
        print(f"\n{'='*60}")
        print(f"PHASE 4: Perturbation restarts (budget={p4_deadline-p4_start:.0f}s)")
        print(f"{'='*60}")

        n_restarts = min(6, max(3, int((p4_deadline - p4_start) / 35)))
        per_restart = (p4_deadline - p4_start) / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > p4_deadline - 10:
                break
            f_pert = perturb_sequence(best_sequence, rng, strategy=r % 5)
            r_dl = min(p4_deadline, time.monotonic() + per_restart * 0.85)

            f2, s2 = lp_optimize(f_pert, min(r_dl, time.monotonic() + per_restart * 0.45),
                                  max_iters=60, step_init=0.08)
            f2, s2 = subgradient_optimize(f2, r_dl, max_iters=3000, lr_start=0.004)
            f2, s2 = coordinate_polish(f2, r_dl, max_rounds=10)

            if s2 < best_score:
                best_score, best_sequence = s2, f2.copy()
                print(f"  Restart {r} (strat={r%5}): {s2:.7f} (NEW BEST)")
            else:
                print(f"  Restart {r} (strat={r%5}): {s2:.7f}")

    # ===== PHASE 5: Final intensive polish (remaining time) =====
    if best_sequence is not None and time.monotonic() < deadline - 10:
        p5_deadline = deadline - 5
        p5_time = p5_deadline - time.monotonic()
        print(f"\n{'='*60}")
        print(f"PHASE 5: Final polish (budget={p5_time:.0f}s, current={best_score:.7f})")
        print(f"{'='*60}")

        if p5_time > 10:
            # Subgradient with very small learning rate
            f = best_sequence.copy()
            f, s = subgradient_optimize(f, min(p5_deadline, time.monotonic() + p5_time * 0.5),
                                        max_iters=10000, lr_start=0.003, verbose_prefix="F-SG")
            if s < best_score:
                best_score, best_sequence = s, f.copy()

        if time.monotonic() < p5_deadline - 5:
            # LP with tiny steps
            f, s = lp_optimize(best_sequence, p5_deadline - 5, max_iters=50,
                                step_init=0.03, verbose_prefix="F-LP")
            if s < best_score:
                best_score, best_sequence = s, f.copy()

        # Final coordinate polish
        if time.monotonic() < deadline - 3:
            f, s = coordinate_polish(best_sequence, deadline - 3, max_rounds=15)
            if s < best_score:
                best_score, best_sequence = s, f.copy()

        print(f"  Phase 5 final: {best_score:.7f}")

    # Safety
    if best_sequence is None:
        best_sequence = tukey_window(512, 0.3)
        best_score = evaluate_sequence(best_sequence.tolist())

    best_sequence = np.clip(best_sequence, 0, 1000)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_sequence)}, score={best_score:.7f}, time={elapsed:.1f}s")
    print(f"{'='*60}")

    # Verify with actual evaluate_sequence
    final_score = evaluate_sequence([float(x) for x in best_sequence])
    print(f"Verified via evaluate_sequence: {final_score:.7f}")

    return [float(x) for x in best_sequence]
