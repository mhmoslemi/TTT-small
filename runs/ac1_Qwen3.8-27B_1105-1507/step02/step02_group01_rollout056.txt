Targets objective below 1.5027.

import numpy as np
from scipy.optimize import minimize as sp_minimize
import time


def compute_obj(a):
    """Compute the objective function value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def soft_max_and_weights(a, T):
    """Compute soft-max of convolution and the weight distribution."""
    conv = np.convolve(a, a)
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    sm_val = max_c + T * np.log(sum_exp)
    return sm_val, weights


def lbgfgs_optimize(a_init, T_values, max_time, maxiter_per_T=80):
    """
    Optimize sequence using L-BFGS-B with T-annealing on soft-max surrogate.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    bounds = [(0.0, 1.0)] * n

    for T in T_values:
        if time.monotonic() - start > max_time:
            break

        def obj_func(x):
            x = np.maximum(x, 0.0)
            s = np.sum(x)
            if s < 1e-10:
                return 1e10
            x_norm = x / s
            sm_val, _ = soft_max_and_weights(x_norm, T)
            return 2.0 * n * sm_val

            obj_func, a, method='L-BFGS-B',

                     'maxls': 50}
        )

            a_new = np.clip(result.x, 0.0, None)

def coordinate_descent(a_init, max_time, tol=1e-10, max_rounds=100):
    """
    Coordinate descent: for each position, optimize its value to reduce
    the maximum of the convolution.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for rnd in range(max_rounds):
        if time.monotonic() - start > max_time:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - start > max_time:
                break
            # Try to find optimal a[i] given others fixed
            # The convolution as a function of a[i] is quadratic:
            # conv[k] = (sum_{j!=i, j+k-i valid} a[j]*a[k-j]) + a[i]*a[k-i] + a[k-i]*a[i]
            # More precisely, let b = a with a[i] set to 0, and c[k] = a[k-i] contribution
            # conv[k] = base[k] + a[i] * coeff[k]

            # Simpler approach: just try a few candidate values

            if old_val < 1e-12:

                j = peak_idx - i

                    # To reduce peak: set a[i] = old_val * target_peak / current_peak_contribution
                    # This is approximate

            for cv in candidates:

                a = np.clip(a, 0.0, None)

def block_perturb_refine(a_init, rng, max_time):
    """Block perturbation refinement with adaptive step size."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    sigma = 0.05
    it = 0
    no_improve = 0

    while time.monotonic() - start < max_time and it < 50000:
        width = max(1, int(n * rng.uniform(0.005, 0.08)))
        pos = int(rng.integers(0, n - width + 1))
        candidate = a.copy()
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            it += 1
            continue
        candidate = candidate / s
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate.copy()
            best_obj = obj
            a = candidate
            no_improve = 0
            sigma = min(sigma * 1.1, 0.3)
        else:
            no_improve += 1
            if no_improve > 200:
                sigma = max(0.01, sigma * 0.7)
                no_improve = 0
        it += 1

def init_bump(n, rng, center_frac=None, width_frac=None):
