    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    sg = (conv_full[:n] + corr) / len(active)
    return sg, b, active


def run_subgradient(a, max_iters, deadline, tracker, eta_init=0.05):
    """Run subgradient descent with adaptive step size."""
    a = np.maximum(a, 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    best_a = a.copy()
    best_J = current_J
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        eta = eta_init / np.sqrt(it + 1)
        sg, b, active = subgradient_step_fft(a)
        a_new = a - eta * sg
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        J_new = compute_J(a_new)
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
        a = a_new
        if it % 100 == 0:
            tracker.update(a)

    tracker.update(best_a)
    return best_a


def run_lp_iterations(a, max_iters, deadline, tracker):
    """Run LP mirror descent with line-search step size."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    prev_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Line search: try multiple step sizes
        base_t = 1.0 / np.sqrt(it + 2)
        if it < max_iters // 4:
            base_t = max(base_t, 0.3)
        elif it < max_iters // 2:
            base_t = max(base_t, 0.15)

        candidates_t = [base_t, min(1.0, base_t * 2.0), max(0.01, base_t * 0.5),
                        max(0.01, base_t * 0.25), 1.0]
        candidates_t = sorted(set(round(t, 6) for t in candidates_t if 0 < t <= 1.0),
                              reverse=True)

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            # Try momentum step
            t_small = 0.05
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-15:
                a = a_trial
                current_J = J_trial
            else:
                break
        else:
            prev_a = a.copy()
            a = best_a
            current_J = best_J

        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def active_set_balancing(a, deadline, tracker, n_rounds=5):
    """Balance the top active lags by solving targeted LPs."""
    a = np.maximum(a, 0.001).copy()
    a /= a.sum()

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        max_b = b.max()
        n = len(a)

        # Identify top lags (within 1% of max)
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) < 2:
            active = np.argsort(b)[::-1][:3]

        # Try to reduce the max by solving LP with tighter bound on active set
        M_tight = max_b * (1.0 - 0.01 * (rnd + 1))
        M_tight = max(M_tight, max_b * 0.9)

        g = lp_best_response(a, M_tight)
        if g is None:
            # Try original M
            g = lp_best_response(a, max_b)
            if g is None:
                break
            M_tight = max_b

        g = np.maximum(g, 0.0)
        if g.sum() < 1e-10:
            break
        g /= g.sum()

        # Mix with current
        for t in [0.3, 0.5, 0.2, 0.7, 0.1]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < compute_J(a):
                a = a_trial
                break

        tracker.update(a)

    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []

    if parent is not None and parent.size > 0:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n, dtype=np.float64))
    inits.append(rng.uniform(0.3, 1.0, n).astype(np.float64))

    t_arr = np.linspace(0, 1, n, endpoint=False)
    for freq in [1, 2, 3, 5, 7, 10]:
        inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr), 0.01))

    inits.append(np.maximum(1.0 + 0.4 * np.sin(2.0 * np.pi * t_arr ** 2), 0.01))
    inits.append(np.maximum(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.1, 0.01))
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.2)
    inits.append(np.maximum(1.0 + 0.5 * np.cos(4.0 * np.pi * t_arr) * np.cos(2.0 * np.pi * t_arr), 0.01))

    # Two-lobe structure
    inits.append(np.maximum(np.cos(np.pi * t_arr) ** 2 + 0.1, 0.01))
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * t_arr) + 0.2 * np.cos(4 * np.pi * t_arr), 0.01))

    # Sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 50)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same') + 0.05, 0.01))

    # Power law decay from center
    center = n / 2.0
    inits.append(np.maximum(np.exp(-1.5 * np.abs(t_arr - 0.5) * 3) + 0.05, 0.01))

    # Alternating with decay
    inits.append(np.maximum(0.5 + 0.3 * np.cos(2 * np.pi * t_arr * n / 2) + 0.1, 0.01))

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * NEW BEST: J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer:
    Phase 1: LP mirror descent across multiple lengths with line search
    Phase 2: Subgradient polish for longer sequences
    Phase 3: Active set balancing
    Phase 4: L-BFGS-B + final LP polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        parent = None

    # === Phase 1: LP Mirror Descent ===
    print(f"\n{'='*60}", flush=True)
    print("PHASE 1: LP Mirror Descent", flush=True)
    print(f"{'='*60}", flush=True)

    # Start with parent length and neighbors
    if parent is not None:
        parent_n = parent.size
        # Find nearest powers of 2
        lengths = sorted(set([64, 128, 256, 512, 1024, 2048]))
        # Put parent length region first
        lengths.sort(key=lambda x: abs(x - parent_n))
    else:
        lengths = [128, 256, 512, 1024]

    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n  --- LP n={n} ---", flush=True)

        inits = generate_initializations(n, rng, parent)
        # Number of iterations: more for shorter sequences
        iters_per_init = max(30, min(200, int(50000 // n)))
        n_inits = min(len(inits), max(4, int(60000 // n)))

        best_local = None
        best_local_J = float('inf')

        for idx in range(n_inits):
            if time.monotonic() > deadline:
                break
            a0 = inits[idx % len(inits)].copy()
            a0 = np.maximum(a0, 0.001)
            s0 = a0.sum()
            if s0 < 0.01:
                continue
            a0 /= s0

            a = run_lp_iterations(a0, iters_per_init, deadline, tracker)
            J_local = compute_J(a)
            if J_local < best_local_J:
                best_local_J = J_local
                best_local = a.copy()

            elapsed_init = time.monotonic() - t0
            if idx % 3 == 0:
                print(f"    init {idx}: J={J_local:.8f} elapsed={elapsed_init:.1f}s", flush=True)

        if best_local is not None:
            # Run a few more LP iterations on the best local
            if time.monotonic() < deadline:
                extra_iters = max(20, iters_per_init // 2)
                best_local = run_lp_iterations(best_local, extra_iters, deadline, tracker)

        tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Subgradient polish on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 2: Subgradient Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        # Run subgradient on the best sequence
        a_sub = run_subgradient(tracker.best.copy(), max_iters=2000, deadline=deadline,
                                tracker=tracker, eta_init=0.03)

        # Also try subgradient from slightly perturbed versions
        for restart in range(3):
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best.copy() + rng.normal(0, 0.02, n)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            a_sub = run_subgradient(a_pert, max_iters=500, deadline=deadline,
                                    tracker=tracker, eta_init=0.05)

        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Active Set Balancing ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 3: Active Set Balancing", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        a_bal = active_set_balancing(tracker.best.copy(), deadline, tracker, n_rounds=8)

        # Try balancing from slightly different starting points
        for restart in range(2):
            if time.monotonic() > deadline:
                break
            a_pert = tracker.best.copy() + rng.normal(0, 0.01, tracker.best.size)
            a_pert = np.maximum(a_pert, 0.001)
            a_pert /= a_pert.sum()
            a_bal = active_set_balancing(a_pert, deadline, tracker, n_rounds=4)

        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: L-BFGS-B + Final LP ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 4: L-BFGS-B + Final LP", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = tracker.best.size

        # L-BFGS-B on the max convolution
        def lbb_objective(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-8:
                return 1e10
            ax = ax / s
            b = fast_conv(ax)
            return float(b.max())

        def lbb_gradient(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-8:
                return np.zeros_like(x)
            ax = ax / s
            b = fast_conv(ax)
            max_b = b.max()
            active = np.where(b >= max_b * 0.9999)[0]

            # Subgradient
            n_x = len(ax)
            grad = np.zeros(n_x)
            for k in active:
                for j in range(n_x):
                    idx1 = k - j
                    idx2 = j - k
                    if 0 <= idx1 < n_x:
                        grad[j] += ax[idx1]
                    if 0 <= idx2 < n_x:
                        grad[j] += ax[idx2]
            grad /= len(active)
            # Chain rule for normalization
            grad -= grad * ax / s
            return grad / (s * s)  # Approximate

        bounds_list = [(0.0, None)] * n
        try:
            res = minimize(lbb_objective, tracker.best.astype(np.float64),
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-15,
                                    'gtol': 1e-10})
            if res.fun < 1e10:
                tracker.update(res.x)
                print(f"  L-BFGS-B: J={tracker.best_score:.8f}", flush=True)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)

        # Final LP polish
        if time.monotonic() < deadline and n <= 1024:
            a_final = run_lp_iterations(tracker.best.copy(), max_iters=100,
                                        deadline=deadline, tracker=tracker)

        # One more subgradient pass
        if time.monotonic() < deadline:
            a_final = run_subgradient(tracker.best.copy(), max_iters=500,
                                      deadline=deadline, tracker=tracker, eta_init=0.02)

        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Try slightly different lengths (resample) ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print("PHASE 5: Length Exploration via Resampling", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        base_n = tracker.best.size
        for factor in [0.75, 1.25, 1.5, 2.0, 0.5]:
            if time.monotonic() > deadline:
                break
            new_n = int(round(base_n * factor))
            if new_n < 32 or new_n > 4096:
                continue
            a_res = resample_seq(tracker.best, new_n)
            a_res = np.maximum(a_res, 0.001)
            a_res /= a_res.sum()
            tracker.update(a_res)
            # Quick LP polish
            if new_n <= 512 and time.monotonic() < deadline:
                a_res = run_lp_iterations(a_res, max_iters=30, deadline=deadline,
                                          tracker=tracker)

        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128, dtype=np.float64)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print("FINAL RESULT", flush=True)
    print(f"{'='*60}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
