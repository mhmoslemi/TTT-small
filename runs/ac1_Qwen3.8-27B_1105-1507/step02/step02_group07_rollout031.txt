```python
def subgradient_optimize(f_init, deadline, verbose=False):
    """Optimize using subgradient descent with adaptive step size."""
    f = f_init.copy().astype(np.float64)
    S = f.sum()
    if S < 1e-10:
        return f_init, evaluate_sequence(f_init.tolist())
    f = f / S  # normalize to sum = 1
    n = len(f)
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    # Polyak-like step size
    gap = None
    lr = 1.0
    momentum = np.zeros(n)
    mu = 0.9  # momentum coefficient
    
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        C = np.convolve(f, f)
        M = C.max()
        current_obj = M  # since sum(f) = 1, score = 2n * M
        
        # Polyak step size: lr = gap / ||g||^2
        # gap = current_obj - optimal_obj (we don't know optimal, use approximate)
        # For now, use adaptive lr
        
        # Find active set
        threshold = M - 0.05 * M
        active_mask = C >= threshold
        S_indices = np.where(active_mask)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in S_indices:
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            if j_hi >= j_lo:
                indices = np.arange(j_lo, j_hi + 1)
                grad[indices] += f[k - indices]
        grad /= max(len(S_indices), 1)
        
        # Momentum update
        momentum = mu * momentum + grad
        update = momentum
        
        # Line search
        best_new_f = f.copy()
        best_new_obj = current_obj
        for trial_lr in [lr, lr*0.5, lr*2, lr*0.1, lr*0.8]:
            new_f = f - trial_lr * update
            new_f = np.clip(new_f, 0, 1000)
            ns = new_f.sum()
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            new_C = np.convolve(new_f, new_f)
            new_M = new_C.max()
            if new_M < best_new_obj:
                best_new_obj = new_M
                best_new_f = new_f.copy()
                lr = trial_lr
        
        score = 2 * n * best_new_obj  # since sum = 1
        if score < best_score:
            f = best_new_f
            best_f = f.copy()
            best_score = score
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"  SG iter {it}: obj={score:.6f}")
        else:
            no_improve += 1
            if no_improve > 20:
                lr *= 0.7
                no_improve = 0
                if lr < 1e-8:
                    break
        
        it += 1
    
    return best_f, best_score
```

```python
"""
Subgradient-based optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses momentum subgradient descent with adaptive active set, combined with
LP refinement and spectral initialization from Zadoff-Chu-like sequences.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    if A_cache is None:
        A = build_conv_matrix_sparse(f)
    else:
        A = A_cache
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def compute_subgradient(f, active_indices):
    """Compute subgradient of max(C) w.r.t. f at the given active set."""
    n = len(f)
    grad = np.zeros(n)
    for k in active_indices:
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_hi >= j_lo:
            indices = np.arange(j_lo, j_hi + 1)
            grad[indices] += f[k - indices]
    grad /= max(len(active_indices), 1)
    return grad

    f = f / S

    mu = 0.85  # momentum coefficient

    no_improve = 0
    it = 0

    step_size = 0.05
    no_improve = 0
    it = 0

        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))

        it += 1
        
        if g is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            no_improve += 1
            continue
        
        g = g / S  # normalize to same sum as f
        
        # Line search
        best_t = step_size
        best_s = best_score
        for trial_t in [step_size, step_size * 0.5, step_size * 2]:
            new_f = (1.0 - trial_t) * f + trial_t * g
            new_f = np.clip(new_f, 0.0, 1000.0)
            ns = float(np.sum(new_f))
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = trial_t
        
        if best_s < best_score:
            f = (1.0 - best_t) * f + best_t * g
            f = np.clip(f, 0.0, 1000.0)
            S = float(np.sum(f))
            f = f / S
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            A_cache = build_conv_matrix_sparse(f)  # rebuild
        else:
            no_improve += 1
            step_size *= 0.7
            if step_size < 0.01 or no_improve > 20:
                break
    
    return best_f, best_score


def zadoff_chu_init(n, dc_offset=0.5):
    """Generate a Zadoff-Chu inspired initialization (real, non-negative)."""
    indices = np.arange(n, dtype=np.float64)
    # Zadoff-Chu: z[n] = exp(-j*pi*n*(2n+1)/N)
    # Take real part: cos(pi*n*(2n+1)/N)
    phase = np.pi * indices * (2 * indices + 1) / n
    real_part = np.cos(phase)
    # Add DC offset to make non-negative
    f = real_part + dc_offset
    f = np.abs(f)  # ensure non-negative
    f = f + 0.01  # small floor to avoid zeros
    return f / f.sum()


def windowed_cosine_init(n, freq=0.3, width=0.8):
    """Generate a windowed cosine initialization."""
    indices = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    # Window
    x = (indices - center) / (n * width / 2.0)
    window = np.exp(-x ** 2)
    # Cosine oscillation
    cos_part = np.cos(2 * np.pi * freq * indices / n)
    # Combine (ensure non-negative)
    f = window * (1.0 + 0.5 * cos_part)
    f = np.maximum(f, 0.0)
    if f.sum() < 1e-10:
        f = window
    return f / f.sum()


def smooth_random_init(n, seed=42):
    """Generate a smooth random initialization."""
    rng = np.random.default_rng(seed)
    x = rng.standard_normal(n)
    # Gaussian smoothing
    sigma = max(1.0, n / 20.0)
    kernel_size = int(6 * sigma) | 1  # make odd
    kernel = np.exp(-0.5 * (np.arange(-kernel_size//2, kernel_size//2+1) / sigma) ** 2)
    kernel /= kernel.sum()
    smooth_x = np.convolve(x, kernel, mode='same')
    f = np.abs(smooth_x) + 0.05
    # Apply a mild window to reduce edge effects
    edge = n // 10
    if edge > 0:
        for i in range(edge):
            w = i / edge
            f[i] *= w
            f[n-1-i] *= w
    f = np.maximum(f, 0.0)
    return f / f.sum()
```

```python
    z = f - lr * grad

    # Sort z in decreasing order
    z_sorted = np.sort(z)[::-1]
    # Find the largest k such that z_sorted[k] > (sum(z_sorted[:k+1]) - 1) / (k+1)
    cumsum = np.cumsum(z_sorted)

    k = np.max(np.where(cond)[0]) + 1 if np.any(cond) else 0
    if k == 0:

    else:
        tau = (cumsum[k-1] - 1.0) / k
    f_new = np.maximum(z - tau, 0.0)
    return f_new
```

```python
"""
Subgradient + Proximal Water-Filling + LP hybrid optimization for minimizing
2n*max(conv(f,f))/sum(f)^2. Uses momentum subgradient with proximal step,
adaptive active set, Zadoff-Chu inspired initialization, and periodic LP refinement.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    if A_cache is None:
        A = build_conv_matrix_sparse(f)
    else:
        A = A_cache
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def compute_subgradient(f, active_indices):
    """Compute subgradient of max_k C_k w.r.t. f. d(C_k)/d(f_j) = 2*f_{k-j}."""
    n = len(f)
    grad = np.zeros(n)
    for k in active_indices:
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_hi >= j_lo:
            indices = np.arange(j_lo, j_hi + 1)
            grad[indices] += 2.0 * f[k - indices]
    grad /= max(len(active_indices), 1)
    return grad


def proximal_waterfill(z, target_sum=1.0):
    """Proximal step: argmin 0.5||x - z||^2 s.t. x >= 0, sum(x) = target_sum."""
    n = len(z)
    z_sorted = np.sort(z)[::-1]
    cumsum = np.cumsum(z_sorted)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = z_sorted > (cumsum - target_sum) / rho
    if np.any(cond):
        k = int(np.max(np.where(cond)[0])) + 1
    else:
        k = 0
    if k == 0:
        tau = z_sorted[0] - target_sum / n
    else:
        tau = (cumsum[k - 1] - target_sum) / k
    x = np.maximum(z - tau, 0.0)
    return x

            no_improve += 1
            if no_improve > 100:
                lr *= 0.85
                mu = max(0.5, mu * 0.95)
                no_improve = 0
                if lr < 1e-12:
                    break

        t_k = t_next
        it += 1

    return best_f, best_score


def lp_refine(f_init, deadline, max_iters=40):
    """LP-based refinement."""
    f = f_init.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f_init, evaluate_sequence(f_init.tolist())
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

    A_cache = build_conv_matrix_sparse(f)
    step_size = 0.08
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))

        g = solve_lp_direction(f, M, A_cache=A_cache)
        it += 1

        if g is None:
            no_improve += 1
            if no_improve > 15:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            no_improve += 1
            continue

        g = g / S

        best_t = 0.0
        best_s = best_score
        for trial_t in [step_size, step_size * 0.5, step_size * 1.5, step_size * 0.25]:
            new_f = (1.0 - trial_t) * f + trial_t * g
            new_f = np.clip(new_f, 0.0, 1000.0)
            ns = float(np.sum(new_f))
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = trial_t

        if best_t > 0 and best_s < best_score:
            f = (1.0 - best_t) * f + best_t * g
            f = np.clip(f, 0.0, 1000.0)
            S = float(np.sum(f))
            f = f / S
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            A_cache = build_conv_matrix_sparse(f)
        else:
            no_improve += 1
            step_size *= 0.7
            if step_size < 0.005 or no_improve > 25:
                break

    return best_f, best_score


def zadoff_chu_init(n, dc_offset=0.6):
    """Zadoff-Chu inspired real non-negative initialization."""
    indices = np.arange(n, dtype=np.float64)
    phase = np.pi * indices * (2.0 * indices + 1.0) / n
    real_part = np.cos(phase)
    f = real_part + dc_offset
    f = np.maximum(f, 0.01)
    return f / f.sum()


def windowed_cosine_init(n, freq=0.3):
    """Windowed cosine initialization."""
    indices = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (indices - center) / (n * 0.4)
    window = np.exp(-x ** 2)
    cos_part = np.cos(2.0 * np.pi * freq * indices / n)
    f = window * (1.0 + 0.4 * cos_part)
    f = np.maximum(f, 0.0)
    if f.sum() < 1e-10:
        f = window
    return f / f.sum()


def smooth_random_init(n, seed=42):
    """Smooth random initialization with edge tapering."""
    rng = np.random.default_rng(seed)
    x = rng.standard_normal(n)
    sigma = max(1.0, n / 25.0)
    ks = int(6 * sigma) | 1
    kernel = np.exp(-0.5 * (np.arange(-ks // 2, ks // 2 + 1) / sigma) ** 2)
    kernel /= kernel.sum()
    smooth_x = np.convolve(x, kernel, mode='same')
    f = np.abs(smooth_x) + 0.1
    edge = n // 8
    if edge > 0:
        for i in range(edge):
            w = 0.1 + 0.9 * (i / edge)
            f[i] *= w
            f[n - 1 - i] *= w
    f = np.maximum(f, 0.0)
    return f / f.sum()


def symmetric_init(n, base):
    """Make a sequence symmetric."""
    f = base.copy()[:n]
    f = np.concatenate([f[:n // 2 + 1], f[:n // 2][::-1]])[:n]
    return f / f.sum()


def combined_optimize(f_init, deadline, verbose_prefix=""):
    """Run combined subgradient + LP optimization."""
    f_init = f_init.copy().astype(np.float64)
    S = float(np.sum(f_init))
    if S < 1e-10:
        return f_init, evaluate_sequence(f_init.tolist())
    f_init = f_init / S

    best_f = f_init.copy()
    best_score = evaluate_sequence(f_init.tolist())
    t0 = time.monotonic()
    total = max(deadline - t0, 1.0)

    # Phase A: Subgradient (55% of time)
    p1_deadline = t0 + total * 0.55
    if time.monotonic() < p1_deadline:
        f1, s1 = subgradient_optimize(f_init, p1_deadline,
                                       verbose_prefix=verbose_prefix + "A ")
        if s1 < best_score:
            best_f, best_score = f1, s1

    # Phase B: LP refinement (25% of time)
    p2_deadline = t0 + total * 0.80
    if time.monotonic() < p2_deadline:
        f2, s2 = lp_refine(best_f, p2_deadline, max_iters=30)
        if s2 < best_score:
            best_f, best_score = f2, s2

    # Phase C: Fine subgradient polish (20% of time)
    if time.monotonic() < deadline - 2:
        f3, s3 = subgradient_optimize(best_f, deadline - 2,
                                       verbose_prefix=verbose_prefix + "C ",
                                       lr_start=0.005, band_frac=0.03)
        if s3 < best_score:
            best_f, best_score = f3, s3

    return best_f, best_score


def perturb(f, rng, strength=0.04):
    """Add multiplicative noise."""
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.0, 3.0)
    new_f = np.clip(new_f, 0.0, 1000.0)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S


def propose_candidate(budget_s=1500, **kwargs):
    """Main search function. Returns best sequence found within budget."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)

    # Try parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                score = evaluate_sequence(parent.tolist())
                if np.isfinite(score):
                    best_score = score
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    # Candidate lengths and time allocation
    lengths_to_try = [512, 768, 1024]
    total_budget = deadline - t0
    time_fractions = {512: 0.40, 768: 0.25, 1024: 0.20}

    for n in lengths_to_try:
        if time.monotonic() > deadline - 80:
            print("Time running out, stopping length exploration")
            break

        n_budget = total_budget * time_fractions.get(n, 0.15)
        n_deadline = time.monotonic() + n_budget
        print(f"\n=== n={n} (budget {n_budget:.0f}s) ===")

        # Generate initializations
        inits = []

        if parent is not None:
            if len(parent) == n:
                inits.append(('parent', parent.copy()))
            else:
                x_old = np.linspace(0, 1, len(parent))
                x_new = np.linspace(0, 1, n)
                p = np.interp(x_new, x_old, parent)
                p = np.clip(p, 0, None)
                if p.sum() > 0.01:
                    inits.append(('parent_interp', p / p.sum()))

        # Zadoff-Chu variants
        inits.append(('zac_05', zadoff_chu_init(n, dc_offset=0.5)))
        inits.append(('zac_08', zadoff_chu_init(n, dc_offset=0.8)))

        # Windowed cosine
        inits.append(('wcos_02', windowed_cosine_init(n, freq=0.2)))
        inits.append(('wcos_04', windowed_cosine_init(n, freq=0.4)))

        # Smooth random
        inits.append(('srand1', smooth_random_init(n, seed=7)))
        inits.append(('srand2', smooth_random_init(n, seed=13)))

        # Symmetric variants
        inits.append(('sym_zac', symmetric_init(n, zadoff_chu_init(n, 0.6))))
        inits.append(('sym_srand', symmetric_init(n, smooth_random_init(n, seed=3))))

        # Uniform baseline
        inits.append(('uniform', np.ones(n) / n))

        # Filter and sort by initial score (optimize best inits first)
        scored_inits = []
        for label, f_init in inits:
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            S = float(np.sum(f_init))
            if S < 0.01:
                continue
            sc = evaluate_sequence(f_init.tolist())
            if np.isfinite(sc):
                scored_inits.append((sc, label, f_init))
                if sc < best_score:
                    best_score = sc
                    best_sequence = f_init.copy()

        scored_inits.sort(key=lambda x: x[0])

        # Time per init
        n_valid = len(scored_inits)
        if n_valid == 0:
            continue
        per_init_time = n_budget / n_valid * 0.80

        # Optimize each init (skip if already at or near best)
        for sc, label, f_init in scored_inits:
            if time.monotonic() > n_deadline - 15:
                break

            # Skip if this init is already much worse than best
            if best_score < float('inf') and sc > best_score * 1.02 and n_valid > 3:
                print(f"  [{n}/{label}] skipping (init={sc:.6f} > best*1.02)")
                continue

            init_deadline = min(n_deadline, time.monotonic() + per_init_time)
            if time.monotonic() >= init_deadline - 5:
                continue

            print(f"  [{n}/{label}] init={sc:.6f}, optimizing...", end=" ", flush=True)
            try:
                bf, bs = combined_optimize(f_init, init_deadline,
                                           verbose_prefix=f"    [{n}/{label}] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"-> {bs:.6f} *NEW BEST*")
                else:
                    print(f"-> {bs:.6f}")
            except Exception as e:
                print(f"error: {e}")

    # Perturbation restarts
    if best_sequence is not None and time.monotonic() < deadline - 40:
        print(f"\n=== Perturbation restarts (best: {best_score:.6f}) ===")
        restart_budget = deadline - time.monotonic() - 15
        n_restarts = min(6, max(2, int(restart_budget / 35)))
        per_restart = restart_budget / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break

            strength = 0.02 + 0.015 * r
            f_pert = perturb(best_sequence, rng, strength=strength)

            try:
                bf, bs = combined_optimize(f_pert, time.monotonic() + per_restart * 0.75,
                                           verbose_prefix=f"  [R{r}] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  Restart {r} (str={strength:.3f}): {bs:.6f} *NEW BEST*")
                else:
                    print(f"  Restart {r}: {bs:.6f}")
            except Exception as e:
                print(f"  Restart {r} error: {e}")

    # Final fine-grained polish
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print(f"\n=== Final polish ===")
        try:
            bf, bs = subgradient_optimize(best_sequence, deadline - 5,
                                           verbose_prefix="[final] ",
                                           lr_start=0.003, band_frac=0.02,
                                           mu_start=0.7)
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()

            bf, bs = lp_refine(best_sequence, deadline - 3, max_iters=15)
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()
            print(f"  Final: {best_score:.6f}")
        except Exception as e:
            print(f"  Final error: {e}")

    if best_sequence is None:
        best_sequence = np.ones(256) / 256.0
        best_score = evaluate_sequence(best_sequence.tolist())

    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256) / 256.0
        best_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
```
