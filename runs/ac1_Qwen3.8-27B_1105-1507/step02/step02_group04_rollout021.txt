import numpy as np

def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalized_conv(a):
    """Compute convolution and normalize sequence to sum=1 first."""
    s = float(np.sum(a))
    if s < 1e-15:
        return None
    an = a / s
    return an


def smoothed_grad(conv, a, tau):
    """Compute smoothed subgradient of max(conv) using log-sum-exp."""
    n = len(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[:n]
    return grad


def peak_subgrad(a, conv):
    """Compute exact subgradient at the argmax of conv."""
    n = len(a)
    k_star = int(np.argmax(conv))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            if i == j:
                grad[i] = a[i]
            else:
                grad[i] = 2.0 * a[j]
    return grad


def subgradient_optimize(a, max_iter, tau_start, tau_end, step_init, use_peak_grad=False):
    """Run smoothed subgradient method. Returns best (a, obj)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    step = step_init
    stall = 0
    patience = 300

    for it in range(max_iter):
        frac = it / max(1, max_iter)
        tau = tau_start + (tau_end - tau_start) * frac

        conv = compute_conv(a)
        current_max = float(np.max(conv))
        obj = 2.0 * n * current_max  # since sum=1

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:

                if step < 1e-10:

                a = a + noise * step
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s < 1e-10:

                a = a / s
                continue

            grad = peak_subgrad(a, conv)
        else:

        a = a - step * grad_centered

        s = np.sum(a)

        a = a / s

def build_toeplitz(a):
    """Build Toeplitz matrix A where (A*g)[k] = (a*g)[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        # (a*g)[k] = sum_j a[j]*g[k-j] where 0<=j<n and 0<=k-j<n
        # So j ranges from max(0, k-n+1) to min(n-1, k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def lp_step(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns g/sum(g) or None."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g > 1e-15:
                return g / sum_g
    except Exception:
        pass
    return None

    stall = 0

        conv = compute_conv(a)
        M = float(np.max(conv))

        if sum_g <= 0:

        s_new = np.sum(a_new)
        if s_new < 1e-10:

        obj_new = compute_obj(a_new)

            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

        # Adaptively decrease t if improvement is small
        if it > 20:

def coordinate_descent(a, n_sweeps=3, tol=0.01):
    """Coordinate descent with golden section search on each coordinate.
    Optimizes max(conv) for each coordinate individually."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        return a
    a = a / s
    best_obj = compute_obj(a)

    golden_ratio = (1.0 + np.sqrt(5.0)) / 2.0
    inv_phi = 1.0 / golden_ratio

    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            # Define f(x) = max conv when a[i] = x, others fixed, then normalize
            x_current = a[i]
            # Search range: [0, 3 * avg] where avg = 1/n
            lo = 0.0
            hi = max(3.0 / n, 2.0 * x_current + 1e-6)

            def eval_f(x_val):
                a_test = a.copy()
                a_test[i] = x_val
                s_test = np.sum(a_test)
                if s_test < 1e-15:
                    return np.inf
                a_test = a_test / s_test
                conv = compute_conv(a_test)
                return float(np.max(conv))

            # Golden section search
            c = hi - inv_phi * (hi - lo)
            d = lo + inv_phi * (hi - lo)
            fc = eval_f(c)
            fd = eval_f(d)
            for _ in range(30):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - inv_phi * (hi - lo)
                    fc = eval_f(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + inv_phi * (hi - lo)
                    fd = eval_f(d)

            x_opt = (lo + hi) / 2.0
            new_a = a.copy()
            new_a[i] = x_opt
            s_new = np.sum(new_a)
            if s_new < 1e-15:
                continue
            new_a = new_a / s_new
            new_obj = compute_obj(new_a)
            if new_obj < best_obj - 1e-12:
                a = new_a
                best_obj = new_obj
                improved = True

        if remaining > 20:
            lp_iters = int(min(300, remaining / 0.1))
            a_lp = lp_optimize(a_sg, max_iter=lp_iters, t=0.4, time_limit=3)
            obj_lp = compute_obj(a_lp)
            if obj_lp < obj_sg:
                a_sg = a_lp
                obj_sg = obj_lp

        # Phase 3: Coordinate descent
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            n_sweeps = int(min(10, remaining / 2))
            a_cd = coordinate_descent(a_sg, n_sweeps=max(1, n_sweeps))
            obj_cd = compute_obj(a_cd)
            if obj_cd < obj_sg:
                a_sg = a_cd
                obj_sg = obj_cd

        # Phase 4: Final subgradient polish with peak gradient
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a_pol, _ = subgradient_optimize(a_sg, max_iter=int(remaining * 3),
                                            tau_start=20.0, tau_end=80.0,
                                            step_init=0.02, use_peak_grad=True)
            obj_pol = compute_obj(a_pol)
            if obj_pol < obj_sg:
                a_sg = a_pol
                obj_sg = obj_pol

        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
            print(f"    n={n} init{init_idx}: obj={best_obj:.6f}")

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)
    return best_a, best_obj


def polish_sequence(a, time_budget):
    """Intensive polishing of a single sequence using all techniques."""
    start = time.monotonic()
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    print(f"  Polish start: n={n}, obj={best_obj:.8f}")

    iteration = 0
    while time.monotonic() - start < time_budget - 5:
        iteration += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        # Strategy alternates: LP -> subgrad -> coord desc -> subgrad peak
        strategy = iteration % 4

        if strategy == 0:
            # LP step
            if remaining > 10 and n <= 500:
                a_lp = lp_optimize(a, max_iter=int(min(200, remaining / 0.05)),
                                   t=0.3, time_limit=3)
                obj_lp = compute_obj(a_lp)
                if obj_lp < best_obj:
                    best_obj = obj_lp
                    best_a = a_lp.copy()
                    a = a_lp
                    print(f"  Polish {iteration} LP: obj={best_obj:.8f}")

        elif strategy == 1:
            # Smoothed subgradient
            iters = int(min(5000, remaining * 5))
            a_sg, _ = subgradient_optimize(a, max_iter=iters,
                                           tau_start=10.0, tau_end=60.0,
                                           step_init=0.03)
            obj_sg = compute_obj(a_sg)
            if obj_sg < best_obj:
                best_obj = obj_sg
                best_a = a_sg.copy()
                a = a_sg
                print(f"  Polish {iteration} SG: obj={best_obj:.8f}")

        elif strategy == 2:
            # Coordinate descent
            n_sweeps = int(min(5, remaining / 3))
            a_cd = coordinate_descent(a, n_sweeps=max(1, n_sweeps))
            obj_cd = compute_obj(a_cd)
            if obj_cd < best_obj:
                best_obj = obj_cd
                best_a = a_cd.copy()
                a = a_cd
                print(f"  Polish {iteration} CD: obj={best_obj:.8f}")

        else:
            # Peak subgradient (exact)
            iters = int(min(3000, remaining * 4))
            a_pk, _ = subgradient_optimize(a, max_iter=iters,
                                           tau_start=50.0, tau_end=200.0,
                                           step_init=0.015, use_peak_grad=True)
            obj_pk = compute_obj(a_pk)
            if obj_pk < best_obj:
                best_obj = obj_pk
                best_a = a_pk.copy()
                a = a_pk
                print(f"  Polish {iteration} PK: obj={best_obj:.8f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function. Uses LP power iteration, smoothed subgradient,
    coordinate descent, and peak subgradient in an alternating scheme.
    Focuses on n=250 (parent length) with exploration of nearby lengths.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Try parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # Determine n values to explore
    n_values = [150, 200, 250, 300, 350, 400]
    if parent is not None:
        n_par = len(parent)
        n_values = sorted(set(n_values + [n_par, max(50, n_par - 30), min(500, n_par + 30)]))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    # Phase 1: Quick exploration of each n
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Allocate time based on proximity to parent
        if parent is not None and n == len(parent):
            alloc = min(120.0, remaining * 0.25)
        elif abs(n - 250) <= 50:
            alloc = min(60.0, remaining * 0.15)
        else:
            alloc = min(40.0, remaining * 0.1)

        alloc = max(15.0, alloc)
        print(f"\nN={n}: {remaining:.0f}s remaining, allocating {alloc:.0f}s")

        a_opt, obj = optimize_for_n(n, seed=seed + n * 7919, time_budget=alloc)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  UPDATE: n={n}, obj={best_score:.8f}")

    # Phase 2: Intensive polishing of the best sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== Intensive polishing (n={n_best}), {remaining:.0f}s remaining ===")

        a_pol, obj_pol = polish_sequence(best_seq, time_budget=remaining)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()

    # Phase 3: Try a few more n values if time remains
    if time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        additional_n = [int(len(best_seq) * 0.8), int(len(best_seq) * 1.2)]
        additional_n = [n for n in additional_n if 50 <= n <= 600 and n != len(best_seq)]
        for n in additional_n[:2]:
            if time.monotonic() > deadline - 20:
                break
            alloc = min(40.0, deadline - time.monotonic() - 20)
            if alloc < 15:
                break
            print(f"\nTrying additional n={n}, {alloc:.0f}s")
            # Start from best_seq resampled to new n
            from numpy import interp
            old_n = len(best_seq)
            new_idx = np.linspace(0, old_n - 1, n)
            a_init = np.interp(new_idx, np.arange(old_n), best_seq)
            a_init = np.maximum(a_init, 0)
            s = np.sum(a_init)
            if s < 1e-10:
                a_init = np.ones(n)
                s = n
            a_init = a_init / s

            a_opt, obj = optimize_for_n(n, seed=seed + 99999, time_budget=alloc)
            # Also try from resampled parent
            a_opt2, obj2 = optimize_for_n(n, seed=seed + 88888, time_budget=alloc)
            # Actually just call once with the resampled as init
            # Let's just do a quick polish from resampled
            a_res, obj_res = polish_sequence(a_init, time_budget=alloc // 2)
            obj_best_local = min(obj, obj_res)
            if obj_best_local < best_score:
                best_score = obj_best_local
                best_seq = a_opt.copy() if obj < obj_res else a_res.copy()
                print(f"  UPDATE: n={n}, obj={best_score:.8f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    print(f"\nFinal: n={len(result)}, internal_score={best_score:.8f}, time={elapsed:.1f}s")

    # Validate with the actual evaluate_sequence
    validation = evaluate_sequence(result)
    if np.isfinite(validation):
        if validation < best_score:
            print(f"  Validation gives better: {validation:.8f}")
        else:
            print(f"  Validation: {validation:.8f} (internal: {best_score:.8f})")
        best_score = validation
    else:
        print(f"  WARNING: Validation returned inf!")
        # Fallback to simple sequence
        best_seq = np.ones(100)
        result = [1.0] * 100
        best_score = compute_obj(np.ones(100))

    return result
