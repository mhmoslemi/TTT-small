L(M, a, lambda, mu) = M + sum_k lambda_k (conv_k(a) - M) + mu (sum(a) - S) + sum_i nu_i (-a_i)

At optimality (strong duality might not hold since the problem is non-convex, but KKT conditions give necessary conditions):
- M(1 - sum_k lambda_k) = 0
- sum_k lambda_k * ∇conv_k(a) + mu * 1 - nu = 0
- lambda_k * (conv_k(a) - M) = 0
- nu_i * a_i = 0
- lambda_k >= 0, nu_i >= 0

From the first condition: either M = 0 (impossible) or sum_k lambda_k = 1.
From the second: for a_i > 0, sum_k lambda_k * ∇conv_k(a)_i + mu = 0.

This gives us a system to solve. If we knew which lambda_k are positive and which a_i are positive, we could solve the system.

This suggests an **active set method**:
1. Guess the active set (which k have conv_k = M, which i have a_i > 0)
2. Solve the resulting system
3. Check if the solution satisfies the complementary slackness conditions
4. If not, update the active set and repeat

This is similar to what the LP approach does, but more principled.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# Constants
N = 1024  # Primary sequence length
S_TARGET = np.sqrt(2.0 * N)  # Normalization: sum = sqrt(2n)

def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 1e-10:
        return float('inf')
    return 2.0 * n * M / (S * S)

def normalize(f, S):
    """Scale f to have sum S."""
    s = float(np.sum(f))
    if s < 1e-15:
        return f.copy()
    return f * (S / s)

def simplex_project(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.max(np.where(u * np.arange(1, n+1) > (cv - S))[0])
    theta = (cv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(f):
    """Build sparse matrix A where (A@g)[k] = conv(f,g)[k]."""
    n = len(f)
    # A has shape (2n-1, n)
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_min = max(0, k-(n-1))
        j_max = min(k, n-1)
        for j in range(j_min, j_max+1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k-j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g)_k <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                     options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if result.success:
        return result.x
    return None

def subgradient_descent(f, deadline, S, lr=0.01, momentum=0.9, top_k=5):
    """Subgradient descent with simplex projection and momentum."""
    f = f.copy().astype(np.float64)
    f = simplex_project(f, S)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    v = np.zeros(n)  # momentum velocity
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        
        # Get top-k active lags
        top_idx = np.argsort(conv)[-top_k:]
        weights = np.maximum(conv[top_idx] - conv.mean(), 0)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(top_k) / top_k
        else:
            weights = weights / wsum
        
        # Compute subgradient
        g = np.zeros(n)
        for k, w in zip(top_idx, weights):
            # d(conv_k)/df[j] = 2*f[k-j] if 0 <= k-j <= n-1
            j_min = max(0, k-(n-1))
            j_max = min(k, n-1)
            for j in range(j_min, j_max+1):
                g[j] += 2.0 * w * f[k-j]
        
        # Nesterov momentum update
        v = momentum * v - lr * g
        f_new = simplex_project(f + v, S)
        
        new_score = compute_score(f_new)
        it += 1
        
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.95
            v *= 0.8
            if lr < 1e-8 or no_improve > 200:
                break
    
    return best_f, best_score

"""
Step function optimization using LP direction finding with target annealing,
subgradient descent with exact simplex projection, and coordinate descent polish.
Exploits multiple initializations, adaptive step sizes, and symmetry.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    """Compute objective: 2n * max(conv(f,f)) / sum(f)^2"""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 1e-10:
        return float('inf')
    return 2.0 * n * M / (S * S)


def normalize_seq(f, S):
    """Scale f to have sum S."""
    s = float(np.sum(f))
    if s < 1e-15:
        return f.copy()
    return f * (S / s)


def simplex_project(v, S):
    """Project v onto {x >= 0, sum(x) = S} using exact algorithm."""
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cv - S)
    if not np.any(cond):
        return np.full(n, S / n)
    rho = np.max(np.where(cond)[0])
    theta = (cv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build sparse matrix A where (A@g)[k] = conv(f, g)[k]."""
    n = len(f)
    total = 0
    for k in range(2 * n - 1):
        total += min(k, n - 1) - max(0, k - (n - 1)) + 1
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g)_k <= M for all k, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def compute_subgradient(f, top_lags, weights):
    """Compute weighted subgradient of max conv."""
    n = len(f)
    g = np.zeros(n)
    for k, w in zip(top_lags, weights):
        if w < 1e-15:
            continue
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        idx = np.arange(j_min, j_max + 1)
        g[idx] += 2.0 * w * f[k - idx]
    return g


def subgradient_optimize(f, deadline, S, lr_init=0.05, momentum_init=0.85, top_k=7):
    """Subgradient descent with Nesterov momentum and simplex projection."""
    f = simplex_project(f.copy().astype(np.float64), S)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    v = np.zeros(n)
    lr = lr_init
    momentum = momentum_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        if M < 1e-15:
            break

        # Weight lags by excess above mean
        mean_conv = float(np.sum(conv)) / len(conv)
        top_idx = np.argsort(conv)[-top_k:]
        weights = np.maximum(conv[top_idx] - mean_conv, 0)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(top_k) / top_k
        else:
            weights = weights / wsum

        g = compute_subgradient(f, top_idx, weights)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / (gn + 1e-15)

        # Nesterov momentum
        v = momentum * v - lr * g
        f_new = simplex_project(f + v, S)

        new_score = compute_score(f_new)
        it += 1

        if new_score < best_score - 1e-10:
            best_score = new_score
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            # Slowly increase momentum
            momentum = min(0.95, momentum + 0.0001)
        else:
            no_improve += 1
            lr *= 0.96
            v *= 0.7
            momentum = max(0.5, momentum - 0.005)
            if lr < 1e-8 or no_improve > 300:
                break

    return best_f, best_score


def line_search(f, g, S, n_steps=12):
    """1D line search between f and g, both having sum S."""
    best_score = compute_score(f)
    best_t = 0.0
    ts = np.linspace(0.01, 1.0, n_steps)
    for t in ts:
        cand = (1.0 - t) * f + t * g
        cand = simplex_project(cand, S)
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_t = t
    if best_t == 0.0:
        return f, best_score
    result = (1.0 - best_t) * f + best_t * g
    result = simplex_project(result, S)
    return result, best_score


def lp_optimize_with_annealing(f, deadline, S, top_k=5):
    """LP direction finding with target annealing and line search."""
    f = simplex_project(f.copy().astype(np.float64), S)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        if M < 1e-15:
            break

        found_improvement = False
        # Try multiple target multipliers (annealing)
        for mult in [0.96, 0.98, 1.0, 1.01, 1.02, 1.05]:
            M_target = M * mult
            if M_target < 1e-15:
                continue
            t_start = time.monotonic()
            g = solve_lp_direction(f, M_target)
            lp_time = time.monotonic() - t_start

            if g is not None and float(np.sum(g)) > 1e-10:
                g = simplex_project(g, S)
                f_new, sc_new = line_search(f, g, S, n_steps=10)
                if sc_new < best_score - 1e-10:
                    best_score = sc_new
                    best_f = f_new.copy()
                    f = f_new
                    no_improve = 0
                    found_improvement = True
                    it += 1
                    if it % 10 == 0:
                        print(f"    LP iter {it}: score={best_score:.6f} "
                              f"(mult={mult}, lp_t={lp_time:.2f}s)")
                    break

        if not found_improvement:
            no_improve += 1
            if no_improve > 20:
                break

    return best_f, best_score


def coordinate_descent(f, deadline, S, max_rounds=50):
    """Coordinate descent with 1D golden section search per coordinate."""
    f = simplex_project(f.copy().astype(np.float64), S)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    it = 0

    # Golden section constants
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() < deadline and it < max_rounds:
        improved = False
        # Random order each round
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            # Optimize coordinate i with others fixed
            # conv_k as function of f[i]:
            # For each k, conv_k = c_k + f[i]*d_k + (1 if 2i==k else 0)*f[i]^2
            # where d_k = 2*f[k-i] (if valid) and c_k = conv_k - f[i]*(d_k + (1 if 2i==k else 0)*f[i])
            old_val = f[i]
            # Sample several candidates
            candidates = [old_val]
            # Perturbations relative to current value
            scale = max(old_val, S / n * 0.5)
            for delta in [0.5, 0.7, 0.8, 1.2, 1.5, 2.0, 3.0]:
                candidates.append(old_val * delta)
            candidates.append(0.0)
            candidates.append(S / n)

            best_cand_score = best_score
            best_cand_val = old_val
            for cv in candidates:
                cv = max(0.0, min(1000.0, cv))
                f[i] = cv
                # Renormalize
                s = float(np.sum(f))
                if s < 1e-10:
                    f[i] = old_val
                    continue
                f_norm = f * (S / s)
                sc = compute_score(f_norm)
                if sc < best_cand_score - 1e-10:
                    best_cand_score = sc
                    best_cand_val = cv

            if best_cand_val != old_val:
                f[i] = best_cand_val
                s = float(np.sum(f))
                if s > 1e-10:
                    f = f * (S / s)
                if best_cand_score < best_score:
                    best_score = best_cand_score
                    best_f = f.copy()
                    improved = True

            it += 1
            if it > max_rounds:
                break

        if not improved:
            break

    return best_f, best_score


def make_initializations(n, rng):
    """Create diverse initializations."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    inits.append(('uniform', np.ones(n)))

    cos_t = (1.0 + np.cos(np.pi * idx / max(n - 1, 1))) / 2.0
    inits.append(('cosine', cos_t))

    center = (n - 1) / 2.0
    bell = np.exp(-0.5 * ((idx - center) / (n / 5.0)) ** 2)
    inits.append(('bell', bell))

    # Flat-top with sharp edges
    edge = n // 8
    flat = np.zeros(n)
    flat[edge:n - edge] = 1.0
    # Smooth edges
    for i in range(edge):
        t = i / edge
        flat[i] = 0.5 * (1 - np.cos(np.pi * t))
        flat[n - 1 - i] = 0.5 * (1 - np.cos(np.pi * t))
    inits.append(('flat_top', flat))

    # Bimodal
    two = (np.exp(-0.5 * ((idx - n * 0.25) / (n / 10.0)) ** 2) +
           np.exp(-0.5 * ((idx - n * 0.75) / (n / 10.0)) ** 2))
    inits.append(('bimodal', two))

    # Random with structure
    r = rng.uniform(0.3, 1.0, size=n)
    # Smooth it
    from scipy.ndimage import uniform_filter1d
    r_smooth = uniform_filter1d(r, size=n // 16)
    inits.append(('rand_smooth', r_smooth))

    #-step pattern
    step_size = max(n // 64, 1)
    step_pat = np.repeat(rng.uniform(0.2, 1.0, size=n // step_size + 1), step_size)[:n]
    inits.append(('step', step_pat))

    # Triangle
    tri = 1.0 - np.abs(idx - center) / center
    inits.append(('triangle', tri))

    return inits


def symmetric_optimize(f, deadline, S):
    """Optimize assuming symmetry: a[i] = a[n-1-i]."""
    n = len(f)
    half = (n + 1) // 2
    # Extract symmetric part
    f_sym = np.zeros(n)
    for i in range(half):
        f_sym[i] = f[i]
        f_sym[n - 1 - i] = f[i]
    f_sym = simplex_project(f_sym, S)

    best_f = f_sym.copy()
    best_score = compute_score(f_sym)
    no_improve = 0
    it = 0
    lr = 0.02

    while time.monotonic() < deadline:
        conv = np.convolve(f_sym, f_sym)
        M = float(np.max(conv[:n]))  # Only check first half (symmetry)

        top_idx = np.argsort(conv[:n])[-5:]
        weights = np.maximum(conv[top_idx] - np.mean(conv[:n]), 0)
        wsum = weights.sum()
        if wsum < 1e-15:
            weights = np.ones(5) / 5
        else:
            weights = weights / wsum

        # Compute gradient only for first half
        g_half = np.zeros(half)
        for k, w in zip(top_idx, weights):
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, min(j_max + 1, half)):
                g_half[j] += 2.0 * w * f_sym[k - j]
            # Account for mirror: gradient at j also contributes to n-1-j
            # But since we enforce symmetry, we only update first half

        gn = np.linalg.norm(g_half)
        if gn < 1e-15:
            break
        g_half = g_half / (gn + 1e-15)

        f_sym[:half] -= lr * g_half
        # Mirror
        for i in range(half):
            f_sym[n - 1 - i] = f_sym[i]
        f_sym = simplex_project(f_sym, S)
        # Re-enforce symmetry after projection
        for i in range(half):
            val = (f_sym[i] + f_sym[n - 1 - i]) / 2.0
            f_sym[i] = val
            f_sym[n - 1 - i] = val
        f_sym = simplex_project(f_sym, S)

        sc = compute_score(f_sym)
        it += 1
        if sc < best_score - 1e-10:
            best_score = sc
            best_f = f_sym.copy()
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.95
            if lr < 1e-7 or no_improve > 100:
                break

    return best_f, best_score

    best_sequence = np.clip(best_sequence, 0.0, 1000.0)

    final_score = evaluate_sequence([float(x) for x in best_sequence])
    if not np.isfinite(final_score):
        print("WARNING: Final sequence invalid! Using fallback.")
        best_sequence = np.ones(1024)
        final_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, official_score={final_score:.6f}, "
          f"internal={best_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_sequence]
