One more thought: the LP in the previous code solves:
maximize sum(b) subject to convolve(a, b) <= M, b >= 0

This finds the "flattest" b that is "compatible" with a under the bound M. Then we blend a towards b.

An alternative LP formulation:
minimize max_k (a*a)_k subject to sum(a) = 1, a >= 0

This is not a linear program (because of the a*a term), but we can linearize it by introducing the constraint (a*a)_k <= t for all k, and minimizing t. The problem becomes:
minimize t
subject to: sum_j a_j * a_{k-j} <= t for all k
         sum_j a_j = 1
         a_j >= 0

This is still non-convex (bilinear constraints), but we can use the alternating approach:
- Fix a, compute M = max(a*a)
- Solve: minimize t subject to convolve(a, b) <= t, sum(b) = 1, b >= 0
  (This IS a linear program!)
- Update a <- b
- Repeat

This is essentially what the previous code does (though it maximizes sum(b) instead of minimizing t, which is equivalent up to scaling).

"""
Optimized step function search using spectral initialization, L-BFGS with
analytical gradients on log-sum-exp smoothed objective, LP alternating
refinement, and coordinate descent. Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from numpy.lib.stride_tricks import sliding_window_view
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_obj_and_grad(x, T):
    """Objective and gradient for log-sum-exp smoothed max of convolution."""
    n = len(x)
    conv = np.convolve(x, x)
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_s = np.exp(shifted)
    sum_exp = float(np.sum(exp_s))
    p = exp_s / sum_exp
    lse = max_c + T * np.log(sum_exp)

    sum_a = float(np.sum(x))
    if sum_a < 1e-8:
        return 1e10, np.zeros(n)

    val = 2.0 * n * lse / (sum_a * sum_a)

    # Gradient: d(lse)/d(a_j) = sum_k p_k * d(conv_k)/d(a_j)
    # d(conv_k)/d(a_j) = 2*a_{k-j} if 0<=k-j<n else 0
    # (J^T @ p)_j = 2 * sum_{m=0}^{n-1} a[m] * p[j+m]
    windows = sliding_window_view(p, n)
    jac_T_p = 2.0 * (windows @ x)

    # val = 2n * lse / sum_a^2
    # d(val)/d(a_j) = 2n/sum_a^2 * jac_T_p[j] - 4n*lse/sum_a^3
    grad = (2.0 * n) / (sum_a * sum_a) * jac_T_p - (4.0 * n * lse) / (sum_a ** 3)

    return val, grad


def spectral_init(n, rng, iters=40):
    """Initialize using flat-spectrum (Gerchberg-Saxton style) method."""
    a = np.ones(n) / n
    for _ in range(iters):
        A = np.fft.rfft(a)
        mag = np.abs(A)
        # Blend towards flat spectrum
        target_mag = np.full(len(mag), mag.mean())
        alpha = 0.5
        new_mag = (1 - alpha) * mag + alpha * target_mag
        A_new = new_mag * np.exp(1j * np.angle(A))
        a_new = np.real(np.fft.irfft(A_new, n=n))
        a_new = np.maximum(0.0, a_new)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new /= s
        if np.max(np.abs(a_new - a)) < 1e-10:
            break
        a = a_new
    return a


def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        # Spectral (flat autocorrelation target)
        return spectral_init(n, rng)
    elif mode == 1:
        # Near-uniform with multiplicative noise
        a = np.ones(n) * (1.0 + 0.3 * rng.standard_normal(n))
        a = np.maximum(0.01, a)
        return a / a.sum()
    elif mode == 2:
        # Edge-heavy
        c = 1.0 / np.sqrt(2.0 * n - 1.0)
        a = np.ones(n) * 0.3
        a[0] = c * 2
        a[1] = c * 1.5
        a[-1] = c * 2
        a[-2] = c * 1.5
        a = np.maximum(0.0, a)
        return a / a.sum()
    elif mode == 3:
        # Periodic pattern
        period = max(2, int(n / rng.integers(5, 20)))
        pattern = np.maximum(0.1, rng.random(period))
        a = np.tile(pattern, n // period + 1)[:n]
        return a / a.sum()
    elif mode == 4:
        # Gaussian bump + noise
        center = n / 2.0
        width = n * rng.uniform(0.2, 0.5)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        a += 0.2
        return a / a.sum()
    elif mode == 5:
        # Two bumps
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * 0.1
        a = np.exp(-0.5 * ((np.arange(n) - c1) / w) ** 2) + np.exp(-0.5 * ((np.arange(n) - c2) / w) ** 2)
        a += 0.1
        return a / a.sum()
    else:
        # Pure random
        a = np.maximum(0.0, rng.random(n))
        s = a.sum()
        return a / s if s > 1e-10 else np.ones(n) / n


def lbfgs_optimize(a, time_budget, T_init=None):
    """Optimize using L-BFGS-B with adaptive temperature."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    if T_init is None:
        conv = np.convolve(a, a)
        T_init = 0.05 * float(np.max(conv))

    T = T_init
    best_a = a.copy()
    best_val = compute_obj(a)
    start = time.monotonic()

    num_lbfgs_calls = [0]

    def obj_wrapper(x):
        num_lbfgs_calls[0] += 1
        val, grad = smooth_obj_and_grad(x, T)
        return val, grad

    # Multiple L-BFGS rounds with decreasing T
    max_rounds = 8
    for round_idx in range(max_rounds):
        if time.monotonic() - start > time_budget:
            break

        # Set constraints: sum(x) = 1 handled by working on simplex
        # L-BFGS-B with bounds x >= 0
        bounds = [(0.0, None)] * n

        # Use equality constraint for sum = 1
        from scipy.optimize import NonlinearConstraint
        sum_constraint = NonlinearConstraint(lambda x: x.sum(), 1.0, 1.0)

        remaining = time_budget - (time.monotonic() - start)
        maxiter = min(200, max(20, int(remaining * 50)))

        res = minimize(
            obj_wrapper, a, jac=True, method='L-BFGS-B',
            bounds=bounds, constraints=[sum_constraint],
            options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8, 'maxfun': 5000}
        )

        if res.x is not None:
            a_new = np.clip(res.x, 0.0, None)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                obj = compute_obj(a_new)
                if obj < best_val:
                    best_val = obj
                    best_a = a_new.copy()
                    a = a_new

        # Decrease temperature
        T = T * 0.4
        if T < 1e-6 * float(np.max(np.convolve(a, a))):
            T = 1e-6 * float(np.max(np.convolve(a, a)))

    return best_a, best_val


def build_conv_matrix(a):
    """Build matrix A such that convolve(a, b) can be computed as A @ b."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def lp_refine(a, time_budget):
    """LP alternating refinement."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(200):
        if time.monotonic() - start > time_budget:
            break

        conv = np.convolve(a, a)
        M = float(np.max(conv))
        A = build_conv_matrix(a)

        # LP: maximize sum(b) subject to A @ b <= M, b >= 0
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, 10.0)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'time_limit': 5.0, 'presolve': True})
        except Exception:
            break

        if res.status != 0:
            break

        g = res.x
        g_sum = float(-res.fun)

        if g_sum <= 1.0 + 1e-6:
            stall += 1
            if stall > 10:
                break
            continue
        stall = 0

        g = g / g_sum

        # Line search for blending parameter
        improved = False
        for t in [0.8, 0.5, 0.3, 0.1]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
                improved = True
                break

        if not improved:
            # Try smaller steps
            for t in [0.05, 0.02, 0.01]:
                a_new = (1.0 - t) * a + t * g
                a_new = np.clip(a_new, 0.0, None)
                s = a_new.sum()
                if s < 1e-10:
                    continue
                a_new = a_new / s
                obj = compute_obj(a_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new.copy()
                    a = a_new
                    break

    return best_a, best_obj


def coord_descent_refine(a, rng, time_budget):
    """Coordinate descent / local search refinement."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    max_iters = 50000

    while it < max_iters and (time.monotonic() - start) < time_budget:
        # Random coordinate perturbation
        width = max(1, int(n * rng.uniform(0.005, 0.08)))
        pos = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()

        # Multiplicative perturbation
        sigma = 0.02 + 0.05 * max(0.0, 1.0 - it / max_iters)
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.clip(candidate, 0.0, None)
        s = candidate.sum()
        if s < 1e-10:
            it += 1
            continue
        candidate /= s

        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj
        it += 1

    return best_a, best_obj


def full_pipeline(a_init, time_budget, rng):
    """Full optimization pipeline for a single initialization."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: L-BFGS
    lbfgs_time = time_budget * 0.4
    if time_budget > 2:
        a1, obj1 = lbfgs_optimize(a, lbfgs_time)
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()

    # Phase 2: LP refinement
    elapsed = time.monotonic() - start
    if time_budget - elapsed > 3:
        lp_time = min(time_budget - elapsed - 1, time_budget * 0.3)
        a2, obj2 = lp_refine(best_a, lp_time)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()

    # Phase 3: Another L-BFGS round with lower T
    elapsed = time.monotonic() - start
    if time_budget - elapsed > 3:
        lbfgs_time2 = min(time_budget - elapsed - 1, time_budget * 0.2)
        a3, obj3 = lbfgs_optimize(best_a, lbfgs_time2, T_init=0.01 * float(np.max(np.convolve(best_a, best_a))))
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()

    # Phase 4: Coordinate descent refinement
    elapsed = time.monotonic() - start
    if time_budget - elapsed > 2:
        cd_time = time_budget - elapsed - 0.5
        a4, obj4 = coord_descent_refine(best_a, rng, cd_time)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()

    # Phase 5: Final LP

    if time_budget - elapsed > 2:
        lp_time2 = min(time_budget - elapsed - 0.5, 10.0)
        a5, obj5 = lp_refine(best_a, lp_time2)
        if obj5 < best_obj:
