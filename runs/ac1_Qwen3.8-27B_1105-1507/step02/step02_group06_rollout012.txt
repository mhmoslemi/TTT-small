def compute_score(f):
    n = len(f)

def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = sum_j f[k-j]*g[j] = conv(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):

        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((np.array(vals), (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))),
                       shape=(2 * n - 1, n))

def solve_lp_direction(f, M):
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M * 1.001)  # Slightly relaxed
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

    return None

"""
Optimization of step function heights to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smoothed gradient descent, hard subgradient method, and LP polish.
Explores multiple sequence lengths with diverse initializations.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    """Fast score computation (no type checks)."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def project_simplex(x):
    """Euclidean projection onto the probability simplex."""
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1)
    cond = u > (cssv - 1.0) / ind
    if not np.any(cond):
        return np.ones(n) / n
    i = np.where(cond)[0][-1]
    tau = (cssv[i] - 1.0) / (i + 1)
    return np.maximum(x - tau, 0.0)


def lpse_gradient(conv, T):
    """Compute gradient of T*log(sum exp(conv_k/T)) = logsumexp.
    Returns the gradient as a vector of length n (implicitly via weights)."""
    # Numerically stable logsumexp
    max_c = np.max(conv)
    shifted = conv - max_c
    exp_vals = np.exp(shifted / T)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp  # softmax weights
    return weights, max_c + T * np.log(sum_exp)


def subgradient_from_weights(f, weights):
    """Compute sum_k weights[k] * 2*f[k-i] (for valid indices).
    This gives the gradient of sum_k weights[k] * conv(f,f)_k w.r.t. f."""
    n = len(f)
    grad = np.zeros(n)
    active = np.where(weights > 1e-15)[0]
    for k in active:
        w = weights[k]
        if w < 1e-15:
            continue
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            grad[i] += 2.0 * w * f[k - i]
    return grad


def subgradient_from_weights_fast(f, weights):
    """Faster version using numpy operations."""
    n = len(f)
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for k in active:
        w = weights[k]
        if w < 1e-12:
            continue
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        if i_max >= i_min:
            grad[i_min:i_max+1] += 2.0 * w * f[k - i_max:k - i_min - 1:-1]
    return grad


def lpse_optimize(f_init, deadline, T_init=0.001, T_decay=0.9995,
                  lr_init=0.001, verbose_prefix=""):
    """Log-sum-exp smoothed gradient descent with momentum."""
    f = f_init.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-15:
        return f, float('inf')
    f = f / S  # normalize to sum=1
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    best_conv_max = float(np.max(np.convolve(f, f)))

    momentum = np.zeros(n)
    lr = lr_init
    T = T_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        current_max = float(np.max(conv))

        # Compute LSE gradient
        weights, lse_val = lpse_gradient(conv, T)
        grad = subgradient_from_weights_fast(f, weights)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            break
        grad = grad / grad_norm

        # Momentum update
        momentum = 0.9 * momentum - lr * grad
        f_new = f + momentum
        f_new = f_new / np.sum(f_new)  # renormalize
        f_new = project_simplex(f_new)

        new_conv = np.convolve(f_new, f_new)
        new_max = float(np.max(new_conv))
        new_score = 2.0 * n * new_max

        it += 1

        if new_max < best_conv_max:
            best_conv_max = new_max
            best_score = new_score
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            # Slowly decrease T (anneal towards hard max)
            T = max(T * T_decay, 1e-6)
            # Slowly decrease lr
            lr *= 0.999
            if it % 1000 == 0:
                print(f"    {verbose_prefix}LSE iter {it}: score={best_score:.6f} T={T:.6f} lr={lr:.6f}")
        else:
            no_improve += 1
            lr *= 0.95
            momentum *= 0.5
            f = f  # don't update
            if lr < 1e-9 or no_improve > 2000:
                break

    return best_f, best_score


def hard_subgradient_optimize(f_init, deadline, top_k=5, lr_init=0.005,
                              verbose_prefix=""):
    """Hard subgradient descent on the max with momentum."""
    f = f_init.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-15:
        return f, float('inf')
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    best_max = float(np.max(np.convolve(f, f)))

    momentum = np.zeros(n)
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        current_max = float(np.max(conv))

        # Top-k subgradient
        top_indices = np.argsort(conv)[-top_k:]
        top_vals = conv[top_indices]
        mean_val = np.sum(conv) / len(conv)
        excess = np.maximum(top_vals - mean_val, 0)
        if np.sum(excess) < 1e-18:
            excess = np.ones(top_k)
        weights = excess / np.sum(excess)

        grad = np.zeros(n)
        for k, w in zip(top_indices, weights):
            if w < 1e-12:
                continue
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            if i_max >= i_min:
                grad[i_min:i_max+1] += 2.0 * w * f[k - i_max:k - i_min - 1:-1]

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            break
        grad = grad / grad_norm

        momentum = 0.85 * momentum - lr * grad
        f_new = f + momentum
        f_new = project_simplex(f_new)

        new_conv = np.convolve(f_new, f_new)
        new_max = float(np.max(new_conv))
        new_score = 2.0 * n * new_max

        it += 1
        if new_max < best_max:
            best_max = new_max
            best_score = new_score
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr *= 0.999
            if it % 2000 == 0:
                print(f"    {verbose_prefix}SUB iter {it}: score={best_score:.6f} lr={lr:.6f}")
        else:
            no_improve += 1
            lr *= 0.97
            momentum *= 0.5
            if lr < 1e-10 or no_improve > 3000:
                break

    return best_f, best_score


def build_conv_matrix(f):
    """Build dense matrix A where (A@g)[k] = conv(f,g)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        A[k, j_min:j_max+1] = f[k - j_max:k - j_min - 1:-1]
    return A


def solve_lp_direction(f, M, verbose=False):
    """Solve LP: max sum(g) s.t. g>=0, conv(f,g)_k <= M."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M * 1.0001)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_price': 100})
        if result.success:
            return result.x
    except Exception as e:
        if verbose:
            print(f"    LP error: {e}")
    return None


def lp_optimize(f_init, deadline, step_init=0.1, verbose_prefix=""):
    """LP-based direction finding with line search."""
    f = f_init.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-15:
        return f, float('inf')
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    best_max = float(np.max(np.convolve(f, f)))

    step = step_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break

        t_start = time.monotonic()
        g = solve_lp_direction(f, M)
        lp_time = time.monotonic() - t_start

        if g is None:
            no_improve += 1
            if no_improve > 50:
                break
            step *= 0.5
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-15:
            no_improve += 1
            continue

        g = g / sum_g  # normalize to sum=1

        best_step_max = best_max
        best_step_f = None
        steps_to_try = [step, step * 0.5, step * 1.5, step * 0.25, step * 2.0]
        for alpha in steps_to_try:
            alpha = min(alpha, 1.0)
            if alpha < 0.001:
                continue
            candidate = (1.0 - alpha) * f + alpha * g
            candidate = project_simplex(candidate)
            cand_conv = np.convolve(candidate, candidate)
            cand_max = float(np.max(cand_conv))
            if cand_max < best_step_max:
                best_step_max = cand_max
                best_step_f = candidate

        it += 1
        if best_step_f is not None:
            f = best_step_f
            best_f = f.copy()
            best_max = best_step_max
            best_score = 2.0 * n * best_max
            no_improve = 0
            step *= 0.99
            step = max(step, 0.01)
            if it % 10 == 0:
                print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f} "
                      f"(step={step:.3f}, time={lp_time:.2f}s)")
        else:
            no_improve += 1
            step *= 0.7
            if step < 0.005 or no_improve > 100:
                break

    return best_f, best_score


def make_initializations(n, rng, parent=None):
    """Create diverse initializations for length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Uniform
    inits.append(('uniform', np.ones(n)))

    # Cosine taper
    cos_t = (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0
    inits.append(('cosine', cos_t))

    # Triangular
    tri = 1.0 - np.abs(idx - center) / center
    inits.append(('triangular', tri))

    # Exponential decay
    inits.append(('exp_dec', np.exp(-3.0 * idx / n)))

    # Bell (narrow)
    bell = np.exp(-4.0 * ((idx - center) / (n / 3.0)) ** 2)
    inits.append(('bell', bell))

    # Bimodal
    bim = (np.exp(-8.0 * ((idx - n * 0.25) / (n / 6.0)) ** 2) +
            np.exp(-8.0 * ((idx - n * 0.75) / (n / 6.0)) ** 2))
    inits.append(('bimodal', bim))

    # Random
    inits.append(('random', rng.uniform(0.3, 2.0, size=n)))

    # S-curve
    s_curve = 1.0 / (1.0 + np.exp(-6.0 * (idx / (n - 1) - 0.5)))
    inits.append(('sigmoid', s_curve))

    # Quadratic
    quad = 1.0 - (2.0 * idx / (n - 1) - 1.0) ** 2
    quad = np.clip(quad, 0.05, None)
    inits.append(('quadratic', quad))

    # Parent interpolation
    if parent is not None and len(parent) > 1:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        parent_interp = np.interp(x_new, x_old, parent)
        inits.append(('parent_interp', parent_interp))
    elif parent is not None and len(parent) == n:
        inits.append(('parent', parent.copy()))

    return inits

    lengths_to_try = []
    for n in [512, parent_n, max(512, parent_n * 2), min(4096, parent_n * 3)]:
        if n not in lengths_to_try and n >= 64:
            lengths_to_try.append(n)
    lengths_to_try = sorted(set(lengths_to_try))

    # Also try some non-power-of-2 lengths
    for extra_n in [768, 1536, 3072]:
        if extra_n not in lengths_to_try:
            lengths_to_try.append(extra_n)
    lengths_to_try = sorted(lengths_to_try)

    candidates = []

    for n in lengths_to_try:
        if time.monotonic() > exploration_deadline - 20:
            break
        print(f"\n--- Exploring n={n} ---")
        inits = make_initializations(n, rng, parent)

        for label, f_init in inits:
            if time.monotonic() > exploration_deadline - 5:
                break
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            S = float(np.sum(f_init))
            if S < 1e-10:
                continue
            f_init = f_init / S  # normalize

            score = compute_score(f_init)
            if not np.isfinite(score):
                continue

            # Quick LSE optimization
            quick_deadline = min(exploration_deadline, time.monotonic() + 12.0)
            f_opt, s_opt = lpse_optimize(f_init, quick_deadline, T_init=0.002,
                                          lr_init=0.002)
            if s_opt < score:
                score = s_opt
                f_init = f_opt

            if score < best_score:
                best_score = score
                best_sequence = f_init.copy()
                print(f"  NEW BEST: n={n} {label}: score={score:.6f}")

            candidates.append((score, f_init.copy(), n, label))
            print(f"  {label}: init={compute_score(f_init):.6f} -> {score:.6f}")

    # Sort by score
    candidates.sort(key=lambda x: x[0])
    print(f"\nTop 5 after exploration:")
    for sc, seq, n, lbl in candidates[:5]:
        print(f"  n={n} {lbl}: {sc:.6f}")
    print(f"Best so far: {best_score:.6f}")

    # Phase 2: Focused optimization
    print(f"\n=== Phase 2: Focused Optimization (best={best_score:.6f}) ===")
    focus_deadline = deadline - 200.0

    # Select top candidates (unique by n)
    seen_n = set()
    top_candidates = []
    for sc, seq, n, lbl in candidates:
        if n not in seen_n:
            top_candidates.append((sc, seq, n, lbl))
            seen_n.add(n)
        if len(top_candidates) >= 4:
            break

    for idx_tc, (sc, f_init, n, lbl) in enumerate(top_candidates):
        remaining_time = focus_deadline - time.monotonic()
        if remaining_time < 30:
            break
        print(f"\n--- Focused: n={n} {lbl} (start={sc:.6f}), budget={remaining_time:.0f}s ---")

        # Budget allocation
        time_per_candidate = remaining_time / max(1, len(top_candidates) - idx_tc)

        # Phase 2a: LSE optimization (60% of budget)
        lse_deadline = time.monotonic() + time_per_candidate * 0.5
        f_opt, s_opt = lpse_optimize(f_init, lse_deadline, T_init=0.001,
                                      T_decay=0.9995, lr_init=0.001,
                                      verbose_prefix=f"[n={n}] ")
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()
            print(f"  LSE improved to: {best_score:.6f}")

        # Phase 2b: Hard subgradient (20% of budget)
        sub_deadline = time.monotonic() + time_per_candidate * 0.3
        f_opt2, s_opt2 = hard_subgradient_optimize(f_opt, sub_deadline, top_k=5,
                                                    lr_init=0.003,
                                                    verbose_prefix=f"[n={n}] ")
        if s_opt2 < best_score:
            best_score = s_opt2
            best_sequence = f_opt2.copy()
            print(f"  SUB improved to: {best_score:.6f}")

        # Phase 2c: LP polish (remaining budget for this candidate)
        if time.monotonic() < focus_deadline - 10:
            lp_dl = min(focus_deadline, time.monotonic() + time_per_candidate * 0.3)
            f_opt3, s_opt3 = lp_optimize(f_opt2, lp_dl, step_init=0.08,
                                          verbose_prefix=f"[n={n}] ")
            if s_opt3 < best_score:
                best_score = s_opt3
                best_sequence = f_opt3.copy()
                print(f"  LP improved to: {best_score:.6f}")

    # Phase 3: Final intensive polish on the best
    print(f"\n=== Phase 3: Final Polish (best={best_score:.6f}) ===")
    if best_sequence is not None and time.monotonic() < deadline - 10:
        n_best = len(best_sequence)
        print(f"Polishing n={n_best}...")

        # Multiple rounds of alternating LSE + SUB + LP
        polish_deadline = deadline - 5
        rounds = 3
        round_deadline = (polish_deadline - time.monotonic()) / rounds

        for r in range(rounds):
            rd = time.monotonic() + round_deadline
            if time.monotonic() >= rd:
                break

            # LSE round
            f_p, s_p = lpse_optimize(best_sequence, rd, T_init=0.0005,
                                      T_decay=0.999, lr_init=0.0005,
                                      verbose_prefix=f"[polish{r}-LSE] ")
            if s_p < best_score:
                best_score = s_p
                best_sequence = f_p.copy()

            # Hard subgradient round
            rd = time.monotonic() + round_deadline * 0.8
            f_p2, s_p2 = hard_subgradient_optimize(best_sequence, rd, top_k=7,
                                                    lr_init=0.001,
                                                    verbose_prefix=f"[polish{r}-SUB] ")
            if s_p2 < best_score:
                best_score = s_p2
                best_sequence = f_p2.copy()

            # LP round
            if time.monotonic() < polish_deadline - 5:
                f_p3, s_p3 = lp_optimize(best_sequence, polish_deadline, step_init=0.04,
                                          verbose_prefix=f"[polish{r}-LP] ")
                if s_p3 < best_score:
                    best_score = s_p3
                    best_sequence = f_p3.copy()

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(512)
        best_score = evaluate_sequence(best_sequence.tolist())

    # Verify with official evaluator
    final_list = [float(x) for x in best_sequence]
    final_score = evaluate_sequence(final_list)
    if not np.isfinite(final_score):
        print("WARNING: Failed evaluation! Using fallback.")
        final_list = [float(x) for x in np.ones(256)]
        final_score = evaluate_sequence(final_list)

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(final_list)}, score={final_score:.6f}, time={elapsed:.1f}s")

    return final_list
