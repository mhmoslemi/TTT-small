"""
Optimized step function sequence generator using LP-based Frank-Wolfe with line search,
subgradient polishing, and coordinate descent refinement.
"""

import numpy as np
from scipy.optimize import linprog
import time


def normalize(x):
    """Normalize x to sum to 1, handling edge cases."""
    s = x.sum()
    if s < 1e-15:
        return np.zeros_like(x)
    return x / s


def score_of(f):
    """Compute the score: 2*n*max(conv)/sum^2"""
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (s * s)


def build_conv_matrix(f):
    """Build matrix A such that A @ g = np.convolve(f, g, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def solve_lp(f, M, time_limit=3):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit, 'dual_feasibility_tolerance': 1e-8})
        if res.success and res.x is not None:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None


def line_search(f, g, n_trials=15):
    """Find best step size t in [0,1] that minimizes score of normalized (1-t)f + t*g."""
    n = len(f)
    s_g = g.sum()
    if s_g < 1e-15:
        return 0.0, score_of(f)
    g_norm = g / s_g
    
    best_t = 0.0
    best_score = score_of(f)
    
    # Try a grid first
    trials = np.linspace(0.01, 1.0, n_trials)
    scores = []
    for t in trials:
        f_new = (1 - t) * f + t * g_norm
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            scores.append(np.inf)
        else:
            f_new = f_new / s
            scores.append(score_of(f_new))
    
    scores = np.array(scores)
    min_idx = np.argmin(scores)
    if scores[min_idx] >= best_score:
        return 0.0, best_score
    
    best_t = trials[min_idx]
    best_score = scores[min_idx]
    
    # Refine with binary search around the best point
    lo = trials[max(0, min_idx - 1)]
    hi = trials[min(min_idx + 1, len(trials) - 1)]
    
    for _ in range(10):
        mid = (lo + hi) / 2
        f_mid = (1 - mid) * f + mid * g_norm
        f_mid = np.clip(f_mid, 0, None)
        s = f_mid.sum()
        if s < 1e-10:
            hi = mid
            continue
        f_mid = f_mid / s
        s_mid = score_of(f_mid)
        if s_mid < best_score:
            best_score = s_mid
            best_t = mid
        # Compare mid with lo
        f_lo = (1 - lo) * f + lo * g_norm
        f_lo = np.clip(f_lo, 0, None)
        if f_lo.sum() > 1e-10:
            f_lo = f_lo / f_lo.sum()
            s_lo = score_of(f_lo)
        else:
            s_lo = np.inf
        if s_lo < s_mid:
            hi = mid
        else:
            lo = mid
    
    return best_t, best_score


def fw_optimize(f, time_budget, initial_lr=0.5):
    """Frank-Wolfe optimization with line search."""
    n = len(f)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    
    best_f = f.copy()
    best_score = score_of(f)
    t = initial_lr
    start = time.monotonic()

    no_improve = 0
    
    while time.monotonic() - start < time_budget:

        M = conv.max()
        
        # Solve LP
        remaining = time_budget - (time.monotonic() - start)
        lp_limit = max(0.5, min(3.0, remaining * 0.1))
        g = solve_lp(f, M, time_limit=lp_limit)
        
        if g is None or g.sum() < 1e-10:
            break
        
        g = normalize(g)
        
        # Line search
        best_t, new_score = line_search(f, g)

            best_score = new_score

            print(f"  FW iter {iterations}: score={best_score:.6f}, t={t:.3f}, elapsed={elapsed:.0f}s")

def subgradient_polish(f, n_iter=500, alpha=0.01, rng=None):
    """Subgradient descent using all active peaks."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(f)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    
    best_f = f.copy()
    best_score = score_of(f)
    
    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Use peaks above 90% of max
        threshold = 0.9 * M
        peak_k = np.where(conv >= threshold)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in peak_k:
            # conv[k] = sum_i f[i]*f[k-i]
            # d(conv[k])/df[j] = f[k-j] (if 0 <= k-j < n)
            idx = k - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            if valid.any():
                grad[np.where(valid)[0]] += f[idx[valid]]
        
        if grad.max() > 0:
            grad = grad / grad.max()
        
        # Adaptive step size
        alpha_eff = alpha / (1.0 + 0.01 * it)
        
        f_new = f - alpha_eff * grad
        f_new = np.clip(f_new, 0, None)
        f_new = normalize(f_new)
        
        s = score_of(f_new)
        if s < best_score - 1e-12:
            best_score = s
            best_f = f_new.copy()
            f = f_new
        else:
            # Try smaller step
            f_new = f - alpha_eff * 0.5 * grad
            f_new = np.clip(f_new, 0, None)
            f_new = normalize(f_new)
            s = score_of(f_new)
            if s < best_score - 1e-12:
                best_score = s
                best_f = f_new.copy()
                f = f_new
    
    return best_f, best_score


def coordinate_descent_polish(f, n_sweeps=20, rng=None):
    """Coordinate descent: for each element, find optimal value via ternary/grid search."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(f)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    
    best_score = score_of(f)
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        improved = False
        for i in order:
            original = f[i]
            if original < 1e-12:
                continue
            
            best_val = original
            best_local = best_score
            
            # Try multiple ratios
            for ratio in [0.1, 0.3, 0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.0/0.5, 1.0/0.3, 3.0]:
                f[i] = original * ratio
                s = f.sum()
                if s < 1e-10:
                    continue
                conv = np.convolve(f, f)
                local_score = 2.0 * n * conv.max() / (s * s)
                if local_score < best_local:
                    best_local = local_score
                    best_val = f[i]
            
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
                improved = True
        
        f = normalize(f)
        if not improved:
            break

def diverse_init(n, rng):
