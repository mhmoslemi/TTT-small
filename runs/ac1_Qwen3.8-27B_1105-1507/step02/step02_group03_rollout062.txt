"""
Step function optimization using LP-based direction finding, 
adaptive pressure updates, and efficient coordinate descent.
Targets score < 1.5027 for n=512 and explores n=1024.

import numpy as np
from scipy.optimize import linprog
import time


def compute_score_normalized(f):
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        m_start = max(0, k - n + 1)
        m_end = min(k, n - 1)
        A[k, m_start:m_end+1] = f[k-m_start:k-m_end-1:-1]
    return A


def lp_direction(f, M, time_limit=2.0):
    """Solve LP to find best direction b that keeps conv(f,b) <= M."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    b = np.clip(res.x, 0, None)
    if b.sum() < 1e-10:
        return None
    return b / b.sum()


def pressure_based_update(f, eta, n_active=10):
    """Pressure-based multiplicative update using all near-max convolution positions."""
    conv = np.convolve(f, f)
    M = conv.max()
    # Find positions where conv is within 5% of max
    threshold = 0.95 * M
    active_positions = np.where(conv >= threshold)[0]
    
    pressure = np.zeros(len(f))
    for k in active_positions:
        for i in range(len(f)):
            j = k - i
            if 0 <= j < len(f):
                pressure[i] += f[j]
    
    if pressure.max() > 0:
        pressure = pressure / pressure.max()
    
    f_new = f * np.exp(-eta * pressure)
    f_new = np.clip(f_new, 1e-15, None)
    return f_new / f_new.sum()


def coordinate_descent_1d(f, i, old_conv, old_max_outside):
    """Find optimal value for f[i] given others fixed. Returns best v and best score."""
    n = len(f)
    v = f[i]
    S = f.sum()
    
    # Changing f[i] from v to v':
    # new_conv(k) = old_conv(k) + (v'-v)*f[k-i] for k in [i, min(i+n-1, 2n-2)]
    # The max_outside is the max of old_conv for k NOT in [i, i+n-1]
    
    # For k in [i, i+n-1], j = k-i ranges from 0 to min(n-1, 2n-2-i)
    j_max = min(n-1, 2*n-1-i)
    # new_conv(i+j) = old_conv(i+j) + (v'-v)*f[j] for j=0,...,j_max
    # = (old_conv(i+j) - v*f[j]) + v'*f[j]
    
    intercepts = old_conv[i:i+j_max+1] - v * f[0:j_max+1]
    slopes = f[0:j_max+1].copy()
    
    def score_at(vp):
        new_sum = S - v + vp
        if new_sum < 1e-15:
            return np.inf
        # new_max = max(old_max_outside, max_j(intercepts[j] + vp*slopes[j]))
        inner_max = np.max(intercepts + vp * slopes)
        new_max = max(old_max_outside, inner_max)
        return 2.0 * n * new_max / (new_sum * new_sum)
    
    # Golden section search on [0, upper]
    # upper: try 3*v as initial upper bound
    upper = max(3.0 * v, 0.1 / n)
    lower = 0.0
    
    # First, check a few points to bracket

        if s < best_s:

    for c in grid:
        s = score_at(c)
        if s < best_s:
            best_s = s
            best_v = c
    
    return best_v, best_s


def coordinate_descent(f, n_sweeps=5, rng=None):
    """Full coordinate descent with efficient 1D optimization."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score_normalized(f)
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            old_conv = np.convolve(f, f)
            # Compute max_outside for coordinate i
            k_start = i
            k_end = min(i + n - 1, 2*n-2)
            if k_start == 0 and k_end == 2*n-2:
                old_max_outside = 0.0
            else:
                mask = np.ones(2*n-1, dtype=bool)
                mask[k_start:k_end+1] = False
                old_max_outside = np.max(old_conv[mask]) if np.any(mask) else 0.0
            
            best_v, best_s = coordinate_descent_1d(f, i, old_conv, old_max_outside)
            f[i] = best_v
            f = np.clip(f, 0, None)
            s = f.sum()
            if s < 1e-10:
                break
            f = f / s
            if best_s < best_score:
                best_score = best_s
        # Renormalize after each sweep
        f = normalize(f)
    
    return f, best_score


def lp_optimize(f, time_budget, t0=0.5, rng=None):
    """LP-based optimization with adaptive step size."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    t = t0
    start = time.monotonic()
    stuck_count = 0
    
    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Solve LP
        g = lp_direction(f, M, time_limit=2.0)
        if g is None:
            break
        
        # Try step sizes
        improved = False
        for step in [min(1.0, t * 2), t, t * 0.5, t * 0.25, t * 0.1]:
            f_new = (1 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_score = compute_score_normalized(f_new)
            if new_score < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                t = min(1.0, t * 1.15)
                improved = True
                stuck_count = 0
                break
        
        if not improved:
            stuck_count += 1
            t *= 0.7
            if t < 0.01:
                # Try pressure update as escape
                f_pert = pressure_based_update(f, eta=0.1)
                score_pert = compute_score_normalized(f_pert)
                if score_pert < best_score:
                    f = f_pert
                    best_f = f_pert.copy()
                    best_score = score_pert
                    t = 0.3
                    stuck_count = 0
                else:
                    # Try random perturbation
                    f_pert = f * (1 + 0.1 * rng.normal(0, 1, n))
                    f_pert = np.clip(f_pert, 0, None)
                    f_pert = normalize(f_pert)
                    score_pert = compute_score_normalized(f_pert)
                    if score_pert < best_score:
                        f = f_pert
                        best_f = f.copy()
                        best_score = score_pert
                        t = 0.2
                        stuck_count = 0
                    else:
                        break
    
    return best_f, best_score

import numpy as np

def compute_score_normalized(f):
    """Compute score for normalized sequence (sum=1): 2*n*max(conv)."""
    n = len(f)

def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def build_conv_matrix(f):
    """Build matrix A (2n-1 x n) such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        m_start = max(0, k - n + 1)
        m_end = min(k, n - 1)
        if m_start <= m_end:
            A[k, m_start:m_end + 1] = f[k - m_start:k - m_end - 1:-1]
    return A


def lp_direction(f, M, time_limit=3.0):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0. Returns normalized b or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    b = np.clip(res.x, 0, None)
    s = b.sum()
    if s < 1e-10:
        return None
    return b / s

                if has_quad:
                    if k_start < k_2i:
                        conv[k_start:k_2i] += 2 * delta * old_f[0:i]
                    conv[k_2i] += best_vp * best_vp - v * v
                    if k_2i < k_end:
                        conv[k_2i + 1:k_end + 1] += 2 * delta * old_f[i + 1:j_max + 1]
                else:
                    conv[k_start:k_end + 1] += 2 * delta * old_f
                f[i] = best_vp
                if best_s < best_score:
                    best_score = best_s
    
    f = np.clip(f, 0, None)
    s = f.sum()
    if s > 0:
        f = f / s
    return f, best_score


def lp_optimize(f, time_budget, t0=0.3, rng=None, verbose=False):
    """LP-based optimization with adaptive step size and pressure escapes."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    t = t0
    start = time.monotonic()
    stuck = 0
    
    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        
        g = lp_direction(f, M, time_limit=3.0)
        if g is None:
            break
        
        # Try step sizes from large to small
        improved = False
        for step in [min(1.0, t * 2.5), min(1.0, t * 1.5), t, t * 0.5, t * 0.2, t * 0.1]:
            if step <= 0.001:
                continue
            f_new = (1 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_score = compute_score_normalized(f_new)
            if new_score < best_score - 1e-13:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                t = min(1.0, t * 1.1)
                improved = True
                stuck = 0
                break
        
        if not improved:
            stuck += 1
            t *= 0.65
            if t < 0.005:
                # Escape: try pressure update
                f_p = pressure_update(f, eta=0.08, n_top=5, rng=rng)
                sp = compute_score_normalized(f_p)
                if sp < best_score:
                    f = f_p
                    best_f = f_p.copy()
                    best_score = sp
                    t = 0.25
                    stuck = 0
                else:
                    # Try random multiplicative perturbation
                    f_p = f * np.exp(rng.normal(0, 0.15, n))
                    f_p = np.clip(f_p, 0, None)
                    f_p = normalize(f_p)
                    sp = compute_score_normalized(f_p)
                    if sp < best_score:
                        f = f_p
                        best_f = f.copy()
                        best_score = sp
                        t = 0.2
                        stuck = 0
                    else:
                        break
                if verbose:
                    print(f"    LP stuck, escaped. score={best_score:.6f}, t={t:.3f}")
        
        if verbose and (stuck == 0) and int((time.monotonic() - start) % 10) == 0:
            print(f"    LP iter: score={best_score:.6f}, t={t:.3f}, elapsed={time.monotonic()-start:.0f}s")
    
    return best_f, best_score


def spectral_project(f, n_iter=200, rng=None):
    """FFT-based spectral flattening: project onto flat-magnitude spectrum."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    
    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Target: equalize magnitudes (keep DC)
        n_bins = len(mag)
        current_energy = np.sum(f ** 2)
        target_mag = np.sqrt(current_energy * n_fft / n_bins)
        
        mag_new = np.full_like(mag, target_mag)
        mag_new[0] = 1.0  # DC fixed (sum=1)
        
        # Small phase perturbation every 30 iterations
        if it % 30 == 29:
            phase = phase + rng.normal(0, 0.03, len(phase))
        
        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s
        
        score = compute_score_normalized(f_new)
        if score < best_score - 1e-11:
            best_score = score
            best_f = f_new.copy()
        f = f_new
    
    return best_f, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    r = rng.random()
    if r < 0.2:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.15, 0.4)
        f = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    elif r < 0.4:
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    elif r < 0.6:
        freq = rng.uniform(1, 3)
        f = np.abs(np.sin(np.pi * freq * np.arange(n) / n)) + 0.1
    elif r < 0.8:
        f = np.ones(n)
        for _ in range(4):
            center = int(rng.integers(0, n))
            width = max(2, int(n // 20))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
            f *= (1 + 0.3 * rng.standard_normal() * kernel)
        f = np.abs(f) + 0.05
    else:
        # Low-freq smooth random
        freqs = np.arange(n_fft_small := 1 << (n // 8 + 1).bit_length())
        spec = rng.normal(0, 1, len(freqs)) + rng.normal(0, 1, len(freqs)) * 1j
        spec[0] = abs(spec[0]) + 1.0
        f = np.fft.irfft(spec * np.fft.rfftfreq(len(freqs), 1.0 / len(freqs)), n)[:n]
        f = np.abs(f) + 0.05
    
    f = np.clip(f, 0.01, None)
    f = f * (1 + 0.03 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    return normalize(f)


def simulated_annealing(f, rng, n_iter=3000, T0=0.03, T_end=0.0001):
    """Simulated annealing with diverse moves."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    cur_score = best_score
    no_improve = 0
    
    for it in range(n_iter):
        T = T0 * (T_end / T0) ** (it / max(1, n_iter))
        f_new = f.copy()
        r = rng.random()
        
        if r < 0.25:
            # Block scaling
            width = max(1, int(rng.integers(1, max(2, n // 12))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.3))
        elif r < 0.4:
            # Single element
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] + rng.normal(0, 0.03))
        elif r < 0.55:
            # Swap
            i, j = rng.integers(0, n, 2)
            if i != j:
                f_new[i], f_new[j] = f_new[j], f_new[i]
        elif r < 0.7:
            # Gaussian bump addition/subtraction
            center = int(rng.integers(0, n))
            width = max(2, int(n // 40))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
            f_new += rng.normal(0, 0.05) * kernel
        elif r < 0.85:
            # Inverse (complement) perturbation
            f_new = f.max() - f_new + f.min()
        else:
            # Large random reweight
            f_new *= np.exp(rng.normal(0, 0.25, n))
        
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score_normalized(f_new)
        
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > n_iter // 2 and it > n_iter // 3:
            break
    
    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: LP optimization + pressure + CD + SA across n=512 and n=1024.
    Starts from parent if available, explores multiple initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0
    
    best_overall = None
    best_score_overall = np.inf
    
    # Try parent sequence
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None
    
    if parent is not None:
        parent = np.clip(parent, 0, None)
        if parent.sum() > 0.01:
            parent_n = len(parent)
            parent = normalize(parent)
            parent_score = compute_score_normalized(parent)
            print(f"Parent: n={parent_n}, score={parent_score:.6f}")
            best_overall = parent.copy()
            best_score_overall = parent_score
    
    # Define lengths to try
    lengths = [512, 1024]
    if parent is not None and len(parent) not in lengths:
        lengths.insert(0, len(parent))
    
    print(f"Lengths to try: {lengths}, budget: {budget_s}s, target: 1.5027")
    
    for n in lengths:
        remaining = deadline - time.monotonic()
        if remaining < 120:
            break
        
        print(f"\n=== n={n} (remaining: {remaining:.0f}s) ===")
        
        # Phase 1: Spectral + Pressure (fast initialization refinement)
        if remaining > 180:
            # Use parent if correct length, else diverse init
            if parent is not None and len(parent) == n:
                f0 = parent.copy()
            else:
                # Interpolate parent or use diverse init
                if parent is not None:
                    f0 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
                    f0 = np.clip(f0, 0.01, None)
                    f0 = normalize(f0)
                else:
                    f0 = diverse_init(n, rng)
            
            # Spectral flattening
            t_spec = min(30, remaining * 0.08)
            f1, s1 = spectral_project(f0, n_iter=int(t_spec * 40), rng=rng)
            print(f"  After spectral: {s1:.6f}")
            if s1 < best_score_overall:
                best_score_overall = s1
                best_overall = f1.copy()
            
            # Pressure update
            t_press = min(20, remaining * 0.05)
            f2, s2 = pressure_update(f1, eta=0.12, n_top=5, rng=rng)
            s2 = compute_score_normalized(f2)
            # Do a few pressure iterations
            for _ in range(min(50, int(t_press * 5))):
                f2 = pressure_update(f2, eta=0.08, n_top=5, rng=rng)
                s2_new = compute_score_normalized(f2)
                if s2_new >= s2:
                    f2 = f2 * (1 + 0.05 * rng.normal(0, 1, n))
                    f2 = np.clip(f2, 0, None)
                    f2 = normalize(f2)
                    s2 = compute_score_normalized(f2)
                    break
                s2 = s2_new
            print(f"  After pressure: {s2:.6f}")
            if s2 < best_score_overall:
                best_score_overall = s2
                best_overall = f2.copy()
        
        # Phase 2: LP optimization (main workhorse)
        remaining = deadline - time.monotonic()
        if remaining > 80:
            # Start LP from best so far at this length
            if parent is not None and len(parent) == n:
                f_lp_start = parent.copy()
            else:
                f_lp_start = best_overall.copy() if best_overall is not None else diverse_init(n, rng)
            
            t_lp = min(remaining * 0.55, 400)
            f3, s3 = lp_optimize(f_lp_start, t_lp, t0=0.3, rng=rng, verbose=True)
            print(f"  After LP: {s3:.6f}")
            if s3 < best_score_overall:
                best_score_overall = s3
                best_overall = f3.copy()
        
        # Phase 3: Coordinate descent polishing
        remaining = deadline - time.monotonic()
        if remaining > 40 and best_overall is not None and len(best_overall) == n:
            t_cd = min(remaining * 0.25, 100)
            n_sweeps = max(3, int(t_cd // 2))
            f4, s4 = coordinate_descent(best_overall, n_sweeps=n_sweeps, rng=rng)
            print(f"  After CD: {s4:.6f}")
            if s4 < best_score_overall:
                best_score_overall = s4
                best_overall = f4.copy()
        
        # Phase 4: Simulated annealing
        remaining = deadline - time.monotonic()
        if remaining > 30 and best_overall is not None and len(best_overall) == n:
            n_iter_sa = min(5000, int(remaining * 200))
            f5, s5 = simulated_annealing(best_overall, rng, n_iter=n_iter_sa, T0=0.02, T_end=0.0001)
            print(f"  After SA: {s5:.6f}")
            if s5 < best_score_overall:
                best_score_overall = s5
                best_overall = f5.copy()
        
        # Phase 5: Final LP + CD cycle
        remaining = deadline - time.monotonic()
        if remaining > 60 and best_overall is not None and len(best_overall) == n:
            t_final = min(remaining * 0.6, 120)
            f6, s6 = lp_optimize(best_overall, t_final, t0=0.2, rng=rng)
            if s6 < best_score_overall:
                best_score_overall = s6
                best_overall = f6.copy()
            
            remaining = deadline - time.monotonic()
            if remaining > 20:
                f7, s7 = coordinate_descent(best_overall, n_sweeps=max(3, int(remaining // 3)), rng=rng)
                if s7 < best_score_overall:
                    best_score_overall = s7
                    best_overall = f7.copy()
                print(f"  Final polish: {best_score_overall:.6f}")
    
    # Final verification
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        
        if final_score < best_score_overall:
            best_score_overall = final_score
        else:
            # evaluate_sequence might give different due to clipping
            print(f"  (internal score was {best_score_overall:.6f})")
        return final_list
    
    # Fallback
    print("FALLBACK: using uniform sequence")
    return [1.0] * 512
