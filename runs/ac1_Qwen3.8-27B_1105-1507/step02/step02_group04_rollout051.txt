"""
Optimized step function search combining:
1. FFT-accelerated convolution
2. Exact coordinate descent with incremental updates
3. LP-based global moves with M-bisection for tighter directions
4. Smoothed subgradient with adaptive parameters

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full linear convolution a*a using FFT. Length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj_normalized(a):
    """Compute 2*n*max(conv)/(sum)^2 assuming sum(a)=1, returns 2*n*max(conv)."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * np.max(conv)


def build_toeplitz(a, M_target=None):
    """Build constraint matrix A where A @ g gives (a*g) for all lags."""
    n = len(a)
    n_constraints = 2 * n - 1
    A = np.zeros((n_constraints, n))
    for k in range(n_constraints):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = a[j]
    return A


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns (g, sum_g) or (None, None) if infeasible."""
    n = len(a)
    A = build_toeplitz(a, M)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None


def lp_step(a, M=None, t=0.3, max_lp_iters=1):
    """One LP-based step. Returns new a and improvement flag."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    if s < 1e-10:
        return a, False
    a = a / s

    if M is None:
        conv = compute_conv(a)
        M = float(np.max(conv))

    g, sum_g = solve_lp(a, M)
    if g is None:
        return a, False

    if sum_g <= 1.0 + 1e-6:
        return a, False

    g = g / sum_g
    a_new = (1.0 - t) * a + t * g
    a_new = np.maximum(a_new, 0)
    s_new = np.sum(a_new)
    if s_new < 1e-10:
        return a, False
    a_new = a_new / s_new
    return a_new, True


def lp_bisection_step(a, t=0.3):

    s = np.sum(a)

        # np.correlate(weights, a, 'full')[k] = sum_j weights[j+k] * a[j] (for valid range)
        # We want sum_j a[j] * weights[i0+j] = np.correlate(weights, a, 'full')[i0]
        # Wait: np.correlate(A, B, 'full')[n] = sum_m A[m] * B[m+n-len(A)+1]... 
        # Let me just verify with a small example mentally.
        # np.correlate([w0,w1,w2], [a0,a1], 'full') has length 4
        # [0]: w0*a0
        # [1]: w0*a1 + w1*a0  
        # [2]: w1*a1 + w2*a0
        # [3]: w2*a1
        # So index i gives sum_j weights[j] * a[j+i-1]... 
        # Actually: output[n] = sum_m A[m] * conj(B[m+n+1-len(B)])
        # For real: output[n] = sum_m A[m] * B[m+n+1-len(B)]
        
        # I want: sum_j a[j] * weights[i0 + j] for j = 0..n-1
        # Let's set A = weights (length 2n-1), B = a (length n)
        # output[n] = sum_m weights[m] * a[m + n + 1 - n] = sum_m weights[m] * a[m+1]
        # Hmm, not quite.
        
        # Let me just use a direct approach with FFT for the gradient:
        # grad[i0] = 2 * sum_{j=0}^{n-1} a[j] * weights[i0+j]
        # Let's define: for fixed i0, this is the dot product of a with weights[i0:i0+n]
        # We can compute all i0 at once using:
        # Create a circulant-like computation
        
        # Actually, the cleanest way:
        # grad = 2 * ifft(fft(a, 2n) * conj(fft(shifted_weights, 2n)))  ... no
        
        # Let me just use the convolution interpretation:
        # sum_j a[j] * weights[i0+j] = (a convolved with reversed weights) at position i0+n-1
        # Or: define b = weights reversed, then sum_j a[j]*b[n-1-(i0+j)] ... getting confused.
        
        # Simplest correct approach using numpy:
        # grad[i0] = 2 * np.dot(a, weights[i0:i0+n])
        # This is O(n^2) total which is fine for n up to 500.
        
        # For speed, use the Toeplitz structure or just accept O(n^2):
        grad = np.zeros(n)
        for i0 in range(n):
            grad[i0] = 2.0 * np.dot(a, weights[i0:i0 + n])
            # Correction for the k=2*i0 term (double counting):
            if 2 * i0 < len(weights):
                grad[i0] -= weights[2 * i0] * a[i0]
        
        # Center for sum constraint
        grad_centered = grad - np.mean(grad)
        
        a_new = a - step * grad_centered
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a = a_new / s_new
        
        if it > 200:
            step *= 0.9998

    return best_a


def coordinate_descent(a, n_sweeps=30, n_candidates=12):
    """Coordinate descent with candidate evaluation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    conv = compute_conv(a)
    
    factors = np.array([0.0, 0.05, 0.2, 0.4, 0.6, 0.8, 0.95, 1.0, 1.05, 1.2, 1.5, 2.0])
    
    for sweep in range(n_sweeps):
        any_change = False
        for i in range(n):
            old_ai = a[i]
            candidates = np.maximum(old_ai * factors, 0)
            if old_ai < 1e-8:
                candidates = np.array([0.0, 0.01/n, 0.05/n, 0.1/n, 0.2/n])
            
            best_x = old_ai
            best_peak = np.max(conv)
            
            for x in candidates:
                if abs(x - old_ai) < 1e-12:
                    continue
                a[i] = x
                # Update conv incrementally
                # Change in conv[k] for k in [i, i+n-1]:
                # For k where j=k-i is valid:
                #   if k != 2*i: delta = 2*(x - old_ai)*a[j]
                #   if k == 2*i: delta = x^2 - old_ai^2
                k_start = i
                k_end = min(i + n - 1, 2 * n - 2)
                js = np.arange(k_start, k_end + 1) - i  # j values
                
                # Apply linear change
                conv[k_start:k_end+1] += 2.0 * (x - old_ai) * a[js]
                # Correct for k=2i (the quadratic term)
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i
                    # We added 2*(x-old_ai)*a[i] but should add x^2 - old_ai^2
                    conv[idx] -= 2.0 * (x - old_ai) * a[i]
                    conv[idx] += x * x - old_ai * old_ai
                
                peak = np.max(conv)
                if peak < best_peak:
                    best_peak = peak
                    best_x = x
            
            # Revert and apply best
            if best_x != old_ai:
                # Revert
                k_start = i
                k_end = min(i + n - 1, 2 * n - 2)
                js = np.arange(k_start, k_end + 1) - i
                conv[k_start:k_end+1] -= 2.0 * (best_x - old_ai) * a[js]
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i
                    conv[idx] += 2.0 * (best_x - old_ai) * a[i]
                    conv[idx] -= best_x * best_x - old_ai * old_ai
                # We're back to original conv, now apply best_x properly
                # Actually we already have a[i] = x from the loop, reset it
                a[i] = old_ai  # reset
                # Now apply best_x
                a[i] = best_x
                conv[k_start:k_end+1] += 2.0 * (best_x - old_ai) * a[js]
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i
                    conv[idx] -= 2.0 * (best_x - old_ai) * a[i]
                    conv[idx] += best_x * best_x - old_ai * old_ai
                any_change = True
            else:
                a[i] = old_ai
        
        if not any_change:
            break
        
        # Renormalize
        s = np.sum(a)
        if s > 1e-10:
            scale = 1.0 / s
            a = a * scale
            conv = conv * scale * scale
    
    return a


def spectral_init(n, alpha=0.5, seed=0):
    """Generate a spectrally-motivated initialization."""
    rng = np.random.default_rng(seed)
    x = np.arange(n)
    center = (n - 1) / 2.0
    # Various spectral-motivated windows
    choice = rng.integers(0, 6)
    if choice == 0:
        a = 1.0 + alpha * np.cos(2 * np.pi * x / n)
    elif choice == 1:
        a = 1.0 + alpha * np.cos(2 * np.pi * (x - center) / n)
    elif choice == 2:
        a = 1.0 / (1.0 + alpha * ((x - center) / center) ** 2)
    elif choice == 3:
        # Broader spectral peak
        a = 1.0 + 0.8 * np.cos(2 * np.pi * x / n) + 0.3 * np.cos(4 * np.pi * x / n)
    elif choice == 4:
        # Geometric
        r = rng.uniform(0.5, 0.9)
        a = np.exp(-alpha * np.abs(x - center) / center)
    else:
        # Random smooth
        a = rng.random(n)
        # Smooth
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
    return a / np.sum(a)


def try_parent():
    """Try to use parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            s = np.sum(parent)
            if s > 1e-10:
                parent = parent / s
                return parent.copy()
    except Exception:
        pass
    return None


def optimize_for_n(n, seed, time_budget, verbose=False):
    """Run full optimization pipeline for a given n."""
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    
    best_a = None
    best_obj = np.inf
    
    # Generate initializations
    inits = []
    
    # Try parent if n matches
    parent = try_parent()
    if parent is not None and len(parent) == n:
        inits.append(parent.copy())
    
    # Spectral initializations
    for k in range(4):
        inits.append(spectral_init(n, alpha=rng.uniform(0.2, 1.0), seed=seed + k))
    
    # Cosine-based
    a_cos = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n + rng.uniform(0, 2*np.pi)))
    a_cos = np.maximum(a_cos, 0)
    if np.sum(a_cos) > 1e-10:
        inits.append(a_cos / np.sum(a_cos))
    
    # Random smooth
    a_rand = rng.random(n)
    a_rand = np.convolve(a_rand, np.ones(7)/7, mode='same')
    a_rand = np.maximum(a_rand, 0)
    if np.sum(a_rand) > 1e-10:
        inits.append(a_rand / np.sum(a_rand))
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Try each initialization
    n_inits = len(inits)
    time_per_init = time_budget / n_inits
    
    for idx, a0 in enumerate(inits):
        if time.monotonic() - start > time_budget:
            break
        t_remaining = time_budget - (time.monotonic() - start)
        if t_remaining < 3:
            break
        
        a = a0.copy()
        
        # Phase 1: Coordinate descent (fast local improvement)
        t1 = min(5.0, t_remaining * 0.2)
        if t1 > 1:
            n_cd_sweeps = int(min(50, t1 * 2))
            a = coordinate_descent(a, n_sweeps=max(5, n_cd_sweeps))
        
        # Phase 2: LP steps
        t2 = min(30.0, t_remaining * 0.3)
        steps_done = 0
        while time.monotonic() - start < start + time_budget - t_remaining * 0.3 and steps_done < 30:
            t_remaining = time_budget - (time.monotonic() - start)
            if t_remaining < 5:
                break
            a_new, improved = lp_bisection_step(a, t=0.3)
            if improved:
                a = a_new
                steps_done += 1
                # Follow up with coordinate descent
                a = coordinate_descent(a, n_sweeps=10)
            else:
                # Try plain LP step
                a_new, improved = lp_step(a, t=0.2)
                if improved:
                    a = a_new
                    steps_done += 1
                    a = coordinate_descent(a, n_sweeps=10)
                else:
                    break
        
        # Phase 3: Subgradient refinement
        t_remaining = time_budget - (time.monotonic() - start)
        if t_remaining > 5:
            sg_iters = int(min(3000, t_remaining * 2))
            a = smoothed_subgradient(a, max_iter=max(500, sg_iters),
                                      tau_start=15.0, tau_end=80.0, step_init=0.04)
            # Final coordinate descent
            a = coordinate_descent(a, n_sweeps=20)
        
        # Evaluate
        conv = compute_conv(a)
        obj = 2.0 * n * np.max(conv)  # since sum=1
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if verbose:
                print(f"    n={n} init{idx}: obj={obj:.6f}")
    
    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = 2.0 * n * np.max(compute_conv(best_a))
    
    return best_a, best_obj


def polish_sequence(a, time_budget, verbose=False):
    """Intensively polish a sequence using all available methods."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = 2.0 * n * np.max(compute_conv(a))
    start = time.monotonic()
    rng = np.random.default_rng(99999)
    
    round_num = 0
    while time.monotonic() - start < time_budget - 3:
        round_num += 1
        t_rem = time_budget - (time.monotonic() - start)
        if t_rem < 3:
            break
        
        # LP step
        a_new, improved = lp_bisection_step(a, t=0.25)
        if improved:
            a = a_new
            a = coordinate_descent(a, n_sweeps=15)
            obj = 2.0 * n * np.max(compute_conv(a))
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                if verbose:
                    print(f"    Polish R{round_num} LP: obj={best_obj:.8f}")
        else:
            # Try subgradient
            sg_iters = int(min(2000, t_rem))
            a_new = smoothed_subgradient(a, max_iter=max(500, sg_iters),
                                          tau_start=20.0, tau_end=100.0, step_init=0.03)
            a_new = coordinate_descent(a_new, n_sweeps=15)
            obj = 2.0 * n * np.max(compute_conv(a_new))
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new.copy()
                if verbose:
                    print(f"    Polish R{round_num} SG: obj={best_obj:.8f}")
            else:
                # Random perturbation
                a_pert = a * np.exp(rng.normal(0, 0.03, n))
                a_pert = np.maximum(a_pert, 0)
                s_p = np.sum(a_pert)
                if s_p > 1e-10:
                    a_pert = a_pert / s_p
                    a_pert = coordinate_descent(a_pert, n_sweeps=10)
                    obj_p = 2.0 * n * np.max(compute_conv(a_pert))
                    if obj_p < best_obj:
                        best_obj = obj_p
                        best_a = a_pert.copy()
                        a = a_pert.copy()
                        if verbose:
                            print(f"    Polish R{round_num} pert: obj={best_obj:.8f}")
                    else:
                        a = a  # keep current
    
    return best_a


def propose_candidate(seed=42, budget_s=1500):
    """
    Search for optimal step function sequence minimizing 2*n*max(conv)/(sum)^2.
    Uses coordinate descent, LP-based moves, subgradient method, and spectral inits.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    
    best_seq = None
    best_score = np.inf
    
    # Try parent
    parent = try_parent()
    if parent is not None:
        conv_p = compute_conv(parent)
        s_p = np.sum(parent)
        if s_p > 1e-10:
            parent_norm = parent / s_p
            score_p = 2.0 * len(parent_norm) * np.max(compute_conv(parent_norm))
            best_score = score_p
            best_seq = parent_norm.copy()
            print(f"Parent (n={len(parent_norm)}): obj={score_p:.8f}")
    
    # Sequence lengths to explore
    n_values = [100, 150, 200, 250, 300, 350, 400, 450, 500]
    
    if parent is not None:
        n_p = len(parent)
        n_values = sorted(set(n_values + [n_p, max(50, n_p - 50), min(600, n_p + 50)]))
    
    print(f"Exploring n values: {n_values}")
    
    for n in n_values:
        if time.monotonic() > deadline:
            break
        t_rem = deadline - time.monotonic()
        if t_rem < 20:
            break
        
        # Time allocation based on n
        if n <= 200:
            alloc = min(60.0, t_rem * 0.1)
        elif n <= 350:
            alloc = min(100.0, t_rem * 0.15)
        else:
            alloc = min(120.0, t_rem * 0.2)
        alloc = max(20.0, alloc)
        
        print(f"N={n}: {t_rem:.0f}s remaining, allocating {alloc:.0f}s")
        
        a_opt, obj = optimize_for_n(n, seed=seed + n * 104729, time_budget=alloc,
                                     verbose=(n in [200, 250, 300]))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  UPDATE: n={n}, obj={best_score:.8f}")
    
    # Intensive polishing on best sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        t_rem = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\nPolishing best (n={n_best}), {t_rem:.0f}s remaining")
        
        best_seq = polish_sequence(best_seq, t_rem, verbose=True)
        conv_b = compute_conv(best_seq)
        s_b = np.sum(best_seq)
        final_obj = 2.0 * n_best * np.max(conv_b)
        if final_obj < best_score:
            best_score = final_obj
        print(f"After polish: obj={best_score:.8f}")
    
    # If we still have time, try a few more n values near the best
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        t_rem = deadline - time.monotonic()
        # Try neighbors
        for n_try in [n_best - 25, n_best + 25, n_best - 50, n_best + 50]:
            if n_try < 50 or n_try > 600:
                continue
            if time.monotonic() > deadline - 30:
                break
            alloc = min(60.0, t_rem * 0.3)
            if alloc < 15:
                continue
            print(f"Trying n={n_try} (near best), {alloc:.0f}s")
            a_try, obj_try = optimize_for_n(n_try, seed=seed + n_try * 31,
                                             time_budget=alloc)
            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  UPDATE: n={n_try}, obj={best_score:.8f}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = 2.0
    
    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s")
    
    # Convert to list for evaluation
    result = [float(x) for x in best_seq]
    
    # Validate against the actual evaluation function
    try:
        validation = evaluate_sequence(result)
        if np.isfinite(validation) and validation < best_score:
            print(f"  Validation gave lower: {validation:.8f}")
            best_score = validation
        elif np.isfinite(validation):
            best_score = validation
    except Exception as e:
        print(f"  Validation error: {e}")
    
    return result
