"""
Hybrid LP + log-sum-exp penalty optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses FFT-accelerated convolution, iterative LP direction finding, smooth penalty
gradient descent, and diversification to escape local optima. Explores multiple
sequence lengths with aggressive time management.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_autoconvolve(f):
    """Fast autocorrelation via FFT. Returns full convolution of f with itself."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    f_hat = np.fft.rfft(f, n=fft_size)
    conv = np.fft.irfft(f_hat * f_hat, n=fft_size)[:size]
    return conv


def compute_score(f):
    """Compute the objective value: 2*n*max(conv)/sum^2."""
    n = len(f)
    conv = fast_autoconvolve(f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where A[k,j] = f[k-j] if valid, else 0."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, conv(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_step(f, S, aggressive=False):
    """Perform one LP-based optimization step. Returns new f or None."""
    n = len(f)
    conv_f = fast_autoconvolve(f)
    M = float(np.max(conv_f))
    if M <= 0:
        return None

    M_target = M * (0.95 if aggressive else 1.0)
    g = solve_lp_direction(f, M_target)
    if g is None:
        g = solve_lp_direction(f, M * 1.001)
    if g is None:
        return None

    sum_g = float(np.sum(g))
    if sum_g < 1e-10:
        return None

    # Rescale to match sum
    g = g * (S / sum_g)
    return g


def lp_optimize(f, deadline, verbose_prefix="", max_iters=500):
    """LP-based optimization with adaptive step size and line search."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    step = 0.5
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        g = lp_step(f, S, aggressive=(it % 10 == 9))
        if g is None:
            no_improve += 1
            if no_improve > 25:
                break
            continue

        # Line search
        best_step_score = best_score
        best_step_f = None
        alphas = [step, step * 0.5, step * 1.5, step * 0.25, 0.8, 0.6, 0.3]
        for alpha in alphas:
            alpha = min(max(alpha, 0.05), 0.95)
            candidate = (1.0 - alpha) * f + alpha * g
            candidate = np.clip(candidate, 0.0, None)
            cand_sum = float(np.sum(candidate))
            if cand_sum < 1e-10:
                continue
            candidate = candidate * (S / cand_sum)
            cand_score = compute_score(candidate)
            if np.isfinite(cand_score) and cand_score < best_step_score:
                best_step_score = cand_score
                best_step_f = candidate

        it += 1

        if best_step_f is not None and best_step_score < best_score - 1e-9:
            f = best_step_f
            best_f = f.copy()
            best_score = best_step_score
            no_improve = 0
            if it % 10 == 0:
                print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f}")
            step = min(step * 1.03, 0.9)
        else:
            no_improve += 1
            step *= 0.85
            if step < 0.02 or no_improve > 35:
                break

    return best_f, best_score


def logsumexp_gradient(f, alpha, S):
    """Compute gradient of sum_k exp(alpha*(p*p)[k]) where p = f/S."""
    n = len(f)
    conv = fast_autoconvolve(f)
    # Work with normalized p = f/S
    p = f / S
    conv_p = conv / (S * S)

    # Log-sum-exp for numerical stability
    max_c = np.max(conv_p)
    shifted = alpha * (conv_p - max_c)
    # Clip for numerical safety
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    weights = weights / (np.sum(weights) + 1e-30)

    # Gradient of sum_k weights[k] * (p*p)[k] w.r.t. p
    # d/dp_i [(p*p)[k]] = 2 * p[k-i] (with bounds checking)
    grad_p = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad_p[j] += 2.0 * w * p[k - j]

    # Chain rule: d/dx_i = (1/S) * d/dp_i - (p_i/S^2) * sum_j d/dp_j
    # Simplified: just use grad_p / S as approximate gradient direction
    return grad_p / S


def logsumexp_optimize(f, deadline, alpha_init=10.0, alpha_growth=1.02,
                       lr_init=0.1, verbose_prefix=""):

    f = f.copy().astype(np.float64)
    best_f = f.copy()

    n = len(f)
    S = float(np.sum(f))
    alpha = alpha_init
    lr = lr_init
    momentum = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        grad = logsumexp_gradient(f, alpha, S)
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-15:
            break
        grad = grad / norm_g

        momentum = 0.85 * momentum - lr * grad
        new_f = f + momentum
        new_f = np.clip(new_f, 0.0, None)
        new_sum = float(np.sum(new_f))
        if new_sum < 1e-10:
            lr *= 0.5
            no_improve += 1
            if no_improve > 200:
                break
            it += 1
            continue
        new_f = new_f * (S / new_sum)

        new_score = compute_score(new_f)
        it += 1

        if np.isfinite(new_score) and new_score < best_score - 1e-9:
            best_f = new_f.copy()
            best_score = new_score
            no_improve = 0
            f = new_f
            # Gradually increase alpha
            alpha *= alpha_growth
            if it % 200 == 0:
                print(f"    {verbose_prefix}LSE iter {it}: score={best_score:.6f} alpha={alpha:.1f}")
        else:
            f = new_f
            no_improve += 1
            lr *= 0.97
            momentum *= 0.5
            if no_improve > 300 or lr < 1e-6:
                # Increase alpha to focus on the peak more
                alpha *= 1.5
                no_improve = 0
                lr = max(lr * 2, 0.01)
                if alpha > 10000:
                    break

    return best_f, best_score

def make_initializations(n, rng):
    """Create diverse initializations for length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    inits.append(('uniform', np.ones(n)))

    # Cosine window
    cos_w = 0.5 * (1.0 + np.cos(np.pi * idx / (n - 1)))
    inits.append(('cosine', cos_w + 0.01))

    # Gaussian
    center = (n - 1) / 2.0
    bell = np.exp(-0.5 * ((idx - center) / (n / 8.0)) ** 2)
    inits.append(('gaussian', bell + 0.01))

    # Bimodal
    two = (np.exp(-0.5 * ((idx - n * 0.3) / (n / 12.0)) ** 2) +
           np.exp(-0.5 * ((idx - n * 0.7) / (n / 12.0)) ** 2))
    inits.append(('bimodal', two + 0.01))

    # Smooth random (low-frequency)
    base = rng.uniform(0.3, 1.7, size=max(2, n // 32))
    x_old = np.linspace(0, 1, len(base))
    x_new = np.linspace(0, 1, n)
    smooth_rand = np.interp(x_new, x_old, base)
    inits.append(('smooth_rand', smooth_rand + 0.01))

    # Multi-frequency
    freqs = rng.uniform(1, 8, size=4)
    amps = rng.uniform(0.1, 0.4, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    multi = np.ones(n)
    for fr, am, ph in zip(freqs, amps, phases):
        multi += am * np.cos(2 * np.pi * fr * idx / n + ph)
    inits.append(('multi_freq', np.clip(multi, 0.05, None)))

    # Sigmoid
    s_curve = 1.0 / (1.0 + np.exp(-8.0 * (idx / (n - 1) - 0.5)))
    inits.append(('sigmoid', s_curve))

    # Triangular
    tri = 1.0 - 2.0 * np.abs(idx / (n - 1) - 0.5)
    inits.append(('triangular', tri + 0.01))

    return inits


def diversify(f, strength=0.1, rng=None):
    """Add structured perturbation to escape local optima."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(f)
    # Low-frequency perturbation
    n_modes = max(1, n // 64)
    coeffs = rng.normal(0, strength, size=n_modes)
    perturbation = np.zeros(n)
    for i, c in enumerate(coeffs):
        perturbation += c * np.cos(2 * np.pi * (i + 1) * np.arange(n) / n)
    new_f = f + perturbation * np.std(f)
    new_f = np.clip(new_f, 0.0, None)
    s = float(np.sum(new_f))
    if s < 1e-10:
        return f.copy()
    return new_f * (float(np.sum(f)) / s)

            f_init = np.clip(f_init.astype(np.float64), 0.01, 1000.0)
            S_init = float(np.sum(f_init))
            if S_init < 0.01:
                continue
            score_init = compute_score(f_init)
            if not np.isfinite(score_init):
                continue

            # Quick logsumexp to improve
            quick_dl = min(exploration_deadline, time.monotonic() + 12.0)
            f_opt, s_opt = logsumexp_optimize(f_init, quick_dl, alpha_init=15.0,
                                              alpha_growth=1.05, lr_init=0.15)
            if s_opt < score_init:
                score_init = s_opt
                f_init = f_opt

            if score_init < best_score:
                best_score = score_init
                best_sequence = f_init.copy()
                print(f"  NEW BEST: n={n} {label}: {score_init:.6f}")

            candidate_pool.append((score_init, f_init.copy(), n))
            print(f"  n={n} {label}: {score_init:.6f}")

    candidate_pool.sort(key=lambda x: x[0])
    print(f"\nTop candidates after exploration:")
    for sc, seq, n in candidate_pool[:6]:
        print(f"  n={n}: {sc:.6f}")
    print(f"Best so far: {best_score:.6f}")

    # Phase 2: Focused optimization (next ~1000s)
    print("\n=== Phase 2: Focused Optimization ===")
    focus_deadline = deadline - 250.0

    # Select top diverse candidates
    seen_n = set()
    top_candidates = []
    for sc, seq, n in candidate_pool:
        if n not in seen_n:
            top_candidates.append((sc, seq, n))
            seen_n.add(n)
        if len(top_candidates) >= 3:
            break

    # Also ensure we try the absolute best
    if candidate_pool:
        best_sc, best_seq, best_n = candidate_pool[0]
        if best_n not in seen_n:
            top_candidates.append((best_sc, best_seq, best_n))

    for sc, f_init, n in top_candidates:
        if time.monotonic() > focus_deadline - 100:
            break
        print(f"\n--- Mixed optimization: n={n} (start={sc:.6f}) ---")

        f_opt, s_opt = mixed_optimize(f_init, focus_deadline,
                                       verbose_prefix=f"[n={n}] ")
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()
            print(f"  IMPROVED TO: {best_score:.6f}")

    # Phase 3: Polish and diversify (last ~250s)
    print(f"\n=== Phase 3: Polish (best={best_score:.6f}) ===")
    if best_sequence is not None and time.monotonic() < deadline - 30:
        polish_deadline = deadline - 30
        n_best = len(best_sequence)
        print(f"\nPolishing n={n_best}...")

        # Alternating polish with diversification
        while time.monotonic() < polish_deadline - 30:
            # LP polish
            seg_dl = min(polish_deadline, time.monotonic() + 50.0)
            f_opt, s_opt = lp_optimize(best_sequence, seg_dl,
                                        verbose_prefix="[polish-LP] ", max_iters=30)
            if s_opt < best_score - 1e-9:
                best_score = s_opt
                best_sequence = f_opt.copy()
                print(f"  LP polish -> {best_score:.6f}")

            # Log-sum-exp polish
            seg_dl = min(polish_deadline, time.monotonic() + 30.0)
            f_opt, s_opt = logsumexp_optimize(best_sequence, seg_dl,
                                              alpha_init=300.0,
                                              alpha_growth=1.05,
                                              lr_init=0.03,
                                              verbose_prefix="[polish-LSE] ")
            if s_opt < best_score - 1e-9:
                best_score = s_opt
                best_sequence = f_opt.copy()
                print(f"  LSE polish -> {best_score:.6f}")

            # Subgradient polish
            seg_dl = min(polish_deadline, time.monotonic() + 20.0)
            f_opt, s_opt = subgradient_optimize(best_sequence, seg_dl, lr_init=0.005,
                                                 verbose_prefix="[polish-sub] ")
            if s_opt < best_score - 1e-9:
                best_score = s_opt
                best_sequence = f_opt.copy()
                print(f"  Sub polish -> {best_score:.6f}")

            # Diversification attempt
            if time.monotonic() < polish_deadline - 60:
                for strength in [0.05, 0.1, 0.2]:
                    if time.monotonic() >= polish_deadline - 30:
                        break
                    f_div = diversify(best_sequence, strength=strength, rng=rng)
                    seg_dl = min(polish_deadline, time.monotonic() + 25.0)
                    f_opt, s_opt = lp_optimize(f_div, seg_dl,
                                                verbose_prefix=f"[div-{strength}] ",
                                                max_iters=15)
                    if s_opt < best_score - 1e-9:
                        best_score = s_opt
                        best_sequence = f_opt.copy()
                        print(f"  DIVERSIFY({strength}) -> {best_score:.6f}")
                        break

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(512)

    final_score = evaluate_sequence([float(x) for x in best_sequence])
    if not np.isfinite(final_score):
        print("WARNING: Final sequence failed! Using fallback.")
        best_sequence = np.ones(256)
        final_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, "
          f"internal={best_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_sequence]
