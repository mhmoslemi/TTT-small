import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


class ConvSolver:
    """Precomputes sparse structure for LP constraint matrix."""

    def __init__(self, n):
        self.n = n
        # Precompute row/col pattern
        rows = []
        cols = []
        for k in range(2 * n - 1):
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
        self.rows = np.array(rows, dtype=np.int32)
        self.cols = np.array(cols, dtype=np.int32)
        self.nnz = len(rows)

    def build(self, a):
        """Build sparse matrix with current a values."""
        n = self.n
        vals = np.empty(self.nnz, dtype=np.float64)
        for idx in range(self.nnz):
            k = self.rows[idx]
            j = self.cols[idx]
            vals[idx] = a[k - j]
        A = lil_matrix((2 * n - 1, n))
        A[self.rows, self.cols] = vals
        return A.tocsr()


def lp_best_response(a, solver, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = solver.build(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):

def subgradient_step(a, lr, n_active=5):
    """One subgradient step for minimizing max_k (a*a)[k].
    Gradient of (a*a)[k] w.r.t. a[i] = 2*a[k-i] if 0<=k-i<n, else 0."""
    n = len(a)
    b = fast_conv(a)
    max_b = b.max()
    # Find top n_active peaks
    top_idx = np.argsort(b)[-n_active:]
    # Compute average gradient over active peaks
    grad = np.zeros(n)
    for k in top_idx:
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += 2.0 * a[k - i_hi:k - i_lo + 1][::-1]
    grad /= len(top_idx)

    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s

    inits.append(0.5 + 0.3 * np.sin(10 * np.pi * t))  # 5-lobe sine
    # Gaussian variants
    inits.append(np.exp(-2.0 * (t - 0.5) ** 2) + 0.1)
    inits.append(np.exp(-4.0 * (t - 0.25) ** 2) + np.exp(-4.0 * (t - 0.75) ** 2) + 0.1)
    inits.append(np.exp(-1.0 * (t - 0.5) ** 2) + 0.3)
    # Polynomial
    inits.append(1.0 + 0.5 * (2 * t - 1) ** 2)
    inits.append(1.0 - 0.3 * (2 * t - 1) ** 2)
    # Random
    inits.append(rng.uniform(0.3, 1.0, n))
    inits.append(rng.uniform(0.0, 1.0, n) + 0.1)

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    LP mirror descent + subgradient + coordinate descent to minimize
    the peak-to-sum ratio of self-convolution. Multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Quick LP on n=512 ===
    t0 = time.monotonic()
    n1 = 512
    print(f"\n{'='*50} Phase 1: LP n={n1} {'='*50}", flush=True)
    solver1 = ConvSolver(n1)
    inits = generate_initializations(n1, rng, parent)
    for idx, a0 in enumerate(inits[:5]):  # Top 5 inits for speed
        if time.monotonic() > deadline:
            break
        a0 = np.maximum(a0.astype(np.float64), 1e-6)
        a0 = a0 / a0.sum()
        a = run_lp_iterations(a0, 30, deadline, tracker, solver1)
    tracker.report(f"Phase1 n={n1}: {time.monotonic()-t0:.1f}s")

    # === Phase 2: LP on n=1024 ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n2 = 1024
        print(f"\n{'='*50} Phase 2: LP n={n2} {'='*50}", flush=True)
        solver2 = ConvSolver(n2)
        # Use top sequence resampled + a few inits
        inits = [resample_seq(tracker.best, n2)]
        inits.extend(generate_initializations(n2, rng, parent)[1:4])
        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 = a0 / a0.sum()
            a = run_lp_iterations(a0, 80, deadline, tracker, solver2)
        # Subgradient refinement
        if tracker.best is not None:
            n_best = tracker.best.size
            if n_best == n2:
                sg = subgradient_phase(tracker.best, 500, 0.05, 0.001,
                                       deadline, tracker)
                tracker.update(sg)
        tracker.report(f"Phase2 n={n2}: {time.monotonic()-t0:.1f}s")

    # === Phase 3: LP on n=2048 ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n3 = 2048
        print(f"\n{'='*50} Phase 3: LP n={n3} {'='*50}", flush=True)
        solver3 = ConvSolver(n3)
        inits = [resample_seq(tracker.best, n3)]
        if parent is not None:
            inits.append(resample_seq(parent, n3))
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n3, endpoint=False)))
        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 = a0 / a0.sum()
            iters = 150 if idx == 0 else 60
            a = run_lp_iterations(a0, iters, deadline, tracker, solver3)
        # Subgradient refinement
        if tracker.best is not None and tracker.best.size == n3:
            sg = subgradient_phase(tracker.best, 1000, 0.03, 0.001,
                                   deadline, tracker)
            tracker.update(sg)
        # Coordinate descent
        if tracker.best is not None and tracker.best.size == n3:
            cd = coordinate_descent(tracker.best, max_sweeps=15,
                                    deadline=deadline, tracker=tracker)
            tracker.update(cd)
        tracker.report(f"Phase3 n={n3}: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Try n=4096 if time allows ===
    remaining = deadline - time.monotonic()
    if tracker.best is not None and remaining > 200:
        t0 = time.monotonic()
        n4 = 4096
        print(f"\n{'='*50} Phase 4: LP n={n4} (time remaining: {remaining:.0f}s) {'='*50}", flush=True)
        solver4 = ConvSolver(n4)
        inits = [resample_seq(tracker.best, n4)]
        a0 = np.maximum(inits[0].astype(np.float64), 1e-6)
        a0 = a0 / a0.sum()
        iters = 80
        a = run_lp_iterations(a0, iters, deadline, tracker, solver4)
        # Subgradient
        if tracker.best is not None and tracker.best.size == n4:
            sg = subgradient_phase(tracker.best, 500, 0.02, 0.001,
                                   deadline, tracker)
            tracker.update(sg)
        tracker.report(f"Phase4 n={n4}: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final refinement on best found ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase 5: Final refinement n={n_best} {'='*50}", flush=True)

        # More LP iterations
        solver_f = ConvSolver(n_best)
        a = run_lp_iterations(tracker.best, 100, deadline, tracker, solver_f)

        # Subgradient
        if tracker.best is not None and tracker.best.size == n_best:
            sg = subgradient_phase(tracker.best, 2000, 0.02, 0.0005,
                                   deadline, tracker)
            tracker.update(sg)

        # Coordinate descent
        if tracker.best is not None and tracker.best.size == n_best:
            cd = coordinate_descent(tracker.best, max_sweeps=20,
                                    deadline=deadline, tracker=tracker)
            tracker.update(cd)

        # One more LP pass
        if tracker.best is not None and tracker.best.size == n_best and time.monotonic() < deadline:
            a = run_lp_iterations(tracker.best, 50, deadline, tracker, solver_f)

        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
