def generate_power_law_parent(n_min=128, n_max=4096, exponent_min=-5, exponent_max=5):
    # This produces a candidate that decays or grows according to a power law.
    seq = []
    length_candidates = [n_min, n_max]
    for length in length_candidates:
        t_arr = np.linspace(0, 1, length)
        exponent = exponent_min + (exponent_max - exponent_min) * np.random.random()
        # To keep in a bounded range [0,1000]
        amplitude = np.exp(np.random.rand()) * 10.0
        candidates = t_arr.astype(np.float32)
        # Shift the power law curve so that it is centered around 0.5 and stays non-negative.
        candidates = amplitude * (0.5 + candidates**exponent)
        candidates = np.maximum(candidates, 0.0)  # Could numerically go negative, so protect.
        candidates = np.minimum(candidates, 1000.0)  # Clamping to avoid evaluation function issues.
        seq.append(candidates)
    return seq
