import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

Next compute $g_0=(b_0,\dots,b_{n-1})$ by solving a linear program, i.e.\ maximizing $\sum_j b_j$ subject to $b_j\ge0$ and $\|f*g_0\|_\infty\le M$; as is standard, the optimum is attained at an extreme point determined by an active set of binding inequalities, here corresponding to important constraints where the convolution bound $(f*g_0)(x)\le M$ is tight and limiting. def compute_score_normalized(f):
    """Compute score for a normalized sequence (sum=1): 2*n*max(conv)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def spectral_flat(f, n_iter=500, rng=None, target_ratio=1.0):
    """Iterative spectral flattening: project onto flat-magnitude spectrum + non-negative simplex."""
    if rng is None:
        rng = np.random.default_rng(0)

        phase = np.angle(F)

        # DC is fixed at 1 (sum = 1)
        # Target magnitude for non-DC bins
        n_bins = len(mag)
        # Total energy: sum(f^2) = (1/n_fft)*sum(|F|^2)
        # For flat spectrum: sum(|F|^2) = n_bins * target_mag^2

        target_energy = current_energy  # Keep same energy

def pressure_update(f, n_iter=200, eta=0.1, rng=None):
    """Fast pressure-based update: reduce elements that contribute to convolution peaks."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score_normalized(f)
    stall_count = 0

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()

        # Compute pressure: for each i, how much does increasing f[i] increase the max?
        # The convolution at position j contributed by f[i] is f[i] * f[j-i]
        # The max is at j*, so pressure on f[i] is f[j*-i]
        # But we should consider all positions near the max
        # Use: pressure_i = max over j of f[j] * f[|j-i|] weighted by proximity to argmax

        # Simpler: use the subgradient at the argmax
        j_star = np.argmax(conv)
        pressure = np.zeros(n)
        for i in range(n):
            idx = j_star - i
            if 0 <= idx < n:
                pressure[i] = f[idx]

        # Also consider the second-highest peak
        conv_sorted = np.sort(conv)[::-1]
        if len(conv_sorted) > 1:
            second_peak = conv_sorted[1]
            if second_peak > 0.9 * M:
                # Find second peak position
                j_star2 = np.argmax(np.where(np.isclose(conv, second_peak), conv, -1))
                if j_star2 != j_star:
                    for i in range(n):
                        idx = j_star2 - i
                        if 0 <= idx < n:
                            pressure[i] += 0.5 * f[idx]

        if pressure.max() > 0:
            pressure = pressure / pressure.max()

        # Multiplicative update
        eta_eff = eta * (1.0 / (1.0 + 0.01 * it))
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        score = compute_score_normalized(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 100:
                # Perturb to escape
                f_pert = f * (1.0 + 0.1 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score_normalized(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score_normalized(f)
                    stall_count = 0
                else:
                    break

    return best_f, best_score


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def lp_refine(f, time_budget, t0=0.5, rng=None, lp_time_limit=3):
    """LP-based refinement with adaptive step size."""
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)

    best_f = f.copy()
    best_score = compute_score_normalized(f)
    t = t0
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()

        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': lp_time_limit})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break

        g = b / b_sum

        # Try multiple step sizes (larger first)
        improved = False
        for step in [min(1.0, t * 1.5), t, t * 0.5, t * 0.25]:
            f_new = (1 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s

            new_score = compute_score_normalized(f_new)
            if new_score < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                t = min(1.0, t * 1.1)
                improved = True
                break

        if not improved:
            t *= 0.6
            if t < 0.005:
                # Try perturbation
                f_pert = f + rng.normal(0, 0.02, n)
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score_normalized(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score_normalized(f)
                    t = 0.1
                else:
                    break

    return best_f, best_score


def coordinate_descent(f, n_sweeps=10, rng=None):
    """Coordinate descent polishing."""
    if rng is None:
        rng = np.random.default_rng(0)

    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score_normalized(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-10:
                continue
            best_val = original
            best_local = best_score

            # Try geometric spread of values
            for factor in [0.3, 0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.0/0.5, 1.0/0.3]:
                f[i] = max(0.0, original * factor)
                conv = np.convolve(f, f)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]

            f[i] = best_val
            if best_local < best_score:
                best_score = best_local

        f = np.clip(f, 0, None)
        f = normalize(f)

    return f, best_score


def simulated_annealing(f, rng, n_iter=5000, T0=0.05, T_end=0.0001):
    """Simulated annealing with diverse moves."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    cur_score = best_score
    no_improve = 0

    for it in range(n_iter):
        T = T0 * (T_end / T0) ** (it / n_iter)
        f_new = f.copy()

        r = rng.random()
        if r < 0.3:
            # Block scaling
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.4))
        elif r < 0.5:
            # Single element perturbation
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] + rng.normal(0, 0.05))
        elif r < 0.7:
            # Two-element swap
            i, j = rng.integers(0, n, 2)
            if i != j:
                f_new[i], f_new[j] = f_new[j], f_new[i]
        elif r < 0.85:
            # Smooth perturbation (Gaussian kernel)
            center = int(rng.integers(0, n))
            width = max(2, int(n // 32))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (width ** 2))
            f_new += rng.normal(0, 0.1) * kernel
        else:
            # Large move: random reweight
            f_new *= np.exp(rng.normal(0, 0.3, n))

        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score_normalized(f_new)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 2000 and it > n_iter // 2:
            break

    return best_f, best_score

    f_large = np.interp(x_large, x_small, f_small)
    f_large = np.clip(f_large, 0.01, None)
    f_large = normalize(f_large)

def diverse_init(n, rng):
    """Generate a diverse initial sequence."""
    r = rng.random()
    if r < 0.25:
        # Gaussian bump
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.4)
        f = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    elif r < 0.5:
        # Cosine window
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    elif r < 0.75:
        # Product of sines ( creates interesting structure)
        f = np.abs(np.sin(np.pi * np.arange(n) / n * rng.uniform(1, 3)))
        f += 0.1
    else:
        # Random smooth
        f = np.ones(n)
        for _ in range(5):
            center = int(rng.integers(0, n))
            width = int(rng.integers(3, n // 4))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / max(1, width ** 2))
            f *= (1 + 0.5 * rng.standard_normal() * kernel)
        f = np.abs(f) + 0.05

    f = np.clip(f, 0.01, None)
    f = normalize(f)
    # Add small noise to break symmetry
    f = f * (1 + 0.05 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f
