            a = np.maximum(a, 0)
            record(a, f"deep_{trial}")

        # Coordinate descent with finer steps
        cd_deadline2 = min(deadline, time.monotonic() + 300.0)
        a_cd2, J_cd2 = coordinate_descent(best_seq, cd_deadline2, [0.01, -0.01, 0.005, -0.005])
        record(a_cd2, "fine_cd")

    # Final fallback
    if best_seq is None:
        best_seq = np.ones(128)
        best_score = 2.0

    elapsed = time.monotonic() - t_start
    print(f"\n=== FINAL: J={best_score:.10f} n={best_seq.size} time={elapsed:.0f}s ===", flush=True)
    return [float(x) for x in best_seq]
