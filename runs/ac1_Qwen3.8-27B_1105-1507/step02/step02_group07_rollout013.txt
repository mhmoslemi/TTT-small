"""
Multi-strategy optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses fast soft-max gradient with Adam, peak shaving, LP polish,
and diverse multi-start with adaptive length selection.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(f):
    """Compute full convolution using FFT for large n."""
    n = len(f)
    if n <= 512:
        return np.convolve(f, f)
    # Use FFT
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    F = np.fft.rfft(f, fft_size)
    C = np.fft.irfft(F * F, fft_size)[:size]
    return C


def compute_soft_max_grad(f, beta):
    """Compute gradient of L_beta(f) = (1/beta) * log(sum exp(beta * C[k]))."""
    n = len(f)
    C = fast_conv(f)
    max_C = C.max()
    # Numerical stability
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros(n), max_C
    p = exp_C / total
    grad = 2.0 * np.correlate(p, f, mode='valid')
    soft_val = max_C + np.log(total) / beta
    return grad, soft_val, C


def top_k_grad(f, C, k=5):
    """Compute gradient weighted by top-k peaks."""
    n = len(f)
    # Get top k values
    top_indices = np.argsort(C)[-k:]
    top_vals = C[top_indices]
    max_val = C.max()
    # Weight by how much each exceeds the mean
    mean_val = C.mean()
    weights = np.zeros(len(C))
    for idx in top_indices:
        w = max(0, C[idx] - mean_val) / (max_val - mean_val + 1e-30)
        weights[idx] = w
    weights /= weights.sum()
    grad = 2.0 * np.correlate(weights, f, mode='valid')
    return grad


def adam_step(f, grad, m, v, beta1, beta2, it, lr):
    """Adam update step."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** it)
    v_hat = v / (1 - beta2 ** it)
    update = m_hat / (np.sqrt(v_hat) + 1e-8)
    new_f = f - lr * update
    new_f = np.clip(new_f, 0.0, 1000.0)
    S = float(np.sum(new_f))
    if S < 1e-10:
        new_f = np.clip(f, 0.0, 1000.0)
        S = float(np.sum(new_f))
    new_f = new_f / S
    return new_f, m, v

    n = len(f)
    k_star = int(np.argmax(C))
    # Find which f[j] contribute most to C[k_star]
    # C[k_star] = sum_j f[j] * f[k_star - j]
    # Contribution of f[j] is f[j] * f[k_star - j]
    J = []
    for j in range(n):
        if 0 <= k_star - j < n:
            J.append((j, f[j] * f[k_star - j]))
    J.sort(key=lambda x: -x[1])
    # Reduce the top contributors
    total_contrib = sum(c for _, c in J)
    if total_contrib < 1e-30:
        return f
    threshold = total_contrib * (1 - fraction)
    cumulative = 0
    for j, c in J:
        if cumulative >= threshold:
            break
        # How much to reduce f[j]?
        # If we reduce f[j] by delta, the contribution reduces by delta * f[k_star - j]
        # (approximately, ignoring second-order effects)
        if j != k_star - j:
            delta = min(f[j], c * fraction / (f[k_star - j] + 1e-30))
        else:
            delta = min(f[j], np.sqrt(c) * fraction)
        f[j] -= delta
        cumulative += delta * f[k_star - j]
    f = np.clip(f, 0.0, 1000.0)
    S = float(np.sum(f))
    if S < 1e-10:
        return f  # shouldn't happen
    f = f / S
    return f

            cols.append(j)
            vals.append(f[k - j])

            break

            break

                j_min = max(0, k - (n - 1))

            step_size *= 0.7
            if no_improve > 15:

def tukey_window(n, r=0.3):
    w = np.ones(n)
    m = max(1, int(np.floor(r * n / 2)))
    for i in range(m):
        w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
        w[n - 1 - i] = w[i]
    return w


def hann_window(n):
    w = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    return w


def flat_top_window(n):
    """Flat-top window (maximally flat in frequency domain)."""
    a = [1.0, 1.93, 1.29, 0.388, 0.028]
    w = np.zeros(n)
    for i in range(n):
        x = i / (n - 1) * 2 - 1  # -1 to 1
        w[i] = a[0] - a[1] * np.cos(np.pi * x) + a[2] * np.cos(2 * np.pi * x) \
               - a[3] * np.cos(3 * np.pi * x) + a[4] * np.cos(4 * np.pi * x)
    w = np.abs(w)
    w = w / w.max()
    return w


def cosine_tapered(n, gamma=2.0):
    """Cosine tapered to the gamma power."""
    x = np.arange(n) / (n - 1) * 2 - 1  # -1 to 1
    w = np.cos(np.pi * x / 2) ** gamma
    return np.clip(w, 0, None)


def perturb(f, rng, strength=0.05):

"""
Multi-phase optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Combines fast soft-max gradient descent (Adam), top-k peak targeting,
LP-based direction finding, diverse multi-start initialization,
and adaptive length selection. Key improvements over previous:
- Much faster per-iteration (no LP in main loop) enabling 10x more steps
- Adaptive soft-max temperature with top-k blending
- Hybrid gradient+LP refinement to escape plateaus
- Diverse window initializations (flat-top, cosmetic-tapered, super-Gaussian)

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(f):
    """Compute full linear convolution, using FFT for large n."""
    n = len(f)
    if n <= 600:
        return np.convolve(f, f)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    F = np.fft.rfft(f, fft_size)
    C = np.fft.irfft(F * F, fft_size)[:size]
    return C


def compute_soft_max_grad(f, beta):
    """Compute gradient and value of L_beta = (1/beta)*log(sum exp(beta*C[k]))."""
    n = len(f)
    C = fast_conv(f)
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros(n), max_C, C
    p = exp_C / total
    # grad[j] = 2 * sum_m p[j+m] * f[m] = 2 * correlate(p, f, 'valid')[j]
    grad = 2.0 * np.correlate(p, f, mode='valid')
    soft_val = max_C + np.log(total) / beta
    return grad, soft_val, C


def top_k_weighted_grad(f, C, k=5):
    """Gradient weighted toward top-k peaks of the convolution."""
    n = len(f)
    sorted_idx = np.argsort(C)
    top_idx = sorted_idx[-k:]
    max_val = C.max()
    mean_val = C.mean()
    if max_val - mean_val < 1e-30:
        return 2.0 * np.correlate(C / C.sum(), f, mode='valid')
    weights = np.zeros(len(C))
    for idx in top_idx:
        weights[idx] = (C[idx] - mean_val) / (max_val - mean_val)
    if weights.sum() < 1e-30:
        return np.zeros(n)
    weights /= weights.sum()
    return 2.0 * np.correlate(weights, f, mode='valid')

            if no_improve > patience:
                lr *= 0.6
                no_improve = 0
                patience = int(patience * 1.2)
                if lr < 1e-7:
                    break

    return best_f, best_score


def build_conv_matrix(f):
    """Build sparse convolution matrix A where (A@g)[k] = conv(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((np.array(vals, dtype=np.float64),
                       (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))),
                      shape=(2 * n - 1, n))


def lp_polish(f, deadline, max_iters=50, step_init=0.1, verbose=""):
    """LP-based direction finding for final polishing."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, float('inf')
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    A = build_conv_matrix(f)
    step = step_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        C = fast_conv(f)
        M = float(C.max())

        try:
            result = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                             bounds=[(0, None)] * n, method='highs')
            if not result.success:
                break
            g = result.x
        except Exception:
            break

        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            break
        g = g / S

        improved = False
        for t in [step, step * 0.5, step * 2, step * 0.7, step * 1.5, step * 0.3]:
            if t <= 0 or t > 0.95:
                continue
            new_f = (1 - t) * f + t * g
            new_f = np.clip(new_f, 0.0, 1000.0)
            ns = float(np.sum(new_f))
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_score:
                f = new_f
                best_f = f.copy()
                best_score = sc
                improved = True
                A = build_conv_matrix(f)
                break

        it += 1
        if improved:
            no_improve = 0
            if it % 10 == 0:
                print(f"    {verbose} LP iter={it} score={best_score:.6f}")
        else:
            no_improve += 1
            step *= 0.65
            if no_improve > 20:
                break

    return best_f, best_score


def hybrid_optimize(f_init, deadline, verbose=""):
    """
    Hybrid: gradient + periodic LP kicks to escape plateaus.
    """
    f = f_init.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f_init, float('inf')
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

    # Cycle: 60% gradient, 20% LP, 20% gradient
    cycle_time = 60.0
    grad_frac = 0.4
    lp_frac = 0.35

    while time.monotonic() < deadline:
        cycle_start = time.monotonic()
        remaining = deadline - cycle_start

        # Gradient phase
        grad_deadline = cycle_start + remaining * grad_frac
        if time.monotonic() < grad_deadline:
            bf, bs = gradient_optimize(best_f, grad_deadline,
                                       beta_start=10.0, beta_end=300.0,
                                       lr_start=0.015, verbose=verbose + "G")
            if bs < best_score:
                best_score = bs
                best_f = bf.copy()

        # LP phase
        lp_start = max(grad_deadline, time.monotonic())
        lp_deadline = min(deadline, lp_start + remaining * lp_frac)
        if time.monotonic() < lp_deadline - 2:
            bf, bs = lp_polish(best_f, lp_deadline, max_iters=20,
                               step_init=0.08, verbose=verbose + "L")
            if bs < best_score:
                best_score = bs
                best_f = bf.copy()

    return best_f, best_score


def tukey_window(n, r=0.3):
    w = np.ones(n)
    m = max(1, int(np.floor(r * n / 2)))
    for i in range(m):
        w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
        w[n - 1 - i] = w[i]
    return w


def hann_window(n):
    return 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))


def flat_top_window(n):
    a = [1.0, 1.93, 1.29, 0.388, 0.028]
    x = np.arange(n) / (n - 1) * 2 - 1
    w = a[0] - a[1] * np.cos(np.pi * x) + a[2] * np.cos(2 * np.pi * x) \
        - a[3] * np.cos(3 * np.pi * x) + a[4] * np.cos(4 * np.pi * x)
    w = np.abs(w)
    if w.max() > 1e-10:
        w = w / w.max()
    return w


def cosine_tapered(n, gamma=2.0):
    x = np.arange(n) / (n - 1) * 2 - 1
    w = np.cos(np.pi * x / 2) ** gamma
    return np.clip(w, 0, None)


def super_gaussian(n, alpha=2.0):
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    return np.exp(-((idx - center) / (n / 2.0)) ** (2 * alpha))


def random_smooth(n, rng, seed_offset=0):
    r = np.random.default_rng(seed_offset)
    x = r.standard_normal(n)
    kernel = np.exp(-0.5 * (np.arange(-8, 9) / 2.5) ** 2)
    x = np.convolve(x, kernel, mode='same')
    x = np.abs(x) + 0.05
    return x


def perturb(f, rng, strength=0.04):
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.1, 2.0)
    new_f = np.clip(new_f, 0.0, 1000.0)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S


def propose_candidate(budget_s=1500):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 12.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(7)

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                score = evaluate_sequence(parent.tolist())
                if np.isfinite(score):
                    best_score = score
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    # Phase 1: Quick gradient from parent (120s)
    if best_sequence is not None:
        p1_deadline = t0 + 120
        print(f"\n=== Phase 1: Gradient from parent (n={len(best_sequence)}) ===")
        try:
            bf, bs = gradient_optimize(best_sequence, p1_deadline,
                                       beta_start=5.0, beta_end=150.0,
                                       lr_start=0.02, verbose="[P1]")
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()
                print(f"  P1: {bs:.6f}")
        except Exception as e:
            print(f"  P1 error: {e}")

    # Phase 2: Hybrid optimization on best + diverse inits (500s)
    p2_deadline = t0 + 650
    print(f"\n=== Phase 2: Hybrid + diverse inits ===")

    # First: hybrid on current best
    if best_sequence is not None and time.monotonic() < p2_deadline - 100:
        hybrid_deadline = min(p2_deadline - 100, time.monotonic() + 250)
        try:
            bf, bs = hybrid_optimize(best_sequence, hybrid_deadline, verbose="[P2-best]")
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()
                print(f"  P2 hybrid best: {bs:.6f}")
        except Exception as e:
            print(f"  P2 hybrid error: {e}")

    # Try diverse initializations at key lengths
    target_lengths = [512, 768, 1024]
    if best_sequence is not None:
        target_lengths.insert(0, len(best_sequence))
    # Remove duplicates preserving order
    seen = set()
    target_lengths = [x for x in target_lengths if not (x in seen or seen.add(x))]

    per_length_budget = (p2_deadline - time.monotonic()) / max(len(target_lengths) - 1, 1)

    for n in target_lengths[1:]:  # Skip the first (already done)
        if time.monotonic() > p2_deadline - 20:
            break
        n_deadline = time.monotonic() + per_length_budget
        print(f"\n  --- n={n} (budget {per_length_budget:.0f}s) ---")

        inits = [
            ('tukey30', tukey_window(n, 0.3)),
            ('tukey50', tukey_window(n, 0.5)),
            ('hann', hann_window(n)),
            ('flattop', flat_top_window(n)),
            ('cos2', cosine_tapered(n, 2.0)),
            ('cos3', cosine_tapered(n, 3.0)),
            ('gauss15', super_gaussian(n, 1.5)),
            ('gauss25', super_gaussian(n, 2.5)),
        ]

        # Add parent interpolation if available
        if best_sequence is not None:
            from numpy import interp
            pi = interp(np.linspace(0, 1, n), np.linspace(0, 1, len(best_sequence)), best_sequence)
            pi = np.clip(pi, 0, None)
            if pi.sum() > 0.01:
                inits.insert(0, ('parent_interp', pi))

        n_best_s = float('inf')
        n_best_f = None
        per_init = per_length_budget / max(len(inits), 1) * 0.6

        for label, f_init in inits:
            if time.monotonic() > n_deadline - 3:
                break
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            S = float(np.sum(f_init))
            if S < 0.01:
                continue
            f_init = f_init / S
            sc = evaluate_sequence(f_init.tolist())
            if not np.isfinite(sc):
                continue
            if sc < n_best_s:
                n_best_s = sc
                n_best_f = f_init.copy()

            # Optimize if promising
            if sc < best_score * 1.08:
                i_deadline = min(n_deadline, time.monotonic() + per_init)
                if time.monotonic() < i_deadline - 2:
                    try:
                        bf, bs = gradient_optimize(f_init, i_deadline,
                                                   beta_start=5.0, beta_end=200.0,
                                                   lr_start=0.015, verbose=f"[n={n}]")
                        if bs < n_best_s:
                            n_best_s = bs
                            n_best_f = bf.copy()
                    except Exception:
                        pass

        if n_best_f is not None and n_best_s < best_score:
            best_score = n_best_s
            best_sequence = n_best_f.copy()
            print(f"  n={n} improved to: {best_score:.6f}")

    # Phase 3: Aggressive LP polish (200s)
    if best_sequence is not None and time.monotonic() < deadline - 50:
        print(f"\n=== Phase 3: LP polish (best={best_score:.6f}) ===")
        lp_deadline = min(deadline - 50, time.monotonic() + 180)
        try:
            bf, bs = lp_polish(best_sequence, lp_deadline, max_iters=60,
                               step_init=0.06, verbose="[P3-LP]")
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()
                print(f"  P3 LP: {bs:.6f}")
        except Exception as e:
            print(f"  P3 LP error: {e}")

    # Phase 4: Perturbation restarts + final gradient (remaining time)
    if best_sequence is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Phase 4: Restart + final polish ===")
        remaining = deadline - time.monotonic() - 15
        n_restarts = max(3, int(remaining / 50))
        per_restart = remaining / (n_restarts + 1)

        for r in range(n_restarts):
            if time.monotonic() > deadline - 25:
                break
            strength = 0.015 + 0.015 * r
            f_p = perturb(best_sequence, rng, strength=strength)
            r_deadline = time.monotonic() + per_restart * 0.55

            try:
                bf, bs = gradient_optimize(f_p, r_deadline,
                                           beta_start=50.0, beta_end=350.0,
                                           lr_start=0.01, verbose=f"[R{r}]")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  Restart {r} (str={strength:.3f}): {bs:.6f}")
            except Exception:
                pass

        # Absolute final LP
        if time.monotonic() < deadline - 10:
            try:
                bf, bs = lp_polish(best_sequence, deadline - 5, max_iters=15,
                                   step_init=0.03, verbose="[final-LP]")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
            except Exception:
                pass

        # Final gradient polish
        if time.monotonic() < deadline - 5:
            try:
                bf, bs = gradient_optimize(best_sequence, deadline - 3,
                                           beta_start=200.0, beta_end=800.0,
                                           lr_start=0.003, verbose="[final-G]")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
            except Exception:
                pass

    # Fallback
    if best_sequence is None:
        best_sequence = tukey_window(512, 0.3)
        best_score = evaluate_sequence(best_sequence.tolist())

    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
