"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response iteration with warm starts and adaptive step sizes
2. Multi-start from diverse initializations including parent sequence
3. LSE-smoothed L-BFGS-B refinement for final polish
4. Perturbation-based local search to escape local optima
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_conv_matrix(a):
    """Build matrix A where (A @ x) = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def lp_solves(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns (g, y_dual) or (None, None)."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        # Get dual variables (shadow prices)
        y = res.ineqlin.marginals if hasattr(res, 'ineqlin') else None
        return res.x, y
    return None, None


def lp_iteration(a, max_iters, deadline, verb=False):
    """LP best-response iteration with adaptive step size and backtracking."""
    best_local = a.copy()
    best_score = compute_score(a)
    a = a.copy()
    a = np.clip(a, 1e-8, None)
    a = a / a.sum()
    
    n_stalls = 0
    base_t = 0.6
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g, _ = lp_solves(a, M)
        if g is None:
            break
        g = g / g.sum()
        
        # Adaptive step size with backtracking
        t = base_t
        improved = False
        for attempt in range(5):
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_new = a_new / s
            score = compute_score(a_new)
            if score < best_score - 1e-12:
                best_score = score
                best_local = a_new.copy()
                a = a_new
                improved = True
                n_stalls = 0
                break
            else:
                t *= 0.5
        
        if not improved:
            n_stalls += 1
            if n_stalls >= 20:
                break
        else:
            # Gradually reduce step size
            base_t = max(0.05, base_t * 0.999)
    
    return best_local, best_score


def lse_objective(x, beta, n):
    """Smoothed objective using log-sum-exp."""
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 0.01:
        return 1e10
    x_norm = x_pos / s
    b = fast_conv(x_norm)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    log_sum = b_max + np.log(np.exp(shifted).sum()) / beta
    return float(2.0 * n * log_sum)


def smooth_refine(a, deadline, verb=False):
    """Refine using LSE-smoothed objective with L-BFGS-B and beta annealing."""
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    
    betas = [50, 200, 1000, 5000, 20000]
    for beta in betas:
        if time.monotonic() > deadline:
            break
        x0 = best.copy()
        x0 = np.clip(x0, 1e-6, None)
        x0 = x0 / x0.sum()
        
        try:
            res = minimize(lse_objective, x0, args=(beta, n),
                           method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': 200, 'maxfun': 3000,
                                    'ftol': 1e-18, 'gtol': 1e-16})
            if res.fun < 1e10:
                res_x = np.clip(res.x, 0, None)
                if res_x.sum() > 0.01:
                    score = compute_score(res_x)
                    if score < best_score:
                        best = res_x.copy()
                        best_score = score
                        if verb:
                            print(f"    beta={beta}: J={score:.8f}", flush=True)
        except Exception:
            pass
    
    return best, best_score

        if r.sum() > 0.01:
            inits.append(r / r.sum())
    
    inits.append(np.ones(n) / n)
    
    # Various cosine patterns
    for freq, amp in [(1, 0.2), (1, 0.4), (2, 0.3), (3, 0.2), (4, 0.2)]:
        c = 1.0 + amp * np.cos(2 * np.pi * freq * t)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())
    
    # Sine patterns
    for freq, amp in [(1, 0.3), (2, 0.2)]:
        c = 1.0 + amp * np.sin(2 * np.pi * freq * t)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())
    
    # Quadratic

    # Random

    # Triangular (inverted)
    c = np.abs(t - 0.5) + 0.1
    inits.append(c / c.sum())
    
    return inits

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response with full-step backtracking (primary method)
2. Multi-start from diverse initializations  
3. Structured perturbation (sinusoidal + Gaussian) to escape local optima
4. LSE-smoothed L-BFGS-B for final polish
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_conv_matrix(a):
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def compute_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def lp_solve(a, M):
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None


def lp_best_response_loop(a, max_iters, deadline):
    """LP iteration with full-step backtracking."""
    best = a.copy()
    best_score = compute_score(a)
    a = np.clip(a, 1e-10, None)
    a = a / a.sum()
    
    stall_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M)
        if g is None:
            break
        g = g / g.sum()
        
        # Backtracking line search
        t = 1.0
        found = False
        for _ in range(8):
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_new = a_new / s
            sc = compute_score(a_new)
            if sc < best_score - 1e-14:
                best_score = sc
                best = a_new.copy()
                a = a_new
                found = True
                break
            t *= 0.5
        
        if not found:
            stall_count += 1
            if stall_count > 30:
                break
        else:
            stall_count = 0
    
    return best, best_score


def lse_objective(x, beta, n):
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 0.01:
        return 1e10
    x_norm = x_pos / s
    b = fast_conv(x_norm)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -700.0, 700.0)
    log_sum = b_max + np.log(np.exp(shifted).sum()) / beta
    return float(2.0 * n * log_sum)


def smooth_refine(a, deadline):
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    
    for beta in [30, 100, 500, 2000, 10000]:
        if time.monotonic() > deadline:
            break
        x0 = np.clip(best, 1e-8, None)
        x0 = x0 / x0.sum()
        try:
            res = minimize(lse_objective, x0, args=(beta, n),
                           method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': 300, 'maxfun': 5000,
                                    'ftol': 1e-18, 'gtol': 1e-18})
            if res.fun < 1e10:
                rx = np.clip(res.x, 0, None)
                if rx.sum() > 0.01:
                    sc = compute_score(rx)
                    if sc < best_score:
                        best = rx.copy()
                        best_score = sc
        except Exception:
            pass
    
    return best, best_score


def get_initializations(n, rng, parent):
    inits = []
    t = np.linspace(0, 1, n)
    
    # Parent (resampled if needed)
    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        r = np.maximum(np.interp(t, t_old, parent), 0)
        if r.sum() > 0.01:
            inits.append(r / r.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine patterns
    for freq, amp in [(1, 0.3), (2, 0.3), (3, 0.2), (1, 0.5), (2, 0.5)]:
        c = 1.0 + amp * np.cos(2 * np.pi * freq * t)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())
    
    # Random
    inits.append(rng.uniform(0.2, 1.0, n) / n)
    
    # Inverted parabola
    c = 1.0 - 0.5 * (2 * t - 1) ** 2
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response with full-step backtracking (primary workhorse)
2. Multi-start from diverse initializations (cosine, random, parent)
3. Structured perturbation (Gaussian + sinusoidal) to escape local optima
4. LSE-smoothed L-BFGS-B refinement for final polish
5. Multi-length search (primarily parent length, also tries doubling)
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_conv_matrix(a):
    """Build matrix A where (A @ x) = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None


def lp_loop(a, max_iters, deadline):
    """LP best-response iteration with full-step backtracking."""
    best = a.copy()
    best_score = compute_score(a)
    a = np.clip(a, 1e-10, None)
    a = a / a.sum()

    stall = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_solve(a, M)
        if g is None:
            break
        g = g / g.sum()

        # Backtracking: try full step, halve until improvement
        t = 1.0
        found = False
        for _ in range(10):
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_new = a_new / s
            sc = compute_score(a_new)
            if sc < best_score - 1e-14:
                best_score = sc
                best = a_new.copy()
                a = a_new
                found = True
                break
            t *= 0.5

        if not found:
            stall += 1
            if stall > 50:
                break
        else:
            stall = 0

    return best, best_score


def lse_obj(x, beta, n):
    """Log-sum-exp smoothed objective."""
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 0.01:
        return 1e10
    x_norm = x_pos / s
    b = fast_conv(x_norm)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -700.0, 700.0)
    log_sum = b_max + np.log(np.exp(shifted).sum()) / beta
    return float(2.0 * n * log_sum)


def smooth_refine(a, deadline):
    """Refine using LSE-smoothed objective with beta annealing."""
    n = len(a)
    best = a.copy()

        if time.monotonic() > deadline:
            break

                           bounds=[(0, None)] * n,
                           options={'maxiter': 300, 'maxfun': 5000,
                                    'ftol': 1e-18, 'gtol': 1e-18})
            if res.fun < 1e10:

                    sc = compute_score(rx)
                    if sc < best_score:
                        best = rx.copy()
                        best_score = sc
        except Exception:
            pass

    return best, best_score

    c = 1.0 - 0.4 * (2 * t - 1) ** 2
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())

    return inits
