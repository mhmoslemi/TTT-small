20. Let me also think about **coordinate descent**:
    - For a given coordinate `i`, the objective as a function of `a_i` (with all other `a_j` fixed) is:
      `f(a_i) = 2n * max_k (base_k + 2*a_i * b_k(i) + a_i^2 * c_k(i)) / (S - a_i_old + a_i)^2`
      
      where `base_k` is `conv_k` with the terms involving `a_i` removed, `b_k(i) = a_{k-i}` (if valid), and `c_k(i) = 1` if `k = 2i`.

    - This is a 1D problem that can be solved with golden section search or by finding the critical point analytically (since it's a ratio of a piecewise quadratic to a quadratic).

    - The issue is that the max makes it piecewise, but we can still do projected 1D line search.

"""
Optimized step function sequence search using smooth log-sum-exp gradient descent
with Adam optimizer, adaptive temperature annealing, periodic LP restarts,
and coordinate descent polishing. Targets multiple sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2"""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def smooth_objective(a, tau):
    """Compute tau * log(sum(exp(conv_k / tau))) as smooth approx to max(conv)."""
    conv = np.convolve(a, a)
    max_c = np.max(conv)
    return max_c + tau * np.log(np.sum(np.exp((conv - max_c) / tau)))


def compute_smooth_gradient(a, tau):
    """Compute gradient of smooth objective w.r.t. a."""
    n = len(a)
    conv = np.convolve(a, a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / np.sum(exp_shifted)
    
    # grad_i = sum_j p[i+j] * a[j] + p[2i] * a[i] (with p out of range = 0)
    # Vectorized using matrix
    idx = np.arange(n)
    ij = idx[:, None] + idx[None, :]  # ij[i,j] = i+j
    valid = ij < 2 * n - 1
    M = np.where(valid, p[ij], 0.0)  # M[i,j] = p[i+j] if valid else 0
    grad1 = M @ a
    
    # Term 2: p[2i] * a[i]
    idx_2i = 2 * idx
    valid_2i = idx_2i < 2 * n - 1
    p_2i = np.where(valid_2i, p[idx_2i], 0.0)
    grad2 = p_2i * a
    
    return grad1 + grad2


def adam_nesterov_step(a, m, v, t, lr, beta1, beta2, eps, tau, tau_decay):
    """One step of Adam with Nesterov momentum on the smooth objective."""
    n = len(a)
    
    # Nesterov lookahead
    a_lookahead = a - beta1 * m
    
    # Compute gradient at lookahead point
    grad = compute_smooth_gradient(a_lookahead, tau)
    
    # Normalize gradient (since we care about relative changes)
    # Actually, let's work with the normalized objective:
    # We want to minimize max(conv) / sum(a)^2
    # By scale invariance, fix sum(a) = sqrt(2n) and minimize max(conv)
    # So we can just minimize the smooth approx of max(conv) subject to sum(a) = S
    
    # Update momentum
    m = beta1 * m + (1 - beta1) * grad
    m_hat = m / (1 - beta1 ** t)
    
    v = beta2 * v + (1 - beta2) * grad ** 2
    v_hat = v / (1 - beta2 ** t)
    
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    
    # Enforce non-negativity
    a_new = np.clip(a_new, 0, None)
    
    # Renormalize to maintain sum
    S = np.sqrt(2 * n)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new * (S / s)
    
    return a_new, m, v, t + 1


def lp_direction(a, M_target):
    """Solve LP: max sum(g) s.t. g >= 0, conv(a,g)_k <= M_target."""
    n = len(a)
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success:
            return result.x
    except Exception:
        pass
    return None

But the actual objective is `2n * max(conv) / S^2`. Since we're normalizing `S = sqrt(2n)`, the objective becomes `max(conv)`. So minimizing the smooth approx of `max(conv)` is correct.

The smooth objective is: `F(a) = tau * log(sum_k exp(conv_k(a) / tau))`

"""
Optimized step function sequence search using log-sum-exp smoothed gradient
descent with Adam optimizer and Nesterov momentum, adaptive temperature annealing,
periodic LP-based direction steps, and coordinate descent polishing.
Explores multiple sequence lengths and initializations.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / sum(a)^2"""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def compute_smooth_gradient(a, tau):
    """Compute gradient of tau*log(sum(exp(conv_k/tau))) w.r.t. a."""
    n = len(a)
    conv = np.convolve(a, a)
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / tau
    shifted = np.clip(shifted, -100.0, 100.0)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / np.sum(exp_shifted)

    # grad_i = sum_j p[i+j]*a[j] + p[2i]*a[i]
    idx = np.arange(n)
    ij = idx[:, None] + idx[None, :]
    valid = ij < 2 * n - 1
    M_mat = np.where(valid, p[ij], 0.0)
    grad1 = M_mat @ a

    idx_2i = 2 * idx
    valid_2i = idx_2i < 2 * n - 1
    p_2i = np.where(valid_2i, p[idx_2i], 0.0)
    grad2 = p_2i * a

    return grad1 + grad2


def lp_direction(a, M_target):
    """Solve LP: max sum(g) s.t. g>=0, conv(a,g)_k <= M_target for all k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 20})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def coordinate_descent(a, max_sweeps=3, verbose=""):
    """Coordinate descent polish: try multiple candidate values per coordinate."""
    n = len(a)
    S = float(np.sum(a))
    a = a.copy().astype(np.float64)

    for sweep in range(max_sweeps):
        t_sweep_start = time.monotonic()
        improved_count = 0
        for i in range(n):
            old_val = a[i]
            if old_val < 1e-10:
                continue

            conv = np.convolve(a, a)
            current_max = float(np.max(conv))

            candidates = [0.0, old_val * 0.25, old_val * 0.5, old_val * 0.75,
                          old_val * 1.0, old_val * 1.25, old_val * 1.5, old_val * 2.0]

            max_k = int(np.argmax(conv))
            j_comp = max_k - i
            if 0 <= j_comp < n and j_comp != i:
                candidates.append(a[j_comp])
            j_comp2 = max_k - i
            if 0 <= j_comp2 < n:
                candidates.append(0.5 * (a[i] + a[j_comp2]))

            best_val = old_val
            best_max_conv = current_max

            for new_val in candidates:
                if new_val < -1e-10:
                    continue
                new_val = max(0.0, new_val)
                a[i] = new_val
                new_conv = np.convolve(a, a)
                new_max = float(np.max(new_conv))
                if new_max < best_max_conv - 1e-10:
                    best_max_conv = new_max
                    best_val = new_val

            a[i] = best_val
            if best_val != old_val:
                improved_count += 1

        s = float(np.sum(a))
        if s > 1e-10:
            a = a * (S / s)

        if verbose and sweep % 2 == 0:
            sc = compute_score(a)
            print(f"    {verbose}CD sweep {sweep}: improved {improved_count} coords, "
                  f"score={sc:.6f}, time={time.monotonic()-t_sweep_start:.1f}s")

        if improved_count == 0:
            break

    return a


def main_optimize(a_init, deadline, verbose=""):
    """Full optimization: Adam+Nesterov on smooth objective + periodic LP + CD polish."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    S = float(np.sqrt(2 * n))
    a = a * (S / max(1e-10, float(np.sum(a))))

    best_a = a.copy()
    best_score = compute_score(a)

    # Adam parameters
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 1

    # Temperature annealing
    tau = 2.0 / np.sqrt(n)
    tau_final = 0.05 / np.sqrt(n)
    tau_decay = 0.997

    no_improve = 0
    it = 0
    lp_interval = max(100, n // 2)
    last_lp_time = time.monotonic()
    lp_cd_interval = 10

    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 30 and it > 100:
            break

        # Adam + Nesterov step
        a_look = a - beta1 * m
        a_look = np.clip(a_look, 0, None)
        a_look_s = float(np.sum(a_look))
        if a_look_s > 1e-10:
            a_look = a_look * (S / a_look_s)
        else:
            a_look = a.copy()

        grad = compute_smooth_gradient(a_look, tau)

        m = beta1 * m + (1.0 - beta1) * grad
        m_hat = m / (1.0 - beta1 ** t)
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        v_hat = v / (1.0 - beta2 ** t)

        step = m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - lr * step
        a_new = np.clip(a_new, 0.0, None)
        s_new = float(np.sum(a_new))
        if s_new > 1e-10:
            a_new = a_new * (S / s_new)
        else:
            a_new = a.copy()

        a = a_new
        t += 1
        it += 1

        # Anneal temperature
        tau = max(tau_final, tau * tau_decay)

        # Decay learning rate slowly
        lr = max(0.001, lr * 0.9999)

        # Check improvement periodically
        if it % 50 == 0:
            cur_score = compute_score(a)
            if cur_score < best_score - 1e-10:
                best_score = cur_score
                best_a = a.copy()
                no_improve = 0
                if it % 500 == 0:
                    print(f"  {verbose}iter {it}: score={best_score:.6f} "
                          f"tau={tau:.5f} lr={lr:.5f}")
            else:
                no_improve += 50

            if no_improve > 3000:
                # Try a restart with perturbation
                perturb = a + 0.1 * S / np.sqrt(n) * np.random.randn(n)
                perturb = np.clip(perturb, 0, None)
                ps = float(np.sum(perturb))
                if ps > 1e-10:
                    perturb = perturb * (S / ps)
                    p_score = compute_score(perturb)
                    if p_score < best_score:
                        a = perturb
                        m = np.zeros(n)
                        v = np.zeros(n)
                        t = 1
                        lr = 0.05
                        no_improve = 0
                    else:
                        # Even worse, try LP
                        conv = np.convolve(a, a)
                        M_cur = float(np.max(conv))
                        g = lp_direction(a, M_cur * 1.01)
                        if g is not None:
                            sg = float(np.sum(g))
                            if sg > 1e-10:
                                g = g * (S / sg)
                                for alpha in [0.3, 0.5, 0.7]:
                                    cand = (1 - alpha) * a + alpha * g
                                    cand = np.clip(cand, 0, None)
                                    cs = float(np.sum(cand))
                                    if cs > 1e-10:
                                        cand = cand * (S / cs)
                                        cs_sc = compute_score(cand)
                                        if cs_sc < best_score:
                                            a = cand
                                            best_a = cand.copy()
                                            best_score = cs_sc
                                            m = np.zeros(n)
                                            v = np.zeros(n)
                                            t = 1
                                            lr = 0.05
                                            no_improve = 0
                                            break

            if no_improve > 8000:
                break

        # Periodic LP step
        if it % lp_interval == 0 and it > 200:
            now = time.monotonic()
            if now - last_lp_time > max(10, remaining * 0.05):
                conv = np.convolve(a, a)
                M_cur = float(np.max(conv))
                g = lp_direction(a, M_cur)
                last_lp_time = now
                if g is not None:
                    sg = float(np.sum(g))
                    if sg > 1e-10:
                        g = g * (S / sg)
                        for alpha in [0.1, 0.2, 0.3, 0.5]:
                            cand = (1.0 - alpha) * a + alpha * g
                            cand = np.clip(cand, 0.0, None)
                            cs = float(np.sum(cand))
                            if cs > 1e-10:
                                cand = cand * (S / cs)
                                cs_score = compute_score(cand)
                                if cs_score < best_score:
                                    best_score = cs_score
                                    best_a = cand.copy()
                                    a = cand.copy()
                                    m = np.zeros(n)
                                    v = np.zeros(n)
                                    t = 1
                                    no_improve = 0
                                    break

    # Final coordinate descent polish if time permits
    remaining = deadline - time.monotonic()
    if remaining > 20:
        n_sweeps = min(5, max(1, int(remaining // 30)))
        a_pol = coordinate_descent(best_a, max_sweeps=n_sweeps, verbose=verbose)
        pol_score = compute_score(a_pol)
        if pol_score < best_score:
            best_score = pol_score
            best_a = a_pol
            print(f"  {verbose}CD polish: {best_score:.6f}")

    return best_a, best_score


def make_initializations(n, rng):
    """Create diverse initializations for length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(('uniform', np.ones(n)))

    # Cosine taper
    cos_t = (1.0 + np.cos(np.pi * idx / max(1, n - 1))) / 2.0
    inits.append(('cosine', cos_t))

    # Triangle (peaked center)
    center = (n - 1) / 2.0
    tri = 1.0 - np.abs(idx - center) / center
    tri = np.clip(tri, 0.01, None)
    inits.append(('triangle', tri))

    # Inverse triangle (peaked edges)
    inv_tri = np.abs(idx - center) / center
    inv_tri = np.clip(inv_tri, 0.01, None)
    inits.append(('inv_tri', inv_tri))

    # Random
    r = rng.uniform(0.1, 1.0, size=n)
    inits.append(('random', r))

    # Binary-like alternating
    binary = np.where(np.arange(n) % 2 == 0, 1.0, 0.15)
    inits.append(('binary', binary))

    # Fourier magnitude pattern
    for k in [3, 7, 13]:
        fourier = np.abs(1.0 + np.exp(2j * np.pi * k * idx / n))
        inits.append((f'fourier{k}', fourier))

    # Smoothed random (low-pass)
    base = rng.uniform(0.2, 1.0, size=n)
    # Simple moving average as low-pass
    win = min(15, n // 3)
    if win >= 3:
        kernel = np.ones(win) / win
        smoothed = np.convolve(base, kernel, mode='same')
    else:
        smoothed = base
    inits.append(('smooth_rand', smoothed))

    # Exponential decay from center
    exp_c = np.exp(-1.5 * np.abs(idx - center) / (n * 0.3))
    inits.append(('exp_center', exp_c))

    return inits
