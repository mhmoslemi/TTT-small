def spectral_init(n, seed=0):

    rng = np.random.default_rng(seed)

def build_conv_A(f):
    """Build sparse matrix A where (A@g)[k] = sum_i f[k-i]*g[i] for valid indices."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_min_peak(f, A=None):
    """Solve: min t s.t. A@g - t <= 0, sum(g) = 1, g >= 0.
    Returns (g, t) or (None, None)."""
    n = len(f)
    if A is None:
        A = build_conv_A(f)
    
    # Variables: [g_0, ..., g_{n-1}, t]
    # Objective: minimize t -> c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A@g - t <= 0 for each row
    # A is (2n-1) x n, we need to add -1 for the t column
    A_eq_extra = np.zeros((2 * n - 1, 1))
    A_eq_extra[:, 0] = -1.0
    A_ub = scipy_sparse_hstack(A, A_eq_extra)
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    bounds = [(0, None)] * n + [(None, None)]  # g >= 0, t free
    
    try:

                         bounds=bounds, method='highs')
        if result.success:

            t = result.x[n]
            return g, t

        pass
    return None, None

def build_A(f):
    """Build sparse matrix for conv(f, .)"""
    n = len(f)
    rows, cols, vals = [], [], []
    for k in range(2*n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    return csr_matrix((np.array(vals), (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))), shape=(2*n-1, n))

def lp_min_peak(f):

    n = len(f)

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, csc_matrix, hstack


def fast_score(f):
    """Compute score = 2n * max(conv(f,f)) / sum(f)^2 directly."""
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * n * np.max(conv) / (S * S))


def build_conv_matrix(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            rows[idx] = k
            cols[idx] = i
            vals[idx] = f[k - i]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_min_peak(f):
    """Solve: min t s.t. conv(f,g)_k <= t for all k, sum(g)=1, g>=0.
    Returns optimal g or None."""
    n = len(f)
    A = build_conv_matrix(f)
    # Add column for t: constraint becomes A@g - t <= 0
    t_col = csc_matrix(-np.ones((2 * n - 1, 1)))
    A_full = hstack([A, t_col], format='csc')
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = A_full
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            g = res.x[:n]
            g = np.maximum(g, 0)  # numerical safety
            S = g.sum()
            if S < 1e-10:
                return None
            return g / S
    except Exception:
        pass
    return None


def solve_lp_max_volume(f, M):
    """Solve: max sum(g) s.t. conv(f,g)_k <= M, g>=0."""
    n = len(f)
    A = build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            S = g.sum()
            if S < 1e-10:
                return None
            return g / S
    except Exception:
        pass
    return None


def lp_sca_phase(f, deadline, max_iters=300, method='min_peak'):
    """LP-SCA optimization. method='min_peak' or 'max_volume'."""
    f = f.copy().astype(np.float64)
    f = np.clip(f, 0, 1000)
    S = f.sum()
    if S < 0.01:
        return f, fast_score(f)
    f = f / S
    best_f = f.copy()
    best_score = fast_score(f)
    no_improve = 0
    step_base = 0.15
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        if method == 'min_peak':
            g = solve_lp_min_peak(f)
        else:
            conv = np.convolve(f, f)
            M = float(np.max(conv))
            g = solve_lp_max_volume(f, M)
        
        if g is None:
            no_improve += 1
            if no_improve > 15:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_s = best_score
        alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
        for alpha in alphas:
            trial = (1.0 - alpha) * f + alpha * g
            trial = np.clip(trial, 0.0, 1000.0)
            ts = trial.sum()
            if ts < 1e-10:
                continue
            trial = trial / ts
            s = fast_score(trial)
            if s < best_s:
                best_s = s
                best_alpha = alpha
        
        if best_alpha > 0:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.clip(f, 0.0, 1000.0)
            f = f / f.sum()
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            if it % 20 == 0:
                print(f"    LP-SCA({method}) iter {it}: score={best_score:.6f}, alpha={best_alpha:.3f}")
        else:

            step_base *= 0.8
            if step_base < 0.01 or no_improve > 25:
                break
    
    return best_f, best_score

Let me also add one more technique that's different from the previous code: **iteratively reweighted least squares (IRLS) on the active constraints**.

The idea: at each step, identify which conv[k] values are "active" (close to the max). Then solve a QP that explicitly equalizes these active constraints:

min t
s.t. conv(f_new, f_new)[k] ≈ t for active k
     sum(f_new) = 1
     f_new >= 0
     ||f_new - f|| <= ε (trust region for convexity)

This is a trust-region QP, which can be solved with scipy. But it's complex to set up.

A simpler variant: use the KKT conditions. At the optimum, there exist multipliers λ_k >= 0 for each constraint conv_k <= t such that:
- grad_obj = 2 * (sum_k λ_k * grad(conv_k)) = λ_sum * 1 (gradient of sum constraint)
- complementarity: λ_k * (t - conv_k) = 0

This gives us a system of equations we can solve iteratively:
1. Given current f, compute conv(f,f)
2. Identify active set A = {k : t - conv_k < tolerance}
3. Solve for λ on the active set: the KKT condition becomes
   2 * sum_{k in A} λ_k * (2 * f_shifted_k) = μ * 1
   where f_shifted_k[j] = f[k-j] and μ is the Lagrange multiplier for sum constraint
   Also: sum_{k in A} λ_k * t = μ (from the t constraint)
   
   This is a linear system in (λ, μ)!

"""
Optimization for minimizing 2n*max(conv(f,f))/sum(f)^2 using:
1. Spectral-based initialization (rectified chirps, asymmetric windows)
2. LP-SCA with peak-minimization objective (different from previous max-volume)
3. Softmax gradient with Adam and Nesterov momentum
4. Active-set identification and equalization
5. Multi-length, multi-restart with adaptive time allocation
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, csc_matrix, hstack


def fast_score(f):
    """Compute score = 2n * max(conv(f,f)) / sum(f)^2 directly."""
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * n * np.max(conv) / (S * S))


def build_conv_matrix(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            rows_list.append(k)
            cols_list.append(i)
            vals_list.append(f[k - i])
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    vals = np.array(vals_list, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_min_peak(f):
    """Solve: min t s.t. conv(f,g)_k <= t for all k, sum(g)=1, g>=0."""
    n = len(f)
    A = build_conv_matrix(f)
    t_col = csc_matrix(-np.ones((2 * n - 1, 1)))
    A_full = hstack([A, t_col], format='csc')
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = A_full
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            g = res.x[:n]
            g = np.maximum(g, 0.0)
            S = g.sum()
            if S < 1e-10:
                return None
            return g / S
    except Exception:
        pass
    return None


def solve_lp_max_volume(f, M):
    """Solve: max sum(g) s.t. conv(f,g)_k <= M, g>=0."""
    n = len(f)
    A = build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            g = np.maximum(g, 0.0)
            S = g.sum()
            if S < 1e-10:
                return None
            return g / S
    except Exception:
        pass
    return None


def softmax_gradient_fast(f, beta):
    """Compute gradient of smooth-max approximation to max(conv(f,f))."""
    n = len(f)
    C = np.convolve(f, f)
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros(n), 0.0
    p = exp_C / total
    
    row_idx = np.arange(n)[:, np.newaxis]
    col_idx = np.arange(n)[np.newaxis, :]
    T = p[row_idx + col_idx]
    grad = 2.0 * (T @ f)
    
    soft_val = max_C + np.log(total) / beta
    return grad, soft_val


def lp_sca_phase(f, deadline, max_iters=500, alternate=False):
    """LP-SCA optimization with optional alternating LP types."""
    f = f.copy().astype(np.float64)
    f = np.clip(f, 0, 1000)
    S = f.sum()
    if S < 0.01:
        return f, fast_score(f)
    f = f / S
    best_f = f.copy()
    best_score = fast_score(f)
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        if alternate:
            if it % 3 == 2:
                conv = np.convolve(f, f)
                M = float(np.max(conv))
                g = solve_lp_max_volume(f, M)
            else:
                g = solve_lp_min_peak(f)
        else:
            g = solve_lp_min_peak(f)
        
        if g is None:
            no_improve += 1
            if no_improve > 15:
                break
            continue
        
        best_alpha = 0.0
        best_s = best_score
        alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
        for alpha in alphas:
            trial = (1.0 - alpha) * f + alpha * g
            trial = np.clip(trial, 0.0, 1000.0)
            ts = trial.sum()
            if ts < 1e-10:
                continue
            trial = trial / ts
            s = fast_score(trial)
            if s < best_s:
                best_s = s
                best_alpha = alpha
        
        if best_alpha > 0:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.clip(f, 0.0, 1000.0)
            f = f / f.sum()
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            if it % 30 == 0:
                print(f"    LP-SCA iter {it}: score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 20:
                break
    
    return best_f, best_score
