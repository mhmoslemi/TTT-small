"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A

def solve_lp_g(a, M):
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_alternate(a, max_iter=200, relax=1.0):
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    ema = a.copy()

    for it in range(max_iter):
        conv = np.convolve(a, a)
        M = float(np.max(conv)) * relax
        g = solve_lp_g(a, M)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum

        ema = 0.6 * ema + 0.4 * g

        obj_g = compute_obj(g)
        obj_ema = compute_obj(ema)

        if obj_g < best_obj:
            best_a = g.copy()
            best_obj = obj_g
        if obj_ema < best_obj:
            best_a = ema.copy()
            best_obj = obj_ema

        t = 0.5
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            a = g
        else:
            a = a_new / s

    return best_a, best_obj

def peak_shaving_gradient(a, n_steps=5000, momentum=0.9):
    """Momentum-based gradient descent on the smoothed max of convolution."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s  # normalize to sum=1

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    v = np.zeros(n)  # momentum

    for step in range(n_steps):
        conv = np.convolve(a, a)
        max_conv = np.max(conv)
        
        # Softmax for smooth max
        T = 50.0 + 50.0 * (step / n_steps)  # increase sharpness over time
        scaled = T * (conv - max_conv)  # shift for numerical stability
        exp_scaled = np.exp(scaled)
        softmax_weights = exp_scaled / np.sum(exp_scaled)
        
        # Gradient of smoothed max w.r.t. a
        # d(smoothed_max)/da[j] = sum_k softmax_weights[k] * d(conv[k])/da[j]
        # d(conv[k])/da[j] = a[k-j] if 0 <= k-j < n
        grad = np.zeros(n)
        for j in range(n):
            # conv[k] depends on a[j] when 0 <= k-j < n, i.e., j <= k <= j+n-1
            k_start = j
            k_end = min(2*n-2, j+n-1)
            # d(conv[k])/da[j] = a[k-j]
            # k-j ranges from 0 to n-1
            indices = np.arange(k_start, k_end+1)
            a_indices = indices - j  # these range from 0 to min(n-1, ...)
            a_indices = np.clip(a_indices, 0, n-1)
            grad[j] = np.sum(softmax_weights[k_start:k_end+1] * a[a_indices])
        
        # Full gradient of objective: 2*n*smoothed_max / (sum(a))^2
        # = 2*n * smoothed_max (since sum=1)
        # But we also need to normalize, so let's just use the gradient of smoothed_max
        # and project onto the simplex
        
        # Normalize gradient (it's the gradient of the smoothed max part)
        # The objective is 2*n*smoothed_max (with sum(w)=1 constraint)
        # Gradient on the simplex: project grad to be orthogonal to all-ones
        grad_mean = np.mean(grad)
        grad_centered = grad - grad_mean
        
        # Adaptive step size
        lr = 0.1 / (1.0 + 0.001 * step)
        
        v = momentum * v - lr * grad_centered
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj

def subgradient_polish(a, n_steps=3000):
    """Subgradient method with adaptive tolerance."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    v = np.zeros(n)
    momentum = 0.85

    for step in range(n_steps):
        h = np.convolve(a, a)
        max_h = np.max(h)
        if max_h < 1e-15:
            break

        tol = max_h * (0.0005 + 0.01 * step / n_steps)
        active = h >= max_h - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            active = h >= max_h - 1e-10
            n_active = 1

        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            vals = a[k - j_end:k - j_start + 1][::-1]
            grad[j_start:j_end + 1] += 2.0 * vals
        grad /= n_active
        
        # Center the gradient (for simplex constraint)
        grad = grad - np.mean(grad)
        
        lr = 0.3 / (1.0 + 0.005 * step)
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj

def coord_refine(a, n_rounds=5):
    """Coordinate descent with finer steps."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)

    for rnd in range(n_rounds):
        improved = False
        for j in range(n):
            old_val = a[j]
            best_x = old_val
            best_local = best_obj
            
            # Try multipliers on a log scale
            for mult in [0.0, 0.1, 0.3, 0.5, 0.7, 0.85, 0.95, 1.0, 1.05, 1.1, 1.2, 1.4, 1.7, 2.0, 3.0]:
                x = old_val * mult
                if x < 0:
                    x = 0.0
                a[j] = x
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_x = x
            
            if best_x != old_val:
                a[j] = best_x
                improved = True
            else:
                a[j] = old_val

        if compute_obj(a) < best_obj:
            best_obj = compute_obj(a)
            best_a = a.copy()
        
        if not improved:
            break

    return best_a, best_obj

"""
Multi-phase convex optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Exploits the convexity of the problem (max of quadratics over probability simplex).
Uses smoothed gradient with adaptive temperature, Frank-Wolfe, LP alternation,
momentum subgradient, and coordinate descent in a carefully scheduled pipeline.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A_matrix(a):
    """Build constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_g(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_alternate(a, max_iter=200, relax=1.0):
    """LP alternating optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    ema = a.copy()

    for it in range(max_iter):
        conv = np.convolve(a, a)
        M = float(np.max(conv)) * relax
        g = solve_lp_g(a, M)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum

        ema = 0.6 * ema + 0.4 * g

        obj_g = compute_obj(g)
        obj_ema = compute_obj(ema)

        if obj_g < best_obj:
            best_a = g.copy()
            best_obj = obj_g
        if obj_ema < best_obj:
            best_a = ema.copy()
            best_obj = obj_ema

        t = 0.5
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            a = g
        else:
            a = a_new / s

    return best_a, best_obj


def smoothed_gradient_descent(a, n_steps=5000, T_start=20.0, T_end=200.0):
    """Smoothed gradient descent with adaptive temperature and momentum."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    v = np.zeros(n)
    momentum = 0.92

    for step in range(n_steps):
        frac = step / max(1, n_steps - 1)
        T = T_start + (T_end - T_start) * frac

        conv = np.convolve(a, a)
        max_conv = np.max(conv)

        # Softmax weights
        shifted = T * (conv - max_conv)
        shifted = np.clip(shifted, -500, 500)
        exp_shifted = np.exp(shifted)
        softmax_w = exp_shifted / np.sum(exp_shifted)

        # Gradient: grad[m] = sum_k softmax_w[k] * (a[m] + a[k-m]) for valid k-m
        grad = np.zeros(n)
        for j in range(n):
            k_start = j
            k_end = min(2 * n - 2, j + n - 1)
            a_idx = np.arange(k_start, k_end + 1) - j
            # Clip to valid range (should already be 0..n-1)
            a_idx = np.clip(a_idx, 0, n - 1)
            # d(conv[k])/da[j] = a[j] + a[k-j] but since we sum over k where k-j is valid
            # Actually: d(conv[k])/da[j] = a[k-j] (from first a) + a[j] (from second a)
            # But k-j ranges over valid indices, and a[j] is constant w.r.t. k
            # So: sum_k w[k]*(a[k-j] + a[j]) = sum_k w[k]*a[k-j] + a[j]*sum_k w[k]
            # = sum_k w[k]*a[k-j] + a[j] (since sum of w = 1)
            g1 = np.sum(softmax_w[k_start:k_end + 1] * a[a_idx])
            g2 = a[j] * np.sum(softmax_w[k_start:k_end + 1])
            grad[j] = g1 + g2

        # Center gradient for simplex constraint
        grad = grad - np.mean(grad)

        # Adaptive learning rate
        lr = 0.08 / (1.0 + 0.002 * step)

        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def frank_wolfe_polish(a, n_steps=5000):
    """Frank-Wolfe (conditional gradient) for the convex problem."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for step in range(n_steps):
        conv = np.convolve(a, a)
        max_conv = np.max(conv)

        # Active set with tolerance
        tol = max_conv * (0.001 + 0.01 * step / n_steps)
        active = conv >= max_conv - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            active = conv >= max_conv - 1e-12
            n_active = 1

        active_idx = np.where(active)[0]

        # Subgradient: average of gradients of active components
        grad = np.zeros(n)
        for k in active_idx:
            for j in range(n):
                km = k - j
                if 0 <= km < n:
                    grad[j] += a[j] + a[km]
        grad /= n_active

        # Frank-Wolfe: move towards argmin of gradient
        j_min = int(np.argmin(grad))

        # Step size: standard FW uses t = 2/(step+2), but we can use line search
        t = 2.0 / (step + 2.0)
        # Also try a fixed step
        t_fixed = 0.05 / (1.0 + 0.01 * step)

        # Try both
        a_fw = (1.0 - t) * a
        a_fw[j_min] += t
        a_fw = np.maximum(a_fw, 0.0)
        s_fw = np.sum(a_fw)
        if s_fw > 1e-10:
            a_fw = a_fw / s_fw
            obj_fw = compute_obj(a_fw)

        a_fix = (1.0 - t_fixed) * a
        a_fix[j_min] += t_fixed
        a_fix = np.maximum(a_fix, 0.0)
        s_fix = np.sum(a_fix)
        if s_fix > 1e-10:
            a_fix = a_fix / s_fix
            obj_fix = compute_obj(a_fix)

        if obj_fw <= obj_fix:
            a_new = a_fw
            obj_new = obj_fw
        else:
            a_new = a_fix
            obj_new = obj_fix

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def subgradient_momentum(a, n_steps=5000):
    """Subgradient method with momentum and adaptive tolerance."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    v = np.zeros(n)
    momentum = 0.88

    for step in range(n_steps):
        h = np.convolve(a, a)
        max_h = np.max(h)

        tol = max_h * (0.0005 + 0.005 * step / n_steps)
        active = h >= max_h - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            active = h >= max_h - 1e-12
            n_active = 1

        active_idx = np.where(active)[0]
        grad = np.zeros(n)
        for k in active_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            vals = a[k - j_end:k - j_start + 1][::-1]
            # d(conv[k])/da[j] for j in [j_start, j_end] is a[j] + a[k-j]
            # a[k-j] for j from j_start to j_end is a[k-j_end] to a[k-j_start] (reversed)
            # = vals
            # a[j] for j from j_start to j_end is a[j_start:j_end+1]
            grad[j_start:j_end + 1] += (a[j_start:j_end + 1] + vals)
        grad /= n_active

        grad = grad - np.mean(grad)

        lr = 0.06 / (1.0 + 0.003 * step)
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def coord_descent(a, n_rounds=5):
    """Coordinate descent with multiple trial values per coordinate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)

    multipliers = [0.0, 0.1, 0.3, 0.5, 0.7, 0.85, 0.95, 1.0, 1.05, 1.1, 1.3, 1.5, 2.0, 3.0]

    for rnd in range(n_rounds):
        improved = False
        for j in range(n):
            old_val = a[j]
            best_x = old_val
            best_local = best_obj

            for mult in multipliers:
                x = old_val * mult
                if x < 0:
                    x = 0.0
                a[j] = x
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_x = x

            if best_x != old_val:
                a[j] = best_x
                improved = True
            else:
                a[j] = old_val

        cur_obj = compute_obj(a)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        if not improved:
            break

    return best_a, best_obj


def generate_init(n, rng, method=0):
    """Generate diverse initializations."""
    if method == 0:
        # Palindromic random
        half = (n + 1) // 2
        x = rng.random(half) + 0.05
        if n % 2 == 0:
            a = np.concatenate([x[:half], x[:half][::-1]])
        else:
            a = np.concatenate([x, x[-2::-1]])
    elif method == 1:
        # Cosine
        freq = rng.uniform(0.3, 2.5)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0, np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
    elif method == 2:
        # Gaussian
        sigma = rng.uniform(0.1, 0.4)
        t = np.arange(n) / n - 0.5
        a = np.exp(-t**2 / (2 * sigma**2))
    elif method == 3:
        # Two-lobe
        mid = n // 2
        a = np.ones(n)
        a[:mid] = rng.uniform(0.3, 1.2)
        a[mid:] = rng.uniform(0.3, 1.2)
    elif method == 4:
        # Smoothed random
        a = rng.random(n)
        k = np.ones(5) / 5
        a = np.convolve(a, k, mode='same')
    elif method == 5:
        # Quartic window
        t = np.arange(n) / n - 0.5
        a = np.maximum(0, 1.0 - (2.0 * t)**4)
    elif method == 6:
        # Periodic
        period = int(rng.integers(max(2, n // 15), max(3, n // 5)))
        pattern = np.maximum(0, rng.random(period) - 0.1)
        a = np.tile(pattern, n // period + 1)[:n]
    elif method == 7:
        # Random
        a = rng.random(n)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s


def optimize_single(a_init, time_limit):
    """Run full optimization pipeline on a single initialization."""
    start = time.monotonic()
    a = a_init.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)

    # Phase 1: Smoothed gradient (quick progress)
    elapsed = time.monotonic() - start
    if elapsed < time_limit * 0.3:
        n_steps = int(min(3000, (time_limit * 0.3 - elapsed) * 20))
        n_steps = max(n_steps, 1000)
        a, obj = smoothed_gradient_descent(a, n_steps=n_steps)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    # Phase 2: LP alternating
    elapsed = time.monotonic() - start
    if elapsed < time_limit * 0.5:
        max_iter = int(min(80, (time_limit * 0.5 - elapsed) / 0.3))
        max_iter = max(max_iter, 20)
        a, obj = lp_alternate(a, max_iter=max_iter)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    # Phase 3: Subgradient with momentum (polishing)
    elapsed = time.monotonic() - start
    if elapsed < time_limit * 0.75:
        n_steps = int(min(3000, (time_limit * 0.75 - elapsed) * 25))
        n_steps = max(n_steps, 1000)
        a, obj = subgradient_momentum(best_a, n_steps=n_steps)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    # Phase 4: Frank-Wolfe polish
    elapsed = time.monotonic() - start
    if elapsed < time_limit * 0.9:
        n_steps = int(min(3000, (time_limit * 0.9 - elapsed) * 40))
        n_steps = max(n_steps, 500)
        a, obj = frank_wolfe_polish(best_a, n_steps=n_steps)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    # Phase 5: Final subgradient
    elapsed = time.monotonic() - start
    if elapsed < time_limit * 0.97:
        n_steps = int(min(2000, (time_limit * 0.97 - elapsed) * 20))
        n_steps = max(n_steps, 500)
        a, obj = subgradient_momentum(best_a, n_steps=n_steps)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining smoothed gradient, Frank-Wolfe, LP alternation,
    momentum subgradient, and coordinate descent with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent_obj = np.inf
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent_obj = compute_obj(parent)
            if parent_obj < best_score:
                best_score = parent_obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={parent_obj:.6f}")
    except Exception:
        parent = None

    # Define search schedule: (n, time_fraction, n_restarts)
    # Focus more time on n values near the best
    n_candidates = [200, 250, 300, 350, 400]
    # Allocate time proportionally, giving more to n=300
    time_weights = [0.12, 0.15, 0.30, 0.20, 0.23]

    total_time = deadline - time.monotonic()
    if total_time < 60:
        total_time = 60

    for idx, n in enumerate(n_candidates):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        alloc = min(remaining, total_time * time_weights[idx])
        alloc = max(alloc, 30)
        n_restarts = max(1, int(alloc / 40))
        n_restarts = min(n_restarts, 5)
        per_restart = alloc / n_restarts

        print(f"N={n}: alloc={alloc:.0f}s, restarts={n_restarts}, best={best_score:.6f}")

        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break
            phase_deadline = time.monotonic() + per_restart

            # Choose initialization
            if r == 0 and best_seq is not None and len(best_seq) == n:
                a_init = best_seq.copy()
            else:
                method = (r + idx) % 8
                a_init = generate_init(n, rng, method)

            # Run optimization pipeline
            a_best, obj = optimize_single(a_init, per_restart * 0.85)

            if obj < best_score:
                best_score = obj
                best_seq = a_best.copy()
                print(f"  n={n} r={r} m={method}: obj={obj:.6f}")

    # Final intensive polishing on the best solution
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\nFinal polishing: n={n_best}, obj={best_score:.6f}, {remaining:.0f}s left")

        # Extended subgradient
        a, obj = subgradient_momentum(best_seq, n_steps=int(min(5000, remaining * 15)))
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  SG polish: obj={best_score:.6f}")

        # Smoothed gradient with high T
        if time.monotonic() < deadline - 10:
            a, obj = smoothed_gradient_descent(best_seq, n_steps=int(min(3000, (deadline - time.monotonic()) * 15)),
                                               T_start=50.0, T_end=300.0)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  SG high-T: obj={best_score:.6f}")

        # Coordinate descent
        if time.monotonic() < deadline - 10 and n_best <= 400:
            rounds = int(min(4, (deadline - time.monotonic()) / 15.0))
            if rounds >= 1:
                a, obj = coord_descent(best_seq, n_rounds=rounds)
                if obj < best_score:
                    best_score = obj
                    best_seq = a.copy()
                    print(f"  Coord: obj={best_score:.6f}")

        # LP on best
        if time.monotonic() < deadline - 15 and n_best <= 350:
            max_iter = int(min(80, (deadline - time.monotonic()) / 0.4))
            if max_iter > 15:
                a, obj = lp_alternate(best_seq, max_iter=max_iter)
                if obj < best_score:
                    best_score = obj
                    best_seq = a.copy()
                    print(f"  Final LP: obj={best_score:.6f}")

        # Another round of subgradient after LP
        if time.monotonic() < deadline - 5:
            a, obj = subgradient_momentum(best_seq, n_steps=int(min(2000, (deadline - time.monotonic()) * 20)))
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  Final SG2: obj={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
