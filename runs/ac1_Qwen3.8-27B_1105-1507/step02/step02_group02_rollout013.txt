    while s < 2 * n - 1:

    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

    if s < 0.01:

    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))

def build_conv_matrix(a):
    """Build matrix A where A @ x = convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):

        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh+1] = a[k-jh:k-jl+1][::-1]
    return A

def lp_iteration(a, max_iters, deadline, base_t=0.5):

def peak_targeted_perturbation(a, n_peaks=5, scale=0.1, rng=None):
    """Create a perturbation targeting the highest peaks in the convolution."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    
    # Find lags near the maximum
    threshold = b_max * 0.95
    active_lags = np.where(b >= threshold)[0]
    
    # For each active lag, find the top contributing index pairs
    # and create a perturbation that reduces those products
    d = np.zeros(n)
    for k in active_lags[:n_peaks]:
        # The gradient of (a*a)_k w.r.t. a_j is a_{k-j} (for valid indices)
        # We want to reduce (a*a)_k, so we want to follow -gradient
        # gradient_k[j] = a[k-j] if 0 <= k-j < n, else 0
        grad_k = np.zeros(n)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                grad_k[j] = a[idx]
        # Scale the perturbation
        d -= scale * grad_k / (np.linalg.norm(grad_k) + 1e-10)
    
    # Also add some random noise for diversity
    d += scale * 0.1 * rng.normal(0, 1, n)
    
    # Apply perturbation and normalize
    a_pert = a + d
    a_pert = np.clip(a_pert, 0, None)
    s = a_pert.sum()
    if s < 0.01:
        return a.copy()
    return a_pert / s

**Alternative: Alternating Direction Method of Multipliers (ADMM)**

We can write the problem as:
$$\min_a \max_k (a*a)_k \quad \text{s.t.} \quad a \geq 0, \sum a = 1$$

Introduce a variable $t \geq (a*a)_k$ for all $k$:
$$\min_{a,t} t \quad \text{s.t.} \quad (a*a)_k \leq t \; \forall k, \; a \geq 0, \sum a = 1$$

This is a non-convex problem (the constraints $(a*a)_k \leq t$ are quadratic). But we can use ADMM:
- Split: $\min_{a,t} t + \frac{\rho}{2} \sum_k \max(0, (a*a)_k - t)^2$
- Alternate:
  - Update $a$: $\min_a \frac{\rho}{2} \sum_k \max(0, (a*a)_k - t)^2 + \text{regularization}$ s.t. $a \geq 0, \sum a = 1$
  - Update $t$: $t = \max_k (a*a)_k$ (or a smoothed version)

The $a$-update is still non-convex. But we can linearize:
- At the current $a$, compute the gradient of $\sum_k (a*a)_k^2$ (or the max)
- Take a gradient step with projection onto the simplex

This is essentially what the smooth refinement does. So I'll stick with the smooth refinement approach.

**One more idea: Use the structure of the convolution matrix.**

6. Final precision polish with progressive beta annealing

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2 directly."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build Toeplitz-like matrix A where (A @ x) = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M, partial_lags=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M (optionally only on subset of lags)."""
    n = len(a)
    A = build_conv_matrix(a)
    if partial_lags is not None:
        A_sub = A[partial_lags, :]
        b_ub = np.full(len(partial_lags), M)
    else:
        A_sub = A
        b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    res = linprog(c, A_ub=A_sub, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None


def smooth_objective(x, beta, n):
    """Log-sum-exp smoothed objective: approximates 2n*max(conv(a,a)) for normalized a."""
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 0.01:
        return 1e10
    x_norm = x_pos / s
    b = fast_conv(x_norm)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -60.0, 60.0)
    log_sum = b_max * beta + np.log(np.exp(shifted).sum())
    return float(2.0 * n * log_sum)

        x0 = np.clip(x0, 1e-8, None)

        b_max_val = b.max()

        if g is None:
            break
        g = g / g.sum()

        # Adaptive step size
        t = base_t * max(0.1, 1.0 - 0.3 * it / max_iters)
        t = min(t, 0.9)

        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        score = compute_score(a_new)

        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            a = a_new
            no_improve_count = 0
        else:
            # Try smaller step
            t2 = t * 0.3
            a_new2 = (1.0 - t2) * a + t2 * g
            a_new2 = np.clip(a_new2, 0, None)
            s2 = a_new2.sum()
            if s2 > 1e-10:
                a_new2 = a_new2 / s2
                score2 = compute_score(a_new2)
                if score2 < best_score:
                    best_score = score2
                    best_local = a_new2.copy()
                    no_improve_count = 0
                a = a_new2
            else:
                a = a_new
            no_improve_count += 1
            if no_improve_count > 30:
                break

    return best_local, best_score


def coordinate_descent_polish(a, n_sweeps=10, deadline=None):
    """Coordinate descent with uniform redistribution for final polish."""
    best = a.copy()
    best_score = compute_score(a)
    n = len(a)

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            # Try scaling a_j by various factors
            old_val = best[j]
            if old_val < 1e-12:
                continue
            for factor in [0.85, 0.92, 0.96, 1.04, 1.08, 1.15]:
                new_val = old_val * factor
                if new_val < 0:
                    continue
                # Redistribute sum change uniformly to other coordinates
                delta = new_val - old_val
                a_try = best.copy()
                a_try[j] = new_val
                # Distribute -delta to others
                other_idx = np.arange(n) != j
                a_try[other_idx] -= delta / (n - 1)
                a_try = np.clip(a_try, 0, None)
                s = a_try.sum()
                if s < 0.01:
                    continue
                a_try = a_try / s
                score = compute_score(a_try)
                if score < best_score - 1e-12:
                    best = a_try.copy()
                    best_score = score
                    old_val = new_val
                    improved = True
                    break
        if not improved:
            break

    return best, best_score


def peak_targeted_perturbation(a, n_peaks=5, scale=0.1, rng=None):
    """Create a perturbation targeting the highest peaks in the convolution."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()

    # Find lags near the maximum
    threshold = b_max * 0.9
    active_lags = np.where(b >= threshold)[0]

    # For each active lag, the gradient of (a*a)_k w.r.t. a is a reversed/shifted copy of a
    d = np.zeros(n)
    for k in active_lags[:n_peaks]:
        grad_k = np.zeros(n)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                grad_k[j] = a[idx]
        norm = np.linalg.norm(grad_k)
        if norm > 1e-12:
            d -= scale * grad_k / norm

    # Add some random noise for diversity
    d += scale * 0.15 * rng.normal(0, 1, n)

    a_pert = a + d
    a_pert = np.clip(a_pert, 0, None)
    s = a_pert.sum()
    if s < 0.01:
        return a.copy()
    return a_pert / s


def resample_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    t_old = np.linspace(0, 1, len(a))
    t_new = np.linspace(0, 1, new_n)
    a_new = np.interp(t_new, t_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 0.01:
        a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
    1. LP best-response (full and partial constraint) with adaptive step sizes
    2. High-precision smooth log-sum-exp refinement with L-BFGS-B
    3. Coordinate descent polish
    4. Peak-targeted perturbation with re-optimization
    5. Multi-length exploration
    6. Final precision polish with progressive beta annealing
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 25.0

    best = None
    best_score = float('inf')

    # Try parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.clip(p, 0.0, 1000.0)
            sc = compute_score(p)
            if sc < best_score:
                best = p.copy()
                best_score = sc
                print(f"Parent: J={sc:.8f} (n={p.size})", flush=True)
    except (NameError, AttributeError):
        pass

    if best is None:
        best = np.ones(512) / 512
        best_score = compute_score(best)
        print(f"Init uniform: J={best_score:.8f} (n=512)", flush=True)

    n = best.size
    print(f"\nStarting search: n={n}, J={best_score:.8f}", flush=True)
    print(f"Budget: {total_budget}s, deadline at {deadline - time.monotonic():.1f}s remaining", flush=True)

    # === PHASE 1: LP polish (12% of budget) ===
    lp_time = total_budget * 0.12
    lp_deadline = time.monotonic() + lp_time
    print(f"\n=== Phase 1: LP Polish (budget={lp_time:.0f}s) ===", flush=True)

    a = best.copy() / best.sum()
    max_iters = max(200, int(lp_time / 0.04))
    a_result, score = lp_iteration(a, max_iters, lp_deadline)
    if score < best_score:
        best = a_result.copy()
        best_score = score
        print(f"  LP: J={best_score:.8f}", flush=True)
    else:
        print(f"  LP: no improvement (J={best_score:.8f})", flush=True)

    # Quick coordinate descent after LP
    if time.monotonic() < deadline:
        a_result, score = coordinate_descent_polish(best, n_sweeps=5,
                                                     deadline=min(lp_deadline + 10, deadline))
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  CoordDesc: J={best_score:.8f}", flush=True)

    # === PHASE 2: Main smooth refinement (35% of budget) ===
    smooth_time = total_budget * 0.35
    smooth_deadline = time.monotonic() + smooth_time
    print(f"\n=== Phase 2: Smooth Refinement (budget={smooth_time:.0f}s) ===", flush=True)

    a_result, score = smooth_refine(best, smooth_deadline,
                                     min_beta=10, max_beta=100000)
    if score < best_score:
        best = a_result.copy()
        best_score = score
        print(f"  Smooth: J={best_score:.8f}", flush=True)

    # Coordinate descent polish
    if time.monotonic() < deadline:
        a_result, score = coordinate_descent_polish(best, n_sweeps=15,
                                                     deadline=min(smooth_deadline + 15, deadline))
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  CoordDesc: J={best_score:.8f}", flush=True)

    # === PHASE 3: Perturbation search (25% of budget) ===
    pert_time = total_budget * 0.25
    pert_deadline = time.monotonic() + pert_time
    print(f"\n=== Phase 3: Perturbation Search (budget={pert_time:.0f}s) ===", flush=True)

    n = best.size
    n_tries = max(5, int(pert_time / 20))
    n_tries = min(n_tries, 40)

    for trial in range(n_tries):
        if time.monotonic() > pert_deadline:
            break

        # Various perturbation strategies
        strategy = trial % 4
        scale = 0.03 + 0.02 * (trial % 5)

        if strategy == 0:
            # Peak-targeted
            a_pert = peak_targeted_perturbation(best, n_peaks=3, scale=scale, rng=rng)
        elif strategy == 1:
            # Random noise
            noise = rng.normal(0, scale, n)
            a_pert = best + noise
            a_pert = np.clip(a_pert, 0, None)
            a_pert = a_pert / a_pert.sum()
        elif strategy == 2:
            # Localized perturbation (gaussian bump)
            center = rng.integers(0, n)
            width = max(3, n // 20)
            t = np.arange(n)
            bump = np.exp(-0.5 * ((t - center) / width) ** 2)
            a_pert = best + scale * bump * rng.normal(1, 0.3, n)
            a_pert = np.clip(a_pert, 0, None)
            a_pert = a_pert / a_pert.sum()
        else:
            # Anti-symmetric perturbation
            half = n // 2
            pert = rng.normal(0, scale, half)
            if n % 2 == 0:
                full_pert = np.concatenate([pert, -pert[::-1]])
            else:
                full_pert = np.concatenate([pert, [0.0], -pert[::-1]])
            a_pert = best + full_pert[:n]
            a_pert = np.clip(a_pert, 0, None)
            s = a_pert.sum()
            if s < 0.01:
                continue
            a_pert = a_pert / s

        # Quick smooth refinement from perturbed point
        remaining = pert_deadline - time.monotonic()
        sub_time = min(remaining * 0.6, 18.0)
        sub_deadline = time.monotonic() + sub_time

        a_result, score = smooth_refine(a_pert, sub_deadline,
                                         min_beta=200, max_beta=20000, verbose=False)
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  Pert {trial} (strat={strategy}, scale={scale:.3f}): J={best_score:.8f}",
                  flush=True)

    # === PHASE 4: Multi-length exploration (15% of budget) ===
    multi_time = total_budget * 0.15
    multi_deadline = time.monotonic() + multi_time
    print(f"\n=== Phase 4: Multi-Length (budget={multi_time:.0f}s) ===", flush=True)

    n_base = best.size
    # Try lengths around the current one
    lengths_to_try = sorted([l for l in [256, 320, 384, 448, 512, 576, 640, 768, 896, 1024]
                             if l != n_base and 128 <= l <= 1024],
                            key=lambda x: abs(x - n_base))

    for n2 in lengths_to_try[:5]:
        if time.monotonic() > multi_deadline:
            break
        if multi_deadline - time.monotonic() < 20:
            break

        a2 = resample_sequence(best, n2)
        if a2 is None or a2.sum() < 0.01:
            continue

        remaining = multi_deadline - time.monotonic()
        sub_deadline = time.monotonic() + min(remaining * 0.5, 25.0)
        a_result, score = smooth_refine(a2, sub_deadline,
                                         min_beta=100, max_beta=50000, verbose=False)
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  Multi n={n2}: J={best_score:.8f}", flush=True)

    # === PHASE 5: Final precision polish (remaining time) ===
    print(f"\n=== Phase 5: Final Polish ===", flush=True)
    if time.monotonic() < deadline:
        # High-precision smooth refinement
        a_result, score = smooth_refine(best, deadline - 10,
                                         min_beta=5000, max_beta=200000)
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  Final smooth: J={best_score:.8f}", flush=True)

        # Final coordinate descent
        if time.monotonic() < deadline - 5:
            a_result, score = coordinate_descent_polish(best, n_sweeps=20,
                                                         deadline=deadline - 5)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  Final coord: J={best_score:.8f}", flush=True)

        # One more LP pass
        if time.monotonic() < deadline - 15:
            remaining = deadline - time.monotonic() - 10
            max_iters = max(50, int(remaining / 0.04))
            a_result, score = lp_iteration(best, max_iters, deadline - 10)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  Final LP: J={best_score:.8f}", flush=True)

            # Coordinate descent after LP
            if time.monotonic() < deadline - 3:
                a_result, score = coordinate_descent_polish(best, n_sweeps=10,
                                                             deadline=deadline - 3)
                if score < best_score:
                    best = a_result.copy()
                    best_score = score
                    print(f"  Final coord2: J={best_score:.8f}", flush=True)

    # Final validation
    if best is None:
        best = np.ones(256) / 256
        best_score = compute_score(best)

    # Make sure we return valid values
    best = np.clip(best, 0.0, 1000.0)
    final_score = float(evaluate_sequence([float(x) for x in best]))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: J={best_score:.8f}, n={best.size}", flush=True)
    return [float(x) for x in best]
