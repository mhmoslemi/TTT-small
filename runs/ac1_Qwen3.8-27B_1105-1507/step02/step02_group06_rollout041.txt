def get_ns(L, rng):
    """Generate non-sequential indices [0..L-1] via xorshift."""
    idx = np.arange(L, dtype=np.int64)
    s = np.array([0, 1, 5], dtype=np.int64)
    for _ in range(4):
        s = (s ^ (s << 13)) & 0x7FFFFFFFFFFFFFFF
        s = (s ^ (s >> 7)) & 0x7FFFFFFFFFFFFFFF
        s = (s ^ (s << 17)) & 0x7FFFFFFFFFFFFFFF
    s = s % L
    idx[s] = np.arange(L)
    return idx


def make_xor_init(n, rng):
    """Create structured initialization from xorshift permutation of bits."""
    ns = get_ns(n, rng)
    L = ns[:-2]
    a = np.zeros(n, dtype=np.float64)
    sign = np.where((L ^ (L >> 1)) & 1, 1.0, -1.0)
    for t in range(len(sign) // 2 + 1):
        if 2 * t < len(sign) and 2 * t + 1 < len(sign):
            a[ns[2 * t]] = sign[2 * t]
            a[ns[2 * t + 1]] = sign[2 * t + 1]
        elif 2 * t < len(sign):
            a[ns[2 * t]] = sign[2 * t]
    a = np.abs(a)
    if a.max() > 0:
        a = a / a.max()
    return a


def make_lfsr_init(n, rng):
    """Generate an LFSR (m-sequence like) binary pattern."""
    # Use a simple LFSR with a maximal-length tap
    # For n up to 4096, use a 12-bit or 13-bit LFSR
    bits = np.zeros(n, dtype=np.float64)
    state = rng.integers(1, 256)  # 8-bit state
    for i in range(n):
        bits[i] = state & 1
        new_bit = ((state >> 0) ^ (state >> 3) ^ (state >> 5) ^ (state >> 7)) & 1
        state = ((state >> 1) | (new_bit << 7)) & 0xFF
    return bits


def make_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n, dtype=np.float64))

    # Cosine taper
    cos_t = 0.5 * (1.0 + np.cos(np.pi * idx / max(n - 1, 1)))
    inits.append(cos_t)

    # XOR-based
    inits.append(make_xor_init(n, rng))

    inits.append(make_lfsr_init(n, rng))

    # Random binary
    inits.append(rng.random(n).copy())
    inits[-1][inits[-1] < 0.5] = 0

    # Sine pattern
    inits.append(0.5 + 0.5 * np.sin(2 * np.pi * idx / n * 3))

    # Two-scale pattern

    period = max(n // 7, 1)

    inits.append(1.0 - np.exp(-0.5 * ((idx - center) / sigma) ** 2))

    y = a.copy()
    v = np.zeros(n, dtype=np.float64)
    t_val = 1.0
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        if no_improve > 300:
            break

        # Compute convolution at y (Nesterov lookahead)
        C_y = np.convolve(y, y)
        M_y = float(np.max(C_y))
        S_y = float(np.sum(y))

        # Compute top-k subgradient
        top_indices = np.argsort(C_y)[-top_k:]
        mean_C = (S_y * S_y) / (2 * n - 1)
        weights = np.maximum(C_y[top_indices] - mean_C, 0.0)
        if weights.sum() < 1e-15:
            weights = np.ones(top_k) / top_k

        # Build subgradient
        g = np.zeros(n, dtype=np.float64)
        for k, w in zip(top_indices, weights):
            k_int = int(k)
            # g[p] += w * 2 * a_y[k-p] for valid k-p
            lo = 0
            hi = n
            k_lo = max(0, k_int - (n - 1))  # smallest p such that k-p >= 0
            k_hi = min(n - 1, k_int)  # largest p such that k-p < n
            for p in range(k_lo, k_hi + 1):
                g[p] += w * 2.0 * y[k_int - p]

        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        # Nesterov update
        mu = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        t_val += 1.0

        v = (t_val - 1.0) / (t_val + mu) * v + lr * g
        y = a - v

        # Project: non-negative, sum = 1
        y = np.maximum(y, 0.0)
        S_y = float(np.sum(y))
        if S_y < 1e-15:
            y = np.ones(n) / n
            S_y = 1.0
        else:
            y = y / S_y

        a = y

        # Evaluate
        score = compute_score(a)
        it += 1

        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if verbose_prefix and it % 500 == 0:
                print(f"    {verbose_prefix}nesterov iter {it}: score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.9

        # Decay learning rate
        lr = lr_init / (1.0 + 0.01 * it)

    return best_a, best_score


def lp_optimize(f, deadline, step_init=0.08):
    """LP-based direction finding (from previous approach, kept as backup)."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    step = step_init
    no_improve = 0
    it = 0

    # Pre-build sparse conv matrix
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    A = csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))

    while time.monotonic() < deadline:
        if no_improve > 40:
            break

        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        try:
            result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if not result.success:
                no_improve += 1
                continue
            g = result.x
        except Exception:
            no_improve += 1
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-15:
            no_improve += 1
            continue

        g = g * (S / sum_g)

        best_step_score = best_score
        best_step_f = None
        for alpha in [step, step * 0.5, step * 1.5]:
            alpha = min(alpha, 1.0)
            if alpha < 0.005:
                continue
            candidate = (1.0 - alpha) * f + alpha * g
            candidate = np.clip(candidate, 0.0, None)
            cand_sum = float(np.sum(candidate))
            if cand_sum < 1e-15:
                continue
            candidate = candidate * (S / cand_sum)
            cand_score = compute_score(candidate)
            if cand_score < best_step_score:
                best_step_score = cand_score
                best_step_f = candidate

        it += 1
        if best_step_f is not None:
            f = best_step_f
            best_f = f.copy()
            best_score = best_step_score
            no_improve = 0
            step = max(step * 0.995, 0.01)
            # Update sparse matrix
            for k in range(2 * n - 1):
                j_min = max(0, k - (n - 1))
                j_max = min(k, n - 1)
                for j in range(j_min, j_max + 1):
                    pass  # Rebuilding sparse matrix each iter is expensive
            # Skip matrix update for simplicity (use same structure)
        else:
            no_improve += 1
            step *= 0.7
            if step < 0.005:
                break

    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    """
    Main entry point: hybrid binary search + continuous optimization.
    """
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(42)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            score = evaluate_sequence(parent.tolist())
            if np.isfinite(score):
                best_score = score
                best_sequence = parent.copy()
                print(f"Parent n={parent.size}: score={score:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # Decide lengths to try
    parent_n = len(best_sequence) if best_sequence is not None else 1024
    lengths = sorted(set([parent_n, max(512, parent_n // 2),
                          min(4096, parent_n * 2)]))

    print(f"\nLengths to try: {lengths}")
    print(f"Time budget: {budget_s}s")

    # Phase 1: Quick exploration (first 15% of time)
    print("\n=== Phase 1: Quick Exploration ===")
    explore_deadline = t0 + (deadline - t0) * 0.15
    candidates = []  # (score, sequence, n)

    for n in lengths:
        if time.monotonic() > explore_deadline - 20:
            break
        print(f"\n--- n={n} ---")

        inits = make_structured_inits(n, rng)

        # Add parent interpolation if different length
        if best_sequence is not None:
            if len(best_sequence) == n:
                inits.insert(0, best_sequence.copy())
            else:
                x_old = np.linspace(0, 1, len(best_sequence))
                x_new = np.linspace(0, 1, n)
                parent_interp = np.interp(x_new, x_old, best_sequence)
                inits.insert(0, parent_interp)

        for init_idx, f_init in enumerate(inits):
            if time.monotonic() > explore_deadline - 5:
                break
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            if np.sum(f_init) < 1e-10:
                continue

            score = compute_score(f_init)
            if not np.isfinite(score):
                continue

            if score < best_score:
                best_score = score
                best_sequence = f_init.copy()

            # Quick Nesterov (5 seconds)
            quick_deadline = min(explore_deadline, time.monotonic() + 5.0)
            f_opt, s_opt = nesterov_optimize(f_init, quick_deadline,
                                             lr_init=0.008, top_k=3)
            if s_opt < score:
                score = s_opt

            if score < best_score:
                best_score = score
                best_sequence = f_opt.copy()

            candidates.append((score, f_opt.copy(), n))
            print(f"  init {init_idx}: {s_opt:.6f}")

    candidates.sort(key=lambda x: x[0])
    print(f"\nAfter exploration: best={best_score:.6f}")
    for sc, seq, nl in candidates[:5]:
        print(f"  n={nl}: {sc:.6f}")

    # Phase 2: Binary local search (next 40% of time)
    print("\n=== Phase 2: Binary Local Search ===")
    binary_deadline = t0 + (deadline - t0) * 0.55

    # Take top candidates and round to binary
    seen_n = set()
    binary_seeds = []
    for sc, seq, nl in candidates:
        if nl not in seen_n:
            # Round to binary with threshold
            a_bin = (seq > np.median(seq[seq > 0]) if np.sum(seq > 0) > 0
                     else np.ones(nl) * 0.5).astype(np.float64)
            a_bin = (a_bin > 0.5).astype(np.float64)
            if np.sum(a_bin) > 2 and np.sum(a_bin) < nl - 2:
                binary_seeds.append((sc, a_bin, nl))
            seen_n.add(nl)
        if len(binary_seeds) >= 3:
            break

    # Also add pure random binary seeds
    for n in lengths[:2]:
        a_rand = (rng.random(n) > 0.5).astype(np.float64)
        binary_seeds.append((compute_score(a_rand), a_rand, n))

    for sc, a_bin, n in binary_seeds:
        if time.monotonic() > binary_deadline - 10:
            break
        print(f"\n  Binary LS n={n}, init_score={sc:.6f}, ones={int(a_bin.sum())}")

        a_opt, s_opt = binary_local_search(a_bin, min(binary_deadline,
                                                       time.monotonic() + 120.0),
                                           verbose_prefix=f"[n={n}] ")
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = a_opt.copy()
            print(f"  Improved to: {best_score:.6f}")

        # Also try continuous refinement from the binary solution
        if time.monotonic() < binary_deadline - 10:
            cont_deadline = min(binary_deadline, time.monotonic() + 30.0)
            f_cont, s_cont = nesterov_optimize(a_opt, cont_deadline,
                                               lr_init=0.005, top_k=3)
            if s_cont < best_score:
                best_score = s_cont
                best_sequence = f_cont.copy()
                print(f"  Continuous refine: {best_score:.6f}")

    # Phase 3: Focused continuous optimization (next 30% of time)
    print(f"\n=== Phase 3: Continuous Optimization (best={best_score:.6f}) ===")
    cont_deadline = t0 + (deadline - t0) * 0.85

    if best_sequence is not None:
        n_best = len(best_sequence)
        print(f"Optimizing n={n_best}...")

        # Nesterov with different hyperparameters
        f1, s1 = nesterov_optimize(best_sequence, min(cont_deadline,
                                                      time.monotonic() + 100.0),
                                   lr_init=0.005, top_k=5)
        if s1 < best_score:
            best_score = s1
            best_sequence = f1.copy()
            print(f"  Nesterov top5: {best_score:.6f}")

        # Try LP optimization
        if time.monotonic() < cont_deadline - 30:
            lp_deadline = min(cont_deadline, time.monotonic() + 80.0)
            f2, s2 = lp_optimize(best_sequence, lp_deadline, step_init=0.05)
            if s2 < best_score:
                best_score = s2
                best_sequence = f2.copy()
                print(f"  LP: {best_score:.6f}")

    # Phase 4: Final polish (last 15% of time)
    print(f"\n=== Phase 4: Final Polish (best={best_score:.6f}) ===")

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Multiple Nesterov passes with decreasing lr
        for lr_try, topk_try in [(0.003, 7), (0.002, 5), (0.001, 3)]:
            if time.monotonic() > deadline - 10:
                break
            f_pol, s_pol = nesterov_optimize(best_sequence, deadline,
                                             lr_init=lr_try, top_k=topk_try)
            if s_pol < best_score:
                best_score = s_pol
                best_sequence = f_pol.copy()
                print(f"  Polish lr={lr_try}, topk={topk_try}: {best_score:.6f}")

        # Try binary local search from current best
        if time.monotonic() < deadline - 20:
            a_bin_final = (best_sequence > np.median(best_sequence[best_sequence > 0])
                          ).astype(np.float64) if np.sum(best_sequence > 0) > 0 \
                else (best_sequence > 0.5).astype(np.float64)
            if 2 < a_bin_final.sum() < n_best - 2:
                a_ls, s_ls = binary_local_search(a_bin_final, deadline - 5,
                                                 verbose_prefix="[final] ")
                if s_ls < best_score:
                    best_score = s_ls
                    best_sequence = a_ls.copy()
                    print(f"  Final binary LS: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(1024)
        best_score = evaluate_sequence(best_sequence.tolist())

    # Final verification
    final_score = evaluate_sequence([float(x) for x in best_sequence])
    if not np.isfinite(final_score):
        print("WARNING: Final sequence failed! Using fallback.")
        best_sequence = np.ones(512)
        final_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, "
          f"internal={best_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_sequence]
