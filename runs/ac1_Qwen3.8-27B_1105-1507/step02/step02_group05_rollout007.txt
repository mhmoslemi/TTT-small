import numpy as np

from scipy.sparse import csr_matrix

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_sparse(a):
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-12})
