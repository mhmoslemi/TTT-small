"""
Optimization for minimizing 2n*max(f*f)/(sum f)^2 using:
1. Adaptive LP refinement with constraint generation
2. Smooth-max gradient descent
3. Coordinate descent polishing
4. Multi-length exploration
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute 2*n*max(conv) for f normalized to sum=1."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def normalize(f):
    """Normalize f to sum 1."""
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def build_conv_matrix(f):
    """Build matrix A such that A @ g = np.convolve(f, g, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        for j in range(js, je):
            A[k, j] = f[k - j]
    return A


def build_conv_matrix_fast(f):
    """Build convolution matrix using Toeplitz structure (faster for large n)."""
    n = len(f)
    # A[k,j] = f[k-j] if 0 <= k-j <= n-1, else 0
    # This is a Toeplitz matrix
    # Row k has f[k], f[k-1], ..., going diagonally
    # Use np.eye approach or direct construction
    
    # For efficiency, use the fact that each row is a shifted version of f
    # A[k, :] = [0...0, f[k], f[k-1], ..., f[0], 0, ..., 0] (shifted)
    
    # Actually, let's use a different approach:
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1
    # For row k: the non-zero entries are at positions j where 0 <= k-j <= n-1
    # i.e., k-(n-1) <= j <= k, intersected with [0, n-1]
    
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        A[k, j_start:j_end] = f[k - j_start : k - j_end + 1][::-1]
    return A


def adaptive_lp_solve(f, M, max_constraints=500, max_iter=50, time_limit=10):
    """
    Solve max sum(g) s.t. A@f_g <= M, g >= 0 using constraint generation.
    where A@f_g means convolve(f, g).
    Returns g or None.
    """
    n = len(f)
    A = build_conv_matrix(f)
    
    # Initial constraint set: top peaks
    conv_f = A @ f  # = f*f
    M_local = conv_f.max()
    
    # Select initial constraints
    n_init = min(max_constraints, len(conv_f))
    top_idx = np.argsort(conv_f)[::-1][:n_init]
    active = list(top_idx)
    
    start_time = time.monotonic()
    
    for it in range(max_iter):
        if time.monotonic() - start_time > time_limit:
            return None
        
        A_act = np.vstack([A[i, :] for i in active])
        c_obj = -np.ones(n)
        b_ub = np.full(len(active), M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A_act, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception:
            return None
        
        if not res.success or res.x is None:
            return None
        
        g = np.clip(res.x, 0, None)
        
        # Check all constraints
        conv_g = A @ g
        violated = np.where(conv_g > M * (1 + 1e-8) + 1e-10)[0]
        
        if len(violated) == 0:
            return g
        
        # Add most violated
        n_add = min(100, len(violated))
        most_violated = violated[np.argsort(conv_g[violated])[::-1]][:n_add]
        active.extend(most_violated.tolist())
        active = list(set(active))
        # Cap total constraints
        if len(active) > max_constraints * 3:
            # Keep only the most important ones
            conv_g_active = conv_g[active]
            keep = np.argsort(conv_g_active)[::-1][:max_constraints * 3]
            active = [active[i] for i in keep]
    
    return None


def lp_refine(f, time_budget, t0=0.3, rng=None):
    """LP-based refinement loop."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    n = len(f)
    f = normalize(f.copy().astype(np.float64))
    best_f = f.copy()
    best_score = compute_score(f)
    t = t0
    start = time.monotonic()
    no_improve_count = 0
    
    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        
        g = adaptive_lp_solve(f, M, max_constraints=min(300, 2*n-1),
                             max_iter=30, time_limit=min(8, time_budget/5))
        
        if g is None:
            no_improve_count += 1
            if no_improve_count > 5:
                break
            t *= 0.7
            if t < 0.01:
                break
            continue
        
        g = normalize(g)
        
        # Try different step sizes
        improved = False
        for step in [min(1.0, t * 1.2), t, t * 0.5, t * 0.25, 0.1, 0.05]:
            f_new = (1 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_score = compute_score(f_new)
            
            if new_score < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                if step < t:
                    t = min(1.0, t * 1.1)
                else:
                    t = min(1.0, t * 1.05)
                improved = True
                no_improve_count = 0
                break
        
        if not improved:
            t *= 0.6
            no_improve_count += 1
            if t < 0.005 or no_improve_count > 10:
                # Try perturbation
                f_pert = f * (1 + 0.1 * rng.standard_normal(n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    t = 0.1
                    no_improve_count = 0
                else:
                    break
    
    return best_f, best_score


def smooth_max_gradient(f, alpha=50.0, lr=0.05, n_iter=2000, rng=None):
    """Gradient descent on smooth approximation of max convolution peak."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    n = len(f)
    f = normalize(f.copy().astype(np.float64))
    best_f = f.copy()
    best_score = compute_score(f)
    start = time.monotonic()
    
    # Precompute index matrix for gradient
    # grad[i] = sum_k weights[k] * f[k-i] (with bounds)
    # = sum_{j=0}^{n-1} weights[i+j] * f[j] (with i+j <= 2n-2)
    idx = np.arange(n)[:, None] + np.arange(n)[None, :]  # idx[i,j] = i+j
    
    no_improve = 0
    
    for it in range(n_iter):
        if time.monotonic() - start > 600:  # Time cap
            break
        
        conv = np.convolve(f, f)
        
        # Smooth max weights
        conv_max = conv.max()
        shifted = conv - conv_max
        weights = np.exp(alpha * shifted)
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights = weights / wsum
        
        # Compute gradient: grad[i] = sum_j weights[i+j] * f[j] (with bounds)
        W_idx = np.where((idx >= 0) & (idx <= 2*n-2), weights[idx], 0.0)
        grad = W_idx @ f
        
        # Gradient descent with projection
        f_new = f - lr * grad
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            lr *= 0.5
            continue
        f_new = f_new / s
        
        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            # Slightly increase lr if making good progress
            if it % 100 == 0 and it > 0:
                lr = min(0.1, lr * 1.1)
        else:
            no_improve += 1
            lr *= 0.95
            if no_improve > 500:
                # Perturb
                f = f * (1 + 0.05 * rng.standard_normal(n))
                f = np.clip(f, 0, None)
                f = normalize(f)
                lr = 0.05
                no_improve = 0
                if time.monotonic() - start > 300:
                    break
    
    return best_f, best_score


def coordinate_descent(f, n_sweeps=5, rng=None, time_limit=60):
    """Coordinate descent: try changing one element at a time."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    n = len(f)
    f = normalize(f.copy().astype(np.float64))
    best_score = compute_score(f)
    start = time.monotonic()
    
    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_limit:
            break
        
        order = rng.permutation(n)
        improved_any = False
        
        for i in order:
            if time.monotonic() - start > time_limit:
                    break
            
            orig = f[i]
            if orig < 1e-12:
                continue
            
            best_val = orig
            best_local = best_score
            
            # Try geometric factors
            for factor in [0.3, 0.5, 0.7, 0.85, 1/0.85, 1/0.7, 1/0.5, 1.5, 2.0]:
                f[i] = orig * factor
                s = f.sum()
                if s < 1e-10:
                    f[i] = orig
                    continue
                conv = np.convolve(f, f)
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = orig * factor
            
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
                improved_any = True
        
        f = normalize(f)
        if not improved_any:
            break
    
    return f, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    r = rng.random()
    if r < 0.2:
        # Gaussian
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.15, 0.35)
        f = np.exp(-0.5 * (np.arange(n) - center)**2 / width**2)
    elif r < 0.4:
        # Cosine window
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        f[f < 0] = 0.01
    elif r < 0.6:
        # Sum of Gaussians
        f = np.ones(n) * 0.1
        for _ in range(rng.integers(2, 5)):
            c = int(rng.integers(0, n))
            w = int(rng.integers(n//8, n//3))
            f += np.exp(-0.5 * (np.arange(n) - c)**2 / max(1, w**2))
    elif r < 0.8:
        # Random smooth
        f = np.ones(n)
        for _ in range(rng.integers(3, 8)):
            c = int(rng.integers(0, n))
            w = int(rng.integers(3, n//4))
            f *= (1 + 0.3 * rng.standard_normal() * np.exp(-0.5*(np.arange(n)-c)**2/max(1,w**2)))
        f = np.abs(f) + 0.05
    else:
        # Power law
        f = (np.arange(n) + 1) ** rng.uniform(-1, 1)
        f = np.abs(f) + 0.05
    
    f = np.clip(f, 0.01, None)
    f = f * (1 + 0.1 * rng.standard_normal(n))
    f = np.clip(f, 0.001, None)
    return normalize(f)

Explores multiple sequence lengths with diverse initializations.

import numpy as np

def compute_score(f):
    """Compute 2*n*max(conv)(f,f) for f (assumed sum=1)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def normalize(f):
    """Normalize f to sum 1."""
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def build_conv_matrix(f):
    """Build matrix A such that A @ g = np.convolve(f, g, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def adaptive_lp_solve(f, A, M, max_constraints=400, max_iter=40, time_limit=10.0):
    """

    Returns g or None.

    n_init = min(max_constraints, len(conv_f))
    top_idx = np.argsort(conv_f)[::-1][:n_init]
    active = list(top_idx)
