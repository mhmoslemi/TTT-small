import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    n = len(a)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # Then (A @ g)[k] = sum_j A[k,j] * g[j] = sum_j a[k-j] * g[j]
    # which is the k-th element of conv(a, g)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            A[k, j_lo:j_hi+1] = a[k - j_hi : k - j_lo + 1][::-1]
    return A

def lp_solve_g(a, M):
    """Maximize sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_alternate(a, max_iter=200, relax=1.0):
    """LP-based alternating optimization."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        conv = np.convolve(a, a)
        M = np.max(conv) * relax
        g = lp_solve_g(a, M)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Try different blend factors
        for t in [0.5, 0.7, 1.0]:
            a_new = (1 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s = np.sum(a_new)
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
        
        a = np.maximum(0.5 * a + 0.5 * g)
        s = np.sum(a)
        a = a / s
        
        if it % 50 == 0:
            print(f"    LP iter {it}: obj={best_obj:.6f}")
    
    return best_a, best_obj

def subgradient_opt(a, n_steps=5000, verbose=False):
    """Subgradient optimization with adaptive step size."""
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for step in range(n_steps):
        conv = np.convolve(a, a)
        max_val = np.max(conv)
        
        # Active set
        tol = max_val * 0.01 * (1 + step / n_steps)
        active = conv >= max_val - tol
        n_active = np.sum(active)
        if n_active == 0:
            active = conv >= max_val - 1e-10
            n_active = 1
        
        # Compute subgradient
        # d/max(conv) / da[j] = (2/n_active) * sum_{k in active} (da*da/d a[j])[k]
        # (a*a)[k] = sum_i a[i] * a[k-i]
        # d(a*a)[k] / da[j] = a[k-j] + a[j-k] (with bounds)
        # For the "autocorrelation" interpretation: d(a*a)[k]/da[j] = 2*a[k-j] if 0<=k-j<n, or a[j] if k-j=j (i.e., k=2j)
        
        # Actually, (a*a)[k] = sum_{i} a[i] * a[k-i] where a[m] = 0 for m outside [0,n-1]
        # d(a*a)[k] / da[j] = a[k-j] + a[j-k] (if both are in range)
        # But since a is only defined on [0,n-1]:
        # d(a*a)[k] / da[j] = a[k-j] * [0<=k-j<n] + a[j-k] * [0<=j-k<n]
        
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            for j in range(n):
                dk = k - j
                val = 0.0
                if 0 <= dk < n:
                    val += a[dk]
                dk2 = j - k
                if 0 <= dk2 < n:
                    val += a[dk2]
                grad[j] += val
        
        grad /= n_active
        
        # Adaptive step size
        lr = 1.0 / (1.0 + 0.05 * step)
        
        # Multiplicative update
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 1e-15)  # Keep positive
        s = np.sum(a_new)
        a_new = a_new / s
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj
