def compute_score_fast(a):
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    S = float(np.sum(a))
    return 2.0 * n * M / (S * S)

def build_Q(p, n):
    """Build the QP matrix Q(p) = sum_k p_k A_k."""
    # A_k has (i,j) entry = 1 if i+j = k
    # Q(p)_{i,j} = sum_k p_k * 1[i+j=k] = p_{i+j} (if i+j < 2n-1)

        for j in range(n):
            k = i + j
            if k < len(p):
                Q[i, j] = p[k]
    return Q

def proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]

    return np.maximum(v - theta, 0.0)
