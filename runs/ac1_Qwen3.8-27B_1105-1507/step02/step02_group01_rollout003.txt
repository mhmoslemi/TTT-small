import numpy as np

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_sparse_A(a):
    """Build sparse A matrix for LP constraints: (a*g)[k] <= M for all k."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. A@G <= M, g >= 0."""
    n = len(a)
    A = build_sparse_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun)
    return None, None

"""
Optimized step function coefficient search using sparse LP alternating method,
analytical coordinate descent, diverse multi-start initialization, and adaptive
time allocation. Targets objective below 1.5027 by exploiting fast sparse LP
solves and smart 1D coordinate optimization with critical point candidates.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def compute_obj(a):
    """Compute the objective function value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_sparse_A(a):
    """Build sparse constraint matrix for LP: row k has a[k-j] at column j."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_sparse_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return np.clip(res.x, 0.0, None), float(-res.fun)
    except Exception:
        pass
    return None, None


def lp_optimize(a, max_time, verbose=False):
    """Iterative LP optimization with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)

        a = np.ones(n) / n

        a = a / s

    for it in range(500):
        if time.monotonic() - start > max_time:
            break

        conv = np.convolve(a, a)
        M = float(np.max(conv))

        g, g_sum = solve_lp(a, M)
        if g is None:
            break

        gs = float(np.sum(g))
        if gs < 1e-10:
            break
        g = g / gs

        # Try step sizes in decreasing order of aggressiveness
        improved = False
        for tt in [t, t * 0.7, t * 0.4, t * 0.2]:
            a_new = (1.0 - tt) * a + tt * g
            a_new = np.clip(a_new, 0.0, None)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-12:
                best_a = a_new.copy()
                best_obj = obj
                a = a_new
                improved = True
                stall_count = 0
                break

        if not improved:
            stall_count += 1
            t = max(0.05, t * 0.85)

                if time.monotonic() - start > max_time * 0.8:

        if verbose and it % 20 == 0:

def coordinate_descent(a, max_time, verbose=False):
    """Coordinate descent with analytical candidate generation."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for epoch in range(200):
        if time.monotonic() - start > max_time:
            break

        epoch_improved = False

        for i in range(n):
            if time.monotonic() - start > max_time:
                break

            ai = a[i]
            conv = np.convolve(a, a)
            s = float(np.sum(a))

            # Generate candidates
            candidates = [0.0, ai * 0.3, ai * 0.6, ai * 1.4, ai * 1.8,
                          ai * 2.5, ai * 0.1]

            # Try to equalize top convolution entries
            top_ks = np.argsort(conv)[-4:]
            for idx1 in range(len(top_ks)):
                for idx2 in range(idx1 + 1, len(top_ks)):
                    k1 = top_ks[idx1]
                    k2 = top_ks[idx2]
                    ak1 = a[k1 - i] if 0 <= k1 - i < n else 0.0
                    ak2 = a[k2 - i] if 0 <= k2 - i < n else 0.0
                    sq1 = 1.0 if k1 == 2 * i else 0.0
                    sq2 = 1.0 if k2 == 2 * i else 0.0

                    if abs(sq1 - sq2) > 1e-10:
                        A_c = sq1 - sq2
                        B_c = 2.0 * (ak1 - ak2)

                        disc = B_c * B_c - 4.0 * A_c * C_c
                        if disc >= 0:
                            sqrt_disc = np.sqrt(disc)
                            d1 = (-B_c + sqrt_disc) / (2.0 * A_c)
                            d2 = (-B_c - sqrt_disc) / (2.0 * A_c)
                            for d in [d1, d2]:
                                if d > -ai - 1e-9:
                                    candidates.append(max(0.0, min(d, 5.0 * max(ai, 1.0 / n))))
                    else:
                        if abs(ak1 - ak2) > 1e-10:
                            d = (conv[k2] - conv[k1]) / (2.0 * (ak1 - ak2))
                            if d > -ai - 1e-9:
                                candidates.append(max(0.0, min(d, 5.0 * max(ai, 1.0 / n))))

            # Evaluate candidates
            best_val = ai
            best_obj_i = compute_obj(a)

                    best_obj_i = obj
                    best_val = val

            a[i] = best_val
            if best_val != ai:
                epoch_improved = True

        obj_now = compute_obj(a)
        if obj_now < best_obj - 1e-13:
            best_a = a.copy()
            best_obj = obj_now

        if not epoch_improved:
            break

    return best_a, best_obj


def block_refine(a, rng, max_time):
    """Refine by perturbing random blocks."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    sigma = 0.08

    while it < 200000 and (time.monotonic() - start) < max_time:
        # Adapt sigma based on progress
        progress = (time.monotonic() - start) / max_time
        sigma = 0.08 * (1.0 - 0.5 * progress)

        if rng.random() < 0.3:
            width = max(1, int(n * rng.uniform(0.005, 0.04)))
        else:
            width = max(1, int(n * rng.uniform(0.02, 0.12)))
        pos = int(rng.integers(0, n - width + 1))

        candidate = best_a.copy()
        strat = rng.random()
        if strat < 0.5:
            factor = np.exp(rng.normal(0, sigma, width))
            candidate[pos:pos + width] *= factor
        elif strat < 0.8:
            candidate[pos:pos + width] += rng.normal(0, sigma * np.mean(best_a) + 1e-6, width)
        else:
            t = np.exp(rng.normal(0, sigma))
            candidate[pos:pos + width] *= t

        candidate = np.clip(candidate, 0.0, None)
        s = float(np.sum(candidate))
        if s < 1e-10:
            it += 1
            continue
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj
        it += 1

    return best_a, best_obj


def diverse_init(n, rng, init_type):
    """Generate diverse initializations of type init_type."""
    if init_type == 0:
        # Edge-concentrated with decay

        a += 0.1 * rng.random(n)
    elif init_type == 1:
        # Gaussian center
        center = n / 2.0
        width = n * rng.uniform(0.15, 0.4)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        a += 0.05 * rng.random(n)
    elif init_type == 2:
        # Two peaks
        p1 = n * rng.uniform(0.2, 0.4)
        p2 = n * rng.uniform(0.6, 0.8)

            np.exp(-0.5 * ((np.arange(n) - p2) / w) ** 2)
        a += 0.05 * rng.random(n)
    elif init_type == 3:
        # Cosine window
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / n))
        a += 0.1 * rng.random(n)
    elif init_type == 4:
        # Multi-frequency
        freqs = rng.uniform(0.3, 2.0, 3)
        phases = rng.uniform(0, 2 * np.pi, 3)
        a = np.ones(n)
        for f, p in zip(freqs, phases):
            a += 0.3 * np.cos(2 * np.pi * f * np.arange(n) / n + p)
        a = np.maximum(0, a)
    elif init_type == 5:
        # Smooth random
        raw = rng.random(n)
        kernel = np.ones(7) / 7
        a = np.convolve(raw, kernel, mode='same')
        a = np.maximum(0.01, a)
    elif init_type == 6:
        # Plateau with edges

        edge_w = max(1, n // 10)

        a += 0.1 * rng.random(n)
    elif init_type == 7:
        # Pure random
        a = np.maximum(0.01, rng.exponential(0.5, n))
    else:
        a = np.ones(n) / n

    a = np.maximum(0.0, a)
    s = float(np.sum(a))
    if s < 1e-10:

                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    print(f"  Parent CD: obj={obj_cd:.6f}")
    except Exception as e:
        print(f"Parent opt failed: {e}")

    # === Main search ===
    n_values = [60, 80, 100, 120, 150, 180, 200, 250, 300, 350, 400]

    for n in n_values:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 50:
            break

        num_remaining_n = len([x for x in n_values if x >= n])
        time_per_n = remaining / max(1, num_remaining_n)
        time_per_n = min(time_per_n, 130.0)

        # Decide number of restarts
        num_restarts = max(2, min(5, int(time_per_n / 28)))
        time_per_restart = time_per_n / num_restarts

        print(f"\nN={n}: {remaining:.0f}s left, {num_restarts} restarts ({time_per_restart:.0f}s each)")

        for r in range(num_restarts):
            if time.monotonic() > deadline - 50:
                break

            init_type = int(rng.integers(0, 8))
            a_init = diverse_init(n, rng, init_type)

            a_opt, obj_opt = full_pipeline(a_init, time_per_restart, rng)

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} r={r} init={init_type}: obj={obj_opt:.6f} ** NEW BEST **")
            else:
                print(f"  n={n} r={r} init={init_type}: obj={obj_opt:.6f}")

    # === Final refinement ===
    if best_seq is not None and time.monotonic() < deadline - 8:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: {remaining:.0f}s, current best={best_score:.6f}")

        # LP refinement
        lp_time = min(remaining * 0.3, 60.0)
        a_f, obj_f = lp_optimize(best_seq.copy(), lp_time)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  After LP: obj={best_score:.6f}")

        # Coordinate descent
        if time.monotonic() < deadline - 5:
            cd_time = min(remaining * 0.25, 45.0)
            a_f, obj_f = coordinate_descent(best_seq.copy(), cd_time)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  After CD: obj={best_score:.6f}")

        # Block perturbation
        if time.monotonic() < deadline - 4:
            bp_time = deadline - time.monotonic() - 3.0
            a_f, obj_f = block_refine(best_seq, rng, bp_time)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  After BP: obj={best_score:.6f}")

        # Final LP
        if time.monotonic() < deadline - 3:
            lp_time = deadline - time.monotonic() - 2.0
            a_f, obj_f = lp_optimize(best_seq.copy(), lp_time)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  After final LP: obj={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
