"""
LP-based iterative minimax optimization for self-convolution peak reduction.
Uses linearized LP to find optimal perturbation directions, with line search
and multiple fallback strategies for robustness.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_score(f):
    """Compute the score for a sequence."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (s * s)

def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s

def lp_minimax_direction(f):
    """
    Solve the LP to find the direction δ that minimizes the linearized peak
    of the self-convolution.
    
    LP: minimize t
        subject to 2*(f*δ)[k] - t <= -conv(f,f)[k] for all k
                 sum(δ) = 0
                 δ >= -f
    """
    n = len(f)
    conv_ff = np.convolve(f, f)
    
    # Build A matrix: A[k,i] = 2*f[k-i] if 0 <= k-i < n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            A[k, i] = 2.0 * f[k - i]
    
    # LP formulation
    # Variables: [δ_0, ..., δ_{n-1}, t]
    c = np.append(np.zeros(n), 1.0)
    
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = -conv_ff  # A@δ - t <= -conv_ff[k]
    
    A_eq = np.hstack([np.ones(n), 0.0])
    b_eq = np.array([0.0])
    
    bounds = [(lo, None) for lo in (-f)] + [(-np.inf, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5})
    
    if not res.success or res.x is None:
        return None, None
    return res.x[:n], res.x[n]  # δ*, t*

def line_search(f, delta, max_alpha=1.0):
    """Find the best step size α for f + α*delta."""
    n = len(f)
    cur_score = compute_score(f)
    
    # Maximum α for non-negativity
    neg_mask = delta < 0
    if neg_mask.any():
        alpha_nn = (f[neg_mask] / (-delta[neg_mask])).min()
        max_alpha = min(max_alpha, alpha_nn * 0.99)
    
    if max_alpha <= 0:
        return 0.0, None
    
    # Try several step sizes
    candidates = []
    for frac in [0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
        a = frac * max_alpha
        if 0 < a <= max_alpha:
            candidates.append(a)
    
    if not candidates:
        return 0.0, None
    
    best_alpha = 0.0
    best_f = None
    best_score = cur_score
    
    for a in sorted(candidates, reverse=True):  # Try larger steps first
        f_new = f + a * delta
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 0.01:
            continue
        f_new = f_new / s
        sc = compute_score(f_new)
        if sc < best_score:
            best_score = sc
            best_f = f_new
            best_alpha = a
    
    return best_alpha, best_f

def pressure_update_multipeak(f, n_iter=200, eta=0.15, n_peaks=5, rng=None):
    """Multi-peak pressure-based update."""
    if rng is None:

    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0
    
    for it in range(n_iter):

        # Find top peaks
        peak_vals = np.sort(conv)[::-1][:n_peaks]
        threshold = peak_vals[-1] * 0.8  # Consider peaks above 80% of the n-th peak

        pressure = np.zeros(n)
        for k in range(2*n-1):
            if conv[k] < threshold:
                continue
            weight = (conv[k] - threshold) / max(peak_vals[0] - threshold, 1e-10)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += weight * f[j]
        
        if pressure.max() < 1e-10:
            break
        
        pressure /= pressure.max()
        
        # Multiplicative update
        eta_eff = eta / (1.0 + 0.05 * it)

        f_new = np.clip(f_new, 1e-15, None)

        target_mag = np.sqrt(np.sum(mag**2) * n_fft / n_bins)  # Preserve total energy
        # Actually, let's target a specific flatness

        # For other bins, target a flat magnitude

def coordinate_descent(f, n_sweeps=5, rng=None):
    """Coordinate descent polishing."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)
    
    for sweep in range(n_sweeps):
        for i in rng.permutation(n):
            orig = f[i]
            if orig < 1e-12:
                continue
            best_val = orig
            best_local = best_score
            for factor in [0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.5, 2.0]:
                f[i] = max(0.0, orig * factor)
                sc = compute_score(f)
                if sc < best_local:
                    best_local = sc
                    best_val = f[i]
            f[i] = best_val
            best_score = min(best_score, best_local)
        f = normalize(f)
    
    return f, best_score

"""
LP-based iterative minimax optimization for self-convolution peak reduction.
Uses a linearized LP subproblem to find optimal perturbation directions that
minimize the peak of the self-convolution in the max sense, combined with
line search, multi-peak pressure updates, spectral flattening, and coordinate
descent for robustness and polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute the score: 2*n*max(conv(f,f))/sum(f)^2."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (s * s)


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def build_conv_matrix(f):
    """Build matrix A such that A @ delta = 2*(f*delta) for linear convolution."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            A[k, i] = 2.0 * f[k - i]
    return A


def lp_minimax_direction(f, time_limit=5):
    """
    Solve: minimize t
           s.t. 2*(f*delta)[k] - t <= -conv(f,f)[k]  for all k
                sum(delta) = 0
                delta >= -f
    Returns (delta, t_opt) or (None, None).
    """
    n = len(f)
    conv_ff = np.convolve(f, f)

    A = build_conv_matrix(f)

    c = np.append(np.zeros(n), 1.0)
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = -conv_ff.copy()
    A_eq = np.hstack([np.ones(n), 0.0])
    b_eq = np.array([0.0])
    bounds = [(-f[i], None) for i in range(n)] + [(-np.inf, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None, None

    if not res.success or res.x is None:
        return None, None
    return res.x[:n], res.x[n]


def line_search(f, delta, max_alpha=0.8):
    """Find best step size alpha for f + alpha*delta via multi-evaluation search."""
    cur_score = compute_score(f)

    neg_mask = delta < -1e-15
    if neg_mask.any():
        alpha_nn = (f[neg_mask] / (-delta[neg_mask])).min()
        max_alpha = min(max_alpha, alpha_nn * 0.99)

    if max_alpha <= 1e-6:
        return 0.0, None

    best_alpha = 0.0
    best_f = None
    best_score = cur_score

    # Geometric sequence of step sizes
    fracs = [0.1, 0.2, 0.35, 0.5, 0.65, 0.8, 1.0]
    for frac in sorted(fracs, reverse=True):
        a = frac * max_alpha
        if a <= 0:
            continue
        f_new = f + a * delta
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 0.01:
            continue
        f_new = f_new / s
        sc = compute_score(f_new)
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            best_alpha = a

    return best_alpha, best_f


def pressure_multipeak(f, n_iter=200, eta=0.15, n_peaks=5, rng=None):
    """Multi-peak pressure update: reduce elements contributing to top convolution peaks."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()

        # Consider peaks above a fraction of max
        threshold = M * 0.7
        pressure = np.zeros(n)
        for k in range(2 * n - 1):
            if conv[k] < threshold:
                continue
            weight = (conv[k] - threshold) / max(M - threshold, 1e-10)
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += weight * f[j]

        pmax = pressure.max()
        if pmax < 1e-12:
            break
        pressure /= pmax

        eta_eff = eta / (1.0 + 0.03 * it)
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        sc = compute_score(f_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 40:
                f_pert = f * (1 + 0.15 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    stall = 0
                else:
                    break

    return best_f, best_score


def spectral_flat(f, n_iter=100, rng=None):
    """Spectral flattening: project onto flat-magnitude spectrum + non-negative constraint."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)
        n_bins = len(mag)

        current_energy = np.sum(f ** 2)
        target_sq = (n_fft * current_energy - 1.0) / max(n_bins - 1, 1)
        if target_sq < 1e-10:
            target_sq = current_energy * n_fft / n_bins

        mag_new = np.full(n_bins, np.sqrt(target_sq))
        mag_new[0] = 1.0

        if it % 60 == 59:
            phase = phase + rng.normal(0, 0.1, n_bins)

        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 0.01:
            phase = phase + rng.normal(0, 0.3, n_bins)
            F_new = mag_new * np.exp(1j * phase)
            f_new = np.fft.irfft(F_new, n_fft)[:n]
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 0.01:
                stall += 1
                if stall > 80:
                    break
                continue

        f_new = f_new / s
        sc = compute_score(f_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break

    return best_f, best_score


def coordinate_descent(f, n_sweeps=5, rng=None):
    """Coordinate descent polishing."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)

    for sweep in range(n_sweeps):
        for i in rng.permutation(n):
            orig = f[i]
            if orig < 1e-12:
                continue
            best_val = orig
            best_local = best_score
            for factor in [0.4, 0.6, 0.8, 0.92, 1.0 / 0.92, 1.25, 1.6, 2.5]:
                f[i] = max(0.0, orig * factor)
                sc = compute_score(f)
                if sc < best_local:
                    best_local = sc
                    best_val = f[i]
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
        f = normalize(f)

    return f, best_score


def diverse_init(n, rng):
    """Generate a diverse initial sequence."""
    r = rng.random()
    x = np.arange(n, dtype=np.float64)
    if r < 0.25:
        f = np.ones(n)
        for _ in range(12):
            c = rng.uniform(0, n)
            w = max(2.0, n * rng.uniform(0.05, 0.3))
            amp = rng.uniform(0.1, 0.5)
            f += amp * np.exp(-0.5 * (x - c) ** 2 / w ** 2) * rng.standard_normal()
    elif r < 0.5:
        freq = rng.uniform(1, 4)
        phi = rng.uniform(0, 2 * np.pi)
        f = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n + phi)
    elif r < 0.75:
        c1 = n * rng.uniform(0.1, 0.3)
        c2 = n * rng.uniform(0.7, 0.9)
        w = n * rng.uniform(0.1, 0.25)
        f = np.exp(-0.5 * (x - c1) ** 2 / w ** 2) + np.exp(-0.5 * (x - c2) ** 2 / w ** 2)
    else:
        # Low-pass random
        spec = np.fft.rfft(rng.standard_normal(n))
        freqs = np.fft.rfftfreq(n)
        spec *= np.exp(-5.0 * (freqs * n) ** 2)
        f = np.fft.irfft(spec, n)

    f = np.abs(f) + 0.01
    f = normalize(f)
    f = f * (1 + 0.05 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    return normalize(f)

        print(f"Final result: score={final_score:.6f}, n={len(final)}")
        print(f"Target: 1.5027, Previous best: 1.512006")
        return final

    # Fallback
    print("FALLBACK: uniform sequence")
    return [1.0] * 256
