def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0

        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)

            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    rows = rows[:idx]

    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def make_initializations(n, rng):
    """Create diverse initializations for length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(('uniform', np.ones(n)))

    # Cosine taper
    cos_t = (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0
    inits.append(('cosine', cos_t))

    # Linear ramp (decreasing)
    ramp = np.linspace(1.0, 0.1, n)
    inits.append(('ramp', ramp))

    # Linear ramp (increasing)
    inits.append(('ramp_inc', np.linspace(0.1, 1.0, n)))

    # Exponential decay
    exp_dec = np.exp(-2.0 * idx / n)
    inits.append(('exp_decay', exp_dec))

    # Exponential decay (other side)
    exp_dec2 = np.exp(-2.0 * (n - 1 - idx) / n)
    inits.append(('exp_decay2', exp_dec2))

    # Bell (centered)
    center = (n - 1) / 2.0
    bell = np.exp(-0.5 * ((idx - center) / (n / 4.0)) ** 2)
    inits.append(('bell', bell))

    # Bimodal
    two_peak = (np.exp(-0.5 * ((idx - n * 0.25) / (n / 8.0)) ** 2) +
                np.exp(-0.5 * ((idx - n * 0.75) / (n / 8.0)) ** 2))
    inits.append(('bimodal', two_peak))

    # Random
    r = rng.uniform(0.2, 2.0, size=n)
    inits.append(('random', r))

    # S-curve (sigmoid)
    s_curve = 1.0 / (1.0 + np.exp(-4.0 * (idx / (n - 1) - 0.5)))
    inits.append(('sigmoid', s_curve))

    # Quadratic
    quad = 1.0 - (2.0 * idx / (n - 1) - 1.0) ** 2
    quad = np.clip(quad, 0.05, None)
    inits.append(('quadratic', quad))

    return inits
