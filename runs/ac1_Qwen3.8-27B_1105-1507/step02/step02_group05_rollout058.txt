"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-scale L-BFGS-B with log-sum-exp smoothing and analytic gradients
2. LP-based mirror descent for exact max refinement
3. Diverse initializations and random restarts
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def lse_grad(a, tau):
    """
    Compute the gradient of LSE(conv(a,a), tau) with respect to a.
    Returns (lse_value, gradient).
    """
    n = len(a)
    b = fast_conv(a)
    
    # Log-sum-exp
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    sum_exp = exp_b.sum()
    lse_val = (b_max + np.log(sum_exp)) / tau
    
    # Softmax weights
    p = exp_b / sum_exp
    
    # Gradient: grad_i = 2 * sum_{d=0}^{n-1} p[i+d] * a[d]
    # This is a cross-correlation of p and a
    S = 1
    while S < (2 * n - 1) + n - 1:
        S <<= 1
    
    P = np.fft.rfft(p, S)
    A = np.fft.rfft(a, S)
    cross_corr = np.fft.irfft(P * np.conj(A), S)
    
    grad = 2.0 * cross_corr[:n]
    return lse_val, grad


def objective_and_grad(x, tau):
    """Objective and gradient for L-BFGS-B. Minimize 2n*LSE/S^2."""
    n = len(x)
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    
    lse_val, g = lse_grad(a, tau)
    
    # J = 2n * lse_val / s^2
    obj = 2.0 * n * lse_val / (s * s)
    
    # dJ/da_i = (2n/s^2) * (g_i - 2*max_b/s)
    # where max_b ≈ lse_val
    # dJ/dx_i = dJ/da_i if x_i > 0, else 0 (subgradient)
    max_b_approx = lse_val
    grad_a = (2.0 * n / (s * s)) * (g - 2.0 * max_b_approx / s)
    
    # Chain rule for ReLU
    grad_x = grad_a * (x > 0)
    
    return obj, grad_x


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def run_lbb(a_init, tau_start, tau_end, max_iter, deadline, tracker):
    """Run L-BFGS-B optimization with adaptive tau."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    
    if a.sum() < 1e-10:
        return a
    
    # Run L-BFGS-B with increasing tau
    tau = tau_start
    n_taus = 5  # number of tau levels
    
    for ti in range(n_taus):
        if time.monotonic() > deadline:
            break
        tau = tau_start + (tau_end - tau_start) * ti / max(1, n_taus - 1)
        
        def fun(x):
            obj, _ = objective_and_grad(x, tau)
            return obj
        
        def grad(x):
            _, g = objective_and_grad(x, tau)
            return g
        
        bounds = [(0.0, None)] * n
        result = minimize(fun, a, jac=grad, method='L-BFGS-B',
                         bounds=bounds, options={'maxiter': max_iter // n_taus,
                                                 'max_fun': 20000, 'ftol': 1e-18})
        if result.fun < 1e10:
            a = np.maximum(result.x, 0.0)
        
        tracker.update(a)
    
    return a


def build_constraint_matrix(a):
    """Build the constraint matrix for LP: (a*g)_k <= M for all k."""
    n = len(a)
    # A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    # Using the structure: for row k, valid j are max(0, k-n+1) to min(k, n-1)
    from scipy.sparse import coo_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_constraint_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_iterations(a, max_iters, deadline, tracker):
    """Run LP mirror descent iterations."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Adaptive step
        t_candidates = [0.1, 0.3, 0.5, 0.7, 1.0]
        best_a = None
        best_J = compute_J(a)
        
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            if a_trial.sum() < 1e-10:

            J_trial = compute_J(a_trial)
            if J_trial < best_J:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            break
        a = best_a
    
    tracker.update(a)
    return a

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-scale L-BFGS-B with log-sum-exp smoothing and analytic gradients
2. LP-based mirror descent for exact max refinement
3. Diverse initializations, random restarts, and adaptive hyperparameters

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import coo_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def lse_value_and_p(b, tau):
    """Compute LSE value and softmax probability weights."""
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    sum_exp = exp_b.sum()
    lse_val = (b_max + np.log(sum_exp)) / tau
    p = exp_b / sum_exp
    return lse_val, p


def compute_lse_grad(a, tau):
    """
    Compute gradient of LSE(conv(a,a), tau) w.r.t. a.
    Returns (lse_value, gradient_vector).
    """
    n = len(a)
    b = fast_conv(a)
    lse_val, p = lse_value_and_p(b, tau)

    # grad_i = 2 * sum_{d=0}^{n-1} p[i+d] * a[d]
    # Cross-correlation via FFT
    S = 1
    while S < (2 * n - 1) + n - 1:
        S <<= 1

    P = np.fft.rfft(p, S)
    A = np.fft.rfft(a, S)
    cross_corr = np.fft.irfft(P * np.conj(A), S)

    grad = 2.0 * cross_corr[:n]
    return lse_val, grad


def _obj_func(x, tau):
    """Objective for L-BFGS-B: 2n * LSE / S^2 with ReLU on x."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    lse_val, _ = compute_lse_grad(a, tau)
    return 2.0 * len(x) * lse_val / (s * s)


def _grad_func(x, tau):
    """Gradient for L-BFGS-B."""
    n = len(x)
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros(n)
    lse_val, g = compute_lse_grad(a, tau)
    # dJ/da_i = (2n/S^2) * (g_i - 2*LSE/S)
    grad_a = (2.0 * n / (s * s)) * (g - 2.0 * lse_val / s)
    # Chain rule for ReLU: zero out where x < 0
    mask = (x > 0).astype(np.float64)
    return grad_a * mask


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def build_lp_matrix(a):
    """Build sparse LP constraint matrix: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            rows.extend([k] * (jh - jl + 1))
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    return coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_lp_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_from(a, max_iters, deadline, tracker):
    """Run LP mirror descent from initial a."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g /= gs

        best_a = None
        best_J = compute_J(a)
        for t in [0.1, 0.25, 0.5, 0.75, 1.0]:

                best_a = trial
        if best_a is None:

        a = best_a

    return a

def make_inits(n, rng, parent):

    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.1, 1.0, n))

    t = np.linspace(0, 1, n, endpoint=False)

    # Cosine-based

    inits.append(0.3 + 0.5 * np.cos(2 * np.pi * 3 * t))
    inits.append(0.4 + 0.4 * np.cos(2 * np.pi * 5 * t))

    inits.append(1.0 - 0.4 * np.exp(-20 * (t - 0.5) ** 2))

    # Smooth slightly
    k = np.ones(3) / 3.0

    amps = rng.uniform(0.15, 0.45, size=4)

    inits.append(np.maximum(sig, 0.01))

    # Triangular (inverted)
    inits.append(1.0 - np.abs(2 * t - 1) * 0.6)

    return inits

            if n_best <= 2048:
                a = run_lp_from(tracker.best, 100, deadline, tracker)

        tracker.report(f"Polish: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*40} FINAL {'='*40}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
