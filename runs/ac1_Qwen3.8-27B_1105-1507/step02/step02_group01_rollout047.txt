import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    """Fast objective computation."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def subgradient_step(a, T=0.01):
    """Compute a subgradient-like direction to reduce the max convolution."""
    n = len(a)
    conv = np.convolve(a, a)
    max_c = float(np.max(conv))
    # Soft-max weights
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    w = w / np.sum(w)
    
    # Gradient of soft-max w.r.t. a
    # d(soft_max)/d(a[i]) = sum_k w[k] * d(c[k])/d(a[i])
    # d(c[k])/d(a[i]) = a[k-i] if 0 <= k-i < n, else 0
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-10:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # c[k] = sum_{i=i_start}^{i_end} a[i] * a[k-i]
        # d(c[k])/d(a[i]) = a[k-i]
        for i in range(i_start, i_end + 1):
            grad[i] += w[k] * a[k - i]
    
    return grad, max_c

The effect on the convolution:

So the change is proportional to the difference in the "neighborhood" of `i` and `j` as seen from position `k`.

- Find the `j` that contributes least to `c[k*]` (or doesn't contribute at all).
- Transfer mass from `i` to `j`.

    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for _ in range(num_swaps):
        k_star = int(np.argmax(conv))
        # Find i that contributes most to c[k_star]
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        contributions = [a[i] * a[k_star - i] for i in range(i_start, i_end + 1)]
        i_best = i_start + int(np.argmax(contributions))
        
        # Find j that contributes least to c[k_star]
        j_candidates = [j for j in range(n) if j != i_best]
        # Prefer j where a[k_star - j] is small (if valid)
        min_contrib = float('inf')
        j_best = j_candidates[0]
        for j in j_candidates:
            if 0 <= k_star - j < n:
                c = a[i_best] * a[k_star - j]
            else:
                c = 0
            if c < min_contrib:
                min_contrib = c
                j_best = j

        delta_options = [0.01, 0.05, 0.1, 0.2, 0.3] * (np.random.random() * 0.5 + 0.5)

            a_new[i_best] -= delta

targeted swap moves, LP-based global steps, and simulated annealing refinement.
Targets objective below 1.5027 by improving on the parent (1.5117).

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute the objective function value for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s} in Euclidean distance."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - s))[0]
    if len(rho) == 0:
        return np.full(n, s / n)
    rho = rho[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    w_sum = np.sum(w)
    if w_sum > 1e-15:
        w = w * s / w_sum
    else:
        w = np.full(n, s / n)
    return w


def multi_subgrad(a, top_k=10):
    """Compute averaged subgradient from top-k convolution peaks."""
    n = len(a)
    conv = np.convolve(a, a)
    top_indices = np.argsort(conv)[-top_k:]
    grad = np.zeros(n)
    for k in top_indices:

        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):

    grad /= len(top_indices)
    return grad, conv, top_indices

    """Subgradient descent with momentum and simplex projection."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

def build_A_matrix(a):
    """Build the constraint matrix for the LP formulation."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def lp_step(a, max_time, upper_bound=10.0):
    """Solve LP to find a direction that reduces the max convolution."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': max_time})
    if res.status == 0:
        g = res.x
        g_sum = float(np.sum(g))
        if g_sum > 1e-10:
            g = g / g_sum
            return g, M
    return None, M

    stall = 0

    while time.monotonic() - start < max_time:
        lp_time = min(5.0, max_time * 0.3)
        g, M = lp_step(a, lp_time)
        if g is None:
            break
        if np.sum(g) <= 1.0 + 1e-4:
            stall += 1
            if stall > 5:
                break
            continue
        stall = 0
        for t in [0.5, 0.3, 0.1, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0, None)
            s = np.sum(a_new)
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_a = a_new.copy()
                best_obj = obj
                a = a_new
                break

    return best_a, best_obj


def targeted_swap_refine(a, max_time, rng):
    """Refine using targeted swap moves guided by convolution peaks."""
    n = len(a)
    best_a = a.copy().astype(np.float64)
    best_a = best_a / np.sum(best_a)
    best_obj = compute_obj(best_a)
    conv = np.convolve(best_a, best_a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        k_star = int(np.argmax(conv))
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)

        # Find hottest contributor
        contribs = np.array([best_a[i] * best_a[k_star - i]
                            for i in range(i_start, i_end + 1)])
        i_hot = i_start + int(np.argmax(contribs))

        # Find cold position (low value, far from hot in convolution sense)
        candidates = []
        for j in range(n):
            if j == i_hot:
                continue
            # Score: how much j contributes to c[k_star]
            if 0 <= k_star - j < n:
                score = best_a[i_hot] * best_a[k_star - j]
            else:
                score = 0
            # Also penalize if j has high value (hard to subtract from)
            score += 0.1 * best_a[j]
            candidates.append((score, j))

        candidates.sort()
        j_cold = candidates[0][1]

        # Try different transfer amounts
        improved = False
        base_delta = best_a[i_hot]
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
            delta = base_delta * frac * rng.uniform(0.5, 1.5)
            delta = min(delta, best_a[i_hot] * 0.9)
            if delta < 1e-12:
                continue
            a_new = best_a.copy()
            a_new[i_hot] -= delta
            a_new[j_cold] += delta
            a_new = np.clip(a_new, 0, None)
            s = np.sum(a_new)
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new
                conv = np.convolve(best_a, best_a)
                improved = True
                break

        if not improved:
            # Random swap as exploration
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            delta = best_a[i] * rng.uniform(0.05, 0.3)
            if delta < 1e-12:
                continue
            a_new = best_a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = np.clip(a_new, 0, None)
            s = np.sum(a_new)
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new
                conv = np.convolve(a_new, a_new)

    return best_a, best_obj


def block_anneal(a, max_time, rng):
    """Simulated annealing with block perturbations."""
    n = len(a)
    best_a = a.copy().astype(np.float64)
    best_a = best_a / np.sum(best_a)
    best_obj = compute_obj(best_a)
    start = time.monotonic()

    T = 0.02
    T_min = 0.0001
    cooling = 0.999

    current_a = best_a.copy()
    current_obj = best_obj
    no_improve = 0

    while time.monotonic() - start < max_time and T > T_min:
        # Perturb a random block
        block_size = max(2, int(n * rng.uniform(0.02, 0.15)))
        pos = int(rng.integers(0, n - block_size + 1))

        a_new = current_a.copy()
        noise = rng.normal(1.0, 0.3, block_size)
        a_new[pos:pos + block_size] *= noise
        a_new = np.clip(a_new, 0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            T *= cooling
            continue
        a_new = a_new / s
        obj_new = compute_obj(a_new)

        delta = obj_new - current_obj
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_a = a_new
            current_obj = obj_new
            if obj_new < best_obj:
                best_a = a_new.copy()
                best_obj = obj_new
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        T *= cooling
        if no_improve > 500:
            T *= 0.9

    return best_a, best_obj


def edge_init(n, rng):
    """Initialize with mass concentrated at edges."""
    a = np.zeros(n)
    c = 1.0 / np.sqrt(2.0 * n - 1.0)
    a[0] = c * rng.uniform(0.8, 1.5)
    a[1] = c * rng.uniform(0.5, 1.2)
    a[-1] = c * rng.uniform(0.8, 1.5)
    a[-2] = c * rng.uniform(0.5, 1.2)
    remaining = 1.0 - a[0] - a[1] - a[-1] - a[-2]
    if remaining > 0:
        interior = np.ones(n - 4) * (remaining / (n - 4))
        noise = rng.normal(0, 0.3, n - 4)
        interior *= np.exp(noise)
        interior = np.clip(interior, 0, None)
        s = np.sum(interior)
        if s > 1e-10:
            interior = interior * remaining / s
        a[2:n-2] = interior
    a = np.clip(a, 0, None)
    a = a / np.sum(a)
    return a


def sinusoidal_init(n, rng):
    """Initialize with sinusoidal structure."""
    freq = rng.uniform(0.3, 2.0)
    phase = rng.uniform(0, 2 * np.pi)
    a = np.maximum(0, 1.0 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase))
    a += 0.1 * rng.random(n)
    a = a / np.sum(a)
    return a


def two_bump_init(n, rng):
    """Initialize with two bumps."""
    center1 = n * rng.uniform(0.2, 0.4)
    center2 = n * rng.uniform(0.6, 0.8)
    width1 = n * rng.uniform(0.05, 0.15)
    width2 = n * rng.uniform(0.05, 0.15)
    a = (np.exp(-0.5 * ((np.arange(n) - center1) / width1) ** 2) +
         np.exp(-0.5 * ((np.arange(n) - center2) / width2) ** 2))
    a += 0.05 * rng.random(n)
    a = a / np.sum(a)
    return a


def random_structured_init(n, rng):
    """Random initialization with structure."""
    a = np.maximum(0, rng.normal(1.0, 0.5, n))
    # Add some smoothness
    kernel = np.ones(3) / 3.0
    a = np.convolve(a, kernel, mode='same')
    a = a / np.sum(a)
    return a


def optimize_single(a_init, time_budget, rng):
    """Run full optimization pipeline on a single initialization."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: Subgradient descent (fast, local)
    t1 = min(time_budget * 0.3, 60.0)
    a, obj = subgrad_descent(a, t1, top_k=10, lr0=0.005)
    if obj < best_obj:
        best_a = a.copy()
        best_obj = obj

    # Phase 2: LP step (global jump)
    elapsed = time.monotonic() - start
    if elapsed < time_budget * 0.6:
        t2 = min(time_budget * 0.2, 30.0)
        a, obj = lp_phase(a, t2)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj
            print(f"    LP improved: {best_obj:.6f}")

    # Phase 3: More subgradient from LP result
    elapsed = time.monotonic() - start
    if elapsed < time_budget * 0.8:
        t3 = min(time_budget * 0.2, 30.0)
        a, obj = subgrad_descent(a, t3, top_k=15, lr0=0.003)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj

    # Phase 4: Targeted swap refinement
    elapsed = time.monotonic() - start
    if elapsed < time_budget * 0.95:
        t4 = min(time_budget * 0.15, 30.0)
        a, obj = targeted_swap_refine(a, t4, rng)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj

    # Phase 5: Simulated annealing
    elapsed = time.monotonic() - start
    if elapsed < time_budget * 0.99:
        t5 = time_budget - elapsed - 1.0
        if t5 > 5:
            a, obj = block_anneal(a, t5, rng)
            if obj < best_obj:
                best_a = a.copy()
                best_obj = obj

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-strategy search: subgradient descent + LP + targeted swaps + annealing.
    Tries parent optimization and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 8.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        parent = None

    # Optimize parent
    if parent is not None and parent.size <= 500:
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining * 0.3, 300.0)
        if t_alloc > 30:
            print(f"Optimizing parent with {t_alloc:.0f}s budget...")
            a_opt, obj_opt = optimize_single(parent.copy(), t_alloc, rng)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Parent optimized: obj={best_score:.6f}")

    # Try other sequence lengths
    n_values = [200, 250, 300, 350, 400, 500]

    for n in n_values:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        num_remaining = len([x for x in n_values if x >= n])
        time_per_n = remaining / max(1, num_remaining)
        time_per_n = min(time_per_n, 200.0)

        num_restarts = max(1, min(3, int(time_per_n / 80)))
        time_per_restart = time_per_n / num_restarts

        print(f"\nN={n}: remaining={remaining:.0f}s, {num_restarts} restarts, "
              f"{time_per_restart:.0f}s each")

        for r in range(num_restarts):
            if time.monotonic() > deadline - 30:
                break

            choice = (r + n) % 4
            if choice == 0:
                a_init = edge_init(n, rng)
            elif choice == 1:
                a_init = sinusoidal_init(n, rng)
            elif choice == 2:
                a_init = two_bump_init(n, rng)
            else:
                a_init = random_structured_init(n, rng)

            a_opt, obj_opt = optimize_single(a_init, time_per_restart, rng)

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} r={r}: obj={obj_opt:.6f} ** NEW BEST **")
            else:
                print(f"  n={n} r={r}: obj={obj_opt:.6f}")

    # Final deep refinement on best found
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: {remaining:.0f}s remaining")

        # Additional subgradient pass
        if remaining > 60:
            a_ref, obj_ref = subgrad_descent(best_seq.copy(), min(remaining * 0.3, 120.0),
                                            top_k=20, lr0=0.002)
            if obj_ref < best_score:
                best_score = obj_ref
                best_seq = a_ref.copy()
                print(f"  After final subgrad: obj={best_score:.6f}")

        # Additional swap refinement
        if time.monotonic() < deadline - 30:
            t_swap = min(deadline - time.monotonic() - 10, 60.0)
            a_ref, obj_ref = targeted_swap_refine(best_seq, t_swap, rng)
            if obj_ref < best_score:
                best_score = obj_ref
                best_seq = a_ref.copy()
                print(f"  After final swaps: obj={best_score:.6f}")

        # Final annealing
        if time.monotonic() < deadline - 15:
            t_anneal = deadline - time.monotonic() - 5
            a_ref, obj_ref = block_anneal(best_seq, t_anneal, rng)
            if obj_ref < best_score:
                best_score = obj_ref
                best_seq = a_ref.copy()
                print(f"  After final anneal: obj={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
