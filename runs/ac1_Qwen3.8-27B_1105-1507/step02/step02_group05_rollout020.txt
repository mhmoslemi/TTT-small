= 2n/S^3 * (S * dM/da[m] - 2M)

At the optimum (interior point, a[m] > 0): dJ/da[m] = 0 for all m.
This gives: S * dM/da[m] = 2M for all m with a[m] > 0.
And S * dM/da[m] >= 2M for m with a[m] = 0 (KKT condition).

This is a system of equations that characterizes the optimum. We could try to solve it directly!

At the optimum:

- For m with a[m] > 0: S * sum_{i=1}^p 2*a[k_i - m]*[0<=k_i-m<n] = 2M

This is a system of nonlinear equations (quadratic in a) coupled with the linear constraint sum(a) = S (or we can set S = 1).

"""
Minimizes J(a) = 2*n*max(conv(a,a))/sum(a)^2 using:
1. LP mirror descent with sparse constraints for broad exploration
2. Smooth log-sum-exp objective with L-BFGS-B for local refinement
3. Coordinate descent with KKT-guided search for final polish
4. Diverse restarts and multi-length exploration
Key improvements: smooth optimization with spectral gradient, adaptive tau annealing,
and efficient O(n log n) gradient computation via FFT.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[jl: jh + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_lp_iterations(a, max_iters, deadline, tracker):
    """Run LP mirror descent iterations with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        base_t = max(0.05, 0.5 * (1.0 - it / float(2 * max_iters)))
        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.02, base_t * 0.5), 0.3, 0.1]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        else:
            a = best_a
            current_J = best_J

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def smooth_objective_and_grad(x, tau):
    """Compute smoothed J and its gradient using log-sum-exp smoothing.
    
    J_smooth = 2n * (1/tau) * logsumexp(tau * conv(a,a)) / sum(a)^2
    """
    a = np.maximum(x, 0.0)
    n = len(a)
    S = a.sum()
    if S < 1e-8:
        return 1e10, np.zeros(n)
    
    a_norm = a / S
    c = fast_conv(a_norm)
    
    # Log-sum-exp for numerical stability
    max_c = c.max()
    exp_c = np.exp(tau * (c - max_c))
    sum_exp = exp_c.sum()
    if sum_exp < 1e-300:
        return 1e10, np.zeros(n)
    log_sum_exp = max_c + np.log(sum_exp)
    M_smooth = log_sum_exp / tau
    
    J_val = 2.0 * n * M_smooth
    
    # Compute gradient: dM_smooth/da[m]
    # softmax weights
    w = exp_c / sum_exp  # length 2n-1
    
    # grad_M[m] = 2 * sum_j w[m+j] * a_norm[j] for j where m+j < 2n-1
    # This is (reverse(a_norm) * w)[n-1+m]
    a_rev = a_norm[::-1]
    # Convolve a_rev with w
    s_fft = 1
    while s_fft < n + (2 * n - 1) - 1:
        s_fft <<= 1
    fa = np.fft.rfft(a_rev, s_fft)
    fw = np.fft.rfft(w, s_fft)
    conv_result = np.fft.irfft(fa * fw, s_fft)
    
    # conv_result[n-1+m] gives us sum_j a_rev[j] * w[n-1+m-j]
    # = sum_j a[n-1-j] * w[m+j] ... let me verify
    # (a_rev * w)[k] = sum_j a_rev[j] * w[k-j] = sum_j a[n-1-j] * w[k-j]
    # Let i = n-1-j: = sum_i a[i] * w[k-n+1+i]
    # We want: sum_i a[i] * w[m+i]
    # So k-n+1 = m, i.e., k = m+n-1
    grad_M = 2.0 * conv_result[n - 1: n - 1 + n]
    
    # dJ/da[m] = 2n/S^3 * (S * grad_M[m] - 2*M_smooth) [since J = 2n*M/S^2 and M depends on a/S]
    # Actually, let's be more careful. J = 2n * M_smooth(a/S) where M_smooth is the smoothed max of conv(a/S, a/S)
    # Let b = a/S. Then J = 2n * M_smooth(b).
    # dJ/da[m] = 2n * sum_k (dM_smooth/db[k]) * (db[k]/da[m])
    # db[k]/da[m] = (delta_km * S - b[k] * 1) / S^2 = (delta_km - b[k]) / S
    # dM_smooth/db[k] = tau * sum_c w[c] * dc[lag]/db[k] ... this is getting complex.
    # 
    # Simpler: J(a) = 2n * M_smooth(a) / S^2 where M_smooth(a) = (1/tau)*logsumexp(tau*conv(a,a))
    # dJ/da[m] = 2n * (S^2 * dM/da[m] - M_smooth * 2S) / S^4
    # = 2n/S^2 * (dM/da[m] - 2*M_smooth/S)
    # where dM/da[m] = tau * sum_k w[k] * dc[k]/da[m] = tau * grad_M[m] (computed above with a_norm)
    # But we need dM/da[m] not dM/db[m]. Since a = S*b:
    # M_smooth(a) = (1/tau)*logsumexp(tau*conv(a,a)) = S^2 * (1/tau)*logsumexp(tau*conv(b,b)) = S^2 * M_smooth(b)
    # So dM/da[m] = 2S*b[m]*M_smooth(b) + S^2 * dM_smooth(b)/db[m]
    # = 2S*a_norm[m]*M_smooth + S^2 * tau * grad_M[m]
    # 
    # This is getting complicated. Let me just use the formula directly.
    # J = 2n * M(S*a_norm) / S^2 where M(x) = (1/tau)*logsumexp(tau*conv(x,x))
    # M is homogeneous of degree 2: M(cx) = c^2 M(x)
    # So M(a) = M(S*a_norm) = S^2 * M(a_norm)
    # J = 2n * S^2 * M(a_norm) / S^2 = 2n * M(a_norm)
    # So J doesn't depend on S! It only depends on the direction a/S.
    # 
    # dJ/da[m] = 2n * dM(a_norm)/da[m] = 2n * (dM/db)[m] * db/da[m]
    # db/da[m] = (delta_m - b[m])/S = (delta_m - a[m])/S
    # dM/db[m] = tau * sum_k w[k] * dc[k]/db[m] = tau * 2 * sum_k w[k]*b[k-m]*[0<=k-m<n]
    # = tau * grad_M[m] (computed above)
    # 
    # So dJ/da[m] = 2n * tau * grad_M[m] * (delta_m - a[m])/S
    # But this doesn't seem right either because grad_M already incorporates the scaling.
    
    # Let me just compute it numerically-safe way:
    # J(a) = 2n * (1/tau) * log(sum_k exp(tau * c_k(a))) / S^2
    # where c_k(a) = sum_i a[i]*a[k-i]
    # 
    # dJ/da[m] = 2n * [S^2 * (1/tau) * (1/sum_exp) * sum_k tau*exp(tau*c_k)*dc_k/da[m] - 2S*M_unscaled] / S^4
    # where M_unscaled = (1/tau)*logsumexp(tau*c)
    # = 2n/S^4 * [S^2 * sum_k w_k * dc_k/da[m] - 2S * M_unscaled]
    # = 2n/S^2 * [sum_k w_k * dc_k/da[m] - 2*M_unscaled/S]
    # 
    # dc_k/da[m] = 2*a[k-m] if 0<=k-m<n (as derived before)
    # So sum_k w_k * dc_k/da[m] = 2 * sum_k w_k * a[k-m]*[0<=k-m<n]
    # = 2 * (the cross-correlation we computed, but with a instead of a_norm)
    
    # Recompute with a (not a_norm):
    a_rev_full = a[::-1]
    s_fft2 = 1
    while s_fft2 < n + (2 * n - 1) - 1:
        s_fft2 <<= 1
    fa2 = np.fft.rfft(a_rev_full, s_fft2)
    fw2 = np.fft.rfft(w, s_fft2)
    conv_result2 = np.fft.irfft(fa2 * fw2, s_fft2)
    sum_w_dc = 2.0 * conv_result2[n - 1: n - 1 + n]
    
    M_unscaled = (1.0 / tau) * log_sum_exp
    
    grad_J = 2.0 * n / (S * S) * (sum_w_dc - 2.0 * M_unscaled / S)
    
    return float(J_val), grad_J


def smooth_optimize(a_start, tau_schedule, deadline, tracker):
    """Optimize using L-BFGS-B on the smoothed objective with tau annealing."""
    a = np.maximum(a_start.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S
    
    current_best = compute_J(a)
    tracker.update(a)
    
    for tau in tau_schedule:
        if time.monotonic() > deadline:
            break
        
        def obj(x):
            val, _ = smooth_objective_and_grad(x, tau)
            return val
        
        def obj_grad(x):
            _, g = smooth_objective_and_grad(x, tau)
            return g
        
        bounds = [(0.0, None)] * len(a)
        try:
            res = minimize(obj_grad, a.copy(), method='L-BFGS-B',
                           bounds=bounds, jac=True,
                           options={'maxiter': 200, 'max_fun': 5000, 'ftol': 1e-18})
            if res.x is not None:
                a_trial = np.maximum(res.x, 0.0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    J_trial = compute_J(a_trial)
                    if J_trial < current_best:
                        current_best = J_trial
                        a = a_trial
                        tracker.update(a)
        except Exception:
            pass
    
    return a


def coordinate_descent(a, max_rounds, deadline, tracker):
    """Coordinate descent: optimize each coordinate by 1D search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a / S
    current_J = compute_J(a)
    tracker.update(a)
    n = len(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for m in range(n):
            if time.monotonic() > deadline:
                break
            # Try perturbing a[m]
            old_val = a[m]
            # Try several candidate values
            candidates = [0.0, old_val * 0.5, old_val * 0.75, old_val * 1.25, old_val * 1.5,
                          2.0 / n, 1.0 / n, 0.001, 0.01]
            # Also try values inspired by the KKT condition
            if old_val > 0:
                candidates.append(old_val * 0.9)
                candidates.append(old_val * 1.1)
            
            best_val = old_val
            best_J = current_J
            
            for cv in candidates:
                if cv < 0:
                    continue
                a[m] = cv
                s = a.sum()
                if s < 0.01:
                    a[m] = old_val
                    continue
                J_new = compute_J(a)
                if J_new < best_J - 1e-12:
                    best_J = J_new
                    best_val = cv
            a[m] = best_val
            if best_val != old_val:
                improved = True
                current_J = best_J
                a = a / a.sum()
        
        if not improved:
            break
        if rnd % 5 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    
    if parent is not None:
        inits.append(resample_seq(parent, n))
    
    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.3, 1.0, n))
    inits[-1] /= inits[-1].sum()
    
    t_arr = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 4, 5, 6, 7, 8, 10, 12]:
        init = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t_arr)
        init = np.maximum(init, 0.01)
        inits.append(init / init.sum())
    
    inits.append(np.abs(np.sin(np.pi * t_arr)) ** 3 + 0.01)
    inits[-1] /= inits[-1].sum()
    
    # Bell-shaped with offset
    for center in [0.3, 0.5, 0.7]:
        init = np.exp(-10.0 * (t_arr - center) ** 2) + 0.01
        inits.append(init / init.sum())
    
    # Two-hump
    init = np.exp(-20.0 * (t_arr - 0.25) ** 2) + np.exp(-20.0 * (t_arr - 0.75) ** 2) + 0.01
    inits.append(init / init.sum())
    
    # Frequency-modulated
    init = 1.0 + 0.3 * np.sin(4.0 * np.pi * t_arr * t_arr)
    init = np.maximum(init, 0.01)
    inits.append(init / init.sum())
    
    # Sawtooth
    init = np.abs(2.0 * t_arr - 1.0) + 0.1
    inits.append(init / init.sum())
    
    # Random with smoothing
    for _ in range(3):
        raw = rng.random(n)
        kernel = np.ones(5) / 5.0
        sm = np.convolve(raw, kernel, mode='same')
        sm = np.maximum(sm, 0.01)
        inits.append(sm / sm.sum())
    
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: LP mirror descent across lengths for broad exploration
    Phase 2: Smooth L-BFGS-B with log-sum-exp annealing
    Phase 3: Coordinate descent polish
    Phase 4: Final LP + smooth refinement on best
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass
    
    # === Phase 1: LP Mirror Descent ===
    lengths = [64, 128, 256, 512, 1024, 2048]
    
    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n{'='*50} LP Phase n={n} {'='*50}", flush=True)
        
        inits = generate_initializations(n, rng, parent)
        iters_per_init = max(30, min(150, int(60000 // n)))
        
        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 /= a0.sum()
            a = run_lp_iterations(a0, iters_per_init, deadline, tracker)
        
        elapsed = time.monotonic() - t0
        tracker.report(f"Phase1 n={n}: {elapsed:.1f}s")
    
    if tracker.best is None:
        tracker.best = np.ones(128) / 128
        tracker.best_score = compute_J(tracker.best)
    
    # === Phase 2: Smooth L-BFGS-B with tau annealing ===
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Smooth Opt n={n} {'='*50}", flush=True)
        
        # Try multiple tau schedules
        tau_schedules = [
            [100.0, 50.0, 20.0, 10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.08, 0.05],
            [50.0, 20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1],
        ]
        
        for ts in tau_schedules:
            if time.monotonic() > deadline:
                break
            a_opt = smooth_optimize(tracker.best, ts, deadline, tracker)
        
        # Also try from perturbed versions
        for restart in range(5):
            if time.monotonic() > deadline:
                break
            noise_level = 0.05 * (restart + 1)
            a_pert = tracker.best + rng.normal(0, noise_level, n)
            a_pert = np.maximum(a_pert, 0.0)
            sp = a_pert.sum()
            if sp > 1e-10:
                a_pert /= sp
            a_opt = smooth_optimize(a_pert, tau_schedules[1], deadline, tracker)
        
        tracker.report(f"Smooth: {time.monotonic()-t0:.1f}s")
    
    # === Phase 3: Coordinate Descent ===
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Coord Descent n={n} {'='*50}", flush=True)
        
        max_rounds = max(5, min(50, int(120.0 / max(1, (time.monotonic() - tracker.t_start) / budget_s))))
        # Limit coordinate descent based on remaining time
        remaining = deadline - time.monotonic()
        if n <= 256:
            max_rounds = min(max_rounds, 30)
        elif n <= 512:
            max_rounds = min(max_rounds, 15)
        else:
            max_rounds = min(max_rounds, 8)
        
        a_cd = coordinate_descent(tracker.best, max_rounds, deadline, tracker)
        tracker.report(f"CoordDesc: {time.monotonic()-t0:.1f}s")
    
    # === Phase 4: Final refinement ===
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Final Refine n={n} {'='*50}", flush=True)
        
        # More LP iterations
        remaining_iters = max(20, int((deadline - time.monotonic()) * 2))
        a_final = run_lp_iterations(tracker.best, min(remaining_iters, 200), deadline, tracker)
        
        # One more smooth pass with very small tau
        if time.monotonic() < deadline:
            a_final = smooth_optimize(a_final, [1.0, 0.5, 0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03], deadline, tracker)
        
        # Very fine coordinate descent
        if time.monotonic() < deadline and n <= 256:
            a_final = coordinate_descent(a_final, 20, deadline, tracker)
        
        tracker.report(f"Final: {time.monotonic()-t0:.1f}s")
    
    # === Phase 5: Try different lengths from best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Length Exploration {'='*50}", flush=True)
        
        best_n = tracker.best.size
        nearby_lengths = [l for l in [best_n // 2, best_n * 2, best_n - 32, best_n + 32,
                                       best_n - 64, best_n + 64]
                         if 32 <= l <= 4096 and l != best_n]
        
        for nl in nearby_lengths[:4]:
            if time.monotonic() > deadline:
                break
            a_resam = resample_seq(tracker.best, nl)
            a_resam = np.maximum(a_resam, 0.001)
            a_resam /= a_resam.sum()
            # Quick LP
            a_try = run_lp_iterations(a_resam, 50, deadline, tracker)
            # Smooth
            if time.monotonic() < deadline:
                a_try = smooth_optimize(a_try, [5.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1], deadline, tracker)
        
        tracker.report(f"LengthExpl: {time.monotonic()-t0:.1f}s")
    
    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
    
    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    
    return [float(x) for x in tracker.best]
