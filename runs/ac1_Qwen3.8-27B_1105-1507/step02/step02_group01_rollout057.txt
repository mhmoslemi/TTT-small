Multi-phase optimization for step function coefficient search.
Uses direct sub-gradient descent on max-convolution, LP alternating,
coordinate descent, and simulated annealing refinement.
Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end+1] = a[k - j_end : k - j_start + 1][::-1]
    return A

def solve_lp(a, max_iter_lp=100):
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, 10.0)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M

def normalize(a):
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s

def subgrad_phase(a, max_time, lr_init=0.01):
    """Direct sub-gradient descent on max convolution."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 0, None)
    a = normalize(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    patience = 0
    
    for it in range(100000):
        if time.monotonic() - start > max_time:
            break
        
        conv = np.convolve(a, a)
        k_star = int(np.argmax(conv))
        max_b = conv[k_star]

        grad = np.zeros(n)
        for m in range(n):

            if 0 <= idx < n:

        # Update

        a_new = np.clip(a_new, 0, None)
        a_new = normalize(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            patience = 0
        else:

            patience += 1

                if lr < 1e-8:

    start = time.monotonic()
    stall = 0

    for it in range(n_iters_max):
        if time.monotonic() - start > max_time:
            break

        if use_minmax:
            g, t_opt = solve_minmax_lp(a)
            if g is None:
                break
        else:
            conv = np.convolve(a, a, mode='full')
            M = float(np.max(conv))
            g, g_sum = solve_maxsum_lp(a, M)
            if g is None:
                break
            if g_sum > 1e-10:
                g = g / g_sum
            else:
                break

        # Try several blend factors
        improved = False
        best_local_obj = best_obj
        best_local_a = None
        alphas = [0.3, 0.5, 0.7, 1.0]
        if stall > 5:
            alphas = [0.2, 0.3, 0.5, 0.7, 1.0]

        for alpha in alphas:
            a_new = (1.0 - alpha) * a + alpha * g
            a_new = np.clip(a_new, 0.0, None)
            s = np.sum(a_new)
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_local_obj:
                best_local_obj = obj
                best_local_a = a_new

        if best_local_a is not None:
            a = best_local_a
            if best_local_obj < best_obj:
                best_a = a.copy()
                best_obj = best_local_obj
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > 20:
            # Aggressive reset: try full replacement
            a_try = g.copy()
            s = np.sum(a_try)
            if s > 1e-10:
                a_try = a_try / s
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_a = a_try.copy()
                    best_obj = obj_try
                    a = a_try
                    stall = 0
                else:
                    break

    return best_a, best_obj


def subgrad_phase(a, max_time):
    """Direct sub-gradient descent on max convolution (for non-symmetric a)."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 0, None)
    a = normalize(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.005
    patience = 0

    for it in range(50000):
        if time.monotonic() - start > max_time:
            break

        conv = np.convolve(a, a, mode='full')
        k_star = int(np.argmax(conv))

        # Sub-gradient: grad[m] = 2*a[k_star - m] for valid m
        grad = np.zeros(n)
        m_start = max(0, k_star - n + 1)
        m_end = min(n - 1, k_star)
        grad[m_start:m_end + 1] = 2.0 * a[k_star - m_end: k_star - m_start + 1][::-1]

        a_new = a - lr * grad
        a_new = np.clip(a_new, 0.0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            patience = 0
        else:
            a = a_new
            patience += 1
            if patience > 80:
                lr *= 0.85
                patience = 0
                if lr < 1e-9:
                    break

    return best_a, best_obj


def softmax_phase(a, max_time, T_init=None):
    """Soft-max smoothed gradient descent."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 0, None)
    a = normalize(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if T_init is None:
        conv = np.convolve(a, a, mode='full')
        T_init = 0.05 * float(np.max(conv))

    T = T_init
    lr = 0.01
    patience = 0

    for it in range(30000):
        if time.monotonic() - start > max_time:
            break

        conv = np.convolve(a, a, mode='full')
        max_c = float(np.max(conv))
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -400.0, 400.0)
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        if sum_exp < 1e-300:
            break
        p = exp_vals / sum_exp

        # Gradient of soft-max w.r.t. a
        idx = np.arange(n)
        M = p[idx[:, None] + idx[None, :]]
        grad = (2.0 / T) * (M @ a)

        a_new = a - lr * grad
        a_new = np.clip(a_new, 0.0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            patience = 0
            if T > 0.001:
                T = max(0.001, T * 0.99)
        else:
            patience += 1
            lr *= 0.95
            if patience > 100:
                lr = 0.005
                T = max(0.001, T * 0.7)
                patience = 0
                if lr < 1e-10:
                    break

    return best_a, best_obj


def coordinate_descent(a, max_time, n_passes=10):
    """Coordinate descent: perturb individual coordinates."""
    n = len(a)
    best_a = a.copy().astype(np.float64)
    best_a = np.clip(best_a, 0, None)
    best_a = normalize(best_a)
    best_obj = compute_obj(best_a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    deltas = [0.1, -0.1, 0.05, -0.05, 0.2, -0.2, 0.3, -0.3]

    for pass_num in range(n_passes):
        if time.monotonic() - start > max_time:
            break
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - start > max_time:
                break
            best_local = best_obj
            best_val = best_a[j]
            for d in deltas:
                candidate = best_a.copy()
                candidate[j] = best_a[j] * (1.0 + d)
                if candidate[j] < 0:
                    candidate[j] = 0.0
                obj = compute_obj(candidate)
                if obj < best_local:
                    best_local = obj
                    best_val = candidate[j]
            if best_val != best_a[j]:
                best_a[j] = best_val
                best_a = normalize(best_a)
                if best_local < best_obj:
                    best_obj = best_local

    return best_a, best_obj


def block_anneal(a, max_time, rng=None):
    """Block perturbation with simulated annealing."""
    if rng is None:
        rng = np.random.default_rng(99999)
    n = len(a)
    best_a = a.copy().astype(np.float64)
    best_a = np.clip(best_a, 0, None)
    best_a = normalize(best_a)
    best_obj = compute_obj(best_a)
    current_a = best_a.copy()
    current_obj = best_obj
    start = time.monotonic()
    T_init = 0.001
    T = T_init

    it = 0
    while time.monotonic() - start < max_time:
        T = T_init * 0.995 ** it
        width = max(1, int(n * rng.uniform(0.02, 0.15)))
        pos = int(rng.integers(0, n - width + 1))
        candidate = current_a.copy()
        sigma = 0.05 + 0.1 * max(0.0, 1.0 - it / 50000.0)
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            it += 1
            continue
        candidate = candidate / s
        obj = compute_obj(candidate)

        delta = obj - current_obj
        if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            current_a = candidate
            current_obj = obj
            if obj < best_obj:
                best_a = candidate.copy()
                best_obj = obj

        it += 1

    return best_a, best_obj


def full_pipeline(a_init, time_budget):
    """Run the full optimization pipeline on a single initialization."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.clip(a, 0, None)
    a = normalize(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: LP alternating (main workhorse)
    lp_time = min(time_budget * 0.5, 60.0)
    if n <= 200:
        max_lp_iters = 200
    elif n <= 400:
        max_lp_iters = 100
    else:
        max_lp_iters = 50
    a, obj = lp_alternating(a, lp_time, use_minmax=True, n_iters_max=max_lp_iters)
    if obj < best_obj:
        best_a = a.copy()
        best_obj = obj

    # Phase 2: Second LP round (max-sum variant)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        lp_time2 = min(remaining * 0.25, 20.0)
        a2, obj2 = lp_alternating(a, lp_time2, use_minmax=False,
                                  n_iters_max=max(20, max_lp_iters // 2))
        if obj2 < best_obj:
            best_a = a2.copy()
            best_obj = obj2
            a = a2

    # Phase 3: Sub-gradient (useful for non-symmetric)
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 5:
        sg_time = min(remaining * 0.2, 15.0)
        a3, obj3 = subgrad_phase(a, sg_time)
        if obj3 < best_obj:
            best_a = a3.copy()
            best_obj = obj3
            a = a3

    # Phase 4: Soft-max gradient
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        sm_time = min(remaining * 0.2, 10.0)
        a4, obj4 = softmax_phase(a, sm_time)
        if obj4 < best_obj:
            best_a = a4.copy()
            best_obj = obj4
            a = a4

    # Phase 5: Final LP polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 3:
        lp_time3 = min(remaining * 0.5, 15.0)
        a5, obj5 = lp_alternating(a, lp_time3, use_minmax=True,
                                  n_iters_max=max(20, max_lp_iters // 3))
        if obj5 < best_obj:
            best_a = a5.copy()
            best_obj = obj5

    return best_a, best_obj


def refine_best(a, max_time, rng):
    """Intensive refinement on the best candidate found."""
    n = len(a)
    best_a = a.copy().astype(np.float64)
    best_a = np.clip(best_a, 0, None)
    best_a = normalize(best_a)
    best_obj = compute_obj(best_a)
    start = time.monotonic()

    # Coordinate descent
    cd_time = min(max_time * 0.3, 30.0)
    a_cd, obj_cd = coordinate_descent(best_a, cd_time, n_passes=20)
    if obj_cd < best_obj:
        best_a = a_cd.copy()
        best_obj = obj_cd

    # Another LP round
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        lp_time = min(remaining * 0.3, 30.0)
        a_lp, obj_lp = lp_alternating(best_a, lp_time, use_minmax=True,
                                      n_iters_max=100)
        if obj_lp < best_obj:
            best_a = a_lp.copy()
            best_obj = obj_lp

    # More coordinate descent
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        cd_time2 = min(remaining * 0.3, 20.0)
        a_cd2, obj_cd2 = coordinate_descent(best_a, cd_time2, n_passes=30)
        if obj_cd2 < best_obj:
            best_a = a_cd2.copy()
            best_obj = obj_cd2

    # Block annealing
    remaining = max_time - (time.monotonic() - start)
    if remaining > 5:
        ba_time = min(remaining * 0.5, 40.0)
        a_ba, obj_ba = block_anneal(best_a, ba_time, rng=rng)
        if obj_ba < best_obj:
            best_a = a_ba.copy()
            best_obj = obj_ba

    # Final LP polish
    remaining = max_time - (time.monotonic() - start)
    if remaining > 3:
        a_fin, obj_fin = lp_alternating(best_a, remaining * 0.7,
                                        use_minmax=True, n_iters_max=50)
        if obj_fin < best_obj:
            best_a = a_fin.copy()
            best_obj = obj_fin

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: diverse initializations -> LP alternating -> sub-gradient
    -> coordinate descent -> block annealing. Tries multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf

    # Try parent construction
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}", flush=True)
    except Exception:
        pass

    # Optimize parent quickly if available
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all() and parent.size <= 400:
            p_time = min(30.0, float(budget_s) * 0.05)
            p_opt, p_obj = full_pipeline(parent.copy(), p_time)
            if p_obj < best_score:
                best_score = p_obj
                best_seq = p_opt.copy()
                print(f"Parent optimized (n={parent.size}): obj={p_obj:.6f}", flush=True)
    except Exception:
        pass

    # Main search: try multiple n values
    n_values = [40, 60, 80, 100, 120, 150, 200, 250, 300, 400]
    variants_per_n = 3

    print(f"\nStarting main search. Deadline in {deadline - time.monotonic():.0f}s", flush=True)

    for ni, n in enumerate(n_values):
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        n_remaining = len(n_values) - ni
        time_per_n = remaining / max(1, n_remaining)
        time_per_n = min(time_per_n, 100.0)

        if time_per_n < 10:
            continue

        num_restarts = min(variants_per_n, max(1, int(time_per_n / 25)))
        time_per_restart = time_per_n / max(1, num_restarts) - 2

        print(f"\nN={n}: {num_restarts} restarts, ~{time_per_restart:.0f}s each", flush=True)

        for r in range(num_restarts):
            if time.monotonic() > deadline - 80:
                break
            variant = (r + ni) % 7
            a_init = get_initialization(n, variant, rng)
            obj_init = compute_obj(a_init)

            a_opt, obj_opt = full_pipeline(a_init, time_per_restart)

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} variant={variant}: obj={obj_opt:.6f} ** NEW BEST **",
                      flush=True)
            else:
                print(f"  n={n} variant={variant}: obj={obj_opt:.6f} (init={obj_init:.4f})",
                      flush=True)

    # Cross-n perturbation: take best and try small n changes
    if best_seq is not None and time.monotonic() < deadline - 60:
        best_n = len(best_seq)
        for delta_n in [-20, -10, 10, 20, -5, 5]:
            new_n = best_n + delta_n
            if new_n < 10 or new_n > 500:
                continue
            if time.monotonic() > deadline - 60:
                break
            # Pad or truncate
            if new_n > best_n:
                a_try = np.zeros(new_n)
                a_try[:best_n] = best_seq
                a_try[best_n:] = best_seq[-1] * 0.5
            else:
                a_try = best_seq[:new_n].copy()
            a_try = normalize(a_try)
            t_budget = min(20.0, (deadline - time.monotonic()) * 0.3)
            a_opt, obj_opt = full_pipeline(a_try, t_budget)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Cross-n perturbation (n={new_n}): obj={obj_opt:.6f} ** NEW BEST **",
                      flush=True)

    # Final intensive refinement on the best candidate
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: {remaining:.0f}s budget, n={len(best_seq)}", flush=True)
        a_ref, obj_ref = refine_best(best_seq, remaining, rng)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  After refinement: obj={best_score:.6f}", flush=True)
        else:
            print(f"  Refinement: no improvement (obj={obj_ref:.6f})", flush=True)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_obj(best_seq)

    # Final validation
    final_score = compute_obj(best_seq)
    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={final_score:.6f}, time={elapsed:.1f}s", flush=True)

    return [float(x) for x in best_seq]
