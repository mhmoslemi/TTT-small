"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Softmax-smoothed L-BFGS-B for smooth navigation
2. LP best-response for non-smooth refinement
3. Subgradient descent on the true objective
4. Multi-scale approach (coarse to fine) with diverse asymmetric inits
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_max(C, tau):
    """Softmax approximation of max(C)."""
    C_max = C.max()
    return C_max + tau * np.log(np.sum(np.exp((C - C_max) / tau)))


def smoothed_objective(a, tau):
    """Smoothed version of max(conv(a,a)) for gradient-based optimization."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    a = a / s
    C = fast_conv(a)
    return softmax_max(C, tau)


def smoothed_gradient(a, tau):
    """Gradient of smoothed objective w.r.t. a (before normalization)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.zeros_like(a)
    a_norm = a / s
    C = fast_conv(a_norm)
    # Softmax weights
    C_max = C.max()
    w = np.exp((C - C_max) / tau)
    w /= w.sum()
    # d(softmax_max)/dC = w
    # dC(k)/da(m) = a(k-m) + a(m) (with boundary handling)
    # So gradient w.r.t. a_norm is: sum_k w(k) * (a_norm(k-m) + a_norm(m))
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            k_minus_j = k - j
            if 0 <= k_minus_j < n and k_minus_j != j:
                grad[j] += w[k] * a_norm[k_minus_j]
                grad[k_minus_j] += w[k] * a_norm[j]
            elif 0 <= k_minus_j < n and k_minus_j == j:
                grad[j] += 2 * w[k] * a_norm[j]
    # Chain rule: d/dam of f(a/s) where a is pre-normalization
    # = (1/s) * grad[m] - (a[m]/s^2) * sum(grad)
    grad = grad / s - a * (grad @ a) / (s * s)
    return grad

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Softmax-annealed L-BFGS-B with analytical gradient (FFT-based cross-correlation)
2. LP best-response refinement (sparse HiGHS)
3. Projected subgradient descent on the true objective
4. Multi-scale search with diverse asymmetric initializations
5. Random perturbation restarts from best solution
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def softmax_max(C, tau):
    """Compute softmax approximation of max(C) and the softmax weights."""
    C_max = C.max()
    exp_C = np.exp((C - C_max) / tau)
    w = exp_C / exp_C.sum()
    val = C_max + tau * np.log(exp_C.sum())
    return val, w


def smooth_obj_grad(x, tau):
    """Analytical objective and gradient for softmax-smoothed max-conv problem."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    b = a / s
    C = fast_conv(b)
    C_max = C.max()
    exp_C = np.exp((C - C_max) / tau)
    w = exp_C / exp_C.sum()
    obj = float(C_max + tau * np.log(exp_C.sum()))
    
    n = len(b)
    N = 1
    while N < 3 * n:
        N <<= 1
    W = np.fft.rfft(w, N)
    B = np.fft.rfft(b, N)
    corr = np.fft.irfft(W * np.conj(B), N)[:n]
    g_b = 2.0 * corr
    gb = float(np.dot(g_b, b))
    grad = np.where(x > 0, (g_b - gb) / s, 0.0)
    return obj, grad


def lbfgs_smooth(x0, tau_seq, deadline, tracker):
    """Run L-BFGS-B with annealed softmax temperature."""
    x = x0.copy().astype(np.float64)
    x = np.maximum(x, 0.01)
    
    for tau in tau_seq:
        if time.monotonic() > deadline:
            break
        
        # Local function for minimize
        def fun(x):
            return smooth_obj_grad(x, tau)[0]
        
        def fprime(x):
            return smooth_obj_grad(x, tau)[1]
        
        res = minimize(fun, x, method='L-BFGS-B', jac=fprime,
                       bounds=[(0.0, None)] * len(x),
                       options={'maxiter': 300, 'max_fun': 5000, 'ftol': 1e-14, 'gtol': 1e-10})
        
        if res.x is not None:
            x_new = np.maximum(res.x, 0.0)
            if x_new.sum() > 0.01:
                x = x_new
                tracker.update(x)
    
    return x


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    if not rows:
        return lil_matrix((2*n-1, n))
    A = lil_matrix((2 * n - 1, n))
    A[rows, cols] = vals
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None
