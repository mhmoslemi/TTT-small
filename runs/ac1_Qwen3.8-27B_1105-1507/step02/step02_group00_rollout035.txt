import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def subgradient_of_max_conv(a, tol=0.01):
    """Compute subgradient of max_k (a*a)_k at point a."""
    n = len(a)
    h = np.convolve(a, a)
    max_h = np.max(h)
    if max_h < 1e-15:
        return np.zeros(n), max_h
    # Active set
    active = h >= max_h * (1 - tol)
    n_active = np.sum(active)
    if n_active == 0:
        active = np.zeros(len(h), dtype=bool)
        active[np.argmax(h)] = True
        n_active = 1
    
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        # d/d(a_i) (a*a)_k = 2*a_{k-i} for valid i
        # i ranges from max(0, k-n+1) to min(n-1, k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # a[k-i] for i in [i_start, i_end]
        # k-i ranges from k-i_end to k-i_start
        vals = a[k - i_end : k - i_start + 1][::-1]
        grad[i_start:i_end+1] += 2.0 * vals
    grad /= n_active
    return grad, max_h

def proj_simplex(v, target_sum=1.0):
    """Project v onto {a : sum(a) = target_sum, a >= 0}."""
    n = len(v)
    # Standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, n+1) > (cssv - target_sum))
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def frank_wolfe(a, n_iter=10000, tol=0.01):
    """Frank-Wolfe method for min max_conv(a) s.t. a in simplex."""
    n = len(a)
    a = a / np.sum(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(n_iter):
        grad, max_h = subgradient_of_max_conv(a, tol=tol)
        # Frank-Wolfe: linear minimization over simplex
        # min <grad, v> s.t. v in simplex => v = e_{argmin grad}
        j = np.argmin(grad)
        # Step size
        if it % 100 == 0:
            t = 2.0 / (it + 2)
        else:
            t = 0.1  # smaller step for stability
        a_new = (1 - t) * a + t * np.eye(n)[j]
        # Hmm, this is O(n) per step to create the unit vector, but we can be smarter
        a_new = (1 - t) * a
        a_new[j] += t
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s > 0:
            a_new /= s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    return best_a, best_obj

"""
Multi-phase optimization: LP with constraint generation + accelerated subgradient + coordinate refinement.
Targets objective < 1.50 for sequences of length ~300.
"""
import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A_rows(a, rows):
    """Build constraint matrix for specific rows of the convolution constraint."""
    n = len(a)
    A = np.zeros((len(rows), n), dtype=np.float64)
    for idx, k in enumerate(rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[idx, j_start:j_end+1] = a[k - j_end:k - j_start + 1][::-1]
    return A

def solve_lp_cg(a, max_cg=15, init_tol=0.3):
    """LP with constraint generation: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    conv_aa = np.convolve(a, a)
    M = float(np.max(conv_aa))
    if M < 1e-15:
        return None
    
    # Initial active set
    active_mask = conv_aa >= M * (1 - init_tol)
    active_rows = np.where(active_mask)[0]
    
    g = None
    for cg in range(max_cg):
        A = build_A_rows(a, active_rows)
        c = -np.ones(n)
        b_ub = np.full(len(active_rows), M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        except:
            return None
        if res.status != 0:
            return None
        g = res.x
        if g is None or np.sum(g) < 1e-15:
            return None
        
        # Check all constraints
        conv_ag = np.convolve(a, g)
        violated = conv_ag > M * (1 + 1e-10)
        if not np.any(violated):
            return g
        # Add violated constraints
        new_active = active_mask | violated
        if np.sum(new_active) == np.sum(active_mask):
            return g  # No new constraints to add
        active_mask = new_active
        active_rows = np.where(active_mask)[0]
    
    return g

def lp_alternate(a, max_iter=200, deadline=None):
    """LP alternation with geometric mean blending."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        g = solve_lp_cg(a, max_cg=15, init_tol=max(0.05, 0.3 * (1 - it/max_iter)))
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Adaptive blending: geometric mean with decreasing alpha
        alpha = 0.3 + 0.3 * (it / max_iter)  # goes from 0.3 to 0.6
        a_new = np.power(a, alpha) * np.power(g, 1 - alpha)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / np.sum(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj

def subgradient_polish(a, n_steps=50000, deadline=None, verbose=False):
    """Accelerated subgradient with Nesterov momentum."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / np.sum(a)
    
    y = a.copy()  # momentum variable
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute subgradient at y
        h = np.convolve(y, y)
        max_h = np.max(h)
        if max_h < 1e-15:
            break
        
        # Active set with adaptive tolerance
        tol_frac = 0.01 + 0.04 * (step / n_steps)
        active = h >= max_h * (1 - tol_frac)
        if not np.any(active):
            active = np.zeros(len(h), dtype=bool)
            active[np.argmax(h)] = True
        n_active = int(np.sum(active))
        
        # Subgradient
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            vals = y[k - i_end:k - i_start + 1][::-1]
            grad[i_start:i_end+1] += 2.0 * vals
        grad /= n_active
        
        # Nesterov step size
        t = 2.0 / (step + 2.0)
        y_new = y - t * grad
        y_new = np.maximum(y_new, 0.0)
        s = np.sum(y_new)
        if s < 1e-10:
            break
        y_new = y_new / s
        
        # Momentum update
        beta = (step + 1.5) / (step + 2.5)  # standard Nesterov momentum
        y = beta * y_new + (1 - beta) * a
        y = np.maximum(y, 0.0)
        s = np.sum(y)
        if s < 1e-10:
            break
        y = y / s
        
        a = y_new
        
        if step % 1000 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
            if verbose:
                print(f"    SG step {step}: obj={best_obj:.6f}")
    
    # Final check
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()
    
    return best_a, best_obj

def coord_refine(a, n_rounds=5, deadline=None):
    """Coordinate descent with golden section search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            old_val = a[j]
            # Try several values
            candidates = [0.0, old_val * 0.3, old_val * 0.5, old_val * 0.7,
                         old_val, old_val * 1.3, old_val * 1.5, old_val * 2.0]
            best_x = old_val
            best_local = best_obj
            
            for x in candidates:
                a[j] = max(0.0, x)
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_x = max(0.0, x)
            a[j] = best_x
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj

"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP with constraint generation, accelerated subgradient, and coordinate refinement.

adaptive exponent, Nesterov-momentum subgradient, and block coordinate LP alternation.

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A_rows(a, rows):
    """Build constraint matrix A where A[idx,j] = a[rows[idx]-j] if valid."""
    n = len(a)
    A = np.zeros((len(rows), n), dtype=np.float64)
    for idx, k in enumerate(rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[idx, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_full(a, fixed_mask=None):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M, g>=0, optionally with fixed zeros.
    Uses constraint generation for speed."""
    n = len(a)
    conv_aa = np.convolve(a, a)
    M = float(np.max(conv_aa))
    if M < 1e-18:
        return None

    # Initial active set: top constraints
    active_mask = conv_aa >= M * 0.7
    if not np.any(active_mask):
        active_mask = np.zeros(len(conv_aa), dtype=bool)
        active_mask[np.argmax(conv_aa)] = True

    g = None
    for cg_iter in range(20):
        active_rows = np.where(active_mask)[0]
        A = build_A_rows(a, active_rows)
        c = -np.ones(n)
        b_ub = np.full(len(active_rows), M)
        bounds = [(0.0, None)] * n
        if fixed_mask is not None:
            for j in range(n):
                if fixed_mask[j]:
                    bounds[j] = (0.0, 0.0)
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            return None
        if res.status != 0 or res.x is None:
            return None
        g = res.x
        if g is None or np.sum(g) < 1e-15:
            return None

        # Check all constraints
        conv_ag = np.convolve(a, g)
        violated = conv_ag > M * (1 + 1e-9)
        if not np.any(violated):
            return g
        new_mask = active_mask | violated
        if np.sum(new_mask) == np.sum(active_mask):
            return g
        active_mask = new_mask

    return g


def lp_alternate(a, max_iter=200, deadline=None, verbose=False):
    """LP alternation with adaptive geometric-mean blending."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        g = solve_lp_full(a)
        if g is None:
            break
        gs = np.sum(g)
        if gs < 1e-10:
            break
        g = g / gs

        alpha = 0.4 + 0.35 * min(1.0, it / max_iter)

def compute_subgradient(y, n, active_tol=0.02):
    """Compute subgradient of max_k (y*y)_k at point y."""
    h = np.convolve(y, y)
    max_h = float(np.max(h))
    if max_h < 1e-18:
        return np.zeros(n), max_h

    active = h >= max_h * (1 - active_tol)
    if not np.any(active):
        active = np.zeros(len(h), dtype=bool)
        active[np.argmax(h)] = True
    n_active = int(np.sum(active))

    grad = np.zeros(n)
    active_idx = np.where(active)[0]
    for k in active_idx:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        vals = y[k - i_end:k - i_start + 1][::-1]
        grad[i_start:i_end + 1] += 2.0 * vals
    grad /= n_active
    return grad, max_h


def subgradient_polish(a, n_steps=50000, deadline=None, verbose=False):
    """
    Modified Frank-Wolfe with subgradient descent.
    Uses: a_new = proj_simplex(a - step * subgrad), with adaptive step size.
    Also uses "best iterate" tracking.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    # We'll use a combination: subgradient step + occasional "pull toward uniform"
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Adaptive tolerance: start loose, tighten
        tol = 0.05 * (1 - step / n_steps) + 0.005
        grad, max_h = compute_subgradient(a, n, active_tol=tol)

        # Step size: decreasing with exploration
        base_step = 0.15 / (1.0 + 0.005 * step)
        # Add small random perturbation for exploration
        if step % 500 == 0 and step > 0:
            perturbation = np.random.default_rng(step).normal(0, 0.01, n)
            grad = grad + perturbation / (np.max(np.abs(grad)) + 1e-10)

        a_new = a - base_step * grad
        a_new = np.maximum(a_new, 0.0)
        sn = np.sum(a_new)
        if sn < 1e-10:
            break
        a_new = a_new / sn

        # Track best
        if step % 50 == 0:
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose and step % 500 == 0:
                    print(f"    SG step={step}: obj={best_obj:.7f}")
        a = a_new

    # Final evaluation
    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a, best_obj


def fw_polish(a, n_steps=30000, deadline=None):
    """
    Frank-Wolfe: at each step, move toward the vertex e_j where j=argmin(subgrad).
    Uses backtracking line search for step size.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        tol = 0.02 * (1 - step / n_steps) + 0.003
        grad, _ = compute_subgradient(a, n, active_tol=tol)

        # FW direction: move toward e_j where j = argmin(grad)
        j = int(np.argmin(grad))
        direction = np.zeros(n)
        direction[j] = 1.0
        direction = direction - a  # direction from a to e_j

        # Backtracking line search
        best_t = 0.0
        best_obj_trial = compute_obj(a)
        t = 0.5
        for _ in range(8):
            a_trial = a + t * direction
            a_trial = np.maximum(a_trial, 0.0)
            st = np.sum(a_trial)
            if st < 1e-10:
                t /= 2
                continue
            a_trial = a_trial / st
            obj_trial = compute_obj(a_trial)
            if obj_trial < best_obj_trial:
                best_obj_trial = obj_trial
                best_t = t
            t /= 2

        if best_t < 1e-6:
            # No improvement found; try a smaller gradient step
            step_size = 0.05 / (1.0 + 0.01 * step)
            a_new = a - step_size * grad
            a_new = np.maximum(a_new, 0.0)
            sn = np.sum(a_new)
            if sn < 1e-10:
                break
            a_new = a_new / sn
        else:
            a_new = a + best_t * direction
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / np.sum(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_obj


def coord_refine(a, n_rounds=3, deadline=None):
    """Coordinate descent with multiple trial values per coordinate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-15:
                # Try turning on this coordinate
                candidates = [0.0, np.max(a) * 0.1, np.max(a) * 0.3, np.max(a) * 0.5]
            else:
                candidates = [0.0, old_val * 0.2, old_val * 0.5, old_val * 0.8,
                             old_val, old_val * 1.2, old_val * 1.5, old_val * 2.0]
            best_x = old_val
            best_local = best_obj
            for x in candidates:
                a[j] = max(0.0, x)
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_x = max(0.0, x)
            a[j] = best_x

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj

    s = np.sum(a)

        print(f"  Phase 1 result: obj={best_score:.7f}")
    else:
        print(f"  Phase 1 no improvement (lp={obj1:.7f})")
        # Try block LP
        a1b, obj1b = block_lp_alternate(parent, max_iter=100, deadline=phase1_deadline)
        if obj1b < best_score:
            best_score = obj1b
            best_seq = a1b.copy()
            print(f"  Block LP result: obj={best_score:.7f}")

    # === Phase 2: More LP from the best so far (15% budget) ===
    phase2_deadline = start_time + total_budget * 0.55
    if best_score > 1.45:
        print(f"\nPhase 2: Further LP (n={len(best_seq)})")
        a2, obj2 = lp_alternate(best_seq, max_iter=300, deadline=phase2_deadline, verbose=True)
        if obj2 < best_score:
            best_score = obj2
            best_seq = a2.copy()
            print(f"  Phase 2 result: obj={best_score:.7f}")

    # === Phase 3: Subgradient + FW polishing (20% budget) ===
    phase3_deadline = start_time + total_budget * 0.75
    print(f"\nPhase 3: Subgradient polishing (n={len(best_seq)})")
    a3, obj3 = subgradient_polish(best_seq, n_steps=100000, deadline=phase3_deadline, verbose=True)
    if obj3 < best_score:
        best_score = obj3
        best_seq = a3.copy()
        print(f"  SG result: obj={best_score:.7f}")

    # FW polish
    if time.monotonic() < phase3_deadline - 30:
        a3b, obj3b = fw_polish(best_seq, n_steps=50000, deadline=phase3_deadline)
        if obj3b < best_score:
            best_score = obj3b
            best_seq = a3b.copy()
            print(f"  FW result: obj={best_score:.7f}")

    # === Phase 4: Coordinate refinement (10% budget) ===
    phase4_deadline = start_time + total_budget * 0.85
    n_best = len(best_seq)
    if n_best <= 600:
        print(f"\nPhase 4: Coordinate refinement (n={n_best})")
        rounds = 5 if n_best <= 300 else 2
        a4, obj4 = coord_refine(best_seq, n_rounds=rounds, deadline=phase4_deadline)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  Coord result: obj={best_score:.7f}")

    # === Phase 5: Try alternative n values (15% budget) ===
    remaining = deadline - time.monotonic()
    if remaining > 45:
        print(f"\nPhase 5: Alternative lengths (budget={remaining:.0f}s)")
        n_options = []
        for n_try in [200, 250, 350, 400, 500]:
            if n_try != n_parent:
                n_options.append(n_try)

        per_option = remaining / (len(n_options) + 1)
        for n_try in n_options:
            if time.monotonic() > deadline - 15:
                break
            sub_deadline = min(deadline - 10, time.monotonic() + per_option - 5)
            # Initialize by interpolation from best
            a_init = np.interp(
                np.linspace(0, 1, n_try),
                np.linspace(0, 1, len(best_seq)),
                best_seq
            )
            a_init = np.maximum(a_init, 0.0)
            a_init = a_init / np.sum(a_init)

            a_try, obj_try = lp_alternate(a_init, max_iter=80, deadline=sub_deadline)
            if time.monotonic() < sub_deadline - 5:
                a_try, obj_try = subgradient_polish(a_try, n_steps=5000, deadline=sub_deadline)

            if obj_try < best_score:
                best_score = obj_try
                best_seq = a_try.copy()
                print(f"  n={n_try}: obj={best_score:.7f} (NEW BEST!)")
            else:
                print(f"  n={n_try}: obj={obj_try:.7f} (no improvement)")

    # === Final: one more subgradient pass ===
    if time.monotonic() < deadline - 3:
        a_final, obj_final = subgradient_polish(best_seq, n_steps=30000, deadline=deadline - 1)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"\nFinal SG: obj={best_score:.7f}")

    elapsed = time.monotonic() - start_time
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_seq)}, score={best_score:.7f}, time={elapsed:.1f}s")
    print(f"{'='*50}")

    # Ensure valid output
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    if np.sum(best_seq) < 0.01:
        best_seq = np.ones(100) * 0.01

    return [float(x) for x in best_seq]
