Actually, let me reconsider the 1D subproblem solver. Instead of a general 1D search, I can exploit the structure:

$G(t) = \max_k (r_k + 2t w_k + t^2 v_k) / (S+t)^2$

The numerator $N(t) = \max_k (r_k + 2t w_k + t^2 v_k)$ is the max of $O(n)$ affine functions (plus one quadratic if $v_k = 1$ for some $k$).

The optimal $t^*$ satisfies $G'(t^*) = 0$ (interior) or is at a boundary.

$G'(t) = [N'(t)(S+t)^2 - N(t) \cdot 2(S+t)] / (S+t)^4 = [N'(t)(S+t) - 2N(t)] / (S+t)^3$

So $G'(t) = 0$ iff $N'(t)(S+t) = 2N(t)$.

If the active constraint at the optimum is a single line $g_k(t) = r_k + 2tw_k$ (with $v_k = 0$), then $N(t) = r_k + 2tw_k$ and $N'(t) = 2w_k$, so:
$2w_k(S+t) = 2(r_k + 2tw_k)$
$w_k S + w_k t = r_k + 2tw_k$
$w_k S - r_k = tw_k$
$t = (w_k S - r_k) / w_k = S - r_k/w_k$

So if a single line is active, the optimal $t$ is $S - r_k/w_k$ (if positive and the line is indeed active there).

If the active constraint is the parabola $g_{2i}(t) = r_{2i} + t^2$, then $N(t) = r_{2i} + t^2$ and $N'(t) = 2t$:
$2t(S+t) = 2(r_{2i} + t^2)$
$tS + t^2 = r_{2i} + t^2$
$tS = r_{2i}$
$t = r_{2i}/S$

If two lines are active (both at the maximum), say $g_j$ and $g_k$:
$r_j + 2tw_j = r_k + 2tw_k$ at the crossing point
$t = (r_k - r_j) / (2(w_j - w_k))$
And we need to check that at this $t$, these two are indeed the maximum.

So the algorithm for the 1D subproblem is:
1. For each pair of "potentially active" lags, compute the candidate $t$
2. Also check $t = 0$ and large $t$
3. Evaluate $G(t)$ at all candidates and pick the best

But there are $O(n^2)$ pairs, which is too many. Instead, we can:
1. Identify the lags that are near the maximum (within some tolerance)
2. Only consider pairs among those
3. Also use a coarse grid + refinement

"""
Optimized step function coefficient search using coordinate descent with exact 1D
subproblem solving, LP-based refinement, and block perturbation.
Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute the objective function value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def solve_coord_1d(a, i):
    """
    Solve the 1D subproblem for coordinate i:
    min_{t >= 0} max_k (a(t)*a(t))_k / (sum a(t))^2
    
    where a(t) is a with a[i] replaced by t.
    Returns optimal t.
    """
    n = len(a)
    a_i = a[i]
    S = float(np.sum(a) - a_i)  # sum of all other elements
    
    if S < 1e-15:
        return a_i
    
    # Current autocorrelation
    conv = np.convolve(a, a)
    
    # Compute r[k] = autocorrelation with a[i] removed
    # r[k] = conv[k] - 2*a[i]*a[k-i] (for k != 2i, valid range)
    # r[k] = conv[k] - a[i]^2 (for k = 2i)
    r = conv.copy()
    w = np.zeros(2*n-1)
    v = np.zeros(2*n-1)
    
    # For k where 0 <= k-i < n and k != 2i:
    #   conv[k] includes 2*a[i]*a[k-i], so r[k] = conv[k] - 2*a[i]*a[k-i]
    #   and (a(t)*a(t))[k] = r[k] + 2*t*a[k-i], so w[k] = a[k-i]
    # For k = 2i:
    #   conv[k] includes a[i]^2, so r[k] = conv[k] - a[i]^2
    #   and (a(t)*a(t))[k] = r[k] + t^2, so v[k] = 1
    
    # Valid range for k-i: 0 <= k-i < n => i <= k <= i+n-1
    k_start = max(0, i)
    k_end = min(2*n-2, i+n-1)
    
    for k in range(k_start, k_end+1):
        j = k - i  # = k-i, this is the index in a
        if j == i:  # k = 2i
            r[k] = conv[k] - a_i * a_i
            v[k] = 1.0
        else:
            r[k] = conv[k] - 2.0 * a_i * a[j]
            w[k] = a[j]
    
    # Now solve min_{t>=0} max_k (r[k] + 2*t*w[k] + t^2*v[k]) / (S+t)^2
    # Since we're comparing ratios, and the 2n factor is constant, we minimize:
    # F(t) = max_k (r[k] + 2*t*w[k] + t^2*v[k]) / (S+t)^2
    
    # Candidate t values
    candidates = set()
    candidates.add(0.0)
    candidates.add(a_i)
    candidates.add(0.1 * a_i)
    candidates.add(10.0 * a_i)
    candidates.add(S * 0.01)
    candidates.add(S * 0.1)
    candidates.add(S)
    
    # Find the top lags at t=0
    max_r = float(np.max(r))
    threshold = max_r - 0.05 * max_r  # lags within 5% of max
    active_lags = np.where(r >= threshold)[0]
    
    for k in active_lags[:30]:  # limit to 30 for efficiency
        if w[k] > 1e-15:
            t_cand = S - r[k] / w[k]
            if t_cand > 0 and t_cand < 100 * S:
                candidates.add(float(t_cand))
        if v[k] > 1e-15:
            t_cand = r[k] / S
            if t_cand > 0 and t_cand < 100 * S:
                candidates.add(float(t_cand))
    
    # Also try pairs of active lags
    for idx1 in range(min(len(active_lags), 15)):
        for idx2 in range(idx1+1, min(len(active_lags), 15)):
            k1, k2 = active_lags[idx1], active_lags[idx2]
            # r[k1] + 2*t*w[k1] + t^2*v[k1] = r[k2] + 2*t*w[k2] + t^2*v[k2]
            # (2*w[k1]-2*w[k2])*t + (v[k1]-v[k2])*t^2 = r[k2]-r[k1]
            A = v[k1] - v[k2]
            B = 2.0*(w[k1] - w[k2])
            C = r[k2] - r[k1]
            if abs(A) > 1e-15:
                # Quadratic: A*t^2 + B*t - C = 0
                disc = B*B + 4*A*C
                if disc >= 0:
                    t1 = (-B + np.sqrt(disc)) / (2*A)
                    t2 = (-B - np.sqrt(disc)) / (2*A)
                    for t_cand in [t1, t2]:
                        if t_cand > 0 and t_cand < 100*S:
                            candidates.add(float(t_cand))
            elif abs(B) > 1e-15:
                t_cand = C / B
                if t_cand > 0 and t_cand < 100*S:
                    candidates.add(float(t_cand))
    
    candidates = np.array([t for t in candidates if t >= 0 and np.isfinite(t)])
    if len(candidates) == 0:
        candidates = np.array([0.0, a_i])
    
    # Evaluate F(t) at all candidates
    best_t = candidates[0]
    best_F = np.inf
    
    for t in candidates:
        val = r + 2.0*t*w + t*t*v
        max_val = float(np.max(val))
        F = max_val / (S + t)**2
        if F < best_F:
            best_F = F
            best_t = t
    
    # Golden section refinement around best_t
    # Determine bracket
    lo = max(0.0, best_t * 0.5 - 0.1 * S)
    hi = best_t * 2.0 + 0.1 * S
    if hi <= 0:
        hi = S
    if lo < 0:
        lo = 0.0
    
    # Make sure best_t is in [lo, hi]
    mid = (lo + hi) / 2
    for _ in range(50):
        gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
        x1 = lo + (1 - gr) * (hi - lo)
        x2 = lo + gr * (hi - lo)
        
        F1 = float(np.max(r + 2*x1*w + x1*x1*v)) / (S+x1)**2
        F2 = float(np.max(r + 2*x2*w + x2*x2*v)) / (S+x2)**2
        
        if F1 < F2:
            hi = x2
        else:
            lo = x1
        
        if hi - lo < 1e-12 * max(1, S):
            break
    
    best_t = (lo + hi) / 2
    best_F = float(np.max(r + 2*best_t*w + best_t*best_t*v)) / (S+best_t)**2
    
    # Check boundaries
    F_0 = float(np.max(r)) / (S)**2
    if F_0 < best_F:
        best_t = 0.0
    
    return float(best_t)


def coord_descent(a, max_time, order='random', rng=None):
    """
    Coordinate descent: iterate over coordinates, solving 1D subproblem for each.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if rng is None:
        rng = np.random.default_rng()
    
    it = 0
    stall = 0
    order_list = np.arange(n)
    
    while time.monotonic() - start < max_time:
        if order == 'random':
            order_list = rng.permutation(n)
        elif order == 'cyclic':
            order_list = np.arange(n)
        
        improved = False
        for i in order_list:
            if time.monotonic() - start > max_time:
                break
            t_opt = solve_coord_1d(a, i)
            if abs(t_opt - a[i]) > 1e-12:
                a[i] = max(0.0, t_opt)
                s = np.sum(a)
                if s < 1e-10:
                    a[i] = 0.0
                    continue
                a = a / s
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
        
        it += 1
        if not improved:
            stall += 1
            if stall > 3:
                break
        else:
            stall = 0
    
    return best_a, best_obj


def build_A_matrix(a):
    """Build the constraint matrix for the LP: (f*g)[k] <= M for all k."""
    n = len(a)
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n), dtype=np.float64)
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_phase(a, max_time, t_start=0.5, t_min=0.05):
    """LP-based direction finding and update."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    it = 0
    
    while time.monotonic() - start < max_time:
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        A = build_A_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0.0, 10.0)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.status == 0:
            g = res.x
            g_sum = float(np.sum(g))
            if g_sum <= 1.0 + 1e-6:
                stall += 1
            else:
                stall = 0
                g = g / g_sum
                t = max(t_min, t_start * (1.0 - it / 100.0))
                for tt in [t, t*0.5, t*0.25, t*2.0]:
                    a_new = (1.0 - tt) * a + tt * g
                    a_new = np.clip(a_new, 0.0, None)
                    s = np.sum(a_new)
                    if s < 1e-10:
                        continue
                    a_new = a_new / s
                    obj = compute_obj(a_new)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a_new.copy()
                        a = a_new.copy()
                        break
        else:
            stall += 1
        
        it += 1
        if stall > 10:
            break
    
    return best_a, best_obj


def block_perturb(a, rng, max_time):
    """Block perturbation refinement."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        width = max(1, int(n * rng.uniform(0.02, 0.15)))
        pos = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        sigma = 0.02 + 0.06 * (1.0 - (time.monotonic() - start) / max_time)
        candidate[pos:pos+width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_obj = obj
            best_a = candidate
    
    return best_a, best_obj

"""
Optimized step function coefficient search using coordinate descent with exact 1D
subproblem solving, LP-based refinement, and block perturbation.
Key improvement over previous: direct coordinate descent on the peak autocorrelation
with exact 1D minimax solution per coordinate.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute the objective function value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def solve_coord_1d(a, i, n_candidates_grid=20):
    """
    Solve the 1D subproblem for coordinate i:
    min_{t >= 0} max_k (a(t)*a(t))_k / (sum a(t))^2
    where a(t) replaces a[i] with t.
    Returns optimal t.
    """
    n = len(a)
    a_i = a[i]
    S = float(np.sum(a) - a_i)

    if S < 1e-15:
        return a_i

    # Current autocorrelation
    conv = np.convolve(a, a)
    num_lags = 2 * n - 1

    # Compute r[k], w[k], v[k]
    # (a(t)*a(t))[k] = r[k] + 2*t*w[k] + t^2*v[k]
    r = conv.copy()
    w = np.zeros(num_lags)
    v = np.zeros(num_lags)

    k_start = max(0, i)
    k_end = min(num_lags - 1, i + n - 1)

    for k in range(k_start, k_end + 1):
        j = k - i
        if 0 <= j < n:
            if j == i:  # k = 2i
                r[k] = conv[k] - a_i * a_i
                v[k] = 1.0
            else:
                r[k] = conv[k] - 2.0 * a_i * a[j]
                w[k] = a[j]

    def eval_F(t):
        val = r + 2.0 * t * w + t * t * v
        return float(np.max(val)) / (S + t) ** 2

    # Grid search for good initial bracket
    t_min = 0.0
    t_max = max(10.0 * S, 2.0 * a_i * 10.0, 1.0)

    grid = np.concatenate(([0.0], np.logspace(-4, np.log10(t_max), n_candidates_grid)))
    grid = np.clip(grid, 0.0, t_max)

    best_t_grid = 0.0
    best_F_grid = np.inf
    for t in grid:
        F = eval_F(t)
        if F < best_F_grid:
            best_F_grid = F
            best_t_grid = t

    # Add analytical candidates from active lags
    max_r = float(np.max(r))
    threshold = max_r - 0.1 * max_r
    active_lags = np.where(r >= threshold)[0]
    active_lags = active_lags[:20]  # limit

    candidates = [best_t_grid, 0.0, a_i]
    for k in active_lags:
        if w[k] > 1e-15:
            t_cand = S - r[k] / w[k]
            if 0 < t_cand < t_max:
                candidates.append(float(t_cand))
        if v[k] > 1e-15:
            t_cand = r[k] / S
            if 0 < t_cand < t_max:
                candidates.append(float(t_cand))

    # Pair candidates

        for idx2 in range(idx1 + 1, min(len(active_lags), 12)):

            A = v[k1] - v[k2]
            B = 2.0 * (w[k1] - w[k2])

                if disc >= 0:

                    for t_cand in [(-B + sq) / (2 * A), (-B - sq) / (2 * A)]:
                        if 0 < t_cand < t_max:

                if 0 < t_cand < t_max:
                    candidates.append(float(t_cand))

    # Evaluate all candidates

def coord_descent(a, max_time, rng=None, verbose=True):
    """
    Coordinate descent: iterate over coordinates, solving 1D subproblem for each.
    Uses random ordering and normalizes after each full pass.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if rng is None:
        rng = np.random.default_rng()

    stall = 0
    it = 0
    min_improve = 1e-9

    while time.monotonic() - start < max_time:
        order = rng.permutation(n)
        obj_before = best_obj
        any_change = False

        for i in order:
            if time.monotonic() - start > max_time:
                break
            t_opt = solve_coord_1d(a, i)
            if abs(t_opt - a[i]) > 1e-10:
                a[i] = max(0.0, t_opt)
                any_change = True

        # Normalize
        s = np.sum(a)
        if s < 1e-10:
            break
        a = a / s

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        it += 1
        if not any_change or (best_obj - obj_before) < min_improve:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - start
            print(f"    CD iter {it}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def build_A_matrix(a):
    """Build constraint matrix: (f*g)[k] for all k."""
    n = len(a)
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n), dtype=np.float64)
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_phase(a, max_time, verbose=True):
    """LP-based direction finding and update."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - start < max_time:
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        A = build_A_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, 10.0)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 10.0})
        except Exception:
            break

        if res.status != 0:
            stall += 1
            if stall > 5:
                break
            it += 1
            continue

        g = res.x
        g_sum = float(np.sum(g))
        if g_sum <= 1.0 + 1e-7:
            stall += 1
        else:
            stall = 0
            g = g / g_sum
            t = 0.3 * (1.0 - min(it / 50.0, 1.0)) + 0.05
            found = False
            for tt in [t, t * 0.5, t * 0.25, t * 2.0, t * 0.75]:
                a_new = (1.0 - tt) * a + tt * g
                a_new = np.clip(a_new, 0.0, None)
                s = np.sum(a_new)
                if s < 1e-10:
                    continue
                a_new = a_new / s
                obj = compute_obj(a_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new.copy()
                    a = a_new.copy()
                    found = True
                    break
            if not found:
                stall += 1

        it += 1
        if stall > 15:
            break
        if verbose and it % 3 == 0:
            print(f"    LP iter {it}: obj={best_obj:.8f}")

    return best_a, best_obj


def block_perturb(a, rng, max_time, verbose=True):
    """Block multiplicative perturbation."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        progress = (time.monotonic() - start) / max(max_time, 1e-6)
        width = max(1, int(n * rng.uniform(0.01, 0.12)))
        pos = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        sigma = 0.01 + 0.05 * (1.0 - progress)
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            it += 1
            continue
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_obj = obj
            best_a = candidate
        it += 1

    return best_a, best_obj

                best_seq = a_opt.copy()
                print(f"  After LP: obj={best_score:.6f}")

        # === Phase 3: More coordinate descent ===
        if time.monotonic() < deadline - 30:
            cd_time2 = min(200.0, float(budget_s) * 0.25)
            print(f"\nPhase 3: CD round 2, budget={cd_time2:.0f}s")
            a_opt, obj_opt = coord_descent(best_seq.copy(), cd_time2, rng=rng, verbose=True)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  After CD2: obj={best_score:.6f}")

    # === Phase 4: Try different lengths ===
    n_values = [150, 200, 250, 350, 400, 500]
    for n in n_values:
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        time_per_n = min(50.0, remaining / max(1, len(n_values) - n_values.index(n)))

        for attempt in range(2):
            if time.monotonic() > deadline - 30:
                break
            a_init = structured_init(n, rng)
            cd_budget = time_per_n * 0.6
            a_opt, obj_opt = coord_descent(a_init, cd_budget, rng=rng, verbose=False)

            if time.monotonic() < deadline - 20:
                lp_budget = time_per_n * 0.3
                a_opt2, obj_opt2 = lp_phase(a_opt, lp_budget, verbose=False)
                if obj_opt2 < obj_opt:
                    a_opt = a_opt2
                    obj_opt = obj_opt2

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"n={n} attempt={attempt}: obj={obj_opt:.6f} ** NEW BEST **")
            else:
                print(f"n={n} attempt={attempt}: obj={obj_opt:.6f}")

    # === Phase 5: Final refinement on best ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        print(f"\nPhase 5: Final refinement, budget={remaining:.0f}s")

        # Block perturbation
        bp_time = remaining * 0.3
        if bp_time > 5:
            a_opt, obj_opt = block_perturb(best_seq.copy(), rng, bp_time, verbose=False)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  After BP: obj={best_score:.6f}")

        # Final coordinate descent
        if time.monotonic() < deadline - 5:
            cd_final = deadline - time.monotonic() - 3.0
            a_opt, obj_opt = coord_descent(best_seq.copy(), cd_final, rng=rng, verbose=False)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  After final CD: obj={best_score:.6f}")

        # One more LP
        if time.monotonic() < deadline - 3:
            lp_final = min(30.0, deadline - time.monotonic() - 2.0)
            if lp_final > 5:
                a_opt, obj_opt = lp_phase(best_seq.copy(), lp_final, verbose=False)
                if obj_opt < best_score:
                    best_score = obj_opt
                    best_seq = a_opt.copy()
                    print(f"  After final LP: obj={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
