```python
"""
Smoothed optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smoothing of the max convolution, with gradient computed
via FFT in O(n log n). Combined with Nesterov-accelerated gradient descent,
T-annealing, and subgradient polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def softmax(x):
    x_shifted = x - np.max(x)
    e = np.exp(x_shifted)
    return e / np.sum(e)


def compute_score_fast(a):
    n = len(a)
    c = np.convolve(a, a)
    M = np.max(c)
    S = np.sum(a)
    if S < 1e-15:
        return np.inf
    return 2.0 * n * M / (S * S)


def smooth_objective_and_grad(w, T):
    """Smoothed objective F_T(w) = (1/T)*log(sum exp(T*c_k)) and gradient."""
    n = len(w)
    a = softmax(w)
    c = np.convolve(a, a)
    
    Tc = T * c
    Tc_max = np.max(Tc)
    exp_terms = np.exp(Tc - Tc_max)
    Z = np.sum(exp_terms)
    F_T = Tc_max + np.log(Z) / T
    p = exp_terms / Z
    
    E_p_c = np.dot(p, c)
    
    # cross[m] = sum_j a[j] * p[j+m] for m=0,...,n-1
    # Equivalent to: cross[m] = self.dot(p[m:m+n], a)
    cross = np.empty(n)
    for m in range(n):
        cross[m] = np.dot(p[m:m+n], a)
    
    grad = 2.0 * a * (cross - E_p_c)
    return F_T, grad


def smooth_gradient_descent(w_init, deadline, T_schedule, lr_init=0.1, 
                            momentum_init=0.9, max_iter_per_T=500):
    """Nesterov gradient descent with T-annealing."""
    w = w_init.copy().astype(np.float64)
    n = len(w)
    best_w = w.copy()
    best_score = compute_score_fast(softmax(w))
    
    v = np.zeros(n)  # velocity for momentum
    
    for T in T_schedule:
        if time.monotonic() > deadline:
            break
        
        lr = lr_init
        momentum = momentum_init
        v = np.zeros(n)
        
        for it in range(max_iter_per_T):
            if time.monotonic() > deadline:
                break
            
            F, grad = smooth_objective_and_grad(w, T)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-10:
                break
            
            # Nesterov step
            v = momentum * v + lr * grad
            w_new = w - v
            
            # Check improvement
            a_new = softmax(w_new)
            score = compute_score_fast(a_new)
            
            if score < best_score:
                best_score = score
                best_w = w_new.copy()
                w = w_new
            else:
                # Reduce learning rate
                lr *= 0.8
                v *= 0.5
                if lr < 1e-6:
                    break
            
            # Optional: print progress
            if it % 100 == 0:
                print(f"    T={T:.0f} iter={it} score={best_score:.6f} lr={lr:.4f}")
    
    return best_w, best_score, softmax(best_w)
```

But wait, the Nesterov step I wrote above is not exactly right. The proper Nesterov method is:

y = w + (t_k - 1) / t_{k+1} * (w - w_prev)
w_new = y - (1/L) * grad(y)
t_{k+1} = (1 + sqrt(1 + 4*t_k^2)) / 2

But a simpler variant (which often works well in practice) is:
v = momentum * v + lr * grad
w = w - v

This is "heavy ball" or "momentum" method, not exactly Nesterov, but similar.

Let me use the simpler momentum method for now.
