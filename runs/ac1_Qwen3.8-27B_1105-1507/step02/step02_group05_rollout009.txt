import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smoothed_max_and_grad(a, beta):
    """Compute smoothed max of conv(a,a) and its gradient.
    Returns (obj, grad) where grad is the gradient of the smoothed max w.r.t. a.
    """
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    b_shifted = b - b_max
    exp_b = np.exp(beta * b_shifted)
    p = exp_b / exp_b.sum()
    obj = b_max + np.log(exp_b.sum()) / beta

    # Gradient: grad[j] = (2/beta) * sum_{k: valid} p[k] * a[k-j]
    # For each j, k ranges over [j, j+n-1] intersect [0, 2n-2]
    grad = np.zeros(n)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        l_end = k_end - j
        grad[j] = np.dot(p[k_start:k_end], a[:l_end])
    grad *= 2.0 / beta
    return obj, grad


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def adam_smoothed_max(a, beta_schedule, n_iters_per_beta, deadline, tracker):
    """Adam optimizer on the smoothed max objective with beta annealing."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_count = 0
    lr = 0.05
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8

    for beta_idx, beta in enumerate(beta_schedule):
        for it in range(n_iters_per_beta):
            if time.monotonic() > deadline:
                return a, tracker

            obj, grad = smoothed_max_and_grad(a, beta)
            s = a.sum()
            # Normalize gradient for scale invariance
            grad = grad / (s * s)

            t_count += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t_count)
            v_hat = v / (1 - beta2 ** t_count)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                a_new = a.copy()
                continue
            a_new = a_new / s_new

            # Check improvement
            J_new = compute_J(a_new)
            J_old = compute_J(a)
            if J_new < J_old + 1e-12:
                a = a_new
            else:
                # Reduce learning rate
                lr *= 0.8
                if lr < 0.001:
                    break

        if beta_idx % 3 == 0:
            tracker.update(a)

    tracker.update(a)
    return a, tracker


def lp_mirror_descent(a, max_iters, deadline, tracker):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    stalled = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step: try multiple step sizes
        base_t = max(0.05, 0.6 * (1.0 - it / float(max(2 * max_iters, 1))))
        candidates_t = [base_t, min(1.0, base_t * 1.8), max(0.02, base_t * 0.4), base_t * 2.5]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            if t > 1.0:

            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)

            if s < 1e-10:

            a_trial = a_trial / s
            J_trial = compute_J(a_trial)

            # Try tiny step

            J_trial = compute_J(a_trial)

                a = a_trial
                current_J = J_trial
            else:
                break
        else:
            a = best_a
            current_J = best_J
            stalled = 0

        if it % 20 == 0:
            tracker.update(a)

    return a

    t_arr = np.linspace(0, 1, n, endpoint=False)
    for freq in [1, 2, 3, 4, 5, 7]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2.0 * np.pi * freq * t_arr), 0.01))

    inits.append(np.maximum(1.0 + 0.4 * np.sin(2.0 * np.pi * t_arr ** 2 * 4), 0.01))
    inits.append(np.maximum(np.exp(-4.0 * (t_arr - 0.3) ** 2) + 0.05, 0.01))
    inits.append(np.maximum(np.exp(-4.0 * (t_arr - 0.7) ** 2) + 0.05, 0.01))
    inits.append(np.maximum(1.0 - 0.5 * np.abs(2 * t_arr - 1), 0.01))

    # Anti-palindromic tendency
    anti = np.maximum(1.0 + 0.6 * (2 * t_arr - 1), 0.01)
    inits.append(anti)
    anti2 = np.maximum(1.0 - 0.6 * (2 * t_arr - 1), 0.01)
    inits.append(anti2)

    # Two-peak
    two_peak = np.maximum(np.exp(-20 * (t_arr - 0.25) ** 2) + np.exp(-20 * (t_arr - 0.75) ** 2) + 0.02, 0.01)
    inits.append(two_peak)

    # Sparse
    sparse = np.zeros(n)
    n_spikes = max(4, n // 20)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_width = max(1, n // 50)
    kernel = np.ones(kernel_width * 2 + 1) / (kernel_width * 2 + 1)
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.02)

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Multi-method optimization: smoothed max + Adam, LP mirror descent, L-BFGS-B, active set polish."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()
    t_start = time.monotonic()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Smoothed max + Adam (primary method) ===
    print(f"\n{'='*50} Phase 1: Smoothed Max + Adam {'='*50}", flush=True)
    t0 = time.monotonic()

    n = 128
    if parent is not None:
        n = parent.size

    inits = generate_inits(n, rng, parent)

    for idx, a0 in enumerate(inits):
        if time.monotonic() > deadline:
            break
        a0 = np.maximum(a0.astype(np.float64), 0.001)
        a0 = a0 / a0.sum()

        # Beta schedule: start smooth, get sharper
        beta_schedule = [3.0, 5.0, 8.0, 12.0, 18.0, 25.0, 35.0, 50.0]
        iters_per_beta = max(30, min(80, 3000 // n))

        a, tracker = adam_smoothed_max(a0, beta_schedule, iters_per_beta, deadline, tracker)

        # Follow up with LP on the result
        if time.monotonic() < deadline:
            a = lp_mirror_descent(a, 30, deadline, tracker)

    tracker.report(f"Phase 1 done: {time.monotonic()-t0:.1f}s")

    # === Phase 2: LP mirror descent from multiple starts ===
    if time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 2: LP Mirror Descent {'='*50}", flush=True)
        t0 = time.monotonic()

        # Start from current best and several other inits
        start_points = []
        if tracker.best is not None:
            start_points.append(tracker.best.copy())
        for a0 in inits[:5]:
            start_points.append(np.maximum(a0, 0.001))

        for sp in start_points:
            if time.monotonic() > deadline:
                break
            a = lp_mirror_descent(sp, 100, deadline, tracker)

        tracker.report(f"Phase 2 done: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Simulated annealing with LP restarts ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 3: SA + LP Restarts {'='*50}", flush=True)
        t0 = time.monotonic()
        n_best = tracker.best.size
        temperature = 0.12
        n_restarts = 80

        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break
            a = tracker.best.copy().astype(np.float64)
            noise = rng.normal(0, temperature, n_best)
            a = a + noise
            a = np.maximum(a, 0.001)
            a = a / a.sum()

            # Short LP run
            iters = max(15, min(40, 60 - r // 2))
            a = lp_mirror_descent(a, iters, deadline, tracker)

            temperature *= 0.96

        tracker.report(f"Phase 3 done: {time.monotonic()-t0:.1f}s")

    # === Phase 4: L-BFGS-B refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 4: L-BFGS-B {'='*50}", flush=True)
        t0 = time.monotonic()

        a_refined = lbb_refine(tracker.best, deadline, tracker)
        if a_refined is not None:
            # Follow up with LP
            a_refined = lp_mirror_descent(a_refined, 50, deadline, tracker)

        tracker.report(f"Phase 4 done: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Active set polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 5: Active Set Polish {'='*50}", flush=True)
        t0 = time.monotonic()

        a_polished = active_set_polish(tracker.best, deadline, tracker)

        tracker.report(f"Phase 5 done: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Final slow LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        n = tracker.best.size
        if n <= 512:
            print(f"\n{'='*50} Phase 6: Final LP Polish {'='*50}", flush=True)
            t0 = time.monotonic()
            a_final = lp_mirror_descent(tracker.best, 200, deadline, tracker)
            tracker.report(f"Phase 6 done: {time.monotonic()-t0:.1f}s")

    # === Try different lengths for final improvement ===
    if tracker.best is not None and time.monotonic() < deadline:
        print(f"\n{'='*50} Phase 7: Multi-length exploration {'='*50}", flush=True)
        t0 = time.monotonic()
        best_n = tracker.best.size

        for try_n in [best_n, best_n * 2, best_n // 2]:
            if try_n < 32 or try_n > 1024:
                continue
            if time.monotonic() > deadline:
                break

            a_init = resample_seq(tracker.best, try_n)
            a_init = np.maximum(a_init, 0.001)
            a_init = a_init / a_init.sum()

            # Quick smoothed max
            beta_sched = [5.0, 10.0, 20.0]
            a_try, tracker = adam_smoothed_max(a_init, beta_sched, 40, deadline, tracker)

            # Quick LP
            a_try = lp_mirror_descent(a_try, 40, deadline, tracker)

        tracker.report(f"Phase 7 done: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
