import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a, b=None):
    """Full convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    s = 1
    while s < n + m - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fb = np.fft.rfft(b, s)
    return np.fft.irfft(fa * fb, s)[:n + m - 1]

def self_conv(a):
    """Self-convolution a*a via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = self_conv(a)
    return float(2.0 * n * b.max() / (s * s))

def mirrored_gradient(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*q_k)) where q = conv(a,a).
    Returns the gradient vector (same length as a).
    """
    n = len(a)
    q = self_conv(a)  # length 2n-1
    # Softmax
    shifted = beta * (q - q.max())
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    # Gradient: 2 * cross_correlation(p, a)
    # (p ⋆ a)_i = sum_l p_{l+i} * a_l
    # = conv(p, reverse(a))[i + n - 1]
    a_rev = a[::-1]
    # conv(p, a_rev) has length (2n-1) + n - 1 = 3n-2
    conv_pa = fast_conv(p, a_rev)
    grad = 2.0 * conv_pa[n-1 : 2*n-1]
    return grad

def mirror_descent_step(a, grad, eta):
    """Multiplicative update: a_new = a * exp(-eta * grad), normalized."""
    # For numerical stability
    grad_centered = grad - grad.mean()  # Remove mean to preserve sum approximately
    log_update = -eta * grad_centered
    log_update -= log_update.max()  # Numerical stability
    a_new = a * np.exp(log_update)
    s = a_new.sum()
    if s < 1e-300:
        return a  # Underflow
    a_new = a_new / s
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 0.01:
        return a
    return a_new / s

def optimize_mirror(a, beta_schedule, eta0, deadline):
    """Run mirror descent with adaptive step size and beta annealing."""
    a = np.clip(a, 1e-10, None).astype(np.float64)
    a = a / a.sum()
    best = a.copy()
    best_score = compute_score(a)
    
    n = len(a)
    eta = eta0
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        for it in range(2000):
            if time.monotonic() > deadline:
                break
            
            q = self_conv(a)
            q_max = q.max()
            
            # Compute gradient
            grad = mirrored_gradient(a, beta)
            
            # Try the step
            a_new = mirror_descent_step(a, grad, eta)
            score_new = compute_score(a_new)
            
            if score_new < best_score:
                best = a_new.copy()
                best_score = score_new
                a = a_new
                # Maybe increase eta slightly
                eta = min(eta * 1.05, 1.0)
            else:
                # Decrease eta
                eta *= 0.7
                if eta < 1e-8:
                    break
    
    return best, best_score

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated mirror descent (multiplicative weights) with adaptive step size
2. Beta annealing for the log-sum-exp smoothing
3. Multi-start with diverse initializations and sequence lengths
4. Periodic large-move perturbation (simulated annealing style)
5. Final coordinate-wise polish
"""

import numpy as np
import time

def fast_convolution(a, b):
    """Full linear convolution via FFT."""
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:na + nb - 1]

def self_convolution(a):
    """Self-convolution a*a via FFT (optimized)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = self_convolution(a)
    return float(2.0 * n * b.max() / (s * s))

def compute_gradient(a, beta):
    """
    Compute gradient of smoothed objective (1/beta)*log(sum exp(beta*q_k)).
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    q = self_convolution(a)
    # Softmax weights
    shifted = beta * (q - q.max())
    exp_vals = np.exp(np.clip(shifted, -500, 500))
    p = exp_vals / exp_vals.sum()
    # Gradient = 2 * cross_correlation(p, a)
    # (p star a)_i = sum_l p[l+i] * a[l] = conv(p, a[::-1])[i + n - 1]
    a_rev = a[::-1]
    pa = fast_convolution(p, a_rev)
    grad = 2.0 * pa[n - 1: 2 * n - 1]
    return grad

def mirror_step(a, grad, eta):
    """Multiplicative weights update with normalization."""
    # Center the gradient to approximately preserve the sum
    grad_c = grad - grad.mean()
    # Scale for numerical stability
    scale = np.abs(grad_c).max()
    if scale < 1e-15:
        return a.copy()
    log_ratios = -eta * grad_c / (scale * len(a))
    log_ratios -= log_ratios.max()
    weights = np.exp(log_ratios)
    a_new = a * weights
    s = a_new.sum()
    if s < 1e-100:
        return a.copy()
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def optimize_sequence(a0, deadline, verbose=False):
    """
    Optimize a sequence using mirror descent with beta annealing.
    Returns (best_sequence, best_score).
    """
    n = len(a0)
    a = np.clip(a0, 1e-10, None).astype(np.float64)
    a = a / a.sum()
    
    best = a.copy()
    best_score = compute_score(a)
    
    # Beta schedule: start smooth, end sharp
    beta_list = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0, 5000.0]
    
    # For each beta, do several rounds of mirror descent
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        
        eta = 10.0  # Step size (will be adapted)
        no_improve_count = 0
        max_no_improve = 50
        
        for it in range(5000):
            if time.monotonic() > deadline:
                break
            if no_improve_count > max_no_improve:
                break

def generate_initialization(n, rng, parent=None):
    """Generate a diverse initialization."""
    t = np.linspace(0, 1, n)
    
    choices = []
    
    # Resampled parent
    if parent is not None and len(parent) > 1:
        t_old = np.linspace(0, 1, len(parent))
        resampled = np.interp(t, t_old, parent)
        resampled = np.maximum(resampled, 0)
        if resampled.sum() > 0.01:
            choices.append('parent')
    
    # Uniform
    choices.append('uniform')
    
    # Cosine bump (peaked in middle)
    choices.append('cosine')
    
    # Quadratic bump
    choices.append('quadratic')
    
    # Sine (one period)
    choices.append('sine')
    
    # Random smooth
    choices.append('random')
    
    # Bimodal
    choices.append('bimodal')
    
    choice = rng.choice(choices)
    
    if choice == 'parent' and parent is not None:
        a = np.interp(t, np.linspace(0, 1, len(parent)), parent)
    elif choice == 'uniform':
        a = np.ones(n)
    elif choice == 'cosine':
        a = 1.0 + 0.8 * np.cos(2 * np.pi * (t - 0.5))
        a = np.maximum(a, 0.01)
    elif choice == 'quadratic':
        a = 1.0 + 1.5 * np.cos(np.pi * (t - 0.5)) ** 2
        a = np.maximum(a, 0.01)
    elif choice == 'sine':
        a = 1.0 + 0.5 * np.sin(2 * np.pi * t)
        a = np.maximum(a, 0.01)
    elif choice == 'random':
        # Smooth random via low-pass filtering
        raw = rng.random(n)
        # Simple smoothing
        kernel = np.ones(5) / 5
        a = np.convolve(raw, kernel, mode='same')
        a = np.maximum(a, 0.01)
    elif choice == 'bimodal':
        a = 1.0 + 0.6 * np.cos(4 * np.pi * (t - 0.5))
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n)
        s = n
    return a / s

def perturbation(a, rng, strength=0.1):
    """Add random perturbation to escape local minima."""
    n = len(a)
    # Random perturbation
    noise = rng.normal(0, strength, n)
    a_new = a + noise
    # Make some elements zero (sparsity)
    if rng.random() < 0.3:
        n_zeros = int(n * 0.1)
        zero_idx = rng.choice(n, n_zeros, replace=False)
        a_new[zero_idx] = 0
    
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 0.01:
        return a.copy()
    return a_new / s

def large_move(a, rng):
    """Make a large random move (for simulated annealing)."""
    n = len(a)
    if rng.random() < 0.5:
        # Replace a random block
        block_size = max(1, n // 10)
        start = rng.integers(0, n - block_size)
        a_new = a.copy()
        a_new[start:start+block_size] = rng.random(block_size) * 0.5
    else:
        # Multiply by a random smooth factor
        t = np.linspace(0, 1, n)
        freq = rng.integers(1, 5)
        phase = rng.random() * 2 * np.pi
        factor = 1.0 + 0.5 * np.cos(2 * np.pi * freq * t + phase)
        a_new = a * np.maximum(factor, 0.1)
    
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 0.01:
        return a.copy()
    return a_new / s

def coordinate_polish(a, n_steps=100, deadline=None):
    """Try small coordinate-wise perturbations."""
    n = len(a)
    a = a.copy()
    score = compute_score(a)
    rng = np.random.default_rng(42)
    
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        # Try a random subset of coordinates
        n_try = max(1, n // 20)
        idx = rng.choice(n, n_try, replace=False)
        for i in idx:
            for delta in [0.1, -0.1, 0.05, -0.05, 0.01, -0.01]:

                a_new[i] = max(0, a[i] + delta * a.mean())

                if sc < score - 1e-10:
                    a = a_norm
                    score = sc
                    improved = True
                    break
        if not improved and step > 10:
            break
    
    return a, score

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated mirror descent (multiplicative weights) with adaptive step size
2. Log-sum-exp smoothing with beta annealing, plus random subgradient method
3. Multi-start search with diverse initializations and sequence lengths
4. Simulated annealing style perturbation to escape local minima
5. Final coordinate-wise polish
"""

import numpy as np
import time


def fast_convolution(a, b):
    """Full linear convolution via FFT, O(n log n)."""
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:na + nb - 1]


def self_convolution(a):
    """Self-convolution a * a via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n * max(conv(a,a)) / (sum(a))^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = self_convolution(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_gradient(a, beta):
    """
    Gradient of (1/beta)*log(sum_k exp(beta*q_k(a))) where q = self_conv(a).
    Computed in O(n log n) via FFT.
    """
    n = len(a)
    q = self_convolution(a)
    shifted = beta * (q - q.max())
    exp_vals = np.exp(np.clip(shifted, -700, 700))
    p = exp_vals / exp_vals.sum()
    # grad_i = 2 * sum_k p_k * a_{k-i} (for valid k-i)
    # = 2 * (cross_correlation of p and a)_i
    # cross_correlation(p, a)_i = (conv(p, a[::-1]))[i + n - 1]
    pa = fast_convolution(p, a[::-1])
    grad = 2.0 * pa[n - 1: 2 * n - 1]
    return grad


def subgradient(a, eps, rng):
    """
    Random subgradient of max_k q_k(a).
    Picks a random weighted combination of gradients at near-maximal k's.
    """
    n = len(a)
    q = self_convolution(a)
    q_max = q.max()
    active = np.where(q >= q_max - eps)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(q))])
    
    w = rng.dirichlet(np.ones(len(active)))
    
    p = np.zeros(2 * n - 1)
    for idx, wv in zip(active, w):
        p[idx] = wv
    
    pa = fast_convolution(p, a[::-1])
    grad = 2.0 * pa[n - 1: 2 * n - 1]
    return grad


def mirror_update(a, grad, eta, floor=1e-15):
    """
    Mirror descent (multiplicative weights) update.
    a_new_i = a_i * exp(-eta * grad_i) / Z
    """
    a_safe = np.maximum(a, floor)
    log_vals = np.log(a_safe) - eta * grad
    log_vals -= log_vals.max()
    new_a = np.exp(log_vals)
    s = new_a.sum()
    if s < 1e-30:
        return a.copy()
    return new_a / s


def optimize_mirror(a0, deadline, verbose=False, tag=""):
    """
    Optimize using mirror descent with beta annealing and adaptive step size.
