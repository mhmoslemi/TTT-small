from scipy.optimize import linprog
import time

def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    # Binary search for the threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def fast_convolve(f, g):
    """Fast linear convolution using FFT."""
    n = len(f)
    m = len(g)
    n_fft = 1 << (n + m - 1).bit_length()
    F = np.fft.rfft(f, n_fft)
    G = np.fft.rfft(g, n_fft)
    return np.fft.irfft(F * G, n_fft)[:n + m - 1]

def compute_score(f):
    """Compute 2*n*max(conv) / sum^2."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = fast_convolve(f, f)
    return 2.0 * n * conv.max() / (s * s)

def subgradient_step(f, eta, k=5, rng=None):
    """One step of projected subgradient with top-k averaging."""
    n = len(f)
    conv = fast_convolve(f, f)
    M = conv.max()

    if k >= len(conv):
        top_k = np.arange(len(conv))
    else:
        top_k = np.argsort(conv)[::-1][:k]
    
    # Weighted subgradient
    w = conv[top_k] / (conv[top_k].sum() + 1e-30)
    
    g = np.zeros(n)

        i_start = max(0, j - n + 1)

import time

def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""

        return np.full(n, s / n)
    rho = np.where(cond)[0][-1]

    return np.maximum(v - theta, 0)

    G = np.fft.rfft(g, n_fft)

"""
Optimization of step function coefficients to minimize 2n*max(conv)/(sum)^2.
Uses projected subgradient + LP with line search + coordinate descent polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time

def project_simplex(v, s=1.0):
    """Project v onto the non-negative simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def fast_convolve(f, g):
    """Fast linear convolution using FFT."""
    n = len(f)
    m = len(g)
    n_fft = 1 << (n + m - 1).bit_length()
    F = np.fft.rfft(f, n_fft)
    G = np.fft.rfft(g, n_fft)
    return np.fft.irfft(F * G, n_fft)[:n + m - 1]

def score_from_conv(conv, n):
    """Compute score from convolution (assuming sum=1)."""
    return 2.0 * n * conv.max()

def line_search(f, g, conv_ff, conv_fg, conv_gg, n):
    """Find best t in [0,1] minimizing score along f_t = (1-t)f + tg."""
    best_t = 0.0
    best_score = score_from_conv(conv_ff, n)
    
    # Ternary search
    lo, hi = 0.0, 1.0
    for _ in range(40):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        
        conv_m1 = (1-m1)**2 * conv_ff + 2*m1*(1-m1) * conv_fg + m1**2 * conv_gg
        conv_m2 = (1-m2)**2 * conv_ff + 2*m2*(1-m2) * conv_fg + m2**2 * conv_gg
        
        s1 = conv_m1.max()
        s2 = conv_m2.max()
        
        if s1 < s2:
            hi = m2
        else:
            lo = m1
        
        # Check intermediate points
        for t_check in [m1, m2, (m1+m2)/2]:
            conv_tc = (1-t_check)**2 * conv_ff + 2*t_check*(1-t_check) * conv_fg + t_check**2 * conv_gg
            sc = conv_tc.max()
            if sc < best_score:
                best_score = sc
                best_t = t_check
    
    return best_t, best_score

def solve_lp_direction(f, M=None, time_limit=2):
    """Solve LP to find direction g that minimizes max_j (f*g)(j)."""
    n = len(f)
    m = 2 * n - 1
    
    # Build constraint matrix
    A = np.zeros((m, n))
    for j in range(m):
        i_start = max(0, j - n + 1)
        i_end = min(n, j + 1)
        for i in range(i_start, i_end):
            A[j, i] = f[j - i]
    
    # LP: minimize t subject to A@g - t <= 0, sum(g) = 1, g >= 0, t >= 0
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    
    if not res.success or res.x is None:
        return None
    
    g = np.clip(res.x[:n], 0, None)
    g_sum = g.sum()
    if g_sum < 1e-10:
        return None
    g = g / g_sum
    
    return g

def lp_with_line_search(f, time_budget, initial_t_limit=3, rng=None):
    """Iterative LP with line search."""
    n = len(f)
    f = f.copy().astype(np.float64)
    s = f.sum()
    f = f / s
    
    best_f = f.copy()
    best_score = score_from_conv(fast_convolve(f, f), n)
    t_limit = initial_t_limit
    start = time.monotonic()
    
    iteration = 0
    while time.monotonic() - start < time_budget:
        g = solve_lp_direction(f, time_limit=t_limit)
        if g is None:
            break
        
        # Precompute convolutions for line search
        conv_ff = fast_convolve(f, f)
        conv_fg = fast_convolve(f, g)
        conv_gg = fast_convolve(g, g)
        
        t_opt, score_opt = line_search(f, g, conv_ff, conv_fg, conv_gg, n)
        
        if t_opt > 0 and score_opt < best_score:
            f_new = (1 - t_opt) * f + t_opt * g
            f_new = project_simplex(f_new, s=1.0)
            # Recompute actual score (line search used precomputed convs, might have small error)
            actual_score = score_from_conv(fast_convolve(f_new, f_new), n)
            if actual_score < best_score:
                best_f = f_new.copy()
                best_score = actual_score
                f = f_new
                t_limit = min(5.0, t_limit * 1.1)
            else:
                # Accept the line search result
                f = f_new
                t_limit *= 0.8
        else:
            t_limit *= 0.7
            if t_limit < 0.3:
                # Try a perturbation
                f_pert = f * (1 + 0.05 * rng.standard_normal(n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = project_simplex(f_pert, s=1.0)
                pert_score = score_from_conv(fast_convolve(f_pert, f_pert), n)
                if pert_score < best_score:
                    f = f_pert
                    best_f = f_pert.copy()
                    best_score = pert_score
                    t_limit = 1.0
                else:
                    break
        
        iteration += 1
    
    return best_f, best_score, iteration

def subgradient_optimize(f, n_iter=5000, eta0=0.1, k=5, beta=0.9, rng=None):
    """Projected subgradient with momentum and top-k averaging."""
    n = len(f)
    f = f.copy().astype(np.float64)
    f = project_simplex(f, s=1.0)
    
    best_f = f.copy()
    best_score = score_from_conv(fast_convolve(f, f), n)
    v = np.zeros(n)
    
    for t in range(1, n_iter + 1):
        conv = fast_convolve(f, f)
        M = conv.max()
        
        # Top-k peaks
        if k < len(conv):
            top_indices = np.argpartition(conv, -k)[-k:]
            top_indices = top_indices[np.argsort(conv[top_indices])[::-1]]
        else:
            top_indices = np.arange(len(conv))
        
        weights = conv[top_indices]
        w_sum = weights.sum()
        if w_sum < 1e-30:
            break
        weights = weights / w_sum
        
        # Compute subgradient (vectorized)
        g = np.zeros(n)
        for idx, j in enumerate(top_indices):
            j = int(j)
            i_start = max(0, j - n + 1)
            i_end = min(n, j + 1)
            if i_start < i_end:
                g[i_start:i_end] += weights[idx] * f[j - np.arange(i_start, i_end)][::-1]
        
        # Momentum update
        eta = eta0 / np.sqrt(t)
        v = beta * v - eta * g
        f_new = f + v
        f_new = project_simplex(f_new, s=1.0)
        
        score = score_from_conv(fast_convolve(f_new, f_new), n)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
        
        f = f_new
    
    return best_f, best_score

def coordinate_descent(f, n_sweeps=10, rng=None):
    """Coordinate descent polishing."""
    n = len(f)
    f = f.copy().astype(np.float64)
    f = project_simplex(f, s=1.0)
    best_score = score_from_conv(fast_convolve(f, f), n)
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-10:
                continue
            best_val = original
            best_local = best_score
            
            for factor in [0.3, 0.5, 0.7, 0.85, 1/0.85, 1/0.7, 1/0.5, 1/0.3]:
                f[i] = max(0.0, original * factor)
                conv = fast_convolve(f, f)
                sc = conv.max()
                if sc < best_local:
                    best_local = sc
                    best_val = f[i]
            
            f[i] = best_val
        
        f = project_simplex(f, s=1.0)
        best_score = score_from_conv(fast_convolve(f, f), n)
    
    return f, best_score

def diverse_init(n, rng):
    """Generate a diverse initial sequence."""
    r = rng.random()
    if r < 0.2:
        # Uniform with noise
        f = np.ones(n) * (1 + 0.1 * rng.standard_normal(n))
    elif r < 0.4:
        # Gaussian bump
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.15, 0.4)
        f = np.exp(-0.5 * (np.arange(n) - center)**2 / width**2)
    elif r < 0.6:
        # Cosine window with modulation
        f = 0.5 * (1 + np.cos(2*np.pi*np.arange(n)/n))
        f *= (1 + 0.3 * np.sin(4*np.pi*np.arange(n)/n * rng.uniform(0.5, 2)))
    elif r < 0.8:
        # Product of bumps
        f = np.ones(n)
        for _ in range(3):
            c = rng.uniform(0, n)
            w = n * rng.uniform(0.1, 0.3)
            f *= np.exp(-0.5 * (np.arange(n) - c)**2 / w**2)
        f += 0.05
    else:
        # Random Fourier features
        freqs = rng.uniform(-2, 2, size=n//4)
        phases = rng.uniform(0, 2*np.pi, size=n//4)
        f = np.ones(n)
        for freq, phase in zip(freqs, phases):
            f += 0.3 * np.cos(2*np.pi*freq*np.arange(n)/n + phase)
        f = np.abs(f) + 0.1
    
    f = np.clip(f, 0.01, None)
    f = project_simplex(f, s=1.0)
    return f

def propose_candidate(seed=42, budget_s=1500, **kwargs):

"""
Optimizes step function coefficients to minimize 2n*max(conv)/(sum)^2.
Uses: diverse init -> projected subgradient -> LP with line search -> CD polishing.
Key innovations: min-peak LP formulation, line search along LP direction,
top-k subgradient averaging with momentum.
"""

import numpy as np
from scipy.optimize import linprog
import time


def project_simplex(v, s=1.0):
    """Project v onto the non-negative simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def fast_convolve(f, g):
    """Fast linear convolution using FFT."""
    n = len(f)
    m = len(g)
    n_fft = 1 << (n + m - 1).bit_length()
    F = np.fft.rfft(f, n_fft)
    G = np.fft.rfft(g, n_fft)
    return np.fft.irfft(F * G, n_fft)[:n + m - 1]


def score_from_conv(conv, n):
    """Compute 2n*max(conv) assuming the sequence is normalized to sum=1."""
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build matrix A (len=2n-1, n) such that A @ g = convolve(f, g)."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(m):
        i_start = max(0, j - n + 1)
        i_end = min(n, j + 1)
        for i in range(i_start, i_end):
            A[j, i] = f[j - i]
    return A


def solve_lp_min_peak(f, time_limit=2.0):
    """
    Solve LP: minimize t s.t. (f*g)(j) <= t for all j, sum(g)=1, g>=0.
    Returns g (normalized) or None on failure.
    """
    n = len(f)
    m = 2 * n - 1
    
    A = build_conv_matrix(f)
    
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None
    
    if not res.success or res.x is None:
        return None
    
    g = np.clip(res.x[:n], 0, None)
    g_sum = g.sum()
    if g_sum < 1e-10:
        return None
    g = g / g_sum
    
    return g


def line_search_conv(conv_ff, conv_fg, conv_gg, n):
    """
    Find best t in [0,1] minimizing max_j [(1-t)^2*ff(j) + 2t(1-t)*fg(j) + t^2*gg(j)].
    """
    best_t = 0.0
    best_val = conv_ff.max()
    
    lo, hi = 0.0, 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        
        v1 = ((1-m1)**2 * conv_ff + 2*m1*(1-m1) * conv_fg + m1**2 * conv_gg).max()
        v2 = ((1-m2)**2 * conv_ff + 2*m2*(1-m2) * conv_fg + m2**2 * conv_gg).max()
        
        # Check these points
        for tc, vc in [(m1, v1), (m2, v2)]:
            if vc < best_val:
                best_val = vc
                best_t = tc
        
        if v1 < v2:
            hi = m2
        else:
            lo = m1
        
        if hi - lo < 1e-10:
            break
    
    return best_t, best_val


def lp_with_line_search(f, time_budget, initial_t_limit=3.0, rng=None):

    start_time = time.monotonic()

    iteration = 0
    
    while time.monotonic() - start_time < time_budget:
        # Check time per iteration
        elapsed = time.monotonic() - start_time
        if elapsed > time_budget * 0.8:
            # Only do quick LPs now
            t_limit = min(t_limit, 1.0)
        if elapsed > time_budget * 0.95:
            break
        
        g = solve_lp_min_peak(f, time_limit=t_limit)
        if g is None:
            break
        
        # Check if g is close to f (stagnation)
        if np.linalg.norm(g - f, 1) < 1e-6:
            stall_count += 1
            if stall_count > 5:
                break
            # Perturb f
            f = f * (1 + 0.02 * rng.standard_normal(n))
            f = project_simplex(f, s=1.0)
            t_limit = max(1.0, t_limit * 0.8)

        conv_ff = fast_convolve(f, f)

        if t_opt < 1e-8:
            # No improvement possible along this direction
            t_limit *= 0.7
            stall_count += 1
            if stall_count > 10:
                break
            continue
        
        # Compute new point
        f_new = (1 - t_opt) * f + t_opt * g
        f_new = project_simplex(f_new, s=1.0)
        
        # Verify with actual computation
        actual_val = score_from_conv(fast_convolve(f_new, f_new), n)
        
        if actual_val < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = actual_val
            f = f_new
            stall_count = 0
            t_limit = min(5.0, t_limit * 1.05)
        else:

            f = f_new

                    best_f = f.copy()
                    best_score = score_from_conv(fast_convolve(f, f), n)
                t_limit = 1.5
                stall_count = 0
        
        iteration += 1
    
    return best_f, best_score, iteration


def subgradient_optimize(f, n_iter, eta0=0.05, k=5, beta=0.9, rng=None):
    """
    Projected subgradient with momentum and top-k peak averaging.
    """
    n = len(f)
    f = f.copy().astype(np.float64)
    f = project_simplex(f, s=1.0)
    
    best_f = f.copy()
    best_score = score_from_conv(fast_convolve(f, f), n)
    v = np.zeros(n)  # momentum
    f_prev = f.copy()
    
    for t in range(1, n_iter + 1):
        conv = fast_convolve(f, f)
        
        # Top-k peaks
        if k < len(conv):
            top_indices = np.argpartition(conv, -k)[-k:]
        else:
            top_indices = np.arange(len(conv))
        
        weights = conv[top_indices]
        w_sum = weights.sum()
        if w_sum < 1e-30:
            break
        weights = weights / w_sum
        
        # Subgradient
        g = np.zeros(n)
        for idx, j in enumerate(top_indices):
            j = int(j)
            i_start = max(0, j - n + 1)
            i_end = min(n, j + 1)
            if i_start < i_end:
                # g[i] += weights[idx] * f[j-i] for i in [i_start, i_end)
                g[i_start:i_end] += weights[idx] * f[j - np.arange(i_start, i_end)][::-1]
        
        # Nesterov momentum
        eta = eta0 / np.sqrt(t)
        v = beta * v + beta * (f - f_prev) - eta * g
        f_prev = f.copy()
        f_new = f + v
        f_new = project_simplex(f_new, s=1.0)
        
        score = score_from_conv(fast_convolve(f_new, f_new), n)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
        
        f = f_new
    
    return best_f, best_score
