Step function coefficient optimization targeting minimum of 2*n*max(conv)/(sum^2).
Uses uniform baseline, parent optimization, structured initializations,
LP-based alternating, and gradient refinement.

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute the objective function for a given sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_conv_matrix(a):
    """Build the matrix A such that A @ g = a * g (convolution)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        seg = a[k - j_end:k - j_start + 1]
        A[k, j_start:j_end + 1] = seg[::-1]
    return A


def solve_lp_for_g(a, upper_bound=100.0):
    """Solve LP: maximize sum(g) subject to a*g <= M elementwise, g >= 0."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M


def lp_phase(a, max_time, t_start=0.5, t_min=0.05, max_iters=500):

def coordinate_descent(a, max_time, max_iters=100000):

        best_local = a[i]

        obj = compute_obj(a)

def structured_init(n, rng):
    """Generate various structured initializations."""
    choice = int(rng.integers(0, 8))
    if choice == 0:
        # Uniform
        a = np.ones(n)
    elif choice == 1:
        # Cosine bump
        phase = rng.uniform(0, 2 * np.pi)
        freq = rng.uniform(0.1, 0.5)
        a = np.maximum(0.01, np.cos(2 * np.pi * freq * np.arange(n) + phase) + 1)
    elif choice == 2:
        # Gaussian bell
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.2, 0.5)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2) + 0.1
    elif choice == 3:
        # Edge-concentrated
        a = np.ones(n)
        decay = rng.uniform(0.85, 0.98)
        for i in range(n):
            dist = min(i, n - 1 - i)
            a[i] = decay ** dist
    elif choice == 4:
        # Periodic
        period = int(rng.integers(max(2, n // 20), max(3, n // 3)))
        pattern = np.maximum(0.1, rng.random(period))
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 5:
        # Random smooth
        a = np.maximum(0.1, rng.normal(1.0, 0.3, n))
        # Smooth
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif choice == 6:
        # Power law
        a = (np.arange(n) + 1) ** (-rng.uniform(0, 0.5))
    else:
        # Mixed
        a = 0.5 * np.ones(n) + 0.5 * np.cos(2 * np.pi * rng.uniform(0.1, 0.8) * np.arange(n))
        a = np.maximum(0.01, a)

    s = np.sum(a)
    return a / s if s > 1e-10 else np.ones(n) / n
