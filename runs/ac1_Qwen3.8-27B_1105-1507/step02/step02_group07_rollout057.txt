The key insight is: for large n, the "effective" problem is about the shape of the density function. The convolution peak at the center is approximately ∫f(x)f(1-x)dx, and we want to minimize this (and other peaks). """
Multi-phase optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses exact coordinate descent (1D minimax), subgradient with adaptive step,
and LP direction finding. Employs power-law and exponential initializations.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def score_of(f):
    """Compute the score without calling evaluate_sequence (faster for internal use)."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    conv = np.convolve(f, f)
    max_b = float(np.max(conv))
    return float(2 * n * max_b / (S * S))


def coordinate_descent(f, deadline, max_passes=50, verbose=False, verbose_prefix=""):
    """
    Exact coordinate descent: for each coordinate, find the best value
    by minimizing the 1D function h(t) = score with f[j]=t.
    """
    f = f.copy().astype(np.float64)
    n = len(f)
    best_score = score_of(f)
    best_f = f.copy()
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            
            # 1D minimax: find t that minimizes score with f[j] = t
            # score(t) = 2n * max_k conv_k(t) / (sum(f with f[j]=t))^2
            # We need to find min over t >= 0.
            
            fj = f[j]
            
            # Precompute: for each k, how does conv[k] change when f[j] changes?
            # conv[k] = sum_i f[i]*f[k-i]
            # The term involving f[j]:
            #   if 0 <= k-j < n and k != 2*j: contributes 2*f[j]*f[k-j]
            #   if k == 2*j: contributes f[j]^2
            #   otherwise: no contribution
            
            # For the 1D search, we need max_k conv_k as a function of t.
            # This is expensive to compute for each t. Instead, use a
            # smaller set of candidate t values.
            
            # Strategy: try a few candidate values of t and pick the best.
            # Candidates: 0, f[j]*0.5, f[j], f[j]*1.5, and the value
            # that would equalize the top few peaks.
            
            # Quick approach: use the current convolution and estimate
            # the optimal t.
            
            conv_curr = np.convolve(f, f)
            current_max = float(np.max(conv_curr))
            current_sum = float(np.sum(f))
            
            # Try several values of t
            candidates = [0.0, f[j] * 0.25, f[j] * 0.5, f[j] * 0.75,
                         f[j] * 1.25, f[j] * 1.5, f[j] * 2.0]
            
            # Also try the value that would make the peak involving f[j]
            # equal to the second-highest peak
            top_idx = np.argmax(conv_curr)
            second_max = np.max(np.delete(conv_curr, top_idx))
            
            # If the peak at top_idx involves f[j], try to reduce it
            if 0 <= top_idx - j < n:
                # The peak involves f[j] with weight f[top_idx-j] (or 2*f[...] or f[j])
                if top_idx == 2 * j:
                    # conv[top_idx] includes f[j]^2
                    # To make conv[top_idx] = second_max:
                    # conv[top_idx] - f[j]^2 + t^2 = second_max
                    # t^2 = second_max - conv[top_idx] + f[j]^2
                    t_sq = second_max - float(conv_curr[top_idx]) + f[j]**2
                    if t_sq > 0:
                        candidates.append(np.sqrt(t_sq))
                else:
                    # conv[top_idx] includes 2*f[j]*f[top_idx-j]
                    # conv[top_idx] - 2*f[j]*f[top_idx-j] + 2*t*f[top_idx-j] = second_max
                    # 2*t*f[top_idx-j] = second_max - conv[top_idx] + 2*f[j]*f[top_idx-j]
                    w = float(f[top_idx - j])
                    if w > 1e-15:
                        t_val = (second_max - float(conv_curr[top_idx]) + 2 * f[j] * w) / (2 * w)
                        if t_val >= 0:
                            candidates.append(t_val)
            
            # Evaluate each candidate
            best_t = f[j]
            best_s = score_of(f)
            
            for t in candidates:
                t = max(0.0, t)
                f[j] = t
                s = score_of(f)
                if s < best_s:
                    best_s = s
                    best_t = t
            
            f[j] = best_t
            
            if best_s < best_score - 1e-12:
                best_score = best_s
                best_f = f.copy()
                improved = True
        
        if verbose and p % 5 == 0:
            print(f"    {verbose_prefix}CD pass {p}: score={best_score:.6f}")
        
        if not improved and p > 2:
            break
    
    return best_f, best_score

            break

            fj = f[j]
            
            # Affected positions: k in [j, j+n-1]
            k_min = j
            k_max = min(j + n - 1, 2 * n - 2)

            # And identify if 2j is in the affected range
            has_quad = (2 * j >= k_min and 2 * j <= k_max)

            # We can precompute this, but for simplicity, just compute max of full conv
            # and note that only affected entries change.
            # For efficiency, compute max_unaffected once per pass.
            
            # For now, just do a simple ternary search on delta in [-fj, 3*fj]
            # (or [0, 3*fj] if we want to keep f[j] non-negative... actually we can
            # go down to 0, so delta in [-fj, 3*fj] means new f[j] in [0, 3*fj])
            
            lo = -fj  # f[j] becomes 0

            if hi <= lo:
                hi = lo + 1.0 / n

            for _ in range(30):  # 30 iterations gives good precision
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                
                # Compute h(m1) and h(m2)
                s1 = _eval_score_delta(f, conv, fsum, j, m1, n)
                s2 = _eval_score_delta(f, conv, fsum, j, m2, n)
                
                if s1 < s2:
                    hi = m2
                else:
                    lo = m1
            
            delta = (lo + hi) / 2.0
            new_fj = fj + delta
            new_fj = max(0.0, new_fj)
            delta = new_fj - fj
            
            if abs(delta) > 1e-12:
                # Update f and conv
                f[j] = new_fj
                fsum += delta
                # Update conv
                for k in range(k_min, k_max + 1):
                    if k == 2 * j:
                        conv[k] += 2 * fj * delta + delta * delta
                    else:
                        conv[k] += 2 * delta * f[k - j]

                if new_score < best_score - 1e-12:

                elif new_score > best_score * 1.05:

                    f[j] = fj

def _eval_score_delta(f, conv, fsum, j, delta, n):
    """Evaluate the score if f[j] changes by delta (without actually modifying f or conv)."""
    fj = f[j]
    new_fsum = fsum + delta
    if new_fsum < 1e-10:
        return float('inf')
    
    k_min = j
    k_max = min(j + n - 1, 2 * n - 2)
    
    # Max of unchanged entries
    # This is expensive to compute every time. For now, use a bound:
    # max_unchanged <= max(conv) (the current max)
    # This is an upper bound, which is fine for ternary search (we're minimizing).
    
    # Actually, for correctness of the ternary search, I need the exact value.
    # Let me compute it properly.
    
    # The max is max(max of unchanged, max of changed)
    # Changed entries: k in [k_min, k_max]
    # For k in [k_min, k_max], k != 2j: new_val = conv[k] + 2*delta*f[k-j]
    # For k = 2j (if in range): new_val = conv[2j] + 2*fj*delta + delta^2
    
    # I'll compute the max of changed entries:
    max_changed = 0.0
    for k in range(k_min, k_max + 1):
        if k == 2 * j:
            val = conv[k] + 2 * fj * delta + delta * delta
        else:
            val = conv[k] + 2 * delta * f[k - j]
        if val > max_changed:
            max_changed = val
    
    # Max of unchanged: this is max(conv[k]) for k not in [k_min, k_max]
    # For efficiency, precompute this. But for now, just use max(conv) as an upper bound.
    # This will make the ternary search slightly suboptimal but still converge.
    
    # Actually, let me be more careful. The unchanged max is:
    max_unchanged = 0.0
    for k in range(2 * n - 1):
        if k < k_min or k > k_max:
            if conv[k] > max_unchanged:
                max_unchanged = conv[k]
    
    total_max = max(max_changed, max_unchanged)
    return 2.0 * n * total_max / (new_fsum * new_fsum)
