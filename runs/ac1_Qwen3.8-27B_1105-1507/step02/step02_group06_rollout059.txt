def build_conv_matrix(f):
    """Build sparse matrix A where (A@g)[k] = sum_j A[k,j]*g[j] = conv(f,g)[k]"""
    n = len(f)
    # A is (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []

        j_min = max(0, k-(n-1))
        j_max = min(k, n-1)
        for j in range(j_min, j_max+1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k-j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

    n = len(f)

    """Iterative reciprocal method to flatten convolution."""
    a = a.copy().astype(np.float64)
    S = float(np.sum(a))
    for _ in range(n_iters):
        conv = np.convolve(a, a)
        # Compute "pressure" at each position
        # pressure_j = sum_k w_k * a_{k-j} where w_k ∝ max(0, conv_k - mean)
        excess = np.maximum(conv - np.mean(conv), 0)
        if np.sum(excess) < 1e-15:
            break
        w = excess / np.sum(excess)
        # pressure_j = sum_k w_k * a_{k-j}
        pressure = np.zeros(len(a))

            if wk < 1e-15:

            j_max = min(k, len(a)-1)

        # Update: a_j = 1 / (1 + pressure_j)
        a = 1.0 / (1.0 + pressure)
        a = a * (S / np.sum(a))

def lse_gradient(f, tau=None):
    """Compute the gradient of the log-sum-exp smoothed max of conv(f,f)."""
    n = len(f)
    conv = np.convolve(f, f)
    if tau is None:
        tau = 2.0 / max(np.max(conv), 1e-10)
    
    # Numerically stable softmax
    shifted = tau * conv - np.max(tau * conv)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    # Gradient: (1/tau) * sum_k weights[k] * (d conv_k / d f_j)

Optimized search for sequences minimizing 2n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
- Log-sum-exp smoothed gradient for stable subgradient steps
- Dual-variable extraction from LP for informed gradient direction

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def build_conv_matrix(f):
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def solve_lp(f, M):
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            dual = None
            try:
                dual = result.ineqlin.marginals
            except:
                pass
            return result.x, dual
    except:
        pass
    return None, None


def lse_subgradient(f, tau):
    n = len(f)
    conv = np.convolve(f, f)
    shifted = tau * conv - np.max(tau * conv)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-15:
            continue
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        grad[j_min:j_max + 1] += wk * 2.0 * f[k - j_max:k - j_min + 1][::-1]
    return grad / tau


def project_simplex(x, S):
    """Project x onto {a >= 0, sum(a) = S}."""
    n = len(x)
    # Binary search for the threshold
    lo = float(np.min(x)) - S
    hi = float(np.max(x))
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.maximum(x - mid, 0.0)
        s = float(np.sum(proj))
        if s > S:
            lo = mid
        else:
            hi = mid
    proj = np.maximum(x - (lo + hi) / 2.0, 0.0)
    # Fix numerical issues
    proj = np.maximum(proj, 0.0)
    s = float(np.sum(proj))
    if s > 0:
        proj = proj * (S / s)
    return proj


def subgradient_optimize(f, deadline, tau=5.0, lr_init=0.01, momentum_decay=0.8, max_iter=100000):
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    lr = lr_init
    momentum = np.zeros(n)
    it = 0
    stale = 0
    
    while time.monotonic() < deadline and it < max_iter:
        grad = lse_subgradient(f, tau)
        momentum = momentum_decay * momentum - lr * grad
        new_f = project_simplex(f + momentum, S)
        new_score = compute_score(new_f)
        it += 1
        if new_score < best_score - 1e-12:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            stale = 0
        else:
            stale += 1
            lr *= 0.95
            momentum *= 0.5
            if lr < 1e-8 or stale > 200:
                break
    return best_f, best_score

def iterative_flattening(a, n_iters=30):
    a = a.copy().astype(np.float64)
    S = float(np.sum(a))
    n = len(a)
    for _ in range(n_iters):
        conv = np.convolve(a, a)
        # Smoothed max to get weights
        tau = 5.0 / max(np.max(conv), 1e-10)
        shifted = tau * conv - np.max(tau * conv)
        exp_vals = np.exp(shifted)
        weights = exp_vals / np.sum(exp_vals)
        # Compute pressure at each position
        pressure = np.zeros(n)
        for k in range(2 * n - 1):
            if weights[k] < 1e-15:
                continue
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            pressure[j_min:j_max + 1] += weights[k] * 2.0 * a[k - j_max:k - j_min + 1][::-1]
        # Update: reduce high-pressure positions
        a = 1.0 / (1.0 + 0.1 * pressure)
        a = np.clip(a, 1e-6, None)
        a = a * (S / np.sum(a))
    return a

"""

Key improvements over prior approach:
- Log-sum-exp smoothed gradient for stable subgradient steps
- Dual-variable extraction from LP for informed direction

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    """Fast internal score computation (no type checks)."""
    f = np.clip(f, 0.0, 1000.0)
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * M / (S * S)


def build_conv_matrix(f):
    """Build sparse matrix A where (A@g)[k] = conv(f,g)[k]."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. A_f @ g <= M*1, g >= 0. Returns (primal, dual)."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            dual = None
            try:
                if hasattr(result, 'ineqlin') and result.ineqlin is not None:
                    dual = np.asarray(result.ineqlin.marginals, dtype=np.float64)
            except Exception:
                pass
            return result.x, dual
    except Exception:
        pass
    return None, None


def lse_subgradient(f, tau):
    """Compute gradient of log-sum-exp smoothed max of conv(f,f)."""
    n = len(f)
    conv = np.convolve(f, f)
    tconv = tau * conv
    max_tconv = np.max(tconv)
    exp_vals = np.exp(tconv - max_tconv)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-15:
            continue
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        grad[j_min:j_max + 1] += wk * 2.0 * f[k - j_max:k - j_min + 1][::-1]
    return grad / tau


def project_simplex(x, S):
    """Project x onto {a >= 0, sum(a) = S} via bisection."""
    n = len(x)
    lo = float(np.min(x)) - S
    hi = float(np.max(x))
    for _ in range(64):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.maximum(x - mid, 0.0)))
        if s > S:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    proj = np.maximum(x - tau, 0.0)
    s = float(np.sum(proj))
    if s > 1e-15:
        proj = proj * (S / s)
    return proj


def subgradient_optimize(f, deadline, tau=5.0, lr_init=0.01, momentum_decay=0.85, max_iter=100000):
    """Subgradient descent with momentum and LSE smoothing."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    lr = lr_init
    momentum = np.zeros(n)
    it = 0
    stale = 0
    while time.monotonic() < deadline and it < max_iter:
        grad = lse_subgradient(f, tau)
        momentum = momentum_decay * momentum - lr * grad
        new_f = project_simplex(f + momentum, S)
        new_score = compute_score(new_f)
        it += 1
        if new_score < best_score - 1e-13:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            stale = 0
        else:
            stale += 1
            lr *= 0.95
            momentum *= 0.5
            if lr < 1e-9 or stale > 300:
                break
    return best_f, best_score

def iterative_flattening(a, n_iters=40):
    """Iterative reciprocal method to flatten convolution."""
    a = a.copy().astype(np.float64)
    S = float(np.sum(a))
    n = len(a)
    for _ in range(n_iters):
        conv = np.convolve(a, a)
        tau = 5.0 / max(np.max(conv), 1e-10)
        tconv = tau * conv
        exp_vals = np.exp(tconv - np.max(tconv))
        weights = exp_vals / np.sum(exp_vals)
        pressure = np.zeros(n)
        for k in range(2 * n - 1):
            wk = weights[k]
            if wk < 1e-15:
                continue
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            pressure[j_min:j_max + 1] += wk * 2.0 * a[k - j_max:k - j_min + 1][::-1]
        a = 1.0 / (1.0 + 0.1 * pressure)
        a = np.clip(a, 1e-8, None)
        a = a * (S / np.sum(a))
    return a
