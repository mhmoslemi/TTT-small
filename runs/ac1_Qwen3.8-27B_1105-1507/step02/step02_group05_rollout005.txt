2. L-BFGS-B with adaptive beta (smoothness parameter)

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def soft_max_value(b, beta):
    """Numerically stable soft-max."""
    b_max = b.max()
    return float(b_max + np.log(np.sum(np.exp(beta * (b - b_max)))) / beta)


def soft_max_weights(b, beta):
    """Weights for soft-max gradient."""
    b_max = b.max()
    e = np.exp(beta * (b - b_max))
    return e / e.sum()


def soft_max_grad_wrt_a(a, w):
    """
    Compute gradient of soft_max(conv(a,a)) w.r.t. a.
    Uses FFT-based cross-correlation for O(n log n) computation.
    grad[j] = 2 * sum_k w[k] * a[k-j] (for valid indices)
    """
    n = len(a)
    N = 1
    while N < 3 * n:
        N <<= 1
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    cross = np.fft.irfft(fw * np.conj(fa), N)[:n]
    return 2.0 * cross


def objective_and_grad(x, beta):
    """
    Objective and gradient for L-BFGS-B.
    F(a) = 2n * soft_max(conv(a,a)) / sum(a)^2
    """
    a = np.maximum(x, 1e-12)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    b = fast_conv(a)
    sm = soft_max_value(b, beta)
    w = soft_max_weights(b, beta)
    grad_sm = soft_max_grad_wrt_a(a, w)
    F = 2.0 * n * sm / (s * s)
    grad = (2.0 * n / (s * s)) * (grad_sm - 2.0 * sm / s)
    return F, grad

def coord_descent_polish(a, max_rounds=50, deadline=None):
    """
    Coordinate descent: for each coordinate, try values that equalize
    the top peaks in the convolution.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    n = len(a)
    current_J = compute_J(a)

    for rnd in range(max_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        b = fast_conv(a)
        s = a.sum()

        # Find top peaks
        num_peaks = min(5, 2 * n - 1)
        peak_indices = np.argsort(b)[-num_peaks:]

        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break

            # Candidate values for a[j]:
            # 1. Current value
            # 2. Zero
            # 3. Values that would equalize the contribution to top peaks
            candidates = [a[j], 0.0]

            for k in peak_indices:
                # b[k] = sum_i a[i]*a[k-i], the term involving a[j] is a[j]*a[k-j] (if valid)
                # + a[j]*a[j-k] (if valid, and j != k-j)
                contrib = 0.0
                if 0 <= k - j < n:
                    contrib += a[k - j]
                if 0 <= j - k < n and j != k - j:
                    contrib += a[j - k]
                # Actually more carefully: b[k] includes a[j]*a[k-j] when 0<=k-j<n
                # and a[k-j]*a[j] is the same term. Plus if j appears as the second index:
                # b[k] = sum_i a[i]*a[k-i]. a[j] appears when i=j (giving a[j]*a[k-j])
                # and when k-i=j, i.e., i=k-j (giving a[k-j]*a[j]).
                # So the total contribution of a[j] to b[k] is:
                #   a[k-j] (from i=j) + a[k-j] (from i=k-j) = 2*a[k-j] if j != k-j
                #   a[j] * 2 (from i=j=k-j) if j = k-j
                #   a[k-j] (only from i=j) if k-j is out of [0,n) range for i=k-j
                # This is getting complex. Let me just use a simpler approach.

                # Simple approach: try a[j] values that would make b[k] equal to
                # the second-highest peak (to reduce the max)
                if contrib > 1e-10:
                    # To reduce b[k] to target, we need a[j] such that
                    # b[k] - a[j]*contrib + new_a[j]*contrib = target
                    # new_a[j] = a[j] + (target - b[k])/contrib
                    target = np.median(b)
                    new_val = a[j] + (target - b[k]) / max(contrib, 1e-10)
                    if new_val > 0:
                        candidates.append(new_val)

            # Also try small perturbations
            candidates.append(max(0, a[j] * 0.8))
            candidates.append(max(0, a[j] * 1.2))

            best_val = a[j]
            best_F = current_J
            s_new_base = s - a[j]

            for cv in candidates:
                if cv < 0 or not np.isfinite(cv):
                    continue
                s_new = s_new_base + cv
                if s_new < 1e-10:
                    continue
                # Quick estimate: just check if this coordinate change helps
                # by computing the new max conv value
                a_new = a.copy()
                a_new[j] = cv
                b_new = fast_conv(a_new)
                F_new = 2.0 * n * b_new.max() / (s_new * s_new)
                if F_new < best_F - 1e-14:
                    best_F = F_new
                    best_val = cv

            if best_val != a[j]:
                a[j] = best_val
                current_J = best_F
                improved = True

        if not improved:
            break

    return a


def l_bfgsb_soft_max(a, beta_start, beta_end, max_iter, deadline):
    """
    Run L-BFGS-B with increasing beta (adaptive smoothness).
    """
    a = a.astype(np.float64).copy()
    n = len(a)
    bounds = [(0.0, None)] * n

    # Split iterations into phases with increasing beta
    num_phases = 4
    iters_per_phase = max(10, max_iter // num_phases)

    for phase in range(num_phases):
        if time.monotonic() > deadline:
            break
        t_frac = phase / max(1, num_phases - 1)
        beta = beta_start * (beta_end / beta_start) ** t_frac

        def obj(x):
            return objective_and_grad(x, beta)

        try:
            res = minimize(
                obj, a, method='L-BFGS-B', jac=True,
                bounds=bounds,
                options={'maxiter': iters_per_phase, 'max_fun': iters_per_phase * n,
                         'ftol': 1e-18, 'gtol': 1e-10, 'iprint': 0}
            )
            if res.x is not None:
                a = np.maximum(res.x, 0.0)
        except Exception:
            pass

    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        # Resample parent to length n
        if parent.size != n:
            t_old = np.linspace(0, 1, parent.size)
            p_resampled = np.interp(t_arr, t_old, parent)
        else:
            p_resampled = parent.copy()
        inits.append(np.maximum(p_resampled, 0.01))

    # Uniform
    inits.append(np.ones(n) * 0.5)

    # Random
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine waves
    for freq in [1, 2, 3, 4, 5, 7, 11]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr + rng.uniform(0, 6.28)))

    # Chirp
    inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * (t_arr ** 2) * 5))

    # Gaussian
    inits.append(np.exp(-2.0 * (t_arr - 0.5) ** 2) + 0.05)

    # Sine absolute
    inits.append(np.abs(np.sin(np.pi * t_arr * 3)) + 0.1)

    # Power law
    inits.append((t_arr + 0.05) ** rng.uniform(-0.5, 1.5) + 0.05)

    # Sparse spikes smoothed
    sparse = np.zeros(n)
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Alternating pattern
    alt = np.ones(n) * 0.5
    alt[::2] = 1.0
    inits.append(alt)

    # Tukey window
    r = 0.3
    tukey = np.ones(n)
    m = int(n * r / 2)
    for i in range(m):
        val = 0.5 * (1 + np.cos(np.pi * (i / m - 1)))
        tukey[i] = val
        tukey[n - 1 - i] = val
    inits.append(tukey)

    # Hamming window
    ham = 0.54 - 0.46 * np.cos(2 * np.pi * t_arr)
    inits.append(ham + 0.1)

    # Random + structure
    base = rng.uniform(0.5, 1.0, n)
    base += 0.2 * np.cos(2 * np.pi * rng.uniform(2, 8) * t_arr)
    inits.append(np.maximum(base, 0.1))

    return inits


class ScoreTracker:

    def update(self, a):

            elapsed = time.monotonic() - self.t_start

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)

        pass

    # === Phase 1: Soft-max L-BFGS-B across lengths ===
    lengths = [128, 256, 512, 1024, 2048]

    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1 L-BFGS-B n={n} {'='*50}", flush=True)

        inits = generate_initializations(n, rng, parent)

        # Choose number of initializations based on available time
        time_remaining = deadline - time.monotonic()
        est_time_per_init = max(5, 60 * n / 512)
        max_inits = max(3, int(time_remaining / (est_time_per_init * 3)))
        inits = inits[:max_inits]

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.01)
            s = a0.sum()
            if s < 1e-10:
                continue
            a0 = a0 / s

            # Adaptive L-BFGS-B with increasing beta
            beta_start = 5.0
            beta_end = 50.0
            max_iter = max(50, min(200, int(30000 // n)))
            a = l_bfgsb_soft_max(a0, beta_start, beta_end, max_iter, deadline)
            a = np.maximum(a, 0.0)

            # Quick coord descent polish
            a = coord_descent_polish(a, max_rounds=20, deadline=deadline)
            tracker.update(a)

            if idx % 3 == 0:
                print(f"    init {idx}: J={compute_J(a):.8f} t={time.monotonic()-t0:.1f}s", flush=True)

        tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Focused improvement on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2 Focused n={n} {'='*50}", flush=True)

        # Run more iterations from the best
        a = tracker.best.copy()
        a = a / a.sum()

        # Higher precision soft-max
        a = l_bfgsb_soft_max(a, 20.0, 100.0, 300, deadline)
        a = np.maximum(a, 0.0)
        tracker.update(a)

        # Coordinate descent
        a = coord_descent_polish(a, max_rounds=50, deadline=deadline)
        tracker.update(a)

        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Perturbation restarts ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3 Restarts n={n} {'='*50}", flush=True)

        base = tracker.best.copy()
        base = base / base.sum()
        temperature = 0.1
        num_restarts = 30

        for r in range(num_restarts):
            if time.monotonic() > deadline:
                break

            # Gaussian perturbation
            a = base + rng.normal(0, temperature, n)
            a = np.maximum(a, 0.01)
            a = a / a.sum()

            # Quick L-BFGS-B refinement
            a = l_bfgsb_soft_max(a, 10.0, 50.0, 80, deadline)
            a = np.maximum(a, 0.0)
            tracker.update(a)

            # Also try sparse perturbation (zero out random subset)
            if r % 3 == 2:
                a2 = base.copy()
                mask = rng.random(n) < 0.1
                a2[mask] = 0.0
                a2 = np.maximum(a2, 0.01)
                a2 = a2 / a2.sum()
                a2 = l_bfgsb_soft_max(a2, 10.0, 50.0, 50, deadline)
                a2 = np.maximum(a2, 0.0)
                tracker.update(a2)

            temperature *= 0.95

        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Try different lengths from best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase4 Multi-length {'='*50}", flush=True)

        best_n = tracker.best.size
        # Try nearby lengths
        for n_try in [best_n // 2, best_n * 2, best_n + 64, best_n - 64]:
            if time.monotonic() > deadline:
                break
            if n_try < 32 or n_try > 4096:
                continue

            # Resample best to new length
            t_old = np.linspace(0, 1, tracker.best.size)
            t_new = np.linspace(0, 1, n_try)
            a = np.interp(t_new, t_old, tracker.best)
            a = np.maximum(a, 0.01)
            a = a / a.sum()

            a = l_bfgsb_soft_max(a, 10.0, 60.0, 150, deadline)
            a = np.maximum(a, 0.0)
            a = coord_descent_polish(a, max_rounds=30, deadline=deadline)
            tracker.update(a)

        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final high-precision polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase5 Final Polish n={n} {'='*50}", flush=True)

        a = tracker.best.copy()
        a = a / a.sum()

        # Very high beta for near-exact max
        a = l_bfgsb_soft_max(a, 50.0, 200.0, 200, deadline)
        a = np.maximum(a, 0.0)
        tracker.update(a)

        # Final coordinate descent
        a = coord_descent_polish(a, max_rounds=100, deadline=deadline)
        tracker.update(a)

        # One more L-BFGS-B pass
        a = l_bfgsb_soft_max(a, 100.0, 300.0, 100, deadline)
        a = np.maximum(a, 0.0)
        tracker.update(a)

        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256) * 0.5
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
