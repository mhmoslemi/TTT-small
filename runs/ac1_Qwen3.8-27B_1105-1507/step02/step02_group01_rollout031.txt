```python
"""
Step function coefficient optimization using softmax parameterized
gradient descent with Adam, LP-based refinement, coordinate descent,
and block perturbation. Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute the objective function."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def soft_max_value_and_grad(conv, T):
    """Compute soft-max approximation of max and its gradient w.r.t. conv."""
    max_c = float(np.max(conv))
    shifted = np.clip((conv - max_c) / T, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    sm_val = max_c + T * np.log(sum_exp)
    p = exp_vals / sum_exp
    return sm_val, p


def grad_objective_wrt_a(a, p):
    """
    Compute gradient of sum_k p[k]*conv[k] w.r.t. a.
    conv[k] = sum_j a[j]*a[k-j]
    d(conv[k])/d(a[i]) = 2*a[k-i] if 0 <= k-i <= n-1
    grad_a[i] = 2 * sum_k p[k] * a[k-i] * I(0<=k-i<=n-1)
    = 2 * (cross-correlation of p and a) at appropriate indices
    """
    n = len(a)
    # p has length 2n-1
    # grad_a[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    # = 2 * np.correlate(p, a, 'full')[i + n - 1]
    corr = np.correlate(p, a, mode='full')
    grad_a = 2.0 * corr[n-1:2*n-1]
    return grad_a


def softmax(x):
    """Stable softmax."""
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    s = np.sum(exp_x)
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / s


def softmax_gd(a_init, max_time, T_init=None, verbose=True):
    """
    Gradient descent in softmax parameterization with Adam optimizer.
    Minimizes 2n * soft_max(conv(a,a), T) subject to a = softmax(x), sum(a)=1.
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Initialize x from a
    x = np.log(a + 1e-15)
    x = x - np.max(x)  # for numerical stability
    
    conv = np.convolve(a, a)
    if T_init is None:
        T_init = 0.05 * float(np.max(conv))
    T = T_init
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adam parameters
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = 5.0
    t = 0
    last_improve = 0
    patience = 200
    
    while time.monotonic() - start < max_time:
        t += 1
        
        # Compute current a from x
        a = softmax(x)
        conv = np.convolve(a, a)
        
        # Soft-max gradient
        sm_val, p = soft_max_value_and_grad(conv, T)
        grad_a = grad_objective_wrt_a(a, p)
        
        # Convert to gradient w.r.t. x
        # d(obj)/dx[j] = a[j] * (grad_a[j] - sum_i grad_a[i]*a[i])
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)
        
        # Scale by 2n (the objective is 2n * sm_val)
        grad_x *= 2.0 * n
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * grad_x * grad_x
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate
        a_new = softmax(x)
        obj = compute_obj(a_new)
        
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            last_improve = t
            # Gradually decrease T
            T = max(0.001 * float(np.max(conv)), T * 0.998)
        else:
            if t - last_improve > patience:
                # Restart with perturbed state
                x = np.log(best_a + 1e-15) - np.max(np.log(best_a + 1e-15))
                T = max(0.001, T * 0.5)
                lr = min(lr * 1.3, 20.0)
                m = np.zeros(n)
                v = np.zeros(n)
                last_improve = t
                patience = int(patience * 0.8)
            lr = max(0.1, lr * 0.995)
    
    return best_a, best_obj


def build_conv_matrix(a):
    """Build the matrix A such that (A @ g)[k] = (a * g)[k] for linear convolution."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_refine(a, max_iters=50, max_time=30.0, verbose=True):
    """LP-based refinement: find direction g that reduces max convolution."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s  # normalize
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    # Pre-build the matrix (it changes with a, so we rebuild each iteration)
    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break
        
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        A = build_conv_matrix(a)
        
        # LP: max sum(g) s.t. A@g <= M, g >= 0
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), 
                     bounds=[(0.0, 10.0)]*n, method='highs')
        
        if res.status != 0:
            break
        
        g = res.x
        g_sum = float(-res.fun)
        
        if g_sum <= 1.0 + 1e-4:
            stall += 1
            if stall > 5:
                break
            continue
        stall = 0
        
        # Normalized direction
        g = g / g_sum
        
        # Line search for best step size
        best_t = 0.0
        best_obj_ls = best_obj
        for t in [0.8, 0.5, 0.3, 0.2, 0.1, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, None)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t
        
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.clip(a, 0.0, None)
            s = np.sum(a)
            a = a / s
            if best_obj_ls < best_obj:
                best_a = a.copy()
                best_obj = best_obj_ls
            if verbose and it % 10 == 0:
                print(f"    LP iter {it}: obj={best_obj:.6f}, t={best_t:.2f}")
    
    return best_a, best_obj


def coordinate_descent(a, max_time=20.0, rng=None):
    """Coordinate descent: optimize one element at a time."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        improved = False
        # Random order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start > max_time:
                break
            # Try to optimize a[i] while keeping others fixed
            # The objective is 2n * max(conv) / (sum)^2
            # If we fix all a[j] for j != i, and let a[i] = v:
            # conv[k] = conv_base[k] + v * (a[k-i] + a[i] if k != 2i, or 2a[i] if ...)
            # Actually: conv[k] = sum_j a[j]*a[k-j]
            # Terms involving a[i]=v: v*a[k-i] (from j=i) + a[k-i]*v (from k-j=i, j=k-i)
            # If k-i = i (k=2i): v*a[i] = v^2, and the other term is a[i]*v = v^2, but they're the same term
            # So: conv[k] = conv_other[k] + v * b[k] where b[k] = a[k-i] + (a[i] if k=2i else a[k-i])... 
            # Let me just do it numerically.
            
            # Precompute conv with a[i] = 0
            a_zero = a.copy()

def block_refine(a, rng, max_time=30.0):

    n = len(a)

    best_obj = compute_obj(a)

        pos = int(rng.integers(0, max(1, n - width + 1)))
        candidate = best_a.copy()

        sigma = 0.02 + 0.08 * (1.0 - min(1.0, it / 50000.0))
        log_perturb = rng.normal(0, sigma, width)
        candidate[pos:pos+width] *= np.exp(log_perturb)
        candidate = np.clip(candidate, 0.0, None)

            candidate[src] -= amount
            candidate[pos:pos+width] += amount / width
            candidate = np.clip(candidate, 0.0, None)

        if s < 1e-10:
            continue
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj
```

```python
"""
Step function coefficient optimization using softmax parameterized gradient
descent with Adam, LP-based refinement, and adaptive block perturbation.
Explores multiple sequence lengths and initialization strategies.
Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def soft_max_value_and_grad(conv, T):
    """Compute soft-max of conv and gradient weights p (proportional to exp((conv-max)/T))."""
    max_c = float(np.max(conv))
    shifted = np.clip((conv - max_c) / T, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    sm_val = max_c + T * np.log(sum_exp)
    return sm_val, p


def grad_objective_wrt_a(a, p):
    """
    Compute d(sum_k p[k]*conv[k])/d(a[i]) for linear self-convolution.
    Result: 2 * (cross-correlation of p and a) at indices [n-1, 2n-2].
    """
    n = len(a)
    corr = np.correlate(p, a, mode='full')
    return 2.0 * corr[n-1:2*n-1]


def softmax(x):
    """Numerically stable softmax."""
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    s = np.sum(exp_x)
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / s


def softmax_gd(a_init, max_time, T_init=None, verbose=True):
    """
    Gradient descent in softmax parameterization with Adam optimizer.
    Minimizes 2n * soft_max(conv(a,a), T) where a = softmax(x), sum(a)=1.
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    conv_init = np.convolve(a, a)
    if T_init is None:
        T_init = 0.05 * float(np.max(conv_init))
    T = T_init

    # Initialize x from a (log parameterization, shifted for stability)
    x = np.log(a + 1e-15)
    x = x - np.max(x)

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Adam parameters
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = 5.0
    t = 0
    last_improve = 0
    patience = 300

    while time.monotonic() - start < max_time:
        t += 1

        a = softmax(x)
        conv = np.convolve(a, a)

        sm_val, p = soft_max_value_and_grad(conv, T)
        grad_a = grad_objective_wrt_a(a, p)

        # Convert to gradient w.r.t. x via chain rule for softmax
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot) * (2.0 * n)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax(x)
        obj = compute_obj(a_new)

        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            last_improve = t
            # Gradually anneal temperature
            T = max(0.001 * float(np.max(conv)), T * 0.997)
        else:
            if t - last_improve > patience:
                # Periodic restart from best with perturbation
                x = np.log(best_a + 1e-15)
                x = x - np.max(x)
                x += rng_perturb(x, 0.01)
                T = max(0.0005, T * 0.4)
                lr = min(lr * 1.5, 30.0)
                m = np.zeros(n)
                v = np.zeros(n)
                last_improve = t
                patience = max(50, int(patience * 0.7))
            lr = max(0.05, lr * 0.997)

    return best_a, best_obj


def rng_perturb(x, scale):
    """Deterministic-ish perturbation for reproducibility within a call."""
    # Use a simple hash-based approach to avoid needing rng
    seed_val = int(np.sum(np.abs(x)) * 1e6) % (2**31)
    rng_local = np.random.default_rng(seed_val)
    return rng_local.standard_normal(len(x)) * scale


def build_conv_matrix(a):
    """Build matrix A where (A @ g)[k] = (a * g)[k] for linear convolution."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_refine(a, max_iters=40, max_time=30.0, verbose=True):
    """LP-based refinement: find direction g reducing max convolution."""
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break

        conv = np.convolve(a, a)
        M = float(np.max(conv))
        A = build_conv_matrix(a)

        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0.0, 10.0)] * n, method='highs')

        if res.status != 0:
            break

        g = res.x
        g_sum = float(-res.fun)

        if g_sum <= 1.0 + 1e-4:
            stall += 1
            if stall > 6:
                break
            continue
        stall = 0

        g = g / g_sum

        # Line search
        best_t = 0.0
        best_obj_ls = best_obj
        for t_candidate in [0.8, 0.5, 0.3, 0.2, 0.12, 0.06]:
            a_new = (1.0 - t_candidate) * a + t_candidate * g
            a_new = np.clip(a_new, 0.0, None)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_obj_ls:
                best_obj_ls = obj_new
                best_t = t_candidate

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.clip(a, 0.0, None)
            s = np.sum(a)
            a = a / s
            best_a = a.copy()
            best_obj = best_obj_ls
            if verbose and it % 5 == 0:
                print(f"    LP iter {it}: obj={best_obj:.6f}, t={best_t:.3f}")
        else:
            stall += 1
            if stall > 8:
                break

    return best_a, best_obj


def block_refine(a, rng, max_time=30.0):
    """Adaptive block perturbation with mass transfer."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    total_iters = int(max_time / 0.002)  # estimate
    total_iters = min(total_iters, 200000)

    while time.monotonic() - start < max_time and it < total_iters:
        it += 1
        width = max(1, int(n * rng.uniform(0.02, 0.12)))
        pos = int(rng.integers(0, max(1, n - width + 1)))
        candidate = best_a.copy()

        # Gaussian log-perturbation
        sigma = 0.02 + 0.06 * (1.0 - min(1.0, it / max(1, total_iters * 0.5)))
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))

        # Occasional mass transfer
        if rng.random() < 0.25:
            src = int(rng.integers(0, n))
            amount = best_a[src] * rng.uniform(0.02, 0.15)
            candidate[src] = max(0, candidate[src] - amount)
            target = int(rng.integers(0, n))
            candidate[target] += amount

        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj

    return best_a, best_obj


def init_edge_concentrated(n, rng):
    """Concentrate mass at edges with smooth interior."""
    a = np.zeros(n)
    c = 1.0 / np.sqrt(2.0 * n - 1.0)
    a[0] = c * 1.8
    a[1] = c
    a[-1] = c * 1.8
    a[-2] = c
    remaining = 1.0 - a[0] - a[1] - a[-1] - a[-2]
    if n > 4 and remaining > 0:
        x = np.linspace(0, 1, n - 4)
        envelope = 1.0 - 0.5 * np.cos(np.pi * x)
        a[2:n - 2] = remaining * envelope / np.sum(envelope)
    a = np.abs(a)
    a += 0.005 * rng.random(n)
    s = np.sum(a)
    return a / s


def init_cosine(n, rng):
    """Raised cosine envelope."""
    x = np.linspace(0, np.pi, n)
    a = np.cos(x) ** 2 + 0.05
    a += 0.03 * rng.random(n)
    a = np.maximum(0, a)
    s = np.sum(a)
    return a / s if s > 1e-10 else np.ones(n) / n


def init_random_gaussian(n, rng):
    """Random gaussian positive values."""
    a = np.maximum(0, rng.normal(1.0, 0.6, n))
    a += 0.01
    s = np.sum(a)
    return a / s if s > 1e-10 else np.ones(n) / n


def init_periodic(n, rng):
    """Periodic pattern with noise."""
    period = int(rng.integers(max(4, n // 15), max(5, n // 4)))
    pattern = np.maximum(0.05, rng.random(period))
    a = np.tile(pattern, n // period + 1)[:n]
    a += 0.02 * rng.random(n)
    s = np.sum(a)
    return a / s if s > 1e-10 else np.ones(n) / n


def init_bipolar(n, rng):
    """Two clusters of high values."""
    a = np.full(n, 0.1)
    c1 = int(n * rng.uniform(0.1, 0.3))
    c2 = int(n * rng.uniform(0.6, 0.8))
    a[c1 - 2:c1 + 3] += 1.0
    a[c2 - 2:c2 + 3] += 1.0
    a = np.maximum(0, a + 0.02 * rng.random(n))
    s = np.sum(a)
    return a / s


def full_pipeline(a_init, time_budget, rng):
    """Run the complete optimization pipeline with time management."""
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: Quick LP to get shape right
    if time_budget > 15:
        lp_time1 = min(12.0, time_budget * 0.2)
        a1, obj1 = lp_refine(a, max_iters=25, max_time=lp_time1, verbose=False)
        if obj1 < best_obj:
            best_a, best_obj = a1, obj1
            a = a1

    # Phase 2: Softmax GD (main workhorse)
    remaining = time_budget - (time.monotonic() - start)
    gd_time = min(25.0, remaining * 0.4)
    if gd_time > 5:
        a2, obj2 = softmax_gd(a, max_time=gd_time)
        if obj2 < best_obj:
            best_a, best_obj = a2, obj2
            a = a2

    # Phase 3: LP refinement from GD result
    remaining = time_budget - (time.monotonic() - start)
    lp_time2 = min(10.0, remaining * 0.2)
    if lp_time2 > 3:
        a3, obj3 = lp_refine(a, max_iters=25, max_time=lp_time2, verbose=False)
        if obj3 < best_obj:
            best_a, best_obj = a3, obj3
            a = a3

    # Phase 4: Fine GD with low temperature
    remaining = time_budget - (time.monotonic() - start)
    gd_time2 = min(10.0, remaining * 0.25)
    if gd_time2 > 3:
        a4, obj4 = softmax_gd(a, max_time=gd_time2, T_init=0.003)
        if obj4 < best_obj:
            best_a, best_obj = a4, obj4
            a = a4

    # Phase 5: Block perturbation (polish)
    remaining = time_budget - (time.monotonic() - start) - 1.0
    if remaining > 3:
        a5, obj5 = block_refine(a, rng, max_time=remaining)
        if obj5 < best_obj:
            best_a, best_obj = a5, obj5

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization: structured init -> LP -> softmax GD -> LP ->
    fine GD -> block perturbation. Explores multiple n values and restarts.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf

    # Evaluate parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Optimize parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all() and parent.size <= 500:
            p_time = min(50.0, float(budget_s) * 0.12)
            p_opt, p_obj = full_pipeline(parent.copy(), p_time, rng)
            if p_obj < best_score:
                best_score = p_obj
                best_seq = p_opt.copy()
                print(f"Parent optimized: obj={p_obj:.6f}")
    except Exception as e:
        print(f"Parent opt failed: {e}")

    # Main search over n values
    n_values = [200, 250, 300, 350, 400, 500]
    init_fns = [init_edge_concentrated, init_cosine, init_random_gaussian,
                init_periodic, init_bipolar]

    for n in n_values:
        if time.monotonic() > deadline - 25:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        num_remaining = len([x for x in n_values if x >= n])
        time_per_n = min(remaining / max(1, num_remaining), 100.0)
        num_restarts = max(1, min(3, int(time_per_n / 35)))
        time_per_restart = time_per_n / num_restarts

        print(f"\nN={n}: rem={remaining:.0f}s, {num_restarts} restarts ({time_per_restart:.0f}s each)")

        for r in range(num_restarts):
            if time.monotonic() > deadline - 25:
                break

            init_fn = init_fns[(r + n) % len(init_fns)]

            if best_seq is not None and len(best_seq) == n and r >= 1:
                # Perturb best found
                a_init = best_seq * np.exp(rng.normal(0, 0.05, n))
                a_init = np.clip(a_init, 0, None)
                s_init = np.sum(a_init)
                if s_init > 1e-10:
                    a_init = a_init / s_init
                else:
                    a_init = init_fn(n, rng)
            else:
                a_init = init_fn(n, rng)

            a_opt, obj_opt = full_pipeline(a_init, time_per_restart, rng)

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} r={r}: obj={obj_opt:.6f} ** NEW BEST **")
            else:
                print(f"  n={n} r={r}: obj={obj_opt:.6f}")

    # Final polish on best
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        print(f"\nFinal polish: {remaining:.0f}s remaining")

        # Full pipeline on best
        a_ref, obj_ref = full_pipeline(best_seq.copy(), min(remaining * 0.5, 60.0), rng)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  After pipeline: obj={best_score:.6f}")

        # Extended block perturbation
        if time.monotonic() < deadline - 5:
            bp_time = deadline - time.monotonic() - 3.0
            a_bp, obj_bp = block_refine(best_seq, rng, max_time=bp_time)
            if obj_bp < best_score:
                best_score = obj_bp
                best_seq = a_bp.copy()
                print(f"  After block refine: obj={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(200) / 200
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
```
