"""
Optimization of non-negative step function to minimize 2n*max(conv(f,f))/sum(f)^2.
Key insight: U-shaped (bowl) sequences outperform hill-shaped (window) sequences.
Uses multiplicative exponential updates with exact subgradient, LP refinement,
and multi-start with diverse U-shaped initializations.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def u_shape_init(n, curvature=1.0, power=2.0):
    """Generate U-shaped initialization: high at edges, low in middle."""
    x = np.linspace(-1, 1, n)
    return 1.0 + curvature * np.abs(x) ** power


def v_shape_init(n, curvature=1.0):
    """V-shaped: high at edges, linear dip in middle."""
    x = np.linspace(-1, 1, n)
    return 1.0 + curvature * np.abs(x)


def exp_u_init(n, sharpness=2.0):
    """Exponential U: f[j] = exp(sharpness * (2j/(n-1) - 1)^2)"""
    x = np.linspace(-1, 1, n)
    return np.exp(sharpness * x ** 2)


def tukey_window(n, r=0.3):
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def compute_exact_subgradient(f, top_k=5):
    """
    Compute subgradient of max(conv(f,f)) using top-k positions.
    Returns (gradient, max_value, top_positions).
    """
    n = len(f)
    C = np.convolve(f, f)
    # Find top-k positions
    k = min(top_k, len(C))
    top_idx = np.argsort(C)[-k:][::-1]
    top_vals = C[top_idx]
    max_val = C[top_idx[0]]
    
    # Subgradient at position p: grad[i] = 2 * sum of f[j] where i+j=p (with bounds)
    # For the top-k weighted subgradient:
    weights = top_vals / top_vals.sum()
    grad = np.zeros(n)
    for w, p in zip(weights, top_idx):
        j_min = max(0, p - (n - 1))
        j_max = min(p, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += w * f[p - j]
    grad *= 2.0
    return grad, float(max_val)


def multiplicative_optimize(f, deadline, eta_init=0.5, top_k=5, 
                            max_iters=5000, verbose_prefix=""):
    """
    Optimize using multiplicative (exponential) updates with exact subgradient.
    f[i] *= exp(-eta * grad[i] / norm)
    """
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    f = f / S
    n = len(f)
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    eta = eta_init
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        grad, max_val = compute_exact_subgradient(f, top_k)
        
        # Normalize gradient for stability
        grad_max = np.max(np.abs(grad))
        if grad_max < 1e-15:
            break
        grad = grad / grad_max
        
        # Multiplicative update
        log_factor = -eta * grad
        # Clip to avoid overflow/underflow
        log_factor = np.clip(log_factor, -5.0, 5.0)
        new_f = f * np.exp(log_factor)
        new_f = np.clip(new_f, 0.0, 1000.0)
        ns = float(np.sum(new_f))
        if ns < 1e-10:
            new_f = f
            ns = S
        new_f = new_f / ns
        
        sc = evaluate_sequence(new_f.tolist())
        it += 1
        
        if np.isfinite(sc) and sc < best_score:
            f = new_f
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            if it % 200 == 0:
                print(f"    {verbose_prefix}Mult iter {it}: score={best_score:.6f} (eta={eta:.4f})")
        else:
            no_improve += 1
            if no_improve > 50:
                eta *= 0.7
                no_improve = 0
                if eta < 1e-6:
                    break
    
    return best_f, best_score


def softmax_gradient_optimize(f, deadline, beta_start=10.0, beta_end=500.0,
                               lr_start=0.03, max_iters=5000, verbose_prefix=""):
    """Softmax gradient descent with Adam."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        C = np.convolve(f, f)
        max_C = C.max()
        exp_C = np.exp(beta_start * (1 + (beta_end/beta_start - 1) * min(1.0, it / (max_iters * 0.7))) * (C - max_C))
        # Use adaptive beta
        it_frac = min(1.0, it / (max_iters * 0.7))
        beta = beta_start * (beta_end / beta_start) ** it_frac
        exp_C = np.exp(beta * (C - max_C))
        total = exp_C.sum()
        if total < 1e-300:
            break
        p = exp_C / total
        
        corr = np.correlate(p, f, mode='valid')
        grad = 2.0 * corr
        
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        new_f = f - lr * update
        new_f = np.clip(new_f, 0.0, 1000.0)
        ns = float(np.sum(new_f))
        if ns < 1e-10:
            new_f = f
            ns = S
        new_f = new_f / ns
        
        sc = evaluate_sequence(new_f.tolist())
        if np.isfinite(sc) and sc < best_score:
            f = new_f
            best_f = f.copy()
            best_score = sc
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-7:
                    break
    
    return best_f, best_score


def lp_iterate(f, deadline, step_size_init=0.1, max_iters=100, verbose_prefix=""):
    """LP-based iterations."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    f = f / S
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    it = 0
    no_improve = 0
    step_size = step_size_init
    
    # Build matrix once
    A = build_conv_matrix_sparse(f)
    
    while time.monotonic() < deadline and it < max_iters:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break
        
        # Solve LP: max sum(g) s.t. g >= 0, A@g <= M
        c = -np.ones(len(f))
        b_ub = np.full(2 * len(f) - 1, M)
        bounds = [(0, None)] * len(f)
        
        try:
            result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if result.success:
                g = result.x
            else:
                g = None
        except Exception:
            g = None
        it += 1
        
        if g is None or float(np.sum(g)) < 1e-15:
            no_improve += 1
            if no_improve > 15:
                break
            continue
        
        g = g / S  # normalize to same sum
        
        # Line search
        best_t = step_size
        best_s = best_score
        for trial_t in [step_size, step_size * 1.5, step_size * 0.5]:
            new_f = (1.0 - trial_t) * f + trial_t * g
            new_f = np.clip(new_f, 0.0, 1000.0)
            ns = float(np.sum(new_f))
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = trial_t
        
        if best_s < best_score:
            f = (1.0 - best_t) * f + best_t * g
            f = np.clip(f, 0.0, 1000.0)
            f = f / float(np.sum(f))
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            if it % 20 == 0:
                print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f}")
        else:
            no_improve += 1
            step_size *= 0.6
            if step_size < 0.005 or no_improve > 30:
                break
    
    return best_f, best_score


def perturb(f, rng, strength=0.05):
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.1, 2.0)
    new_f = np.clip(new_f, 0.0, 1000.0)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S


def full_optimize(f_init, deadline, verbose_prefix=""):
    """Multi-phase optimization."""
    f_init = f_init.copy().astype(np.float64)
    S = float(np.sum(f_init))
    if S < 1e-10:
        return f_init, evaluate_sequence(f_init.tolist())
    f_init = f_init / S
    
    best_f = f_init.copy()
    best_score = evaluate_sequence(f_init.tolist())
    t0 = time.monotonic()
    total = deadline - t0
    
    # Phase 1: Multiplicative updates (fast)
    p1_deadline = t0 + total * 0.35
    if time.monotonic() < p1_deadline:
        f1, s1 = multiplicative_optimize(f_init, p1_deadline, eta_init=0.5, top_k=5,
                                          max_iters=3000, verbose_prefix=verbose_prefix + "M1")
        if s1 < best_score:
            best_f, best_score = f1, s1
    
    # Phase 2: Softmax gradient (medium)
    p2_deadline = t0 + total * 0.6
    if time.monotonic() < p2_deadline:
        f2, s2 = softmax_gradient_optimize(best_f, p2_deadline, beta_start=10.0,
                                            beta_end=300.0, lr_start=0.03,
                                            max_iters=3000, verbose_prefix=verbose_prefix + "SM")
        if s2 < best_score:
            best_f, best_score = f2, s2
    
    # Phase 3: LP refinement (slow but precise)
    if time.monotonic() < deadline - 3:
        f3, s3 = lp_iterate(best_f, deadline - 3, step_size_init=0.05,
                            max_iters=30, verbose_prefix=verbose_prefix + "LP")
        if s3 < best_score:
            best_f, best_score = f3, s3
    
    return best_f, best_score

"""
Multi-phase optimization minimizing 2n*max(conv(f,f))/sum(f)^2 for non-negative sequences.
Key innovation: U-shaped (bowl) initializations exploit asymmetry to reduce peak autocorrelation

subgradient, LP direction finding, and softmax gradient with Adam.
"""
import numpy as np

def u_shape_init(n, curvature=1.0, power=2.0):
    """Generate U-shaped (bowl) initialization: high at edges, low in middle."""
    x = np.linspace(-1, 1, n)
    return 1.0 + curvature * np.abs(x) ** power


def v_shape_init(n, curvature=1.0):
    """V-shaped initialization."""
    x = np.linspace(-1, 1, n)
    return 1.0 + curvature * np.abs(x)


def exp_u_init(n, sharpness=2.0):
    """Exponential U-shape: exp(sharpness * x^2)."""
    x = np.linspace(-1, 1, n)
    return np.exp(sharpness * x ** 2)


def perfect_conv_init(n):
    """
    Initialize using the 'perfect convolution' sequence.
    For infinite support, conv = constant requires f satisfying a recurrence.
    We compute the first n terms and use that as initialization.
    """
    f = np.zeros(n)
    f[0] = 1.0
    M = 1.0  # target convolution value (= f[0]^2)
    for k in range(1, n):
        # C[k] = sum_{j} f[j]*f[k-j] = M (target)
        # C[k] = 2*f[k]*f[0] + sum_{j=1}^{k-1} f[j]*f[k-j] (for k < n)
        # But also: C[k] = f[k]*f[0] + f[0]*f[k] + sum_{j=1}^{k-1} f[j]*f[k-j]
        #         = 2*f[k]*f[0] + sum_{j=1}^{k-1} f[j]*f[k-j]
        # So: f[k] = (M - sum_{j=1}^{k-1} f[j]*f[k-j]) / (2*f[0])
        s = 0.0
        for j in range(1, k):
            if k - j < n:
                s += f[j] * f[k - j]
        f[k] = (M - s) / (2.0 * f[0])
        if f[k] < 0:
            f[k] = 0.0
    # Also try the reverse
    f_rev = f[::-1].copy()
    # Blend: use the one with lower score
    s1 = evaluate_sequence(f.tolist())
    s2 = evaluate_sequence(f_rev.tolist())
    if s2 < s1:
        return f_rev
    return f


def dual_peaked_init(n, sep=0.7, width=0.15):
    """Two Gaussian peaks at positions sep and 1-sep."""
    x = np.linspace(0, 1, n)
    w = np.exp(-0.5 * ((x - sep) / width) ** 2) + np.exp(-0.5 * ((x - (1 - sep)) / width) ** 2)
    return w + 0.05


def tukey_window(n, r=0.3):
    """Tukey (tapered cosine) window."""
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def compute_exact_subgradient(f, top_k=5):
    """
    Compute weighted subgradient of max(conv(f,f)) using top-k peak positions.
    Returns (gradient, max_value).
    """
    n = len(f)
    C = np.convolve(f, f)
    k = min(top_k, len(C))
    top_idx = np.argsort(C)[-k:][::-1]
    top_vals = C[top_idx]
    max_val = float(C[top_idx[0]])
    
    # Weight by proximity to max
    weights = top_vals / top_vals.sum()
    grad = np.zeros(n)
    for w, p in zip(weights, top_idx):
        j_min = max(0, p - (n - 1))
        j_max = min(p, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += w * f[p - j]
    grad *= 2.0
    return grad, max_val


def multiplicative_optimize(f, deadline, eta_init=0.5, top_k=5,

    S = float(np.sum(f))

        return f, evaluate_sequence(f.tolist())

    best_score = evaluate_sequence(f.tolist())

    no_improve = 0
    it = 0

        new_f = np.maximum(new_f, floor_val)

def softmax_gradient_optimize(f, deadline, beta_start=10.0, beta_end=400.0,
                               lr_start=0.03, max_iters=5000, verbose_prefix=""):
    """Softmax-smax gradient descent with Adam optimizer."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        it_frac = min(1.0, it / (max_iters * 0.7))
        beta = beta_start * (beta_end / beta_start) ** it_frac
        
        C = np.convolve(f, f)
        max_C = C.max()
        exp_C = np.exp(beta * (C - max_C))
        total = exp_C.sum()
        if total < 1e-300:
            break
        p = exp_C / total
        
        corr = np.correlate(p, f, mode='valid')
        grad = 2.0 * corr
        
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        new_f = f - lr * update
        new_f = np.clip(new_f, 1e-10, 1000.0)
        ns = float(np.sum(new_f))
        if ns < 1e-8:
            new_f = f.copy()
            ns = S
        new_f = new_f / ns
        
        sc = evaluate_sequence(new_f.tolist())
        if np.isfinite(sc) and sc < best_score:
            f = new_f
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            if it % 500 == 0:
                print(f"    {verbose_prefix}SM iter {it}: score={best_score:.6f} (beta={beta:.1f})")
        else:
            no_improve += 1
            if no_improve > 150:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-7:
                    break
    
    return best_f, best_score

    while time.monotonic() < deadline and it < max_iters:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 1e-15:
            break
        
        if it % rebuild_every == rebuild_every - 1:
            A = build_conv_matrix_sparse(f)
        
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            g = result.x if result.success else None
        except Exception:
            g = None
        it += 1
        
        if g is None or float(np.sum(g)) < 1e-15:
            no_improve += 1
            if no_improve > 15:
                break
            continue
        
        g = g / S
        
        best_t = step_size
        best_s = best_score
        for trial_t in [step_size, step_size * 1.5, step_size * 0.5, step_size * 0.8]:
            new_f = (1.0 - trial_t) * f + trial_t * g
            new_f = np.clip(new_f, 1e-10, 1000.0)
            ns = float(np.sum(new_f))
            if ns < 1e-8:
                continue
            new_f = new_f / ns
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = trial_t
        
        if best_s < best_score:
            f = (1.0 - best_t) * f + best_t * g
            f = np.clip(f, 1e-10, 1000.0)
            f = f / float(np.sum(f))
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            if it % 10 == 0:
                print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f}")
        else:
            no_improve += 1
            step_size *= 0.65
            if step_size < 0.003 or no_improve > 25:
                break
    
    return best_f, best_score


def perturb(f, rng, strength=0.05):
    """Add multiplicative noise and renormalize."""
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.05, 3.0)
    new_f = np.clip(new_f, 1e-10, 1000.0)
    S = float(np.sum(new_f))
    if S < 1e-8:
        return f.copy()
    return new_f / S


def full_optimize(f_init, deadline, verbose_prefix=""):
    """Run multi-phase optimization: multiplicative -> softmax -> LP."""
    f_init = f_init.copy().astype(np.float64)
    S = float(np.sum(f_init))
    if S < 1e-10:
        return f_init, evaluate_sequence(f_init.tolist())
    f_init = f_init / S
    
    best_f = f_init.copy()
    best_score = evaluate_sequence(f_init.tolist())
    t0 = time.monotonic()
    total = max(deadline - t0, 1.0)
    
    # Phase 1: Multiplicative updates (fast, many iterations)
    p1_deadline = t0 + total * 0.35
    if time.monotonic() < p1_deadline:
        f1, s1 = multiplicative_optimize(f_init, p1_deadline, eta_init=0.5, top_k=5,
                                          max_iters=5000, verbose_prefix=verbose_prefix + "M1")
        if s1 < best_score:
            best_f, best_score = f1, s1
    
    # Phase 2: Softmax gradient (medium speed)
    p2_deadline = t0 + total * 0.65
    if time.monotonic() < p2_deadline:
        f2, s2 = softmax_gradient_optimize(best_f, p2_deadline, beta_start=10.0,
                                            beta_end=350.0, lr_start=0.03,
                                            max_iters=5000, verbose_prefix=verbose_prefix + "SM")
        if s2 < best_score:
            best_f, best_score = f2, s2
    
    # Phase 3: LP refinement (slow but precise)
    if time.monotonic() < deadline - 3:
        f3, s3 = lp_iterate(best_f, deadline - 3, step_size_init=0.05,
                            max_iters=40, verbose_prefix=verbose_prefix + "LP")
        if s3 < best_score:
            best_f, best_score = f3, s3
    
    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    """
    Main search function. Explores U-shaped and other non-standard initializations,
    optimizes with multiplicative/softmax/LP methods across multiple sequence lengths.
    """
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 12.0)
    
    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)
    
    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                score = evaluate_sequence(parent.tolist())
                if np.isfinite(score):
                    best_score = score
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")
    
    # === Phase A: Explore U-shaped initializations at n=512 ===
    n_main = 512
    inits = []
    
    # U-shaped with various parameters
    for curv in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 4.0]:
        inits.append((f'u2_{curv}', u_shape_init(n_main, curvature=curv, power=2.0)))
    for curv in [0.5, 1.0, 2.0, 3.0]:
        inits.append((f'v_{curv}', v_shape_init(n_main, curvature=curv)))
    for sharp in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        inits.append((f'eu_{sharp}', exp_u_init(n_main, sharpness=sharp)))
    
    # Dual peaked
    inits.append(('dual_5', dual_peaked_init(n_main, sep=0.6, width=0.12)))
    inits.append(('dual_7', dual_peaked_init(n_main, sep=0.7, width=0.1)))
    inits.append(('dual_3', dual_peaked_init(n_main, sep=0.5, width=0.2)))
    
    # Perfect convolution init
    inits.append(('perf', perfect_conv_init(n_main)))
    inits.append(('perf_rev', perfect_conv_init(n_main)[::-1].copy()))
    
    # Inverted windows
    x = np.linspace(0, 1, n_main)
    inits.append(('inv_hann', 0.5 * (1 - (1 + np.cos(2 * np.pi * x)) / 2 / 0.5 * 0.5 + 1)))
    # Simpler: 1 - normalized hann
    hann = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(('inv_hann2', 2.0 - hann))
    
    # If parent matches length, include it
    if best_sequence is not None and len(best_sequence) == n_main:
        inits.insert(0, ('parent', best_sequence.copy()))
    
    # Clip all inits to be positive
    inits = [(lbl, np.clip(arr.astype(np.float64), 0.01, 100.0)) for lbl, arr in inits]
    
    print(f"\n=== Phase A: n={n_main}, {len(inits)} initializations ===")
    
    # Quick-evaluate all
    scored = []
    for label, f_init in inits:
        sc = evaluate_sequence(f_init.tolist())
        if np.isfinite(sc):
            scored.append((sc, label, f_init))
            if sc < best_score:
                best_score = sc
                best_sequence = f_init.copy()
                print(f"  New best from init [{label}]: {sc:.6f}")
    
    scored.sort(key=lambda x: x[0])
    print("Top 8 initializations:")
    for sc, label, _ in scored[:8]:
        print(f"  {label}: {sc:.6f}")
    
    # Optimize top initializations with time budget
    n_top = min(6, len(scored))
    remaining = deadline - time.monotonic() - 120  # reserve time for later phases
    per_init = max(remaining / n_top, 20.0)
    
    for i in range(n_top):
        if time.monotonic() > deadline - 120:
            break
        sc, label, f_init = scored[i]
        # Spend more time on better inits
        time_alloc = per_init * (1.0 if i == 0 else 0.8 if i < 3 else 0.6)
        init_dl = min(time.monotonic() + time_alloc, deadline - 110)
        if time.monotonic() >= init_dl - 5:
            continue
        print(f"\n  --- Optimizing [{label}] (score={sc:.6f}, time={time_alloc:.0f}s) ---")
        try:
            bf, bs = full_optimize(f_init, init_dl, verbose_prefix=f"[{label}]")
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()
                print(f"    [{label}] => {bs:.6f} (NEW BEST)")
            else:
                print(f"    [{label}] => {bs:.6f}")
        except Exception as e:
            print(f"    [{label}] error: {e}")
    
    # === Phase B: Try n=1024 ===
    if best_sequence is not None and time.monotonic() < deadline - 180:
        n2 = 1024
        print(f"\n=== Phase B: n={n2} ===")
        
        u2 = u_shape_init(n2, curvature=1.0, power=2.0)
        u2b = u_shape_init(n2, curvature=2.0, power=2.0)
        eu2 = exp_u_init(n2, sharpness=1.5)
        
        # Interpolate best
        from numpy import interp
        interp_best = interp(np.linspace(0, 1, n2),
                             np.linspace(0, 1, len(best_sequence)),
                             best_sequence)
        interp_best = np.clip(interp_best, 0.01, None)
        
        candidates_1024 = [
            ('u_10', u2),
            ('u_20', u2b),
            ('eu_10', eu2),
            ('interp', interp_best),
        ]
        
        # Quick score
        best_1024_init = None
        best_1024_sc = float('inf')
        for lbl, arr in candidates_1024:
            sc = evaluate_sequence(arr.tolist())
            if np.isfinite(sc) and sc < best_1024_sc:
                best_1024_sc = sc
                best_1024_init = arr
            if sc < best_score:
                best_score = sc
                best_sequence = arr.copy()
                print(f"  n=1024 [{lbl}]: {sc:.6f} (NEW BEST)")
        
        if best_1024_init is not None:
            dl_1024 = deadline - 60
            try:
                bf, bs = full_optimize(best_1024_init, dl_1024, verbose_prefix="[n1024]")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  n=1024 optimized: {bs:.6f}")
            except Exception as e:
                print(f"  n=1024 error: {e}")
    
    # === Phase C: Perturbation restarts on best ===
    if best_sequence is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Phase C: Restarts (best={best_score:.6f}) ===")
        restart_dl = deadline - 30
        n_restarts = 4
        per_restart = (restart_dl - time.monotonic()) / n_restarts
        
        for r in range(n_restarts):
            if time.monotonic() > restart_dl:
                break
            strengths = [0.03, 0.05, 0.08, 0.12]
            f_p = perturb(best_sequence, rng, strength=strengths[r])
            r_dl = min(time.monotonic() + per_restart, restart_dl)
            if time.monotonic() >= r_dl - 5:
                continue
            try:
                bf, bs = full_optimize(f_p, r_dl, verbose_prefix=f"[R{r}]")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  Restart {r} (str={strengths[r]}): {bs:.6f} (NEW BEST)")
                else:
                    print(f"  Restart {r} (str={strengths[r]}): {bs:.6f}")
            except Exception as e:
                print(f"  Restart {r} error: {e}")
    
    # === Phase D: Final refinement ===
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print(f"\n=== Phase D: Final refinement ===")
        try:
            bf, bs = multiplicative_optimize(best_sequence, deadline - 8, eta_init=0.08,
                                              top_k=3, max_iters=5000, verbose_prefix="[fin_M]")
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()
            
            bf, bs = lp_iterate(best_sequence, deadline - 4, step_size_init=0.03,
                                 max_iters=20, verbose_prefix="[fin_LP]")
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()
            
            bf, bs = softmax_gradient_optimize(best_sequence, deadline - 2,
                                                beta_start=200.0, beta_end=800.0,
                                                lr_start=0.005, max_iters=3000,
                                                verbose_prefix="[fin_SM]")
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()
            
            print(f"  Final: {best_score:.6f}")
        except Exception as e:
            print(f"  Final error: {e}")
    
    # Fallback
    if best_sequence is None:
        best_sequence = u_shape_init(512, curvature=1.0)
        best_score = evaluate_sequence(best_sequence.tolist())
    
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = evaluate_sequence(best_sequence.tolist())
    
    elapsed = time.monotonic() - t0
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    print(f"{'='*60}")
    
    return [float(x) for x in best_sequence]
