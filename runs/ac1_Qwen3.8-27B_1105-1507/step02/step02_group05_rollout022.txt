def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            # a[k-j] for j in [jl, jh]
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):

    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_min_M(a):
    """LP: min M s.t. (a*g)_k <= M, sum(g)=1, g>=0, M>=0.
    Returns (g, M_opt) or (None, None)."""
    n = len(a)
    # Variables: [g_0, ..., g_{n-1}, M]
    # Objective: minimize M => c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: (a*g)_k - M <= 0 for k=0,...,2n-2
    # A_ub has shape (2n-1, n+1)
    A_g = build_A_sparse(a)  # shape (2n-1, n)
    # Add -M column
    A_ub = lil_matrix((2 * n - 1, n + 1))
    A_ub[:, :n] = A_g
    A_ub[:, n] = -1.0
    A_ub = A_ub.tocsr()
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x[:n], 0.0)
        M_opt = float(r.x[n])
        if g.sum() > 1e-10:
            g = g / g.sum()
            # Recompute M_opt with normalized g
            ag = np.convolve(a, g)
            M_opt = float(ag.max())
            return g, M_opt
    return None, None


def lp_best_response_fixed_M(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def coordinate_decrease(a, n_sweeps, deadline):
    """Coordinate descent: decrease each a_j to its maximum allowable value."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)
    conv = fast_conv(a)
    M_current = conv.max()
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        total_decreased = 0.0
        for j in range(n):
            if time.monotonic() > decrease deadline:
                break
            # Find max a_j such that max(conv) <= M_current
            # a_j affects conv[k] for k in [j, j+n-1]
            k_lo = j
            k_hi = min(j + n - 1, 2 * n - 2)
            
            # M_other = max conv outside [k_lo, k_hi]
            M_other = 0.0
            if k_lo > 0:
                M_other = max(M_other, conv[:k_lo].max())
            if k_hi < 2 * n - 2:
                M_other = max(M_other, conv[k_hi+1:].max())
            
            target = min(M_current, M_other)
            
            # Find max a_j such that for all k in [k_lo, k_hi]:
            # c_k + contribution(a_j) <= target
            # where c_k = conv[k] - current_contribution(a_j)
            
            current_val = a[j]
            max_allowed = current_val
            
            for k in range(k_lo, k_hi + 1):
                if k == 2 * j:
                    c_k = conv[k] - current_val ** 2
                    if c_k >= target:
                        max_allowed = 0.0
                        break
                    bound = np.sqrt(target - c_k)
                else:
                    c_k = conv[k] - 2.0 * current_val * a[k - j]
                    if c_k >= target:
                        max_allowed = 0.0
                        break
                    slope = 2.0 * a[k - j]
                    if slope < 1e-15:
                        continue
                    bound = (target - c_k) / slope
                max_allowed = min(max_allowed, bound)
            
            max_allowed = max(0.0, max_allowed)
            if max_allowed < current_val:
                delta = current_val - max_allowed
                a[j] = max_allowed
                total_decreased += delta
                # Update conv
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * j:
                        conv[k] = (conv[k] - current_val**2) + max_allowed**2
                    else:
                        conv[k] = (conv[k] - 2*current_val*a[k-j]) + 2*max_allowed*a[k-j]
        
        if total_decreased > 1e-12:
            # Redistribute freed mass proportionally
            a += total_decreased / n  # uniform redistribution
            # Recompute conv
            conv = fast_conv(a)
            M_current = conv.max()
    
    return a

def golden_section_line_search(a, g, conv_a, conv_ag, conv_g, n_
evals=50):
    """Find t in [0,1] that minimizes max(conv((1-t)*a + t*g))."""
    def f(t):
        # conv = (1-t)^2 * conv_a + 2t(1-t) * conv_ag + t^2 * conv_g
        c = (1-t)**2 * conv_a + 2*t*(1-t) * conv_ag + t*t * conv_g
        return c.max()
    
    # Golden section in [0, 1]
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    a_lo, a_hi = 0.0, 1.0
    c = a_lo + (1 - gr) * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc = f(c)
    fd = f(d)
    
    for _ in range(n_
evals):
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_lo + (1 - gr) * (a_hi - a_lo)
            fc = f(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = f(d)
    
    t_best = (a_lo + a_hi) / 2
    return t_best

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using LP mirror descent
with exact golden-section line search, coordinate descent polish,
smooth gradient refinement, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def cross_conv(a, b):
    """Linear cross-convolution a*b."""
    n = len(a)
    m = len(b)
    length = n + m - 1
    s = 1
    while s < length:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fb = np.fft.rfft(b, s)
    return np.fft.irfft(fa * fb, s)[:length]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                A[k, j] = a[k - j]
    return A.tocsr()


def lp_min_M(a):
    """Solve: min M s.t. (a*g)_k <= M, sum(g)=1, g>=0, M>=0.
    Returns (g, M_opt)."""
    n = len(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_g = build_A_sparse(a)
    A_ub = lil_matrix((2 * n - 1, n + 1))
    A_ub[:, :n] = A_g.tolil()
    A_ub[:, n] = -1.0
    A_ub = A_ub.tocsr()
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(0.0, None)]
    
    r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                bounds=bounds, method='highs')
    if r.success:
        g = np.maximum(r.x[:n], 0.0)
        if g.sum() > 1e-10:
            g = g / g.sum()
            return g
    return None


def golden_line_search(conv_a, conv_ag, conv_g, n_
evals=40):
    """Find t in [0,1] minimizing max((1-t)^2*conv_a + 2t(1-t)*conv_ag + t^2*conv_g)."""
    gr = (np.sqrt(5) - 1) / 2
    
    def f(t):
        c = (1.0 - t) ** 2 * conv_a + 2.0 * t * (1.0 - t) * conv_ag + t * t * conv_g
        return c.max()
    
    lo = 0.0
    hi = 1.0
    c_ = lo + (1.0 - gr) * (hi - lo)
    d_ = lo + gr * (hi - lo)
    fc = f(c_)
    fd = f(d_)
    
    for _ in range(n_
evals):
        if fc < fd:
            hi = d_
            d_ = c_
            fd = fc
            c_ = lo + (1.0 - gr) * (hi - lo)
            fc = f(c_)
        else:
            lo = c_
            c_ = d_
            fc = fd
            d_ = lo + gr * (hi - lo)
            fd = f(d_)
    
    return (lo + hi) / 2.0


def lp_mirror_descent(a0, max_iters, deadline, tracker=None):
    """LP mirror descent with golden section line search."""
    a = np.maximum(a0.astype(np.float64), 1e-8)
    a = a / a.sum()
    n = len(a)
    
    conv_a = fast_conv(a)
    current_M = conv_a.max()

        conv_g = fast_conv(g)
        
        # Golden section line search
        t = golden_line_search(conv_a, conv_ag, conv_g, n_
evals=30)
        
        # Update
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:

        a_new = a_new / s

        M_new = conv_new.max()
        
        if M_new >= current_M * (1.0 - 1e-10):

            t = t * 0.5
            if t < 1e-6:

            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            conv_new = fast_conv(a_new)
            M_new = conv_new.max()
            if M_new >= current_M:
                break
        
        a = a_new
        conv_a = conv_new
        current_M = M_new

    if tracker:
        tracker.update(a)
    return a
