"""
Coordinate descent with exact 1D subproblem solution for minimizing
2*n*max(conv(a,a))/(sum(a))^2.
Solves each 1D subproblem by evaluating the objective at analytically derived
candidate points (where individual constraints become binding).
Combined with log-sum-exp smoothed gradient polishing and diverse initializations.
"""

import numpy as np
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def solve_1d_fast(a, j, n_cand_extra=5, rng=None):
    """Solve 1D subproblem for coordinate j: minimize obj with a[j]=x >= 0."""
    n = len(a)
    x_old = a[j]
    S = float(np.sum(a) - x_old)
    
    if S < 1e-15:
        return x_old
    
    # Compute c_base: convolution contribution excluding a[j]
    # c[k] = sum_i a[i]*a[k-i]
    # Remove a[j]'s contribution:
    #   k=2j: subtract x_old^2
    #   0<=k-j<n, k!=2j: subtract 2*x_old*a[k-j]
    
    conv = np.convolve(a, a)
    c_base = conv - 0.0  # copy
    js = j
    for k in range(2 * n - 1):
        ks = k - js
        if 0 <= ks < n:
            if k == 2 * js:
                c_base[k] -= x_old * x_old
            else:
                c_base[k] -= 2.0 * x_old * a[ks]
    
    # Precompute constant part (k where a[j] doesn't appear)
    # f_k(x) = x^2 if k==2j, 2*x*a[k-j] if 0<=k-j<n and k!=2j, 0 otherwise
    # Constant base: max over k where f_k = 0
    const_mask = np.ones(2 * n - 1, dtype=bool)
    const_mask[2 * js] = False  # quadratic
    for k in range(2 * n - 1):
        ks = k - js
        if 0 <= ks < n and k != 2 * js:
            const_mask[k] = False
    C_const = float(np.max(c_base[const_mask])) if np.any(const_mask) else 0.0
    
    # Generate candidates
    candidates = [0.0, x_old]
    
    # Quadratic binding: x = c_base[2j] / S
    xq = c_base[2 * js] / S
    if xq > 1e-12:
        candidates.append(xq)
    
    # Linear binding: x = S - c_base[k] / a[k-j]
    for k in range(2 * n - 1):
        ks = k - js
        if 0 <= ks < n and k != 2 * js and a[ks] > 1e-12:
            xl = S - c_base[k] / a[ks]
            if xl > 1e-12:
                candidates.append(xl)
    
    # Add a few extra points
    if candidates:
        max_cand = max(candidates)
        if max_cand > 0:
            candidates.append(max_cand * 0.5)
            candidates.append(max_cand * 1.5)
            candidates.append(max_cand * 0.1)
        if rng is not None and n_cand_extra > 0:
            for _ in range(n_cand_extra):
                candidates.append(np.exp(rng.uniform(np.log(max(1e-10, max_cand * 0.01)), np.log(max_cand * 10))))
    
    # Evaluate objective at each candidate (vectorized where possible)
    best_x = x_old
    best_obj = np.inf
    
    # For efficiency, precompute the linear part arrays
    # valid_k: indices k where 0<=k-j<n and k!=2j
    valid_ks = []
    valid_akj = []
    for k in range(2 * n - 1):
        ks = k - js
        if 0 <= ks < n and k != 2 * js:
            valid_ks.append(k)
            valid_akj.append(a[ks])
    valid_ks = np.array(valid_ks, dtype=np.int64) if valid_ks else np.array([], dtype=np.int64)
    valid_akj = np.array(valid_akj, dtype=np.float64) if valid_akj else np.array([], dtype=np.float64)
    
    base_valid = c_base[valid_ks] if len(valid_ks) > 0 else np.array([])
    
    for x in candidates:
        if x < 0:
            x = 0.0
        s = S + x
        if s < 1e-15:
            continue
        
        # N(x) = max(C_const, c_base[2j] + x^2, max_valid(c_base[k] + 2*x*a[k-j]))
        val_quad = c_base[2 * js] + x * x
        if len(valid_ks) > 0:
            val_lin = float(np.max(base_valid + 2.0 * x * valid_akj))
        else:
            val_lin = -np.inf
        N = max(C_const, val_quad, val_lin)
        obj = 2.0 * n * N / (s * s)
        
        if obj < best_obj:
            best_obj = obj
            best_x = x
    
    a[j] = best_x
    return best_x


def coord_descent(a, max_sweeps=200, tol=1e-12, verbose=False, rng=None):
    """Coordinate descent with exact 1D subproblem solution."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for sweep in range(max_sweeps):
        max_change = 0.0
        for j in range(n):
            old = a[j]
            solve_1d_fast(a, j, rng=rng)
            max_change = max(max_change, abs(a[j] - old))
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if verbose and sweep % 20 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.8f}, max_change={max_change:.2e}")
        
        if max_change < tol:
            break
    
    return best_a, best_obj


def smooth_gradient_step(a, n_steps=500, beta_start=10.0, beta_end=100.0,
                         lr_start=0.1, lr_end=0.01, verbose=False):
    """Log-sum-exp smoothed gradient descent with projection."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s  # normalize to sum=1
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for step in range(n_steps):
        frac = step / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        conv = np.convolve(a, a)
        
        # Numerical stability for log-sum-exp
        conv_shifted = conv - np.max(conv)
        weights = np.exp(beta * conv_shifted)
        weights /= np.sum(weights)
        
        # Gradient of log(sum(exp(beta*conv))) w.r.t. a[j]
        # = 2 * beta * sum_k weights[k] * d(conv[k])/d(a[j])
        # d(conv[k])/d(a[j]) = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n) ... 
        # Actually: conv[k] = sum_i a[i]*a[k-i]
        # d/d(a[j]) conv[k] = a[k-j] (if 0<=k-j<n) + a[j-(k-j)]... 
        # More carefully: d/d(a[j]) sum_i a[i]a[k-i] = a[k-j] (term i=j) + a[j] (term k-i=j, i.e., i=k-j)
        # = a[k-j] (if 0<=k-j<n) + a[k-j] (if 0<=k-j<n) = 2*a[k-j] if k!=2j
        # = a[j] (if k=2j, only one term)
        
        # So grad[j] = 2*beta * sum_k weights[k] * (2*a[k-j] if valid and k!=2j, a[j] if k=2j, 0 otherwise)
        # = 2*beta * (sum over valid k!=2j of weights[k]*2*a[k-j] + weights[2j]*a[j] if 0<=2j<2n-1)
        
        # This is: 4*beta * (weights * a)_shifted + 2*beta*weights[2j]*a[j]
        # where (weights*a)_shifted[j] = sum_k weights[k]*a[k-j] for valid k
        
        # Compute efficiently: this is a correlation
        # grad[j] = 2*beta * sum_k weights[k] * kernel_j(k)
        # where kernel_j(k) = 2*a[k-j] if 0<=k-j<n and k!=2j, a[j] if k=2j, 0 otherwise
        
        # Simplify: let b[k] = weights[k]
        # sum_k b[k]*a[k-j] for valid k (0<=k-j<n) is (b * a)[j] (cross-correlation)
        # We need: 2*beta * (2*(b*a)[j] - b[2j]*a[j] (to fix the double-count at k=2j) - 0 (for invalid))
        # Wait, let me be more careful.
        
        # (b * a_corr)[j] = sum_k b[k] * a[k-j] where 0<=k-j<n, 0<=k<2n-1
        # = sum_{ks=0}^{n-1} b[j+ks] * a[ks]
        
        # The gradient contribution from the "2*a[k-j]" terms (k!=2j):
        # sum over k where 0<=k-j<n and k!=2j of b[k]*2*a[k-j]
        # = 2 * sum_{ks=0}^{n-1, j+ks != 2j} b[j+ks]*a[ks]
        # = 2 * (sum_{ks=0}^{n-1} b[j+ks]*a[ks] - b[2j]*a[j] if 0<=2j<2n-1)
        # = 2 * corr[j] - 2*b[2j]*a[j] (if 2j is valid)
        
        # Plus the "a[j]" term at k=2j: b[2j]*a[j] (if valid)
        
        # Total inner sum = 2*corr[j] - 2*b[2j]*a[j] + b[2j]*a[j] = 2*corr[j] - b[2j]*a[j]
        
        # So grad[j] = 2*beta*(2*corr[j] - b[2j]*a[j])
        
        # Compute corr[j] = sum_{ks=0}^{n-1} b[j+ks]*a[ks]
        # This is a valid range of b: b[j] to b[j+n-1], but b has length 2n-1.
        # For j=0: b[0..n-1] * a[0..n-1]
        # For j=n-1: b[n-1..2n-2] * a[0..n-1]
        # This is just a sliding dot product.
        
        # Use FFT for efficiency, or direct computation for small n.
        if n <= 500:
            # Direct computation
            corr = np.zeros(n)
            for j in range(n):
                # corr[j] = sum_{ks=0}^{n-1} weights[j+ks] * a[ks]
                corr[j] = np.dot(weights[j:j+n], a)
        else:
            # FFT-based
            # corr[j] = sum_{ks=0}^{n-1} weights[j+ks]*a[ks]
            # This is the cross-correlation of weights and a
            # corr = np.cross_correlation... let's use fft
            N = 2 * n
            W = np.zeros(N)
            W[:2*n-1] = weights
            A = np.zeros(N)
            A[:n] = a
            corr = np.real(np.ifft(np.fft.fft(W) * np.conj(np.fft.fft(A))))
            # corr[i] = sum_j W[j] * A[(i-j) mod N]
            # We want: sum_{ks} weights[j+ks]*a[ks] = sum_{ks} weights[j+ks]*a[ks]
            # Let m = j+ks, then ks = m-j: sum_m weights[m]*a[m-j] for m in [j, j+n-1]
            # This is the cross-correlation at shift j.
            # Using the FFT formula: corr_ft[m] = sum_j W[j]*A[j-m] = sum_j W[j]*A[(j-m) mod N]
            # We need sum_m W[m]*A[m-j] = sum_m W[m]*A[(m-j) mod N]
            # This is corr_ft[-j mod N] = corr_ft[N-j] (for j>0), or corr_ft[0] for j=0.
            # Actually, let me just use the direct formula:
            # corr[j] = <weights[j:j+n], a> 
            # Which is the cross-correlation of weights with a at lag j.
            # np.correlate(weights, a, mode='valid') gives sum_{i} weights[i]*a[i+j] for j=0..(2n-1-n)=n-1
            # That's not quite right. Let me just use the direct loop for safety.
            corr = np.zeros(n)
            for j in range(n):
                corr[j] = np.dot(weights[j:j+n], a)
        
        w2j = np.zeros(n)
        valid_2j = (2 * np.arange(n) < 2 * n - 1)
        w2j[valid_2j] = weights[2 * np.arange(n)[valid_2j]]
        
        grad = 2.0 * beta * (2.0 * corr - w2j * a)
        
        # Also need gradient of -2*log(s) where s = sum(a) = 1 (since we normalize)
        # If we fix sum=1, we project onto the simplex after each step.
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and step % 100 == 0:
            print(f"    SG step {step}: beta={beta:.1f}, obj={best_obj:.8f}")
    
    return best_a, best_obj


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 8))
    t = np.arange(n, dtype=np.float64) / n
    if choice == 0:
        # Cosine window
        freq = rng.uniform(1.0, 3.0)
        a = np.maximum(0, np.cos(np.pi * freq * t + rng.uniform(0, np.pi)))
    elif choice == 1:
        # Two-lobe (high at edges, low in middle or vice versa)
        if rng.random() < 0.5:
            a = np.exp(-4.0 * (t - 0.5) ** 2)
        else:
            a = 1.0 - np.exp(-4.0 * (t - 0.5) ** 2) + 0.1
    elif choice == 2:
        # Periodic
        period = int(rng.integers(max(2, n // 50), max(3, n // 5)))
        pattern = np.abs(np.sin(2 * np.pi * np.arange(period) / period * rng.uniform(1, 4))) + 0.1
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 3:
        # Quadratic bowl
        a = 0.3 + 0.7 * (1 - (2 * t - 1) ** 2)
    elif choice == 4:
        # Sine burst
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.4)
        a = np.exp(-((t - center) / width) ** 2)
    elif choice == 5:
        # Smooth random
        k = int(rng.integers(2, max(3, n // 20)))
        coeffs = rng.standard_normal(k)
        a = np.abs(np.polynomial.polymul(coeffs, np.polynomial.polymul(coeffs, coeffs[::-1]))[np.arange(n) % len(coeffs)]) + 0.1
        # Simpler: sin mixing
        a = 1.0 + 0.5 * np.sin(2 * np.pi * rng.uniform(1, 5) * t + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
    elif choice == 6:
        # Triangle wave
        freq = int(rng.integers(1, max(2, n // 10)))
        a = 1.0 - np.abs(2 * (t * freq % 1.0) - 1.0)
    else:
        # Random smooth
        a = 0.5 + 0.5 * rng.random(n)
        # Smooth with a box filter
        k = max(1, n // 100)
        if k > 1:
            kernel = np.ones(k) / k
            a = np.convolve(a, kernel, mode='same')
    
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def try_length(n, rng, time_budget, best_obj_ref, verbose=False):
    """Try to find a good sequence of length n within the given time budget."""
    results = []
    
    n_inits = 4 if n <= 300 else 2
    
    for init_i in range(n_inits):
        if time_budget <= 10:
            break
        
        if init_i == 0:
            # Start from all-ones (known to give obj=2)
            a_init = np.ones(n) / n
        elif init_i == 1:
            a_init = structured_init(n, rng)
        elif init_i == 2:
            a_init = structured_init(n, rng)
        else:
            # Perturbed ones
            a_init = np.ones(n) / n
            a_init += rng.normal(0, 0.1, n)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / np.sum(a_init)
        
        # Phase 1: Smooth gradient step
        if time_budget > 30:
            n_sgs = min(300, int(time_budget / 0.1))
            if n_sgs > 50:
                a1, obj1 = smooth_gradient_step(a_init, n_steps=n_sgs,
                                                beta_start=5.0, beta_end=50.0,
                                                lr_start=0.05, lr_end=0.005,
                                                verbose=False)
                time_budget -= 0.1 * n_sgs
            else:
                a1, obj1 = a_init, compute_obj(a_init)
        else:
            a1, obj1 = a_init, compute_obj(a_init)
        
        # Phase 2: Coordinate descent
        if time_budget > 20:
            max_sweeps = min(100, int(time_budget / 0.5))
            if max_sweeps > 10:
                a2, obj2 = coord_descent(a1, max_sweeps=max_sweeps, verbose=False, rng=rng)
                time_budget -= 0.5 * max_sweeps
            else:
                a2, obj2 = a1, obj1
        else:
            a2, obj2 = a1, obj1
        
        results.append((a2, obj2))
        
        # Also try direct CD from the initialization
        if time_budget > 20 and init_i < 2:
            max_sweeps2 = min(50, int(time_budget / 0.5))
            if max_sweeps2 > 10:
                a3, obj3 = coord_descent(a_init, max_sweeps=max_sweeps2, verbose=False, rng=rng)
                time_budget -= 0.5 * max_sweeps2
                results.append((a3, obj3))
    
    if not results:
        a = np.ones(n) / n
        return a, compute_obj(a)
    
    best = min(results, key=lambda x: x[1])
    return best[0], best[1]

            break
        
        alloc = min(time_per_n[n], remaining * 0.4)
        print(f"N={n}: allocating {alloc:.0f}s, current best={best_score:.8f}")
        
        a, obj = try_length(n, rng, alloc, best_score, verbose=False)
        
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  New best: n={n}, obj={obj:.8f}")
        
        # If this is significantly better, also try to refine
        if obj - best_score < 0.01 and obj < 2.0:
            if time.monotonic() < deadline - 60:
                a_r, obj_r = refine_best(a, min(60, deadline - time.monotonic() - 30), verbose=True)
                if obj_r < best_score:
                    best_score = obj_r
                    best_seq = a_r.copy()
                    print(f"  Refined: n={n}, obj={obj_r:.8f}")
    
    # Final refinement of the best solution
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: n={len(best_seq)}, obj={best_score:.8f}, {remaining:.0f}s left")
        a_final, obj_final = refine_best(best_seq, remaining * 0.8, verbose=True)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final: obj={best_score:.8f}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)
    
    # Ensure valid output
    best_seq = np.clip(best_seq, 0, 1000.0)
    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s")
    
    return [float(x) for x in best_seq]
