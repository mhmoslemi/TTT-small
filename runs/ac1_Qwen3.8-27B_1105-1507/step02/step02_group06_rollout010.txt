sum_l D_{j,l} * a_l = 0
D_{j,j} * a_j = -sum_{l≠j} D_{j,l} * a_l
a_j = -sum_{l≠j} D_{j,l} * a_l / D_{j,j}

But D_{j,l} = w_{j+l} >= 0, so a_j = -sum_{l≠j} w_{j+l} * a_l / w_{2j} <= 0 (since all terms are non-negative).

This means the unconstrained minimum of a^T D a is at a = 0 (the trivial solution)! Of course, since D is PSD, a^T D a >= 0 with equality at a = 0.

So without the sum constraint, the minimum is trivially 0. The sum constraint is what makes the problem interesting.

With the sum constraint sum(a) = S, the problem is non-trivial. Let me use the Lagrangian approach:

KKT: 2Da - lambda*1 = 0 (for positive a_j)

```python
"""
Iterative reweighting with FFT-based QP solve for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses power-weighted reweighting
where weights are the current convolution values raised to a power p,
and the QP is solved via conjugate gradient with FFT-based matvec.
"""
import numpy as np
import time


def compute_score(f):
    """Fast score computation without type checks."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def next_power_of_2(n):
    return 1 << (n - 1).bit_length()


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    L = next_power_of_2(n)
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    return np.fft.irfft(A * B, L)[:n]


def make_init(n, init_type, parent):
    """Create initial sequence of length n."""
    idx = np.arange(n, dtype=np.float64)
    
    if init_type == 'parent' and parent is not None:
        # Interpolate parent to length n
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        a = np.interp(x_new, x_old, parent)
        return np.clip(a, 0, None)
    elif init_type == 'uniform':
        return np.ones(n)
    elif init_type == 'cosine':
        return (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0
    elif init_type == 'bell':
        center = (n - 1) / 2.0
        return np.exp(-0.5 * ((idx - center) / (n / 4.0)) ** 2)
    elif init_type == 'bimodal':
        return (np.exp(-0.5 * ((idx - n * 0.25) / (n / 8.0)) ** 2) +
                np.exp(-0.5 * ((idx - n * 0.75) / (n / 8.0)) ** 2))
    elif init_type == 'random':
        return np.random.default_rng(42).uniform(0.1, 1.0, n)
    else:
        return np.ones(n)


def solve_linear_system_D_w(w, rhs, n_cg=300, tol=1e-10):
    """Solve D_w @ x = rhs where D_w(j,l) = w[j+l].
    Uses conjugate gradient with FFT-based matvec.
    w has length 2n-1, rhs has length n.
    Returns x of length n.
    """
    n = len(rhs)
    # Need L >= (2n-1) + n - 1 = 3n-2 for the cross-correlation
    L = next_power_of_2(3 * n - 2)
    
    # Precompute FFT of w
    w_pad = np.zeros(L)
    w_pad[:2*n-1] = w
    W = np.fft.rfft(w_pad)
    
    def matvec(v):
        """Compute D_w @ v = cross_correlation(w, v)."""
        v_pad = np.zeros(L)
        v_pad[:n] = v
        V = np.fft.rfft(v_pad)
        # Cross-correlation: ifft(fft(w) * conj(fft(v)))
        result = np.fft.irfft(W * np.conj(V), n=L)
        return result[:n]
    
    # CG iteration
    x = np.ones(n) / n  # initial guess
    r = rhs - matvec(x)
    p = r.copy()
    rs_old = np.dot(r, r)
    
    for i in range(n_cg):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if abs(pAp) < 1e-30:
            break
        alpha = rs_old / pAp
        x += alpha * p
        r -= alpha * Ap
        rs_new = np.dot(r, r)
        if np.sqrt(rs_new) < tol * np.linalg.norm(rhs):
            break
        p = r + (rs_new / rs_old) * p
        rs_old = rs_new
    
    return x


def reweighting_optimize(a, deadline, p=2.0, max_iter=50000, verbose=True):
    """Iterative reweighting optimization.
    
    At each step:
    1. Compute conv = a * a
    2. Set weights w = conv^p
    3. Solve QP: min b^T D_w b s.t. sum(b) = sum(a), b >= 0
       (where D_w(j,l) = w[j+l])
    4. Update: a_new = blend(a, b)
    """
    a = a.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return a, float('inf')
    
    best_a = a.copy()
    best_score = compute_score(a)
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iter:
        # Compute convolution
        conv = fft_convolve(a, a)
        M = float(np.max(conv))
        current_score = 2 * n * M / (S * S)
        
        # Compute weights
        # Avoid zero weights (add small epsilon)
        w = np.maximum(conv, 1e-10) ** p
        
        # Solve QP: D_w @ b = 1, then scale to sum = S
        rhs = np.ones(n)
        b = solve_linear_system_D_w(w, rhs, n_cg=200, tol=1e-8)
        
        # Project to non-negative
        b = np.maximum(b, 0)
        b_sum = float(np.sum(b))
        if b_sum < 1e-10:
            # Fallback: use previous a
            no_improve += 1
            if no_improve > 100:
                break
            it += 1
            continue
        b = b * (S / b_sum)
        
        # Compute score of b
        b_score = compute_score(b)
        
        # Blend
        t = 0.5  # blend factor
        a_new = (1 - t) * a + t * b
        a_new = np.maximum(a_new, 0)
        a_new_sum = float(np.sum(a_new))
        if a_new_sum < 1e-10:
            no_improve += 1
            it += 1
            continue
        a_new = a_new * (S / a_new_sum)
        
        a_new_score = compute_score(a_new)
        
        it += 1
        if a_new_score < best_score:
            best_score = a_new_score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    iter {it}: score={best_score:.8f}")
        elif b_score < best_score:
            best_score = b_score
            best_a = b.copy()
            a = b  # Take the QP solution directly
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    iter {it} (direct): score={best_score:.8f}")
        else:
            no_improve += 1
            if no_improve > 200:
                # Try a bigger step
                a = b
                no_improve = 0
                if no_improve > 500:
                    break
    
    return best_a, best_score


def spectral_project(a, deadline, alpha=2.0, max_iter=5000):
    """Spectral projection: flatten the autocorrelation by peaking the power spectrum."""
    a = a.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    L = next_power_of_2(2 * n - 1)
    
    best_a = a.copy()
    best_score = compute_score(a)
    it = 0
    
    while time.monotonic() < deadline and it < max_iter:
        # FFT
        a_pad = np.zeros(L)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        
        # Power spectrum
        P = np.abs(A) ** 2
        
        # Peak the spectrum (increase alpha to make more peaked)
        P_peak = P ** alpha
        
        # Reconstruct with modified magnitude, keep phase
        phase = np.angle(A)
        A_new = np.sqrt(P_peak) * np.exp(1j * phase)
        
        # Inverse FFT
        a_new_full = np.fft.irfft(A_new, n=L).real
        a_new = a_new_full[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        a_new_sum = float(np.sum(a_new))
        if a_new_sum < 1e-10:
            break
        a_new = a_new * (S / a_new_sum)
        
        a_new_score = compute_score(a_new)
        
        if a_new_score < best_score:
            best_score = a_new_score
            best_a = a_new.copy()
            a = a_new
            if it % 500 == 0:
                print(f"    spectral iter {it}: score={best_score:.8f}")
        
        it += 1
    
    return best_a
