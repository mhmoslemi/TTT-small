    return 2.0 * n * conv.max()

def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s

def build_conv_matrix(f):
    """Build A such that A @ g = convolve(f, g, 'full')."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k, j] = f[i]
    return A

    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.

    c = -np.ones(n)

    except Exception:

    return None

"""
LP-centric optimization with adaptive M-shrinking, line search, and multi-strategy
initialization. Focuses on the LP direction-finding step (which is the most powerful
local optimizer) with improved step-size selection and M-schedule.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute 2*n*max(conv) for normalized f (sum=1)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def build_conv_matrix(f):
    """Build A such that A @ g = np.convolve(f, g, 'full')."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_min = max(0, k - n + 1)
        j_max = min(n, k + 1)
        for j in range(j_min, j_max):
            A[k, j] = f[k - j]
    return A


def lp_direction(f, M, max_time=3.0):
    """Solve LP: max sum(g) s.t. A_f @ g <= M, g >= 0. Returns normalized g or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max_time})
        if res.success and res.x is not None:
            g = np.clip(res.x, 0, None)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def lp_line_search(f, g, best_score, max_evals=12):
    """Line search along direction g. Returns (best_f, best_score, best_t)."""
    f = f.copy()
    g = g.copy()
    n = len(f)
    best_t = 0.0
    best_f = f.copy()
    best_s = best_score

    # Try a range of step sizes, focusing on larger ones first
    steps = [0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.7, 0.9, 1.0]

    for t in steps:
        f_new = (1.0 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        score = compute_score(f_new)
        if score < best_s - 1e-13:
            best_s = score
            best_f = f_new.copy()
            best_t = t

    return best_f, best_s, best_t

def spectral_flat(f, n_iter=200, rng=None):
    """Spectral flattening: project onto flat-magnitude spectrum."""
    if rng is None:

    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)

        current_energy = np.sum(f ** 2)
        n_bins = len(mag)
        target_mag = np.sqrt(current_energy * n_fft / n_bins)

        mag_new = np.full_like(mag, target_mag)
        mag_new[0] = 1.0

        if it % 80 == 79:
            phase = phase + rng.normal(0, 0.08, len(phase))

        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 1e-15, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s

        score = compute_score(f_new)
        if score < best_score - 1e-10:
            best_score = score
            best_f = f_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break
        f = f_new

    return best_f, best_score


def pressure_update(f, n_iter=150, eta=0.15, rng=None):
    """Pressure-based multiplicative update."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        j_star = np.argmax(conv)

        pressure = np.zeros(n)
        for i in range(n):
            idx = j_star - i
            if 0 <= idx < n:
                pressure[i] = f[idx]

        # Add secondary peaks
        conv_sorted_idx = np.argsort(conv)[::-1]
        for k in range(1, min(3, len(conv_sorted_idx))):
            j2 = conv_sorted_idx[k]
            if conv[j2] > 0.85 * M:
                for i in range(n):
                    idx = j2 - i
                    if 0 <= idx < n:
                        pressure[i] += 0.3 * f[idx]

        if pressure.max() > 0:
            pressure = pressure / pressure.max()

        eta_eff = eta * max(0.1, 1.0 - 0.5 * it / n_iter)
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 80:
                f_pert = f * np.exp(rng.normal(0, 0.1, n))
                f_pert = np.clip(f_pert, 1e-15, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    stall = 0
                else:
                    break

    return best_f, best_score


def coordinate_descent(f, n_sweeps=5, rng=None):
    """Coordinate descent polishing."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-12:
                continue
            best_val = original
            best_local = best_score

            for factor in [0.4, 0.6, 0.8, 0.9, 1.1, 1.3, 1.6, 2.0, 0.2]:
                f[i] = max(0.0, original * factor)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                conv = np.convolve(f, f)
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local - 1e-13:
                    best_local = score
                    best_val = f[i]

            f[i] = best_val
            if best_local < best_score:
                best_score = best_local

        f = np.clip(f, 0, None)
        f = normalize(f)

    return f, best_score

                n_sa = len(f_sa)
                best_sa = best_overall.copy()
                best_sa_score = best_score_overall
                cur_score = best_sa_score

                for it_sa in range(n_iter_sa):
                    if time.monotonic() > deadline:
                        break
                    T = T0 * (T_end / T0) ** (it_sa / n_iter_sa)
                    f_try = f_sa.copy()
                    r = rng.random()
                    if r < 0.4:
                        # Local perturbation
                        width = max(1, n_sa // 32)
                        center = int(rng.integers(0, n_sa))
                        idx = np.arange(n_sa)
                        kernel = np.exp(-0.5 * (idx - center) ** 2 / (width ** 2))
                        f_try += T * 2 * rng.standard_normal(n_sa) * kernel
                    elif r < 0.7:
                        # Single element
                        i = int(rng.integers(0, n_sa))
                        f_try[i] = max(0, f_try[i] * np.exp(rng.normal(0, 0.3)))
                    else:
                        # Block scaling
                        w = max(1, int(rng.integers(1, max(2, n_sa // 16))))
                        st = int(rng.integers(0, n_sa - w))
                        f_try[st:st + w] *= np.exp(rng.normal(0, 0.3, w))

                    f_try = np.clip(f_try, 0, None)
                    s_try = f_try.sum()
                    if s_try < 1e-10:
                        continue
                    f_try = f_try / s_try
                    new_score = compute_score(f_try)

                    if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
                        f_sa = f_try
                        cur_score = new_score
                        if new_score < best_sa_score:
                            best_sa_score = new_score
                            best_sa = f_try.copy()

                if best_sa_score < best_score_overall:
                    best_score_overall = best_sa_score
                    best_overall = best_sa.copy()
                    print(f"  After SA: {best_sa_score:.6f}")

    # Final coordinate descent
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 5:
            f_final, s_final = coordinate_descent(best_overall, n_sweeps=3, rng=rng)
            if s_final < best_score_overall:
                best_score_overall = s_final
                best_overall = f_final.copy()
                print(f"  Final CD: {s_final:.6f}")

    # Verify and return
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal: evaluate_score={final_score:.6f}, internal_score={best_score_overall:.6f}, n={len(final_list)}")
        return final_list

    # Fallback
    print("Using fallback")
    f = np.ones(256)
    return [float(x) for x in f]
