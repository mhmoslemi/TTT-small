def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

        A_ub[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
        A_ub[k, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize t
    c = np.zeros(n_vars)
    c[n] = 1.0
    
    bounds = [(0.0, None)] * n + [(-np.inf, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n].copy(), float(res.x[n])
    return None, None


def lp_alternate_balanced(a, max_iter=1000, deadline=None, verbose=False):
    """LP alternating with balanced formulation."""
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones_like(a)
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        g, t = solve_lp_balanced(a)
        if g is None:
            break
        
        gs = np.sum(g)
        if gs < 1e-10:
            break
        g = g / gs
        
        # Adaptive blend: start aggressive, decrease over time
        if it < 100:
            blend = 0.5
        elif it < 500:
            blend = 0.3
        else:
            blend = 0.15
        
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a = a_new / s
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if verbose and it % 200 == 0:
            print(f"    LP-bal it={it}: t={t:.6f}, obj={best_obj:.6f}")
    
    return best_a, best_obj


def subgrad_momentum(a, n_steps=10000, deadline=None, verbose=False):
    """Subgradient method with heavy ball momentum."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    v = np.zeros(n)
    beta = 0.9  # momentum
    f_lower = best_obj / (2.0 * n)  # lower bound on max_b (with sum=1)
    
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        h = np.convolve(a, a)
        max_h = float(np.max(h))
        
        # Active set with tolerance
        tol = max_h * 0.005
        active = h >= max_h - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            idx = int(np.argmax(h))
            active = np.zeros(len(h), dtype=bool)
            active[idx] = True
            n_active = 1
        
        # Compute subgradient (average of active gradients)
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            # grad_k[j] = a[k-j] for j in [j_start, j_end]
            grad[j_start:j_end + 1] += a[k - j_end:k - j_start + 1][::-1]
            # Diagonal correction: if k = 2j, add extra a[j]
            if k % 2 == 0:
                j_diag = k // 2
                if 0 <= j_diag < n:
                    grad[j_diag] += a[j_diag]
        
        grad /= n_active
        
        # Polyak step size
        grad_norm_sq = float(np.dot(grad, grad))
        if grad_norm_sq > 1e-20:
            gamma = (max_h - f_lower) / grad_norm_sq
            gamma = min(gamma, 0.5)
        else:
            gamma = 0.01
        
        v = beta * v + gamma * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = a
            v = np.zeros(n)
        else:
            a = a_new / s
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            # Update lower bound
            f_lower = max(max_h * 0.99, f_lower)
        
        if verbose and step % 500 == 0:
            print(f"    SG it={step}: max_h={max_h:.6f}, obj={best_obj:.6f}")
    
    return best_a, best_obj


def coord_search(a, n_rounds=5, deadline=None):
    """Coordinate descent with binary search for optimal value."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        for j in range(n):
            old_val = a[j]
            # Try a range of values using logarithmic scaling
            if old_val > 1e-10:
                candidates = [old_val * m for m in [0.1, 0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]]
            else:
                candidates = [0.0, 0.001, 0.01, 0.1]
            candidates.append(0.0)
            
            best_x = old_val
            best_local = best_obj
            
            for x in candidates:
                x = max(0.0, x)
                a[j] = x
                s = np.sum(a)
                if s < 1e-10:
                    a[j] = old_val
                    continue
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_x = x
            
            a[j] = best_x
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    choice = int(rng.integers(0, 10))
    x = np.arange(n, dtype=np.float64)
    xn = x / max(1, n - 1)
    
    if choice == 0:
        # Cosine family
        freq = rng.uniform(1.0, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        power = rng.uniform(0.5, 4.0)
        a = np.abs(np.cos(np.pi * freq * xn + phase)) ** power
    elif choice == 1:
        # Gaussian bump
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.1, 0.5)
        a = np.exp(-0.5 * ((xn - center) / width) ** 2)
    elif choice == 2:
        # Periodic
        period = int(rng.integers(2, max(3, n // 5)))
        pattern = rng.random(period)
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 3:
        # Anti-bump (high at edges)
        a = (0.5 + 0.5 * np.sin(np.pi * xn / 2)) ** rng.uniform(1, 5)
    elif choice == 4:
        # Two bumps
        c1, c2 = sorted(rng.uniform(0, 1, 2))
        w = rng.uniform(0.05, 0.3)
        a = np.exp(-((xn - c1) / w) ** 2) + np.exp(-((xn - c2) / w) ** 2)
    elif choice == 5:
        # Smooth random
        k = max(2, n // 10)
        raw = rng.random(k)
        a = np.convolve(raw, np.hanning(k), mode='same')[:n]
        a = np.maximum(a, 0)
    elif choice == 6:
        # Power law
        p1 = rng.uniform(0.3, 3.0)
        p2 = rng.uniform(0.3, 3.0)
