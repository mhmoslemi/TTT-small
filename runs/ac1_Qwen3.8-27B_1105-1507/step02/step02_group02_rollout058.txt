```python
"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FFT-accelerated smooth gradient descent with Adam on the simplex
2. QP-based best-response iterations (minimize ||conv(a,g)||_2^2)
3. Multi-length and perturbation search
4. Beta annealing for the softmax approximation of max
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def cross_correl(a, w):
    """Compute result[j] = sum_{m=0}^{n-1} a[m] * w[m+j] for j=0,...,n-1.
    a has length n, w has length 2n-1."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fw = np.fft.rfft(w, s)
    cc = np.fft.irfft(fw * np.conj(fa), s)
    return cc[:n]

def softmax(x, beta):
    """Numerically stable softmax with temperature 1/beta."""
    x_shifted = beta * (x - x.max())
    x_shifted = np.clip(x_shifted, -700, 700)
    e = np.exp(x_shifted)
    return e / e.sum()

def proj_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u * np.arange(1, n + 1) > cssv)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_conv_matrix(a):
    """Build Toeplitz-like matrix A where (A @ x) = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A

def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2 directly."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))

def smooth_gradient(a, beta):
    """Compute gradient of (2n/beta)*log(sum exp(beta*conv(a,a))) w.r.t. a.
    Returns gradient vector of same shape as a."""
    n = len(a)
    c = fast_conv(a)
    w = softmax(c, beta)
    # grad[j] = 2 * cross_correl(a, w)[j] - a[j] * w[2j] (if 2j < 2n-1)
    cc = cross_correl(a, w)
    grad = 2.0 * cc
    # Correction term
    idx = 2 * np.arange(n)
    valid = idx < 2 * n - 1
    grad[valid] -= a[valid] * w[idx[valid]]
    return 2.0 * n * grad  # Include the 2n factor

def adam_simplex(a, beta_schedule, lr_schedule, n_steps, deadline, n):
    """Adam optimization on the simplex for the smooth objective."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
    else:
        a = a / s
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    best = a.copy()
    best_score = compute_score(a)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        # Beta and learning rate schedules
        progress = step / max(1, n_steps - 1)
        beta = beta_schedule(progress)
        lr = lr_schedule(progress)
        
        # Compute gradient
        grad = smooth_gradient(a, beta)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex
        a_new = proj_simplex(a_new)
        
        a = a_new
        
        # Evaluate every 10 steps (to save time)
        if step % 10 == 0 or step == n_steps - 1:
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best = a.copy()
    
    return best, best_score

def qp_best_response(a):
    """Solve QP: min ||conv(a,g)||_2^2 s.t. sum(g)=1, g>=0.
    Uses the fact that H = A^T A is Toeplitz."""
    n = len(a)
    # Build H (Toeplitz) from autocorrelation of a
    # H[i,j] = sum_k A[k,i]*A[k,j] where A is conv matrix of a
    # H is symmetric Toeplitz with first row = [r0, r1, ..., r_{n-1}]
    # where r[d] = sum_{p=d}^{n-1} a[p]*a[p-d] for d >= 0
    c = fast_conv(a)  # c[k] = sum_i a[i]*a[k-i], length 2n-1
    # r[d] for d=0,...,n-1: 
    # r[d] = sum_{p=d}^{n-1} a[p] * a[p-d]
    # This equals c[n-1+d]... let me verify
    # c[n-1+d] = sum_i a[i]*a[n-1+d-i], let j=n-1+d-i, i=n-1+d-j:
    # = sum_j a[n-1+d-j]*a[j], for j=0,...,d (since n-1+d-j < n means j > d-1, and i>=0 means j<=n-1+d)
    # Actually for d>=0: c[n-1+d] = sum_{j=0}^{d} a[n-1+d-j]*a[j]
    # And r[d] = sum_{p=d}^{n-1} a[p]*a[p-d], let q=p-d: sum_{q=0}^{n-1-d} a[q+d]*a[q]
    # These are different! c[n-1+d] has d+1 terms, r[d] has n-d terms.
    # 
    # Let me just compute r directly.
    r = np.zeros(n)
    for d in range(n):
        r[d] = np.sum(a[d:] * a[:n-d])
    
    # Build H as Toeplitz matrix
    from scipy.linalg import toeplitz
    H = toeplitz(r)
    
    # Solve QP: min (1/2) g^T H g s.t. 1^T g = 1, g >= 0
    # Use scipy optimize
    from scipy.optimize import minimize
    
    x0 = np.ones(n) / n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 1,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0, None)] * n
    
    res = minimize(
        lambda x: 0.5 * x @ H @ x,
        x0, jac=lambda x: H @ x,
        method='SLSQP', bounds=bounds, constraints=constraints,
        options={'maxiter': 500, 'ftol': 1e-12}
    )
    
    if res.success:
        g = np.maximum(res.x, 0)
        g = g / g.sum()
        return g
    return None
```

```python
        # Let's just use scipy

        H = toeplitz(r)
```
