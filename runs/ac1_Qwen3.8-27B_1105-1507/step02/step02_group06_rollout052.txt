```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w
```

```python
    p = p.copy().astype(np.float64)

    best_score = float(2 * n * np.max(np.convolve(p, p)))
    
    momentum = np.zeros(n)

    momentum_decay = 0.9
    top_k = 15
    it = 0

        conv = np.convolve(p, p)

        top_indices = np.argsort(conv)[::-1][:top_k]
        top_vals = conv[top_indices]
```

```python
"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using projected subgradient descent
with Nesterov momentum on the probability simplex, combined with periodic
LP-based direction finding. Fixed length n=1024 for maximum depth.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import sys

def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    if n == 0:
        return float('inf')
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))

def normalized_score(p):
    """Score for normalized sequence (sum to 1)."""
    n = len(p)
    conv = np.convolve(p, p)
    return float(2 * n * np.max(conv))

def project_simplex(v):
    """Project onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v.copy()
    # Algorithm from Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_max_conv(p, top_k=15, temp_frac=0.05):
    """Compute weighted subgradient of max_k (p*p)[k]."""
    n = len(p)
    conv = np.convolve(p, p)
    M = float(np.max(conv))
    if M <= 0:
        return np.zeros(n), 0.0
    
    top_idx = np.argsort(conv)[::-1][:top_k]
    top_vals = conv[top_idx]
    
    # Softmax weight
    spread = max(top_vals[0] - top_vals[-1], M * 1e-6)
    temp = max(spread * temp_frac, 1e-12)
    w = np.exp((top_vals - top_vals[0]) / temp)
    w = w / w.sum()
    
    grad = np.zeros(n)
    for k, wi in zip(top_idx, w):
        if wi < 1e-12:
            continue
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        seg = p[k - j_hi : k - j_lo + 1][::-1]
        grad[j_lo:j_hi+1] += 2.0 * wi * seg
    
    return grad, M

def build_conv_matrix_sparse(p):
    """Build sparse matrix A where (A@g)[k] = convolve(p,g)[k]."""
    n = len(p)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(p[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp_direction(p, M, time_limit=5.0):
    """Solve: max sum(g) s.t. conv(p,g)[k] <= M for all k, g >= 0."""
    n = len(p)
    try:
        A = build_conv_matrix_sparse(p)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        t_start = time.monotonic()
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            g = result.x
            if np.sum(g) > 1e-10:
                return g / np.sum(g)  # normalize to sum to 1
    except Exception:
        pass
    return None

def nesterov_subgradient(p, deadline, lr_init=0.03, momentum_decay=0.9,
                          top_k=15, temp_frac=0.05, verbose=False):
    """Nesterov accelerated projected subgradient on the simplex."""
    p = p.copy().astype(np.float64)
    n = len(p)
    best_p = p.copy()
    best_score = normalized_score(p)
    
    y = p.copy()  # Nesterov look-ahead
    momentum = np.zeros(n)
    lr = lr_init
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        # Evaluate gradient at Nesterov point
        grad, M = subgradient_max_conv(y, top_k=top_k, temp_frac=temp_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        
        # Update
        momentum = momentum_decay * momentum - lr * grad
        new_p = project_simplex(p + momentum)
        
        # Nesterov update
```

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 via projected subgradient descent 
with momentum on the simplex, combined with periodic LP-based direction 
finding and perturbation restarts. Fixed n=1024.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

N = 1024

def compute_score(a):
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))

def norm_score(p):
    """Score for unit-sum sequence."""
    n = len(p)
    return float(2 * n * np.max(np.convolve(p, p)))

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgrad(p, top_k=15, temp_frac=0.03):
    """Weighted subgradient of max convolution."""
    n = len(p)
    conv = np.convolve(p, p)
    M = float(np.max(conv))
    if M <= 0:
        return np.zeros(n), 0.0
    tidx = np.argsort(conv)[::-1][:top_k]
    tvals = conv[tidx]
    spread = max(tvals[0] - tvals[-1], M * 1e-8)
    temp = spread * temp_frac
    w = np.exp((tvals - tvals[0]) / temp)
    w /= w.sum()
    g = np.zeros(n)
    for k, wi in zip(tidx, w):
        if wi < 1e-14:
            continue
        jlo = max(0, k - n + 1)
        jhi = min(k, n - 1)
        g[jlo:jhi+1] += 2.0 * wi * p[k-jhi:k-jlo+1][::-1]
    return g, M

def build_conv_mat(p):
    """Sparse convolve matrix."""
    n = len(p)
    rows = np.empty(2*n*(n+1)//2, dtype=np.int32)
    cols = np.empty(2*n*(n+1)//2, dtype=np.int32)
    vals = np.empty(2*n*(n+1)//2, dtype=np.float64)
    idx = 0
    for k in range(2*n-1):
        jlo = max(0, k-n+1)
        jhi = min(k, n-1)
        cnt = jhi - jlo + 1
        rows[idx:idx+cnt] = k
        cols[idx:idx+cnt] = np.arange(jlo, jhi+1)
        vals[idx:idx+cnt] = p[k - np.arange(jhi, jlo-1, -1)]
        idx += cnt
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2*n-1, n))

def lp_direction(p, M):
    """Solve LP for direction."""
    n = len(p)
    try:
        A = build_conv_mat(p)
        c = -np.ones(n)
        b = np.full(2*n-1, M)
        bounds = [(0, None)]*n
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            s = float(np.sum(res.x))
            if s > 1e-10:
                return res.x / s
    except:
        pass
    return None

def subgrad_optimize(p, deadline, lr=0.02, mom_decay=0.9, top_k=15, 
                     temp_frac=0.03, verbose=False, prefix=""):
    """Projected subgradient with heavy ball momentum."""
    p = p.copy()
    n = len(p)
    best_p = p.copy()
    best_s = norm_score(p)
    momentum = np.zeros(n)
    cur_lr = lr
    it = 0
    stall = 0
    
    while time.monotonic() < deadline:
        g, M = subgrad(p, top_k, temp_frac)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        momentum = mom_decay * momentum - cur_lr * g
        new_p = project_simplex(p + momentum)
        s = norm_score(new_p)
        it += 1
        if s < best_s:
            p = new_p
            best_p = p.copy()
            best_s = s
            stall = 0
            cur_lr *= 0.9999
        else:
            stall += 1
            cur_lr *= 0.97
            momentum *= 0.5
            if stall > 300:
                cur_lr = max(lr * 0.3, 1e-5)
                momentum = np.zeros(n)
                stall = 0
                if cur_lr < 1e-7:
                    break
        if verbose and it % 10000 == 0:
            print(f"  {prefix}it={it} score={best_s:.8f} lr={cur_lr:.6f}")
    return best_p, best_s

def do_lp_step(p, time_budget=8.0):
    """One LP-based step with line search."""
    n = len(p)
    conv = np.convolve(p, p)
    M = float(np.max(conv))
    g = lp_direction(p, M)
    if g is None:
        return p, norm_score(p)
    best_s = norm_score(p)
    best_a = 0.0
    for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
        h = (1-alpha)*p + alpha*g
        h = np.clip(h, 0, None)
        s_ = float(np.sum(h))
        if s_ < 1e-10:
            continue
        h /= s_
        sc = norm_score(h)
        if sc < best_s:
            best_s = sc
            best_a = alpha
    if best_a > 0:
        h = (1-best_a)*p + best_a*g
        h = np.clip(h, 0, None)
        s_ = float(np.sum(h))
        if s_ > 1e-10:
            h /= s_
            return h, best_s
    return p, best_s

def make_windows(n, rng):
    """Generate diverse initializations."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(('uniform', np.ones(n)))
    
    # Hann
    inits.append(('hann', 0.5 + 0.5*np.cos(2*np.pi*idx/(n-1))))
    
    # Hamming
    inits.append(('hamming', 0.54 - 0.46*np.cos(2*np.pi*idx/(n-1))))
    
    # Blackman
    inits.append(('blackman', 0.42 - 0.5*np.cos(2*np.pi*idx/(n-1)) + 0.08*np.cos(4*np.pi*idx/(n-1))))
    
    # Raised cosine
    inits.append(('rcos', 0.5*(1+np.cos(np.pi*idx/(n-1)))))
    
    # Cosine squared
    inits.append(('cos2', np.cos(np.pi*idx/(2*(n-1)))**2))
    
    # Gaussian (various widths)
    center = (n-1)/2.0
    for sigma_frac in [0.25, 0.35, 0.45, 0.5]:
        g = np.exp(-0.5*((idx-center)/(n*sigma_frac))**2)
        inits.append((f'gauss_{sigma_frac}', g))
    
    # Kaiser (approximate)
    for beta in [2.0, 3.0, 5.0, 8.0]:
        x = 2*idx/(n-1) - 1
        # Approximate Kaiser using Bessel function series
        from scipy.special import i0
        k = i0(beta * np.sqrt(np.maximum(1-x**2, 0)))
        inits.append((f'kaiser_{beta}', k))
    
    # Random
    for _ in range(3):
        r = rng.uniform(0.1, 1.0, size=n)
        inits.append((f'rand_{rng.integers(1e6)}', r))
    
    # Symmetric noise on uniform
    base = np.ones(n)
    noise = rng.normal(0, 0.1, size=n)
    noise = (noise + noise[::-1]) / 2  # make symmetric
    inits.append(('sym_noise', np.maximum(base + noise, 0.01)))
    
    return inits
