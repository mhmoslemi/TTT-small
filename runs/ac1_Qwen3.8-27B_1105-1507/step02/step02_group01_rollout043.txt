import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_obj(a):
    """Compute the objective function value."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smooth_max_obj(a, T):
    """Smooth max of autocorrelation, with a normalized to sum=1."""
    s = np.sum(a)
    if s < 1e-10:
        return np.inf, np.zeros_like(a)
    a_norm = a / s
    conv = np.convolve(a_norm, a_norm)
    shifted = conv / T
    shifted = np.clip(shifted, -500, 500)
    max_shifted = np.max(shifted)
    exp_vals = np.exp(shifted - max_shifted)
    sum_exp = np.sum(exp_vals)
    val = T * (max_shifted + np.log(sum_exp))
    # Gradient
    p = exp_vals / sum_exp
    a_rev = a_norm[::-1]
    conv_p = np.convolve(p, a_rev, mode='full')
    n = len(a_norm)
    grad = conv_p[n-1:n-1+n] / s  # chain rule for normalization
    return val, grad

def lp_solve(a, M=None, max_g=10.0):
    """Solve the LP: maximize sum(g) s.t. a*g <= M, 0 <= g <= max_g."""
    n = len(a)
    if M is None:
        conv = np.convolve(a, a)
        M = np.max(conv)
    # Build constraint matrix
    # (a*g)[k] = sum_j a[j] * g[k-j] <= M for k=0,...,2n-2
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[k, j] = a[k-j]
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, max_g)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, -res.fun, M
    return None, None, M

def lp_phase(a, max_time, max_iters=500):
    """LP-based alternating optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    t = 0.5
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break
        
        g, g_sum, M = lp_solve(a)
        if g is None:
            break
        if g_sum <= 1.0 + 1e-6:
            stall += 1
            if stall > 10:
                break
            # Try a smaller step or perturbation
            continue
        stall = 0
        
        g = g / g_sum  # normalize to sum=1
        
        # Backtracking line search
        accepted = False
        for tt in [t, t*0.5, t*0.25, t*0.1, t*0.05]:
            a_new = (1-tt) * a + tt * g
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_a = a_new.copy()
                best_obj = obj
                a = a_new
                accepted = True
                t = min(1.0, t * 1.1)  # increase step size slightly
                break
        
        if not accepted:
            t = max(0.01, t * 0.5)
            if t < 0.01:
                break
    
    return best_a, best_obj

def gd_phase(a, max_time, T_init=None, max_iters=5000):
    """Gradient descent on smooth max."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if T_init is None:
        conv = np.convolve(a, a)
        T_init = 0.05 * np.max(conv)
    T = T_init
    lr = 0.05
    
    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break
        
        val, grad = smooth_max_obj(a, T)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        
        # Gradient step with normalization
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)
        
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            T = max(0.001, T * 0.998)
            lr = min(0.1, lr * 1.01)
        else:
            lr *= 0.9
            if lr < 1e-8:

"""
Optimized step function coefficient search using aggressive LP alternating,
corrected smooth-max gradient descent, scipy L-BFGS-B optimization,
diverse initialization, and block perturbation refinement.
Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smooth_max_and_grad(a, T):
    """Compute smooth max of autocorrelation and its gradient (a normalized to sum=1)."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf, np.zeros(n)
    a_n = a / s
    conv = np.convolve(a_n, a_n)
    shifted = conv / T
    shifted = np.clip(shifted, -500.0, 500.0)
    max_s = np.max(shifted)
    exp_vals = np.exp(shifted - max_s)
    sum_exp = float(np.sum(exp_vals))
    val = T * (max_s + np.log(sum_exp))
    p = exp_vals / sum_exp
    # Gradient: d(val)/da[m] = 2 * sum_k p[k] * a_n[k-m] * I[0<=k-m<n]
    # = 2 * (p * a_n_rev)[n-1+m]
    a_rev = a_n[::-1]
    conv_pa = np.convolve(p, a_rev, mode='full')
    grad = 2.0 * conv_pa[n - 1:n - 1 + n]
    # Correct for normalization: a_n = a/s, so da_n/da = (I - a_n a_n^T)/s
    # grad_total = (grad - a_n * dot(grad, a_n)) / s
    grad = (grad - a_n * np.dot(grad, a_n)) / s
    return val, grad


def lp_solve(a):
    """Solve LP: max sum(g) s.t. a*g <= max(a*a), 0<=g<=bound. Returns (g, g_sum, M)."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    # Build constraint matrix A where (A @ g)[k] = (a*g)[k]
    # (a*g)[k] = sum_j a[k-j]*g[j] for valid j
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # A[k, j] = a[k-j] for j in [j_start, j_end]
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, 100.0)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M


def lp_phase(a, max_time, max_iters=2000):
    """Aggressive LP-based alternating optimization with adaptive step size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    t = 0.3
    stall = 0
    consecutive_no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break

        g, g_sum, M = lp_solve(a)
        if g is None:
            break
        if g_sum <= 1.0 + 1e-5:
            stall += 1
            if stall > 15:
                break
            # Perturb to escape
            a = a + 0.01 * np.random.default_rng(it).normal(0, 1, n)
            a = np.maximum(a, 0)
            s = np.sum(a)
            a = a / s
            consecutive_no_improve = 0
            continue
        stall = 0

        g = np.maximum(g, 0)
        g_sum_actual = np.sum(g)
        if g_sum_actual < 1e-10:
            break
        g = g / g_sum_actual

        # Backtracking line search
        accepted = False
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            tt = t * factor
            a_new = (1.0 - tt) * a + tt * g
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_a = a_new.copy()
                best_obj = obj
                a = a_new
                accepted = True
                consecutive_no_improve = 0
                t = min(0.8, t * 1.05)
                break

        if not accepted:
            t = max(0.005, t * 0.6)
            consecutive_no_improve += 1
            if consecutive_no_improve > 50:
                break

    a = best_a
    return a, best_obj


def gd_phase(a, max_time, T_init=None, max_iters=10000):
    """Gradient descent on smooth max with adaptive temperature and learning rate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if T_init is None:
        conv = np.convolve(a, a)
        T_init = 0.05 * float(np.max(conv))
    T = T_init
    lr = 0.1
    last_improve = 0
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break

        _, grad = smooth_max_and_grad(a, T)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)

        if obj < best_obj - 1e-10:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            last_improve = it
            T = max(0.0005, T * 0.997)
            lr = min(0.2, lr * 1.005)
        else:
            lr *= 0.92
            if lr < 1e-10:
                # Reheat with perturbation
                a_pert = a + 0.02 * rng.normal(0, 1, n)
                a_pert = np.maximum(a_pert, 0)
                s_p = np.sum(a_pert)
                if s_p > 1e-10:
                    a = a_pert / s_p
                lr = 0.05
                T = max(T, 0.02 * float(np.max(np.convolve(a, a))))
                last_improve = it
                if it - last_improve > 5000:
                    break
            if it - last_improve > 3000:
                break

    return best_a, best_obj


def scipy_phase(a, max_time, T):
    """Use scipy L-BFGS-B to minimize smooth max objective."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a, np.inf
    a = a / s

    stop_flag = [False]
    start = time.monotonic()

    def obj_fn(x):
        if stop_flag[0]:
            return 1e10
        x_pos = np.maximum(x, 0)
        s = np.sum(x_pos)
        if s < 1e-10:
            return 1e10
        x_n = x_pos / s
        conv = np.convolve(x_n, x_n)
        shifted = conv / T
        shifted = np.clip(shifted, -500, 500)
        max_s = np.max(shifted)
        return T * (max_s + np.log(np.sum(np.exp(shifted - max_s))))

    def grad_fn(x):
        if stop_flag[0]:
            return np.zeros(n)
        x_pos = np.maximum(x, 0)
        s = np.sum(x_pos)
        if s < 1e-10:
            return np.zeros(n)
        x_n = x_pos / s
        conv = np.convolve(x_n, x_n)
        shifted = conv / T
        shifted = np.clip(shifted, -500, 500)
        max_s = np.max(shifted)
        exp_vals = np.exp(shifted - max_s)
        p = exp_vals / np.sum(exp_vals)
        a_rev = x_n[::-1]
        conv_pa = np.convolve(p, a_rev, mode='full')
        grad = 2.0 * conv_pa[n - 1:n - 1 + n]
        grad = (grad - x_n * np.dot(grad, x_n)) / s
        return grad

    def callback(xk):
        if time.monotonic() - start > max_time:
            stop_flag[0] = True

    try:
        result = minimize(obj_fn, a, jac=grad_fn, method='L-BFGS-B',
                          bounds=[(0, None)] * n,
                          options={'maxiter': 5000, 'maxfun': 10000,
                                   'ftol': 1e-12, 'gtol': 1e-8, 'iprint': -1},
                          callback=callback)
        a_opt = np.maximum(result.x, 0)
        s_opt = np.sum(a_opt)
        if s_opt > 1e-10:
            a_opt = a_opt / s_opt
        obj_opt = compute_obj(a_opt)
        return a_opt, obj_opt
    except Exception:
        return a, compute_obj(a)


def block_refine(a, rng, max_time):
    """Block perturbation refinement with adaptive parameters."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time and it < 500000:
        progress = (time.monotonic() - start) / max_time
        width = max(1, int(n * rng.uniform(0.01, 0.12)))
        pos = int(rng.integers(0, max(1, n - width + 1)))
        candidate = best_a.copy()
        sigma = 0.015 + 0.05 * (1.0 - progress)
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.maximum(candidate, 0)
        s = np.sum(candidate)
        if s < 1e-10:
            it += 1
            continue
        candidate = candidate / s
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj
        it += 1

    return best_a, best_obj


def diverse_init(n, rng, init_type=None):
    """Generate diverse initializations for optimization."""
    if init_type is None:
        init_type = int(rng.integers(0, 10))

    if init_type == 0:
        # Increasing ramp
        a = np.linspace(0.05, 1.0, n)
    elif init_type == 1:
        # Decreasing ramp
        a = np.linspace(1.0, 0.05, n)
    elif init_type == 2:
        # Two bumps (asymmetric)
        c1, c2 = n * 0.2, n * 0.7
        w = n * 0.08
        a = np.exp(-0.5 * ((np.arange(n) - c1) / w) ** 2) * 1.5
        a += np.exp(-0.5 * ((np.arange(n) - c2) / w) ** 2) * 0.8
        a += 0.03
    elif init_type == 3:
        # High frequency cosine
        freq = rng.uniform(0.2, 0.7)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.45 * np.cos(2 * np.pi * freq * np.arange(n) + phase)
        a = np.maximum(a, 0.01)
    elif init_type == 4:
        # Edge heavy
        a = np.ones(n) * 0.05
        a[0] = 0.8
        a[-1] = 0.6
        a[1] = 0.3
        a[-2] = 0.3
        if n > 3:
            a[2] = 0.15
            a[-3] = 0.15
    elif init_type == 5:
        # Anti-symmetric ramp with noise
        a = np.linspace(1.0, 0.02, n)
        a += 0.08 * rng.random(n)
    elif init_type == 6:
        # Gaussian bump off-center
        center = n * rng.uniform(0.2, 0.4)
        width = n * rng.uniform(0.1, 0.25)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        a += 0.04
    elif init_type == 7:
        # Periodic pattern
        period = max(2, int(n / rng.integers(20, 60)))
        pattern = np.maximum(0.1, rng.random(period))
        a = np.tile(pattern, n // period + 1)[:n]
        a += 0.02 * rng.random(n)
    elif init_type == 8:
        # Smooth random (low-pass filtered noise)
        raw = rng.random(n)
        kernel = np.ones(5) / 5
        a = np.convolve(raw, kernel, mode='same')
        a = np.maximum(a, 0.01)
    else:
        # Pure random
        a = np.maximum(0, rng.normal(0.5, 0.3, n))

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def full_pipeline(a_init, time_budget, rng=None):
    """Run full optimization pipeline: LP -> GD -> LP -> GD -> scipy -> block."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: LP (most important)
    lp_time = min(time_budget * 0.40, 40.0)
    a, obj = lp_phase(a, lp_time)
    if obj < best_obj:
        best_a = a.copy()
        best_obj = obj

    # Phase 2: GD on smooth max
    if time.monotonic() - start < time_budget * 0.65:
        gd_time = min(time_budget * 0.20, 20.0)
        a, obj = gd_phase(a, gd_time)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj

    # Phase 3: Second LP round
    if time.monotonic() - start < time_budget * 0.80:
        lp_time2 = min(time_budget * 0.12, 12.0)
        a, obj = lp_phase(a, lp_time2)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj

    # Phase 4: Scipy optimization
    if time.monotonic() - start < time_budget * 0.90:
        sc_time = min(time_budget * 0.08, 10.0)
        conv = np.convolve(a, a)
        T_sc = 0.01 * float(np.max(conv))
        a_sc, obj_sc = scipy_phase(a, sc_time, T_sc)
        if obj_sc < best_obj:
            best_a = a_sc.copy()
            best_obj = obj_sc

    # Phase 5: Fine GD
    if time.monotonic() - start < time_budget * 0.95:
        gd_time2 = min(time_budget * 0.05, 8.0)
        a, obj = gd_phase(a, gd_time2, T_init=0.003 * float(np.max(np.convolve(a, a))))
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj

    return best_a, best_obj
