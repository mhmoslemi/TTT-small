2. Update `a` to satisfy `T_λ(a) = constant` on the support.
3. Update `λ` to satisfy the complementary slackness (λ_k > 0 only where conv[k] = M).
4. Iterate.

def fft_conv(a):
    """Fast linear autocorrelation via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    conv = fft_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def subgradient_max_conv(conv, a, tau):
    """Compute smoothed gradient of max(conv) using log-sum-exp with temperature tau."""
    n = len(a)
    cmax = np.max(conv)
    shifted = tau * (conv - cmax)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum
    # Gradient: d(conv[k])/d(a[i]) = 2*a[k-i] if 0<=k-i<n
    # Sum over k: 2 * sum_k weights[k] * a[k-i] * 1[0<=k-i<n]
    # This is 2 * correlate(weights, a, 'full')[n-1 + i] ... let me verify.
    # np.correlate(v, a, 'full')[m] = sum_j v[j] * a[j+m-n] approximately
    # Let's use direct computation for correctness.
    # grad[i] = 2 * sum_{k: 0<=k-i<n} weights[k] * a[k-i]
    # Let j = k-i, so k = j+i, j from 0 to n-1:
    # grad[i] = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j]  (if i+j < 2n-1)
    # This is 2 * (weights * a) in a specific way.
    # Actually: grad[i] = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j]
    # This is the cross-correlation: np.correlate(weights, a, 'full')[i + n - 1]
    # Let's verify: np.correlate(w, a, 'full')[k] = sum_j w[j+k-(n-1)] * a[j] for 'full'
    # Hmm, let me just use a different approach.
    # grad[i] = 2 * sum_j a[j] * weights[i+j]
    # Define reverse a: ar[j] = a[n-1-j]
    # Then sum_j a[j] * weights[i+j] = sum_j ar[n-1-j] * weights[i+j]
    # Let k = n-1-j: sum_k ar[k] * weights[i+n-1-k]
    # This is (ar * weights)[i+n-1] in convolution sense = np.convolve(ar, weights, 'full')[i+n-1]
    # But actually simpler: just use np.correlate
    # np.correlate(weights, a, mode='full') gives cross-correlation
    # cross_corr[m] = sum_n weights[n+m] * a[n] (for valid ranges)
    # We want sum_j a[j] * weights[i+j] = cross_corr with offset i
    # Actually np.correlate(weights, a, 'full')[i + (n-1)] might not be right.
    # Let me just compute directly using a Toeplitz-like approach or FFT.
    
    # Direct: grad[i] = 2 * sum_{j=0}^{n-1} min(n-1, 2n-2-i) >= i+j
    # For i in [0, n-1], j in [0, n-1], i+j ranges from i to i+n-1, all < 2n-1.
    # So grad[i] = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j]
    # This is a matrix-vector product where M[i,j] = weights[i+j]
    # M is a Hankel matrix. We can compute this via FFT.
    
    # Alternative: use the fact that this is a cross-correlation
    # sum_j a[j] * weights[i+j] = <a, shift_i(weights)>
    # where shift_i shifts weights to the right by i
    # = np.dot(a, weights[i:i+n])  (since i+j goes from i to i+n-1, all valid)
    
    # For efficiency, use FFT-based convolution:
    # sum_j a[j] * weights[i+j] for i = 0,...,n-1
    # This is the "anti-diagonal" sum, which is a cross-correlation.
    # np.correlate(weights, a, 'full') has length (2n-1) + (n-1) = 3n-2... no.
    # weights has length 2n-1, a has length n.
    # np.correlate(weights, a, 'full') has length (2n-1) + n - 1 = 3n-2.
    # cross_corr[k] = sum_j weights[j+k] * a[j] (for valid j, meaning j in [0,n-1] and j+k in [0, 2n-2])
    # For k = i, j in [0, n-1], j+i in [i, i+n-1], all in [0, 2n-2] since i < n and i+n-1 < 2n-1.
    # So cross_corr[i] = sum_{j=0}^{n-1} weights[i+j] * a[j] = what we want!
    
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[:n]
    return grad


def subgradient_step(a, conv, step_size, tau):
    """One step of subgradient descent with projection onto simplex (sum=1, a>=0)."""
    n = len(a)
    grad = subgradient_max_conv(conv, a, tau)
    # Center to maintain sum
    grad_centered = grad - np.mean(grad)
    a_new = a - step_size * grad_centered
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def active_set_subgradient(a, conv, step_size, threshold_frac=0.95):
    """Subgradient using only the active set (lags near the maximum)."""
    n = len(a)
    cmax = np.max(conv)
    active = np.where(conv >= cmax * threshold_frac)[0]
    if len(active) == 0:
        active = np.array([np.argmax(conv)])
    
    # Equal weights on active set
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / len(active)
    
    # Gradient: 2 * sum_k w[k] * a[k-i] * 1[0<=k-i<n]
    # = 2 * correlate(w, a, 'full')[:n]
    c = np.correlate(w, a, mode='full')
    grad = 2.0 * c[:n]
    grad_centered = grad - np.mean(grad)
    
    a_new = a - step_size * grad_centered
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def pairwise_swap(a, conv, max_swaps=100):
    """Try pairwise mass transfers to reduce the convolution peak."""
    n = len(a)
    cmax = np.max(conv)
    a_new = a.copy()
    improved = False
    rng = np.random.default_rng(42)
    
    # Find the hottest lag
    k_hot = int(np.argmax(conv))
    
    # Try moving mass from high-contributing positions to low-contributing ones
    for attempt in range(max_swaps):
        # Find the pair (i, j) with i+j most related to k_hot that has large product
        # and a position p that's "cold" (not contributing much to any hot lag)
        candidates_i = []
        for i in range(max(0, k_hot - n + 1), min(n, k_hot + 1)):
            j = k_hot - i
            if 0 <= j < n:
                candidates_i.append((a[i] * a[j], i, j))
        
        if not candidates_i:
            break
        candidates_i.sort(reverse=True)
        
        best_improvement = 0
        best_move = None
        
        for prod, i, j in candidates_i[:5]:
            # Try moving some mass from i (or j) to a random cold position
            for _ in range(2):
                p = rng.integers(0, n)
                if p == i or p == j:
                    continue
                for source in [i, j]:
                    # Try moving delta from source to p
                    delta = a_new[source] * 0.1
                    if delta < 1e-12:
                        continue
                    a_test = a_new.copy()
                    a_test[source] -= delta
                    a_test[p] += delta
                    if a_test[source] < 0:
                        continue

                            best_improvement = cmax - new_max

        if best_move is None:
            break
        
        source, p, delta = best_move
        a_new[source] -= delta
        a_new[p] += delta
        improved = True
    
    if improved:
        s = a_new.sum()
        if s > 1e-15:

    return a_new, improved


def lp_best_response(a, M, max_iter_lp=50):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0.
    Returns normalized g or None."""
    n = len(a)
    
    # Build constraint matrix: (a*g)_k = sum_j a[j]*g[k-j] <= M
    # For each k in [0, 2n-2], the constraint is sum_{j: valid} a[j]*g[k-j] <= M
    # Let i = k-j, so j = k-i: sum_i a[k-i]*g[i] <= M (with 0<=k-i<n and 0<=i<n)
    # This is a Toeplitz-like matrix.
    
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = a[j]
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            g = res.x
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def self_consistent_field(a, n_iters=200, verbose=False):
    """Self-consistent field iteration:
    1. Given lambda (dual variables), solve for a: T_lambda(a) = const on support
    2. Given a, update lambda based on tightness of constraints
    """
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s
    
    # Initialize lambda uniformly
    lam = np.ones(2 * n - 1) / (2 * n - 1)
    
    for it in range(n_iters):

        conv = fft_conv(a)
        M = np.max(conv)
        
        # Update lambda: emphasize tight constraints
        # Use power weighting for "soft" active set
        tight_frac = 0.95
        lam = np.where(conv >= M * tight_frac, 1.0, 0.1)
        lam = lam / lam.sum()
        
        # Solve for a: 2 * sum_k lam[k] * a[k-i] * 1[valid] = mu (constant) on support
        # This is a linear system: T_lam @ a = mu * ones (on support)
        # T_lam[i, j] = 2 * lam[i+j] (if i+j < 2n-1) -- wait, let me re-derive
        # The gradient condition: for i in support(a):
        #   2 * sum_k lam[k] * a[k-i] * 1[0<=k-i<n] = mu
        # Let j = k-i: 2 * sum_j lam[i+j] * a[j] = mu
        # So: sum_j (2 * lam[i+j]) * a[j] = mu
        # This is (H_lam @ a) = mu * 1, where H_lam[i,j] = 2*lam[i+j]
        
        # Build H_lam (Hankel matrix)
        H = np.zeros((n, n))
        for i in range(n):
            for j in range(n):
                H[i, j] = 2.0 * lam[i + j]
        
        # Solve H@a = mu*e on the support, with a[i]=0 outside support
        # For simplicity, solve the full system with a[i]>=0 constraint
        # Use: H@a = mu*e, sum(a)=1, a>=0
        # From sum(a)=1: mu = sum_i (H@a)_i / n ... not quite.
        # Actually, sum over support: (H@a) = mu on support.
        # Multiply by a^T: a^T H a = mu * sum(a) = mu (since sum=1)
        # Also a^T H a = sum_{i,j} a[i]*H[i,j]*a[j] = 2*sum_{i,j} lam[i+j]*a[i]*a[j] = 2*sum_k lam[k]*conv[k]
        
        # Let's just solve: minimize ||H@a - 1||^2 s.t. sum(a)=1, a>=0
        # This is a QP. For speed, use a simpler approach:
        # a_i proportional to (H^{-1} @ 1)_i (if H is invertible and the solution is non-negative)
        
        try:
            # Try to solve H@a = 1 (setting mu=1 for now, we'll renormalize)
            a_trial = np.linalg.solve(H + 1e-10 * np.eye(n), np.ones(n))
            if np.all(a_trial >= -1e-10):

                if s_t > 1e-10:
                    a = a_trial / s_t
        except np.linalg.LinAlgError:
            # Fallback: use power iteration or skip
            pass
    
    return a

    a_cur = best_seq.copy()
    a_best = best_seq.copy()
    score_best = best_score
    
    polish_start = time.monotonic()
    polish_deadline = deadline - 30
    
    method_cycle = 0
    last_improvement = time.monotonic()
    
    while time.monotonic() < polish_deadline:
        remaining = polish_deadline - time.monotonic()
        if remaining < 5:
            break
        
        method_cycle += 1
        
        if method_cycle % 1 == 0:
            # Subgradient with adaptive tau
            conv = fft_conv(a_cur)
            cmax = np.max(conv)
            n_active = np.sum(conv >= cmax * 0.98)
            tau = max(5.0, min(200.0, 50.0 / max(1, n_active)))
            step = 0.03 * (1.0 + 0.5 * np.sin(method_cycle * 0.01))
            a_new = subgradient_step(a_cur, conv, step, tau)
            a_cur = a_new
        elif method_cycle % 2 == 0:
            # Active set with smaller step
            conv = fft_conv(a_cur)
            a_cur = active_set_subgradient(a_cur, conv, 0.02)
        elif method_cycle % 5 == 0 and n_best <= 300:
            # LP step
            conv = fft_conv(a_cur)
            s_cur = a_cur.sum()
            M = float(np.max(conv / s_cur ** 2))
            g = lp_best_response(a_cur / s_cur, M)
            if g is not None:
                t_blend = 0.2 + 0.1 * np.sin(method_cycle * 0.05)
                a_new = (1 - t_blend) * a_cur / s_cur + t_blend * g
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s > 1e-10:
                    a_cur = a_new / s
        elif method_cycle % 10 == 0:
            # Self-consistent field
            if n_best <= 250:
                a_cur = self_consistent_field(a_cur, n_iters=30)
        elif method_cycle % 20 == 0:
            # Pairwise swap
            conv = fft_conv(a_cur)
            a_new, _ = pairwise_swap(a_cur, conv, max_swaps=100)
            a_cur = a_new
        
        # Check improvement
        obj = compute_obj(a_cur)
        if obj < score_best - 1e-10:
            score_best = obj
            a_best = a_cur.copy()
            last_improvement = time.monotonic()
            if obj < best_score:
                best_score = obj
                best_seq = a_best.copy()
                print(f"  Polish iter {method_cycle}: NEW GLOBAL BEST obj={best_score:.6f}")
        
        # If stuck for a while, perturb
        if time.monotonic() - last_improvement > 60:
            rng = np.random.default_rng(method_cycle)
            noise_scale = min(0.2, 0.05 + 0.01 * (time.monotonic() - last_improvement) / 60)
            a_pert = a_cur * np.exp(rng.normal(0, noise_scale, n_best))
            a_pert = np.maximum(a_pert, 0)
            s_p = a_pert.sum()
            if s_p > 1e-10:
                a_pert = a_pert / s_p
            if compute_obj(a_pert) < compute_obj(a_cur):
                a_cur = a_pert
            last_improvement = time.monotonic()
    
    # Phase 4: Final fine-tuning
    print("\n=== Phase 4: Final Fine-tuning ===")
    if time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        a_final = a_best.copy()
        for _ in range(int(min(5000, remaining * 5))):
            if time.monotonic() > deadline - 5:
                break
            conv = fft_conv(a_final)
            a_final = subgradient_step(a_final, conv, 0.005, 50.0)
            obj = compute_obj(a_final)
            if obj < best_score:
                best_score = obj
                best_seq = a_final.copy()
    
    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)
    
    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation != best_score and validation < np.inf:
        print(f"  Validation: computed={best_score:.6f}, validated={validation:.6f}")
        best_score = validation
    
    return result
