    if n == 0:

def compute_gradient(a, T):

    n = len(a)
    conv = np.convolve(a, a)

    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    
    # grad_i = sum_{k=i}^{i+n-1} p[k] * a[k-i] (for valid k)
    grad = np.zeros(n)
    for i in range(n):

        k_end = min(2*n-1, i+n)
        if k_start >= 2*n-1:
            break

    # The gradient of 2*n*max_b/S^2 w.r.t. a_i (with S = sum(a)):
    # d/d a_i [2n * max_b / S^2] = 2n * (grad_i * S^2 - max_b * 2*S) / S^4

    # So the effective gradient is grad_i - mean(grad) (project onto sum=0 hyperplane)

"""

method, and multi-length exploration. Targets objective below 1.5027.

import numpy as np
from numpy.lib.stride_tricks import sliding_window_view
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_softmax_gradient(a, T):
    """Compute soft-max value and its gradient w.r.t. a (normalized to sum=1)."""
    n = len(a)
    conv = np.convolve(a, a)
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp

    # grad[i] = sum_{k=i}^{i+n-1} p[k] * a[k-i]
    # Using sliding window: windows[i] = p[i:i+n], grad = windows @ a
    windows = sliding_window_view(p, n)
    grad = windows @ a

    # Add the self-contribution factor (db_k/da_i = 2*a[k-i] when valid)
    # The sliding window already gives sum p[k]*a[k-i], multiply by 2
    grad *= 2.0

    sm_val = max_c + T * np.log(sum_exp)
    return sm_val, grad, p, conv


def mirror_descent_phase(a, max_time, T_init=None, lr_init=0.3):
    """Mirror descent on soft-max of convolution with adaptive parameters."""
    n = len(a)
    a = a.astype(np.float64).copy()
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv = np.convolve(a, a)
    if T_init is None:
        T_init = 0.005 * float(np.max(conv))
    T = T_init
    lr = lr_init
    stagnation = 0
    max_stagnation = 300

    for it in range(200000):
        if time.monotonic() - start > max_time:
            break

        sm_val, grad, p, conv = compute_softmax_gradient(a, T)
        max_c = float(np.max(conv))

        # Project gradient onto sum=0 hyperplane (to maintain normalization)
        grad_centered = grad - np.mean(grad)

        # Mirror descent update: a_i <- a_i * exp(-lr * grad_centered_i)
        update = np.clip(-lr * grad_centered, -40.0, 40.0)
        a_new = a * np.exp(update)
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            lr *= 0.5
            if lr < 1e-15:
                break
            continue
        a_new = a_new / s_new

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            stagnation = 0
            # Cool temperature gradually
            T = max(1e-7, T * 0.9995)
            # Slightly increase lr if making good progress
            if stagnation == 0 and lr < lr_init:
                lr = min(lr * 1.02, lr_init)
        else:
            stagnation += 1
            lr *= 0.97
            if stagnation > max_stagnation:

                a = best_a * np.exp(noise_scale * np.random.randn(n))
                a = np.maximum(a, 0.0)

                T = max(T * 0.5, 1e-7)

def subgradient_phase(a, max_time):

def block_refine(a, rng, max_time):

                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Phase A: Aggressively optimize parent
    if best_seq is not None:
        n_parent = len(best_seq)
        parent_budget = min(350.0, float(budget_s) * 0.4)
        print(f"\n=== Phase A: Optimizing parent n={n_parent}, budget={parent_budget:.0f}s ===")
        a_opt, obj_opt = full_optimize(best_seq.copy(), parent_budget, rng)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"Parent optimized: obj={best_score:.6f}")

    # Phase B: Multiple restarts with parent's length (or 300)
    if time.monotonic() < deadline - 60:
        n_main = len(best_seq) if best_seq is not None else 300
        remaining = deadline - time.monotonic()
        num_restarts = max(3, min(6, int(remaining / 70)))
        time_per = remaining / (num_restarts + 1)

        print(f"\n=== Phase B: {num_restarts} restarts at n={n_main} ===")
        for r in range(num_restarts):
            if time.monotonic() > deadline - 60:
                break
            a_init = structured_init(n_main, rng)
            t_start = time.monotonic()
            a_opt, obj_opt = full_optimize(a_init, time_per * 0.9, rng)
            elapsed = time.monotonic() - t_start
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Restart {r}: obj={best_score:.6f} ** NEW BEST ** ({elapsed:.0f}s)")
            else:
                print(f"  Restart {r}: obj={obj_opt:.6f} ({elapsed:.0f}s)")

    # Phase C: Try different lengths
    if time.monotonic() < deadline - 80:
        n_current = len(best_seq) if best_seq is not None else 300
        other_ns = [150, 200, 400, 500, 700, 1000]
        other_ns = [x for x in other_ns if abs(x - n_current) > 80]

        remaining = deadline - time.monotonic()
        time_per_n = remaining / max(1, len(other_ns) + 1)

        print(f"\n=== Phase C: Other lengths: {other_ns} ===")
        for n in other_ns:
            if time.monotonic() > deadline - 40:
                break
            a_init = structured_init(n, rng)
            t_start = time.monotonic()
            a_opt, obj_opt = full_optimize(a_init, time_per_n * 0.85, rng)
            elapsed = time.monotonic() - t_start
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n}: obj={best_score:.6f} ** NEW BEST ** ({elapsed:.0f}s)")
            else:
                print(f"  n={n}: obj={obj_opt:.6f} ({elapsed:.0f}s)")

    # Phase D: Final intensive polish on best
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 5
        print(f"\n=== Phase D: Final polish, {remaining:.0f}s ===")

        # Run full optimize again on best
        a_f, obj_f = full_optimize(best_seq.copy(), remaining * 0.6, rng)
        if obj_f < best_score:
            best_score = obj_f
            best_seq = a_f.copy()
            print(f"  After polish: obj={best_score:.6f}")

        # Block refine
        rem2 = deadline - time.monotonic() - 5
        if rem2 > 5:
            a_b, obj_b = block_refine(best_seq, rng, rem2 * 0.5)
            if obj_b < best_score:
                best_score = obj_b
                best_seq = a_b.copy()
                print(f"  After block: obj={best_score:.6f}")

        # Coordinate refine
        rem3 = deadline - time.monotonic() - 3
        if rem3 > 3:
            a_c, obj_c = coordinate_refine(best_seq, rem3 * 0.7)
            if obj_c < best_score:
                best_score = obj_c
                best_seq = a_c.copy()
                print(f"  After coord: obj={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    # Ensure non-negative and valid
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    final_score = compute_obj(best_seq)
    if final_score != best_score:
        best_score = final_score

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
