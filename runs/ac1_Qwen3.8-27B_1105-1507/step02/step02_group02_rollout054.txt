import numpy as np
from scipy.optimize import minimize, linprog

def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_conv2(x, y):
    """Full linear convolution of x and y via FFT."""
    n = len(x) + len(y) - 1
    size = 1
    while size < n:
        size <<= 1
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    return np.fft.irfft(fx * fy, size)[:n]


def compute_J(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if np.ndim(v) != 1 or n == 0:
        return np.ones(n) / max(n, 1)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def safe_softmax(x):

    x_shifted = x - x.max()

    s = exp_x.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return exp_x / s

        # Find active set (near-max indices)
        tol = max_b * 1e-10
        S = np.where(b >= max_b - tol)[0]

        # Compute subgradient
        grad = np.zeros(n)
        for k in S:
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            if i_min <= i_max:
                idx = np.arange(i_min, i_max + 1)
                grad[idx] += 2.0 * a[k - idx]
        grad /= len(S)

        # Adaptive step size
        step = 0.05 / (1.0 + 0.005 * it)
        a_new = project_simplex(a - step * grad)
        score = compute_J(a_new)

        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            step2 = step * 0.3
            a_new2 = project_simplex(a - step2 * grad)
            score2 = compute_J(a_new2)
            if score2 < best_score:
                best_score = score2
                best_a = a_new2.copy()
            a = a_new2 if score2 <= score else a

    return best_a, best_score


def lp_best_response(a, M, n):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    if n > 2048:
        return None
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(m, M),
                      bounds=[(0, None)] * n, method='highs')
        if res.success and res.x.sum() > 1e-10:
            return res.x
    except Exception:
        pass
    return None


def lp_iteration(a, deadline, max_iters=2000):
    """LP best-response iteration."""
    n = len(a)
    best_a = a.copy()
    best_score = compute_J(a)
    a = a.copy()
    a = np.maximum(a, 1e-6)
    a = a / a.sum()

    t = 0.5
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, n)
        if g is None:
            break
        g = g / g.sum()

        t = max(0.02, 0.5 * (1.0 - 0.5 * it / max_iters))
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        score = compute_J(a_new)

        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            t2 = t * 0.3
            a_new2 = (1.0 - t2) * a + t2 * g
            a_new2 = np.maximum(a_new2, 0)
            s2 = a_new2.sum()
            if s2 > 1e-10:
                a_new2 = a_new2 / s2
                score2 = compute_J(a_new2)
                if score2 < best_score:
                    best_score = score2
                    best_a = a_new2.copy()
                a = a_new2

    return best_a, best_score


def init_uniform(n):
    return np.ones(n) / n


def init_cosine(n, freq, amp, phase=0.0):
    t = np.arange(n, dtype=np.float64) / n
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def init_cazac_real(n, m=1):
    """Inspired by Zadoff-Chu: |cos(pi*m*t*(t+1)/n)| pattern."""
    t = np.arange(n, dtype=np.float64)
    phase = np.pi * m * t * (t + 1.0) / n
    a = np.abs(np.cos(phase))
    a = np.maximum(a, 0.01)
    return a / a.sum()


def init_multi_peak(n, n_peaks, width_scale=3.0):
    t = np.arange(n, dtype=np.float64) / n
    a = np.ones(n) * 0.05
    for k in range(n_peaks):
        center = (k + 0.5) / n_peaks
        width = 1.0 / (n_peaks * width_scale)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    return a / a.sum()


def init_bimodal(n, ratio=0.15):
    """Two Gaussians at positions ratio and 1-ratio."""
    t = np.arange(n, dtype=np.float64) / n
    w = 0.08
    a = np.exp(-0.5 * ((t - ratio) / w) ** 2) + np.exp(-0.5 * ((t - (1 - ratio)) / w) ** 2)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def init_asymmetric(n, skew=0.3):
    """Asymmetric: more mass on one side."""
    t = np.arange(n, dtype=np.float64) / n
    a = 1.0 + skew * np.sin(np.pi * t) + 0.2 * np.cos(2 * np.pi * t)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def init_mersenne_like(n, rng):
    """Mersenne-like: random with structure."""
    a = rng.uniform(0.2, 1.0, n)
    # Add periodic modulation
    t = np.arange(n, dtype=np.float64) / n
    a *= 1.0 + 0.3 * np.cos(2 * np.pi * rng.integers(1, 5) * t)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search function: multi-phase optimization to minimize J(a)."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 25.0

    best = None
    best_score = float('inf')

    # Try parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.maximum(p, 0.0)
            sc = compute_J(p)
            if sc < best_score:
                best = p.copy()
                best_score = sc
                print(f"Parent: J={sc:.6f} (n={p.size})", flush=True)
    except (NameError, AttributeError, ValueError):
        pass

    # Define lengths to try
    lengths = [256, 384, 512, 640, 768, 1024]
    if best is not None:
        pl = best.size
        lengths = [pl] + sorted([l for l in lengths if l != pl],
                                key=lambda x: abs(x - pl))

    # === PHASE 1: L-BFGS-B smooth refinement from diverse inits ===
    print("\n=== Phase 1: Smooth L-BFGS-B with continuation ===", flush=True)
    phase1_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.35

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        # Skip very long if not enough time
        if n >= 1024 and (phase1_deadline - time.monotonic()) < 60:
            continue
        if n >= 768 and (phase1_deadline - time.monotonic()) < 30:
            continue

        inits = [
            init_uniform(n),
            init_cosine(n, 1, 0.4),
            init_cosine(n, 2, 0.35),
            init_cosine(n, 3, 0.3),
            init_cosine(n, 1, 0.4, np.pi / 3),
            init_cazac_real(n, 1),
            init_cazac_real(n, 2),
            init_cazac_real(n, 3),
            init_multi_peak(n, 3),
            init_multi_peak(n, 4),
            init_multi_peak(n, 5),
            init_bimodal(n, 0.12),
            init_bimodal(n, 0.2),
            init_asymmetric(n, 0.3),
            init_mersenne_like(n, rng),
        ]

        # Add resampled parent if available
        if best is not None:
            t_old = np.linspace(0, 1, best.size)
            t_new = np.linspace(0, 1, n)
            resampled = np.maximum(np.interp(t_new, t_old, best), 0.01)
            resampled = resampled / resampled.sum()
            inits.insert(0, resampled)

        # Limit inits based on time
        max_inits = min(len(inits), max(3, int((phase1_deadline - time.monotonic()) / 8)))

        for i, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > phase1_deadline:
                break
            betas = [2, 10, 50, 200, 1000, 5000, 20000]
            a_result, score = lbfgsb_smooth(a0, n, phase1_deadline, betas)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  NEW BEST (smooth): J={score:.6f} (n={n}, init={i})", flush=True)

    # === PHASE 2: LP best-response on best ===
    print("\n=== Phase 2: LP best-response ===", flush=True)
    phase2_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.25

    if best is not None and best.size <= 1024:
        a_result, score = lp_iteration(best, phase2_deadline, max_iters=5000)
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  NEW BEST (LP): J={score:.6f}", flush=True)

        # Also try LP from a few perturbed versions
        n = best.size
        for pert_i in range(3):
            if time.monotonic() > phase2_deadline:
                break
            noise = rng.normal(0, 0.05, n)
            a_pert = best + noise
            a_pert = project_simplex(a_pert)
            a_result, score = lp_iteration(a_pert, phase2_deadline, max_iters=2000)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  NEW BEST (LP pert {pert_i}): J={score:.6f}", flush=True)

    # === PHASE 3: Subgradient refinement ===
    print("\n=== Phase 3: Subgradient refinement ===", flush=True)
    if best is not None and time.monotonic() < deadline:
        phase3_deadline = min(time.monotonic() + 120, deadline)
        a_result, score = subgradient_refine(best, best.size, phase3_deadline, max_iters=10000)
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  NEW BEST (subgrad): J={score:.6f}", flush=True)

    # === PHASE 4: Perturbation + quick optimization ===
    print("\n=== Phase 4: Perturbation search ===", flush=True)
    if best is not None and time.monotonic() < deadline:
        n = best.size
        remaining = deadline - time.monotonic()
        n_tries = max(5, min(30, int(remaining / 12)))

        for trial in range(n_tries):
            if time.monotonic() > deadline:
                break
            # Vary perturbation scale
            scale = [0.03, 0.06, 0.1, 0.02, 0.15][trial % 5]
            noise = rng.normal(0, scale, n)
            a_pert = best + noise
            a_pert = project_simplex(a_pert)

            # Quick smooth + subgrad
            sub_deadline = min(deadline, time.monotonic() + remaining / (n_tries - trial) * 0.9)
            if sub_deadline - time.monotonic() > 3:
                a_res, sc = lbfgsb_smooth(a_pert, n, sub_deadline,
                                          betas=[50, 500, 5000])
                if sc < best_score:
                    best = a_res.copy()
                    best_score = sc
                    print(f"  NEW BEST (pert {trial}): J={sc:.6f}", flush=True)

    # === PHASE 5: Final polish ===
    print("\n=== Phase 5: Final polish ===", flush=True)
    if best is not None and time.monotonic() < deadline:
        # Subgradient
        a_result, score = subgradient_refine(best, best.size, deadline, max_iters=10000)
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  NEW BEST (final subgrad): J={score:.6f}", flush=True)

        # One more L-BFGS-B pass
        if time.monotonic() < deadline:
            a_result, score = lbfgsb_smooth(best, best.size, deadline,
                                            betas=[1000, 10000, 50000])
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  NEW BEST (final smooth): J={score:.6f}", flush=True)

    # Final validation
    if best is None:
        best = np.ones(256) / 256
        best_score = compute_J(best)

    # Scale to ensure evaluate_sequence works well (avoid clipping issues)
    best = np.maximum(best, 0.0)
    best = np.minimum(best, 1000.0)
    if best.sum() < 0.01:
        best = np.ones(256) / 256

    final_seq = [float(x) for x in best]
    final_score = evaluate_sequence(final_seq)
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return final_seq
