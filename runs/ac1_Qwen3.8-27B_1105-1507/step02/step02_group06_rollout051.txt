import time

def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))

def project_to_simplex(f, S):
    """Project f onto {x >= 0, sum(x) = S}."""
    # f should already be >= 0, just need to fix the sum
    # Actually, we need the full projection
    # min ||x - f||^2 subject to x >= 0, sum(x) = S
    
    # Use bisection on the Lagrange multiplier
    n = len(f)
    lo = -max(f) - S  # Large negative
    hi = max(f) + S  # Large positive
    
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.maximum(0, f - mid)
        s = np.sum(x)
        if s > S:
            lo = mid
        else:
            hi = mid
    
    x = np.maximum(0, f - (lo+hi)/2)
    # Adjust for floating point
    s = np.sum(x)
    if s > 0:
        x = x * S / s
    return x

def smooth_gradient(f, beta):
    """Compute gradient of smooth approximation to max(conv(f,f)) / sum(f)^2."""
    n = len(f)
    S = np.sum(f)
    conv = np.convolve(f, f)
    # Normalize conv by S^2 to get scale-invariant weights
    conv_norm = conv / (S * S)
    # Compute weights (softmax with temperature 1/beta)
    conv_shifted = conv_norm - np.max(conv_norm)
    w = np.exp(beta * conv_shifted)
    w /= np.sum(w)
    # Gradient of sum_k w_k * conv_k
    grad = 2.0 * np.correlate(w, f, mode='valid')
    return grad

def optimize_smooth(f, deadline, beta_start=1.0, beta_end=50.0, 
                     lr_start=0.1, verbose_prefix=""):
    """Optimize using smooth subgradient with acceleration."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = np.sum(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    
    # Nesterov momentum
    y = f.copy()
    v = np.zeros(n)
    t = 1.0
    beta = beta_start
    
    it = 0
    while time.monotonic() < deadline:
        # Update beta (increase over time to focus on max)
        progress = min(1.0, it / 5000.0)
        beta = beta_start + (beta_end - beta_start) * progress
        
        # Compute gradient at y
        grad_y = smooth_gradient(y, beta)
        
        # Nesterov update
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        momentum = (t - 1) / t_new
        
        # Step size (decreases over time)
        lr = lr_start / (1 + 0.01 * it)
        
        # Update
        y = np.maximum(0, y - lr * grad_y)
        y = project_to_simplex(y, S)
        
        # Update f with momentum
        f_new = y + momentum * (y - f)
        f_new = project_to_simplex(f_new, S)
        
        t = t_new
        f = f_new
        it += 1
        
        # Check improvement (not every iteration to save time)
        if it % 10 == 0:
            score = compute_score(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
                if it % 100 == 0:
                    print(f"  {verbose_prefix}iter {it}: score={score:.6f} beta={beta:.1f}")
    
    return best_f, best_score

def compute_score(f):
    """Fast score computation."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def project_simplex(f, S):
    """Project f onto {x >= 0, sum(x) = S} using bisection."""
    n = len(f)
    # We need to find lambda such that sum(max(0, f - lambda)) = S
    f_max = np.max(f)
    lo = f_max - S  # If we subtract this, sum becomes at most n*(f_max - (f_max-S)) = n*S... 
    # Actually, let me think more carefully.
    # sum(max(0, f_i - lambda)) is a decreasing function of lambda
    # At lambda = min(f): sum = S_target (the sum of elements > min(f))
    # At lambda = max(f): sum = 0
    # We want sum = S
    
    # Binary search
    lo = float(np.min(f)) - S  # Very negative, sum will be large
    hi = float(np.max(f)) + S  # Very positive, sum will be 0
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.maximum(0.0, f - mid)
        s = np.sum(x)
        if s > S:
            lo = mid
        else:

    x = np.maximum(0.0, f - lam)
    s = np.sum(x)
    if s < 1e-15:
        # Fallback: uniform
        return np.full(n, S / n)
    x *= S / s
    return x


def smooth_gradient(f, beta):
    """
    Compute gradient of soft-max approximation to max(conv(f,f)).
    Uses the identity: grad of sum_k w_k * conv_k = 2 * correlate(w, f, 'valid')
    where w_k = softmax(beta * conv_k / S^2).
    """
    n = len(f)
    S = float(np.sum(f))
    if S < 1e-10:
        return np.zeros(n)
    conv = np.convolve(f, f)
    conv_norm = conv / (S * S)
    # Softmax with temperature
    conv_shifted = conv_norm - np.max(conv_norm)
    w = np.exp(beta * conv_shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # Gradient: grad_m = 2 * sum_k w_k * f_{k-m} for valid k
    # This is 2 * np.correlate(w, f, mode='valid')
    # len(w) = 2n-1, len(f) = n, so 'valid' gives n elements
    grad = 2.0 * np.correlate(w, f, mode='valid')
    return grad


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': 30, 'mip_rel_gap': 0.1})
        if result.success and result.x is not None:
            return result.x
    except Exception:
        pass
    return None


def optimize_nesterov(f_init, deadline, beta_start=2.0, beta_end=100.0,
                       lr_init=0.5, verbose_prefix="", perturb_every=500,
                       perturb_scale=0.01):
    """
    Nesterov-accelerated smooth subgradient method.
    """
    f = f_init.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    
    best_f = f.copy()
    best_score = compute_score(f)
    
    # Nesterov state
    y = f.copy()
    t = 1.0
    it = 0
    rng = np.random.default_rng(42)
    
    last_improve = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        # Adaptive beta
        frac = min(1.0, it / 3000.0)
        beta = beta_start + (beta_end - beta_start) * frac
        
        # Learning rate schedule
        lr = lr_init / (1.0 + 0.005 * it)
        
        # Compute gradient at y
        grad_y = smooth_gradient(y, beta)
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        
        # Step
        y = y - lr * grad_y
        y = project_simplex(y, S)
        
        # Momentum update
        f = y + momentum * (y - f)
        f = project_simplex(f, S)
        t = t_new
        it += 1
        
        # Periodic perturbation to escape local optima
        if it % perturb_every == 0 and no_improve > 100:
            perturbation = rng.normal(0, perturb_scale, size=n)
            f = f + perturbation
            f = project_simplex(f, S)
            y = f.copy()
            no_improve = 0  # Reset counter
        
        # Check score periodically
        if it % 5 == 0:
            score = compute_score(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
                last_improve = it
                if it % 50 == 0:
                    print(f"  {verbose_prefix}iter {it}: score={score:.6f} beta={beta:.1f} lr={lr:.4f}")
        
        no_improve = it - last_improve
        if no_improve > 5000:
            break
    
    return best_f, best_score

def make_initializations(n, rng, parent_seq=None):

    return inits

"""
Optimized search for minimum 2n*max(conv(a,a))/sum(a)^2 using:
- Smooth (log-sum-exp) subgradient with Nesterov acceleration
- Proper simplex projection via bisection
- Adaptive smoothness parameter (beta annealing)
- Periodic LP-based direction refinement for polishing
- Multiple diverse initializations including parent interpolation
- Optional symmetric constraint exploitation
- Coordinate descent final polish
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    """Fast score computation: 2n*max(conv)/sum^2."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def project_simplex(f, S):
    """Project f onto {x >= 0, sum(x) = S} using bisection on Lagrange multiplier."""
    n = len(f)
    if n == 0:
        return f
    f_max = float(np.max(f))
    f_min = float(np.min(f))
    lo = f_min - S
    hi = f_max + S
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.maximum(0.0, f - mid)
        s = float(np.sum(x))
        if s > S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.maximum(0.0, f - lam)
    s = float(np.sum(x))
    if s < 1e-15:
        return np.full(n, S / n)
    x *= S / s
    return x


def smooth_gradient(f, beta):
    """
    Compute gradient of soft-max approximation to max(conv(f,f)).
    grad_m = 2 * sum_k w[k] * f[k-m] = 2 * np.correlate(w, f, 'valid')[m]
    where w = softmax(beta * conv / S^2).
    """
    n = len(f)
    S = float(np.sum(f))
    if S < 1e-10:
        return np.zeros(n)
    conv = np.convolve(f, f)
    conv_norm = conv / (S * S)
    conv_shifted = conv_norm - np.max(conv_norm)
    w = np.exp(beta * conv_shifted)
    w_sum = float(np.sum(w))
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    grad = 2.0 * np.correlate(w, f, mode='valid')
    return grad


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, timeout=60):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': timeout})
        if result.success and result.x is not None:
            return result.x
    except Exception:
        pass
    return None


def optimize_nesterov(f_init, deadline, beta_start=2.0, beta_end=100.0,
                      lr_init=0.5, verbose_prefix="", perturb_every=800,
                      perturb_scale=0.01, score_check_interval=5):
    """
    Nesterov-accelerated smooth subgradient method with adaptive beta.
    """
    f = f_init.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    if S < 1e-10:
        S = np.sqrt(2.0 * n)
        f = f * (S / float(np.sum(f)) + 1e-15)

    best_f = f.copy()
    best_score = compute_score(f)

    y = f.copy()
    t = 1.0
    it = 0
    rng = np.random.default_rng(12345)
    last_improve = 0
    no_improve = 0

    while time.monotonic() < deadline:
        frac = min(1.0, it / 5000.0)
        beta = beta_start + (beta_end - beta_start) * frac
        lr = lr_init / (1.0 + 0.003 * it)

        grad_y = smooth_gradient(y, beta)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new

        y = y - lr * grad_y
        y = project_simplex(y, S)

        f = y + momentum * (y - f)
        f = project_simplex(f, S)
        t = t_new
        it += 1

        if it % perturb_every == 0 and no_improve > 200:
            perturbation = rng.normal(0, perturb_scale, size=n)
            f = f + perturbation
            f = project_simplex(f, S)
            y = f.copy()
            no_improve = 0

        if it % score_check_interval == 0:
            score = compute_score(f)
            if score < best_score - 1e-10:
                best_score = score
                best_f = f.copy()
                last_improve = it
                if it % (score_check_interval * 20) == 0:
                    print(f"  {verbose_prefix}iter {it}: score={score:.6f} beta={beta:.1f} lr={lr:.5f}")

        no_improve = it - last_improve
        if no_improve > 8000:
            break

    return best_f, best_score


def optimize_lp_polish(f_init, deadline, step_init=0.1, max_iters=40, verbose_prefix=""):
    """LP-based polishing for the final refinement phase."""
    f = f_init.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = compute_score(f)
    step = step_init
    it = 0
    no_improve = 0

    while time.monotonic() < deadline and it < max_iters:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))

        t_start = time.monotonic()
        g = solve_lp_direction(f, M, timeout=30)
        lp_time = time.monotonic() - t_start
        if g is None:
            no_improve += 1
            if no_improve > 5:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-15:
            break
        g = g * (S / sum_g)

        best_cand_score = best_score
        best_cand_f = None
        for alpha in [step, step * 0.5, step * 2.0, step * 0.25, step * 4.0]:
            alpha = min(alpha, 1.0)
            if alpha < 0.005:
                continue
            cand = (1.0 - alpha) * f + alpha * g
            cand = np.clip(cand, 0.0, None)
            cand_sum = float(np.sum(cand))
            if cand_sum < 1e-15:
                continue
            cand = cand * (S / cand_sum)
            cand_score = compute_score(cand)
            if cand_score < best_cand_score:
                best_cand_score = cand_score
                best_cand_f = cand
        it += 1

        if best_cand_f is not None:
            f = best_cand_f
            best_f = f.copy()
            best_score = best_cand_score
            no_improve = 0
            step *= 0.92
            print(f"  {verbose_prefix}LP iter {it}: score={best_score:.6f} (lp_time={lp_time:.1f}s)")
        else:
            no_improve += 1
            step *= 0.6
            if step < 0.005 or no_improve > 10:
                break

    return best_f, best_score


def coordinate_descent_polish(f_init, deadline, verbose_prefix=""):
    """
    Cycle through coordinates, doing small gradient-based updates.
    For each coordinate m, compute the directional derivative and take a step.
    """
    f = f_init.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = compute_score(f)
    it = 0

    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        S2 = S * S
        active_mask = conv > 0.9 * np.max(conv)
        active_indices = np.where(active_mask)[0]

        for m in range(n):
            if time.monotonic() > deadline:
                break
            # Compute derivative of max conv w.r.t. f[m]
            # d(conv_k)/d(f_m) = 2*f[k-m] if k-m in [0,n-1]
            # Weighted by active lags
            effect = np.zeros(n)
            for k in active_indices:
                km = k - m
                if 0 <= km < n:
                    effect[m] += 2.0 * f[km]

            if len(active_indices) > 0:
                # Approximate gradient as weighted average over active lags
                w_k = conv[active_indices]
                w_k = w_k / (np.sum(w_k) + 1e-15)
                grad_m = 0.0
                for k in active_indices:
                    km = k - m
                    if 0 <= km < n:
                        grad_m += w_k[list(active_indices).index(k)] * 2.0 * f[km]
            else:
                grad_m = 0.0

            # Step in negative gradient direction
            lr = 0.01 * S / (n + 1)
            f[m] -= lr * grad_m
            f[m] = max(0.0, f[m])

        # Renormalize
        f_sum = float(np.sum(f))
        if f_sum > 1e-15:
            f *= S / f_sum

        it += 1
        if it % 10 == 0:
            score = compute_score(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
                if it % 50 == 0:
                    print(f"  {verbose_prefix}CD iter {it}: score={score:.6f}")

    return best_f, best_score


def make_initializations(n, rng, parent_seq=None):
    """Create diverse initializations for length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    inits.append(('uniform', np.ones(n)))

    if n > 1:
        cos_t = (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0
        inits.append(('cosine', cos_t))

    r = rng.uniform(0.1, 1.5, size=n)
    inits.append(('random', r))

    # Sum of random sinusoids (structured random)
    sig = np.zeros(n)
    for _ in range(8):
        freq = rng.uniform(0.3, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 2.0)
        sig += amp * np.cos(2 * np.pi * freq * idx / n + phase)
    sig = np.abs(sig) + 0.05
    inits.append(('random_sig', sig))

    # Bimodal
    inits.append(('bimodal',
                  np.exp(-0.5 * ((idx - n * 0.25) / (n / 6.0)) ** 2) +
                  np.exp(-0.5 * ((idx - n * 0.75) / (n / 6.0)) ** 2) + 0.05))

    # Triangular
    if n > 1:
        tri = 1.0 - np.abs(2.0 * idx / (n - 1) - 1.0) * 0.8
        inits.append(('triangular', np.maximum(tri, 0.05)))

    # Random symmetric
    half = rng.uniform(0.1, 1.5, size=n // 2 + 1)
    sym = np.concatenate([half, half[-2::-1]])
    if len(sym) > n:
        sym = sym[:n]
    elif len(sym) < n:
        sym = np.concatenate([sym, rng.uniform(0.1, 1.5, size=n - len(sym))])
    inits.append(('random_sym', sym))

    # Parent
    if parent_seq is not None:
        if len(parent_seq) == n:
            inits.append(('parent', parent_seq.copy()))
        else:
            x_old = np.linspace(0, 1, len(parent_seq))
            x_new = np.linspace(0, 1, n)
            inits.append(('parent_interp', np.interp(x_new, x_old, parent_seq)))

    return inits
