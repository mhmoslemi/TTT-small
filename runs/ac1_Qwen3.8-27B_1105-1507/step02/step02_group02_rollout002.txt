Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FFT-accelerated projected gradient with annealed soft-max weights (fast exploration)
2. LP best-response iteration for exact local optimization (polishing)
3. Multi-start perturbation search
4. Smooth log-sum-exp L-BFGS refinement

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def hankel_mv(w, a):
    """Compute H @ a where H[i,j] = 2*w[i+j], using FFT cross-correlation.
    w has length 2n-1, a has length n. Result has length n.
    (H @ a)[i] = sum_j 2*w[i+j]*a[j] for i=0,...,n-1
    """
    n = len(a)
    # We need: q[i] = sum_j a[j] * w[i+j] for i=0,...,n-1
    # This is the cross-correlation of a and w at lags 0,...,n-1
    # Cross-correlation: r[k] = sum_j a[j] * w[j+k]
    # Using FFT: r = ifft(fft(a, N) * conj(fft(w, N)))
    # where N >= n + len(w) - 1 = n + 2n - 1 - 1 = 3n - 2
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    r = np.fft.irfft(A * np.conj(W), N)[:n]
    return 2.0 * r


def proj_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Algorithm from Duchi et al. 2012
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2 directly."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))
