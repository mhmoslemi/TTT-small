def compute_obj(a):

    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def softmax_obj_and_grad(a, T):
    n = len(a)
    conv = np.convolve(a, a)
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    sm_val = max_c + T * np.log(sum_exp)
    idx = np.arange(n)
    M = p[idx[:, None] + idx[None, :]]
    grad = (2.0 / T) * (M @ a)
    return sm_val, grad

    start = time.monotonic()
    multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 0.1, 3.0, 5.0]

    rnd = 0
    while time.monotonic() - start < time_budget and rnd < 50:
        rnd += 1
        improved = False
        for j in range(n):
            if time.monotonic() - start > time_budget:
                break
            original = a[j]
            if original < 1e-15:
                # Try adding mass
                for val in [best_obj * 0.001, best_obj * 0.01, 1.0 / n]:
                    candidate = a.copy()
                    candidate[j] = val
                    s = np.sum(candidate)
                    candidate = candidate / s
                    obj = compute_obj(candidate)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = candidate.copy()
                        a = candidate
                        improved = True
                        break
            else:
                for mult in multipliers:
                    candidate = a.copy()
                    candidate[j] = original * mult
                    s = np.sum(candidate)
                    if s < 1e-10:
                        continue
                    candidate = candidate / s
                    obj = compute_obj(candidate)
                    if obj < best_obj:
                        best_obj = obj
                        best_a = candidate.copy()
                        a = candidate
                        improved = True
                        break
        if not improved:
            # Random small perturbations
            rng = np.random.default_rng(rnd * 12345)
            any_imp = False
            for _ in range(20):
                if time.monotonic() - start > time_budget:
                    break
                j = int(rng.integers(0, n))
                candidate = a.copy()
                candidate[j] *= np.exp(rng.normal(0, 0.08))
                candidate = np.clip(candidate, 0.0, None)
                s = np.sum(candidate)
                if s < 1e-10:
                    continue
                obj = compute_obj(candidate)
                if obj < best_obj:
                    best_obj = obj
                    best_a = candidate.copy()
                    a = candidate
                    improved = True
                    any_imp = True
            if not any_imp:
                break

    return best_a, best_obj


def simulated_annealing(a, time_budget, seed=42):
    """Simulated annealing with adaptive block size."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(seed)
    T_current = 0.05

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / time_budget)
        T_current = 0.05 * (0.01) ** progress

        width = max(1, int(n * rng.uniform(0.02, 0.15)))
        pos = int(rng.integers(0, n - width + 1))
        candidate = a.copy()
        sigma = 0.05 + 0.3 * (1.0 - progress)
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate = candidate / s

        obj = compute_obj(candidate)
        if obj < best_obj:
            best_obj = obj
            best_a = candidate.copy()
            a = candidate
        elif rng.uniform() < np.exp(-(obj - a_obj) / max(T_current, 1e-12)):
            a = candidate
            a_obj = obj

    return best_a, best_obj


def simulated_annealing_v2(a, time_budget, seed=42):
    """Simulated annealing v2 - tracks current obj separately."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    cur_obj = best_obj
    start = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - start < time_budget:
        elapsed = time.monotonic() - start
        progress = min(1.0, elapsed / max(time_budget, 1e-6))
        T_current = 0.05 * (0.005) ** progress

        width = max(1, int(n * rng.uniform(0.02, 0.15)))
        pos = int(rng.integers(0, n - width + 1))
        candidate = a.copy()
        sigma = 0.05 + 0.3 * (1.0 - progress)
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.clip(candidate, 0.0, None)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate = candidate / s

        obj = compute_obj(candidate)
        if obj < best_obj:
            best_obj = obj
            best_a = candidate.copy()
            a = candidate
            cur_obj = obj
        elif rng.uniform() < np.exp(-(obj - cur_obj) / max(T_current, 1e-12)):
            a = candidate
            cur_obj = obj

    return best_a, best_obj


def nesterov_softmax(a, time_budget, T_init=0.01):
    """Nesterov-accelerated gradient descent on soft-max."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    y = a.copy()
    t_val = 1.0
    T = T_init
    lr = 0.01

    while time.monotonic() - start < time_budget:
        sm_val, grad = softmax_obj_and_grad(y, T)
        z = y - lr * grad
        z = np.clip(z, 0.0, None)
        s = np.sum(z)
        if s < 1e-10:
            lr *= 0.5
            continue
        z = z / s

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val ** 2)) / 2.0
        y = z + ((t_val - 1.0) / t_new) * (z - a)
        y = np.clip(y, 0.0, None)
        s = np.sum(y)
        if s > 1e-10:
            y = y / s
        a = z
        t_val = t_new

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            if T > 0.001:
                T = max(0.001, T * 0.999)
        else:
            lr *= 0.98
            if T > 0.001:
                T = max(0.001, T * 0.995)
            if lr < 1e-7:
                break

    return best_a, best_obj


def mixed_pipeline(a_init, time_budget, seed=42):
    """Full mixed-method optimization pipeline."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: Mirror descent (good for simplex constraints)
    md_time = min(15.0, time_budget * 0.25)
    if md_time > 2:
        a, obj = mirror_descent(a, md_time)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"    MD: {best_obj:.6f}")

    # Phase 2: LP alternating
    if time.monotonic() - start < time_budget * 0.6:
        lp_time = min(20.0, time_budget * 0.3)
        if lp_time > 2:
            a, obj = lp_alternating(a, lp_time)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                print(f"    LP: {best_obj:.6f}")

    # Phase 3: Nesterov soft-max
    if time.monotonic() - start < time_budget * 0.8:
        nv_time = min(15.0, time_budget * 0.2)
        if nv_time > 2:
            a, obj = nesterov_softmax(a, nv_time)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                print(f"    Nesterov: {best_obj:.6f}")

    # Phase 4: Coordinate descent
    if time.monotonic() - start < time_budget * 0.95:
        cd_time = min(10.0, time_budget * 0.15)
        if cd_time > 1:
            a, obj = coordinate_descent(a, cd_time)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                print(f"    CD: {best_obj:.6f}")

    # Phase 5: Simulated annealing polish
    if time.monotonic() - start < time_budget - 1:
        sa_time = time_budget - (time.monotonic() - start) - 0.5
        if sa_time > 2:
            a, obj = simulated_annealing_v2(a, sa_time, seed)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                print(f"    SA: {best_obj:.6f}")

    return best_a, best_obj


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 10))
    if choice == 0:
        # Edge-heavy with decay
        a = np.ones(n)
        decay = np.linspace(1.0, 0.3, n)
        a = decay + np.flip(decay)
        a += 0.1
        s = np.sum(a)
        return a / s
    elif choice == 1:
        # Cosine with random phase
        phase = rng.uniform(0, 2 * np.pi)
        freq = rng.uniform(0.05, 0.4)
        a = np.maximum(0.05, np.cos(2 * np.pi * freq * np.arange(n) + phase) + 0.6)
        s = np.sum(a)
        return a / s
    elif choice == 2:
        # Beta-like distribution
        alpha = rng.uniform(0.5, 3.0)
        beta = rng.uniform(0.5, 3.0)
        x = np.linspace(0.01, 0.99, n)
        a = x ** (alpha - 1) * (1 - x) ** (beta - 1)
        s = np.sum(a)
        return a / s
    elif choice == 3:
        # Two-peak
        p1 = int(n * rng.uniform(0.15, 0.35))
        p2 = int(n * rng.uniform(0.65, 0.85))
        w = max(2, n * 0.08)
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - p1) / w) ** 2) + np.exp(-0.5 * ((x - p2) / w) ** 2) + 0.05
        s = np.sum(a)
        return a / s
    elif choice == 4:
        # Periodic with harmonics
        base_period = max(3, int(n / rng.uniform(5, 20)))
        a = 1.0 + 0.3 * np.cos(2 * np.pi * np.arange(n) / base_period)
        a += 0.15 * np.cos(4 * np.pi * np.arange(n) / base_period + rng.uniform(0, 3))
        a = np.maximum(0.01, a)
        s = np.sum(a)
        return a / s
    elif choice == 5:
        # Power-law centered
        center = n / 2.0
        a = 1.0 / (1.0 + 0.3 * np.abs(np.arange(n) - center) / (n * 0.3)) ** 2
        s = np.sum(a)
        return a / s
    elif choice == 6:
        # Smoothed random
        a = rng.random(n)
        kernel_size = max(3, n // 20)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, 'same')
        a = np.maximum(0.01, a)
        s = np.sum(a)
        return a / s
    elif choice == 7:
        # Sigmoid mixture
        x = np.linspace(0, 1, n)
        c1 = rng.uniform(0.2, 0.5)
        c2 = rng.uniform(0.5, 0.8)
        a = 1.0 / (1.0 + np.exp(-20 * (x - c1))) + 1.0 / (1.0 + np.exp(-20 * (c2 - x))) + 0.1
        s = np.sum(a)
        return a / s
    elif choice == 8:
        # Alternating with envelope
        envelope = np.linspace(0.5, 1.0, n)
        a = envelope * (1.0 + 0.4 * np.cos(2 * np.pi * np.arange(n) / max(4, n // 8)))
        a = np.maximum(0.01, a)
        s = np.sum(a)
        return a / s
    else:
        # Log-normal
        a = np.exp(rng.normal(0, 0.4, n))
        s = np.sum(a)
        return a / s


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-method optimization: mirror descent, LP alternating, Nesterov soft-max,
    coordinate descent, and simulated annealing. Focuses on n=250-400 range with
    diverse initializations and the parent sequence.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf

    # Try parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Optimize parent aggressively
    if parent is not None and parent.size > 0:
        p_time = min(120.0, float(budget_s) * 0.25)
        print(f"Optimizing parent with {p_time:.0f}s budget...")
        p_opt, p_obj = mixed_pipeline(parent.copy(), p_time, seed=seed)
        if p_obj < best_score:
            best_score = p_obj
            best_seq = p_opt.copy()
            print(f"Parent optimized: obj={p_obj:.6f} ** NEW BEST **")
        else:
            print(f"Parent optimized: obj={p_obj:.6f}")

    # Also try a second pass on the best parent result with different seed
    if parent is not None and parent.size > 0 and time.monotonic() < deadline - 60:
        second_time = min(60.0, float(budget_s) * 0.1)
        p_opt2, p_obj2 = mixed_pipeline(best_seq.copy(), second_time, seed=seed + 999)
        if p_obj2 < best_score:
            best_score = p_obj2
            best_seq = p_opt2.copy()
            print(f"Parent 2nd pass: obj={p_obj2:.6f} ** NEW BEST **")

    # Main search over n values - focus on 200-500
    n_values = [200, 250, 300, 350, 400, 500]

    for n in n_values:
        if time.monotonic() > deadline - 40:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        num_remaining = len([x for x in n_values if x >= n])
        time_per_n = remaining / max(1, num_remaining)
        time_per_n = min(time_per_n, 60.0)

        num_restarts = max(2, min(4, int(time_per_n / 15)))
        time_per_restart = time_per_n / num_restarts

        print(f"\nN={n}: remaining={remaining:.0f}s, {num_restarts} restarts")

        for r in range(num_restarts):
            if time.monotonic() > deadline - 40:
                break
            a_init = structured_init(n, rng)
            a_opt, obj_opt = mixed_pipeline(a_init, time_per_restart, seed=seed + n * 100 + r)

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} r={r}: obj={obj_opt:.6f} ** NEW BEST **")
            else:
                print(f"  n={n} r={r}: obj={obj_opt:.6f}")

    # Final intensive refinement on best found
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: {remaining:.0f}s")

        # Mirror descent
        md_time = min(remaining * 0.25, 30.0)
        if md_time > 3:
            a_ref, obj_ref = mirror_descent(best_seq.copy(), md_time)
            if obj_ref < best_score:
                best_score = obj_ref
                best_seq = a_ref.copy()
                print(f"  Final MD: obj={best_score:.6f}")

        # LP
        if time.monotonic() < deadline - 10:
            lp_time = min(remaining * 0.2, 20.0)
            if lp_time > 3:
                a_ref, obj_ref = lp_alternating(best_seq.copy(), lp_time)
                if obj_ref < best_score:
                    best_score = obj_ref
                    best_seq = a_ref.copy()
                    print(f"  Final LP: obj={best_score:.6f}")

        # Nesterov
        if time.monotonic() < deadline - 8:
            nv_time = min(remaining * 0.15, 15.0)
            if nv_time > 2:
                a_ref, obj_ref = nesterov_softmax(best_seq.copy(), nv_time)
                if obj_ref < best_score:
                    best_score = obj_ref
                    best_seq = a_ref.copy()
                    print(f"  Final Nesterov: obj={best_score:.6f}")

        # Coordinate descent
        if time.monotonic() < deadline - 5:
            cd_time = min(remaining * 0.2, 15.0)
            if cd_time > 2:
                a_ref, obj_ref = coordinate_descent(best_seq.copy(), cd_time)
                if obj_ref < best_score:
                    best_score = obj_ref
                    best_seq = a_ref.copy()
                    print(f"  Final CD: obj={best_score:.6f}")

        # SA polish
        if time.monotonic() < deadline - 2:
            sa_time = deadline - time.monotonic() - 1.0
            if sa_time > 2:
                a_ref, obj_ref = simulated_annealing_v2(best_seq.copy(), sa_time, seed=seed + 777)
                if obj_ref < best_score:
                    best_score = obj_ref
                    best_seq = a_ref.copy()
                    print(f"  Final SA: obj={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
