import time


def fast_conv(a):
    """Full linear self-convolution via FFT for large n, direct for small n."""
    n = len(a)
    if n <= 512:
        return np.convolve(a, a, mode='full')
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def build_conv_matrix(a):
    """Build matrix A where A @ x = convolve(a, x) (full)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def lp_iteration(a, max_iters, deadline, step_size=0.4):
    """LP best-response with fixed step size and backtracking."""
    best_local = a.copy()
    best_score = compute_score(a)
    a = np.clip(a, 0.0001, None)
    a = a / a.sum()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()

        # Try full step first, then backtrack
        accepted = False
        for t in [step_size, step_size * 0.5, step_size * 0.25, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            score = compute_score(a_new)
            if score < best_score:
                best_score = score
                best_local = a_new.copy()
                a = a_new
                no_improve = 0
                accepted = True
                break
        if not accepted:
            no_improve += 1
            if no_improve > 30:
                break
        # Gradually reduce step size
        if it % 200 == 199:
            step_size = max(0.1, step_size * 0.85)

    return best_local, best_score


def smooth_objective_and_grad(x, beta, n):
    """Compute smoothed objective and its gradient for normalized x (sum=1)."""
    x_pos = np.maximum(x, 1e-12)
    s = x_pos.sum()
    xn = x_pos / s
    b = fast_conv(xn)  # length 2n-1
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()

    # Objective: 2n * (b_max + log(sum(exp))/beta)
    lse = b_max + np.log(exp_shifted.sum()) / beta
    obj = 2.0 * n * lse

    # Gradient: 2n * cross_correlation(p, xn) projected to sum=0
    # cross_corr(j) = sum_k p[k] * xn[k-j] for k-j in [0,n-1]
    a_flip = xn[::-1]
    c = np.convolve(p, a_flip, mode='full')
    cross = c[n - 1:n - 1 + n]  # cross-correlation at lags 0..n-1
    grad = 2.0 * n * cross

    # Project gradient to simplex tangent space (sum of grad = 0)
    grad = grad - grad.mean()

    return obj, grad


def smooth_refine(a, deadline, max_total_iters=600):
    """Refine using log-sum-exp with beta annealing and L-BFGS-B."""
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    best = np.clip(best, 1e-6, None)
    best = best / best.sum()

    betas = [50, 200, 1000, 5000, 20000]
    iters_per_beta = max(30, max_total_iters // len(betas))

    for beta in betas:
        if time.monotonic() > deadline:
            break
        x0 = best.copy()

        def obj_func(x):
            o, _ = smooth_objective_and_grad(x, beta, n)
            return o

        def grad_func(x):
            _, g = smooth_objective_and_grad(x, beta, n)
            return g

        try:
            res = minimize(obj_func, x0, jac=grad_func,
                           method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': iters_per_beta,
                                    'maxfun': iters_per_beta * 10,
                                    'ftol': 1e-18, 'gtol': 1e-14})
            if hasattr(res, 'x') and res.x is not None:
                rx = np.clip(res.x, 0, None)
                if rx.sum() > 0.01:
                    score = compute_score(rx)
                    if score < best_score:
                        best = rx.copy()
                        best_score = score
        except Exception:
            pass

    return best, best_score


def multiplicative_update(a, p, eta):
    """Exponential/multiplicative update: a_new = a * exp(-eta * T^T p)."""
    n = len(a)
    # T^T @ p[j] = sum_k p[k] * a[k-j] (cross-correlation)
    a_flip = a[::-1]
    c = np.convolve(p, a_flip, mode='full')
    cross = c[n - 1:n - 1 + n]
    log_update = -eta * cross
    log_update -= log_update.mean()  # keep sum roughly constant
    a_new = a * np.exp(log_update)
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s


def mul_update_iteration(a, max_iters, deadline):
    """Iterative multiplicative update method."""
    best_local = a.copy()
    best_score = compute_score(a)
    a = np.clip(a, 0.001, None)
    a = a / a.sum()
    n = len(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        # Softmax with high temperature to focus on peaks
        beta = 100.0 + 50.0 * it
        shifted = beta * (b - b_max)
        shifted = np.clip(shifted, -500, 0)
        p = np.exp(shifted)
        p = p / p.sum()

        eta = 0.1 / (1.0 + 0.01 * it)
        a_new = multiplicative_update(a, p, eta)
        a_new = np.clip(a_new, 0, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_local = a_new.copy()
        a = a_new

    return best_local, best_score


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n)

    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        resampled = np.maximum(np.interp(t, t_old, parent), 0)
        if resampled.sum() > 0.01:
            inits.append(resampled / resampled.sum())

    inits.append(np.ones(n) / n)
    for harm in [1, 2, 3, 4]:
        c = 1.0 + 0.3 * np.cos(2 * np.pi * harm * t)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())
    # Quadratic
    c = 1.0 + 0.5 * (4.0 * (t - 0.5) ** 2 - 0.5)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())
    # Triangular
    c = 1.0 - np.abs(t - 0.5)
    c = np.maximum(c, 0.05)
    inits.append(c / c.sum())
    # Bimodal
    c = 1.0 + 0.4 * np.cos(2 * np.pi * t) ** 2
    inits.append(c / c.sum())
    # Random
    inits.append(rng.uniform(0.2, 1.0, n) / n)
    # Power law
    c = (t + 0.1) ** rng.uniform(-1, 1)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search function combining LP, smooth, and multiplicative methods."""
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 10.0

    best = None
    best_score = float('inf')

    # Try parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.clip(p, 0.0, 1000.0)
            sc = compute_score(p)
            if sc < best_score:
                best = p.copy()
                best_score = sc
                print(f"Parent: J={sc:.6f} (n={p.size})", flush=True)
    except (NameError, AttributeError):
        pass

    # === PHASE 1: LP best-response ===
    t0 = time.monotonic()
    print(f"\n=== Phase 1: LP Best-Response ===", flush=True)
    lp_budget = total_budget * 0.40
    lp_deadline = time.monotonic() + lp_budget

    if best is not None:
        parent_len = best.size
    else:
        parent_len = 512

    lengths = [parent_len]
    for l in [256, 384, 512, 768, 1024, 2048]:
        if l not in lengths:
            lengths.append(l)
    # Sort: parent first, then by proximity to parent
    lengths.sort(key=lambda l: abs(l - parent_len))

    for n in lengths:
        if time.monotonic() > lp_deadline:
            break
        # Time estimate per LP iteration
        if n <= 512:
            est_time = 0.02
        elif n <= 1024:
            est_time = 0.05
        else:
            est_time = 0.15

        remaining = lp_deadline - time.monotonic()
        if remaining < est_time * 100:
            continue

        print(f"  LP n={n}, est_time/iter={est_time*1000:.0f}ms", flush=True)
        inits = generate_initializations(n, rng, best)

        # Limit initializations based on time
        max_inits = min(len(inits), max(2, int(remaining / (est_time * 200))))

        for i, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > lp_deadline:
                break
            a0 = np.clip(a0, 0.001, None)
            a0 = a0 / a0.sum()

            remaining = lp_deadline - time.monotonic()
            max_iters = max(50, int(remaining * 0.7 / est_time))
            max_iters = min(max_iters, 8000)

            a_result, score = lp_iteration(a0, max_iters, lp_deadline, step_size=0.4)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  NEW BEST (LP): J={score:.6f} (n={n}, init={i})", flush=True)

    print(f"  Phase 1 done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === PHASE 2: Smooth refinement ===
    t0 = time.monotonic()
    print(f"\n=== Phase 2: Smooth Refinement ===", flush=True)
    smooth_budget = total_budget * 0.25
    smooth_deadline = time.monotonic() + smooth_budget

    if best is not None:
        n = best.size
        iters = max(200, int(smooth_budget * 30))
        a_result, score = smooth_refine(best, smooth_deadline, max_total_iters=iters)
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  NEW BEST (smooth): J={score:.6f} (n={n})", flush=True)

        # Try at neighboring lengths
        for n2 in [max(128, n // 2), min(2048, n * 2)]:
            if time.monotonic() > smooth_deadline:
                break
            if n2 == n:
                continue
            t_old = np.linspace(0, 1, n)
            t_new = np.linspace(0, 1, n2)
            a2 = np.maximum(np.interp(t_new, t_old, best), 0)
            if a2.sum() > 0.01:
                a2 = a2 / a2.sum()
                a_result2, score2 = smooth_refine(a2, smooth_deadline, max_total_iters=300)
                if score2 < best_score:
                    best = a_result2.copy()
                    best_score = score2
                    print(f"  NEW BEST (smooth n={n2}): J={score2:.6f}", flush=True)

    print(f"  Phase 2 done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === PHASE 3: Multiplicative updates + perturbation ===
    t0 = time.monotonic()
    print(f"\n=== Phase 3: Perturbation + Mul Updates ===", flush=True)
    if best is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            n = best.size
            n_tries = int(remaining / 10)
            n_tries = max(5, min(50, n_tries))

            for trial in range(n_tries):
                if time.monotonic() > deadline:
                    break
                # Vary perturbation scale
                scale = 0.05 + 0.1 * (trial % 3)
                noise = rng.normal(0, scale, n)
                a_pert = best + noise
                a_pert = np.clip(a_pert, 0, None)
                if a_pert.sum() < 0.01:
                    continue
                a_pert = a_pert / a_pert.sum()

                remaining = deadline - time.monotonic()
                # Quick LP or mul update
                if remaining > 15:
                    a_result, score = lp_iteration(a_pert, 500, deadline, step_size=0.3)
                else:
                    a_result, score = mul_update_iteration(a_pert, 200, deadline)

                if score < best_score:
                    best = a_result.copy()
                    best_score = score
                    print(f"  NEW BEST (pert {trial}): J={score:.6f}", flush=True)

    print(f"  Phase 3 done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === PHASE 4: Final polish ===
    t0 = time.monotonic()
    print(f"\n=== Phase 4: Final Polish ===", flush=True)
    if best is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            n = best.size
            # Final LP pass
            if remaining > 60:
                est_time = 0.02 if n <= 512 else 0.05 if n <= 1024 else 0.1
                max_iters = max(100, int(remaining * 0.5 / est_time))
                a_result, score = lp_iteration(best, max_iters, time.monotonic() + remaining * 0.4, step_size=0.2)
                if score < best_score:
                    best = a_result.copy()
                    best_score = score
                    print(f"  NEW BEST (polish LP): J={score:.6f}", flush=True)

            # Final smooth pass
            if time.monotonic() < deadline:
                a_result, score = smooth_refine(best, deadline, max_total_iters=400)
                if score < best_score:
                    best = a_result.copy()
                    best_score = score
                    print(f"  NEW BEST (polish smooth): J={score:.6f}", flush=True)

    print(f"  Phase 4 done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # Final validation
    if best is None:
        best = np.ones(256) / 256

    final_score = float(evaluate_sequence([float(x) for x in best]))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
