import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(n, a):
    """Build sparse constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_best_response(a, M, method='highs'):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method=method)
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def subgradient_step(a, step_size):
    """Subgradient descent step for minimizing max(conv(a,a))."""
    b = fast_conv(a)
    M = b.max()
    # Subgradient: 2 * sum of shifts of a corresponding to argmax positions
    active = (b >= M - 1e-12)
    # For each active k, the gradient w.r.t. a[i] is 2*a[k-i] (if valid)
    grad = np.zeros_like(a)
    for k in np.where(active)[0]:
        for i in range(len(a)):
            ki = k - i
            if 0 <= ki < len(a):
                grad[i] += 2.0 * a[ki]
    # Normalize gradient
    gnorm = np.linalg.norm(grad)
    if gnorm > 1e-15:
        grad /= gnorm
    return grad, M

def subgradient_descent(a, max_iters, deadline, tracker):
    """Subgradient descent with diminishing step size."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    current_M = fast_conv(a).max()
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = b.max()
        active = (b >= M - 1e-12 * M)
        active_indices = np.where(active)[0]
        
        # Compute subgradient
        grad = np.zeros_like(a)
        for k in active_indices:
            # d/d a[i] (a*a)[k] = 2*a[k-i] if 0 <= k-i < n
            valid_i = np.maximum(0, k - len(a) + 1)
            valid_j = np.minimum(len(a) - 1, k)
            for i in range(valid_i, valid_j + 1):
                grad[i] += 2.0 * a[k - i]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Polyak step: t = (f(a) - f*) / ||g||^2, but we don't know f*
        # Use diminishing step: t = c / sqrt(it+1)
        t = 0.5 / (np.sqrt(it + 1))
        
        a_new = a - t * grad / gnorm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        
        new_M = fast_conv(a_new).max()
        if new_M < M - 1e-15:
            a = a_new
            current_M = new_M
        # else: keep old a (backtrack)
    
    tracker.update(a)
    return a

2. Progressive length extension from shorter to longer sequences
3. Simulated annealing restarts for escaping local optima

import numpy as np
from scipy.optimize import linprog, minimize

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(n, a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if 0<=k-j<n."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0. Returns g or None."""
    n = len(a)
    A = build_A_sparse(n, a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

    return inits


def _lbb_objective(x):
    """Objective for L-BFGS-B: minimize max(conv(x,x)) with x>=0, sum=1."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    return float(b.max())


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search:
    Phase 1: LP mirror descent on multiple lengths (progressive)
    Phase 2: Subgradient polish on best
    Phase 3: Extended LP on best
    Phase 4: SA restarts
    Phase 5: Length extension attempts
    Phase 6: L-BFGS-B final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 20.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128)
        tracker.update(parent)

    # === Phase 1: LP Mirror Descent across lengths ===
    # Start with parent length and nearby, then expand
    parent_n = parent.size
    # Build length list: include parent_n and neighbors, then larger
    base_lengths = sorted(set([
        max(64, parent_n), parent_n,
        parent_n * 2, parent_n * 3,
        128, 192, 256, 320, 384, 512, 640, 768, 1024
    ]))
    # Filter to reasonable range and sort
    base_lengths = [n for n in base_lengths if 64 <= n <= 2048]
    base_lengths = sorted(set(base_lengths))

    # Allocate time: shorter sequences get more time (LP is faster)
    total_time = deadline - time.monotonic()
    # Weight by 1/n (shorter = more time)
    weights = [1.0 / n for n in base_lengths]
    total_w = sum(weights)

    current_best = parent.copy()

    for n in base_lengths:
        time_budget = 0.6 * total_time * (1.0 / n) / total_w  # 60% for Phase 1
        phase_deadline = time.monotonic() + max(30, time_budget)
        if phase_deadline > deadline:
            phase_deadline = deadline - 10

        if time.monotonic() > deadline - 120:
            break

        t0 = time.monotonic()
        print(f"\n{'='*40} Phase1 LP n={n} budget={max(0, phase_deadline - t0):.0f}s {'='*40}", flush=True)

        inits = generate_initializations(n, rng, current_best)
        n_inits = len(inits)

        # Estimate LP time for this n
        est_lp_time = 0.01 * (n / 128) ** 2  # Rough estimate in seconds
        max_iters = max(20, min(500, int(max(30, phase_deadline - time.monotonic()) / (est_lp_time * n_inits * 0.5))))

        best_a_for_n = current_best if current_best.size == n else resample_seq(current_best, n)
        best_J_for_n = compute_J(best_a_for_n)

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase_deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-6)
            a0 /= a0.sum()

            iters = max(15, min(max_iters, int((phase_deadline - time.monotonic()) / max(0.01, est_lp_time))))
            a = run_lp_mirror_descent(a0, iters, phase_deadline, tracker)

            J = compute_J(a)
            if J < best_J_for_n:
                best_J_for_n = J
                best_a_for_n = a.copy()

        if best_a_for_n is not None and best_J_for_n < compute_J(current_best):
            current_best = best_a_for_n.copy()

        tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s, local_best={best_J_for_n:.6f}")

    # === Phase 2: Subgradient polish ===
    if tracker.best is not None and time.monotonic() < deadline - 100:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*40} Phase2 Subgradient n={n} {'='*40}", flush=True)
        a = subgradient_polish(tracker.best, 200, min(deadline - 100, time.monotonic() + 120))
        tracker.update(a)
        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Extended LP on current best ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*40} Phase3 LP Extended n={n} {'='*40}", flush=True)
        a = run_lp_mirror_descent(tracker.best, 1000, deadline - 80, tracker)
        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: SA Restarts ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*40} Phase4 SA Restarts n={n} {'='*40}", flush=True)

        temperature = 0.08
        n_restarts = 50
        est_lp_time = 0.01 * (n / 128) ** 2

        for r in range(n_restarts):
            if time.monotonic() > deadline - 60:
                break
            a = tracker.best.copy().astype(np.float64)
            noise_scale = temperature * (1.0 + 0.5 * np.sin(r * 0.3))
            a = a + rng.normal(0, noise_scale * 0.05, n)
            a = np.maximum(a, 0.001)
            a /= a.sum()

            iters = max(10, min(60, int((deadline - 60 - time.monotonic()) / (est_lp_time * (n_restarts - r)))))
            a = run_lp_mirror_descent(a, iters, deadline - 60, tracker)
            temperature *= 0.96

        tracker.report(f"Phase4 SA: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Try extending to longer length ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        base_n = tracker.best.size
        print(f"\n{'='*40} Phase5 Extension base_n={base_n} {'='*40}", flush=True)

        for mult in [2, 3]:
            new_n = base_n * mult
            if new_n > 4096 or time.monotonic() > deadline - 60:
                continue
            a_ext = resample_seq(tracker.best, new_n)
            a_ext = np.maximum(a_ext, 1e-6)
            a_ext /= a_ext.sum()

            est_lp_time = 0.01 * (new_n / 128) ** 2
            iters = max(20, min(200, int((deadline - 60 - time.monotonic()) / max(0.01, est_lp_time))))
            a = run_lp_mirror_descent(a_ext, iters, deadline - 60, tracker)
            tracker.update(a)

        tracker.report(f"Phase5 Extension: {time.monotonic()-t0:.1f}s")

    # === Phase 6: L-BFGS-B final polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*40} Phase6 L-BFGS-B n={n} {'='*40}", flush=True)

        bounds_list = [(0.0, None)] * n
        try:
            res = minimize(_lbb_objective, tracker.best.astype(np.float64),
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': 500, 'max_fun': 100000, 'ftol': 1e-16})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)

        tracker.report(f"Phase6 L-BFGS-B: {time.monotonic()-t0:.1f}s")

    # Final: one more LP pass if time remains
    if tracker.best is not None and time.monotonic() < deadline - 5:
        print(f"\n{'='*40} Final LP pass {'='*40}", flush=True)
        a = run_lp_mirror_descent(tracker.best, 200, deadline, tracker)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*40} FINAL {'='*40}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
