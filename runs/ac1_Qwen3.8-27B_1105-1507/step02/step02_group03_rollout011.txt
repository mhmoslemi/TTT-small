"""
SCP-based optimization with LP direction finding, subgradient momentum,
spectral initialization, and adaptive peak weighting for minimizing
2*n*max(convolve(f,f))/(sum(f))^2 over non-negative sequences.

Key innovations over previous approach:
1. Sequential Convex Programming (SCP): linearize the full conv(g,g) and
   solve a clean LP to get the steepest descent direction for the peak.
2. Adaptive peak weighting: weight taller peaks more in the subgradient.
3. Multi-scale spectral initialization with triangular window correction.
4. Aggressive global moves with diversity-preserving restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute score for sequence f (no normalization needed, scale-invariant)."""
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (s * s)


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def compute_conv_and_grad(f):
    """Compute convolution and gradient of each conv[k] w.r.t. f."""
    n = len(f)
    conv = np.convolve(f, f)
    return conv


def subgradient_max_conv(f, conv, n_active=None, weight_power=2.0, eps=1e-10):
    """
    Compute weighted subgradient of max(conv) w.r.t. f.
    Peaks are weighted by (conv[k]/M)^weight_power for emphasis on tallest peaks.
    """
    n = len(f)
    M = conv.max()
    if M < eps:
        return np.zeros(n)

    # Find peaks (within 1% of max for robustness)
    threshold = M * (1.0 - 1e-4)
    active = np.where(conv >= threshold)[0]

    if n_active is not None and len(active) > n_active:
        # Keep only the tallest n_active peaks
        order = np.argsort(conv[active])[::-1][:n_active]
        active = active[order]

    # Compute weighted subgradient
    sgrad = np.zeros(n)
    total_weight = 0.0
    for k in active:
        w = (conv[k] / M) ** weight_power
        total_weight += w
        # grad of conv[k] w.r.t. f[j] = f[k-j] if valid (times 1 or 2)
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            idx = k - j
            if 0 <= idx < n:
                if j == idx:
                    sgrad[j] += w * f[idx]
                else:
                    sgrad[j] += w * f[idx]

    if total_weight > 0:
        sgrad /= total_weight
    return sgrad


def scp_lp_direction(f, conv):
    """
    Solve the SCP LP:
    min t subject to grad_k . g - t <= conv_f[k] for all k
                      g >= 0, sum(g) = 1
    Returns g* (the LP solution) or None if infeasible.
    """
    n = len(f)
    n_conv = 2 * n - 1
    M = conv.max()

    # Build constraint matrix: for each k, grad_k . g - t <= conv_f[k]
    # Variables: [g_0, ..., g_{n-1}, t] (total n+1 variables)
    # Constraint k: sum_j grad_k[j] * g_j - 1 * t <= conv_f[k]

    A_ub = np.zeros((n_conv, n + 1))
    b_ub = np.zeros(n_conv)

    for k in range(n_conv):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            idx = k - j
            if 0 <= idx < n:
                if j == idx:
                    A_ub[k, j] = f[idx]  # factor of 1 when j == k-j
                else:
                    A_ub[k, j] = f[idx]  # factor of 1 (the other term adds to grad_k[idx])
        # Actually, the gradient of conv[k] w.r.t. f[j] is:
        # f[k-j] if 0<=k-j<n, counting both i=j and i=k-j terms
        # So it's f[k-j] * (1 + (j != k-j and 0<=k-j<n))
        # But we already handled this above. Let me redo properly.

    # Redo gradient computation properly
    A_ub = np.zeros((n_conv, n + 1))
    b_ub = np.zeros(n_conv)

    for k in range(n_conv):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                if j == idx:
                    A_ub[k, j] = f[idx]
                else:
                    A_ub[k, j] = f[idx]  # will be added to when we hit i=idx
        # We need to be more careful. Let's compute grad directly.
        # grad_k[j] = d(conv[k])/d(f[j])
        # conv[k] = sum_{i valid} f[i] * f[k-i]
        # d/d(f[j]) = f[k-j] (if valid, from i=j) + f[k-j] (if valid, from i=k-j, i.e., k-i=j means i=k-j)
        # So grad_k[j] = f[k-j] + f[k-j] if j != k-j, else f[k-j]
        #                   (if both valid)

    # Let me just compute it cleanly
    A_ub = np.zeros((n_conv, n + 1))
    b_ub = np.zeros(n_conv)

    for k in range(n_conv):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                # contribution from term f[j]*f[idx] when i=j
                # contribution from term f[idx]*f[j] when i=idx (i.e., k-i=j means i=k-j=idx)
                if j == idx:
                    A_ub[k, j] = f[idx]
                else:
                    A_ub[k, j] = 2.0 * f[idx]
        A_ub[k, n] = -1.0  # coefficient of t
        b_ub[k] = conv[k]

    # Objective: minimize t => c = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    # Bounds: g_j >= 0, t unbounded below (but effectively bounded by constraints)
    bounds = [(0, None)] * n + [(None, None)]

    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success and res.x is not None:
            g = np.clip(res.x[:n], 0, None)
            s = g.sum()
            if s < 1e-10:
                return None
            return g / s
    except Exception:
        pass
    return None


def fast_subgradient_direction(f, conv, weight_power=3.0):
    """
    Faster direction: use weighted subgradient and project to simplex.
    Find g = argmin ||g - (f - eta*sgrad)||_2 s.t. g >= 0, sum(g) = 1
    """
    n = len(f)
    sgrad = subgradient_max_conv(f, conv, weight_power=weight_power)

    #Simplex projection: find g that minimizes ||g - p||^2 s.t. g >= 0, sum(g) = 1
    #where p = f - eta * sgrad
    M = conv.max()
    if M < 1e-15:
        return f.copy()

    #Adaptive step size
    eta = 0.3 * n / (M + 1e-10) * f.sum() ** 2 / n
    eta = min(eta, 1.0)

    p = f - eta * sgrad

    # Project p onto simplex {g >= 0, sum(g)=1}
    # Using the standard algorithm
    u = np.sort(p)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    g = np.maximum(p - theta, 0)
    s = g.sum()
    if s > 1e-10:
        g = g / s
    else:
        g = f.copy()
    return g


def momentum_subgradient(f, conv, n_iter=200, momentum=0.9, rng=None):
    """Subgradient descent with momentum on the simplex."""
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    v = np.zeros(n)
    best_f = f.copy()
    best_score = compute_score(f)
    base_lr = 1.0

    for it in range(n_iter):
        lr = base_lr / (1.0 + 0.01 * it)
        conv = np.convolve(f, f)
        sgrad = subgradient_max_conv(f, conv, weight_power=2.0)

        #Momentum update
        v = momentum * v - lr * sgrad

        #Project to simplex
        p = f + v
        u = np.sort(p)[::-1]
        cssv = np.cumsum(u)
        rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
        if len(rho) == 0:
            rho_arr = np.array([0])
        else:
            rho_arr = rho[-1:]
        rho_val = rho_arr[0]
        theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
        f_new = np.maximum(p - theta, 0)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s

        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = score
            f = f_new
        else:
            #If no improvement, dampen momentum
            v *= 0.5

    return best_f, best_score


def spectral_flat_triangular(f, n_iter=200, rng=None):
    """
    Spectral flattening with triangular window correction.
    For linear convolution, the "natural" spectrum has a triangular envelope.
    We target a flat spectrum after dividing by the triangular window.
    """
    if rng is None:
        rng = np.random.default_rng(0)

    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()

    #Triangular window in frequency domain (from linear convolution structure)
    freqs = np.fft.rfftfreq(n_fft) * n_fft
    #The triangular window is 1 at DC, decreasing linearly
    tri_window = np.ones(len(freqs))
    for i, fr in enumerate(freqs[1:]):
        tri_window[i + 1] = max(0.1, 1.0 - abs(fr) / (n_fft / 2.0))

    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)

        # Target: mag / tri_window should be flat
        corrected = mag / tri_window
        target_mag = np.full_like(mag, corrected.mean())
        target_mag[0] = mag[0]  # Keep DC fixed

        desired_mag = target_mag * tri_window
        desired_mag[0] = mag[0]

        F_new = desired_mag * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s

        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break
        f = f_new

    return best_f, best_score


def diverse_init(n, rng):

    elif r < 0.4:

    elif r < 0.6:
        #Product of sinusoids
        freq1 = rng.uniform(1, 4)

    elif r < 0.8:

        f = np.exp(-rate * x / n) + 0.1
    else:
        #Random smooth (low-frequency noise)

        f = np.zeros(n)

    f = normalize(f)
    #Small noise to break symmetry
    f = f * (1 + 0.03 * rng.normal(0, 1, n))

                print(f"Polish improved! New best: {best_score_overall:.6f}")

    #Final verification
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal: score={final_score:.6f}, n={len(final_list)}")
        return final_list

    #Fallback
    print("Fallback: uniform sequence")
    return [1.0] * 256
