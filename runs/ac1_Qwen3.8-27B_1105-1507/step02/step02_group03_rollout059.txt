"""
Improved multi-strategy optimization for minimizing peak autocorrelation of non-negative sequences.
Uses enhanced spectral flattening, improved pressure updates, LP refinement, and diverse search.

import numpy as np
from scipy.optimize import linprog
import time

def compute_score(f):
    """Compute score = 2*n*max(conv) / (sum)^2 for sequence f."""
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (s * s)

def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s

def spectral_flat_v2(f, n_iter=300, rng=None, alpha=0.5):
    """
    Improved spectral flattening with soft target and progressive refinement.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()

    best_f = f.copy()
    best_score = compute_score(f)
    stall_count = 0

    for it in range(n_iter):
        # Progressive alpha: start aggressive, become gentle
        alpha_eff = alpha * (1.0 / (1.0 + 0.02 * it))
        
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Target: geometric mean of non-DC magnitudes
        # This preserves total energy better than arithmetic mean
        if len(mag) > 1:
            log_mags = np.log(np.maximum(mag[1:], 1e-15))
            target_log = np.mean(log_mags)
            target_mag = np.exp(target_log)
        else:
            target_mag = mag[0] / np.sqrt(n)
        
        # Soft update: move towards target
        mag_new = mag.copy()
        mag_new[1:] = (1 - alpha_eff) * mag[1:] + alpha_eff * target_mag
        mag_new[0] = mag[0]  # DC fixed (sum = 1)
        
        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        
        # Enforce non-negativity with soft clipping
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        
        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall_count = 0
        else:
            # Accept with some probability (like simulated annealing)
            if rng.random() < 0.1:
                f = f_new
            stall_count += 1
            if stall_count > 150:
                break

    return best_f, best_score

def pressure_update_v2(f, n_iter=200, eta=0.15, rng=None, top_k=5):
    """
    Improved pressure update considering multiple near-max positions.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    stall_count = 0

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Find all positions where conv is within 90% of max
        threshold = 0.9 * M
        near_max_positions = np.where(conv >= threshold)[0]
        
        # Compute pressure using weighted sum over near-max positions
        pressure = np.zeros(n)
        weights = np.exp(10 * (conv[near_max_positions] - M) / M)  # softmax-like
        weights /= weights.sum()
        
        for j in near_max_positions:
            for i in range(n):
                idx = j - i
                if 0 <= idx < n:
                    pressure[i] += weights[np.where(near_max_positions == j)[0][0]] * f[idx]
        
        if pressure.max() > 0:
            pressure = pressure / pressure.max()
        
        eta_eff = eta * (1.0 / (1.0 + 0.005 * it))
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)
        
        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 80:

                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    stall_count = 0
                else:
                    break

    return best_f, best_score

def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A

def lp_refine_v2(f, time_budget, t0=0.5, rng=None, lp_time_limit=2):
    """LP-based refinement with adaptive step size and multiple LP targets."""
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)

    best_f = f.copy()
    best_score = compute_score(f)
    t = t0
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()

        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        # Use slightly tighter bound to encourage movement
        b_ub = np.full(2 * n - 1, M * 0.999)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': lp_time_limit})
        except Exception:
            break

        if not res.success or res.x is None:
            # Try with looser bound
            b_ub = np.full(2 * n - 1, M)
            try:
                res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                              method='highs', options={'presolve': True, 'time_limit': lp_time_limit})
            except Exception:
                break
            if not res.success or res.x is None:
                break

        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break

        g = b / b_sum

        improved = False
        for step in [min(1.0, t * 1.5), t, t * 0.5, t * 0.25, t * 0.1]:
            f_new = (1 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s

            new_score = compute_score(f_new)
            if new_score < best_score - 1e-13:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                t = min(1.0, t * 1.05)
                improved = True
                break

        if not improved:
            t *= 0.7
            if t < 0.003:
                f_pert = f + rng.normal(0, 0.01, n)
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    t = 0.1
                else:

def coordinate_descent_v2(f, n_sweeps=10, rng=None):
    """Improved coordinate descent with wider search range."""
    if rng is None:
        rng = np.random.default_rng(0)

    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-12:
                continue
            best_val = original
            best_local = best_score

            # Wider geometric spread
            for factor in [0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.0/0.5, 1.0/0.3, 1.0/0.2, 1.0/0.1]:
                f[i] = max(0.0, original * factor)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                conv = np.convolve(f, f)
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]

            f[i] = best_val
            if best_local < best_score:
                best_score = best_local

        f = np.clip(f, 0, None)
        f = normalize(f)

    return f, best_score

def simulated_annealing_v2(f, rng, n_iter=5000, T0=0.03, T_end=0.00001):
    """Simulated annealing with improved move proposals."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    cur_score = best_score
    no_improve = 0

    for it in range(n_iter):
        T = T0 * (T_end / T0) ** (it / n_iter)
        f_new = f.copy()

        r = rng.random()
        if r < 0.25:
            # Block scaling with varying widths
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.5))
        elif r < 0.45:
            # Single element
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] * np.exp(rng.normal(0, 0.5)))
        elif r < 0.6:
            # Two-element interaction
            i, j = rng.integers(0, n, 2)
            if i != j:
                # Transfer mass
                transfer = rng.uniform(0, 0.3) * min(f_new[i], f_new[j])
                f_new[i] -= transfer
                f_new[j] += transfer
        elif r < 0.8:
            # Smooth bump (add or remove a Gaussian bump)
            center = int(rng.integers(0, n))
            width = max(2, int(n * rng.uniform(0.02, 0.1)))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (width ** 2))
            amplitude = rng.uniform(-0.3, 0.3)
            f_new += amplitude * kernel
        else:
            # Large reweighting
            f_new *= np.exp(rng.normal(0, 0.3, n))

        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score(f_new)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 3000 and it > n_iter // 3:
            break

    return best_f, best_score

def chirp_init(n, rng):
    """Initialize with chirp-like structure (quadratic phase made non-negative)."""
    x = np.arange(n)
    # Try different chirp rates
    rate = rng.uniform(0.5, 3.0)
    # The magnitude of exp(j*pi*rate*x^2) is 1, so we use the real part pattern
    # Instead, create a pattern based on cosine of quadratic
    f = 0.5 + 0.5 * np.cos(np.pi * rate * (x / n - 0.5) ** 2 * n)
    f = np.clip(f, 0.01, None)
    # Add noise
    f = f * (1 + 0.1 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    return normalize(f)

def random_phase_init(n, rng):
    """Initialize from random phase spectrum (made non-negative)."""
    n_fft = 1 << (2 * n - 1).bit_length()
    phase = rng.uniform(0, 2 * np.pi, n_fft // 2 + 1)
    phase[0] = 0
    mag = np.ones(n_fft // 2 + 1)
    # Ensure Hermitian symmetry for real signal
    F = mag * np.exp(1j * phase)
    f = np.fft.irfft(F, n_fft)[:n]
    f = np.abs(f) + 0.01  # Make non-negative
    f = normalize(f)
    return f

The subgradient of max_k (f*f)[k] with respect to f is:

- If the max is attained at multiple k values, the subgradient is a convex combination of the individual subgradients.

def subgradient_descent(f, n_iter=10000, rng=None, eta0=0.01):
    """Subgradient descent with AdaGrad on max convolution."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    
    # AdaGrad state
    G = np.zeros(n)  # accumulated squared gradients
    
    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Compute subgradient: weighted average of f[j-i] over argmax positions
        # Use softmax weighting for robustness
        logits = 10 * (conv - M)  # negative, max is 0
        weights = np.exp(logits)
        weights /= weights.sum()
        
        # Subgradient: g[i] = sum_j weights[j] * f[j-i]
        g = np.zeros(n)
        for j in range(2 * n - 1):
            if weights[j] < 1e-8:
                continue
            for i in range(n):
                idx = j - i
                if 0 <= idx < n:
                    g[i] += weights[j] * f[idx]
        
        # Normalize gradient
        if g.max() > 0:
            g = g / g.max()
        
        # AdaGrad update
        eta = eta0 / np.sqrt(G + g**2 + 1e-8)
        f_new = f - eta * g
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            f_new = f
        else:
            f_new = f_new / s
        
        G += g**2
        
        score = compute_score(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
        f = f_new
    
    return best_f, best_score

Now, let me also think about whether there's a way to use the LP more effectively.

The LP:
maximize sum(b)
subject to: A @ b <= M (where A is the convolution matrix and M is the current max)
           b >= 0

This finds the "best" direction to move that keeps the convolution bounded by M.

But M is the current max, so the current f is already feasible (A @ f = conv, and max(conv) = M). The LP finds another feasible point b with maximum sum.

If b is very different from f, the direction (b - f) might not be good (it might increase some other peak).

A better approach: instead of maximizing sum(b), maximize sum(b) - lambda * ||b - f||^2 (a quadratic regularization). But that makes it a QP, not an LP.

Alternatively, we can solve multiple LPs:
1. LP1: max sum(b) s.t. A@b <= M, b >= 0 (original)
2. LP2: max sum(b) s.t. A@b <= M, b >= 0, ||b - f||_1 <= epsilon (add L1 constraint)

The second LP is more conservative and might give better directions.

But the L1 constraint makes it more expensive. Let me stick with the original LP but use a better step size strategy.

**Better step size strategy**: Instead of a fixed decay, use a line search:
- Start with a large step
- If the score increases, reduce the step and try again
- Use bisection or golden section to find the optimal step

This is what the previous code does, but we can make it more efficient.

Instead of using M = max(conv), we can:
1. Solve LP with M (feasible, current f is on the boundary)
2. Solve LP with M * 0.95 (infeasible for current f, but might find a "direction")
3. Use the solution of LP2 as a target, and project from f towards it

Actually, if M * 0.95 is infeasible, the LP won't have a feasible solution (the current f violates it). So we'd need to add a slack variable or use a different formulation.

Let me think of another approach: **Frank-Wolfe (Conditional Gradient) method**.

The Frank-Wolfe algorithm for minimizing a convex function f(x) over a convex set C:
1. Start at x_0 in C
2. At each step:
   a. Solve the linearization: s = argmin_{y in C} <∇f(x), y>
   b. Update: x_new = (1-t)x + t*s where t is chosen by line search

In our case, the function is NOT convex (it's the max of bilinear forms), so Frank-Wolfe doesn't directly apply. But we can use a "non-convex Frank-Wolfe" variant where we use subgradients instead of gradients.

This is essentially what the LP approach does: the LP solves the linearization subproblem, and the convex combination is the update.

So the LP approach IS a form of Frank-Wolfe for the non-convex problem. Good.

"""
Multi-strategy optimization for minimizing peak autocorrelation of non-negative sequences.
Uses efficient subgradient computation, spectral flattening, LP refinement, and diverse search.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute score = 2*n*max(conv) / (sum)^2 for non-negative sequence f."""
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full'). Vectorized."""
    n = len(f)
    k = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    idx = k - j
    A = np.where((idx >= 0) & (idx < n), f[idx], 0.0)
    return A


def compute_subgradient(f, conv, temperature=5.0):
    """Compute subgradient of max(conv) w.r.t. f using softmax weighting over near-max positions."""
    n = len(f)
    M = conv.max()
    # Softmax weights over convolution positions
    logits = temperature * (conv - M)
    logits -= logits.max()
    weights = np.exp(logits)
    wsum = weights.sum()
    if wsum < 1e-15:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= wsum

    # g[i] = sum_j weights[j] * f[j-i] for valid j-i
    # Use correlate: g[i] = cross_correlate(weights_padded, f)[i]
    weights_padded = np.zeros(2 * n)
    weights_padded[:2 * n - 1] = weights
    # np.correlate(a, v, 'valid')[m] = sum_k v[k] * a[m+k]
    # We want g[i] = sum_k f[k] * weights_padded[i+k]
    # So: result = np.correlate(weights_padded, f, 'valid')
    # result[m] = sum_k f[k] * weights_padded[m+k] for m = 0..(2n-n)=n
    # So g = result[:n]
    result = np.correlate(weights_padded, f, mode='valid')
    g = result[:n]
    return g


def subgradient_descent(f, n_iter=2000, rng=None, eta0=0.05, use_adagrad=True):
    """Subgradient descent with optional AdaGrad on max convolution value."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype np.float64)
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    G = np.zeros(n) if use_adagrad else None
    stall_count = 0

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()

        g = compute_subgradient(f, conv, temperature=5.0)
        gmax = g.max()
        if gmax < 1e-15:
            break
        g = g / gmax

        if use_adagrad:
            eta = eta0 / np.sqrt(G + g * g + 1e-8)
        else:
            eta = eta0 / (1.0 + 0.01 * it)

        f_new = f - eta * g
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / s

        if use_adagrad:
            G += g * g

        score = compute_score(f_new)
        if score < best_score - 1e-13:
            best_score = score
            best_f = f_new.copy()
            stall_count = 0
        else:
            # Accept with small probability to explore
            if rng.random() < 0.05:
                pass  # still update f for exploration
            stall_count += 1
            if stall_count > 300:
                # Perturb
                f_pert = f * (1.0 + 0.08 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                sp = f_pert.sum()
                if sp > 1e-10:
                    f_pert = f_pert / sp
                    if compute_score(f_pert) < best_score:
                        f = f_pert
                        best_f = f.copy()
                        best_score = compute_score(f)
                        G = np.zeros(n)
                        stall_count = 0
                        continue
                break

        f = f_new

    return best_f, best_score


def spectral_flat_v2(f, n_iter=400, rng=None, alpha=0.5):
    """Spectral flattening with soft target and progressive refinement."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()

    best_f = f.copy()
    best_score = compute_score(f)
    stall_count = 0

    for it in range(n_iter):
        alpha_eff = alpha * (1.0 / (1.0 + 0.01 * it))

        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)

        # Geometric mean of non-DC magnitudes as target
        if len(mag) > 1:
            log_mags = np.log(np.maximum(mag[1:], 1e-15))
            target_mag = np.exp(np.mean(log_mags))
        else:
            target_mag = 1.0 / np.sqrt(n)

        mag_new = mag.copy()
        mag_new[1:] = (1.0 - alpha_eff) * mag[1:] + alpha_eff * target_mag
        mag_new[0] = mag[0]

        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]

        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s

        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall_count = 0
        else:
            stall_count += 1
            if rng.random() < 0.08:
                f = f_new  # explore
            if stall_count > 200:
                break

    return best_f, best_score


def pressure_update_v2(f, n_iter=300, eta=0.1, rng=None):
    """Pressure-based update using efficient subgradient computation."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    stall_count = 0

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()

        g = compute_subgradient(f, conv, temperature=10.0)
        gmax = g.max()
        if gmax < 1e-15:
            break
        g = g / gmax

        eta_eff = eta * (1.0 / (1.0 + 0.005 * it))
        f_new = f * np.exp(-eta_eff * g)
        f_new = np.clip(f_new, 1e-15, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s

        score = compute_score(f_new)

            f = f_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 100:
                f_pert = f * (1.0 + 0.08 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                sp = f_pert.sum()
                if sp > 1e-10:
                    f_pert = f_pert / sp
                    if compute_score(f_pert) < best_score:
                        f = f_pert
                        best_f = f.copy()
                        best_score = compute_score(f)

def lp_refine_v2(f, time_budget, t0=0.5, rng=None, lp_time_limit=2):
    """LP-based refinement with adaptive step size."""
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)

    best_f = f.copy()
    best_score = compute_score(f)
    t = t0
    start = time.monotonic()
    it_count = 0

    while time.monotonic() - start < time_budget:
        it_count += 1
        conv = np.convolve(f, f)
        M = conv.max()

        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': lp_time_limit})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break

        g = b / b_sum

        # Check if g is too similar to f (no useful direction)
        if np.allclose(g, f, atol=1e-6):
            t *= 0.8
            if t < 0.005:
                break
            continue

        improved = False
        for step in [min(1.0, t * 1.2), t, t * 0.5, t * 0.25, t * 0.1]:
            f_new = (1.0 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s

            new_score = compute_score(f_new)
            if new_score < best_score - 1e-13:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                t = min(1.0, t * 1.03)
                improved = True
                break

        if not improved:
            t *= 0.65
            if t < 0.003:
                # Try perturbation
                f_pert = f + rng.normal(0, 0.01, n)
                f_pert = np.clip(f_pert, 0, None)
                sp = f_pert.sum()
                if sp > 1e-10:
                    f_pert = f_pert / sp
                    if compute_score(f_pert) < best_score:
                        f = f_pert
                        best_f = f.copy()
                        best_score = compute_score(f)
                        t = 0.15
                    else:
                        break

    return best_f, best_score


def coordinate_descent_v2(f, n_sweeps=8, rng=None):
    """Coordinate descent with wider search."""
    if rng is None:
        rng = np.random.default_rng(0)

    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)

    factors = [0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.0/0.5, 1.0/0.3, 1.0/0.2, 1.0/0.1]

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-12:
                continue
            best_val = original
            best_local = best_score

            for factor in factors:
                f[i] = max(0.0, original * factor)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                conv = np.convolve(f, f)
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]

            f[i] = best_val
            if best_local < best_score:
                best_score = best_local

        f = np.clip(f, 0, None)
        f = normalize(f)

    return f, best_score


def simulated_annealing_v2(f, rng, n_iter=8000, T0=0.02, T_end=0.00001):
    """Simulated annealing with diverse moves."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    cur_score = best_score
    no_improve = 0

    for it in range(n_iter):
        T = T0 * (T_end / T0) ** (it / n_iter)
        f_new = f.copy()

        r = rng.random()
        if r < 0.25:
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.4))
        elif r < 0.45:
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] * np.exp(rng.normal(0, 0.4)))
        elif r < 0.6:
            i, j = rng.integers(0, n, 2)
            if i != j:
                transfer = rng.uniform(0, 0.2) * min(f_new[i], f_new[j])
                f_new[i] -= transfer
                f_new[j] += transfer
        elif r < 0.8:
            center = int(rng.integers(0, n))
            width = max(2, int(n * rng.uniform(0.02, 0.08)))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (width ** 2))
            f_new += rng.uniform(-0.2, 0.2) * kernel
        else:
            f_new *= np.exp(rng.normal(0, 0.25, n))

        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score(f_new)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 3000 and it > n_iter // 3:
            break

    return best_f, best_score


def chirp_init(n, rng):
    """Chirp-like initialization."""
    x = np.arange(n)
    rate = rng.uniform(1.0, 4.0)
    f = 0.5 + 0.5 * np.cos(np.pi * rate * (x / n - 0.5) ** 2 * n * 0.1)
    f = np.clip(f, 0.01, None)
    f = f * (1.0 + 0.1 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    return normalize(f)


def random_phase_init(n, rng):
    """Initialize from random phase spectrum."""
    n_fft = 1 << (2 * n - 1).bit_length()
    half = n_fft // 2 + 1
    phase = rng.uniform(0, 2 * np.pi, half)
    phase[0] = 0
    N = np.ones(half)
    F = N * np.exp(1j * phase)
    # Ensure Hermitian symmetry
    F_full = np.zeros(n_fft, dtype=complex)
    F_full[:half] = F
    F_full[half:] = np.conj(F[1:half][::-1]) if half > 1 else 0
    f = np.fft.ifft(F_full).real[:n]
    f = np.abs(f) + 0.01
    return normalize(f)


def diverse_init_v2(n, rng):
    """Diverse initialization combining multiple strategies."""
    r = rng.random()
    if r < 0.2:
        f = chirp_init(n, rng)
    elif r < 0.4:
        f = random_phase_init(n, rng)
    elif r < 0.6:
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        f = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    elif r < 0.8:
        f = np.ones(n)
        for _ in range(int(rng.integers(2, 5))):
            freq = rng.uniform(0.5, 3.0)
            f *= 0.5 + 0.5 * np.cos(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
        f = np.abs(f) + 0.05
    else:
        f = np.ones(n)
        for _ in range(int(rng.integers(3, 7))):
            center = int(rng.integers(0, n))
            width = max(3, int(n * rng.uniform(0.03, 0.15)))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (width ** 2))
            f *= (1.0 + 0.3 * rng.standard_normal() * kernel)
        f = np.abs(f) + 0.05

    f = np.clip(f, 0.01, None)
    f = f * (1.0 + 0.05 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    return normalize(f)

            t_alloc = min(30, remaining * 0.12)
            n_iter_sg = max(500, int(t_alloc * 50))
            f2, s2 = subgradient_descent(f1, n_iter=n_iter_sg, rng=rng, eta0=0.04)
            print(f"  [A] After subgrad ({n_iter_sg} iters): {s2:.6f}")
            if s2 < best_score_overall:
                best_score_overall = s2
                best_overall = f2.copy()

            # Pressure update as second pass
            f2b, s2b = pressure_update_v2(f2, n_iter=200, eta=0.08, rng=rng)
            print(f"  [A] After pressure: {s2b:.6f}")
            if s2b < best_score_overall:
                best_score_overall = s2b
                best_overall = f2b.copy()

            # LP refinement (main workhorse)
            remaining2 = deadline - time.monotonic()
            t_lp = min(remaining2 * 0.4, 180)
            if t_lp > 20:
                f3, s3 = lp_refine_v2(f2b, t_lp, t0=0.5, rng=rng, lp_time_limit=2)
                print(f"  [A] After LP ({t_lp:.0f}s): {s3:.6f}")
                if s3 < best_score_overall:
                    best_score_overall = s3
                    best_overall = f3.copy()

        # --- Pipeline B: Different init + subgradient + LP ---
        remaining = deadline - time.monotonic()
        if remaining > 150:
            f0b = diverse_init_v2(n, rng)

            n_iter_spec_b = max(80, int(min(20, remaining * 0.1) * 60))
            f1b, s1b = spectral_flat_v2(f0b, n_iter=n_iter_spec_b, rng=rng, alpha=0.4)

            n_iter_sg_b = max(300, int(min(25, remaining * 0.1) * 50))
            f2b2, s2b2 = subgradient_descent(f1b, n_iter=n_iter_sg_b, rng=rng, eta0=0.03)
            if s2b2 < best_score_overall:
                best_score_overall = s2b2
                best_overall = f2b2.copy()
                print(f"  [B] After subgrad: {s2b2:.6f}")

            remaining3 = deadline - time.monotonic()
            t_lp_b = min(remaining3 * 0.3, 90)
            if t_lp_b > 15:
                f3b, s3b = lp_refine_v2(f2b2, t_lp_b, t0=0.4, rng=rng, lp_time_limit=2)
                if s3b < best_score_overall:
                    best_score_overall = s3b
                    best_overall = f3b.copy()
                    print(f"  [B] After LP: {s3b:.6f}")

        # --- Pipeline C: Refine parent ---
        if parent is not None and len(parent) == n:
            remaining = deadline - time.monotonic()
            if remaining > 60:
                for scale in [0.05, 0.1, 0.15]:
                    fp = parent * (1.0 + scale * rng.normal(0, 1, n))
                    fp = np.clip(fp, 0, None)
                    fp = normalize(fp)
                    fp, sp = subgradient_descent(fp, n_iter=500, rng=rng, eta0=0.03)
                    if sp < best_score_overall:
                        best_score_overall = sp
                        best_overall = fp.copy()
                        print(f"  [C] Parent perturb({scale}) + subgrad: {sp:.6f}")

                    remaining = deadline - time.monotonic()
                    if remaining > 30 and sp < best_score_overall + 0.005:
                        t_lp_c = min(remaining * 0.2, 30)
                        if t_lp_c > 10:
                            fp2, sp2 = lp_refine_v2(fp, t_lp_c, t0=0.3, rng=rng, lp_time_limit=2)
                            if sp2 < best_score_overall:
                                best_score_overall = sp2
                                best_overall = fp2.copy()
                                print(f"  [C] Parent perturb({scale}) + LP: {sp2:.6f}")

    # --- Final polishing phase ---
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 25:
            print(f"\n{'='*60}")
            print(f"Polishing phase, remaining: {remaining:.0f}s")
            print(f"Best so far: {best_score_overall:.6f}, n={len(best_overall)}")
            print(f"{'='*60}")

            # Coordinate descent
            if remaining > 20:
                n_sweeps = max(3, int(min(6, remaining * 0.1)))
                f_cd, s_cd = coordinate_descent_v2(best_overall, n_sweeps=n_sweeps, rng=rng)
                if s_cd < best_score_overall:
                    best_score_overall = s_cd
                    best_overall = f_cd.copy()
                    print(f"  After CD ({n_sweeps} sweeps): {best_score_overall:.6f}")

            # Simulated annealing
            remaining = deadline - time.monotonic()
            if remaining > 20:
                n_iter_sa = min(8000, int(remaining * 150))
                f_sa, s_sa = simulated_annealing_v2(best_overall, rng, n_iter=n_iter_sa,
                                                     T0=0.015, T_end=0.00001)
                if s_sa < best_score_overall:
                    best_score_overall = s_sa
                    best_overall = f_sa.copy()
                    print(f"  After SA ({n_iter_sa} iters): {best_score_overall:.6f}")

            # Extra spectral pass
            remaining = deadline - time.monotonic()
            if remaining > 15:
                f_spec2, s_spec2 = spectral_flat_v2(best_overall, n_iter=300, rng=rng, alpha=0.2)
                if s_spec2 < best_score_overall:
                    best_score_overall = s_spec2
                    best_overall = f_spec2.copy()
                    print(f"  After extra spectral: {best_score_overall:.6f}")

            # More subgradient
            remaining = deadline - time.monotonic()
            if remaining > 15:
                f_sg2, s_sg2 = subgradient_descent(best_overall, n_iter=2000, rng=rng, eta0=0.02)
                if s_sg2 < best_score_overall:
                    best_score_overall = s_sg2
                    best_overall = f_sg2.copy()
                    print(f"  After extra subgrad: {best_score_overall:.6f}")

            # Final CD
            remaining = deadline - time.monotonic()
            if remaining > 15:
                f_cd2, s_cd2 = coordinate_descent_v2(best_overall, n_sweeps=4, rng=rng)
                if s_cd2 < best_score_overall:
                    best_score_overall = s_cd2
                    best_overall = f_cd2.copy()
                    print(f"  Final CD: {best_score_overall:.6f}")

            # Final LP
            remaining = deadline - time.monotonic()
            if remaining > 20:
                f_lp, s_lp = lp_refine_v2(best_overall, min(remaining * 0.5, 60),
                                           t0=0.15, rng=rng, lp_time_limit=2)
                if s_lp < best_score_overall:
                    best_score_overall = s_lp
                    best_overall = f_lp.copy()
                    print(f"  Final LP: {best_score_overall:.6f}")

    # Verify and return
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*60}")
        return final_list

    # Fallback
    print("Using fallback uniform sequence")
    return [1.0] * 512
