"""
Optimized step function search using active-set L-BFGS-B, 
coordinate descent, and simulated annealing.
Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import minimize
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def softmax_obj(a, T):
    conv = np.convolve(a, a)
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    shift = (conv - np.max(conv)) / T
    shift = np.clip(shift, -500, 500)
    sm = np.max(conv) + T * np.log(np.sum(np.exp(shift)))
    return 2.0 * len(a) * sm / (S * S)

def softmax_grad(a, T):
    conv = np.convolve(a, a)
    S = np.sum(a)
    n = len(a)
    max_c = np.max(conv)
    shift = (conv - max_c) / T
    shift = np.clip(shift, -500, 500)
    exp_vals = np.exp(shift)
    sum_exp = np.sum(exp_vals)
    p = exp_vals / sum_exp
    
    # Gradient of soft_max w.r.t. a[i]
    # d(sm)/d(a[i]) = sum_k p[k] * d(c[k])/d(a[i])
    # d(c[k])/d(a[i]) = 2*a[k-i] if 0 <= k-i < n
    grad_sm = np.zeros(n)
    for i in range(n):
        # sum over k where 0 <= k-i < n, i.e., k in [i, i+n-1]
        k_min = max(i, 0)
        k_max = min(i + n - 1, len(conv) - 1)
        # p[k] * 2*a[k-i] for k in [k_min, k_max]
        weights = p[k_min:k_max+1]
        a_slice = a[0:k_max - k_min + 1]  # a[k-i] for k in [k_min, k_max]

"""
Optimized step function coefficient search using:
1. Soft-max L-BFGS-B with temperature annealing
2. Active-set exact peak tracking with L-BFGS-B
3. Coordinate descent with scan-based 1D optimization
4. Simulated annealing for stochastic refinement

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_obj(a):
    """Compute the objective function value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def softmax_value_and_grad(a, T):
    """Compute soft-max of convolution and its gradient."""
    n = len(a)
    conv = np.convolve(a, a)
    S = float(np.sum(a))
    max_c = float(np.max(conv))
    shift = (conv - max_c) / T
    shift = np.clip(shift, -500.0, 500.0)
    exp_vals = np.exp(shift)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    sm_val = max_c + T * np.log(sum_exp)

    # grad_sm[i] = 2 * sum_j p[i+j] * a[j] for valid j
    grad_sm = np.zeros(n)
    for i in range(n):
        k_max = min(i + n - 1, 2 * n - 2)
        if k_max >= i:
            p_slice = p[i:k_max + 1]
            a_slice = a[:len(p_slice)]
            grad_sm[i] = 2.0 * np.dot(p_slice, a_slice)

    # f = 2n * sm / S^2
    # df/d(a[i]) = 2n * (grad_sm[i] * S - 2*sm_val) / S^3
    grad_f = 2.0 * n * (grad_sm * S - 2.0 * sm_val) / (S ** 3)
    return sm_val, grad_sm, grad_f


def softmax_objective(a, T):
    """Objective function for L-BFGS-B."""
    n = len(a)
    conv = np.convolve(a, a)
    S = float(np.sum(a))
    if S < 1e-10:
        return 1e10
    max_c = float(np.max(conv))
    shift = (conv - max_c) / T
    shift = np.clip(shift, -500.0, 500.0)
    sm_val = max_c + T * np.log(float(np.sum(np.exp(shift))))
    return 2.0 * n * sm_val / (S * S)


def softmax_gradient(a, T):
    """Gradient for L-BFGS-B."""
    n = len(a)
    conv = np.convolve(a, a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.ones(n)
    max_c = float(np.max(conv))
    shift = (conv - max_c) / T
    shift = np.clip(shift, -500.0, 500.0)
    exp_vals = np.exp(shift)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    sm_val = max_c + T * np.log(sum_exp)

    grad_sm = np.zeros(n)
    for i in range(n):
        k_max = min(i + n - 1, 2 * n - 2)
        if k_max >= i:
            p_slice = p[i:k_max + 1]
            a_slice = a[:len(p_slice)]
            grad_sm[i] = 2.0 * np.dot(p_slice, a_slice)

    return 2.0 * n * (grad_sm * S - 2.0 * sm_val) / (S ** 3)


def lbfgsb_softmax(a_init, T_start, T_end, max_time, max_iter=500):
    """Run L-BFGS-B on soft-max objective with temperature annealing."""
    n = len(a_init)
    a = a_init.copy()
    start = time.monotonic()

    # Normalize
    S = np.sum(a)
    if S > 1e-10:
        a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    T = T_start

    while time.monotonic() - start < max_time and T > T_end:
        # Run L-BFGS-B at current T
        bounds = [(0.0, None)] * n
        res = minimize(
            softmax_objective, a, args=(T,),
            jac=softmax_gradient, method='L-BFGS-B',
            bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        if res.fun < 1e10:
            a_new = np.clip(res.x, 0.0, None)
            S = np.sum(a_new)
            if S > 1e-10:
                a_new = a_new / S
                obj = compute_obj(a_new)
                if obj <= best_obj * 1.001:  # Accept if not much worse
                    a = a_new
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()

        # Anneal temperature
        T = max(T_end, T * 0.6)

    return best_a, best_obj


def build_A_matrix(a):
    """Build the matrix A such that A @ g = convolve(a, g) (truncated to n entries)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_alternation(a_init, max_time, max_iters=200):
    """LP-based alternating optimization."""
    n = len(a_init)
    a = a_init.copy()
    S = np.sum(a)
    if S > 1e-10:
        a = a / S
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    A = build_A_matrix(a)

    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        if M < 1e-15:
            break

        # Solve LP: max sum(g) s.t. A @ g <= M, g >= 0
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': max_time - (time.monotonic() - start)})
        if res.status != 0:
            break

        g = res.x
        g_sum = float(-res.fun)
        if g_sum <= 0:
            break

        g = g / g_sum

        # Try blending
        improved = False
        for t in [0.5, 0.3, 0.15, 0.08, 0.04, 0.1, 0.2]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.clip(a_try, 0.0, None)
            S_try = np.sum(a_try)
            if S_try < 1e-10:
                continue
            a_try = a_try / S_try
            obj = compute_obj(a_try)
            if obj < best_obj:
                best_obj = obj
                best_a = a_try.copy()
                a = a_try
                improved = True
                break

        if not improved:
            # Rebuild A with new a
            A = build_A_matrix(a)

    return best_a, best_obj


def coordinate_descent(a_init, max_time, n_sweep=7):
    """Coordinate descent with value scanning."""
    n = len(a_init)
    a = a_init.copy()
    S = np.sum(a)
    if S > 1e-10:
        a = a / S
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Sweep multipliers
    sweeps = [0.0, 0.1, 0.3, 0.5, 0.7, 1.0, 1.4, 1.8, 2.5, 4.0]
    sweeps = sweeps[:n_sweep + 1]

    rounds = 0
    while time.monotonic() - start < max_time:
        rounds += 1
        improved = False
        for i in range(n):
            if time.monotonic() - start > max_time:
                break
            current = a[i]
            if current < 1e-15:
                candidates = [0.0, 0.01, 0.02, 0.05]
            else:
                candidates = [current * s for s in sweeps]
                candidates.append(0.0)

            local_best_t = current
            local_best_obj = compute_obj(a)

            for t in candidates:
                if t == current:
                    continue
                a[i] = t
                S_new = np.sum(a)
                if S_new < 1e-10:
                    a[i] = current
                    continue
                a_norm = a / S_new
                obj = compute_obj(a_norm)
                if obj < local_best_obj:
                    local_best_obj = obj
                    local_best_t = t
                a[i] = current

            if local_best_t != current:
                a[i] = local_best_t
                S = np.sum(a)
                a = a / S
                improved = True
                if local_best_obj < best_obj:
                    best_obj = local_best_obj
                    best_a = a.copy()

        if not improved:
            break

    return best_a, best_obj


def simulated_annealing(a_init, max_time, rng):
    """Simulated annealing with peak-directed perturbations."""
    n = len(a_init)
    a = a_init.copy()

    if S > 1e-10:
        a = a / S
    best_a = a.copy()

    start = time.monotonic()

    T = 0.05
    T_min = 0.001
    cool_rate = 0.9995
    it = 0

    while time.monotonic() - start < max_time and T > T_min:
        it += 1
        # Decide perturbation type
        r = rng.random()
        if r < 0.4:
            # Single element perturbation
            i = int(rng.integers(0, n))
            delta = rng.normal(0, 0.1)

        elif r < 0.7:

            else:

                pass  # a is already updated, need to revert... 
                # Actually let's do it differently - keep a separate candidate

        T = max(T_min, T * cool_rate)

    return best_a, best_obj


def simulated_annealing_v2(a_init, max_time, rng):
    """Improved simulated annealing with proper accept/reject."""
    n = len(a_init)
    a = a_init.copy()
    S = np.sum(a)
    if S > 1e-10:
        a = a / S
    best_a = a.copy()
    best_obj = compute_obj(a)
    current_obj = best_obj
    start = time.monotonic()

    T = 0.03
    T_min = 0.0005
    cool_rate = 0.9997
    it = 0
    no_improve = 0

    while time.monotonic() - start < max_time and T > T_min:
        it += 1
        if it % 1000 == 0:
            print(f"    SA: it={it}, T={T:.4f}, best={best_obj:.6f}")

        # Generate candidate
        a_cand = a.copy()
        r = rng.random()
        sigma_scale = 0.05 + 0.15 * (T / 0.03)

        if r < 0.35:
            # Gaussian perturbation on random element
            i = int(rng.integers(0, n))
            a_cand[i] = max(0.0, a_cand[i] + rng.normal(0, sigma_scale))
        elif r < 0.55:
            # Two-element swap-like
            i, j = rng.integers(0, n, 2)
            if i != j:
                delta = rng.normal(0, sigma_scale)
                a_cand[i] = max(0.0, a_cand[i] + delta)
                a_cand[j] = max(0.0, a_cand[j] - delta)
        elif r < 0.75:
            # Block multiplicative
            width = max(1, int(n * rng.uniform(0.03, 0.12)))
            pos = int(rng.integers(0, n - width + 1))
            a_cand[pos:pos + width] = np.maximum(0, a_cand[pos:pos + width] * np.exp(rng.normal(0, sigma_scale, width)))
        else:
            # Peak-directed (reduce contribution to peak)
            conv = np.convolve(a_cand, a_cand)
            peak_idx = int(np.argmax(conv))
            # Find elements contributing most to the peak
            contribs = np.zeros(n)
            for j in range(n):
                ki = peak_idx - j
                if 0 <= ki < n:
                    contribs[j] = 2.0 * a_cand[j] * a_cand[ki]
            top_idx = np.argsort(contribs)[-5:]
            i = int(top_idx[int(rng.integers(0, len(top_idx)))])
            a_cand[i] = max(0.0, a_cand[i] * (1.0 - rng.uniform(0.05, 0.3)))

        S_cand = np.sum(a_cand)
        if S_cand < 1e-10:
            continue
        a_cand = a_cand / S_cand
        cand_obj = compute_obj(a_cand)

        # Accept/reject
        delta = cand_obj - current_obj
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_cand
            current_obj = cand_obj
            if cand_obj < best_obj:
                best_obj = cand_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Cool
        T = max(T_min, T * cool_rate)
        if no_improve > 5000:
            T = max(T_min, T * 0.5)
            no_improve = 0

    return best_a, best_obj


def structured_init(n, rng, init_type=0):
    """Generate structured initializations."""
    if init_type == 0:
        # Edge-weighted
        a = np.zeros(n)
        c = 1.0 / np.sqrt(2.0 * n)
        a[0] = c
        a[1] = c * 0.8
        a[-1] = c
        a[-2] = c * 0.8
        if n > 4:
            interior = np.ones(n - 4) * 0.3
            interior += rng.normal(0, 0.1, n - 4)
            a[2:n-2] = np.maximum(0, interior)
        s = np.sum(a)
        return a / s

    elif init_type == 1:
        # Cosine window
        phase = rng.uniform(0, 2 * np.pi)
        freq = rng.uniform(0.1, 0.5)
        a = np.maximum(0, np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
        a += 0.05
        s = np.sum(a)
        return a / s

    elif init_type == 2:
        # Gaussian bump
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        a += 0.05
        s = np.sum(a)
        return a / s

    elif init_type == 3:
        # Random positive
        a = np.abs(rng.normal(1.0, 0.5, n))
        s = np.sum(a)
        return a / s

    elif init_type == 4:
        # Periodic
        period = max(2, int(n / rng.integers(5, 30)))
        pattern = np.abs(rng.normal(1.0, 0.3, period))
        a = np.tile(pattern, n // period + 1)[:n]
        s = np.sum(a)
        return a / s

    else:
        # Triangular (high in middle)
        x = np.linspace(-1, 1, n)
        a = np.maximum(0, 1 - x ** 2)
        a += 0.05 * np.abs(rng.normal(0, 1, n))
        s = np.sum(a)
        return a / s

    n = len(a_init)

    a = a_init.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: L-BFGS-B on soft-max
    if time.monotonic() - start < time_budget * 0.4:
        conv = np.convolve(a, a)
        T_start = 0.1 * float(np.max(conv))
        T_end = 0.005 * float(np.max(conv))
        t1 = time_budget * 0.35
        a, obj = lbfgsb_softmax(a, T_start, T_end, t1)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"    After L-BFGS-B: {obj:.6f}")

    # Phase 2: LP alternation
    if time.monotonic() - start < time_budget * 0.7:
        t2 = time_budget * 0.2
        a, obj = lp_alternation(best_a, t2, max_iters=100)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"    After LP: {obj:.6f}")

    # Phase 3: Coordinate descent
    if time.monotonic() - start < time_budget * 0.9:
        t3 = time_budget * 0.2
        a, obj = coordinate_descent(best_a, t3)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"    After CoordDes: {obj:.6f}")

    # Phase 4: Simulated annealing
    if time.monotonic() - start < time_budget * 0.98:
        t4 = time_budget * 0.18
        a, obj = simulated_annealing_v2(best_a, t4, rng)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"    After SA: {obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization with diverse initializations and sequence lengths.
    Uses L-BFGS-B (soft-max), LP alternation, coordinate descent, and simulated annealing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 8.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")

            # Optimize parent
            parent_time = min(80.0, (deadline - time.monotonic()) * 0.15)
            if parent_time > 20:
                p_opt, p_obj = full_optimize(parent.copy(), parent_time, seed=0)
                if p_obj < best_score:
                    best_score = p_obj
                    best_seq = p_opt.copy()
                    print(f"Parent optimized: obj={p_obj:.6f}")
    except Exception as e:
        print(f"Parent loading failed: {e}")

    # Main search: try various n and initializations
    n_values = [100, 150, 200, 250, 300, 350, 400]
    init_types = [0, 1, 2, 3, 4, 5]

    # Allocate time
    num_configs = 0
    for n in n_values:
        for it in init_types:
            num_configs += 1

    # We can only afford a subset
    # Prioritize: parent n with various inits, then other n
    configs = []
    parent_n = 300
    try:
        parent_n = len(np.asarray(height_sequence_1))
    except:
        pass

    # Add parent-n configs first
    for it in init_types:
        configs.append((parent_n, it))
    # Then other n values
    for n in n_values:
        if n == parent_n:
            continue
        for it in init_types[:3]:
            configs.append((n, it))

    # Shuffle non-parent configs
    parent_configs = [c for c in configs if c[0] == parent_n]
    other_configs = [c for c in configs if c[0] != parent_n]
    rng.shuffle(other_configs)
    configs = parent_configs + other_configs

    print(f"\nSearch plan: {len(configs)} configs, deadline in {deadline - time.monotonic():.0f}s")

    # Determine how many configs we can try
    time_per_config = 60.0  # target
    max_configs = int((deadline - time.monotonic()) / time_per_config)
    max_configs = max(3, min(max_configs, len(configs)))
    configs = configs[:max_configs]

    for idx, (n, it) in enumerate(configs):
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Time allocation: give more time to promising configs
        if idx < 3:
            t_alloc = min(80.0, remaining * 0.25)
        elif idx < 6:
            t_alloc = min(60.0, remaining * 0.2)
        else:
            t_alloc = min(45.0, remaining / max(1, len(configs) - idx))

        if t_alloc < 15:
            break

        print(f"\nConfig {idx+1}/{len(configs)}: n={n}, init={it}, time={t_alloc:.0f}s")

        a_init = structured_init(n, rng, it)
        a_opt, obj_opt = full_optimize(a_init, t_alloc, seed=seed + idx)

        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  ** NEW BEST: n={n}, init={it}, obj={obj_opt:.6f}")
        else:
            print(f"  n={n}, init={it}, obj={obj_opt:.6f}")

    # Final refinement on best found
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: {remaining:.0f}s, current best={best_score:.6f}")

        # Run full_optimize on the best with remaining time
        t_final = min(remaining * 0.6, 120.0)
        a_ref, obj_ref = full_optimize(best_seq.copy(), t_final, seed=seed + 999)
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  After final opt: {best_score:.6f}")

        # Additional coordinate descent
        if time.monotonic() < deadline - 10:
            t_cd = deadline - time.monotonic() - 8.0
            if t_cd > 5:
                a_cd, obj_cd = coordinate_descent(best_seq, min(t_cd, 60.0))
                if obj_cd < best_score:
                    best_score = obj_cd
                    best_seq = a_cd.copy()
                    print(f"  After final CD: {best_score:.6f}")

        # Additional SA
        if time.monotonic() < deadline - 5:
            t_sa = deadline - time.monotonic() - 5.0
            if t_sa > 5:
                a_sa, obj_sa = simulated_annealing_v2(best_seq, min(t_sa, 30.0), rng)
                if obj_sa < best_score:
                    best_score = obj_sa
                    best_seq = a_sa.copy()
                    print(f"  After final SA: {best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
