    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 0.01:

- If `0 <= k-j < n`: both terms contribute, total = `a[k-j] + a[k-j] = 2*a[k-j]`
- If `0 <= k-j < n` is false (i.e., k-j is out of bounds): contribution is 0
- Special case: if `j = k-j` (k=2j), it's still `a[j] + a[j] = 2*a[j] = 2*a[k-j]` ✓

So: **`d(conv[k])/d(a[j]) = 2*a[k-j]` if `0 <= k-j < n`, else 0.**

Or equivalently: `d(conv[k])/d(a[j]) = 2*a[k-j]` if `max(0, k-n+1) <= j <= min(n-1, k)`, else 0.

This is much simpler than what I had before! The gradient of the max (subgradient) averaged over active lags is:

`subgrad[j] = (1/n_active) * sum_{k in active} 2*a[k-j]` (where `0 <= k-j < n`)

Let me rewrite the subgradient code with this correct formula.

Now, let me also think about the **objective gradient** (not just the conv subgradient). The objective is:
`f(a) = 2*n*max_k conv[a,a][k] / (sum(a))^2`

The gradient is more complex because of the normalization. But since we're always normalizing `a` to have sum 1, we can just minimize `max_k conv[a,a][k]` subject to `sum(a) = 1, a >= 0`, and the subgradient of the max conv (projected onto the simplex) is what we need.

Actually, for the subgradient method, we're minimizing `max_k conv[k]` on the simplex `{a: sum(a)=1, a>=0}`. The subgradient of `max_k conv[k]` at point `a` is the average of the gradients of the active functions. Then we project back to the simplex (which is: clip to non-negative, then normalize to sum 1... actually, the Euclidean projection onto the simplex is more complex, but for non-negative vectors, clipping + normalizing is a good approximation).

OK let me now write the final, clean implementation. I'll focus on correctness and efficiency.

One more idea: **instead of the LP, I can use a simpler "natural gradient" approach**:

At each step:
1. Find the active lag(s) k* where conv is maximum
2. Compute the subgradient
3. Take a step in the negative subgradient direction
4. Project onto the simplex (a >= 0, sum = 1)

This is essentially what the subgradient method does, and it's much faster than the LP (no LP solve needed). The LP gives a better direction, but the subgradient might be sufficient if we use proper step size decay.

Let me combine both: use the LP for the first phase (to get close to the optimum), then switch to subgradient for fine-tuning.

def compute_obj(a):
    n = len(a)
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (sum_a * sum_a)


def solve_lp_g(a, M):
    n = len(a)
    # Build constraint matrix
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        idx = np.arange(i_lo, i_hi + 1)
        A[k, idx] = a[k - idx]
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x, 0.0)
        s = g.sum()
        if s > 1e-15:
            return g / s
    return None


def lp_alternate(a_init, max_iter=5000, deadline=None):

        best_local = best_obj

def subgrad_polish(a_init, n_steps=10000, deadline=None):
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    s = a.sum()
    a /= s
    best_a = a.copy()
    best_obj = compute_obj(a)
    momentum = np.zeros(n)
    mu = 0.85
    
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        conv = np.convolve(a, a)
        max_c = float(np.max(conv))
        tol = max_c * 0.002
        active = conv >= max_c - tol
        n_act = np.sum(active)
        if n_act == 0:
            active = np.zeros(2*n-1, dtype=bool)
            active[np.argmax(conv)] = True
            n_act = 1
        
        # Subgradient: for each active k, grad[j] += 2*a[k-j] (if 0<=k-j<n)
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            kp = k - np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi+1] += 2.0 * a[kp]
        
        grad /= n_act
        
        # Momentum
        lr = 0.05 / (1.0 + step * 0.0001)
        momentum = mu * momentum + (1 - mu) * grad
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-10:
            break
        a_new /= ss
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_obj


def coordinate_descent(a_init, n_rounds=10, deadline=None):
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        for j in range(n):
            old = a[j]
            best_val = old
            best_local = best_obj
            for factor in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
                a[j] = old * factor
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_val = a[j]
            a[j] = best_val
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj
