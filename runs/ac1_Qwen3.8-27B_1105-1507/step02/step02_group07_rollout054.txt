Let me also think about the "excess energy" approach more carefully:

J_T(f) = sum_k [max(0, C[k] - T)]²

∂J_T/∂f[j] = 2 * sum_{k: C[k]>T} (C[k] - T) * 2 * f[k-j] (for valid k-j)

This is a weighted sum of f values, where the weights emphasize the "hot" lags. The gradient points in the direction that most increases J, so we want to go in the negative gradient direction.

The intuition: if C[k] is high, we want to reduce the contributions to C[k], which means reducing f[i]*f[k-i] for various i. The gradient tells us which f[j] to reduce.

This is similar to the "water filling" approach in communications.

```python
def gradient_excess(f, T):
    """Gradient of J_T(f) = sum_k [max(0, conv(f,f)[k] - T)]^2"""
    n = len(f)
    C = np.convolve(f, f)
    excess = np.maximum(0, C - T)
    # grad[j] = 4 * sum_k excess[k] * f[k-j] for valid k-j
    # = 4 * (excess convolved with f, but only the "valid" part)
    # Actually: sum_k excess[k] * f[k-j] for 0<=k-j<n
    # = sum_{k=j}^{j+n-1} excess[k] * f[k-j]
    # = sum_{m=0}^{n-1} excess[j+m] * f[m]
    # This is a cross-correlation of excess and f
    # np.correlate(excess, f, 'valid') gives sum_m excess[m+i]*f[m] for i=0..len(excess)-len(f)
    # excess has length 2n-1, f has length n
    # So 'valid' gives (2n-1)-n+1 = n values
    # That's exactly what we want!
    grad = 4.0 * np.correlate(excess, f, mode='valid')
    return grad
```

```python
"""
Optimization of non-negative sequence to minimize 2n*max(conv(f,f))/sum(f)^2.
Uses excess-energy gradient, softmax subgradient, and simplex projection.
Focuses on n=1024-4096 for best results.
"""
import numpy as np
import time

def project_simplex(f):
    """Project f onto {x >= 0, sum(x) = 1} using Euclidean projection."""
    n = len(f)
    if np.sum(f) < 1e-30:
        return np.ones(n) / n
    # Bisection for tau
    lo, hi = f.min() - 1, 0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.maximum(f - mid, 0))
        if s > 1:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.maximum(f - tau, 0)

def conv_and_max(f):
    """Compute convolution and its max."""
    C = np.convolve(f, f)
    return C, C.max()

def grad_excess(f, C, T):
    """Gradient of sum_k [max(0, C[k]-T)]^2 w.r.t. f."""
    excess = np.maximum(0, C - T)
    if excess.max() < 1e-15:
        return np.zeros_like(f)
    return 4.0 * np.correlate(excess, f, mode='valid')

def grad_softmax(f, C, beta):
    """Gradient of (1/beta)*log(sum exp(beta*C)) w.r.t. f."""
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros_like(f)
    p = exp_C / total
    return 2.0 * np.correlate(p, f, mode='valid')

def optimize_excess(f, deadline, T_start_frac=0.8, T_end_frac=1.01, 
                     lr=0.01, max_iters=50000, verbose=False):
    """
    Optimize using excess-energy gradient with adaptive threshold.
    T goes from T_start_frac*max_C to T_end_frac*mean_C.
    """
    f = f.copy().astype(np.float64)
    f = project_simplex(f)
    n = len(f)
    
    best_f = f.copy()
    C0, max0 = conv_and_max(f)
    best_score = 2.0 * n * max0  # since sum=1
    
    if verbose:
        print(f"  Excess opt start: score={best_score:.6f}, n={n}")
    
    m = np.zeros(n)  # Adam momentum
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    no_improve = 0
    
    while time.monotonic() < deadline and t < max_iters:
        t += 1
        C, max_C = conv_and_max(f)
        mean_C = 1.0 / (2*n - 1)  # since sum(f)=1, sum(C)=1
        
        # Adaptive threshold
        frac = min(1.0, t / (max_iters * 0.8))
        T = T_start_frac * (max_C - mean_C) + mean_C
        T = T * (1 - frac) + T_end_frac * mean_C * frac
        # Actually let's be more explicit:
        # Early: T is high (only reduce the worst peaks)
        # Late: T is closer to mean (flatten everything)
        T = (1 - frac) * (mean_C + 0.7 * (max_C - mean_C)) + frac * mean_C * 1.001
        
        g = grad_excess(f, C, T)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        # Adaptive learning rate
        adaptive_lr = lr / (1 + 0.01 * t)
        new_f = f - adaptive_lr * update
        new_f = project_simplex(new_f)
        
        C_new, max_new = conv_and_max(new_f)
        score_new = 2.0 * n * max_new
        
        if score_new < best_score:
            f = new_f
            best_f = f.copy()
            best_score = score_new
            no_improve = 0
            if verbose and t % 5000 == 0:
                print(f"    Excess t={t}: score={best_score:.6f}, T={T:.6f}")
        else:
            no_improve += 1
            if no_improve > 5000:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-8:
                    break
    
    return best_f, best_score

def optimize_subgradient(f, deadline, K=5, lr=0.001, max_iters=10000, verbose=False):
    """
    Subgradient method at top-K lags.
    """
    f = f.copy().astype(np.float64)
    f = project_simplex(f)
    n = len(f)
    
    best_f = f.copy()
    C0, max0 = conv_and_max(f)
    best_score = 2.0 * n * max0
    
    if verbose:
        print(f"  Subgradient start: score={best_score:.6f}")
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    no_improve = 0
    current_lr = lr
    
    while time.monotonic() < deadline and t < max_iters:
        t += 1
        C, max_C = conv_and_max(f)
        
        # Find top-K lags
        top_k_idx = np.argsort(C)[-K:]
        # Weights: proportional to (C[k] - mean_C) for top-K
        mean_C = 1.0 / (2*n - 1)
        weights = np.maximum(0, C[top_k_idx] - mean_C)
        if weights.sum() < 1e-15:
            weights = np.ones(K) / K
        else:
            weights = weights / weights.sum()
        
        # Gradient: sum of subgradients at top-K lags
        # Subgradient at lag k: 2*f[k-j] for valid j
        g = np.zeros(n)
        for wk, k in zip(weights, top_k_idx):
            # f[k-j] for j=0..n-1, valid when 0<=k-j<n
            j_min = max(0, k - (n-1))
            j_max = min(k, n-1)
            if j_min <= j_max:
                g[j_min:j_max+1] += 2.0 * wk * f[k - np.arange(j_min, j_max+1)]
        
        # Adam
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        new_f = f - current_lr * update
        new_f = project_simplex(new_f)
        
        C_new, max_new = conv_and_max(new_f)
        score_new = 2.0 * n * max_new
        
        if score_new < best_score:
            f = new_f
            best_f = f.copy()
            best_score = score_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                current_lr *= 0.5
                no_improve = 0
                if current_lr < 1e-10:
                    break
    
    return best_f, best_score

def raised_cosine(n, width=0.5):
    """Raised cosine window."""
    x = np.arange(n) / (n - 1)
    # Center at 0.5, width controls how much is raised
    w = 0.5 + 0.5 * np.cos(2 * np.pi * (x - 0.5)) ** (1/width)
    return w

def super_gaussian(n, alpha=2.0):
    """Super-Gaussian: exp(-(x/σ)^(2α))"""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n / 4.0
    w = np.exp(-((x - center) / sigma) ** (2 * alpha))
    return w

def two_hump(n, gap_frac=0.2):
    """Two-hump initialization: energy at edges and center."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n * gap_frac
    w = np.exp(-((x - 0) / (sigma*2))**2) + np.exp(-((x - (n-1)) / (sigma*2))**2) + 0.5*np.exp(-((x - center) / sigma)**2)
    return w
```

Multi-phase optimization minimizing 2n*max(conv(f,f))/sum(f)^2 for non-negative f.
Uses excess-energy gradient, softmax subgradient, LP direction finding,
and simplex projection. Focuses on n=1024-4096 for best theoretical gains.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def project_simplex(f):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(f)
    if np.sum(f) < 1e-30:
        return np.ones(n) / n
    if np.all(f >= -1e-12) and abs(np.sum(f) - 1.0) < 1e-8:
        return np.maximum(f, 0) / np.sum(f)
    lo = np.min(f) - 1.0
    hi = np.max(f) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(f - mid, 0))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    result = np.maximum(f - tau, 0)
    s = np.sum(result)
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(n) / n
    return result


def conv_and_max(f):
    """Compute linear convolution of f with itself and return (conv, max)."""
    C = np.convolve(f, f)
    return C, float(C.max())


def grad_excess(f, C, T):
    """Gradient of sum_k [max(0, C[k]-T)]^2 w.r.t. f (up to constant)."""
    excess = np.maximum(0, C - T)
    if excess.max() < 1e-15:
        return np.zeros_like(f)
    return 4.0 * np.correlate(excess, f, mode='valid')


def grad_softmax(f, C, beta):
    """Gradient of soft-max approximation to max(C) w.r.t. f (up to constant)."""
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros_like(f)
    p = exp_C / total
    return 2.0 * np.correlate(p, f, mode='valid')
