import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def softmax_objective(a, T):
    """Smooth approximation of J using softmax of convolution peak."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(a)
    a_norm = a / s
    b = fast_conv(a_norm)
    # Softmax smooth approximation of max
    b_shifted = b - b.max()
    log_sum_exp = b.max() + np.log(np.sum(np.exp(T * b_shifted)) / len(b))
    J = 2 * len(a) * log_sum_exp
    # Gradient (approximate)
    # dJ/da_j = (2n/s^2) * [max_b - sum_k c_k * s_k / sum(s_k)] * something
    # This is complex; let's use numerical gradient or a simpler approach
    return J, None

def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            # a[k-j] for j in [jl, jh]
            idx = k - np.arange(jl, jh + 1)
            vals.extend(a[idx].tolist())
    A = lil_matrix((2 * n - 1, n))
    for i in range(len(rows)):
        A[rows[i], cols[i]] = vals[i]
    return A.tocsr()

def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def lp_mirr_descent(a, max_iters, deadline, tracker, tol=1e-10):
    """LP mirror descent with adaptive step and convergence detection."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    patience = 0
    best_local_J = current_J
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Try multiple step sizes
        base_t = 0.3 * (1.0 + np.cos(np.pi * it / max(max_iters, 1))) / 2
        candidates_t = [base_t, min(1.0, base_t * 2), max(0.05, base_t * 0.5), 1.0]
        
        best_a = None
        best_J = current_J
        
        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-12:
                best_J = J_trial
                best_a = a_trial
        
        if best_a is None:
            patience += 1
            if patience > 15:
                break
        else:
            a = best_a
            current_J = best_J
            patience = 0
            if current_J < best_local_J - 1e-10:
                best_local_J = current_J
                if it % 50 == 0 or it < 5:
                    tracker.update(a)
    
    tracker.update(a)
    return a

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    inits.append(np.exp(-20 * (tm - 0.25) ** 2) + np.exp(-20 * (tm - 0.75) ** 2) + 0.01)
    
    # Random
    for _ in range(3):
        inits.append(rng.uniform(0.2, 1.0, n))
    
    # Random smooth
    for _ in range(3):
        freqs = rng.uniform(1, 8, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        amps = rng.uniform(0.1, 0.4, size=5)
        sig = np.ones(n)
        for f, p, a in zip(freqs, phases, amps):
            sig += a * np.cos(2 * np.pi * f * tm + p)
        inits.append(np.maximum(sig, 0.01))
    
    # Spike-based (sparse with smoothing)
    for nspike in [4, 8, 16]:
        spikes = np.zeros(n)
        positions = rng.choice(n, size=nspike, replace=False)
        spikes[positions] = 1.0
        kernel_size = max(3, n // 16)
        kernel = np.exp(-0.5 * (np.arange(-kernel_size//2, kernel_size//2+1) / (kernel_size//4)) ** 2)
        kernel /= kernel.sum()
        inits.append(np.convolve(spikes, kernel, mode='same') + 0.02)
    
    return inits

def smooth_gradient_descent(a, max_iters, deadline, tracker, T_start=10, T_end=500):
    """Refine using smooth objective with increasing T (continuation)."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    T = T_start
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Gradually increase T
        T = T_start * (T_end / T_start) ** (it / max(max_iters, 1))
        
        # Compute smooth objective and gradient numerically
        a_pos = np.maximum(a, 1e-8)
        a_norm = a_pos / a_pos.sum()
        b = fast_conv(a_norm)
        b_max = b.max()
        b_shifted = T * (b - b_max)
        weights = np.exp(b_shifted)
        weights = weights / weights.sum()
        J_smooth = 2 * n * (b_max + np.log(weights.sum() / len(b)) / T)
        
        # Numerical gradient (central differences on a few coordinates)
        # Actually, let's use a more efficient approach
        # The gradient of max_k c[k] w.r.t. a[j] where c = a*a
        # If k* is the argmax, then d c[k*]/d a[j] = 2 * a[k*-j] (for valid index)
        # But with softmax, it's a weighted average
        
        # Compute gradient of smooth max w.r.t. a
        # d/da_j [sum_k w_k * c_k] where w_k = exp(T*(c_k - c_max)) / sum(exp(...))
        # = sum_k w_k * d c_k / d a_j
        # d c_k / d a_j = a[k-j] + a[j-k] (if indices valid)
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if weights[k] < 1e-10:
                continue
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                partner = k - j
                if 0 <= partner < n:
                    grad[j] += weights[k] * a_norm[partner]
        
        grad *= 2 * n / (a_norm.sum() ** 2)  # Account for normalization (approximate)
        
        # Actually, let me just do a simpler gradient step
        # Step in direction -grad with momentum
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        # Check if it improved
        J_new = compute_J(a_new)
        J_old = compute_J(a)
        if J_new < J_old:
            a = a_new
            if it % 20 == 0:
                tracker.update(a)
        else:
            lr *= 0.95
            v *= 0.5
        
        # Periodically reset T and learning rate
        if it % 50 == 0 and it > 0:
            lr = max(0.001, 0.01 * (1 - it / max_iters))
    
    tracker.update(a)
    return a

def lbfgs_refine(a, deadline, tracker, max_fun=20000):
    """L-BFGS-B refinement on the smooth objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    
    def objective(x):
        ax = np.maximum(x, 0)
        s = ax.sum()
        if s < 1e-8:
            return 1e10
        ax = ax / s
        # Use softmax with moderate T for smoothness
        b = fast_conv(ax)
        T = 50
        b_shifted = T * (b - b.max())
        lse = b.max() + np.log(np.sum(np.exp(b_shifted)) / len(b)) / T
        return 2 * n * lse
    
    def gradient(x):
        ax = np.maximum(x, 0)
        s = ax.sum()
        if s < 1e-8:
            return np.zeros(n)
        ax = ax / s
        b = fast_conv(ax)
        T = 50
        b_shifted = T * (b - b.max())
        w = np.exp(b_shifted)
        w = w / w.sum()
        
        # Gradient of sum_k w_k * c_k w.r.t. a_j
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                partner = k - j
                if 0 <= partner < n:
                    grad[j] += w[k] * ax[partner]
        
        return 2 * n * grad / (s * s)
    
    bounds_list = [(0.0, None)] * n
    try:
        res = minimize(objective, a, method='L-BFGS-B', 
                      jac=gradient,
                      bounds=bounds_list,
                      options={'maxiter': 200, 'max_fun': max_fun, 'ftol': 1e-15, 'gtol': 1e-10})
        if res.fun < 1e10:
            tracker.update(res.x)
            return res.x
    except Exception as e:
        print(f"  L-BFGS-B error: {e}", flush=True)
    
    return a

def sa_lp_search(a_start, n_restarts, deadline, tracker):
    """Simulated annealing with LP-assisted moves."""
    a = a_start.astype(np.float64).copy()
    a = a / a.sum()
    current_J = compute_J(a)
    
    temperature = 0.1
    best_local = a.copy()
    best_local_J = current_J
    
    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        
        # Perturb
        perturbation = rng_global.normal(0, temperature, a.size)
        a_trial = a + perturbation
        a_trial = np.maximum(a_trial, 1e-6)
        a_trial = a_trial / a_trial.sum()
        
        # Run a few LP iterations
        a_trial = run_short_lp(a_trial, min(10, 30 - r // 2), deadline)
        
        J_trial = compute_J(a_trial)
        
        # Accept or reject
        if J_trial < current_J or rng_global.random() < np.exp(-(J_trial - current_J) / temperature):
            a = a_trial
            current_J = J_trial
            if current_J < best_local_J:
                best_local = a.copy()
                best_local_J = current_J
                tracker.update(a)
        
        temperature *= 0.96
    
    return best_local

Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse initialization (window functions, analytical profiles, parent)
2. LP-based mirror descent with adaptive steps and convergence detection
3. Frank-Wolfe (conditional gradient) refinement

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            idx = k - np.arange(jl, jh + 1)
            vals.extend(a[idx].tolist())
    A = lil_matrix((2 * n - 1, n))
    for i in range(len(rows)):
        A[rows[i], cols[i]] = vals[i]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)

    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lp_mirror_descent(a, max_iters, deadline, tracker=None, tol=1e-12):

def frank_wolfe_refine(a, max_iters, deadline, tracker=None):
    """Frank-Wolfe (conditional gradient) for minimax autocorrelation."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = len(a)
    current_J = compute_J(a)
    if tracker is not None:
        tracker.update(a)

    patience = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        k_star = int(np.argmax(b))

        # Gradient of c[k_star] w.r.t. a[j] = a[k_star - j] (for valid j)
        # Directional derivative in direction e_j - a is:
        #   grad_j - mean(grad)
        # where grad_j = a[k_star - j] + a[j - k_star] (if valid, else 0)
        grad = np.zeros(n)
        jl = max(0, k_star - n + 1)
        jh = min(n - 1, k_star)
        for j in range(jl, jh + 1):
            partner = k_star - j
            if 0 <= partner < n:
                grad[j] += a[partner]
        # Also the symmetric part if j - k_star is valid
        jl2 = max(0, k_star)
        jh2 = min(n - 1, k_star + n - 1)
        for j in range(jl2, jh2 + 1):
            partner = j - k_star
            if 0 <= partner < n and j != k_star:
                grad[j] += a[partner]

        # Frank-Wolfe: pick j with minimum directional derivative
        # dir_grad[j] = grad[j] - dot(grad, a)
        mean_grad = np.dot(grad, a)
        dir_grad = grad - mean_grad
        j_min = int(np.argmin(dir_grad))

        # Step size: use 2/(it+2) (standard FW) or line search
        t_fw = 2.0 / (it + 2.0)

        # Move towards e_{j_min}
        a_new = (1.0 - t_fw) * a + t_fw * np.zeros(n)
        a_new[j_min] += t_fw
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-12:
            a = a_new
            current_J = J_new
            patience = 0
        else:
            patience += 1
            if patience > 30:
                break
            # Try smaller step
            t_small = 0.05 / (it + 1)
            a_small = (1.0 - t_small) * a
            a_small[j_min] += t_small
            a_small = np.maximum(a_small, 0.0)
            s_small = a_small.sum()
            if s_small >= 1e-10:
                a_small = a_small / s_small
                J_small = compute_J(a_small)
                if J_small < current_J - 1e-12:
                    a = a_small
                    current_J = J_small
                    patience = 0

        if tracker is not None and it % 30 == 0:
            tracker.update(a)

    if tracker is not None:
        tracker.update(a)
    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initial sequences for optimization."""
    inits = []
    tm = np.linspace(0, 1, n)  # includes endpoints

    if parent is not None:
        inits.append(resample_seq(parent, n))

    # Classic window functions
    inits.append(np.ones(n))
    inits.append(np.sin(np.pi * tm) ** 2)
    inits.append(np.maximum(0.54 - 0.46 * np.cos(2 * np.pi * tm), 0.01))
    inits.append(np.maximum(0.42 - 0.5 * np.cos(2 * np.pi * tm) + 0.08 * np.cos(4 * np.pi * tm), 0.01))
    inits.append(np.exp(-6.0 * (tm - 0.5) ** 2) + 0.03)
    inits.append(1.0 - np.abs(2 * tm - 1))
    inits.append(np.maximum(1.0 - (2 * tm - 1) ** 4, 0.01))
    inits.append(np.sqrt(np.maximum(1.0 - (2 * tm - 1) ** 2, 0.0)) + 0.02)
    inits.append(np.maximum(1.0 + 0.25 * np.cos(2 * np.pi * tm), 0.01))
    inits.append(np.maximum(1.0 + 0.15 * np.cos(4 * np.pi * tm), 0.01))
    inits.append(np.maximum(1.0 + 0.1 * np.cos(6 * np.pi * tm), 0.01))

    # Power-law tapers
    for alpha in [0.3, 0.5, 0.7, -0.3, -0.5, 1.0]:
        inits.append((tm + 0.02) ** alpha)

    # Asymmetric
    inits.append((1.0 - tm) ** 1.5 + 0.03)
    inits.append(tm ** 1.5 + 0.03)
    inits.append(np.exp(-4.0 * tm) + 0.02)
    inits.append(np.exp(-4.0 * (1.0 - tm)) + 0.02)

    # Two peaks
    inits.append(np.exp(-40 * (tm - 0.3) ** 2) + np.exp(-40 * (tm - 0.7) ** 2) + 0.02)
    inits.append(np.exp(-25 * (tm - 0.25) ** 2) + np.exp(-25 * (tm - 0.75) ** 2) + 0.02)
    inits.append(np.exp(-15 * (tm - 0.2) ** 2) + np.exp(-15 * (tm - 0.5) ** 2) + np.exp(-15 * (tm - 0.8) ** 2) + 0.01)

    # Random smooth signals
    for _ in range(4):
        freqs = rng.uniform(1, 10, size=4)
        phases = rng.uniform(0, 2 * np.pi, size=4)
        amps = rng.uniform(0.05, 0.35, size=4)
        sig = np.ones(n)
        for f, p, amp in zip(freqs, phases, amps):
            sig = sig + amp * np.cos(2 * np.pi * f * tm + p)
        inits.append(np.maximum(sig, 0.02))

    # Random uniform
    for _ in range(3):
        inits.append(rng.uniform(0.1, 1.0, n))

    # Sparse with Gaussian smoothing
    for nspike in [4, 8, 16, 32]:
        spikes = np.zeros(n)
        positions = rng.choice(n, size=nspike, replace=False)
        spikes[positions] = 1.0
        ksize = max(5, n // 12)
        kx = np.arange(-ksize // 2, ksize // 2 + 1)
        kernel = np.exp(-0.5 * (kx / (ksize / 4.0)) ** 2)
        kernel = kernel / kernel.sum()
        smoothed = np.convolve(spikes, kernel, mode='same')
        inits.append(smoothed + 0.03)

    # Cosine-modulated (Kaiser-like)
    beta = rng.uniform(4, 12)
    inits.append(np.i0(beta * np.sqrt(np.maximum(1 - (2 * tm - 1) ** 2, 0))) / np.i0(beta) + 0.01)

    return inits


def lbfgs_smooth_refine(a, deadline, tracker, T_start=20.0, T_end=200.0, max_iter_phases=3):
    """L-BFGS-B on softmax-smoothed approximation with continuation on T."""
    n = len(a)
    a_best = a.astype(np.float64).copy()

    for phase in range(max_iter_phases):
        if time.monotonic() > deadline:
            break
        T = T_start * (T_end / T_start) ** (phase / max(max_iter_phases - 1, 1))

        def obj(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-8:
                return 1e10
            ax = ax / s
            b = fast_conv(ax)
            b_max = b.max()
            b_shift = T * (b - b_max)
            lse = b_max + np.log(np.sum(np.exp(b_shift)) / len(b)) / T
            return 2 * n * lse

        def grad(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-8:
                return np.zeros(n)
            ax = ax / s
            b = fast_conv(ax)
            b_max = b.max()
            b_shift = T * (b - b_max)
            w = np.exp(b_shift)
            w = w / w.sum()
            # grad of sum_k w_k * c_k w.r.t. a_j
            g = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-12:
                    continue
                jl = max(0, k - n + 1)
                jh = min(n - 1, k)
                for j in range(jl, jh + 1):
                    partner = k - j
                    if 0 <= partner < n:
                        g[j] += w[k] * ax[partner]
            return 2 * n * g / (s * s)

        bounds_list = [(0.0, None)] * n
        try:
            res = minimize(obj, a_best, method='L-BFGS-B', jac=grad,
                           bounds=bounds_list,
                           options={'maxiter': min(100, 300 - 100 * phase),
                                    'max_fun': 8000, 'ftol': 1e-14, 'gtol': 1e-8})
            if res.fun < 1e10:
                J_res = compute_J(res.x)
                J_best = compute_J(a_best)
                if J_res < J_best:
                    a_best = res.x
                    if tracker is not None:
                        tracker.update(a_best)
        except Exception:
            pass

    if tracker is not None:
        tracker.update(a_best)
    return a_best


def run_short_lp(a, max_iters, deadline):
    """Quick LP iterations without tracking."""
    a = np.maximum(a.astype(np.float64), 1e-8)
    a = a / a.sum()
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        t = 0.25
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        if compute_J(a_new) < compute_J(a) - 1e-12:
            a = a_new
    return a


def sa_with_lp(a_start, n_restarts, deadline, tracker, rng):
    """Simulated annealing with LP-assisted short runs."""
    a = a_start.astype(np.float64).copy()
    a = a / a.sum()
    current_J = compute_J(a)
    best_local = a.copy()
    best_local_J = current_J
    temperature = 0.12

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break

        # Perturb with decreasing temperature
        noise = rng.normal(0, temperature, a.size)
        a_trial = a + noise
        a_trial = np.maximum(a_trial, 1e-6)
        a_trial = a_trial / a_trial.sum()

        # Short LP refinement
        n_lp = max(3, 15 - r // 5)
        a_trial = run_short_lp(a_trial, n_lp, deadline)

        J_trial = compute_J(a_trial)

        # Metropolis acceptance
        if J_trial < current_J or rng.random() < np.exp(-(J_trial - current_J) / max(temperature, 1e-6)):
            a = a_trial
            current_J = J_trial
            if current_J < best_local_J:
                best_local = a.copy()
                best_local_J = current_J
                if tracker is not None:
                    tracker.update(a)

        temperature *= 0.955

    return best_local


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization combining diverse LP search, Frank-Wolfe,
    L-BFGS-B smooth refinement, and simulated annealing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Diverse LP search at n=128 ===
    n = 128
    t0 = time.monotonic()
    print(f"\n{'='*50} Phase 1: LP Diverse n={n} {'='*50}", flush=True)

    inits = generate_diverse_inits(n, rng, parent)
    print(f"  {len(inits)} initializations", flush=True)

    iters_per_init = max(50, min(200, int(100000 // n)))

    for idx, a0 in enumerate(inits):
        if time.monotonic() > deadline:
            break
        a0 = np.maximum(a0.astype(np.float64), 1e-8)
        s0 = a0.sum()
        if s0 < 1e-10:
            continue
        a0 = a0 / s0
        a_result = lp_mirror_descent(a0, iters_per_init, deadline, tracker)
        if idx % 15 == 0:
            print(f"  [{idx}/{len(inits)}] t={time.monotonic()-t0:.0f}s J={tracker.best_score:.8f}", flush=True)

    tracker.report(f"Phase 1 ({time.monotonic()-t0:.0f}s)")

    # === Phase 2: Frank-Wolfe on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 2: Frank-Wolfe {'='*50}", flush=True)
        a_fw = frank_wolfe_refine(tracker.best, 200, deadline, tracker)
        # Then LP polish on FW result
        if tracker.best is not None:
            a_polished = lp_mirror_descent(a_fw, 80, deadline, tracker)
        tracker.report(f"Phase 2 ({time.monotonic()-t0:.0f}s)")

    # === Phase 3: L-BFGS-B smooth refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 3: L-BFGS-B Smooth {'='*50}", flush=True)
        a_lbfgs = lbfgs_smooth_refine(tracker.best, deadline, tracker,
                                       T_start=15.0, T_end=150.0, max_iter_phases=3)
        # Polish with LP
        if tracker.best is not None and time.monotonic() < deadline:
            lp_mirror_descent(a_lbfgs, 60, deadline, tracker)
        tracker.report(f"Phase 3 ({time.monotonic()-t0:.0f}s)")

    # === Phase 4: Try n=256 ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n2 = 256
        print(f"\n{'='*50} Phase 4: LP n={n2} {'='*50}", flush=True)
        inits2 = generate_diverse_inits(n2, rng, tracker.best)
        iters2 = max(30, min(100, int(60000 // n2)))
        for idx, a0 in enumerate(inits2[:20]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 1e-8)
            s0 = a0.sum()
            if s0 < 1e-10:
                continue
            a0 = a0 / s0
            lp_mirror_descent(a0, iters2, deadline, tracker)
            if idx % 10 == 0:
                print(f"  [{idx}/{min(20,len(inits2))}] t={time.monotonic()-t0:.0f}s", flush=True)
        tracker.report(f"Phase 4 ({time.monotonic()-t0:.0f}s)")

    # === Phase 5: Simulated Annealing ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 5: SA {'='*50}", flush=True)
        remaining = deadline - time.monotonic()
        n_sa = int(min(80, max(30, remaining / 8)))
        a_sa = sa_with_lp(tracker.best, n_sa, deadline, tracker, rng)
        # Final LP on SA best
        if tracker.best is not None and time.monotonic() < deadline:
            lp_mirror_descent(a_sa, 100, deadline, tracker)
        tracker.report(f"Phase 5 ({time.monotonic()-t0:.0f}s)")

    # === Phase 6: Final polish cycle ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 6: Final Polish {'='*50}", flush=True)
        # LP
        a1 = lp_mirror_descent(tracker.best, 150, deadline, tracker)
        # FW
        if time.monotonic() < deadline:
            a2 = frank_wolfe_refine(a1, 100, deadline, tracker)
            # LP again
            if time.monotonic() < deadline:
                lp_mirror_descent(a2, 80, deadline, tracker)
        # Smooth L-BFGS
        if time.monotonic() < deadline:
            lbfgs_smooth_refine(tracker.best, deadline, tracker,
                                T_start=50.0, T_end=300.0, max_iter_phases=2)
        tracker.report(f"Phase 6 ({time.monotonic()-t0:.0f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
