def raised_cosine(n, power=2.0):
    """Generate a raised cosine window with configurable power."""
    x = np.arange(n, dtype=np.float64)
    t = np.pi * x / (n - 1) - np.pi / 2  # range [-pi/2, pi/2]
    w = np.cos(t) ** power
    return w


def generalized_hamming(n, alpha=1.0):
    """Generalized Hamming window."""
    w = np.ones(n)
    m = n // 2
    for i in range(m):
        t = (i + 0.5) / m
        # Blend between rectangular (alpha=0) and Hann (alpha=1) and Blackman (alpha~2)
        w[i] = 0.5 * (1 + (alpha * np.cos(2*np.pi*t/(n-1)) + (1-alpha) * 1))
        w[n-1-i] = w[i]
    return w

"""
Multi-method optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses multiplicative (exponentiated gradient) descent, hard subgradient descent,
active-set refinement, spectral domain design, and multi-start strategy.
Key differences from prior approach: multiplicative updates (natural non-negativity),
hard subgradient on active set, spectral flattening, and two-scale optimization.
"""
import numpy as np
import time
import math


def compute_grad_soft(f, beta):
    """Compute soft-max gradient of max(conv(f,f)). Returns (grad, soft_max_val)."""
    C = np.convolve(f, f)
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros(len(f)), 0.0
    p = exp_C / total
    corr = np.correlate(p, f, mode='valid')
    grad = 2.0 * corr
    soft_val = max_C + np.log(total) / beta
    return grad, soft_val


def get_active_lags(C, top_k=5, threshold_frac=0.98):
    """Get lags near the maximum of convolution."""
    max_C = C.max()
    threshold = threshold_frac * max_C
    active = np.where(C >= threshold)[0]
    top_indices = np.argsort(C)[::-1][:top_k]
    combined = np.unique(np.concatenate([active, top_indices]))
    return combined


def multiplicative_descent(f, deadline, beta_start=10.0, beta_end=500.0,
                           lr_start=0.5, lr_end=0.01, max_iters=10000,
                           verbose_prefix=""):
    """Multiplicative (exponentiated gradient) descent with soft-max surrogate."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    it = 0
    no_improve = 0
    lr = lr_start
    step_clip = 1.5

    while time.monotonic() < deadline and it < max_iters:
        it_frac = min(1.0, it / (max_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** it_frac
        lr = lr_start * (lr_end / lr_start) ** it_frac

        grad, _ = compute_grad_soft(f, beta)

        step = np.clip(lr * grad, -step_clip, step_clip)
        f_new = f * np.exp(-step)
        f_new = np.clip(f_new, 1e-15, 1000.0)

        S_new = float(np.sum(f_new))
        it += 1
        if S_new < 1e-10:
            no_improve += 1
            if no_improve > 200:
                break
            continue
        f_new = f_new / S_new

        sc = evaluate_sequence(f_new.tolist())

        if np.isfinite(sc) and sc < best_score:
            f = f_new
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            if it % 1000 == 0:
                print(f"    {verbose_prefix}MD iter {it}: score={best_score:.6f} (beta={beta:.1f}, lr={lr:.4f})")
        else:
            no_improve += 1
            if no_improve > 300:
                step_clip *= 0.7
                no_improve = 0
                if step_clip < 0.01:
                    break

    return best_f, best_score


def subgradient_descent(f, deadline, lr_start=0.1, lr_end=0.005,
                        top_k=3, threshold_frac=0.99, max_iters=10000,
                        verbose_prefix=""):
    """Hard subgradient descent on the max of convolution (active set method)."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    it = 0
    no_improve = 0
    lr = lr_start

    while time.monotonic() < deadline and it < max_iters:
        it_frac = min(1.0, it / (max_iters * 0.7))
        lr = lr_start * (lr_end / lr_start) ** it_frac

        C = np.convolve(f, f)
        active = get_active_lags(C, top_k=top_k, threshold_frac=threshold_frac)

        # Uniform average over active lags (subgradient of max)
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / len(active)
        grad = 2.0 * np.correlate(p, f, mode='valid')

        # Projected gradient step (additive with projection to non-negative)
        f_new = f - lr * grad
        f_new = np.clip(f_new, 0.0, 1000.0)
        S_new = float(np.sum(f_new))
        it += 1
        if S_new < 1e-10:
            no_improve += 1
            lr *= 0.5
            if no_improve > 200 or lr < 1e-7:
                break

        f_new = f_new / S_new

        sc = evaluate_sequence(f_new.tolist())

        if np.isfinite(sc) and sc < best_score:
            f = f_new
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            if it % 1000 == 0:
                print(f"    {verbose_prefix}SG iter {it}: score={best_score:.6f} (lr={lr:.5f}, active={len(active)})")
        else:
            no_improve += 1
            if no_improve > 200:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-7:
                    break

    return best_f, best_score


def blended_active_descent(f, deadline, beta_base=50.0, top_k=3,
                           lr_start=0.3, lr_end=0.01, max_iters=10000,
                           verbose_prefix=""):
    """Blend between soft-max (broad) and hard max (focused) using adaptive mixing."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)

    it = 0
    no_improve = 0
    lr = lr_start

        it_frac = min(1.0, it / (max_iters * 0.7))
        lr = lr_start * (lr_end / lr_start) ** it_frac

        C = np.convolve(f, f)
        max_C = C.max()
        active = get_active_lags(C, top_k=top_k, threshold_frac=0.97)

        # Blended probability: mix of uniform-over-active and soft-max
        # Sharp soft-max on active set
        active_vals = C[active]
        beta_eff = beta_base * (1.0 + 5.0 * it_frac)  # sharpen over time
        exp_vals = np.exp(beta_eff * (active_vals - active_vals.max()))
        p_active = exp_vals / exp_vals.sum()

        p = np.zeros(2 * n - 1)
        p[active] = p_active

        grad = 2.0 * np.correlate(p, f, mode='valid')

        # Multiplicative update
        step = np.clip(lr * grad, -1.0, 1.0)
        f_new = f * np.exp(-step)
        f_new = np.clip(f_new, 1e-15, 1000.0)
        S_new = float(np.sum(f_new))
        it += 1
        if S_new < 1e-10:
            no_improve += 1
            if no_improve > 200:
                break
            continue
        f_new = f_new / S_new

        sc = evaluate_sequence(f_new.tolist())

        if np.isfinite(sc) and sc < best_score:
            f = f_new
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            if it % 1000 == 0:
                print(f"    {verbose_prefix}BA iter {it}: score={best_score:.6f} (beta={beta_eff:.1f}, active={len(active)})")
        else:
            no_improve += 1
            if no_improve > 300:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-6:
                    break

    return best_f, best_score


def spectral_flatten(f, deadline, verbose_prefix=""):
    """Iteratively flatten the power spectrum in frequency domain."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    it = 0
    lr = 0.2

    while time.monotonic() < deadline and it < 500:
        F = np.fft.fft(f, n=2 * n)  # Use zero-padded FFT for circular conv approximation
        mag = np.abs(F)
        phase = np.angle(F)

        # Target: flat magnitude (white spectrum -> flat autocorrelation)
        # But only modify the positive-frequency half meaningfully
        target_mag = np.ones_like(mag) * mag.mean()

        # Gentle blend
        new_mag = (1.0 - lr) * mag + lr * target_mag
        new_F = new_mag * np.exp(1j * phase)
        new_f = np.real(np.fft.ifft(new_F, n=2 * n))[:n]  # Take first n samples

        # Project to non-negative
        new_f = np.clip(new_f, 0.0, 1000.0)
        S_new = float(np.sum(new_f))
        it += 1
        if S_new < 1e-10:
            continue
        new_f = new_f / S_new

        sc = evaluate_sequence(new_f.tolist())

        if np.isfinite(sc) and sc < best_score:
            f = new_f
            best_f = f.copy()
            best_score = sc
            if it % 100 == 0:
                print(f"    {verbose_prefix}Spec iter {it}: score={best_score:.6f}")
        else:
            lr *= 0.7
            if lr < 1e-4:
                break

    return best_f, best_score


def coord_descent_step(f, lr=0.1):
    """One pass of cyclic coordinate descent on the soft-max objective."""
    n = len(f)
    C = np.convolve(f, f)
    beta = 50.0
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    p = exp_C / exp_C.sum()

    for i in range(n):
        old_val = f[i]
        # Gradient at coordinate i
        # dC[k]/df[i] = 2*f[k-i] for valid k, so grad[i] = 2*sum_k p[k]*f[k-i]
        # = 2 * np.correlate(p, f, mode='valid')[i]
        # But we already have p, so let's compute directly
        k_range = np.arange(2 * n - 1)
        valid = (k_range - i) >= 0
        valid &= (k_range - i) < n
        if not valid.any():
            continue
        grad_i = 2.0 * np.sum(p[valid] * f[k_range[valid] - i])
        f[i] = max(0.0, f[i] - lr * grad_i)

    S = float(np.sum(f))
    if S > 1e-10:
        f = f / S
    return f


def raised_cosine(n, power=2.0):
    """Raised cosine window with configurable power."""
    x = np.arange(n, dtype=np.float64)
    t = np.pi * x / (n - 1) - np.pi / 2
    w = np.cos(t) ** power
    return w


def super_gaussian(n, alpha=2.0):
    """Super-Gaussian window."""
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    w = np.exp(-((idx - center) / (n * 0.45)) ** (2 * alpha))
    return w


def flat_topped(n, w=0.5):
    """Flat-topped window: flat center, tapered edges."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    half_width = w * n / 2.0
    t = np.abs(x - center) / half_width
    w_out = np.ones(n)
    mask = t > 1.0
    t_ext = t[mask] - 1.0
    w_out[mask] = 0.5 * (1 + np.cos(np.pi * t_ext / (1.0 - w + 1e-10)))
    return w_out


def perturb_sequence(f, rng, mode='mul', strength=0.05):
    """Perturb a sequence in various ways."""
    n = len(f)
    if mode == 'mul':
        noise = 1.0 + strength * rng.standard_normal(n)
        new_f = f * np.clip(noise, 0.3, 2.0)
    elif mode == 'blend':
        new_w = raised_cosine(n, 1.5 + rng.random() * 2.0)
        new_w = new_w / new_w.sum()
        mix = 0.8 + 0.2 * rng.random()
        new_f = mix * f + (1 - mix) * new_w
    elif mode == 'shift':
        shift = int(rng.integers(3, max(4, n // 8)))
        new_f = np.roll(f, shift)
        new_f = np.clip(new_f, 0, None)
    else:
        new_f = f * np.clip(1.0 + strength * rng.standard_normal(n), 0.5, 1.5)
    return np.clip(new_f, 0.0, 1000.0)


def optimize_pipeline(f_init, deadline, verbose_prefix=""):
    """Run full optimization pipeline: MD -> SG -> blended -> spectral -> final MD."""
    f_init = f_init.copy().astype(np.float64)
    S = float(np.sum(f_init))
    if S < 1e-10:
        return f_init, evaluate_sequence(f_init.tolist())
    f_init = f_init / S

    best_f = f_init.copy()
    best_score = evaluate_sequence(f_init.tolist())
    t0 = time.monotonic()
    remaining = deadline - t0

    # Phase 1: Multiplicative descent (broad exploration)
    if remaining > 20:
        p1_dl = t0 + remaining * 0.30
        f, s = multiplicative_descent(f_init, p1_dl,
                                       beta_start=10.0, beta_end=300.0,
                                       lr_start=0.5, lr_end=0.02,
                                       verbose_prefix=verbose_prefix + "P1")
        if s < best_score:
            best_f, best_score = f, s

    # Phase 2: Subgradient descent (focused)
    if time.monotonic() < deadline - 10:
        p2_dl = t0 + remaining * 0.55
        f, s = subgradient_descent(best_f, p2_dl,
                                    lr_start=0.1, lr_end=0.005,
                                    top_k=3, threshold_frac=0.99,
                                    verbose_prefix=verbose_prefix + "P2")
        if s < best_score:
            best_f, best_score = f, s

    # Phase 3: Blended active descent (transitional)
    if time.monotonic() < deadline - 10:
        p3_dl = t0 + remaining * 0.75
        f, s = blended_active_descent(best_f, p3_dl,
                                       beta_base=50.0, top_k=4,
                                       lr_start=0.3, lr_end=0.01,
                                       verbose_prefix=verbose_prefix + "P3")
        if s < best_score:
            best_f, best_score = f, s

    # Phase 4: Spectral flattening
    if time.monotonic() < deadline - 10:
        p4_dl = t0 + remaining * 0.88
        f, s = spectral_flatten(best_f, p4_dl,
                                 verbose_prefix=verbose_prefix + "P4")
        if s < best_score:
            best_f, best_score = f, s

    # Phase 5: Final intensive MD with high beta
    if time.monotonic() < deadline - 3:
        f, s = multiplicative_descent(best_f, deadline - 3,
                                       beta_start=200.0, beta_end=2000.0,
                                       lr_start=0.1, lr_end=0.001,
                                       verbose_prefix=verbose_prefix + "P5")
        if s < best_score:
            best_f, best_score = f, s

    return best_f, best_score


def quick_optimize(f_init, deadline, verbose_prefix=""):
    """Quick 2-phase optimization for less promising initializations."""
    f_init = f_init.copy().astype(np.float64)
    S = float(np.sum(f_init))
    if S < 1e-10:
        return f_init, evaluate_sequence(f_init.tolist())
    f_init = f_init / S

    best_f = f_init.copy()
    best_score = evaluate_sequence(f_init.tolist())

    # Phase 1: MD
    if time.monotonic() < deadline - 5:
        f, s = multiplicative_descent(f_init, (time.monotonic() + deadline) / 2,
                                       beta_start=10.0, beta_end=200.0,
                                       lr_start=0.5, lr_end=0.05,
                                       verbose_prefix=verbose_prefix + "q1")
        if s < best_score:
            best_f, best_score = f, s

    # Phase 2: SG
    if time.monotonic() < deadline - 2:
        f, s = subgradient_descent(best_f, deadline - 2,
                                    lr_start=0.1, lr_end=0.01,
                                    top_k=3,
                                    verbose_prefix=verbose_prefix + "q2")
        if s < best_score:
            best_f, best_score = f, s

    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                parent = parent / S
                score = evaluate_sequence(parent.tolist())
                if np.isfinite(score):
                    best_score = score
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")
        print(e)

    # Determine primary length
    if best_sequence is not None:
        n_primary = len(best_sequence)
    else:
        n_primary = 512

    # Define lengths to try
    n_values = [n_primary]
    if n_primary <= 512:
        n_values.extend([768, 1024])
    elif n_primary <= 768:
        n_values.extend([512, 1024])
    elif n_primary <= 1024:
        n_values.extend([768, 2048])
    else:
        n_values.extend([n_primary // 2, n_primary * 2])
    n_values = [n for n in n_values if 128 <= n <= 4096]
    n_values = list(dict.fromkeys(n_values))

    total_budget = deadline - t0
    # Weight: primary gets 55%, others split remaining
    weights = {}
    weights[n_values[0]] = 0.55
    for nv in n_values[1:]:
        weights[nv] = 0.45 / max(1, len(n_values) - 1)

    print(f"\nPlanning: lengths={n_values}, budget={total_budget:.0f}s")

    for n in n_values:
        if time.monotonic() > deadline - 80:
            break

        n_budget = total_budget * weights.get(n, 0.1)
        n_deadline = time.monotonic() + n_budget
        print(f"\n{'='*60}")
        print(f"=== n={n} (budget {n_budget:.0f}s) ===")
        print(f"{'='*60}")

        # Generate diverse initializations
        inits = []

        # Parent (if available and matches length)
        if best_sequence is not None:
            if len(best_sequence) == n:
                inits.append(('parent', best_sequence.copy(), True))
            else:
                from numpy import interp
                p_interp = interp(np.linspace(0, 1, n),
                                  np.linspace(0, 1, len(best_sequence)),
                                  best_sequence)
                p_interp = np.clip(p_interp, 0, None)
                if p_interp.sum() > 0.01:
                    inits.append(('parent_interp', p_interp, True))

        # Various analytical windows
        inits.append(('rc2', raised_cosine(n, 2.0), False))
        inits.append(('rc25', raised_cosine(n, 2.5), False))
        inits.append(('rc3', raised_cosine(n, 3.0), False))
        inits.append(('sg3', super_gaussian(n, 3.0), False))
        inits.append(('sg4', super_gaussian(n, 4.0), False))
        inits.append(('ft05', flat_topped(n, 0.5), False))
        inits.append(('ft07', flat_topped(n, 0.7), False))

        # Random smooth
        for seed in [7, 13]:
            r_rng = np.random.default_rng(seed + n * 100)
            x = r_rng.standard_normal(n)
            kernel_size = max(5, n // 40)
            kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (kernel_size * 0.4)) ** 2)
            smooth_x = np.convolve(x, kernel, mode='same')
            smooth_x = np.abs(smooth_x) + 0.05
            inits.append((f'rand{seed}', smooth_x, False))

        # Sort initializations: parent first, then by initial score
        scored_inits = []
        for label, f_init, is_parent in inits:
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            S = float(np.sum(f_init))
            if S < 0.01:
                continue
            f_init = f_init / S
            sc = evaluate_sequence(f_init.tolist())
            if np.isfinite(sc):
                scored_inits.append((label, f_init, sc, is_parent))
                if sc < best_score:
                    best_score = sc
                    best_sequence = f_init.copy()

        scored_inits.sort(key=lambda x: (not x[3], x[2]))
        # Parents first, then by score

        per_init = n_budget / max(len(scored_inits), 1) * 0.85

        for label, f_init, init_sc, is_parent in scored_inits:
            if time.monotonic() > n_deadline:
                break

            # Decide optimization depth
            if is_parent or init_sc < best_score * 1.01:
                # Full pipeline
                init_dl = min(n_deadline, time.monotonic() + per_init)
                if time.monotonic() < init_dl - 15:
                    try:
                        bf, bs = optimize_pipeline(f_init, init_dl,
                                                   verbose_prefix=f"[n={n} {label}] ")
                        if bs < best_score:
                            best_score = bs
                            best_sequence = bf.copy()
                            print(f"  [{n}/{label}] NEW BEST: {bs:.6f}")
                    except Exception as e:
                        print(f"  [{n}/{label}] error: {e}")
            elif init_sc < best_score * 1.03:
                # Quick optimization
                init_dl = min(n_deadline, time.monotonic() + per_init * 0.5)
                if time.monotonic() < init_dl - 5:
                    try:
                        bf, bs = quick_optimize(f_init, init_dl,
                                                 verbose_prefix=f"[n={n} {label}]q ")
                        if bs < best_score:
                            best_score = bs
                            best_sequence = bf.copy()
                            print(f"  [{n}/{label}] quick improved: {bs:.6f}")
                    except Exception as e:
                        pass
            else:
                # Skip - not promising enough
                print(f"  [{n}/{label}] skip (score={init_sc:.6f} > {best_score*1.03:.6f})")

    # === Perturbation restarts on best ===
    if best_sequence is not None and time.monotonic() < deadline - 100:
        print(f"\n=== Perturbation restarts (best: {best_score:.6f}, n={len(best_sequence)}) ===")
        restart_budget = deadline - time.monotonic() - 40
        n_restarts = min(6, max(2, int(restart_budget / 50)))
        per_restart = restart_budget / n_restarts

        modes = ['mul', 'blend', 'shift', 'mul', 'blend', 'mul']
        strengths = [0.03, 0.10, 0.0, 0.05, 0.15, 0.04]

        for r in range(n_restarts):
            if time.monotonic() > deadline - 40:
                break

            mode = modes[r % len(modes)]
            strength = strengths[r % len(strengths)]
            f_pert = perturb_sequence(best_sequence, rng, mode=mode, strength=strength)
            S = float(np.sum(f_pert))
            if S < 0.01:
                continue
            f_pert = f_pert / S

            r_dl = time.monotonic() + per_restart * 0.8
            try:
                if r < 3:
                    bf, bs = optimize_pipeline(f_pert, r_dl,
                                               verbose_prefix=f"[r{r} {mode}] ")
                else:
                    bf, bs = quick_optimize(f_pert, r_dl,
                                             verbose_prefix=f"[r{r} {mode}] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  Restart {r} ({mode}): NEW BEST {bs:.6f}")
                else:
                    print(f"  Restart {r} ({mode}): {bs:.6f}")
            except Exception as e:
                print(f"  Restart {r} error: {e}")

    # === Final intensive refinement ===
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Final refinement (score={best_score:.6f}) ===")

        # Subgradient with very tight active set
        if time.monotonic() < deadline - 8:
            f, s = subgradient_descent(best_sequence, deadline - 6,
                                        lr_start=0.05, lr_end=0.001,
                                        top_k=2, threshold_frac=0.999,
                                        verbose_prefix="[fin_SG] ")
            if s < best_score:
                best_score = s
                best_sequence = f.copy()
                print(f"  SG final: {best_score:.6f}")

        # Very high beta multiplicative
        if time.monotonic() < deadline - 3:
            f, s = multiplicative_descent(best_sequence, deadline - 3,
                                           beta_start=500.0, beta_end=5000.0,
                                           lr_start=0.05, lr_end=0.0005,
                                           max_iters=5000,
                                           verbose_prefix="[fin_MD] ")
            if s < best_score:
                best_score = s
                best_sequence = f.copy()
                print(f"  MD final: {best_score:.6f}")

        # Active blended
        if time.monotonic() < deadline - 2:
            f, s = blended_active_descent(best_sequence, deadline - 2,
                                           beta_base=200.0, top_k=2,
                                           lr_start=0.1, lr_end=0.001,
                                           verbose_prefix="[fin_BA] ")
            if s < best_score:
                best_score = s
                best_sequence = f.copy()
                print(f"  BA final: {best_score:.6f}")

    # Ensure validity
    if best_sequence is None:
        best_sequence = raised_cosine(512, 2.0)
        best_score = evaluate_sequence(best_sequence.tolist())

    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    print(f"{'='*60}")
    return [float(x) for x in best_sequence]
