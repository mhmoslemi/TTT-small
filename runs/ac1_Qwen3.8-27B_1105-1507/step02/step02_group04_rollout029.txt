We want to minimize: `2 * n * max(conv(a,a)) / (sum(a))^2`

where `conv(a,a)` is the full linear convolution of `a` with itself.

Key observations:
1. The objective is scale-invariant (we can normalize sum(a) to any value)
2. The objective depends on the ratio of the maximum autocorrelation value to the square of the sum
3. For a uniform sequence `a = [1, 1, ..., 1]` of length n: conv(a,a) = [1, 2, 3, ..., n, n-1, ..., 1], so max = n, sum = n, obj = 2*n*n/n^2 = 2.
4. We want to go below 2, and ideally below 1.5.

The previous approach used:
- Smoothed subgradient (log-sum-exp) for the max function
- LP (alternating direction) where given `a`, solve for `g` that maximizes sum(g) subject to (a*g)_k <= M
- Multiple initializations with different patterns

def simplex_project(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    # Scale to work with sum=1, then rescale
    v = v / s
    # Algorithm from Wang & Carreira-Perpinan
    mu = np.zeros(n)
    sorted_v = np.sort(v)[::-1]
    cssv = np.cumsum(sorted_v)
    rho = np.nonzero(sorted_v * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    proj = np.maximum(v - theta, 0)
    return proj * s

Minimize 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed subgradient with variance regularization
2. Efficient coordinate descent with precomputed base convolution
3. Simplex projection and backtracking
4. Diverse initializations (structured + random)
5. Adaptive exploration of n from 30 to 3000

import numpy as np
from scipy.optimize import minimize_scalar
import time


def fft_convolve(a):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    conv = fft_convolve(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def lse_weights(conv, tau):
    """Compute softmax-like weights for log-sum-exp smoothing."""
    max_conv = np.max(conv)
    shifted = (conv - max_conv) / tau
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    w_sum = np.sum(exp_shifted)
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w = exp_shifted / w_sum
    return w


def gradient_lse(a, weights):
    """Gradient of log-sum-exp smoothing of conv(a,a) w.r.t. a."""
    n = len(a)
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1: n - 1 + n]
    return grad


def gradient_variance(a, conv):
    """Gradient of ||conv||^2/(sum(a))^4 w.r.t. a."""
    n = len(a)
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.zeros(n)
    sum_a4 = sum_a ** 4
    conv_sq = float(np.sum(conv ** 2))

    c = np.correlate(conv, a, mode='full')
    cross = c[n - 1: n - 1 + n]

    grad = 2.0 * (cross - 2.0 * conv_sq / sum_a) / sum_a4
    return grad


def smoothed_subgradient_optimize(a, max_time=30.0, beta=0.05, tau_start=3.0,
                                   tau_end=60.0, verbose=False):
    """Optimize using smoothed subgradient with variance regularization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_obj = compute_objective(best_a)

    start = time.monotonic()
    step = 0.15
    stall = 0
    patience = 150
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, it / 3000.0)
        tau = tau_start + (tau_end - tau_start) * frac

        conv = fft_convolve(a)
        current_obj = compute_objective(a)

        if current_obj < best_obj - 1e-13:
            best_obj = current_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                step *= 0.6
                if step < 1e-10:
                    break
                rng = np.random.default_rng(hash((it, n)) % (2**32))
                a = a + rng.normal(0, 0.03, n)
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s < 1e-10:
                    break
                a = a / s
                stall = 0
                continue

        weights = lse_weights(conv, tau)
        grad = gradient_lse(a, weights)

        if beta > 0.0:
            grad_var = gradient_variance(a, conv)
            scale = beta * tau / tau_start
            grad = grad + scale * grad_var

        grad_centered = grad - np.mean(grad)

        # Backtracking line search
        a_cur = a.copy()
        accepted = False
        for _ in range(15):
            candidate = a_cur - step * grad_centered
            candidate = np.maximum(candidate, 0)
            s_c = np.sum(candidate)
            if s_c < 1e-10:
                step *= 0.5
                continue
            candidate = candidate / s_c
            obj_c = compute_objective(candidate)
            if obj_c <= current_obj * (1.0 - 1e-10) - 1e-14:
                a_cur = candidate
                accepted = True
                break
            step *= 0.5

        if not accepted:
            # Try a smaller fixed step
            candidate = a - max(step, 1e-8) * grad_centered
            candidate = np.maximum(candidate, 0)
            s_c = np.sum(candidate)
            if s_c > 1e-10:
                candidate = candidate / s_c
                obj_c = compute_objective(candidate)
                if obj_c < current_obj:
                    a_cur = candidate

        a = a_cur
        if it % 200 == 0:
            step = min(step * 1.02, 0.4)

    return best_a, best_obj


def coordinate_descent(a, max_time=15.0, verbose=False):
    """Efficient coordinate descent using precomputed base convolution."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_obj = compute_objective(a)
    start = time.monotonic()
    n_passes = 0

    while time.monotonic() - start < max_time:
        n_passes += 1
        improved = False
        mean_val = np.sum(a) / n

        for m in range(n):
            if time.monotonic() - start > max_time:
                break

            a_m_old = a[m]
            # Build base vector with a[m] = 0
            b = a.copy()
            b[m] = 0.0
            G = fft_convolve(b)  # O(n log n)
            Sb = float(np.sum(b))  # = sum(a) - a[m]

            # conv[k] = G[k] + 2*t*a[k-m] (if 0<=k-m<n, k!=2m) + t^2 (if k=2m)
            # A[k] = a[k-m] if 0<=k-m<n and k!=2m, else 0
            A = np.zeros(2 * n - 1)
            k_min = m
            k_max = min(m + n - 1, 2 * n - 2)
            for k in range(k_min, k_max + 1):
                if k != 2 * m:
                    A[k] = a[k - m]
            B = np.zeros(2 * n - 1)
            if 0 <= 2 * m <= 2 * n - 2:
                B[2 * m] = 1.0

            def obj_1d(t):
                conv_k = G + 2.0 * t * A + t * t * B
                max_c = np.max(conv_k)
                s_t = Sb + t
                if s_t < 1e-10:
                    return 1e10
                return 2.0 * n * max_c / (s_t * s_t)

            t_max = max(4.0 * mean_val, a_m_old * 3.0 + 1e-6)
            try:
                res = minimize_scalar(obj_1d, bounds=(0.0, t_max), method='bounded',
                                      options={'xatol': 1e-10, 'maxiter': 40})
                if res.fun < best_obj - 1e-11:
                    a[m] = res.x
                    best_obj = res.fun
                    improved = True
                else:
                    a[m] = a_m_old
            except Exception:
                a[m] = a_m_old

        if verbose:
            print(f"    CD pass {n_passes}: obj={best_obj:.8f}")
        if not improved:
            break

    return a, best_obj


def generate_initializations(n, idx):
    """Generate diverse initializations for sequence length n."""
    rng = np.random.default_rng(idx * 7919 + n)
    t = np.arange(n)

    if idx == 0:
        a = np.ones(n)
    elif idx == 1:
        a = 0.5 * (1 + np.cos(2 * np.pi * (t + rng.integers(0, n)) / n))
    elif idx == 2:
        a = rng.random(n)
    elif idx == 3:
        a = 1.0 - 2.0 * np.abs(t / n - 0.5)
    elif idx == 4:
        a = (rng.random(n) > 0.5).astype(float)
    elif idx == 5:
        freq = rng.uniform(0.5, 3.0)
        a = 0.5 * (1 + np.cos(2 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi)))
    elif idx == 6:
        # Quadratic residues mod p (where p is prime near n)
        p = n
        while p < n + 20:
            if all(p % i != 0 for i in range(2, int(np.sqrt(p)) + 1)):
                break
            p += 1
        a = np.zeros(n)
        for j in range(n):
            r = (j * j) % p
            # Check if r is quadratic residue mod p
            a[j] = 1.0 if (pow(r, (p - 1) // 2, p) == 1 or r == 0) else 0.0
    elif idx == 7:
        # Comb pattern
        period = max(2, int(rng.integers(2, max(3, n // 4)))
        a = np.zeros(n)
        a[::period] = 1.0
    elif idx == 8:
        # Gaussian bump
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.4)
        a = np.exp(-0.5 * ((t - center) / width) ** 2)
    elif idx == 9:
        # Modulated random
        a = rng.random(n) * (0.5 + 0.5 * np.sin(2 * np.pi * rng.uniform(1, 5) * t / n))
    else:
        a = rng.random(n) ** rng.uniform(0.5, 3.0)

    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def try_parent_sequence():
    """Try to use the parent sequence as a starting point."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_objective(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def run_for_n(n, seed, max_time=30.0, use_cd=True, verbose=False):
    """Run optimization for a specific sequence length."""
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    n_inits = max(2, int(max_time / 15))
    n_inits = min(n_inits, 8)

    for init_idx in range(n_inits):
        if time.monotonic() - start > max_time:
            break
        remaining = max_time - (time.monotonic() - start)
        if remaining < 5:
            break

        a_init = generate_initializations(n, init_idx + seed % 100)
        init_obj = compute_objective(a_init)

        # Phase 1: Smoothed subgradient
        sg_time = remaining * 0.6
        a_sg, obj_sg = smoothed_subgradient_optimize(
            a_init, max_time=sg_time, beta=0.05,
            tau_start=3.0, tau_end=50.0, verbose=False)
        obj_sg = compute_objective(a_sg)

        # Phase 2: Coordinate descent (if time and n allow)
        a_best = a_sg
        obj_best = obj_sg
        if use_cd and n <= 800:
            remaining2 = max_time - (time.monotonic() - start)
            if remaining2 > 5:
                a_cd, obj_cd = coordinate_descent(a_sg, max_time=remaining2 * 0.7)
                if obj_cd < obj_best:
                    obj_best = obj_cd
                    a_best = a_cd

        # Phase 3: Final subgradient polish
        remaining3 = max_time - (time.monotonic() - start)
        if remaining3 > 3:
            a_pol, obj_pol = smoothed_subgradient_optimize(
                a_best, max_time=remaining3 * 0.8, beta=0.02,
                tau_start=20.0, tau_end=100.0, verbose=False)
            if obj_pol < obj_best:
                obj_best = obj_pol
                a_best = a_pol

        if obj_best < best_obj:
            best_obj = obj_best
            best_a = a_best.copy()
            if verbose:
                print(f"    n={n} init{init_idx}: obj={best_obj:.8f}")

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_objective(best_a)

    return best_a, best_obj
