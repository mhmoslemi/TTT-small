def compute_score(f):
    """Compute the objective value directly."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def build_conv_matrix(f):
    """Build dense matrix A where (A@g)[k] = convolve(f, g)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = f[k - j]
    return A


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f, g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, use_sparse=True):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        if use_sparse:
            A = build_conv_matrix_sparse(f)
            result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        else:
            A = build_conv_matrix(f)
            result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
        elif result.status == 2:  # iteration limit
            return result.x  # return best found
    except Exception:
        pass
    return None

def subgradient_direction(f, top_k=5):
    """Compute subgradient for the max convolution term."""
    n = len(f)
    conv = np.convolve(f, f)
    top_indices = np.argsort(conv)[-top_k:]
    max_val = conv[top_indices[0]]
    mean_val = float(np.sum(f)) ** 2 / (2 * n - 1)
    
    # Soft weighting: exponential of (conv - max) / temperature
    temp = max(max_val - mean_val, 1e-10)
    weights = np.exp((conv[top_indices] - max_val) / temp)
    weights = weights / np.sum(weights)
    
    grad = np.zeros(n)
    for k, w in zip(top_indices, weights):
        if w < 1e-10:
            continue
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += 2.0 * w * f[k - j]
    
    return grad

                         top_k=5, verbose=False, prefix=""):

    best_f = f.copy()

    n = len(f)
    S = float(np.sum(f))
    lr = lr_init
    momentum = np.zeros(n)
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < 100000:
        grad = subgradient_direction(f, top_k=top_k)
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-15:
            break
        
        grad = grad / (norm_grad + 1e-15)
        momentum = momentum_decay * momentum - lr * grad
        new_f = f + momentum
        new_f = np.clip(new_f, 0.0, None)
        new_sum = float(np.sum(new_f))
        if new_sum < 1e-15:
            lr *= 0.5
            no_improve += 1
            if no_improve > 200:
                break
            it += 1
            continue
        new_f = new_f * (S / new_sum)
        new_score = compute_score(new_f)
        it += 1
        
        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score

"""
Minimizes 2n*max(conv(f,f))/sum(f)^2 using LP-based direction finding with
adaptive step sizes, multi-length exploration, soft-max gradient polishing,
and restart strategies. Focuses computational budget on the most promising
sequence lengths identified during a quick exploration phase.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.ndimage import gaussian_filter1d


def compute_score(f):
    """Compute 2*n*max(conv(f,f))/sum(f)^2 efficiently."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))


def build_conv_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f, g)[k]."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def solve_lp(f, M, timeout=None):
    """Solve max sum(g) s.t. g>=0, conv(f,g)<=M. Returns g or None."""
    n = len(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    options = {'presolve': True}
    if timeout is not None:
        options['time_limit'] = timeout
    try:
        A = build_conv_sparse(f)
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options=options)
        if result.success or result.status in (2,):
            x = result.x
            if x is not None and np.all(np.isfinite(x)):
                return np.clip(x, 0, None)
    except Exception:
        pass
    return None


def lp_optimize(f, deadline, verbose=False, prefix=""):
    """LP-based iterative optimization with adaptive step and restarts."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    alpha = 0.15
    no_improve = 0
    it = 0
    rng = np.random.default_rng(7)

    while time.monotonic() < deadline and no_improve < 80:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 1e-18:
            break

        t0 = time.monotonic()
        lp_timeout = max(1.0, min(30.0, (deadline - time.monotonic()) * 0.3))
        g = solve_lp(f, M, timeout=lp_timeout)
        lp_t = time.monotonic() - t0

        if g is None:
            no_improve += 1
            if no_improve > 15:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-15:
            no_improve += 1
            continue

        ratio = sum_g / S
        g = g * (S / sum_g)

        # Adaptive step: larger when ratio >> 1, smaller when ratio ~ 1
        base_alpha = min(0.4, max(0.05, 0.3 / (ratio ** 0.5)))

        # Line search over several step sizes
        best_cand_score = best_score
        best_cand_f = None
        candidates_alpha = [base_alpha, base_alpha * 0.5, base_alpha * 1.5,
                            base_alpha * 0.25, 0.5, 0.8]
        for a in candidates_alpha:

            cand = cand * (S / cs)

def softmax_gradient(f, beta=50.0):
    """Compute smoothed gradient of max(conv(f,f)) using soft-max."""
    n = len(f)
    conv = np.convolve(f, f)
    # Soft-max weights
    conv_shifted = conv - np.max(conv)
    weights = np.exp(beta * conv_shifted)
    wsum = float(np.sum(weights))
    if wsum < 1e-300:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
    else:
        weights = weights / wsum

    grad = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += 2.0 * weights[k] * f[k - j]

    return grad


def softmax_optimize(f, deadline, beta=50.0, lr_init=0.01, verbose=False, prefix=""):
    """Optimize using soft-max approximation of max with backtracking."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    lr = lr_init
    momentum = np.zeros(n)
    mu = 0.9
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < 50000:
        grad = softmax_gradient(f, beta=beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        momentum = mu * momentum - lr * grad
        new_f = f + momentum
        new_f = np.clip(new_f, 0.0, None)
        ns = float(np.sum(new_f))
        if ns < 1e-15:
            lr *= 0.5
            no_improve += 1
            if no_improve > 200:
                break
            it += 1
            continue
        new_f = new_f * (S / ns)
        new_score = compute_score(new_f)
        it += 1

        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
            lr = min(lr * 1.02, 0.1)
            # Decay beta to spread gradients more
            beta = max(beta * 0.999, 5.0)
        else:
            no_improve += 1
            lr *= 0.96
            if lr < 1e-9 or no_improve > 300:

    return best_f, best_score


def subgradient_optimize(f, deadline, top_k=5, lr_init=0.01, verbose=False, prefix=""):
    """Subgradient descent with momentum."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    lr = lr_init
    momentum = np.zeros(n)
    mu = 0.85
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < 100000:
        conv = np.convolve(f, f)
        top_idx = np.argsort(conv)[-top_k:]
        max_val = conv[top_idx[0]]
        mean_val = S * S / (2 * n - 1)
        temp = max(max_val - mean_val, 1e-10)

        w = np.exp((conv[top_idx] - max_val) / temp)
        w = w / np.sum(w)

        grad = np.zeros(n)
        for k, wi in zip(top_idx, w):
            if wi < 1e-10:
                continue
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                grad[j] += 2.0 * wi * f[k - j]

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad = grad / gn

        momentum = mu * momentum - lr * grad
        new_f = f + momentum
        new_f = np.clip(new_f, 0.0, None)
        ns = float(np.sum(new_f))
        if ns < 1e-15:
            lr *= 0.5
            no_improve += 1
            if no_improve > 200:
                break
            it += 1
            continue
        new_f = new_f * (S / ns)
        new_score = compute_score(new_f)
        it += 1

        if np.isfinite(new_score) and new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.97
            if lr < 1e-9 or no_improve > 400:
                break

    return best_f, best_score


def make_inits(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    inits.append(('uniform', np.ones(n)))
    inits.append(('cosine', (1.0 + np.cos(np.pi * idx / max(n - 1, 1))) / 2.0))
    inits.append(('gauss_c', np.exp(-((idx - center) ** 2) / (2 * (n / 5.0) ** 2))))
    inits.append(('u_shape', ((idx - center) / max(center, 1)) ** 2 + 0.05))
    inits.append(('exp_sym', np.exp(-np.abs(idx - center) / (n / 8.0))))
    inits.append(('random', rng.uniform(0.1, 1.5, n)))
    inits.append(('pert_uniform', np.ones(n) + rng.normal(0, 0.15, n)))

    if parent is not None and len(parent) > 0:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        inits.append(('parent', np.interp(x_new, x_old, parent)))

    return inits
