import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_matrix(a):
    """Build dense constraint matrix for LP: A[k,j] = a[j] if j+k in [0,n-1]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_min_peak(a, A):
    """LP: min t s.t. A@g <= t, sum(g)=1, g>=0. Returns (g, t_opt)."""
    n = len(a)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.hstack([np.eye(n), np.zeros((1, 1))])
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    r = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x[:n], 0.0)
        if g.sum() > 1e-15:
            g /= g.sum()
            return g, float(r.x[-1])
    return None, None


def lp_max_sum(a, A, M):
    """LP: max sum(g) s.t. A@g <= M, g>=0. Returns g."""
    n = len(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def gradient_smooth_step(a, p=4, lr=0.01):
    """One gradient step minimizing sum(b^p) with projection to simplex."""
    n = len(a)
    b = fast_conv(a)
    if b.max() <= 0:
        return a.copy()
    # d/d(a_j) sum_k b[k]^p = p * sum_k b[k]^(p-1) * a[k-j] (valid indices)
    # This is p * (b^(p-1) convolved with a, at position n-1+j)
    bp1 = b ** (p - 1)
    # grad_j = p * sum_k bp1[k] * a[k-j] where 0<=k-j<n
    # = p * (bp1 * a_reversed)[n-1-j] ... let's compute via FFT
    s = 1
    while s < 2 * n - 1 + 2 * n - 1:
        s <<= 1
    fbp1 = np.fft.rfft(bp1, s)
    fa = np.fft.rfft(a, s)
    conv_full = np.fft.irfft(fbp1 * fa, s)
    # conv_full has length s; we want the part corresponding to positions n-1-j for j=0..n-1
    # conv_full[i] = sum_k bp1[k] * a[i-k] for valid indices
    # We want grad_j = p * sum_k bp1[k] * a[k-j] = p * conv_full[n-1-j + n-1] ... 
    # Actually let's just compute it directly for clarity
    grad = np.zeros(n)
    for j in range(n):
        # grad_j = p * sum_k bp1[k] * a[k-j] where 0<=k-j<n and 0<=k<2n-1
        # k-j in [0,n-1] means k in [j, j+n-1]
        k_lo = j
        k_hi = min(2 * n - 2, j + n - 1)
        if k_lo <= k_hi:
            grad[j] = np.dot(bp1[k_lo:k_hi + 1], a[0:k_hi - k_lo + 1])
    grad *= p
    # Projection: a_new = max(0, a - lr*grad), then normalize
    a_new = np.maximum(a - lr * grad, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new /= s
    return a_new


def coordinate_descent(a, max_iters=200, deadline=None):
    """Local search: try adjusting each coordinate."""
    n = len(a)
    current_J = compute_J(a)
    best = a.copy()
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            for delta in [0.01, 0.05, 0.1, 0.01, 0.05, 0.1]:
                sign = 1 if delta > 0 else -1
                a_trial = a.copy()
                a_trial[j] += sign * delta * a.max()
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:

                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-12:
                    a = a_trial
                    current_J = J_trial
                    improved = True
                    break
        if not improved:
            break
    return a, current_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

            a_trial /= s
            J_trial = compute_J(a_trial)

            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:

    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        # Symmetrized parent
        p = resample_seq(parent, n)
        inits.append((p + p[::-1]) / 2.0)

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.5, 1.0, n))

    # Various smooth profiles
    for freq in [1, 2, 3, 4, 5, 7, 8]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr))
        inits.append(1.0 + 0.5 * np.sin(2.0 * np.pi * freq * t_arr + rng.uniform(0, 6.28)))

    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.1)
    inits.append(1.0 + 0.4 * (2.0 * t_arr - 1.0) ** 2)
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.2)
    inits.append(1.0 / (1.0 + np.exp(-10 * (t_arr - 0.5))))
    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * t_arr ** 2))

    # Sparse + smoothed
    sparse = np.zeros(n)
    n_spikes = max(3, n // 20)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 10)
    kernel = np.exp(-((np.arange(kernel_size) - kernel_size // 2) ** 2) / (kernel_size / 4.0))
    kernel /= kernel.sum()
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Block structure
    n_blocks = rng.integers(3, 8)
    block_val = rng.uniform(0.3, 1.5, n_blocks)
    edges = np.sort(rng.uniform(0, 1, n_blocks - 1))
    blocks = np.zeros(n)
    prev = 0
    for i, e in enumerate(edges):
        idx = np.arange(int(prev * n), int(e * n))
        if len(idx) > 0:
            blocks[idx] = block_val[i]
        prev = e
    idx = np.arange(int(prev * n), n)
    if len(idx) > 0:
        blocks[idx] = block_val[-1]
    inits.append(blocks + 0.1)

    # Power law
    inits.append((t_arr + 0.05) ** rng.uniform(-1.5, 1.5) + 0.1)

    return inits

        # Another LP pass
        if time.monotonic() < deadline:
            a = tracker.best.copy()
            a = a / a.sum()
            a = run_lp_descent(a, 300, deadline, tracker, use_min_peak=True)

        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Try resampling to slightly different lengths ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*40} Resample search {'='*40}", flush=True)

        for factor in [0.9, 1.1, 1.25, 0.75, 1.5]:
            if time.monotonic() > deadline - 30:
                break
            n_new = int(n * factor)
            n_new = max(16, min(2048, n_new))
            if n_new == n:
                continue
            a = resample_seq(tracker.best, n_new)
            a = np.maximum(a, 1e-6)
            a /= a.sum()
            a = run_lp_descent(a, 150, deadline, tracker, use_min_peak=True)
            tracker.update(a)

        tracker.report(f"Phase6: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        print(f"Fallback: J={tracker.best_score:.8f}", flush=True)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
