    """Find optimal value of f[j] given all other coordinates fixed."""
    n = len(f)
    f_j = f[j]
    f[j] = 0
    C_0 = np.convolve(f, f)  # autocorrelation without coordinate j
    f[j] = f_j
    
    # s[k] = f[k-j] + f[j+k] (with boundary)
    s = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        val = 0
        if 0 <= k - j < n:
            val += f[k - j]
        if 0 <= j + k < n:
            val += f[j + k]
        s[k] = val
    
    # d[k] = 1 if k == 2j
    d = (np.arange(2*n - 1) == 2*j).astype(float)
    
    # Find x that minimizes max_k (C_0[k] + x*s[k] + x^2*d[k])
    # Use ternary search or bisection on derivative
    
    def h(x):
        return np.max(C_0 + x*s + x*x*d)
    
    # Bisection on the derivative
    # h'(x) = max of (s[k] for k where C_0[k]+x*s[k]+x^2*d[k] is max) + 2x*(d[k] for max k)
    # Actually h'(x) = max_{k in active} (s[k] + 2x*d[k])
    
    lo = 0.0
    # Upper bound: find where h(x) > h(0)
    h0 = h(0)
    # Upper bound estimate
    max_s = np.max(s)
    if max_s < 0:
        return 0.0  # all s are negative, so increasing x from 0 increases h
    # Actually if all s[k] < 0, then h is decreasing initially, so we should increase x
    # Let me just use a generous upper bound
    x_hi = max_s if max_s > 0 else 1.0
    # Double x_hi until h(x_hi) > h0
    while h(x_hi) < h0 and x_hi < 1000:
        x_hi *= 2
    
    # Now bisect to find where h'(x) = 0
    # h'(x) = max_{k: g_k(x) = h(x)} (s[k] + 2x*d[k])

        if deriv < 0:

        else:

    return (lo + x_hi) / 2

"""
Optimization for minimizing 2n*max(conv(f,f))/sum(f)^2 using
coordinate descent, Nesterov soft-max gradient, and LP refinement.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = (convolve(f,g))[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    if A_cache is None:
        A = build_conv_matrix_sparse(f)
    else:
        A = A_cache
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None

def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = (convolve(f,g))[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])

    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    if A_cache is None:

    else:
        A = A_cache
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:

    except Exception:

    return None


def compute_soft_max_weights(C, beta):
    """Compute soft-max weights p[k] proportional to exp(beta*C[k])."""

            # So C[k] = C_0[k] + 2*old*f_wj[k-j] + old^2*[k==2j]
            # => C_0[k] = C[k] - 2*old*f_wj[k-j] - old^2*[k==2j]
            # where f_wj[k-j] = f[k-j] if (0<=k-j<n and k-j != j) else 0
            #                   = f[k-j] - old*[k==2j] (if 0<=k-j<n) else 0
            
            # New: C_new[k] = C_0[k] + 2*x*f_wj[k-j] + x^2*[k==2j]
            #         = C[k] - 2*old*f_wj[k-j] - old^2*[k==2j] + 2*x*f_wj[k-j] + x^2*[k==2j]
            #         = C[k] + 2*(x-old)*f_wj[k-j] + (x^2-old^2)*[k==2j]
            
            # h(x) = max_k (C[k] + 2*(x-old)*f_wj[k-j] + (x^2-old^2)*[k==2j])
            # Let u = x - old (so x = old + u, x >= 0 means u >= -old)
            # h = max_k (C[k] + 2*u*f_wj[k-j] + (u*(2*old+u))*[k==2j])
            #   = max_k (C[k] + 2*u*f_wj[k-j] + u*(2*old+u)*[k==2j])
            
            # For the bisection, I'll work directly with x.
            # h(x) is convex in x >= 0.
            # h'(x) = max_{k active} (2*f_wj[k-j] - 2*x*[k==2j])
            #        = max_{k active, k in [j,j+n-1], k!=2j} 2*f[k-j]
            #          vs -2x if k=2j is active
            #          vs 0 if any k outside [j,j+n-1] is active
            
            # Efficient computation:
            # The affected lags are k in [j, min(j+n-1, 2n-2)]
            # For those k: C_new[k] = C[k] + 2*(x-old)*f_wj[k-j] + (x^2-old^2)*[k==2j]
            # For unaffected k: C_new[k] = C[k]
            
            # Let me find the optimal x using subgradient bisection.
            
            # Precompute: for k in [j, j+n-1], coeff[k] = f_wj[k-j]
            # coeff[k] = f[k-j] if k != 2j else 0
            # quad_coeff = 1 (at k=2j), 0 elsewhere
            
            # Lower bound: x = 0
            # Upper bound: need h(x_hi) > h(0) to bracket the minimum
            
            # Compute h(0):
            # h(0) = max(C[k] for k not affected, C[k] - 2*old*f_wj[k-j] + old^2*[k==2j] for affected k)
            
            # Let me just compute the max of unaffected C:
            # Unaffected lags: k < j or k >= j+n
            unaffected_max = 0.0
            if j > 0:
                unaffected_max = max(unaffected_max, float(np.max(C[:j])))
            if j + n < 2 * n - 1:
                unaffected_max = max(unaffected_max, float(np.max(C[j+n:])))
            
            # For affected lags, at x=0:
            # C_new[k] = C[k] - 2*old*f_wj[k-j] + old^2*[k==2j]
            # f_wj[k-j] = f[k-j] if k != 2j else 0
            k_start = j
            k_end = min(j + n, 2 * n - 1)
            
            if k_start < k_end:
                # shifted values: f[k-j] for k in [k_start, k_end)
                shifted = f[:k_end - k_start].copy()
                # Zero out the position where k-j == j, i.e., k == 2j
                pos_2j = 2 * j - k_start
                if 0 <= pos_2j < len(shifted):
                    shifted[pos_2j] = 0.0
                vals_at_0 = C[k_start:k_end].copy()
                vals_at_0 -= 2.0 * old * shifted
                if 2 * j >= k_start and 2 * j < k_end:
                    vals_at_0[2 * j - k_start] += old * old
                affected_max_at_0 = float(np.max(vals_at_0))
            else:
                affected_max_at_0 = 0.0
            
            h_0 = max(unaffected_max, affected_max_at_0)
            
            # Find upper bound
            x_hi = max(2.0 * old + 1.0, h_0 / max(np.max(f) + 1e-15, 0.1))
            x_hi = min(x_hi, 1000.0)
            
            # Check if h is increasing at x=0 (then x=0 is optimal)
            # h'(0) = max of (2*f_wj[k-j]) for active k (in affected range)
            #        vs 0 for unaffected k
            # At x=0, active k are those with C_new[k] = h_0
            if unaffected_max >= affected_max_at_0 - 1e-12:
                # Unaffected lags are active (or tied), so h'(0) >= 0
                # This means x=0 is optimal
                pass
            else:
                # Only affected lags are active
                # h'(0) = max(2*f_wj[k-j] for active affected k)
                # If this is >= 0, x=0 is still optimal
                # If < 0, we should increase x
                # Compute: max of 2*f[k-j] over active affected k
                # For simplicity, just check: if max(2*f_wj[k-j]) > 0, 
                # the function could be decreasing at 0, so we need to search.
                max_coeff = 0.0
                if k_start < k_end:
                    shifted2 = f[:k_end - k_start].copy()
                    pos_2j = 2 * j - k_start
                    if 0 <= pos_2j < len(shifted2):
                        shifted2[pos_2j] = 0.0
                    max_coeff = float(np.max(shifted2)) * 2.0
                if max_coeff <= 1e-12:
                    # Function is non-decreasing at x=0, so x=0 is optimal
                    pass
            
            # Bisection on h'(x) = 0
            lo = 0.0
            hi = x_hi
            
            # Ensure we have a valid bracket
            # If h'(0) > 0, the minimum might be at 0. Let's check h(hi) vs h(0).
            # For safety, just do the bisection.
            
            new_x = old  # default: keep current value
            for _iter in range(40):
                mid = (lo + hi) / 2.0
                # Compute C_new at x = mid
                # C_new[k] for affected k: C[k] + 2*(mid-old)*f_wj[k-j] + (mid^2-old^2)*[k==2j]
                if k_start < k_end:
                    vals = C[k_start:k_end].copy()
                    shifted = f[:k_end - k_start].copy()
                    pos_2j = 2 * j - k_start
                    if 0 <= pos_2j < len(shifted):
                        shifted[pos_2j] = 0.0
                    vals += 2.0 * (mid - old) * shifted
                    if 2 * j >= k_start and 2 * j < k_end:
                        vals[2 * j - k_start] += (mid * mid - old * old)
                    mid_max = max(float(np.max(vals)), unaffected_max)
                else:
                    mid_max = unaffected_max
                
                # Find active set and compute subgradient
                if k_start < k_end:
                    vals_full = C[k_start:k_end].copy()
                    shifted = f[:k_end - k_start].copy()
                    pos_2j = 2 * j - k_start
                    if 0 <= pos_2j < len(shifted):
                        shifted[pos_2j] = 0.0
                    vals_full += 2.0 * (mid - old) * shifted
                    if 2 * j >= k_start and 2 * j < k_end:
                        vals_full[2 * j - k_start] += (mid * mid - old * old)
                    active_max = float(np.max(vals_full))
                else:
                    active_max = 0.0
                
                # Derivative: max of (2*f_wj[k-j] - 2*mid*[k==2j]) for active k
                # Unaffected k have derivative 0
                deriv = 0.0  # from unaffected k
                if k_start < k_end:
                    tol_act = 1e-10 * max(1.0, abs(active_max))
                    # Active affected k:
                    active_mask = (np.abs(vals_full - active_max) <= tol_act)
                    if np.any(active_mask):
                        for idx in np.where(active_mask)[0]:
                            k = idx + k_start
                            if k == 2 * j:
                                dk = -2.0 * mid
                            else:
                                dk = 2.0 * f[k - j]
                            deriv = max(deriv, dk)
                
                if deriv < 0:
                    lo = mid
                else:
                    hi = mid
            
            new_x = (lo + hi) / 2.0
            new_x = max(0.0, min(1000.0, new_x))
            
            # Update C
            if k_start < k_end:
                shifted = f[:k_end - k_start].copy()
                pos_2j = 2 * j - k_start
                if 0 <= pos_2j < len(shifted):
                    shifted[pos_2j] = 0.0
                C[k_start:k_end] += 2.0 * (new_x - old) * shifted
                if 2 * j >= k_start and 2 * j < k_end:
                    C[2 * j] += (new_x * new_x - old * old)
            
            f[j] = new_x
            max_change = max(max_change, abs(new_x - old))
        
        if max_change < tol * max(1.0, S):
            break
    
    return f


def nesterov_softmax_optimize(f, deadline, beta_start=10.0, beta_end=500.0,
                               lr_start=0.05, max_iters=3000, verbose_prefix=""):
    """
    Optimize using soft-max gradient with Nesterov momentum.
    Minimizes 2n * soft_max(conv(f,f)) / sum(f)^2 (scale invariant, fix sum=1).
    """
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    # Nesterov state
    y = f.copy()  # lookahead
    v = np.zeros(n)  # velocity
    mu = 0.9  # momentum
    lr = lr_start
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        it_frac = min(1.0, it / (max_iters * 0.6))
        beta = beta_start * (beta_end / beta_start) ** it_frac
        
        # Compute gradient at y (lookahead point)
        C_y = np.convolve(y, y)
        p, _ = compute_soft_max_weights(C_y, beta)
        grad = compute_gradient_from_weights(p, y)
        
        # Update velocity
        v = mu * v - lr * grad
        
        # Update y (Nesterov step)
        new_mu = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
        y_new = y + ((1.0 - mu) / new_mu) * v + (mu * (mu - 1.0) / new_mu) * (y - f)
        
        # Clip and normalize
        y_new = np.clip(y_new, 0.0, 1000.0)
        ns = float(np.sum(y_new))
        if ns < 1e-10:
            y_new = f
            ns = S
        y_new = y_new / ns
        
        # Evaluate
        sc = evaluate_sequence(y_new.tolist())
        if np.isfinite(sc) and sc < best_score:
            f = y_new
            y = y_new
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            if it % 200 == 0:
                print(f"    {verbose_prefix}NesSM iter {it}: score={best_score:.6f} (beta={beta:.1f}, lr={lr:.5f})")
        else:
            # Revert: keep y = f
            y = f.copy()
            no_improve += 1
            if no_improve > 100:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-7:
                    break
        
        it += 1
    
    return best_f, best_score


def adam_softmax_optimize(f, deadline, beta_start=10.0, beta_end=500.0,
                           lr_start=0.03, max_iters=5000, verbose_prefix=""):
    """
    Optimize using soft-max gradient with Adam optimizer.
    """
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        it_frac = min(1.0, it / (max_iters * 0.6))
        beta = beta_start * (beta_end / beta_start) ** it_frac
        
        C = np.convolve(f, f)
        p, _ = compute_soft_max_weights(C, beta)
        grad = compute_gradient_from_weights(p, f)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        new_f = f - lr * update
        new_f = np.clip(new_f, 0.0, 1000.0)
        ns = float(np.sum(new_f))
        if ns < 1e-10:
            new_f = f
            ns = S
        new_f = new_f / ns
        
        sc = evaluate_sequence(new_f.tolist())
        if np.isfinite(sc) and sc < best_score:
            f = new_f
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            if it % 500 == 0:
                print(f"    {verbose_prefix}Adam iter {it}: score={best_score:.6f} (beta={beta:.1f}, lr={lr:.5f})")
        else:
            no_improve += 1
            if no_improve > 300:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-8:
                    break
    
    return best_f, best_score


def lp_iterate(f, deadline, step_size_init=0.1, max_iters=50, verbose_prefix=""):
    """Run LP-based iterations with line search."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    it = 0
    no_improve = 0
    step_size = step_size_init
    
    # Build A cache
    A_cache = build_conv_matrix_sparse(f)
    
    while time.monotonic() < deadline and it < max_iters:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 1e-15:
            break
        
        g = solve_lp_direction(f, M, A_cache=A_cache)
        it += 1
        
        if g is None:
            no_improve += 1
            if no_improve > 15:
                break
            continue
        
        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue
        
        g = g / sum_g  # normalize g to have sum=1 (same as f)
        
        # Line search
        best_t = 0.0
        best_s = best_score
        for trial_t in [step_size, step_size * 1.5, step_size * 0.5, step_size * 2, step_size * 0.25]:
            new_f = (1.0 - trial_t) * f + trial_t * g
            new_f = np.clip(new_f, 0.0, 1000.0)
            ns = float(np.sum(new_f))
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = trial_t
        
        if best_t > 0 and best_s < best_score:
            f = (1.0 - best_t) * f + best_t * g
            f = np.clip(f, 0.0, 1000.0)
            S = float(np.sum(f))
            f = f / S
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            # Rebuild A cache since f changed
            A_cache = build_conv_matrix_sparse(f)
            if it % 10 == 0:
                print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f} (t={best_t:.3f})")
        else:
            no_improve += 1
            step_size *= 0.6
            if step_size < 0.003 or no_improve > 30:
                break
    
    return best_f, best_score


def tukey_window(n, r=0.3):
    """Generate a Tukey (tapered cosine) window."""
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def hann_window(n):
    """Generate a Hann window."""
    if n < 2:
        return np.ones(n)
    w = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    return w


def psf_like_window(n, alpha=2.0):
    """Generate a super-Gaussian (PSF-like) window."""
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    w = np.exp(-((idx - center) / (n / 2.0)) ** (2 * alpha))
    return w


def block_window(n, n_blocks=8, profile='cosine'):
    """Generate a block structure window."""
    if profile == 'cosine':
        # Cosine profile across blocks
        block_heights = 0.5 * (1 + np.cos(np.pi * np.arange(n_blocks) / max(n_blocks - 1, 1)))
    elif profile == 'ramp':
        block_heights = np.linspace(0.3, 1.0, n_blocks)
    elif profile == 'v_shaped':
        mid = (n_blocks - 1) / 2.0
        block_heights = 0.3 + 0.7 * np.abs(np.arange(n_blocks) - mid) / max(mid, 1)
        block_heights = 1.0 / block_heights  # invert: high in middle
    else:
        block_heights = np.ones(n_blocks)
    
    w = np.zeros(n)
    block_size = n / n_blocks
    for b in range(n_blocks):
        start = int(b * block_size)
        end = int((b + 1) * block_size)
        if end > n:
            end = n
        w[start:end] = block_heights[b]
    return w


def random_smooth_window(n, seed=42):
    """Generate a random smooth non-negative window."""
    rng = np.random.default_rng(seed)
    x = rng.standard_normal(n)
    kernel_size = max(3, n // 16)
    kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (kernel_size / 2.0)) ** 2)
    kernel /= kernel.sum()
    smooth_x = np.convolve(x, kernel, mode='same')
    smooth_x = np.abs(smooth_x) + 0.1
    # Apply a slight taper
    taper = np.ones(n)
    m = max(1, n // 20)
    for i in range(m):
        taper[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
        taper[n - 1 - i] = taper[i]
    smooth_x *= taper
    return smooth_x


def perturb_sequence(f, rng, strength=0.05):
    """Add multiplicative noise and renormalize."""
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.1, 3.0)
    new_f = np.clip(new_f, 0.0, 1000.0)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S


def interpolate_to_length(f, target_n):
    """Interpolate a sequence to a target length."""
    if len(f) == target_n:
        return f.copy()
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, target_n)
    new_f = np.interp(x_new, x_old, f)
    return np.clip(new_f, 0.0, None)


def full_optimize_pipeline(f_init, deadline, phase_weights=(0.35, 0.35, 0.2, 0.1),
                           verbose_prefix=""):
    """
    Run full multi-phase optimization on a starting point.
    Phase 1: Coordinate descent
    Phase 2: Soft-max gradient (Adam)
    Phase 3: LP refinement
    Phase 4: Final coordinate descent + soft-max polish
    """
    f_init = f_init.copy().astype(np.float64)
    S = float(np.sum(f_init))
    if S < 1e-10:
        return f_init, evaluate_sequence(f_init.tolist())
    f_init = f_init / S
    
    best_f = f_init.copy()
    best_score = evaluate_sequence(f_init.tolist())
    t0 = time.monotonic()
    total_budget = deadline - t0
    if total_budget < 10:
        return best_f, best_score
    
    n = len(f_init)
    
    # Phase 1: Coordinate descent
    p1_deadline = t0 + total_budget * phase_weights[0]
    if time.monotonic() < p1_deadline - 2:
        # Use symmetric ordering for first few sweeps, then random
        n_cd_sweeps = min(15, max(3, int((p1_deadline - time.monotonic()) / 2)))
        f1 = coordinate_descent_sweep(f_init, n_sweeps=n_cd_sweeps,
                                        order='symmetric', deadline=p1_deadline - 1)
        s1 = evaluate_sequence(f1.tolist())
        if s1 < best_score:
            best_f, best_score = f1, s1
            print(f"    {verbose_prefix}P1(CD): {best_score:.6f}")
        # Also try forward ordering
        if time.monotonic() < p1_deadline - 2:
            f1b = coordinate_descent_sweep(best_f, n_sweeps=5,
                                            order='forward', deadline=p1_deadline - 1)
            s1b = evaluate_sequence(f1b.tolist())
            if s1b < best_score:
                best_f, best_score = f1b, s1b
    
    # Phase 2: Adam soft-max gradient
    p2_deadline = t0 + total_budget * (phase_weights[0] + phase_weights[1])
    if time.monotonic() < p2_deadline - 3:
        f2, s2 = adam_softmax_optimize(best_f, p2_deadline - 2,
                                        beta_start=10.0, beta_end=400.0,
                                        lr_start=0.03, max_iters=5000,
                                        verbose_prefix=verbose_prefix + "P2")
        if s2 < best_score:
            best_f, best_score = f2, s2
            print(f"    {verbose_prefix}P2(Adam): {best_score:.6f}")
    
    # Phase 3: LP refinement
    p3_deadline = t0 + total_budget * (phase_weights[0] + phase_weights[1] + phase_weights[2])
    if time.monotonic() < p3_deadline - 3 and n <= 1024:
        # LP is slow for large n, only use for n <= 1024
        f3, s3 = lp_iterate(best_f, p3_deadline - 2, step_size_init=0.08,
                             max_iters=30, verbose_prefix=verbose_prefix + "P3")
        if s3 < best_score:
            best_f, best_score = f3, s3
            print(f"    {verbose_prefix}P3(LP): {best_score:.6f}")
    
    # Phase 4: Final polish - coordinate descent + short gradient
    if time.monotonic() < deadline - 2:
        f4 = coordinate_descent_sweep(best_f, n_sweeps=5, order='random',
                                       deadline=deadline - 3)
        s4 = evaluate_sequence(f4.tolist())
        if s4 < best_score:
            best_f, best_score = f4, s4
        
        if time.monotonic() < deadline - 2:
            f5, s5 = adam_softmax_optimize(best_f, deadline - 1,
                                            beta_start=200.0, beta_end=800.0,
                                            lr_start=0.01, max_iters=2000,
                                            verbose_prefix=verbose_prefix + "P4")
            if s5 < best_score:
                best_f, best_score = f5, s5
    
    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    """
    Main search function. Tries multiple lengths and initializations,
    using coordinate descent, soft-max gradient with Adam, LP refinement,
    and perturbation restarts.
    """
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)
    
    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)
    
    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                score = evaluate_sequence(parent.tolist())
                if np.isfinite(score):
                    best_score = score
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")
    
    # Lengths to try with time budgets
    # Focus on lengths that might give the best tradeoff
    lengths_budgets = [
        (512, 0.28),    # Proven good length, refine heavily
        (768, 0.15),    # Between 512 and 1024
        (1024, 0.18),   # Longer, potentially better asymptotic
        (2048, 0.12),   # Much longer
        (384, 0.08),    # Shorter, quick check
        (640, 0.07),    # Odd length
    ]
    
    total_budget = deadline - t0
    
    for n_len, budget_frac in lengths_budgets:
        if time.monotonic() > deadline - 60:
            print("Time running out, stopping length exploration")
            break
        
        n_budget = total_budget * budget_frac
        n_deadline = time.monotonic() + n_budget
        print(f"\n=== n={n_len} (budget {n_budget:.0f}s, current best={best_score:.6f}) ===")
        
        # Generate initializations
        inits = []
        
        # If we have a best sequence, interpolate to this length
        if best_sequence is not None:
            if len(best_sequence) == n_len:
                inits.append(('best', best_sequence.copy()))
            else:
                inits.append(('best_interp', interpolate_to_length(best_sequence, n_len)))
        
        # Tukey windows
        for r in [0.2, 0.35, 0.5]:
            inits.append((f'tukey_r{r}', tukey_window(n_len, r)))
        
        # Hann
        inits.append(('hann', hann_window(n_len)))
        
        # Super-Gaussian
        for alpha in [2.0, 3.0, 4.0]:
            inits.append((f'psf_a{alpha}', psf_like_window(n_len, alpha)))
        
        # Block structures
        for nb in [4, 8, 16]:
            inits.append((f'block_{nb}', block_window(n_len, nb, 'cosine')))
        
        # Random smooth
        for seed in [7, 13]:
            inits.append((f'rand_{seed}', random_smooth_window(n_len, seed)))
        
        # Per-init time
        n_inits = len(inits)
        per_init_time = n_budget / max(n_inits, 1) * 0.75
        
        for label, f_init in inits:
            if time.monotonic() > n_deadline:
                break
            
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            S = float(np.sum(f_init))
            if S < 0.01:
                continue
            
            init_score = evaluate_sequence(f_init.tolist())
            if not np.isfinite(init_score):
                continue
            
            if init_score < best_score:
                best_score = init_score
                best_sequence = f_init.copy()
                print(f"  [n={n_len} {label}] init: {best_score:.6f}")
            
            # Only do full optimization if this init has promise
            # or if it's the best/interpolated init
            is_promise = (init_score < best_score * 1.02) or (label in ('best', 'best_interp'))
            if is_promise or n_inits <= 6:
                init_deadline = min(n_deadline, time.monotonic() + per_init_time)
                if time.monotonic() < init_deadline - 5:
                    try:
                        bf, bs = full_optimize_pipeline(f_init, init_deadline,
                                                        verbose_prefix=f"[n={n_len} {label}] ")
                        if bs < best_score:
                            best_score = bs
                            best_sequence = bf.copy()
                            print(f"  [n={n_len} {label}] improved to {bs:.6f}")
                    except Exception as e:
                        print(f"  [n={n_len} {label}] error: {e}")
    
    # Perturbation restarts on the best sequence found
    if best_sequence is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Perturbation restarts (current best: {best_score:.6f}) ===")
        restart_budget = deadline - time.monotonic() - 20
        n_restarts = min(8, max(2, int(restart_budget / 40)))
        per_restart = restart_budget / n_restarts
        
        for r in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            
            # Vary perturbation strength
            strength = 0.02 + 0.02 * r
            f_pert = perturb_sequence(best_sequence, rng, strength=strength)
            f_pert_deadline = time.monotonic() + per_restart * 0.8
            
            try:
                bf, bs = full_optimize_pipeline(f_pert, f_pert_deadline,
                                                 phase_weights=(0.4, 0.3, 0.15, 0.15),
                                                 verbose_prefix=f"[restart {r}] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  Restart {r} (str={strength:.3f}): {bs:.6f}")
                else:
                    print(f"  Restart {r} (str={strength:.3f}): no improvement ({bs:.6f})")
            except Exception as e:
                print(f"  Restart {r} error: {e}")
    
    # Final intensive refinement on the absolute best
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Final refinement (n={len(best_sequence)}, score={best_score:.6f}) ===")
        final_deadline = deadline - 5
        n_fin = len(best_sequence)
        
        try:
            # More coordinate descent sweeps
            n_cd = min(20, max(5, int((final_deadline - time.monotonic()) / 3)))
            f_coarse = coordinate_descent_sweep(best_sequence, n_sweeps=n_cd,
                                                  order='symmetric', deadline=final_deadline - 10)
            s_coarse = evaluate_sequence(f_coarse.tolist())
            if s_coarse < best_score:
                best_score = s_coarse
                best_sequence = f_coarse
                print(f"  Final CD: {best_score:.6f}")
            
            # Short LP if n is small enough
            if n_fin <= 768 and time.monotonic() < final_deadline - 15:
                f_lp, s_lp = lp_iterate(best_sequence, time.monotonic() + 30,
                                         step_size_init=0.04, max_iters=20,
                                         verbose_prefix="[final_LP] ")
                if s_lp < best_score:
                    best_score = s_lp
                    best_sequence = f_lp
                    print(f"  Final LP: {best_score:.6f}")
            
            # Final gradient polish
            if time.monotonic() < final_deadline - 5:
                f_grad, s_grad = adam_softmax_optimize(best_sequence, final_deadline - 2,
                                                        beta_start=300.0, beta_end=1000.0,
                                                        lr_start=0.005, max_iters=3000,
                                                        verbose_prefix="[final_ADAM] ")
                if s_grad < best_score:
                    best_score = s_grad
                    best_sequence = f_grad
                    print(f"  Final Adam: {best_score:.6f}")
            
            # Very final coordinate descent
            if time.monotonic() < final_deadline - 2:
                f_final = coordinate_descent_sweep(best_sequence, n_sweeps=5,
                                                    order='random', deadline=final_deadline - 1)
                s_final = evaluate_sequence(f_final.tolist())
                if s_final < best_score:
                    best_score = s_final
                    best_sequence = f_final
                    print(f"  Final CD2: {best_score:.6f}")
                    
        except Exception as e:
            print(f"  Final refinement error: {e}")
    
    if best_sequence is None:
        # Fallback
        best_sequence = tukey_window(512, 0.3)
        best_score = evaluate_sequence(best_sequence.tolist())
    
    # Ensure all values are valid
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = evaluate_sequence(best_sequence.tolist())
    
    # Final verification
    final_score = evaluate_sequence([float(x) for x in best_sequence])
    if np.isfinite(final_score):
        best_score = final_score
    
    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
