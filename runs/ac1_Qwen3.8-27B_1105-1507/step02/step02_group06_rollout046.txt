def smoothed_max_and_grad(f, temperature):
    """Compute smoothed max of convolution and its gradient.
    Returns (smoothed_max_value, gradient_vector, softmax_weights)."""
    conv = np.convolve(f, f)
    max_conv = np.max(conv)
    # Numerically stable softmax
    shifted = temperature * (conv - max_conv)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    smoothed_val = max_conv + np.log(np.sum(exp_shifted)) / temperature

    # Gradient: grad[j] = 2 * sum_k weights[k] * f[k-j] (with 0 for out-of-range)
    n = len(f)
    grad = np.zeros(n)
    # For each j, grad[j] = 2 * sum_{k=j}^{min(j+n-1, 2n-2)} weights[k] * f[k-j]
    # This is a cross-correlation: grad = 2 * (weights corr f) at appropriate lags
    # Use FFT for speed
    n FFT = 1
    while nFFT < 2 * n:
        nFFT *= 2
    F_weights = np.fft.rfft(weights, nFFT)
    F_f = np.fft.rfft(f, nFFT)
    # Cross-correlation: corr[m] = sum_k weights[k] * f[k-m] (with appropriate shifts)
    # We want: for j in [0, n-1], grad[j] = 2 * sum_k weights[k] * f[k-j]
    # This is the cross-correlation of weights with f, evaluated at lag j
    # np.correlate(weights, f, mode='full') gives sum_k weights[k] * conj(f[k - lag])
    # But since everything is real: sum_k weights[k] * f[k - lag]
    # The 'full' correlate has length len(weights) + len(f) - 1 = (2n-1) + n - 1 = 3n-2
    # The lag j in our case corresponds to index... let me just do it with FFT
    # corr[j] = sum_k weights[k] * f[k-j] for k in [0, 2n-2], k-j in [0, n-1]
    # This equals: ifft(fft(weights) * conj(fft(f))) at the right index
    corr = np.fft.irfft(F_weights * np.conj(F_f), nFFT)
    # The correlation at "zero lag" is at index 0
    # corr[m] = sum_k weights[k+m] * f[k] (circular)
    # Hmm, this is getting confusing. Let me just do it directly for n=1024.
    # O(n^2) with n=1024 is ~1M ops, which is fast.
    grad = np.zeros(n)
    for j in range(n):
        # grad[j] = 2 * sum_k weights[k] * f[k-j] where 0 <= k-j < n
        # => k in [j, j+n-1], and k in [0, 2n-2]
        k_start = j
        k_end = min(j + n, 2 * n - 1)  # exclusive
        if k_end > k_start:
            grad[j] = 2.0 * np.dot(weights[k_start:k_end], f[0:k_end - k_start])
    return smoothed_val, grad, weights, conv


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def fista_smoothed(f, deadline, S, temp_init=5.0, temp_end=50.0,
                   n_anneal=200, verbose_prefix=""):
    """FISTA on smoothed max-convolution with projection.
    Returns (best_f, best_score)."""
    f = f.copy().astype(np.float64)
    f = np.clip(f, 0.0, 1000.0)
    # Normalize to sum = S
    cur_sum = np.sum(f)
    if cur_sum < 1e-10:
        f = np.ones(len(f)) * S / len(f)
        cur_sum = S
    else:
        f = f * (S / cur_sum)

    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)

    # FISTA state
    y = f.copy()
    t_moment = 1.0
    lr = 1.0  # Will be adapted
    no_improve = 0

    it = 0
    while time.monotonic() < deadline:
        # Adaptive temperature
        frac = min(1.0, it / n_anneal)
        temperature = temp_init * (temp_end / temp_init) ** frac

        # Compute gradient at y
        smooth_val, grad, weights, conv = smoothed_max_and_grad(y, temperature)
        M = float(np.max(conv))

        # Gradient step
        y_new = y - lr * grad

        # Project onto {v >= 0, sum(v) = S}
        a_new = project_onto_simplex(y_new, S)

        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_moment * t_moment)) / 2.0
        y_new_fista = a_new + ((t_moment - 1.0) / t_new) * (a_new - f)
        y = np.clip(y_new_fista, 0.0, None)
        # Re-project to maintain feasibility for y (optional but helps)
        y = project_onto_simplex(y, S)
        t_moment = t_new

        f = a_new
        it += 1

        # Compute actual score
        if it % 5 == 0:
            score = compute_score(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Adaptive learning rate
        if no_improve > 50:
            lr *= 0.5
            no_improve = 0
            # Also reset momentum
            y = f.copy()
            t_moment = 1.0
        elif no_improve == 0:
            lr *= 1.02
            lr = min(lr, 10.0)

        # Safety: if lr too small, stop
        if lr < 1e-8:
            break

        if it % 100 == 0:
            print(f"    {verbose_prefix}FISTA iter {it}: score={best_score:.6f} "
                  f"temp={temperature:.1f} lr={lr:.4f}")

    # Final check
    final_score = compute_score(f)
    if final_score < best_score:
        best_score = final_score
        best_f = f.copy()

    return best_f, best_score


def lp_step(f, S, target_scale=0.99, verbose_prefix=""):
    """One LP-based direction step. Returns (new_f, new_score)."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))

    # Try LP with slightly reduced M target
    M_target = M * target_scale
    g = solve_lp_direction(f, M_target)
    if g is None:
        return f, compute_score(f)

    sum_g = np.sum(g)
    if sum_g < 1e-15:
        return f, compute_score(f)
    g = g * (S / sum_g)

    # Try several step sizes
    best_f = f
    best_score = compute_score(f)
    for alpha in [0.3, 0.5, 0.7, 1.0, 0.15, 0.85]:
        candidate = (1.0 - alpha) * f + alpha * g
        candidate = np.clip(candidate, 0.0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (S / cand_sum)
        cand_score = compute_score(candidate)
        if cand_score < best_score:
            best_score = cand_score
            best_f = candidate.copy()

    return best_f, best_score


def lp_optimize_loop(f, deadline, S, target_scale=0.99,
                     lp_every=3, verbose_prefix=""):
    """Alternating LP and FISTA steps."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    it = 0

    while time.monotonic() < deadline:
        # LP step
        t_start = time.monotonic()
        f_new, sc = lp_step(f, S, target_scale=target_scale)
        lp_time = time.monotonic() - t_start

        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
        else:
            # Try with less aggressive target
            f_new2, sc2 = lp_step(f, S, target_scale=1.0)
            if sc2 < best_score:
                best_score = sc2
                best_f = f_new2.copy()
                f = f_new2

        it += 1
        if it % 10 == 0:
            print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f} "
                  f"(lp_time={lp_time:.2f}s)")

        # Periodic FISTA polish
        if it % lp_every == 0:
            remaining = deadline - time.monotonic()
            if remaining > 5.0:
                fista_deadline = time.monotonic() + min(30.0, remaining * 0.1)
                f_fista, sc_fista = fista_smoothed(f, fista_deadline, S,
                                                    temp_init=10.0, temp_end=50.0,
                                                    n_anneal=100,
                                                    verbose_prefix=verbose_prefix + "fista ")
                if sc_fista < best_score:
                    best_score = sc_fista
                    best_f = f_fista.copy()
                    f = f_fista

    return best_f, best_score


def make_initializations(n, rng):
    """Create diverse initializations for length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(('uniform', np.ones(n)))

    # Cosine taper
    cos_t = (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0
    inits.append(('cosine', cos_t))

    # Linear ramp
    inits.append(('ramp', np.linspace(1.0, 0.2, n)))

    # Exponential decay
    inits.append(('exp', np.exp(-2.0 * idx / n)))

    # Bell
    center = (n - 1) / 2.0
    bell = np.exp(-0.5 * ((idx - center) / (n / 5.0)) ** 2)
    inits.append(('bell', bell))

    # W-shaped (low in middle)
    w_shape = 1.0 - 0.7 * np.exp(-0.5 * ((idx - center) / (n / 6.0)) ** 2)
    inits.append(('w_shape', w_shape))

    # Random
    inits.append(('random', rng.uniform(0.3, 1.0, size=n)))

    # Sparse peaks
    sparse = np.zeros(n)
    n_peaks = max(2, n // 64)
    for _ in range(n_peaks):
        pos = rng.integers(0, n)
        width = max(1, n // (4 * n_peaks))
        for d in range(-width, width + 1):
            p = (pos + d) % n
            sparse[p] += np.exp(-0.5 * (d / (width * 0.5)) ** 2)
    inits.append(('sparse_peaks', sparse + 0.05))

    return inits


def propose_candidate(budget_s=1500, **kwargs):
    """Main entry point: search for the best sequence minimizing the objective."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 5.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)

    # Try parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            score = evaluate_sequence(parent.tolist())
            if np.isfinite(score):
                best_score = score
                best_sequence = parent.copy()
                print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    # Determine length to focus on
    if best_sequence is not None:
        n_main = len(best_sequence)
    else:
        n_main = 1024

    # Try a few lengths
    lengths = [n_main]
    if n_main <= 1024:
        lengths.append(n_main * 2)
    lengths.append(max(256, n_main // 2))
    lengths = sorted(set(l for l in lengths if 128 <= l <= 4096))

    print(f"\nWill try lengths: {lengths}")
    print(f"Initial best: {best_score:.6f}")

    # Phase 1: Quick exploration (~200s)
    print("\n=== Phase 1: Exploration ===")
    expl_deadline = t0 + min(200.0, (deadline - t0) * 0.15)

    all_candidates = []  # (score, sequence, n)

    for n in lengths:
        if time.monotonic() > expl_deadline - 30:
            break
        print(f"\n--- n={n} ---")
        S = np.sqrt(2.0 * n)

        inits = make_initializations(n, rng)

        # Add parent-based init
        if parent is not None:
            if len(parent) == n:
                inits.append(('parent', parent.copy()))
            else:
                x_old = np.linspace(0, 1, len(parent))
                x_new = np.linspace(0, 1, n)
                p_interp = np.interp(x_new, x_old, parent)
                inits.append(('parent_interp', p_interp))

        for label, f_init in inits:
            if time.monotonic() > expl_deadline - 10:
                break
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            if np.sum(f_init) < 1e-10:
                continue

            # Normalize
            f_init = f_init * (S / np.sum(f_init))
            score = compute_score(f_init)

            # Quick FISTA
            quick_dl = min(expl_deadline, time.monotonic() + 20.0)
            f_opt, s_opt = fista_smoothed(f_init, quick_dl, S,
                                           temp_init=5.0, temp_end=30.0,
                                           n_anneal=50, verbose_prefix=f"[{label}] ")
            if s_opt < score:
                score = s_opt
                f_init = f_opt

            if score < best_score:
                best_score = score
                best_sequence = f_init.copy()

            all_candidates.append((score, f_init.copy(), n))
            print(f"  {label}: {score:.6f}")

    all_candidates.sort(key=lambda x: x[0])
    print(f"\nTop after exploration:")
    for sc, seq, n in all_candidates[:5]:
        print(f"  n={n}: {sc:.6f}")
    print(f"Best so far: {best_score:.6f}")

    # Phase 2: Focused optimization (~1000s)
    print(f"\n=== Phase 2: Focused Optimization (target < {best_score:.6f}) ===")
    focus_deadline = deadline - 200.0

    # Pick top unique-length candidates
    seen_n = set()
    top_cands = []
    for sc, seq, n in all_candidates:
        if n not in seen_n:
            top_cands.append((sc, seq, n))
            seen_n.add(n)
        if len(top_cands) >= 2:
            break

    # If we have a parent, always include it
    if parent is not None and n_main not in seen_n:
        top_cands.append((best_score, parent.copy(), n_main))

    for sc, f_start, n in top_cands:
        if time.monotonic() > focus_deadline - 60:
            break
        print(f"\n--- Optimize n={n} (start={sc:.6f}) ---")
        S = np.sqrt(2.0 * n)
        f_start = f_start * (S / np.sum(f_start))

        # Alternating LP + FISTA
        f_opt, s_opt = lp_optimize_loop(f_start, focus_deadline, S,
                                         target_scale=0.99, lp_every=5,
                                         verbose_prefix=f"[n={n}] ")
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()
            print(f"  Improved to: {best_score:.6f}")

    # Phase 3: Final polish (~200s)
    print(f"\n=== Phase 3: Final Polish (best={best_score:.6f}) ===")
    if best_sequence is not None and time.monotonic() < deadline - 10:
        n_best = len(best_sequence)
        S = np.sqrt(2.0 * n_best)
        print(f"Polishing n={n_best}...")

        # Intensive FISTA with high temperature range
        f_pol, s_pol = fista_smoothed(best_sequence, time.monotonic() + 60.0, S,
                                       temp_init=20.0, temp_end=100.0,
                                       n_anneal=200, verbose_prefix="[polish] ")
        if s_pol < best_score:
            best_score = s_pol
            best_sequence = f_pol.copy()

        # LP polish
        if time.monotonic() < deadline - 20:
            f_pol, s_pol = lp_optimize_loop(best_sequence, deadline, S,
                                             target_scale=0.995, lp_every=3,
                                             verbose_prefix="[polishLP] ")
            if s_pol < best_score:
                best_score = s_pol
                best_sequence = f_pol.copy()

        # One more FISTA
        if time.monotonic() < deadline - 10:
            f_pol, s_pol = fista_smoothed(best_sequence, deadline, S,
                                           temp_init=50.0, temp_end=200.0,
                                           n_anneal=100, verbose_prefix="[final] ")
            if s_pol < best_score:
                best_score = s_pol
                best_sequence = f_pol.copy()

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(1024)
        best_score = evaluate_sequence(best_sequence.tolist())

    # Final verification
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    final_score = evaluate_sequence([float(x) for x in best_sequence])
    if not np.isfinite(final_score):
        print("WARNING: Final failed! Using fallback.")
        best_sequence = np.clip(best_sequence * 0.5, 0.0, 1000.0)
        final_score = evaluate_sequence([float(x) for x in best_sequence])

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, "
          f"internal={best_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_sequence]
