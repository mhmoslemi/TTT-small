Optimized step function search using smoothed gradient methods with temperature annealing,
fast O(n) coordinate descent with incremental convolution updates, and diverse initializations.
Combines Adam optimizer on log-sum-exp smoothed objective with exact-coordinate polishing.

import numpy as np
from scipy.optimize import minimize
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fast_convolve(a):
    """Compute full convolution of a with itself."""
    return np.convolve(a, a)


def smooth_max_val(v, T):
    """Numerically stable log-sum-exp approximation of max."""
    v_max = float(np.max(v))
    if T < 1e-6:
        return v_max
    shifted = T * (v - v_max)
    shifted = np.clip(shifted, -80, 80)
    return v_max + float(np.log(np.sum(np.exp(shifted)))) / T


def smooth_max_grad(v, T):
    """Gradient of log-sum-exp smoothing w.r.t. v."""
    v_max = float(np.max(v))
    shifted = T * (v - v_max)
    shifted = np.clip(shifted, -80, 80)
    weights = np.exp(shifted)
    weights = weights / np.sum(weights)
    return weights


def obj_smoothed(a, T, n):
    """Smoothed objective: 2*n*smooth_max(conv(a,a)) with sum(a)=1."""
    conv = np.convolve(a, a)
    conv_max = float(np.max(conv))
    if conv_max < 1e-15:
        return 0.0
    return 2.0 * n * smooth_max_val(conv, T)

    """Gradient of smoothed objective w.r.t. a (a is on simplex, sum=1)."""
    conv = np.convolve(a, a)
    v_max = float(np.max(conv))
    shifted = T * (conv - v_max)
    shifted = np.clip(shifted, -80, 80)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    w = w / w_sum  # softmax weights over conv values

    # Actually: conv[k] = sum_i a[i]*a[k-i]

        s = n
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    lr = 0.01
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    for T in T_schedule:
        step_in_T = 0
        while time.monotonic() < time_limit and step_in_T < 300:
            t += 1
            step_in_T += 1

            obj = obj_smoothed(a, T, n)
            grad = obj_smoothed_grad(a, T, n)

            # Project gradient onto simplex tangent space
            grad_mean = np.mean(grad)
            grad_tangent = grad - grad_mean

            m = beta1 * m + (1 - beta1) * grad_tangent
            v = beta2 * v + (1 - beta2) * grad_tangent ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step
            a_new = np.maximum(a_new, 0)
            s = np.sum(a_new)
            if s < 1e-10:
                a_new = a
            else:
                a_new = a_new / s

            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

            a = a_new

        # Adjust learning rate
        lr *= 0.85

    return best_a, best_obj


def fast_coord_descent(a, c, n_rounds=10, time_limit=None, verbose=False):
    """
    Fast coordinate descent with incremental convolution update.
    a: the sequence (sum=1)
    c: current convolution np.convolve(a,a)
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    c = c.copy().astype(np.float64)
    best_obj = 2.0 * n * np.max(c)  # since sum(a)=1
    c = np.convolve(a, a)  # ensure c is correct
    best_obj = 2.0 * n * float(np.max(c))
    best_a = a.copy()
    best_c = c.copy()

    for rnd in range(n_rounds):
        improved = 0
        for j in range(n):
            if time_limit is not None and time.monotonic() > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-15:
                candidates = [0.0, 0.5 / n, 1.0 / n, 2.0 / n, 4.0 / n]
            else:
                candidates = [0.0, old_val * 0.2, old_val * 0.5, old_val * 0.75,
                              old_val, old_val * 1.33, old_val * 1.75, old_val * 2.5]

            best_local_obj = best_obj
            best_local_val = old_val
            best_local_c = c.copy()

            for new_val in candidates:
                new_val = max(0.0, new_val)
                if abs(new_val - old_val) < 1e-12:
                    continue

                delta = new_val - old_val
                # Update convolution: c[k] += 2*delta*a[k-j] for k in [j, j+n-1]
                # Special: c[2j] gets additional delta^2
                c_new = c.copy()
                k_start = j
                k_end = min(2 * n - 2, j + n - 1)
                c_new[k_start:k_end + 1] += 2.0 * delta * a[k_start - j:k_end - j + 1]
                if 0 <= 2 * j <= 2 * n - 2:
                    c_new[2 * j] += delta * delta

                obj_new = 2.0 * n * float(np.max(c_new))
                if obj_new < best_local_obj - 1e-12:
                    best_local_obj = obj_new
                    best_local_val = new_val
                    best_local_c = c_new

            if best_local_val != old_val:
                a[j] = best_local_val
                c = best_local_c
                improved += 1
                if best_local_obj < best_obj:
                    best_obj = best_local_obj
                    best_a = a.copy()
                    best_c = c.copy()

        if verbose:
            print(f"    CD round {rnd}: improved {improved} coords, obj={best_obj:.6f}")

        if improved == 0:
            break

    return best_a, best_c, best_obj


def diverse_init(n, rng, init_type):
    """Generate diverse initializations."""
    if init_type == 0:
        # Exponential decay
        ratio = rng.uniform(0.6, 0.95)
        a = ratio ** np.arange(n)
    elif init_type == 1:
        # Exponential growth
        ratio = rng.uniform(0.6, 0.95)
        a = ratio ** np.arange(n - 1, -1, -1)
    elif init_type == 2:
        # Gaussian bump (off-center)
        center = n * rng.uniform(0.25, 0.75)
        width = n * rng.uniform(0.05, 0.25)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
    elif init_type == 3:
        # Linear ramp with noise
        a = np.linspace(0.1, 1.0, n) * (1 + 0.3 * rng.standard_normal(n))
        a = np.maximum(a, 0)
    elif init_type == 4:
        # Two-peak
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w1 = n * rng.uniform(0.03, 0.1)
        w2 = n * rng.uniform(0.03, 0.1)
        a = np.exp(-0.5 * ((np.arange(n) - c1) / w1) ** 2) + \
            rng.uniform(0.3, 1.0) * np.exp(-0.5 * ((np.arange(n) - c2) / w2) ** 2)
    elif init_type == 5:
        # Power law
        alpha = rng.uniform(0.5, 3.0)
        a = (np.arange(n) + 1) ** (-alpha)
    elif init_type == 6:
        # Sinusoidal
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0, np.sin(2 * np.pi * freq * np.arange(n) / n + phase))
    elif init_type == 7:
        # Piecewise constant
        n_blocks = int(rng.integers(2, 8))
        a = np.concatenate([np.full(max(1, n // n_blocks), rng.uniform(0.1, 2.0))
                            for _ in range(n_blocks)])[:n]
    else:
        # Random
        a = rng.random(n)

    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s


def lp_step(a, c, n, M_relax=1.0):
    """
    LP step: find g maximizing sum(g) subject to conv(a,g) <= M, g >= 0.
    Return the solution g (normalized to sum=1) or None.
    """
    from scipy.optimize import linprog

    M = float(np.max(c)) * M_relax
    if M < 1e-15:
        return None

    # Build constraint matrix A where A[k,j] = a[k-j] if 0<=k-j<n else 0
    # conv(a,g)[k] = sum_j a[k-j]*g[j] <= M for k=0,...,2n-2
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A_ub[k, j_start:j_end + 1] = a[j_end:j_start - 1: -1 if j_start > j_end else 1]
        # More carefully:
        # conv(a,g)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[k-j] * g[j]
        # For j from max(0,k-n+1) to min(n-1,k):
        #   term is a[k-j] * g[j]
    # Rebuild properly
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A_ub[k, j] = a[k - j]

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        g = res.x
        g_sum = np.sum(g)
        if g_sum > 1e-10:
            return g / g_sum
    return None


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-strategy search: smoothed gradient (Adam) with temperature annealing,
    fast coordinate descent with O(n) updates, LP alternating, and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0

    best_seq = None
    best_score = np.inf

    # Evaluate parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.all(np.isfinite(parent)):
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent: n={parent.size}, obj={obj:.6f}")
    except Exception:
        pass

    # N values to try (prioritize around parent size)
    n_values = [200, 250, 300, 350, 400, 500, 700, 1000]

    # Temperature schedules
    T_schedules = [
        [5, 10, 20, 50, 100],
        [10, 30, 80, 200],
        [3, 8, 20, 50, 100, 200],
    ]

    strategies_per_n = [8, 6, 6, 5, 4, 3, 3, 2]

    total_strategies = sum(strategies_per_n)

    for idx_n, n in enumerate(n_values):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 45:
            break

        n_strat = strategies_per_n[idx_n]
        time_per_strat = (remaining - 10) / n_strat
        print(f"\nN={n} ({remaining:.0f}s left, {n_strat} strategies, ~{time_per_strat:.0f}s each)")

        for s_idx in range(n_strat):
            if time.monotonic() > deadline:
                break
            strat_time = min(time_per_strat, deadline - time.monotonic() - 5)
            if strat_time < 5:
                break

            # Choose initialization
            if s_idx == 0 and best_seq is not None and len(best_seq) == n:
                a_init = best_seq.copy()
                init_type = -1
            else:
                init_type = (s_idx + idx_n) % 9
                a_init = diverse_init(n, rng, init_type)

            # Strategy: Adam smoothed + CD
            T_sched = T_schedules[s_idx % len(T_schedules)]
            adam_limit = time.monotonic() + strat_time * 0.5
            a_opt, obj_opt = adam_smoothed(a_init, n, T_sched, adam_limit)

            # Coordinate descent polish
            c_opt = np.convolve(a_opt, a_opt)
            cd_limit = min(time.monotonic() + strat_time * 0.35, deadline - 10)
            a_cd, c_cd, obj_cd = fast_coord_descent(a_opt, c_opt,
                                                     n_rounds=15, time_limit=cd_limit)

            if obj_cd < obj_opt:
                a_opt, obj_opt = a_cd, obj_cd

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} s={s_idx} (init={init_type}): obj={obj_opt:.6f}")

            # Try LP step
            if time.monotonic() < deadline - 20 and n <= 400:
                c_cur = np.convolve(a_opt, a_opt)
                g = lp_step(a_opt, c_cur, n, M_relax=1.0)
                if g is not None:
                    # Blend and re-optimize
                    for blend in [0.3, 0.5, 0.7]:
                        a_mixed = (1 - blend) * a_opt + blend * g
                        a_mixed = np.maximum(a_mixed, 0)
                        sm = np.sum(a_mixed)
                        if sm < 1e-10:
                            continue
                        a_mixed = a_mixed / sm
                        c_mixed = np.convolve(a_mixed, a_mixed)
                        a_lp, c_lp, obj_lp = fast_coord_descent(
                            a_mixed, c_mixed, n_rounds=5,
                            time_limit=min(time.monotonic() + 5, deadline - 5))
                        if obj_lp < best_score:
                            best_score = obj_lp
                            best_seq = a_lp.copy()
                            print(f"  n={n} LP blend={blend}: obj={obj_lp:.6f}")
                            a_opt = a_lp.copy()

    # Final intensive polishing
    if best_seq is not None and time.monotonic() < deadline:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        print(f"\nFinal polish: n={n_best}, obj={best_score:.6f}, {remaining:.0f}s left")

        # Extended coordinate descent
        c_best = np.convolve(best_seq, a_best if False else best_seq)
        c_best = np.convolve(best_seq, best_seq)
        a_final, c_final, obj_final = fast_coord_descent(
            best_seq, c_final if False else np.convolve(best_seq, best_seq),
            n_rounds=50, time_limit=deadline - 5)

        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  Final CD: obj={obj_final:.6f}")

        # One more Adam pass from the polished solution
        if time.monotonic() < deadline - 20:
            T_final = [50, 100, 200, 500]
            a_am, obj_am = adam_smoothed(best_seq, n_best, T_final,
                                         time_limit=deadline - 5)
            c_am = np.convolve(a_am, a_am)
            a_am2, c_am2, obj_am2 = fast_coord_descent(a_am, c_am,
                                                       n_rounds=20,
                                                       time_limit=deadline - 3)
            if obj_am2 < best_score:
                best_score = obj_am2
                best_seq = a_am2.copy()
                print(f"  Final Adam+CD: obj={obj_am2:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)

    # Final safety check
    final_score = compute_obj(best_seq)
    if np.isinf(final_score) or np.isnan(final_score):
        best_seq = np.ones(300)
        final_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={final_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
