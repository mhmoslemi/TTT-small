import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

    """Compute subgradient of max(conv(a,a)) w.r.t. a."""
    b = fast_conv(a)
    n = len(a)
    # Find all positions where b achieves its maximum
    max_val = b.max()
    argmax_positions = np.where(np.isclose(b, max_val, rtol=0, atol=1e-10))[0]
    
    subgrad = np.zeros(n)
    for k in argmax_positions:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                subgrad[i] += a[j]
    return subgrad / len(argmax_positions)

def subgradient_descent(a, max_iters, deadline, lr_init, tracker):
    """Subgradient descent with Polyak step size."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)
    
    # Polyak step size: needs a lower bound on optimal
    # J >= 2 (for uniform), actual lower bound is higher
    J_lower = 1.45  # conservative lower bound
    
    lr = lr_init
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        subgrad = subgradient_of_max_conv(a)
        
        # Polyak step size
        current_J = compute_J(a)
        if current_J - J_lower > 1e-10:
            lr = (current_J - J_lower) / (np.dot(subgrad, subgrad) + 1e-10)
            lr = min(lr, 1.0)
        
        a_new = a - lr * subgrad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        new_J = compute_J(a_new)
        if new_J < best_J:
            best_J = new_J
            best_a = a_new.copy()
            tracker.update(a_new)
        
        a = a_new
    
    # Return the best found, not necessarily the last
    return best_a

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def peak_shaving(a, max_iters, deadline, tracker):

    best_J = current_J

        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        # Use top-5 peaks for subgradient
        top_idx = np.argsort(b)[-5:][::-1]
        weights = np.array([1.0, 0.4, 0.2, 0.1, 0.05])

        s = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    s[i] += w * a[j]

        if s_norm < 1e-12:
            break

        a_new = np.maximum(a_new, 0.0)

        tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Peak Shaving ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Peak Shaving n={n} {'='*50}", flush=True)

        a = peak_shaving(tracker.best, 1000, deadline, tracker)

        # Try on resampled to slightly larger sizes
        for factor in [1.25, 1.5, 2.0]:
            n2 = int(n * factor)
            if n2 <= 768 and time.monotonic() < deadline:
                a2 = resample_seq(tracker.best, n2)
                a2 = peak_shaving(a2, 500, deadline, tracker)

        tracker.report(f"Peak Shaving: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Simulated Annealing with LP + Peak Shaving ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} SA n={n} {'='*50}", flush=True)

        temperature = 0.08
        n_restarts = 50

        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break

            a = tracker.best.copy().astype(np.float64)
            # Mix of Gaussian noise and random perturbation
            noise = rng.normal(0, temperature, n)
            # Occasionally do a larger perturbation
            if rng.random() < 0.15:
                idx = rng.choice(n, size=max(1, n // 10), replace=False)
                noise[idx] += rng.normal(0, temperature * 3, len(idx))
            a = a + noise
            a = np.maximum(a, 0.001)
            a = a / a.sum()

            iters = max(10, min(50, 60 - r // 2))
            a = run_lp_mirror_descent(a, iters, deadline, tracker)

            # Quick peak shaving on result
            if time.monotonic() < deadline:
                a = peak_shaving(a, 30, deadline, tracker)

            temperature *= 0.96

        tracker.report(f"SA: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Smooth L-BFGS-B ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} L-BFGS-B n={n} {'='*50}", flush=True)

        # Try multiple beta values for log-sum-exp
        for beta in [30.0, 60.0, 100.0]:
            if time.monotonic() > deadline:
                break

            # Define objective as top-level compatible (using a wrapper)
            _lbb_beta = beta
            _lbb_n = n

            def _lbb_obj(x, beta_val=_lbb_beta):
                ax = np.maximum(x, 0.0)
                s = ax.sum()
                if s < 1e-8:
                    return 1e10
                ax = ax / s
                b = fast_conv(ax)
                max_b = b.max()
                # Soft max
                shifted = b - max_b
                soft_max = max_b + np.log(np.sum(np.exp(shifted * beta_val))) / beta_val
                return float(2 * len(ax) * soft_max)

            bounds_list = [(0.0, None)] * n
            x0 = tracker.best.astype(np.float64)
            res = minimize(_lbb_obj, x0, method='L-BFGS-B',
                           bounds=bounds_list,
                           options={'maxiter': 200, 'max_fun': 30000, 'ftol': 1e-18})
            if res.fun < 1e10:
                tracker.update(res.x)

        tracker.report(f"L-BFGS-B: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final LP Polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        n = tracker.best.size
        if n <= 512:
            t0 = time.monotonic()
            print(f"\n{'='*50} Final LP n={n} {'='*50}", flush=True)
            a = run_lp_mirror_descent(tracker.best, 400, deadline, tracker)
            a = peak_shaving(a, 200, deadline, tracker)
            tracker.report(f"Final: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
