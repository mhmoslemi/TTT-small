```python
"""
Minimax LP-based optimization for step function coefficients.
Uses successive linearization of the peak-convolution minimization,
combined with line search, block refinement, and multi-scale search.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    s = np.sum(a)
    if s < 1e-15:
        return a
    return a / s


def lp_minimax_step(a, top_k=50):
    """
    One step of the successive linearization minimax LP.
    Returns the LP solution (unnormalized) and the LP objective value.
    """
    n = len(a)
    L = 2 * n - 1
    conv = np.convolve(a, a)
    
    # Find active constraints (top-k largest convolution values)
    if top_k >= L:
        active = np.arange(L)
    else:
        # Use top_k values
        top_indices = np.argsort(conv)[::-1][:top_k]
        active = top_indices
    
    # Build the constraint matrix
    # For each active k: G[k,:] @ a - t <= c_k
    # where G[k,i] = d(c_k)/d(a_i)
    
    # Efficient computation of G rows for active k values
    # G[k,i] = 2*a[k-i] if 0 <= k-i <= n-1 and k != 2i
    #         2*a[i]   if k == 2i (and 0 <= i <= n-1)
    #         a[k-i]   if exactly one of the conditions holds
    # Actually, as derived: G[k,i] = 2*a[k-i] if 0<=k-i<=n-1, unless k=2i where it's 2*a[i]
    # But 2*a[k-i] when k=2i gives 2*a[i], so it's the same!
    # So: G[k,i] = 2*a[k-i] if 0 <= k-i <= n-1, else 0.
    
    # Build A_ub matrix: for each active k, row is [G[k,0], ..., G[k,n-1], -1]
    # RHS: c_k for each active k
    
    A_ub = np.zeros((len(active), n + 1))
    b_ub = np.zeros(len(active))
    
    for idx, k in enumerate(active):
        # G[k, i] = 2*a[k-i] for i in [max(0,k-n+1), min(n-1, k)]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        A_ub[idx, i_start:i_end+1] = 2.0 * a[k - i_end:k - i_start + 1][::-1]
        A_ub[idx, n] = -1.0
        b_ub[idx] = conv[k]
    
    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: a_i >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x[:n], float(res.x[n])
    return None, None


def lp_minimax_phase(a, max_time, top_k=50, verbose=True):
    """
    Phase of successive linearization minimax LP.
    """
    a = a.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    a = normalize(a)
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(10000):
        if time.monotonic() - start > max_time:
            break
        
        a_new, t_new = lp_minimax_step(a, top_k)
        if a_new is None:
            break
        
        a_new = np.clip(a_new, 0.0, None)
        a_new = normalize(a_new)
        
        # Line search
        best_tau = 0.0
        best_tau_obj = best_obj
        for tau in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
            a_cand = (1.0 - tau) * a + tau * a_new
            a_cand = np.clip(a_cand, 0.0, None)
            a_cand = normalize(a_cand)
            obj = compute_obj(a_cand)
            if obj < best_tau_obj:
                best_tau_obj = obj
                best_tau = tau
        
        if best_tau == 0.0:
            # Try smaller steps
            for tau in [0.02, 0.01, 0.005, 0.001]:
                a_cand = (1.0 - tau) * a + tau * a_new
                a_cand = np.clip(a_cand, 0.0, None)
                a_cand = normalize(a_cand)
                obj = compute_obj(a_cand)
                if obj < best_tau_obj:
                    best_tau_obj = obj
                    best_tau = tau
        
        if best_tau > 0:
            a = (1.0 - best_tau) * a + best_tau * a_new
            a = np.clip(a, 0.0, None)
            a = normalize(a)
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                if verbose and it % 50 == 0:
                    print(f"    LP iter {it}: obj={obj:.6f}, tau={best_tau:.3f}")
        else:
            # No improvement, stop
            if it > 10:
                break
    
    return best_a, best_obj


def soft_max_gradient(a, T):
    """
    Compute the gradient of the soft-max of the convolution values.
    """
    n = len(a)
    L = 2 * n - 1
    conv = np.convolve(a, a)
    max_c = float(np.max(conv))
    
    # Soft-max weights
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p = exp_vals / np.sum(exp_vals)
    
    # Soft-max value
    sm_val = max_c + T * np.log(np.sum(exp_vals))
    
    # Gradient: d(sm_val)/d(a_i) = sum_k p_k * G[k,i]
    # = sum_k p_k * 2*a[k-i] (for valid k-i)
    # = 2 * (p_convolved_with_a) ... let's compute efficiently
    
    # grad_i = sum_k p[k] * 2 * a[k-i] for 0<=k-i<=n-1
    # = 2 * sum_k p[k] * a[k-i]
    # This is 2 * (p * a)_i where * is correlation (not convolution)
    # (p * a)_i = sum_k p[k] a[k-i] = (p a^R)_i where a^R is reversed
    # Actually: sum_k p[k] a[k-i] for k-i in [0,n-1]
    # = sum_j p[j+i] a[j] for j in [0,n-1] and j+i in [0,L-1]
    # = (flip(p) * a)_i ... 
    
    # Let me just compute it directly using the convolution structure
    # grad_i = 2 * sum_{k: 0<=k-i<=n-1} p[k] * a[k-i]
    # Let j = k-i, then k = j+i, j in [0, n-1], j+i in [0, L-1]
    # grad_i = 2 * sum_j p[j+i] * a[j] for j in [max(0,-i), min(n-1, L-1-i)]
    #        = 2 * sum_j a[j] * p[j+i]
    
    # This is 2 * (a * flip(p))_i where flip(p)_j = p[L-1-j]
    # Actually: (a * q)_i = sum_j a[j] q[i-j]
    # We want sum_j a[j] p[j+i] = sum_j a[j] p[i-(-j)] ... 
    
    # Let me use a different approach: 
    # grad = 2 * (correlation of p with a)
    # corr_i = sum_j p[j+i] * a[j]
    # This is the cross-correlation of p and a at lag i.
    # In numpy: np.correlate(p, a, mode='full') gives cross-correlation
    # np.correlate(p, a, 'full')[i + n - 1] = sum_j p[j+i] * a[j] (approximately)
    
    # Actually, let me just use the direct computation:
    # For each i, grad_i = 2 * sum_{j=0}^{n-1} a[j] * p[j+i] (if j+i < L)
    
    # Use FFT for efficiency when n is large
    if n > 50:
        # Pad to next power of 2
        size = 1
        while size < L + n:
            size *= 2
        fa = np.fft.rfft(a, size)
        fp = np.fft.rfft(p, size)
        # We want sum_j a[j] p[j+i] = (a * flip(p))[i + something]
        # flip(p)[j] = p[L-1-j]
        # (a * flip(p))[k] = sum_j a[j] flip(p)[k-j] = sum_j a[j] p[L-1-k+j]
        # We want sum_j a[j] p[j+i] = sum_j a[j] p[L-1-(L-1-j-i)] 
        # = (a * flip(p))[L-1-i]
        # So: grad_i = 2 * (a * flip(p))[L-1-i]
        flipped_p = p[::-1]
        corr = np.fft.irfft(fa * np.fft.rfft(flipped_p, size), size)[:L]
        grad = 2.0 * corr[L-1::-1][:n]  # grad[i] = 2*corr[L-1-i]
    else:
        grad = np.zeros(n)
        for i in range(n):
            s = 0.0
            for j in range(n):
                ki = j + i
                if ki < L:
                    s += a[j] * p[ki]
            grad[i] = 2.0 * s
    
    return sm_val, grad
