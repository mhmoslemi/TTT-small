def spectral_time_projection(f, n_iter=2000, rng=None, time_budget=30.0):
    """
    Alternate between flattening |F| (frequency) and enforcing f >= 0 (time).
    This is a projection algorithm that should converge to a local minimum.
    """
    if rng is None:
        rng = np.random.default_rng(42)
    
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()
    
    best_f = f.copy()
    best_score = compute_score(f)
    start = time.monotonic()
    
    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break
        
        # Frequency domain: flatten magnitude
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Target: make mag as flat as possible (except DC)
        # Current energy: sum(|F|^2) / n_fft = sum(f^2)
        # For flat: each non-DC bin has magnitude = sqrt(sum(|F|^2) / (len-1))

        # Interpolate between current and target (soft projection)
        alpha = 0.3
        mag_new = (1 - alpha) * mag + alpha * np.full(n_bins, target_mag)
        mag_new[0] = mag[0]  # DC fixed

"""
Optimization of non-negative step function to minimize 2*n*max(conv(f,f))/sum(f)^2.
Uses: (1) spectral-time projection for global flattening, (2) convolution-based
soft subgradient flow with momentum, (3) LP refinement with line search,
(4) coordinate descent polishing. Multi-phase with annealed softness.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute score: 2*n*max(conv(f,f))/sum(f)^2"""
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (s * s)


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def spectral_time_projection(f, n_iter=3000, alpha=0.3, rng=None, time_budget=30.0):
    """
    Alternate between flattening |F| (frequency) and enforcing f >= 0 (time).
   Projection algorithm that converges to a local minimum of max autocorrelation.
    """
    if rng is None:
        rng = np.random.default_rng(42)

    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()

    best_f = f.copy()
    best_score = compute_score(f)
    start = time.monotonic()
    stall = 0

    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break

        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)

        # Target: flatten magnitude spectrum (keep DC fixed)
        total_power = np.sum(mag ** 2)
        n_bins = len(mag)
        target_mag = np.sqrt(max(0, total_power - mag[0] ** 2) / (n_bins - 1))

        # Soft interpolation
        mag_new = (1 - alpha) * mag + alpha * np.full(n_bins, target_mag)
        mag_new[0] = mag[0]

        # Periodic randomization to escape local minima
        if it % 100 == 99:
            noise = 1 + 0.02 * rng.normal(0, 1, n_bins - 1)
            mag_new[1:] = np.maximum(mag_new[1:] * noise, 0)

        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]

        # Non-negativity projection
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s

        score = compute_score(f_new)
        if score < best_score - 1e-10:
            best_score = score
            best_f = f_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break

        f = f_new

    return best_f, best_score


def compute_subgradient(f, weights):
    """
    Compute gradient of max_k (f*f)[k] with soft weights.
    grad[i] = sum_k weights[k] * f[k-i] for valid indices.
    Uses convolution for efficiency: O(n log n).
    """
    n = len(f)
    # grad[i] = sum_k weights[k] * f[k-i]
    # This equals np.convolve(weights, f[::-1], 'full')[n-1 : 2*n-1]
    temp = np.convolve(weights, f[::-1], mode='full')
    grad = temp[n - 1:2 * n - 1]
    return grad


def soft_subgradient_flow(f, n_iter=20000, eta0=0.05, momentum=0.9,
                          softness_start=0.1, softness_end=0.001,
                          rng=None, time_budget=40.0):
    """
    Subgradient descent with annealed softness and momentum.
    Softness controls how many peaks contribute to the gradient.
    """
    if rng is None:
        rng = np.random.default_rng(42)

    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    v = np.zeros(n)
    eta = eta0
    start = time.monotonic()
    no_improve = 0

    for it in range(n_iter):
        if time.monotonic() - start > time_budget:
            break

        # Anneal softness
        progress = it / n_iter
        softness = softness_start * (softness_end / softness_start) ** progress

        conv = np.convolve(f, f)
        M = conv.max()

        # Soft weights (exponential decay from peak)
        weights = np.exp((conv - M) / max(softness, 1e-6))
        # Threshold to keep computation reasonable
        active = weights > 0.001
        if not active.any():
            active[np.argmax(conv)] = True

        # Compute gradient via convolution (O(n log n))
        w_active = weights.copy()
        w_active[~active] = 0
        grad = compute_subgradient(f, w_active)

        if grad.max() > 0:
            grad = grad / grad.max()

        # Momentum update
        v = momentum * v + (1 - momentum) * grad
        f_new = f - eta * v
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s

        score = compute_score(f_new)

        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            no_improve = 0
            eta = min(eta * 1.005, 0.3)
        else:
            no_improve += 1
            eta = max(eta * 0.99, 0.0005)

        f = f_new

        if no_improve > 3000:
            # Perturb to escape
            f_pert = f * (1 + 0.08 * rng.normal(0, 1, n))
            f_pert = np.maximum(f_pert, 0)
            f_pert = normalize(f_pert)
            if compute_score(f_pert) < best_score:
                f = f_pert
                best_f = f.copy()
                best_score = compute_score(f)
                eta = 0.05
                v = np.zeros(n)
                no_improve = 0
            elif it > n_iter * 0.7:
                break

    return best_f, best_score


def build_conv_matrix(f):
    """Build A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def lp_refine(f, time_budget=120.0, t0=0.3, rng=None, relaxation=0.0):
    """LP-based refinement with line search. Solves max sum(b) s.t. conv(f,b)<=M, b>=0."""
    if rng is None:
        rng = np.random.default_rng(42)

    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)

    best_f = f.copy()
    best_score = compute_score(f)
    t = t0
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max() * (1.0 + relaxation)

        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': 5})
        except Exception:
            break

        if not res.success:
            break

        g = np.maximum(res.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Line search: try multiple step sizes
        improved = False
        for step in [t, t * 0.5, t * 0.75, t * 1.5, t * 0.25, t * 2.0]:
            if step <= 0 or step > 2.0:
                continue
            f_new = (1 - step) * f + step * g
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s

            new_score = compute_score(f_new)
            if new_score < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                improved = True
                t = min(1.5, t * 1.08)
                break

        if not improved:
            t *= 0.6
            if t < 0.003:
                # Perturb
                f_pert = f * (1 + 0.03 * rng.normal(0, 1, n))
                f_pert = np.maximum(f_pert, 0)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    t = 0.1
                else:
                    break

        it += 1
        if it % 5 == 0:
            print(f"    LP iter {it}: score={best_score:.6f}, t={t:.4f}")

    return best_f, best_score


def coordinate_descent(f, n_sweeps=20, rng=None, time_budget=30.0):
    """Coordinate descent polishing with geometric search."""
    if rng is None:
        rng = np.random.default_rng(42)

    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)
    start = time.monotonic()

    for sweep in range(n_sweeps):
        if time.monotonic() - start > time_budget:
            break
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-10:
                continue
            best_val = original
            best_local = best_score

            for factor in [0.1, 0.25, 0.5, 0.75, 1.33, 1.67, 2.0, 4.0]:
                f[i] = max(0.0, original * factor)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                score = compute_score(f)
                if score < best_local:
                    best_local = score
                    best_val = f[i]

            f[i] = best_val
            if best_local < best_score:
                best_score = best_local

        f = np.maximum(f, 0)
        s = f.sum()
        if s > 1e-15:
            f = f / s

    return f, best_score


def diverse_init(n, rng):
    """Generate a diverse initial sequence."""
    r = rng.random()
    if r < 0.2:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.3)
        f = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    elif r < 0.4:
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    elif r < 0.6:
        f = np.ones(n)
        for _ in range(3):
            c = rng.uniform(0.2, 0.8) * n
            w = rng.uniform(0.05, 0.2) * n
            f *= np.exp(-0.5 * (np.arange(n) - c) ** 2 / w ** 2)
        f += 0.1
    elif r < 0.8:
        f = np.ones(n)
        for _ in range(10):
            c = int(rng.integers(0, n))
            w = int(rng.integers(3, max(4, n // 5)))
            kernel = np.exp(-0.5 * (np.arange(n) - c) ** 2 / max(1, w ** 2))
            f *= (1 + 0.3 * rng.standard_normal() * kernel)
        f = np.abs(f) + 0.05
    else:
        freq = rng.uniform(1, 4)
        f = np.abs(np.sin(np.pi * freq * np.arange(n) / n)) + 0.2

    f = np.maximum(f, 0.01)
    f = normalize(f)
    f = f * (1 + 0.02 * rng.normal(0, 1, n))
    f = np.maximum(f, 0.001)
    f = normalize(f)
    return f


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: multi-phase optimization combining spectral-time projection,
    annealed subgradient flow, LP refinement, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0

    best_overall = None
    best_score_overall = np.inf

    # Try parent sequence
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None

    if parent is not None:
        parent = np.maximum(parent, 0)
        if parent.sum() > 0.01:
            parent = normalize(parent)
            parent_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
            best_overall = parent.copy()
            best_score_overall = parent_score

    # Determine lengths to try
    lengths = [512]
    if parent is not None:
        lengths = [len(parent)]
        if len(parent) > 256:
            lengths.append(256)
        if len(parent) < 1024:
            lengths.append(1024)
    # Remove duplicates while preserving order
    seen = set()
    unique_lengths = []
    for L in lengths:
        if L not in seen:
            seen.add(L)
            unique_lengths.append(L)
    lengths = unique_lengths

    print(f"Lengths to try: {lengths}")

    for n in lengths:
        remaining = deadline - time.monotonic()
        if remaining < 90:
            break

        print(f"\n=== n={n}, remaining={remaining:.0f}s ===")

        # Get starting sequence
        if parent is not None and len(parent) == n:
            f0 = parent.copy()
        else:
            f0 = diverse_init(n, rng)

        current = f0.copy()
        current_score = compute_score(current)
        print(f"  Initial score: {current_score:.6f}")

        # Phase 1: Spectral-time projection
        t1 = min(35, remaining * 0.12)
        if t1 > 5:
            f1, s1 = spectral_time_projection(current, n_iter=5000, alpha=0.3,
                                              rng=rng, time_budget=t1)
            print(f"  After spectral ({t1:.0f}s): {s1:.6f}")
            if s1 < best_score_overall:
                best_score_overall = s1
                best_overall = f1.copy()
            current = f1

        # Phase 2: Annealed soft subgradient flow (coarse)
        remaining2 = deadline - time.monotonic()
        t2 = min(50, remaining2 * 0.18)
        if t2 > 5:
            f2, s2 = soft_subgradient_flow(current, n_iter=30000, eta0=0.06,
                                           momentum=0.85, softness_start=0.1,
                                           softness_end=0.01, rng=rng, time_budget=t2)
            print(f"  After coarse subgrad ({t2:.0f}s): {s2:.6f}")
            if s2 < best_score_overall:
                best_score_overall = s2
                best_overall = f2.copy()
            current = f2

        # Phase 3: LP refinement (most powerful local method)
        remaining3 = deadline - time.monotonic()
        t3 = min(remaining3 * 0.35, 120)
        if t3 > 10:
            f3, s3 = lp_refine(current, time_budget=t3, t0=0.3, rng=rng)
            print(f"  After LP ({t3:.0f}s): {s3:.6f}")
            if s3 < best_score_overall:
                best_score_overall = s3
                best_overall = f3.copy()
            current = f3

        # Phase 4: Fine subgradient (tight softness)
        remaining4 = deadline - time.monotonic()
        t4 = min(35, remaining4 * 0.15)
        if t4 > 5:
            f4, s4 = soft_subgradient_flow(current, n_iter=30000, eta0=0.02,
                                           momentum=0.92, softness_start=0.02,
                                           softness_end=0.001, rng=rng, time_budget=t4)
            print(f"  After fine subgrad ({t4:.0f}s): {s4:.6f}")
            if s4 < best_score_overall:
                best_score_overall = s4
                best_overall = f4.copy()
            current = f4

        # Phase 5: Second LP pass (refined)
        remaining5 = deadline - time.monotonic()
        t5 = min(60, remaining5 * 0.25)
        if t5 > 10:
            f5, s5 = lp_refine(current, time_budget=t5, t0=0.15, rng=rng)
            print(f"  After LP2 ({t5:.0f}s): {s5:.6f}")
            if s5 < best_score_overall:
                best_score_overall = s5
                best_overall = f5.copy()
            current = f5

        # Phase 6: Coordinate descent
        remaining6 = deadline - time.monotonic()
        t6 = min(25, remaining6 * 0.2)
        if t6 > 5:
            f6, s6 = coordinate_descent(current, n_sweeps=30, rng=rng, time_budget=t6)
            print(f"  After CD ({t6:.0f}s): {s6:.6f}")
            if s6 < best_score_overall:
                best_score_overall = s6
                best_overall = f6.copy()
            current = f6

    # Final polishing on the best sequence found
    if best_overall is not None:
        remaining_final = deadline - time.monotonic()
        if remaining_final > 20:
            print(f"\n=== Final polish, remaining={remaining_final:.0f}s ===")

            # Fine subgradient
            t_sg = min(30, remaining_final * 0.3)
            if t_sg > 5:
                f_sg, s_sg = soft_subgradient_flow(best_overall, n_iter=30000, eta0=0.01,
                                                   momentum=0.95, softness_start=0.005,
                                                   softness_end=0.0005, rng=rng,
                                                   time_budget=t_sg)
                if s_sg < best_score_overall:
                    best_score_overall = s_sg
                    best_overall = f_sg.copy()
                    print(f"  Final subgrad: {s_sg:.6f}")

            # LP
            remaining_lp = deadline - time.monotonic()
            if remaining_lp > 15:
                t_lp = min(40, remaining_lp * 0.4)
                f_lp, s_lp = lp_refine(best_overall, time_budget=t_lp, t0=0.08, rng=rng)
                if s_lp < best_score_overall:
                    best_score_overall = s_lp
                    best_overall = f_lp.copy()
                    print(f"  Final LP: {s_lp:.6f}")

            # CD
            remaining_cd = deadline - time.monotonic()
            if remaining_cd > 10:
                f_cd, s_cd = coordinate_descent(best_overall, n_sweeps=15, rng=rng,
                                                time_budget=min(15, remaining_cd * 0.5))
                if s_cd < best_score_overall:
                    best_score_overall = s_cd
                    best_overall = f_cd.copy()
                    print(f"  Final CD: {s_cd:.6f}")

    # Verify with evaluate_sequence
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal: score={final_score:.6f}, n={len(final_list)}")
        return final_list

    # Fallback
    print("No improvement found, using fallback")
    if parent is not None:
        return [float(max(0, x)) for x in parent]
    return [1.0] * 512
