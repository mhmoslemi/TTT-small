"""
Step function sequence optimizer using adaptive LP + accelerated subgradient methods.
Key improvements: line search for LP blending, active-set LP acceleration,
Nesterov-style momentum, diverse structured initializations, wider length exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_grad(conv, a, tau):
    """Smoothed gradient of max(conv) using log-sum-exp weighting."""
    n = len(a)
    max_c = np.max(conv)
    shifted = np.clip(tau * (conv - max_c), -500, 500)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    
    # u[i] = sum_j a[j]*w[i+j] for i=0..n-1
    # = convolve(a[::-1], w, 'full')[n-1 : 2n-1]
    # v[i] = sum_j a[j]*w[i-j] for i=0..n-1 (w[x]=0 for x<0)
    # = convolve(a, w, 'full')[:n]
    
    # Use FFT for speed
    size = 1
    while size < n + len(w) - 1:
        size <<= 1
    
    a_flip_fft = np.fft.rfft(a[::-1], size)
    w_fft = np.fft.rfft(w, size)
    u_full = np.fft.irfft(a_flip_fft * w_fft, size)
    u = u_full[n-1:2*n-1]
    
    a_fft = np.fft.rfft(a, size)
    v_full = np.fft.irfft(a_fft * w_fft, size)
    v = v_full[:n]
    
    return 2.0 * (u + v)


def subgradient_optimize(a, max_iter=50000, tau_start=5.0, tau_end=100.0,
                         step_init=0.05, momentum=0.9):
    """Accelerated subgradient method with momentum."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    
    step = step_init
    v = np.zeros(n)  # momentum
    stall = 0
    patience = 500
    
    for it in range(max_iter):
        frac = it / max(1, max_iter)
        tau = tau_start + (tau_end - tau_start) * frac * frac  # quadratic schedule
        
        conv = compute_conv(a)
        cur_max = float(np.max(conv))
        
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                step *= 0.5
                if step < 1e-10:
                    break
                # Random kick
                rng = np.random.default_rng(it * 1000 + n)
                a = normalize(a + step * rng.normal(0, 0.1, n))
                v = np.zeros(n)
                stall = 0
                continue
        
        grad = compute_grad(conv, a, tau)
        grad_centered = grad - grad.mean()
        
        # Nesterov-like update
        v = momentum * v - step * grad_centered
        a_new = a + v
        a_new = normalize(a_new)
        a = a_new
        
        # Decrease step
        if it > 200:
            step *= 0.9997
    
    return best_a


def build_A_from_active(a, conv, M, tol=0.02):
    """Build constraint matrix using only near-active constraints."""
    n = len(a)
    # Find active constraints: conv[k] >= M * (1 - tol)
    active = np.where(conv >= M * (1.0 - tol))[0]
    
    A = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[idx, j_min:j_max+1] = a[k - j_max:k - j_min + 1][::-1]
    return A, active


def solve_lp(a, M, tol=0.02, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for active k, g >= 0."""
    n = len(a)
    conv = compute_conv(a)
    
    # Use active constraints only
    A, active = build_A_from_active(a, conv, M, tol)
    
    if len(A) == 0:
        return None, None
    
    c = -np.ones(n)
    b_ub = np.full(len(A), M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None


def lp_line_search(a, g, M, n_points=20):
    """Line search over blend parameter t in (0, 1)."""
    n = len(a)
    best_t = 0.0
    best_max = float(np.max(compute_conv(a)))
    
    # Try a range of t values
    ts = np.linspace(0.01, 0.95, n_points)
    for t in ts:
        a_t = (1.0 - t) * a + t * g
        a_t = normalize(a_t)
        max_t = float(np.max(compute_conv(a_t)))
        if max_t < best_max:
            best_max = max_t
            best_t = t
    
    if best_t > 0:
        # Refine around best_t
        lo = max(0.01, best_t - 0.05)
        hi = min(0.95, best_t + 0.05)
        ts_fine = np.linspace(lo, hi, 10)
        for t in ts_fine:
            a_t = normalize((1.0 - t) * a + t * g)
            max_t = float(np.max(compute_conv(a_t)))
            if max_t < best_max:
                best_max = max_t
                best_t = t
    
    return best_t, best_max


def lp_optimize(a, max_iter=100, tol=0.02, time_limit=5):
    """LP optimization with line search."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    
    stall = 0
    for it in range(max_iter):
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g, sum_g = solve_lp(a, M, tol=tol, time_limit=time_limit)
        if g is None:
            break
        
        if sum_g <= 1.0 + 1e-5:
            # Try full constraint set
            g, sum_g = solve_lp(a, M, tol=0.0, time_limit=time_limit)
            if g is None or sum_g <= 1.0 + 1e-5:
                stall += 1
                if stall > 30:
                    break
                # Perturb and retry
                rng = np.random.default_rng(it + n)
                a = normalize(a + 0.01 * rng.normal(0, 1, n))
                continue
            stall = 0
        
        g = normalize(g)
        
        # Line search
        t, new_max = lp_line_search(a, g, M)
        if t > 0 and new_max < best_max:
            a = normalize((1.0 - t) * a + t * g)
            best_max = new_max
            best_a = a.copy()
        else:
            stall += 1
            if stall > 30:
                break
    
    return best_a, best_max


def generate_initializations(n, rng):
    """Generate diverse initializations for sequence of length n."""
    inits = []
    x = np.arange(n)
    
    # 1. Uniform
    inits.append(np.ones(n))
    
    # 2. Gaussian centered
    for sigma in [0.2, 0.3, 0.5]:
        inits.append(np.exp(-0.5 * ((x - n/2) / (sigma * n))**2))
    
    # 3. Cosine-modulated
    for freq in [1, 2, 3, 5]:
        for phase in [0, np.pi/2, np.pi]:
            sig = np.exp(-0.5 * ((x - n/2) / (0.3*n))**2)
            inits.append(np.maximum(sig * (1 + 0.5*np.cos(2*np.pi*freq*x/n + phase)), 0))
    
    # 4. Chirp (quadratic phase)
    for rate in [0.5, 1.0, 2.0, 5.0]:
        sig = np.exp(-0.5 * ((x - n/2) / (0.3*n))**2)
        inits.append(np.maximum(sig * (1 + 0.3*np.cos(np.pi*rate*x**2/n)), 0))
    
    # 5. Random
    for _ in range(3):
        inits.append(rng.random(n))
    
    # 6. Two bumps
    for frac in [0.25, 0.33]:
        inits.append(np.exp(-0.5*((x/n - frac)/0.05)**2) + np.exp(-0.5*((x/n - (1-frac))/0.05)**2))
    
    # 7. Power law
    for p in [0.5, 1.0, 2.0]:
        mid = n / 2
        inits.append((1 + 0.1*np.abs(x/n - 0.5)**p)**(-1))
    
    return inits
