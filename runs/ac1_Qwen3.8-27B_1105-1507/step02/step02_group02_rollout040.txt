Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Multi-start FISTA on smoothed log-sum-exp objective with beta annealing
2. LP best-response refinement (full and top-K constrained)
3. Multiplicative weights (mirror descent) updates for diversity
4. Adaptive time budget allocation across phases and lengths
5. Perturbation-based local search to escape local minima

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2 directly."""
    a = np.clip(a, 0.0, 1000.0)

    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))

    """Gradient of (2n/beta)*log(sum exp(beta*conv(a,a))) for normalized a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(beta * (b - b_max))
    w = exp_b / exp_b.sum()
    grad = 2.0 * n * np.correlate(w, a, mode='valid')
    return grad

    n = len(v)

    n = len(a)

        jh = min(n - 1, k)

        y_new = z_proposed + ((t - 1.0) / t_new) * (z_proposed - z)
        z = z_proposed
        t = t_new
        y = y_new

        if k % 20 == 0 or k == max_iters - 1:
            score = compute_score(z)
            if score < best_score:
                best_score = score
                best = z.copy()

    score = compute_score(z)
    if score < best_score:
        best_score = score
        best = z.copy()

    return best, best_score


def grad_descent_phase(a, beta, max_iters, deadline, lr=None):
    """Simple gradient descent on smoothed objective."""
    n = len(a)
    a = project_simplex(a.copy())
    best = a.copy()
    best_score = compute_score(a)

    if lr is None:
        lr = 1.0 / (4.0 * n * (1.0 + 2.0 * beta * 0.3))

    for k in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad = compute_grad_smoothed(a, beta)
        a_new = project_simplex(a - lr * grad)
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best = a_new.copy()
        a = a_new

    return best, best_score


def multiplicative_weights_phase(a, beta, max_iters, deadline, eta0=0.1):
    """Mirror descent / multiplicative weights on smoothed objective."""
    n = len(a)
    a = project_simplex(a.copy())
    best = a.copy()
    best_score = compute_score(a)
    eta = eta0

    for k in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad = compute_grad_smoothed(a, beta)
        log_a = np.full(n, -100.0)
        mask = a > 1e-15
        log_a[mask] = np.log(a[mask])
        update = log_a - eta * grad
        update_max = update.max()
        new_a = np.exp(update - update_max)
        new_sum = new_a.sum()
        if new_sum < 1e-15:
            new_a = np.ones(n) / n
        else:
            new_a /= new_sum
        score = compute_score(new_a)
        if score < best_score:
            best_score = score
            best = new_a.copy()
            a = new_a
            eta = min(eta * 1.02, 1.0)
        else:
            eta *= 0.7

    return best, best_score


def lp_refine(a, max_iters, deadline, topk=None):
    """LP best-response iterations."""
    n = len(a)
    a = np.clip(a, 1e-6, None)
    a = a / a.sum()
    best = a.copy()
    best_score = compute_score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, topk=topk)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size
        t = max(0.05, 0.5 * (1.0 - it / max(float(max_iters), 1)))
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best = a_new.copy()
            a = a_new
        else:
            t2 = t * 0.3
            a_new2 = (1.0 - t2) * a + t2 * g
            a_new2 = np.clip(a_new2, 0.0, None)
            s2 = a_new2.sum()
            if s2 > 1e-10:
                a_new2 = a_new2 / s2
                score2 = compute_score(a_new2)
                if score2 < best_score:
                    best_score = score2
                    best = a_new2.copy()
            a = a_new2 if a_new2.sum() > 0.01 else a

    return best, best_score


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    t_arr = np.linspace(0, 1, n)

    # Resampled parent
    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        resampled = np.maximum(np.interp(t_arr, t_old, parent), 0)
        if resampled.sum() > 0.01:
            inits.append(resampled / resampled.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-enhanced (power law from edges)
    d = np.minimum(t_arr, 1.0 - t_arr) + 1e-10
    c = d ** 0.3
    inits.append(c / c.sum())

    # Power law alpha=0.5 from left
    c = 1.0 / (np.arange(n) + 1)**0.5
    inits.append(c / c.sum())
    inits.append(c[::-1] / c.sum())

    # Symmetric power law
    c = 1.0 / (np.arange(n) + 1)**0.5
    cs = np.minimum(c, c[::-1])
    inits.append(cs / cs.sum())

    # Cosine harmonics
    for k in [1, 2, 3]:
        c = 1.0 + 0.4 * np.cos(2 * np.pi * k * t_arr)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    # Bimodal
    c = np.exp(-10 * (t_arr - 0.2)**2) + np.exp(-10 * (t_arr - 0.8)**2) + 0.05
    inits.append(c / c.sum())

    # Triangular inverse
    c = 1.0 + np.abs(t_arr - 0.5)
    inits.append(c / c.sum())

    # Random smooth
    r = rng.standard_normal(n)
    # Smooth it
    from numpy import fft
    R = fft.rfft(r)
    freqs = fft.rfftfreq(n)
    R *= np.exp(-2 * freqs**2)
    r_smooth = np.maximum(fft.irfft(R, n), 0.01)
    inits.append(r_smooth / r_smooth.sum())

    # Random
    r = rng.uniform(0.2, 1.0, n)
    inits.append(r / r.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: multi-start FISTA + LP + multiplicative weights.
    Uses beta annealing and adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 30.0

    best = None
    best_score = float('inf')

    # Try parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.clip(p, 0.0, 1000.0)
            sc = compute_score(p)
            if sc < best_score:
                best = p.copy()
                best_score = sc
                print(f"Parent: J={sc:.6f} (n={p.size})", flush=True)
    except (NameError, AttributeError):
        pass

    # Primary lengths to try
    primary_n = 512
    if best is not None:
        primary_n = best.size

    # === PHASE 1: Multi-start FISTA + LP (60% of time) ===
    phase1_deadline = time.monotonic() + total_budget * 0.55
    print(f"\n=== Phase 1: Multi-start (budget {phase1_deadline - time.monotonic():.0f}s) ===", flush=True)

    # Try primary length first with all initializations
    n = primary_n
    inits = generate_initializations(n, rng, best)

    # Allocate time per initialization
    time_per_init = (phase1_deadline - time.monotonic()) / max(len(inits), 1)
    time_per_init = min(time_per_init, 120.0)  # Cap at 120s per init

    for i, a0 in enumerate(inits):
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 20:
            break

        a0 = np.clip(a0, 1e-4, None)
        a0 = a0 / a0.sum()
        init_score = compute_score(a0)
        print(f"  Init {i}: J={init_score:.6f}", flush=True)

        # Sub-phase 1a: FISTA with beta annealing
        sub_deadline = time.monotonic() + min(time_per_init * 0.4, remaining * 0.4)
        betas = [20, 50, 100, 300]
        for beta in betas:
            if time.monotonic() > sub_deadline:
                break
            max_it = 100
            a0, sc = fista_phase(a0, beta, max_it, sub_deadline)
            if sc < best_score:
                best_score = sc
                best = a0.copy()
                print(f"    FISTA beta={beta}: J={sc:.6f} *", flush=True)

        # Sub-phase 1b: LP refinement
        sub_deadline2 = min(time.monotonic() + min(time_per_init * 0.4, remaining * 0.35), phase1_deadline - 10)
        a0, sc = lp_refine(a0, 500, sub_deadline2)
        if sc < best_score:
            best_score = sc
            best = a0.copy()
            print(f"    LP: J={sc:.6f} *", flush=True)

        # Sub-phase 1c: FISTA with higher beta
        if time.monotonic() < phase1_deadline - 10:
            sub_deadline3 = min(time.monotonic() + min(time_per_init * 0.2, remaining * 0.2), phase1_deadline - 5)
            for beta in [500, 1000]:
                if time.monotonic() > sub_deadline3:
                    break
                a0, sc = fista_phase(a0, beta, 50, sub_deadline3)
                if sc < best_score:
                    best_score = sc
                    best = a0.copy()
                    print(f"    FISTA hi-beta={beta}: J={sc:.6f} *", flush=True)

        # Sub-phase 1d: More LP
        if time.monotonic() < phase1_deadline - 5:
            a0, sc = lp_refine(a0, 300, phase1_deadline - 5)
            if sc < best_score:
                best_score = sc
                best = a0.copy()
                print(f"    LP final: J={sc:.6f} *", flush=True)

    # === PHASE 2: Try other lengths (15% of time) ===
    phase2_deadline = min(time.monotonic() + total_budget * 0.15, deadline)
    if best is not None and phase2_deadline > time.monotonic() + 30:
        print(f"\n=== Phase 2: Other lengths ===", flush=True)
        other_lengths = [384, 640, 256, 768]
        for n2 in other_lengths:
            if time.monotonic() > phase2_deadline:
                break
            if n2 == best.size:
                continue
            # Resample best to new length
            t_old = np.linspace(0, 1, best.size)
            t_new = np.linspace(0, 1, n2)
            a2 = np.maximum(np.interp(t_new, t_old, best), 0.001)
            a2 = a2 / a2.sum()

            # Quick FISTA + LP
            sub_dl = min(time.monotonic() + 60, phase2_deadline - 5)
            a2, sc = fista_phase(a2, 100, 100, sub_dl)
            a2, sc = lp_refine(a2, 200, sub_dl)
            if sc < best_score:
                best_score = sc
                best = a2.copy()
                print(f"  n={n2}: J={sc:.6f} *", flush=True)

    # === PHASE 3: Focused refinement of best (20% of time) ===
    if best is not None:
        phase3_deadline = min(time.monotonic() + total_budget * 0.20, deadline)
        print(f"\n=== Phase 3: Focused refinement (budget {phase3_deadline - time.monotonic():.0f}s) ===", flush=True)
        n = best.size

        # Alternate FISTA and LP
        a = best.copy()
        cycles = 5
        per_cycle = (phase3_deadline - time.monotonic()) / cycles
        for cyc in range(cycles):
            if time.monotonic() > phase3_deadline:
                break
            sub_dl = time.monotonic() + per_cycle

            # FISTA with high beta
            beta = 200 + cyc * 200
            a, sc = fista_phase(a, beta, 80, min(sub_dl * 0.4, phase3_deadline))
            if sc < best_score:
                best_score = sc
                best = a.copy()

            # LP
            a, sc = lp_refine(a, 300, sub_dl)
            if sc < best_score:
                best_score = sc
                best = a.copy()
                print(f"  Cycle {cyc}: J={sc:.6f} *", flush=True)

    # === PHASE 4: Perturbation search (5% of time) ===
    if best is not None and time.monotonic() < deadline:
        phase4_deadline = min(time.monotonic() + total_budget * 0.05, deadline)
        print(f"\n=== Phase 4: Perturbation search ===", flush=True)
        n = best.size
        n_tries = int((phase4_deadline - time.monotonic()) / 5)
        n_tries = max(3, min(30, n_tries))

        for trial in range(n_tries):
            if time.monotonic() > phase4_deadline:
                break
            scale = 0.05 + 0.1 * rng.random()
            noise = rng.normal(0, scale, n)
            a_pert = best + noise
            a_pert = np.clip(a_pert, 0, None)
            if a_pert.sum() < 0.01:
                continue
            a_pert = a_pert / a_pert.sum()

            # Quick LP
            a_res, sc = lp_refine(a_pert, 100, phase4_deadline)
            if sc < best_score:
                best_score = sc
                best = a_res.copy()
                print(f"  Pert {trial}: J={sc:.6f} *", flush=True)

    # === PHASE 5: Final long LP polish ===
    if best is not None and time.monotonic() < deadline:
        print(f"\n=== Phase 5: Final LP polish (budget {deadline - time.monotonic():.0f}s) ===", flush=True)
        a_res, sc = lp_refine(best, 5000, deadline)
        if sc < best_score:
            best_score = sc
            best = a_res.copy()
            print(f"  Final LP: J={sc:.6f} *", flush=True)

        # One more FISTA pass
        if time.monotonic() < deadline - 10:
            a_res, sc = fista_phase(best, 500, 200, deadline)
            if sc < best_score:
                best_score = sc
                best = a_res.copy()

    # Final validation
    if best is None:
        best = np.ones(256) / 256

    final_list = [float(max(0, x)) for x in best]
    final_score = float(evaluate_sequence(final_list))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return final_list
