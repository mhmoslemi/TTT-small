"""
Optimization of non-negative sequence to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-accelerated subgradient descent with momentum, periodic reduced LP,
pairwise swap local search, and multiple initializations with random restarts.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve_auto(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    out_len = len(a) + len(b) - 1
    size = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:out_len]


def score_fast(a):
    """2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    conv = fft_convolve_auto(a)
    M = float(conv.max())
    S = float(a.sum())
    if S < 1e-10:
        return float('inf')
    return 2.0 * n * M / (S * S)


def subgrad_lag(a, k):
    """Subgradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    jlo = max(0, k - n + 1)
    jhi = min(k, n - 1)
    g = np.zeros(n)
    for j in range(jlo, jhi + 1):
        p = k - j
        if 0 <= p < n:
            if p == j:
                g[j] += 2.0 * a[j]
            else:
                g[j] += a[p]
                g[p] += a[j]
    return g


def subgrad_topk(a, top_k=3):
    """Weighted sum of subgradients at top-k lags."""
    n = len(a)
    conv = fft_convolve_auto(a)
    S = float(a.sum())
    mean_v = S * S / (2 * n - 1)
    idx = np.argsort(conv)[-top_k:]
    w = np.maximum(conv[idx] - mean_v, 0.0)
    ws = float(w.sum())
    if ws < 1e-15:
        w = np.full(top_k, 1.0 / top_k)
    else:
        w = w / ws
    g = np.zeros(n)
    for k, wi in zip(idx, w):
        g += wi * subgrad_lag(a, int(k))
    return g, conv


def reduced_lp_direction(f, M, top_m=50):
    """Solve reduced LP: max sum(g) s.t. g>=0, top-m conv constraints."""
    n = len(f)
    conv_f = fft_convolve_auto(f)
    # Pick top-m lags
    lags = np.argsort(conv_f)[-top_m:]
    # Build constraint matrix for these lags
    rows, cols, vals = [], [], []
    for k in lags:
        k = int(k)
        jlo = max(0, k - n + 1)
        jhi = min(k, n - 1)
        for j in range(jlo, jhi + 1):
            p = k - j
            if 0 <= p < n:
                if p == j:
                    rows.append(k)
                    cols.append(j)
                    vals.append(2.0 * f[j])
                else:
                    rows.append(k)
                    cols.append(j)
                    vals.append(f[p])
                    rows.append(k)
                    cols.append(p)
                    vals.append(f[j])
    # Map original k indices to 0..top_m-1
    k_to_new = {int(k): i for i, k in enumerate(lags)}
    rows = [k_to_new[r] for r in rows]
    A = csr_matrix((vals, (rows, cols)), shape=(top_m, n))
    c = -np.ones(n)
    b_ub = np.full(top_m, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 5.0})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def optimize_subgrad(f, deadline, lr=0.01, mom=0.9, top_k=3,
                     lp_every=200, verbose=False, tag=""):
    """Main optimization: subgradient + momentum + periodic LP."""
    f = f.copy()
    n = len(f)
    S0 = float(f.sum())
    best = f.copy()
    best_s = score_fast(f)
    v = np.zeros(n)
    lr_cur = lr
    it = 0
    no_imp = 0
    t_start = time.monotonic()
    
    while time.monotonic() < deadline:
        g, conv = subgrad_topk(f, top_k)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        
        # LP direction every lp_every iterations
        if lp_every > 0 and it % lp_every == 0 and it > 0:
            M = float(conv.max())
            glp = reduced_lp_direction(f, M, top_m=min(30, 2*n-1))
            if glp is not None:
                s_glp = float(glp.sum())
                if s_glp > 1e-10:
                    glp = glp * (S0 / s_glp)
                    # Mix subgrad and LP direction
                    g = 0.5 * g / gn + 0.5 * glp / (float(np.linalg.norm(glp)) + 1e-14)
        
        v = mom * v - lr_cur * (g / (gn + 1e-14))
        f_new = np.clip(f + v, 0.0, None)
        sn = float(f_new.sum())
        if sn < 1e-10:
            lr_cur *= 0.5
            no_imp += 1
            if no_imp > 200:
                break
            it += 1
            continue
        f_new *= S0 / sn
        s_new = score_fast(f_new)
        it += 1
        
        if s_new < best_s:
            best = f_new.copy()
            best_s = s_new
            f = f_new
            no_imp = 0
            if verbose and it % 500 == 0:
                el = time.monotonic() - t_start
                print(f"  {tag} it={it} s={best_s:.6f} lr={lr_cur:.5f} t={el:.0f}s")
        else:
            f = f_new  # still move
            no_imp += 1
            if no_imp % 100 == 0:
                lr_cur *= 0.9
            if no_imp > 1000:
                # Reheat
                lr_cur = max(lr_cur, 0.005)
                v *= 0.1
                no_imp = 0
        if lr_cur < 1e-7:
            break
    
    return best, best_s


def pairwise_swap_local(a, max_swaps=2000):
    """Local search: move small mass from high-grad to low-grad positions."""
    a = a.copy()
    n = len(a)
    S = float(a.sum())
    best_s = score_fast(a)
    eps = 0.01 * S / n
    
    for _ in range(max_swaps):
        conv = fft_convolve_auto(a)
        k_star = int(np.argmax(conv))
        g = subgrad_lag(a, k_star)
        # Move from highest gradient to lowest gradient
        i_src = int(np.argmax(g * a))  # position contributing most
        i_dst = int(np.argmin(g))       # position contributing least
        if i_src == i_dst:
            break
        if a[i_src] < eps:
            break
        a_new = a.copy()
        a_new[i_src] -= eps
        a_new[i_dst] += eps
        a_new = np.clip(a_new, 0, None)
        s_new = score_fast(a_new)
        if s_new < best_s:
            a = a_new
            best_s = s_new
        else:
            eps *= 0.5
            if eps < 1e-8:
                break
    
    return a, best_s


def make_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(('uniform', np.ones(n)))
    
    # Linear tilt
    inits.append(('tilt', np.linspace(2.0, 0.5, n)))
    inits.append(('tilt2', np.linspace(0.5, 2.0, n)))
    
    # Exponential
    inits.append(('exp1', np.exp(-1.5 * idx / n)))
    inits.append(('exp2', np.exp(-1.5 * (n - 1 - idx) / n)))
    inits.append(('exp3', np.exp(-3.0 * idx / n)))
    inits.append(('exp4', np.exp(-0.5 * idx / n)))
    
    # Cosine (raised, zero at edges)
    inits.append(('cos', np.sin(np.pi * (idx + 0.5) / n) ** 2))
    
    # Tilted cosine
    t = np.sin(np.pi * (idx + 0.5) / n) ** 2
    mod = 1.0 + 0.3 * (idx / (n - 1) - 0.5)
    inits.append(('cos_tilt', t * mod))
    
    # Gaussian
    c = (n - 1) / 2.0
    for sigma_frac in [0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((idx - c) / (n * sigma_frac)) ** 2)
        inits.append((f'gau_{sigma_frac}', g))
    
    # Asymmetric Gaussian
    for alpha in [0.5, 1.0, 2.0]:
        g = np.exp(-0.5 * ((idx - c) / (n * 0.3)) ** 2) * np.exp(-alpha * (idx - c) / n)
        inits.append((f'asym_gau_{alpha}', g))
    
    # Power of cosine
    for p in [1, 2, 4]:
        t = np.sin(np.pi * (idx + 0.5) / n) ** (2 * p)
        inits.append((f'cos_p{p}', t))
    
    # Random smooth
    r = rng.uniform(0, 1, n)
    # Smooth with a kernel
    kernel = np.exp(-0.5 * (np.arange(-20, 21) / 5.0) ** 2)
    kernel /= kernel.sum()
    r_smooth = np.convolve(r, kernel, 'same')
    inits.append(('rand_smooth', r_smooth))
    
    return inits
