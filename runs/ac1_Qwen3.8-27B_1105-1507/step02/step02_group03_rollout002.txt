"""
Optimized step function search using LP-based game iteration, spectral flattening,
multi-peak subgradient descent, and coordinate descent polishing.
Combines multiple strategies across sequence lengths to minimize 2n*max(conv)/sum^2.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute 2n * max(conv) / sum^2 for non-negative f."""
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (s * s)


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def build_conv_matrix(f):
    """Build sparse-friendly dense matrix A such that A @ b = convolve(f, b, 'full')."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(n):
        # Row k: contributions from f[0..k] * b[k..0]
        A[k, 0:k+1] = f[k:-k-1:-1] if k > 0 else f[k]
    for k in range(n, m):
        # Row k: contributions from f[k-n+1..n-1] * b[n-1..k-n+1]
        i_start = k - n + 1
        A[k, i_start:n] = f[n-1:i_start-1:-1]
    return A


def diverse_init(n, rng, seed_variant=0):

        f += 0.05
    elif r < 0.6:

        for _ in range(rng.integers(2, 6)):

            f = f_new

    if rng is None:

    n = len(f)

    best_score = compute_score(f)

    stall = 0

        M = conv.max()
        
        # Build constraint matrix: A @ g <= M for all k

        except Exception:
            break

                best_score = new_score
                f = f_new
                t = min(1.0, t * 1.2)

            stall = 0

            stall += 1
            t *= 0.7

                # Perturbation
                f_pert = f + 0.05 * rng.normal(0, 1, n)
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    t = 0.2
                    stall = 0
                else:
                    break
    
    return best_f, best_score


def coordinate_descent(f, n_sweeps=10, rng=None):
    """Polishing via coordinate descent with geometric search."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)
    
    factors = [0.5, 0.7, 0.85, 0.95, 1.0/0.95, 1.0/0.85, 1.0/0.7, 1.0/0.5]
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        improved = False
        for i in order:
            original = f[i]
            if original < 1e-8:
                f[i] = 0.001  # Try to activate
                continue
            
            best_val = original
            best_local = compute_score(f)
            
            for fac in factors:
                f[i] = max(0.0, original * fac)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                score = compute_score(f)
                if score < best_local:
                    best_local = score
                    best_val = f[i]
            
            f[i] = best_val
        
        f = normalize(f)
        current_score = compute_score(f)
        if current_score < best_score:
            best_score = current_score
            improved = True
        
        if not improved and sweep > 2:
            break
    
    return f.copy(), best_score


def block_refinement(f, n_blocks=16, rng=None):
    """Refine by optimizing blocks of consecutive elements."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)
    block_size = max(1, n // n_blocks)
    
    for _ in range(5):
        improved = False
        for start in range(0, n - block_size, block_size // 2):
            end = min(start + block_size, n)
            original_block = f[start:end].copy()
            
            # Try scaling the whole block
            for scale in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                f[start:end] = np.clip(original_block * scale, 0, None)
                s = f.sum()
                if s < 1e-10:
                    f[start:end] = original_block
                    continue
                score = compute_score(f)
                if score < best_score:
                    best_score = score
                    improved = True
                else:
                    f[start:end] = original_block
            
            # Try adding/removing a Gaussian bump in the block
            center = (start + end) // 2
            width = max(1, block_size // 3)
            bump = np.exp(-0.5 * ((np.arange(start, end) - center) ** 2) / width ** 2)
            for amp in [0.1, 0.3, -0.1, -0.3]:
                f_try = f.copy()
                f_try[start:end] = np.clip(original_block + amp * bump, 0, None)
                s = f_try.sum()
                if s < 1e-10:
                    continue
                score = compute_score(f_try)
                if score < best_score:
                    best_score = score
                    f = f_try
                    improved = True
        
        if not improved:
            break
    
    return f.copy(), best_score


def optimize_length(n, rng, time_budget, parent=None):
    """Run the full optimization pipeline for a given sequence length."""
    start_time = time.monotonic()
    
    # Initialize
    if parent is not None and len(parent) == n:
        f = parent.copy()
        f = normalize(f)
    else:
        f = diverse_init(n, rng)
    
    best_f = f.copy()
    best_score = compute_score(f)
    print(f"  n={n} init: score={best_score:.6f}")
    
    # Phase 1: Quick subgradient + spectral (fast, ~30s)
    phase1_time = min(30.0, time_budget * 0.15)
    f, s = multi_peak_subgradient(f, n_iter=200, eta0=0.2, rng=rng)
    if s < best_score:
        best_score = s
        best_f = f.copy()
        print(f"  After subgrad: {s:.6f}")
    
    f, s = spectral_flat(f, n_iter=80, rng=rng)
    if s < best_score:
        best_score = s
        best_f = f.copy()
        print(f"  After spectral: {s:.6f}")
    
    elapsed = time.monotonic() - start_time
    remaining = time_budget - elapsed
    if remaining < 10:
        return best_f, best_score
    
    # Phase 2: LP game iteration (main workhorse)
    phase2_time = min(remaining * 0.5, 300.0)
    n_iter_lp = max(20, int(phase2_time / max(0.5, n / 512)))  # Scale iterations
    f, s = lp_game_iteration(f, n_iter=n_iter_lp, t0=0.4, rng=rng,
                            lp_time_limit=max(0.5, min(5.0, phase2_time // max(1, n_iter_lp))))
    if s < best_score:
        best_score = s
        best_f = f.copy()
        print(f"  After LP: {s:.6f}")
    
    # Phase 3: Alternating subgradient + spectral for escape
    elapsed = time.monotonic() - start_time
    remaining = time_budget - elapsed
    if remaining > 20:
        for round_idx in range(3):
            f, s = multi_peak_subgradient(best_f, n_iter=50, eta0=0.15, rng=rng)
            if s < best_score:
                best_score = s
                best_f = f.copy()
            f, s = spectral_flat(f, n_iter=40, rng=rng)
            if s < best_score:
                best_score = s
                best_f = f.copy()
            # Quick LP
            f, s = lp_game_iteration(f, n_iter=20, t0=0.3, rng=rng, lp_time_limit=2.0)
            if s < best_score:
                best_score = s
                best_f = f.copy()
            print(f"  Round {round_idx}: {best_score:.6f}")
            
            elapsed = time.monotonic() - start_time
            if time_budget - elapsed < 20:
                break
    
    # Phase 4: Polishing
    elapsed = time.monotonic() - start_time
    remaining = time_budget - elapsed
    if remaining > 10:
        f, s = coordinate_descent(best_f, n_sweeps=min(15, int(remaining * 0.3)), rng=rng)
        if s < best_score:
            best_score = s
            best_f = f.copy()
            print(f"  After CD: {s:.6f}")
    
    elapsed = time.monotonic() - start_time
    remaining = time_budget - elapsed
    if remaining > 10:
        f, s = block_refinement(best_f, n_blocks=max(8, n // 32), rng=rng)
        if s < best_score:
            best_score = s
            best_f = f.copy()
            print(f"  After block: {s:.6f}")
    
    # Final LP polish
    elapsed = time.monotonic() - start_time
    remaining = time_budget - elapsed
    if remaining > 5:
        f, s = lp_game_iteration(best_f, n_iter=max(5, int(remaining / 3)), t0=0.2, rng=rng,
                                lp_time_limit=max(0.3, min(3.0, remaining // max(1, int(remaining/3)))))
        if s < best_score:
            best_score = s
            best_f = f.copy()
            print(f"  Final LP: {s:.6f}")
    
    return best_f, best_score


def propose_candidate():
    """
    Main search: multi-length LP game optimization with spectral and subgradient methods.
    Starts from parent sequence, explores multiple initializations and lengths.
    """
    start_time = time.monotonic()
    budget = 1480.0  # Leave 20s buffer
    rng = np.random.default_rng(12345)
    
    best_overall = None
    best_score_overall = np.inf
    
    # Try to load parent
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except NameError:
        parent = None
    except Exception:
        parent = None
    
    if parent is not None:
        parent = np.clip(parent, 0, None)
        if parent.sum() > 0.01:
            parent = normalize(parent)
            p_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_overall = parent.copy()
            best_score_overall = p_score
    
    # Sequence lengths to try, with time allocation
    n_parent = len(parent) if parent is not None else 512
    
    # Prioritize parent length, then explore others
    lengths_to_try = []
    if parent is not None:
        lengths_to_try.append(n_parent)
    
    # Add other lengths (sorted by expected time cost)
    for n in [256, 384, 512, 768, 1024]:
        if n not in lengths_to_try:
            lengths_to_try.append(n)
    
    # Sort: parent first, then by proximity to parent, then by size
    lengths_to_try.sort(key=lambda x: (0 if x == n_parent else 1, abs(x - n_parent)))
    
    print(f"Lengths to try: {lengths_to_try}")
    print(f"Budget: {budget}s")
    
    # Allocate time: parent gets 40%, others share 60%
    n_lengths = len(lengths_to_try)
    if n_lengths <= 1:
        allocations = {lengths_to_try[0]: budget * 0.95}
    else:
        remaining_alloc = budget * 0.55 / (n_lengths - 1)
        allocations = {}
        allocations[lengths_to_try[0]] = budget * 0.40
        for n in lengths_to_try[1:]:
            allocations[n] = min(remaining_alloc, budget / max(1, n_lengths))
        
        # Ensure total doesn't exceed budget
        total = sum(allocations.values())
        if total > budget * 0.95:
            scale = budget * 0.95 / total
            for n in allocations:
                allocations[n] *= scale
    
    # Run optimization for each length
    for n in lengths_to_try:
        elapsed = time.monotonic() - start_time
        remaining = budget - elapsed
        if remaining < 30:
            print(f"Time running out, skipping n={n}")
            break
        
        t_alloc = min(allocations.get(n, remaining * 0.3), remaining * 0.8)
        print(f"\n{'='*50}")
        print(f"Starting n={n}, allocation={t_alloc:.0f}s, remaining={remaining:.0f}s")
        
        # If this is the parent length, use parent as init
        p = parent if (parent is not None and len(parent) == n) else None
        f_best, s_best = optimize_length(n, rng, t_alloc, parent=p)
        
        if s_best < best_score_overall:
            best_score_overall = s_best
            best_overall = f_best.copy()
            print(f"  NEW BEST: n={n}, score={s_best:.6f}")
    
    # Final global polish on the best found
    if best_overall is not None:
        elapsed = time.monotonic() - start_time
        remaining = budget - elapsed
        if remaining > 15:
            print(f"\n{'='*50}")
            print(f"Final polish, remaining={remaining:.0f}s")
            n_best = len(best_overall)
            rng2 = np.random.default_rng(99999)
            
            # Additional LP iterations
            f, s = lp_game_iteration(best_overall, n_iter=max(5, int(remaining / 4)),
                                    t0=0.15, rng=rng2, lp_time_limit=1.0)
            if s < best_score_overall:
                best_score_overall = s
                best_overall = f.copy()
                print(f"  Polish LP: {s:.6f}")
            
            remaining = budget - (time.monotonic() - start_time)
            if remaining > 10:
                f, s = coordinate_descent(best_overall, n_sweeps=min(10, int(remaining / 2)), rng=rng2)
                if s < best_score_overall:
                    best_score_overall = s
                    best_overall = f.copy()
                    print(f"  Polish CD: {s:.6f}")
    
    # Verify with evaluate_sequence
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        return final_list
    
    # Fallback
    print("Using fallback")
    return [1.0] * 512
