- Update a ← (1-t)a + tg

This is a best-response dynamics / fictitious play approach. It works because:
- If we're at a Nash equilibrium of the "game" where both players choose a sequence and the cost is max(conv(a,g)), then the equilibrium should be good.
- At equilibrium, max(conv(a,a)) is minimized in some sense.

def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def fast_conv_ab(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a) + len(b) - 1
    s = 1
    while s < n:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fb = np.fft.rfft(b, s)
    return np.fft.irfft(fa * fb, s)[:n]

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # Water-filling
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cv - target_sum))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0)

def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))

def smooth_gradient(a, beta):
    """Compute gradient of smooth objective (assuming sum(a)=1)."""
    n = len(a)
    c = fast_conv(a)  # (2n-1)-vector
    # Softmax weights
    c_shifted = beta * (c - c.max())
    c_shifted = np.clip(c_shifted, -50, 50)
    w = np.exp(c_shifted)
    w = w / w.sum()  # normalized softmax (sum to 1)
    # Gradient: 2 * conv(a, w_reversed) but we need to be careful with indexing
    # ∇_a[i] = 2 * sum_k w[k] * a[k-i] (with a[m]=0 outside [0,n-1])

    # We only need the first n elements (indices 0 to n-1)
    w_rev = w[::-1]
    grad = 2.0 * fast_conv_ab(a, w_rev)[:n]
    return grad, c.max()

def adam_smooth_descent(a, n_betas, steps_per_beta, lr_base=0.01, deadline=None):
    """Adam-based smooth descent with beta annealing."""
    a = a.copy().astype(np.float64)
    a = np.clip(a, 1e-10, None)
    a = a / a.sum()
    n = len(a)
    
    best = a.copy()
    best_score = compute_score(a)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    for beta_idx, beta in enumerate(n_betas):
        if deadline is not None and time.monotonic() > deadline:
            break
        for step in range(steps_per_beta):
            if deadline is not None and time.monotonic() > deadline:
                break
            t += 1
            grad, c_max = smooth_gradient(a, beta)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            
            lr = lr_base / np.sqrt(t + 1)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Project onto simplex
            a_new = project_simplex(a_new, 1.0)
            
            # Check score
            if step % 10 == 0 or step == steps_per_beta - 1:
                score = compute_score(a_new)
                if score < best_score:
                    best = a_new.copy()
                    best_score = score
            a = a_new
    
    return best, best_score

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    # Build the constraint matrix
    # conv(a,g)[k] = sum_j a[j] * g[k-j] for valid j
    # We need this <= M for all k
    # This is a linear constraint on g
    A = np.zeros((m, n))
    for k in range(m):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(m, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None

                if score2 < best_score:

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Fast Adam-based smooth gradient descent with beta annealing
2. LP best-response refinement for precise optimization
3. Multi-start with diverse asymmetric initializations
4. Adaptive time budget allocation
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def fast_conv_ab(a, b):
    """Full linear convolution of a and b via FFT."""
    total_len = len(a) + len(b) - 1
    size = 1
    while size < total_len:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:total_len]


def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum} using water-filling."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cv - target_sum)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def smooth_grad_a(a, beta):
    """
    Compute gradient of smooth objective w.r.t. a.
    Returns (grad_a, c_max) where grad_a = ∇_a [(1/β)log Σ exp(β*c_k)]
    and c = conv(a,a).
    """
    n = len(a)
    c = fast_conv(a)
    c_max = c.max()
    c_shifted = beta * (c - c_max)
    c_shifted = np.clip(c_shifted, -50.0, 50.0)
    w = np.exp(c_shifted)
    w = w / w.sum()
    
    # grad_a[i] = 2 * sum_j a[j] * w[i+j] for i=0,...,n-1
    # = 2 * cross_correlation(a, w)[i]
    # Computed via: conv(a, w_rev)[2n-2-i]
    w_rev = w[::-1]
    conv_full = fast_conv_ab(a, w_rev)
    idx = 2 * n - 2 - np.arange(n)
    grad_a = 2.0 * conv_full[idx]
    
    return grad_a, c_max


def adam_smooth_descent(a, beta_schedule, steps_per_beta, lr_base=0.1, deadline=None):
    """
    Adam-based smooth gradient descent with beta annealing.
    a is normalized to sum to 1.
    """
    a = a.copy().astype(np.float64)
    a = np.clip(a, 1e-10, None)
    a = a / a.sum()
    n = len(a)
    
    best = a.copy()
    best_score = compute_score(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    
    for beta_idx, beta in enumerate(beta_schedule):
        if deadline is not None and time.monotonic() > deadline:
            break
        for step in range(steps_per_beta):
            if deadline is not None and time.monotonic() > deadline:
                break
            t += 1
            
            grad_a, c_max = smooth_grad_a(a, beta)
            
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_a
            v = beta2 * v + (1.0 - beta2) * grad_a * grad_a
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr_base / np.sqrt(1.0 + 0.01 * t)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = project_simplex(a_new, 1.0)
            
            # Check score periodically
            if step % 20 == 0 or step == steps_per_beta - 1:
                score = compute_score(a_new)
                if score < best_score:
                    best_score = score
                    best = a_new.copy()
            
            a = a_new
    
    return best, best_score


def build_conv_matrix(a):
    """Build matrix A where (A @ g) = full convolve(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs')
        if res.success and res.x.sum() > 1e-10:
            return res.x
    except Exception:
        pass
    return None


def lp_iteration(a, max_iters, deadline, base_t=0.5):
    """Run LP best-response iterations with adaptive step size."""
    best_local = a.copy()
    best_score = compute_score(a)
    a = a.copy()
    a = np.clip(a, 0.001, None)
    a = a / a.sum()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()
        
        t = base_t * (1.0 - 0.5 * it / max(max_iters, 1))
        t = max(0.05, min(0.8, t))
        
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            a = a_new
        else:
            t2 = t * 0.3
            a_new2 = (1.0 - t2) * a + t2 * g
            a_new2 = np.clip(a_new2, 0.0, None)
            s2 = a_new2.sum()
            if s2 > 1e-10:
                a_new2 = a_new2 / s2
                score2 = compute_score(a_new2)
                if score2 < best_score:
                    best_score = score2
                    best_local = a_new2.copy()
                a = a_new2
    
    return best_local, best_score
