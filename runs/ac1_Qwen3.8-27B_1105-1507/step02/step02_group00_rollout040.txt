```python
        A[k, j_start:j_end + 1] = a[k - (j_start:j_end + 1)]
    return A


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_alternate(a_init, max_iter=300, verbose=False, time_limit=None):
    """LP alternating with line search."""
    start = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        if time_limit is not None and (time.monotonic() - start) > time_limit:
            break
        
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        g = solve_lp(a, M)
        if g is None:
            g = solve_lp(a, M * 1.001)
            if g is None:
                break
        
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Line search
        best_t = 0.5
        best_obj_t = best_obj
        for t in [0.2, 0.4, 0.5, 0.6, 0.8]:
            a_trial = (1 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0)
            s_t = np.sum(a_trial)
            if s_t < 1e-10:
                continue
            a_trial = a_trial / s_t
            obj_t = compute_obj(a_trial)
            if obj_t < best_obj_t:
                best_obj_t = obj_t
                best_t = t
        
        a_new = (1 - best_t) * a + best_t * g
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a = g
        else:
            a = a_new / s_new
        
        obj_new = compute_obj(a)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()
            if verbose:
                print(f"    LP it={it}: obj={best_obj:.8f}")
    
    return best_a, best_obj
```

```python
"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating with line search, smooth-max gradient polish,
and coordinate descent. Explores multiple sequence lengths with time-aware budgeting.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_conv_matrix(a):
    """Build matrix A where (A@g)_k = convolve(a, g)_k for all k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - np.arange(j_start, j_end + 1)]
    return A


def solve_lp(a, M, time_limit_lp=5.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': time_limit_lp})
    if res.status == 0:
        return res.x
    return None

    """LP alternating optimization with adaptive line search."""
    start = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    prev_obj = best_obj

    for it in range(max_iter):
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 5.0:
            break

        conv = np.convolve(a, a)
        M = float(np.max(conv))

        g = solve_lp(a, M, time_limit_lp=10.0)
        if g is None:
            g = solve_lp(a, M * 1.01, time_limit_lp=10.0)
            if g is None:
                break

        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive line search
        best_t = 0.5
        best_obj_t = best_obj
        for t in [0.15, 0.3, 0.5, 0.7, 0.85]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s_t = np.sum(a_trial)
            if s_t < 1e-10:
                continue
            a_trial = a_trial / s_t
            obj_t = compute_obj(a_trial)
            if obj_t < best_obj_t:
                best_obj_t = obj_t
                best_t = t

        a_new = (1.0 - best_t) * a + best_t * g
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a = g
        else:
            a = a_new / s_new

        obj_new = compute_obj(a)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()

        # Convergence check
        if it > 20 and abs(prev_obj - best_obj) < 1e-9:
            if verbose:
                print(f"    LP converged at it={it}, obj={best_obj:.8f}")
            break
        prev_obj = best_obj

        if verbose and it % 50 == 0:
            print(f"    LP it={it}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj
```

"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses improved LP alternating with line search, smooth-max gradient polishing
with momentum, and coordinate descent. Explores multiple sequence lengths
with adaptive time allocation based on initial quick scan.

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_conv_matrix(a):
    """Build matrix A where (A@g) = convolve(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - np.arange(j_start, j_end + 1)]
    return A


def solve_lp(a, M, tl=5.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': tl})
    if res.status == 0:
        return res.x
    return None
