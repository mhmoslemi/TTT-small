"""
Optimization of step function heights to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses subgradient descent with adaptive step size, LP-based direction finding,
smoothing, and random perturbation for n=1024.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

N = 1024

def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))

def normalize(f):
    """Normalize to sum=1."""
    s = np.sum(f)
    if s < 1e-15:
        return f / max(np.max(f), 1e-15)
    return f / s

def get_active_lags(f, threshold=0.99):
    """Get lags where conv is close to the max."""
    conv = np.convolve(f, f)
    M = np.max(conv)
    if M <= 0:
        return [len(conv)//2]
    # Lags within threshold of max
    active = np.where(conv >= threshold * M)[0]
    # Also always include the actual max
    max_idx = np.argmax(conv)
    if max_idx not in active:
        active = np.append(active, max_idx)
    return list(active)

def subgradient_obj(f, active_lags):
    """Compute subgradient of the objective (assuming sum=1 normalization).
    Objective: 2n * M (since sum=1, score = 2n*M)
    Subgradient: 2n * (1/|K|) * sum_{k in K} grad_conv[k]
    """
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    
    grad = np.zeros(n)
    K = len(active_lags)
    for k in active_lags:
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            idx = np.arange(i_min, i_max + 1)
            grad[idx] += 2.0 * f[k - idx]
    
    grad *= (2.0 * n) / K
    return grad, M

def subgradient_step(f, step_size, momentum_vec, momentum_decay=0.9):
    """One step of subgradient descent with momentum."""
    n = len(f)
    active = get_active_lags(f)
    grad, M = subgradient_obj(f, active)
    
    # Add the gradient contribution from the normalization constraint
    # We're minimizing 2n*M subject to sum=1, a>=0
    # The constraint gradient is just 1 (constant)
    
    momentum_vec = momentum_decay * momentum_vec - step_size * grad
    f_new = f + momentum_vec
    f_new = np.clip(f_new, 0.0, None)
    f_new = normalize(f_new)
    
    return f_new, momentum_vec

def smoothing_step(f, sigma=1.0, alpha=0.01):
    """Apply a small amount of Gaussian smoothing."""
    n = len(f)
    idx = np.arange(n)
    center = (n-1)/2
    kernel = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
    kernel /= kernel.sum()
    f_smooth = np.convolve(f, kernel, mode='same')
    f_new = (1 - alpha) * f + alpha * f_smooth
    f_new = np.clip(f_new, 0, None)
    return normalize(f_new)

def build_conv_matrix(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception as e:
        pass
    return None

def lp_optimize(f, deadline, max_iters=2000, verbose=False):

    n = len(f)
    it = 0
    no_improve = 0

        conv = np.convolve(f, f)
        M = float(np.max(conv))
        
        g = lp_direction(f, M)
        if g is None or np.sum(g) < 1e-15:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        g = normalize(g)
        it += 1
        
        # Line search

        best_cand = None
        for alpha in [0.1, 0.05, 0.2, 0.01, 0.5, 0.005]:
            cand = (1 - alpha) * f + alpha * g

            cand = normalize(cand)
            score = compute_score(cand)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_alpha = alpha
                best_cand = cand
        
        if best_cand is not None:
            f = best_cand
            best_f = f.copy()
            no_improve = 0

    return best_f, best_score

def subgrad_optimize(f, deadline, lr_init=0.01, momentum_decay=0.9,

    f = normalize(f.copy())

    n = len(f)
    momentum = np.zeros(n)
    lr = lr_init
    it = 0
    no_improve = 0

        it += 1
        
        if np.isfinite(score) and score < best_score:
            f = f_new
            best_f = f.copy()
            best_score = score
            no_improve = 0
            # Slowly decay learning rate

            if verbose and it % 5000 == 0:

            lr *= (0.99 if no_improve < 50 else 0.95)
            momentum *= 0.5

"""
Minimizes 2n*max(conv(a,a))/sum(a)^2 for n=1024 using:
- Asymmetric initializations (key insight: optimal is non-symmetric)
- LP-based direction finding with dual-weighted subgradient hybrid
- Adaptive step sizes, momentum, and random perturbation

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

N = 1024

def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 1e-10:
        return float('inf')
    return float(2.0 * n * M / (S * S))

def normalize(f):
    s = np.sum(f)
    if s < 1e-15:
        return f / max(np.max(f), 1e-15)
    return f / s

def get_active_lags(f, fraction=0.98):
    conv = np.convolve(f, f)
    M = np.max(conv)
    if M <= 0:
        return [len(conv) // 2]
    active = np.where(conv >= fraction * M)[0]
    return list(active)

def build_conv_matrix(f):
    n = len(f)
    nnz = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        nnz += (j_max - j_min + 1)
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    vals = np.empty(nnz, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def solve_lp(f, M):
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None

def weighted_subgradient(f, active_lags, weights=None):
    n = len(f)
    K = len(active_lags)
    if weights is None:
        weights = np.ones(K) / K
    else:
        wsum = np.sum(weights)
        if wsum < 1e-15:
            weights = np.ones(K) / K
        else:
            weights = weights / wsum
    grad = np.zeros(n)
    for k, w in zip(active_lags, weights):
        if w <= 0:
            continue
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            idx = np.arange(i_min, i_max + 1)
            grad[idx] += w * 2.0 * f[k - idx]
    return grad

    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)

    lr = lr_init
    no_improve = 0
    it = 0
    conv_cache = np.convolve(f, f)
    M_cache = float(np.max(conv_cache))
    active_weights = None

    while time.monotonic() < deadline and it < max_steps:

                else:
                    conv_f = np.convolve(f, f)
                    M_f = float(np.max(conv_f))

def lp_descent(f, deadline, max_iters=5000, verbose_prefix=""):

    while time.monotonic() < deadline and it < max_iters:
        conv = np.convolve(f, f)
        M = float(np.max(conv))

        t_start = time.monotonic()
        g = solve_lp(f, M)
        lp_time = time.monotonic() - t_start

        if g is None or np.sum(g) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        g = normalize(g)
        it += 1

        best_alpha = 0
        best_cand = None
        for alpha in step_sizes:
            cand = (1.0 - alpha) * f + alpha * g
            cand = np.clip(cand, 0.0, None)
            cs = np.sum(cand)
            if cs < 1e-10:
                continue
            cand = cand / cs
            score = compute_score(cand)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_alpha = alpha
                best_cand = cand

        if best_cand is not None:

        else:

def perturb_search(f, deadline, n_trials=100, scale=0.03, verbose_prefix=""):

    best_f = f.copy()
    best_score = compute_score(f)
    rng = np.random.default_rng(7)
    n = len(f)

    for i in range(n_trials):
        if time.monotonic() > deadline:
            break
        noise = rng.normal(0, scale, size=n)
        f_p = f + noise
        f_p = np.clip(f_p, 0, None)
        sp = np.sum(f_p)
        if sp < 1e-10:
            continue
        f_p = f_p / sp
        score = compute_score(f_p)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_f = f_p.copy()
            f = f_p
            scale = max(0.005, scale * 0.9)
            if verbose_prefix and i % 10 == 0:
                print(f"    {verbose_prefix}perturb {i}: {best_score:.6f}")

    return best_f, best_score

def smooth_blend(f, alpha=0.05, sigma=3.0):
    n = len(f)
    idx = np.arange(n, dtype=np.float64)
    kernel = np.exp(-0.5 * ((idx - n / 2) / sigma) ** 2)
    kernel /= kernel.sum()
    f_s = np.convolve(f, kernel, mode='same')
    f_new = (1 - alpha) * f + alpha * f_s
    f_new = np.clip(f_new, 0, None)
    return normalize(f_new)

def make_tilted_bell(n, tilt=0.1, sigma=0.3, sharp=1.0):
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0 * (1.0 + tilt)
    x = (idx - (n - 1) / 2.0) / (n * sigma)
    f = np.exp(-sharp * x * x) * np.exp(-tilt * x * 2.0)
    return normalize(f)

def make_exponentially_tilted(n, tilt_rate=0.3, base_sigma=0.35):
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (idx - center) / (n * base_sigma)
    f = np.exp(-x * x) * np.exp(-tilt_rate * x)
    return normalize(f)

def make_bimodal_tilted(n, offset=0.03):
    idx = np.arange(n, dtype=np.float64)
    c1 = (n - 1) / 2.0 * (1.0 - offset)
    c2 = (n - 1) / 2.0 * (1.0 + offset)
    s = n * 0.15
    f = 0.6 * np.exp(-0.5 * ((idx - c1) / s) ** 2) + 0.4 * np.exp(-0.5 * ((idx - c2) / s) ** 2)
    return normalize(f)

def make_paren年多(n, p=1.5):
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = np.abs(idx - center) / (n / 2.0)
    f = np.power(np.maximum(1.0 - x, 0.0), p)
    if np.max(f) < 1e-10:
        f = np.ones(n)
    return normalize(f)

def propose_candidate(budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            if parent.size != N:
                x_old = np.linspace(0, 1, parent.size)
                x_new = np.linspace(0, 1, N)
                parent = np.interp(x_new, x_old, parent)
            score = compute_score(parent)
            if np.isfinite(score):
                best_score = score
                best_sequence = parent.copy()
                print(f"Parent (n={N}): score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"No parent: {e}")

    # Build initializations
    inits = []
    if parent is not None:
        inits.append(('parent', normalize(parent.copy())))
        # Slightly perturbed parent
        rng0 = np.random.default_rng(0)
        for pert_scale in [0.02, 0.05, 0.1]:
            p = np.clip(parent + rng0.normal(0, pert_scale, N), 0, None)
            inits.append((f'parent_p{pert_scale}', normalize(p)))

    # Asymmetric initializations (key: optimal is non-symmetric!)
    inits.append(('tilt_b1', make_tilted_bell(N, tilt=0.05, sigma=0.35, sharp=1.0)))
    inits.append(('tilt_b2', make_tilted_bell(N, tilt=0.1, sigma=0.3, sharp=1.5)))
    inits.append(('tilt_b3', make_tilted_bell(N, tilt=0.15, sigma=0.35, sharp=1.0)))
    inits.append(('tilt_b4', make_tilted_bell(N, tilt=-0.08, sigma=0.32, sharp=1.2)))
    inits.append(('tilt_b5', make_tilted_bell(N, tilt=0.03, sigma=0.4, sharp=0.7)))
    inits.append(('tilt_b6', make_tilted_bell(N, tilt=0.2, sigma=0.25, sharp=2.0)))
    inits.append(('exp_t1', make_exponentially_tilted(N, tilt_rate=0.2, base_sigma=0.35)))
    inits.append(('exp_t2', make_exponentially_tilted(N, tilt_rate=0.5, base_sigma=0.3)))
    inits.append(('exp_t3', make_exponentially_tilted(N, tilt_rate=0.8, base_sigma=0.35)))
    inits.append(('bimodal', make_bimodal_tilted(N, offset=0.03)))
    inits.append(('bimodal2', make_bimodal_tilted(N, offset=0.05)))
    inits.append(('pow_15', make_paren年多(N, p=1.5)))
    inits.append(('pow_2', make_paren年多(N, p=2.0)))
    inits.append(('pow_05', make_paren年多(N, p=0.5)))

    # Evaluate and sort
    print("\n=== Initializations ===")
    scored_inits = []
    for label, f_init in inits:
        s = compute_score(f_init)
        scored_inits.append((s, label, f_init))
        if s < best_score:
            best_score = s
            best_sequence = f_init.copy()
    scored_inits.sort(key=lambda x: x[0])
    for s, lbl, _ in scored_inits[:8]:
        print(f"  {.lbl}: {s:.6f}")

    # Phase 1: Quick optimization of top initializations (300s)
    print(f"\n=== Phase 1: Quick Opt (best={best_score:.6f}) ===")
    p1_end = t0 + 300.0
    candidates = []

    for s_init, label, f_init in scored_inits[:6]:
        if time.monotonic() > p1_end - 30:
            break
        print(f"  [{label}] init={s_init:.6f}", end="")
        # Quick subgradient
        t_quick = min(p1_end, time.monotonic() + 40.0)
        f_opt, s_opt = subgrad_descent(f_init, t_quick, lr_init=0.015,
                                        momentum_decay=0.9, verbose_prefix="")
        # Quick LP
        if time.monotonic() < p1_end - 10:
            t_lp = min(p1_end, time.monotonic() + 20.0)
            f_opt, s_opt2 = lp_descent(f_opt, t_lp, max_iters=30)
            if s_opt2 < s_opt:
                s_opt = s_opt2
        print(f" -> {s_opt:.6f}")
        candidates.append((s_opt, f_opt, label))
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()

    candidates.sort(key=lambda x: x[0])
    print(f"\nTop candidates: {[f'{l}:{s:.4f}' for s, _, l in candidates[:4]]}")

    # Phase 2: Deep optimization on top candidates (800s)
    print(f"\n=== Phase 2: Deep Opt (best={best_score:.6f}) ===")
    p2_end = t0 + 1100.0
    seen_labels = set()
    top_cands = []
    for s, f_c, lbl in candidates:
        if lbl not in seen_labels:
            top_cands.append((s, f_c, lbl))
            seen_labels.add(lbl)
        if len(top_cands) >= 3:
            break

    for s_start, f_c, lbl in top_cands:
        if time.monotonic() > p2_end - 60:
            break
        print(f"\n  Deep: [{lbl}] start={s_start:.6f}")

        # Long subgradient with LP weights
        t_sg = time.monotonic() + min(250, p2_end - time.monotonic() - 100)
        f_opt, s_opt = subgrad_descent(f_c, t_sg, lr_init=0.008,
                                        momentum_decay=0.92,
                                        use_lp_weights=True, lp_interval=300,
                                        verbose_prefix=f"[{lbl}]")
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()
            print(f"    subgrad improved to {best_score:.6f}")

        # LP phase
        if time.monotonic() < p2_end - 60:
            t_lp = min(p2_end, time.monotonic() + 200.0)
            f_opt2, s_opt2 = lp_descent(f_opt, t_lp, max_iters=500,
                                         verbose_prefix=f"[{lbl}-LP]")
            if s_opt2 < best_score:
                best_score = s_opt2
                best_sequence = f_opt2.copy()
                print(f"    LP improved to {best_score:.6f}")
            f_opt = f_opt2

        # Smooth + subgradient
        if time.monotonic() < p2_end - 30:
            f_smooth = smooth_blend(f_opt, alpha=0.08, sigma=2.0)
            t_sm = min(p2_end, time.monotonic() + 80.0)
            f_opt3, s_opt3 = subgrad_descent(f_smooth, t_sm, lr_init=0.004,
                                              momentum_decay=0.93,
                                              verbose_prefix=f"[{lbl}-sm]")
            if s_opt3 < best_score:
                best_score = s_opt3
                best_sequence = f_opt3.copy()

    print(f"\nBest after Phase 2: {best_score:.6f}")

    # Phase 3: Final polish (remaining time)
    print(f"\n=== Phase 3: Final Polish (best={best_score:.6f}) ===")
    if best_sequence is not None:
        # Fine subgradient
        t_fine = deadline - 120
        f_pol, s_pol = subgrad_descent(best_sequence, t_fine, lr_init=0.002,
                                        momentum_decay=0.95,
                                        use_lp_weights=True, lp_interval=200,
                                        verbose_prefix="[final-sg]")
        if s_pol < best_score:
            best_score = s_pol
            best_sequence = f_pol.copy()

        # LP polish
        if time.monotonic() < deadline - 60:
            f_pol2, s_pol2 = lp_descent(best_sequence, deadline - 30,
                                         max_iters=200, verbose_prefix="[final-LP]")
            if s_pol2 < best_score:
                best_score = s_pol2
                best_sequence = f_pol2.copy()

        # Perturbation
        f_pol3, s_pol3 = perturb_search(best_sequence, deadline,
                                         n_trials=50, scale=0.02)
        if s_pol3 < best_score:
            best_score = s_pol3
            best_sequence = f_pol3.copy()

    # Fallback
    if best_sequence is None:
        best_sequence = make_tilted_bell(N, tilt=0.05, sigma=0.35, sharp=1.0)

    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    final_score = compute_score(best_sequence)
    if not np.isfinite(final_score):
        best_sequence = np.ones(N) / N
        final_score = compute_score(best_sequence)

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, internal_score={final_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
