```python
def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Algorithm from Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.find_last((u * np.arange(1, n + 1)) > (cssv - 1))
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

        return v

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_convolve(a, b):
    """Full linear convolution using FFT."""
    n_out = len(a) + len(b) - 1
    size = 1 << (n_out - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n_out]


def fast_autocorr(f):
    """Compute conv(f, f) using FFT."""
    return fast_convolve(f, f)


def fast_cross_corr(p, f):
    """
    Compute correlate(p, f, 'valid').
    Result[i] = sum_{j=0}^{len(f)-1} p[i+j] * f[j]
    Requires len(p) >= len(f).
    """
    n_p = len(p)
    n_f = len(f)
    n_full = n_p + n_f - 1
    size = 1 << (n_full - 1).bit_length()
    f_rev = f[::-1]
    fp = np.fft.rfft(p, size)
    ff = np.fft.rfft(f_rev, size)
    conv = np.fft.irfft(fp * ff, size)[:n_full]
    return conv[n_f - 1: n_p]

def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = conv(f, g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def perturbation_restart(f, rng, strength=0.05):

    new_f = f * np.clip(noise, 0.05, 3.0)

    S = float(np.sum(new_f))
    if S < 1e-10:

    return new_f / S
