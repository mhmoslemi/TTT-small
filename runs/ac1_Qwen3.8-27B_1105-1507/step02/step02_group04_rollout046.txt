"""
Nesterov-momentum smoothed gradient + peak-chasing + LP alternating optimization
for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous: Nesterov momentum, backtracking line search,
peak-chasing greedy heuristic, frequency-domain guided initialization,
wider n range (50-600), more diverse initializations, adaptive time allocation.

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute autocorrelation conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    """Normalize so sum = 1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return a
    return a / s


def compute_smoothed_grad(a, conv, tau):
    """Compute gradient of LSE-smoothed max(conv) w.r.t. a.
    Uses the identity: d(conv[k])/d(a[j]) = 2*a[k-j] if 0<=k-j<n.
    So grad[j] = 2 * sum_m weights[j+m] * a[m]."""
    n = len(a)
    max_conv = float(np.max(conv))
    shifted = (conv - max_conv) / tau
    shifted = np.clip(shifted, -50.0, 500.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights = weights / w_sum
    # grad[j] = 2 * sum_{m=0}^{n-1} weights[j+m] * a[m]
    # This is cross-correlation of weights with a, first n values
    corr = np.correlate(weights, a, mode='full')
    grad = 2.0 * corr[:n].copy()
    return grad

    n = len(a)
    a = a.copy().astype(np.float64)
    a = normalize(a)
    best_a = a.copy()
    conv0 = compute_conv(a)
    best_max = float(np.max(conv0))
    
    v = np.zeros(n)
    step = step_init
    stall = 0
    patience = 400
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        frac = min(1.0, it / max(1, max_iter * 0.8))
        tau = tau_start + (tau_end - tau_start) * frac ** 0.6
        
        conv = compute_conv(a)
        current_max = float(np.max(conv))
        
        if current_max < best_max - 1e-12:
            best_max = current_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                step *= 0.6
                if step < 1e-11:
                    break

                a = a + noise * step
                a = normalize(a)
                v = np.zeros(n)
                stall = 0
                continue

        grad_c = grad - np.mean(grad)

        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)

            break
        a_new = a_new / s

        obj_new = float(np.max(compute_conv(a_new)))

            v = v * 0.5

            if it > 200:
                step *= 0.9997
    
    return best_a, best_max


def peak_chase(a, max_iter=5000, deadline=None, verbose=False):
    """Greedy peak-reduction: reduce coordinates that contribute most to the peak lag."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = normalize(a)
    best_a = a.copy()
    conv = compute_conv(a)
    best_max = float(np.max(conv))
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        conv = compute_conv(a)
        peak_idx = int(np.argmax(conv))
        current_max = float(conv[peak_idx])
        
        if current_max < best_max - 1e-12:
            best_max = current_max
            best_a = a.copy()
        
        # peak_idx corresponds to lag k = peak_idx - (n-1) in correlation terms
        # conv[peak_idx] = sum_i a[i] * a[peak_idx - i] for valid i
        # Contribution of a[j] to conv[peak_idx]: a[peak_idx - j] if 0 <= peak_idx-j < n
        # Gradient contribution: 2 * a[peak_idx - j]
        
        lag = peak_idx  # in conv indexing
        # contribution[j] = 2 * a[lag - j] if 0 <= lag-j < n
        contrib = np.zeros(n)
        j_min = max(0, lag - n + 1)
        j_max = min(n - 1, lag)
        if j_min <= j_max:
            contrib[j_min:j_max+1] = 2.0 * a[lag - j_max:lag - j_min + 1][::-1]
        
        # Reduce high-contribution coordinates, increase low-contribution ones
        # Use a softmax-like weighting
        max_contrib = float(np.max(contrib))
        if max_contrib < 1e-15:
            break
        
        # Compute "pressure" - coordinates with high contrib should decrease
        pressure = contrib / max_contrib
        # Small step
        eta = 0.02
        a_new = a * (1.0 - eta * pressure)
        # Also add mass to low-pressure coordinates
        low_mask = pressure < 0.3
        if np.any(low_mask):
            a_new[low_mask] += eta * 0.5 * a[low_mask].mean()
        
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-15:
            break
        a_new = a_new / s
        
        obj_new = float(np.max(compute_conv(a_new)))
        if obj_new < current_max:
            a = a_new
        else:
            # Try smaller step
            eta = 0.005
            a_new = a * (1.0 - eta * pressure)
            if np.any(low_mask):
                a_new[low_mask] += eta * 0.5 * a[low_mask].mean()
            a_new = np.maximum(a_new, 0.0)
            s = np.sum(a_new)
            if s < 1e-15:
                break
            a_new = a_new / s
            obj_new = float(np.max(compute_conv(a_new)))
            if obj_new < current_max:
                a = a_new
            else:
                # Stuck, break
                break
    
    return best_a, best_max


def build_A_matrix(a):
    """Build Toeplitz matrix where A[k,j] gives contribution of a[j] to conv[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            # A[k, j] = 2 * a[k-j] for j in valid range (from gradient formula)
            # But for the LP constraint conv[k] = sum_j A[k,j] * g[j] <= M
            # conv[k] = sum_j a[k-j] * g[j] (from the convolution definition)
            # Wait: conv = np.convolve(a, g), so conv[k] = sum_i a[i]*g[k-i]
            # For fixed k: conv[k] = sum_{i: valid} a[i] * g[k-i]
            # Let j = k-i, so i = k-j: conv[k] = sum_{j: valid} a[k-j] * g[j]
            # So A[k, j] = a[k-j] if 0 <= k-j < n
            A[k, j_min:j_max+1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def solve_lp(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    if n > 400:
        return None, None
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None


def lp_alternate(a, max_iter=30, t=0.25, deadline=None, verbose=False):
    """LP-based alternating optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = normalize(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    stall = 0
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        g, sum_g = solve_lp(a, M, time_limit=15)
        if g is None:
            break
        
        if sum_g <= 1.0 + 1e-3:
            stall += 1
            if stall > 10:
                break
            continue
        stall = 0
        
        g = g / sum_g
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-15:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose:
                print(f"    LP it={it}: obj={best_obj:.6f}")
        a = a_new
    
    return best_a, best_obj


def gen_init_cosine(n, rng):
    """Cosine window with random phase."""
    t = np.arange(n) / n
    phase = rng.uniform(0, 2 * np.pi)
    a = 0.5 * (1 + np.cos(2 * np.pi * t + phase))
    return a


def gen_init_gaussian(n, rng):
    """Gaussian window with random center and width."""
    center = rng.uniform(0.2, 0.8) * n
    width = rng.uniform(0.1, 0.5) * n
    t = np.arange(n)
    a = np.exp(-((t - center) ** 2) / (2 * width ** 2))
    return a


def gen_init_binary_smoothed(n, rng):
    """Random binary sequence, then smoothed."""
    b = rng.integers(0, 2, n).astype(np.float64)
    # Smooth with a small kernel
    kernel = np.array([0.25, 0.5, 0.25])
    a = np.convolve(b, kernel, mode='same')
    return a


def gen_init_barker_like(n, rng):
    """Barker-like: start with random signs, map to non-negative, smooth."""
    signs = rng.choice([-1, 1], n).astype(np.float64)
    # Map to non-negative: (signs + 1) / 2 gives {0, 1}
    a = (signs + 1) * 0.5
    # Add some structure
    t = np.arange(n) / n
    a = a * (0.5 + 0.5 * np.cos(2 * np.pi * rng.uniform(0.5, 3.0) * t))
    return a


def gen_init_random(n, rng):

    return rng.random(n)


def gen_init_frequency_flat(n, rng):
    """Initialize to have flat power spectrum (ideal autospectral flatness)."""
    # Start with random phase, uniform magnitude
    mags = np.ones(n)
    mags[0] = np.sqrt(n)  # DC component
    phases = rng.uniform(0, 2 * np.pi, n)
    A = mags * np.exp(1j * phases)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:

    return a / s

    # Phase 2: Peak chasing
    alloc2 = min(max_time * 0.15, 10.0)
    a2, _ = peak_chase(a1, max_iter=3000, deadline=start_time + alloc1 + alloc2)
    obj2 = compute_obj(a2)
    
    # Phase 3: Second Nesterov pass with different params
    best_a = a2 if obj2 < obj1 else a1
    best_obj = min(obj1, obj2)
    
    alloc3 = min(max_time * 0.3, 25.0)
    a3, _ = nesterov_descent(best_a, max_iter=20000, tau_start=15.0, tau_end=100.0,
                             step_init=0.03, deadline=start_time + alloc1 + alloc2 + alloc3)
    obj3 = compute_obj(a3)
    if obj3 < best_obj:
        best_a = a3
        best_obj = obj3
    
    # Phase 4: LP polish (only for moderate n)
    if n <= 300:
        alloc4 = min(max_time * 0.25, 20.0)
        a4, obj4 = lp_alternate(best_a, max_iter=20, t=0.2,
                                deadline=start_time + alloc1 + alloc2 + alloc3 + alloc4)
        if obj4 < best_obj:
            best_a = a4
            best_obj = obj4
    
    # Phase 5: Final quick polish
    remaining = deadline - time.monotonic()
    if remaining > 3:
        a5, _ = nesterov_descent(best_a, max_iter=int(remaining * 2),
                                 tau_start=30.0, tau_end=120.0, step_init=0.02,
                                 deadline=deadline)
        obj5 = compute_obj(a5)
        if obj5 < best_obj:
            best_a = a5
            best_obj = obj5
    
    if verbose:
        print(f"    n={n} init_type={init_type}: final_obj={best_obj:.6f}")
    
    return best_a, best_obj


def try_parent_sequence():
    """Try to use the parent sequence as a starting point."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def polish_best(a, deadline, verbose=True):
    """Intensive polishing of the best sequence found."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = normalize(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    if verbose:
        print(f"\nIntensive polish: n={n}, initial obj={best_obj:.6f}")
    
    round_num = 0
    while time.monotonic() < deadline - 5:
        round_num += 1
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        
        # Nesterov with high tau (peak-focused)
        alloc = min(remaining * 0.3, 30.0)
        a_cur, _ = nesterov_descent(a, max_iter=30000, tau_start=20.0, tau_end=150.0,
                                    step_init=0.02, deadline=time.monotonic() + alloc)
        obj = compute_obj(a_cur)
        if obj < best_obj:
            best_obj = obj
            best_a = a_cur.copy()
            a = a_cur
            if verbose:
                print(f"  Polish R{round_num} nest: obj={best_obj:.6f}")
        
        # Peak chase
        remaining = deadline - time.monotonic()
        if remaining > 5:
            alloc = min(remaining * 0.2, 10.0)
            a_pc, _ = peak_chase(a, max_iter=3000, deadline=time.monotonic() + alloc)
            obj_pc = compute_obj(a_pc)
            if obj_pc < best_obj:
                best_obj = obj_pc
                best_a = a_pc.copy()
                a = a_pc
                if verbose:
                    print(f"  Polish R{round_num} peak: obj={best_obj:.6f}")
        
        # LP if feasible
        remaining = deadline - time.monotonic()
        if remaining > 15 and n <= 300:
            a_lp, obj_lp = lp_alternate(a, max_iter=15, t=0.2,
                                        deadline=time.monotonic() + min(remaining*0.3, 20.0))
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
                a = a_lp
                if verbose:
                    print(f"  Polish R{round_num} LP: obj={best_obj:.6f}")
        
        # Random perturbation to escape local minima
        remaining = deadline - time.monotonic()
        if remaining > 10:
            rng = np.random.default_rng(round_num * 12345 + n)
            for pert_scale in [0.03, 0.05, 0.08]:
                a_pert = a * np.exp(rng.normal(0, pert_scale, n))
                a_pert = normalize(a_pert)
                obj_p = compute_obj(a_pert)
                if obj_p < best_obj:
                    best_obj = obj_p
                    best_a = a_pert.copy()
                    a = a_pert
                    if verbose:
                        print(f"  Polish R{round_num} perturb({pert_scale}): obj={best_obj:.6f}")
                    break
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Combined Nesterov + peak-chasing + LP optimization.
    Explores sequence lengths 50-600 with diverse initializations.
    Returns the best sequence found within the time budget.
    """
    global start_time
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    
    best_seq = None
    best_score = np.inf
    
    # Try parent sequence
    parent, parent_obj = try_parent_sequence()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
    
    # Sequence lengths to explore
    n_values = [50, 60, 80, 100, 120, 150, 180, 200, 250, 300, 350, 400, 500, 600]
    
    # If parent exists, include its neighbors
    if parent is not None:
        n_parent = len(parent)
        extra = [max(30, n_parent - 30), n_parent, min(700, n_parent + 30)]
        n_values = sorted(set(n_values + extra))
    
    print(f"N values to explore: {n_values}")
    
    # Time allocation: more time for middle-range n values
    total_n = len(n_values)
    # We'll do 2-3 inits per n value
    n_inits_per_n = 3
    
    for n_idx, n in enumerate(n_values):
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break
        
        # Allocate time based on n (moderate n gets more time)
        if n <= 80:
            per_init = 15.0
        elif n <= 150:
            per_init = 25.0
        elif n <= 250:
            per_init = 35.0
        elif n <= 350:
            per_init = 30.0
        else:
            per_init = 20.0
        
        max_time_for_n = per_init * n_inits_per_n
        max_time_for_n = min(max_time_for_n, remaining * 0.3)
        max_time_for_n = max(10.0, max_time_for_n)
        
        if verbose_check(n):
            print(f"\nN={n}: {remaining:.0f}s remaining, {max_time_for_n:.0f}s allocated, {n_inits_per_n} inits")
        
        for init_idx in range(n_inits_per_n):
            if time.monotonic() > deadline - 30:
                break
            init_type = (init_idx + n) % 6
            a_opt, obj = run_single(n, init_type, seed + n * 7919 + init_idx * 104729,
                                    max_time=max_time_for_n / n_inits_per_n,
                                    deadline=deadline - 30,
                                    verbose=False)
            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  UPDATE: n={n}, init={init_type}, obj={best_score:.6f}")
    
    # Intensive polishing on the best
    if best_seq is not None and time.monotonic() < deadline - 10:
        best_seq, best_score = polish_best(best_seq, deadline, verbose=True)
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(100) / 100.0
        best_score = compute_obj(best_seq)
    
    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    
    # Validate
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation != best_score and validation < np.inf:
        print(f"  Validation: computed={best_score:.6f}, validated={validation:.6f}")
        best_score = validation
    
    return result


def verbose_check(n):
    """Return True if we should print for this n."""
    return n in [100, 150, 200, 250, 300, 400]
