Actually, I realize there's a subtlety. The LP finds g that minimizes max_k (a*g)_k subject to sum(g) = 1, g >= 0. This g is the "best" sequence to convolve with a to get a flat result. But when we update a <- (1-tau)*a + tau*g, the new self-convolution (a_new * a_new) is NOT the same as (a * g). So the LP solution is a direction, not the exact solution.

This is why the LP best-response is an ITERATIVE method - it converges to a fixed point where a = g (i.e., a is its own best response).

At the fixed point, a satisfies: max_k (a*a)_k = min_g max_k (a*g)_k (among unit-sum g). This is a "Nash equilibrium" condition for the game where the "environment" picks k and the "player" picks g.

import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def fast_conv_diff(a, g):
    """Full linear convolution of a and g via FFT."""
    n = len(a)
    m = len(g)
    s = 1
    while s < n + m - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fg = np.fft.rfft(g, s)
    return np.fft.irfft(fa * fg, s)[:n + m - 1]

def build_conv_matrix(a):
    """Build matrix A where A @ g = conv(a, g)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for i in range(n):
        A[i:i+n, i] = a
    return A

def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))

def lp_best_response(a, M):
    """Solve LP: min t s.t. conv(a,g) <= t, sum(g)=1, g>=0. Returns (g, t)."""
    n = len(a)
    A = build_conv_matrix(a)
    # Variables: g[0..n-1], t
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # A_ub: A @ g - t <= 0, i.e., [A | -1] @ [g; t] <= 0
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # A_eq: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        g = res.x[:n]
        t = res.x[n]
        return g, t
    return None, None

def lp_iteration(a, max_iters, deadline, base_tau=0.5):
    """Run LP best-response iterations with backtracking step size."""
    best_local = a.copy()
    best_score = compute_score(a)
    a = a.copy()
    a = np.clip(a, 0.001, None)
    a = a / a.sum()
    
    stagnation_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g, t_lp = lp_best_response(a, M)
        if g is None:
            break
        
        g = np.clip(g, 0, None)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Backtracking step size
        improved = False
        for tau in [base_tau, 0.3, 0.1, 0.05, 0.02, 0.01]:
            a_new = (1.0 - tau) * a + tau * g
            a_new = np.clip(a_new, 0, None)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            score = compute_score(a_new)
            if score < best_score:
                best_score = score
                best_local = a_new.copy()
                a = a_new
                improved = True
                stagnation_count = 0
                break
            elif score < compute_score(a):
                # Even if not better than best, if better than current, accept
                a = a_new
                break
        if not improved:
            stagnation_count += 1
            if stagnation_count > 20:
                break
    
    return best_local, best_score
