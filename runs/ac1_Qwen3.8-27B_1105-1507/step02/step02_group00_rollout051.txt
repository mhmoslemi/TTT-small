def softmax_subgradient(a_init, n_steps=10000, tau_start=0.1, tau_end=0.01, verbose=False):
    """Subgradient method with softmax approximation of max."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    # Nesterov momentum
    y = a.copy()
    beta = 0.0
    
    for step in range(n_steps):
        # Anneal tau
        progress = step / n_steps
        tau = tau_start * (tau_end / tau_start) ** progress
        
        conv = np.convolve(y, y)
        
        # Softmax weights
        weights = np.exp(conv / tau - np.max(conv) / tau)  # shifted for numerical stability
        weights /= np.sum(weights)
        
        # Gradient: grad_j = (1/tau) * sum_k weights_k * d conv_k / da_j
        # = (1/tau) * sum_k weights_k * [a_{k-j} + a_{j-k}] (with bounds, excluding diagonal)
        # This is a weighted cross-correlation
        
        # Compute using direct method (efficient for n <= 400)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k]
            if w < 1e-10:
                continue
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                other = k - j
                if other == j:
                    grad[j] += w * a[j]
                else:
                    grad[j] += w * a[other]
        grad /= tau
        
        # Nesterov update
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        t_k = (1.0 + beta) / (1.0 + beta_new)
        
        step_size = 0.3 / np.sqrt(step + 1.0)
        
        a_new = y - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        y = a_new + (beta_new - 1.0) / (1.0 + beta_new) * (a_new - y)
        y = np.maximum(y, 0.0)
        s = np.sum(y)
        if s < 1e-10:
            y = a_new.copy()
        else:
            y = y / s
        
        beta = beta_new
        a = a_new
        
        if step % 500 == 0:
            obj_new = compute_obj(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            if verbose:
                print(f"    Softmax it={step}: tau={tau:.4f}, obj={best_obj:.6f}")
    
    obj_final = compute_obj(a)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = a.copy()
    
    return best_a, best_obj
