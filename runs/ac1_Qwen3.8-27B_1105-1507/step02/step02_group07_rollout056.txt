"""
Multi-strategy optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Combines Sionkin LP iterations, soft-max gradient with Adam, coordinate descent,
spectral initialization, and perturbation restarts. Focuses on n=512 with
aggressive local search, plus exploration of nearby lengths.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score_fast(f):
    """Fast score computation without safety checks."""
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return np.inf
    C = np.convolve(f, f)
    return 2.0 * n * float(np.max(C)) / (S * S)


def normalize(f):
    S = float(np.sum(f))
    if S < 1e-15:
        return f
    return f / S


def build_conv_sparse(f):
    """Build sparse matrix A where (A@g)[k] = (f*g)[k] for linear convolution."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_sionkin(f, M):
    """Solve: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def sionkin_optimize(f, deadline, max_iters=200, step_init=0.1, verbose=False):
    """Sionkin LP-based iteration with adaptive step size."""
    f = normalize(f.astype(np.float64).copy())
    best_f = f.copy()
    best_score = compute_score_fast(f)
    step = step_init
    no_improve = 0
    A_cache = None

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        C = np.convolve(f, f)
        M = float(np.max(C))
        g = lp_sionkin(f, M)
        if g is None:
            no_improve += 1
            if no_improve > 25:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            no_improve += 1
            continue
        g = g / sum_g

        improved = False
        candidates = [step, step * 0.5, step * 1.5, step * 0.1, 0.2, 0.4, 0.6]
        for t in candidates:
            if t <= 0 or t >= 1:
                continue
            new_f = normalize(np.maximum((1.0 - t) * f + t * g, 0.0))
            s = compute_score_fast(new_f)
            if s < best_score:
                f = new_f
                best_f = f.copy()
                best_score = s
                improved = True
                if verbose and it % 20 == 0:
                    print(f"      Sionkin it={it}: score={best_score:.6f} (t={t:.3f})")
                break

        if improved:
            no_improve = 0
            step = min(step * 1.1, 0.5)
        else:
            no_improve += 1
            step *= 0.6
            if step < 0.005 or no_improve > 35:
                break

    return best_f, best_score


def softmax_gradient(f, beta):
    """Compute gradient of soft-max approximation of max(conv(f,f))."""
    C = np.convolve(f, f)
    max_C = float(np.max(C))
    p = np.exp(beta * (C - max_C))
    p /= p.sum()
    grad = 2.0 * np.correlate(p, f, mode='valid')
    return grad


def softmax_adam_optimize(f, deadline, beta_start=20.0, beta_end=600.0,
                          lr_start=0.05, max_iters=8000, verbose=False):
    """Soft-max gradient descent with Adam."""
    f = normalize(f.astype(np.float64).copy())
    n = len(f)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    best_f = f.copy()
    best_score = compute_score_fast(f)
    step = 0
    no_improve = 0
    lr = lr_start

    while time.monotonic() < deadline and step < max_iters:
        frac = min(1.0, step / (max_iters * 0.75))
        beta = beta_start * (beta_end / beta_start) ** frac

        grad = softmax_gradient(f, beta)

        step += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad ** 2)
        m_hat = m / (1.0 - b1 ** step)
        v_hat = v / (1.0 - b2 ** step)
        update = m_hat / (np.sqrt(v_hat) + eps)

        new_f = np.maximum(f - lr * update, 0.0)
        new_f = normalize(new_f)
        s = compute_score_fast(new_f)

        if s < best_score:
            f = new_f
            best_f = f.copy()
            best_score = s
            no_improve = 0
            if verbose and step % 500 == 0:
                print(f"      SoftMax it={step}: score={best_score:.6f} (beta={beta:.1f}, lr={lr:.6f})")
        else:
            no_improve += 1
            f = new_f  # still take the step for Adam momentum
            if no_improve > 200:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-7:
                    break

    return best_f, best_score


def coordinate_descent(f, deadline, max_sweeps=50):
    """Try changing each element slightly to reduce the score."""
    f = normalize(f.astype(np.float64).copy())
    n = len(f)
    best_f = f.copy()
    best_score = compute_score_fast(f)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            orig = f[j]
            for delta in [0.01, -0.01, 0.05, -0.05, 0.1, -0.1]:
                new_val = orig + delta * best_f.sum() / n
                if new_val < 0:
                    continue
                f[j] = new_val
                nf = normalize(f)
                s = compute_score_fast(nf)
                if s < best_score:
                    best_score = s
                    best_f = nf.copy()
                    improved = True
                    break
                else:
                    f[j] = orig
        if not improved:
            break

    return best_f, best_score


def tukey_window(n, r=0.3):
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def gauss_window(n, sigma_frac=0.4):
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n / 2.0
    return np.exp(-0.5 * ((idx - center) / sigma) ** 2)


def raised_cosine(n, order=2):
    idx = np.arange(n, dtype=np.float64)
    x = 2.0 * idx / (n - 1) - 1.0
    return (1.0 - x ** order) ** order


def random_smooth_init(n, seed):
    rng = np.random.default_rng(seed)
    x = rng.standard_normal(n)
    kernel_size = max(3, n // 30)
    kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (kernel_size // 2 + 1)) ** 2)
    x = np.convolve(x, kernel, mode='same')
    x = np.abs(x) + 0.05
    return x


def perturb(f, rng, strength=0.05):
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.1, 3.0)
    new_f = np.clip(new_f, 0.0, 1000.0)
    return normalize(new_f)


def full_pipeline(f_init, deadline, lp_frac=0.35, sm_frac=0.50, cd_frac=0.15, verbose=False):
    """Run the full optimization pipeline on an initialization."""
    f_init = normalize(f_init.astype(np.float64).copy())
    best_f = f_init.copy()
    best_score = compute_score_fast(f_init)

    t0 = time.monotonic()
    total = deadline - t0

    # Phase 1: Sionkin LP
    d1 = t0 + total * lp_frac
    if time.monotonic() < d1:
        f1, s1 = sionkin_optimize(f_init, d1, max_iters=80, step_init=0.12, verbose=verbose)
        if s1 < best_score:
            best_f, best_score = f1, s1
            if verbose:
                print(f"        After LP: {best_score:.6f}")

    # Phase 2: Soft-max Adam
    d2 = t0 + total * (lp_frac + sm_frac)
    if time.monotonic() < d2:
        f2, s2 = softmax_adam_optimize(best_f, d2, beta_start=20.0, beta_end=500.0,
                                        lr_start=0.04, max_iters=6000, verbose=verbose)
        if s2 < best_score:
            best_f, best_score = f2, s2
            if verbose:
                print(f"        After SoftMax: {best_score:.6f}")

    # Phase 3: Short Sionkin + coordinate descent
    if time.monotonic() < deadline - 2:
        d3 = deadline - 2
        remaining = d3 - time.monotonic()
        if remaining > 10:
            f3, s3 = sionkin_optimize(best_f, time.monotonic() + remaining * 0.4,
                                       max_iters=30, step_init=0.05)
            if s3 < best_score:
                best_f, best_score = f3, s3
        if time.monotonic() < d3 - 5:
            f4, s4 = coordinate_descent(best_f, d3, max_sweeps=20)
            if s4 < best_score:
                best_f, best_score = f4, s4

    return best_f, best_score

    elapsed = time.monotonic() - t0
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_f)}, score={best_score:.6f}, time={elapsed:.1f}s")
    print(f"{'='*50}")

    return [float(x) for x in best_f]
