    """Fast score computation (assumes f is non-negative, sum > 0)."""
    n = len(f)
    C = np.convolve(f, f)
    S = np.sum(f)
    return 2.0 * n * np.max(C) / (S * S)


def build_conv_rows(f, lags):
    """Build rows of the conv matrix for specific lags.
    Returns (K, n) matrix where row k corresponds to constraint (f*g)(lags[k]) <= M.
    """
    n = len(f)
    K = len(lags)
    A = np.zeros((K, n))
    for ki, k in enumerate(lags):
        # (f*g)(k) = sum_j f[k-j] * g[j] for valid j
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[ki, j] = f[k - j]
    return A


def solve_reduced_lp(f, M, active_lags):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)(k) <= M for k in active_lags."""
    n = len(f)
    K = len(active_lags)
    if K == 0:
        return np.ones(n) / n
    
    A = build_conv_rows(f, active_lags)
    c = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_constraint_generation(f, M, max_k=50, max_rounds=5):
    """LP with constraint generation: start with few constraints, add violated ones."""
    n = len(f)
    C = np.convolve(f, f)
    
    # Start with top-5 lags
    top_k = min(5, 2 * n - 1)
    active = np.argsort(C)[::-1][:top_k]
    
    for round in range(max_rounds):
        g = solve_reduced_lp(f, M, active)
        if g is None:
            return None
        
        # Check which constraints are violated
        new_C = np.convolve(f, g)
        violated = np.where(new_C > M * (1 + 1e-10))[0]
        
        if len(violated) == 0:
            return g
        
        # Add violated constraints (up to max_k total)
        new_active = np.union1d(active, violated)
        # Keep only the most violated
        if len(new_active) > max_k:
            viol_amount = new_C[new_active] - M
            order = np.argsort(viol_amount)[::-1]
            new_active = new_active[order[:max_k]]
        active = new_active
    
    return None


def lp_optimize(f, deadline, max_iters=200, step_init=0.1, verbose=""):
    """LP-based optimization with constraint generation."""
    f = f.copy().astype(np.float64)
    S = np.sum(f)
    f = f / S
    best_f = f.copy()
    best_score = compute_score_fast(f)
    step = step_init
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        C = np.convolve(f, f)
        M = np.max(C)
        
        g = lp_constraint_generation(f, M, max_k=30, max_rounds=3)
        it += 1
        if g is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        sum_g = np.sum(g)
        if sum_g < 1e-15:
            no_improve += 1
            continue
        
        g = g / sum_g  # normalize to sum=1
        
        # Line search
        best_t = step
        best_s = best_score
        for t in [step, step*0.5, step*1.5, step*0.25, step*2.0]:
            new_f = (1-t)*f + t*g
            new_f = np.clip(new_f, 0, None)
            ns = np.sum(new_f)
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            s = compute_score_fast(new_f)
            if s < best_s:
                best_s = s
                best_t = t
        
        if best_s < best_score - 1e-12:
            f = (1-best_t)*f + best_t*g
            f = np.clip(f, 0, None)
            f = f / np.sum(f)
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            if it % 20 == 0:
                print(f"  {verbose}LP[{it}]: {best_score:.6f}")
        else:
            no_improve += 1
            step *= 0.7
            if step < 0.003 or no_improve > 30:
                break
    
    return best_f, best_score


def softmax_gradient(f, beta):
    """Compute gradient of soft-max objective."""
    C = np.convolve(f, f)
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros(len(f)), max_C
    p = exp_C / total
    grad = 2.0 * np.correlate(p, f, mode='valid')
    soft_val = max_C + np.log(total) / beta
    return grad, soft_val


def softmax_adam_optimize(f, deadline, beta_start=10.0, beta_end=500.0,
                          lr_start=0.03, max_iters=5000, verbose=""):
    """Soft-max gradient descent with Adam."""
    f = f.copy().astype(np.float64)
    f = f / np.sum(f)
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score_fast(f)
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    lr = lr_start
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        frac = min(1.0, it / (0.8 * max_iters))
        beta = beta_start * (beta_end / beta_start) ** frac
        
        grad, _ = softmax_gradient(f, beta)
        it += 1
        
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*(grad**2)
        m_hat = m / (1 - b1**it)
        v_hat = v / (1 - b2**it)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        new_f = f - lr * update
        new_f = np.clip(new_f, 0, None)
        ns = np.sum(new_f)
        if ns < 1e-10:
            new_f = f.copy()
        new_f = new_f / ns
        
        s = compute_score_fast(new_f)
        if s < best_score - 1e-12:
            f = new_f
            best_f = f.copy()
            best_score = s
            no_improve = 0
            if it % 500 == 0:
                print(f"  {verbose}SM[{it}]: {best_score:.6f} (β={beta:.0f})")
        else:
            no_improve += 1
            if no_improve > 300:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-7:
                    break
    
    return best_f, best_score


def equalization_optimize(f, deadline, max_iters=300, verbose=""):
    """
    Peak equalization: iteratively identify the peak lag and push energy
    away from it using a QP-like step.
    """
    f = f.copy().astype(np.float64)
    f = f / np.sum(f)
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score_fast(f)
    
    it = 0
    no_improve = 0
    step = 0.05
    
    while time.monotonic() < deadline and it < max_iters:
        C = np.convolve(f, f)
        M = C.max()
        
        # Get the p distribution (soft-max over lags)
        beta = max(50.0, 200.0 * it / max_iters)
        exp_C = np.exp(beta * (C - M))
        p = exp_C / exp_C.sum()
        
        # Gradient
        grad = 2.0 * np.correlate(p, f, mode='valid')
        
        # Mirror descent step (multiplicative weights)
        # f_new[j] ∝ f[j] * exp(-step * grad[j])
        # This naturally maintains non-negativity!
        log_step = step * 0.5  # scale factor
        multiplier = np.exp(-log_step * grad)
        new_f = f * multiplier
        new_f = np.clip(new_f, 0, None)
        ns = np.sum(new_f)
        if ns < 1e-10:
            new_f = f.copy()
        new_f = new_f / ns
        
        it += 1
        s = compute_score_fast(new_f)
        if s < best_score - 1e-12:
            f = new_f
            best_f = f.copy()
            best_score = s
            no_improve = 0
            if it % 50 == 0:
                print(f"  {verbose}EQ[{it}]: {best_score:.6f} (β={beta:.0f})")
        else:
            no_improve += 1
            step *= 0.9
            if no_improve > 50 or step < 0.001:
                break
    
    return best_f, best_score


def generate_inits(n):
    """Generate diverse initializations."""
    inits = []
    
    # Raised cosine
    x = np.arange(n)
    center = (n-1)/2
    for width in [0.5, 0.8, 1.0, 1.5, 2.0]:
        w = np.exp(-((x-center)/(n*width*0.5))**2)
        inits.append(w)
    
    # Super-Gaussian
    for alpha in [1.5, 2.0, 2.5, 3.0, 4.0]:
        w = np.exp(-np.abs((x-center)/(n*0.4))**(2*alpha))
        inits.append(w)
    
    # Raised cosine (Hann-like)
    w = 0.5*(1+np.cos(2*np.pi*x/n))
    inits.append(w)
    
    # Tukey windows
    for r in [0.3, 0.5, 0.7]:
        w = np.ones(n)
        m = int(r*n/4)
        if m > 0:
            for i in range(m):
                w[i] = 0.5*(1+np.cos(np.pi*(2*i/(2*m)-1)))
                w[n-1-i] = w[i]
        inits.append(w)
    
    # Flat with noise
    for seed in range(3):
        rng = np.random.default_rng(seed)
        w = 1.0 + 0.3*rng.standard_normal(n)
        w = np.abs(w)
        inits.append(w)
    
    # "Two-bump" initialization
    w = np.zeros(n)
    w[:n//2] = np.exp(-((x[:n//2] - n//4)/(n*0.1))**2)
    w[n//2:] = np.exp(-((x[n//2:] - 3*n//4)/(n*0.1))**2)
    inits.append(w)
    
    # Triangular (should give score ~2)
    w = 1.0 - np.abs(x - center) / center
    inits.append(w)
    
    return inits


def perturb(f, rng, strength=0.05):
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.1, 3.0)
    new_f = np.clip(new_f, 0, None)
    S = np.sum(new_f)
    if S < 1e-10:
        return f.copy()
    return new_f / S


def interp_to_length(f, n):
    """Interpolate sequence to new length."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, n)
    return np.interp(x_new, x_old, f)
