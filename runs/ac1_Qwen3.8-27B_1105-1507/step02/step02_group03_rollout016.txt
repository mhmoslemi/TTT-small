For the score to decrease, we need:
2*n*(M(1-t^2) + t^2 * M_g) / (1 + t*(sum(g)-1))^2 < 2*n*M
=> (M(1-t^2) + t^2 * M_g) < M * (1 + t*(sum(g)-1))^2
=> M - t^2*M + t^2*M_g < M*(1 + 2t*(sum(g)-1) + t^2*(sum(g)-1)^2)
=> -t^2*M + t^2*M_g < 2tM*(sum(g)-1) + t^2*M*(sum(g)-1)^2
=> t^2*(M_g - M) < 2tM*(sum(g)-1) + t^2*M*(sum(g)-1)^2
=> t*(M_g - M) < 2M*(sum(g)-1) + t*M*(sum(g)-1)^2 (dividing by t > 0)
=> t*(M_g - M - M*(sum(g)-1)^2) < 2M*(sum(g)-1)

If M_g < M + M*(sum(g)-1)^2, then the left coefficient is negative, and the inequality holds for all t > 0. Specifically:

Since sum(g) >= 1 (as g=f is feasible and gives sum=1, and the max is >= 1), if M_g <= M, then we're guaranteed improvement.

"""

4. Multiple sequence lengths and diverse starts

import numpy as np
from scipy.optimize import linprog
import time


def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def score_of(f):
    """Compute 2*n*max(conv(f,f)) for normalized f (sum=1)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A such that A @ g = np.convolve(f, g)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def spectral_init(n, n_iter=2000, rng=None):
    """
    Iterative flat-spectrum algorithm:
    Project onto flat |FFT| magnitude + non-negative + length-n support.
    """
    if rng is None:
        rng = np.random.default_rng(42)
    
    # Start with a slightly perturbed uniform
    f = np.ones(n) / n
    f += 0.01 * rng.standard_normal(n)
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    
    N = 1 << (2 * n - 1).bit_length()  # Next power of 2 >= 2n-1
    
    best_f = f.copy()
    best_score = score_of(f)
    stall = 0
    
    for it in range(n_iter):
        F = np.fft.rfft(f, N)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Target: flat magnitude (except DC)
        # Mean |F|^2 = N * sum(f^2) / N = sum(f^2) (by Parseval for length-N DFT)
        # Actually: sum_k |F[k]|^2 = N * sum_j |f_padded[j]|^2 = N * sum(f^2)
        # So mean |F[k]|^2 = sum(f^2)
        # DC: |F[0]| = sum(f) = 1
        # Target for non-DC: |F[k]| = sqrt((N * sum(f^2) - 1) / (N/2 - 1)) for k=1,...,N/2
        energy = np.sum(f ** 2)
        n_bins = N // 2  # number of rfft bins
        target_mag = np.sqrt(max(0, N * energy - 1) / max(1, n_bins - 1))
        
        mag_new = np.full(n_bins, target_mag)
        mag_new[0] = 1.0  # DC fixed
        
        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, N)[:n]
        
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        
        sc = score_of(f_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_f = f_new.copy()
            stall = 0
        else:
            stall += 1
        
        f = f_new
        
        if stall > 500:
            break
        
        # Occasionally perturb phase to escape local minima
        if it % 200 == 199:
            phase_pert = phase + rng.normal(0, 0.1, n_bins)
            F_pert = mag_new * np.exp(1j * phase_pert)
            f_pert = np.fft.irfft(F_pert, N)[:n]
            f_pert = np.clip(f_pert, 0, None)
            s_pert = f_pert.sum()
            if s_pert > 1e-10:
                f_pert /= s_pert
                if score_of(f_pert) < best_score:
                    best_score = score_of(f_pert)
                    best_f = f_pert.copy()
                    f = f_pert
                    stall = 0
    
    return best_f, best_score


def lp_refine(f, time_budget, rng=None, verbose=True):
    """
    Successive LP refinement: repeatedly solve LP to find best direction,
    then take optimal step.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(f)
    f = normalize(f.copy().astype(np.float64))
    best_f = f.copy()
    best_score = score_of(f)
    start_time = time.monotonic()
    
    # Pre-build conv matrix structure (will update when f changes)
    A = build_conv_matrix(f)
    
    iter_count = 0
    while time.monotonic() - start_time < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Solve LP: maximize sum(g) s.t. A@g <= M, g >= 0
        c_obj = -np.ones(n)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M),
                         bounds=[(0, None)] * n, method='highs',
                         options={'presolve': True})
        except Exception:
            break
        
        if not res.success or res.x is None:
            break
        
        g = np.clip(res.x, 0, None)
        g_sum = g.sum()
        if g_sum < 1.0 + 1e-9:
            # LP found g = f (or close), no improvement possible with this M
            # Try to reduce M: use subgradient step
            S = np.where(np.isclose(conv, M, atol=1e-12))[0]
            subgrad = A[S].mean(axis=0)
            v = f - 0.01 * subgrad
            # Project onto simplex
            tau = np.sort(v)[::-1]
            w = np.arange(1, n+1)
            rho = np.where(tau - (w-1)/n > 0)[0][-1] if n > 1 else 0
            threshold = (v[:rho+1].sum() - 1) / (rho + 1)
            f_new = np.clip(v - threshold, 0, None)
            f_new = normalize(f_new)
            sc = score_of(f_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_f = f_new.copy()
                f = f_new
                A = build_conv_matrix(f)
            else:
                break
            iter_count += 1
            continue
        
        # Find optimal step by trying several values
        best_t = 0
        best_new_score = best_score
        for t in [0.5, 0.3, 0.2, 0.1, 0.7, 0.8, 0.4]:
            f_new = (1 - t) * f + t * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            sc = score_of(f_new)
            if sc < best_new_score:
                best_new_score = sc
                best_t = t
        
        if best_t > 0:
            f_new = (1 - best_t) * f + best_t * g
            f_new = np.clip(f_new, 0, None)
            f_new = normalize(f_new)
            best_score = best_new_score
            best_f = f_new.copy()
            f = f_new
            A = build_conv_matrix(f)
        else:
            break
        
        iter_count += 1
        if verbose and iter_count % 10 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    LP iter {iter_count}: score={best_score:.6f}, time={elapsed:.1f}s")
    
    if verbose:
        print(f"  LP done: {iter_count} iters, score={best_score:.6f}")
    
    return best_f, best_score


def subgradient_polish(f, time_budget, rng=None, verbose=True):
    """Projected subgradient method for fine polishing."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(f)
    f = normalize(f.copy().astype(np.float64))
    best_f = f.copy()
    best_score = score_of(f)
    start_time = time.monotonic()
    
    A = build_conv_matrix(f)
    eta = 0.1
    iter_count = 0
    stall = 0
    avg_f = f.copy()
    
    while time.monotonic() - start_time < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        S = np.where(np.isclose(conv, M, rtol=1e-10, atol=1e-14))[0]
        
        # Subgradient: average of rows of A corresponding to argmax
        subgrad = A[S].mean(axis=0)
        
        # Decreasing step size
        eta = max(0.001, 0.1 / (1 + 0.01 * iter_count))
        
        v = f - eta * subgrad
        # Project onto simplex {x: sum(x)=1, x>=0}
        # Use water-filling
        v_sorted = np.sort(v)[::-1]
        w = np.arange(1, n + 1)
        cssv = np.cumsum(v_sorted)
        rho = np.where(v_sorted - (w - 1) / n > 0)[0]
        if len(rho) == 0:
            break
        rho = rho[-1]
        threshold = (v_sorted[:rho+1].sum() - 1) / (rho + 1)
        f_new = np.clip(v - threshold, 0, None)
        f_new = normalize(f_new)
        
        avg_f += f_new
        sc = score_of(f_new)
        
        if sc < best_score - 1e-14:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
            A = build_conv_matrix(f)
            stall = 0
        else:
            # Use average iterate (Nesterov averaging)
            avg_f_norm = normalize(avg_f / (iter_count + 1))
            sc_avg = score_of(avg_f_norm)
            if sc_avg < best_score - 1e-14:
                best_score = sc_avg
                best_f = avg_f_norm.copy()
            f = f_new  # Keep current iterate for subgradient
            A = build_conv_matrix(f)
            stall += 1
        
        iter_count += 1
        if stall > 1000:
            # Perturb to escape
            f_pert = f * (1 + 0.05 * rng.standard_normal(n))
            f_pert = np.clip(f_pert, 0, None)
            f_pert = normalize(f_pert)
            if score_of(f_pert) < best_score:
                f = f_pert
                best_f = f.copy()
                best_score = score_of(f)
                A = build_conv_matrix(f)
                stall = 0
            else:
                break
        
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    SG iter {iter_count}: score={best_score:.6f}, time={elapsed:.1f}s")
    
    if verbose:
        print(f"  SG done: {iter_count} iters, score={best_score:.6f}")
    
    return best_f, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    r = rng.random()
    if r < 0.25:
        # Ramp
        f = np.linspace(0.1, 1.0, n)
    elif r < 0.5:
        # Anti-ramp (non-symmetric)
        f = np.linspace(1.0, 0.1, n)
    elif r < 0.75:
        # Product of two bumps at different locations
        c1 = n * rng.uniform(0.2, 0.5)
        c2 = n * rng.uniform(0.5, 0.8)
        w = n * 0.15
        f = np.exp(-0.5*(np.arange(n)-c1)**2/w**2) + np.exp(-0.5*(np.arange(n)-c2)**2/w**2)
    else:
        # Random smooth
        f = 1.0 + 0.3 * rng.standard_normal(n)
        # Smooth
        kernel = np.ones(max(1, n//50)) / max(1, n//50)
        f = np.convolve(f, kernel, 'same')
    
    f = np.clip(f, 0.01, None)
    f = normalize(f)
    # Break symmetry
    f *= (1 + 0.05 * rng.standard_normal(n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f
