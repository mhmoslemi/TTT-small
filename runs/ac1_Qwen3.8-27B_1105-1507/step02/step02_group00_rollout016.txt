def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):

    n = len(a)

    for k in range(2 * n - 1):
        # g_j contributes to lag k when j + something = k
        # (a*g)_k = sum_j a_j * g_{k-j}

def binary_init(n, rng):
    """Generate a binary-like initialization with good autocorrelation properties."""

        # Random binary with about n/2 ones

        a = np.zeros(n)
        ones = rng.choice(n, size=n_ones, replace=False)
        a[ones] = 1.0
    elif choice == 1:
        # M-sequence like: use a primitive polynomial structure
        # Simplified: create a pattern based on bit-reversal
        period = 1 << (int(np.log2(n)) )
        pattern = np.zeros(period)
        for i in range(period):
            # Count trailing zeros (a simple structure)
            pattern[i] = 1.0 if (i & -i) != i else 0.0  # Not quite m-sequence
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 2:
        # Fourier-based: choose random phases, take real part
        freqs = rng.choice(range(1, n), size=min(n//2, n-1), replace=False)
        a = np.ones(n)  # DC component
        for f in freqs:
            a += np.cos(2 * np.pi * f * np.arange(n) / n + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0)
    else:
        # Weighted random: spike train with random spacing
        a = np.zeros(n)
        n_spikes = n // 3
        positions = np.sort(rng.choice(n, size=n_spikes, replace=False))
        a[positions] = 1.0
        # Add small background
        a += 0.1 * rng.random(n)
    
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

def conv_max(a):
    """Compute max of full convolution using FFT for speed."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    power = A * np.conj(A)
    result = np.fft.irfft(power, fft_size)[:2 * n - 1]
    return float(np.max(result))

    """Compute full convolution using FFT."""
    n = len(a)

    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            return res.x, res
        return None, None
    except Exception:
        return None, None


def solve_lp_dual(a, M):
    """Solve LP and return dual variables if available."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            dual = res.ineqlin.marginals if hasattr(res.ineqlin, 'marginals') else None
            return res.x, dual
        return None, None
    except Exception:
        return None, None


def anti_palindromic_init(n, rng):
    """Create initialization that is anti-correlated with its reverse."""
    choice = rng.integers(0, 5)
    if choice == 0:
        # Left-heavy: mass concentrated on left side with exponential decay
        decay = rng.uniform(1.0, 3.0)
        a = np.exp(-decay * np.arange(n) / n)
        # Add small right side
        a[n // 2:] += 0.3 * rng.random(n // 2)
    elif choice == 1:
        # Ramp: linearly increasing
        a = np.linspace(0.1, 1.0, n)
        noise = rng.normal(0, 0.1, n)
        a = np.maximum(a + noise, 0)
    elif choice == 2:
        # Two separated blobs (not symmetric)
        center1 = n * rng.uniform(0.1, 0.3)
        center2 = n * rng.uniform(0.6, 0.8)
        sigma1 = n * rng.uniform(0.05, 0.15)
        sigma2 = n * rng.uniform(0.05, 0.15)
        a = np.exp(-((np.arange(n) - center1) ** 2) / (2 * sigma1 ** 2))
        a += 0.7 * np.exp(-((np.arange(n) - center2) ** 2) / (2 * sigma2 ** 2))
    elif choice == 3:
        # Fourier-based: choose to cancel middle peak
        # Use odd harmonics to create anti-symmetric pattern
        a = np.ones(n) * 0.5
        n_terms = max(3, n // 10)
        for k in range(1, n_terms + 1):
            freq = rng.choice([2 * m + 1 for m in range(1, max(2, n // 4))])
            phase = rng.uniform(0, 2 * np.pi)
            a += 0.3 / k * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        a = np.maximum(a, 0)
    else:
        # Random with left-right asymmetry
        a_left = rng.power(2.0, n // 2)  # Heavier tail on left
        a_right = rng.power(3.0, n - n // 2) * 0.5
        a = np.concatenate([a_left, a_right])
        a = a[rng.permutation(n)]  # Shuffle to avoid trivial structure
        # Re-apply asymmetry after shuffle
        a[:n // 3] *= rng.uniform(1.0, 2.0)
        a[2 * n // 3:] *= rng.uniform(0.3, 0.7)
    
    s = np.sum(a)
    if s < 1e-10:
        a = np.linspace(0.1, 1.0, n)
        s = np.sum(a)
    return a / s


def lp_alternate(a_init, max_iter=200, m_tightening=0.0, verbose=False):
    """
    LP alternating optimization with adaptive M tightening.
    m_tightening: fraction of M to reduce each iteration (encourages flatter solutions).
    """
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    current_obj = best_obj
    
    for it in range(max_iter):
        conv_a = conv_full(a)
        M = float(np.max(conv_a))
        
        # Adaptive M tightening: try slightly smaller M to encourage spread
        if m_tightening > 0 and it > 10:
            M_eff = M * (1.0 - m_tightening * (1.0 - it / max_iter))
            g, _ = solve_lp_g(a, M_eff)
            if g is None or np.sum(g) < 1e-10:
                M_eff = M
                g, _ = solve_lp_g(a, M_eff)
        else:
            M_eff = M
            g, _ = solve_lp_g(a, M_eff)
        
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Try multiple blend factors, pick best
        best_t_obj = current_obj
        best_t_a = a.copy()
        
        for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_t_obj:
                best_t_obj = obj_try
                best_t_a = a_try.copy()
        
        if best_t_obj < best_obj:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
        
        a = best_t_a
        current_obj = compute_obj(a)
        
        if verbose and (it % 25 == 0):
            print(f"    LP it={it}: best={best_obj:.6f}, curr={current_obj:.6f}, M={M:.6f}")
    
    return best_a, best_obj


def mirror_descent(a_init, max_iter=2000, verbose=False):
    """
    Mirror descent (multiplicative weights) for the convex problem:
    min max_k (a*a)_k s.t. sum(a)=1, a>=0.
    Uses exponential tilting to maintain non-negativity.
    """
    n = len(a)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(max_iter):
        # Compute subgradient: gradient of max active constraint
        h = conv_full(a)
        max_h = np.max(h)
        
        # Active set with tolerance
        tol = max_h * 0.05
        active = h >= max_h - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            idx = np.argmax(h)
            active = np.zeros(2 * n - 1, dtype=bool)
            active[idx] = True
            n_active = 1
        
        # Compute subgradient
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                j_range = np.arange(j_start, j_end + 1)
                grad[j_start:j_end + 1] += 2.0 * a[k - j_range]
        grad /= n_active
        
        # Mirror descent step (multiplicative)
        lr = 0.1 / (1.0 + 0.05 * it)
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
        
        if verbose and (it % 500 == 0):
            print(f"    MD it={it}: best={best_obj:.6f}")
    
    return best_a, best_obj


def subgradient_polish(a, n_steps=3000, verbose=False):
    """
    Subgradient method with kinetic step-size rule.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    # Best-so-far objective for step size
    best_so_far = best_obj
    
    for step in range(n_steps):
        h = conv_full(a)
        max_h = np.max(h)
        
        # Active set
        tol = max_h * (0.005 + 0.02 * step / n_steps)
        active = h >= max_h - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            idx = np.argmax(h)
            active = np.zeros(2 * n - 1, dtype=bool)
            active[idx] = True
            n_active = 1
        
        # Subgradient
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                j_range = np.arange(j_start, j_end + 1)
                grad[j_start:j_end + 1] += 2.0 * a[k - j_range]
        grad /= n_active
        
        # Normalized subgradient for projection onto simplex
        # Use step size: t_k = alpha / sqrt(k+1)
        alpha = 0.5
        t = alpha / np.sqrt(step + 1)
        
        a_new = a - t * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            best_so_far = best_obj
        a = a_new
    
    return best_a, best_obj


def diverse_init(n, rng):
    """Generate diverse initializations."""
    choice = rng.integers(0, 4)
    if choice == 0:
        return anti_palindromic_init(n, rng)
    elif choice == 1:
        # Cosine with random frequency (asymmetric via phase)
        freq = rng.uniform(1, 4)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0, np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
        if np.sum(a) < 1e-10:
            a = np.ones(n)
    elif choice == 2:
        # Random with smoothing
        a = rng.random(n)
        kernel_size = max(3, n // 50)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0)
    else:
        # Power-law distribution
        alpha = rng.uniform(0.5, 3.0)
        a = (np.arange(n) + 1) ** (-alpha)
        if rng.random() > 0.5:
            a = a[::-1]
    
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search: diverse init -> LP alternation -> mirror descent ->
    subgradient polish. Multiple sequence lengths and restarts.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 15.0
    
    best_seq = None
    best_score = np.inf
    
    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            s = np.sum(parent)
            if s > 0.01:
                obj = compute_obj(parent)
                if obj < best_score:
                    best_score = obj
                    best_seq = parent.copy()
                print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass
    
    # Sequence lengths to try (focus on 300-500 based on previous results)
    n_values = [300]
    if best_seq is not None:
        n_parent = len(best_seq)
        if n_parent not in n_values and 150 <= n_parent <= 600:
            n_values.insert(0, n_parent)
    n_values.extend([350, 400, 450, 500])
    n_values = sorted(set(n_values))
    
    print(f"N values: {n_values}, initial best={best_score:.6f}")
    
    # Phase 1: LP alternation with diverse initializations
    for n in n_values:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        print(f"\n--- N={n}: {remaining:.0f}s left, best={best_score:.6f} ---")
        
        n_restarts = max(2, min(5, int(remaining / 60)))
        
        for restart in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            
            # Choose initialization
            if restart == 0 and best_seq is not None and len(best_seq) == n:
                a_init = best_seq.copy()
            elif restart == 1:
                a_init = anti_palindromic_init(n, rng)
            else:
                a_init = diverse_init(n, rng)
            
            # LP alternation
            max_iter = max(30, min(150, int((deadline - time.monotonic()) / (n_restarts - restart + 1) / 2.0)))
            t0 = time.monotonic()
            a, obj = lp_alternate(a_init, max_iter=max_iter)
            lp_time = time.monotonic() - t0
            
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  LP n={n} r={restart}: obj={obj:.6f} ({lp_time:.1f}s)")
            
            # Mirror descent polish
            if time.monotonic() < deadline - 15 and obj < best_score + 0.05:
                ns_md = int(min(1000, (deadline - time.monotonic()) / 0.3))
                if ns_md > 100:
                    t0 = time.monotonic()
                    a_md, obj_md = mirror_descent(a, max_iter=ns_md)
                    md_time = time.monotonic() - t0
                    if obj_md < best_score:
                        best_score = obj_md
                        best_seq = a_md.copy()
                        print(f"  MD n={n} r={restart}: obj={obj_md:.6f} ({md_time:.1f}s)")
                    a = a_md if obj_md < obj else a
                    obj = min(obj, obj_md)
            
            # Second LP round from best
            if time.monotonic() < deadline - 15:
                max_iter2 = max(20, min(100, int((deadline - time.monotonic()) / 20.0)))
                a2, obj2 = lp_alternate(a, max_iter=max_iter2)
                if obj2 < best_score:
                    best_score = obj2
                    best_seq = a2.copy()
                    print(f"  LP2 n={n} r={restart}: obj={obj2:.6f}")
    
    # Phase 2: Final refinement
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n--- Final refinement: n={n_best}, obj={best_score:.6f}, {remaining:.0f}s ---")
        
        # Long subgradient polish
        ns_sg = int(min(5000, remaining / 0.05))
        if ns_sg > 200:
            t0 = time.monotonic()
            a_sg, obj_sg = subgradient_polish(best_seq.copy(), n_steps=ns_sg)
            sg_time = time.monotonic() - t0
            if obj_sg < best_score:
                best_score = obj_sg
                best_seq = a_sg.copy()
            print(f"  SG: obj={obj_sg:.6f} ({sg_time:.1f}s)")
        
        # Final LP pass
        if time.monotonic() < deadline - 5:
            max_iter = int(min(50, (deadline - time.monotonic()) / 1.0))
            if max_iter > 10:
                a_lp, obj_lp = lp_alternate(best_seq.copy(), max_iter=max_iter)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                print(f"  Final LP: obj={obj_lp:.6f}")
        
        # Quick mirror descent
        if time.monotonic() < deadline - 3:
            ns = int(min(500, (deadline - time.monotonic()) / 0.2))
            if ns > 50:
                a_md, obj_md = mirror_descent(best_seq.copy(), max_iter=ns)
                if obj_md < best_score:
                    best_score = obj_md
                    best_seq = a_md.copy()
                print(f"  Final MD: obj={obj_md:.6f}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)
    
    # Validate
    best_seq = np.clip(best_seq, 0, 1000.0)
    s = np.sum(best_seq)
    if s < 0.01:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)
    
    # Verify with evaluate_sequence if available
    try:
        eval_score = evaluate_sequence([float(x) for x in best_seq])
        if eval_score < best_score:
            best_score = eval_score
        print(f"  Eval check: {eval_score:.6f}")
    except Exception:
        pass
    
    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
