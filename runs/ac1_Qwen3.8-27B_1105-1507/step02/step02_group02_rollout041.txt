def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based best-response (Frank-Wolfe) with backtracking line search
2. Projected subgradient method with Nesterov momentum
3. Low-dimensional cosine parameterization with L-BFGS-B
4. Multi-length, multi-start search with dynamic time allocation
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time

# Top-level helper functions

def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build matrix A where (A @ x) = conv(a, x) (full linear convolution)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        # conv(a, x)[k] = sum_j a[j] * x[k-j]
        # Valid j: 0 <= j < n and 0 <= k-j < n
        # So: max(0, k-n+1) <= j <= min(n-1, k)
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            # A[k, j] = a[k - j] for j in [jl, jh]
            # When j = jl: a[k - jl], when j = jh: a[k - jh]
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A

def lp_best_response(a, M, prev_g=None):
    """
    Solve LP: max sum(g) s.t. A@g <= M, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                           'primal_feasibility_tolerance': 1e-9})
    if res.success and res.x is not None and res.x.sum() > 1e-10:
        g = np.maximum(res.x, 0)
        g = g / g.sum()
        return g
    return None

def subgradient_step(a, eta):
    """
    One projected subgradient step for minimizing max(conv(a,a)) on the simplex.
    Returns (new_a, subgrad).
    """
    n = len(a)
    b = fast_conv(a)
    k_star = int(np.argmax(b))
    # Subgradient: grad[j] = a[k_star - j] (with bounds)
    # conv(a, e_j)[k_star] = a[k_star - j] if 0 <= k_star - j < n, else 0
    grad = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            grad[j] = a[idx]
    # Projected step
    a_new = project_simplex(a - eta * grad)
    return a_new, grad, k_star

def nesterov_subgradient(a, max_iters, deadline, eta0=0.01, momentum0=0.9):
    """
    Nesterov-accelerated projected subgradient method.
    """
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    
    x = a.copy()
    y = a.copy()
    v = np.zeros(n)
    
    for t in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov momentum
        beta = (1.0 + 3.0 * np.sqrt(t + 1)) / (2.0 + 3.0 * np.sqrt(t + 1))
        
        # Compute subgradient at y
        b = fast_conv(y)
        k_star = int(np.argmax(b))
        grad = np.zeros(n)
        jl = max(0, k_star - n + 1)
        jh = min(n - 1, k_star)
        if jl <= jh:
            grad[jl:jh+1] = y[k_star - jh:k_star - jl + 1][::-1]
        
        # Update velocity
        v = beta * v + grad
        
        # Step size (decreasing)
        eta = eta0 / (1.0 + 0.01 * t)
        
        # Update
        x_new = project_simplex(y - eta * v)
        
        # Check score
        score = compute_score(x_new)
        if score < best_score:
            best_score = score
            best = x_new.copy()
        
        # Nesterov extrapolation
        y = x_new + beta * (x_new - x)
        y = project_simplex(y)
        x = x_new

def cosine_parameterize(n, coefficients):
    """
    Generate a sequence from cosine coefficients.
    coefficients: array of (alpha_0, alpha_1, phi_1, alpha_2, phi_2, ...)
    a[j] = alpha_0 + sum_k alpha_k * cos(2*pi*k*j/n + phi_k)
    """
    t = np.arange(n)
    a = np.full(n, coefficients[0])
    for k in range(1, len(coefficients) // 2 + 1):
        if 2*k < len(coefficients):
            alpha = coefficients[2*k - 1]
            phi = coefficients[2*k]
            a += alpha * np.cos(2 * np.pi * k * t / n + phi)
    return np.maximum(a, 0.0)

def cosine_optimize(a_init, n, max_harmonics, deadline):
    """
    Optimize a cosine-series parameterization of the sequence.
    """
    t = np.arange(n)
    
    # Initialize coefficients from a_init via DCT-like projection
    coeffs = [a_init.mean()]
    for k in range(1, max_harmonics + 1):
        cos_k = np.cos(2 * np.pi * k * t / n)
        sin_k = np.sin(2 * np.pi * k * t / n)
        alpha = 2.0 * np.sum(a_init * cos_k) / n  # rough init
        phi = 0.0
        # Better: fit a*cos + b*sin
        A_mat = np.column_stack([cos_k, sin_k])
        try:
            sol, _, _, _ = np.linalg.lstsq(A_mat, a_init, rcond=None)
            alpha = np.sqrt(sol[0]**2 + sol[1]**2)
            phi = np.arctan2(-sol[1], sol[0])
        except:
            pass
        coeffs.extend([alpha, phi])
    
    def objective(params):
        a = cosine_parameterize(n, params)
        s = a.sum()
        if s < 0.01:
            return 1e10
        a_norm = a / s
        b = fast_conv(a_norm)
        return float(2.0 * n * b.max())
    
    x0 = np.array(coeffs)
    bounds = [(0, None)] +  # alpha_0 >= 0
             [(None, None)] * (2 * max_harmonics)  # alpha_k, phi_k free
    
    # Actually phi should be bounded
    bounds = [(0, None)]  # alpha_0
    for k in range(max_harmonics):
        bounds.append((None, None))  # alpha_k
        bounds.append((-np.pi, np.pi))  # phi_k
    
    try:
        res = minimize(objective, x0, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 200, 'maxfun': 5000,
                                                'ftol': 1e-14, 'gtol': 1e-10})
        if res.fun < 1e10:
            a_opt = cosine_parameterize(n, res.x)
            a_opt = np.maximum(a_opt, 0)
            if a_opt.sum() > 0.01:
                a_opt = a_opt / a_opt.sum()
                return a_opt
    except:
        pass
    return None

def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t_arr = np.linspace(0, 1, n)
    
    # Resampled parent
    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        resampled = np.maximum(np.interp(t_arr, t_old, parent), 0)
        if resampled.sum() > 0.01:
            inits.append(resampled / resampled.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine (1st harmonic) - bell shape
    c = 1.0 + 0.3 * np.cos(2 * np.pi * t_arr)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())
    
    # Stronger bell
    c = 1.0 + 0.5 * np.cos(2 * np.pi * t_arr)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())
    
    # 2nd harmonic
    c = 1.0 + 0.3 * np.cos(4 * np.pi * t_arr)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())
    
    # 4th harmonic
    c = 1.0 + 0.2 * np.cos(8 * np.pi * t_arr)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())
    
    # Random
    inits.append(rng.uniform(0.3, 1.0, n) / n)
    
    # Gaussian-like bump
    c = np.exp(-2.0 * (t_arr - 0.5)**2)
    inits.append(c / c.sum())
    
    # U-shape (inverse Gaussian)
    c = 1.0 / (np.exp(-5.0 * (t_arr - 0.5)**2) + 0.1)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())
    
    # Multi-modal
    c = 1.0 + 0.3 * np.cos(2 * np.pi * t_arr) + 0.15 * np.cos(4 * np.pi * t_arr)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())
    
    return inits

For this to be an improvement, we need max(conv(a_new, a_new)) < max(conv(a, a)).

Note that conv(a_new, a_new) = (1-t)^2 conv(a,a) + 2t(1-t) conv(a,g) + t^2 conv(g,g).

The max of this is <= (1-t)^2 * M + 2t(1-t) * M + t^2 * max(conv(g,g))
= (1-t)^2 * M + 2t(1-t) * M + t^2 * max(conv(g,g))

For this to be < M:
(1-t)^2 * M + 2t(1-t) * M + t^2 * max(conv(g,g)) < M
M * ((1-t)^2 + 2t(1-t)) + t^2 * max(conv(g,g)) < M
M * (1-t^2) + t^2 * max(conv(g,g)) < M
t^2 * max(conv(g,g)) < M * t^2
max(conv(g,g)) < M

So we need max(conv(g,g)) < M for the update to be an improvement (for small t).

But the LP only constrains conv(a,g) <= M, not conv(g,g). So the improvement is not guaranteed by the LP alone.

However, in practice, the LP tends to find g that spreads mass in a way that reduces the peak. The backtracking line search ensures we only take steps that actually improve.

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based best-response with backtracking line search and FW-gap step sizing
2. Projected subgradient method with Nesterov momentum (fast per-iteration)
3. Low-dimensional cosine parameterization with smooth log-sum-exp objective
4. Multi-length, multi-start search with dynamic time allocation
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    v = v.copy()
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ x) = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success and res.x is not None and res.x.sum() > 1e-10:
            g = np.maximum(res.x, 0)
            g = g / g.sum()
            return g
    except Exception:
        pass
    return None


def get_subgradient(a, k_star):
    """Compute subgradient of max(conv(a,a)) at k_star. Returns grad vector."""
    n = len(a)
    grad = np.zeros(n)
    jl = max(0, k_star - n + 1)
    jh = min(n - 1, k_star)
    if jl <= jh:
        grad[jl:jh + 1] = 2.0 * a[k_star - jh:k_star - jl + 1][::-1]
    return grad


def lp_iteration_improved(a, max_iters, deadline):
    """
    LP best-response with backtracking line search.
    Returns (best_a, best_score).
    """
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    a = np.clip(a, 1e-10, None)
    a = a / a.sum()

    no_improve = 0
    consecutive_fail = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        g = lp_best_response(a, M)
        if g is None:
            consecutive_fail += 1
            if consecutive_fail > 5:
                break
            continue
        consecutive_fail = 0

        # Backtracking line search
        best_t = 0.0
        best_val = M + 1e-10
        a_best_try = a.copy()

        for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03]:
            a_try = (1.0 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            b_try = fast_conv(a_try)
            f_try = float(b_try.max())
            if f_try < best_val:
                best_val = f_try
                best_t = t
                a_best_try = a_try.copy()

        if best_t == 0.0:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        a = a_best_try
        no_improve = 0

        score = compute_score(a)
        if score < best_score:
            best_score = score
            best = a.copy()

    return best, best_score


def subgradient_nesterov(a, max_iters, deadline, eta0=0.01):
    """
    Nesterov-accelerated projected subgradient method.
    Returns (best_a, best_score).
    """
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    x = np.clip(a, 1e-10, None)
    x = x / x.sum()
    y = x.copy()
    v = np.zeros(n)

    for t in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Nesterov momentum
        if t == 0:
            beta = 0.0
        else:
            beta = (1.0 + 3.0 * np.sqrt(t)) / (2.0 + 3.0 * np.sqrt(t))

        # Subgradient at y
        b = fast_conv(y)
        k_star = int(np.argmax(b))
        grad = get_subgradient(y, k_star)

        # Update
        eta = eta0 / (1.0 + 0.05 * t)
        v = beta * v + grad
        x_new = project_simplex(y - eta * v)

        # Evaluate
        score = compute_score(x_new)
        if score < best_score:
            best_score = score
            best = x_new.copy()

        # Extrapolation
        y_new = x_new + beta * (x_new - x)
        y = project_simplex(y_new)
        x = x_new

    return best, best_score


def cosine_parameterize(n, params):
    """
    Generate sequence from cosine parameters.
    params = [dc, alpha_1, phi_1, alpha_2, phi_2, ...]
    a[j] = dc + sum_k alpha_k * cos(2*pi*k*j/n + phi_k)
    """
    t = np.arange(n)
    a = np.full(n, params[0])
    for k in range(1, (len(params) - 1) // 2 + 1):
        idx_a = 2 * k - 1
        idx_p = 2 * k
        if idx_p < len(params):
            a += params[idx_a] * np.cos(2.0 * np.pi * k * t / n + params[idx_p])
    return np.maximum(a, 0.0)


def smooth_objective(params, n, beta):
    """Log-sum-exp smoothed objective for cosine parameterization."""
    a = cosine_parameterize(n, params)
    s = a.sum()
    if s < 0.01:
        return 1e10
    a = a / s
    b = fast_conv(a)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50.0, 0.0)
    log_sum = b_max + np.log(np.exp(shifted).sum()) / beta
    return float(2.0 * n * log_sum)


def cosine_optimize(a_init, n, max_harmonics, deadline):
    """
    Optimize cosine-series parameterization using L-BFGS-B.
    Returns optimized sequence or None.
    """
    t = np.arange(n)

    # Initialize from DCT-like projection
    params = [float(a_init.mean())]
    for k in range(1, max_harmonics + 1):
        cos_k = np.cos(2.0 * np.pi * k * t / n)
        sin_k = np.sin(2.0 * np.pi * k * t / n)
        A_mat = np.column_stack([cos_k, sin_k])
        try:
            sol, _, _, _ = np.linalg.lstsq(A_mat, a_init, rcond=None)
            alpha = float(np.sqrt(sol[0] ** 2 + sol[1] ** 2))
            phi = float(np.arctan2(-sol[1], sol[0]))
        except Exception:
            alpha, phi = 0.0, 0.0
        params.extend([alpha, phi])

    def obj(p):
        return smooth_objective(p, n, 500.0)

    bounds = [(0.0, None)]  # dc >= 0
    for _ in range(max_harmonics):
        bounds.append((None, None))   # alpha
        bounds.append((-np.pi, np.pi))  # phi

    x0 = np.array(params)
    try:
        res = minimize(obj, x0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 300, 'maxfun': 10000,
                                'ftol': 1e-14, 'gtol': 1e-8})
        if res.fun < 1e10:
            a_opt = cosine_parameterize(n, res.x)
            a_opt = np.maximum(a_opt, 0)
            if a_opt.sum() > 0.01:
                a_opt = a_opt / a_opt.sum()
                return a_opt
    except Exception:
        pass
    return None
