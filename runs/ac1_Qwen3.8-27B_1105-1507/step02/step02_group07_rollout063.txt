def fast_convolve_squared(f):
    """Compute conv(f,f) using FFT for speed."""
    n = len(f)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    F = np.fft.rfft(f, N)
    C = np.fft.irfft(F * F, N)
    return C[:2*n-1]


def softmax_gradient_step(f, beta):
    """Compute soft-max gradient of max(conv(f,f))."""
    n = len(f)
    C = np.convolve(f, f)
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros(n), max_C
    p = exp_C / total
    grad = 2.0 * np.correlate(p, f, mode='valid')
    soft_max_val = max_C + np.log(total) / beta
    return grad, soft_max_val, p


def nesterov_softmax_optimize(f, deadline, beta_start=10.0, beta_end=500.0,
                               lr_start=0.02, max_iters=10000, verbose_prefix=""):
    """Nesterov-accelerated soft-max gradient descent."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    # Nesterov state
    y = f.copy()  # extrapolated point
    t = 1.0
    lr = lr_start
    no_improve = 0
    it = 0
    best_it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        # Adaptive beta
        it_frac = min(1.0, it / (max_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** it_frac
        
        # Compute gradient at y (extrapolated point)
        y_clipped = np.clip(y, 0.0, 1000.0)
        Sy = float(np.sum(y_clipped))
        if Sy < 1e-10:
            y = f.copy()
            y_clipped = f
            Sy = S
        y_norm = y_clipped / Sy
        
        grad, soft_val, p = softmax_gradient_step(y_norm, beta)
        
        # Nesterov update
        it += 1
        t_old = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        alpha = (t_old - 1.0) / t  # Nesterov momentum
        
        # Gradient step

        f_new = np.clip(f_new, 0.0, 1000.0)

        if S_new < 1e-10:

            S_new = S
        f_new = f_new / S_new
        
        # Nesterov extrapolation

        y = np.clip(y, 0.0, 1000.0)
        Sy = float(np.sum(y))
        if Sy < 1e-10:
            y = f_new.copy()
        
        f = f_new

                # Reset Nesterov state

                t = 1.0

                no_improve -= 100

def topk_gradient_step(f, k_frac=0.1):
    """Gradient of the mean of top-k% values of conv(f,f)."""
    n = len(f)
    C = np.convolve(f, f)
    m = len(C)
    k = max(1, int(k_frac * m))
    # Find top k values
    thresholds = np.partition(C, m-k)[m-k]
    # Weight: 1 for top k, 0 for rest (subgradient of top-k mean)
    p = (C >= thresholds).astype(np.float64) / k
    grad = 2.0 * np.correlate(p, f, mode='valid')
    return grad, float(np.mean(C[C >= thresholds]))

def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_refine(f, deadline, max_iters=100, verbose_prefix=""):
    """LP-based local refinement."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    step_size = 0.1
    no_improve = 0
    it = 0
    
    # Build A once
    A = build_conv_matrix_sparse(f)
    
    while time.monotonic() < deadline and it < max_iters:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break
        
        # Solve LP: max sum(g) s.t. A@g <= M, g >= 0
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        try:
            result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if result.success:
                g = result.x
            else:
                no_improve += 1
                if no_improve > 10:
                    break
                step_size *= 0.7
                continue
        except Exception:
            no_improve += 1
            if no_improve > 10:
                break
            continue
        
        it += 1
        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue
        g = g / S  # same normalization as f
        
        # Line search
        best_t = step_size
        best_s = best_score
        for trial_t in [step_size, step_size * 1.5, step_size * 0.5, step_size * 2]:
            new_f = (1.0 - trial_t) * f + trial_t * g
            new_f = np.clip(new_f, 0.0, 1000.0)
            ns = float(np.sum(new_f))
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = trial_t
        
        if best_s < best_score:
            f = (1.0 - best_t) * f + best_t * g
            f = np.clip(f, 0.0, 1000.0)
            S = float(np.sum(f))
            f = f / S
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            if it % 20 == 1:
                print(f"  {verbose_prefix}LP iter {it}: score={best_score:.6f}")
            # Rebuild A
            A = build_conv_matrix_sparse(f)
        else:
            no_improve += 1
            step_size *= 0.6
            if step_size < 0.005 or no_improve > 30:
                break
    
    return best_f, best_score


def asymmetric_gaussian(n, center_frac=0.4, width=0.3, left_taper=1.0, right_taper=1.0):
    """Asymmetric Gaussian initialization."""
    idx = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    sigma = width * n
    f = np.exp(-((idx - center) / sigma) ** 2)
    # Apply edge tapers
    if left_taper > 0:
        f = f * (idx / (n - 1) + 0.01) ** left_taper
    if right_taper > 0:
        f = f * ((n - 1 - idx) / (n - 1) + 0.01) ** right_taper
    f = np.maximum(f, 0.0)
    S = f.sum()
    if S > 0:
        f = f / S
    return f


def skewed_window(n, skew=0.3, alpha=2.0):
    """Skewed window: a power-law taper on one side, Gaussian on the other."""
    idx = np.arange(n, dtype=np.float64)
    # Center shifted
    center = 0.5 * (1 - skew) * (n - 1)
    # Width
    sigma = 0.35 * n
    f = np.exp(-((idx - center) / sigma) ** 2)
    # Additional skew: power taper from right
    f = f * np.maximum(1.0 - (idx - center) / (n - center + 1), 0) ** (1 + alpha * skew)
    f = np.maximum(f, 0.0)
    S = f.sum()
    if S > 0:
        f = f / S
    return f


def perturbed(f, rng, strength=0.05):
    """Add multiplicative noise."""
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.1, 2.0)
    new_f = np.clip(new_f, 0.0, 1000.0)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S

def softmax_gradient_step(f, beta):
    """Compute soft-max gradient of max(conv(f,f)). Returns (grad, soft_max_val, p)."""
    n = len(f)
    C = np.convolve(f, f)
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros(n), max_C, np.ones(len(C)) / len(C)
    p = exp_C / total
    grad = 2.0 * np.correlate(p, f, mode='valid')
    soft_max_val = max_C + np.log(total) / beta
    return grad, soft_max_val, p

        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old))
        alpha = (t_old - 1.0) / t

        f_new = y_norm - lr * grad
        f_new = np.clip(f_new, 0.0, 1000.0)
        S_new = float(np.sum(f_new))
        if S_new < 1e-10:
            f_new = f
            S_new = S
        f_new = f_new / S_new

        y = f_new + alpha * (f_new - y_norm)
        y = np.clip(y, 0.0, 1000.0)
        Sy = float(np.sum(y))
        if Sy < 1e-10:
            y = f_new.copy()

        f = f_new

        sc = evaluate_sequence(f.tolist())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f.copy()
            no_improve = 0
            if it % 500 == 0:
                print(f"  {verbose_prefix}Nest it{it}: {best_score:.6f} b={beta:.0f} lr={lr:.5f}")
        else:
            no_improve += 1
            if no_improve > 300:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-7:
                    break
                y = f.copy()
                t = 1.0
            elif no_improve > 100:
                lr *= 0.8
                no_improve = 0
                y = 0.5 * y + 0.5 * f
                t = max(1.0, t * 0.7)

    return best_f, best_score


def plain_gradient_optimize(f, deadline, beta_start=50.0, beta_end=2000.0,
                              lr_start=0.005, max_iters=20000, verbose_prefix=""):
    """Plain gradient descent with adaptive step, high beta for precision."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

    lr = lr_start
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it_frac = min(1.0, it / max(max_iters * 0.7, 1))
        beta = beta_start * (beta_end / beta_start) ** it_frac

        grad, soft_val, p = softmax_gradient_step(f, beta)
        it += 1

        f_new = f - lr * grad
        f_new = np.clip(f_new, 0.0, 1000.0)
        S_new = float(np.sum(f_new))
        if S_new < 1e-10:
            f_new = f
            S_new = S
        f_new = f_new / S_new

        sc = evaluate_sequence(f_new.tolist())
        if np.isfinite(sc) and sc < best_score:
            f = f_new
            best_score = sc
            best_f = f.copy()
            no_improve = 0
            if it % 2000 == 0:
                print(f"  {verbose_prefix}Grad it{it}: {best_score:.6f} b={beta:.0f} lr={lr:.6f}")
        else:
            no_improve += 1
            if no_improve > 200:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-8:
                    break
            elif no_improve > 80:
                lr *= 0.85
                no_improve = 0

    return best_f, best_score


def topk_optimize(f, deadline, k_frac=0.05, lr_start=0.008,
                  max_iters=10000, verbose_prefix=""):
    """Optimize using top-k mean objective for flatness."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

    lr = lr_start
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        C = np.convolve(f, f)
        m = len(C)
        k = max(1, int(k_frac * m))
        threshold = np.partition(C, m - k)[m - k]
        p = (C >= threshold).astype(np.float64) / k
        grad = 2.0 * np.correlate(p, f, mode='valid')

        f_new = f - lr * grad
        f_new = np.clip(f_new, 0.0, 1000.0)
        S_new = float(np.sum(f_new))
        if S_new < 1e-10:
            f_new = f
            S_new = S
        f_new = f_new / S_new

        sc = evaluate_sequence(f_new.tolist())
        if np.isfinite(sc) and sc < best_score:
            f = f_new
            best_score = sc
            best_f = f.copy()
            no_improve = 0
            if it % 3000 == 0:
                print(f"  {verbose_prefix}TopK it{it}: {best_score:.6f} lr={lr:.6f}")
        else:
            no_improve += 1
            if no_improve > 300:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-8:
                    break
            elif no_improve > 150:
                lr *= 0.85
                no_improve = 0

    return best_f, best_score


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_refine(f, deadline, max_iters=80, verbose_prefix=""):
    """LP-based local refinement."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    step_size = 0.08
    no_improve = 0
    it = 0

    A = build_conv_matrix_sparse(f)

    while time.monotonic() < deadline and it < max_iters:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        try:
            result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if not result.success:
                no_improve += 1
                if no_improve > 8:
                    break
                step_size *= 0.7
                continue
            g = result.x
        except Exception:
            no_improve += 1
            if no_improve > 8:
                break
            continue

        it += 1
        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue
        g = g / S

        best_t = step_size
        best_s = best_score
        for trial_t in [step_size, step_size * 1.5, step_size * 0.5, step_size * 2.0, step_size * 0.3]:
            new_f = (1.0 - trial_t) * f + trial_t * g
            new_f = np.clip(new_f, 0.0, 1000.0)
            ns = float(np.sum(new_f))
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = trial_t

        if best_s < best_score:
            f = (1.0 - best_t) * f + best_t * g
            f = np.clip(f, 0.0, 1000.0)
            S = float(np.sum(f))
            f = f / S
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            if it % 15 == 0:
                print(f"  {verbose_prefix}LP it{it}: {best_score:.6f}")
            A = build_conv_matrix_sparse(f)
        else:
            no_improve += 1
            step_size *= 0.6
            if step_size < 0.003 or no_improve > 25:
                break

    return best_f, best_score


def asymmetric_gaussian(n, center_frac=0.4, width=0.3, left_taper=0.5, right_taper=0.5):
    """Asymmetric Gaussian with edge tapers - key new initialization."""
    idx = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    sigma = width * n
    f = np.exp(-((idx - center) / sigma) ** 2)
    if left_taper > 0:
        f = f * (idx / max(n - 1, 1) + 0.05) ** left_taper
    if right_taper > 0:
        f = f * ((n - 1 - idx) / max(n - 1, 1) + 0.05) ** right_taper
    f = np.maximum(f, 0.0)
    S = f.sum()
    if S > 0:
        f = f / S
    return f


def skewed_power_window(n, skew=0.3, alpha=2.0, width=0.35):
    """Skewed window with power-law taper on one side."""
    idx = np.arange(n, dtype=np.float64)
    center = 0.5 * (1 - skew) * (n - 1)
    sigma = width * n
    f = np.exp(-((idx - center) / sigma) ** 2)
    right_extent = n - 1 - center
    if right_extent > 0:
        right_frac = np.maximum(1.0 - (idx - center) / (right_extent + 1), 0)
        f = f * right_frac ** (0.5 + alpha * skew)
    left_extent = center
    if left_extent > 0:
        left_frac = np.maximum(1.0 - (center - idx) / (left_extent + 1), 0)
        f = f * left_frac ** (0.3 * skew)
    f = np.maximum(f, 0.0)
    S = f.sum()
    if S > 0:
        f = f / S
    return f


def tukey_window(n, r=0.3):
    """Generate a Tukey window."""
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def perturbed(f, rng, strength=0.04):
    """Add multiplicative noise and renormalize."""
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.1, 2.0)
    new_f = np.clip(new_f, 0.0, 1000.0)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S


def multi_phase_optimize(f_init, deadline, verbose_prefix=""):
    """Run multi-phase optimization: Nesterov -> TopK -> HighBeta -> LP."""
    best_f = f_init.copy()
    best_score = evaluate_sequence(f_init.tolist())
    t0 = time.monotonic()
    remaining = deadline - t0

    # Phase 1: Nesterov gradient (60% of time)
    p1_deadline = t0 + remaining * 0.50
    if time.monotonic() < p1_deadline:
        bf, bs = nesterov_softmax_optimize(
            best_f, p1_deadline, beta_start=10.0, beta_end=300.0,
            lr_start=0.012, max_iters=20000, verbose_prefix=verbose_prefix + "N")
        if bs < best_score:
            best_f, best_score = bf, bs

    # Phase 2: Top-k objective (15% of time)
    p2_deadline = t0 + remaining * 0.65
    if time.monotonic() < p2_deadline:
        bf, bs = topk_optimize(
            best_f, p2_deadline, k_frac=0.05, lr_start=0.005,
            max_iters=8000, verbose_prefix=verbose_prefix + "T")
        if bs < best_score:
            best_f, best_score = bf, bs

    # Phase 3: High-beta gradient polish (15% of time)
    p3_deadline = t0 + remaining * 0.80
    if time.monotonic() < p3_deadline:
        bf, bs = plain_gradient_optimize(
            best_f, p3_deadline, beta_start=100.0, beta_end=2000.0,
            lr_start=0.003, max_iters=10000, verbose_prefix=verbose_prefix + "G")
        if bs < best_score:
            best_f, best_score = bf, bs

    # Phase 4: LP refinement (15% of time)
    if time.monotonic() < deadline - 2:
        bf, bs = lp_refine(best_f, deadline - 2, max_iters=40,
                           verbose_prefix=verbose_prefix + "L")
        if bs < best_score:
            best_f, best_score = bf, bs

    # Phase 5: Final high-beta polish
    if time.monotonic() < deadline - 1:
        bf, bs = plain_gradient_optimize(
            best_f, deadline - 1, beta_start=500.0, beta_end=5000.0,
            lr_start=0.001, max_iters=5000, verbose_prefix=verbose_prefix + "F")
        if bs < best_score:
            best_f, best_score = bf, bs

    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                score = evaluate_sequence(parent.tolist())
                if np.isfinite(score):
                    best_score = score
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    total_budget = deadline - t0
    print(f"Total budget: {total_budget:.0f}s")

    # Lengths to try - focus near 512 with some variation
    lengths_to_try = [512, 500, 520, 480, 550, 448]
    time_weights = {512: 0.28, 500: 0.10, 520: 0.10, 480: 0.10, 550: 0.08, 448: 0.07}

    for n in lengths_to_try:
        if time.monotonic() > deadline - 80:
            print("Time running out, stopping length exploration")
            break

        n_budget = total_budget * time_weights.get(n, 0.08)
        n_deadline = time.monotonic() + n_budget
        print(f"\n=== n={n} (budget {n_budget:.0f}s) ===")

        # Generate diverse initializations
        inits = []

        # Key: asymmetric Gaussians with various parameters
        for cf in [0.30, 0.35, 0.40, 0.45]:
            for w in [0.25, 0.30, 0.35, 0.40]:
                for lt in [0.3, 0.7]:
                    for rt in [0.3, 0.7]:
                        inits.append((f'asym_{cf}_{w}_{lt}_{rt}',
                                      asymmetric_gaussian(n, cf, w, lt, rt)))

        # Skewed power windows
        for s in [0.2, 0.3, 0.4, 0.5]:
            for a in [1.0, 2.0]:
                inits.append((f'skp_{s}_{a}', skewed_power_window(n, s, a)))

        # Tukey (symmetric, for comparison)
        for r in [0.3, 0.5]:
            inits.append((f'tuk_{r}', tukey_window(n, r)))

        # Parent interpolation
        if best_sequence is not None:
            if len(best_sequence) == n:
                inits.insert(0, ('parent', best_sequence.copy()))
            else:
                parent_interp = np.interp(
                    np.linspace(0, 1, n),
                    np.linspace(0, 1, len(best_sequence)),
                    best_sequence)
                parent_interp = np.clip(parent_interp, 0, None)
                inits.insert(0, ('parent_int', parent_interp))

        # Score all inits
        init_scores = []
        for label, f_init in inits:
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            S = float(np.sum(f_init))
            if S < 0.01:
                continue
            f_init = f_init / S
            sc = evaluate_sequence(f_init.tolist())
            if np.isfinite(sc):
                init_scores.append((sc, label, f_init))
                if sc < best_score:
                    best_score = sc
                    best_sequence = f_init.copy()

        init_scores.sort(key=lambda x: x[0])
        if init_scores:
            print(f"  Top init: {init_scores[0][1]} = {init_scores[0][0]:.6f}")
            if len(init_scores) > 3:
                print(f"  4th init: {init_scores[3][1]} = {init_scores[3][0]:.6f}")

        # Optimize top initializations
        top_k = min(3, len(init_scores))
        per_init_time = n_budget / max(top_k + 0.5, 1.5)

        for rank in range(top_k):
            if time.monotonic() > n_deadline - 15:
                break

            sc, label, f_init = init_scores[rank]
            # If this init is already good, give it more time
            if sc < best_score * 1.01:
                per_t = per_init_time * 1.5
            else:
                per_t = per_init_time

            init_deadline = min(n_deadline - 5, time.monotonic() + per_t)
            if time.monotonic() >= init_deadline - 5:
                continue

            print(f"  Opt [{label}] init={sc:.6f}...")
            try:
                bf, bs = multi_phase_optimize(f_init, init_deadline,
                                              verbose_prefix=f"[{label}] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  [{label}] => {bs:.6f} *** NEW BEST ***")
                else:
                    print(f"  [{label}] best={bs:.6f}")
            except Exception as e:
                print(f"  [{label}] error: {e}")

    # Extended LP refinement on best
    if best_sequence is not None and time.monotonic() < deadline - 50:
        n_best = len(best_sequence)
        print(f"\n=== Extended LP (n={n_best}, best={best_score:.6f}) ===")

        for rnd in range(4):
            if time.monotonic() > deadline - 30:
                break
            lp_dl = min(deadline - 20 - 5 * (3 - rnd), time.monotonic() + 100)
            try:
                bf, bs = lp_refine(best_sequence, lp_dl, max_iters=30,
                                   verbose_prefix=f"[XLP{rnd}]")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  XLP{rnd}: {best_score:.6f}")
            except Exception as e:
                print(f"  XLP{rnd} error: {e}")

    # Perturbation restarts
    if best_sequence is not None and time.monotonic() < deadline - 50:
        print(f"\n=== Restarts (best={best_score:.6f}) ===")
        restart_budget = deadline - time.monotonic() - 30
        n_restarts = min(5, max(2, int(restart_budget / 50)))
        per_restart = restart_budget / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break

            strength = 0.03 + 0.025 * r
            f_pert = perturbed(best_sequence, rng, strength=strength)

            # Also try a "skew perturbation": shift center of mass
            if r % 2 == 1 and len(best_sequence) > 30:
                n_p = len(best_sequence)
                center_shift = int(n_p * (0.03 * rng.random() + 0.01)) * (1 if rng.random() > 0.5 else -1)
                f_shift = np.copy(f_pert)
                if center_shift > 0:
                    f_shift[center_shift:] = f_pert[:-center_shift]
                    f_shift[:center_shift] = 0
                else:
                    f_shift[:center_shift] = f_pert[-center_shift:]
                    f_shift[center_shift:] = 0
                S_sh = f_shift.sum()
                if S_sh > 0.01:
                    f_pert = f_shift / S_sh

            r_deadline = time.monotonic() + per_restart * 0.85
            try:
                bf, bs = multi_phase_optimize(f_pert, r_deadline,
                                              verbose_prefix=f"[R{r}] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  R{r}: {bs:.6f} *** NEW BEST ***")
                else:
                    print(f"  R{r}: {bs:.6f}")
            except Exception as e:
                print(f"  R{r} error: {e}")

    # Final precision polish
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print(f"\n=== Final polish (best={best_score:.6f}) ===")
        try:
            bf, bs = plain_gradient_optimize(
                best_sequence, deadline - 3, beta_start=500.0, beta_end=10000.0,
                lr_start=0.0005, max_iters=5000, verbose_prefix="[FIN]")
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()
            # One more LP
            if time.monotonic() < deadline - 2:
                bf2, bs2 = lp_refine(best_sequence, deadline - 2, max_iters=15,
                                     verbose_prefix="[FIN LP]")
                if bs2 < best_score:
                    best_score = bs2
                    best_sequence = bf2.copy()
            print(f"  Final: {best_score:.6f}")
        except Exception as e:
            print(f"  Final error: {e}")

    if best_sequence is None:
        best_sequence = asymmetric_gaussian(512, 0.4, 0.3)
        best_score = evaluate_sequence(best_sequence.tolist())

    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
