def compute_conv(a):

    size = 1
    while size < 2 * n:

    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)

def subgradient_smooth(conv, a, tau):

"""
Multi-scale optimization using mirror descent (multiplicative weights), LP refinement,
and structured window initializations to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: entropy mirror descent for natural non-negativity handling,
adaptive tau/eta schedules, focused exploration of longer sequences,
and iterative LP-based redistribution.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (s * s)


def smoothed_subgradient(conv, a, tau):
    """Compute smoothed subgradient of max(conv) using softmax weighting.
    Returns (gradient of length n, weights of length 2n-1)."""
    n = len(a)
    max_cv = np.max(conv)
    shifted = (conv - max_cv) / tau
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    # grad[i] = 2 * sum_j a[j] * w[i+j] for valid indices
    h = np.correlate(a, w, mode='valid')
    grad = 2.0 * h
    return grad, w


def mirror_descent_optimize(a0, max_time=30.0, verbose=False):
    """Mirror descent (multiplicative weights) to minimize max(conv(a,a)) on simplex.
    Uses adaptive step size and smoothing parameter."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:

        s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    eta = 2.0 / np.sqrt(n)
    tau_init = max(2.0, n / 5.0)
    stall = 0
    patience = 150
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        conv = compute_conv(a)
        cur_max = float(np.max(conv))

        if cur_max < best_max - 1e-13:

            best_a = a.copy()
            stall = 0
        else:

                rng = np.random.default_rng(it * 7919 + n)

                s = a.sum()
                if s < 1e-10:

                a = a / s
                stall = 0
                continue

        # Adaptive tau: exponential decay from tau_init to 1

        tau_cur = max(1.0, tau_init * (1.0 / tau_init) ** frac)

        grad_c = grad - grad.mean()

        log_update = -eta * grad_c
        log_update -= log_update.max()
        multiplier = np.exp(log_update)
        a_new = a * multiplier

        s_new = a_new.sum()

            print(f"    MD it={it}: max={cur_max:.8f}, eta={eta:.4f}, tau={tau_cur:.2f}, t={elapsed:.1f}s")

def build_A_matrix(a):
    """Build Toeplitz matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            A[k, jmin:jmax + 1] = a[k - jmax:k - jmin + 1][::-1]
    return A


def lp_refine(a0, max_iters=30, time_limit=30.0, verbose=False):
    """LP-based alternating optimization. Solves max sum(g) s.t. conv(a,g) <= max(conv(a,a))."""
    start = time.monotonic()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(compute_conv(a)))
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_conv(a)
        M = float(np.max(conv))

        A = build_A_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        remaining_time = max(1.0, time_limit - (time.monotonic() - start))

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': remaining_time})
            if res.status != 0:
                break
            g = res.x
            sum_g = float(np.sum(g))
            if sum_g <= 1.0 + 1e-5:
                stall += 1
                if stall > 10:
                    break
                continue
            stall = 0

            g = g / sum_g
            t_blend = 0.35
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                break
            a_new = a_new / s_new

            obj_new = 2.0 * n * float(np.max(compute_conv(a_new)))
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
                if verbose:
                    print(f"      LP it={it}: obj={best_obj:.6f}")
            a = a_new

        except Exception:
            break

    return best_a, best_obj


def generate_initializations(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Hann window
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    inits.append(hann / hann.sum())

    # Blackman window
    if n > 2:
        blackman = (0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
                    + 0.08 * np.cos(4.0 * np.pi * x / (n - 1)))
        blackman = np.maximum(blackman, 0)
        if blackman.sum() > 0:
            inits.append(blackman / blackman.sum())

    # Triangular
    center = (n - 1) / 2.0
    tri = 1.0 - np.abs(x - center) / (center + 1e-10)
    tri = np.maximum(tri, 0)
    if tri.sum() > 0:
        inits.append(tri / tri.sum())

    # Raised cosine (cos^2)
    cos_win = np.cos(np.pi * x / max(1, n - 1)) ** 2
    inits.append(cos_win / cos_win.sum())

    # Sine (half period)
    sine_win = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine_win / sine_win.sum())

    # Parabolic
    para = 1.0 - ((x - center) / (center + 1e-10)) ** 2
    para = np.maximum(para, 0)
    if para.sum() > 0:
        inits.append(para / para.sum())

    # Random smooth
    freq = rng.uniform(0.5, 3.0)
    phase = rng.uniform(0, 2 * np.pi)
    rand_smooth = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n + phase)
    inits.append(rand_smooth / rand_smooth.sum())

    # Bimodal
    w = max(1, n // 10)
    bimod = (np.exp(-((x - n * 0.25) ** 2) / (2.0 * w * w)) +
             np.exp(-((x - n * 0.75) ** 2) / (2.0 * w * w)))
    inits.append(bimod / bimod.sum())

    return inits


def full_optimize(a, max_time=30.0, use_lp=True, n_lp_limit=500, verbose=False):
    """Full optimization pipeline: mirror descent + LP + final polish."""
    start = time.monotonic()
    a = a.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones_like(a)
        s = a.sum()

    n = len(a)

    # Phase 1: Mirror descent (50% of time)
    md_time = max_time * 0.5
    a_md, obj_md = mirror_descent_optimize(a, max_time=md_time, verbose=verbose)

    best_a = a_md.copy()
    best_obj = obj_md

    # Phase 2: LP refinement (30% of time)
    remaining = max_time - (time.monotonic() - start)
    if use_lp and remaining > 3 and n <= n_lp_limit:
        lp_time = min(remaining * 0.6, 60.0)
        a_lp, obj_lp = lp_refine(a_md, max_iters=25, time_limit=lp_time, verbose=verbose)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()

    # Phase 3: Final mirror descent polish (remaining time)
    remaining = max_time - (time.monotonic() - start)
    if remaining > 2:
        a_pol, obj_pol = mirror_descent_optimize(best_a, max_time=remaining * 0.85, verbose=False)
        if obj_pol < best_obj:
            best_obj = obj_pol
            best_a = a_pol.copy()

    return best_a, best_obj
