import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_convolution_matrix(a):
    """Build the matrix A such that (A @ b)[k] = (a*b)[k] for all k."""
    n = len(a)
    # A has shape (2n-1, n)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        # (a*b)[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i] * b[k-i]
        # Let j = k-i, so i = k-j
        # j ranges over: max(0, k-(n-1)) <= j <= min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # For each j in [j_start, j_end], the coefficient of b[j] is a[k-j]
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def solve_lp_minimize_peak(a):
    """
    Given a, find b that minimizes max_k (a*b)_k subject to sum(b)=1, b>=0.
    Returns (b, t_optimal).
    """
    n = len(a)
    A = build_convolution_matrix(a)  # shape (2n-1, n)
    # Variables: [b_0, ..., b_{n-1}, t]
    # Minimize t: c = [0, ..., 0, 1]
    num_vars = n + 1
    c = np.zeros(num_vars)
    c[-1] = 1.0
    
    # Constraints:
    # A @ b - t <= 0  =>  [A, -1] @ [b; t] <= 0
    # sum(b) = 1  =>  [1, ..., 1, 0] @ [b; t] = 1
    # b >= 0, t free (or t >= 0)
    
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    A_eq = np.zeros((1, num_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # t >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.status == 0:
        b = res.x[:n]
        t = res.x[-1]
        return b, t
    return None, None

def compute_obj(a):
    """Compute the objective function value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    """Build matrix A such that (A @ b)[k] = sum_i a[i]*b[k-i] (full conv cross term)."""
    n = len(a)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def solve_lp_min_peak(a):
