def smooth_gradient_descent(a, deadline, tau=50, lr=0.01, momentum=0.9):
    n = len(a)
    S0 = np.sum(a)
    a = a / S0 * np.sqrt(2*n)  # normalize
    best_a = a.copy()
    best_score = compute_score(a)
    
    # FISTA parameters
    v = a.copy()  # momentum variable
    t = 1.0
    
    # Precompute FFT setup
    L = 1  # power of 2
    while L < 3*n:
        L *= 2
    
    it = 0
    while time.monotonic() < deadline:
        # Compute convolution
        conv = np.convolve(a, a)  # length 2n-1
        M = np.max(conv)
        
        # Compute softmax weights
        # Use stable softmax
        shifted = tau * (conv - M)
        p = np.exp(shifted)
        p /= np.sum(p)
        
        # Compute gradient using FFT
        a_pad = np.zeros(L)
        a_pad[:n] = a
        p_flip = p[::-1]  # reverse p, length 2n-1
        p_pad = np.zeros(L)
        p_pad[:2*n-1] = p_flip
        
        A_fft = np.fft.rfft(a_pad)
        P_fft = np.fft.rfft(p_pad)
        C = np.fft.irfft(A_fft * P_fft, n=L)
        
        # grad[i] = 2 * C[2n-2-i]
        grad = 2 * C[2*n-2 : n-1 : -1]
        # Make sure we get exactly n elements
        # C[2n-2], C[2n-3], ..., C[n-1] -> that's n elements
        
        # Nesterov update
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        a_new = v - lr * grad
        a_new = np.maximum(a_new, 0)  # project onto non-negative
        # Renormalize
        S_new = np.sum(a_new)
        if S_new > 1e-10:
            a_new = a_new * (np.sqrt(2*n) / S_new)
        
        v = a_new + (t / t_new) * (a_new - a)
        t = t_new
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        it += 1
        
        # Adapt learning rate
        if it % 1000 == 0:
            lr *= 0.95
        
        if it % 10000 == 0:
            print(f"  iter {it}: score={best_score:.6f}, M={M:.4f}, lr={lr:.6f}")
    
    return best_a, best_score

"""
Optimization of step function heights to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses smooth FISTA (log-sum-exp) with FFT-based gradients as the main optimizer,
with adaptive tau schedules, Nesterov momentum, and LP-based direction finding
for final polish. Targets n=1024 sequence length.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    """Fast internal score computation (no type checks)."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = sum_j f[k-j]*g[j]."""
    n = len(f)
    rows = np.empty(n * (n + 1) // 2 + n * (n - 1) // 2, dtype=np.int32)
    cols = np.empty_like(rows)
    vals = np.empty_like(rows, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            if idx < len(rows):
                rows[idx] = k
                cols[idx] = j
                vals[idx] = f[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 60})
        if result.success and result.x is not None:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None

    """Compute gradient of smooth log-sum-exp objective using FFT.
    grad[i] = 2 * sum_k p[k] * a[k-i] for valid indices.

    p_pad[:2 * n - 1] = p_flip
    C = np.fft.irfft(np.fft.rfft(a_pad) * np.fft.rfft(p_pad), n=L)
    grad = 2.0 * C[2 * n - 2: n - 2: -1]
    return grad

    return best_a, best_score


def lp_optimize(a, deadline, verbose_prefix="", max_iters=100):
    """LP-based direction finding with line search."""
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    S = float(np.sum(a))
    a = np.maximum(a, 0.0) * (target_sum / S)

    best_a = a.copy()
    best_score = compute_score(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv_f = np.convolve(a, a)
        M = float(np.max(conv_f))

        t_start = time.monotonic()
        g = solve_lp_direction(a, M)
        lp_time = time.monotonic() - t_start

        if g is None:
            no_improve += 1
            if no_improve > 5:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            break

        g = g * (target_sum / sum_g)

        # Line search with multiple alpha values
        best_alpha = 0.0
        best_cand_score = best_score
        best_cand = None
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            cand = (1.0 - alpha) * a + alpha * g
            cand = np.maximum(cand, 0.0)
            S_cand = float(np.sum(cand))
            if S_cand < 1e-10:
                continue
            cand = cand * (target_sum / S_cand)
            s_cand = compute_score(cand)
            if s_cand < best_cand_score:
                best_cand_score = s_cand
                best_alpha = alpha
                best_cand = cand

        if best_cand is not None:
            a = best_cand
            best_a = a.copy()
            best_score = best_cand_score
            no_improve = 0
            print(f"  {verbose_prefix}LP iter {it}: score={best_score:.6f} "
                  f"(alpha={best_alpha:.2f}, time={lp_time:.1f}s)")
        else:
            no_improve += 1
            if no_improve > 10:
                break

    return best_a, best_score


def coordinate_descent(a, deadline, verbose_prefix="", max_sweeps=200000):
    """Coordinate descent: try small perturbations of individual coordinates."""
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    S = float(np.sum(a))
    a = np.maximum(a, 0.0) * (target_sum / S)

    best_a = a.copy()
    best_score = compute_score(a)
    conv = np.convolve(a, a)

    rng = np.random.default_rng(42)
    sweep = 0
    last_improve_sweep = 0

    while time.monotonic() < deadline and sweep < max_sweeps:
        # Random coordinate order
        order = rng.permutation(n)

        for i in order:
            # Try small decrease
            if a[i] <= 1e-10:
                continue
            delta = a[i] * 0.05
            a[i] -= delta
            S_new = S - delta
            if S_new < 0.01:
                a[i] += delta
                continue
            a_scaled = a * (target_sum / S_new)
            conv_new = np.convolve(a_scaled, a_scaled)
            score = compute_score(a_scaled)
            if score < best_score:
                best_score = score
                best_a = a_scaled.copy()
                conv = conv_new
                S = float(np.sum(a))
                a = a_scaled
                last_improve_sweep = sweep
            else:
                a[i] += delta  # revert
                S = S + delta - 0  # S unchanged since delta removed

        sweep += 1
        if sweep % 5000 == 0 and sweep > last_improve_sweep + 2000:
            # No improvement in a while, reduce perturbation
            break
        if sweep % 10000 == 0:
            print(f"  {verbose_prefix}coord sweep {sweep}: score={best_score:.6f}")

    return best_a, best_score


def multi_tau_fista(a, deadline, verbose_prefix=""):
    """Run FISTA with multiple tau values, keeping the best result."""
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    overall_best = a.copy()
    overall_best_score = compute_score(a)

    tau_values = [10.0, 20.0, 50.0, 100.0]
    # Allocate time: 30%, 30%, 25%, 15%
    time_fracs = [0.30, 0.30, 0.25, 0.15]
    remaining = deadline - time.monotonic()

    for idx, (tau, frac) in enumerate(zip(tau_values, time_fracs)):
        phase_deadline = time.monotonic() + remaining * frac
        if time.monotonic() > phase_deadline - 5:
            phase_deadline = time.monotonic() + 30

        print(f"  {verbose_prefix}FISTA tau={tau:.0f} "
              f"(time={frac*100:.0f}%), starting score={overall_best_score:.6f}")
        a_opt, s_opt = smooth_fista(overall_best, phase_deadline,
                                     tau_init=tau,
                                     lr_init=2.0 / np.sqrt(n),
                                     tau_schedule=30000,
                                     verbose_prefix=f"{verbose_prefix}[tau={tau:.0f}] ")
        if s_opt < overall_best_score:
            overall_best = a_opt
            overall_best_score = s_opt

    return overall_best, overall_best_score


def propose_candidate(budget_s=1500, **kwargs):
    """
    Main entry point: search for the best sequence of non-negative heights.
    Strategy:
    1. Initialize from parent (interpolated to n=1024) or uniform.
    2. Multi-tau smooth FISTA optimization (main workhorse).
    3. LP-based direction finding for refinement.
    4. Final FISTA polish with high tau.
    5. Verify with official evaluator.
    """
    t0 = time.monotonic()
    deadline = t0 + max(10.0, float(budget_s) - 30.0)  # 30s safety margin

    n = 1024
    target_sum = np.sqrt(2.0 * n)

    # Load parent sequence
    a_init = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            if parent.size == n:
                a_init = parent
            else:
                x_old = np.linspace(0.0, 1.0, parent.size)
                x_new = np.linspace(0.0, 1.0, n)
                a_init = np.interp(x_new, x_old, parent)
                a_init = np.clip(a_init, 0.0, None)
    except (NameError, Exception):
        pass

    if a_init is None or np.sum(a_init) < 1e-10:
        a_init = np.ones(n)

    best_a = a_init.copy()
    best_score = compute_score(a_init)
    print(f"Initial: n={n}, score={best_score:.6f}, sum={np.sum(a_init):.4f}")

    # Also try a few alternative initializations
    init_pool = [a_init.copy()]

    # Cosine window
    idx = np.arange(n, dtype=np.float64)
    cos_w = (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0
    init_pool.append(cos_w)

    # Slightly(random) perturbed parent
    rng = np.random.default_rng(7)
    pert = a_init * (1.0 + 0.1 * rng.standard_normal(n))
    pert = np.clip(pert, 0.0, None)
    init_pool.append(pert)

    # Evaluate all inits
    init_scores = []
    for i_init in init_pool:
        sc = compute_score(i_init)
        init_scores.append(sc)
        if sc < best_score:
            best_score = sc
            best_a = i_init.copy()

    print(f"Best init: score={best_score:.6f}")

    # Phase 1: Multi-tau FISTA (60% of budget)
    phase1_deadline = t0 + (deadline - t0) * 0.60
    print(f"\n=== Phase 1: Multi-tau FISTA (until t={phase1_deadline - t0:.0f}s) ===")

    # Use the best initial for the main optimization
    start_a = best_a.copy()
    a_opt, s_opt = multi_tau_fista(start_a, phase1_deadline, verbose_prefix="[P1] ")
    if s_opt < best_score:
        best_score = s_opt
        best_a = a_opt.copy()
        print(f"  Phase 1 improved to: {best_score:.6f}")

    # Phase 2: LP polish (25% of budget)
    phase2_deadline = t0 + (deadline - t0) * 0.85
    print(f"\n=== Phase 2: LP Polish (until t={phase2_deadline - t0:.0f}s) ===")
    a_opt, s_opt = lp_optimize(best_a, phase2_deadline, verbose_prefix="[P2] ",
                                max_iters=80)
    if s_opt < best_score:
        best_score = s_opt
        best_a = a_opt.copy()
        print(f"  Phase 2 improved to: {best_score:.6f}")

    # Phase 3: Final FISTA with high tau (10% of budget)
    phase3_deadline = deadline - 10.0
    print(f"\n=== Phase 3: Final Polish (until t={phase3_deadline - t0:.0f}s) ===")
    a_opt, s_opt = smooth_fista(best_a, phase3_deadline,
                                 tau_init=100.0,
                                 lr_init=1.0 / np.sqrt(n),
                                 tau_schedule=20000,
                                 verbose_prefix="[P3] ")
    if s_opt < best_score:
        best_score = s_opt
        best_a = a_opt.copy()
        print(f"  Phase 3 improved to: {best_score:.6f}")

    # One more LP if time permits
    if time.monotonic() < deadline - 30.0:
        a_opt, s_opt = lp_optimize(best_a, deadline - 15.0,
                                    verbose_prefix="[P3b] ", max_iters=10)
        if s_opt < best_score:
            best_score = s_opt
            best_a = a_opt.copy()
            print(f"  Phase 3b improved to: {best_score:.6f}")

    # Final: ensure valid and verify
    best_a = np.clip(best_a, 0.0, 1000.0)
    if np.sum(best_a) < 0.01:
        best_a = np.ones(n)

    final_score = evaluate_sequence([float(x) for x in best_a])
    if not np.isfinite(final_score):
        print("WARNING: Final sequence failed! Using fallback.")
        best_a = np.ones(n)
        final_score = evaluate_sequence([float(x) for x in best_a])

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_a)}, score={final_score:.6f}, "
          f"internal={best_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_a]
