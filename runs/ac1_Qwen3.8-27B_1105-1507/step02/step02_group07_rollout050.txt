    """Minimize max(conv(f,f)) using subgradient method with momentum."""

    S = np.sum(f)
    f = f / S  # normalize to sum=1
    n = len(f)

    best_score = evaluate_sequence(f.tolist())
    
    # Momentum
    v = np.zeros(n)
    beta_mom = 0.9  # momentum coefficient
    lr = 0.1  # initial learning rate
    lr_decay = 0.999
    
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline and it < max_iters:
        C = np.convolve(f, f)
        max_C = C.max()
        # Find active lags
        threshold = max_C * (1 - 1e-6)
        active = np.where(C >= threshold)[0]
        
        # Subgradient: average over active lags
        sg = np.zeros(n)
        for k in active:
            # dC[k]/df[j] = f[k-j] (if valid) + f[j-k] (if valid)
            for j in range(n):
                if 0 <= k-j < n:
                    sg[j] += f[k-j]
                if 0 <= j-k < n:
                    sg[j] += f[j-k]
        sg /= len(active)
        
        # Momentum update
        v = beta_mom * v + lr * sg
        f_new = f - v
        f_new = np.clip(f_new, 0, None)
        S_new = f_new.sum()
        if S_new < 1e-10:
            f_new = f
        else:
            f_new = f_new / S_new
        
        # Check improvement
        sc = evaluate_sequence(f_new.tolist())
        if np.isfinite(sc) and sc < best_score:
            f = f_new
            best_f = f.copy()
            best_score = sc
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-8:
                    break
        
        lr *= lr_decay
        it += 1
        
        if verbose and it % 1000 == 0:
            print(f"    SubGrad iter {it}: score={best_score:.6f}, lr={lr:.6f}, max_C={max_C:.6e}")
    
    return best_f, best_score

    p: weights of length 2n-1 (normalized, sum to 1)
    Returns gradient of sum_k p[k]*C[k] with respect to f.

        if pk < 1e-15:
            continue

        # For the first term: j ranges from max(0, k-n+1) to min(k, n-1)
        j1_start = max(0, k - n + 1)
        j1_end = min(k, n - 1)
        if j1_start <= j1_end:

            indices = np.arange(j1_start, j1_end + 1)

        j2_end = min(n - 1, n - 1 + k)
        if j2_start <= j2_end:
            indices = np.arange(j2_start, j2_end + 1)
            grad[indices] += pk * f[indices - k]
    return grad

def evaluate_obj(f):

    n = len(f)

    max_C = C.max()
    S = f.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * max_C / (S * S)


def compute_softmax_gradient(f, beta):

    n = len(f)
    C = np.convolve(f, f)
    max_C = C.max()

    # Numerical stability

    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros(n), 0.0
    p = exp_C / total
    
    # Compute gradient: grad[j] = 2 * sum_k p[k] * (f[k-j] + f[j-k]) (valid indices)

def softmax_adam_optimize(f, deadline, beta_start=5.0, beta_end=300.0,
                          lr_start=0.05, max_iters=10000):
    """Optimize using soft-max gradient with Adam."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        it_frac = min(1.0, it / (max_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** it_frac
        
        grad, _ = compute_softmax_gradient(f, beta)
        
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        new_f = f - lr * update
        new_f = np.clip(new_f, 0.0, None)
        ns = float(np.sum(new_f))

        else:

            if no_improve > 150:

                no_improve = 0
