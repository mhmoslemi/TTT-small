    n = len(f)
    a = f[i]

    # conv[j] = sum_k f[k]*f[j-k]

The LP is: max 1^T b subject to A b <= M, b >= 0
where A is the (2n-1) x n convolution matrix and M is the current max convolution.

The dual is: min M * 1^T y subject to A^T y >= 1, y >= 0

By strong duality, at the optimum, the primal and dual are both tight.

The complementary slackness conditions tell us:
- If b[i] > 0, then (A^T y)[i] = 1
- If y[j] > 0, then (A b)[j] = M

This means the optimal b is supported on the set of i where (A^T y)[i] = 1, and the optimal y is supported on the set of j where (A b)[j] = M.

This gives us a way to characterize the solution: we need to find a set of "active" positions J (where convolution equals M) and "active" coordinates I (where b > 0), such that:
- For j in J: sum_{i in I} A[j][i] * b[i] = M
- For i in I: sum_{j in J} A[j][i] * y[j] = 1
- y >= 0, b >= 0

This is a system of linear equations that can be solved if |J| = |I| (or we have enough constraints).

In practice, the LP solution tends to have a small number of active constraints (by the fundamental theorem of LP), so this could be efficient.

"""
LP-driven optimization with pressure updates and coordinate descent for
minimizing the autocorrelation peak of non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
import time


def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def score(f):
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)


def build_conv_matrix(n):
    """Build A such that A @ b = np.convolve(ones, b) shifted... 
    Actually, we need A_f such that A_f @ g = np.convolve(f, g).
    But f changes, so we need to rebuild each time.
    For the LP, we fix f and solve for g."""
    pass  # We'll build it inline


def build_conv_matrix_for_f(f):
    """Build matrix A (2n-1 x n) such that A @ g = np.convolve(f, g, mode='full')."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # g[j] contributes to conv[j+i] for i in range(n), with coefficient f[i]
        # Actually: conv[k] = sum_i f[i] * g[k-i]
        # So g[j] contributes to conv[k] where k-j = i, i.e., k = j+i
        # Contribution is f[i] * g[j] to conv[j+i]
        # So A[j+i, j] = f[i] for i in range(n)
        A[j:j+n, j] = f
    return A


def lp_step(f, M, lp_time=2):
    """Solve LP: max sum(g) s.t. A_f @ g <= M, g >= 0. Returns normalized g."""
    n = len(f)
    A = build_conv_matrix_for_f(f)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': lp_time})
    
    if not res.success or res.x is None:
        return None
    g = np.clip(res.x, 0, None)
    g_sum = g.sum()
    if g_sum < 1e-15:

    return g / g_sum

def exact_coordinate_descent(f, n_sweeps=50, rng=None):
    """Coordinate descent with exact 1D optimization."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(f)
    f = f.copy().astype(np.float64)
    conv = np.convolve(f, f)
    best_s = score(f)
    
    for sweep in range(n_sweeps):
        improved = False
        order = rng.permutation(n) if rng else np.arange(n)
        
        for i in order:
            a_old = f[i]
            if a_old < 1e-12:
                continue
            
            S = f.sum() - a_old
            if S < 1e-12:
                continue
            
            # D[j] = f[j-i] for 0 <= j-i < n, else 0
            D = np.zeros(2*n-1)
            D[i:i+n] = f
            
            # base[j] = conv[j] - a_old * D[j]
            base = conv - a_old * D
            
            # Minimize phi(a) = max_j(base[j] + a*D[j]) / (S+a)^2
            # Grid search
            # The optimal a is likely in [0, 3*a_old]
            a_vals = np.concatenate([
                [0.0],
                np.linspace(0, a_old, 10),
                np.linspace(a_old, 2*a_old, 5),
                np.linspace(2*a_old, 3*a_old, 3),
            ])
            a_vals = np.unique(a_vals)
            
            best_a = a_old
            best_local = 2*n*conv.max() / f.sum()**2
            
            for a_new in a_vals:
                # max_j(base[j] + a_new * D[j])
                # = max of (base + a_new * D)
                # We can compute this efficiently
                vals = base + a_new * D
                max_val = vals.max()
                sum_new = S + a_new
                if sum_new < 1e-12:
                    continue
                s_new = 2*n*max_val / (sum_new * sum_new)
                if s_new < best_local - 1e-12:
                    best_local = s_new
                    best_a = a_new
            
            if best_a != a_old:
                f[i] = best_a
                conv = conv + (best_a - a_old) * D
                improved = True
        
        s = 2*n*conv.max() / f.sum()**2
        if s < best_s:
            best_s = s
        if not improved:
            break
    
    return f, best_s


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    r = rng.random()
    if r < 0.3:
        # Smooth random
        f = np.ones(n)
        for _ in range(3 + int(rng.integers(0, 5))):
            center = rng.uniform(0.1*n, 0.9*n)
            width = rng.uniform(0.05*n, 0.3*n)
            kernel = np.exp(-0.5*(np.arange(n)-center)**2/width**2)
            f *= (1 + rng.uniform(0.3, 1.0) * rng.standard_normal() * kernel)
        f = np.abs(f) + 0.01
    elif r < 0.6:
        # Cosine-based
        f = 0.5*(1 + np.cos(2*np.pi*np.arange(n)/n * rng.uniform(0.5, 2.0)))
        f = f + 0.05*np.abs(rng.standard_normal(n))
    else:
        # Piecewise constant
        n_blocks = int(rng.integers(3, 10))
        f = np.repeat(rng.uniform(0.5, 1.5, n_blocks), n//n_blocks + 1)[:n]
    
    f = np.clip(f, 0.01, None)
    f = normalize(f)
    return f


def lp_driven_optimize(f, time_budget, rng, lp_time=3, max_iter=200):
    """LP-driven optimization with adaptive step size."""
    n = len(f)
    f = f.copy().astype(np.float64)
    best_f = f.copy()

    t = 0.5  # blending parameter
    start = time.monotonic()
    
    for it in range(max_iter):
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 5:
            break
        
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Solve LP
        g = lp_step(f, M, lp_time=lp_time)
        if g is None:
            break
        
        # Try multiple step sizes
        improved = False
        for step in sorted([1.0, 0.7, 0.5, 0.3, 0.2, 0.1], reverse=True):
            f_new = (1-step)*f + step*g
            f_new = np.clip(f_new, 0, None)
            s_new = f_new.sum()
            if s_new < 1e-10:
                continue
            f_new = f_new / s_new

            if s < best_s - 1e-10:
                best_s = s
                best_f = f_new.copy()
                f = f_new
                t = min(1.0, step * 1.2)
                improved = True
                break
        
        if not improved:
            t *= 0.5
            if t < 0.01:

                f_pert = np.clip(f_pert, 0, None)

                    f = f_pert
                    best_f = f.copy()

                    t = 0.3
                else:
                    break
        
        if it % 10 == 0:
            print(f"  LP iter {it}: score={best_s:.6f}, t={t:.3f}, elapsed={elapsed:.0f}s")
    
    return best_f, best_s


def fast_pressure(f, n_iter=1000, eta=0.05, rng=None):
    """Fast pressure update using FFT-based pressure computation."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(f)
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_s = score(f)
    stall = 0
    
    n_fft = 1 << (2*n).bit_length()
    
    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Binary threshold: mark positions where conv > M * threshold
        threshold = M * 0.95
        mask = (conv > threshold).astype(np.float64)
        
        # Pressure via FFT cross-correlation
        # pressure[i] = sum_j mask[j] * conv[j] * f[j-i] (weighted)
        # Simpler: pressure[i] = sum_j (conv[j] - threshold)+ * f[j-i]
        exceed = np.clip(conv - threshold, 0, None)
        
        # Compute pressure using FFT
        # pressure[i] = sum_j exceed[j] * f[j-i] for 0 <= j-i < n
        # = cross-correlation
        F = np.fft.rfft(f, n_fft)
        Ex = np.fft.rfft(exceed, n_fft)
        # The cross-correlation R[i] = sum_j a[j] * b[j-i]
        # In frequency: this is IFFT(conj(A) * B) gives sum_j a[j-i] * b[j]... 
        # Let me be more careful.
        # Define: R[i] = sum_j exceed[j] * f[j-i] (with 0<=j-i<n)
        # = sum_m exceed[m+i] * f[m] (substituting m=j-i)
        # This is the cross-correlation of exceed and f at lag i
        # 
        # Using FFT: if we zero-pad both to length N >= 2n-1:
        # C = ifft(conj(FFT(exceed)) * FFT(f)) gives circular correlation
        # C[k] = sum_n exceed[n] * f[(n-k) mod N]
        # For k = i: C[i] = sum_n exceed[n] * f[(n-i) mod N]
        # With zero-padding, (n-i) mod N = n-i for n >= i (since N >= 2n-1 > i)
        # So C[i] = sum_{n>=i} exceed[n] * f[n-i] = sum_{m>=0} exceed[m+i] * f[m] ✓
        
        # So: pressure = ifft(conj(FFT(exceed_padded)) * FFT(f_padded))[:n]
        F_f = np.fft.rfft(f, n_fft)
        F_ex = np.fft.rfft(exceed, n_fft)
        pressure_full = np.fft.irfft(np.conj(F_ex) * F_f, n_fft)[:n]
        
        if pressure_full.max() > 0:
            eta_eff = eta / (1.0 + 0.01*it)
            f = f * np.exp(-eta_eff * pressure_full / pressure_full.max())
            f = np.clip(f, 1e-15, None)
            f = normalize(f)
        
        s = score(f)
        if s < best_s - 1e-10:
            best_s = s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                break
    
    return best_f, best_s

"""
LP-driven optimization with FFT-based pressure updates and exact coordinate descent
for minimizing the autocorrelation peak of non-negative sequences.
Strategy: Alternate between LP-based direction finding and fast multiplicative
pressure updates, finishing with coordinate descent for fine-tuning.
"""

import numpy as np
from scipy.optimize import linprog
import time


def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def score(f):
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)


def build_conv_matrix_for_f(f):
    """Build A (2n-1 x n) such that A @ g = np.convolve(f, g, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_solve(f, M, lp_time=3):
    """Solve LP: max sum(g) s.t. A_f @ g <= M, g >= 0. Returns normalized g or None."""
    n = len(f)
    A = build_conv_matrix_for_f(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': lp_time})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    g = np.clip(res.x, 0, None)
    g_sum = g.sum()
    if g_sum < 1e-15:
        return None
    return g / g_sum


def fast_pressure(f, n_iter=2000, eta=0.05, threshold_frac=0.9, rng=None):
    """FFT-based multiplicative pressure update."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(f)
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_s = score(f)
    stall = 0
    n_fft = 1 << (2 * n).bit_length()

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        threshold = M * threshold_frac
        exceed = np.clip(conv - threshold, 0, None)

        # Pressure via FFT cross-correlation
        F_f = np.fft.rfft(f, n_fft)
        F_ex = np.fft.rfft(exceed, n_fft)
        pressure_full = np.fft.irfft(np.conj(F_ex) * F_f, n_fft)[:n]
        pressure_full = np.clip(pressure_full, 0, None)

        if pressure_full.max() > 1e-15:
            eta_eff = eta / (1.0 + 0.005 * it)
            f = f * np.exp(-eta_eff * pressure_full / pressure_full.max())
            f = np.clip(f, 1e-15, None)
            f = normalize(f)

        s = score(f)
        if s < best_s - 1e-10:
            best_s = s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                # Perturb to escape
                f_pert = f * (1.0 + 0.08 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 1e-15, None)
                f_pert = normalize(f_pert)
                if score(f_pert) < best_s:
                    f = f_pert
                    best_f = f.copy()
                    best_s = score(f)
                    stall = 0
                else:
                    break

    return best_f, best_s


def lp_driven_optimize(f, time_budget, rng, lp_time=3, max_iter=500):
    """LP-driven optimization with adaptive blending."""
    n = len(f)
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_s = score(f)
    t = 0.5
    start = time.monotonic()
    no_improve = 0

    for it in range(max_iter):
        elapsed = time.monotonic() - start
        if elapsed > time_budget - 5:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        g = lp_solve(f, M, lp_time=lp_time)
        if g is None:
            break

        # Try multiple step sizes (larger first for faster convergence)
        improved = False
        step_options = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]
        for step in step_options:
            f_new = (1 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s_new = f_new.sum()
            if s_new < 1e-10:
                continue
            f_new = f_new / s_new
            s = score(f_new)
            if s < best_s - 1e-10:
                best_s = s
                best_f = f_new.copy()
                f = f_new
                improved = True
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                # Try larger perturbation

                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if score(f_pert) < best_s:
                    f = f_pert
                    best_f = f.copy()
                    best_s = score(f)
                    no_improve = 0
                elif no_improve >= 15:
                    break
            # Also try element-wise product direction
            if no_improve == 3:
                f_prod = f * g
                f_prod = np.clip(f_prod, 0, None)
                sp = f_prod.sum()
                if sp > 1e-10:
                    f_prod = f_prod / sp
                    s_prod = score(f_prod)
                    if s_prod < best_s:
                        best_s = s_prod
                        best_f = f_prod.copy()
                        f = f_prod
                        improved = True
                        no_improve = 0

        if it % 20 == 0 and it > 0:
            print(f"    LP iter {it}: score={best_s:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_s


def exact_coordinate_descent(f, n_sweeps=50, rng=None):
    """Coordinate descent with exact 1D optimization per coordinate."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(f)
    f = f.copy().astype(np.float64)
    conv = np.convolve(f, f)
    best_s = score(f)
    total_sum = f.sum()

    for sweep in range(n_sweeps):
        improved = False
        order = rng.permutation(n)

        for i in order:
            a_old = f[i]
            if a_old < 1e-12:
                continue

            S = total_sum - a_old
            if S < 1e-12:
                continue

            # D[j] = f[j-i] for 0 <= j-i < n
            D = np.zeros(2 * n - 1)
            D[i:i + n] = f

            # base[j] = conv[j] - a_old * D[j]
            base = conv - a_old * D

            # Minimize phi(a) = max(base + a*D) / (S+a)^2 over a >= 0
            # Grid search
            a_candidates = np.concatenate([
                [0.0],
                np.linspace(0.0, a_old, 8),
                np.linspace(a_old, 2.0 * a_old, 4),
            ])
            a_candidates = np.unique(np.clip(a_candidates, 0, None))

            best_a = a_old
            best_local = 2 * n * conv.max() / (total_sum * total_sum)

            for a_new in a_candidates:
                max_val = (base + a_new * D).max()
                sum_new = S + a_new
                if sum_new < 1e-12:
                    continue
                s_new = 2 * n * max_val / (sum_new * sum_new)
                if s_new < best_local - 1e-12:
                    best_local = s_new
                    best_a = a_new

            # Refine around best
            if best_a > 0 and best_a < a_old * 1.5:
                a_fine = np.linspace(max(0, best_a * 0.7), best_a * 1.3, 10)
                for a_new in a_fine:
                    max_val = (base + a_new * D).max()
                    sum_new = S + a_new
                    if sum_new < 1e-12:
                        continue
                    s_new = 2 * n * max_val / (sum_new * sum_new)
                    if s_new < best_local - 1e-12:
                        best_local = s_new
                        best_a = a_new

            if best_a != a_old:
                conv = base + best_a * D
                total_sum = S + best_a
                f[i] = best_a
                improved = True

        s = 2 * n * conv.max() / (total_sum * total_sum)
        if s < best_s:
            best_s = s
        if not improved:
            break

    return f, best_s


def diversified_lp_search(f, n_tries=3, time_budget=300, rng=None, lp_time=3):
    """Try multiple LP directions with different M values."""
    if rng is None:

        # Try M * factor for different factors

        rng.shuffle(factors)

        for factor in factors:
            elapsed = time.monotonic() - start
            if elapsed > time_budget - 5:
                break

            M_eff = M * factor
            g = lp_solve(best_f, M_eff, lp_time=lp_time)
            if g is None:

            # Try blending
            for step in [0.5, 0.3, 0.7, 0.2]:
                f_new = (1 - step) * best_f + step * g
                f_new = np.clip(f_new, 0, None)
                s_new = f_new.sum()
                if s_new < 1e-10:

                f_new = f_new / s_new
                s = score(f_new)
                if s < best_s - 1e-10:
                    best_s = s
                    best_f = f_new.copy()
                    break

    return best_f, best_s
