from scipy.optimize import linprog, minimize
import time

    """FFT-based full linear self-convolution."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (S * S))


def build_conv_matrix(a):
    """Build matrix A where (A @ x)[k] = (conv(a, x))[k]."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):

        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            A[k, i_lo:i_hi + 1] = a[k - i_hi:k - i_lo + 1][::-1]
    return A

            break
        x0 = best.copy()
        x0 = np.maximum(x0, 1e-8)
        x0 = x0 / x0.sum()

        try:
            res = minimize(smooth_objective, x0, args=(beta, n),
                           method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': 300, 'maxfun': 10000,
                                    'ftol': 1e-20, 'gtol': 1e-20,
                                    'maxls': 50})
            if res.fun < 1e10:
                res_x = np.maximum(res.x, 0.0)
                if res_x.sum() > 0.01:
                    score = compute_score(res_x)
                    if score < best_score:
                        best = res_x.copy()
                        best_score = score
        except Exception:
            pass

    return best, best_score


def coordinate_descent_polish(a, deadline, n_sweeps=50):
    """Coordinate descent: for each coordinate, try local improvements."""
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 0.01:
        return a, compute_score(a)
    a = a / S
    n = len(a)
    best_score = compute_score(a)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        for i in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-15:
                # Try setting to small positive
                candidates = [1e-4, 3e-4, 1e-3, 3e-3]
            else:
                candidates = [old_val * 0.3, old_val * 0.6, old_val * 0.8,
                              old_val * 1.2, old_val * 1.5, old_val * 2.0,
                              0.0]
            best_save = a.copy()
            best_local = best_score
            for cv in candidates:
                a[i] = max(0.0, cv)
                s = a.sum()
                if s < 0.01:
                    a = best_save.copy()
                    continue
                score = compute_score(a)
                if score < best_local:
                    best_local = score
                    best_save = a.copy()
            a = best_save.copy()
            if best_local < best_score:
                best_score = best_local

    return a, best_score


def resample_sequence(a, n_new):
    """Resample a sequence to a new length using linear interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    t_old = np.linspace(0, 1, n_old)
    t_new = np.linspace(0, 1, n_new)
    a_new = np.interp(t_new, t_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0.01:
        a_new = a_new / s
    return a_new


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n)

    if parent is not None and len(parent) > 1:
        p = resample_sequence(parent, n)
        inits.append(p)

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine (various frequencies)
    for freq in [1, 2, 3, 4]:
        c = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    # Bimodal (two bumps)
    for center in [0.25, 0.35]:
        c = np.exp(-50 * (t - center) ** 2) + np.exp(-50 * (t - (1 - center)) ** 2)
        inits.append(c / c.sum())

    # Quartic bump
    c = 1.0 - 4.0 * (t - 0.5) ** 2
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())

    # Random smooth
    x = rng.uniform(0.2, 1.0, n)
    # Smooth with a small kernel
    kernel = np.array([0.25, 0.5, 0.25])
    x_sm = np.convolve(x, kernel, mode='same')
    x_sm = np.maximum(x_sm, 0.01)
    inits.append(x_sm / x_sm.sum())

    # Triangular
    c = 1.0 - np.abs(2 * t - 1)
    c = np.maximum(c, 0.05)
    inits.append(c / c.sum())

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Uses LP best-response with line search,
    smooth refinement, coordinate descent, and perturbation restarts.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 30.0

    best = None
    best_score = float('inf')

    # Try parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.clip(p, 0.0, 1000.0)
            sc = compute_score(p)
            if sc < best_score:
                best = p.copy()
                best_score = sc
                parent = p.copy()
                print(f"Parent: J={sc:.6f} (n={p.size})", flush=True)
    except (NameError, AttributeError):
        pass

    # === PHASE 1: Multi-start LP across lengths (45% time) ===
    lp_budget = total_budget * 0.45
    lp_deadline = time.monotonic() + lp_budget
    print(f"\n=== Phase 1: Multi-start LP (budget {lp_budget:.0f}s) ===", flush=True)

    # Determine lengths to try
    lengths = [512, 256, 768, 384, 1024, 640]
    if parent is not None:
        plen = parent.size
        lengths = [plen] + [l for l in lengths if l != plen]

    for n in lengths:
        if time.monotonic() > lp_deadline:
            break
        remaining = lp_deadline - time.monotonic()
        # Skip if not enough time for this length
        min_time_for_n = 30 if n <= 512 else 60 if n <= 1024 else 120
        if remaining < min_time_for_n:
            continue

        t0 = time.monotonic()
        print(f"  n={n}, remaining={remaining:.0f}s", flush=True)

        inits = generate_initializations(n, rng, best)

        # Limit initializations based on time
        n_inits = min(len(inits), max(2, int(remaining / 40)))

        for i, a0 in enumerate(inits[:n_inits]):
            if time.monotonic() > lp_deadline:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()

            remaining = lp_deadline - time.monotonic()
            # Run LP with both symmetric and asymmetric modes
            use_sym = (i % 2 == 0)
            a_res, score = lp_iteration(a0, lp_deadline, max_iters=20000,
                                        symmetric=use_sym, verbose=(i == 0))
            if score < best_score:
                best = a_res.copy()
                best_score = score
                print(f"    NEW BEST: J={score:.6f} (n={n}, init={i}, sym={use_sym})", flush=True)

        print(f"  n={n} done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === PHASE 2: Smooth refinement (20% time) ===
    smooth_budget = total_budget * 0.20
    smooth_deadline = min(time.monotonic() + smooth_budget, deadline)
    if best is not None and time.monotonic() < smooth_deadline:
        t0 = time.monotonic()
        print(f"\n=== Phase 2: Smooth Refinement (budget {smooth_deadline - time.monotonic():.0f}s) ===", flush=True)

        a_res, score = smooth_refine(best, smooth_deadline)
        if score < best_score:
            best = a_res.copy()
            best_score = score
            print(f"    NEW BEST (smooth): J={score:.6f} (n={best.size})", flush=True)

        # Try at a neighboring length
        n_alt = best.size * 2 if best.size <= 1024 else best.size // 2
        n_alt = max(128, min(2048, n_alt))
        if n_alt != best.size and time.monotonic() < smooth_deadline - 60:
            a_alt = resample_sequence(best, n_alt)
            a_res2, score2 = smooth_refine(a_alt, smooth_deadline)
            if score2 < best_score:
                best = a_res2.copy()
                best_score = score2
                print(f"    NEW BEST (smooth n={n_alt}): J={score2:.6f}", flush=True)

        print(f"  Smooth done in {time.monotonic()-t0:.1f}s", flush=True)

    # === PHASE 3: Perturbation + LP restart (25% time) ===
    perturb_budget = total_budget * 0.25
    perturb_deadline = min(time.monotonic() + perturb_budget, deadline)
    if best is not None and time.monotonic() < perturb_deadline:
        t0 = time.monotonic()
        print(f"\n=== Phase 3: Perturbation Search (budget {perturb_deadline - time.monotonic():.0f}s) ===", flush=True)
        n = best.size

        # Estimate time per perturbation trial
        est_per_trial = 15 if n <= 512 else 30 if n <= 1024 else 60
        n_trials = max(5, min(50, int((perturb_deadline - time.monotonic()) / est_per_trial)))

        for trial in range(n_trials):
            if time.monotonic() > perturb_deadline:
                break

            # Different perturbation strategies
            strategy = trial % 4
            if strategy == 0:
                # Gaussian noise
                scale = 0.05 + 0.05 * rng.random()
                a_pert = best + rng.normal(0, scale, n)
            elif strategy == 1:
                # Random spike (increase a random subset)
                a_pert = best.copy()
                idx = rng.choice(n, size=n // 8, replace=False)
                a_pert[idx] *= (1.0 + 0.5 * rng.random(len(idx)))
            elif strategy == 2:
                # Random dip (decrease a random subset)
                a_pert = best.copy()
                idx = rng.choice(n, size=n // 8, replace=False)
                a_pert[idx] *= (1.0 - 0.5 * rng.random(len(idx)))
            else:
                # Smooth noise (low-pass filtered)
                noise = rng.normal(0, 0.1, n)
                kernel_size = max(3, n // 16)
                if kernel_size % 2 == 0:
                    kernel_size += 1
                kernel = np.ones(kernel_size) / kernel_size
                noise = np.convolve(noise, kernel, mode='same')
                a_pert = best + noise

            a_pert = np.maximum(a_pert, 0.0)
            s = a_pert.sum()
            if s < 0.01:
                continue
            a_pert = a_pert / s

            # Short LP refinement
            remaining = perturb_deadline - time.monotonic()
            max_iters = max(30, int(remaining / (n_trials - trial) * 0.7 / (0.02 if n <= 512 else 0.05)))
            max_iters = min(max_iters, 5000)

            a_res, score = lp_iteration(a_pert, perturb_deadline, max_iters=max_iters)
            if score < best_score:
                best = a_res.copy()
                best_score = score
                print(f"    NEW BEST (pert {trial}): J={score:.6f} (n={n})", flush=True)

        print(f"  Perturb done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === PHASE 4: Final polish (15% time) ===
    polish_deadline = deadline
    if best is not None and time.monotonic() < polish_deadline:
        t0 = time.monotonic()
        print(f"\n=== Phase 4: Final Polish (budget {polish_deadline - time.monotonic():.0f}s) ===", flush=True)
        n = best.size

        # Extended LP
        a_res, score = lp_iteration(best, polish_deadline, max_iters=50000,
                                     symmetric=False)
        if score < best_score:
            best = a_res.copy()
            best_score = score
            print(f"    NEW BEST (LP polish): J={score:.6f}", flush=True)

        # Coordinate descent
        if time.monotonic() < polish_deadline - 30:
            a_res, score = coordinate_descent_polish(best, polish_deadline,
                                                      n_sweeps=200)
            if score < best_score:
                best = a_res.copy()
                best_score = score
                print(f"    NEW BEST (coord desc): J={score:.6f}", flush=True)

        # Final smooth pass
        if time.monotonic() < polish_deadline - 10:
            a_res, score = smooth_refine(best, polish_deadline)
            if score < best_score:
                best = a_res.copy()
                best_score = score
                print(f"    NEW BEST (final smooth): J={score:.6f}", flush=True)

        print(f"  Polish done in {time.monotonic()-t0:.1f}s", flush=True)

    # === Final validation ===
    if best is None:
        best = np.ones(256) / 256
        best_score = compute_score(best)

    # Ensure all values are valid
    best = np.clip(best, 0.0, 1000.0)
    final_score = float(evaluate_sequence([float(x) for x in best]))
    if np.isfinite(final_score):
        best_score = min(best_score, final_score)

    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
