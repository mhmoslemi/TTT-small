```python
def multiplicative_update(a, num_iters, eta):
    n = len(a)
    for t in range(num_iters):
        conv = np.convolve(a, a)
        M = np.max(conv)
        # Find active set (positions close to max)
        active = conv >= M * 0.99
        # Subgradient: average gradient over active positions
        # grad of q_k at a: for each j, d/d a_j q_k = sum of a_{k-j} for valid k-j
        # Actually, d/d a_j (a*a)[k] = a_{k-j} if 0 <= k-j < n, else 0
        # So the gradient of max is the average of a reversed at active positions
        grad = np.zeros(n)
        for k in np.where(active)[0]:
            # (a*a)[k] = sum_j a_j * a[k-j] where 0<=j<n and 0<=k-j<n
            # d/d a_j = a[k-j] if 0<=k-j<n
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            grad[j_start:j_end+1] += a[k - j_end:k - j_start+1][::-1]
        grad /= np.sum(active)
        # Also subtract the gradient of the normalization (since we're on the simplex)
        # Actually, let's just normalize after the update
        a = a * np.exp(-eta * grad)
        a = np.clip(a, 1e-15, None)
        a = a / np.sum(a)
    return a
```

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_conv_matrix(a):
    """Build the matrix A such that A @ a = a * a (convolution)."""
    n = len(a)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        # (a*a)[k] = sum_j a_j * a[k-j]
        # d/d a_j of (a*a)[k] = a[k-j] for 0 <= k-j < n
        # So A[k, j] = a[k-j] if 0 <= k-j < n
        for j in range(j_start, j_end+1):
            A[k, j] = a[k-j]
    return A

def mirror_descent(a, max_time, T_init=0.1, eta_init=0.5):
    """Mirror descent with soft-max smoothing."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    conv = np.convolve(a, a)
    best_obj = 2.0 * n * float(np.max(conv)) / (float(np.sum(a))**2)
    start = time.monotonic()
    T = T_init * float(np.max(conv))
    eta = eta_init
    last_improve = 0
    
    for it in range(50000):
        if time.monotonic() - start > max_time:
            break
        
        conv = np.convolve(a, a)
        max_c = float(np.max(conv))
        sum_a = float(np.sum(a))
        obj = 2.0 * n * max_c / (sum_a * sum_a)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            last_improve = it
        
        if it - last_improve > 2000:
            break
        
        # Soft-max gradient
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        p = exp_vals / float(np.sum(exp_vals))
        
        # Gradient of soft-max
        # grad_j = (1/T) * sum_k p_k * a[k-j] (for valid k-j)
        grad = np.zeros(n)
        for k in range(2*n-1):
            if p[k] < 1e-15:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n-1, k)
            # a[k-j] for j in [j_start, j_end]
            indices = k - np.arange(j_start, j_end+1)
            grad[j_start:j_end+1] += p[k] * a[indices]
        grad /= T
        
        # Multiplicative update
        log_step = -eta * grad
        log_step -= np.max(log_step)  # for numerical stability
        a = a * np.exp(log_step)
        a = np.clip(a, 1e-15, None)
        a = a / float(np.sum(a))
        
        # Adaptive T and eta
        if obj < best_obj * 1.001:
            T = max(0.001 * max_c, T * 0.99)
            eta = min(2.0, eta * 1.01)
        else:
            eta = max(0.01, eta * 0.8)
            T = T * 1.2
    
    return best_a, best_obj

def lp_alternate(a, max_time):
    """LP-based alternating optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    conv = np.convolve(a, a)
    best_obj = 2.0 * n * float(np.max(conv)) / (float(np.sum(a))**2)
    start = time.monotonic()
    
    for it in range(200):
        if time.monotonic() - start > max_time:
            break
        
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        sum_a = float(np.sum(a))
        
        # LP: maximize sum(b) subject to (a*b)[k] <= M, b >= 0
        # (a*b)[k] = sum_j a[k-j] * b[j] (where a[k-j] is 0 if out of range)
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    A[k, j] = a[idx]
        
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status != 0:
            break
        
        g = res.x
        g_sum = float(np.sum(g))
        if g_sum < 1.0 + 1e-6:
            break
        
        # The LP optimum has sum = g_sum > 1 (since the current a has sum 1
        # and satisfies the constraints, so the LP can do at least as well)
        # Actually, a itself is feasible with sum = sum_a, so g_sum >= sum_a.
        # If we normalize a to sum 1, then g_sum >= 1.
        
        # Rescale g to have the same sum as a
        g = g / g_sum * sum_a
        
        # Line search
        best_t = 0
        best_candidate_obj = best_obj
        for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_new = (1-t) * a + t * g
            a_new = np.clip(a_new, 0, None)
            s = float(np.sum(a_new))
            if s < 1e-10:
                continue
            a_new = a_new / s * sum_a  # keep same sum
            conv_new = np.convolve(a_new, a_new)
            obj_new = 2.0 * n * float(np.max(conv_new)) / (sum_a * sum_a)
            if obj_new < best_candidate_obj:
                best_candidate_obj = obj_new
                best_t = t
        
        if best_t > 0:
            a_new = (1-best_t) * a + best_t * g
            a_new = np.clip(a_new, 0, None)
            s = float(np.sum(a_new))
            if s > 1e-10:
                a = a_new / s * sum_a
            else:
                a = a_new
        
        conv = np.convolve(a, a)
        obj = 2.0 * n * float(np.max(conv)) / (sum_a * sum_a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj

def coord_descent(a, max_time, n_passes=50):
    """Simple cyclic coordinate descent."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for _ in range(n_passes):
        if time.monotonic() - start > max_time:
            break
        for j in range(n):
            if time.monotonic() - start > max_time:
                break
            best_t = a[j]
            best_obj_local = best_obj
            for scale in [0.5, 0.7, 1.3, 2.0]:
                a_test = a.copy()
                a_test[j] = a[j] * scale
                obj_test = compute_obj(a_test)
                if obj_test < best_obj_local:
                    best_obj_local = obj_test
                    best_t = a[j] * scale
            a[j] = best_t
            if best_obj_local < best_obj:
                best_obj = best_obj_local
                best_a = a.copy()
    
    return best_a, best_obj

def structured_init(n, rng):
    """Generate structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        # Edge-concentrated
        a = np.zeros(n)
        w = max(1, n // 10)
        a[:w] = 1.0
        a[-w:] = 1.0
        a += 0.05 * rng.random(n)
    elif choice == 1:
        # Gaussian centered
        center = n / 2.0
        width = n * rng.uniform(0.15, 0.4)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width)**2)
    elif choice == 2:
        # Off-center Gaussian
        center = n * rng.uniform(0.25, 0.75)
        width = n * rng.uniform(0.1, 0.3)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width)**2)
    elif choice == 3:
        # Two bumps
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * 0.1
        a = np.exp(-0.5*((np.arange(n)-c1)/w)**2) + np.exp(-0.5*((np.arange(n)-c2)/w)**2)
    elif choice == 4:
        # Random periodic
        period = int(rng.integers(max(2, n//20), max(3, n//5)))
        pattern = np.maximum(0, rng.normal(0.5, 0.3, period))
        a = np.tile(pattern, n//period + 1)[:n]
    elif choice == 5:
        # Cosine window
        a = np.maximum(0, np.cos(np.pi * np.arange(n) / n))
    elif choice == 6:
        # Linear ramp
        a = np.linspace(0.1, 1.0, n)
    else:
        # Random positive
        a = np.abs(rng.normal(1.0, 0.5, n))
    
    a = np.maximum(0, a)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

            break

            eta = min(3.0, eta * 1.005)
        else:
            eta = max(0.05, eta * 0.9)
    
    return best_a, best_obj

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def fft_cross_corr(p, a, N):
    """Compute g_j = sum_k p[k] * a[k-j] for j=0..n-1 using FFT."""
    n = len(a)
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(P * np.conj(A), N)
    return corr[:n]

def coord_descent(a, max_time, n_scales=4):
    """Cyclic coordinate descent with multiplicative perturbations."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    scales = [0.5, 0.75, 1.25, 1.5, 2.0]
    
    while time.monotonic() - start < max_time:
        improved = False
        for j in range(n):
            if time.monotonic() - start > max_time:
                return best_a, best_obj
            orig = a[j]
            best_obj_local = best_obj
            best_val = orig
            for sc in scales:
                a[j] = orig * sc
                obj_local = compute_obj(a)
                if obj_local < best_obj_local:
                    best_obj_local = obj_local
                    best_val = orig * sc
            a[j] = best_val
            if best_obj_local < best_obj:
                best_obj = best_obj_local
                best_a = a.copy()
                improved = True
        if not improved:
            break
    
    return best_a, best_obj
