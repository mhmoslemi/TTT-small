import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_conv_matrix(a, n):
    """Build the matrix A such that (A @ g)_k = (a * g)_k for all k."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def solve_lp(a, M):
    """Solve the LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def softmax_grad(a, T):
    """Compute gradient of softmax approximation of max(c_k)."""
    n = len(a)
    conv = np.convolve(a, a)
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    
    # grad_i = sum_k p_k * a_{k-i} (for valid indices)
    grad = np.zeros(n)
    for k in range(len(conv)):
        if p[k] < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            grad[j] += p[k] * a[k - j]
    return grad

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

"""
Optimized step function coefficient search using LP-based alternating with adaptive
tightening, projected gradient with backtracking on softmax approximation,
diverse structured initializations, and convex combination polishing.
Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp(a, M, method='highs'):
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method=method)
    if res.status == 0:
        return res.x, float(-res.fun)
    return None, None


def softmax_obj_and_grad(a, T):
    n = len(a)
    conv = np.convolve(a, a)
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    sm_val = max_c + T * np.log(sum_exp)
    grad = np.correlate(p, a, mode='valid')
    return sm_val, grad, p, conv


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    s = np.sum(v)
    if s <= 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidate = np.arange(1, n + 1)
    mask = u * rho_candidate > (cssv - 1)
    if not np.any(mask):
        return np.ones(n) / n
    rho = int(np.nonzero(mask)[0][-1])
    theta = (cssv[rho - 1] - 1.0) / rho
    w = np.maximum(v - theta, 0.0)
    s2 = np.sum(w)
    if s2 < 1e-15:
        return np.ones(n) / n
    return w / s2


def lp_phase(a, max_iters, max_time, t_start=0.5, tighten=False):
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    tighten_factor = 1.0

    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break

        conv = np.convolve(a, a)
        M = float(np.max(conv))

        M_lp = M * tighten_factor
        g, g_sum = solve_lp(a, M_lp)
        if g is None:
            if tighten_factor < 1.0:
                tighten_factor = min(1.0, tighten_factor * 1.01)
                continue
            break

        if g_sum <= 1e-10:
            break

        improvement = g_sum - 1.0
        if improvement < 1e-6:
            stall += 1
            if tighten_factor > 0.95:
                tighten_factor = max(0.95, tighten_factor - 0.01)
            if stall > 20:
                break
            continue
        stall = 0

        g_norm = g / g_sum
        best_t = 0.0
        best_obj_t = best_obj

        for t in [t_start, t_start * 0.5, t_start * 0.25, t_start * 0.1, t_start * 0.05]:
            a_new = (1.0 - t) * a + t * g_norm
            a_new = np.clip(a_new, 0.0, None)
            s = float(np.sum(a_new))
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj_t:
                best_obj_t = obj
                best_t = t

        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * g_norm
            a_new = np.clip(a_new, 0.0, None)
            a_new = a_new / float(np.sum(a_new))
            a = a_new
            if best_obj_t < best_obj:
                best_a = a.copy()
                best_obj = best_obj_t
                tighten_factor = 1.0
        else:
            if tighten_factor > 0.95:
                tighten_factor = max(0.95, tighten_factor - 0.02)
            else:
                stall += 1
                if stall > 10:
                    break

    return best_a, best_obj


def gd_phase(a, max_time, T_init=None):
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if T_init is None:
        conv = np.convolve(a, a)
        T_init = 0.01 * float(np.max(conv))

    T = T_init
    lr = 0.5
    patience = 0
    best_lr = lr

    for it in range(100000):
        if time.monotonic() - start > max_time:
            break

        sm_val, grad, p, conv = softmax_obj_and_grad(a, T)
        max_c = float(np.max(conv))
        T = max(0.0005 * max_c, T * 0.997)

        grad_mean = np.mean(grad)
        grad_centered = grad - grad_mean

        a_new = a - lr * grad_centered
        a_new = project_simplex(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            patience = 0
            best_lr = lr
            lr = min(2.0, lr * 1.05)
        else:
            patience += 1
            lr *= 0.8
            if patience > 50:
                lr = best_lr * 0.1
                patience = 0
            if lr < 1e-12:
                break

    return best_a, best_obj


def diverse_init(n, rng, variant=0):
    if variant == 0:
        a = np.ones(n) / n
        a += 0.1 * rng.normal(0, 1, n)
        a = np.clip(a, 0, None)
    elif variant == 1:
        freq = rng.uniform(0.03, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0.01, np.cos(2 * np.pi * freq * np.arange(n) + phase))
    elif variant == 2:
        center = n / 2.0
        width = n * rng.uniform(0.15, 0.5)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        a += 0.05
    elif variant == 3:
        a = np.maximum(0, rng.normal(0.5, 0.35, n))
    elif variant == 4:
        period = max(2, int(n * rng.uniform(0.02, 0.15)))
        pattern = np.maximum(0.01, rng.random(period) * 0.5 + 0.5)
        a = np.tile(pattern, n // period + 1)[:n]
    elif variant == 5:
        a = np.ones(n)
        a[0] = rng.uniform(0.5, 2.0)
        a[-1] = rng.uniform(0.5, 2.0)
        a *= np.exp(rng.normal(0, 0.2, n))
    elif variant == 6:
        a = np.maximum(0, 1.0 + rng.uniform(-0.8, 0.8) * np.sin(np.pi * np.arange(n) / n))
    elif variant == 7:
        a = 1.0 / (1.0 + np.exp(-rng.uniform(-3, 3) * (np.arange(n) / n - 0.5)))
    else:
        a = np.maximum(0.01, rng.power(2.0, n) * rng.uniform(0.5, 2.0))

    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def full_pipeline(a_init, time_budget):
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    lp_time = min(25.0, time_budget * 0.35)
    max_iter = max(100, int(lp_time / 0.01))
    a, obj = lp_phase(a, max_iter, lp_time, t_start=0.6, tighten=True)
    if obj < best_obj:
        best_a = a.copy()
        best_obj = obj

    if time.monotonic() - start < time_budget * 0.65:
        gd_time = min(15.0, time_budget * 0.2)
        a, obj = gd_phase(a, gd_time)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj

    if time.monotonic() - start < time_budget * 0.82:
        lp_time2 = min(12.0, time_budget * 0.12)
        max_iter2 = max(50, int(lp_time2 / 0.01))
        a, obj = lp_phase(a, max_iter2, lp_time2, t_start=0.3, tighten=True)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj

    if time.monotonic() - start < time_budget * 0.95:
        gd_time2 = min(8.0, time_budget * 0.08)
        a, obj = gd_phase(a, gd_time2, T_init=0.003 * float(np.max(np.convolve(a, a))))
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj

    return best_a, best_obj


def convex_combo_improve(seq_list, time_budget, rng):
    if len(seq_list) < 2:
        return seq_list[0] if seq_list else None

    best_a = seq_list[0].copy()
    best_obj = compute_obj(best_a)
    start = time.monotonic()

    for _ in range(200):
        if time.monotonic() - start > time_budget:
            break
        i = int(rng.integers(0, len(seq_list)))
        j = int(rng.integers(0, len(seq_list)))
        if i == j:
            continue
        if len(seq_list[i]) != len(seq_list[j]):
            continue
        lam = rng.uniform(0.2, 0.8)
        a_combo = lam * seq_list[i] + (1 - lam) * seq_list[j]
        a_combo = np.clip(a_combo, 0, None)
        s = float(np.sum(a_combo))
        if s < 1e-10:
            continue
        a_combo = a_combo / s
        obj = compute_obj(a_combo)
        if obj < best_obj:
            best_a = a_combo.copy()
            best_obj = obj

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 8.0

    best_seq = None
    best_score = np.inf
    all_good_solutions = []

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all() and parent.size <= 600:
            p_time = min(80.0, float(budget_s) * 0.12)
            p_opt, p_obj = full_pipeline(parent.copy(), p_time)
            if p_obj < best_score:
                best_score = p_obj
                best_seq = p_opt.copy()
                print(f"Parent optimized: obj={p_obj:.6f}")
            if p_obj < 1.52:
                all_good_solutions.append(p_opt.copy())
    except Exception as e:
        print(f"Parent opt error: {e}")

    n_values = [40, 60, 80, 100, 120, 150, 180, 200, 250, 300, 400, 500]

    for n in n_values:
        if time.monotonic() > deadline - 35:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        num_remaining = len([x for x in n_values if x >= n])
        time_per_n = remaining / max(1, num_remaining)
        time_per_n = min(time_per_n, 100.0)

        num_restarts = max(2, min(4, int(time_per_n / 22)))
        time_per_restart = (time_per_n - 2.0) / num_restarts

        print(f"\nN={n}: remaining={remaining:.0f}s, {num_restarts} restarts, {time_per_restart:.1f}s each")

        for r in range(num_restarts):
            if time.monotonic() > deadline - 35:
                break
            variant = int(rng.integers(0, 8))
            a_init = diverse_init(n, rng, variant)
            a_opt, obj_opt = full_pipeline(a_init, time_per_restart)

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} r={r} v={variant}: obj={obj_opt:.6f} ** NEW BEST **")
            else:
                print(f"  n={n} r={r} v={variant}: obj={obj_opt:.6f}")

            if obj_opt < 1.52 and len(all_good_solutions) < 10:
                all_good_solutions.append(a_opt.copy())

    if best_seq is not None and time.monotonic() < deadline - 12:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: {remaining:.0f}s")

        a_ref, obj_ref = full_pipeline(best_seq.copy(), min(remaining * 0.5, 50.0))
        if obj_ref < best_score:
            best_score = obj_ref
            best_seq = a_ref.copy()
            print(f"  After refinement: obj={best_score:.6f}")

        if len(all_good_solutions) >= 2 and time.monotonic() < deadline - 5:
            combo_best = convex_combo_improve(all_good_solutions, deadline - time.monotonic() - 3.0, rng)
            if combo_best is not None:
                combo_obj = compute_obj(combo_best)
                if combo_obj < best_score:
                    best_score = combo_obj
                    best_seq = combo_best.copy()
                    print(f"  After combo: obj={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
