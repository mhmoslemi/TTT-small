def compute_obj(a):

    if n == 0:

    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

        a[i] = x_current  # restore

    return best_x

"""

with convolution tracking, peak-targeted refinement, multi-scale initialization,

import numpy as np
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def update_conv(conv, a, i, old_val, new_val):
    """Update convolution when a[i] changes from old_val to new_val. In-place."""
    n = len(a)
    if abs(new_val - old_val) < 1e-15:
        return
    delta = new_val - old_val
    # Affected k range: k-i in [0, n-1], so k in [i, i+n-1]
    k_lo = max(i, 0)
    k_hi = min(i + n - 1, 2 * n - 2)
    # a[k-i] for k in [k_lo, k_hi]
    idx_lo = k_lo - i
    idx_hi = k_hi - i
    # Linear update
    conv[k_lo:k_hi + 1] += 2.0 * delta * a[idx_lo:idx_hi + 1]
    # Fix k=2i (should be quadratic, not linear)
    if 2 * i >= k_lo and 2 * i <= k_hi:
        conv[2 * i] -= 2.0 * delta * a[i]
        conv[2 * i] += new_val * new_val - old_val * old_val


def coordinate_descent(a, max_time, rng, use_annealing=True):
    """Vectorized coordinate descent with conv tracking."""
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = np.convolve(a, a).astype(np.float64).copy()
    current_sum = float(a.sum())
    current_max = float(conv.max())
    best_obj = 2.0 * n * current_max / (current_sum * current_sum)
    best_a = a.copy()
    start = time.monotonic()
    temp = 0.1
    no_improve = 0

    while time.monotonic() - start < max_time:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start > max_time:
                break
            old_val = a[i]
            S = current_sum - old_val
            if S <= 1e-10:
                continue

            # Generate candidates
            candidates = [0.0, old_val]
            # Golden ratio scaled
            for mult in [0.3, 0.6, 0.8, 1.2, 1.5, 2.0, 3.0]:
                candidates.append(old_val * mult)
            # Critical point for quadratic (k=2i)
            if 2 * i < 2 * n - 1 and S > 1e-10:
                # We need conv at k=2i with a[i]=0
                # conv_base[2i] = current conv[2i] - old_val^2 (since at k=2i, contribution is a[i]^2)
                # Actually: conv[2i] includes old_val^2 from the a[i]*a[i] term
                # So conv_base[2i] = conv[2i] - old_val^2
                conv_base_2i = conv[2 * i] - old_val * old_val
                x_q = conv_base_2i / S
                if 0 < x_q < 10 * current_sum / n:
                    candidates.append(x_q)

            # For linear terms near the max: try to equalize
            max_k = int(np.argmax(conv))
            diff = max_k - i
            if 0 <= diff < n and diff != i and a[diff] > 1e-12:
                # conv[max_k] = base + 2*old_val*a[diff], want to reduce
                # x that makes conv[max_k] = second_max
                second_max = float(np.partition(conv, -2)[-2])
                target = max(second_max, 0)
                base_mk = conv[max_k] - 2.0 * old_val * a[diff]
                x_target = (target - base_mk) / (2.0 * a[diff])
                if 0 < x_target < 5 * current_sum / n:
                    candidates.append(x_target)

            best_local_obj = best_obj
            best_x = old_val
            best_local_conv = None

            for x in candidates:
                x = max(0.0, float(x))
                if x == old_val:
                    obj = best_obj
                else:
                    # Save conv
                    conv_save = conv.copy()
                    # Apply change
                    delta = x - old_val
                    k_lo = max(i, 0)
                    k_hi = min(i + n - 1, 2 * n - 2)
                    idx_lo = k_lo - i
                    idx_hi = k_hi - i
                    conv[k_lo:k_hi + 1] += 2.0 * delta * a[idx_lo:idx_hi + 1]
                    if 2 * i >= k_lo and 2 * i <= k_hi:
                        conv[2 * i] -= 2.0 * delta * a[i]
                        conv[2 * i] += x * x - old_val * old_val

                    new_sum = current_sum + delta
                    if new_sum > 0.01:
                        new_max = float(conv.max())
                        obj = 2.0 * n * new_max / (new_sum * new_sum)
                    else:
                        obj = np.inf

                    # Restore
                    conv[:] = conv_save

                if obj < best_local_obj - 1e-12:
                    best_local_obj = obj
                    best_x = x

            # Apply best
            if best_x != old_val:
                conv_save = conv.copy()
                update_conv(conv, a, i, old_val, best_x)
                a[i] = best_x
                current_sum += best_x - old_val
                current_max = float(conv.max())
                new_obj = 2.0 * n * current_max / (current_sum * current_sum)

                if new_obj < best_obj + 1e-12:
                    if new_obj < best_obj - 1e-12:
                        best_obj = new_obj
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                else:
                    # Revert
                    a[i] = old_val
                    conv[:] = conv_save
                    current_sum -= (best_x - old_val)
                    current_max = float(conv.max())
                    no_improve += 1

        if not improved:
            temp *= 0.9
            no_improve += 1
            if no_improve > 30 and temp < 0.001:
                break
        else:
            no_improve = 0
            temp = min(0.1, temp * 1.02)

    return best_a, best_obj


def peak_targeted_refine(a, max_time, rng):
    """Target the peak lag(s) directly."""
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = np.convolve(a, a).astype(np.float64).copy()
    current_sum = float(a.sum())
    best_obj = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
    best_a = a.copy()
    start = time.monotonic()

    iteration = 0
    while time.monotonic() - start < max_time and iteration < 10000:
        iteration += 1
        current_max = float(conv.max())
        # Find all peaks (within 5% of max)
        threshold = current_max * 0.95
        peak_lags = np.where(conv >= threshold)[0]

        for k in peak_lags:
            if time.monotonic() - start > max_time:
                break
            # conv[k] = sum_j a[j]*a[k-j] for valid j
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start > j_end:
                continue
            #-contributors = a[j_start:j_end+1] * a[k-j_end:k-j_start+1][::-1]
            # Find the pair with largest product
            pairs = a[j_start:j_end + 1] * a[k - j_end:k - j_start + 1][::-1]
            # Find the largest contributor
            max_idx = int(np.argmax(pairs))
            j1 = j_start + max_idx
            j2 = k - j1

            # Try reducing the larger of a[j1], a[j2]
            if j1 == j2:
                # Quadratic term: reduce a[j1]
                old_val = a[j1]
                # Target: make this pair's contribution = second largest
                target = current_max * 0.9
                new_val = np.sqrt(max(0.0, target))
                for frac in [0.3, 0.5, 0.7, new_val / old_val if old_val > 0 else 0]:
                    x = old_val * frac if old_val > 0 else 0
                    x = max(0.0, min(x, old_val))
                    if x == old_val:
                        continue
                    conv_save = conv.copy()
                    update_conv(conv, a, j1, old_val, x)
                    a[j1] = x
                    ns = current_sum + x - old_val
                    if ns > 0.01:
                        nm = float(conv.max())
                        obj = 2.0 * n * nm / (ns * ns)
                        if obj < best_obj:
                            best_obj = obj
                            best_a = a.copy()
                            current_sum = ns
                            continue
                    # Revert
                    a[j1] = old_val
                    conv[:] = conv_save
            else:
                # Reduce the larger element
                if a[j1] >= a[j2]:
                    target_idx = j1
                else:
                    target_idx = j2
                old_val = a[target_idx]
                if old_val < 1e-12:
                    continue
                # Try reducing
                target_mult = current_max * 0.9
                other = a[k - target_idx]
                if other > 1e-12:
                    x_needed = target_mult / (2.0 * other)
                else:
                    x_needed = 0.0
                for x in [0.0, old_val * 0.3, old_val * 0.5, old_val * 0.7, x_needed]:
                    x = max(0.0, min(float(x), old_val))
                    if x == old_val:
                        continue
                    conv_save = conv.copy()
                    update_conv(conv, a, target_idx, old_val, x)
                    a[target_idx] = x
                    ns = current_sum + x - old_val
                    if ns > 0.01:
                        nm = float(conv.max())
                        obj = 2.0 * n * nm / (ns * ns)
                        if obj < best_obj:
                            best_obj = obj
                            best_a = a.copy()
                            current_sum = ns
                            continue
                    # Revert
                    a[target_idx] = old_val
                    conv[:] = conv_save

    return best_a, best_obj


def init_two_bump(n, rng):
    """Two bumps at 1/4 and 3/4 positions."""
    center1 = n * 0.25
    center2 = n * 0.75
    width = n * rng.uniform(0.1, 0.25)
    a = np.exp(-0.5 * ((np.arange(n) - center1) / width) ** 2)
    a += np.exp(-0.5 * ((np.arange(n) - center2) / width) ** 2)
    a += 0.05 * np.ones(n)
    s = a.sum()
    return a / s


def init_u_shaped(n, rng):
    """U-shaped: high at edges, low in middle."""
    x = np.linspace(0, 1, n)
    a = (2 * x - 1) ** 2 + 0.1
    noise = rng.normal(1.0, 0.1, n)
    a = a * np.maximum(0, noise)
    s = a.sum()
    return a / s


def init_cosine_mod(n, rng):
    """Cosine-modulated sequence."""
    freq = rng.uniform(0.5, 3.0)
    phase = rng.uniform(0, 2 * np.pi)
    a = np.maximum(0, 0.5 + 0.5 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
    a += 0.1 * np.ones(n)
    s = a.sum()
    return a / s


def init_random_smooth(n, rng):
    """Smooth random sequence."""
    # Generate in frequency domain
    freq = np.fft.rfft(np.ones(n))
    mask = rng.random(len(freq))
    freq *= mask
    a = np.fft.irfft(freq, n=n)
    a = np.maximum(0, a)
    a += 0.1 * np.ones(n)
    s = a.sum()
    return a / s


def init_parent(n, parent):
    """Resample parent to length n."""
    if len(parent) == 0:
        return np.ones(n) / n
    xp = np.linspace(0, 1, len(parent))
    x = np.linspace(0, 1, n)
    a = np.interp(x, xp, parent)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_structured(n, rng, parent=None):

    elif choice == 5:

    else:

    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: Coordinate descent
    cd_time1 = min(time_budget * 0.5, 40.0)
    a, obj = coordinate_descent(a, cd_time1, rng)
    if obj < best_obj:
        best_a = a.copy()
        best_obj = obj

    if time.monotonic() - start < time_budget * 0.8:
        pt_time = min(time_budget * 0.2, 20.0)
        a, obj = peak_targeted_refine(a, pt_time, rng)

            best_obj = obj

    # Phase 3: More coordinate descent

        remaining = time_budget - (time.monotonic() - start)
        a, obj = coordinate_descent(a, remaining * 0.5, rng)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj

    return best_a, best_obj


def block_perturbation_refine(a, rng, max_time):
    """Perturb blocks of elements for fine-tuning."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    sigma = 0.05

    while time.monotonic() - start < max_time and it < 100000:
        it += 1
        width = max(1, int(n * rng.uniform(0.005, 0.08)))
        pos = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.maximum(0, candidate)
        s = candidate.sum()
        if s < 0.01:

        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj
            sigma = min(0.1, sigma * 1.05)
        else:
            sigma = max(0.005, sigma * 0.98)

        except Exception:
            pass

    # Main search
    n_values = [100, 120, 150, 180, 200, 250, 300, 350, 400]
    # Sort by expected benefit (medium sizes tend to work well)
    n_values = [200, 250, 150, 300, 180, 400, 100, 120, 350]

    for n in n_values:
        if time.monotonic() > deadline - 50:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        num_remaining = len([x for x in n_values if x >= n and n_values.index(x) >= n_values.index(n)])
        time_per_n = remaining / max(1, num_remaining)
        time_per_n = min(time_per_n, 120.0)

        num_restarts = max(2, min(4, int(time_per_n / 30)))
        time_per_restart = time_per_n / num_restarts

        best_for_n = np.inf
        print(f"\nN={n}: remaining={remaining:.0f}s, {num_restarts} restarts")

        for r in range(num_restarts):
            if time.monotonic() > deadline - 50:
                break
            a_init = init_structured(n, rng, parent)
            a_opt, obj_opt = full_pipeline(a_init, time_per_restart, rng)

            if obj_opt < best_for_n:
                best_for_n = obj_opt
                if obj_opt < best_score:
                    best_score = obj_opt
                    best_seq = a_opt.copy()
                    print(f"  n={n} r={r}: obj={obj_opt:.6f} ** NEW BEST **")
                else:
                    print(f"  n={n} r={r}: obj={obj_opt:.6f}")
            else:
                print(f"  n={n} r={r}: obj={obj_opt:.6f}")

    # Final refinement on best found
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: {remaining:.0f}s left")

        # More coordinate descent
        cd_time = remaining * 0.4
        a_opt, obj_opt = coordinate_descent(best_seq, cd_time, rng)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            print(f"  After final CD: obj={best_score:.6f}")

        # Peak targeting
        if time.monotonic() < deadline - 10:
            pt_time = min(remaining * 0.25, 60.0)
            a_opt, obj_opt = peak_targeted_refine(best_seq, pt_time, rng)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  After peak target: obj={best_score:.6f}")

        # Block perturbation
        if time.monotonic() < deadline - 5:
            bp_time = deadline - time.monotonic() - 3.0
            a_opt, obj_opt = block_perturbation_refine(best_seq, rng, bp_time)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  After block pert: obj={best_score:.6f}")

        # One more CD pass
        if time.monotonic() < deadline - 3:
            a_opt, obj_opt = coordinate_descent(best_seq, 5.0, rng)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  After last CD: obj={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    # Ensure non-negative and finite
    best_seq = np.clip(np.asarray(best_seq, dtype=np.float64), 0.0, 1000.0)
    best_seq = best_seq[np.isfinite(best_seq)]

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
