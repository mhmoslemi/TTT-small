"""
Minimax LP + multi-peak pressure + coordinate descent optimization for
minimum-peak self-convolution of non-negative sequences.
Key improvement: direct minimax LP formulation, asymmetric initialization,
top-K peak pressure, and adaptive step sizes.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute score for normalized sequence (sum=1): 2*n*max(conv)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def build_conv_matrix(a):
    """Build matrix A such that A @ b = np.convolve(a, b, mode='full')."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A[k, j] = a[k - j]
    return A


def minimax_lp_direction(a, time_limit=5.0):
    """
    Solve LP: minimize t subject to (a*b)[k] <= t for all k, sum(b)=1, b>=0.
    Returns (b, t_opt) or (None, None) on failure.
    """
    n = len(a)
    A = build_conv_matrix(a)
    # Variables: b[0..n-1], t  (total n+1 variables)
    # Objective: minimize t -> c = [0,...,0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Inequality constraints: A @ b - t <= 0 for each row
    # Build: [A, -I_rows] @ [b, t] <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    idx = np.arange(2 * n - 1)
    A_ub[idx, idx] = -1.0
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            b = np.clip(res.x[:n], 0, None)
            t_val = res.x[-1]
            bs = b.sum()
            if bs < 1e-12:
                return None, None
            b = b / bs
            return b, t_val
    except Exception:
        pass
    return None, None


def maxsum_lp_direction(a, M, time_limit=5.0):
    """
    Solve LP: maximize sum(b) subject to (a*b)[k] <= M for all k, b >= 0.
    Returns b (normalized) or None.
    """
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'preserve_sectivity': True,
                      'time_limit': time_limit})
        if res.success and res.x is not None:
            b = np.clip(res.x, 0, None)
            bs = b.sum()
            if bs < 1e-12:
                return None
            return b / bs
    except Exception:
        pass
    return None


def multi_peak_pressure_update(f, n_iter=300, eta=0.15, top_k=5, rng=None):
    """Pressure-based update considering top-K peaks."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()

        # Find top-K peak positions
        k = min(top_k, len(conv))
        top_idx = np.argsort(conv)[::-1][:k]
        top_vals = conv[top_idx]
        top_max = top_vals[0]
        if top_max < 1e-15:
            break

        # Compute weighted pressure
        # For each position j in sequence, pressure = sum over peaks of
        # the contribution to that peak from f[j]
        pressure = np.zeros(n)
        for pk in range(k):
            j_star = top_idx[pk]
            weight = top_vals[pk] / top_max  # weight by peak height
            # Contribution of f[i] to conv[j_star] is f[j_star - i] if valid
            for i in range(n):
                idx = j_star - i
                if 0 <= idx < n:
                    pressure[i] += weight * f[idx]

        if pressure.max() > 0:
            pressure = pressure / pressure.max()

        # Multiplicative update with decay
        eta_eff = eta * (1.0 / (1.0 + 0.05 * it))
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 150:
                # Perturb
                f_pert = f * (1.0 + 0.2 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    stall = 0
                else:
                    break

    return best_f, best_score


def lp_optimize(f, time_budget, t0=0.5, rng=None, use_minimax=True):
    """
    LP-based optimization loop using minimax or maxsum LP.
    """
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)

    best_f = f.copy()
    best_score = compute_score(f)
    t_step = t0
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 8:
            break

        conv = np.convolve(f, f)
        M = conv.max()
        lp_limit = min(4.0, remaining * 0.2)

        # Get LP direction
        if use_minimax:
            g, t_opt = minimax_lp_direction(f, time_limit=lp_limit)
        else:
            g = maxsum_lp_direction(f, M, time_limit=lp_limit)
            t_opt = None

        if g is None:
            break

        # Try multiple step sizes
        improved = False
        steps = [min(1.0, t_step * 1.5), t_step, t_step * 0.5, t_step * 0.25, t_step * 0.1]
        for alpha in steps:
            f_new = (1.0 - alpha) * f + alpha * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_score = compute_score(f_new)
            if new_score < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                t_step = min(1.0, t_step * 1.1)
                improved = True
                break

        if not improved:
            t_step *= 0.6
            if t_step < 0.01:
                # Perturb to escape
                f_pert = f + rng.normal(0, 0.03, n)
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    t_step = 0.3
                else:
                    break

    return best_f, best_score


def coordinate_descent(f, n_sweeps=8, rng=None):
    """Coordinate descent polishing with geometric search."""
    if rng is None:
        rng = np.random.default_rng(0)

    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            orig = f[i]
            if orig < 1e-10:
                continue
            best_val = orig
            best_local = best_score

            # Geometric spread
            for factor in [0.3, 0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.0/0.5, 1.0/0.3, 2.0, 0.1]:
                f[i] = max(0.0, orig * factor)
                s = f.sum()
                if s < 1e-10:
                    f[i] = orig
                    continue
                conv = np.convolve(f, f)
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]

            f[i] = best_val
            if best_local < best_score:
                best_score = best_local

        f = np.clip(f, 0, None)
        f = normalize(f)

    return f, best_score


def simulated_annealing(f, rng, n_iter=5000, T0=0.04, T_end=0.0001):
    """Simulated annealing with diverse moves."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    cur_score = best_score
    no_improve = 0

    for it in range(n_iter):
        T = T0 * (T_end / T0) ** (it / n_iter)
        f_new = f.copy()
        r = rng.random()

        if r < 0.25:
            # Block multiplicative
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.5))
        elif r < 0.45:
            # Single element
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] * np.exp(rng.normal(0, 0.4)))
        elif r < 0.6:
            # Two-element swap-like
            i, j = rng.integers(0, n, 2)
            if i != j:
                f_new[i], f_new[j] = f_new[j], f_new[i]
        elif r < 0.8:
            # Gaussian kernel perturbation
            center = int(rng.integers(0, n))
            width = max(2, int(n // 24))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
            f_new += rng.normal(0, 0.15) * kernel
        else:
            # Large multiplicative
            f_new *= np.exp(rng.normal(0, 0.35, n))

        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score(f_new)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 3000 and it > n_iter // 2:
            break

    return best_f, best_score


def init_tilted(n, tilt, rng=None):
    """Create a tilted (asymmetric) initial sequence."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.linspace(0, 1, n)
    if tilt == 0:
        f = np.ones(n)
    elif tilt == 1:
        f = 1.0 + tilt * x
    elif tilt == 2:
        f = np.exp(tilt * x)
    elif tilt == 3:
        f = 1.0 + tilt * x * (1 - x) * 4  # bump in middle
    elif tilt == 4:
        # Two peaks at edges
        f = np.exp(-3.0 * x) + np.exp(-3.0 * (1 - x)) + 0.3
    elif tilt == 5:
        # One-sided exponential
        f = np.exp(-2.0 * (1 - x))
    else:
        f = 1.0 + tilt * x
    f = np.clip(f, 0.01, None)
    f = normalize(f)
    # Small noise to break symmetry
    f = f * (1 + 0.03 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f


def init_from_parent(parent, rng):
    """Perturb parent sequence."""
    f = parent.copy()
    f = normalize(f)
    n = len(f)
    f = f * (1 + 0.1 * rng.normal(0, 1, n))
    f = np.clip(f, 0, None)
    f = normalize(f)
    return f


def spectral_refine(f, n_iter=200, rng=None):
    """Spectral flattening: project toward flat |FT|^2."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()

    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)
        n_bins = len(mag)

        # Target: make |F|^2 proportional to (1/mag) to flatten the convolution
        # Actually, we want conv = IDFT(|F|^2) to be flat
        # So we want |F|^2 to be as flat as possible -> mag constant
        # But DC is fixed at sum(f) = 1
        current_energy = np.sum(f ** 2)
        target_mag = np.sqrt(current_energy * n_fft / (n_bins - 1))

        mag_new = np.full_like(mag, target_mag)
        mag_new[0] = 1.0

        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s

        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break

    return best_f, best_score
