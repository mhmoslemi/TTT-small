"""
Optimization of non-negative step function to minimize 2n*max(conv(f,f))/sum(f)^2.
Uses adaptive soft-max gradient with backtracking line search, LP direction finding,
structured perturbation restarts, and multi-length exploration.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f, g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    """Solve: max sum(g) s.t. g >= 0, convolve(f, g)[k] <= M for all k."""
    n = len(f)
    if A_cache is None:
        A = build_conv_matrix_sparse(f)
    else:
        A = A_cache
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def compute_softmax_gradient(f, beta):
    """Compute gradient and soft-max value of the smoothed max of conv(f,f)."""
    n = len(f)
    C = np.convolve(f, f)
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros(n), 0.0
    p = exp_C / total
    grad = 2.0 * np.correlate(p, f, mode='valid')
    soft_val = max_C + np.log(total) / beta
    return grad, soft_val


def soft_max_value(f, beta):
    """Compute the soft-max approximation of max(conv(f,f))."""
    C = np.convolve(f, f)
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    return max_C + np.log(exp_C.sum()) / beta


def tukey_window(n, r=0.3):
    """Generate a Tukey (tapered cosine) window."""
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def hann_window(n):
    """Generate a Hann window."""
    n = max(n, 2)
    w = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    return w


def psf_window(n, alpha=2.0):
    """Generate a super-Gaussian (PSF-like) window."""
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    w = np.exp(-((idx - center) / (n / 2.0)) ** (2 * alpha))
    return w


def asymmetric_window(n, skew=0.3):
    """Generate an asymmetric window that breaks left-right symmetry."""
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0 * (1 - skew)
    width = n / 2.5
    w = np.exp(-((idx - center) / width) ** 2)
    return w


def random_smooth_init(n, seed, sigma=0.15):
    """Generate a random smooth initialization."""
    rng = np.random.default_rng(seed)
    x = rng.standard_normal(n)
    kernel_len = max(3, int(n * 0.1))
    kernel = np.exp(-0.5 * (np.arange(-kernel_len, kernel_len + 1) / (kernel_len / 3.0)) ** 2)
    smooth_x = np.convolve(x, kernel, mode='same')
    smooth_x = np.abs(smooth_x) + 0.1
    return smooth_x


def perturb_structured(f, rng, grad, strength=0.05, grad_weight=0.1):
    """Structured perturbation: mix random noise with gradient direction."""
    n = len(f)
    noise = rng.standard_normal(n) * strength
    grad_dir = grad / (np.linalg.norm(grad) + 1e-10) * strength * grad_weight
    new_f = f * np.clip(1.0 + noise, 0.1, 2.0)
    new_f = new_f + grad_dir
    new_f = np.clip(new_f, 0.0, 1000.0)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S


def softmax_adam_optimize(f_init, deadline, beta_start=10.0, beta_end=500.0,
                          lr_start=0.04, max_iters=5000, verbose=""):
    """
    Optimize using soft-max gradient with Adam and backtracking line search.
    Uses adaptive beta that increases over iterations.
    """
    f = f_init.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f_init.tolist())

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start
    it = 0
    no_improve = 0
    best_soft = soft_max_value(f, beta_start)

    while time.monotonic() < deadline and it < max_iters:
        # Adaptive beta: increase as we iterate
        it_frac = min(1.0, it / (max_iters * 0.6))
        beta = beta_start * (beta_end / beta_start) ** it_frac

        grad, soft_val = compute_softmax_gradient(f, beta)

        # Check if we're at a point where beta is high enough to be accurate
        if beta > 100:
            exact_max = np.max(np.convolve(f, f))
            if abs(soft_val - exact_max) / (exact_max + 1e-10) < 0.01:
                # Use exact subgradient for more accurate direction
                C = np.convolve(f, f)
                k_max = np.argmax(C)
                subgrad = np.zeros(n)
                for j in range(n):
                    if 0 <= k_max - j <= 2 * n - 2:
                        subgrad[j] = 2.0 * f[k_max - j]
                # Blend soft-max gradient with subgradient
                grad = 0.5 * grad + 0.5 * subgrad

        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        update = m_hat / (np.sqrt(v_hat) + eps)

        new_f = None
        step = lr
        base_soft = soft_val
        base_grad_dot = np.dot(grad, f)

                break

            if step < 1e-8:
                break

            f = new_f
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            if it % 500 == 0:
                print(f"    {verbose}SM iter {it}: score={best_score:.6f} (beta={beta:.1f})")

            no_improve += 1
            if no_improve > 150:

                no_improve = 0

    if sum_g <= 1e-15:
        return f, evaluate_sequence(f.tolist())
    g = g * (S / sum_g)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

        sc = evaluate_sequence(new_f.tolist())
        if np.isfinite(sc) and sc < best_score:

    return best_f, best_score

            if bs < best_score:
                print(f"    {verbose}B: LP improved to {best_score:.6f}")
        else:
            # Try smaller step
            bf, bs = lp_step(best_f, A_cache, step_size=0.03)
            if bs < best_score:
                best_f, best_score = bf, bs
                A_cache = build_conv_matrix_sparse(best_f)
            else:
                break

    # Phase C: Final soft-max polish (20% of time)
    if time.monotonic() < deadline - 2:
        f_c, s_c = softmax_adam_optimize(best_f, deadline - 2,
                                         beta_start=100.0, beta_end=800.0,
                                         lr_start=0.01, max_iters=2000,
                                         verbose=verbose + "C")
        if s_c < best_score:
            best_f, best_score = f_c, s_c

    return best_f, best_score


def optimize_with_restarts(f_init, deadline, n_restarts=3, verbose=""):
    """Optimize with structured perturbation restarts."""
    best_f = f_init.copy().astype(np.float64)
    S = float(np.sum(best_f))
    best_f = best_f / S
    best_score = evaluate_sequence(best_f.tolist())

    rng = np.random.default_rng(42)
    n = len(best_f)

    # First, do a solid optimization
    per_phase_time = (deadline - time.monotonic()) / (n_restarts + 1)
    phase_deadline = time.monotonic() + per_phase_time

    bf, bs = optimize_single(best_f, phase_deadline, verbose=verbose + "main")
    if bs < best_score:
        best_f, best_score = bf, bs

    # Then, perturbation restarts
    for r in range(n_restarts):
        if time.monotonic() > deadline - 10:
            break
        remaining = deadline - time.monotonic()
        per_restart = remaining / (n_restarts - r) * 0.8

        strengths = [0.03, 0.02, 0.05]
        strength = strengths[r % len(strengths)]

        # Compute gradient for structured perturbation
        beta_cur = 50.0
        grad, _ = compute_softmax_gradient(best_f, beta_cur)
        f_pert = perturb_structured(best_f, rng, grad, strength=strength)

        restart_deadline = time.monotonic() + per_restart
        bf, bs = optimize_single(f_pert, restart_deadline, verbose=f"{verbose}r{r}")
        if bs < best_score:
            best_f, best_score = bf, bs
            print(f"    {verbose}Restart {r} (str={strength:.3f}): {bs:.6f}")
        else:
            print(f"    {verbose}Restart {r}: no improvement ({bs:.6f})")

    return best_f, best_score


def propose_candidate(budget_s=1500):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)

    best_sequence = None
    best_score = float('inf')
    best_n = 0
    rng = np.random.default_rng(123)

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                score = evaluate_sequence(parent.tolist())
                if np.isfinite(score):
                    best_score = score
                    best_sequence = parent.copy()
                    best_n = parent.size
                    print(f"Parent n={best_n}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    # Strategy:
    # 1. Quick exploration of multiple lengths (30% of time)
    # 2. Deep optimization on best length (55% of time)
    # 3. Final polish (15% of time)

    total_time = deadline - t0
    explore_deadline = t0 + total_time * 0.30
    deep_deadline = t0 + total_time * 0.85
    final_deadline = deadline

    # Phase 1: Quick exploration
    print(f"\n=== Phase 1: Quick exploration (until t={explore_deadline - t0:.0f}s) ===")

    lengths_to_try = [512, 640, 768, 384, 1024]
    if best_n in lengths_to_try:
        lengths_to_try.remove(best_n)
        lengths_to_try.insert(0, best_n)  # Prioritize parent length

    # For each length, try a quick optimization with 2 initializations
    for n in lengths_to_try:
        if time.monotonic() > explore_deadline - 10:
            break

        n_budget = (explore_deadline - time.monotonic()) / max(1, len(lengths_to_try) - lengths_to_try.index(n))
        n_deadline = time.monotonic() + n_budget

        print(f"\n  --- n={n} (budget {n_budget:.0f}s) ---")

        # Generate initializations
        inits = []

        if best_sequence is not None and len(best_sequence) == n:
            inits.append(('parent', best_sequence.copy()))
        elif best_sequence is not None:
            # Interpolate parent to this length
            parent_interp = np.interp(np.linspace(0, 1, n),
                                      np.linspace(0, 1, len(best_sequence)),
                                      best_sequence)
            parent_interp = np.clip(parent_interp, 0, None)
            inits.append(('parent_interp', parent_interp))

        # Add diverse init
        inits.append((f'tukey_{n}', tukey_window(n, 0.35)))
        inits.append((f'asym_{n}', asymmetric_window(n, 0.2)))

        # Quick-optimize the best init
        for label, f_init in inits[:2]:  # Only top 2 inits for speed
            if time.monotonic() > n_deadline - 5:
                break

            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            S = float(np.sum(f_init))
            if S < 0.01:
                continue

            init_score = evaluate_sequence(f_init.tolist())
            if not np.isfinite(init_score):
                continue

            if init_score < best_score:
                best_score = init_score
                best_sequence = f_init.copy()
                best_n = n
                print(f"  [n={n} {label}] init score={init_score:.6f} (new best)")

            # Quick optimization
            init_deadline = min(n_deadline, time.monotonic() + n_budget * 0.45)
            if time.monotonic() < init_deadline - 3:
                try:
                    bf, bs = optimize_single(f_init, init_deadline, verbose=f"[n={n} {label}]")
                    if bs < best_score:
                        best_score = bs
                        best_sequence = bf.copy()
                        best_n = n
                        print(f"  [n={n} {label}] optimized to {bs:.6f} (new best)")
                    else:
                        print(f"  [n={n} {label}] best={bs:.6f}")
                except Exception as e:
                    print(f"  [n={n} {label}] error: {e}")

    # Phase 2: Deep optimization on the best
    print(f"\n=== Phase 2: Deep optimization on n={best_n} (score={best_score:.6f}) ===")
    print(f"    Deadline: t={deep_deadline - t0:.0f}s")

    if best_sequence is not None:
        deep_budget = deep_deadline - time.monotonic()
        if deep_budget > 30:
            try:
                bf, bs = optimize_with_restarts(best_sequence, deep_deadline,
                                                n_restarts=3, verbose="[deep] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  Deep optimization: {bs:.6f}")
                else:
                    print(f"  Deep optimization: {bs:.6f} (no improvement)")
            except Exception as e:
                print(f"  Deep optimization error: {e}")

    # Phase 3: Try one more length if time permits
    if time.monotonic() < final_deadline - 60:
        # Pick a length we haven't deeply optimized
        try_lengths = [768, 640, 896, 1024]
        for n_try in try_lengths:
            if n_try == best_n:
                continue
            if time.monotonic() > final_deadline - 60:
                break

            print(f"\n  --- Trying n={n_try} ---")
            # Interpolate from best
            f_try = np.interp(np.linspace(0, 1, n_try),
                              np.linspace(0, 1, len(best_sequence)),
                              best_sequence)
            f_try = np.clip(f_try, 0, None)
            S = float(np.sum(f_try))
            if S < 0.01:
                continue
            f_try = f_try / S

            try_score = evaluate_sequence(f_try.tolist())
            print(f"  Interp init score: {try_score:.6f}")

            try_deadline = min(final_deadline - 30, time.monotonic() + 80)
            if try_score < best_score * 1.02:  # Only optimize if promising
                try:
                    bf, bs = optimize_single(f_try, try_deadline, verbose=f"[n={n_try}]")
                    if bs < best_score:
                        best_score = bs
                        best_sequence = bf.copy()
                        best_n = n_try
                        print(f"  [n={n_try}] improved to {bs:.6f} (new best!)")
                    else:
                        print(f"  [n={n_try}] best={bs:.6f}")
                except Exception as e:
                    print(f"  [n={n_try}] error: {e}")
            break  # Only try one extra length

    # Phase 4: Final polish
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print(f"\n=== Phase 4: Final polish (n={best_n}, score={best_score:.6f}) ===")
        try:
            # LP polish
            A_cache = build_conv_matrix_sparse(best_sequence)
            for _ in range(5):
                if time.monotonic() > deadline - 5:
                    break
                bf, bs = lp_step(best_sequence, A_cache, step_size=0.05)
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    A_cache = build_conv_matrix_sparse(best_sequence)
                else:
                    break

            # Soft-max polish with high beta
            if time.monotonic() < deadline - 5:
                bf, bs = softmax_adam_optimize(best_sequence, deadline - 3,
                                               beta_start=200.0, beta_end=1000.0,
                                               lr_start=0.003, max_iters=1000,
                                               verbose="[final] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()

            print(f"  Final polish: {best_score:.6f}")
        except Exception as e:
            print(f"  Final polish error: {e}")

    # Ensure validity
    if best_sequence is None:
        best_sequence = tukey_window(512, 0.3)
        best_score = evaluate_sequence(best_sequence.tolist())

    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = evaluate_sequence(best_sequence.tolist())

    # Final score check
    final_score = evaluate_sequence(best_sequence.tolist())
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
