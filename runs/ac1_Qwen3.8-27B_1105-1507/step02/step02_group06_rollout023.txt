"""
Optimization of non-negative step function to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses Nesterov accelerated gradient with softmax-max approximation,
LP-based direction finding, and coordinate descent for final polish.
Explores multiple sequence lengths with focus on n=1024-4096.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    """Compute the objective value directly (fast, no type checks)."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def softmax_gradient(f, temperature):
    """Compute gradient of softmax approximation of max(conv(f,f)).
    Returns gradient vector and the softmax-max value."""
    n = len(f)
    conv = np.convolve(f, f)
    # Softmax probabilities
    conv_shifted = conv - np.max(conv)  # numerical stability
    exp_conv = np.exp(conv_shifted / temperature)
    p = exp_conv / np.sum(exp_conv)
    soft_max_val = temperature * (np.log(np.sum(exp_conv)) + np.max(conv_shifted))
    
    # Gradient: grad[i] = 2 * sum_k p[k] * f[k-i] (if 0 <= k-i < n)
    # This is equivalent to: grad = 2 * cross_correlation(p, f)
    # where cross_correlation(p, f)[i] = sum_{k-i valid} p[k] * f[k-i]
    # = sum_j p[i+j] * f[j] for valid i+j
    
    # Use FFT for efficiency
    # p has length 2n-1, f has length n
    # We want: grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * f[j]
    # This is a cross-correlation
    
    # Pad p to length 4n for FFT
    size = 4 * n
    P = np.fft.rfft(p, size)
    F = np.fft.rfft(f, size)
    # Cross-correlation: ifft(conj(P) * F) gives sum_k p[k] * f[k-i] ... 
    # Actually let me think more carefully.
    # We want g[i] = sum_{j} p[i+j] * f[j]
    # Let's define q[m] = p[m] for m in [0, 2n-2], 0 otherwise
    # g[i] = sum_j q[i+j] * f[j]
    # This is a cross-correlation of q and f, specifically:
    # g = cross_corr(q, f) where cross_corr(q, f)[i] = sum_j q[i+j] * f[j]
    # In terms of convolution: if we reverse f, then (q * f_rev)[i + n - 1] = sum_j q[i+j] * f_rev[n-1-j] = sum_j q[i+j] * f[j]
    # So g = (q * f_rev)[n-1 : 2n-1 + n-1] ... hmm this is getting complicated.
    
    # Let me just use the direct formula with FFT:
    # g[i] = sum_j q[i+j] * f[j] where q has length 2n-1
    # This is the "anti-convolution" or cross-correlation
    # Using FFT: g = ifft(FFT(q) * conj(FFT(f))) shifted appropriately
    # Actually, cross-correlation of q and f: (q ⋆ f)[i] = sum_j q[j] * f[j+i]
    # That's not quite what we want either.
    
    # Let me just do it directly for now since n=1024 is manageable
    # For i in range(n):
    #     g[i] = dot(p[i:i+n], f)  # p has length 2n-1, so p[i:i+n] has at most n elements
    
    # Actually, p has length 2n-1 (indices 0 to 2n-2)
    # For g[i] = sum_{j=0}^{n-1} p[i+j] * f[j], we need i+j <= 2n-2, i.e., j <= 2n-2-i
    # For i >= 0, i+j goes from i to i+n-1, and we need i+n-1 <= 2n-2, i.e., i <= n-1. ✓
    # So for all i in [0, n-1], the range i to i+n-1 is within [0, 2n-2]. ✓
    
    # Use strided matrix multiplication for efficiency
    # Or just use the fact that this is a matrix-vector product
    # Build the matrix M where M[i,j] = p[i+j]
    # This is a Hankel matrix
    
    # For n=1024, a direct computation would be O(n^2) which is ~1M operations, fine.
    # But let's use FFT for larger n.
    
    # FFT approach:
    # g[i] = sum_j p[i+j] * f[j]
    # Let's define h[m] = p[m] for m in [0, 2n-2]
    # g = sum_j h[i+j] * f[j]
    # This is the cross-correlation of h and f evaluated at specific lags.
    # cross_corr(h, f)[lag] = sum_k h[k] * f[k+lag]
    # We want cross_corr(h, f)[i] for i in [0, n-1]
    # cross_corr(h, f)[i] = sum_k h[k] * f[k+i]
    # = sum_k h[k] * f[k+i] which is what we want (with k playing the role of j, and h[k]=p[k], f[k+i] = f[j] where j = k+i... wait no)
    
    # Let me re-derive: g[i] = sum_{j=0}^{n-1} p[i+j] * f[j]
    # Let k = i+j, so j = k-i, and k ranges from i to i+n-1
    # g[i] = sum_{k=i}^{i+n-1} p[k] * f[k-i]
    # This is sum_k p[k] * f[k-i] which is the cross-correlation of p and f at lag i.
    # cross_corr(p, f)[i] = sum_k p[k] * f[k-i] ✓
    
    # Using FFT: cross_corr(p, f) = ifft(conj(FFT(p)) * FFT(f))
    # (with appropriate zero-padding)
    
    size = 1
    while size < 2 * (2 * n - 1) - 1:
        size *= 2
    # Actually we need size >= (2n-1) + n - 1 = 3n - 2
    size = 1
    while size < 3 * n:
        size *= 2
    
    P = np.fft.rfft(p, size)
    F = np.fft.rfft(f, size)
    cross = np.fft.irfft(np.conj(P) * F, size)
    # cross[i] = sum_k p[k] * f[k-i] for i in [0, size-1]
    # But due to circular correlation, we need to be careful
    # For i in [0, n-1], cross[i] should give us what we want
    # (since p is non-zero only for k in [0, 2n-2] and f is non-zero only for k-i in [0, n-1])
    
    grad = 2.0 * cross[:n]
    return grad, soft_max_val


def nesterov_optimize(f, deadline, n_temp_start=10.0, n_temp_end=0.1,
                      lr_init=0.01, momentum_init=0.9, verbose_prefix=""):
    """Nesterov accelerated gradient with adaptive softmax temperature."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = compute_score(f)
    
    # Nesterov variables
    y = f.copy()
    momentum = np.zeros(n)
    lr = lr_init
    no_improve = 0
    it = 0
    
    total_time = deadline - time.monotonic()
    
    while time.monotonic() < deadline and it < 100000:
        # Adaptive temperature
        progress = min(1.0, it / max(1, int(0.8 * 100000)))
        temperature = n_temp_start * (n_temp_end / n_temp_start) ** progress
        
        # Compute gradient at y
        y_clipped = np.clip(y, 0, None)
        y_sum = np.sum(y_clipped)
        if y_sum < 1e-15:
            break
        y_norm = y_clipped * (S / y_sum)
        
        grad, soft_val = softmax_gradient(y_norm, temperature)
        
        # Nesterov update
        momentum = momentum_init * momentum - lr * grad
        new_y = y + momentum
        new_y = np.clip(new_y, 0.0, None)
        new_y_sum = np.sum(new_y)
        if new_y_sum < 1e-15:
            lr *= 0.5
            continue
        new_y = new_y * (S / new_y_sum)
        
        # Update Nesterov position
        beta = (1.0 + np.sqrt(1.0 + 4.0 * momentum_init * momentum_init)) / 2.0
        y = new_y + momentum_init * (new_y - f)
        
        # Check improvement
        new_score = compute_score(new_y)
        if np.isfinite(new_score) and new_score < best_score:
            f = new_y.copy()
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
            if it % 500 == 0:
                print(f"    {verbose_prefix}Nesterov iter {it}: score={best_score:.6f} "
                      f"temp={temperature:.3f} lr={lr:.5f}")
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.9
                momentum_init = max(0.5, momentum_init * 0.99)
            if lr < 1e-8 or no_improve > 2000:
                break
        
        # Decay learning rate
        lr *= 0.9999
        it += 1
    
    return best_f, best_score


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_optimize(f, deadline, step_init=0.1, verbose_prefix=""):
    """LP-based optimization with adaptive step size."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    step = step_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break

        t_start = time.monotonic()
        g = solve_lp_direction(f, M)
        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        lp_time = time.monotonic() - t_start
        
        if lp_time > 30:
            # LP is too slow, skip
            no_improve += 1
            if no_improve > 5:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-15:
            no_improve += 1
            continue

        g = g * (S / sum_g)

        # Line search
        best_step_score = best_score
        best_step_f = None
        steps_to_try = [step, step * 0.5, step * 1.5, step * 0.25, step * 2.0]
        for alpha in steps_to_try:
            alpha = min(alpha, 1.0)
            if alpha < 0.001:
                continue
            candidate = (1.0 - alpha) * f + alpha * g
            candidate = np.clip(candidate, 0.0, None)
            cand_sum = float(np.sum(candidate))
            if cand_sum < 1e-15:
                continue
            candidate = candidate * (S / cand_sum)
            cand_score = compute_score(candidate)
            if np.isfinite(cand_score) and cand_score < best_step_score:
                best_step_score = cand_score
                best_step_f = candidate

        it += 1

        if best_step_f is not None:
            f = best_step_f
            best_f = f.copy()
            best_score = best_step_score
            no_improve = 0
            if it % 10 == 0:
                print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f} "
                      f"(step={step:.3f}, lp_time={lp_time:.2f}s)")
            step *= 0.99
            step = max(step, 0.01)
        else:
            no_improve += 1
            step *= 0.8
            if step < 0.005 or no_improve > 30:
                break

    return best_f, best_score


def coordinate_descent(f, deadline, verbose_prefix=""):
    """Coordinate descent: optimize one coordinate at a time."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = compute_score(f)
    
    no_improve = 0
    
    while time.monotonic() < deadline and no_improve < 50:
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                    break

def make_structured_init(n, rng):
    """Create diverse structured initializations."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    # Uniform
    inits.append(('uniform', np.ones(n)))
    
    # Cosine window
    inits.append(('cosine', (1.0 + np.cos(2 * np.pi * idx / n)) / 2.0 + 0.1))
    
    # Raised cosine
    inits.append(('raised_cos', 0.5 * (1 + np.cos(np.pi * idx / (n - 1)))))
    
    # Chirp-like (quadratic phase, made non-negative)
    phase = (idx / n) ** 2
    inits.append(('chirp_abs', np.abs(np.cos(4 * np.pi * phase)) + 0.1))
    
    # Gaussian
    inits.append(('gaussian', np.exp(-0.5 * ((idx - center) / (n / 4)) ** 2)))
    
    # Bimodal
    inits.append(('bimodal', np.exp(-((idx - n*0.25)/(n*0.1))**2) + np.exp(-((idx - n*0.75)/(n*0.1))**2)))
    
    # Random with structure
    smooth_noise = np.convolve(rng.uniform(0.5, 1.5, n), np.ones(10)/10, mode='same')
    inits.append(('smooth_rand', smooth_noise))
    
    # Periodic
    for period in [n//4, n//3, n//5]:
        if period > 2:
            inits.append((f'period_{period}', 0.5 + 0.5 * np.cos(2 * np.pi * idx / period)))
    
    # Logarithmic chirp
    inits.append(('log_chirp', 0.5 + 0.5 * np.sin(2 * np.pi * idx * np.log1p(idx) / n)))
    
    return [(name, arr.astype(np.float64)) for name, arr in inits if np.all(arr >= 0)]

    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))


def softmax_gradient_fast(f, temperature):
    """Compute gradient of softmax(t * log(sum(exp(conv/t)))) using FFT cross-correlation.
    Returns (gradient, soft_max_value, convolution)."""
    n = len(f)
    conv = np.convolve(f, f)
    conv_max = np.max(conv)
    # Numerically stable softmax
    shifted = (conv - conv_max) / temperature
    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals))
    p = exp_vals / np.sum(exp_vals)
    soft_val = temperature * log_sum + conv_max

    # Gradient: grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * f[j]
    # = 2 * cross_corr(p, f)[i] for i in [0, n-1]
    # Using FFT: cross_corr = ifft(conj(FFT(p)) * FFT(f))
    size = 1
    while size < 3 * n:
        size *= 2
    P = np.fft.rfft(p, size)
    F = np.fft.rfft(f, size)
    cross = np.fft.irfft(np.conj(P) * F, size)
    grad = 2.0 * cross[:n]
    return grad, soft_val, conv


def nesterov_optimize(f, deadline, temp_start=15.0, temp_end=0.3,
                      lr_init=0.005, mu_init=0.90, verbose_prefix=""):
    """Nesterov accelerated gradient with adaptive softmax temperature."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = compute_score(f)

    y = f.copy()  # Nesterov lookahead
    v = np.zeros(n)  # velocity
    lr = lr_init
    mu = mu_init
    no_improve = 0
    it = 0
    max_iter = 200000
    total_budget = max(1, max_iter)

    while time.monotonic() < deadline and it < max_iter:
        # Adaptive temperature: geometric decay
        progress = min(1.0, it / (0.85 * total_budget))
        temperature = temp_start * (temp_end / temp_start) ** progress

        # Ensure y is valid
        y = np.clip(y, 0.0, None)
        y_sum = float(np.sum(y))
        if y_sum < 1e-15:
            lr *= 0.5
            it += 1
            continue
        y = y * (S / y_sum)

        # Compute gradient
        grad, soft_val, conv = softmax_gradient_fast(y, temperature)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Nesterov update
        v = mu * v - lr * grad
        new_x = y + v
        new_x = np.clip(new_x, 0.0, None)
        new_sum = float(np.sum(new_x))
        if new_sum < 1e-15:
            lr *= 0.5
            v *= 0.5
            it += 1
            continue
        new_x = new_x * (S / new_sum)

        # Nesterov lookahead update
        y = new_x + mu * (new_x - f)

        # Evaluate
        new_score = compute_score(new_x)
        it += 1

        if np.isfinite(new_score) and new_score < best_score:
            f = new_x.copy()
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
            if it % 1000 == 0:
                print(f"    {verbose_prefix}iter {it}: score={best_score:.6f} "
                      f"temp={temperature:.3f} lr={lr:.2e} mu={mu:.3f}")
        else:
            no_improve += 1
            if no_improve % 200 == 0:
                lr *= 0.92
                mu = max(0.7, mu * 0.995)
            if lr < 1e-9 or no_improve > 3000:
                break

        # Very slow LR decay
        lr *= 0.99995

    return best_f, best_score


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    rows = rows[:idx]
    cols = cols[:idx]
    vals = vals[:idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_optimize(f, deadline, step_init=0.08, verbose_prefix=""):
    """LP-based optimization with adaptive step size."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    step = step_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break

        t_start = time.monotonic()
        g = solve_lp_direction(f, M)
        lp_time = time.monotonic() - t_start

        if g is None:
            no_improve += 1
            if no_improve > 15:
                break
            continue
        if lp_time > 60:
            no_improve += 2
            if no_improve > 10:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-15:
            no_improve += 1
            continue

        g = g * (S / sum_g)

        best_step_score = best_score
        best_step_f = None
        for alpha in [step, step * 0.5, step * 1.5, step * 0.25, step * 2.0]:
            alpha = min(alpha, 1.0)
            if alpha < 0.001:
                continue
            candidate = np.clip((1.0 - alpha) * f + alpha * g, 0.0, None)
            cand_sum = float(np.sum(candidate))
            if cand_sum < 1e-15:
                continue
            candidate = candidate * (S / cand_sum)
            cand_score = compute_score(candidate)
            if np.isfinite(cand_score) and cand_score < best_step_score:
                best_step_score = cand_score
                best_step_f = candidate

        it += 1

        if best_step_f is not None:
            f = best_step_f
            best_f = f.copy()
            best_score = best_step_score
            no_improve = 0
            if it % 5 == 0:
                print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f} "
                      f"(step={step:.3f}, time={lp_time:.1f}s)")
            step = max(0.01, step * 0.98)
        else:
            no_improve += 1
            step *= 0.75
            if step < 0.005 or no_improve > 25:
                break

    return best_f, best_score


def coordinate_descent(f, deadline, verbose_prefix=""):
    """Cyclic coordinate descent with 1D minimax solve per coordinate."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = compute_score(f)
    no_improve = 0
    epoch = 0

    while time.monotonic() < deadline and no_improve < 20:
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            old_val = f[i]
            conv = np.convolve(f, f)
            M = float(np.max(conv))

            # Compute coefficients for 1D problem
            # conv[k] ≈ base_k + c_k * x (linear approx)
            # c_k = 2*f[k-i] if 0 <= k-i < n
            c = np.zeros(2 * n - 1)
            valid_start = max(0, i - (n - 1))
            valid_end = min(2 * n - 2, i + (n - 1))
            for k in range(valid_start, valid_end + 1):
                j = k - i
                if 0 <= j < n:
                    c[k] = 2.0 * f[j]

            base = conv - c * old_val

            # Find optimal x: minimize max_k (base_k + c_k * x) s.t. x >= 0
            # Try intersection of top constraints
            top_idx = np.argsort(base + c * old_val)[-4:]
            candidates = [0.0, old_val, old_val * 0.5, old_val * 1.5, old_val * 0.1]
            for a_i in range(len(top_idx)):
                for b_i in range(a_i + 1, len(top_idx)):
                    ka, kb = top_idx[a_i], top_idx[b_i]
                    if abs(c[ka] - c[kb]) > 1e-12:
                        x_int = (base[kb] - base[ka]) / (c[ka] - c[kb])
                        if 0 <= x_int <= 1000:
                            candidates.append(x_int)

            best_x = old_val
            best_local_max = np.inf
            for x in candidates:
                x = max(0.0, min(x, 1000.0))
                f[i] = x
                new_sum = float(np.sum(f))
                if new_sum < 1e-15:
                    f[i] = old_val
                    continue
                f_tmp = f * (S / new_sum)
                new_conv = np.convolve(f_tmp, f_tmp)
                new_M = float(np.max(new_conv))
                if new_M < best_local_max:
                    best_local_max = new_M
                    best_x = x

            f[i] = best_x
            if abs(best_x - old_val) > 1e-10 * max(1.0, old_val):
                improved = True
                new_sum = float(np.sum(f))
                if new_sum > 1e-15:
                    f = f * (S / new_sum)

        epoch += 1
        if improved:
            new_score = compute_score(f)
            if new_score < best_score:
                best_f = f.copy()
                best_score = new_score
                no_improve = 0
                if epoch % 5 == 0:
                    print(f"    {verbose_prefix}CD epoch {epoch}: score={best_score:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_f, best_score


def perturbation_restart(f, rng, strength=0.1):
    """Add random perturbation to escape local minima."""
    n = len(f)
    S = float(np.sum(f))
    perturbed = f.copy()
    mask = rng.random(n) < 0.3
    perturbed[mask] *= (1.0 + strength * (rng.random(np.sum(mask)) * 2.0 - 1.0))
    perturbed = np.clip(perturbed, 0.0, None)
    psum = float(np.sum(perturbed))
    if psum < 1e-15:
        return f.copy()
    return perturbed * (S / psum)


def make_structured_init(n, rng):
    """Create diverse structured initializations."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    inits.append(('uniform', np.ones(n)))
    inits.append(('cosine', 0.5 * (1.0 + np.cos(np.pi * idx / max(1, n - 1)))))
    inits.append(('gaussian', np.exp(-0.5 * ((idx - center) / (n / 5.0)) ** 2)))
    inits.append(('bimodal', np.exp(-((idx - n * 0.3) / (n * 0.12)) ** 2) +
                np.exp(-((idx - n * 0.7) / (n * 0.12)) ** 2)))
    inits.append(('ramp', np.linspace(0.3, 1.0, n)))
    inits.append(('inv_ramp', np.linspace(1.0, 0.3, n)))

    # Chirp-like (non-negative)
    phase = np.pi * (idx / n) ** 2
    inits.append(('chirp_pos', 0.5 + 0.5 * np.cos(2.0 * phase)))

    # Multi-frequency
    freqs = [1.0, 2.5, 4.0]
    sig = np.zeros(n)
    for freq in freqs:
        sig += 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * idx / n)
    inits.append(('multifreq', sig / len(freqs)))

    # Smooth random
    for t in range(2):
        r = rng.uniform(0.5, 1.5, size=n)
        kernel = np.ones(max(3, n // 50)) / max(3, n // 50)
        r = np.convolve(r, kernel, mode='same')
        inits.append((f'smooth_r{t}', r))

    return [(nm, arr.astype(np.float64)) for nm, arr in inits if np.all(arr >= 0) and np.sum(arr) > 0]


def propose_candidate(budget_s=1500, **kwargs):
    """Main entry point: search for the best sequence minimizing 2n*max(conv)/sum^2."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(7)

    # Load parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            score = compute_score(parent)
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = parent.copy()
                print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    # Determine lengths
    lengths_to_try = []
    if parent is not None:
        lengths_to_try.append(len(parent))
    for n_try in [1024, 2048, 4096]:
        if n_try not in lengths_to_try:
            lengths_to_try.append(n_try)
    lengths_to_try = sorted(set(lengths_to_try))[:4]
    print(f"Lengths to try: {lengths_to_try}")

    # Phase 1: Exploration (first 400s)
    print("\n=== Phase 1: Exploration (400s) ===")
    expl_deadline = min(t0 + 400.0, deadline - 300.0)

    all_candidates = []

    for n in lengths_to_try:
        if time.monotonic() > expl_deadline - 20:
            break
        print(f"\n--- n={n} ---")
        inits = make_structured_init(n, rng)

        if parent is not None:
            if len(parent) == n:
                inits.append(('parent', parent.copy()))
            else:
                x_old = np.linspace(0, 1, len(parent))
                x_new = np.linspace(0, 1, n)
                inits.append(('parent_interp', np.interp(x_new, x_old, parent)))

        for label, f_init in inits:
            if time.monotonic() > expl_deadline - 5:
                break
            f_init = np.clip(f_init, 0.0, 1000.0)
            fs = float(np.sum(f_init))
            if fs < 1e-10:
                continue
            f_init = f_init * (np.sqrt(2.0 * n) / fs)

            score = compute_score(f_init)
            if not np.isfinite(score):
                continue

            # Quick Nesterov
            qd = min(expl_deadline - 2, time.monotonic() + 25.0)
            f_opt, s_opt = nesterov_optimize(f_init, qd,
                                              temp_start=20.0, temp_end=2.0,
                                              lr_init=0.004, mu_init=0.88)
            if s_opt < score:
                score = s_opt
                f_init = f_opt

            if score < best_score:
                best_score = score
                best_sequence = f_init.copy()

            all_candidates.append((score, f_init.copy(), n, label))
            print(f"  {label}: {score:.6f}")

    all_candidates.sort(key=lambda x: x[0])
    print(f"\nTop after exploration:")
    for sc, seq, nn, lbl in all_candidates[:6]:
        print(f"  n={nn} {lbl}: {sc:.6f}")
    print(f"Best so far: {best_score:.6f}")

    # Phase 2: Focused optimization (next ~800s)
    print(f"\n=== Phase 2: Focused Optimization ===")
    focus_deadline = deadline - 250.0

    seen_n = set()
    top_cands = []
    for sc, seq, nn, lbl in all_candidates:
        if nn not in seen_n:
            top_cands.append((sc, seq, nn, lbl))
            seen_n.add(nn)
        if len(top_cands) >= 3:
            break

    for sc, f_init, n, lbl in top_cands:
        if time.monotonic() > focus_deadline - 60:
            break
        print(f"\n--- Focus: n={n} {lbl} (start={sc:.6f}) ---")

        remaining = focus_deadline - time.monotonic()

        # Nesterov main (70% of focus time)
        nest_dl = time.monotonic() + remaining * 0.7
        f_opt, s_opt = nesterov_optimize(f_init, nest_dl,
                                          temp_start=10.0, temp_end=0.5,
                                          lr_init=0.003, mu_init=0.92)
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()
            print(f"  After Nesterov: {best_score:.6f}")

        # Restart with perturbation if stuck
        if time.monotonic() < focus_deadline - 30:
            f_pert = perturbation_restart(f_opt, rng, strength=0.08)
            s_pert = compute_score(f_pert)
            if s_pert < best_score:
                f_r, s_r = nesterov_optimize(f_pert, min(focus_deadline, time.monotonic() + 60),
                                              temp_start=5.0, temp_end=1.0,
                                              lr_init=0.002, mu_init=0.93)
                if s_r < best_score:
                    best_score = s_r
                    best_sequence = f_r.copy()
                    print(f"  After restart: {best_score:.6f}")

        # LP polish (remaining focus time)
        if time.monotonic() < focus_deadline - 10:
            f_opt2, s_opt2 = lp_optimize(f_opt, focus_deadline, step_init=0.06)
            if s_opt2 < best_score:
                best_score = s_opt2
                best_sequence = f_opt2.copy()
                print(f"  After LP: {best_score:.6f}")

    # Phase 3: Final polish (last 250s)
    print(f"\n=== Phase 3: Final Polish (best={best_score:.6f}) ===")
    if best_sequence is not None:
        # Low-temperature Nesterov
        if time.monotonic() < deadline - 60:
            pol_dl = min(deadline - 30, time.monotonic() + 100)
            f_pol, s_pol = nesterov_optimize(best_sequence, pol_dl,
                                              temp_start=1.0, temp_end=0.1,
                                              lr_init=0.0008, mu_init=0.95)
            if s_pol < best_score:
                best_score = s_pol
                best_sequence = f_pol.copy()
                print(f"  Polish Nesterov: {best_score:.6f}")

        # Coordinate descent
        if time.monotonic() < deadline - 20:
            cd_dl = min(deadline - 10, time.monotonic() + 80)
            f_cd, s_cd = coordinate_descent(best_sequence, cd_dl)
            if s_cd < best_score:
                best_score = s_cd
                best_sequence = f_cd.copy()
                print(f"  Coord descent: {best_score:.6f}")

        # Final LP
        if time.monotonic() < deadline - 5:
            f_lp, s_lp = lp_optimize(best_sequence, deadline, step_init=0.03)
            if s_lp < best_score:
                best_score = s_lp
                best_sequence = f_lp.copy()
                print(f"  Final LP: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(1024)
        best_score = evaluate_sequence(best_sequence.tolist())

    # Validate
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    final_score = evaluate_sequence([float(x) for x in best_sequence])
    if not np.isfinite(final_score):
        print("WARNING: Failed evaluation, using fallback!")
        best_sequence = np.ones(512)
        final_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, "
          f"internal={best_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_sequence]
