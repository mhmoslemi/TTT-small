LP-alternating optimization for minimizing 2*n*max(conv(f,f))/(sum(f))^2.
Uses alternating linear programming (min M s.t. A_f@g <= M*1, sum(g)=1, g>=0),
multiplicative weight updates, spectral initialization, and coordinate descent polishing.

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute 2*n*max(conv)/sum^2 for a sequence (assumes sum=1)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def normalize(f):
    """Normalize to sum 1, handling edge cases."""
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def build_conv_matrix(f):
    """Build matrix A such that A @ g = np.convolve(f, g, 'full')."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            A[k, j_lo:j_hi + 1] = f[k - np.arange(j_lo, j_hi + 1)]
    return A


def solve_lp_direction(f, time_limit=5):
    """Solve: min M s.t. A_f @ g <= M*1, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(f)
    m = 2 * n - 1
    A = build_conv_matrix(f)

    # Variables: [g_0, ..., g_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
    except Exception:
        return None, None

    if res.success:
        g = np.clip(res.x[:n], 0, None)
        M = res.x[-1]
        return g, M
    return None, None


def solve_lp_direction_maxsum(f, M_target, time_limit=5):
    """Solve: max sum(g) s.t. A_f @ g <= M_target*1, g>=0. Returns g or None."""
    n = len(f)

    A = build_conv_matrix(f)

    c = -np.ones(n)
    A_ub = A

    """Alternating LP: find g from f, then f from g, iterating."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    start = time.monotonic()
    no_improve = 0

    while time.monotonic() - start < time_budget:
        # Solve LP for direction from current f
        g, M = solve_lp_direction(f, time_limit=5)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        # Check if g is an improvement
        improved = False
        score_g = compute_score(g)
        if score_g < best_score - 1e-12:
            best_score = score_g
            best_f = g.copy()
            f = g.copy()
            no_improve = 0
            improved = True
        else:
            # Try relaxed updates
            for alpha in [0.5, 0.3, 0.1, 0.7, 0.8]:
                f_new = (1 - alpha) * f + alpha * g
                f_new = np.clip(f_new, 0, None)
                s = f_new.sum()
                if s < 1e-10:
                    continue
                f_new = f_new / s
                score = compute_score(f_new)
                if score < best_score - 1e-12:
                    best_score = score
                    best_f = f_new.copy()
                    f = f_new
                    no_improve = 0
                    improved = True
                    break

        if not improved:
            no_improve += 1
            if no_improve > 20:
                # Perturb to escape
                f_pert = f * (1.0 + 0.2 * rng.standard_normal(n))
                f_pert = np.clip(f_pert, 0, None)
                ps = f_pert.sum()
                if ps > 1e-10:
                    f_pert = f_pert / ps
                    if compute_score(f_pert) < best_score:
                        f = f_pert
                        best_f = f.copy()
                        best_score = compute_score(f)
                        no_improve = 0
                        continue
                break

        # Now solve backward: given g (or updated f), find best f
        if time.monotonic() - start < time_budget:
            g2, M2 = solve_lp_direction(f, time_limit=5)
            if g2 is not None:
                g2s = g2.sum()
                if g2s > 1e-10:
                    g2 = g2 / g2s
                    for alpha in [0.5, 0.3, 0.1]:
                        f_new = (1 - alpha) * f + alpha * g2
                        f_new = np.clip(f_new, 0, None)
                        s = f_new.sum()
                        if s < 1e-10:
                            continue
                        f_new = f_new / s
                        score = compute_score(f_new)
                        if score < best_score - 1e-12:
                            best_score = score
                            best_f = f_new.copy()
                            f = f_new
                            no_improve = 0
                            break

    return best_f, best_score


def lp_fixed_M(f, M_target, time_budget=60, rng=None):
    """LP with fixed M: repeatedly find best g with (f*g)<=M, update f."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    start = time.monotonic()
    no_improve = 0

    # Start with M slightly above current max
    conv = np.convolve(f, f)
    M = M_target * (1 + 1e-6)

    while time.monotonic() - start < time_budget:
        g = solve_lp_direction_maxsum(f, M, time_limit=5)
        if g is None:
            break
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs

        improved = False
        for alpha in [1.0, 0.5, 0.3, 0.1]:
            f_new = (1 - alpha) * f + alpha * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            score = compute_score(f_new)
            if score < best_score - 1e-12:
                best_score = score
                best_f = f_new.copy()
                f = f_new
                no_improve = 0
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 15:
                # Tighten M
                conv_new = np.convolve(f, f)
                M_new = conv_new.max() * (1 + 1e-6)
                if M_new < M:
                    M = M_new
                    no_improve = 0
                else:
                    break

    return best_f, best_score


def multiplicative_weights(f, n_iter=500, eta=0.3, rng=None):
    """Multiplicative weight updates targeting convolution peaks."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    no_improve = 0

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        j_star = np.argmax(conv)

        # Pressure: for each i, how much does f_i contribute to the peak?
        # Contribution of f_i to conv[j_star] is f_i * f[j_star - i] (if valid)
        pressure = np.zeros(n)
        for i in range(n):
            idx = j_star - i
            if 0 <= idx < n:
                pressure[i] = f[idx]

        # Also consider secondary peaks
        conv_sorted = np.sort(conv)[::-1]
        for p in range(1, min(5, len(conv_sorted))):
            if conv_sorted[p] < 0.85 * M:
                break
            j_p = np.where(np.isin(conv, conv_sorted[p]))[0]
            if len(j_p) == 0:
                continue
            j_p = j_p[0]
            for i in range(n):
                idx = j_p - i
                if 0 <= idx < n:
                    pressure[i] += 0.3 * f[idx]

        if pressure.max() > 0:
            pressure = pressure / pressure.max()

        eta_eff = eta * max(0.1, 1.0 - 0.5 * it / n_iter)
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                f_pert = f * (1 + 0.15 * rng.standard_normal(n))
                f_pert = np.clip(f_pert, 0, None)
                ps = f_pert.sum()
                if ps > 1e-10:
                    f_pert = f_pert / ps
                    if compute_score(f_pert) < best_score:
                        f = f_pert
                        best_f = f.copy()
                        best_score = compute_score(f)
                        no_improve = 0
                    else:
                        break
                else:
                    break

    return best_f, best_score


def spectral_init(n, rng, mode=0):
    """Generate initial sequence via spectral shaping."""
    if mode == 0:
        # Start with random phases, flat magnitude
        phases = rng.standard_normal(n)
        mags = np.ones(n)
        F = mags * np.exp(1j * phases)
        f = np.fft.irfft(F * np.fft.rfft(np.ones(n)) / n, n)
    elif mode == 1:
        # Cosine window
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    elif mode == 2:
        # Asymmetric ramp with smoothing
        t = np.arange(n) / (n - 1)
        f = 0.3 + 0.7 * t
    elif mode == 3:
        # Two-bump asymmetric
        c1, c2 = 0.25, 0.7
        w1, w2 = 0.15, 0.2
        t = np.arange(n) / (n - 1)
        f = np.exp(-0.5 * ((t - c1) / w1) ** 2) + 0.8 * np.exp(-0.5 * ((t - c2) / w2) ** 2)
    else:
        f = np.ones(n)

    f = np.clip(f, 0.01, None)
    # Add small noise to break symmetry
    f = f * (1 + 0.05 * rng.standard_normal(n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f


def spectral_refine(f, n_iter=200, rng=None):
    """Iterative spectral flattening with proper simplex projection."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    no_improve = 0

    for it in range(n_iter):
        FFT = np.fft.fft(f, 2 * n)
        mag = np.abs(FFT)
        phase = np.angle(FFT)

        # Target: flatten magnitude (keep DC)
        avg_mag = mag[1:].mean()
        mag_target = np.full_like(mag, avg_mag)
        mag_target[0] = mag[0]  # Keep DC (sum is fixed)

        FFT_new = mag_target * np.exp(1j * phase)
        f_new = np.real(np.fft.ifft(FFT_new, 2 * n))[:n]

        # Project onto non-negative simplex {x >= 0, sum(x) = 1}
        f_new = project_simplex(f_new)

        score = compute_score(f_new)
        if score < best_score - 1e-10:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            # Occasionally perturb
            if no_improve > 50:
                if it % 10 == 0:
                    phase = phase + rng.normal(0, 0.1, len(phase))
                    FFT_new = mag_target * np.exp(1j * phase)
                    f_new = np.real(np.fft.ifft(FFT_new, 2 * n))[:n]
                    f_new = project_simplex(f_new)
                    score = compute_score(f_new)
                    if score < best_score:
                        best_score = score
                        best_f = f_new.copy()
                        f = f_new
                        no_improve = 0
            if no_improve > 200:
                break

    return best_f, best_score


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    ws = w.sum()
    if ws < 1e-15:
        return np.ones(n) / n
    return w / ws


def coordinate_descent(f, n_sweeps=5, rng=None):
    """Polish with coordinate descent."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            orig = f[i]
            if orig < 1e-12:
                continue
            best_val = orig
            best_local = best_score
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                f[i] = max(0.0, orig * factor)
                s = f.sum()
                if s < 1e-10:
                    f[i] = orig
                    continue
                score = compute_score(f / s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local

    f = np.clip(f, 0, None)
    f = normalize(f)
    return f, compute_score(f)


def diverse_random_init(n, rng):
    """Create a diverse random initial sequence."""
    mode = rng.integers(0, 5)
    if mode == 0:
        f = np.abs(rng.standard_normal(n)) + 0.1
    elif mode == 1:
        f = np.exp(-np.abs(np.arange(n) - n * rng.uniform(0.3, 0.7)) / (n * rng.uniform(0.1, 0.3)))
        f = np.clip(f, 0.01, None)
    elif mode == 2:
        # Asymmetric
        t = np.arange(n) / n
        f = (1 - t) ** rng.uniform(0.5, 3) + (t) ** rng.uniform(0.2, 1)
    elif mode == 3:
        # Periodic with asymmetry
        f = 1 + 0.5 * np.sin(2 * np.pi * rng.uniform(1, 4) * np.arange(n) / n + rng.uniform(0, 2*np.pi))
        f = np.clip(f, 0.01, None)
    else:
        # Smooth random
        f = np.ones(n)
        for _ in range(rng.integers(2, 6)):
            center = rng.integers(0, n)
            width = max(3, int(n * rng.uniform(0.05, 0.3)))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
            f *= (1 + 0.5 * rng.standard_normal() * kernel)
        f = np.clip(f, 0.01, None)

    f = f * (1 + 0.03 * rng.standard_normal(n))
    f = np.clip(f, 0.001, None)
    return normalize(f)


def full_pipeline(f0, rng, time_budget):
    """Run the full optimization pipeline on an initial sequence."""
    f = f0.copy()
    n = len(f)
    start = time.monotonic()
    best_f = f.copy()
    best_score = compute_score(f)

    # Phase 1: Spectral refinement
    remaining = time_budget - (time.monotonic() - start)
    t_spec = min(30, remaining * 0.15)
    if t_spec > 5:
        f, s = spectral_refine(f, n_iter=int(t_spec * 20), rng=rng)
        if s < best_score:
            best_score = s
            best_f = f.copy()
        print(f"    After spectral: {s:.6f}")

    # Phase 2: Multiplicative weights
    remaining = time_budget - (time.monotonic() - start)
    t_mw = min(30, remaining * 0.15)
    if t_mw > 5:
        f, s = multiplicative_weights(f, n_iter=int(t_mw * 20), eta=0.3, rng=rng)
        if s < best_score:
            best_score = s
            best_f = f.copy()
        print(f"    After MW: {s:.6f}")

    # Phase 3: LP alternating (main workhorse)
    remaining = time_budget - (time.monotonic() - start)
    t_lp = min(remaining * 0.5, 120)
    if t_lp > 10:
        f, s = lp_alternating(f, time_budget=t_lp, rng=rng)
        if s < best_score:
            best_score = s
            best_f = f.copy()
        print(f"    After LP-alt: {s:.6f}")

    # Phase 4: More LP with fixed M
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 20:
        t_lp2 = min(remaining * 0.4, 60)
        if t_lp2 > 10:
            conv = np.convolve(best_f, best_f)
            M_target = conv.max()
            f2, s2 = lp_fixed_M(best_f, M_target, time_budget=t_lp2, rng=rng)
            if s2 < best_score:
                best_score = s2
                best_f = f2.copy()
            print(f"    After LP-fixedM: {s2:.6f}")

    # Phase 5: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 10:
        f3, s3 = coordinate_descent(best_f, n_sweeps=min(10, int(remaining * 0.3)), rng=rng)
        if s3 < best_score:
            best_score = s3
            best_f = f3.copy()
        print(f"    After CD: {s3:.6f}")

    # Phase 6: Final LP push
    remaining = time_budget - (time.monotonic() - start)
    if remaining > 15:
        f4, s4 = lp_alternating(best_f, time_budget=min(remaining, 30), rng=rng)
        if s4 < best_score:
            best_score = s4
            best_f = f4.copy()
        print(f"    Final LP: {s4:.6f}")

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: LP-alternating + multiplicative weights + spectral + CD,
    across multiple sequence lengths and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0

    best_overall = None
    best_score_overall = np.inf

    # Try parent sequence
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 and len(parent) > 0 and np.isfinite(parent).all():
            parent = np.clip(parent, 0, None)
            if parent.sum() > 0.01:
                parent = normalize(parent)
                parent_score = compute_score(parent)
                print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
                best_overall = parent.copy()
                best_score_overall = parent_score
    except Exception:
        parent = None

    # Sequence lengths to try
    lengths = [512, 384, 256]
    if parent is not None and len(parent) not in lengths:
        lengths.insert(0, len(parent))

    # Deduplicate and sort by priority
    lengths = sorted(set(lengths), key=lambda x: (x != 512, x))
    print(f"Lengths to try: {lengths}")
    print(f"Budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")
    print(f"Target: 1.5027, Current best: {best_score_overall:.6f}")

    for n in lengths:
        remaining = deadline - time.monotonic()
        if remaining < 60:
            print(f"Not enough time for n={n} ({remaining:.0f}s left)")
            break

        print(f"\n=== n={n} (remaining: {remaining:.0f}s) ===")

        # Allocate time: try up to 3 initializations
        n_inits = min(3, max(1, int(remaining / 120)))
        t_per_init = min(remaining / n_inits - 5, 200)

        for init_idx in range(n_inits):
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

            # Generate initial sequence
            if init_idx == 0 and parent is not None and len(parent) == n:
                # Use perturbed parent
                f0 = parent * (1 + 0.1 * rng.standard_normal(n))
                f0 = np.clip(f0, 0, None)
                f0 = normalize(f0)
                print(f"  Init {init_idx}: perturbed parent")
            elif init_idx == 1:
                f0 = spectral_init(n, rng, mode=2)  # Asymmetric ramp
                print(f"  Init {init_idx}: asymmetric ramp")
            elif init_idx == 2:
                f0 = spectral_init(n, rng, mode=3)  # Two-bump
                print(f"  Init {init_idx}: two-bump")
            else:
                f0 = diverse_random_init(n, rng)
                print(f"  Init {init_idx}: diverse random")

            print(f"    Initial score: {compute_score(f0):.6f}")

            # Run pipeline
            f_best, s_best = full_pipeline(f0, rng, time_budget=t_per_init)

            if s_best < best_score_overall:
                best_score_overall = s_best
                best_overall = f_best.copy()
                print(f"    *** NEW BEST: {s_best:.6f} (n={n}, init={init_idx}) ***")

    # Final global polish on the best found
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            print(f"\n=== Final polish (remaining: {remaining:.0f}s) ===")
            n_best = len(best_overall)

            # One more LP pass
            if remaining > 15:
                f_fin, s_fin = lp_alternating(best_overall, time_budget=min(remaining - 5, 40), rng=rng)
                if s_fin < best_score_overall:
                    best_score_overall = s_fin
                    best_overall = f_fin.copy()
                print(f"  After final LP: {s_fin:.6f}")

            # Coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 10:
                f_cd, s_cd = coordinate_descent(best_overall, n_sweeps=min(5, int(remaining * 0.4)), rng=rng)
                if s_cd < best_score_overall:
                    best_score_overall = s_cd
                    best_overall = f_cd.copy()
                print(f"  After final CD: {s_cd:.6f}")

    # Verify with evaluate_sequence
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0.0, 1000.0)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal result: score={final_score:.6f}, n={len(final_list)}")
        return final_list

    # Fallback
    print("FALLBACK: returning uniform sequence")
    return [1.0] * 256
