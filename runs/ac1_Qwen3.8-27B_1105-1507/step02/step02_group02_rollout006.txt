    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def fast_cross_conv(a, w):
    """Cross-convolution of a (len n) and w (len m) via FFT, returns first n elements of full conv."""
    n = len(a)
    m = len(w)
    s = 1
    while s < n + m - 1:
        s <<= 1
    FA = np.fft.rfft(a, s)
    FW = np.fft.rfft(w, s)
    result = np.fft.irfft(FA * FW, s)
    return result[:n]

def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))

def soft_max_value(b, beta):
    """Compute (1/beta)*log(sum(exp(beta*b)))."""
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -100.0, 100.0)
    return b_max + np.log(np.exp(shifted).sum()) / beta

def smooth_objective_value(a, beta):
    """2*n*soft_max(conv(a,a), beta) for normalized a."""
    n = len(a)
    b = fast_conv(a)
    return 2.0 * n * soft_max_value(b, beta)
