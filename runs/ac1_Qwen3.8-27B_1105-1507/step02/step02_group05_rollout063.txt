```python
from scipy.optimize import minimize, linprog
from scipy.sparse import lil_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2*n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n-1]

def compute_gradient(a, tau):
    """Compute gradient of 2n*logsumexp(conv(a,a)/tau)/(sum(a))^2"""
    n = len(a)
    b = fast_conv(a)
    S = a.sum()
    
    # Numerically stable logsumexp
    b_scaled = b / tau
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    sum_exp = exp_b.sum()
    logsumexp = b_max + np.log(sum_exp)
    w = exp_b / sum_exp  # softmax weights
    
    # Gradient of logsumexp w.r.t. a:
    # d(dlogsumexp)/da_m = (2/tau) * sum_j w[j+m] * a[j]
    # = (2/tau) * (conv(w, a_rev))[m+n-1]
    
    a_rev = a[::-1]
    # Compute conv(w, a_rev) via FFT
    s = 1
    while s < len(w) + n - 1:
        s <<= 1
    fw = np.fft.rfft(w, s)
    fa_r = np.fft.rfft(a_rev, s)
    conv_w_ar = np.fft.irfft(fw * fa_r, s)[:len(w) + n - 1]

    # dJ/da = 2n * (grad_logsumexp * S^2 - logsumexp * 2*S) / S^4
    #       = 2n * (grad_logsumexp / S - 2*logsumexp / S^2)

    a = np.maximum(a, 0.001)
    a /= a.sum()
    
    tracker.update(a)
    
    step = 0.01

            break
        
        # Anneal tau

        a_new = np.maximum(a_new, 0.0)

        if J_new < J - 1e-15:
            a = a_new
        else:
            step *= 0.7
        
    tracker.update(a)
    return a
```

```python
"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))² using a hybrid approach:
1. LP mirror descent with adaptive steps (initial exploration)
2. Log-sum-exp smoothed gradient descent with L-BFGS-B (refinement)
3. Active-set subgradient polishing (final precision)
4. Length adaptation and restart strategies
Key innovation: smooth gradient via log-sum-exp enables quasi-Newton convergence.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def compute_smooth_objective_and_grad(a, tau):
    """
    Compute J_smooth(a) = 2n * logsumexp(conv(a,a)/tau) / (sum(a))^2
    and its gradient using FFT-based cross-correlation.
    """
    n = len(a)
    b = fast_conv(a)
    S = a.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)

    # Stable logsumexp
    b_scaled = b / tau
    b_max = float(b_scaled.max())
    exp_b = np.exp(b_scaled - b_max)
    sum_exp = float(exp_b.sum())
    w = exp_b / sum_exp
    L = b_max + np.log(sum_exp)  # logsumexp value

    # Gradient of L w.r.t. a:
    # dL/da_m = (2/tau) * sum_{j=0}^{n-1} w[j+m] * a[j]
    # Compute c[m] = sum_j w[j+m] * a[j] via FFT-based convolution
    # c[m] = conv(w_rev, a)[(len(w)-1) - m]
    w_rev = w[::-1]  # length 2n-1
    L_fft = 1
    while L_fft < (2 * n - 1) + n:
        L_fft <<= 1
    Fw = np.fft.rfft(w_rev, L_fft)
    Fa = np.fft.rfft(a, L_fft)
    conv_full = np.fft.irfft(Fw * Fa, L_fft)[:3 * n - 2]
    # c[m] = conv_full[2n-2-m] for m=0..n-1
    c = conv_full[n - 1:2 * n - 1][::-1]

    grad_L = (2.0 / tau) * c

    # J = 2n * L / S^2
    J_val = float(2.0 * n * L / (S * S))
    # dJ/da_m = 2n * (grad_L[m] * S - 2*L) / S^3
    grad_J = 2.0 * n * (grad_L * S - 2.0 * L) / (S ** 3)

    return J_val, grad_J


def run_lp_iterations(a, max_iters, deadline, tracker, verbose=False):
    """Run LP mirror descent iterations with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Adaptive step with backtracking
        best_a = None
        best_J = current_J
        t_candidates = [0.3, 0.2, 0.1, 0.05]
        for t in t_candidates:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 20 == 0 and verbose:
            tracker.update(a)

    tracker.update(a)
    return a


def run_logsumexp_lbfgs(a, deadline, tracker, tau_start=10.0, tau_end=0.01):
    """
    Optimize using L-BFGS-B on the log-sum-exp smoothed objective.
    Anneal tau from tau_start to tau_end.
    """
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.001)
    a /= a.sum()
    tracker.update(a)

    n = len(a)
    total_budget = max(1, int(deadline - time.monotonic()))
    # Split into stages with decreasing tau
    n_stages = 8
    stage_durations = []
    for s in range(n_stages):
        frac = (s + 1) ** 2 / n_stages ** 2
        stage_durations.append(frac * total_budget)

    current_tau = tau_start
    current_J = compute_J(a)

    for stage in range(n_stages):
        if time.monotonic() > deadline:
            break
        stage_deadline = min(deadline, time.monotonic() + stage_durations[stage])
        tau = tau_start * (tau_end / tau_start) ** ((stage + 1) / n_stages)

        print(f"    L-BFGS-B stage {stage+1}/{n_stages}: tau={tau:.4f}, J={current_J:.8f}", flush=True)

        def obj_and_grad(x):
            x = np.maximum(x, 0.0)
            s = x.sum()
            if s < 1e-10:
                return 1e10, np.zeros_like(x)
            x_norm = x / s
            val, grad = compute_smooth_objective_and_grad(x_norm, tau)
            # Chain rule: d/dx [f(x/sum(x))]
            # Simplification: just use the gradient as-is since we normalize
            return val, grad

        def obj_only(x):
            x = np.maximum(x, 0.0)
            s = x.sum()
            if s < 1e-10:
                return 1e10
            x_norm = x / s
            b = fast_conv(x_norm)
            return float(2 * n * b.max())

        bounds_list = [(0.0, None)] * n
        max_iter = max(50, min(500, int(30000 // n)))

        res = minimize(obj_and_grad, a.copy(), method='L-BFGS-B',
                       bounds=bounds_list, jac=True,
                       options={'maxiter': max_iter, 'max_fun': max_iter * 3,
                                'ftol': 1e-18, 'gtol': 1e-10, 'iprint': -1})

        if res.fun < 1e10:
            a_new = np.maximum(res.x, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new /= s
                J_new = compute_J(a_new)
                if J_new < current_J:
                    a = a_new
                    current_J = J_new
                    tracker.update(a)

    tracker.update(a)
    return a


def run_subgradient_refine(a, max_iters, deadline, tracker):
    """
    Subgradient method on the exact max objective.
    At each step, compute active set (k values near max),
    take average subgradient direction, do projected gradient step.
    """
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    n = len(a)

    step = 0.05
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = float(b.max())
        S = a.sum()

        # Find active set: k values where b_k > 0.95 * b_max
        active = np.where(b > 0.95 * b_max)[0]

        # Compute average subgradient over active set
        # Subgradient of b_k at a: 2*a shifted by k
        # d(b_k)/da_m = 2*a_{k-m} if 0<=k-m<n
        grad = np.zeros(n)
        for k in active:
            for m in range(n):
                j = k - m
                if 0 <= j < n:
                    grad[m] += 2.0 * a[j]
        grad /= len(active)

        # Subgradient of J = 2n*b_max/S^2:
        # dJ/da = 2n*(grad/S - 2*b_max/S^2) ... (simplified, ignoring that b_max changes)
        # Use normalized gradient for direction
        grad_dir = grad * S - 2.0 * b_max * np.ones(n)
        grad_dir = grad_dir / (np.linalg.norm(grad_dir) + 1e-10)

        # Projected gradient step
        a_new = a - step * grad_dir
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            step *= 0.5
            continue
        a_new /= s_new

        J_new = compute_J(a_new)
        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            step = min(step * 1.05, 0.15)
        else:
            step *= 0.7

        if it % 10 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def grow_sequence(a, new_n, rng):
    """Grow sequence by inserting elements, maintaining smoothness."""
    n = len(a)
    if new_n <= n:
        return resample_seq(a, new_n)
    # Interpolate and add small noise
    a_new = resample_seq(a, new_n)
    a_new += rng.normal(0, 0.01, new_n)
    a_new = np.maximum(a_new, 0.001)
    a_new /= a_new.sum()
    return a_new


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Hybrid optimizer: LP mirror descent + log-sum-exp L-BFGS-B + subgradient refine.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128)
        tracker.update(parent)

    # === Phase 1: LP Mirror Descent (start from parent, n=128) ===
    t0 = time.monotonic()
    n1 = parent.size if parent.size <= 512 else 256
    print(f"\n{'='*50} Phase 1: LP n={n1} {'='*50}", flush=True)

    a = resample_seq(parent, n1)
    a = np.maximum(a, 0.001)
    a /= a.sum()

    # Run LP for a few hundred iterations
    lp_deadline = min(deadline, time.monotonic() + 250)
    a = run_lp_iterations(a, 400, lp_deadline, tracker, verbose=True)
    print(f"  Phase 1 done: J={tracker.best_score:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 2: Log-sum-exp L-BFGS-B refinement ===
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 2: L-BFGS-B logsumexp {'='*50}", flush=True)

        # Try multiple tau schedules
        for attempt in range(3):
            if time.monotonic() > deadline:
                break
            a_start = tracker.best.copy() if tracker.best is not None else a
            if attempt > 0:
                # Add noise for diversity
                noise = rng.normal(0, 0.03 * (attempt + 1), a_start.size)
                a_start = np.maximum(a_start + noise, 0.001)
                a_start /= a_start.sum()

            a = run_logsumexp_lbfgs(a_start, deadline, tracker,
                                    tau_start=5.0 * (attempt + 1),
                                    tau_end=0.005)

        print(f"  Phase 2 done: J={tracker.best_score:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 3: Subgradient refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 3: Subgradient refine {'='*50}", flush=True)
        a = run_subgradient_refine(tracker.best, 500, deadline, tracker)
        print(f"  Phase 3 done: J={tracker.best_score:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 4: LP polish on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        if n_best <= 1024:
            print(f"\n{'='*50} Phase 4: LP polish n={n_best} {'='*50}", flush=True)
            a = run_lp_iterations(tracker.best, 200, deadline, tracker)
            print(f"  Phase 4 done: J={tracker.best_score:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 5: Try different lengths ===
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 5: Length exploration {'='*50}", flush=True)

        base = tracker.best if tracker.best is not None else np.ones(128)
        for target_n in [128, 192, 256, 384, 512]:
            if time.monotonic() > deadline:
                break
            if target_n == base.size:
                continue

            a_new = resample_seq(base, target_n)
            a_new = np.maximum(a_new + rng.normal(0, 0.02, target_n), 0.001)
            a_new /= a_new.sum()

            # Quick LP + L-BFGS-B
            quick_deadline = min(deadline, time.monotonic() + 80)
            a_new = run_lp_iterations(a_new, 100, quick_deadline, tracker)
            if time.monotonic() < deadline:
                a_new = run_logsumexp_lbfgs(a_new, deadline, tracker,
                                            tau_start=5.0, tau_end=0.01)

        print(f"  Phase 5 done: J={tracker.best_score:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 6: Final multi-method polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 6: Final polish {'='*50}", flush=True)

        # One more round of LP
        if tracker.best.size <= 512:
            a = run_lp_iterations(tracker.best, 150, deadline, tracker)

        # One more L-BFGS-B with very small tau
        if time.monotonic() < deadline and tracker.best is not None:
            a = run_logsumexp_lbfgs(tracker.best, deadline, tracker,
                                    tau_start=1.0, tau_end=0.001)

        # Subgradient
        if time.monotonic() < deadline and tracker.best is not None:
            a = run_subgradient_refine(tracker.best, 300, deadline, tracker)

        print(f"  Phase 6 done: J={tracker.best_score:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
```
