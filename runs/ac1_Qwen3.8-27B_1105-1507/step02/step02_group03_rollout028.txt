def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    return float(2.0 * n * conv.max() / (s * s))


def convolve_fast(f, g):
    n = len(f) + len(g) - 1
    n_fft = 1 << (n - 1).bit_length()
    F = np.fft.rfft(f, n_fft)
    G = np.fft.rfft(g, n_fft)
    return np.fft.irfft(F * G, n_fft)[:n]


def pressure_step(f, beta, eta):

def lp_direction(f, max_time=5.0):
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()

    # Build constraint matrix A where A[k,j] = f[k-j] if valid
    A = np.zeros((2 * n - 1, n))

        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = f[k - j]

    if g_sum < 1e-10:

    eta = eta0
    beta = beta0
    stall = 0

    while time.monotonic() - start < time_budget:
        f_new, M, p = pressure_step(f, beta, eta)
        score = compute_score(f_new)

        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            eta = min(eta * 1.03, 0.8)
            stall = 0
        else:
            eta *= 0.85
            stall += 1
            # Anneal beta downward to consider more of the convolution
            beta = max(5.0, beta * 0.99)

            if stall > 50:
                # Perturb
                f_pert = f * (1.0 + 0.08 * rng.standard_normal(n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                s_pert = compute_score(f_pert)
                if s_pert < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = s_pert
                    eta = eta0 * 0.5
                    stall = 0
                elif stall > 150:
                    break

    return best_f, best_score


def optimize_lp(f, time_budget, rng, t0=0.3):
    n = len(f)
    f = normalize(f)
    best_f = f.copy()
    best_score = compute_score(f)
    start = time.monotonic()
    t = t0
    stall = 0
    lp_count = 0

    while time.monotonic() - start < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 3:
            break

        lp_time = min(4.0, remaining * 0.3)
        g = lp_direction(f, max_time=lp_time)
        lp_count += 1

        if g is None:
            break

        # Try multiple step sizes
        improved = False
        for step in [min(1.0, t * 1.5), t, t * 0.5, t * 0.2, t * 0.1]:
            f_new = (1.0 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            score = compute_score(f_new)

            if score < best_score - 1e-12:
                best_score = score
                best_f = f_new.copy()
                f = f_new
                t = min(1.0, t * 1.1)
                improved = True
                break

        if not improved:
            t *= 0.7
            stall += 1
            if stall > 10:
                # Perturb and reset
                f_pert = f + 0.03 * rng.standard_normal(n)
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                s_pert = compute_score(f_pert)
                if s_pert < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = s_pert
                    t = t0
                    stall = 0
                else:
                    break

    print(f"    LP refinement: {lp_count} LPs, score={best_score:.6f}")
    return best_f, best_score


def coordinate_descent(f, n_sweeps=10, rng=None):
    n = len(f)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    best_score = compute_score(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n) if rng else np.arange(n)
        for i in order:
            original = f[i]
            if original < 1e-8:
                continue
            best_val = original
            best_local = best_score

            for factor in [0.3, 0.5, 0.7, 0.85, 1.18, 1.43, 2.0, 3.33]:
                f[i] = max(0.0, original * factor)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                score = 2.0 * n * np.convolve(f, f).max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]

            f[i] = best_val
            if best_local < best_score:
                best_score = best_local

        f = np.clip(f, 0, None)
        f = normalize(f)

    return f, best_score


def simulated_annealing(f, rng, n_iter=5000, T0=0.02, Tend=0.00005):
    n = len(f)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    best_f = f.copy()
    best_score = compute_score(f)
    cur_score = best_score
    no_improve = 0

    for it in range(n_iter):
        T = T0 * (Tend / T0) ** (it / max(1, n_iter))
        f_new = f.copy()
        r = rng.random()

        if r < 0.25:
            # Block scaling
            width = max(1, int(rng.integers(1, max(2, n // 16))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.3))
        elif r < 0.45:
            # Single element
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] * np.exp(rng.normal(0, 0.3)))
        elif r < 0.65:
            # Smooth Gaussian perturbation
            center = int(rng.integers(0, n))
            width = max(2, int(n // 32))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (width ** 2))
            f_new += rng.normal(0, 0.05) * kernel
        elif r < 0.85:
            # Two-element swap
            i, j = rng.integers(0, n, 2)
            if i != j:
                f_new[i], f_new[j] = f_new[j], f_new[i]
        else:
            # Global reweight
            f_new *= np.exp(rng.normal(0, 0.2, n))

        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score(f_new)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > max(500, n_iter // 3) and it > n_iter // 2:
            break

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0

    best_overall = None
    best_score_overall = np.inf

    # Try parent sequence
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None

    if parent is not None:
        parent = np.clip(parent, 0, None)
        if parent.sum() > 0.01:
            parent_n = normalize(parent)
            parent_score = compute_score(parent_n)
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
            best_overall = parent_n.copy()
            best_score_overall = parent_score

    # Lengths to try
    n_parent = len(parent) if parent is not None else 512
    lengths = sorted(set([n_parent, 256, 512, 1024]))

    print(f"Lengths to try: {lengths}")
    print(f"Initial best: {best_score_overall:.6f}, target: 1.5027")

    for n in lengths:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break

        print(f"\n=== n={n} (remaining: {remaining:.0f}s) ===")

        # Multiple initializations
        inits = []

        # U-shaped with various parameters
        for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
            for ew in [0.5, 1.0, 2.0, 4.0]:
                inits.append(u_shape_init(n, alpha=alpha, edge_weight=ew, rng=rng))

        # Random smooth
        for _ in range(3):
            inits.append(random_smooth_init(n, rng))

        # If parent matches this length, include it
        if parent is not None and len(parent) == n:
            inits.insert(0, normalize(parent))
            # Perturbed parent
            for scale in [0.05, 0.1, 0.2]:
                fp = normalize(parent) * (1.0 + scale * rng.standard_normal(n))
                fp = np.clip(fp, 0, None)
                inits.append(normalize(fp))

        # Quick screen: pressure-optimize each init briefly, keep best 3
        screen_time = min(10.0, remaining * 0.15 / max(1, len(inits)))
        print(f"  Screening {len(inits)} inits, {screen_time:.1f}s each")

        screened = []
        for i, f0 in enumerate(inits):
            if time.monotonic() > deadline - 60:
                break
            f1, s1 = optimize_pressure(f0, time_budget=screen_time, rng=rng,
                                       eta0=0.2, beta0=300.0)
            screened.append((s1, f1))
            if (i + 1) % 8 == 0:
                print(f"    Screened {i+1}/{len(inits)}, best so far: {min(s for s,_ in screened):.6f}")

        screened.sort(key=lambda x: x[0])
        top_k = screened[:min(3, len(screened))]
        print(f"  Top init score: {top_k[0][0]:.6f}")

        # Detailed optimization on top candidates
        n_candidates = min(3, len(top_k))
        time_per_candidate = min(120.0, max(30.0, (remaining - 30) / n_candidates))

        for idx, (s_init, f_init) in enumerate(top_k):
            remaining = deadline - time.monotonic()
            if remaining < 40:
                break

            t_alloc = min(time_per_candidate, remaining - 30)
            print(f"  Candidate {idx+1}: init_score={s_init:.6f}, budget={t_alloc:.0f}s")

            # Phase 1: Extended pressure optimization
            t_press = t_alloc * 0.35
            f_p, s_p = optimize_pressure(f_init, time_budget=t_press, rng=rng,
                                         eta0=0.1, beta0=100.0)
            if s_p < best_score_overall:
                best_score_overall = s_p
                best_overall = f_p.copy()
                print(f"    After pressure: {s_p:.6f}")

            # Phase 2: LP refinement
            remaining = deadline - time.monotonic()
            t_lp = min(t_alloc * 0.45, max(20.0, remaining - 30))
            if t_lp > 15:
                f_l, s_l = optimize_lp(f_p, time_budget=t_lp, rng=rng, t0=0.4)
                if s_l < best_score_overall:
                    best_score_overall = s_l
                    best_overall = f_l.copy()
                    print(f"    After LP: {s_l:.6f}")
                f_p = f_l

            # Phase 3: Coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 15:
                n_sweeps = max(5, int(min(15, remaining * 0.2)))
                f_cd, s_cd = coordinate_descent(f_p, n_sweeps=n_sweeps, rng=rng)
                if s_cd < best_score_overall:
                    best_score_overall = s_cd
                    best_overall = f_cd.copy()
                    print(f"    After CD: {s_cd:.6f}")
                f_p = f_cd

            # Phase 4: Simulated annealing
            remaining = deadline - time.monotonic()
            if remaining > 20:
                n_iter = min(6000, int(remaining * 250))
                f_sa, s_sa = simulated_annealing(f_p, rng, n_iter=n_iter, T0=0.015, Tend=0.00003)
                if s_sa < best_score_overall:
                    best_score_overall = s_sa
                    best_overall = f_sa.copy()
                    print(f"    After SA: {s_sa:.6f}")

    # Final global polishing
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        print(f"\n=== Final polishing (remaining: {remaining:.0f}s) ===")

        # CD
        if remaining > 20:
            n_sweeps = max(8, int(min(20, remaining * 0.25)))
            f_cd, s_cd = coordinate_descent(best_overall, n_sweeps=n_sweeps, rng=rng)
            if s_cd < best_score_overall:
                best_score_overall = s_cd
                best_overall = f_cd.copy()
                print(f"  Final CD: {s_cd:.6f}")

        # More SA
        remaining = deadline - time.monotonic()
        if remaining > 20:
            n_iter = min(10000, int(remaining * 300))
            f_sa, s_sa = simulated_annealing(best_overall, rng, n_iter=n_iter,
                                             T0=0.01, Tend=0.00001)
            if s_sa < best_score_overall:
                best_score_overall = s_sa
                best_overall = f_sa.copy()
                print(f"  Final SA: {s_sa:.6f}")

        # One more CD
        remaining = deadline - time.monotonic()
        if remaining > 10:
            f_cd2, s_cd2 = coordinate_descent(best_overall, n_sweeps=5, rng=rng)
            if s_cd2 < best_score_overall:
                best_score_overall = s_cd2
                best_overall = f_cd2.copy()
                print(f"  Final CD2: {s_cd2:.6f}")

        # Final pressure pass with low beta
        remaining = deadline - time.monotonic()
        if remaining > 15:
            f_pr, s_pr = optimize_pressure(best_overall, time_budget=remaining * 0.5,
                                           rng=rng, eta0=0.05, beta0=20.0)
            if s_pr < best_score_overall:
                best_score_overall = s_pr
                best_overall = f_pr.copy()
                print(f"  Final pressure: {s_pr:.6f}")

    # Verify and return
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\nFINAL: score={final_score:.6f}, n={len(final_list)}, target=1.5027")
        return final_list

    # Fallback
    print("Using fallback: U-shaped n=512")
    f = u_shape_init(512, alpha=2.0, edge_weight=2.0, rng=rng)
    f, s = optimize_pressure(f, time_budget=30, rng=rng)
    print(f"  Fallback score: {s:.6f}")
    return [float(x) for x in np.clip(f, 0, 1000)]
