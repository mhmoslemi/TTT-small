```python
def soft_subgradient_descent(f, eta, tau=0.01, n_iter=1000):
    n = len(f)
    for _ in range(n_iter):
        conv = np.convolve(f, f)  # length 2n-1
        M = conv.max()
        
        # Soft weights (softmax with temperature tau)
        w = np.exp((conv - M) / tau)
        w /= w.sum()
        
        # Gradient
        grad = 2.0 * np.correlate(w, f, mode='valid')  # length n
        
        # Projected gradient step (project onto simplex)
        f_new = f - eta * grad
        # Project onto {x >= 0, sum(x) = 1}
        f_new = project_simplex(f_new)
        
        f = f_new
    
    return f
```

For a sequence f of length n, the convolution has 2n-1 values. The constraint "conv[k] <= M for all k" defines a convex set in the space of M (for fixed f, it's just M >= max conv). But in the space of f, it's non-convex.

However, if we fix n-1 elements and optimize the nth, the constraint becomes a set of quadratic constraints on a single variable, which is solvable!

Specifically, if we fix f[0], ..., f[n-2] and vary f[n-1] = x, then:
conv[k] = sum_{i=0}^{n-2} f[i]*f[k-i] + f[n-1]*f[k-(n-1)] (if k-(n-1) is valid) + f[k-(n-1)]*f[n-1] (if k-(n-1) is valid)
= (known part) + x * f[k-(n-1)] * 2 (if k-(n-1) != n-1) or x^2 (if k-(n-1) == n-1, i.e., k = 2n-2)

So for each k, conv[k] as a function of x is:
- Linear in x (if k != 2n-2 and k != n-1): conv[k] = c_k + x * d_k where d_k = 2*f[k-(n-1)] (if k-(n-1) in [0,n-2]) or f[k-(n-1)] (if k-(n-1) = n-1, which is k=2n-2)
- Quadratic in x (if k = 2n-2): conv[2n-2] = x^2
- If k = n-1: conv[n-1] = c + x*f[0] (since k-(n-1) = 0, and the other term f[n-1]*f[0] = x*f[0])
  Actually: conv[n-1] = sum_{i=0}^{n-1} f[i]*f[n-1-i]. The terms involving x = f[n-1]:
  - i = n-1: f[n-1]*f[0] = x*f[0]
  - i = 0: f[0]*f[n-1] = f[0]*x
  So conv[n-1] = (sum_{i=1}^{n-2} f[i]*f[n-1-i]) + 2*x*f[0]
  
  For k = 2n-2: conv[2n-2] = f[n-1]*f[n-1] = x^2

So the constraint max_k conv[k] <= M becomes:
- For each k, c_k + x*d_k <= M (linear in x) or x^2 <= M (quadratic)
- This gives: x <= (M - c_k)/d_k for d_k > 0, x >= (M - c_k)/d_k for d_k < 0
- And |x| <= sqrt(M)

So the feasible set for x is an interval [lo, hi], and we want to maximize x (to maximize sum f, equivalently to minimize the score for fixed other elements... wait, no.

Actually, the score is 2n*max_conv / sum^2. If we fix all elements except x = f[n-1], then:
- sum = S + x where S = sum_{i=0}^{n-2} f[i]
- max_conv = max(M_known, x^2, max_k(c_k + x*d_k))

This is a piecewise function of x. We want to minimize 2n * max_conv / (S+x)^2.

This is a 1D optimization problem that can be solved by ternary search or by finding the minimum of a piecewise function.

Actually, for coordinate descent, we just need to find the x that minimizes the score. Since the score as a function of x is unimodal (or at least has a manageable structure), we can use a combination of:
1. Grid search to bracket the minimum
2. Ternary search to refine

Let me think about the structure more carefully:
- For small x: max_conv is dominated by the "known" part (the part not involving x), so max_conv ≈ M_known, and score ≈ 2n*M_known/(S+x)^2, which decreases with x.
- For large x: max_conv is dominated by x^2 or linear terms, and score ≈ 2n*x^2/(S+x)^2 ≈ 2n for large x, or 2n*x*d/(S+x)^2 for the linear terms.

So the score first decreases and then increases, giving a unimodal (or nearly unimodal) function. We can find the minimum with ternary search.

```python
import numpy as np
import time

def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s

def score_of(f):
    """Compute score for a normalized (sum=1) non-negative sequence."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()

def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def soft_gradient_direction(f, tau):
    """Compute the soft subgradient direction for the peak autocorrelation."""
    n = len(f)
    conv = np.convolve(f, f)
    M = conv.max()
    w = np.exp((conv - M) / tau)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    # Direction: correlate(w, f, mode='valid')
    direction = np.correlate(w, f, mode='valid')
    return direction, M, w

def soft_subgrad_descent(f, time_budget=30, tau_init=None, rng=None):
    """Soft subgradient descent with adaptive step size and temperature."""
    start = time.monotonic()
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    
    conv = np.convolve(f, f)
    M = conv.max()
    if tau_init is None:
        tau_init = M * 0.05
    
    best_f = f.copy()
    best_score = score_of(f)
    tau = tau_init
    eta = 1.0 / n  # Initial step size (scaled by 1/n since direction values are ~1/n)
    
    no_improve = 0
    while time.monotonic() - start < time_budget:
        direction, M, w = soft_gradient_direction(f, tau)
        current_score = score_of(f)
        
        improved = False
        for eta_try in [eta, eta * 2, eta * 0.5, eta * 0.25]:
            f_new = f - eta_try * direction
            f_new = project_simplex(f_new)
            new_score = score_of(f_new)
            if new_score < current_score - 1e-14:
                f = f_new
                if new_score < best_score:
                    best_score = new_score
                    best_f = f.copy()
                eta = eta_try * 1.1
                improved = True
                break
        
        if not improved:
            eta *= 0.5
            no_improve += 1
            if no_improve > 20:
                # Decrease temperature and reset step size
                tau *= 0.5
                eta = 1.0 / n
                no_improve = 0
                if tau < M * 0.001:
                    break
        
        if tau < M * 0.001:
            # Very low temperature, switch to coordinate descent
            break
    
    return best_f, best_score

def coordinate_descent(f, n_sweeps=5, rng=None, time_budget=None):
    """Coordinate descent: optimize one element at a time."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = score_of(f)
    start = time.monotonic() if time_budget else None
    
    for sweep in range(n_sweeps):
        if time_budget and time.monotonic() - start > time_budget:
            break
        order = rng.permutation(n) if rng else np.arange(n)
        for i in order:
            if time_budget and time.monotonic() - start > time_budget:
                break
            # Find best value for f[i]
            old_val = f[i]
            if old_val < 1e-12:
                continue
            
            # Try a range of values
            # The optimal value is usually near the current value
            candidates = [old_val * 0.3, old_val * 0.5, old_val * 0.7, old_val * 0.85,
                         old_val, old_val * 1.15, old_val * 1.3, old_val * 1.6, old_val * 2.0]
            # Also try 0
            candidates.append(0.0)
            
            best_local = best_score
            best_val = old_val
            
            for x in candidates:
                f[i] = x
                s = f.sum()
                if s < 1e-10:
                    continue
                f_norm = f / s
                sc = score_of(f_norm)
                if sc < best_local:
                    best_local = sc
                    best_val = x
            
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
        
        f = normalize(f)
    
    return f, best_score

def spectral_init(n, rng):
    """Generate initial sequence using spectral method."""
    # Start with flat spectrum
    n_fft = 1 << (2 * n - 1).bit_length()
    F = np.zeros(n_fft // 2 + 1, dtype=complex)
    F[0] = 1.0  # DC
    # Add random phases
    phases = rng.uniform(0, 2*np.pi, n_fft // 2)
    F[1:] = np.exp(1j * phases) / np.sqrt(n_fft // 2)
    f = np.fft.irfft(F, n_fft)[:n]
    f = np.clip(f, 0, None)
    if f.sum() < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f.sum()
    return f

def diverse_init(n, rng):
    """Generate diverse initial sequence."""
    r = rng.random()
    if r < 0.3:
        # Product of sines
        f = np.ones(n)
        for freq in rng.uniform(1, 5, 3):
            f *= 0.5 + 0.5 * np.cos(2 * np.pi * freq * np.arange(n) / n)
        f += 0.1
    elif r < 0.6:
        # Gaussian bumps
        f = np.zeros(n)
        for _ in range(rng.integers(2, 6)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            f += np.exp(-0.5 * (np.arange(n) - center)**2 / width**2)
    else:
        # Cosine window with modulation
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
        f *= 1 + 0.3 * np.sin(2 * np.pi * rng.uniform(2, 8) * np.arange(n) / n)
    
    f = np.clip(f, 0.01, None)
    f = f / f.sum()
    f = f * (1 + 0.1 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    f = f / f.sum()
    return f
```

**Key insight for coordinate descent**: Instead of trying a fixed set of candidate values, we can analytically find the optimal value.

For a fixed i, with all other elements fixed, the score as a function of x = f[i] is:
S(x) = 2n * max_k (base_k + x * d_k + x^2 * e_k) / (S_0 + x)^2

where:
- base_k = conv[k] - 2*f[i]*f[k-i] - f[i]^2*e_k (the part not involving f[i])
- d_k = 2*f[k-i] (if k-i valid and k ≠ 2i), 0 otherwise
- e_k = 1 if k = 2i, 0 otherwise
- S_0 = 1 - f[i] (sum of all elements except f[i])

The function S(x) is a ratio of (piecewise quadratic) to (quadratic). Its minimum can be found by:
1. Identifying the "active" k (the one that achieves the max)
2. For each possible active k, finding the x that minimizes (base_k + x*d_k + x^2*e_k) / (S_0 + x)^2
3. Taking the minimum over all active k

For a fixed active k, the function (a + bx + cx^2) / (S_0 + x)^2 has its minimum where the derivative is zero:
d/dx [(a + bx + cx^2) / (S_0 + x)^2] = 0

This gives: (b + 2cx)(S_0 + x) - 2(a + bx + cx^2) = 0
=> (b + 2cx)(S_0 + x) = 2(a + bx + cx^2)
=> b*S_0 + bx + 2cx*S_0 + 2cx^2 = 2a + 2bx + 2cx^2
=> b*S_0 + bx + 2cx*S_0 = 2a + 2bx
=> b*S_0 + 2cx*S_0 = 2a + bx
=> x(2c*S_0 - b) = 2a - b*S_0
=> x = (2a - b*S_0) / (2c*S_0 - b)

where a = base_k, b = d_k, c = e_k.

If c = 0 (k ≠ 2i):
x = (2*base_k - d_k*S_0) / (-d_k) = (d_k*S_0 - 2*base_k) / d_k (if d_k ≠ 0)

Note: d_{2i} = 0 (since for k=2i, the linear term is 0, only the quadratic term x^2 is present)

But for the coordinate descent, the issue is that we need to evaluate the score for many candidate x values for each i. Each evaluation requires computing the new max of the convolution.

Actually, we can avoid recomputing the full convolution. We just need to update the parts that change:

When f[i] changes from old to new:
- conv[k] changes by: (new - old) * d[k] + (new^2 - old^2) * e[k]
- where d[k] = 2*f[k-i] (for k ≠ 2i, with k-i valid)
- and e[k] = 1 for k = 2i

So the new convolution is: conv_new[k] = conv_old[k] + (new-old)*d[k] + (new^2-old^2)*e[k]

This is an O(n) update per candidate x! So the total per coordinate is O(n * n_candidates) = O(n^2) in the worst case, but typically much less since we only need to find the max.

Even better: we can analytically determine the new max without iterating over all k for each candidate. The new conv is:
conv_new[k] = conv_old[k] + (x - fi) * d[k] (for k ≠ 2i)
conv_new[2i] = conv_old[2i] + (x^2 - fi^2)

The max of conv_new is the max of:
- max_{k≠2i} (conv_old[k] + (x-fi)*d[k])
- conv_old[2i] + (x^2 - fi^2)

The first term is the max of a set of linear functions of x, and the second is a quadratic.

The max of linear functions is a convex piecewise-linear function. We can find its value at specific x points, but finding the overall minimum of the ratio is still the hard part.

For practical purposes, let me just evaluate the score at a modest number of candidate x values (say 10-20) per coordinate. With O(n) evaluation per candidate (for the max), that's O(n * 20) per coordinate, or O(n^2 * 20) per sweep. For n=512: 512^2 * 20 ≈ 5.2M per sweep. At 100M ops/sec, that's about 0.05s per sweep, so we can do many sweeps.

Actually, computing the max of conv_new is O(n) (just iterate over all k and find the max). So per coordinate: O(n_candidates * n) = O(20 * 512) ≈ 10K operations. Per sweep: n * 10K = 512 * 10K ≈ 5M operations. Very fast!

import numpy as np
import time

def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s

def score_of(f):
    """Score for sum=1 non-negative sequence."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    if v.sum() <= 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def soft_subgrad(f, time_budget, tau_init=None, rng=None):
    """Soft subgradient descent."""
    start_t = time.monotonic()
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    
    conv0 = np.convolve(f, f)
    M0 = conv0.max()
    if tau_init is None:
        tau_init = M0 * 0.05
    
    best_f = f.copy()
    best_score = score_of(f)
    tau = tau_init
    eta = 1.0
    no_improve = 0
    
    while time.monotonic() - start_t < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        
        w = np.exp((conv - M) / tau)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        
        # Direction = correlate(w, f, 'valid')
        direction = np.correlate(w, f, mode='valid')
        
        current_score = score_of(f)
        improved = False
        
        for eta_try in [eta, eta * 0.5, eta * 2.0, eta * 0.25]:

            f_new = project_simplex(f_new)
            new_score = score_of(f_new)
