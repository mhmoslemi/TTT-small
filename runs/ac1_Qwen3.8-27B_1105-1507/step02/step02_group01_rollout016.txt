def compute_obj(a):
    """Compute the objective function value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def softmax_grad_step(a, T, lr):
    """Perform one soft-max gradient descent step. Returns new a and objective."""
    n = len(a)
    conv = np.convolve(a, a)
    max_conv = float(np.max(conv))
    
    # Soft-max weights over convolution coefficients
    shifted = (conv - max_conv) / T
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    
    # Gradient: g[i] = 2 * sum_j w[i+j] * a[j]
    g = 2.0 * correlate(w, a, mode='valid')
    
    # Update with projection
    a_new = a - lr * g
    a_new = np.maximum(0.0, a_new)
    s = a_new.sum()
    if s < 1e-15:
        return a, max_conv
    a_new = a_new / s
    return a_new, max_conv

def build_A_matrix(a):
    """Build the A matrix for LP: A[k,j] = a[k-j] if valid, else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)

        if j_end >= j_start:

    return A

def diverse_init(n, method, rng):
    """Generate diverse initializations."""
    if method == 0:
        # Uniform
        a = np.ones(n)
    elif method == 1:
        # Geometric decay
        r = rng.uniform(0.9, 0.995)
        a = r ** np.arange(n)
        # Add some at the other end
        a[n-1:] = max(a[n-1:], 0.1 * a[0])
    elif method == 2:
        # Triangular (high in middle)
        center = n / 2.0
        a = 1.0 - np.abs(np.arange(n) - center) / center
        a = np.maximum(0.1, a)
    elif method == 3:
        # Cosine pattern
        phase = rng.uniform(0, 2 * np.pi)
        freq = rng.uniform(0.3, 1.5)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        a = np.maximum(0.01, a)
    elif method == 4:
        # Two-point concentration
        a = np.ones(n) * 0.01
        frac = rng.uniform(0.2, 0.8)
        a[int(frac * (n - 1))] = 1.0
        a[int((1 - frac) * (n - 1))] = 1.0
    elif method == 5:
        # Exponential bump
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        a += 0.05
    elif method == 6:
        # Random smooth (low-pass filtered noise)
        a = rng.random(n)

    s = a.sum()
    if s < 1e-15:

        a = a / s
