    # = sum_{k=i}^{i+n-1} w[k] * a[k-i]  (since a[k-i] is nonzero only for 0<=k-i<n)

    # = (convolve(w, a))[i + n - 1] ... let me verify.

    # OK let me just use np.correlate.
    # np.correlate(w, a, mode='full')[l] = sum_j w[j+l] * a[j] (for appropriate l)

    # Let me just do it directly but in a semi-vectorized way.
    
    # For the first sum: S1[i] = sum_k w[k] * a[k-i] for i = 0, ..., n-1
    # = cross-correlation of w and a at lag i
    # For the second sum: S2[i] = sum_k w[k] * a[i-k] for i = 0, ..., n-1
    # = cross-correlation of a and w at lag i

    # S1[i] = sum_k w[k]*a[k-i]. Let j = k-i: = sum_j w[j+i]*a[j]

    # But if w is symmetric (w[k] = w[2n-2-k]), then S1[i] = S2[n-1-i]...

    # Then take the appropriate slice.

    # Contribution of lag k to grad[i] is: w[k] * (a[k-i] if 0<=k-i<n else 0 + a[i-k] if 0<=i-k<n else 0)

    # Which is: grad = cross_correlation(w, a) + cross_correlation(a, w) (appropriately sliced)

    # np.correlate(w, a, mode='full')[m] = sum_j w[j] * a[j+m-len(a)+1] ... 
    
    # Ugh, the indexing of np.correlate is confusing. Let me just verify with a small example in my head.
    # w = [w0, w1, w2], a = [a0, a1, a2]

    # k=0: sum_n a[n]*v[n-0+2] (offset by len(v)-1)...

    # approach that I can verify is correct.
    
    # Simple approach: for each i, compute grad[i] directly.
    # grad[i] = sum_k w[k] * (a[k-i] if 0<=k-i<n else 0) + sum_k w[k] * (a[i-k] if 0<=i-k<n else 0)

    # First_sum[i] = sum_{k=i}^{min(i+n-1, 2n-2)} w[k] * a[k-i]
    #               = sum_{j=0}^{min(n-1, 2n-2-i)} w[i+j] * a[j]

    # First_sum[i] = dot(a[:n], w[i:i+n])  (with appropriate truncation)
    # Second_sum[i] = dot(a[:min(i+1,n)], w[max(0,i-n+1):i+1][::-1])

    # even a simple loop over i (1024 iterations, each doing up to 1024 multiplications)
    # is only ~1M operations total, which is fast.
    
    # Let me use a semi-vectorized approach:
    # First_sum = np.array([np.dot(a, w[i:min(i+n, 2*n-1)]) for i in range(n)])
    # Second_sum = np.array([np.dot(a[:min(i+1,n)], w[max(0,i-n+1):i+1][::-1]) for i in range(n)])

    # This is the cross-correlation of a and w_padded, evaluated at lag i (with w shifted right by i)
    # = np.correlate(w_padded, a, 'full')[i + (2n-1) - 1] ... 
    
    # OK let me just use the FFT approach directly.
    # grad[i] = sum_k w[k] * a[k-i] + sum_k w[k] * a[i-k]
    # 
    # Define the DFT: W[k] = FFT(w_padded), A[k] = FFT(a_padded)
    # where w_padded has length 2n-1 (or next power of 2) and a_padded has length n (or same).
    #
    # The cross-correlation R[i] = sum_k w[k] * a[k-i] can be computed as:
    # R = IFFT(FFT(w) * conj(FFT(a)))
    # (with appropriate zero-padding to avoid circular correlation)
    
    # Let me use a length that's at least (2n-1) + n - 1 = 3n-2 for the FFT.
    # For n=1024, that's 3070. Next power of 2 is 4096.

    W = np.fft.rfft(weights, n=fft_len)
    A = np.fft.rfft(a, n=fft_len)

    # This gives a vector of length fft_len//2+1 (for rfft) or fft_len (for full fft)

    # Second term: sum_k w[k] * a[i-k] = sum_k a[k] * w[i-k] (substituting j=i-k)
    # = IFFT(A * conj(W))[i] (by symmetry of cross-correlation)
    # Actually: sum_k w[k] * a[i-k] = sum_j a[j] * w[i-j] = cross_correlation(a, w)[i]
    # = IFFT(conj(A) * W)[i] = conj(IFFT(W * conj(A))[-i mod N])... 
    
    # For real signals: cross_correlation(a, w)[i] = cross_correlation(w, a)[-i]
    # So: sum_k w[k] * a[i-k] = cross1[-i mod fft_len] = cross1[fft_len - i] for i > 0, cross1[0] for i=0
    
    # Actually for real signals, the cross-correlation is symmetric:
    # R_wa[i] = sum_k w[k]*a[k-i]
    # R_aw[i] = sum_k a[k]*w[k-i] = sum_j w[j]*a[j+i] (substituting j=k-i)
    # Hmm, R_aw[i] = sum_k a[k]*w[k-i]. Let j=k-i: = sum_j a[j+i]*w[j] = sum_j w[j]*a[j+i]
    # And R_wa[i] = sum_k w[k]*a[k-i]. Let j=k-i: = sum_j w[j+i]*a[j] = sum_j a[j]*w[j+i]
    # So R_aw[i] = R_wa[-i] (for the cross-correlation).
    # Therefore: sum_k w[k]*a[i-k] = R_aw[i] = R_wa[-i] = cross1[fft_len - i] (for i>0)
    
    # So: grad[i] = cross1[i] + cross1[(fft_len - i) % fft_len]
    
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = cross1[i] + cross1[(fft_len - i) % fft_len]
    
    return grad, M


def softmax_gradient_fast(a, beta=50.0):
    """Compute gradient using FFT-based cross-correlation."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    
    # Softmax weights (numerically stable)
    shifted = conv / max(M, 1e-15) - 1.0  # In [-1, 0]
    weights = np.exp(beta * shifted)
    weights = weights / np.sum(weights)
    
    # FFT-based cross-correlation
    fft_len = 4096  # Must be >= 2n-1 + n - 1 = 3n-2 = 3070 for n=1024
    W = np.fft.rfft(weights, n=fft_len)
    A = np.fft.rfft(a, n=fft_len)
    cross = np.fft.irfft(W * np.conj(A), n=fft_len)
    
    # grad[i] = cross[i] + cross[(-i) mod fft_len]
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = cross[i] + cross[(fft_len - i) % fft_len]
    
    return grad, M, conv


def simplex_project(v, target_sum):
    """Project v onto the simplex {x : x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    
    # Offset method for simplex projection
    mu = np.full(n, 0.0)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def line_search(a, direction, grad, M_init, beta_sm=50.0, max_iter=20):
    """Backtracking line search."""
    n = len(a)
    S = float(np.sum(a))
    score = compute_score_fast(a)
    
    alpha = 1.0
    for _ in range(max_iter):
        new_a = a - alpha * direction
        new_a = simplex_project(new_a, S)
        new_score = compute_score_fast(new_a)
        if new_score < score:
            return new_a, new_score
        alpha *= 0.5
        if alpha < 1e-8:
            break
    return a, score


def optimize_sequence(a_init, deadline, verbose_prefix=""):
    """Main optimization loop with softmax gradient, line search, and restarts."""
    a = normalize(a_init.copy())
    n = len(a)
    S = float(np.sum(a))
    best_a = a.copy()
    best_score = compute_score_fast(a)
    
    print(f"  {verbose_prefix}Initial score: {best_score:.6f}")
    
    no_improve = 0
    it = 0
    beta = 20.0  # Start with moderate softmax sharpness
    base_lr = 0.1
    
    while time.monotonic() < deadline:
        # Compute gradient
        grad, M, conv = softmax_gradient_fast(a, beta=beta)
        
        if np.linalg.norm(grad) < 1e-12:
            break
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        grad = grad / grad_norm
        
        # Try a few step sizes
        improvements = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1]:
            lr = base_lr * lr_mult / n  # Scale by 1/n for stability
            new_a = a - lr * grad
            new_a = simplex_project(new_a, S)
            new_score = compute_score_fast(new_a)
            if new_score < best_score - 1e-10:
                a = new_a
                best_a = a.copy()
                best_score = new_score
                no_improve = 0
                improvements = True
                break
            elif new_score < best_score:
                a = new_a
                best_a = a.copy()
                best_score = new_score
                no_improve = 0
                improvements = True
        
        if not improvements:
            no_improve += 1
            # Increase beta to focus more on the peak
            beta = min(beta * 1.1, 200.0)
            base_lr *= 0.95
            
            if no_improve % 20 == 0 and no_improve < 200:
                # Random perturbation to escape local minimum
                perturbation = np.random.randn(n) * 0.01 * S / n
                perturbed = a + perturbation
                perturbed = simplex_project(perturbed, S)
                pert_score = compute_score_fast(perturbed)
                if pert_score < best_score:
                    a = perturbed
                    best_a = a.copy()
                    best_score = pert_score
                    no_improve = 0
                else:
                    # Accept with some probability (simulated annealing)
                    if np.random.rand() < math.exp(-(pert_score - best_score) / (0.01 / max(no_improve, 1))):
                        a = perturbed
        
        if no_improve > 300:
            break
        
        it += 1
        if it % 500 == 0:
            print(f"  {verbose_prefix}iter={it}, score={best_score:.6f}, beta={beta:.1f}, no_improve={no_improve}")
    
    return best_a, best_score


def optimize_multi_scale(a_init, deadline, verbose_prefix=""):
    """Multi-scale optimization: start coarse, refine fine."""
    a = a_init.copy().astype(np.float64)
    n = len(a)
    
    # Phase 1: Coarse optimization (use averaged blocks)
    # Divide into blocks of size B, optimize block means, then refine
    for B in [32, 16, 8, 4, 2, 1]:
        if time.monotonic() > deadline - 60:
            break
        if B == 1:
            # Full resolution optimization
            a, score = optimize_sequence(a, deadline, verbose_prefix + f"[B=1] ")
        else:
            n_blocks = n // B
            if n_blocks < 4:
                continue
            # Extract block means
            blocks = a[:n_blocks * B].reshape(n_blocks, B)
            block_means = blocks.mean(axis=1)
            
            # Optimize block means
            def block_obj(bm):
                # Reconstruct and evaluate
                recon = np.tile(bm.reshape(-1, 1), (1, B)).flatten()
                return compute_score_fast(recon)
            
            # Simple optimization on block means
            bm_best = block_means.copy()
            bm_score = block_obj(bm_best)
            S_bm = float(np.sum(bm_best))
            
            for it in range(200):
                if time.monotonic() > deadline - 60:
                    break
                bm = np.maximum(bm_best + np.random.randn(n_blocks) * 0.05 * S_bm / n_blocks, 0)
                bm = bm * (S_bm / np.sum(bm))
                sc = block_obj(bm)
                if sc < bm_score:
                    bm_best = bm
                    bm_score = sc
            
            # Reconstruct
            a = np.tile(bm_best.reshape(-1, 1), (1, B)).flatten()
            a = normalize(a)
            print(f"  {verbose_prefix}Block size {B}: score={compute_score_fast(a):.6f}")
    
    return a, compute_score_fast(a)


def try_qp_step(a, deadline, n_steps=50, verbose_prefix=""):
    """Try solving a sequence of LPs (linearized QPs) to refine the solution."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    
    a = normalize(a.copy())
    n = len(a)
    S = float(np.sum(a))
    best_a = a.copy()
    best_score = compute_score_fast(a)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        # Linearize: minimize sum of |x - a| (L1 distance)
        # subject to: linearized conv constraints, sum(x) = S, x >= 0
        # Target: reduce M by a small amount
        target_M = M * 0.999  # Try to reduce max by 0.1%
        
        # Build constraint matrix: for each lag k, 
        # sum_j (a[j] * x[k-j] + a[k-j] * x[j]) <= target_M + conv[k] (linearization)
        # Wait, the linearization of x[j]*x[k-j] around a is:
        # a[j]*x[k-j] + a[k-j]*x[j] - a[j]*a[k-j]
        # So the constraint sum_j a[j]*x[k-j] + a[k-j]*x[j] - a[j]*a[k-j] <= target_M
        # becomes: sum_j (a[j]*x[k-j] + a[k-j]*x[j]) <= target_M + sum_j a[j]*a[k-j] = target_M + conv[k]
        
        # This gives us 2n-1 linear constraints.
        # Plus: sum(x) = S (equality)
        # Plus: x >= 0
        
        # Objective: minimize ||x - a||_1 (to stay close to current solution)
        # This is a LP!
        
        # But 2n-1 = 2047 constraints with 1024 variables... the constraint matrix is sparse.
        
        # Actually, this LP might be too expensive. Let me limit to just the active constraints.
        active_lags = np.where(conv > 0.9 * M)[0]
        
        if len(active_lags) == 0:
            break
        
        # Build sparse constraint matrix for active lags only
        rows = []
        cols = []
        vals = []
        rhs = []
        
        for k in active_lags:
            for j in range(n):
                # Coefficient of x[j] in the linearized constraint for lag k:
                # a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n, i.e., j>=k)
                coeff = 0.0
                kj = k - j
                if 0 <= kj < n:
                    coeff += a[kj]
                jk = j - k
                if 0 <= jk < n:
                    coeff += a[jk]
                if coeff != 0:
                    rows.append(k)  # Using local index
                    cols.append(j)
                    vals.append(coeff)
            rhs.append(target_M + conv[k])
        
        # Convert to local indexing
        lag_to_local = {int(k): idx for idx, k in enumerate(active_lags)}
        rows = [lag_to_local[r] for r in rows]
        
        A_ub = csr_matrix((vals, (rows, cols)), shape=(len(active_lags), n))
        b_ub = np.array(rhs)
        
        # Objective: minimize sum_j |x[j] - a[j]|
        # Introduce t_j >= |x[j] - a[j]| for each j
        # Minimize sum t_j
        # Constraints: x[j] - a[j] <= t_j, a[j] - x[j] <= t_j, t_j >= 0
        # Plus the conv constraints and sum(x) = S, x >= 0
        
        # Total variables: 2n (x and t)
        # This is getting complex. Let me simplify: just minimize -sum(x) or something.
        
        # Actually, let me just use a simpler LP:
        # Minimize: 0 (feasibility problem)
        # Subject to: linearized conv constraints, sum(x) = S, x >= 0
        # Then take the solution as the direction.
        
        # Even simpler: minimize sum of (x[j] - a[j])^2 approximated by 
        # minimizing sum |x[j] - a[j]| = min sum t_j s.t. -t_j <= x[j]-a[j] <= t_j

                x_old = np.linspace(0, 1, len(parent))

                parent = interp(x_new, x_old, parent)

                best_score = score
