            beta = 0
        y = a_new + beta * (a_new - a)
        y = project_onto_simplex(y)
        
        a = a_new
        
        obj_new = compute_obj(a)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()
        
        if verbose and k % 1000 == 0:
            print(f"  SG k={k}: obj={obj_new:.6f}, best={best_obj:.6f}, step={step:.6f}")
    
    return best_a, best_obj

def lp_alternate(a_init, max_iter=200, M_relax=1.0, time_limit=None, verbose=False):
    """LP alternating optimization."""
    n = len(a_init)
    a = project_onto_simplex(a_init.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    start_time = time.monotonic()
    
    for it in range(max_iter):
        if time_limit and time.monotonic() - start_time > time_limit:
            break
        
        # Solve LP: max sum(g) s.t. A@g <= M, g >= 0
        conv = np.convolve(a, a)
        M = np.max(conv) * M_relax
        
        # Build A matrix
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            j_start = max(0, k - n + 1)
            j_end = min(n-1, k)
            A[k, j_start:j_end+1] = a[k-j_end:k-j_start+1][::-1]
        
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0.0, None)] * n
        
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status != 0:
            # Try with relaxed M
            M = np.max(conv) * 1.5
            b_ub = np.full(2*n-1, M)
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if res.status != 0:
                break
        
        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Blend
        t = 0.5
        a_new = (1-t) * a + t * g
        a = project_onto_simplex(a_new)
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Also check g directly
        obj_g = compute_obj(g)
        if obj_g < best_obj:
            best_obj = obj_g
            best_a = g.copy()
        
        if verbose and it % 50 == 0:
            print(f"    LP it={it}: best={best_obj:.6f}")
    
    return best_a, best_obj

def coord_descent(a_init, n_rounds=5, verbose=False, time_limit=None):
    """Coordinate descent with line search."""
    n = len(a_init)
    a = project_onto_simplex(a_init.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    start_time = time.monotonic()
    
    for rnd in range(n_rounds):
        if time_limit and time.monotonic() - start_time > time_limit:
            break
        
        # Process coordinates in random order
        order = np.random.permutation(n)
        
        for idx in order:
            j = order[idx]
            old_val = a[j]
            
            # Try a range of multipliers
            best_x = old_val
            best_local = compute_obj(a)
            
            # Golden section or grid search
            for mult in [0.0, 0.25, 0.5, 0.75, 1.25, 1.5, 2.0, 3.0]:
                if mult == 1.0:
                    continue
                a[j] = old_val * mult
                a = project_onto_simplex(a)
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_x = old_val * mult
            
            a[j] = best_x
            a = project_onto_simplex(a)
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if verbose:
            print(f"  CD round {rnd}: obj={obj:.6f}, best={best_obj:.6f}")
    
    return best_a, best_obj

def structured_init(n, rng):
    """Generate diverse initializations."""

    elif choice == 1:
        # Two-lobe
        mid = n // 2
        a = np.ones(n)
        a[:mid] = rng.uniform(0.3, 1.0)
        a[mid:] = rng.uniform(0.3, 1.0)
    elif choice == 2:
        # Gaussian
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.4)
        x = np.arange(n)
        a = np.exp(-0.5 * ((x-center)/width)**2)
    elif choice == 3:
        # Periodic
        period = int(rng.integers(max(3, n//20), max(4, n//5)))
        pattern = 0.5 + 0.5 * rng.random(period)
        a = np.tile(pattern, n//period + 1)[:n]
    else:
        a = 0.5 + 0.5 * rng.random(n)
    
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

Actually, I just realized something important: the objective is `2*n*max_b/(sum_a)^2`. Since scaling doesn't change the objective, I can normalize `sum_a = 1` and just minimize `2*n*max_b`. So the effective objective is `max(conv(a,a))` (since `2n` is constant for fixed n).

- Given `a`, we want to find `g >= 0` with `sum(g) = 1` that minimizes `max(conv(g,g))`.
- But this is non-convex.
- The LP approach instead finds `g >= 0` maximizing `sum(g)` subject to `conv(a,g) <= M`.
- This is a convex relaxation that captures the "direction" of improvement.

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternation, Nesterov subgradient, and coordinate descent.
"""
import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * np.max(conv) / (s * s)

def subgrad_max_conv(a):
    """Subgradient of max(conv(a,a))."""
    n = len(a)
    conv = np.convolve(a, a)
    M = np.max(conv)
    # Use a relative tolerance for active set
    tol = max(M * 0.05, 1e-12)
    active = np.where(conv >= M - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n-1, k)
        # a[k-j] for j in [lo, hi]
        idx = k - hi : k - lo + 1
        grad[lo:hi+1] += 2.0 * a[idx][::-1]
    
    grad /= len(active)
    return grad, M

def project_simplex(a):
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def nest_sg(a_init, n_steps, time_limit=None, verbose=False):
    """Nesterov accelerated subgradient."""
    n = len(a_init)
    a = project_simplex(a_init)
    best_a = a.copy()
    best_obj = 2.0 * n * np.max(np.convolve(a, a))  # sum=1
    y = a.copy()
    
    start = time.monotonic()
    
    for k in range(n_steps):
        if time_limit and time.monotonic() - start > time_limit:
            break
        
        grad, M = subgrad_max_conv(y)
        obj_y = 2.0 * n * M  # since sum(y)=1
        
        grad_sq = np.dot(grad, grad)
        if grad_sq < 1e-15:
            break
        
        # Polyak step: (f(y) - f*) / ||g||^2
        # Lower bound: 2n/(2n-1)
        f_lower = 2.0 * n / (2*n - 1)
        step = min((obj_y - f_lower) / grad_sq, 5.0 / (1.0 + 0.1*k))
        
        a_new = y - step * grad
        a_new = project_simplex(a_new)
        
        # Nesterov momentum
        t_k = 2.0 / (k + 2)
        t_next = 2.0 / (k + 3)
        beta = max(0.0, (t_k - 1.0) / t_next) if t_next > 0 else 0.0
        y = a_new + beta * (a_new - a)
        y = project_simplex(y)
        a = a_new
        
        obj = 2.0 * n * np.max(np.convolve(a, a))
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        if verbose and k % 2000 == 0:
            print(f"    SG k={k}: obj={obj:.6f}, best={best_obj:.6f}")
    
    return best_a, best_obj

def lp_alt(a_init, max_iter, time_limit=None, relax=1.0, verbose=False):
    """LP alternating."""
    n = len(a_init)
    a = project_simplex(a_init)
    best_a = a.copy()
    best_obj = 2.0 * n * np.max(np.convolve(a, a))
    
    start = time.monotonic()
    
    for it in range(max_iter):
        if time_limit and time.monotonic() - start > time_limit:
            break
        
        conv = np.convolve(a, a)
        M = np.max(conv) * relax
        
        # Build A
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            lo = max(0, k-n+1)
            hi = min(n-1, k)
            A[k, lo:hi+1] = a[k-hi:k-lo+1][::-1]
        
        c = -np.ones(n)
        b = np.full(2*n-1, M)
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0,None)]*n, method='highs')
        if res.status != 0:
            # Try relaxed
            b = np.full(2*n-1, M * 2)
            res = linprog(c, A_ub=A, b_ub=b, bounds=[(0,None)]*n, method='highs')
            if res.status != 0:
                break
        
        g = res.x
        gs = np.sum(g)
        if gs < 1e-10:
            break
        g /= gs
        
        # Check g
        obj_g = 2.0 * n * np.max(np.convolve(g, g))
        if obj_g < best_obj:
            best_obj = obj_g
            best_a = g.copy()
        
        # Blend
        t = 0.5
        a_new = (1-t)*a + t*g
        a = project_simplex(a_new)
        
        obj_a = 2.0 * n * np.max(np.convolve(a, a))

        if verbose and it % 50 == 0:

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    s = np.sum(a)

    conv = np.convolve(a, a)

def subgrad_max_conv(a):
    """Compute subgradient of max(conv(a,a)) at point a.
    Returns (gradient_vector, max_value, active_indices)."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    # Active set with relative tolerance
    tol = max(M * 0.02, 1e-14)
    active = np.where(conv >= M - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # a[k-j] for j in [lo, hi]: indices in a are [k-hi, k-lo]
        seg = a[k - hi : k - lo + 1]
        grad[lo:hi + 1] += 2.0 * seg[::-1]
    
    grad /= len(active)
    return grad, M, active


def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def build_conv_matrix(a):
    """Build matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    return A

            if o_l < best_score:
                best_score = o_l
                best_seq = a_l.copy()
                print(f"  Deep LP(0.95): {o_l:.6f}")
            
            # Final subgradient
            if time.monotonic() < deadline - 5:
                a_f, o_f = nest_sg(best_seq, n_steps=10000,
                                    time_limit=max(1.0, deadline - time.monotonic() - 5))
                if o_f < best_score:
                    best_score = o_f
                    best_seq = a_f.copy()
                    print(f"  Final SG: {o_f:.6f}")
        
        # Final coordinate pass
        if time.monotonic() < deadline - 5:
            a_z, o_z = coord_desc(best_seq, n_rounds=3,
                                   time_limit=max(1.0, deadline - time.monotonic() - 2))
            if o_z < best_score:
                best_score = o_z
                best_seq = a_z.copy()
                print(f"  Final CD: {o_z:.6f}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)
    
    # Ensure validity
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    if np.sum(best_seq) < 0.01:
        best_seq = np.ones(100)
    
    result = [float(x) for x in best_seq]
    final_eval = evaluate_sequence(result)
    
    if not np.isfinite(final_eval):
        print("WARNING: final eval invalid, using fallback")
        result = [0.5] * 100
        final_eval = evaluate_sequence(result)
    
    print(f"\nFINAL: n={len(result)}, internal_obj={best_score:.6f}, eval={final_eval:.6f}, "
          f"time={time.monotonic()-t0:.1f}s")
    return result
