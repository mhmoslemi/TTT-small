"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response with backtracking line search and adaptive peak-tightening
2. FFT-accelerated subgradient steps for escape from local optima
3. Multi-length exploration (512, 1024, 2048)
4. Log-sum-exp smooth refinement with L-BFGS-B
5. Multi-start perturbation search
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time
import scipy.sparse as sp


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2 directly."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build Toeplitz-like matrix A where (A @ x) = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def build_conv_matrix_sparse(a):
    """Build sparse convolve matrix."""
    n = len(a)
    m = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(m, n))
    return A


def lp_best_response(a, M, top_peaks=None, peak_reduction=0.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M (with optional peak tightening), g >= 0."""
    n = len(a)
    m = 2 * n - 1
    if n <= 1024:
        A = build_conv_matrix(a)
        b_ub = np.full(m, M)
        if top_peaks is not None and peak_reduction > 0:
            for k in top_peaks:
                b_ub[int(k)] = M - peak_reduction
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
    else:
        A = build_conv_matrix_sparse(a)
        b_ub = np.full(m, M)
        if top_peaks is not None and peak_reduction > 0:
            for k in top_peaks:
                b_ub[int(k)] = M - peak_reduction
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None


def subgradient_step(a, step_size, n_top=10):
    """One subgradient step for minimizing max(conv(a,a)). Returns new a."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    # Pick top n_top peaks (or all if fewer)
    m = len(b)
    K = min(n_top, m)
    top_indices = np.argsort(b)[-K:]
    # Compute gradient: grad[j] = 2 * sum_{k in top} a[k-j] (when 0<=k-j<n)
    grad = np.zeros(n)
    for k in top_indices:
        k = int(k)
        # (grad b[k])[j] = 2*a[k-j] for 0 <= k-j < n
        # i.e., j ranges from max(0, k-n+1) to min(n-1, k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += 2.0 * a[k - j_hi:k - j_lo + 1][::-1]
    grad /= K
    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0.01:
        a_new /= s
    return a_new


def lp_iteration(a, max_iters, deadline, use_tightening=True, subgrad_every=20):
    """
    LP best-response iterations with backtracking and periodic subgradient steps.
    Returns (best_sequence, best_score).
    """
    best_local = a.copy()
    best_score = compute_score(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 1e-6, None)
    a = a / a.sum()
    n = len(a)

    consecutive_fails = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())

        # Determine peak tightening
        top_peaks = None
        peak_reduction = 0.0
        if use_tightening and it >= 20:
            n_top = min(5, 2 * n - 1)
            top_peaks = np.argsort(b)[-n_top:]
            # Gradually increase tightening
            frac = min(1.0, (it - 20) / 100.0)
            peak_reduction = M * 0.01 * frac

        # Solve LP
        g = lp_best_response(a, M, top_peaks, peak_reduction)
        if g is None:
            # Relax and retry
            g = lp_best_response(a, M * 1.02)
            if g is None:
                g = lp_best_response(a, M * 1.05)
                if g is None:
                    consecutive_fails += 1
                    if consecutive_fails > 5:
                        break
                    continue
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Backtracking line search
        t = 0.6
        improved = False
        for _backtrack in range(12):
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0, None)
            s_new = a_new.sum()
            if s_new < 1e-10:
                t *= 0.5
                continue
            a_new = a_new / s_new
            score = compute_score(a_new)
            if score < best_score * (1 - 1e-10):
                best_score = score
                best_local = a_new.copy()
                a = a_new
                improved = True
                consecutive_fails = 0
                break
            t *= 0.5
            if t < 0.001:
                break

        if not improved:
            consecutive_fails += 1
            # Try subgradient step every few failures
            if consecutive_fails % subgrad_every == 0:
                for sg_attempt in range(3):
                    sg_step = 0.5 / (np.sqrt(it + 1))
                    a_sg = subgradient_step(a, sg_step, n_top=10)
                    score_sg = compute_score(a_sg)
                    if score_sg < best_score:
                        best_score = score_sg
                        best_local = a_sg.copy()
                        a = a_sg
                        improved = True
                        break
                    sg_step *= 0.5
            if consecutive_fails > 40:
                break

    return best_local, best_score


def smooth_objective(x, beta, n):
    """Log-sum-exp smoothed objective for normalized a."""
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 0.01:
        return 1e10
    x_norm = x_pos / s
    b = fast_conv(x_norm)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    log_sum = b_max * beta + np.log(np.exp(shifted).sum()) / beta
    return float(2.0 * n * log_sum)


def smooth_refine(a, deadline):
    """Refine using smooth log-sum-exp objective with L-BFGS-B and beta annealing."""
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)

    betas = [30, 100, 300, 1000, 3000, 10000]
    for beta in betas:
        if time.monotonic() > deadline:
            break
        x0 = best.copy()
        x0 = np.clip(x0, 1e-5, None)
        x0 = x0 / x0.sum()

        try:
            res = minimize(smooth_objective, x0, args=(beta, n),
                           method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': 100, 'maxfun': 3000,
                                    'ftol': 1e-18, 'gtol': 1e-16,
                                    'maxls': 50})
            if res.fun < 1e10:
                res_x = np.clip(res.x, 0, None)
                if res_x.sum() > 0.01:
                    score = compute_score(res_x)
                    if score < best_score:
                        best = res_x.copy()
                        best_score = score
        except Exception:
            pass

    return best, best_score


def resample_sequence(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    new_a = np.maximum(np.interp(t_new, t_old, a), 0.0)
    s = new_a.sum()
    if s > 0.01:
        new_a /= s
    return new_a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    t_arr = np.linspace(0, 1, n)

    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        resampled = np.maximum(np.interp(t_arr, t_old, parent), 0)
        if resampled.sum() > 0.01:
            inits.append(resampled / resampled.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine harmonics
    for harm in [1, 2, 3]:
        amp = 0.3 / harm
        c = 1.0 + amp * np.cos(2 * np.pi * harm * t_arr)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    # Triangular
    c = 1.0 - np.abs(t_arr - 0.5)
    c = np.maximum(c, 0.05)
    inits.append(c / c.sum())

    # Random smooth
    raw = rng.uniform(0.3, 1.0, n)
    # Smooth with a simple moving average
    k = max(1, n // 50)
    kernel = np.ones(k) / k
    smoothed = np.convolve(raw, kernel, mode='same')
    smoothed = np.maximum(smoothed, 0.01)
    inits.append(smoothed / smoothed.sum())

    # Asymmetric (heavier on one side)
    c = 1.0 + 0.5 * t_arr
    inits.append(c / c.sum())

    # Bimodal
    c = 1.0 + 0.4 * np.cos(4 * np.pi * t_arr)
    c = np.maximum(c, 0.05)
    inits.append(c / c.sum())

    return inits

    return [float(x) for x in best]
