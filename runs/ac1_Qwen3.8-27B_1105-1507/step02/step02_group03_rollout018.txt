def normalize(f):
    """Normalize f to sum to 1, ensuring non-negativity."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s

def compute_score(f):
    """Compute the evaluation score for a non-negative sequence."""
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * n * conv.max() / (s * s))

    """Compute LSE-smoothed score (assumes sum(f)=1)."""
    n = len(f)
    conv = np.convolve(f, f)
    max_c = conv.max()
    lse = max_c + tau * np.log(np.sum(np.exp((conv - max_c) / tau)))
    return 2.0 * n * lse

    weights = exps / exps.sum()

    F = np.fft.rfft(f, N)
    # Cross-correlation: ifft(conj(W) * F) gives sum_n W[n+i] * F[n]
    cross = np.fft.irfft(np.conj(W) * F, N)
    grad = 4.0 * n * cross[:n]

    f = normalize(f.copy())
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    
    v = np.zeros(n)  # momentum
    eta = eta0
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Anneal tau
        tau = tau_start * (tau_end / tau_start) ** (it / n_iter)
        
        # Compute gradient
        grad = lse_gradient(f, tau)
        
        # Project gradient onto tangent space (sum=0)
        grad = grad - np.mean(grad)
        
        # Momentum update
        v = beta * v - eta * grad
        f_new = f + v
        
        # Project onto simplex (non-negative, sum=1)
        f_new = np.clip(f_new, 0, None)
        f_new = normalize(f_new)
        
        # Evaluate
        new_score = compute_score(f_new)
        
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
        
        if new_score < lse_objective(f, tau):
            # Accept
            f = f_new
            # Slowly increase eta if improving
            eta = min(eta * 1.01, 1.0)
        else:
            # Reject, reduce eta
            eta *= 0.5
            v = beta * v  # Keep momentum but don't apply update
    
    return best_f, best_score

from scipy.optimize import linprog
import time

def normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s

def compute_score(f):
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * n * conv.max() / (s * s))

def lse_objective(f, tau):
    n = len(f)
    conv = np.convolve(f, f)
    max_c = conv.max()
    lse = max_c + tau * np.log(np.sum(np.exp((conv - max_c) / tau)))
    return 2.0 * n * lse

def lse_gradient(f, tau):
    n = len(f)
    conv = np.convolve(f, f, mode='full')
    max_c = conv.max()
    exps = np.exp((conv - max_c) / tau)
    weights = exps / exps.sum()
    N = 1 << (3 * n).bit_length()
    W = np.fft.rfft(weights, N)
    F = np.fft.rfft(f, N)
    cross = np.fft.irfft(np.conj(W) * F, N)
    grad = 4.0 * n * cross[:n]
    return grad
