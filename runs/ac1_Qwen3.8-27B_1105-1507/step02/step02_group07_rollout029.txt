def compute_eval(f):
    """Fast evaluation without the overhead of evaluate_sequence"""
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:

    return 2.0 * n * M / (S * S)

def subgradient_step(f, lr, active_frac=0.05):
    """One subgradient step on the active set"""
    n = len(f)
    C = np.convolve(f, f)
    M = C.max()
    threshold = M * (1.0 - active_frac)
    active = np.where(C >= threshold)[0]
    
    # Average subgradient over active set
    grad = np.zeros(n)
    for k in active:
        # d/df[m] C[k] = f[k-m] (if valid) + f[m-k] (if valid)

        hi2 = min(n-1, k + n-1)
        if lo2 <= hi2:
            grad[lo2:hi2+1] += f[lo2-k:hi2-k+1]
    
    grad /= len(active)
    
    # Update

    f_new = np.maximum(f_new, 0.0)
    S = f_new.sum()
    if S > 0:
        f_new /= S
    else:
        f_new = f.copy()
    return f_new, M

    exp_C = np.exp(beta * (C - max_C))

    if total < 1e-300:

```python
import numpy as np
import time

def compute_conv(f):
    return np.convolve(f, f)

def compute_grad_active(f, active_lags):
    """Compute average subgradient over active lags."""
    n = len(f)
    grad = np.zeros(n)
    for k in active_lags:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            grad[lo:hi+1] += 2.0 * f[k-hi:k-lo+1][::-1]
    grad /= len(active_lags)
    return grad

def compute_softmax_grad(f, beta):
    """Compute gradient of soft-max of conv(f,f)."""
    n = len(f)
    C = np.convolve(f, f)
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros(n), max_C
    p = exp_C / total
    # grad[j] = 2 * sum_i f[i] * p[j+i]
    # Using the formula: grad = 2 * conv(f[::-1], p)[n-1:2n-1]
    fp = np.convolve(f[::-1], p)
    grad = 2.0 * fp[n-1:2*n-1]
    soft_val = max_C + np.log(total) / beta
    return grad, soft_val

def optimize_sequence(f, deadline, verbose=False):
    """Main optimization loop combining subgradient and softmax gradient."""
    f = f.astype(np.float64).copy()
    S = f.sum()
    f = f / S
    n = len(f)
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0
    
    lr = 0.01
    no_improve = 0
    max_no_improve = 100
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        C = np.convolve(f, f)
        M = C.max()
        score = 2.0 * n * M / S**2  # S=1 since normalized
        
        # Find active set (top lags)
        threshold = M * 0.98
        active = np.where(C >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(C)])
        
        # Compute subgradient
        subgrad = compute_grad_active(f, active)
        
        # Compute softmax gradient (with adaptive beta)
        beta = min(500.0, 10.0 + 5.0 * it / 100.0)
        sm_grad, _ = compute_softmax_grad(f, beta)
        
        # Combine gradients
        # Normalize both to have similar magnitude
        subgrad_norm = np.linalg.norm(subgrad)
        sm_grad_norm = np.linalg.norm(sm_grad)
        if subgrad_norm > 0:
            subgrad = subgrad / subgrad_norm
        if sm_grad_norm > 0:
            sm_grad = sm_grad / sm_grad_norm
        
        # Weight: more subgradient when we have clear active set
        if len(active) <= 5:
            alpha = 0.7  # more subgradient
        elif len(active) <= 20:
            alpha = 0.5
        else:
            alpha = 0.3
        
        combined_grad = alpha * subgrad + (1 - alpha) * sm_grad
        
        # Adam update
        t_adam += 1
        m = beta1 * m + (1 - beta1) * combined_grad
        v = beta2 * v + (1 - beta2) * (combined_grad ** 2)
        m_hat = m / (1 - beta1**t_adam)
        v_hat = v / (1 - beta2**t_adam)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        new_f = f - lr * update
        new_f = np.maximum(new_f, 0.0)
        ns = new_f.sum()
        if ns < 1e-10:
            new_f = f
        else:
            new_f = new_f / ns
        
        # Evaluate
        C_new = np.convolve(new_f, new_f)
        M_new = C_new.max()
        new_score = 2.0 * n * M_new
        
        if new_score < best_score:
            best_score = new_score
            best_f = new_f.copy()
            f = new_f
            no_improve = 0
            # Occasionally increase lr
            if it % 500 == 0:
                lr = min(lr * 1.1, 0.05)
        else:
            f = new_f  # still move, but don't update best
            no_improve += 1
            if no_improve > max_no_improve:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-7:
                    # Try a restart with perturbation
                    rng = np.random.default_rng(it)
                    f = f * (1.0 + 0.05 * rng.standard_normal(n))
                    f = np.maximum(f, 0)
                    f = f / f.sum()
                    lr = 0.01
                    no_improve = 0
                    max_no_improve = 50
        
        if verbose and it % 1000 == 0:
            print(f"  iter {it}: score={best_score:.6f}, M={M:.6f}, lr={lr:.6f}, n_active={len(active)}")
    
    return best_f, best_score

**Approach: Focus on the specific lags that are at the maximum**

At the current optimum, there are probably a handful of lags that are at or very near the maximum. If we can identify these and specifically reduce them, we can improve.

The key observation: at the optimum, the "active" lags (those at the max) should be balanced. If one lag is slightly higher than the others, we can reduce it by adjusting f in the direction that decreases that specific lag.

**Algorithm: Active-set Newton method**

1. Identify the active set A (lags where C[k] is within, say, 0.1% of max)
2. For each active lag k, compute the "leverage" of each f[i] on C[k]:
   leverage[k, i] = 2 * f[k-i] (if valid)
3. Find a direction d that:
   - Decreases C[k] for all k in A simultaneously
   - Maintains sum(f) = 1 and f >= 0
4. Take a step in that direction

The direction d can be found by solving:
minimize ||d||²
subject to leverage[k, :] · d <= -epsilon for all k in A
sum(d) = 0 (to maintain normalization)

This is a quadratic program (QP) with linear constraints! We can solve it with scipy.optimize or cvxpy.

Actually, since we want to decrease ALL active lags, and they might pull in different directions, a good approach is:

Find d that minimizes the maximum increase in any active lag:
minimize t
subject to leverage[k, :] · d + t >= 0 for all k in A (i.e., the change in C[k] is <= -t)
sum(d) = 0
d >= -f (to maintain f + d >= 0)

This is an LP! Let me formulate it:

Minimize: t (we want to make t as negative as possible, i.e., maximize -t)
Equivalently: maximize -t
Constraints:
- For each k in A: sum_i leverage[k,i] * d[i] + t <= 0 (C[k] decreases by at least t)
- sum_i d[i] = 0
- d[i] >= -f[i] for all i
- t >= -some_bound (to prevent unbounded)

I.e., sum_i leverage[k,i] * d[i] <= -delta for all k in A.

To make this feasible, I should maximize delta. So:

  sum_i d[i] = 0
  d[i] >= -f[i] * alpha for all i (where alpha is a step size bound, e.g., alpha = 0.1)
  delta >= 0

This is a linear program! And it directly gives us the best direction to decrease all active lags simultaneously.

def find_shaving_direction(f, active_lags, alpha=0.1):
    """Find direction d that decreases all active lags simultaneously."""
    from scipy.optimize import linprog
    
    n = len(f)
    A_size = len(active_lags)
    # Variables: d[0..n-1], delta
    # Total: n+1 variables
    
    # Objective: maximize delta => minimize -delta
    c = np.zeros(n + 1)
    c[-1] = -1.0  # minimize -delta
    
    # Constraints:
    # 1. For each active lag k: sum_i leverage[k,i] * d[i] + delta <= 0
    #    => leverage[k,:] @ d + [0,...,0,1] @ [d, delta] <= 0
    A_ub = np.zeros((A_size, n + 1))
    b_ub = np.zeros(A_size)
    for idx, k in enumerate(active_lags):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo <= hi:
            A_ub[idx, lo:hi+1] = 2.0 * f[k-hi:k-lo+1][::-1]
        A_ub[idx, n] = 1.0
        b_ub[idx] = 0.0
    
    # 2. sum(d) = 0 => -sum(d) <= 0 and sum(d) <= 0
    #    As two inequalities:
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # 3. d[i] >= -alpha * f[i] => -d[i] <= alpha * f[i]
    #    And d[i] <= alpha (optional upper bound)
    
    bounds = []
    for i in range(n):
        bounds.append((-alpha * f[i], alpha))  # d[i] in [-alpha*f[i], alpha]
    bounds.append((0, None))  # delta >= 0
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n], result.x[-1]  # direction, delta
    
    return None, 0.0
```

This LP has n+1 variables and A_size + 2 + 2n constraints. For n=512 and A_size ≈ 10-50, this is very manageable.

import numpy as np
import time
from scipy.optimize import linprog

def fast_eval(f):
    """Fast evaluation (assumes f is already valid)."""
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    C = np.convolve(f, f)
    M = float(np.max(C))
    return 2.0 * n * M / (S * S)

def get_leverage(f, k):
    """Get leverage vector for lag k: d(C[k])/df = leverage"""
    n = len(f)
    lev = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    if lo <= hi:
        lev[lo:hi+1] = 2.0 * f[hi:lo-1:-1]  # f[k-hi], f[k-hi+1], ..., f[k-lo]
    return lev

def lp_target_direction(f, M):
    """Original LP: find best target g with conv(f,g) <= M, g >= 0, maximize sum(g)."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: A @ g <= M for all lags
    # A[k, j] = f[k-j] (if valid)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lo = max(0, k - (n-1))
        hi = min(k, n-1)
        if lo <= hi:
            A[k, lo:hi+1] = f[hi:lo-1:-1]
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x
    except:
        pass
    return None

def lp_shave_direction(f, active_lags, alpha=0.1):
    """Find direction that reduces all active lags."""
    n = len(f)
    # Variables: d[0..n-1], delta
    c = np.zeros(n+1)
    c[n] = -1.0  # maximize delta => minimize -delta
    
    A_ub_rows = []
    b_ub_vals = []
    for k in active_lags:
        lev = get_leverage(f, k)
        row = np.append(lev, 1.0)
        A_ub_rows.append(row)
        b_ub_vals.append(0.0)
    
    if not A_ub_rows:
        return None, 0.0
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(-alpha * f[i], alpha) for i in range(n)]
    bounds.append((0, None))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 5})
        if res.success:
            return res.x[:n], float(res.x[n])
    except:
        pass
    return None, 0.0

def softmax_grad(f, beta):
    """Compute softmax gradient of conv(f,f)."""
    n = len(f)
    C = np.convolve(f, f)
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros(n), max_C
    p = exp_C / total
    # grad = 2 * conv(f[::-1], p)[n-1:2n-1]
    fp = np.convolve(f[::-1], p)
    grad = 2.0 * fp[n-1:2*n-1]
    soft_val = max_C + np.log(total) / beta
    return grad, soft_val

def tukey_window(n, r=0.3):

            w[i] = 0.5 * (1 + np.cos(np.pi * (2*i/(2*m) - 1)))
            w[n-1-i] = w[i]
    return w
