```python
def subgradient_step(a, k_star, lr):
    n = len(a)
    # Compute subgradient
    g = np.zeros(n)
    for j in range(n):
        # Contribution from a[j] * a[k*-j] (a[j] varies)
        if 0 <= k_star - j < n:
            g[j] += a[k_star - j]
        # Contribution from a[k*-j] * a[j] (a[k*-j] varies, i.e., a[j] is the second factor)
        if 0 <= j - k_star < n:
            g[j] += a[j - k_star]
    g *= 2.0 * n
    # Projected gradient step on simplex
    a_new = a - lr * g
    # Project onto simplex {x >= 0, sum(x) = 1}
    a_new = project_simplex(a_new)
    return a_new
```

```python
"""
Minimizes J(a) = 2n*max(conv(a,a))/(sum(a))^2 using:
1. Analytical-gradient L-BFGS-B on log-sum-exp smoothed objective
2. Adaptive beta annealing (start smooth, end sharp)
3. Multiple diverse initializations (spectral, structured, random)
4. Multi-length exploration
5. Perturbation + re-optimization
6. Final active-set subgradient polishing
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

def compute_score(a):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))

def soft_max_val(b, beta):
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50, 50)
    return b_max + np.log(np.exp(shifted).sum()) / beta

def smooth_fn_and_grad(x, beta, n):
    """Returns (f(x), grad_x) for the smooth objective."""
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    a = x_pos / s
    b = fast_conv(a)
    # Soft max value
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    fm = b_max + np.log(exp_shifted.sum()) / beta
    f = 2.0 * n * fm
    # Gradient
    m_len = 2 * n - 1
    # g[m] = sum_j a[j]*w[j+m] (cross-correlation)
    # h[m] = sum_k w[k]*a[m-k] (linear convolution, first n elements)
    h = np.convolve(w, a, 'full')[:n]
    g = np.convolve(a, w[::-1], 'full')[n-1:2*n-1][::-1]
    grad_a = 2.0 * n * (g + h)
    # df/dx = grad_a / s * 1[x>0]
    mask = (x > 0).astype(float)
    grad_x = grad_a * mask / s
    return f, grad_x

def optimize_smooth(a0, deadline, n, betas=None):
    """Run L-BFGS-B with adaptive beta schedule. Returns (best_a, best_score)."""
    if betas is None:
        betas = [10, 50, 200, 500, 2000, 5000]
    
    best = a0.copy()
    best_score = compute_score(a0)
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        x0 = np.maximum(best, 1e-6)
        
        def fun(x):
            return smooth_fn_and_grad(x, beta, n)[0]
        
        def jac(x):
            return smooth_fn_and_grad(x, beta, n)[1]
        
        maxiter = max(50, min(300, int((deadline - time.monotonic()) / 1.0)))
        
        try:
            res = minimize(fun, x0, jac=jac, method='L-BFGS-B',
                          bounds=[(0, None)] * n,
                          options={'maxiter': maxiter, 'maxfun': maxiter*5,
                                   'ftol': 1e-18, 'gtol': 1e-10})
            if res.fun < 1e10:
                a_new = np.clip(res.x, 0, None)
                if a_new.sum() > 0.01:
                    score = compute_score(a_new)
                    if score < best_score:
                        best = a_new.copy()
                        best_score = score
        except Exception as e:
            pass
    
    return best, best_score

def generate_inits(n, rng, parent=None):

def active_set_polish(a, iterations=100, deadline=None):
    """Polish using subgradient on the active set (where conv is near max)."""
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    a = a.copy()
    a = np.clip(a, 0, None)
    a = a / a.sum()
    
    lr = 0.01
    for it in range(iterations):
        if deadline and time.monotonic() > deadline:
            break
        b = fast_conv(a)
        b_max = b.max()
        # Active set: where b is close to max
        threshold = b_max * 0.95
        active = b >= threshold
        k_star = np.argmax(b)
        
        # Subgradient at k_star
        # grad[m] = a[k_star - m] + a[m - k_star] (with bounds)
        grad = np.zeros(n)
        for m in range(n):
            if 0 <= k_star - m < n:
                grad[m] += a[k_star - m]
            if 0 <= m - k_star < n:
                grad[m] += a[m - k_star]
        grad *= 2.0 * n
        
        # Projected gradient step on simplex
        # Use mirror descent (exponential weights)
        lr_adapt = lr / (1 + it * 0.1)
        a_new = a * np.exp(-lr_adapt * grad / max(grad.max(), 1e-10))
        a_new = np.clip(a_new, 0, None)
        if a_new.sum() < 1e-10:
            break
        a_new = a_new / a_new.sum()
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best = a_new.copy()
            a = a_new
        else:
            lr *= 0.8
    
    return best, best_score
```

One idea I haven't explored: **frequency domain optimization**. 

In the frequency domain, the convolution becomes a pointwise product:
FFT(conv(a,a)) = |FFT(a)|^2 (for the circular convolution of zero-padded a)

So we want: IDFT(|A|^2) to be as flat as possible on the first 2n-1 elements, where A = FFT(a, 2n-1 or more).

This is like: we want |A[k]|^2 to be shaped so that its IDFT is flat on a window.

The IDFT of a constant |A|^2 is a delta. The IDFT of |A|^2 = sinc^2 shape is a triangular window.

We want something in between: a |A|^2 shape whose IDFT is flat on [0, 2n-2] and zero (or small) elsewhere.

The ideal |A|^2 would be: |A[k]|^2 = C for k=0,...,some range, and 0 elsewhere. But that gives a sinc in the time domain, which oscillates.

Actually, for our purposes, we just want the first 2n-1 elements of IDFT(|A|^2) to be as flat as possible. The "tail" (elements 2n-1 onwards) doesn't matter (they correspond to the "wrap-around" in circular conv that doesn't exist in linear conv).

So the problem is: minimize max_{k=0}^{2n-2} (IDFT(|A|^2))[k] subject to A = FFT(a, 2n-1) and a ≥ 0.

This isn't directly helpful for optimization, but it gives insight: we want |A| to be "smooth" (not too peaked in frequency).

A sequence with smooth |A| is one that's "spread out" in time (not concentrated). A uniform sequence has |A[0]| = n and |A[k]| ≈ 0 for k ≠ 0 (like a delta in frequency), giving a peak in the time-domain autocorrelation.

We want something with more uniform |A|, which means a more "oscillatory" time-domain sequence. But we're constrained to a ≥ 0.
