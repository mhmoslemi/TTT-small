"""
Optimized search for step function coefficients minimizing 2n*max(autocorr)/sum^2.
Uses adaptive LP, coordinate descent, and multi-strategy restarts.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_conv_matrix(a):
    """Build matrix A such that (A @ b)[k] = (a*b)[k] for full convolution."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

def solve_lp(a, M, upper_bound=100.0):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M, b >= 0, b <= upper_bound."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun)
    return None, None

def lp_iteration(a, time_budget, max_iters=100):
    """Perform LP-based iterations to reduce max autocorrelation."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        sum_a = float(np.sum(a))
        
        # Try slightly tighter M
        for factor in [1.0, 0.99, 0.98, 0.97]:
            M_try = M * factor
            b, b_sum = solve_lp(a, M_try)
            if b is not None and b_sum > sum_a * 0.99:
                break
        
        if b is None:
            break
        
        # Rescale and mix
        b = b / np.sum(b)
        
        # Try different mixing parameters
        improved = False
        for t in [0.3, 0.5, 0.7, 0.8, 0.9]:
            a_new = (1 - t) * a + t * b
            a_new = np.clip(a_new, 0, None)
            s = np.sum(a_new)
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_a = a_new.copy()
                best_obj = obj
                a = a_new
                improved = True
                break
        
        if not improved:
            # Try smaller steps
            for t in [0.1, 0.05, 0.02]:
                a_new = (1 - t) * a + t * b
                a_new = np.clip(a_new, 0, None)
                s = np.sum(a_new)
                if s < 1e-10:
                    continue
                a_new = a_new / s
                obj = compute_obj(a_new)
                if obj < best_obj:
                    best_a = a_new.copy()
                    best_obj = obj
                    a = a_new
                    break
            else:
                # No improvement, break
                if it > 5:
                    break
    
    return best_a, best_obj

def coordinate_descent(a, time_budget, max_passes=50):
    """Optimize each coordinate using golden section search."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Golden section ratio
    gr = (np.sqrt(5) - 1) / 2
    
    for pass_num in range(max_passes):
        if time.monotonic() - start > time_budget:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            
            # Save current value
            old_val = a[i]
            
            # Define 1D function to minimize
            def f_1d(t):
                a[i] = t
                return compute_obj(a)
            
            # Search range: [0, 3 * old_val] (or [0, max_val] if old_val is 0)
            lo = 0.0
            hi = max(3.0 * old_val, 0.1 / n)
            
            # Golden section search
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = f_1d(c)
            fd = f_1d(d)
            
            for _ in range(30):  # 30 iterations of golden section
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = f_1d(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = f_1d(d)
            
            best_t = (lo + hi) / 2
            a[i] = best_t
            obj = compute_obj(a)
            
            if obj < best_obj:
                best_a = a.copy()
                best_obj = obj
                improved = True
            else:
                a[i] = old_val  # Restore
        
        if not improved and pass_num > 3:
            break
    
    return best_a, best_obj

Wait, the coordinate descent with golden section is going to be VERY expensive. Each evaluation of f_1d requires computing the full convolution, which is O(n^2). With n=300, that's 90000 operations per evaluation, times 30 golden section iterations, times 300 coordinates, times multiple passes. That's 30 * 300 * 90000 = 810 million operations per pass. Way too slow.

I need a faster approach for coordinate descent. Let me think...

When I change a[i] from old to new, only certain convolution values change. Specifically:
conv[k] changes for k in [i, i+n-1] (the range where a[i] participates).

The change in conv[k] is:
- If k = 2i: change = new^2 - old^2
- If k != 2i and k-i is in [0, n-1]: change = 2*(new - old)*a[k-i]
- Otherwise: no change

Wait, let me be more careful. conv[k] = sum_j a[j]*a[k-j]. The term involving a[i] is:
- When j = i: a[i]*a[k-i] (if 0 <= k-i < n)
- When k-j = i, i.e., j = k-i: a[k-i]*a[i] (if 0 <= k-i < n and k-i != i)

So for a given k:
- If k-i is in [0, n-1] and k-i != i: conv[k] gets 2*a[i]*a[k-i]
- If k-i = i (i.e., k = 2i): conv[k] gets a[i]^2
- Otherwise: no contribution from a[i]

When a[i] changes from old to new:
- conv[k] changes by 2*(new-old)*a[k-i] for k-i in [0,n-1], k != 2i
- conv[k=2i] changes by new^2 - old^2

So I can efficiently update the convolution when changing one coordinate. This makes coordinate descent feasible!

Actually wait, I can optimize further. The affected values are base_conv[k] + 2*t*a[k-i] (linear in t for k != 2i) or base_conv[k] + t^2 (for k = 2i). The maximum of a set of linear functions and one quadratic is the upper envelope, which is a convex piecewise-linear + quadratic function. I can find the maximum more efficiently, but for n=300, just scanning is fine.

Actually, let me reconsider. The max over k of (base_conv[k] + contrib[k,t]) where contrib is linear in t (except at one point). The maximum of linear functions is convex in t. So objective_1d is convex in t (it's the max of convex functions divided by a quadratic... well, it's not exactly convex, but it's unimodal in practice).

Let me just use the simple approach: precompute the base conv, then for each t, scan the affected k values and compute the max. With n=300, the affected range is at most n=300 values, so each evaluation is O(n). With 50 evaluations from minimize_scalar, that's O(50n) per coordinate, O(50n^2) per pass = 4.5M for n=300. """
Optimized step function search using adaptive LP with coordinate descent refinement.
Key improvements: efficient coordinate-wise optimization exploiting convolution structure,
adaptive LP bounds, and multi-strategy exploration.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def solve_lp(a, M):
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun)
    return None, None


def lp_phase(a, time_budget, max_iters=200):
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    consecutive_fail = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        sum_a = float(np.sum(a))
        
        b = None
        b_sum = 0
        # Try progressively tighter bounds
        for factor in [1.0, 0.995, 0.99, 0.98]:
            b, b_sum = solve_lp(a, M * factor)
            if b is not None:
                break
        
        if b is None:
            break
        
        b = b / float(np.sum(b))
        
        improved = False
        for t in [0.5, 0.7, 0.8, 0.9, 0.3, 0.4, 0.6]:
            a_new = (1 - t) * a + t * b
            a_new = np.clip(a_new, 0, None)
            s = np.sum(a_new)
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-10:
                best_a = a_new.copy()
                best_obj = obj
                a = a_new
                improved = True
                break
        
        if improved:
            consecutive_fail = 0
        else:
            consecutive_fail += 1
            if consecutive_fail > 10:
                break
    
    return best_a, best_obj


def coord_descent(a, time_budget, max_passes=20):
    """Coordinate descent using efficient convolution updates."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for pass_num in range(max_passes):
        if time.monotonic() - start > time_budget:
            break
        
        pass_improved = False
        
        for i in range(n):
            if time.monotonic() - start > time_budget:
                break
            
            old_val = a[i]
            
            # Compute base convolution (with a[i] = 0)
            a[i] = 0.0
            base_conv = np.convolve(a, a)
            a[i] = old_val
            
            sum_base = float(np.sum(a)) - old_val
            
            # Affected k range
            k_start = i
            k_end = min(2 * n - 2, i + n - 1)
            
            # Precompute: for each affected k, the coefficient of t (linear part)
            # and whether it's the quadratic term
            k_vals = np.arange(k_start, k_end + 1)
            j_vals = k_vals - i  # = 0, 1, ..., n-1
            # Linear coefficient: 2*a[j] for k != 2i, 0 for k = 2i
            lin_coef = 2.0 * a[j_vals]
            quad_mask = (k_vals == 2 * i)
            lin_coef[quad_mask] = 0.0
            base_vals = base_conv[k_vals]
            
            # Max of unaffected values
            if k_start > 0 or k_end < 2*n-2:
                mask = np.ones(2*n-1, dtype=bool)
                mask[k_start:k_end+1] = False
                max_unaffected = float(np.max(base_conv[mask])) if mask.any() else 0.0
            else:
                max_unaffected = 0.0
            
            # Now optimize t
            def f_1d(t):
                # Affected values
                affected = base_vals + lin_coef * t
                if quad_mask.any():
                    affected[quad_mask] = base_vals[quad_mask] + t * t
                max_affected = float(np.max(affected))
                max_total = max(max_affected, max_unaffected)
                sum_total = sum_base + t
                if sum_total < 0.01:
                    return 1e10
                return 2 * n * max_total / (sum_total * sum_total)
            
            # Determine search bounds
            hi = max(3.0 * old_val, 2.0 / n)
            if old_val > 0:
                lo = 0.0
            else:
                lo = 0.0
            
            try:
                result = minimize_scalar(f_1d, bounds=(lo, hi), method='bounded',
                                         options={'xatol': 1e-12})
                new_val = float(result.x)
            except Exception:
                new_val = old_val
            
            a[i] = new_val
            obj = compute_obj(a)
            
            if obj < best_obj - 1e-12:
                best_a = a.copy()
                best_obj = obj
                pass_improved = True
            else:
                a[i] = old_val
        
        if not pass_improved and pass_num >= 2:
            break
    
    return best_a, best_obj


def random_restart_init(n, rng):
    """Generate diverse initializations."""
    choice = int(rng.integers(0, 8))
    
    if choice == 0:
        # Uniform
        a = np.ones(n)
    elif choice == 1:
        # Edge-concentrated
        a = np.ones(n)
        a[0] = 1.0
        a[-1] = 1.0
        a[0:n//4] *= 2.0
        a[3*n//4:] *= 2.0
    elif choice == 2:
        # Gaussian
        center = n / 2.0
        width = n * rng.uniform(0.15, 0.5)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
    elif choice == 3:
        # Two-peak (edges)
        center = n / 2.0
        width = n * 0.1
        a = np.exp(-0.5 * ((np.arange(n) - 0) / width) ** 2) + \
            np.exp(-0.5 * ((np.arange(n) - (n-1)) / width) ** 2)
        a += 0.05
    elif choice == 4:
        # Ramp
        a = np.linspace(0.1, 1.0, n)
        # Or reverse
        if rng.random() < 0.5:
            a = a[::-1]
    elif choice == 5:
        # Periodic
        period = int(rng.integers(max(3, n//20), max(5, n//5)))
        phase = rng.uniform(0, period)
        a = np.maximum(0.01, np.cos(2 * np.pi * (np.arange(n) + phase) / period))
    elif choice == 6:
        # Random with smoothing
        a = np.maximum(0, rng.normal(0.5, 0.3, n))
        # Smooth
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    else:
        # Near-uniform with noise
        a = 1.0 + 0.3 * rng.standard_normal(n)
        a = np.maximum(0.01, a)
    
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def full_optimize(a_init, time_budget):
    """Run full optimization pipeline on a single sequence."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

        if obj2 < best_obj:

            best_obj = obj2
            a = a2
    
    # Phase 3: Second LP round
