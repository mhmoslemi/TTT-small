def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            rows.extend([k] * (jh - jl + 1))
            cols.extend(range(jl, jh + 1))
            for i in range(jl, jh + 1):
                vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

        if g.sum() > 1e-10:

def lp_descent(a, max_iters, deadline, min_J=None):
    """LP mirror descent with backtracking line search."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, n)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Backtracking line search
        t = 1.0
        best_a = None
        best_J = current_J
        for _ in range(20):
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:

        a = best_a
        current_J = best_J

def soft_max_grad(a, T):
    """Compute soft max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    soft_max_val = (Tb_max + np.log(Z)) / T
    p = exp_Tb / Z
    
    # Gradient: grad_j = sum_i a[i] * p[j+i]
    N = 1
    while N < 3 * n:
        N <<= 1
    fa = np.fft.fft(a, N)
    fp = np.fft.fft(p, N)
    corr = np.fft.ifft(np.conj(fa) * fp).real
    grad = corr[:n]
    
    return soft_max_val, grad, p


def smooth_objective(x, T):
    """J with smooth max approximation."""
    a = np.maximum(x, 0.0)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(x)
    n = len(a)
    sm_val, grad_sm, _ = soft_max_grad(a, T)
    obj = 2.0 * n * sm_val / (s * s)
    grad = (2.0 * n / (s * s)) * grad_sm - (4.0 * n * sm_val / (s**3)) * np.ones(n)
    return obj, grad

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for i in range(jl, jh + 1):
                rows.append(k)
                cols.append(i)
                vals.append(a[k - i])
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):

    return A.tocsr()

        res = minimize(_obj, best_a, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, None)] * n,
                       options={'maxiter': 150, 'max_fun': 3000, 'ftol': 1e-18, 'gtol': 1e-14})

        if res.x is not None:
            a_trial = np.maximum(res.x, 0.0)
            s = a_trial.sum()
            if s > 0.01:
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if np.isfinite(J_trial) and J_trial < best_J:
                    best_J = J_trial
                    best_a = a_trial.copy()

    return best_a, best_J


def perturbation_search(a, deadline, rng, n_trials=300):
    """Random perturbation search to escape local minima."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_J = compute_J(a)
    best_a = a.copy()
    scale = 1.0

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        a_trial = best_a.copy()
        # Perturb a random subset
        n_pert = rng.integers(1, max(2, n // 8))
        idx = rng.choice(n, size=n_pert, replace=False)
        a_trial[idx] = np.maximum(a_trial[idx] + rng.normal(0, scale * a.mean(), n_pert), 0.0)

        s = a_trial.sum()
        if s < 0.01:
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)

        if J_trial < best_J:
            best_J = J_trial
            best_a = a_trial.copy()
            scale = max(0.1, scale * 0.9)
        else:
            scale = min(2.0, scale * 1.02)

    return best_a, best_J


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def make_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    t = np.linspace(0, 1, n, endpoint=False)
    inits.append(np.ones(n))

    # Sinusoidal
    for freq in [1, 2, 3, 4, 5, 7, 11, 17]:
        phase = rng.uniform(0, 2 * np.pi)
        inits.append(0.6 + 0.4 * np.cos(2 * np.pi * freq * t + phase))

    # Random
    inits.append(rng.uniform(0.1, 1.0, n))

    # Gaussian-shaped
    inits.append(0.3 + 0.7 * np.exp(-4.0 * (t - 0.5) ** 2))

    # Two-peak
    inits.append(0.2 + 0.6 * np.exp(-8 * (t - 0.3) ** 2) + 0.6 * np.exp(-8 * (t - 0.7) ** 2))

    # Sparse + smooth
    sp = np.zeros(n)
    n_sp = max(3, n // 20)
    sp[rng.choice(n, n_sp, replace=False)] = 1.0
    ksize = max(3, n // 32)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sp, kernel, 'same') + 0.05)

    # Power law
    inits.append(0.3 + 0.7 * (t + 0.1) ** rng.uniform(-0.5, 0.5))

    # Sine wave with noise
    inits.append(0.5 + 0.3 * np.sin(4 * np.pi * t) + 0.1 * rng.standard_normal(n))

    return inits


class Tracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t0 = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t0
            print(f"  *** J={score:.8f} n={a.size} t={elapsed:.0f}s ***", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  [report] {label}: J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t0:.0f}s", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search: LP descent + smooth max gradient + perturbation, multi-length.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 30.0
    tracker = Tracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, TypeError):
        pass

    # === Phase 1: LP descent at multiple lengths ===
    lengths = [128, 256, 512]
    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 1: LP descent n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        inits = make_inits(n, rng, parent)
        # Scale iterations based on length and remaining time
        time_left = deadline - time.monotonic()
        iters_per_init = max(20, min(150, int(200 * time_left / (len(lengths) * len(inits) * n))))

        for i, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 = a0 / a0.sum()
            a, J = lp_descent(a0, iters_per_init, deadline)
            tracker.update(a)

        elapsed = time.monotonic() - t0
        tracker.report(f"LP n={n} ({elapsed:.1f}s)")

    # === Phase 2: Smooth max gradient refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: Smooth max refinement n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        a, J = smooth_refine(tracker.best, deadline)
        tracker.update(a)
        tracker.report(f"Smooth refine ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Perturbation search ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Perturbation search n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        a, J = perturbation_search(tracker.best, deadline, rng, n_trials=400)
        tracker.update(a)
        tracker.report(f"Perturbation ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 512:
            print(f"\n{'='*60}", flush=True)
            print(f"Phase 4: LP polish n={n}", flush=True)
            print(f"{'='*60}", flush=True)
            a, J = lp_descent(tracker.best, 200, deadline)
            tracker.update(a)
            tracker.report(f"LP polish ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Final smooth refine ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: Final smooth refine", flush=True)
        print(f"{'='*60}", flush=True)
        a, J = smooth_refine(tracker.best, deadline)
        tracker.update(a)
        tracker.report(f"Final smooth ({time.monotonic()-t0:.1f}s)")

    # === Phase 6: One more perturbation if time remains ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: Final perturbation", flush=True)
        print(f"{'='*60}", flush=True)
        a, J = perturbation_search(tracker.best, deadline, rng, n_trials=200)
        tracker.update(a)
        tracker.report(f"Final perturb ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128) / 128.0
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t0:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
