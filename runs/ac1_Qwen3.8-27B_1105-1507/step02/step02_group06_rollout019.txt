def optimize_subgradient(a, deadline, top_k=20, lr_init=0.01):
    n = len(a)
    best_a = a.copy()
    best_score = compute_score_fast(a)
    S = float(np.sum(a))
    
    momentum = np.zeros(n)
    momentum_decay = 0.9
    lr = lr_init
    no_improve = 0
    
    a = a.copy()
    
    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        # Compute weighted subgradient
        threshold = float(np.percentile(conv, 90))  # top 10% lags
        excess = np.maximum(conv - threshold, 0)
        total_excess = float(np.sum(excess))
        
        grad = np.zeros(n)
        if total_excess > 1e-15:
            weights = excess / total_excess
            # Only process lags with significant weight
            significant = weights > 1e-8
            for k in np.where(significant)[0]:
                if weights[k] > 1e-10:
                    grad += weights[k] * conv_grad_k(a, k)
        else:
            # Fallback: top-k lags
            top_idx = np.argsort(conv)[-top_k:]
            for k in top_idx:
                grad += conv_grad_k(a, k) / top_k
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        
        # Momentum update
        momentum = momentum_decay * momentum - lr * grad
        new_a = a + momentum
        new_a = np.clip(new_a, 0.0, None)
        
        # Renormalize
        new_S = float(np.sum(new_a))
        if new_S < 1e-10:
            lr *= 0.5
            no_improve += 1
            if no_improve > 100:
                break
            continue
        new_a *= S / new_S
        
        new_score = compute_score_fast(new_a)
        
        if new_score < best_score:
            best_score = new_score
            best_a = new_a.copy()
            a = new_a
            no_improve = 0
            # Slowly decrease lr
            lr *= 0.999
        else:
            no_improve += 1
            lr *= 0.95
            momentum *= 0.5
            if lr < 1e-8 or no_improve > 200:
                break
    
    return best_a, best_score

        if new_S < 1e-10:
            lr *= 0.5
            no_improve += 1
            if no_improve > 50:
                break
            it += 1
            continue
        new_a *= S / new_S

        new_score = compute_score(new_a)
        it += 1
        if new_score < best_score:
            best_score = new_score
            best_a = new_a.copy()
            a = new_a
            no_improve = 0
            lr *= 0.999
        else:
            no_improve += 1
            lr *= 0.97
            momentum *= 0.5
            if lr < 1e-8 or no_improve > 200:
                break

    return best_a, best_score


def lp_optimize(a, deadline, step_init=0.1, verbose_prefix=""):
    """LP-based optimization with line search."""
    n = len(a)
    S = float(np.sum(a))
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = compute_score(a)
    step = step_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        if M <= 0:
            break

        t_start = time.monotonic()
        g = solve_lp_direction(a, M)
        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        lp_time = time.monotonic() - t_start

        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            no_improve += 1
            continue

        g *= S / sum_g

        # Line search
        best_step_score = best_score
        best_step_f = None
        for alpha in [step, step * 0.5, step * 1.5, step * 0.25, 0.01]:
            alpha = min(alpha, 1.0)
            candidate = (1.0 - alpha) * a + alpha * g
            candidate = np.clip(candidate, 0.0, 1000.0)
            cand_S = float(np.sum(candidate))
            if cand_S < 1e-10:
                continue
            candidate *= S / cand_S
            cand_score = compute_score(candidate)
            if np.isfinite(cand_score) and cand_score < best_step_score:
                best_step_score = cand_score
                best_step_f = candidate

        it += 1
        if best_step_f is not None:
            a = best_step_f
            best_a = a.copy()
            best_score = best_step_score
            no_improve = 0
            step *= 0.99
            step = max(step, 0.01)
        else:
            no_improve += 1
            step *= 0.7
            if step < 0.005 or no_improve > 30:
                break

    return best_a, best_score


def make_initializations(n, rng, parent=None):
    """Create diverse initializations for length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(('uniform', np.ones(n)))

    # Cosine taper
    cos_t = (1.0 + np.cos(np.pi * idx / max(1, n - 1))) / 2.0
    inits.append(('cosine', cos_t))

    # Bell (centered)
    center = (n - 1) / 2.0
    bell = np.exp(-0.5 * ((idx - center) / (n / 5.0)) ** 2)
    inits.append(('bell', bell))

    # Bimodal
    two_peak = (np.exp(-0.5 * ((idx - n * 0.2) / (n / 10.0)) ** 2) +
                np.exp(-0.5 * ((idx - n * 0.8) / (n / 10.0)) ** 2))
    inits.append(('bimodal', two_peak))

    # Exponential decay
    inits.append(('exp_dec', np.exp(-3.0 * idx / n)))

    # Random smooth (low-pass filtered noise)
    noise = rng.uniform(0, 1, size=n)
    from scipy.ndimage import uniform_filter1d
    smooth_noise = uniform_filter1d(noise, size=n // 10 + 1)
    inits.append(('smooth_rand', smooth_noise + 0.1))

    # Sine wave
    inits.append(('sine', 0.5 + 0.5 * np.sin(2 * np.pi * 3 * idx / n)))

    # Quadratic (arch)
    quad = 1.0 - (2.0 * idx / max(1, n - 1) - 1.0) ** 2
    inits.append(('quadratic', np.clip(quad, 0.05, None)))

    # Parent interpolated
    if parent is not None and len(parent) != n:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        parent_interp = np.interp(x_new, x_old, parent)
        inits.append(('parent_interp', np.clip(parent_interp, 0.01, None)))
    elif parent is not None and len(parent) == n:
        inits.append(('parent', parent.copy()))

    return inits


def propose_candidate(budget_s=1500, **kwargs):
    """Main entry point: search for the best sequence minimizing the objective."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 10.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).copy()
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = p
            score = evaluate_sequence([float(x) for x in parent])
            if np.isfinite(score):
                best_score = score
                best_sequence = parent.copy()
                print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"No parent available: {e}")

    # Phase 1: Exploration (first 200 seconds)
    print("\n=== Phase 1: Exploration ===")
    explore_deadline = t0 + 200.0

    lengths_to_try = [1024, 2048, 4096]
    if parent is not None:
        lengths_to_try.insert(0, len(parent))

    # Remove duplicates and sort
    lengths_to_try = sorted(set(lengths_to_try))

    candidates = []  # (score, sequence, n, label)

    for n in lengths_to_try:
        if time.monotonic() > explore_deadline - 20:
            break
        print(f"\n--- Exploring n={n} ---")

        inits = make_initializations(n, rng, parent=parent)
        n_inits = len(inits)
        time_per_init = max(3.0, (explore_deadline - time.monotonic() - 5) / max(1, n_inits))

        for label, f_init in inits:
            if time.monotonic() > explore_deadline - 5:
                break
            f_init = np.clip(f_init.astype(np.float64), 0.01, 1000.0)
            if np.sum(f_init) < 1e-10:
                continue
            init_score = compute_score(f_init)

            # Quick optimization
            quick_dl = min(explore_deadline - 3, time.monotonic() + time_per_init)
            f_opt, s_opt = quick_subgradient(f_init, quick_dl, lr_init=0.05)
            if s_opt < best_score:
                best_score = s_opt
                best_sequence = f_opt.copy()

            candidates.append((s_opt, f_opt.copy(), n, label))
            print(f"  {label}: init={init_score:.6f} -> quick={s_opt:.6f}")

    # Sort and show top candidates
    candidates.sort(key=lambda x: x[0])
    print(f"\nTop candidates after exploration:")
    for sc, seq, n, lbl in candidates[:5]:
        print(f"  n={n} {lbl}: {sc:.6f}")
    print(f"Best so far: {best_score:.6f}")

    # Phase 2: Focused optimization (next ~1000 seconds)
    print("\n=== Phase 2: Focused Optimization ===")
    focus_deadline = deadline - 200.0

    # Select top candidates with unique lengths
    seen_n = set()
    top_candidates = []
    for sc, seq, n, lbl in candidates:
        if n not in seen_n:
            top_candidates.append((sc, seq, n, lbl))
            seen_n.add(n)
        if len(top_candidates) >= 4:
            break

    # Allocate time proportionally
    n_focused = len(top_candidates)
    time_per_candidate = (focus_deadline - time.monotonic()) / max(1, n_focused)

    for idx, (sc, f_init, n, lbl) in enumerate(top_candidates):
        remaining = focus_deadline - time.monotonic()
        if remaining < 30:
            break
        alloc_time = min(time_per_candidate, remaining)
        cand_deadline = time.monotonic() + alloc_time
        print(f"\n--- Focused: n={n} {lbl} (start={sc:.6f}, time={alloc_time:.0f}s) ---")

        # For smaller n, use LP-assisted optimization
        if n <= 2048:
            # Phase 2a: Nesterov subgradient (60% of time)
            sub_dl = time.monotonic() + (cand_deadline - time.monotonic()) * 0.6
            f_opt, s_opt = nesterov_optimize(
                f_init, sub_dl, beta_init=5.0, beta_final=40.0,
                lr_init=0.02, momentum_init=0.8, lp_every=200,
                verbose_prefix=f"[n={n} {lbl}] ")
            if s_opt < best_score:
                best_score = s_opt
                best_sequence = f_opt.copy()
                print(f"  Nesterov: {s_opt:.6f}")

            # Phase 2b: LP polish (30% of time)
            if time.monotonic() < cand_deadline - 5:
                lp_dl = time.monotonic() + (cand_deadline - time.monotonic()) * 0.5
                f_opt2, s_opt2 = lp_optimize(f_opt, lp_dl, step_init=0.08,
                                             verbose_prefix=f"[n={n} LP] ")
                if s_opt2 < best_score:
                    best_score = s_opt2
                    best_sequence = f_opt2.copy()
                    print(f"  LP: {s_opt2:.6f}")

            # Phase 2c: Final subgradient (remaining)
            if time.monotonic() < cand_deadline - 2:
                f_opt3, s_opt3 = quick_subgradient(f_opt, cand_deadline, lr_init=0.01)
                if s_opt3 < best_score:
                    best_score = s_opt3
                    best_sequence = f_opt3.copy()
                    print(f"  Final sub: {s_opt3:.6f}")
        else:
            # For larger n, only subgradient (LP too expensive)
            f_opt, s_opt = nesterov_optimize(
                f_init, cand_deadline, beta_init=5.0, beta_final=50.0,
                lr_init=0.015, momentum_init=0.85, lp_every=0,
                verbose_prefix=f"[n={n} {lbl}] ")
            if s_opt < best_score:
                best_score = s_opt
                best_sequence = f_opt.copy()
                print(f"  Nesterov: {s_opt:.6f}")

    # Phase 3: Final polish and restart attempts (last 200 seconds)
    print(f"\n=== Phase 3: Final Polish (best={best_score:.6f}) ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        print(f"Polishing n={n_best}...")

        # Try a few random perturbations to escape local minima
        polish_deadline = deadline - 5
        perturbations = 3
        p_time = (polish_deadline - time.monotonic()) / (perturbations + 2)

        for p_idx in range(perturbations):
            if time.monotonic() > polish_deadline - p_time:
                break
            # Add small random perturbation
            noise = rng.normal(0, 0.05, size=n_best)
            perturbed = best_sequence + noise
            perturbed = np.clip(perturbed, 0.01, 1000.0)
            perturbed *= float(np.sum(best_sequence)) / float(np.sum(perturbed))

            f_opt, s_opt = quick_subgradient(perturbed, time.monotonic() + p_time,
                                              lr_init=0.02)
            if s_opt < best_score:
                best_score = s_opt
                best_sequence = f_opt.copy()
                print(f"  Perturb {p_idx}: {s_opt:.6f}")

        # Final LP polish if n is small enough
        if n_best <= 2048 and time.monotonic() < polish_deadline - 10:
            f_pol, s_pol = lp_optimize(best_sequence, polish_deadline, step_init=0.03)
            if s_pol < best_score:
                best_score = s_pol
                best_sequence = f_pol.copy()
                print(f"  Final LP: {s_pol:.6f}")

        # Final subgradient polish
        if time.monotonic() < polish_deadline - 2:
            f_pol, s_pol = quick_subgradient(best_sequence, polish_deadline,
                                              lr_init=0.005)
            if s_pol < best_score:
                best_score = s_pol
                best_sequence = f_pol.copy()
                print(f"  Final sub: {s_pol:.6f}")

    # Fallback
    if best_sequence is None:
        print("WARNING: No valid sequence found. Using fallback.")
        best_sequence = np.ones(512)
        best_score = evaluate_sequence([float(x) for x in best_sequence])

    # Final verification
    final_score = evaluate_sequence([float(x) for x in best_sequence])
    if not np.isfinite(final_score):
        print("WARNING: Final sequence failed! Using uniform fallback.")
        best_sequence = np.ones(256)
        final_score = evaluate_sequence([float(x) for x in best_sequence])

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, "
          f"internal={best_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_sequence]
