import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max(b, T):
    """Smooth approximation of max(b) using log-sum-exp."""
    b_max = b.max()
    return b_max + T * np.log(np.sum(np.exp((b - b_max) / T)))


def build_toeplitz_rows(a):
    """Build row/column/value arrays for the sparse convolution matrix."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            # a[k-j] for j in [jl, jh], i.e., a[k-jh : k-jl+1] reversed
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    return rows, cols, vals


def lp_best_response(a, M, tol=0.0):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    rows, cols, vals = build_toeplitz_rows(a)
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    A = A.tocsr()
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + tol)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-12:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def run_smoothed_lbfgs(a, T_schedule, deadline, tracker):
    """Run L-BFGS-B with softmax-smoothed objective and adaptive T."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    n = a.size
    tracker.update(a)
    
    for T in T_schedule:
        if time.monotonic() > deadline:
            break
        
        def objective(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-8:
                return 1e10
            ax = ax / s
            b = fast_conv(ax)
            return smooth_max(b, T)
        
        def gradient(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-8:
            return np.zeros(n)
            ax = ax / s
            b = fast_conv(ax)
            b_max = b.max()
            w = np.exp((b - b_max) / T)
            w /= w.sum()
            grad = np.correlate(w, ax, mode='full')[:n]
            return grad
        
        bounds_list = [(0.0, None)] * n
        try:
            res = minimize(objective, a, jac=gradient, method='L-BFGS-B',
                           bounds=bounds_list,
                           options={'maxiter': 200, 'max_fun': 50000, 'ftol': 1e-14, 'gtol': 1e-10})
            if res.fun < 1e10:
                a = np.maximum(res.x, 0.0)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                tracker.update(a)
        except Exception as e:
            print(f"  L-BFGS-B error at T={T}: {e}", flush=True)
            break
    
    return a


def run_lp_iterations_v2(a, max_iters, deadline, tracker, use_momentum=False):
    """Improved LP mirror descent with backtracking and optional momentum."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    # Optional momentum state
    v = np.zeros_like(a) if use_momentum else None
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum
        
        # Backtracking line search
        best_a = None
        best_J = current_J
        t = 0.5
        for _ in range(20):
            if use_momentum:
                v = 0.9 * v + 0.1 * (g - a)
                a_trial = a + t * v
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-15:
                best_J = J_trial
                best_a = a_trial
                t = min(1.0, t * 1.2)  # Try larger step next
            else:
                t *= 0.5
            if t < 0.001:
                break
        
        if best_a is None:
            # Try fixed small steps
            found = False
            for t_try in [0.05, 0.03, 0.01, 0.005]:
                a_trial = (1.0 - t_try) * a + t_try * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-15:
                    a = a_trial
                    current_J = J_trial
                    found = True
                    break
            if not found:
                break
        else:
            a = best_a
            current_J = best_J
        
        if it % 20 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def nesterov_subgradient(a, max_iters, deadline, tracker, T=0.1):
    """Nesterov-accelerated subgradient method on smoothed objective."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = a.size
    
    y = a.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        beta = 2.0 / (it + 2.0)
        a_new = y + beta * (a - y)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        # Compute gradient of smoothed objective
        b = fast_conv(a_new)
        b_max = b.max()
        w = np.exp((b - b_max) / T)
        w /= w.sum()
        grad = np.correlate(w, a_new, mode='full')[:n]
        
        # Step
        step_size = 1.0 / (np.linalg.norm(grad) + 1e-10)
        a_next = a_new - step_size * grad
        a_next = np.maximum(a_next, 0.0)
        s = a_next.sum()
        if s < 1e-10:
            break
        a_next = a_next / s
        
        y = a_next
        a = a_next
        
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
        
        if it % 50 == 0:
            tracker.update(best_a)
    
    tracker.update(best_a)
    return best_a

class ScoreTracker:

    def __init__(self):
        self.best = None
        self.best_score = float('inf')

        try:

                return

                self.best_score = score

            pass

            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size} t={time.monotonic()-self.t_start:.1f}s", flush=True)

- The max of b is at least b_{n-1} (the middle term)

For b_{n-1} = sum_i a_i a_{n-1-i}: this is the "reversed dot product" of a with itself.
By Cauchy-Schwarz: b_{n-1} <= (sum a_i^2) but that's an upper bound.
By the constraint sum a_i = 1: the minimum of b_{n-1} is... if we put all mass at one point, say a_0 = 1, then b_{n-1} = 0 (since a_{n-1} = 0). But then b_0 = 1.

So the tradeoff is between b_0 (= a_0^2), b_{2n-2} (= a_{n-1}^2), and b_{n-1} (the middle).

For a uniform distribution a_i = 1/n: b_{n-1} = n * (1/n)^2 = 1/n, and b_0 = 1/n^2. So max(b) = 1/n, and J = 2n * (1/n) = 2.

For a triangular distribution a_i proportional to min(i+1, n-i): this spreads the mass more in the middle, increasing b_{n-1} but decreasing b_0.

The optimal balance is what we're searching for.

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Multi-resolution LP mirror descent with backtracking and momentum
2. Scale-invariant softmax-smoothed L-BFGS-B with adaptive T schedule
3. Nesterov-accelerated subgradient on smoothed objective
4. Active-set-aware LP polish
5. Basin-hopping exploration with multi-length resampling
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    if n == 0:
        return np.array([])
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smooth_max_value(b, T):
    """Compute T * log(sum(exp(b/T))) with numerical stability."""
    b_max = b.max()
    return float(b_max + T * np.log(np.sum(np.exp((b - b_max) / T))))


def smooth_max_weights(b, T):
    """Compute softmax weights for the smoothing."""
    b_max = b.max()
    w = np.exp((b - b_max) / T)
    w /= w.sum()
    return w


def conv_gradient(w, a):
    """
    Compute gradient of sum_k w_k * b_k where b = conv(a,a).
    grad_j = sum_k w_k * a_{k-j} = np.correlate(w, a, 'full')[:n]
    """
    return np.correlate(w, a, mode='full')[:len(a)]


def build_conv_matrix_sparse(a):
    """Build sparse matrix A where (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    if len(rows) == 0:
        return lil_matrix((2 * n - 1, n)).tocsr()
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-12:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size, endpoint=False)
    t_new = np.linspace(0, 1, dest_n, endpoint=False)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def run_lp_mirror_descent(a, max_iters, deadline, tracker, use_momentum=False):
    """LP mirror descent with backtracking line search and optional momentum."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    v = np.zeros_like(a) if use_momentum else None
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum

        # Backtracking line search
        best_a = None
        best_J = current_J
        t = max(0.05, 0.5 * (1.0 - 0.7 * it / max(1, max_iters)))

        for _bt in range(15):
            if use_momentum:
                v = 0.85 * v + 0.15 * (g - a)
                a_trial = a + t * v
            else:
                a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                t *= 0.5
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial
                t_next = min(1.0, t * 1.3)
            else:
                t_next = t * 0.5
            t = t_next
            if t < 0.001:
                break

        if best_a is None:
            stagnant += 1
            if stagnant > 10:
                break
            # Try a few fixed small steps
            found = False
            for t_try in [0.05, 0.03, 0.01, 0.005, 0.001]:
                a_trial = (1.0 - t_try) * a + t_try * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                J_trial = compute_J(a_trial)
                if J_trial < current_J - 1e-14:
                    a = a_trial
                    current_J = J_trial
                    stagnant = 0
                    found = True
                    break
            if not found:
                break
        else:
            a = best_a
            current_J = best_J
            stagnant = 0

        if it % 15 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_smoothed_lbfgs(a, T_schedule, deadline, tracker, maxiter_per_T=150):
    """L-BFGS-B on scale-invariant smoothed objective."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = a.size
    tracker.update(a)

    for T in T_schedule:
        if time.monotonic() > deadline:
            break
        T = max(T, 0.01)
        print(f"    L-BFGS-B T={T:.4f}", flush=True)

        def obj(x):
            ax = np.maximum(x, 0.0)
            s2 = ax.sum()
            if s2 < 1e-8:
                return 1e10
            b = fast_conv(ax)
            S = smooth_max_value(b, T)
            return S / (s2 * s2)

        def grad(x):
            ax = np.maximum(x, 0.0)
            s2 = ax.sum()
            if s2 < 1e-8:
                return np.zeros(n)
            b = fast_conv(ax)
            w = smooth_max_weights(b, T)
            S = smooth_max_value(b, T)
            dS = conv_gradient(w, ax)
            return (dS - 2.0 * S / s2) / (s2 * s2)

        bounds_list = [(0.0, None)] * n
        try:
            res = minimize(obj, a, jac=grad, method='L-BFGS-B',
                           bounds=bounds_list,
                           options={'maxiter': maxiter_per_T, 'max_fun': 30000,
                                    'ftol': 1e-15, 'gtol': 1e-12})
            if res.fun < 1e10:
                a_new = np.maximum(res.x, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-10:
                    a = a_new / s_new
                    tracker.update(a)
        except Exception as e:
            print(f"    L-BFGS-B error T={T}: {e}", flush=True)
            continue

    return a


def run_nesterov_subgrad(a, max_iters, deadline, tracker, T=0.1):
    """Nesterov-accelerated subgradient on smoothed scale-invariant objective."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    n = a.size
    tracker.update(a)

    y = a.copy()
    best_a = a.copy()
    best_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        beta = 2.0 / (it + 2.0)
        a_curr = np.maximum(y + beta * (a - y), 0.0)
        s_curr = a_curr.sum()
        if s_curr < 1e-10:
            break
        a_curr = a_curr / s_curr

        b = fast_conv(a_curr)
        w = smooth_max_weights(b, T)
        S = smooth_max_value(b, T)
        dS = conv_gradient(w, a_curr)
        grad = (dS - 2.0 * S / s_curr) / (s_curr * s_curr)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step = 1.0 / (gnorm * np.sqrt(it + 1))
        a_next = a_curr - step * grad
        a_next = np.maximum(a_next, 0.0)
        s_next = a_next.sum()
        if s_next < 1e-10:
            break
        a_next = a_next / s_next

        y = a_next
        a = a_next

        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()

        if it % 100 == 0:
            tracker.update(best_a)

    tracker.update(best_a)
    return best_a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))
    inits.append(np.minimum(t, 1.0 - t) + 0.05)
    inits.append(1.0 - 0.6 * (2.0 * t - 1.0) ** 2)
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * t)) + 0.05)
    inits.append(np.exp(-4.0 * (t - 0.5) ** 2) + 0.03)
    inits.append(0.05 + 0.45 * np.exp(-6.0 * (t - 0.25) ** 2) + 0.45 * np.exp(-6.0 * (t - 0.75) ** 2))
    inits.append(0.4 + 0.5 * np.sin(np.pi * t))
    inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * t + rng.uniform(0, 6.28)))
    inits.append(0.8 + 0.3 * np.cos(4.0 * np.pi * t + rng.uniform(0, 6.28)))

    # Random smooth
    sig = np.ones(n)
    for _ in range(3):
        f = rng.integers(1, 8)
        amp = rng.uniform(0.1, 0.4)
        ph = rng.uniform(0, 2 * np.pi)
        sig += amp * np.cos(2.0 * np.pi * f * t + ph)
    inits.append(np.maximum(sig, 0.01))

    # Sparse-ish
    sparse = np.zeros(n)
    n_spikes = max(3, n // 20)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel = np.ones(5) / 5.0
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.03)

    return inits


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        try:
            a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
            if a.size == 0 or a.sum() < 0.01:
                return
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < self.best_score:
                self.best = a.copy()
                self.best_score = score
                elapsed = time.monotonic() - self.t_start
                print(f"  * J={score:.8f} n={a.size} t={elapsed:.1f}s", flush=True)
        except Exception:
            pass

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} J={self.best_score:.8f} n={self.best.size} "
                  f"t={time.monotonic()-self.t_start:.1f}s", flush=True)
