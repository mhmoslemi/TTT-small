    val = smooth_max / (s * s)
    # Gradient of smooth_max w.r.t. x:
    # grad_smooth[i] = 2 * sum_k weights[k] * x[k-i] * 1_{0<=k-i<n}
    c = np.correlate(weights, x, mode='full')
    grad_smooth = 2.0 * c[n - 1:2 * n - 1]
    # Gradient of smooth_max / s^2:
    grad = grad_smooth / (s * s) - 2.0 * smooth_max / (s * s * s)
    return val, grad


def lbgf_continuation(a, tau_values, max_iter_per_tau=5000, verbose=False):
    """L-BFGS-B with temperature continuation. Start with large tau, decrease."""
    n = len(a)
    a = normalize_a(a.copy())
    for idx, tau in enumerate(tau_values):
        def obj_and_grad(x):
            return smooth_obj_and_grad(x, tau, n)

        result = minimize(obj_and_grad, a, method='L-BFGS-B',
                          bounds=[(0.0, None)] * n,
                          options={'maxiter': max_iter_per_tau,
                                   'ftol': 1e-14, 'gtol': 1e-10,
                                   'maxls': 50})
        a = normalize_a(result.x)
        if verbose:
            print(f"      L-BFGS tau={tau:.1f}: smooth_val={result.fun:.6f}, true_obj={compute_obj(a):.6f}")
    return a


def build_A_matrix(a):
    """Build the (2n-1) x n matrix A where conv(a,g)_k = (A @ g)_k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def lp_direction(a, M, time_limit=30):
    """Solve LP: max sum(g) s.t. conv(a,g)_k <= M for all k, g >= 0.
    Returns (g, sum_g) or (None, None) on failure.
    """
    n = len(a)
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = res.x
            sum_g = float(np.sum(g))
            return g, sum_g
    except Exception:
        pass
    return None, None


def lp_optimize(a, max_iter=50, t=0.4, verbose=False, time_budget=60.0):
    """Iterative LP-based optimization. Returns best a found."""
    n = len(a)
    a = normalize_a(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    stall = 0

    for it in range(max_iter):
        if time.monotonic() > time_budget:
            break
        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = lp_direction(a, M, time_limit=20)
        if g is None:
            break

        if sum_g <= 1.0 + 1e-6:
            stall += 1
            if stall > 15:
                break
            continue
        stall = 0

        g = g / sum_g
        a_new = (1.0 - t) * a + t * g
        a_new = normalize_a(a_new)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose:
                print(f"      LP it={it}: obj={best_obj:.6f}, sum_g={sum_g:.4f}")
        a = a_new

    return best_a, best_obj


def random_init(n, rng):
    """Generate a random initialization with diverse structure."""
    choice = rng.integers(0, 6)
    x = np.arange(n)
    if choice == 0:
        a = 0.5 * (1 + np.cos(2 * np.pi * x / n + rng.uniform(0, 2 * np.pi)))
    elif choice == 1:
        a = np.ones(n)
        lo = int(n * rng.uniform(0.2, 0.4))
        hi = int(n * rng.uniform(0.6, 0.8))
        a[lo:hi] = rng.uniform(0.1, 0.5)
    elif choice == 2:
        a = rng.random(n)
        freq = rng.uniform(0.5, 3.0)
        a *= (0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n))
    elif choice == 3:
        center = n / 2.0
        rate = rng.uniform(0.5, 3.0)
        a = np.exp(-rate * np.abs(x - center) / center)
    elif choice == 4:
        # Two-peak structure
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        a = np.exp(-2.0 * (x - c1)**2 / n) + np.exp(-2.0 * (x - c2)**2 / n)
    else:
        a = rng.random(n)
        # Add periodic modulation
        period = int(rng.integers(n // 4, n))
        mod = 0.5 + 0.5 * np.cos(2 * np.pi * x / period)
        a *= mod
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s


def optimize_single_length(n, seed, time_budget, verbose=False):
    """Full optimization pipeline for a single sequence length."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    # Try to use parent if available and matching length
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.size == n:
            parent = p.copy()
    except Exception:
        pass

    num_inits = max(2, int(time_budget / 40))

    for init_idx in range(num_inits):
        if time.monotonic() - start > time_budget - 5:
            break

        # Initialize
        if init_idx == 0 and parent is not None:
            a = normalize_a(parent.copy())
        else:
            a = random_init(n, rng)

        if verbose:
            init_obj = compute_obj(a)
            print(f"    n={n} init{init_idx}: obj={init_obj:.6f}")

        # Phase 1: L-BFGS continuation (smooth -> sharp)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            tau_values = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0]
            max_iter = min(5000, int(remaining * 2))
            a = lbgf_continuation(a, tau_values, max_iter_per_tau=max(200, max_iter // len(tau_values)))

        # Phase 2: LP optimization
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 500:
            t_lp = 0.4 if init_idx == 0 else 0.3
            a_lp, obj_lp = lp_optimize(a, max_iter=min(30, int(remaining // 3)),
                                        t=t_lp, verbose=verbose,
                                        time_budget=time_budget - (time.monotonic() - start) - 10)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()
            a = a_lp

        # Phase 3: Another L-BFGS polish with small tau
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            tau_values = [5.0, 2.0, 1.0, 0.5]
            max_iter = min(3000, int(remaining * 1.5))
            a_pol = lbgf_continuation(a, tau_values, max_iter_per_tau=max(100, max_iter // len(tau_values)))
            obj_pol = compute_obj(a_pol)
            if obj_pol < best_obj:
                best_obj = obj_pol
                best_a = a_pol.copy()
            a = a_pol

        # Phase 4: Final LP polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 400:
            a_final, obj_final = lp_optimize(a, max_iter=min(20, int(remaining // 3)),
                                              t=0.25, verbose=False,
                                              time_budget=time_budget - (time.monotonic() - start) - 5)
            if obj_final < best_obj:
                best_obj = obj_final
                best_a = a_final.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def try_parent_sequence():
    """Try parent sequence as starting point."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < np.inf:
                return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-phase optimization: L-BFGS continuation + LP + polishing.
    Explores sequence lengths 150-500, starting from parent if available.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Try parent
    parent, parent_obj = try_parent_sequence()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        print(f"Parent (n={parent.size}): obj={best_score:.6f}")

    # Determine n values to explore
    n_values = [150, 200, 250, 300, 350, 400]
    if parent is not None:
        n_p = len(parent)
        n_values = sorted(set(n_values + [n_p, max(50, n_p - 50), min(500, n_p + 50)]))

    # Time allocation: more time for the parent length and nearby
    elapsed = time.monotonic() - start_time
    remaining_total = deadline - time.monotonic()
    print(f"Total time budget: {float(budget_s)}s, {remaining_total:.0f}s available")
    print(f"Exploring n values: {n_values}")

    # Sort by priority: parent length first, then nearby
    if parent is not None:
        n_p = len(parent)
        n_values.sort(key=lambda x: abs(x - n_p))

    for n in n_values:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Allocate time based on proximity to parent length and remaining budget
        if parent is not None and n == len(parent):
            alloc = min(remaining * 0.3, 200.0)
        elif parent is not None and abs(n - len(parent)) <= 50:
            alloc = min(remaining * 0.2, 120.0)
        else:
            alloc = min(remaining * 0.15, 80.0)
        alloc = max(15.0, alloc)

        print(f"\nN={n}: {remaining:.0f}s remaining, allocating {alloc:.0f}s")

        a_opt, obj = optimize_single_length(n, seed=seed + n * 7919,
                                             time_budget=alloc, verbose=False)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f}")
        else:
            print(f"  n={n}, obj={obj:.6f} (no improvement)")

    # Final intensive polishing on the best sequence found
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== Final polishing (n={n_best}, {remaining:.0f}s) ===")

        a_cur = best_seq.copy()
        polish_round = 0

        while time.monotonic() < deadline - 8:
            polish_round += 1
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break

            # Round 1: L-BFGS with very small tau
            if remaining > 10:
                tau_vals = [3.0, 1.5, 1.0, 0.5, 0.3]
                max_it = min(4000, int(remaining * 0.8))
                a_cur = lbgf_continuation(a_cur, tau_vals,
                                          max_iter_per_tau=max(200, max_it // len(tau_vals)))
                obj_cur = compute_obj(a_cur)
                if obj_cur < best_score:
                    best_score = obj_cur
                    best_seq = a_cur.copy()
                    print(f"  Polish {polish_round} L-BFGS: obj={best_score:.6f}")

            # Round 2: LP with small t for careful improvement
            remaining = deadline - time.monotonic()
            if remaining > 15 and n_best <= 400:
                a_lp, obj_lp = lp_optimize(a_cur, max_iter=min(30, int(remaining // 3)),
                                            t=0.2, verbose=False,
                                            time_budget=time.monotonic() + remaining - 10)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    print(f"  Polish {polish_round} LP: obj={best_score:.6f}")

            # Round 3: Try perturbation + re-optimize
            remaining = deadline - time.monotonic()
            if remaining > 20:
                for pert_scale in [0.03, 0.01]:
                    a_pert = a_cur * np.exp(rng.normal(0, pert_scale, n_best))
                    a_pert = normalize_a(a_pert)
                    a_pert = lbgf_continuation(a_pert, [2.0, 1.0], max_iter_per_tau=2000)
                    obj_pert = compute_obj(a_pert)
                    if obj_pert < best_score:
                        best_score = obj_pert
                        best_seq = a_pert.copy()
                        a_cur = a_pert.copy()
                        print(f"  Polish {polish_round} perturb({pert_scale}): obj={best_score:.6f}")

            # Check time
            if time.monotonic() > deadline - 10:
                break

    # Absolute final: one more L-BFGS pass
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_final = lbgf_continuation(best_seq, [1.0, 0.5], max_iter_per_tau=int(remaining * 3))
            obj_final = compute_obj(a_final)
            if obj_final < best_score:
                best_score = obj_final
                best_seq = a_final.copy()
                print(f"  Final L-BFGS: obj={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(250)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    # Return as list of floats
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]

    # Validate
    try:
        validation = evaluate_sequence(result)
        if validation < np.inf and validation < best_score:
            print(f"  Validation: {validation:.6f} (lower than computed {best_score:.6f})")
            best_score = validation
        elif validation < np.inf:
            print(f"  Validation: {validation:.6f} (matches computed {best_score:.6f})")
    except Exception:
        pass

    return result
