"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-12:
        return np.ones_like(a) / len(a)
    return a / s


def build_A_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_for_b(a, M=None):
    n = len(a)
    if M is None:
        conv = np.convolve(a, a)
        M = float(np.max(conv))
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, 10.0)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M


def lp_alternating(a, max_time, max_iters=500, t_start=0.7, t_min=0.05):
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        g, g_sum, _ = solve_lp_for_b(a, M)
        if g is None:
            break
        if g_sum <= 1.0 + 1e-6:
            stall_count += 1
            if stall_count > 10:
                break
            continue
        stall_count = 0
        g = g / g_sum
        t = max(t_min, t_start * (1.0 - it / max(float(max_iters), 1)) ** 0.7)
        improved = False
        for tt in [t, t * 0.5, t * 0.25, t * 0.1]:
            a_new = (1.0 - tt) * a + tt * g
            a_new = np.clip(a_new, 0.0, None)
            a_new = normalize(a_new)
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-10:
                best_a = a_new.copy()
                best_obj = obj
                a = a_new
                improved = True
                break
        if not improved:
            a = (1.0 - t) * a + t * g
            a = normalize(a)
            obj = compute_obj(a)
            if obj < best_obj:
                best_a = a.copy()
                best_obj = obj

    return best_a, best_obj


def subgradient_maxconv(a, max_time, lr_init=0.01, max_iters=5000):
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break
        conv = np.convolve(a, a)
        k_star = int(np.argmax(conv))
        M = float(conv[k_star])
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        grad = grad / grad_norm
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0.0, None)
        a_new = normalize(a_new)
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            no_improve = 0
            lr = min(lr * 1.02, 0.1)
        else:
            no_improve += 1
            lr *= 0.92
            if lr < 1e-8:
                break
            if no_improve > 100:
                a = best_a + 0.01 * np.random.randn(n)
                a = normalize(a)
                lr = 0.01
                no_improve = 0

    return best_a, best_obj


def soft_max_gd(a, max_time, T_init=None, max_iters=10000):
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if T_init is None:
        conv = np.convolve(a, a)
        T_init = 0.05 * float(np.max(conv))
    T = T_init
    lr = 0.005
    last_improve = 0

    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break
        conv = np.convolve(a, a)
        max_c = float(np.max(conv))
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -400.0, 400.0)
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        p = exp_vals / sum_exp
        idx = np.arange(n)
        Mmat = p[idx[:, None] + idx[None, :]]
        grad = 2.0 * (Mmat @ a)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0.0, None)
        a_new = normalize(a_new)
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            last_improve = it
            T = max(0.001, T * 0.99)
        else:
            lr *= 0.94
            if lr < 1e-10:
                break
            if it - last_improve > 150:
                T = max(0.001, T * 0.7)
                lr = 0.003
                last_improve = it

    return best_a, best_obj


def pair_redistribute(a, max_time, max_iters=50000):
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break
        conv = np.convolve(a, a)
        k_star = int(np.argmax(conv))
        M = float(conv[k_star])
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        top_idx = np.argsort(contrib)[::-1]
        moved = False
        for _ in range(3):
            if len(top_idx) == 0:
                break
            i = int(top_idx[rng.integers(0, min(5, len(top_idx)))])
            if a[i] < 1e-6:
                continue
            j = k_star - i
            if j == i:
                continue
            delta = a[i] * rng.uniform(0.01, 0.08)
            candidates = list(range(n))
            candidates.remove(i)
            candidates = [c for c in candidates if c != j]
            if not candidates:
                continue
            target = int(rng.choice(candidates, size=min(10, len(candidates)), replace=False))
            for t_pos in target:
                a_try = a.copy()
                a_try[i] -= delta
                a_try[t_pos] += delta
                if a_try[i] < 0:
                    continue
                a_try = np.clip(a_try, 0.0, None)
                a_try = normalize(a_try)
                obj = compute_obj(a_try)
                if obj < best_obj - 1e-12:
                    best_a = a_try.copy()
                    best_obj = obj
                    a = a_try
                    moved = True
                    break
        if not moved and it % 5000 == 0:
            a = best_a + 0.005 * np.random.randn(n)
            a = normalize(a)

    return best_a, best_obj


def block_refine(a, max_time, seed=42):
    n = len(a)
    rng = np.random.default_rng(seed)
    a = normalize(a.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while (time.monotonic() - start) < max_time:
        width = max(1, int(n * rng.uniform(0.005, 0.08)))
        pos = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        sigma = 0.02 + 0.06 * max(0.0, 1.0 - (time.monotonic() - start) / max(max_time, 1))
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.clip(candidate, 0.0, None)
        candidate = normalize(candidate)
        obj = compute_obj(candidate)
        if obj < best_obj - 1e-12:
            best_a = candidate
            best_obj = obj
        it += 1

    return best_a, best_obj


def u_shape_init(n, rng, order=2):
    x = np.linspace(-1, 1, n)
    a = 1.0 + 2.0 * (x ** order)
    a += 0.1 * rng.random(n)
    return normalize(a)


def edge_concentrated_init(n, rng, n_edges=4):
    c = 1.0 / np.sqrt(2.0 * n_edges * n - n_edges * n_edges)
    a = np.zeros(n)
    for i in range(n_edges):
        a[i] = c
        a[n - 1 - i] = c
    if n > 2 * n_edges:
        remaining = 1.0 - 2.0 * n_edges * c
        if remaining < 0:
            remaining = 0.1
        interior = np.ones(n - 2 * n_edges) * remaining / (n - 2 * n_edges)
        noise = rng.normal(0, 0.2, n - 2 * n_edges)
        interior *= np.exp(noise)
        interior = np.maximum(0, interior)
        s = np.sum(interior)
        if s > 1e-10:
            interior = interior * remaining / s
        a[n_edges:n - n_edges] = interior
    return normalize(a)


def cosine_edge_init(n, rng):
    x = np.linspace(0, np.pi, n)
    a = np.cos(x) ** 2 + 0.3
    a += 0.05 * rng.random(n)
    return normalize(a)


def two_blob_init(n, rng):
    center1 = int(n * rng.uniform(0.1, 0.3))
    center2 = n - 1 - center1
    width = n * rng.uniform(0.05, 0.15)
    a = np.zeros(n)
    a += np.exp(-0.5 * ((np.arange(n) - center1) / width) ** 2)
    a += np.exp(-0.5 * ((np.arange(n) - center2) / width) ** 2)
    a += 0.05 * rng.random(n)
    return normalize(a)


def gaussian_center_init(n, rng):
    center = n / 2.0
    width = n * rng.uniform(0.2, 0.5)
    a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
    a += 0.05 * np.ones(n)
    a += 0.02 * rng.random(n)
    return normalize(a)


def random_init(n, rng):

        num_restarts = max(1, min(3, int(time_per_n / 20)))
        time_per_restart = time_per_n / num_restarts

        print(f"\nN={n}: remaining={remaining:.0f}s, {num_restarts} restarts, {time_per_restart:.0f}s each")

        for r in range(num_restarts):
            if time.monotonic() > deadline - 20:
                break
            method = init_methods[(r + n) % len(init_methods)]
            a_init = structured_init(n, rng, method=method)
            a_opt, obj_opt = full_pipeline(
                a_init, time_per_restart,
                use_lp=True, use_gd=True,
                use_sg=(r % 2 == 0),
                use_pair=(r % 3 == 0)
            )

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} r={r} m={method}: obj={obj_opt:.6f} ** NEW BEST **")
            else:
                print(f"  n={n} r={r} m={method}: obj={obj_opt:.6f}")

    # Final refinement on best found
    if best_seq is not None and time.monotonic() < deadline - 8:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: {remaining:.0f}s remaining")

        # Multiple refinement passes
        for pass_name, kwargs_m in [
            ("lp+gd", {"use_lp": True, "use_gd": True, "use_sg": False, "use_pair": False}),
            ("sg+pair", {"use_lp": False, "use_gd": False, "use_sg": True, "use_pair": True}),
            ("block", None),
        ]:
            if time.monotonic() > deadline - 5:
                break
            rem = deadline - time.monotonic() - 3
            if rem < 3:
                break
            alloc = min(rem * 0.4, 40.0)
            if pass_name == "block":
                a_ref, obj_ref = block_refine(best_seq, alloc, seed=int(rng.integers(0, 10000)))
            else:
                a_ref, obj_ref = full_pipeline(best_seq.copy(), alloc, **kwargs_m)
            if obj_ref < best_score:
                best_score = obj_ref
                best_seq = a_ref.copy()
                print(f"  {pass_name}: obj={best_score:.6f} ** NEW BEST **")
            else:
                print(f"  {pass_name}: obj={obj_ref:.6f}")

    # Ensure we have a valid result
    if best_seq is None or not np.all(np.isfinite(best_seq)):
        n_fb = 200
        best_seq = np.ones(n_fb) / n_fb
        best_score = compute_obj(best_seq)

    # Final clipping to ensure validity
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
