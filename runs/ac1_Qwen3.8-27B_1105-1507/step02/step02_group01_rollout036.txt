        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def coord_descent(a, max_passes=50, time_budget=None, start_time=None):
    """Coordinate descent with exact 1D minimization for each coordinate."""
    if start_time is None:

        order = np.random.permutation(n)

                break

            #   j=i: a[i]*a[k-i] (if 0<=k-i<n)
            #   j=k-i: a[k-i]*a[i] (if 0<=k-i<n and k-i != i)
            #   if k=2i: a[i]*a[i] = x^2

            #          = a[k-i] if k=2i (but this is actually the x^2 term)
            # quad[k] = 1 if k=2i and 2i < 2n-1

            # c[k] = (sum over j != i, k-j != i of a[j]*a[k-j]) 
            #        + (if 0<=k-i<n and k-i != i) 2*a[i]*a[k-i]

            #            + a[i]^2 (if k == 2i)

            #              = base[k] + lin[k]*x + quad[k]*x^2

            quad = np.zeros(2*n-1)

            #   if 0 <= k-i < n:
            #     if k == 2*i and 2*i < 2*n-1:
            #       quad[2*i] = 1
            #       lin[k] = 2*a[k-i] - 2*a[i]  # wait, this isn't right
            #     else:
            #       lin[k] = 2*a[k-i]
            
            # Actually, let me think again.
            # c[k] = sum_j a[j]*a[k-j]
            # The terms involving a[i] (= x):
            #   j = i: term = x * a[k-i] (valid if 0 <= k-i < n)
            #   j = k-i: term = a[k-i] * x (valid if 0 <= k-i < n and k-i != i, i.e., k != 2i)

            # So c[k] = base[k] + x*a[k-i] (if 0<=k-i<n) + a[k-i]*x (if 0<=k-i<n and k != 2i) + x^2 (if k==2i)

                #   if k==2i: x*x = x^2

                valid = (np.arange(2*n-1) - i).clip(0, n-1)

                # If k == 2i: subtract x^2

                k_minus_i = k_idx - i
                valid_mask = (k_minus_i >= 0) & (k_minus_i < n)

                contrib = np.zeros(2*n-1)
                not_2i = valid_mask & (k_idx != 2*i)
                contrib[not_2i] = 2 * x * a[k_minus_i[not_2i]]
                if 2*i < 2*n-1:
                    contrib[2*i] = x * x
                c_base = conv_current - contrib
                
                # Add new contribution
                c_new = c_base.copy()
                not_2i_new = valid_mask & (k_idx != 2*i)
                c_new[not_2i_new] += 2 * x_new * a[k_minus_i[not_2i_new]]
                if 2*i < 2*n-1:
                    c_new[2*i] += x_new * x_new
                
                # Compute objective
                sum_new = S_prime + x_new
                if sum_new < 1e-10:
                    continue
                h = 2.0 * n * float(np.max(c_new)) / (sum_new * sum_new)
                if h < best_h:
                    best_h = h
                    best_x = x_new
            
            if best_x != x:
                a[i] = best_x
                S = S_prime + best_x
                new_obj = compute_obj(a)
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj

def coord_descent_fast(a, max_passes=100, time_budget=None, start_time=None, seed=None):

        for i in order:
            if time_budget and (time.monotonic() - start_time) > time_budget:

            # Precompute influence of coordinate i on convolution
            # c[k](x) = base[k] + lin[k]*x + quad[k]*x^2
            # base[k] = conv[k] - (contribution of old x)

            kmi = k_idx - i  # k - i
            valid = (kmi >= 0) & (kmi < n)

            contrib = np.zeros(2*n-1)
            not_2i = valid & (k_idx != 2*i)
            contrib[not_2i] = 2.0 * x * a[kmi[not_2i]]
            if 2*i < 2*n-1:
                contrib[2*i] = x * x
            
            base = conv - contrib
            
            # lin and quad
            lin = np.zeros(2*n-1)
            lin[not_2i] = 2.0 * a[kmi[not_2i]]
            quad = np.zeros(2*n-1)
            if 2*i < 2*n-1:
                quad[2*i] = 1.0
            
            # 1D search for best x_new
            x_lo = 0.0
            x_hi = max(2.0 * x, 0.05 * S / n)
            
            # Grid search
            num_pts = 15
            candidates = np.linspace(x_lo, x_hi, num_pts)
            
            best_x = x
            best_h = np.inf
            
            for x_new in candidates:
                c_new = base + lin * x_new + quad * x_new * x_new
                sum_new = S_prime + x_new
                if sum_new < 1e-10:
                    continue

                if h < best_h:
                    best_h = h
                    best_x = x_new
            
            if best_x != x:
                # Update
                a[i] = best_x
                S = S_prime + best_x
                conv = base + lin * best_x + quad * best_x * best_x

                if new_obj < best_obj:
                    best_obj = new_obj
                    best_a = a.copy()
                    improved_any = True
                else:
                    # Revert
                    a[i] = x
                    S = S_prime + x
                    conv = base + lin * x + quad * x * x
        
        if not improved_any:
            break
    
    return best_a, best_obj

def solve_1d_subproblem(base, lin, quad, S_prime, n, x_current):
    """Solve min_x max_k (base[k] + lin[k]*x + quad[k]*x^2) / (S_prime + x)^2 for x >= 0."""
    # The objective is h(x) = max_k f_k(x) / (S_prime + x)^2
    # where f_k(x) = base[k] + lin[k]*x + quad[k]*x^2
    
    # Critical points: where f_k1(x) = f_k2(x) for some k1, k2
    # AND where the max changes.
    
    # For practical purposes, the max is achieved by a small number of "active" k's.
    # The current active k is argmax f_k(x_current).
    
    # Simple approach: evaluate at x_current, at the point where the active constraint
    # equals the runner-up, and at the boundaries.
    
    # For now, use a dense grid + refinement.
    x_lo = 0.0
    x_hi = max(5.0 * x_current, S_prime / n) if x_current > 0 else S_prime / n
    
    # Dense grid
    grid = np.linspace(x_lo, x_hi, 50)
    
    c_all = base[None, :] + lin[None, :] * grid[:, None] + quad[None, :] * (grid[:, None] ** 2)
    max_c = np.max(c_all, axis=1)
    sums = S_prime + grid
    h = max_c / (sums ** 2)
    
    best_idx = np.argmin(h)
    best_x = grid[best_idx]
    
    # Refine around best_x
    if 0 < best_idx < len(grid) - 1:
        x_ref_lo = grid[best_idx - 1]
        x_ref_hi = grid[best_idx + 1]
        ref_grid = np.linspace(x_ref_lo, x_ref_hi, 30)
        c_ref = base[None, :] + lin[None, :] * ref_grid[:, None] + quad[None, :] * (ref_grid[:, None] ** 2)
        max_c_ref = np.max(c_ref, axis=1)
        sums_ref = S_prime + ref_grid
        h_ref = max_c_ref / (sums_ref ** 2)
        best_ref_idx = np.argmin(h_ref)
        best_x = ref_grid[best_ref_idx]
    
    return best_x

def build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def lp_alternate(a, max_iters=30, time_budget=None, start_time=None):
    n = len(a)
    if start_time is None:
        start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)
    stall = 0

    for it in range(max_iters):
        if time_budget is not None and (time.monotonic() - start_time) > time_budget:
            break
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        A = build_conv_matrix(a)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, M / max(n, 1) * 3.0)] * n
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break
        g = res.x
        g_sum = float(np.sum(g))
        if g_sum <= 1.0 + 1e-4:
            stall += 1
            if stall > 10:
                break
            continue
        stall = 0
        g = g / g_sum
        # Try multiple step sizes
        best_t = 0
        best_obj_t = best_obj
        for t in [0.3, 0.5, 0.7, 0.85, 1.0]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, None)
            s = float(np.sum(a_new))
            if s < 1e-10:
                continue
            a_norm = a_new / s
            obj = compute_obj(a_norm)
            if obj < best_obj_t:
                best_obj_t = obj
                best_t = t
                best_a_candidate = a_norm.copy()

        if best_t > 0:
            a = best_a_candidate
            if best_obj_t < best_obj:
                best_obj = best_obj_t
                best_a = a.copy()

    return best_a, best_obj


def softmax_gd(a, max_time=30.0, start_time=None, T_init=None):
    n = len(a)
    if start_time is None:
        start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(a)

    if T_init is None:
        conv = np.convolve(a, a)
        T_init = 0.05 * float(np.max(conv))

    T = T_init
    lr = 0.05
    last_improve = 0

    for it in range(50000):
        if time.monotonic() - start_time > max_time:
            break
        conv = np.convolve(a, a)
        max_c = float(np.max(conv))
        shifted = np.clip((conv - max_c) / T, -500.0, 500.0)
        exp_vals = np.exp(shifted)
        sum_exp = float(np.sum(exp_vals))
        p = exp_vals / sum_exp

        # Gradient of soft-max w.r.t. a[i]
        idx = np.arange(n)
        M_mat = p[idx[:, None] + idx[None, :]]
        grad = (2.0 / T) * (M_mat @ a)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        a_new = a - lr * grad / grad_norm
        a_new = np.clip(a_new, 0.0, None)
        s = float(np.sum(a_new))
        if s < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)

        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            last_improve = it
            T = max(0.001, T * 0.99)
        else:
            lr *= 0.95
            if lr < 1e-12:
                break
            if it - last_improve > 100:
                T = max(0.001, T * 0.7)
                lr = 0.02
                last_improve = it

    return best_a, best_obj


def simulated_annealing(a, time_budget=60.0, start_time=None, seed=123):
    n = len(a)
    rng = np.random.default_rng(seed)
    if start_time is None:
        start_time = time.monotonic()
    a = a.copy().astype(np.float64)
    S = float(np.sum(a))
    a = a / S
    best_a = a.copy()
    best_obj = compute_obj(a)
    T = 0.05
    T_min = 0.001
    current_obj = best_obj
    iterations = 0

    while time.monotonic() - start_time < time_budget:
        iterations += 1
        # Perturb a random block
        block_size = int(rng.integers(1, max(2, n // 30)))
        pos = int(rng.integers(0, n - block_size + 1))
        a_new = a.copy()
        scale = rng.normal(1.0, 0.3, block_size)
        a_new[pos:pos + block_size] *= np.exp(scale)
        a_new = np.clip(a_new, 0.0, None)
        s = float(np.sum(a_new))
        if s < 1e-10:
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)

        if obj < current_obj or rng.random() < np.exp(-(obj - current_obj) / T):
            a = a_new
            current_obj = obj
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

        # Cooldown
        T = max(T_min, T * 0.9995)

    return best_a, best_obj


def init_edge_profile(n, rng):
    c = 1.0 / np.sqrt(2.0 * n + 1.0)
    a = np.zeros(n)
    a[0] = c
    a[1] = c * 0.8
    a[-1] = c
    a[-2] = c * 0.8
    if n > 4:
        remaining = 1.0 - sum(a)
        if remaining > 0:
            interior = np.ones(n - 4) * (remaining / (n - 4))
            noise = rng.normal(0, 0.2, n - 4)
            interior = np.maximum(0, interior * np.exp(noise))
            s_int = np.sum(interior)
            if s_int > 1e-10:
                interior = interior * remaining / s_int
            a[2:n - 2] = interior
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_u_shape(n, rng):
    t = np.arange(n) / (n - 1)
    center = t - 0.5
    a = 1.0 + 2.0 * center ** 2
    noise = rng.normal(0, 0.1, n)
    a = np.maximum(0, a + noise)
    s = float(np.sum(a))
    return a / s if s > 1e-10 else np.ones(n) / n


def init_cosine(n, rng):
    phase = rng.uniform(0, 2 * np.pi)
    freq = rng.uniform(0.3, 1.5)
    t = np.arange(n) / (n - 1)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase)
    a = np.maximum(0, a)
    a += 0.05 * rng.random(n)
    s = float(np.sum(a))
    return a / s if s > 1e-10 else np.ones(n) / n


def init_spectral(n, rng):
    """Try to initialize with a sequence that has flat-ish magnitude response."""
    # Start with a Hann-like window and add random perturbations
    t = np.arange(n) / (n - 1)
    a = 0.5 * (1 - np.cos(2 * np.pi * t))
    a = np.maximum(0.01, a)
    # Add structured noise
    num_harmonics = int(rng.integers(2, 6))
    for _ in range(num_harmonics):
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.2)
        a += amp * np.cos(2 * np.pi * freq * t + phase)
    a = np.maximum(0, a)
    s = float(np.sum(a))
    return a / s if s > 1e-10 else np.ones(n) / n


def init_random_structured(n, rng):
    choice = int(rng.integers(0, 5))
    if choice == 0:
        return init_edge_profile(n, rng)
    elif choice == 1:
        return init_u_shape(n, rng)
    elif choice == 2:
        return init_cosine(n, rng)
    elif choice == 3:
        return init_spectral(n, rng)
    else:
        a = np.maximum(0, rng.normal(0.5, 0.3, n))
        s = float(np.sum(a))
        return a / s if s > 1e-10 else np.ones(n) / n


def full_refine(a, time_budget, start_time=None, seed=42):
    """Full refinement pipeline: coord descent -> LP -> soft-max GD -> coord descent."""
    if start_time is None:
        start_time = time.monotonic()
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)

    # Phase 1: Coordinate descent (fast)
    cd_time = min(time_budget * 0.3, 60.0)
    a, obj = coord_descent_fast(a, max_passes=200, time_budget=cd_time, start_time=start_time)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    # Phase 2: LP alternating
    if time.monotonic() - start_time < start_time + time_budget * 0.7:
        lp_time = min(time_budget * 0.25, 40.0)
        a, obj = lp_alternate(a, max_iters=20, time_budget=lp_time, start_time=start_time)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    # Phase 3: Soft-max GD
    if time.monotonic() - start_time < start_time + time_budget * 0.85:
        gd_time = min(time_budget * 0.15, 20.0)
        a, obj = softmax_gd(a, max_time=gd_time, start_time=start_time)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    # Phase 4: Final coordinate descent polish
    if time.monotonic() - start_time < start_time + time_budget * 0.95:
        cd_time2 = min(time_budget * 0.1, 30.0)
        a, obj = coord_descent_fast(a, max_passes=100, time_budget=cd_time2, start_time=start_time)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    # Phase 5: Block coordinate descent
    if time.monotonic() - start_time < start_time + time_budget:
        bc_time = min(time_budget * 0.05, 15.0)
        a, obj = block_coord_descent(a, max_passes=30, time_budget=bc_time, start_time=start_time, block_size=2)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj
