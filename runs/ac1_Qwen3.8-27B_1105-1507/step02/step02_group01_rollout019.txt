def project_simplex(v, s=1.0):
    n = len(v)
    if n == 0:
        return v
    b = np.sort(v)[::-1]
    cssv = np.cumsum(b)
    rho = np.argmax((b - cssv/n) * np.arange(1, n+1) > 0)
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Optimized step function coefficient search using power-mean surrogate
gradient descent with Adam, LP alternating, and block refinement.
Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    b = np.sort(v)[::-1]
    cssv = np.cumsum(b)
    rho = np.arange(1, n + 1)
    cond = b * rho > (cssv - s)
    if not np.any(cond):
        theta = (np.sum(v) - s) / n
        return np.maximum(v - theta, 0)
    rho = np.max(rho[cond])
    theta = (cssv[rho - 1] - s) / rho
    return np.maximum(v - theta, 0.0)


def power_mean_value_grad(a, p):
    """Compute power-mean value and gradient.
    h(a) = (1/(2n-1)) * sum_k (conv_k)^p)^(1/p)
    """
    n = len(a)
    conv = np.convolve(a, a)
    u = np.maximum(conv, 1e-30)
    num = 2 * n - 1
    h = (np.sum(u ** p) / num) ** (1.0 / p)
    w = u ** (p - 1.0)
    # grad_j = 2*(p/num)*h^(1-p) * sum_i a[i]*w[j+i]
    corr = np.correlate(w, a, mode='valid')
    grad = 2.0 * (p / num) * (h ** (1.0 - p)) * corr
    return h, grad


def build_A(a):
    """Build the constraint matrix for the LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp(a, M):
    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_alternating(a, max_iters=10, time_budget=30.0, t=0.5):
    """LP alternating method."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        b = solve_lp(a, M)
        if b is None:
            break
        b_sum = np.sum(b)
        if b_sum < 1e-10:
            break
        b = b / b_sum
        # Try different step sizes
        found = False
        for tt in [t, 0.3, 0.1, 0.7, 0.9]:
            a_new = (1.0 - tt) * a + tt * b
            a_new = np.maximum(a_new, 0.0)
            s = np.sum(a_new)
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_a = a_new.copy()
                best_obj = obj
                a = a_new
                found = True
                break
        if not found:
            # Try smaller t
            for tt in [0.05, 0.01, 0.001]:
                a_new = (1.0 - tt) * a + tt * b
                a_new = np.maximum(a_new, 0.0)
                s = np.sum(a_new)
                if s < 1e-10:
                    continue
                a_new = a_new / s
                obj = compute_obj(a_new)
                if obj < best_obj:
                    best_a = a_new.copy()
                    best_obj = obj
                    a = a_new
                    found = True
                    break
        if not found:
            # No improvement, try to continue with smaller t
            a_new = (1.0 - 0.05) * a + 0.05 * b
            a_new = np.maximum(a_new, 0.0)
            s = np.sum(a_new)
            if s > 1e-10:
                a = a_new / s
    
    return best_a


def adam_gd(a, time_budget, p_start=2, p_end=64, N_steps=5000, lr_start=0.01):
    """Gradient descent with Adam on power-mean surrogate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start
    t_adam = 0
    
    step = 0
    while step < N_steps and (time.monotonic() - start) < time_budget:
        # Anneal p
        frac = min(1.0, step / max(1, N_steps * 0.8))
        p = p_start * (p_end / p_start) ** frac
        
        # Compute gradient
        h, grad = power_mean_value_grad(a, p)
        
        # Adam update
        t_adam += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex
        a_new = project_simplex(a_new, 1.0)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            # Slowly decrease lr
            lr *= 0.9999
        else:
            lr *= 0.95
            if lr < 1e-10:
                break
        
        step += 1
    
    return best_a


def block_refine(a, rng, time_budget):
    """Block perturbation refinement."""
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    
    while (time.monotonic() - start) < time_budget:
        width = max(1, int(n * rng.uniform(0.02, 0.15)))
        pos = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        sigma = 0.05 + 0.1 * max(0.0, 1.0 - (time.monotonic() - start) / time_budget)
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.maximum(candidate, 0.0)
        s = np.sum(candidate)
        if s < 1e-10:
            it += 1
            continue
        candidate = candidate / s
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj
        it += 1
    
    return best_a


def coordinate_descent(a, time_budget, steps_per_coord=50):
    """Cycle through coordinates, optimizing each one."""
    n = len(a)
    best_a = a.copy().astype(np.float64)
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    while (time.monotonic() - start) < time_budget:
        improved = False
        for j in range(n):
            if (time.monotonic() - start) < time_budget:
                # Try to optimize a[j] while keeping others fixed
                # The objective is 2n * max_k (a*a)[k] / (sum a)^2
                # With sum a = 1, it's 2n * max_k (a*a)[k]
                # (a*a)[k] is quadratic in a[j]: a[j]^2 * delta + a[j] * linear + const
                # So for each k, it's a convex quadratic in a[j]
                # The max of convex quadratics is convex
                # So we can find the optimal a[j] by ternary search or by
                # finding where the derivative = 0
                
                current = best_a[j]
                # Try a range of values
                best_j_val = current
                best_j_obj = best_obj
                # Golden section or just try a few values
                for factor in [0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.5, 2.0, 0.1, 0.01]:
                    test_val = current * factor
                    if test_val < 0:
                        continue
                    test_a = best_a.copy()
                    test_a[j] = test_val
                    test_s = np.sum(test_a)
                    if test_s < 1e-10:
                        continue
                    obj = compute_obj(test_a)
                    if obj < best_j_obj:

                    if compute_obj(test_a) < best_obj:

"""
Optimized step function coefficient search using power-mean surrogate
gradient descent with Adam, LP alternating, subgradient minimax, and
block refinement. Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_simplex(v, s=1.0):
    n = len(v)
    if n == 0:
        return v
    b = np.sort(v)[::-1]
    cssv = np.cumsum(b)
    rho = np.arange(1, n + 1)
    cond = b * rho > (cssv - s)
    if not np.any(cond):
        theta = (np.sum(v) - s) / n
        return np.maximum(v - theta, 0.0)
    rho = np.max(rho[cond])
    theta = (cssv[rho - 1] - s) / rho
    return np.maximum(v - theta, 0.0)


def power_mean_value_grad(a, p):
    n = len(a)
    conv = np.convolve(a, a)
    u = np.maximum(conv, 1e-30)
    num = 2 * n - 1
    if p == 1.0:
        h = np.sum(u) / num
        w = np.ones_like(u)
    else:
        h = (np.sum(u ** p) / num) ** (1.0 / p)
        w = u ** (p - 1.0)
    corr = np.correlate(w, a, mode='valid')
    grad = 2.0 * (p / num) * (h ** (1.0 - p)) * corr
    return h, grad


def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp(a, M):
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_alternating(a, max_iters=10, time_budget=30.0):
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        b = solve_lp(a, M)
        if b is None:
            break
        b_sum = np.sum(b)
        if b_sum < 1e-10:
            break
        b = b / b_sum
        found = False
        for tt in [0.5, 0.3, 0.1, 0.7, 0.2, 0.05]:
            a_new = (1.0 - tt) * a + tt * b
            a_new = np.maximum(a_new, 0.0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_a = a_new.copy()
                best_obj = obj
                a = a_new
                found = True
                break
        if not found:
            a_new = 0.95 * a + 0.05 * b
            a_new = np.maximum(a_new, 0.0)
            s_new = np.sum(a_new)
            if s_new > 1e-10:
                a = a_new / s_new

    return best_a


def adam_gd(a, time_budget, p_start=2, p_end=128, max_steps=8000, lr_init=0.01):
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = lr_init
    t_adam = 0
    stall = 0
    best_stall_obj = best_obj

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        frac = min(1.0, step / max(1, int(max_steps * 0.7)))
        p = p_start * (p_end / p_start) ** frac

        h, grad = power_mean_value_grad(a, p)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        t_adam += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new, 1.0)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-10:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            stall = 0
            lr = min(lr * 1.001, lr_init)
        else:
            stall += 1
            lr *= 0.9
            if stall > 200:
                lr = lr_init * 0.5
                stall = 0
                if best_obj >= best_stall_obj and lr < lr_init * 0.01:
                    break
            if lr < 1e-12:
                break

    return best_a


def subgradient_minimax(a, time_budget, max_steps=5000):
    a = a.copy().astype(np.float64)
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = 0.1 / np.sqrt(n)

    for step in range(max_steps):
        if time.monotonic() - start > time_budget:
            break
        conv = np.convolve(a, a)
        max_val = np.max(conv)
        # Active set: indices where conv is within 1% of max
        active = np.where(conv >= 0.99 * max_val)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Compute subgradient as average of gradients for active k values
        grad = np.zeros(n)
        for k in active:
            # d(conv[k])/da[j] = 2*a[k-j] if 0 <= k-j < n
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            indices = np.arange(j_start, j_end + 1)
            grad[indices] += 2.0 * a[k - indices]
        grad /= len(active)

        # Normalize
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Step
        a_new = a - lr * grad
        a_new = project_simplex(a_new, 1.0)
        obj = compute_obj(a_new)

        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            lr = min(lr * 1.002, 0.5 / np.sqrt(n))
        else:
            lr *= 0.95
            if lr < 1e-6:
                break

    return best_a


def block_refine(a, rng, time_budget):
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    while (time.monotonic() - start) < time_budget:
        elapsed = time.monotonic() - start
        frac = elapsed / max(1e-10, time_budget)
        width = max(1, int(n * rng.uniform(0.02, 0.12 * (1.0 - 0.5 * frac))))
        pos = int(rng.integers(0, max(1, n - width + 1)))
        candidate = best_a.copy()
        sigma = 0.02 + 0.08 * (1.0 - frac)
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.maximum(candidate, 0.0)
        s = np.sum(candidate)
        if s < 1e-10:
            continue
        candidate = candidate / s
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj

    return best_a


def init_uniform(n, rng):
    return np.ones(n) / n


def init_gaussian(n, rng):
    center = n / 2.0
    width = n * rng.uniform(0.15, 0.45)
    a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
    a += 0.05
    return a / np.sum(a)


def init_edge(n, rng):
    a = np.ones(n) * 0.05
    a[0] = 1.5
    a[1] = 0.8
    a[-1] = 1.5
    a[-2] = 0.8
    if n > 6:
        a[2] += 0.4
        a[-3] += 0.4
    return a / np.sum(a)


def init_periodic(n, rng):
    period = int(rng.integers(max(2, n // 15), max(3, n // 4)))
    pattern = np.maximum(0.05, rng.random(period) - 0.2)
    a = np.tile(pattern, n // period + 1)[:n]
    s = np.sum(a)
    return a / s if s > 1e-10 else np.ones(n) / n


def init_random_structure(n, rng):
    choice = int(rng.integers(0, 4))
    if choice == 0:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.4)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
    elif choice == 1:
        freq = rng.uniform(0.5, 3.0)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 2*np.pi))
    elif choice == 2:
        a = np.exp(-np.abs(np.arange(n) - n/2) / (n * rng.uniform(0.15, 0.5)))
    else:
        a = np.maximum(0, rng.normal(0.5, 0.3, n))
    a = np.maximum(a, 0.01)
    a += 0.03 * rng.random(n)
    return a / np.sum(a)


def init_symmetric(n, rng):
    half = n // 2
    a_half = np.maximum(0.01, rng.random(half + 1))
    a_half += 0.1
    a = np.zeros(n)
    a[:half+1] = a_half
    a[half+1:] = a_half[:n-half-1][::-1]
    return a / np.sum(a)


def full_optimize(a_init, time_budget):
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: LP alternating
    if time_budget > 10:
        lp_time = min(time_budget * 0.2, 25.0)
        a_lp = lp_alternating(a, max_iters=12, time_budget=lp_time)
        obj_lp = compute_obj(a_lp)
        if obj_lp < best_obj:
            best_a = a_lp.copy()
            best_obj = obj_lp

    # Phase 2: Adam GD on power-mean
    if time.monotonic() - start < time_budget * 0.7:
        gd_time = min(time_budget * 0.35, 80.0)
        a_gd = adam_gd(a, gd_time, p_start=2, p_end=128, max_steps=10000, lr_init=0.008)
        obj_gd = compute_obj(a_gd)
        if obj_gd < best_obj:
            best_a = a_gd.copy()
            best_obj = obj_gd

    # Phase 3: Subgradient minimax
    if time.monotonic() - start < time_budget * 0.85:
        sg_time = min(time_budget * 0.15, 30.0)
        a_sg = subgradient_minimax(best_a, sg_time, max_steps=3000)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_a = a_sg.copy()
            best_obj = obj_sg

    # Phase 4: LP polish
    if time.monotonic() - start < time_budget * 0.95:
        lp_time2 = min(time_budget * 0.1, 15.0)
        a_lp2 = lp_alternating(best_a, max_iters=8, time_budget=lp_time2)
        obj_lp2 = compute_obj(a_lp2)
        if obj_lp2 < best_obj:
            best_a = a_lp2.copy()
            best_obj = obj_lp2

    # Phase 5: Fine GD
    if time.monotonic() - start < time_budget * 0.98:
        gd_time2 = min(time_budget * 0.08, 20.0)
        a_gd2 = adam_gd(best_a, gd_time2, p_start=32, p_end=256, max_steps=5000, lr_init=0.003)
        obj_gd2 = compute_obj(a_gd2)
        if obj_gd2 < best_obj:
            best_a = a_gd2.copy()
            best_obj = obj_gd2

    return best_a, best_obj

    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
