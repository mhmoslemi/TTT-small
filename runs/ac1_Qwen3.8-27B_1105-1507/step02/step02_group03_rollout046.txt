"""
Optimized search for flat-autocorrelation non-negative sequences using
LP-based iterative refinement with FFT acceleration, bisection line search,
and multi-scale initialization.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fft_convolve(a, b):
    """Fast convolution using FFT."""
    n = len(a) + len(b) - 1
    n_fft = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, n_fft)
    fb = np.fft.rfft(b, n_fft)
    return np.fft.irfft(fa * fb, n_fft)[:n]

def compute_score(f):
    """Compute 2*n*max(conv) for normalized f (sum=1)."""
    n = len(f)
    conv = fft_convolve(f, f)
    return 2.0 * n * conv.max()

def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s

def build_conv_matrix(f):
    """Build matrix A such that A @ g = fft_convolve(f, g)."""
    n = len(f)
    # A is (2n-1) x n, A[j, i] = f[j-i] if valid
    rows = []
    cols = []
    data = []
    for i in range(n):
        for j in range(i, i + n):  # j-i ranges from 0 to n-1
            rows.append(j)
            cols.append(i)
            data.append(f[j - i])
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A

def solve_lp(f, M, time_limit=2):
    """Solve max sum(g) s.t. A@g <= M, g >= 0. Returns g or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)  # minimize -sum(g)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if not res.success or res.x is None:
        return None
    return np.clip(res.x, 0, None)

def subgradient_iterate(f, n_iter=500, eta0=0.1, rng=None):
    """Subgradient method on the non-convex problem."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    eta = eta0
    
    for it in range(n_iter):
        conv = fft_convolve(f, f)
        M = conv.max()
        # Get all active constraints (within 1% of max)
        active = np.where(conv >= 0.99 * M)[0]
        
        # Subgradient: for each active constraint j, grad[i] = f[j-i] (reversed)
        grad = np.zeros(n)
        for j in active:
            for i in range(n):
                k = j - i
                if 0 <= k < n:
                    grad[i] += f[k]
        
        grad /= max(1, len(active))
        
        # Update
        eta_t = eta0 / (1 + 0.01 * it)
        f_new = f - eta_t * grad
        f_new = np.clip(f_new, 0, None)
        f_new = normalize(f_new)
        
        sc = compute_score(f_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
        else:
            f = f_new  # Still move, but don't update best
    
    return best_f, best_score

def coordinate_descent(f, n_sweeps=10, rng=None):
    """Fast coordinate descent."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n) if rng else np.arange(n)
        for i in order:
            orig = f[i]
            if orig < 1e-12:
                continue
            best_val = orig
            best_local = best_score
            # Try a few factors
            for factor in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                f[i] = orig * factor
                if f[i] < 0:
                    f[i] = 0
                sc = compute_score(f)
                if sc < best_local:
                    best_local = sc
                    best_val = f[i]
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
        f = normalize(f)
    
    return f, best_score

"""
Flat-autocorrelation sequence optimization using LP-based iterative refinement
with FFT acceleration, bisection line search, subgradient pre-conditioning,
and multi-scale exploration. Targets score < 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fft_convolve(a, b):
    """Fast linear convolution via FFT."""
    n = len(a) + len(b) - 1
    n_fft = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, n_fft)
    fb = np.fft.rfft(b, n_fft)
    return np.fft.irfft(fa * fb, n_fft)[:n]


def compute_score(f):
    """Score = 2*n*max(conv) assuming sum(f)=1."""
    n = len(f)
    conv = fft_convolve(f, f)
    return 2.0 * n * conv.max()


def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where A @ g = convolve(f, g)."""
    n = len(f)
    rows = np.arange(2 * n - 1)[:, None]
    cols = np.arange(n)[None, :]
    # A[j, i] = f[j-i] if 0 <= j-i < n else 0
    # For column i, valid j range is [i, i+n-1]
    row_list = []
    col_list = []
    data_list = []
    for i in range(n):
        j_start = i
        j_end = i + n
        row_list.extend(range(j_start, j_end))
        col_list.extend([i] * n)
        data_list.extend(f[0:n])
    A = csr_matrix((data_list, (row_list, col_list)), shape=(2 * n - 1, n))
    return A


def solve_lp_for_g(f, M, time_limit=2.0):
    """Solve max sum(g) s.t. convolve(f,g) <= M, g >= 0. Returns g or None."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-8,
                               'primal_feasibility_tolerance': 1e-8})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    return np.clip(res.x, 0, None)


def lp_iterate(f, time_budget=100.0, rng=None, time_per_lp=2.0):
    """LP-based iterative refinement with bisection line search."""
    if rng is None:

    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_budget:
        it += 1
        conv = fft_convolve(f, f)
        M = conv.max()

        g = solve_lp_for_g(f, M, time_limit=time_per_lp)
        if g is None:
            break

        if g_sum < 1e-10:
            break

def subgradient_iterate(f, n_iter=300, eta0=0.05, rng=None):

        M = conv.max()

        threshold = 0.98 * M
        active = (conv >= threshold).astype(float)

        # Gradient via correlation (vectorized)
        grad = np.correlate(active, f, mode='valid')
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        eta_t = eta0 * n_iter / (it + n_iter)
        f_new = f - eta_t * grad
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s
        sc = compute_score(f_new)

            best_score = sc
            best_f = f_new.copy()
            f = f_new
            stall = 0

                f_pert = np.clip(f_pert, 0, None)

                    f = f_pert

def spectral_precondition(f, n_iter=200, rng=None):

    stall = 0

        mag = np.abs(F)
        phase = np.angle(F)

        # Target: equalize non-DC magnitudes
        target_mag = mag[0] / np.sqrt(len(mag) - 1)  # preserve total energy roughly
        mag_new = np.full_like(mag, target_mag)
        mag_new[0] = 1.0

        # Add phase perturbation periodically

        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        sc = compute_score(f_new)

            best_score = sc
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break

    return best_f, best_score


def coordinate_descent(f, n_sweeps=8, rng=None):
    """Boundary-conditioned coordinate descent."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n) if rng is not None else np.arange(n)
        for i in order:
            orig = f[i]
            if orig < 1e-12:
                continue
            best_val = orig
            best_local = best_score
            for factor in [0.4, 0.6, 0.8, 0.9, 1.1, 1.25, 1.5, 2.0]:
                f[i] = max(0.0, orig * factor)
                sc = compute_score(f)
                if sc < best_local - 1e-12:
                    best_local = sc
                    best_val = f[i]
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
        f = normalize(f)

    return f, best_score

            f3, s3 = coordinate_descent(f2, n_sweeps=5, rng=rng)
        else:
            f3, s3 = f2, s2
        print(f"  Score: {s3:.6f}")
        if s3 < best_score_overall:
            best_score_overall = s3
            best_overall = f3.copy()

    # Strategy 4: Final intensive polishing of best found
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            print(f"\n[Strategy 4: Final polish, remaining={remaining:.0f}s]")
            n_best = len(best_overall)

            # More LP iterations
            if remaining > 40:
                f, s = lp_iterate(best_overall, time_budget=min(remaining * 0.5, 200),
                                  rng=rng, time_per_lp=2.0)
                if s < best_score_overall:
                    best_score_overall = s
                    best_overall = f.copy()
                    print(f"  After LP polish: {s:.6f}")

            # Coordinate descent
            remaining = deadline - time.monotonic()
            if remaining > 15:
                f, s = coordinate_descent(best_overall, n_sweeps=max(5, int(remaining * 0.3)), rng=rng)
                if s < best_score_overall:
                    best_score_overall = s
                    best_overall = f.copy()
                    print(f"  After CD: {s:.6f}")

            # Subgradient with small steps
            remaining = deadline - time.monotonic()
            if remaining > 10:
                n_iter = min(200, int(remaining * 5))
                f, s = subgradient_iterate(best_overall, n_iter=n_iter, eta0=0.01, rng=rng)
                if s < best_score_overall:
                    best_score_overall = s
                    best_overall = f.copy()
                    print(f"  After subgrad: {s:.6f}")

    # Final verification
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000.0)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    print("Using fallback")
    return [1.0] * 512
