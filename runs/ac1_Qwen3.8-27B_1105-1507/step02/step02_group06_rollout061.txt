"""

Uses cyclic and priority-based coordinate descent with golden section search,
combined with LP-based direction finding for polish. Exploits the structure
that each coordinate's contribution to the convolution is piecewise linear/quadratic.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(a):
    """Compute the objective value directly."""
    n = len(a)
    conv = np.convolve(a, a)
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    M = float(np.max(conv))
    return float(2.0 * n * M / (S * S))


def cd_optimize_coordinate(a, i, n, conv, S):
    """Optimize a single coordinate i using golden section search.
    Returns (new_value, new_conv, new_S, new_score)."""
    old_v = a[i]

    # Compute conv without a[i] contribution
    a[i] = 0.0
    conv_0 = np.convolve(a, a)
    S_0 = S - old_v
    a[i] = old_v

    # Determine affected and fixed entries
    k_min = i
    k_max = i + n - 1
    k_affected = np.arange(k_min, k_max + 1)
    k_all = np.arange(2 * n - 1)
    k_fixed = np.setdiff1d(k_all, k_affected)

    if len(k_fixed) > 0:
        max_fixed = float(np.max(conv_0[k_fixed]))
    else:
        max_fixed = 0.0

    # Separate linear and quadratic contributions
    if 2 * i < 2 * n - 1:
        mask_lin = k_affected != 2 * i
    else:
        mask_lin = np.ones(len(k_affected), dtype=bool)

    k_lin = k_affected[mask_lin]
    j_lin = k_affected[mask_lin] - i
    c_lin = conv_0[k_lin].copy()
    a_lin = a[j_lin].copy()
    has_quad = (2 * i < 2 * n - 1)
    c_quad = float(conv_0[2 * i]) if has_quad else 0.0

    # Golden section search range
    max_a = float(np.max(a)) if n > 0 else 1.0
    lo = 0.0
    hi = max(4.0 * S_0 / max(n, 1), 2.0 * max_a + 1e-6, 1.0)

        if new_S < 0.01:
            return float('inf')

            var_max = float(np.max(c_lin + 2.0 * v * a_lin))
        else:
            var_max = 0.0
        M = max(max_fixed, var_max)
        if has_quad:
            M = max(M, c_quad + v * v)
        return 2.0 * n * M / (new_S * new_S)

    # Golden section search
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = f_eval(c_pt)
    fd = f_eval(d_pt)

    for _ in range(35):
        if fc < fd:
            hi = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = hi - gr * (hi - lo)
            fc = f_eval(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = f_eval(d_pt)

    # Check endpoints

    if f0 < f_eval(new_v):

        new_conv[k_fixed] = conv_0[k_fixed]

def build_conv_matrix(f):
    """Build matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_direction(f, M):
    """Solve LP: max sum(g) s.t. g>=0, convolve(f,g)_k <= M."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None

    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    best_a = a.copy()

    S = float(np.sum(a))
    step = 0.1
    it = 0
    no_improve = 0

        lp_time = time.monotonic() - t0

            no_improve += 1
            if no_improve > 20:
                break
            step *= 0.7
            if step < 0.005:
                break
            continue

        sum_g = float(np.sum(g))
        g = g * (S / sum_g)

def make_initializations(n, rng):
    """Create diverse initializations."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(('uniform', np.ones(n)))

    # Cosine taper
    cos_t = (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0
    inits.append(('cosine', cos_t))

    # Linear ramp (increasing - anti-symmetric for min center conv)
    inits.append(('ramp_inc', np.linspace(0.1, 1.0, n)))

    # Linear ramp (decreasing)
    inits.append(('ramp_dec', np.linspace(1.0, 0.1, n)))

    # V-shape (min in center)
    center = (n - 1) / 2.0
    v_shape = np.abs(idx - center) / center
    inits.append(('v_shape', v_shape + 0.1))

    # U-shape (max in center)
    u_shape = 1.0 - np.abs(idx - center) / center
    inits.append(('u_shape', u_shape + 0.1))

    # Bell
    bell = np.exp(-0.5 * ((idx - center) / (n / 3.0)) ** 2)
    inits.append(('bell', bell))

    # Exponential (fast start, slow end)
    inits.append(('exp1', np.exp(-1.5 * idx / n) + 0.1))

    # Exponential (slow start, fast end)
    inits.append(('exp2', np.exp(-1.5 * (n - 1 - idx) / n) + 0.1))

    # Power spectrum of random complex sequence
    z = rng.standard_normal(n) + 1j * rng.standard_normal(n)
    ps = np.abs(np.fft.fft(z, n=2 * n))[:n]
    inits.append(('pwr_spec', ps / np.max(ps) + 0.05))

    # Random
    inits.append(('random', rng.uniform(0.1, 1.0, n)))

    # S-curve
    s_curve = 1.0 / (1.0 + np.exp(-5.0 * (idx / (n - 1) - 0.5)))
    inits.append(('sigmoid', s_curve))

    return inits

    # Final verification
    final_list = [float(x) for x in best_sequence]
    final_score = evaluate_sequence(final_list)
    if not np.isfinite(final_score):
        print("WARNING: Final failed, using fallback")
        best_sequence = np.ones(512)
        final_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, "
          f"internal={best_score:.6f}, time={elapsed:.1f}s")

    return final_list
