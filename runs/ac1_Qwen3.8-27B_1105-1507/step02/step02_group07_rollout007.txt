Focused multi-phase optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Strategy: Start from parent, use LP direction finding with adaptive step sizes,

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    nrows = len(rows)
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(f, M, A=None):
    """Solve LP: max sum(g) s.t. g >= 0, A@g <= M."""
    n = len(f)
    if A is None:
        A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def softmax_grad(f, beta):
    """Compute gradient of soft-max approximation of max(conv(f,f))."""
    n = len(f)
    C = np.convolve(f, f)
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros(n), 0.0
    p = exp_C / total
    # grad[j] = 2 * sum_k p[k] * f[k-j] (with valid indices)
    # This is 2 * correlate(p, f, 'valid')
    grad = 2.0 * np.correlate(p, f, mode='valid')
    soft_val = max_C + np.log(total) / beta
    return grad, soft_val

    v = np.zeros(n)

    t = 0
    no_improve = 0
    
    while time.monotonic() < deadline and t < max_iters:

        beta = beta_start * (beta_end / beta_start) ** frac
        
        grad, _ = softmax_grad(f, beta)
        
        t += 1

        v = b2 * v + (1 - b2) * (grad ** 2)

        # Decay learning rate

def perturb_restart(f, rng, strength=0.05):
    """Multiplicative perturbation."""
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.5, 1.5)
    new_f = np.clip(new_f, 0, 1000)
    S = np.sum(new_f)
    if S < 1e-10:

    return new_f / S

import numpy as np

from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_relaxed(f, M, top_k=20, delta=0.1):
    """Solve LP with relaxed non-peak constraints for faster progress."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    num_constraints = 2 * n - 1
    conv_f = np.convolve(f, f)
    b_ub = np.full(num_constraints, M * (1.0 + delta))
    top_indices = np.argsort(conv_f)[-top_k:]
    b_ub[top_indices] = M
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def softmax_gradient(f, beta):
    """Compute gradient of soft-max approximation of max(conv(f,f))."""
    n = len(f)
    C = np.convolve(f, f)
    max_C = float(C.max())
    shifted = beta * (C - max_C)
    shifted = np.clip(shifted, -700, 700)
    exp_C = np.exp(shifted)
    total = float(exp_C.sum())
    if total < 1e-300:
        return np.zeros(n), 0.0
    p = exp_C / total
    grad = 2.0 * np.correlate(p, f, mode='valid')
    soft_val = max_C + np.log(total) / beta
    return grad, soft_val

        # Cosine decay for learning rate
        lr = lr_start * (1.0 + np.cos(np.pi * frac)) / 2.0
        lr = max(lr, lr_start * 0.005)

        grad, _ = softmax_gradient(f, beta)

        # Gradient clipping
        gnorm = np.linalg.norm(grad)
        if gnorm > 5.0:
            grad = grad / gnorm * 5.0

        t += 1
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad ** 2)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)

        update = m_hat / (np.sqrt(v_hat) + eps)
        new_f = f - lr * update
        new_f = np.clip(new_f, 0.0, 1000.0)
        ns = float(np.sum(new_f))
        if ns < 1e-10:
            continue
        new_f = new_f / ns

        sc = evaluate_sequence(new_f.tolist())
        if np.isfinite(sc) and sc < best_score - 1e-12:
            f = new_f
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            if t % 5000 == 0:
                print(f"    GD it={t}: score={best_score:.6f}, beta={beta:.1f}, lr={lr:.5f}")
        else:
            no_improve += 1
            if no_improve > 3000:
                break

    return best_f, best_score


def perturb_restart(f, rng, strength=0.05):
    """Multiplicative perturbation to escape local minima."""
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.3, 2.0)
    new_f = np.clip(new_f, 0.0, 1000.0)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S


def smooth_perturb(f, rng, sigma=0.3):
    """Gaussian-smoothed perturbation for structured exploration."""
    n = len(f)
    noise = rng.standard_normal(n)
    kernel_size = max(3, n // 20)
    kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / sigma) ** 2)
    kernel = kernel / kernel.sum()
    smooth_noise = np.convolve(noise, kernel, mode='same')
    new_f = f * np.exp(0.05 * smooth_noise / (np.std(smooth_noise) + 1e-10))
    new_f = np.clip(new_f, 0.0, 1000.0)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S


def propose_candidate(budget_s=1500, **kwargs):
    """Main search function: multi-phase optimization starting from parent."""
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 20.0)

    best_f = None
    best_score = float('inf')
    rng = np.random.default_rng(777)

    # Load and evaluate parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        S = float(np.sum(parent))
        if S > 0.01 and np.all(np.isfinite(parent)):
            parent = parent / S
            sc = evaluate_sequence(parent.tolist())
            if np.isfinite(sc):
                best_f = parent.copy()
                best_score = sc
                print(f"Parent n={len(parent)}: score={sc:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    # Fallback if no parent
    if best_f is None:
        n = 512
        t_arr = np.linspace(0.0, 1.0, n)
        r = 0.35
        m = int(r * n / 2)
        best_f = np.ones(n)
        if m > 0:
            for i in range(m):
                w = 0.5 * (1.0 + np.cos(np.pi * (2.0 * i / (2.0 * m) - 1.0)))
                best_f[i] = w
                best_f[n - 1 - i] = w
        best_score = evaluate_sequence(best_f.tolist())
        print(f"Fallback n={n}: score={best_score:.6f}")

    n = len(best_f)
    total_budget = deadline - t0

    # Phase 1: Relaxed LP (30% of budget)
    p1_end = t0 + total_budget * 0.30
    print(f"\n=== Phase 1: Relaxed LP (n={n}, {p1_end-t0:.0f}s) ===")
    try:
        f1, s1 = lp_phase(best_f, p1_end, max_iters=150, use_relaxed=True)
        if s1 < best_score:
            best_f, best_score = f1, s1
            print(f"  After P1 LP: {best_score:.6f}")
    except Exception as e:
        print(f"  P1 error: {e}")

    # Phase 2: Strict LP refinement (15% of budget)
    p2_end = t0 + total_budget * 0.45
    print(f"\n=== Phase 2: Strict LP ({p2_end-p1_end:.0f}s) ===")
    try:
        f2, s2 = lp_phase(best_f, p2_end, max_iters=80, use_relaxed=False)
        if s2 < best_score:
            best_f, best_score = f2, s2
            print(f"  After P2 LP: {best_score:.6f}")
    except Exception as e:
        print(f"  P2 error: {e}")

    # Phase 3: Soft-max gradient with Adam (30% of budget)
    p3_end = t0 + total_budget * 0.75
    print(f"\n=== Phase 3: Gradient ({p3_end-p2_end:.0f}s) ===")
    try:
        f3, s3 = gradient_phase(best_f, p3_end, beta_start=5.0, beta_end=600.0,
                                 lr_start=0.08, max_iters=100000)
        if s3 < best_score:
            best_f, best_score = f3, s3
            print(f"  After P3 GD: {best_score:.6f}")
    except Exception as e:
        print(f"  P3 error: {e}")

    # Phase 4: LP polish (15% of budget)
    p4_end = t0 + total_budget * 0.90
    print(f"\n=== Phase 4: LP polish ({p4_end-p3_end:.0f}s) ===")
    try:
        f4, s4 = lp_phase(best_f, p4_end, max_iters=60, use_relaxed=True)
        if s4 < best_score:
            best_f, best_score = f4, s4
            print(f"  After P4 LP: {best_score:.6f}")
    except Exception as e:
        print(f"  P4 error: {e}")

    # Phase 5: Final gradient polish + perturbation restarts (10% of budget)
    p5_end = deadline
    p5_budget = p5_end - t0 * 0.0  # remaining
    print(f"\n=== Phase 5: Polish + Restarts ({p5_end-p4_end:.0f}s) ===")

    # Quick gradient polish
    grad_deadline = min(p4_end + (p5_end - p4_end) * 0.5, p5_end - 10)
    if time.monotonic() < grad_deadline:
        try:
            f5, s5 = gradient_phase(best_f, grad_deadline, beta_start=200.0,
                                     beta_end=2000.0, lr_start=0.02, max_iters=30000)
            if s5 < best_score:
                best_f, best_score = f5, s5
                print(f"  After P5 GD: {best_score:.6f}")
        except Exception as e:
            print(f"  P5 GD error: {e}")

    # Perturbation restarts
    if time.monotonic() < p5_end - 5:
        restart_budget = p5_end - time.monotonic()
        n_restarts = min(4, max(1, int(restart_budget / 20)))
        per_restart = restart_budget / n_restarts
        print(f"  Trying {n_restarts} restarts...")

        for r in range(n_restarts):
            if time.monotonic() > p5_end - 3:
                break
            strengths = [0.04, 0.08, 0.12, 0.20]
            strength = strengths[r % len(strengths)]

            if r % 2 == 0:
                f_p = perturb_restart(best_f, rng, strength)
            else:
                f_p = smooth_perturb(best_f, rng, sigma=0.3 + 0.2 * r)

            r_deadline = time.monotonic() + per_restart * 0.75
            try:
                # Quick LP on perturbed
                fp1, sp1 = lp_phase(f_p, r_deadline, max_iters=30, use_relaxed=True)
                if sp1 < best_score:
                    best_f, best_score = fp1, sp1
                    print(f"    Restart {r} (mult, str={strength}): {best_score:.6f}")
                else:
                    print(f"    Restart {r} (mult, str={strength}): no improve ({sp1:.6f})")
            except Exception as e:
                print(f"    Restart {r} error: {e}")

    # Final verification
    best_f = np.clip(best_f, 0.0, 1000.0)
    S = float(np.sum(best_f))
    if S < 0.01:
        best_f = np.ones(512) / 512.0

    final_score = evaluate_sequence(best_f.tolist())
    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_f)}, score={final_score:.6f}, time={elapsed:.1f}s")

    if not np.isfinite(final_score):
        # Absolute fallback
        n_fb = 512
        best_f = np.ones(n_fb) / n_fb
        final_score = evaluate_sequence(best_f.tolist())
        print(f"FALLBACK: score={final_score:.6f}")

    return [float(x) for x in best_f]
