def build_conv_matrix_sparse(f):
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp_direction(f, M, A_cache=None):
    n = len(f)
    if A_cache is None:
        A = build_conv_matrix_sparse(f)
    else:
        A = A_cache
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None

def multiplicative_weight_optimize(f, deadline, eta=0.01, max_iters=5000, verbose=""):

    """

    S = float(np.sum(f))

    n = len(f)

        M = float(np.max(conv_f))
        
        # Find "hot" lags (those within a fraction of max)
        threshold = M * 0.95
        hot_lags = np.where(conv_f >= threshold)[0]
        
        if len(hot_lags) == 0:
            break
        
        # Compute pressure vector
        pressure = np.zeros(n)
        for k in hot_lags:
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    pressure[j] += f[kj]
        
        # Normalize pressure
        if pressure.max() > 0:
            pressure = pressure / pressure.max()
        
        # Multiplicative update
        f_new = f * np.exp(-eta * pressure)
        f_new = np.clip(f_new, 1e-10, 1000.0)
        ns = float(np.sum(f_new))
        if ns < 1e-10:
            break
        f_new = f_new / ns
        
        sc = evaluate_sequence(f_new.tolist())
        it += 1
        
        if np.isfinite(sc) and sc < best_score:
            f = f_new
            best_f = f.copy()
            best_score = sc
            if it % 500 == 0:
                print(f"    {verbose}MWU iter {it}: score={best_score:.6f}")
        else:
            # Reduce eta
            eta *= 0.8
            if eta < 1e-6:
                break
            # Try again with smaller step
            f_new = f * np.exp(-eta * pressure)
            f_new = np.clip(f_new, 1e-10, 1000.0)
            ns = float(np.sum(f_new))
            f_new = f_new / ns
            sc = evaluate_sequence(f_new.tolist())
            if np.isfinite(sc) and sc < best_score:

                best_f = f.copy()

def softmax_gradient_optimize(f, deadline, beta_start=5.0, beta_end=500.0,
                               lr_start=0.05, max_iters=5000, verbose=""):
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    f = f / S
    n = len(f)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        it_frac = min(1.0, it / (max_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** it_frac
        
        conv_f = compute_conv(f)
        max_C = conv_f.max()
        exp_C = np.exp(beta * (conv_f - max_C))
        total = exp_C.sum()
        if total < 1e-300:
            break
        p = exp_C / total
        
        # Gradient: 2 * correlate(p, f, 'valid')
        corr = np.correlate(p, f, mode='valid')
        grad = 2.0 * corr
        
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        new_f = f - lr * update
        new_f = np.clip(new_f, 0.0, 1000.0)
        ns = float(np.sum(new_f))
        if ns < 1e-10:
            new_f = f
            ns = 1.0
        new_f = new_f / ns
        
        sc = evaluate_sequence(new_f.tolist())
        if np.isfinite(sc) and sc < best_score:
            f = new_f
            best_f = f.copy()
            best_score = sc
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.5
                no_improve = 0
                if lr < 1e-7:
                    break
    
    return best_f, best_score

def spectral_flatten(f, deadline, max_iters=1000, alpha=0.1, verbose=""):
    """
    Iteratively modify f in frequency domain to flatten the convolution.
    """
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    f = f / S
    n = len(f)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    L = 1  # FFT length (next power of 2 >= 2n-1)
    while L < 2 * n - 1:
        L *= 2
    
    it = 0
    while time.monotonic() < deadline and it < max_iters:
        F = fft(f, n=L)
        
        # We want F^2 to be concentrated at DC
        # Reduce non-DC components of F
        F_mod = F.copy()
        for k in range(1, L):
            # Reduce magnitude of F[k] to make F[k]^2 smaller
            F_mod[k] = F[k] * (1.0 - alpha)
        
        f_new = np.real(ifft(F_mod, n=L)[:n])
        f_new = np.clip(f_new, 0.0, 1000.0)
        ns = float(np.sum(f_new))
        if ns < 1e-10:
            break
        f_new = f_new / ns
        
        sc = evaluate_sequence(f_new.tolist())
        it += 1
        
        if np.isfinite(sc) and sc < best_score:
            f = f_new
            best_f = f.copy()
            best_score = sc
        else:
            alpha *= 0.5
            if alpha < 0.001:
                break
    
    return best_f, best_score

def peak_shave(f, deadline, max_iters=2000, verbose=""):
    """
    Iteratively reduce f values that contribute most to the convolution peak.
    """
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    f = f / S
    n = len(f)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    it = 0
    step = 0.01
    while time.monotonic() < deadline and it < max_iters:
        conv_f = compute_conv(f)
        M = float(np.max(conv_f))
        k_star = int(np.argmax(conv_f))
        
        # Compute pressure: how much each f[j] contributes to conv[k_star]
        pressure = np.zeros(n)
        for j in range(n):
            kj = k_star - j
            if 0 <= kj < n:
                pressure[j] = f[kj]
        
        if pressure.max() == 0:
            break
        
        # Reduce f where pressure is high
        f_new = f - step * pressure
        f_new = np.clip(f_new, 0.0, 1000.0)
        ns = float(np.sum(f_new))
        if ns < 1e-10:
            break
        f_new = f_new / ns

        it += 1
        
        if np.isfinite(sc) and sc < best_score:
            f = f_new
            best_f = f.copy()
            best_score = sc
            if it % 200 == 0:
                print(f"    {verbose}PS iter {it}: score={best_score:.6f}")
        else:
            step *= 0.5
            # Try redistributing: increase low-pressure elements
            f_new = f.copy()
            low_mask = pressure < pressure.mean()
            if low_mask.any():
                f_new[low_mask] += step * 2
                f_new = np.clip(f_new, 0.0, 1000.0)
                ns = float(np.sum(f_new))
                f_new = f_new / ns
                sc = evaluate_sequence(f_new.tolist())
                if np.isfinite(sc) and sc < best_score:
                    f = f_new
                    best_f = f.copy()
                    best_score = sc
            
            if step < 1e-6:
                break
    
    return best_f, best_score

def generate_inits(n, rng):

    # Asymmetric: ramp
    for direction in [1, -1]:
        if direction == 1:
            r = np.linspace(0.1, 1.0, n)
        else:
            r = np.linspace(1.0, 0.1, n)
        inits.append(('ramp' + str(direction), r))
    
    # Asymmetric: exponential
    for tau in [n/4, n/3, n/2]:
        e = np.exp(-np.arange(n) / tau)
        inits.append(('exp_tau' + str(int(tau)), e))
        e2 = np.exp(-np.arange(n-1, -1, -1) / tau)
        inits.append(('exp_rev' + str(int(tau)), e2))
    
    # Asymmetric: S-curve (logistic)
    for center in [0.3, 0.4, 0.6, 0.7]:
        s = 1.0 / (1.0 + np.exp(-5 * (np.arange(n) / n - center)))
        inits.append(('logistic' + str(center), s))
    
    # Asymmetric: two peaks
    for c1, c2 in [(0.25, 0.7), (0.3, 0.65), (0.35, 0.6)]:
        x = np.arange(n) / n
        tp = np.exp(-20*(x-c1)**2) + np.exp(-20*(x-c2)**2)
        inits.append(('wooperand' + str(int(c1*100)) + str(int(c2*100)), tp))
    
    # Random smooth (asymmetric)
    for seed in [42, 123, 7]:
        r_rng = np.random.default_rng(seed + n)
        x = r_rng.standard_normal(n)
        kernel_size = max(3, n // 50)
        kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size+1) / (kernel_size//2+1)) ** 2)
        smooth_x = np.convolve(x, kernel, mode='same')
        smooth_x = np.abs(smooth_x) + 0.05
        inits.append(('rand' + str(seed), smooth_x))
    
    # Comb (alternating)
    comb = np.ones(n)
    comb[::2] = 0.3
    inits.append(('comb', comb))
    
    # Triangle (asymmetric)
    tri = np.zeros(n)
    peak = n // 3
    for j in range(n):
        tri[j] = max(0, 1 - abs(j - peak) / (peak + 1))
    inits.append(('tri_asym', tri))
    
    return inits
