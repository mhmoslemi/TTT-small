What about a_n = cos(π * n * (n+c) / N) + 1? This is non-negative (since cos >= -1, so cos + 1 >= 0). Its autocorrelation might be flatter than uniform.

For α = 0: uniform, J = 2.
For α = 1: a_n = 1 + cos(π * n^2 / N), which ranges from 0 to 2.

The autocorrelation of cos(π * n^2 / N) is related to the chirp signal's autocorrelation, which is known to be relatively flat (this is the basis of chirp radar). So this might give a flatter autocorrelation than uniform!

"""
Minimizes J(a) = 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (Nesterov-accelerated projected gradient) with log-sum-exp smoothing
2. LP best-response for periodic corrections
3. Chirp and harmonic initializations for diverse starting points
4. Beta annealing for progressively sharper max approximation
5. Coordinate descent polish
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def cross_corr(a, w):
    """Compute h_i = sum_j a_j * w_{i+j} for i = 0, ..., n-1.
    where a has length n and w has length 2n-1.
    """
    n = len(a)
    # h_i = sum_j a_j * w_{i+j}
    # = conv(a, w_rev)[2n-2-i] where w_rev = w[::-1]
    # So h = conv(a, w_rev)[n-1:2n-1][::-1]
    w_rev = w[::-1]
    size = 1
    while size < len(a) + len(w_rev) - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w_rev, size)
    tmp = np.fft.irfft(A * W, size)
    # tmp has length size, the convolution has length n + (2n-1) - 1 = 3n-2
    # h[i] = tmp[2n-2-i] for i=0,...,n-1
    # So h = reverse of tmp[n-1:2n-1]
    h = tmp[2*n-2-n+1:2*n-2+1][::-1]
    # Let me double-check: h[0] should be sum_j a_j * w_j
    # conv(a, w_rev)[k] = sum_j a_j * w_rev[k-j] = sum_j a_j * w[2n-2-k+j]
    # h[0] = sum_j a_j * w_j = conv(a, w_rev)[2n-2] 
    # Because conv(a,w_rev)[2n-2] = sum_j a_j * w_rev[2n-2-j] = sum_j a_j * w[j]. ✓
    # h[n-1] = sum_j a_j * w_{n-1+j} = conv(a, w_rev)[2n-2-(n-1)] = conv(a,w_rev)[n-1]
    # Because conv(a,w_rev)[n-1] = sum_j a_j * w_rev[n-1-j] = sum_j a_j * w[2n-2-(n-1-j)] = sum_j a_j * w[n-1+j]. ✓
    # So h[i] = conv(a,w_rev)[2n-2-i], i.e., h = reverse of conv(a,w_rev)[n-1:2n-1]
    h = tmp[n-1:2*n-1][::-1]
    return h


def softmax(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build matrix A where (A @ x) = conv(a, x) (full linear convolution)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None and res.x.sum() > 1e-10:
        return res.x
    return None


def fista_smooth(a, beta, max_iters, deadline, eta_scale=1.0):
    """FISTA with log-sum-exp smoothed objective.
    
    Minimizes f_beta(a) = (1/beta) * log(sum_k exp(beta * (conv(a,a))_k))
    subject to a >= 0, sum(a) = 1.
    """
    n = len(a)
    a = np.clip(a, 0, None).copy()
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    t_k = 1.0
    a_prev = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov momentum
        t_prev = t_k
        t_k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        momentum = (t_prev - 1.0) / t_k
        
        y = a + momentum * (a - a_prev)
        y = np.clip(y, 0, None)
        s_y = y.sum()
        if s_y < 0.01:
            y = a.copy()
        else:
            y = y / s_y
        
        # Compute smooth gradient
        c = fast_conv(y)  # length 2n-1
        bm = beta * c
        w = softmax(bm)  # length 2n-1
        
        # Gradient = 2 * h where h_i = sum_j y_j * w_{i+j}
        h = cross_corr(y, w)
        grad = 2.0 * h
        
        # Step size
        # Lipschitz constant of grad is ~2*beta (conservative)
        eta = eta_scale / (beta + 1.0)
        
        # Gradient step + projection
        a_new = y - eta * grad
        a_new = proj_simplex(a_new)
        
        a_prev = a
        a = a_new
        
        # Track best
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score


def fista_multibeta(a, deadline, betas=None):
    """Run FISTA with annealing beta."""
    if betas is None:
        betas = [5, 20, 50, 100, 200, 500, 1000, 2000]
    
    best = a.copy()
    best_score = compute_score(a)
    
    n = len(a)
    # Estimate time per iteration
    # For n=512: FFT is fast, maybe 0.1ms per iter
    # So we can do ~10000 iters per second roughly
    # Let's just use the deadline
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        # Determine max iterations for this beta
        remaining = deadline - time.monotonic()
        # Rough estimate: ~0.05ms per iter for n<=512
        est_time_per_iter = 0.0001 * (n / 128)  # scaled by n
        max_iters = max(50, int(remaining * 0.7 / max(est_time_per_iter, 0.00005)))
        max_iters = min(max_iters, 50000)
        
        a_result, score = fista_smooth(best, beta, max_iters, deadline)
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"    FISTA beta={beta}: J={score:.8f}", flush=True)
    
    return best, best_score


def lp_iteration(a, max_iters, deadline, base_t=0.5):
    """LP best-response iterations."""
    best_local = a.copy()
    best_score = compute_score(a)
    a = np.clip(a, 0.001, None).copy()
    a = a / a.sum()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()
        
        # Adaptive step size
        t = base_t * max(0.1, 1.0 - 0.5 * it / max_iters)
        t = max(0.02, min(0.8, t))
        
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            a = a_new
        else:
            t2 = t * 0.3
            a_new2 = (1.0 - t2) * a + t2 * g
            a_new2 = np.clip(a_new2, 0, None)
            s2 = a_new2.sum()
            if s2 > 1e-10:
                a_new2 = a_new2 / s2
                score2 = compute_score(a_new2)
                if score2 < best_score:
                    best_score = score2
                    best_local = a_new2.copy()
                a = a_new2
    
    return best_local, best_score
