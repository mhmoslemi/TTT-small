Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Smooth projected gradient descent with log-sum-exp approximation and beta annealing
2. LP-based best response (two variants) for local refinement

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_gradient(a, beta):
    """
    Compute gradient of LSE(beta * conv(a,a)) w.r.t. a.
    Returns (grad, conv_values, peak).
    """
    n = len(a)
    c = fast_conv(a)
    # Softmax for stability
    max_c = c.max()
    shifted = beta * (c - max_c)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    
    # grad[j] = 2*beta * sum_k p[k] * a[k-j] for valid k-j in [0, n-1]
    # = 2*beta * sum_m p[j+m] * a[m] for m in [0, n-1]
    grad = np.zeros(n)
    for j in range(n):
        # p indices: j, j+1, ..., j+n-1 (all valid since max index is (n-1)+(n-1) = 2n-2)
        grad[j] = np.dot(p[j:j+n], a)
    grad *= 2.0 * beta
    return grad, c, max_c


def smooth_gradient_fft(a, beta):
    """

    grad[j] = 2*beta * sum_m p[j+m] * a[m]
    This is the cross-correlation of p and a, which can be done via FFT.

    n = len(a)
    c = fast_conv(a)
    max_c = c.max()
    shifted = beta * (c - max_c)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    
    # We want r[j] = sum_{m=0}^{n-1} p[j+m] * a[m] for j = 0,...,n-1
    # This is the correlation of p and a.
    # r = cross_correlation(p, a) where r[j] = sum_m p[j+m] * a[m]
    # = conv(flip(p), a) at appropriate offset
    # flip(p) has length 2n-1: flip(p)[k] = p[2n-2-k]
    # conv(flip(p), a) has length (2n-1) + n - 1 = 3n-2
    # r[j] = conv(flip(p), a)[j + (2n-2)]... let me verify
    # conv(flip(p), a)[k] = sum_i flip(p)[i] * a[k-i] = sum_i p[2n-2-i] * a[k-i]
    # Let j = 2n-2-k, then: = sum_i p[j+i] * a[k-i]... this is getting confusing.

    for j in range(n):

    grad *= 2.0 * beta
    return grad, c, max_c


def smooth_projected_gradient(a, deadline, verbose=False):
    """
    Run smooth projected gradient with beta annealing.
    a is normalized to sum to 1 before starting.
    Returns (best_a, best_score).
    """
    n = len(a)
    a = np.clip(a, 1e-10, None)
    a = a / a.sum()
    
    best = a.copy()
    best_score = compute_score(a)
    
    betas = [10, 30, 100, 300, 1000, 3000, 10000]
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        # Adaptive number of iterations based on beta
        # Higher beta → sharper → need fewer iterations
        max_iters = max(200, 20000 // beta)
        max_iters = min(max_iters, 5000)
        
        # Initial step size
        c = fast_conv(a)
        alpha = 1.0 / (2.0 * beta * max(c.max(), 1e-10) + 1e-10)
        alpha = min(alpha, 0.5)  # Cap step size
        
        for it in range(max_iters):
            if time.monotonic() > deadline:
                break
            
            grad, c, max_c = smooth_gradient(a, beta)
            
            # Adaptive step size with backtracking
            # Try the step
            a_new = a - alpha * grad
            a_new = project_simplex(a_new)
            
            # Check if score improved (only every few iterations to save time)
            if it % 10 == 0 or it == max_iters - 1:
                score = compute_score(a_new)
                if score < best_score:
                    best_score = score
                    best = a_new.copy()
                    if verbose:
                        print(f"    beta={beta}, iter={it}: J={score:.6f}", flush=True)
            
            a = a_new
            
            # Decay step size slightly
            alpha *= 0.999
            # Or use backtracking
            # if it % 20 == 0:

            #     alpha = 1.0 / (2.0 * beta * max(c_new.max(), 1e-10) + 1e-10)
            #     alpha = min(alpha, 0.5)
        
        # After each beta phase, evaluate

        if score < best_score:

            if verbose:

def build_conv_matrix(a):
    """Build matrix A where (A @ x)[k] = conv(a, x)[k] (full convolution)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response_v1(a, M):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x / res.x.sum()
    return None


def lp_best_response_v2(a):
    """LP: min t s.t. conv(a,g)[k] <= t for all k, sum(g) = 1, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    # Variables: g_0,...,g_{n-1}, t (total n+1)
    # Minimize: t
    # Subject to: A@g - t <= 0 (i.e., A@g <= t*ones)
    #             sum(g) = 1
    #             g >= 0, t free
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality: A@g - t <= 0
    # [A, -1] @ [g; t] <= 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        g = res.x[:n]
        if g.sum() > 1e-10:
            return g / g.sum()
    return None


def lp_iteration_combined(a, max_iters, deadline, use_v1=True, use_v2=True, base_t=0.3):
    """
    Run combined LP iterations using both v1 and v2.
    Returns (best_a, best_score).
    """
    best_local = a.copy()
    best_score = compute_score(a)
    a = a.copy()
    a = np.clip(a, 0.001, None)
    a = a / a.sum()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        candidates = []
        
        if use_v1:
            b = fast_conv(a)
            M = float(b.max())
            g1 = lp_best_response_v1(a, M)
            if g1 is not None:
                candidates.append(g1)
        
        if use_v2:
            g2 = lp_best_response_v2(a)
            if g2 is not None:
                candidates.append(g2)
        
        if not candidates:
            break
        
        # Try each candidate with adaptive step size
        improved = False
        for g in candidates:
            t = base_t
            for _ in range(5):  # Try decreasing step sizes
                a_new = (1.0 - t) * a + t * g
                a_new = np.clip(a_new, 0, None)
                s = a_new.sum()
                if s < 1e-10:
                    break
                a_new = a_new / s
                score = compute_score(a_new)
                if score < best_score:
                    best_score = score
                    best_local = a_new.copy()
                    improved = True
                    break
                t *= 0.5
            if improved:
                a = a_new if 'a_new' in dir() else a
                break
        
        if not improved:
            # Accept the best candidate even without strict improvement (for exploration)
            g = candidates[0]
            t = 0.1
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0, None)
            s = a_new.sum()
            if s > 1e-10:
                a = a_new / s
            else:
                break
    
    return best_local, best_score


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t_arr = np.linspace(0, 1, n)
    
    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        resampled = np.maximum(np.interp(t_arr, t_old, parent), 0)
        if resampled.sum() > 0.01:
            inits.append(resampled / resampled.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Various cosines
    for harmonic in [1, 2, 3, 4, 6]:
        amp = 0.3 / harmonic
        c = 1.0 + amp * np.cos(2 * np.pi * harmonic * t_arr)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())
    
    # Quadratic

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Smooth projected gradient with log-sum-exp and beta annealing (vectorized)
2. LP best-response (two variants: max-sum and min-max)

4. Multi-start perturbation and adaptive time allocation

import numpy as np
from numpy.lib.stride_tricks import sliding_window_view
from scipy.optimize import linprog
import time


def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    if v.sum() <= 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.where(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    # Ensure exact projection
    s = w.sum()
    if s > 0 and abs(s - 1) > 1e-12:
        w = w / s  # should not happen, but safety
    return w


def compute_smooth_gradient(a, beta):
    """
    Compute gradient of 2n * LSE(beta * conv(a,a)) w.r.t. a.
    Returns (gradient, conv, peak_conv).
    """
    n = len(a)
    c = fast_conv(a)
    max_c = c.max()
    shifted = beta * (c - max_c)
    shifted = np.clip(shifted, -100.0, 100.0)
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    
    # grad[j] = 4*n*beta * sum_{m=0}^{n-1} p[j+m] * a[m]
    # = 4*n*beta * (sliding_window_view(p, n)[j] @ a)
    P = sliding_window_view(p, n)  # shape (n, n), P[j,m] = p[j+m]
    grad = 4.0 * n * beta * (P @ a)
    return grad, c, max_c


def smooth_pg(a, deadline, betas=None, max_total_iters=10000):
    """
    Smooth projected gradient with beta annealing.
    Input a is assumed normalized (sum=1) or will be normalized.
    """

    best = a.copy()

    if betas is None:
        betas = [10, 30, 100, 300, 1000, 3000, 10000, 30000]
    
    total_iters = 0
    for beta in betas:
        if time.monotonic() > deadline or total_iters >= max_total_iters:
            break
        
        # Number of iterations for this beta (decreasing with beta)
        iters = min(2000, max(100, 30000 // beta))
        iters = min(iters, max_total_iters - total_iters)
        
        # Initial step size
        c = fast_conv(a)

        alpha = 1.0 / max(L_est, 1e-10)
        alpha = min(alpha, 1.0 / (4.0 * n))  # Cap
        
        for it in range(iters):
            if time.monotonic() > deadline:
                break
            total_iters += 1
            
            grad, c, max_c = compute_smooth_gradient(a, beta)
            
            # Adaptive step size
