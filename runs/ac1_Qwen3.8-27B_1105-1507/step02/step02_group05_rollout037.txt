def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def sliding_dotprod(W, a):
    """
    Compute S[i] = sum_j W[j+i] * a[j] for i = 0,...,n-1.
    W: length 2n-1, a: length n.
    """
    n = len(a)
    L = 1
    while L < 3 * n:
        L <<= 1
    a_pad = np.zeros(L)
    a_pad[:n] = a
    w_rev = np.zeros(L)
    w_rev[:2*n-1] = W[::-1]
    result = np.fft.irfft(np.fft.rfft(a_pad) * np.fft.rfft(w_rev), L)
    return result[n-1:2*n-1][::-1]


def lse_obj_grad(x, beta):
    """
    Log-sum-exp objective and gradient in log-domain.
    a = softmax(x), objective = log(sum(exp(beta * conv(a,a))))
    """
    n = len(x)
    # Softmax
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    Z = exp_x.sum()
    a = exp_x / Z
    
    # Convolution
    b = fast_conv(a)
    
    # Softmax weights
    bb = beta * b
    bb_max = bb.max()
    W = np.exp(bb - bb_max)
    W_sum = W.sum()
    W /= W_sum
    
    # Objective
    obj = bb_max + np.log(W_sum)
    
    # Gradient: (2*a/beta) * (S - E_W)
    E_W = np.dot(W, b)
    S = sliding_dotprod(W, a)
    grad = (2.0 * a / beta) * (S - E_W)
    
    return obj, grad


def softmax_from_x(x):
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    return exp_x / exp_x.sum()


def log_to_softmax(a):
    return np.log(np.maximum(a, 1e-30))


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    while beta < beta_end and time.monotonic() < deadline:

        try:
            res = minimize(make_obj(beta), x, jac=make_grad(beta), method='L-BFGS-B',
                          options={'maxiter': max_iter, 'max_fun': max_iter * 4,
                                   'ftol': 1e-14, 'gtol': 1e-8})
            if res.fun < 1e15:

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Log-sum-exp continuation with L-BFGS-B in log-domain (softmax parametrization)
2. FFT-based O(n log n) gradient computation for the smooth objective
3. Active-set projected gradient for final polish
4. Multiple diverse initializations, restarts, and length exploration
Key innovation: direct smooth optimization via log-sum-exp continuation,
avoiding the LP bottleneck, with clean gradient structure in log-domain.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def sliding_dotprod(W, a):
    """
    Compute S[i] = sum_j W[j+i] * a[j] for i = 0,...,n-1.
    W: length 2n-1, a: length n. Uses FFT.
    """
    n = len(a)
    L = 1
    while L < 3 * n:
        L <<= 1
    a_pad = np.zeros(L)
    a_pad[:n] = a
    w_rev = np.zeros(L)
    w_rev[:2 * n - 1] = W[::-1]
    result = np.fft.irfft(np.fft.rfft(a_pad) * np.fft.rfft(w_rev), L)
    return result[n - 1:2 * n - 1][::-1]


def lse_obj_grad(x, beta):
    """
    Log-sum-exp objective and gradient in log-domain (a = softmax(x)).
    Returns (objective_value, gradient_wrt_x).
    """
    n = len(x)
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    Z = exp_x.sum()
    a = exp_x / Z

    b = fast_conv(a)

    bb = beta * b
    bb_max = bb.max()
    W = np.exp(bb - bb_max)
    W_sum = W.sum()
    W /= W_sum

    obj = bb_max + np.log(W_sum)

    E_W = np.dot(W, b)
    S = sliding_dotprod(W, a)
    grad = (2.0 * a / beta) * (S - E_W)

    return obj, grad


def lse_obj(x, beta):
    """Wrapper for minimize: objective only."""
    return lse_obj_grad(x, beta)[0]


def lse_grad(x, beta):
    """Wrapper for minimize: gradient only."""
    return lse_obj_grad(x, beta)[1]


def softmax_from_x(x):
    """Convert log-domain parameters to probability vector."""
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    return exp_x / exp_x.sum()


def log_to_softmax(a):
    """Convert probability vector to log-domain parameters."""
    return np.log(np.maximum(a, 1e-30))


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def lse_continuation(x, deadline, tracker, beta_start=1.0, beta_end=300.0, beta_mult=1.5):
    """Run log-sum-exp continuation on log-domain parameters."""
    beta = beta_start
    best_a = softmax_from_x(x)
    best_J = compute_J(best_a)
    tracker.update(best_a)
    n = len(x)

    while beta < beta_end and time.monotonic() < deadline:
        max_iter = max(30, min(400, int(40000 / n)))

        try:
            res = minimize(lse_obj, x, args=(beta,), jac=lse_grad, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'max_fun': max_iter * 5,
                                    'ftol': 1e-14, 'gtol': 1e-8})
            if res.fun < 1e15:
                x = res.x
                a = softmax_from_x(x)
                J = compute_J(a)
                if J < best_J:
                    best_J = J
                    best_a = a.copy()
                    tracker.update(a)
        except Exception as ex:
            print(f"    L-BFGS-B error at beta={beta:.1f}: {ex}", flush=True)
            break

        beta *= beta_mult

    return best_a


def active_set_polish(a, deadline, tracker, n_iters=2000):
    """
    Projected gradient descent directly on the max convolution entry.
    Uses the active set (entries near max) for the gradient.
    """
    a = np.maximum(a, 1e-10).copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)

    lr = 1.0 / (2.0 * n)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()

        # Find active set: entries within 1% of max
        active = b > 0.99 * b_max
        if not active.any():
            active = np.zeros(len(b), dtype=bool)
            active[np.argmax(b)] = True

        # Gradient of average over active set
        # d(b[k])/da_i = 2*a_{k-i} if 0<=k-i<n
        # Average gradient = 2 * (mean of shifted a over active k)
        n_active = active.sum()
        # Build gradient efficiently
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            # d(b[k])/da_i = 2*a_{k-i} for valid i
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
        grad /= n_active

        # Gradient step
        a = a - lr * grad

        # Project onto simplex {a >= 0, sum(a) = 1}
        # Using the standard projection algorithm
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a /= s

        # Backtracking if J increased
        J = compute_J(a)
        if J > best_J * 1.01:
            a = best_a.copy()
            lr *= 0.8
        else:
            if J < best_J:
                best_J = J
                best_a = a.copy()
                if it % 100 == 0:
                    tracker.update(a)

    tracker.update(best_a)
    return best_a


def subgradient_avg_polish(a, deadline, tracker, n_iters=3000):
    """
    Smoothed max polish: minimize average of top-k convolution entries.
    More stable than single-active-set.
    """
    a = np.maximum(a, 1e-10).copy()
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_J = compute_J(a)
    tracker.update(a)

    lr = 0.5 / (2.0 * n)
    top_k = max(3, n // 10)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_sorted = np.sort(b)[::-1]
        threshold = b_sorted[min(top_k, len(b_sorted)) - 1] if top_k < len(b_sorted) else b_sorted[-1]
        active = b >= threshold
        n_active = max(1, active.sum())

        # Compute gradient via the sliding dotprod trick
        W = active.astype(np.float64) / n_active
        S = sliding_dotprod(W, a)
        grad = 2.0 * S

        a = a - lr * grad
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a /= s

        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()
            if it % 200 == 0:
                tracker.update(a)
        elif it % 500 == 0:
            lr *= 0.9

    tracker.update(best_a)
    return best_a


def generate_inits(n, rng, parent):
    """Generate diverse initial sequences of length n, all normalized to sum=1."""
    inits = []

    if parent is not None:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent[::-1].copy(), n))
        # Perturbed parent
        p = resample_seq(parent, n)
        inits.append(np.maximum(p + rng.normal(0, 0.1, n), 0.01))

    inits.append(np.ones(n))
    inits.append(rng.dirichlet(np.ones(n)))
    u = rng.uniform(0.1, 1.0, n)
    inits.append(u / u.sum())

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [1, 2, 3, 4, 5, 7, 8]:
        inits.append(1.0 + 0.5 * np.cos(2 * np.pi * freq * t))
    inits.append(1.0 + 0.4 * np.sin(4 * np.pi * t + rng.uniform(0, 6.28)))
    inits.append(1.0 + 0.3 * np.cos(2 * np.pi * t**2 * 5))
    inits.append(np.exp(-5.0 * (t - 0.5)**2) + 0.05)
    inits.append(np.abs(np.sin(np.pi * t)) + 0.05)
    inits.append(np.abs(np.sin(2 * np.pi * t)) * np.exp(-2 * t) + 0.1)

    # Sparse
    sp = np.zeros(n)
    n_spikes = max(3, n // 16)
    sp[rng.choice(n, n_spikes, replace=False)] = 1.0
    inits.append(np.convolve(sp, np.ones(3) / 3.0, mode='same') + 0.05)

    # Periodic
    for period in [2, 3, 4, 5]:
        pat = np.zeros(n)
        pat[::period] = 1.0
        inits.append(pat + 0.1)

    # Triangle
    inits.append(1.0 - np.abs(2.0 * t - 1.0))

    # Random smooth (low-pass filtered noise)
    noise = rng.standard_normal(n)
    kernel = np.exp(-0.5 * (np.arange(-5, 6) / 2.0)**2)
    kernel /= kernel.sum()
    inits.append(np.convolve(noise, kernel, mode='same') + 0.8)

    # Normalize all
    result = []
    for i in inits:
        arr = np.maximum(np.asarray(i, dtype=np.float64), 0.01)
        s = arr.sum()
        if s > 0.01:
            result.append(arr / s)
    return result


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Minimize J(a) = 2n*max(a*a)/(sum a)^2 using log-sum-exp continuation
    with L-BFGS-B in log-domain, active-set polish, and diverse restarts.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Log-sum-exp continuation across lengths ===
    lengths = [128, 256, 512, 1024]
    phase1_deadline = time.monotonic() + (total_deadline - time.monotonic()) * 0.55

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break
        t0 = time.monotonic()
        print(f"\n{'='*40} Phase1 n={n} {'='*40}", flush=True)

        inits = generate_inits(n, rng, parent)

        # Budget per init
        time_remaining = phase1_deadline - time.monotonic()
        time_per_init = max(2.0, time_remaining / max(1, len(inits)))

        best_local = None
        best_local_J = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline - 15:
                break
            x = log_to_softmax(a0)
            init_dl = min(phase1_deadline - 15, time.monotonic() + time_per_init)
            a = lse_continuation(x, init_dl, tracker, beta_start=1.0, beta_end=200.0, beta_mult=1.5)
            J = compute_J(a)
            if J < best_local_J:
                best_local_J = J
                best_local = a.copy()

        if best_local is not None:
            tracker.update(best_local)
        tracker.report(f"Phase1 n={n} ({time.monotonic()-t0:.0f}s)")

    # === Phase 2: Aggressive high-beta continuation from best ===
    if tracker.best is not None and time.monotonic() < total_deadline - 120:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*40} Phase2 (high-beta) n={n} {'='*40}", flush=True)

        x = log_to_softmax(tracker.best)
        beta = 20.0
        phase2_deadline = min(total_deadline - 120, time.monotonic() + 300)

        while beta < 5000 and time.monotonic() < phase2_deadline:
            max_iter = max(50, min(300, int(30000 / n)))
            try:
                res = minimize(lse_obj, x, args=(beta,), jac=lse_grad, method='L-BFGS-B',
                               options={'maxiter': max_iter, 'max_fun': max_iter * 5,
                                        'ftol': 1e-14, 'gtol': 1e-10})
                if res.fun < 1e15:
                    x = res.x
                    a = softmax_from_x(x)
                    J = compute_J(a)
                    if J < tracker.best_score:
                        tracker.update(a)
            except Exception as ex:
                print(f"  Phase2 error beta={beta:.0f}: {ex}", flush=True)
                break
            beta *= 2.0

        tracker.report(f"Phase2 ({time.monotonic()-t0:.0f}s)")

    # === Phase 3: Active-set / subgradient polish ===
    if tracker.best is not None and time.monotonic() < total_deadline - 80:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*40} Phase3 (active-set) n={n} {'='*40}", flush=True)

        a = tracker.best.copy()
        a = subgradient_avg_polish(a, min(total_deadline - 80, time.monotonic() + 200), tracker)
        tracker.report(f"Phase3 ({time.monotonic()-t0:.0f}s)")

    # === Phase 4: Restart with perturbations ===
    if tracker.best is not None and time.monotonic() < total_deadline - 60:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*40} Phase4 (restarts) n={n} {'='*40}", flush=True)

        x_best = log_to_softmax(tracker.best)
        best_x_J = tracker.best_score

        n_restarts = 30
        for r in range(n_restarts):
            if time.monotonic() > total_deadline - 60:
                break

            noise_scale = 0.4 * max(0.3, 1.0 - r / float(n_restarts))
            x_pert = x_best + rng.normal(0, noise_scale, n)

            a = lse_continuation(x_pert, min(total_deadline - 60, time.monotonic() + 20),
                                 tracker, beta_start=10.0, beta_end=300.0, beta_mult=2.0)
            J = compute_J(a)
            if J < best_x_J:
                best_x_J = J
                x_best = log_to_softmax(a)

        tracker.report(f"Phase4 ({time.monotonic()-t0:.0f}s)")

    # === Phase 5: Final polish ===
    if tracker.best is not None and time.monotonic() < total_deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*40} Phase5 (polish) n={n} {'='*40}", flush=True)

        # High beta L-BFGS-B
        x = log_to_softmax(tracker.best)
        for beta in [200, 500, 1000]:
            if time.monotonic() > total_deadline:
                break
            max_iter = max(100, min(500, int(30000 / n)))
            try:
                res = minimize(lse_obj, x, args=(beta,), jac=lse_grad, method='L-BFGS-B',
                               options={'maxiter': max_iter, 'max_fun': max_iter * 5,
                                        'ftol': 1e-15, 'gtol': 1e-12})
                if res.fun < 1e15:
                    x = res.x
                    a = softmax_from_x(x)
                    tracker.update(a)
            except Exception:
                pass

        # Final active-set
        if time.monotonic() < total_deadline:
            a = tracker.best.copy()
            a = active_set_polish(a, total_deadline, tracker, n_iters=1000)

        tracker.report(f"Phase5 ({time.monotonic()-t0:.0f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128) / 128.0
        tracker.best_score = compute_J(tracker.best)
        tracker.update(tracker.best)

    print(f"\n{'='*40} FINAL {'='*40}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} t={time.monotonic()-tracker.t_start:.1f}s", flush=True)

    return [float(v) for v in tracker.best]
