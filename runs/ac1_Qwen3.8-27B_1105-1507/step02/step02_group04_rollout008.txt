"""
Optimization combining projected subgradient, LP alternating, and 
spectral initialization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""
import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_simplex(v, s=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.full(n, s / n)
    rho = idx[-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_subgrad_fast(conv, a, tau):
    n = len(a)
    max_c = np.max(conv)
    shifted = (conv - max_c) / tau
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= ws
    c = np.correlate(w, a, mode='full')
    return 2.0 * c[:n]

def subgradient_optimize(a, max_iter=5000, tau_start=10.0, tau_end=200.0,
                         step_init=0.1, min_step=1e-6):

    a = a.copy().astype(np.float64)
    a = a / np.sum(a)
    best_a = a.copy()

    step = step_init
    stall = 0
    patience = 100
    
    for it in range(max_iter):
        frac = it / max(1, max_iter)
        tau = tau_start + (tau_end - tau_start) * frac
        conv = compute_conv(a)
        cur_max = float(np.max(conv))
        cur_obj = 2.0 * n * cur_max  # since sum=1
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                step *= 0.8
                if step < min_step:
                    break
                a = project_simplex(a + np.random.default_rng(it * 7 + 13).normal(0, 0.02, n))
                continue
        
        grad = smooth_subgrad_fast(conv, a, tau)
        grad_c = grad - grad.mean()
        a_new = a - step * grad_c
        a = project_simplex(a_new, 1.0)
        
        if it > 200:
            step *= 0.999
        
    return best_a, best_obj

def build_A_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A

def lp_optimize_info(a, max_iter=30, t=0.3, time_limit_per_lp=10):
