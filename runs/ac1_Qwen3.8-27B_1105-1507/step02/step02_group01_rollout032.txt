```python
"""
Optimized step function search using subgradient methods, simulated annealing,
LP-based direction finding, and coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))

def normalize(a):
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def topk_sm_grad(a, k, T):
    n = len(a)
    conv = np.convolve(a, a)
    if k >= len(conv):
        k = len(conv)
    sorted_idx = np.argsort(conv)[::-1][:k]
    top_k = conv[sorted_idx]
    max_val = top_k[0]
    shifted = (top_k - max_val) / T
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    
    g = np.zeros(n)
    for i, m in enumerate(sorted_idx):
        wi = p[i]
        if wi < 1e-12:
            continue
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        k_start = m - j_end
        k_end = m - j_start
        g[k_start:k_end+1] += 2.0 * wi * a[j_start:j_end+1][::-1]
    
    sm_val = max_val + T * np.log(exp_vals.sum())
    return sm_val, g

def subgradient_opt(a, time_limit, rng=None):
    a = normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    T = 0.05 * float(np.max(np.convolve(a, a)))
    k = min(20, 2*n-1)
    lr = 0.02
    no_improve = 0
    
    while time.monotonic() - start < time_limit:
        sm_val, g = topk_sm_grad(a, k, T)
        
        # Mirror descent (multiplicative update)
        a_new = a * np.exp(-lr * g)
        a_new = np.clip(a_new, 1e-15, None)
        a_new = a_new / np.sum(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-10:
            best_a = a_new.copy()
            best_obj = obj
            no_improve = 0
            # Slightly increase learning rate
            lr = min(lr * 1.005, 0.1)
            # Decrease temperature
            T = max(T * 0.999, 1e-4)
        else:
            no_improve += 1
            lr *= 0.95
            if no_improve > 100:
                T = max(T * 0.9, 1e-4)
                if T < 1e-6 and lr < 1e-5:
                    break
    
    return best_a, best_obj

def lp_direction(a, M=None):
    n = len(a)
    if M is None:
        M = float(np.max(np.convolve(a, a)))
    
    # Build constraint matrix: conv(a, b)_m <= M for all m
    # conv(a, b)_m = sum_i a[i]*b[m-i] (linear convolution)
    # This is linear in b!
    A = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        for k in range(n):  # k is index in b
            i = m - k  # index in a
            if 0 <= i < n:
                A[m, k] = a[i]
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_optimize(a, time_limit, max_iters=200):
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    t_blend = 0.5
    
    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break
        
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        b = lp_direction(a, M)
        if b is None:
            break
        
        s_b = np.sum(b)
        if s_b <= 1.0 + 1e-6:
            continue
        
        # Try different blend factors
        best_local_a = a
        best_local_obj = best_obj
        for t in [t_blend, t_blend*0.5, t_blend*0.7, t_blend*1.3]:
            a_try = (1-t) * a + t * b / s_b
            a_try = normalize(a_try)
            obj = compute_obj(a_try)
            if obj < best_local_obj:
                best_local_obj = obj
                best_local_a = a_try
        
        if best_local_obj < best_obj:
            best_obj = best_local_obj
            best_a = best_local_a.copy()
            a = best_local_a
        else:
            t_blend *= 0.8
            if t_blend < 0.05:
                break
    
    return best_a, best_obj

def sa_optimize(a, time_limit, rng):
    a = normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    current_obj = best_obj
    start = time.monotonic()
    
    T = 0.001  # temperature
    T_final = 1e-6
    total_iters = int(time_limit * 500)  # rough estimate
    alpha = (T_final / T) ** (1.0 / max(total_iters, 1))
    
    it = 0
    while time.monotonic() - start < time_limit and it < total_iters:
        # Generate a move
        move_type = rng.integers(0, 4)
        a_try = a.copy()
        
        if move_type == 0:
            # Block scale
            block_len = int(rng.integers(1, max(2, n // 5)))
            pos = int(rng.integers(0, n - block_len))
            sigma = 0.05 + 0.1 * T / 0.001
            a_try[pos:pos+block_len] *= np.exp(rng.normal(0, sigma, block_len))
        elif move_type == 1:
            # Single coordinate scale
            k = int(rng.integers(0, n))
            sigma = 0.1 + 0.2 * T / 0.001
            a_try[k] *= np.exp(rng.normal(0, sigma))
        elif move_type == 2:
            # Mass transfer
            block1_len = int(rng.integers(1, max(2, n // 10)))
            block2_len = int(rng.integers(1, max(2, n // 10)))
            pos1 = int(rng.integers(0, n - block1_len))
            pos2 = int(rng.integers(0, n - block2_len))
            # Move some mass from block1 to block2
            mass = np.sum(a_try[pos1:pos1+block1_len]) * rng.uniform(0.05, 0.3)
            a_try[pos1:pos1+block1_len] -= mass / block1_len
            a_try[pos2:pos2+block2_len] += mass / block2_len
        else:
            # Two blocks in opposite direction
            block_len = int(rng.integers(1, max(2, n // 5)))
            pos1 = int(rng.integers(0, n - block_len))
            pos2 = int(rng.integers(0, n - block_len))
            sigma = 0.08
            a_try[pos1:pos1+block_len] *= np.exp(rng.normal(0, sigma, block_len))
            a_try[pos2:pos2+block_len] *= np.exp(rng.normal(0, -sigma, block_len))
        
        a_try = np.clip(a_try, 0.0, None)
        s = np.sum(a_try)
        if s < 1e-10:
            it += 1
            T *= alpha
            continue

def coord_descent(a, time_limit, rng=None):

    n = len(a)

    start = time.monotonic()

                    a = a_norm.copy()

def init_sequence(n, method, rng):

        c = 1.0 / np.sqrt(2.0 * n - 1.0)

        a[1] = c * 1.5
        a[-1] = c * 2
        a[-2] = c * 1.5

    elif method == 1:

        x = np.linspace(0, 1, n)
```

```python
"""
Optimized step function coefficient search using subgradient descent with top-k
soft-max, structured simulated annealing with domain-specific moves, LP-based
direction finding, and coordinate descent. Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a)) / (sum(a))^2"""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def topk_sm_grad(a, k, T):
    """Compute top-k soft-max value and its gradient."""
    n = len(a)
    conv = np.convolve(a, a)
    k = min(k, len(conv))
    sorted_idx = np.argsort(conv)[::-1][:k]
    top_k = conv[sorted_idx]
    max_val = top_k[0]
    shifted = (top_k - max_val) / T
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()

    g = np.zeros(n)
    for i, m in enumerate(sorted_idx):
        wi = p[i]
        if wi < 1e-12:
            continue
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        k_start = m - j_end
        k_end = m - j_start
        g[k_start:k_end + 1] += 2.0 * wi * a[j_start:j_end + 1][::-1]

    sm_val = max_val + T * np.log(max(exp_vals.sum(), 1e-300))
    return sm_val, g


def subgradient_opt(a, time_limit, rng=None):
    """Subgradient descent with top-k soft-max and multiplicative updates."""
    a = normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    conv0 = np.convolve(a, a)
    T = 0.05 * float(np.max(conv0))
    k_top = min(max(5, n // 10), 40)
    lr = 0.03
    no_improve = 0
    stagnation_count = 0

    while time.monotonic() - start < time_limit:
        sm_val, g = topk_sm_grad(a, k_top, T)

        # Multiplicative update (mirror descent)
        a_new = a * np.exp(-lr * g)
        a_new = np.clip(a_new, 1e-15, None)
        s = np.sum(a_new)
        if s < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_a = a_new.copy()
            best_obj = obj
            no_improve = 0
            lr = min(lr * 1.003, 0.15)
            T = max(T * 0.998, 1e-5)
        else:
            no_improve += 1
            lr *= 0.96
            if no_improve > 50:
                stagnation_count += 1
                T = max(T * 0.85, 1e-6)
                lr = max(lr, 0.01)
                no_improve = 0
                if stagnation_count > 15:
                    break

    return best_a, best_obj


def build_conv_matrix_linear(a, n):
    """Build matrix A such that (A @ b)[m] = conv(a, b)[m] for linear convolution."""
    A = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        k_start = m - j_end
        k_end = m - j_start
        A[m, k_start:k_end + 1] = a[j_start:j_end + 1][::-1]
    return A


def lp_direction(a, M):
    """Find direction b maximizing sum(b) subject to conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix_linear(a, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_optimize(a, time_limit, max_iters=300):
    """LP-based alternating optimization."""
    a = normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    t_blend = 0.5
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = np.convolve(a, a)
        M = float(np.max(conv))

        b = lp_direction(a, M)
        if b is None:
            break

        s_b = np.sum(b)
        if s_b <= 1.0 + 1e-5:
            stall += 1
            if stall > 10:
                break
            continue
        stall = 0

        b = b / s_b
        best_local_a = a
        best_local_obj = best_obj

        for t in [t_blend, t_blend * 0.5, t_blend * 0.7, t_blend * 1.2, 0.3, 0.7]:
            a_try = (1.0 - t) * a + t * b
            a_try = normalize(a_try)
            obj = compute_obj(a_try)
            if obj < best_local_obj:
                best_local_obj = obj
                best_local_a = a_try.copy()

        if best_local_obj < best_obj - 1e-10:
            best_obj = best_local_obj
            best_a = best_local_a
            a = best_local_a
        else:
            t_blend *= 0.85
            if t_blend < 0.03:
                break

    return best_a, best_obj


def hot_spot_contributions(a):
    """For each position k, compute its contribution to the max convolution coefficient."""
    n = len(a)
    conv = np.convolve(a, a)
    m_star = int(np.argmax(conv))
    contrib = np.zeros(n)
    for k in range(n):
        j = m_star - k
        if 0 <= j < n:
            contrib[k] = a[k] * a[j]
    return contrib, m_star


def sa_optimize(a, time_limit, rng):

    a = normalize(a.copy())

    current_obj = best_obj
    start = time.monotonic()

    T_min = 1e-7
    it = 0
    no_improve_long = 0
    last_best_it = 0

        if r < 0.35:

            block_len = int(rng.integers(1, max(2, n // 4)))
            pos = int(rng.integers(0, n - block_len + 1))

            a_try = a.copy()
            a_try[pos:pos + block_len] *= np.exp(rng.normal(0, sigma, block_len))
        elif r < 0.55:

            k = int(rng.integers(0, n))
            sigma = 0.05 + 0.2 * (T / 0.002)
            a_try = a.copy()
            a_try[k] *= np.exp(rng.normal(0, sigma))
        elif r < 0.75:
            # Mass transfer
            block_len = int(rng.integers(1, max(2, n // 8)))
            pos1 = int(rng.integers(0, n - block_len + 1))
            pos2 = int(rng.integers(0, n - block_len + 1))
            if pos1 == pos2:
                continue
            a_try = a.copy()
            mass = np.sum(a_try[pos1:pos1 + block_len]) * rng.uniform(0.05, 0.25)
            a_try[pos1:pos1 + block_len] -= mass / block_len
            a_try[pos2:pos2 + block_len] += mass / block_len
        elif r < 0.90:
            # Hot-spot reduction
            a_try = a.copy()
            contrib, m_star = hot_spot_contributions(a)
            # Pick top contributor with some randomness
            top_idx = np.argsort(contrib)[::-1][:max(1, n // 10)]
            k_hot = int(top_idx[int(rng.integers(0, len(top_idx)))])
            k_cold = int(rng.integers(0, n))
            if k_cold == k_hot:
                k_cold = (k_cold + 1) % n
            frac = rng.uniform(0.05, 0.25)
            mass = a_try[k_hot] * frac
            a_try[k_hot] -= mass
            a_try[k_cold] += mass
        else:
            # Two-block opposite perturbation
            block_len = int(rng.integers(1, max(2, n // 6)))

            pos2 = int(rng.integers(0, n - block_len + 1))

            sigma = 0.05
            a_try = a.copy()
            a_try[pos1:pos1 + block_len] *= np.exp(rng.normal(0, sigma, block_len))

        a_try = a_try / s

            current_obj = obj_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

            else:
                no_improve_long += 1
        else:

            a[kick_pos:kick_pos + kick_block] *= np.exp(rng.normal(0, 0.3, kick_block))
            a = normalize(a)
            current_obj = compute_obj(a)
            no_improve_long = 0
            T = max(T, 0.0005)

        it += 1

def coord_descent(a, time_limit):
    """Coordinate descent trying several values per coordinate."""
    a = normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    passes = 0
    while time.monotonic() - start < time_limit:
        improved = False
        for k in range(n):
            if time.monotonic() - start > time_limit:
                break
            orig = a[k]
            if orig < 1e-14:
                continue

            mean_val = np.sum(a) / n
            candidates = [orig * 0.5, orig * 0.7, orig * 1.3, orig * 1.6, mean_val, 0.0]
            for val in candidates:
                a[k] = max(0.0, val)
                s = np.sum(a)
                if s < 1e-10:
                    continue
                a_norm = a / s
                obj = compute_obj(a_norm)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_norm.copy()
                    a = a_norm.copy()
                    improved = True
                    break
            if not improved:
                a[k] = orig

        passes += 1
        if not improved and passes > 1:
            break

    return best_a, best_obj


def fine_sa_polish(a, time_limit, rng):

    a = normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    current_obj = best_obj
    start = time.monotonic()

    T = 0.0001
    T_min = 1e-8

    while time.monotonic() - start < time_limit:
        r = rng.random()
        a_try = a.copy()

        if r < 0.6:
            # Tiny block scale
            block_len = int(rng.integers(1, max(2, n // 10)))
            pos = int(rng.integers(0, n - block_len + 1))
            sigma = 0.01 + 0.03 * (T / 0.0001)
            a_try[pos:pos + block_len] *= np.exp(rng.normal(0, sigma, block_len))
        elif r < 0.85:
            # Single coord tiny
            k = int(rng.integers(0, n))
            sigma = 0.02 + 0.05 * (T / 0.0001)
            a_try[k] *= np.exp(rng.normal(0, sigma))
        else:
            # Hot spot micro-adjust
            contrib, _ = hot_spot_contributions(a)
            top_idx = np.argsort(contrib)[::-1][:max(1, n // 5)]
            k_hot = int(top_idx[0])
            k_cold = int(rng.integers(0, n))
            if k_cold == k_hot:
                k_cold = (k_cold + 1) % n
            mass = a_try[k_hot] * rng.uniform(0.01, 0.05)
            a_try[k_hot] -= mass
            a_try[k_cold] += mass

        a_try = np.clip(a_try, 0.0, None)
        s = np.sum(a_try)
        if s < 1e-10:
            continue
        a_try = a_try / s
        obj_try = compute_obj(a_try)
        delta = obj_try - current_obj

        if delta < 0 or rng.random() < np.exp(-max(delta, 0) / max(T, T_min)):
            a = a_try
            current_obj = obj_try
            if obj_try < best_obj:
                best_obj = obj_try
                best_a = a_try.copy()

        T = max(T * 0.9999, T_min)

    return best_a, best_obj


def init_sequence(n, method, rng):
    """Generate initial sequence of length n using method."""
    if method == 0:
        # Edge-heavy (U-shaped)
        x = np.arange(n) / max(1, n - 1)
        a = 1.0 + 2.0 * (2 * x - 1) ** 2 + 0.2 * rng.random(n)
        return normalize(a)
    elif method == 1:
        # Cosine-centered bump
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2 + 0.1 * rng.random(n)
        return normalize(a)
    elif method == 2:
        # Smooth random (sum of random Gaussians)
        a = np.ones(n) * 0.5
        for _ in range(4):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.1, n * 0.3)
            amp = rng.uniform(-0.5, 0.5)
            a += amp * np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        a = np.maximum(0.01, a)
        return normalize(a)
    elif method == 3:
        # Product structure
        f1 = rng.uniform(1, 4)
        f2 = rng.uniform(2, 8)
        x = np.arange(n)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f1 * x / n) * np.cos(2 * np.pi * f2 * x / n)
        a = np.maximum(0.05, a)
        return normalize(a)
    elif method == 4:
        # Bimodal
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w1 = n * rng.uniform(0.05, 0.15)
        w2 = n * rng.uniform(0.05, 0.15)
        a = np.exp(-0.5 * ((np.arange(n) - c1) / w1) ** 2) + np.exp(-0.5 * ((np.arange(n) - c2) / w2) ** 2)
        a += 0.05
        return normalize(a)
    elif method == 5:
        # Geometric decay from edges
        dec = rng.uniform(0.9, 0.99)
        a_left = dec ** np.arange(n // 2)
        a_right = dec ** np.arange(n - n // 2)
        a = np.concatenate([a_left, a_right])
        a += 0.05 * rng.random(n)
        return normalize(a)
    elif method == 6:
        # Triangle wave
        period = int(rng.integers(max(3, n // 10), max(4, n // 3)))
        x = np.arange(n)
        a = np.abs(((x % period) / period) - 0.5)
        a = 1.0 - a  # inverted triangle (peaks at 1, troughs at 0)
        a = np.maximum(0.05, a)
        return normalize(a)
    else:
        # Random smooth
        a = np.maximum(0.05, rng.normal(0.5, 0.3, n))
        return normalize(a)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization: quick scan -> LP -> subgradient -> SA -> coord descent -> polish.
    Targets objective below 1.5027.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf

    # Try parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        parent = None

    # --- Phase 0: Quick scan (150s) ---
    print("\n=== Phase 0: Quick scan ===")
    scan_deadline = start_time + 150.0
    n_values = [40, 60, 80, 120, 200, 250, 300, 400, 500]
    scan_results = []

    for n in n_values:
        if time.monotonic() > scan_deadline:
            break
        for method in range(4):
            if time.monotonic() > scan_deadline:
                break
            a_init = init_sequence(n, method, rng)
            a_opt, obj_opt = subgradient_opt(a_init, 8.0, rng)
            scan_results.append((obj_opt, n, a_opt.copy()))
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n}, m={method}: obj={obj_opt:.6f} ** NEW BEST **")
            else:
                print(f"  n={n}, m={method}: obj={obj_opt:.6f}")

    # Optimize parent if available
    if parent is not None and parent.size <= 600 and time.monotonic() < scan_deadline:
        p_opt, p_obj = subgradient_opt(parent.copy(), 30.0, rng)
        if p_obj < best_score:
            best_score = p_obj
            best_seq = p_opt.copy()
        print(f"  Parent subgrad: obj={p_obj:.6f}")

    # Select top candidates (diverse n)
    scan_results.sort(key=lambda x: x[0])
    top_candidates = []
    seen_n = set()
    for obj, n, a in scan_results:
        if n not in seen_n:
            top_candidates.append((obj, n, a))
            seen_n.add(n)
        if len(top_candidates) >= 3:
            break

    # Include parent as candidate if it's good
    if parent is not None and parent.size <= 600:
        p_obj = compute_obj(parent)
        top_candidates.append((p_obj, parent.size, parent.copy()))

    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:3]

    # --- Phase 1: LP optimization (200s) ---
    print("\n=== Phase 1: LP optimization ===")
    lp_deadline = start_time + 350.0
    for i, (obj_init, n, a_init) in enumerate(top_candidates):
        if time.monotonic() > lp_deadline:
            break
        lp_time = min(80.0, max(20.0, (lp_deadline - time.monotonic()) / max(1, len(top_candidates) - i)))
        a_lp, obj_lp = lp_optimize(a_init, lp_time)
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  Cand {i} (n={n}): obj={obj_lp:.6f} ** NEW BEST **")
        else:
            print(f"  Cand {i} (n={n}): obj={obj_lp:.6f}")

    # --- Phase 2: Subgradient refinement (200s) ---
    print("\n=== Phase 2: Subgradient refinement ===")
    sg_deadline = start_time + 550.0
    if best_seq is not None and time.monotonic() < sg_deadline:
        sg_time = min(200.0, sg_deadline - time.monotonic())
        a_sg, obj_sg = subgradient_opt(best_seq.copy(), sg_time, rng)
        if obj_sg < best_score:
            best_score = obj_sg
            best_seq = a_sg.copy()
            print(f"  After subgradient: obj={best_score:.6f}")
        else:
            print(f"  After subgradient: obj={obj_sg:.6f}")

    # --- Phase 3: LP again on current best (100s) ---
    print("\n=== Phase 3: Second LP round ===")
    lp2_deadline = start_time + 650.0
    if best_seq is not None and time.monotonic() < lp2_deadline:
        lp2_time = min(100.0, lp2_deadline - time.monotonic())
        a_lp2, obj_lp2 = lp_optimize(best_seq.copy(), lp2_time)
        if obj_lp2 < best_score:
            best_score = obj_lp2
            best_seq = a_lp2.copy()
            print(f"  After 2nd LP: obj={best_score:.6f}")
        else:
            print(f"  After 2nd LP: obj={obj_lp2:.6f}")

    # --- Phase 4: Simulated annealing (400s) ---
    print("\n=== Phase 4: Simulated annealing ===")
    sa_deadline = start_time + 1050.0
    if best_seq is not None and time.monotonic() < sa_deadline:
        sa_time = min(400.0, sa_deadline - time.monotonic())
        a_sa, obj_sa = sa_optimize(best_seq.copy(), sa_time, rng)
        if obj_sa < best_score:
            best_score = obj_sa
            best_seq = a_sa.copy()
            print(f"  After SA: obj={best_score:.6f}")
        else:
            print(f"  After SA: obj={obj_sa:.6f}")

    # --- Phase 5: Coordinate descent (100s) ---
    print("\n=== Phase 5: Coordinate descent ===")
    cd_deadline = start_time + 1150.0
    if best_seq is not None and time.monotonic() < cd_deadline:
        cd_time = min(100.0, cd_deadline - time.monotonic())
        a_cd, obj_cd = coord_descent(best_seq.copy(), cd_time)
        if obj_cd < best_score:
            best_score = obj_cd
            best_seq = a_cd.copy()
            print(f"  After coord descent: obj={best_score:.6f}")
        else:
            print(f"  After coord descent: obj={obj_cd:.6f}")

    # --- Phase 6: Final polish with fine SA + subgradient (remaining time) ---
    print("\n=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 5
        half = remaining / 2.0

        # Fine SA
        a_pol, obj_pol = fine_sa_polish(best_seq.copy(), half, rng)
        if obj_pol < best_score:
            best_score = obj_pol
            best_seq = a_pol.copy()
            print(f"  After fine SA: obj={best_score:.6f}")

        # Final subgradient
        if time.monotonic() < deadline - 5:
            rem2 = deadline - time.monotonic() - 5
            a_fin, obj_fin = subgradient_opt(best_seq.copy(), rem2, rng)
            if obj_fin < best_score:
                best_score = obj_fin
                best_seq = a_fin.copy()
                print(f"  After final subgrad: obj={best_score:.6f}")

    # --- Final: try a few more random restarts if time remains ---
    if time.monotonic() < deadline - 30:
        print("\n=== Extra restarts ===")
        for _ in range(5):
            if time.monotonic() > deadline - 20:
                break
            n_try = int(rng.integers(50, 400))
            m_try = int(rng.integers(0, 8))
            a_init = init_sequence(n_try, m_try, rng)
            a_opt, obj_opt = subgradient_opt(a_init, 15.0, rng)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  Restart n={n_try},m={m_try}: obj={obj_opt:.6f} ** NEW BEST **")

    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
```
