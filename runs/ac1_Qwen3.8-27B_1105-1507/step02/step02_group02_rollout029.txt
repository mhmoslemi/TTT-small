"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Multiplicative Weights / Mirror Descent (fast O(n log n) per iter)
2. LP-based best-response polishing
3. Smooth log-sum-exp refinement
4. Multi-start with diverse initializations
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def hankel_mv(q, a):
    """Compute Q @ a where Q[j,l] = q[j+l], using FFT. O(n log n)."""
    n = len(a)
    # Qa[j] = sum_l q[j+l] * a[l] = (reverse(a) * q)[j] for j in [0, n-1]
    # reverse(a) has length n, q has length 2n-1
    # Convolution of reverse(a) and q has length n + (2n-1) - 1 = 3n - 2
    # We want positions [0, n-1] of this convolution
    ra = a[::-1]
    s = 1
    while s < len(ra) + len(q) - 1:
        s <<= 1
    result = np.fft.irfft(np.fft.rfft(ra, s) * np.fft.rfft(q, s), s)
    return result[:n]


def project_simplex(v, target_sum=1.0):
    """Project v onto {x : sum(x) = target_sum, x >= 0}."""
    n = len(v)
    # Standard algorithm
    if v.sum() <= 0:
        return np.full(n, target_sum / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    # Ensure exact sum
    diff = target_sum - w.sum()
    if abs(diff) > 1e-12:
        w += diff / n
        w = np.maximum(w, 0)
    return w


def mirror_descent_step(a, q, eta):
    """One step of projected gradient: a <- Proj(a - eta * 2*Qa)."""
    grad = 2.0 * hankel_mv(q, a)
    a_new = a - eta * grad
    return project_simplex(a_new, 1.0)


def multiplicative_weights_update(q, conv_result, eta_q):
    """Update q using multiplicative weights."""
    # q_k proportional to exp(-eta_q * conv[k])
    logits = -eta_q * conv_result
    logits -= logits.max()  # numerical stability
    q_new = np.exp(logits)
    q_new /= q_new.sum()
    return q_new


def run_mirror_descent(a0, deadline, max_iters=50000, eta_decay=0.9999, 
                       eta0=None, eta_q0=1.0, eta_q_decay=0.9999,
                       q_init=None, print_interval=5000):
    """Run multiplicative weights / mirror descent iterations."""
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    if q_init is None:
        q = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        q = q_init.copy()
        q = np.maximum(q, 1e-10)
        q = q / q.sum()
    
    if eta0 is None:
        eta0 = 0.1
    
    eta = eta0
    eta_q = eta_q0
    
    best_a = a.copy()
    best_score = float('inf')
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution
        conv = fast_conv(a)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Update a using projected gradient
        a = mirror_descent_step(a, q, eta)
        
        # Update q using multiplicative weights
        conv_new = fast_conv(a)
        q = multiplicative_weights_update(q, conv_new, eta_q)
        
        # Decay step sizes
        eta *= eta_decay
        eta_q *= eta_q_decay
        
        if it % print_interval == 0:
            print(f"    MD iter {it}: J={score:.6f}, eta={eta:.6f}, eta_q={eta_q:.6f}", flush=True)
    
    return best_a, best_score


def build_conv_matrix(a):
    """Build matrix A where (A @ x) = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-12,
                           'primal_feasibility_tolerance': 1e-12})
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))

    best_local = a.copy()
    best_score = compute_score(a)

    a = a / a.sum()

            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()

        t = base_t * (1.0 - 0.7 * it / max_iters)
        t = max(0.02, min(0.8, t))

        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-10:

        a_new = a_new / s

            a = a_new

            a_new2 = (1.0 - t2) * a + t2 * g

def smooth_objective(x, beta, n):
    """Log-sum-exp smoothed objective."""
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 0.01:
        return 1e10
    x_norm = x_pos / s
    b = fast_conv(x_norm)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    log_sum = b_max * beta + np.log(np.exp(shifted).sum())
    return float(2.0 * n * log_sum)


def smooth_refine(a, deadline, betas=None):
    """Refine using smooth log-sum-exp objective with L-BFGS-B."""
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)

    if betas is None:
        betas = [50, 200, 1000, 5000, 20000]
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        x0 = best.copy()
        x0 = np.clip(x0, 0.001, None)
        x0 = x0 / x0.sum()
        try:
            res = minimize(smooth_objective, x0, args=(beta, n),
                           method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': 300, 'maxfun': 5000,
                                    'ftol': 1e-18, 'gtol': 1e-16})
            if res.fun < 1e10:
                res_x = np.clip(res.x, 0, None)
                if res_x.sum() > 0.01:
                    score = compute_score(res_x)
                    if score < best_score:
                        best = res_x.copy()
                        best_score = score
        except Exception:
            pass

    return best, best_score


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t_arr = np.linspace(0, 1, n)

    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        resampled = np.maximum(np.interp(t_arr, t_old, parent), 0)
        if resampled.sum() > 0.01:
            inits.append(resampled / resampled.sum())

    inits.append(np.ones(n) / n)
    
    for freq in [1, 2, 3, 4, 5, 7, 10]:
        amp = 0.2 / np.sqrt(freq)
        c = 1.0 + amp * np.cos(2 * np.pi * freq * t_arr)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    # Sinc-like
    c = np.sinc(t_arr - 0.5) ** 2
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())

    # Gaussian
    c = np.exp(-4.0 * (t_arr - 0.5) ** 2)
    inits.append(c / c.sum())

    # Random
    inits.append(rng.uniform(0.3, 1.0, n) / n)

    # Triangular
    c = 1.0 - np.abs(t_arr - 0.5)
    c = np.maximum(c, 0.05)
    inits.append(c / c.sum())

    return inits

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Multiplicative Weights / Mirror Descent (O(n log n) per iteration via FFT)
2. LP-based best-response polishing (exact Nash equilibrium approximation)
3. Smooth log-sum-exp refinement with L-BFGS-B
4. Multi-start with diverse initializations and perturbation search
Key innovation: Hankel matrix-vector product computed via FFT for fast
projected gradient steps in the mirror descent phase.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def hankel_mv(q, a):
    """Compute Q @ a where Q[j,l] = q[j+l], using FFT. O(n log n).
    
    Verified: (Qa)[j] = sum_l a[l]*q[j+l] = convolve(a_rev, q)[n-1+j]
    """
    n = len(a)
    a_rev = a[::-1]
    s = 1
    while s < n + len(q) - 1:
        s <<= 1
    result = np.fft.irfft(np.fft.rfft(a_rev, s) * np.fft.rfft(q, s), s)
    return result[n - 1:2 * n - 1]


def project_simplex(v, target_sum=1.0):
    """Project v onto {x : sum(x) = target_sum, x >= 0} in O(n log n)."""
    n = len(v)
    if v.sum() <= 0:
        return np.full(n, target_sum / n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - target_sum)
    if not cond.any():
        w = np.full(n, target_sum / n)
        return w
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    diff = target_sum - w.sum()
    if abs(diff) > 1e-12:
        w += diff / n
        w = np.maximum(w, 0)
    return w


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def run_mirror_descent(a0, deadline, max_iters=100000, eta0=0.05, eta_q0=0.5,
                       eta_decay=0.9999, eta_q_decay=0.9999, q_init=None,
                       q_interpolate=0.0, print_interval=20000):
    """Run multiplicative weights / mirror descent for the convolution game.
    
    The game: row player picks position k, column player picks sequence a.
    Payoff: (a*a)[k]. We seek the Nash equilibrium value.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    if q_init is not None:
        q = np.maximum(q_init, 1e-10).astype(np.float64)
        q = q / q.sum()
    else:
        q = np.ones(2 * n - 1) / (2 * n - 1)

    eta = eta0
    eta_q = eta_q0
    best_a = a.copy()
    best_score = float('inf')
    no_improve_count = 0
    patience = 50000

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Compute current convolution and score
        conv = fast_conv(a)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count > patience:
            break

        # Projected gradient step: a <- Proj(a - eta * 2*Qa)
        grad = 2.0 * hankel_mv(q, a)
        a_new = a - eta * grad
        a = project_simplex(a_new, 1.0)

        # Update q via multiplicative weights (with optional interpolation)
        conv_new = fast_conv(a)
        logits = -eta_q * conv_new
        logits -= logits.max()
        q_new = np.exp(logits)
        q_new /= q_new.sum()

        if q_interpolate > 0:
            q = (1.0 - q_interpolate) * q + q_interpolate * q_new
            q = np.maximum(q, 1e-15)
            q /= q.sum()
        else:
            q = q_new

        # Decay step sizes
        eta *= eta_decay
        eta_q *= eta_q_decay

        if it % print_interval == 0:
            elapsed = time.monotonic()
            print(f"    MD iter {it}: J={score:.7f}, eta={eta:.6f}, "
                  f"eta_q={eta_q:.4f}, q_max={q.max():.4f}", flush=True)

    return best_a, best_score


def build_conv_matrix(a):
    """Build matrix A where (A @ x) = full convolve(a, x). Shape (2n-1, n)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True,
                           'dual_feasibility_tolerance': 1e-12,
                           'primal_feasibility_tolerance': 1e-12})
    if res.success and res.x is not None and res.x.sum() > 1e-10:
        return res.x
    return None


def lp_iteration(a, max_iters, deadline, base_t=0.4):
    """Run LP best-response iterations with adaptive step size."""
    best_local = a.copy()
    best_score = compute_score(a)
    a = np.clip(a, 0.001, None).copy()
    a = a / a.sum()

    stall_count = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0)
        if g.sum() < 1e-10:
            break
        g = g / g.sum()

        # Adaptive step size
        t = base_t * max(0.1, 1.0 - 0.8 * it / max_iters)
        t = min(t, 0.8)

        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        score = compute_score(a_new)
        if score < best_score - 1e-8:
            best_score = score
            best_local = a_new.copy()
            a = a_new
            stall_count = 0
        else:
            stall_count += 1
            # Try smaller step
            t2 = t * 0.25
            a_new2 = (1.0 - t2) * a + t2 * g
            a_new2 = np.clip(a_new2, 0.0, None)
            s2 = a_new2.sum()
            if s2 > 1e-10:
                a_new2 = a_new2 / s2
                score2 = compute_score(a_new2)
                if score2 < best_score:
                    best_score = score2
                    best_local = a_new2.copy()
                    stall_count = 0
                a = a_new2
            if stall_count > 50:
                break

    return best_local, best_score


def smooth_objective(x, beta, n):
    """Log-sum-exp smoothed approximation of 2n*max(conv(a,a)) for sum(a)=1."""
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 0.01:
        return 1e10
    x_norm = x_pos / s
    b = fast_conv(x_norm)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    log_sum = b_max * beta + np.log(np.exp(shifted).sum())
    return float(2.0 * n * log_sum)


def smooth_refine(a, deadline, betas=None):
    """Refine using smooth log-sum-exp with L-BFGS-B and beta annealing."""
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)

    if betas is None:
        betas = [50, 200, 1000, 5000, 20000]

    for beta in betas:
        if time.monotonic() > deadline:
            break
        x0 = np.clip(best, 0.001, None)
        x0 = x0 / x0.sum()
        try:
            res = minimize(smooth_objective, x0, args=(beta, n),
                           method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': 400, 'maxfun': 8000,
                                    'ftol': 1e-18, 'gtol': 1e-14})
            if res.fun < 1e10:
                res_x = np.clip(res.x, 0, None)
                if res_x.sum() > 0.01:
                    score = compute_score(res_x)
                    if score < best_score:
                        best = res_x.copy()
                        best_score = score
        except Exception:
            pass

    return best, best_score


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    t_arr = np.linspace(0, 1, n)

    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        resampled = np.maximum(np.interp(t_arr, t_old, parent), 0)
        if resampled.sum() > 0.01:
            inits.append(resampled / resampled.sum())

    inits.append(np.ones(n) / n)

    # Various cosine harmonics
    for freq in [1, 2, 3, 4, 5, 7, 11]:
        amp = 0.15 / np.sqrt(freq)
        c = 1.0 + amp * np.cos(2 * np.pi * freq * t_arr)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    # Raised cosine
    c = 0.5 + 0.5 * np.cos(2 * np.pi * t_arr)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())

    # Gaussian
    c = np.exp(-3.0 * (t_arr - 0.5) ** 2)
    inits.append(c / c.sum())

    # Bimodal
    c = np.exp(-8.0 * (t_arr - 0.25) ** 2) + np.exp(-8.0 * (t_arr - 0.75) ** 2)
    inits.append(c / c.sum())

    # Random
    inits.append(rng.uniform(0.3, 1.0, n) / n)

    # Triangular
    c = 1.0 - np.abs(t_arr - 0.5)
    c = np.maximum(c, 0.05)
    inits.append(c / c.sum())

    # Sinc-like
    c = np.sinc(t_arr - 0.5) ** 2
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())

    return inits

    # === PHASE 3: Smooth Refinement ===
    smooth_frac = 0.15
    smooth_deadline = min(time.monotonic() + total_budget * smooth_frac, deadline)

    if best is not None and time.monotonic() < smooth_deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Smooth Refinement", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()

        a_result, score = smooth_refine(best, smooth_deadline,
                                        betas=[20, 100, 500, 2000, 10000, 50000])
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  NEW BEST (smooth): J={score:.7f}", flush=True)

        print(f"  Phase 3 done in {time.monotonic()-t0:.1f}s, best={best_score:.7f}", flush=True)

    # === PHASE 4: Perturbation + LP ===
    if best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Perturbation Search", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = best.size
        remaining = deadline - time.monotonic()
        n_tries = max(5, min(40, int(remaining / 4)))

        noise_scales = [0.02, 0.04, 0.01, 0.06, 0.015, 0.08, 0.03, 0.01]

        for trial in range(n_tries):
            if time.monotonic() > deadline:
                break
            scale = noise_scales[trial % len(noise_scales)]
            noise = rng.normal(0, scale, n)
            a_pert = best + noise
            a_pert = np.clip(a_pert, 0, None)
            if a_pert.sum() < 0.01:
                continue
            a_pert = a_pert / a_pert.sum()

            # Short LP refinement
            remaining = deadline - time.monotonic()
            est_time = 0.025 if n <= 256 else 0.04 if n <= 512 else 0.07
            max_iters = max(30, int(remaining / (n_tries - trial) / est_time * 0.5))
            max_iters = min(max_iters, 1500)

            a_result, score = lp_iteration(a_pert, max_iters, deadline)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  NEW BEST (pert {trial}): J={score:.7f}", flush=True)

        print(f"  Phase 4 done in {time.monotonic()-t0:.1f}s, best={best_score:.7f}", flush=True)

    # === PHASE 5: Final Polish ===
    if best is not None and time.monotonic() < deadline:
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: Final Polish", flush=True)
        print(f"{'='*60}", flush=True)
        t0 = time.monotonic()
        n = best.size

        # Mirror descent polish with small steps
        est_time = 0.001 * n / 512.0
        md_time = (deadline - time.monotonic()) * 0.3
        max_iters = max(5000, int(md_time / max(est_time, 0.0003)))
        a_result, score = run_mirror_descent(
            best, time.monotonic() + md_time, max_iters=max_iters,
            eta0=0.005, eta_q0=0.1, eta_decay=0.99999,
            eta_q_decay=0.99999, q_interpolate=0.3, print_interval=100000)
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  NEW BEST (MD polish): J={score:.7f}", flush=True)

        # LP polish
        if time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            est_time = 0.025 if n <= 256 else 0.04 if n <= 512 else 0.07
            max_iters = max(100, int(remaining * 0.4 / est_time))
            a_result, score = lp_iteration(best, max_iters, deadline)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  NEW BEST (LP polish): J={score:.7f}", flush=True)

        # Smooth
        if time.monotonic() < deadline - 5:
            a_result, score = smooth_refine(best, deadline - 3,
                                            betas=[1000, 5000, 20000, 100000])
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  NEW BEST (smooth polish): J={score:.7f}", flush=True)

        print(f"  Phase 5 done in {time.monotonic()-t0:.1f}s, best={best_score:.7f}", flush=True)

    # === Final Validation ===
    if best is None:
        best = np.ones(256) / 256
        best_score = compute_score(best)

    final_score = float(evaluate_sequence([float(x) for x in best]))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={best_score:.7f}, n={best.size}", flush=True)
    print(f"{'='*60}", flush=True)
    return [float(x) for x in best]
