def normalize(f):

def compute_score(f):

def fft_convolve(a, b):
    """FFT-based convolution for efficiency."""
    n = len(a) + len(b) - 1
    n_fft = 1 << (n - 1).bit_length()
    A = np.fft.fft(a, n_fft)
    B = np.fft.fft(b, n_fft)
    return np.fft.ifft(A * B).real[:n]


def build_conv_matrix(f):
    """Build the convolution matrix A such that A @ x = convolve(f, x) (full)."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = f[j]
    return A


def build_conv_matrix_fast(f):
    """Build convolution matrix using structured approach."""
    n = len(f)
    m = 2 * n - 1
    # A[k, i] = f[k - i] if 0 <= k-i < n, else 0
    # This is a Hankel-like matrix
    A = np.zeros((m, n))
    for i in range(n):
        # For column i, A[k, i] = f[k-i] for k such that 0 <= k-i < n
        # i.e., k in [i, i+n-1]
        start = i
        end = min(m, i + n)
        A[start:end, i] = f[:end - start]
    return A

                                               'dual_feasibility_tolerance': 1e-8,
                                               'primal_feasibility_tolerance': 1e-8})
    except Exception:
        return None
    
    if not res.success or res.x is None:
        return None
    
    b = np.clip(res.x, 0, None)
    s = b.sum()
    if s < 1e-15:
        return None
    return b / s

def local_search(f, n_iter=500, rng=None):
    """Local search with structured perturbations."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    cur_score = best_score
    
    T = 0.02
    for it in range(n_iter):
        T *= 0.999
        f_new = f.copy()
        r = rng.random()
        
        if r < 0.3:
            # Block scaling
            w = max(1, int(n * rng.uniform(0.01, 0.1)))
            s = int(rng.integers(0, n - w + 1))
            f_new[s:s+w] *= np.exp(rng.normal(0, 0.3))
        elif r < 0.5:
            # Single element
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] * np.exp(rng.normal(0, 0.5)))
        elif r < 0.7:
            # Mirror partial
            i = int(rng.integers(0, n // 2))
            j = n - 1 - i
            t = rng.uniform(0, 1)
            f_new[i], f_new[j] = (1-t)*f_new[i] + t*f_new[j], (1-t)*f_new[j] + t*f_new[i]
        elif r < 0.85:
            # Gaussian kernel perturbation
            c = int(rng.integers(0, n))
            w = max(2, n // 16)
            kernel = np.exp(-0.5 * (np.arange(n) - c) ** 2 / w ** 2)
            f_new += rng.normal(0, 0.05) * kernel
        else:
            # Global gentle perturbation
            f_new *= np.exp(rng.normal(0, 0.1, n))
        
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score(f_new)
        
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
    
    return best_f, best_score

def lp_refine(f, time_budget, rng=None, max_iters=100):
    """LP-based refinement with adaptive M and step size."""

"""
Hybrid optimization for minimizing peak autocorrelation of non-negative sequences.
Uses LP-based direction finding with adaptive bounds, spectral projection,
subgradient descent, coordinate descent, and local search across multiple lengths.
Key innovations: adaptive M relaxation in LP, multi-direction LP combining,
and interleaved spectral smoothing to escape local minima.
"""

import numpy as np
from scipy.optimize import linprog
import time


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def compute_score(f):
    """Compute score: 2*n*max(conv) assuming f is normalized to sum=1."""
    n = len(f)
    conv = fft_convolve(f, f)
    return 2.0 * n * conv.max()


def fft_convolve(a, b):
    """FFT-based full convolution for efficiency."""
    na, nb = len(a), len(b)
    n = na + nb - 1
    n_fft = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, n_fft)
    B = np.fft.rfft(b, n_fft)
    return np.fft.irfft(A * B, n_fft)[:n]


def build_conv_matrix_fast(f):
    """Build convolution matrix A such that A @ x = np.convolve(f, x, mode='full')."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for i in range(n):
        start = i
        end = min(m, i + n)
        A[start:end, i] = f[:end - start]
    return A


def lp_direction(f, M, time_limit=1.0):
    """LP: maximize sum(b) s.t. conv(f,b) <= M, b >= 0. Returns normalized direction or None."""
    n = len(f)
    A = build_conv_matrix_fast(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit,
                               'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
    except Exception:
        return None
    if not res.success or res.x is None:
        return None
    b = np.clip(res.x, 0, None)
    s = b.sum()
    if s < 1e-15:
        return None
    return b / s


def spectral_projection(f, n_iter=15, rng=None, aggressive=0.0):
    """Iterative spectral flattening with optional aggressive targeting."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()
    best_f = f.copy()
    best_score = compute_score(f)

    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)
        n_bins = len(mag)

        # Target flat magnitude for non-DC bins
        energy = np.sum(f ** 2)
        # DC component magnitude is 1 (sum=1)
        # Total: 1^2 + sum_{k=1}^{n_bins-1} mag_k^2 = n_fft * energy (Parseval with rfft scaling)
        # Actually: sum_k |F[k]|^2 / n_fft = sum f^2
        # So sum_{k=0}^{n_bins-1} mag_k^2 = n_fft * energy
        # mag_0 = 1, so sum_{k=1}^{n_bins-1} mag_k^2 = n_fft*energy - 1
        # Target: each = (n_fft*energy - 1) / (n_bins - 1)
        remaining = n_fft * energy - 1.0
        if remaining < 0:
            remaining = 0
        target = np.sqrt(remaining / max(1, n_bins - 1))

        # Blend: some fraction flat, some fraction current
        t_flat = 1.0 - aggressive
        mag_new = np.empty(n_bins)
        mag_new[0] = 1.0
        mag_new[1:] = t_flat * target + (1.0 - t_flat) * mag[1:]

        # Optional: add small random perturbation to phases
        if it % 10 == 9 and rng is not None:
            phase = phase + rng.normal(0, 0.02, len(phase))

        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        score = compute_score(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
        f = f_new

    return best_f, best_score


def pressure_subgradient(f, n_iter=150, eta0=0.12, rng=None):
    """Subgradient descent using pressure at convolution peaks."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0

    for it in range(n_iter):
        conv = fft_convolve(f, f)
        M = conv.max()
        # Find peaks (positions where conv is near max)
        threshold = 0.85 * M
        peak_idx = np.where(conv >= threshold)[0]

        # Compute pressure vector
        pressure = np.zeros(n)
        for k in peak_idx:
            w = (conv[k] - 0.85 * M) / (0.15 * M + 1e-15)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += w * f[j]

        if pressure.max() > 0:
            pressure = pressure / pressure.max()

        eta = eta0 * 1.0 / (1.0 + 0.03 * it)
        f_new = f * np.exp(-eta * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)
        score = compute_score(f_new)

        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 80:
                # Perturb
                f_pert = f * (1 + 0.15 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    stall = 0
                else:
                    break

    return best_f, best_score


def coord_descent(f, n_sweeps=5, rng=None):
    """Greedy coordinate descent with multiplicative steps."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n) if rng.random() > 0.3 else np.arange(n)
        for i in order:
            orig = f[i]
            if orig < 1e-12:
                continue
            best_val = orig
            best_local = best_score
            for factor in [0.3, 0.5, 0.7, 0.85, 1.2, 1.5, 2.0, 3.0]:
                f[i] = max(0.0, orig * factor)
                s = f.sum()
                if s < 1e-10:
                    f[i] = orig
                    continue
                conv = fft_convolve(f, f)
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]
            f[i] = best_val
        f = normalize(f)
        cs = compute_score(f)
        if cs < best_score:
            best_score = cs

    return f, best_score


def local_search(f, n_iter=1000, rng=None):
    """Simulated annealing with diverse structured moves."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    cur_score = best_score
    T = 0.03
    no_improve = 0

    for it in range(n_iter):
        T *= 0.9995
        f_new = f.copy()
        r = rng.random()

        if r < 0.25:
            w = max(1, int(n * rng.uniform(0.01, 0.08)))
            s = int(rng.integers(0, n - w + 1))
            f_new[s:s + w] *= np.exp(rng.normal(0, 0.35))
        elif r < 0.45:
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] * np.exp(rng.normal(0, 0.6)))
        elif r < 0.65:
            i = int(rng.integers(0, n // 2))
            j = n - 1 - i
            t = rng.uniform(0, 1)
            f_new[i], f_new[j] = (1 - t) * f_new[i] + t * f_new[j], \
                                  (1 - t) * f_new[j] + t * f_new[i]
        elif r < 0.8:
            c = int(rng.integers(0, n))
            w = max(3, n // 16)
            kernel = np.exp(-0.5 * (np.arange(n) - c) ** 2 / w ** 2)
            f_new += rng.normal(0, 0.06) * kernel
        else:
            f_new *= np.exp(rng.normal(0, 0.12, n))

        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score(f_new)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 3000 and it > n_iter // 2:
            break

    return best_f, best_score


def lp_refine(f, time_budget, rng=None, max_iters=300):
    """LP-based refinement with adaptive M, multi-direction, and step selection."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    t = 0.4
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_budget:
            break
        remaining_budget = time_budget - elapsed
        # Decrease LP time as budget shrinks
        lp_t = max(0.3, min(2.0, remaining_budget / (3 * max(1, max_iters - it))))

        conv = fft_convolve(f, f)
        M = conv.max()

        # Try multiple M values and collect directions
        directions = []
        for m_factor in [1.0, 1.02, 1.05, 0.98]:
            g = lp_direction(f, M * m_factor, time_limit=lp_t)
            if g is not None:
                directions.append(g)

        if not directions:
            stall += 1
            if stall > 20:
                break
            continue

        # Try each direction with multiple step sizes
        improved = False
        for g in directions:
            for step in [min(1.0, t), t * 0.5, t * 0.2, 0.1]:
                f_new = (1 - step) * f + step * g
                f_new = np.clip(f_new, 0, None)
                s = f_new.sum()
                if s < 1e-10:
                    continue
                f_new = f_new / s
                new_score = compute_score(f_new)
                if new_score < best_score - 1e-12:
                    best_f = f_new.copy()
                    best_score = new_score
                    f = f_new
                    t = min(1.0, t * 1.08)
                    stall = 0
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            t *= 0.7
            if t < 0.02:
                # Perturb to escape
                f_pert = f * (1 + 0.12 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    t = 0.3
                    stall = 0
                elif stall > 30:
                    break

    return best_f, best_score


def init_diverse(n, rng):
    """Generate a diverse initial sequence."""
    r = rng.random()
    if r < 0.2:
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    elif r < 0.4:
        x = np.arange(n) / max(1, n - 1)
        f = 1 - np.abs(2 * x - 1)
    elif r < 0.55:
        c1, c2 = n * 0.3, n * 0.7
        w = n * 0.12
        f = np.exp(-0.5 * (np.arange(n) - c1) ** 2 / w ** 2) + \
            np.exp(-0.5 * (np.arange(n) - c2) ** 2 / w ** 2)
    elif r < 0.7:
        f = np.linspace(0.1, 1.0, n)
    elif r < 0.85:
        f = np.sin(np.pi * np.arange(n) / n) ** 2
    else:
        f = np.ones(n)
        for _ in range(4):
            c = int(rng.integers(0, n))
            w = max(2, n // 10)
            kernel = np.exp(-0.5 * (np.arange(n) - c) ** 2 / w ** 2)
            f *= (1 + rng.normal(0, 0.4) * kernel)
        f = np.abs(f) + 0.05

    f = np.clip(f, 0.01, None)
    f = normalize(f)
    f = f * (1 + 0.03 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    return normalize(f)


def upsample_seq(f, n_new):
    """Upsample a sequence to a new length via interpolation."""
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, n_new)
    f_new = np.interp(x_new, x_old, f)
    return np.clip(f_new, 0.001, None)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Combines LP refinement, spectral projection,
    subgradient descent, coordinate descent, and local search.
    Explores multiple sequence lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 12.0

    best_overall = None
    best_score_overall = np.inf

    # Load parent
    parent = None
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None

    if parent is not None:
        parent = np.clip(parent, 0, None)
        if parent.sum() > 0.01:
            parent = normalize(parent)
            parent_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
            best_overall = parent.copy()
            best_score_overall = parent_score

    # Lengths to explore
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for L in [512, 768, 1024]:
        if L not in lengths:
            lengths.append(L)

    print(f"Lengths to try: {lengths}, budget: {budget_s}s, target: 1.5027")

    for n in lengths:
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        print(f"\n{'='*50}")
        print(f"n={n}, remaining={remaining:.0f}s, current best={best_score_overall:.8f}")
        print(f"{'='*50}")

        # Get initial sequence
        if parent is not None and len(parent) == n:
            f0 = parent.copy()
        else:
            f0 = init_diverse(n, rng)
            # If we have a good solution at a different length, try upsampling
            if best_overall is not None and len(best_overall) != n:
                f_up = upsample_seq(best_overall, n)
                f_up = normalize(f_up)
                if compute_score(f_up) < compute_score(f0):
                    f0 = f_up

        # Phase 1: Quick spectral + subgradient (~15s)
        t1 = min(15, remaining * 0.12)
        if t1 > 5:
            f_s, s_s = spectral_projection(f0, n_iter=int(t1 * 2), rng=rng, aggressive=0.2)
            print(f"  Spectral: {s_s:.8f}")
            if s_s < best_score_overall:
                best_score_overall = s_s
                best_overall = f_s.copy()

            f_sg, s_sg = pressure_subgradient(f_s, n_iter=int(t1 * 8), eta0=0.12, rng=rng)
            print(f"  Subgradient: {s_sg:.8f}")
            if s_sg < best_score_overall:
                best_score_overall = s_sg
                best_overall = f_sg.copy()

        # Phase 2: LP refinement (main driver, ~40% of remaining)
        remaining = deadline - time.monotonic()
        t2 = max(20, min(remaining * 0.4, 150))
        if remaining > t2 + 20:
            f_lp, s_lp = lp_refine(f_sg, time_budget=t2, rng=rng, max_iters=400)
            print(f"  LP round 1: {s_lp:.8f}")
            if s_lp < best_score_overall:
                best_score_overall = s_lp
                best_overall = f_lp.copy()

        # Phase 3: Spectral interlude + LP again
        remaining = deadline - time.monotonic()
        if remaining > 30:
            f_sp2, s_sp2 = spectral_projection(best_overall, n_iter=10, rng=rng, aggressive=0.5)
            if s_sp2 < best_score_overall:
                best_score_overall = s_sp2
                best_overall = f_sp2.copy()

            t3 = max(10, min(remaining * 0.25, 80))
            if remaining > t3 + 10:
                f_lp2, s_lp2 = lp_refine(best_overall, time_budget=t3, rng=rng, max_iters=200)
                print(f"  LP round 2: {s_lp2:.8f}")
                if s_lp2 < best_score_overall:
                    best_score_overall = s_lp2
                    best_overall = f_lp2.copy()

        # Phase 4: Coordinate descent + local search
        remaining = deadline - time.monotonic()
        if remaining > 15:
            n_cd = max(3, int(min(8, remaining * 0.15)))
            f_cd, s_cd = coord_descent(best_overall, n_sweeps=n_cd, rng=rng)
            print(f"  CD: {s_cd:.8f}")
            if s_cd < best_score_overall:
                best_score_overall = s_cd
                best_overall = f_cd.copy()

        remaining = deadline - time.monotonic()
        if remaining > 12:
            n_ls = max(300, int(min(5000, remaining * 80)))
            f_ls, s_ls = local_search(best_overall, n_iter=n_ls, rng=rng)
            print(f"  Local search: {s_ls:.8f}")
            if s_ls < best_score_overall:
                best_score_overall = s_ls
                best_overall = f_ls.copy()

        # Phase 5: Final LP pass
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t5 = max(10, min(remaining * 0.4, 60))
            f_lp3, s_lp3 = lp_refine(best_overall, time_budget=t5, rng=rng, max_iters=200)
            print(f"  LP round 3: {s_lp3:.8f}")
            if s_lp3 < best_score_overall:
                best_score_overall = s_lp3
                best_overall = f_lp3.copy()

    # Final polish
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10:
            print(f"\n{'='*50}")
            print(f"Final polish: remaining={remaining:.0f}s, score={best_score_overall:.8f}")
            print(f"{'='*50}")

            # Quick CD
            f_cd, s_cd = coord_descent(best_overall, n_sweeps=3, rng=rng)
            if s_cd < best_score_overall:
                best_score_overall = s_cd
                best_overall = f_cd.copy()
                print(f"  After CD: {best_score_overall:.8f}")

            remaining = deadline - time.monotonic()
            if remaining > 15:
                f_lp, s_lp = lp_refine(best_overall, time_budget=min(remaining * 0.5, 30),
                                       rng=rng, max_iters=80)
                if s_lp < best_score_overall:
                    best_score_overall = s_lp
                    best_overall = f_lp.copy()
                    print(f"  After final LP: {best_score_overall:.8f}")

            # Short local search
            remaining = deadline - time.monotonic()
            if remaining > 8:
                f_ls, s_ls = local_search(best_overall, n_iter=min(2000, int(remaining * 100)), rng=rng)
                if s_ls < best_score_overall:
                    best_score_overall = s_ls
                    best_overall = f_ls.copy()
                    print(f"  After final LS: {best_score_overall:.8f}")

    # Return best
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000.0)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\nFinal result: score={final_score:.8f}, n={len(final_list)}")
        return final_list

    # Fallback
    print("Fallback: uniform 512")
    return [1.0] * 512
