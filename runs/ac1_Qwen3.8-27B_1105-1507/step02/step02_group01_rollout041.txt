def compute_obj(a):
    """Compute the objective: 2*n*max(conv) / sum^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_conv_matrix_rows(a):
    """Build constraint matrix for LP: A[k, j] gives coefficient of a[j] in (a_fixed * g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_direction(a, upper_bound=50.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0, g <= upper_bound."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_conv_matrix_rows(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tol': 1e-8})

            g = res.x

        if g is None or g_sum <= 1.0 + 1e-4:
            break
        # Try several step sizes
        found = False
        for tt in [t, t * 0.5, t * 0.25, t * 0.1]:
            a_new = (1.0 - tt) * a + tt * g
            a_new = np.maximum(a_new, 0.0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
                found = True
                break
        if not found:
            t *= 0.5
            if t < 0.01:
                break
        else:
            t = max(t * 1.1, 0.1)

    return best_a, best_obj


def block_refine(a, max_time, seed=0):
    """Random block multiplier refinement."""
    rng = np.random.default_rng(seed)
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0

    while time.monotonic() - start < max_time:
        # Try different block sizes
        if it % 10 == 0:
            width = max(1, int(n * rng.uniform(0.005, 0.15)))
        else:
            width = max(1, int(n * rng.uniform(0.01, 0.05)))
        pos = int(rng.integers(0, max(1, n - width + 1)))
        candidate = best_a.copy()
        sigma = 0.02 + 0.06 * max(0.0, 1.0 - it / 100000)
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.maximum(candidate, 0.0)
        s = float(np.sum(candidate))
        if s < 1e-10:
            it += 1
            continue
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate.copy()
            best_obj = obj
        it += 1

    return best_a, best_obj


def coord_refine(a, max_time, seed=0):
    """Coordinate descent: optimize one element at a time."""
    rng = np.random.default_rng(seed)
    n = len(a)
    best_a = a.copy().astype(np.float64)
    best_a = np.maximum(best_a, 0.0)
    s = float(np.sum(best_a))
    if s < 1e-10:
        return best_a, np.inf
    best_a = best_a / s
    best_obj = compute_obj(best_a)
    start = time.monotonic()
    a = best_a.copy()

    while time.monotonic() - start < max_time:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - start > max_time:
                break
            old_val = a[i]
            # Try several candidate values for a[i]
            # The contribution to conv from a[i] is 2*a[i]*a[k-i] at positions k in [i, i+n-1]
            # We want to minimize max_conv / (sum)^2
            base_sum = float(np.sum(a)) - old_val
            # Current conv without a[i]'s contribution at position i
            # Just try a few multipliers
            for mult in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 0.0]:
                a[i] = old_val * mult
                a[i] = max(0.0, a[i])
                s_new = base_sum + a[i]
                if s_new < 1e-10:
                    continue
                obj = compute_obj(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = old_val
            if not improved and (i % 500 == 0):
                # Add small noise to escape
                a[i] = max(0.0, old_val * (1.0 + rng.normal(0, 0.1)))
                s_new = float(np.sum(a))
                if s_new > 1e-10:
                    a = a / s_new

    return best_a, best_obj


def init_uniform(n):
    return np.ones(n) / n


def init_edge_symmetric(n, rng):
    """High at both edges, low in middle (symmetric)."""
    a = np.ones(n) * 0.1
    edge_val = 1.0 / np.sqrt(2 * n)
    a[0] = edge_val
    a[-1] = edge_val
    if n > 2:
        a[1] = edge_val * 0.5
        a[-2] = edge_val * 0.5
    a += rng.normal(0, 0.02, n)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    return a / s if s > 1e-10 else np.ones(n) / n


def init_edge_asymmetric(n, rng, alpha=0.7):
    """High at one edge, lower at the other (asymmetric)."""
    a = np.ones(n) * 0.05
    a[0] = alpha * (1.0 / np.sqrt(n))
    a[-1] = (1 - alpha) * (1.0 / np.sqrt(n))
    # Add a smooth decay from the high edge
    decay = np.exp(-np.arange(n) / (n * 0.3))
    a += 0.3 * decay / n
    a += rng.normal(0, 0.01, n)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    return a / s if s > 1e-10 else np.ones(n) / n


def init_geometric(n, rng):
    """Geometric sequence with random ratio."""
    r = rng.uniform(0.7, 0.98)
    a = r ** np.arange(n)
    a += 0.1 * np.ones(n)
    a += rng.normal(0, 0.02, n)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    return a / s if s > 1e-10 else np.ones(n) / n


def init_two_block(n, rng):
    """Two blocks of different heights."""
    split = int(n * rng.uniform(0.3, 0.7))
    a = np.zeros(n)
    a[:split] = rng.uniform(0.5, 1.5)
    a[split:] = rng.uniform(0.1, 0.5)
    a += rng.normal(0, 0.05, n)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    return a / s if s > 1e-10 else np.ones(n) / n


def init_periodic(n, rng):
    """Periodic pattern with random phase and frequency."""
    freq = rng.uniform(1, 5)
    phase = rng.uniform(0, 2 * np.pi)
    a = np.maximum(0, 1.0 + np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
    a += rng.normal(0, 0.1, n)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    return a / s if s > 1e-10 else np.ones(n) / n


def init_mixed(n, rng):
    """Mix of several initialization strategies."""
    choice = int(rng.integers(0, 6))
    if choice == 0:
        return init_uniform(n)
    elif choice == 1:
        return init_edge_symmetric(n, rng)
    elif choice == 2:
        return init_edge_asymmetric(n, rng, alpha=rng.uniform(0.5, 0.9))
    elif choice == 3:
        return init_geometric(n, rng)
    elif choice == 4:
        return init_two_block(n, rng)
    else:
        return init_periodic(n, rng)


def full_refine(a, time_budget, seed=0):
    """Multi-phase refinement pipeline."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: Soft-max GD (main workhorse)
    gd_time1 = min(time_budget * 0.4, 120.0)
    a, obj = softmax_gd(a, gd_time1, seed=seed)
    if obj < best_obj:
        best_a = a.copy()
        best_obj = obj

    # Phase 2: LP polish
    if time.monotonic() - start < time_budget * 0.7:
        lp_time = min(time_budget * 0.15, 40.0)
        a2, obj2 = lp_refine(best_a, lp_time, max_iters=30)
        if obj2 < best_obj:
            best_a = a2.copy()
            best_obj = obj2

    # Phase 3: Second GD round (different T)
    if time.monotonic() - start < time_budget * 0.85:
        gd_time2 = min(time_budget * 0.15, 60.0)
        a3, obj3 = softmax_gd(best_a, gd_time2, T_init=0.001, seed=seed + 1)
        if obj3 < best_obj:
            best_a = a3.copy()
            best_obj = obj3

    # Phase 4: Coordinate descent
    if time.monotonic() - start < time_budget * 0.95:
        coord_time = min(time_budget * 0.08, 30.0)
        a4, obj4 = coord_refine(best_a, coord_time, seed=seed + 2)
        if obj4 < best_obj:
            best_a = a4.copy()
            best_obj = obj4

    # Phase 5: Block refinement
    if time.monotonic() - start < time_budget * 0.99:
        bp_time = max(1.0, time_budget - (time.monotonic() - start) - 2.0)
        a5, obj5 = block_refine(best_a, bp_time, seed=seed + 3)
        if obj5 < best_obj:
            best_a = a5.copy()
            best_obj = obj5

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization for step function coefficients.
    Uses soft-max gradient descent with momentum, LP polish, coordinate descent,
    and block refinement across multiple sequence lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + float(budget_s) - 8.0

    best_seq = None
    best_score = np.inf

    # Try parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    # Optimize parent
    parent_a = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all() and parent.size <= 1000:
            parent_a = parent.copy()
            p_obj = compute_obj(parent_a)
            print(f"Parent initial: n={parent_a.size}, obj={p_obj:.6f}")
    except Exception:
        parent_a = None

    # Define search space
    # (n, time_per_attempt, num_attempts)
    n_configs = [
        (50, 20, 3),
        (100, 25, 3),
        (150, 30, 3),
        (200, 35, 3),
        (300, 40, 3),
        (500, 50, 2),
        (800, 60, 2),
        (1000, 70, 2),
        (1500, 80, 1),
        (2000, 90, 1),
    ]

    # Also try parent optimization if available
    if parent_a is not None and parent_a.size > 0:
        n_configs.insert(0, (parent_a.size, 30, 1))

    # Explore
    results = []
    for n, time_per, num_attempts in n_configs:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        for att in range(num_attempts):
            if time.monotonic() > deadline - 60:
                break

            if n == parent_a.size and parent_a is not None and att == 0:
                a_init = parent_a.copy()
            else:
                a_init = init_mixed(n, rng)

            t_budget = min(time_per, max(5.0, remaining / max(1, num_attempts - att + 1) - 5.0))
            a_opt, obj_opt = full_refine(a_init, t_budget, seed=seed + n * 10 + att)

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} att={att}: obj={obj_opt:.6f} ** NEW BEST **")
            else:
                print(f"  n={n} att={att}: obj={obj_opt:.6f}")
            results.append((a_opt.copy(), obj_opt, n))

    # Sort results by score
    results.sort(key=lambda x: x[1])

    # Intensive refinement on top candidates
    top_k = min(3, len(results))
    for rank in range(top_k):
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        a_top, obj_top, n_top = results[rank]
        print(f"\nIntensive refinement: rank={rank}, n={n_top}, current_obj={obj_top:.6f}, time_left={remaining:.0f}s")

        # Extended GD
        ext_time = min(remaining * 0.35, 200.0)
        a_ext, obj_ext = softmax_gd(a_top, ext_time, T_init=0.005, seed=seed + 100 + rank)
        if obj_ext < best_score:
            best_score = obj_ext
            best_seq = a_ext.copy()
            print(f"  After ext GD: obj={best_score:.6f}")

        # LP polish
        if time.monotonic() < deadline - 20:
            lp_t = min(remaining * 0.15, 60.0)
            a_lp, obj_lp = lp_refine(best_seq, lp_t, max_iters=40)
            if obj_lp < best_score:
                best_score = obj_lp
                best_seq = a_lp.copy()
                print(f"  After LP: obj={best_score:.6f}")

        # More GD
        if time.monotonic() < deadline - 15:
            gd_t = min(remaining * 0.2, 100.0)
            a_gd2, obj_gd2 = softmax_gd(best_seq, gd_t, T_init=0.002, seed=seed + 200 + rank)
            if obj_gd2 < best_score:
                best_score = obj_gd2
                best_seq = a_gd2.copy()
                print(f"  After GD2: obj={best_score:.6f}")

        # Coordinate descent
        if time.monotonic() < deadline - 15:
            coord_t = min(remaining * 0.15, 60.0)
            a_cd, obj_cd = coord_refine(best_seq, coord_t, seed=seed + 300 + rank)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  After coord: obj={best_score:.6f}")

        # Block refinement
        if time.monotonic() < deadline - 10:
            bp_t = max(5.0, deadline - time.monotonic() - 5.0)
            a_bp, obj_bp = block_refine(best_seq, bp_t, seed=seed + 400 + rank)
            if obj_bp < best_score:
                best_score = obj_bp
                best_seq = a_bp.copy()
                print(f"  After block: obj={best_score:.6f}")

        # Final micro-refinement
        if time.monotonic() < deadline - 5:
            micro_t = max(2.0, deadline - time.monotonic() - 3.0)
            a_micro, obj_micro = softmax_gd(best_seq, micro_t, T_init=0.001,
                                            lr_init=0.002, seed=seed + 500 + rank)
            if obj_micro < best_score:
                best_score = obj_micro
                best_seq = a_micro.copy()
                print(f"  After micro: obj={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - total_start
    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
