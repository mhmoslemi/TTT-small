Wait, there's an issue. When I change a[j], the convolution b changes, so my precomputed b is no longer valid for the next coordinate. I need to either:
1. Recompute b after each coordinate change (expensive: O(n log n) per coordinate)
2. Use the "Gauss-Seidel" approach where I use the updated values
3. Do a "Jacobi" sweep where I use the old b for all coordinates in the sweep

Option 3 is the simplest and should work well enough for a polish step. The key is that we're close to the optimum, so the changes are small.

But actually, there's a subtlety: after changing a[j], the values of b change, which affects the optimization for the next coordinate. For a proper coordinate descent, I should use the updated b. But for efficiency, I'll use the "stale" b and just do multiple sweeps.

Actually, for the coordinate descent to be correct, I need to use the current value of a (including previous updates in the same sweep) when computing the base. Let me just recompute b after each coordinate update. For n=128, fast_self_conv is O(n log n) ≈ 1000 operations, and we have n=128 coordinates, so 128 * 1000 = 128K per sweep. That's very fast.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_self_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def fast_conv(a, b):
    """General convolution via FFT."""
    n = len(a) + len(b) - 1
    s = 1
    while s < n:
        s <<= 1
    fa = np.fft.rfft(a, s)
    fb = np.fft.rfft(b, s)
    return np.fft.irfft(fa * fb, s)[:n]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_self_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                A[k, j] = a[k - j]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-12:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def coordinate_descent_polish(a, n_sweeps=30, tol=1e-13):
    """
    Polish using coordinate descent on max(conv(a,a)).
    For each coordinate j, the function max_k h_k(x) is 1D convex;
    find the minimizer via subgradient bisection.
    """
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s

    for sweep in range(n_sweeps):
        b = fast_self_conv(a)
        max_improvement = 0.0

        for j in range(n):
            # Compute coefficient vector for coordinate j
            coef = np.zeros(2 * n - 1)
            quad = np.zeros(2 * n - 1)
            # k ranges such that k-j is in [0, n-1]: k in [j, j+n-1]
            k_start = max(j, 0)
            k_end = min(j + n - 1, 2 * n - 2)
            if k_start <= k_end:
                l_vals = np.arange(k_start - j, k_end - j + 1)
                coef[k_start:k_end + 1] = 2.0 * a[l_vals]
            # Quadratic term at k = 2j
            if 0 <= 2 * j <= 2 * n - 2:
                coef[2 * j] = 0.0
                quad[2 * j] = 1.0

            aj = a[j]
            # c[k] = b[k] - aj*coef[k] - quad[k]*aj^2
            c = b - aj * coef - quad * aj * aj

            # f(x) = max_k (c[k] + x*coef[k] + quad[k]*x^2)
            f_cur = float(np.max(c + aj * coef + quad * aj * aj))

            # Subgradient (leftmost) at x
            def _subgrad_left(x):
                vals = c + x * coef + quad * x * x
                mx = vals.max()
                active = np.where(vals >= mx - 1e-12 * max(1.0, mx))[0]
                if len(active) == 0:
                    active = np.array([np.argmax(vals)])
                return float(np.min(coef[active] + 2.0 * quad[active] * x))

            # Check boundary x=0
            sg0 = _subgrad_left(0.0)
            if sg0 >= -1e-12:
                x_opt = 0.0
            else:
                # Find upper bracket
                hi = max(aj * 2.0 + 0.5, 1.0)
                sg_hi = _subgrad_left(hi)
                while sg_hi < -1e-12 and hi < 1e4:
                    hi *= 2.0
                    sg_hi = _subgrad_left(hi)
                if sg_hi < -1e-12:
                    x_opt = hi
                else:
                    lo = 0.0
                    for _ in range(60):
                        mid = (lo + hi) * 0.5
                        sg_mid = _subgrad_left(mid)
                        if sg_mid < 0:
                            lo = mid
                        else:
                            hi = mid
                        if hi - lo < 1e-12 * max(1.0, hi):
                            break
                    x_opt = (lo + hi) * 0.5

            x_opt = max(x_opt, 0.0)
            f_new = float(np.max(c + x_opt * coef + quad * x_opt * x_opt))
            if f_new < f_cur - tol:
                a[j] = x_opt
                improvement = f_cur - f_new
                if improvement > max_improvement:
                    max_improvement = improvement
                # Recompute b for next coordinates
                b = fast_self_conv(a)

        # Renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            break
        a /= s

        if max_improvement < 1e-13:
            break

    return a

def run_lp_iterations(a, max_iters, deadline, tracker):
    """LP mirror descent with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_self_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        # Adaptive step with more candidates
        base_t = max(0.03, 0.6 * (1.0 - it / float(3 * max_iters))**1.5)
        candidates_t = [base_t, base_t * 0.5, base_t * 1.5, base_t * 0.25, 0.05, 0.15, 0.3]
        candidates_t = [min(max(t, 0.01), 1.0) for t in candidates_t]
        # Remove duplicates
        seen = set()
        unique_t = []
        for t in candidates_t:
            key = round(t, 4)
            if key not in seen:
                seen.add(key)
                unique_t.append(t)

        best_a = None
        best_J = current_J

        for t in unique_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        else:
            a = best_a
            current_J = best_J

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def generate_initializations(n, rng, parent, diversity=20):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.2, 1.0, n))

    # Cosine waves at various frequencies
    for freq in [1, 2, 3, 4, 5, 6, 8, 10]:
        for amp in [0.2, 0.4, 0.6]:
            phase = rng.uniform(0, 2 * np.pi)
            inits.append(np.maximum(1.0 + amp * np.cos(2.0 * np.pi * freq * t_arr + phase), 0.01))

    # Linear ramps
    for slope in [-0.3, -0.1, 0.1, 0.3]:
        inits.append(np.maximum(1.0 + slope * (2.0 * t_arr - 1.0), 0.01))

    # Quadratic
    for c in [-0.5, 0.5]:
        inits.append(np.maximum(1.0 + c * (2.0 * t_arr - 1.0)**2, 0.01))

    # Gaussian bumps
    for center in [0.25, 0.5, 0.75]:
        for width in [0.1, 0.2, 0.4]:
            inits.append(np.exp(-((t_arr - center)**2) / (2 * width**2)) + 0.05)

    # Double bump
    inits.append(np.exp(-((t_arr - 0.3)**2) * 20) + np.exp(-((t_arr - 0.7)**2) * 20) + 0.03)

    # Sine (non-negative)
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.1)
    inits.append(np.abs(np.sin(2 * np.pi * t_arr)) + 0.1)

    # Sparse spikes
    for n_spikes in [4, 8, 16]:
        sparse = np.zeros(n)
        positions = rng.choice(n, size=min(n_spikes, n), replace=False)
        sparse[positions] = 1.0
        kernel = np.ones(max(2, n // n_spikes)) / max(2, n // n_spikes)
        inits.append(np.maximum(np.convolve(sparse, kernel, mode='same'), 0.01))

    # Power law
    inits.append((t_arr + 0.1) ** rng.uniform(-2, 2))
    inits.append((1 - t_arr + 0.1) ** rng.uniform(-2, 2))

    # Fourier-based (random)
    n_harm = 5
    coeffs = rng.normal(0, 0.3, n_harm)
    inits.append(np.maximum(1.0 + np.abs(np.fft.irfft(np.fft.rfft(coeffs, n)))[:n], 0.01))

    # Triangle wave
    inits.append(np.abs(2 * (t_arr - np.floor(t_arr + 0.5))) + 0.1)

    # Sawtooth
    inits.append(np.abs(t_arr * 4 - 2 * np.floor(t_arr * 2)) + 0.1)

    # Random smooth
    n_ctrl = 8
    ctrl = rng.uniform(0.3, 1.0, n_ctrl)
    t_ctrl = np.linspace(0, 1, n_ctrl)
    inits.append(np.maximum(np.interp(t_arr, t_ctrl, ctrl), 0.01))

    return inits

                tracker.report(f"SA r={r}: {time.monotonic()-t0:.1f}s")

            temperature *= 0.985

        tracker.report(f"Phase3 SA: {time.monotonic()-t0:.1f}s")

    # === Phase 4: L-BFGS-B refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase4 L-BFGS-B n={n} {'='*50}", flush=True)

        def lbb_objective(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-8:
                return 1e10
            ax = ax / s
            b = fast_self_conv(ax)
            return float(b.max())

        bounds_list = [(0.0, None)] * n
        try:
            res = minimize(lbb_objective, tracker.best.astype(np.float64), method='L-BFGS-B',
                           bounds=bounds_list,
                           options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-18, 'gtol': 1e-14})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)

        tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final LP + Coord Descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase5 Final Polish n={n} {'='*50}", flush=True)

        a = run_lp_iterations(tracker.best, 200, deadline, tracker)
        a = coordinate_descent_polish(a, n_sweeps=50)
        tracker.update(a)

        # One more LP round
        if time.monotonic() < deadline:
            a = run_lp_iterations(tracker.best, 100, deadline, tracker)
            a = coordinate_descent_polish(a, n_sweeps=30)
            tracker.update(a)

        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # === Final: heavily polish the absolute best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Final Deep Polish {'='*50}", flush=True)
        a = coordinate_descent_polish(tracker.best, n_sweeps=100)
        tracker.update(a)
        # LP one more time
        if time.monotonic() < deadline:
            a = run_lp_iterations(tracker.best, 100, deadline, tracker)
            a = coordinate_descent_polish(a, n_sweeps=50)
            tracker.update(a)
        tracker.report(f"Final: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
