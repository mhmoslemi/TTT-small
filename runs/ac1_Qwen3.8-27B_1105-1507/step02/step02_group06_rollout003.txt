"""
Focused LP-based optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses sparse LP with backtracking line search, multiple length exploration,

control, precomputed matrix structure, and log-sum-exp gradients for polish.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_score(f):
    """Compute 2n*max(conv)/sum^2 directly."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))


def convolve_with_f(f, g):
    """Compute convolve(f, g) efficiently."""
    return np.convolve(f, g)


def build_sparse_structure(n):
    """Precompute row/col indices for the convolution matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def make_conv_matrix(f, row_idx, col_idx):
    """Build the sparse convolution matrix with current f values."""
    n = len(f)
    # A[k, j] = f[k-j] for valid (k,j) pairs
    vals = f[row_idx - col_idx].astype(np.float64)
    return csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))


def solve_lp_fast(f, M, row_idx, col_idx, time_limit=20.0):
    """Solve max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    A = make_conv_matrix(f, row_idx, col_idx)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs',
                         options={'presolve': True, 'time_limit': time_limit})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def backtrack_line_search(f, g, S, current_score, max_alpha=1.0):
    """Find best alpha in [0, max_alpha] such that score decreases."""
    best_alpha = 0.0
    best_score = current_score
    # Try a set of candidate alphas
    alphas = [1.0, 0.75, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.025, 0.01]
    alphas = [a * max_alpha for a in alphas if a * max_alpha <= max_alpha]
    
    for alpha in alphas:
        if alpha <= 0:
            continue
        candidate = (1.0 - alpha) * f + alpha * g
        candidate = np.clip(candidate, 0.0, None)
        cand_sum = float(np.sum(candidate))
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (S / cand_sum)
        sc = compute_score(candidate)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_alpha = alpha
            best_candidate = candidate
            # Don't break - try smaller alphas too for potentially better
    return best_alpha, best_score, locals().get('best_candidate', None)

def logsumexp_gradient_step(f, t, lr):
    """One step of gradient descent on log-sum-exp approximation."""
    n = len(f)
    conv = np.convolve(f, f)
    S = float(np.sum(f))

    # Softmax weights
    conv_shifted = conv / t - np.max(conv) / t
    exp_vals = np.exp(conv_shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        return f, compute_score(f)
    softmax = exp_vals / total

    # Gradient: d/d(f[i]) log(sum exp(conv/t)) = (2/t) * sum_k softmax[k] * f[k-i]
    # Using cross-correlation
    # cross_corr[i] = sum_j f[j] * softmax[j+i]
    cross = np.correlate(softmax, f, mode='full')[n - 1:2 * n - 1]
    grad = (2.0 / t) * cross - 2.0 / S

    # Gradient step
    f_new = f - lr * grad
    f_new = np.clip(f_new, 0.0, None)
    new_sum = float(np.sum(f_new))
    if new_sum > 1e-10:
        f_new = f_new * (S / new_sum)

    return f_new, compute_score(f_new)

    f = f.copy().astype(np.float64)
    best_f = f.copy()

    S = float(np.sum(f))

            f = f_new

            no_improve = 0

        else:

def subgradient_optimize(f, deadline, lr_init=0.005, top_k=3, verbose_prefix=""):
    """Fast subgradient descent with momentum."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    lr = lr_init
    momentum = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        if M <= 1e-15:
            break

        # Get top-k lags
        top_indices = np.argsort(conv)[-top_k:]
        mean_val = (S * S) / (2 * n - 1)
        weights = np.maximum(conv[top_indices] - mean_val, 0)
        w_sum = np.sum(weights)
        if w_sum < 1e-15:
            weights = np.ones(top_k) / top_k
        else:
            weights = weights / w_sum

        # Compute gradient
        grad = np.zeros(n)
        for k, w in zip(top_indices, weights):
            if w <= 0:
                continue
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                grad[j] += 2.0 * w * f[k - j]

        # Subtract mean gradient (to preserve sum)
        grad -= np.mean(grad)

        # Momentum update
        momentum = 0.85 * momentum - lr * grad
        f_new = f + momentum
        f_new = np.clip(f_new, 0.0, None)
        new_sum = float(np.sum(f_new))
        if new_sum > 1e-10:
            f_new = f_new * (S / new_sum)

        sc = compute_score(f_new)
        it += 1

        if np.isfinite(sc) and sc < best_score:
            f = f_new
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            if it % 1000 == 0:
                print(f"  {verbose_prefix}subgrad iter {it}: score={best_score:.7f}")
        else:
            no_improve += 1
            lr *= 0.95
            momentum *= 0.5
            if lr < 1e-8 or no_improve > 300:
                break

    return best_f, best_score


def make_initializations(n, rng):
    """Create diverse initializations."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    inits.append(('uniform', np.ones(n)))

    # Gaussian

    cos_t = (1.0 + np.cos(2.0 * np.pi * idx / n)) / 2.0
    inits.append(('cosine', np.maximum(cos_t, 0.01)))

    # Random
    inits.append(('random', rng.uniform(0.3, 1.5, size=n)))

    # Bimodal

                np.exp(-0.5 * ((idx - n * 0.75) / (n / 10.0)) ** 2))

    # S-curve
    s_curve = 1.0 / (1.0 + np.exp(-6.0 * (idx / (n - 1) - 0.5)))
    inits.append(('sigmoid', s_curve))

        f_pol, s_pol, _, _ = lp_optimize(best_sequence, polish_deadline,
                                          row_idx, col_idx,
                                          step_init=0.1,
                                          verbose_prefix="[polish-LP] ")
        if s_pol < best_score:
            best_score = s_pol
            best_sequence = f_pol.copy()
            print(f"  LP polish improved to: {best_score:.7f}")

        # Log-sum-exp gradient polish
        lse_deadline = time.monotonic() + 50.0
        f_pol, s_pol = logsumexp_optimize(best_sequence, lse_deadline,
                                           t_init=0.005, lr_init=0.0005,
                                           verbose_prefix="[polish-lse] ")
        if s_pol < best_score:
            best_score = s_pol
            best_sequence = f_pol.copy()
            print(f"  LSE polish improved to: {best_score:.7f}")

        # Subgradient final polish
        sub_deadline = time.monotonic() + 40.0
        f_pol, s_pol = subgradient_optimize(best_sequence, sub_deadline,
                                             lr_init=0.001, top_k=5,
                                             verbose_prefix="[polish-sub] ")
        if s_pol < best_score:
            best_score = s_pol
            best_sequence = f_pol.copy()
            print(f"  Subgrad polish improved to: {best_score:.7f}")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(1024)
        best_score = evaluate_sequence([float(x) for x in best_sequence])

    final_score = evaluate_sequence([float(x) for x in best_sequence])
    if not np.isfinite(final_score):
        print("WARNING: Final sequence failed! Using fallback.")
        best_sequence = np.ones(512)
        final_score = evaluate_sequence([float(x) for x in best_sequence])

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.7f}, "
          f"internal={best_score:.7f}, time={elapsed:.1f}s")

    return [float(x) for x in best_sequence]
