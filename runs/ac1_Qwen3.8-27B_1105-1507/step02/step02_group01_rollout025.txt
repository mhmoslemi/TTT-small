import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    """Compute 2*n*max(conv)/(sum(a))^2"""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def normalize(a):
    """Normalize so sum = 1, clip to non-negative."""
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s

def build_A(a):
    """Build the constraint matrix for the LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        # A[k, j] = a[k - j] for j in [j_lo, j_hi]
        # When j = j_lo: k - j_hi to k - j_lo
        A[k, j_lo:j_hi+1] = a[k - j_hi:k - j_lo + 1][::-1]
    return A

def solve_lp(a):
    """Solve the LP: max sum(b) s.t. (a*b)_k <= M, b >= 0.
    Returns (optimal_b, optimal_sum, M, dual_y)."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status != 0:
        return None, None, M, None
    opt_b = res.x
    opt_sum = float(-res.fun)
    # Get dual variables (shadow prices)
    dual_y = res.ineqlin.marginals if res.ineqlin is not None else None
    return opt_b, opt_sum, M, dual_y

"""
Optimized step function coefficient search using LP-based alternating
optimization with dual-guided gradient descent, population-based search,
and coordinate descent refinement. Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        A[k, j_lo:j_hi+1] = a[k - j_hi:k - j_lo + 1][::-1]
    return A


def lp_solve(a):
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status != 0:
        return None, M
    return res.x, M


def lp_phase(a, max_time, t_start=0.5, t_min=0.05):
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    it = 0
    max_iters = 10000

    while it < max_iters and (time.monotonic() - start) < max_time:
        t0 = time.monotonic()
        g, M = lp_solve(a)
        if g is None:
            break
        
        g_sum = float(np.sum(g))
        if g_sum <= 1.0 + 1e-6:
            stall_count += 1
            if stall_count > 10:
                break
            # Try with slightly relaxed bound
            continue
        stall_count = 0
        
        g = g / g_sum
        # Line search
        t = max(t_min, t_start * (0.95 ** it))
        best_t = None
        for tt in [t, t*0.5, t*1.5, t*0.25]:
            a_new = (1.0 - tt) * a + tt * g
            a_new = normalize(a_new)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                best_t = tt
        if best_t is not None:
            a = best_a.copy()
        else:
            a = normalize((1.0 - t) * a + t * g)
        it += 1
    
    return best_a, best_obj


def softmax_grad(a, T):
    n = len(a)
    conv = np.convolve(a, a)
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    # Gradient: g_i = (2/T) * sum_m p[i+m] * a[m]
    # This is the cross-correlation of p and a
    g = (2.0 / T) * np.correlate(p, a, mode='valid')
    return g


def gd_phase(a, max_time, T_init=None):
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    
    if T_init is None:
        conv = np.convolve(a, a)
        T_init = 0.05 * float(np.max(conv))
    
    T = T_init
    lr = 0.02
    last_improve = 0
    it = 0
    max_iters = 50000
    
    while it < max_iters and (time.monotonic() - start) < max_time:
        g = softmax_grad(a, T)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        
        a_new = a - lr * g / g_norm
        a_new = normalize(a_new)
        obj = compute_obj(a_new)
        
        if obj < best_obj - 1e-10:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            last_improve = it
            T = max(0.0001, T * 0.99)
        else:
            lr *= 0.9
            if lr < 1e-12:
                break
            if it - last_improve > 100:
                T = max(0.0001, T * 0.7)
                lr = 0.01
                last_improve = it
        it += 1
    
    return best_a, best_obj


def coord_descent(a, rng, max_time):
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    
    while (time.monotonic() - start) < max_time:
        # Pick a random coordinate
        i = int(rng.integers(0, n))
        x0 = best_a[i]
        if x0 < 1e-12:
            # Try small positive values
            candidates = [x0 + 0.01, x0 + 0.005, x0 + 0.001]
        else:
            # Try multiplicative perturbations
            candidates = [x0 * f for f in [0.1, 0.3, 0.5, 0.7, 1.3, 1.5, 2.0]]
            candidates.append(x0 + rng.normal(0, x0 * 0.2))
        
        best_local = best_obj
        best_val = x0
        for x_val in candidates:
            x_val = max(0.0, x_val)
            trial = best_a.copy()
            trial[i] = x_val
            trial = normalize(trial)
            obj = compute_obj(trial)
            if obj < best_local:
                best_local = obj
                best_val = x_val
        
        if best_local < best_obj:
            best_a[i] = best_val
            best_a = normalize(best_a)
            best_obj = best_local
        it += 1
    
    return best_a, best_obj


def block_refine(a, rng, max_time):
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    max_iters = 50000
    
    while it < max_iters and (time.monotonic() - start) < max_time:
        width = max(1, int(n * rng.uniform(0.01, 0.1)))
        pos = int(rng.integers(0, n - width + 1))
        candidate = best_a.copy()
        sigma = 0.02 + 0.06 * max(0.0, 1.0 - it / max_iters)
        candidate[pos:pos+width] *= np.exp(rng.normal(0, sigma, width))
        candidate = normalize(candidate)
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj
        it += 1
    
    return best_a, best_obj


def random_init(n, rng):
    choice = int(rng.integers(0, 8))
    if choice == 0:
        # Uniform
        return np.ones(n) / n
    elif choice == 1:
        # Gaussian center
        center = n / 2.0
        width = n * rng.uniform(0.2, 0.5)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        return normalize(a)
    elif choice == 2:
        # Edge-concentrated
        a = np.exp(-np.abs(np.arange(n) - n/2.0) / (n * 0.3))
        a = a ** rng.uniform(1, 5)
        return normalize(a)
    elif choice == 3:
        # Binary-like (random subset)
        a = (rng.random(n) > 0.5).astype(float)
        if np.sum(a) == 0:
            a[rng.integers(0, n)] = 1.0
        return normalize(a)
    elif choice == 4:
        # Chirp-like
        phase = rng.uniform(0, 2*np.pi)
        freq = rng.uniform(0.1, 0.5)
        a = np.abs(np.exp(2j * np.pi * freq * np.arange(n) + phase))
        a = np.ones(n) * (0.5 + 0.5 * np.cos(2*np.pi*freq*np.arange(n) + phase))
        a += 0.1
        return normalize(a)
    elif choice == 5:
        # Sparse random
        k = int(n * rng.uniform(0.2, 0.5))
        a = np.zeros(n)
        idx = rng.choice(n, k, replace=False)
        a[idx] = rng.random(k) + 0.1
        return normalize(a)
    elif choice == 6:
        # Periodic
        period = int(rng.integers(2, max(3, n // 10)))
        pattern = np.maximum(0, rng.random(period) - 0.1)
        a = np.tile(pattern, n // period + 1)[:n]
        a += 0.05
        return normalize(a)
    else:
        # Pure random
        a = np.abs(rng.normal(0.5, 0.4, n))
        return normalize(a)

def pop_search(n, rng, time_budget):
    """Population-based search for a given n."""
    K = 8  # population size
    pop = [random_init(n, rng) for _ in range(K)]
    scores = [compute_obj(p) for p in pop]
    best_a = pop[int(np.argmin(scores))].copy()
    best_score = min(scores)
    start = time.monotonic()
    gen = 0
    max_gens = 200
    
    while gen < max_gens and (time.monotonic() - start) < time_budget:
        # Optimize each member
        per_member_time = (time_budget - (time.monotonic() - start)) / (K + 2)
        
        for i in range(K):
            if time.monotonic() - start > time_budget:
                break
            t_rem = time_budget - (time.monotonic() - start)
            t_alloc = min(per_member_time, t_rem * 0.3)
            if t_alloc < 2.0:
                break
            pop[i], scores[i] = full_optimize(pop[i], t_alloc, rng)
            if scores[i] < best_score:
                best_score = scores[i]
                best_a = pop[i].copy()
        
        # Create new members
        if time.monotonic() - start > time_budget:
            break
        
        # Sort by score
        order = np.argsort(scores)
        
        # Replace worst half
        for i in range(K // 2):
            worst_idx = order[K - 1 - i]
            # Crossover + mutation
            p1 = pop[order[0]].copy()
            p2 = pop[order[1 % K]].copy()
            if rng.random() < 0.5:
                new = normalize(0.5 * p1 + 0.5 * p2)
            else:
                new = pop[order[rng.integers(0, K // 2)]].copy()
                # Mutation
                width = max(1, n // 10)
                pos = int(rng.integers(0, n - width + 1))
                new[pos:pos+width] *= np.exp(rng.normal(0, 0.3, width))
                new = normalize(new)
            pop[worst_idx] = new
            scores[worst_idx] = compute_obj(new)
        
        gen += 1
    
    return best_a, best_score

"""
Optimized step function coefficient search using LP-based alternating optimization
with dual-guided gradient descent, population-based evolutionary search, adaptive
temperature scheduling, and multi-stage refinement. Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        A[k, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
    return A


def lp_solve(a, M_scale=1.0):
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv)) * M_scale
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status != 0:
        return None, float(np.max(conv))
    return res.x, float(np.max(conv))


def lp_phase(a, max_time, t_start=0.5, t_min=0.05):
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall_count = 0
    it = 0
    max_iters = 20000

    while it < max_iters and (time.monotonic() - start) < max_time:
        g, M = lp_solve(a)
        if g is None:
            break
        g_sum = float(np.sum(g))
        if g_sum <= 1.0 + 1e-5:
            stall_count += 1
            if stall_count > 15:
                break
            continue
        stall_count = 0
        g = g / g_sum
        t = max(t_min, t_start * (0.97 ** it))
        best_t = None
        best_t_obj = best_obj
        for tt in [t, t * 0.5, t * 1.5, t * 0.25, t * 2.0]:
            if tt > 1.0:
                tt = 1.0
            if tt < 0.001:
                continue
            a_new = (1.0 - tt) * a + tt * g
            a_new = normalize(a_new)
            obj = compute_obj(a_new)
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = tt
        if best_t is not None:
            a = normalize((1.0 - best_t) * a + best_t * g)
            best_a = a.copy()
            best_obj = best_t_obj
        else:
            a = normalize((1.0 - t) * a + t * g)
        it += 1

    return best_a, best_obj


def softmax_grad(a, T):
    n = len(a)
    conv = np.convolve(a, a)
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    if sum_exp < 1e-300:
        p = np.ones(len(conv)) / len(conv)
    else:
        p = exp_vals / sum_exp
    g = (2.0 / T) * np.correlate(p, a, mode='valid')
    return g


def gd_phase(a, max_time, T_init=None):
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    if T_init is None:
        conv = np.convolve(a, a)
        T_init = 0.05 * float(np.max(conv))

    T = T_init
    lr = 0.03
    last_improve = 0
    it = 0
    max_iters = 100000

    while it < max_iters and (time.monotonic() - start) < max_time:
        g = softmax_grad(a, T)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        a_new = a - lr * g / g_norm
        a_new = normalize(a_new)
        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            last_improve = it
            T = max(1e-5, T * 0.995)
        else:
            lr *= 0.92
            if lr < 1e-14:
                break
            if it - last_improve > 150:
                T = max(1e-5, T * 0.6)
                lr = 0.02
                last_improve = it
        it += 1

    return best_a, best_obj


def coord_descent(a, rng, max_time):
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    max_iters = 200000

    while it < max_iters and (time.monotonic() - start) < max_time:
        i = int(rng.integers(0, n))
        x0 = best_a[i]
        if x0 < 1e-10:
            candidates = [0.005, 0.01, 0.02, 0.001]
        else:
            candidates = [x0 * f for f in [0.1, 0.3, 0.5, 0.7, 1.4, 1.7, 2.5, 0.01]]
            candidates.append(max(0.0, x0 + rng.normal(0, x0 * 0.3)))

        best_local = best_obj
        best_val = x0
        for x_val in candidates:
            x_val = max(0.0, x_val)
            trial = best_a.copy()
            trial[i] = x_val
            trial = normalize(trial)
            obj = compute_obj(trial)
            if obj < best_local:
                best_local = obj
                best_val = x_val

        if best_local < best_obj - 1e-12:
            best_a[i] = best_val
            best_a = normalize(best_a)
            best_obj = best_local
        it += 1

    return best_a, best_obj


def block_refine(a, rng, max_time):
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    it = 0
    max_iters = 100000

    while it < max_iters and (time.monotonic() - start) < max_time:
        width = max(1, int(n * rng.uniform(0.02, 0.15)))
        pos = int(rng.integers(0, max(1, n - width + 1)))
        candidate = best_a.copy()
        sigma = 0.015 + 0.04 * max(0.0, 1.0 - it / max_iters)
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = normalize(candidate)
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj
        it += 1

    return best_a, best_obj


def random_init(n, rng):
    choice = int(rng.integers(0, 10))
    if choice == 0:
        return np.ones(n) / n
    elif choice == 1:
        center = n / 2.0
        width = n * rng.uniform(0.15, 0.5)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        return normalize(a)
    elif choice == 2:
        a = np.ones(n)
        a[0] = 3.0
        a[-1] = 3.0
        if n > 2:
            a[1] = 1.5
            a[-2] = 1.5
        return normalize(a)
    elif choice == 3:
        k = int(n * rng.uniform(0.2, 0.5))
        a = np.zeros(n)
        idx = rng.choice(n, k, replace=False)
        a[idx] = rng.random(k) + 0.05
        return normalize(a)
    elif choice == 4:
        freq = rng.uniform(0.05, 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * np.arange(n) + phase)
        a += 0.1
        return normalize(a)
    elif choice == 5:
        period = int(rng.integers(2, max(3, n // 8)))
        pattern = np.maximum(0, rng.random(period) - 0.15)
        a = np.tile(pattern, n // period + 1)[:n]
        a += 0.08
        return normalize(a)
    elif choice == 6:
        a = np.abs(rng.normal(0.3, 0.4, n))
        return normalize(a)
    elif choice == 7:
        a = np.zeros(n)
        a[0:n // 2] = 1.0
        a[n // 2:] = rng.uniform(0.1, 0.5, n - n // 2)
        return normalize(a)
    elif choice == 8:
        decay = rng.uniform(0.02, 0.15)
        a = np.exp(-decay * np.abs(np.arange(n) - n // 2))
        a += 0.05
        return normalize(a)
    else:
        a = rng.random(n) ** rng.uniform(0.5, 3.0)
        return normalize(a)


def full_optimize(a, time_budget, rng):
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: LP
    lp_time = min(time_budget * 0.35, 40.0)
    if time_budget > 5:
        a, obj = lp_phase(a, lp_time)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj

    # Phase 2: GD
    if time.monotonic() - start < time_budget * 0.65:
        gd_time = min(time_budget * 0.3, 25.0)
        a, obj = gd_phase(a, gd_time)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj

    # Phase 3: Second LP
    if time.monotonic() - start < time_budget * 0.82:
        lp_time2 = min(time_budget * 0.12, 12.0)
        a, obj = lp_phase(a, lp_time2, t_start=0.3, t_min=0.02)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj

    # Phase 4: Fine GD
    if time.monotonic() - start < time_budget * 0.95:
        gd_time2 = min(time_budget * 0.1, 10.0)
        conv_max = float(np.max(np.convolve(a, a)))
        a, obj = gd_phase(a, gd_time2, T_init=0.003 * conv_max)
        if obj < best_obj:
            best_a = a.copy()
            best_obj = obj

def pop_search(n, rng, time_budget):
    K = 6
    pop = [random_init(n, rng) for _ in range(K)]
    scores = [compute_obj(p) for p in pop]
    best_a = pop[int(np.argmin(scores))].copy()
    best_score = min(scores)
    start = time.monotonic()
    gen = 0
    max_gens = 100

    while gen < max_gens and (time.monotonic() - start) < time_budget:
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 10:
            break
        per_member = remaining / (K + 3)

        for i in range(K):
            if time.monotonic() - start > time_budget:
                break
            t_rem = time_budget - (time.monotonic() - start)
            t_alloc = min(per_member, t_rem * 0.4)
            if t_alloc < 3.0:
                break
            pop[i], scores[i] = full_optimize(pop[i], t_alloc, rng)
            if scores[i] < best_score:
                best_score = scores[i]
                best_a = pop[i].copy()

        if time.monotonic() - start > time_budget:
            break

        order = np.argsort(scores)
        for i in range(K // 3):
            worst_idx = order[K - 1 - i]
            r = rng.random()
            if r < 0.4:
                p1 = pop[order[0]].copy()
                p2 = pop[order[1 % K]].copy()
                alpha = rng.uniform(0.3, 0.7)
                new = normalize(alpha * p1 + (1 - alpha) * p2)
            elif r < 0.7:
                src_idx = order[rng.integers(0, max(1, K // 2))]
                new = pop[src_idx].copy()
                width = max(2, n // 8)
                pos = int(rng.integers(0, max(1, n - width + 1)))
                new[pos:pos + width] *= np.exp(rng.normal(0, 0.4, width))
                new = normalize(new)
            else:
                new = random_init(n, rng)
            pop[worst_idx] = new
            scores[worst_idx] = compute_obj(new)
        gen += 1

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-stage optimization combining LP-based alternating, log-sum-exp GD,
    population-based search, and intensive refinement. Targets < 1.5027.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 6.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    n_values = [50, 80, 100, 150, 200, 250, 300, 350, 400, 500, 600]

    # Phase 1: Quick scan
    print("\n=== Phase 1: Quick scan ===")
    for n in n_values:
        if time.monotonic() > deadline - 180:
            break
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        num_scans = max(2, min(3, int(remaining / len(n_values) / 10)))
        for s in range(num_scans):
            if time.monotonic() > deadline - 180:
                break
            a_init = random_init(n, rng)
            t_alloc = min(12.0, max(3.0, (deadline - time.monotonic() - 180) / max(1, num_scans - s)))
            a_opt, obj_opt = full_optimize(a_init, t_alloc, rng)
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} s={s}: obj={obj_opt:.6f} ** NEW BEST **")

    best_n = len(best_seq) if best_seq is not None else 300
    print(f"\nAfter Phase 1: best_n={best_n}, best_score={best_score:.6f}")

    # Phase 2: Focused population search
    print("\n=== Phase 2: Focused search ===")
    focused_ns = sorted(set([best_n, max(50, best_n - 100), min(600, best_n + 100), max(30, best_n // 2)]))
    for n in focused_ns:
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        t_alloc = min(remaining * 0.35, 250.0)
        a_opt, obj_opt = pop_search(n, rng, t_alloc)
        if obj_opt < best_score:
            best_score = obj_opt
            best_seq = a_opt.copy()
            best_n = n
            print(f"  n={n}: obj={obj_opt:.6f} ** NEW BEST **")
        else:
            print(f"  n={n}: obj={obj_opt:.6f}")

    print(f"\nAfter Phase 2: best_n={best_n}, best_score={best_score:.6f}")

    # Phase 3: Intensive refinement
    print("\n=== Phase 3: Intensive refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        num_rounds = max(3, int(remaining / 20))
        num_rounds = min(num_rounds, 10)

        for round_idx in range(num_rounds):
            if time.monotonic() > deadline - 5:
                break
            t_rem = deadline - time.monotonic() - 4
            t_per = t_rem / max(1, num_rounds - round_idx)

            # LP
            if time.monotonic() > deadline - 5:
                break
            a_ref, obj_ref = lp_phase(best_seq.copy(), t_per * 0.25, t_start=0.15, t_min=0.01)
            if obj_ref < best_score:
                best_score = obj_ref
                best_seq = a_ref.copy()
                print(f"  R{round_idx} LP: obj={best_score:.6f}")

            # GD
            if time.monotonic() > deadline - 5:
                break
            conv_max = float(np.max(np.convolve(best_seq, best_seq)))
            a_gd, obj_gd = gd_phase(best_seq.copy(), t_per * 0.25, T_init=0.002 * conv_max)
            if obj_gd < best_score:
                best_score = obj_gd
                best_seq = a_gd.copy()
                print(f"  R{round_idx} GD: obj={best_score:.6f}")

            # Block refine
            if time.monotonic() > deadline - 5:
                break
            a_bp, obj_bp = block_refine(best_seq.copy(), rng, t_per * 0.3)
            if obj_bp < best_score:
                best_score = obj_bp
                best_seq = a_bp.copy()
                print(f"  R{round_idx} BP: obj={best_score:.6f}")

            # Coord descent
            if time.monotonic() > deadline - 5:
                break
            a_cd, obj_cd = coord_descent(best_seq.copy(), rng, t_per * 0.2)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  R{round_idx} CD: obj={best_score:.6f}")

    # Final verify
    if best_seq is not None:
        final_obj = compute_obj(best_seq)
        eval_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"\nFinal: n={len(best_seq)}, compute_obj={final_obj:.6f}, evaluate={eval_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(200) / 200

    elapsed = time.monotonic() - start_time
    print(f"DONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
