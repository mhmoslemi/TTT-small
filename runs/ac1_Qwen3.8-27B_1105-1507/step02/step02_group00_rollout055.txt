```python
import numpy as np
from scipy.optimize import linprog
import time
import cvxpy as cp

def compute_obj(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k-n+1)
        j_end = min(n-1, k)
        A[k, j_start:j_end+1] = a[k-j_start:k-j_end+1][::-1]
    return A

def solve_lp_g(a, M):
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_alternate(n, a_init, max_iter=200):
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        g = solve_lp_g(a, M)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Try different blend factors
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_new = (1-t)*a + t*g
            a_new = np.maximum(a_new, 0)
            s = np.sum(a_new)
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
        
        # Use the best blend for the next iteration
        # Actually, let's just use t=0.5 for simplicity
        a_new = 0.5*a + 0.5*g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            a = g
        else:
            a = a_new / s
    
    return best_a, best_obj

def solve_cvxpy(n, timeout=60):
    """Solve the QCQP directly using cvxpy."""
    a = cp.Variable(n)
    t = cp.Variable()
    constraints = [cp.sum(a) == 1, a >= 0]
    
    for k in range(2*n-1):
        if k == 0:
            constraints.append(a[0]**2 <= t)
        else:
            # conv(a,a)[k] = sum_{j} a[j]*a[k-j]
            # In cvxpy, this is a sum of products of linear terms
            expr = 0
            for j in range(max(0, k-n+1), min(n-1, k)+1):
                expr += a[j] * a[k-j]
            constraints.append(expr <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    # Try different solvers
    for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                return a.value
        except Exception:
            continue
    return None

def coord_descent(a, n_rounds=10):
    n = len(a)
    a = a.copy()
    best_obj = compute_obj(a)
    
    for rnd in range(n_rounds):
        improved = False
        for j in range(n):
            old = a[j]
            # Try multiple values
            candidates = [0.0, 0.1*old, 0.3*old, 0.5*old, 0.7*old, old, 1.3*old, 1.5*old, 2.0*old, 3.0*old]
            best_x = old
            best_local = compute_obj(a)  # current value
            for x in candidates:
                a[j] = x
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_x = x
            a[j] = best_x
            if best_x != old:
                improved = True
        
        if compute_obj(a) < best_obj:
            best_obj = compute_obj(a)
    
    return a, best_obj
```

```python
    a = a.copy()
    conv = np.convolve(a, a)

    sum_a = float(np.sum(a))
    best_obj = 2.0 * n * max_b / (sum_a * sum_a)

            old = a[j]
            conv_before = conv.copy()  # O(2n)

            if old > 0:
                candidates.append(0.0)
            best_x = old
            best_local = best_obj
            
            for x in candidates:
                delta = x - old
                # Update conv
                conv_new = conv_before.copy()
                for k in range(j, min(j + n, 2*n - 1)):

                    if 0 <= other < n:
                        if k == 2 * j:
                            conv_new[k] += delta * (2 * old + delta)
```

```python
def obj_and_grad(a):
    """Compute max(conv(a,a)) and its subgradient."""
    n = len(a)
    r = np.convolve(a, a)
    max_r = float(np.max(r))
    # Active set
    tol = max_r * 0.01
    active = np.where(r >= max_r - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        # grad r[k] / grad a[m] = 2*a[k-m] if 0<=k-m<n
        for m in range(n):
            other = k - m
            if 0 <= other < n:
                grad[m] += 2 * a[other]
    grad /= len(active)
    
    return max_r, grad
```

"""
Multi-method optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternation, scipy-based optimization, and fast coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0 or n is None:
        return np.inf
    a = np.asarray(a, dtype=np.float64)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix for LP: A[k,j] such that (A @ g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # A[k, j] = a[k-j] for j in [j_start, j_end]
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def build_A_fast(a):
    """Faster construction using vectorization."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_g(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_A_fast(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None
