6. Modify F to reduce the peak: e.g., reduce the magnitude of F at frequencies that contribute most to the peak
7. Update f = IFFT(F), project onto non-negative, normalize
8. Repeat

    f = f / S

        G = F ** 2
        g = np.fft.ifft(G).real  # This is conv(f,f) padded to length n
        # Actually, this isn't quite right for the full convolution
        # The full convolution has length 2n-1
        # Let me use a different approach
        
        # Alternative: just work in time domain
        c = np.convolve(f, f)
        peak_val = np.max(c)
        peak_idx = np.argmax(c)
        
        # Compute gradient of peak with respect to f
        # d c[peak_idx] / d f[j] = 2 * f[peak_idx - j] (with bounds)
        grad = np.zeros(n)
        for j in range(n):
            k = peak_idx - j
            if 0 <= k < n:
                grad[j] = 2 * f[k]
        
        # Negative gradient direction (to decrease the peak)
        # But we need to maintain sum = 1, so we need to add back
        # Project: new_f = f - alpha * (grad - mean(grad) * (grad·(1-0)/(n-0)))
        # Simpler: new_f = f - alpha * grad + alpha * sum(grad)/n
        new_f = f - 0.01 * (grad - np.mean(grad))
        new_f = np.clip(new_f, 0, None)
        S = np.sum(new_f)
        if S > 0:
            new_f = new_f / S
        
        if np.max(np.convolve(new_f, new_f)) < peak_val:
            f = new_f
    
    return f

def build_A_sparse(f):
    """Build sparse matrix for conv(f, g) as matrix-vector product."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((np.array(vals, dtype=np.float64),
                       (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))),
                      shape=(2 * n - 1, n))


def solve_lp_dir(f, M, A=None):
    """Solve LP: max sum(g) s.t. A@ <= M, g >= 0."""
    n = len(f)
    if A is None:
        A = build_A_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

            vals.append(f[k - j])
    return csr_matrix((np.array(vals, dtype=np.float64),
                       (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))),
                      shape=(2 * n - 1, n))


def solve_lp_dir(f, M, A=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A is None:
        A = build_A_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fast_score(f):
    """Fast objective without safety checks."""
    n = len(f)
    c = np.convolve(f, f)
    max_c = np.max(c)
    s = np.sum(f)
    if s < 1e-10:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


def nesterov_optimize(f_init, deadline, beta_start=10.0, beta_end=500.0,
                      lr_start=0.06, momentum_init=0.7, verbose=""):
    """Nesterov accelerated gradient on soft-max with LP injection."""
    f = f_init.copy().astype(np.float64)
    f = np.clip(f, 0, None)
    S = np.sum(f)
    if S < 1e-10:
        return f, float('inf')
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = fast_score(f)

    f_prev = f.copy()
    momentum = momentum_init
    lr = lr_start
    no_improve = 0
    it = 0
    lp_interval = 80

    while time.monotonic() < deadline and it < 80000:
        frac = min(1.0, it / 5000.0)
        beta = beta_start * (beta_end / beta_start) ** frac

        # Nesterov lookahead
        y = f + momentum * (f - f_prev)
        y = np.clip(y, 0, None)
        Sy = np.sum(y)
        if Sy > 1e-10:
            y = y / Sy

        # Gradient via soft-max
        c = np.convolve(y, y)
        c_max = np.max(c)
        exp_c = np.exp(beta * (c - c_max))
        total = np.sum(exp_c)
        if total < 1e-300:
            break
        p = exp_c / total
        grad = 2.0 * np.correlate(p, y, mode='valid')

        # Project onto sum=1
        grad = grad - np.mean(grad)

        # Backtracking line search (use fast_score)
        it += 1
        accepted = False
        for _ls in range(8):
            f_new = f - lr * grad
            f_new = np.clip(f_new, 0, None)
            S_new = np.sum(f_new)
            if S_new < 1e-10:
                lr *= 0.5
                continue
            f_new = f_new / S_new
            sc = fast_score(f_new)
            if np.isfinite(sc) and sc < best_score:
                accepted = True
                break
            lr *= 0.5

        if accepted:
            f_prev = f.copy()
            f = f_new
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            if it % 100 == 0:
                lr = min(lr * 1.05, 0.12)
            momentum = min(0.93, momentum + 0.0003)
            if it % 500 == 0:
                print(f"  {verbose}iter {it}: {best_score:.6f} (beta={beta:.1f}, lr={lr:.4f})")
        else:
            no_improve += 1
            lr *= 0.7
            if no_improve > 80:
                lr = max(lr_start * 0.3, lr * 0.5)
                no_improve = 0
                if lr < 5e-5:
                    break

        # Periodic LP injection
        if it > 0 and it % lp_interval == 0:
            c = np.convolve(f, f)
            M = np.max(c)
            if M > 0:
                g = solve_lp_dir(f, M)
                if g is not None:
                    Sg = np.sum(g)
                    if Sg > 1e-10:
                        g = g / Sg
                        for t in [0.15, 0.25, 0.35]:
                            fb = (1 - t) * f + t * g
                            fb = np.clip(fb, 0, None)
                            Sb = np.sum(fb)
                            if Sb < 1e-10:
                                continue
                            fb = fb / Sb
                            sb = fast_score(fb)
                            if np.isfinite(sb) and sb < best_score:
                                f = fb
                                best_f = fb.copy()
                                best_score = sb
                                f_prev = fb.copy()
                                print(f"  {verbose}LP@{it}: {best_score:.6f}")
                                break

    return best_f, best_score


def gen_init(n, kind='tukey', r=0.3, seed=0):
    """Generate diverse initial sequences."""
    if kind == 'uniform':
        return np.ones(n)
    elif kind == 'tukey':
        w = np.ones(n)
        m = max(1, int(r * n / 2))
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
        return w
    elif kind == 'hann':
        return 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / max(n - 1, 1)))
    elif kind == 'raised_cos':
        return 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    elif kind == 'rectified_cos':
        return np.maximum(0.0, np.cos(2 * np.pi * np.arange(n) / n)) + 0.02
    elif kind == 'exp_center':
        x = np.arange(n) - n / 2
        sigma = n / 4.0
        return np.exp(-x**2 / (2 * sigma**2))
    elif kind == 'ramp':
        return np.linspace(0.05, 1.0, n)
    elif kind == 'asym_ramp':
        return np.linspace(1.0, 0.05, n)
    elif kind == 'two_hump':
        x = np.arange(n, dtype=np.float64)
        c1, c2 = n * 0.3, n * 0.7
        sig = n * 0.12
        return np.exp(-(x - c1)**2 / (2 * sig**2)) + np.exp(-(x - c2)**2 / (2 * sig**2))
    elif kind == 'asym_two_hump':
        x = np.arange(n, dtype=np.float64)
        c1, c2 = n * 0.25, n * 0.65
        sig1, sig2 = n * 0.08, n * 0.15
        return 1.5 * np.exp(-(x - c1)**2 / (2 * sig1**2)) + np.exp(-(x - c2)**2 / (2 * sig2**2))
    elif kind == 'random_smooth':
        rng = np.random.default_rng(seed)
        x = rng.standard_normal(n)
        ksize = max(5, n // 15 | 1)
        kernel = np.exp(-0.5 * (np.arange(ksize) - ksize // 2) ** 2 / (ksize / 4.0) ** 2)
        kernel /= np.sum(kernel)
        x = np.convolve(x, kernel, mode='same')
        return np.abs(x) + 0.05
    elif kind == 'block':
        w = np.ones(n)
        mid = n // 2
        w[:mid] = 1.0
        w[mid:] = 0.3
        return w
    else:
        return np.ones(n)


def perturb(f, rng, strength=0.05):
    """Multiplicative perturbation."""
    noise = 1.0 + strength * rng.standard_normal(len(f))
    f_new = f * np.clip(noise, 0.1, 3.0)
    f_new = np.clip(f_new, 0, None)
    S = np.sum(f_new)
    if S < 1e-10:
        return f.copy()
    return f_new / S


def perturb_asymmetric(f, rng, strength=0.08):
    """Perturb first and second halves differently to encourage asymmetry."""
    n = len(f)
    f_new = f.copy()
    half = n // 2
    f_new[:half] *= (1.0 + strength * rng.standard_normal(half))
    f_new[half:] *= (1.0 - 0.5 * strength * rng.standard_normal(n - half))
    f_new = np.clip(f_new, 0, None)
    S = np.sum(f_new)
    if S < 1e-10:
        return f.copy()
    return f_new / S


def coordinate_refine(f, max_iters=2000, verbose=""):
    """Coordinate descent: try changing one element at a time."""
    f = f.copy().astype(np.float64)
    f = np.clip(f, 0, None)
    S = np.sum(f)
    f = f / S
    n = len(f)
    best_score = fast_score(f)
    best_f = f.copy()
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > time.monotonic() + 999999:
            break
        improved = False
        # Try random subset of coordinates for speed
        rng_local = np.random.default_rng(it)
        coords = rng_local.choice(n, size=min(n, 200), replace=False)
        for j in coords:
            old_val = f[j]
            if old_val < 1e-12:
                continue
            # Try: decrease, increase, zero out
            for frac in [0.5, 0.7, 0.9, 1.2, 1.5]:
                f[j] = old_val * frac
                f[j] = max(0.0, f[j])
                S_new = np.sum(f)
                if S_new < 1e-10:
                    f[j] = old_val
                    continue
                sc = fast_score(f)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_f = f.copy()
                    improved = True
                    break
            f[j] = old_val
        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            f = best_f.copy()
            no_improve = 0

    return best_f, best_score


def propose_candidate(budget_s=1500):
    t0 = time.monotonic()
    deadline = t0 + max(0, int(budget_s) - 20)

    rng = np.random.default_rng(777)
    best_score = float('inf')
    best_f = None

    # Load parent
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, None)
        S = np.sum(parent)
        if S > 0.01:
            score = evaluate_sequence(parent.tolist())
            if np.isfinite(score):
                best_score = score
                best_f = parent.copy() / S
                print(f"Parent n={len(parent)}: score={score:.6f}")
    except Exception as e:
        print(f"No parent available: {e}")

    if best_f is None:
        best_f = gen_init(512, 'tukey', r=0.3)
        best_f = best_f / np.sum(best_f)
        best_score = fast_score(best_f)
        print(f"Default init: score={best_score:.6f}")

    n_best = len(best_f)

    # === Phase 1: Aggressive Nesterov on parent (400s) ===
    if time.monotonic() < deadline - 300:
        print("\n=== Phase 1: Nesterov on parent ===")
        p1_deadline = min(deadline - 300, t0 + 380)
        f1, s1 = nesterov_optimize(best_f.copy(), p1_deadline,
                                    beta_start=10, beta_end=600,
                                    lr_start=0.05, momentum_init=0.6,
                                    verbose="P1")
        if s1 < best_score:
            best_score, best_f = s1, f1
            print(f"  Phase 1: {best_score:.6f}")

    # === Phase 2: Coordinate refinement (100s) ===
    if time.monotonic() < deadline - 250:
        print("\n=== Phase 2: Coordinate refinement ===")
        p2_deadline = min(deadline - 250, time.monotonic() + 100)
        f2, s2 = coordinate_refine(best_f, max_iters=5000)
        if s2 < best_score:
            best_score, best_f = s2, f2
            print(f"  Phase 2: {best_score:.6f}")

    # === Phase 3: Nesterov again with higher beta (200s) ===
    if time.monotonic() < deadline - 200:
        print("\n=== Phase 3: Nesterov high-beta ===")
        p3_deadline = min(deadline - 200, time.monotonic() + 200)
        f3, s3 = nesterov_optimize(best_f.copy(), p3_deadline,
                                    beta_start=100, beta_end=2000,
                                    lr_start=0.02, momentum_init=0.5,
                                    verbose="P3")
        if s3 < best_score:
            best_score, best_f = s3, f3
            print(f"  Phase 3: {best_score:.6f}")

    # === Phase 4: Diverse initializations at best length (250s) ===
    if time.monotonic() < deadline - 150:
        print(f"\n=== Phase 4: Diverse inits (n={n_best}) ===")
        inits = [
            ('tukey_015', gen_init(n_best, 'tukey', r=0.15)),
            ('tukey_05', gen_init(n_best, 'tukey', r=0.5)),
            ('hann', gen_init(n_best, 'hann')),
            ('raised_cos', gen_init(n_best, 'raised_cos')),
            ('rect_cos', gen_init(n_best, 'rectified_cos')),
            ('exp', gen_init(n_best, 'exp_center')),
            ('ramp', gen_init(n_best, 'ramp')),
            ('asym', gen_init(n_best, 'asym_ramp')),
            ('2hump', gen_init(n_best, 'two_hump')),
            ('asym2h', gen_init(n_best, 'asym_two_hump')),
            ('rand1', gen_init(n_best, 'random_smooth', seed=10)),
            ('rand2', gen_init(n_best, 'random_smooth', seed=20)),
            ('rand3', gen_init(n_best, 'random_smooth', seed=30)),
            ('block', gen_init(n_best, 'block')),
        ]
        # Add perturbed best
        inits.insert(0, ('best+p', perturb(best_f, rng, 0.03)))
        inits.insert(1, ('best+asym', perturb_asymmetric(best_f, rng, 0.06)))

        remaining = deadline - time.monotonic() - 150
        per_init = remaining / len(inits)

        for label, f_init in inits:
            if time.monotonic() > deadline - 150:
                break
            f_init = np.clip(f_init.astype(np.float64), 0, None)
            S = np.sum(f_init)
            if S < 0.01:
                continue
            f_init = f_init / S
            sc = fast_score(f_init)
            if not np.isfinite(sc):
                continue
            if sc < best_score:
                best_score, best_f = sc, f_init.copy()
                print(f"  [{label}] init: {best_score:.6f}")

            # Optimize if promising (within 8% of best)
            if sc < best_score * 1.08:
                idl = min(deadline - 150, time.monotonic() + per_init * 0.75)
                if time.monotonic() < idl - 3:
                    fo, so = nesterov_optimize(f_init, idl,
                                                beta_start=20, beta_end=400,
                                                lr_start=0.04, momentum_init=0.6,
                                                verbose=f"[{label}]")
                    if so < best_score:
                        best_score, best_f = so, fo
                        print(f"  [{label}] -> {best_score:.6f}")

    # === Phase 5: Try nearby lengths (150s) ===
    if time.monotonic() < deadline - 80:
        print(f"\n=== Phase 5: Length exploration ===")
        for n_try in [480, 544, 560, 576, 600, 640]:
            if time.monotonic() > deadline - 60:
                break
            f_try = gen_init(n_try, 'tukey', r=0.3)
            f_try = f_try / np.sum(f_try)
            sc = fast_score(f_try)
            if np.isfinite(sc) and sc < best_score * 1.03:
                ndl = min(deadline - 60, time.monotonic() + 50)
                fo, so = nesterov_optimize(f_try, ndl,
                                            beta_start=15, beta_end=400,
                                            lr_start=0.05, momentum_init=0.6,
                                            verbose=f"[n={n_try}]")
                if so < best_score:
                    best_score, best_f = so, fo
                    print(f"  n={n_try}: {best_score:.6f}")
            else:
                print(f"  n={n_try}: init={sc:.4f} (skip)")

    # === Phase 6: Perturbation restarts (200s) ===
    if best_f is not None and time.monotonic() < deadline - 40:
        print(f"\n=== Phase 6: Restarts (best={best_score:.6f}) ===")
        n_restarts = min(10, max(3, int((deadline - time.monotonic() - 30) / 25)))
        per_r = (deadline - time.monotonic() - 30) / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            if r % 3 == 0:
                f_p = perturb(best_f, rng, 0.02 + 0.015 * r)
            elif r % 3 == 1:
                f_p = perturb_asymmetric(best_f, rng, 0.04 + 0.01 * r)
            else:
                f_p = perturb(best_f, rng, 0.05 + 0.02 * r)

            rd = time.monotonic() + per_r * 0.7
            fo, so = nesterov_optimize(f_p, rd,
                                        beta_start=50, beta_end=800,
                                        lr_start=0.025, momentum_init=0.5,
                                        verbose=f"[R{r}]")
            if so < best_score:
                best_score, best_f = so, fo
                print(f"  R{r}: {best_score:.6f}")
            else:
                print(f"  R{r}: {so:.6f} (no improve)")

    # === Phase 7: Final LP + coordinate polish ===
    if best_f is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Phase 7: Final polish ===")
        # LP blend
        c = np.convolve(best_f, best_f)
        M = np.max(c)
        A = build_A_sparse(best_f)
        g = solve_lp_dir(best_f, M, A)
        if g is not None:
            Sg = np.sum(g)
            if Sg > 1e-10:
                g = g / Sg
                for t in [0.05, 0.1, 0.15, 0.2, 0.3]:
                    fb = (1 - t) * best_f + t * g
                    fb = np.clip(fb, 0, None)
                    Sb = np.sum(fb)
                    if Sb < 1e-10:
                        continue
                    fb = fb / Sb
                    sb = fast_score(fb)
                    if np.isfinite(sb) and sb < best_score:
                        best_score, best_f = sb, fb
                        print(f"  LP t={t}: {best_score:.6f}")

        # Coordinate refine
        if time.monotonic() < deadline - 5:
            fc, sc = coordinate_refine(best_f, max_iters=3000)
            if sc < best_score:
                best_score, best_f = sc, fc
                print(f"  Coord: {best_score:.6f}")

        # One more Nesterov pass
        if time.monotonic() < deadline - 3:
            fo, so = nesterov_optimize(best_f.copy(), deadline - 3,
                                        beta_start=200, beta_end=3000,
                                        lr_start=0.01, momentum_init=0.3,
                                        verbose="[final]")
            if so < best_score:
                best_score, best_f = so, fo
                print(f"  Final Nest: {best_score:.6f}")

    # Verify with actual evaluate_sequence
    if best_f is not None:
        final_check = evaluate_sequence([float(x) for x in best_f])
        if np.isfinite(final_check):
            if final_check != best_score:
                print(f"  Note: fast_score={best_score:.6f}, eval={final_check:.6f}")

    if best_f is None:
        best_f = np.ones(256)
        best_score = evaluate_sequence(best_f.tolist())

    best_f = np.clip(best_f, 0.0, 1000.0)
    S = np.sum(best_f)
    if S < 0.01:
        best_f = np.ones(256)

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_f)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_f]
