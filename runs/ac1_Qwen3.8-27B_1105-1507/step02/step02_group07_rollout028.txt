"""
Optimization for minimizing 2n*max(conv(f,f))/sum(f)^2 using diverse non-symmetric
initializations (ramps, one-sided windows), projected subgradient with adaptive
step size, LP direction finding, and spectral-aware polishing.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def fast_subgradient_step(f, lr, momentum_buf=None, beta_mom=0.9):
    """
    One step of projected subgradient descent on the simplex.
    Uses the true subgradient at the argmax positions of conv(f,f).
    """
    n = len(f)
    C = np.convolve(f, f)
    max_C = C.max()
    # Subgradient: grad[j] = 2 * sum_{k: C[k]=max} f[k-j]
    # Use soft version: weight by exp(C[k] - max_C)
    diff = C - max_C
    weights = np.exp(diff * 5.0)  # moderate sharpening
    weights /= weights.sum()

    # grad[j] = 2 * sum_k weights[k] * f[k-j]
    grad = 2.0 * np.correlate(weights, f, mode='valid')

    if momentum_buf is None:
        momentum_buf = np.zeros(n)
    momentum_buf = beta_mom * momentum_buf + (1 - beta_mom) * grad

    new_f = f - lr * momentum_buf
    new_f = np.clip(new_f, 0.0, None)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f, None
    new_f = new_f / S
    return new_f, momentum_buf

def softmax_gradient_optimize(f_init, deadline, beta_start=20.0, beta_end=500.0,
                               lr_start=0.02, max_iters=5000, verbose_prefix=""):
    """Soft-max gradient with Adam on the simplex (mirror descent flavor)."""
    f = f_init.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, np.inf
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = lr_start
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        frac = min(1.0, it / (max_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** frac

        C = np.convolve(f, f)
        max_C = C.max()
        exp_C = np.exp(beta * (C - max_C))
        total = exp_C.sum()
        if total < 1e-300:
            break
        p = exp_C / total

        grad = 2.0 * np.correlate(p, f, mode='valid')

        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        update = m_hat / (np.sqrt(v_hat) + eps)

        new_f = f - lr * update
        new_f = np.clip(new_f, 0.0, None)
        ns = float(np.sum(new_f))
        if ns < 1e-10:
            new_f = f
            ns = S
        new_f = new_f / ns

        if it % 5 == 0:
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_score:
                f = new_f
                best_f = f.copy()
                best_score = sc
                no_improve = 0
                if it % 500 == 0:
                    print(f"    {verbose_prefix}SM iter {it}: score={best_score:.6f} (beta={beta:.1f})")
            else:
                no_improve += 1
                if no_improve > 300:
                    lr *= 0.5
                    no_improve = 0
                    if lr < 1e-7:
                        break
        else:
            f = new_f

    # Final evaluation
    sc = evaluate_sequence(f.tolist())
    if np.isfinite(sc) and sc < best_score:

        best_f = f.copy()

def flatten_gradient_optimize(f_init, deadline, lr=0.05, max_iters=3000, verbose_prefix=""):
    """
    Minimize ||C - mean(C)||^2 to flatten the convolution.
    This is complementary to max-minimization.
    """
    f = f_init.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, np.inf
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        C = np.convolve(f, f)
        mean_C = C.mean()
        residual = C - mean_C

        # Gradient of ||C - mean(C)||^2 w.r.t. f[j]:
        # = 2 * sum_k residual[k] * f[k-j] - (2/(2n-1)) * sum_k f[k-j]
        # The second term is constant w.r.t. j (it's 2/(2n-1) * sum_f * (something)),
        # actually sum_k f[k-j] depends on j... let's just use the first term.
        grad = 2.0 * np.correlate(residual, f, mode='valid')

        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        update = m_hat / (np.sqrt(v_hat) + eps)

        new_f = f - lr * update
        new_f = np.clip(new_f, 0.0, None)
        ns = float(np.sum(new_f))
        if ns < 1e-10:
            new_f = f
            ns = S
        new_f = new_f / ns
        f = new_f

        if it % 20 == 0:
            sc = evaluate_sequence(f.tolist())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 400:
                    lr *= 0.5
                    no_improve = 0
                    if lr < 1e-7:
                        break

    sc = evaluate_sequence(f.tolist())
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_f = f.copy()

    return best_f, best_score


def make_ramp(n, power=1.0):
    """Linear (or powered) ramp: f[i] = (i+1)^power."""
    idx = np.arange(n, dtype=np.float64) + 1
    return idx ** power


def make_one_sided_window(n, side='right', window_type='hann'):
    """Window that is 1 on one side and tapered on the other."""
    if window_type == 'hann':
        full = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / max(n - 1, 1)))
    elif window_type == 'tukey':
        r = 0.5
        w = np.ones(n)
        m = int(np.floor(r * n / 2))
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
        full = w
    elif window_type == 'exp':
        idx = np.arange(n, dtype=np.float64)
        full = np.exp(-idx / (n / 3.0))
    else:
        full = np.ones(n)

    if side == 'right':
        # Taper on the right, flat on the left
        half = n // 2
        return np.ones(n) * np.concatenate([np.ones(half), full[half:]])
    else:
        # Taper on the left, flat on the right
        half = n // 2
        return np.concatenate([full[:half], np.ones(n - half)])


def make_bimodal(n, center_ratio=0.3):
    """Two bumps at different heights."""
    c1 = int(center_ratio * n)
    c2 = int((1 - center_ratio) * n)
    w1 = 0.3 * n
    w2 = 0.2 * n
    x = np.arange(n, dtype=np.float64)
    f = np.exp(-0.5 * ((x - c1) / w1) ** 2) + 0.5 * np.exp(-0.5 * ((x - c2) / w2) ** 2)
    return f + 0.01


def perturb(f, rng, strength=0.05):
    """Add multiplicative noise and renormalize."""
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.0, 3.0)
    new_f = np.clip(new_f, 0.0, 1000.0)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S

                    best_sequence = bf.copy()
                    print(f"  Restart {r} (str={strength:.3f}): {bs:.6f} (NEW BEST)")
                else:
                    print(f"  Restart {r} (str={strength:.3f}): {bs:.6f}")
            except Exception as e:
                print(f"  Restart {r} error: {e}")

    # Final refinement
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print(f"\n=== Final refinement (n={len(best_sequence)}, score={best_score:.6f}) ===")
        try:
            # Quick subgradient polish
            bf, bs = projected_subgradient_optimize(best_sequence, deadline - 10,
                                                     lr_init=0.01, max_iters=2000,
                                                     verbose_prefix="[finPG] ")
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()

            # LP polish if short enough
            if len(best_sequence) <= 768 and time.monotonic() < deadline - 10:
                bf, bs = lp_based_optimize(best_sequence, deadline - 10,
                                           verbose_prefix="[finLP] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()

            # Final SM polish
            if time.monotonic() < deadline - 3:
                bf, bs = softmax_gradient_optimize(best_sequence, deadline - 2,
                                                    beta_start=500.0, beta_end=2000.0,
                                                    lr_start=0.003, max_iters=3000,
                                                    verbose_prefix="[finSM] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()

            print(f"  Final score: {best_score:.6f}")
        except Exception as e:
            print(f"  Final refinement error: {e}")

    if best_sequence is None:
        best_sequence = make_ramp(512, 1.0)
        best_score = evaluate_sequence(best_sequence.tolist())

    # Ensure validity
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256, dtype=np.float64)
        best_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
