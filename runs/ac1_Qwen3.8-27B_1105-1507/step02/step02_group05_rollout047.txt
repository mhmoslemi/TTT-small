import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def compute_J_normalized(a):
    """Assume a is already non-negative; returns 2*n*max(conv)/sum^2"""
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    a_norm = a / s
    b = fast_conv(a_norm)
    return float(2 * len(a_norm) * b.max())

def smooth_max(b, T):
    """Smooth approximation of max using log-sum-exp."""
    b_max = b.max()
    return b_max + np.log(np.sum(np.exp(T * (b - b_max)))) / T

def smooth_max_grad(b, T):
    """Gradient of smooth max wrt b."""
    b_max = b.max()
    weights = np.exp(T * (b - b_max))
    weights /= weights.sum()
    return weights

def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def build_A_active(a, M, tol=1e-4):
    """Build sparse constraint matrix only for positions where b_k is close to M."""
    n = len(a)
    b = fast_conv(a)
    # Find positions where b_k is within tol*M of the max
    threshold = M * (1 - tol)
    active_ks = np.where(b >= threshold)[0]
    
    if len(active_ks) == 0:
        active_ks = np.array([np.argmax(b)])
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_ks):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([idx] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    
    A = lil_matrix((len(active_ks), n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr(), active_ks

def lp_best_response(a, M, use_active=True, tol=1e-4):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    
    if use_active:
        A, active_ks = build_A_active(a, M, tol)
        b_ub = np.full(len(active_ks), M)
    else:
        A = build_A_sparse(a)
        b_ub = np.full(2 * n - 1, M)
    
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def run_lp_iterations(a, max_iters, deadline, tracker, use_active=True):
    """Run LP mirror descent with active set and adaptive step size."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, use_active=use_active)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Adaptive step: try multiple step sizes
        base_t = max(0.03, 0.5 * (1.0 - it / float(max(max_iters, 1))))
        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.01, base_t * 0.5), 0.1, 0.3]
        
        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            # Try very small step
            t_small = 0.01
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                break
        else:
            a = best_a
            current_J = best_J

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def coordinate_descent(a, max_iters, deadline, tracker):
    """Coordinate descent: optimize one element at a time."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    n = len(a)
    stuck_count = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Random order for diversity
        order = np.random.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            # Try different values for a[i]
            old_val = a[i]
            best_val = old_val
            best_J = current_J
            
            # Try multiplicative updates
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0, 0.1]:
                trial = a.copy()
                trial[i] = old_val * factor
                trial = np.maximum(trial, 0.0)
                s = trial.sum()
                if s < 1e-10:
                    continue
                J_trial = compute_J(trial)
                if J_trial < best_J - 1e-12:
                    best_J = J_trial
                    best_val = old_val * factor
            
            if best_val != old_val:
                a[i] = best_val
                a = np.maximum(a, 0.0)
                a /= a.sum()
                current_J = best_J
                improved = True
        
        if not improved:
            stuck_count += 1
            if stuck_count > 3:
                break
        else:
            stuck_count = 0
        
        if it % 5 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a

import numpy as np

def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def fft_size(n):
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    return s


def gafsa_iterations(a, n_iters=200, blend=0.5, deadline=None):
    """
    GAFSA-like iterations: alternate between time domain (non-negativity)
    and frequency domain (flat amplitude) to flatten the self-convolution.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s_sum = a.sum()
    if s_sum < 1e-10:
        return a
    a = a / s_sum
    
    n = len(a)
    s = fft_size(n)
    current_J = compute_J(a)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute DFT
        A = np.fft.rfft(a, s)
        
        # Modify magnitude to be flat (for non-DC bins)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Average magnitude of non-DC bins
        if s > 1:
            avg_mag = np.mean(mag[1:])
            mag_new = mag.copy()
            mag_new[1:] = avg_mag
        else:
            mag_new = mag
        
        # Blend with old magnitude to avoid oscillation
        mag_blended = (1 - blend) * mag + blend * mag_new
        
        # Reconstruct
        A_new = mag_blended * np.exp(1j * phase)
        
        # IFT
        a_new = np.fft.irfft(A_new, s)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s_sum = a_new.sum()
        if s_sum < 1e-10:
            break
        a_new = a_new / s_sum
        
        # Blend with old a
        a = (1 - blend) * a + blend * a_new
        a = np.maximum(a, 0.0)
        s_sum = a.sum()
        if s_sum < 1e-10:
            break
        a = a / s_sum
        
        if it % 50 == 0:
            J = compute_J(a)
            if J < current_J:
                current_J = J
    
    return a

    blend = 0.3
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # GAFSA step
        A = np.fft.rfft(a, s)
        mag = np.abs(A)
        phase = np.angle(A)
        
        avg_mag = np.mean(mag[1:])
        mag_new = mag.copy()
        mag_new[1:] = avg_mag
        
        mag_blended = (1 - blend) * mag + blend * mag_new
        A_new = mag_blended * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, s)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_sum = a_new.sum()
        if s_sum < 1e-10:
            break
        a_new /= s_sum
        
        a = (1 - blend) * a + blend * a_new
        a = np.maximum(a, 0.0)
        s_sum = a.sum()
        if s_sum < 1e-10:
            break
        a /= s_sum
        
        # Every 20 iterations, try a gradient step
        if it % 20 == 19:
            b = fast_conv(a)
            M = b.max()
            b_max = M
            T = 10.0
            weights = np.exp(T * (b - b_max))
            weights /= weights.sum()
            
            # Gradient: grad_i = 2 * sum_j a_j * w_{i+j}
            grad = np.zeros(n)
            for i in range(n):
                j_max = min(n, 2 * n - 1 - i)
                if j_max > 0:
                    grad[i] = 2.0 * np.dot(a[:j_max], weights[i:i + j_max])
            
            # Try a small step
            for lr in [0.05, 0.01, 0.005]:
                a_trial = a - lr * grad
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial /= s_t
                J_trial = compute_J(a_trial)
                if J_trial < current_J:
                    a = a_trial
                    current_J = J_trial
                    break
        
        if it % 50 == 0:
            J = compute_J(a)
            if J < current_J:
                current_J = J
            if tracker is not None:
                tracker.update(a)
        
        # Slowly decrease blend
        blend = max(0.1, blend * 0.995)
    
    if tracker is not None:
        tracker.update(a)
    return a


def build_A_sparse(a, active_ks=None):
    """Build sparse constraint matrix."""
    n = len(a)
    if active_ks is None:
        active_ks = range(2 * n - 1)
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_ks):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([idx] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    
    A = lil_matrix((len(list(active_ks)) if not isinstance(active_ks, range) else len(active_ks), n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M, tol=0.05):
    """LP: max sum(g) s.t. (a*g)_k <= M for important k, g >= 0."""
    n = len(a)
    b = fast_conv(a)
    
    # Find active positions (where b_k is close to M)
    threshold = M * (1 - tol)
    active_ks = np.where(b >= threshold)[0]
    if len(active_ks) == 0:
        active_ks = np.array([np.argmax(b)])
    
    c = -np.ones(n)
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_ks):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([idx] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    
    A = lil_matrix((len(active_ks), n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    A = A.tocsr()
    
    b_ub = np.full(len(active_ks), M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_iterations(a, max_iters, deadline, tracker, tol=0.05):
    """LP mirror descent with active set."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M, tol=tol)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Adaptive step
        base_t = max(0.03, 0.4 * (1.0 - it / float(max(max_iters, 1))))
        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.01, base_t * 0.5), 0.1, 0.2, 0.3]
        
        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            t_small = 0.01
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                break
        else:
            a = best_a
            current_J = best_J

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        self.best_score = float('inf')
        self.t_start = time.monotonic()

        if a.size == 0 or a.sum() < 0.01:
