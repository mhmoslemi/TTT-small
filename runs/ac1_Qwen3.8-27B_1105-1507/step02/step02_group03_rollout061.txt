If F = DFT(f, 2n-1) (zero-padded), then the DFT of f*f (zero-padded to 2n-1) is |F|^2.

```python
def subgradient_step(f, n_iter=1000, lr0=0.01, rng=None):
    n = len(f)
    f = f.copy()
    f = f / f.sum()  # normalize to sum=1
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    
    for it in range(n_iter):
        lr = lr0 / (1 + 0.01 * it)
        conv = np.convolve(f, f)
        k_star = np.argmax(conv)
        
        # Compute gradient of (f*f)[k_star] w.r.t. f
        grad = np.zeros(n)
        for j in range(n):
            idx1 = k_star - j
            idx2 = j - k_star
            if 0 <= idx1 < n:
                grad[j] += f[idx1]
            if 0 <= idx2 < n:
                grad[j] += f[idx2]
        
        # Project gradient onto the tangent space of the simplex
        # (subtract the mean to enforce sum(f) = 1)
        grad -= grad.mean()
        
        # Update
        f_new = f - lr * grad
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s
        
        new_score = compute_score_normalized(f_new)
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
        
        f = f_new
    
    return best_f, best_score
```

```python
"""
Optimized step-function coefficient search using adaptive LP with line search,
multi-target LP directions, subgradient polishing, and diverse initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score_normalized(f):
    """Compute score for normalized sequence (sum=1): 2*n*max(conv)."""
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max()


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def build_conv_matrix(f):
    """Build matrix A such that A @ g = np.convolve(f, g, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def convolve_matrix_apply(A, g):
    """Apply the convolution matrix: A @ g."""
    return A @ g


def lp_solve_direction(f, M, n_target=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 5})
        if res.success and res.x is not None:
            g = np.clip(res.x, 0, None)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def line_search_step(f, g, best_score, n_trials=8):
    """Find best alpha in (1-alpha)*f + alpha * g via line search."""
    n = len(f)
    best_alpha = 0.0
    best_local = best_score
    
    # Log-spaced alphas from small to large
    for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]:
        f_new = (1.0 - alpha) * f + alpha * g
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        score = compute_score_normalized(f_new)
        if score < best_local:
            best_local = score
            best_alpha = alpha
    
    if best_alpha > 0:
        f_result = (1.0 - best_alpha) * f + best_alpha * g
        f_result = np.clip(f_result, 0, None)
        f_result = f_result / f_result.sum()
        return f_result, best_local
    return f, best_score


def lp_refine_v2(f, time_budget, rng, verbose=True):
    """Improved LP refinement with multi-target M and line search."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    start = time.monotonic()
    
    # Adaptive M multiplier (start tight, loosen if stuck)
    m_factor = 1.0
    stall_count = 0
    iteration = 0
    
    while time.monotonic() - start < time_budget:
        iteration += 1
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Try multiple M targets (tight to loose)
        best_g = None
        best_g_score = best_score
        
        for mf in [m_factor, m_factor * 1.02, m_factor * 1.05, m_factor * 1.1]:
            M_target = M * mf
            g = lp_solve_direction(f, M_target)
            if g is not None:
                # Line search in direction g
                f_trial, score_trial = line_search_step(f, g, best_score)
                if score_trial < best_g_score:
                    best_g_score = score_trial
                    best_g = g
        
        if best_g is not None and best_g_score < best_score - 1e-10:
            f_new, score_new = line_search_step(f, best_g, best_score)
            f = f_new
            best_score = score_new
            best_f = f.copy()
            stall_count = 0
            # Slightly tighten M factor over time
            m_factor = max(0.95, m_factor * 0.995)
        else:
            stall_count += 1
            # Loosen M to find new directions
            m_factor = min(1.5, m_factor * 1.1)
            
            if stall_count > 30:
                # Try perturbation
                if rng is not None:
                    f_pert = f * (1.0 + 0.05 * rng.normal(0, 1, n))
                    f_pert = np.clip(f_pert, 0, None)
                    f_pert = normalize(f_pert)
                    pert_score = compute_score_normalized(f_pert)
                    if pert_score < best_score:
                        f = f_pert
                        best_f = f.copy()
                        best_score = pert_score
                    m_factor = 1.2
                    stall_count = 0
                else:
                    break
            
            if m_factor > 2.0:
                break
    
    if verbose:
        elapsed = time.monotonic() - start
        print(f"  LP v2: {iteration} iters, {elapsed:.1f}s, score={best_score:.6f}")
    
    return best_f, best_score


def subgradient_polish(f, n_iter=5000, lr0=0.005, rng=None, verbose=True):
    """Subgradient descent on the max-convolution with simplex projection."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    
    for it in range(n_iter):
        # Adaptive learning rate
        lr = lr0 / (1.0 + 0.005 * it)
        
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Find all near-max positions
        threshold = M * 0.999
        max_positions = np.where(conv >= threshold)[0]
        
        # Compute subgradient (average over near-max positions)
        grad = np.zeros(n)
        for k_star in max_positions[:5]:  # Use up to 5 near-max positions
            for j in range(n):
                idx1 = k_star - j
                idx2 = j - k_star
                if 0 <= idx1 < n:
                    grad[j] += f[idx1]
                if 0 <= idx2 < n:
                    grad[j] += f[idx2]
        
        if len(max_positions) > 1:
            grad /= len(max_positions[:5])
        
        # Project onto tangent space of simplex (sum(f)=1)
        grad = grad - grad.mean()
        
        # Update
        f_new = f - lr * grad
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s
        
        new_score = compute_score_normalized(f_new)
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
        
        f = f_new
    
    if verbose:
        print(f"  Subgradient: {n_iter} iters, score={best_score:.6f}")
    
    return best_f, best_score


def spectral_flatten_v2(f, n_iter=300, rng=None):
    """Spectral flattening with adaptive target."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n_fft = 1 << (2 * n - 1).bit_length()
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    stall = 0
    
    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Target: flatten non-DC magnitudes
        # Keep DC (sum) fixed
        dc_val = mag[0]
        non_dc_mag = mag[1:]
        target_mag = np.median(non_dc_mag) * 1.0  # Target at median
        
        # Gradually push toward flat
        alpha = 0.3
        new_mag = (1 - alpha) * mag + alpha * target_mag
        new_mag[0] = dc_val
        
        F_new = new_mag * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        
        score = compute_score_normalized(f_new)
        if score < best_score - 1e-10:
            best_score = score
            best_f = f_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break
        f = f_new
    
    return best_f, best_score


def pressure_v2(f, n_iter=300, eta=0.2, rng=None):
    """Pressure-based update targeting convolution peaks."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    stall = 0
    
    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Find top peaks
        top_k = np.argsort(conv)[::-1][:3]
        
        # Compute pressure
        pressure = np.zeros(n)
        for k_star in top_k:
            weight = conv[k_star] / M
            for j in range(n):
                idx1 = k_star - j
                idx2 = j - k_star
                if 0 <= idx1 < n:
                    pressure[j] += weight * f[idx1]
                if 0 <= idx2 < n:
                    pressure[j] += weight * f[idx2]
        
        if pressure.max() > 0:
            pressure = pressure / pressure.max()
        
        # Multiplicative update
        eta_eff = eta / (1.0 + 0.01 * it)
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s
        
        score = compute_score_normalized(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 150:
                if rng is not None:
                    f_pert = f * (1.0 + 0.1 * rng.normal(0, 1, n))
                    f_pert = np.clip(f_pert, 0, None)
                    f_pert = normalize(f_pert)
                    if compute_score_normalized(f_pert) < best_score:
                        f = f_pert
                        best_f = f.copy()
                        best_score = compute_score_normalized(f)
                        stall = 0
                    else:
                        break
                else:
                    break
    
    return best_f, best_score


def coordinate_descent_v2(f, n_sweeps=10, rng=None):
    """Coordinate descent with geometric search."""
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    best_score = compute_score_normalized(f)
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n) if rng else np.arange(n)
        for i in order:
            original = f[i]
            if original < 1e-12:
                continue
            best_val = original
            best_local = best_score
            
            for factor in [0.2, 0.4, 0.6, 0.8, 1.25, 1.5, 2.0, 5.0, 0.0]:
                f[i] = max(0.0, original * factor) if factor != 0.0 else 0.0
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                conv = np.convolve(f / s, f / s)
                score = 2.0 * n * conv.max()
                if score < best_local:
                    best_local = score
                    best_val = f[i]
            
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
                f = normalize(f)
    
    f = normalize(f)
    return f, best_score


def diverse_init_v2(n, rng):
    """Generate diverse initial sequences."""
    r = rng.random()
    if r < 0.2:
        # Sparse binary pattern (random subset)
        density = rng.uniform(0.3, 0.7)
        f = (rng.random(n) < density).astype(float)
        f += 0.1  # Small floor
    elif r < 0.4:
        # Cosine window
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n * rng.uniform(0.5, 2.0)))
        f = np.clip(f, 0, None) + 0.05
    elif r < 0.6:
        # Gaussian bump
        center = n * rng.uniform(0.2, 0.8)
        width = n * rng.uniform(0.1, 0.4)
        f = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
        f += 0.05
    elif r < 0.8:
        # Product structure
        f = np.ones(n)
        for _ in range(3):
            center = int(rng.integers(0, n))
            width = int(rng.integers(3, n // 3))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / max(1, width ** 2))
            f *= (0.5 + 0.5 * kernel)
        f += 0.05
    else:
        # Random smooth
        f = np.ones(n)
        for _ in range(8):
            center = int(rng.integers(0, n))
            width = int(rng.integers(2, n // 4))
            amp = rng.uniform(-0.3, 0.3)
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / max(1, width ** 2))
            f += amp * kernel
        f = np.abs(f) + 0.05
    
    f = np.clip(f, 0, None)
    f = f * (1.0 + 0.1 * rng.normal(0, 1, n))
    f = np.clip(f, 0, None)
    if f.sum() < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / f.sum()
    return f
```
