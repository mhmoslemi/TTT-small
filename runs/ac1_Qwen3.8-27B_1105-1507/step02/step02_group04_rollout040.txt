def spectral_design(n, iterations=1000):
    """Design a non-negative sequence with flat power spectrum."""
    a = np.ones(n) / n  # initial
    for _ in range(iterations):
        # Compute FFT
        A = np.fft.rfft(a, 2*n)  # sufficient length to avoid circular convolution
        # We want |A|^2 to be flat
        # At each frequency, set |A|^2 = constant = (sum(a)^2) / (2*n) (approximately)
        target_power = np.sum(a)**2  # total power in time domain = integral of |A|^2
        # Actually, by Parseval: sum(a^2) = (1/(2n)) * sum(|A|^2) for the full FFT
        # We want sum(|A|^2) to be evenly distributed
        target_per_freq = target_power / (2*n)
        # Set new A with target magnitude and current phase
        A_new = np.sqrt(target_per_freq) * np.exp(1j * np.angle(A))
        # But we need A[0] to be real and positive (DC component = sum(a))
        A_new[0] = np.sum(a)
        # Inverse FFT
        a_new = np.fft.irfft(A_new, 2*n)[:n]
        # Project onto non-negative
        a_new = np.maximum(a_new, 0)
        # Normalize
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s * np.sum(a)
        a = a_new
    return a

def compute_conv(a):
    """Compute autocorrelation conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smoothed_weights(conv, tau):
    """Compute log-sum-exp weights for smoothing the max."""
    max_val = np.max(conv)
    shifted = tau * (conv - max_val)
    shifted = np.clip(shifted, -500, 500)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    return w

def build_A_matrix(a):
    """Build the Toeplitz constraint matrix for the LP: (a*g)_k for k=0..2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def solve_lp_for_g(a, M, n, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None

def spectral_init(n, seed=42):
    """Initialize using spectral design: flat magnitude in frequency domain."""
    rng = np.random.default_rng(seed)
    # Random phase in frequency domain
    N = 1
    while N < 2 * n:
        N <<= 1
    phases = np.random.default_rng(seed).uniform(0, 2 * np.pi, N // 2 + 1)
    # Flat magnitude
    mag = np.ones(N // 2 + 1)
    A = mag * np.exp(1j * phases)
    A[0] = float(N)  # DC component
    # Inverse FFT
    a = np.fft.ifft(A).real[:n]
    # Project onto non-negative
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return a_new / s

        if verbose:
            print(f"    n={n} init{idx}: obj={init_obj:.6f}")

        # Phase 1: Subgradient (50% of remaining)
        sg_time = remaining * 0.5
        sg_iters = int(min(15000, max(1000, sg_time * 4)))
        a_sg, _ = subgradient_optimize(a_cur, max_iter=sg_iters,
                                       tau_start=8.0, tau_end=80.0,
                                       step_init=0.06)
        obj_sg = compute_obj(a_sg)
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()

        # Phase 2: LP polish (30% of remaining)
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining > 10 and n <= 350:
            lp_time = remaining * 0.3
            lp_iters = int(min(30, max(5, lp_time / 5)))
            a_lp, obj_lp = lp_optimize(a_sg, max_iter=lp_iters, t=0.25)
            if obj_lp < best_obj:
                best_obj = obj_lp
                best_a = a_lp.copy()

        # Phase 3: Fine subgradient (20% of remaining)
        elapsed = time.monotonic() - start
        remaining = time_budget - elapsed
        if remaining > 5:
            a_cur = best_a.copy()
            fin_iters = int(min(8000, max(500, remaining * 3)))
            a_fin, _ = subgradient_optimize(a_cur, max_iter=fin_iters,
                                            tau_start=40.0, tau_end=200.0,
                                            step_init=0.03)
            obj_fin = compute_obj(a_fin)
            if obj_fin < best_obj:
                best_obj = obj_fin
                best_a = a_fin.copy()

        if verbose:
            print(f"    n={n} init{idx} done: best_obj={best_obj:.6f}, time={time.monotonic()-start:.1f}s")

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def try_parent_sequence():
    """Try to use the parent sequence as a starting point."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Multi-strategy optimization: adaptive subgradient + LP + spectral init.
    Explores lengths 150-500 with focus around parent length.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent, parent_obj = try_parent_sequence()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        parent_n = len(parent)
    else:
        parent_n = 250

    # Sequence lengths to explore (focused around parent)
    n_values = sorted(set([
        parent_n - 50, parent_n - 30, parent_n - 10, parent_n,
        parent_n + 10, parent_n + 30, parent_n + 50,
        150, 200, 250, 300, 350, 400, 500
    ]))
    n_values = [n for n in n_values if 20 <= n <= 600]

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    # First pass: quick evaluation of all lengths
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Quick search (small budget)
        alloc = min(30.0, remaining * 0.06)
        alloc = max(10.0, alloc)

        a_opt, obj = optimize_single(n, seed=seed + n * 7919,
                                      time_budget=alloc, verbose=False)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Phase1 UPDATE: n={n}, obj={best_score:.6f}")

    print(f"\nAfter phase 1: best={best_score:.6f}, n={len(best_seq) if best_seq is not None else 0}")

    # Second pass: deeper search on most promising lengths
    if best_seq is not None:
        best_n = len(best_seq)
        # Focus on lengths within 50 of the best
        focused_n = [n for n in n_values if abs(n - best_n) <= 60]
        if best_n not in focused_n:
            focused_n.append(best_n)
        focused_n.sort()
    else:
        focused_n = [200, 250, 300]

    for n in focused_n:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        alloc = min(120.0, remaining * 0.2)
        alloc = max(20.0, alloc)
        print(f"\nPhase2 n={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        a_opt, obj = optimize_single(n, seed=seed + n * 31337 + 1000,
                                      time_budget=alloc, verbose=True)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  Phase2 UPDATE: n={n}, obj={best_score:.6f}")

    # Try stretching the best to nearby lengths
    if best_seq is not None and time.monotonic() < deadline - 60:
        best_n = len(best_seq)
        stretch_ns = [max(30, best_n - 20), max(30, best_n - 10),
                      min(500, best_n + 10), min(500, best_n + 20)]
        stretch_ns = [n for n in stretch_ns if n != best_n and 30 <= n <= 500]

        for n_st in stretch_ns:
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break
            alloc = min(40.0, remaining * 0.3)

            a_stretched = interpolate_sequence(best_seq, n_st)
            # Optimize the stretched version
            elapsed_inner = 0
            a_opt, obj = optimize_single(n_st, seed=seed + n_st * 12345 + 5000,
                                          time_budget=alloc, verbose=False)
            # Also try starting from the stretched version specifically
            # (optimize_single already includes stretching from parent, but let's
            #  also try starting from best_seq stretched)
            if time.monotonic() < deadline - 15:
                a_cur = a_stretched.copy()
                s = np.sum(a_cur)
                a_cur = a_cur / s
                rem2 = deadline - time.monotonic() - 10
                iters = int(min(5000, max(500, rem2 * 2)))
                a_pol, _ = subgradient_optimize(a_cur, max_iter=iters,
                                                 tau_start=20.0, tau_end=150.0,
                                                 step_init=0.04)
                obj_pol = compute_obj(a_pol)
                if obj_pol < obj:
                    a_opt = a_pol
                    obj = obj_pol

            if obj < best_score:
                best_score = obj
                best_seq = a_opt.copy()
                print(f"  Stretch UPDATE: n={n_st}, obj={best_score:.6f}")

    # Final intensive polishing
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"\nFinal polishing: n={len(best_seq)}, current={best_score:.6f}")
        n_best = len(best_seq)
        a_cur = best_seq.copy()

        polish_round = 0
        while time.monotonic() < deadline - 5:
            polish_round += 1
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break

            # Subgradient polish with high tau
            sg_iters = int(min(10000, remaining * 2.5))
            a_cur, _ = subgradient_optimize(a_cur, max_iter=sg_iters,
                                             tau_start=50.0, tau_end=300.0,
                                             step_init=0.02)
            obj = compute_obj(a_cur)
            if obj < best_score:
                best_score = obj
                best_seq = a_cur.copy()
                print(f"  Polish {polish_round} SG: obj={best_score:.6f}")

            # LP step
            remaining = deadline - time.monotonic()
            if remaining > 20 and n_best <= 350:
                a_lp, obj_lp = lp_optimize(a_cur, max_iter=20, t=0.2)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    print(f"  Polish {polish_round} LP: obj={best_score:.6f}")

            # Random perturbation
            remaining = deadline - time.monotonic()
            if remaining > 15 and rng.random() < 0.3:
                noise_scale = 0.03
                a_pert = a_cur * np.exp(rng.normal(0, noise_scale, n_best))
                a_pert = np.maximum(a_pert, 0)
                s_p = np.sum(a_pert)
                if s_p > 1e-10:
                    a_pert = a_pert / s_p
                    obj_p = compute_obj(a_pert)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_pert.copy()
                        a_cur = a_pert.copy()
                        print(f"  Polish {polish_round} PERT: obj={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(50) / 50
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    # Validate using evaluate_sequence
    result = [float(x) for x in best_seq]
    try:
        validation = evaluate_sequence(result)
        if np.isfinite(validation) and validation != best_score:
            print(f"  Validation: computed={best_score:.6f}, actual={validation:.6f}")
            best_score = validation
    except Exception as e:
        print(f"  Validation error: {e}")

    return result
