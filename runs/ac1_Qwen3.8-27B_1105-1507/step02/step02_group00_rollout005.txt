"""

import numpy as np

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A_mat(a):
    """Build constraint matrix A where A[k,j] gives coefficient of a[j] in (a*g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        # (a*g)[k] = sum_j a[k-j]*g[j] where 0<=k-j<n and 0<=j<n
        # So j ranges from max(0, k-n+1) to min(n-1, k)
        # and a[k-j] is the coefficient
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A

def solve_lp_g(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns (g, dual_vars) or (None, None)."""
    n = len(a)
    A = build_A_mat(a)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
    if res.status == 0:
        g = res.x
        # Dual variables (shadow prices) for the inequality constraints
        duals = res.unequality_margin if hasattr(res, 'unequality_margin') else None
        # Actually in HiGHS, the dual values are in res.ineqlin.marginals or similar
        # Let's just return the solution
        return g, res
    return None, None

def project_simplex(v, s=1.0):
    """Project onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if np.sum(v) < 1e-15:
        return np.full(n, s / n)
    # Soft threshold approach
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def fista_smoothed(a, n_steps=2000, deadline=None):
    """FISTA on smoothed version: minimize (1/2)*sum_k max(0, c[k]-t)^2 + indicator(a>=0, sum=1).

    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)

        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute smoothed gradient
        h = np.convolve(a, a)
        max_h = float(np.max(h))
        
        # Smoothed gradient: for each k where h[k] > 0, contribute (h[k]-t)/n_active
        # This is a subgradient of the max
        tol = max_h * 0.05
        active = h >= max_h - tol
        n_active = max(1, int(np.sum(active)))
        
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                if j != k - j:
                    grad[j] += a[k - j]
                    grad[k - j] += a[j]
                else:
                    grad[j] += 2.0 * a[j]
        grad /= n_active
        
        # Step size
        L = 2.0 * n_active  # crude Lipschitz estimate
        step_sz = 1.0 / L
        
        # FISTA update
        a_new = a - step_sz * grad
        a_new = np.maximum(a_new, 0.0)

        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        # FISTA momentum
        k_new = (1.0 + np.sqrt(1.0 + 4.0 * k_ * k_)) / 2.0
        a = a_new + ((k_ - 1.0) / k_new) * (a_new - a)
        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s < 1e-10:
            a = a_new
        else:

        if obj_new < best_obj:

High-performance LP-alternation optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: fast LP with line search, momentum subgradient, local perturbation search,
and aggressive iteration count within time budget.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A_dense(a):
    """Build dense constraint matrix A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def solve_lp_for_g(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_A_dense(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.status == 0:
        return res.x
    return None


def lp_alternate_fast(n, a_init, max_iter, deadline):
    """Fast LP alternation with line search on blend parameter."""
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    stagnant = 0

        conv = np.convolve(a, a)
        M = float(np.max(conv))
        if M < 1e-15:

        g = solve_lp_for_g(a, M)
        if g is None:

        g_sum = float(np.sum(g))
        if g_sum < 1e-10:

        g = g / g_sum

        # Line search on blend factor
        best_t = 0.5
        best_blend_obj = np.inf
        for t in (0.3, 0.4, 0.5, 0.6, 0.7):

            blend = np.maximum(blend, 0.0)

            m_min = max(0, k - n + 1)
            m_max = min(n - 1, k)
            grad[m_min:m_max + 1] += 2.0 * a[k - m_max:k - m_min + 1][::-1]
        grad /= n_active

        # Adaptive step size
        step_size = 1.0 / (best_obj * np.sqrt(step + 1.0))

        v = mu * v + step_size * grad
        a_new = a - v
        a_new = np.maximum(a_new, 0.0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            mu = min(0.99, mu + 0.005)
        else:
            mu = max(0.5, mu - 0.01)

        a = a_new

    return best_a, best_obj


def local_search(a, n_rounds, deadline, seed=99):
    """Local perturbation search: random multiplicative perturbations on subsets."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    rng = np.random.default_rng(seed)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        # Perturb random subset with multiplicative factors
        n_perturb = int(rng.integers(2, max(3, n // 15)))
        indices = rng.choice(n, size=n_perturb, replace=False)
        factors = rng.uniform(0.6, 1.8, size=n_perturb)

        a_trial = a.copy()
        a_trial[indices] *= factors
        a_trial = np.maximum(a_trial, 0.0)
        s = np.sum(a_trial)
        if s < 1e-10:
            continue
        a_trial = a_trial / s

        obj = compute_obj(a_trial)
        if obj < best_obj:
            best_obj = obj
            best_a = a_trial.copy()
            a = a_trial.copy()

    return best_a, best_obj


def lp_then_subgradient(n, a_init, lp_budget_frac, deadline):
    """Run LP alternation for a fraction of the time, then subgradient for the rest."""
    start = time.monotonic()
    total_budget = deadline - start
    lp_deadline = start + total_budget * lp_budget_frac

    # LP phase
    max_iter = 100000  # let it run until deadline
    a, obj = lp_alternate_fast(n, a_init, max_iter, lp_deadline)

    # Subgradient phase
    if time.monotonic() < deadline - 1:
        a, obj = momentum_subgradient(a, 100000, deadline)

    return a, obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Main search function. Uses LP alternation with line search + momentum subgradient
    + local perturbation search. Tries multiple sequence lengths with focus on n=300.
    """
    start_time = time.monotonic()
    deadline = start_time + budget_s - 15.0

    best_seq = None
    best_score = np.inf

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            s = np.sum(parent)
            if s > 0.01:
                obj = compute_obj(parent)
                best_score = obj
                best_seq = parent.copy()
                print(f"Parent: n={parent.size}, obj={obj:.6f}")
    except Exception:
        pass

    rng = np.random.default_rng(seed)

    # Determine primary length
    if best_seq is not None:
        n_primary = len(best_seq)
    else:
        n_primary = 300

    print(f"Budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    # Phase 1: Intensive optimization on primary length
    if best_seq is not None:
        remaining = deadline - start_time
        print(f"Phase 1: n={n_primary}, obj={best_score:.6f}, {remaining:.0f}s")

        # Split time between LP and subgradient
        lp_frac = 0.6
        a1, obj1 = lp_then_subgradient(n_primary, best_seq, lp_frac, deadline - 60)
        if obj1 < best_score:
            best_score = obj1
            best_seq = a1.copy()
            print(f"  Phase 1 result: obj={best_score:.6f}")

        # Local search
        if time.monotonic() < deadline - 10:
            nr = int(min(200, (deadline - time.monotonic() - 5) / 0.05))
            a2, obj2 = local_search(best_seq, nr, deadline - 5, seed=seed)
            if obj2 < best_score:
                best_score = obj2
                best_seq = a2.copy()
                print(f"  Local search: obj={best_score:.6f}")

        # One more LP round on improved solution
        if time.monotonic() < deadline - 30:
            a3, obj3 = lp_alternate_fast(n_primary, best_seq, 50000, deadline - 10)
            if obj3 < best_score:
                best_score = obj3
                best_seq = a3.copy()
                print(f"  Round 2 LP: obj={best_score:.6f}")

            # Quick subgradient
            if time.monotonic() < deadline - 5:
                a4, obj4 = momentum_subgradient(best_seq, 50000, deadline - 1)
                if obj4 < best_score:
                    best_score = obj4
                    best_seq = a4.copy()
                    print(f"  Round 2 SG: obj={best_score:.6f}")

    # Phase 2: Try alternative lengths
    alt_lengths = [250, 350, 400, 500]
    alt_lengths = [x for x in alt_lengths if x != n_primary]

    for n_alt in alt_lengths:
        if time.monotonic() > deadline - 20:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        print(f"Phase 2: trying n={n_alt}, {remaining:.0f}s left")

        # Initialize from best (resample) or from structured init
        if best_seq is not None:
            a_init = np.interp(np.linspace(0, 1, n_alt),
                               np.linspace(0, 1, len(best_seq)), best_seq)
        else:
            freq = rng.uniform(0.5, 2.0)
            a_init = np.maximum(0, np.cos(2 * np.pi * freq * np.arange(n_alt) / n_alt))

        s = np.sum(a_init)
        if s < 1e-10:
            a_init = np.ones(n_alt) / n_alt
        else:
            a_init = a_init / s

        # Run optimization with remaining time (leaving buffer)
        sub_deadline = time.monotonic() + min(remaining - 10, 120)
        a_alt, obj_alt = lp_then_subgradient(n_alt, a_init, 0.5, sub_deadline)

        if obj_alt < best_score:
            best_score = obj_alt
            best_seq = a_alt.copy()
            print(f"  n={n_alt}: obj={best_score:.6f} (NEW BEST!)")
        else:
            print(f"  n={n_alt}: obj={obj_alt:.6f} (not better than {best_score:.6f})")

    # Phase 3: Final polishing on the overall best
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"Phase 3: Final polish n={n_best}, obj={best_score:.6f}, {remaining:.0f}s")

        if remaining > 30:
            # More LP
            a_f, obj_f = lp_alternate_fast(n_best, best_seq, 50000, deadline - 10)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final LP: obj={best_score:.6f}")

        if time.monotonic() < deadline - 5:
            # Final subgradient
            a_f, obj_f = momentum_subgradient(best_seq, 50000, deadline - 2)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final SG: obj={best_score:.6f}")

        # Final local search
        if time.monotonic() < deadline - 2:
            a_f, obj_f = local_search(best_seq, 50, deadline - 1, seed=seed + 1)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"  Final local: obj={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"DONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    # Ensure valid output
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
