def compute_score(f):
    """Compute the objective value directly."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))

    return float(2 * n * M / (S * S))


def subgradient_direction(f, top_k=3):
    """Compute subgradient direction to reduce max convolution."""
    n = len(f)
    conv = np.convolve(f, f)
    S = float(np.sum(f))
    mean_conv = (S * S) / (2 * n - 1)

    top_indices = np.argsort(conv)[-top_k:]
    weights = np.maximum(conv[top_indices] - mean_conv, 0)
    if np.sum(weights) < 1e-15:
        weights = np.ones(top_k)

    grad = np.zeros(n)
    for k, w in zip(top_indices, weights):
        if w <= 0:
            continue
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += w * 2.0 * f[k - j]

    return grad


def fast_subgradient_optimize(f, deadline, lr_init=0.005, momentum_decay=0.85,
                              top_k=5, max_iter=50000):
    """Fast subgradient descent with momentum."""

    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    lr = lr_init
    momentum = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iter:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        if M <= 0:
            break

        grad = subgradient_direction(f, top_k=top_k)
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-15:
            break

        grad = grad / (norm_grad + 1e-15)
        momentum = momentum_decay * momentum - lr * grad
        new_f = f + momentum
        new_f = np.clip(new_f, 0.0, None)

        new_sum = float(np.sum(new_f))
        if new_sum < 1e-15:
            lr *= 0.5
            no_improve += 1
            if no_improve > 100:
                break
            it += 1
            continue
        new_f = new_f * (S / new_sum)

        new_score = compute_score(new_f)
        it += 1
