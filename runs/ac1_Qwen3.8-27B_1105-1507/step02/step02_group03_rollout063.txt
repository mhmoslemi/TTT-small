```python
"""
Candidate generator for low-peak nonnegative step functions.

Strategy:
- Work with normalized nonnegative sequences.
- Optimize a smooth log-sum-exp approximation of the worst convolution peak
  using L-BFGS-B in log-space, with entropy-like softmax parametrization.
- Polish with exact peak subgradient multiplicative updates.
- Apply an active-constraint linear-programming direction:
    maximize sum(b) subject to f*b <= M, b >= 0,
  then line-search along the convex blend f -> (1-t)f + t g.
- Run across several sequence lengths and restarts, keeping the global best.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _normalize(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def _fft_len(n):
    target = max(1, 2 * n - 1, 3 * n - 2)
    return 1 << (target - 1).bit_length()


def _conv_fft(a, N=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if N is None:
        N = _fft_len(n)
    F = np.fft.rfft(a, N)
    c = np.fft.irfft(F * F, N)[:2 * n - 1]
    c = np.clip(c, 0.0, None)
    return c


def _corr_fft(w, a, N=None):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if N is None:
        N = _fft_len(n)
    Fa = np.fft.rfft(a, N)
    Fw = np.fft.rfft(w, N)
    h = np.fft.irfft(Fw * np.conjugate(Fa), N)[:n]
    h = np.clip(h, 0.0, None)
    return h


def _score(a):
    a = _normalize(a)
    n = a.size
    c = _conv_fft(a)
    m = float(c.max()) if c.size else 0.0
    if m <= 0.0 or not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m)


def _softmax_theta(theta):
    theta = np.asarray(theta, dtype=np.float64)
    if theta.size == 0:
        return np.ones(1, dtype=np.float64)
    th = theta - float(np.max(theta))
    e = np.exp(th)
    s = float(e.sum())
    if s <= 1e-300:
        return np.ones_like(theta, dtype=np.float64) / float(theta.size)
    return e / s


def _theta_from(a):
    a = _normalize(a)
    return np.log(np.clip(a, 1e-300, None))


def _perturb_sequence(a, sigma, rng):
    a = _normalize(a)
    theta = _theta_from(a)
    theta = theta + sigma * rng.standard_normal(a.size)
    theta = np.clip(theta, -40.0, 10.0)
    return _softmax_theta(theta)


def _resample_sequence(a, n):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, None)
    if a.size == n:
        return _normalize(a.copy())
    x0 = np.linspace(0.0, 1.0, a.size)
    x1 = np.linspace(0.0, 1.0, n)
    b = np.interp(x1, x0, a)
    return _normalize(np.clip(b, 0.0, None))


def _objective(theta, tau, n, N):
    theta = np.asarray(theta, dtype=np.float64)
    a = _softmax_theta(theta)
    a = _normalize(a)
    F = np.fft.rfft(a, N)
    c = np.fft.irfft(F * F, N)[:2 * n - 1]
    c = np.clip(c, 0.0, None)
    csum = float(c.sum())
    if csum <= 1e-300 or not np.isfinite(csum):
        return (1e300, np.zeros(n, dtype=np.float64))
    c = c / csum

    if tau <= 1e-300:
        return (1e300, np.zeros(n, dtype=np.float64))

    x = c / tau
    mx = float(np.max(x))
    ex = np.exp(x - mx)
    Z = float(ex.sum())
    if Z <= 1e-300:
        return (1e300, np.zeros(n, dtype=np.float64))

    p = ex / Z
    L = float(tau * (mx + np.log(Z)))

    W = np.fft.rfft(p, N)
    h = np.fft.irfft(W * np.conjugate(F), N)[:n]
    h = np.clip(h, 0.0, None)

    g_a = 2.0 * h
    shift = float(np.dot(a, g_a))
    grad = a * (g_a - shift)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

    if not np.isfinite(L):
        return (1e300, np.zeros(n, dtype=np.float64))

    return (L, grad)


def _build_full_conv_matrix(f):
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _build_active_matrix(f, rows):
    f = np.asarray(f, dtype=np.float64)
    rows = np.asarray(rows, dtype=np.int64)
    n = f.size
    A = np.zeros((max(1, rows.size), n), dtype=np.float64)
    if rows.size == 0:
        return A
    for ii in range(rows.size):
        k = int(rows[ii])
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        if je > js:
            A[ii, js:je] = f[k - np.arange(js, je)]
    return A


def _pressure_polish(a, deadline, rng, max_iter=300, initial_eta=0.35):
    a = _normalize(a.copy())
    n = a.size
    N = _fft_len(n)
    cur = a.copy()
    cur_score = _score(cur)

    best = cur.copy()
    best_score = cur_score

    eta = initial_eta
    stall = 0

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        c = _conv_fft(cur, N)
        M = float(c.max())
        if M <= 0.0 or not np.isfinite(M):
            break

        tau = max(1e-15, M * max(1e-6, 2e-4 * (0.95 ** min(it, 250))))
        x = c / tau
        mx = float(np.max(x))
        ex = np.exp(x - mx)
        Z = float(ex.sum())
        if Z <= 1e-300:
            break
        p = ex / Z

        h = _corr_fft(p, cur, N)
        hmax = float(h.max())
        if hmax <= 1e-15:
            break
        h = h / hmax

        improved = False
        for scale in (1.0, 0.5, 0.25, 0.125):
            new = cur * np.exp(-eta * scale * h)
            new = _normalize(new)
            sc = _score(new)
            if sc < cur_score - 1e-12:
                cur = new
                cur_score = sc
                improved = True
                if sc < best_score:
                    best_score = sc
                    best = cur.copy()
                break

        if not improved:
            eta *= 0.7
            stall += 1
            if stall > 20:
                pert = cur * (1.0 + 0.05 * rng.standard_normal(n))
                pert = _normalize(pert)
                psc = _score(pert)
                if psc < best_score:
                    cur = pert
                    cur_score = psc
                    best_score = psc
                    best = cur.copy()
                    eta = initial_eta
                    stall = 0
                else:
                    break

    return best, best_score


def _lp_refine(a, deadline, rng, max_constraints=None, max_iter=None):
    a = _normalize(a.copy())
    n = a.size
    m = 2 * n - 1
    N = _fft_len(n)

    if max_iter is None:
        if n <= 512:
            max_iter = 30
        elif n <= 1024:
            max_iter = 18
        elif n <= 2048:
            max_iter = 10
        else:
            max_iter = 6

    if max_constraints is None:
        if n <= 1024:
            max_constraints = m
        else:
            max_constraints = min(m, 1024)

    K0 = int(max_constraints)
    K0 = max(32, K0)
    K0 = min(m, K0)

    if max_constraints is None or int(max_constraints) >= m:
        Kcap = m
    else:
        Kcap = min(m, max(256, 2 * int(max_constraints), K0))

    cur = a.copy()
    cur_score = _score(cur)
    best = cur.copy()
    best_score = cur_score

    t = 0.4
    K = K0

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        c = _conv_fft(cur, N)
        M = float(c.max())
        if M <= 0.0 or not np.isfinite(M):
            break

        if K >= m:
            rows = np.arange(m, dtype=np.int64)
        else:
            top = np.argsort(c)[-K:]
            thr = np.where(c > 0.9 * M)[0]
            rows = np.unique(np.concatenate((top, thr)))
            if rows.size > Kcap:
                rows = np.argsort(c)[::-1][:Kcap]

        if rows.size == m and n <= 1200:
            A = _build_full_conv_matrix(cur)
            lp_time = 2.0
        else:
            A = _build_active_matrix(cur, rows)
            lp_time = 1.5

        try:
            res = linprog(
                -np.ones(n, dtype=np.float64),
                A_ub=A,
                b_ub=np.full(rows.size, M, dtype=np.float64),
                bounds=[(0.0, None)] * n,
                method='highs',
                options={'presolve': True, 'time_limit': lp_time}
            )
        except Exception:
            break

        if res is None or getattr(res, 'x', None) is None:
            break

        b = np.asarray(res.x, dtype=np.float64)
        if b.size != n or not np.all(np.isfinite(b)):
            break
        b = np.clip(b, 0.0, None)
        S = float(b.sum())
        if S <= 1e-15:
            break

        g = b / S
        if np.allclose(g, cur, rtol=1e-6, atol=1e-12):
            break

        improved = False
        step_candidates = []
        for s in (min(1.0, t * 1.5), t, t * 0.5, t * 0.25, t * 0.1):
            if 0.0 < s <= 1.0 and s not in step_candidates:
                step_candidates.append(s)
        if not step_candidates:
            step_candidates = [0.1]

        for step in step_candidates:
            new = (1.0 - step) * cur + step * g
            new = _normalize(new)
            sc = _score(new)
            if sc < cur_score - 1e-12:
                cur = new
                cur_score = sc
                improved = True
                if sc < best_score:
                    best_score = sc
                    best = cur.copy()
                t = min(0.9, t * 1.15)
                break

        if not improved:
            if K < Kcap:
                K = min(Kcap, int(K * 1.5) + 1)
            else:
                t *= 0.6
                if t < 0.01:
                    break

    return best, best_score


def _optimize_sequence(a, deadline, rng, budget=60.0):
    a = _normalize(a.copy())
    n = a.size
    N = _fft_len(n)

    best_a = a.copy()
    best_score = _score(a)

    lp_reserve = min(30.0, max(5.0, 0.15 * float(budget)))
    min_deadline = max(time.monotonic(), deadline - lp_reserve - 5.0)

    theta = _theta_from(a)
    bounds = [(-40.0, 10.0) for _ in range(n)]

    tau_fracs = [0.12, 0.04, 0.012, 0.0035, 0.0012]
    base_iters = [150, 120, 95, 75, 55]

    budget_scale = min(1.0, max(0.25, float(budget) / 60.0))
    n_scale = 512.0 / max(1, n)

    for stage in range(len(tau_fracs)):
        if time.monotonic() >= min_deadline:
            break

        cmax = max(1e-12, best_score / (2.0 * max(1, n)))
        tau = max(1e-15, float(tau_fracs[stage]) * cmax)
        maxiter = int(base_iters[stage] * budget_scale * n_scale) + 25
        maxiter = max(20, min(maxiter, 400))

        try:
            res = minimize(
                _objective,
                theta,
                method='L-BFGS-B',
                jac=True,
                args=(tau, n, N),
                bounds=bounds,
                options={
                    'maxiter': maxiter,
                    'ftol': 1e-11,
                    'gtol': 1e-8,
                    'maxls': 8,
                    'disp': False
                }
            )
        except Exception:
            break

        if res is not None and getattr(res, 'x', None) is not None:
            try:
                theta_clean = np.asarray(res.x, dtype=np.float64)
                if theta_clean.shape == (n,) and np.all(np.isfinite(theta_clean)):
                    theta_clean = np.clip(theta_clean, -40.0, 10.0)
                    theta = theta_clean
                    a_new = _softmax_theta(theta)
                    a_new = _normalize(a_new)
                    sc = _score(a_new)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_new.copy()
                    a = a_new
            except Exception:
                pass

    now = time.monotonic()
    if now < min_deadline:
        p_deadline = min(min_deadline, now + min(25.0, 0.2 * float(budget)))
        p_iters = int(max(30, min(300, 200 * budget_scale * n_scale)))
        b, s = _pressure_polish(a, p_deadline, rng, max_iter=p_iters)
        if s < best_score:
            best_score = s
            best_a = b.copy()

    now = time.monotonic()
    if now < deadline - 3.0:
        if n <= 1024:
            maxc = 2 * n - 1
        elif n <= 2048:
            maxc = 1024
        else:
            maxc = 1024
        b, s = _lp_refine(best_a, deadline, rng, max_constraints=maxc)
        if s < best_score:
            best_score = s
            best_a = b.copy()

    if time.monotonic() < deadline - 1.0:
        p_deadline = min(deadline, time.monotonic() + 10.0)
        p_iters = int(max(10, 50 * budget_scale * n_scale))
        b, s = _pressure_polish(best_a, p_deadline, rng, max_iter=p_iters)
        if s < best_score:
            best_score = s
            best_a = b.copy()

    return best_a, best_score


def _get_parent_sequence():
    p = globals().get('height_sequence_1', None)
    if p is None:
        return None
    try:
        arr = np.asarray(p, dtype=np.float64)
    except Exception:
        return None
    if arr.ndim != 1 or arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        return None
    return arr / s


def _initial_specs(parent, reference):
    specs = []

    if parent is not None:
        specs.append(('parent', 0.0))
        specs.append(('parent_rev', 0.0))
        specs.append(('parent_pert', 0.10))
        specs.append(('parent_pert', 0.25))
        specs.append(('parent_pert', 0.50))

    if reference is not None:
        specs.append(('ref', 0.0))
        specs.append(('ref_pert', 0.10))
        specs.append(('ref_pert', 0.30))

    specs.extend([
        ('uniform', 0.0),
        ('gauss', 0.35),
        ('gauss', 0.50),
        ('gauss', 0.65),
        ('expdecay', -2.0),
        ('expdecay', -4.0),
        ('expdecay', 2.0),
        ('power', -0.5),
        ('power', -1.0),
        ('power', 0.5),
        ('cos_asym', 0.25),
        ('two_hump', 0.0),
        ('random_smooth', 0.0),
    ])

    seen = set()
    out = []
    for sp in specs:
        key = (sp[0], float(sp[1]))
        if key not in seen:
            seen.add(key)
            out.append(sp)
    return out


def _make_initial(n, spec, rng, parent, reference):
    name = spec[0]
    param = spec[1]
    x = np.linspace(0.0, 1.0, n)
    a = None

    if name == 'uniform':
        a = np.ones(n, dtype=np.float64)
        a = a * (1.0 + 0.12 * rng.standard_normal(n))

    elif name == 'gauss':
        center = float(param)
        width = 0.18
        a = np.exp(-0.5 * ((x - center) / width) ** 2) + 0.02

    elif name == 'expdecay':
        alpha = float(param)
        a = np.exp(alpha * x) + 0.01

    elif name == 'power':
        beta = float(param)
        a = (x + 0.10) ** beta + 0.01

    elif name == 'cos_asym':
        phase = float(param)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x - phase))) + 0.03

    elif name == 'two_hump':
        c1 = rng.uniform(0.15, 0.40)
        c2 = rng.uniform(0.60, 0.85)
        w = rng.uniform(0.10, 0.28)
        a = np.exp(-0.5 * ((x - c1) / w) ** 2) + 0.5 * np.exp(-0.5 * ((x - c2) / w) ** 2)
        a = a + 0.01

    elif name == 'random_smooth':
        a = np.ones(n, dtype=np.float64)
        for _ in range(5):
            center = rng.uniform(0.1, 0.9)
            width = rng.uniform(0.05, 0.35)
            amp = rng.uniform(-0.7, 0.7)
            a *= (1.0 + amp * np.exp(-0.5 * ((x - center) / width) ** 2))
        a = np.abs(a) + 0.01

    elif name == 'parent':
        if parent is not None:
            a = _resample_sequence(parent, n)

    elif name == 'parent_rev':
        if parent is not None:
            a = _resample_sequence(parent[::-1], n)

    elif name == 'parent_pert':
        if parent is not None:
            a = _resample_sequence(parent, n)
            a = _perturb_sequence(a, float(param), rng)

    elif name == 'ref':
        if reference is not None:
            a = _resample_sequence(reference, n)

    elif name == 'ref_pert':
        if reference is not None:
            a = _resample_sequence(reference, n)
            a = _perturb_sequence(a, float(param), rng)

    if a is None:
        return None

    if name != 'random_smooth' and rng.random() < 0.85:
        a = a * (1.0 + 0.01 * rng.standard_normal(n))

    a = np.where(np.isfinite(a), a, 0.0)
    return _normalize(np.clip(a, 0.0, None))


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(120.0, float(budget_s) - 30.0)

    best_a = None
    best_score = np.inf

    parent = _get_parent_sequence()
    if parent is not None:
        parent = _normalize(parent)
        best_a = parent.copy()
        best_score = _score(best_a)
        print("parent n={}, internal score={:.6f}".format(len(best_a), best_score))

    lengths = set()
    if parent is not None:
        lengths.add(int(parent.size))
    lengths.update([512, 1024, 2048])
    if float(budget_s) > 900:
        lengths.add(4096)
    if float(budget_s) > 1300:
        lengths.add(8192)

    lengths = sorted([int(x) for x in lengths if 64 <= int(x) <= 8192])

    final_reserve = min(300.0, max(60.0, 0.15 * float(budget_s)))

    for n in lengths:
        remaining = deadline - time.monotonic()
        if remaining < final_reserve:
            break

        if n <= 128:
            cand_budget = 25
            max_cands = 3
        elif n <= 512:
            cand_budget = 35
            max_cands = 6
        elif n <= 1024:
            cand_budget = 55
            max_cands = 5
        elif n <= 2048:
            cand_budget = 80
            max_cands = 3
        elif n <= 4096:
            cand_budget = 100
            max_cands = 2
        else:
            cand_budget = 140
            max_cands = 1

        specs = _initial_specs(parent, best_a)
        executed = 0

        for spec in specs:
            if executed >= max_cands:
                break
            remaining = deadline - time.monotonic()
            if remaining < cand_budget + 30.0:
                break

            a = _make_initial(n, spec, rng, parent, best_a)
            if a is None:
                continue

            cand_deadline = min(deadline - 5.0, time.monotonic() + float(cand_budget))
            if cand_deadline <= time.monotonic():
                break

            try:
                b, s = _optimize_sequence(a, cand_deadline, rng, budget=cand_budget)
                executed += 1
                if s < best_score:
                    best_score = s
                    best_a = b.copy()
                    print("updated n={} score={:.6f} after candidate {}".format(n, best_score, spec[0]))
            except Exception:
                executed += 1

    if best_a is not None and time.monotonic() < deadline - 20.0:
        print("final polish started, current best={:.6f}, n={}".format(best_score, len(best_a)))

        b, s = _pressure_polish(best_a, deadline, rng, max_iter=300)
        if s < best_score:
            best_score = s
            best_a = b.copy()

        if time.monotonic() < deadline - 25.0:
            n = best_a.size
            if n <= 1024:
                maxc = 2 * n - 1
            elif n <= 2048:
                maxc = 1024
            else:
                maxc = 1024
            b, s = _lp_refine(best_a, deadline, rng, max_constraints=maxc)
            if s < best_score:
                best_score = s
                best_a = b.copy()

        if time.monotonic() < deadline - 2.0:
            b, s = _pressure_polish(best_a, deadline, rng, max_iter=50)
            if s < best_score:
                best_score = s
                best_a = b.copy()

    if best_a is not None:
        best_a = _normalize(best_a)
        final_list = [float(x) for x in np.clip(best_a, 0.0, 1000.0)]
        try:
            final_score = float(evaluate_sequence(final_list))
        except Exception:
            final_score = np.inf

        print("final internal={:.6f}, evaluator={:.6f}, n={}".format(best_score, final_score, len(final_list)))

        if np.isfinite(final_score):
            return final_list

    fallback = [1.0] * 512
    try:
        evaluate_sequence(fallback)
    except Exception:
        pass
    return fallback
```
