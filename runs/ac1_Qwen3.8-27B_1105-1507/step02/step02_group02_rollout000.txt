Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2.
Key insight: for non-negative a, max(conv(a,a)) = sum(a_i^2) always (middle element),
so J = 2n*sum(a_i^2)/(sum(a_i))^2 >= 2 by Cauchy-Schwarz, with equality at uniform.
Strategy: converge to uniform via projected gradient descent, try multiple lengths,
and return the best sequence found.

import numpy as np

def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_score_energy(a):
    """Compute using the insight that max is always sum(a_i^2) for non-neg a."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    return float(2.0 * n * np.sum(a * a) / (s * s))


def projected_gradient_descent(a0, max_iters=5000, lr=0.1, deadline=None):
    """
    Minimize J(a) = 2n*sum(a^2)/(sum(a))^2 via projected gradient descent.
    We parameterize with a on the probability simplex (sum=1, a>=0).
    On the simplex, J = 2n*sum(a^2), so grad = 2n*2a = 4n*a.
    Projected gradient step: a_new = clip(a - lr*grad, 0, inf), then renormalize.
    """
    n = len(a0)
    a = a0.copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best = a.copy()
    best_score = compute_score(a)

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Gradient of 2n*sum(a_i^2) on the simplex w.r.t. a_i is 4n*a_i
        # (after normalization, the objective is 2n*sum(a_i^2))
        grad = 4.0 * n * a

        # Step
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0, None)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new

        # Adaptive learning rate: decrease if oscillating
        score = compute_score_energy(a_new)
        if score < best_score:
            best_score = score
            best = a_new.copy()
            a = a_new
        else:
            # Reduce learning rate
            lr *= 0.9
            if lr < 1e-6:
                break

    return best, best_score
