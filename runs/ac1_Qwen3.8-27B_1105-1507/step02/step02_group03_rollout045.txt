"""
Improved optimization: aggressive LP with tight bounds, peak targeting,
pairwise coordinate descent, and momentum-based LP direction averaging.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute score for sequence: 2*n*max(conv)/sum^2."""
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (s * s)


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def build_conv_matrix(f):
    """Build matrix A such that A @ g = np.convolve(f, g, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def lp_direction(f, M, method='highs', time_limit=5):
    """Solve LP: max sum(g) s.t. A(f)@g <= M, g >= 0. Returns g or None."""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method=method, options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None

def peak_targeting(f, n_iter=200, rng=None):
    """Target specific peaks in the convolution."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    
    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Find top peaks
        peak_positions = np.argsort(conv)[::-1][:5]
        
        # For each peak, identify contributing pairs and reduce them
        reduction = np.zeros(n)
        for p in peak_positions:
            weight = conv[p] / M  # Higher weight for higher peaks
            for i in range(n):
                j = p - i
                if 0 <= j < n:
                    reduction[i] += weight * f[j]
                    # Also reduce f[j] if i != j
                    if i != j:
                        reduction[j] += weight * f[i] * 0.5
        
        if reduction.max() < 1e-15:
            break
        
        # Apply reduction
        eta = 0.1 * (1 - it / n_iter)
        f_new = f * np.exp(-eta * reduction / reduction.max())
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)
        
        new_score = compute_score(f_new)
        if new_score < best_score - 1e-12:
            best_f = f_new.copy()
            best_score = new_score
            f = f_new
        else:
            # Try smaller eta
            for eta_small in [eta * 0.3, eta * 0.1]:
                f_try = f * np.exp(-eta_small * reduction / reduction.max())
                f_try = np.clip(f_try, 1e-15, None)
                f_try = normalize(f_try)
                if compute_score(f_try) < best_score:
                    f = f_try
                    best_f = f.copy()
                    best_score = compute_score(f)
                    break
            else:
                break
    
    return best_f, best_score


def pairwise_cd(f, n_sweeps=20, rng=None):
    """Pairwise coordinate descent."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    best_score = compute_score(f)
    
    for sweep in range(n_sweeps):
        # Random pairs
        for _ in range(n):  # n pairs per sweep
            i, j = rng.integers(0, n, 2)
            if i == j:
                continue
            
            orig_i, orig_j = f[i], f[j]
            if orig_i < 1e-10 and orig_j < 1e-10:
                continue
            
            best_local = best_score
            best_ai, best_aj = orig_i, orig_j
            
            # Try geometric factors
            for fi in [0.3, 0.5, 0.7, 0.85, 1.18, 1.43, 2.0, 3.33]:
                for fj in [0.3, 0.5, 0.7, 0.85, 1.18, 1.43, 2.0, 3.33]:
                    f[i] = max(0.0, orig_i * fi)
                    f[j] = max(0.0, orig_j * fj)
                    s = f.sum()
                    if s < 1e-10:
                        continue
                    score = 2.0 * n * np.convolve(f, f).max() / (s * s)
                    if score < best_local:
                        best_local = score
                        best_ai, best_aj = f[i], f[j]
            
            f[i], f[j] = best_ai, best_aj
            if best_local < best_score:
                best_score = best_local
        
        f = np.clip(f, 0, None)
        f = normalize(f)
    
    return f, best_score

def multi_objective_lp(f, time_budget, rng=None):

        for c_obj in objectives:

                continue

"""
Improved multi-phase optimization: aggressive LP with momentum and tight bounds,
peak targeting, multi-objective LP exploration, pairwise CD, and spectral shaping.
Focuses on reducing the convolution peak below the 1.5027 target.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_score(f):
    """Compute score: 2*n*max(conv)/sum^2."""
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (s * s)


def normalize(f):
    """Normalize f to sum to 1."""
    s = f.sum()
    if s < 1e-15:
        return f
    return f / s


def build_conv_matrix(f):
    """Build matrix A such that A @ g = np.convolve(f, g, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    return A


def solve_lp(f, M, c_obj=None, time_limit=3):
    """Solve LP: max c^T g s.t. A(f)@g <= M, g >= 0. Returns g or None."""
    n = len(f)
    if c_obj is None:
        c_obj = -np.ones(n)
    A = build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.clip(res.x, 0, None)
    except Exception:
        pass
    return None


def aggressive_lp(f, time_budget, rng=None):
    """Aggressive LP iteration with momentum and adaptive M tightening."""
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)

    start = time.monotonic()

    momentum = np.zeros(n)
    momentum_count = 0
    t = 0.25

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()

        # Try tight M first, then relax if needed
        g = solve_lp(f, M * 0.998, time_limit=3)
        if g is None or g.sum() < 1e-10:
            g = solve_lp(f, M, time_limit=3)

        if g is None or g.sum() < 1e-10:
            break

        g = g / g.sum()

        momentum = 0.85 * momentum + 0.15 * g

def peak_targeting(f, time_budget, rng=None):
    """Directly reduce convolution peaks by identifying responsible elements."""
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)

    best_f = f.copy()
    best_score = compute_score(f)
    start = time.monotonic()

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()

        # Find top K peaks
        K = min(8, len(conv))
        peak_indices = np.argsort(conv)[::-1][:K]

        # Compute reduction vector: for each f[i], how much does it contribute to peaks?
        reduction = np.zeros(n)
        for p in peak_indices:
            weight = (conv[p] - 0.5 * M) / M  # Emphasize peaks above half-max
            if weight < 0:
                weight = 0.1
            # f[i] contributes f[i]*f[p-i] to conv[p]
            for i in range(n):
                j = p - i
                if 0 <= j < n:
                    reduction[i] += weight * f[j]

        if reduction.max() < 1e-15:
            break

        reduction = reduction / reduction.max()

        # Apply exponential reduction with adaptive eta
        improved = False
        for eta in [0.15, 0.1, 0.05, 0.03, 0.015]:
            f_new = f * np.exp(-eta * reduction)
            f_new = np.clip(f_new, 1e-15, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_score = compute_score(f_new)
            if new_score < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                improved = True
                break

        if not improved:
            # Add noise to escape
            f_pert = f * (1 + 0.03 * rng.normal(0, 1, n))
            f_pert = np.clip(f_pert, 1e-10, None)
            f_pert = normalize(f_pert)
            if compute_score(f_pert) < best_score:
                f = f_pert
                best_f = f.copy()
                best_score = compute_score(f)
            else:
                break

    return best_f, best_score

                    best_score = new_score

def pairwise_cd(f, time_budget, rng=None):
    """Efficient pairwise coordinate descent."""
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    best_score = compute_score(f)
    start = time.monotonic()

    # Reduced set of factors for speed
    factors = [0.4, 0.65, 0.85, 1.18, 1.5, 2.5]

    while time.monotonic() - start < time_budget:
        improved_sweep = 0
        for _ in range(n):
            if time.monotonic() - start >= time_budget:
                break

            i, j = rng.integers(0, n, 2)
            if i == j:
                continue

            orig_i, orig_j = f[i], f[j]
            if orig_i < 1e-12 and orig_j < 1e-12:
                continue

            best_local = best_score
            best_ai, best_aj = orig_i, orig_j

            # Try a subset of combinations
            for fi in factors:
                for fj in factors[::2]:  # Every other factor for speed
                    f[i] = max(0.0, orig_i * fi)
                    f[j] = max(0.0, orig_j * fj)
                    s = f.sum()
                    if s < 1e-10:
                        continue
                    score = 2.0 * n * np.convolve(f, f).max() / (s * s)
                    if score < best_local:
                        best_local = score
                        best_ai, best_aj = f[i], f[j]

            f[i], f[j] = best_ai, best_aj
            if best_local < best_score:
                best_score = best_local
                improved_sweep += 1

        f = np.clip(f, 0, None)
        f = normalize(f)

        if improved_sweep == 0:
            break

    return f, best_score

def diverse_init(n, rng):

            t5 = min(remaining * 0.08, 80)
            print(f"\nPhase 5: Spectral shaping (budget: {t5:.0f}s)")
            f5, s5 = spectral_shaping(best_overall, t5, rng=rng)
            print(f"  Score: {s5:.6f}")
            if s5 < best_score_overall:
                best_score_overall = s5
                best_overall = f5.copy()

    # Phase 6: Final aggressive LP (most time)
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t6 = min(remaining * 0.35, 400)
            print(f"\nPhase 6: Final aggressive LP (budget: {t6:.0f}s)")
            f6, s6 = aggressive_lp(best_overall, t6, rng=rng)
            print(f"  Score: {s6:.6f}")
            if s6 < best_score_overall:
                best_score_overall = s6
                best_overall = f6.copy()

    # Phase 7: Final pairwise CD polish
    if best_overall is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t7 = min(remaining * 0.5, 100)
            print(f"\nPhase 7: Final pairwise CD (budget: {t7:.0f}s)")
            f7, s7 = pairwise_cd(best_overall, t7, rng=rng)
            print(f"  Score: {s7:.6f}")
            if s7 < best_score_overall:
                best_score_overall = s7
                best_overall = f7.copy()

    # Verify with evaluate_sequence
    if best_overall is not None:
        best_overall = np.clip(best_overall, 0, 1000)
        final_list = [float(x) for x in best_overall]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"Final: score={final_score:.6f}, n={len(final_list)}")
        return final_list

    # Fallback: try diverse init
    print("Using diverse init fallback")
    f_fb = diverse_init(n, rng)
    f_fb, s_fb = aggressive_lp(f_fb, 60, rng=rng)
    f_fb = np.clip(f_fb, 0, 1000)
    final_list = [float(x) for x in f_fb]
    final_score = evaluate_sequence(final_list)
    print(f"Final (fallback): score={final_score:.6f}")
    return final_list
