"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternation, Adam smoothed gradient descent, spectral shaping,
and coordinate refinement with diverse initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix for LP: A[k,j] gives coefficient of a[j] in conv[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            # conv[k] = sum_j a[j] * a[k-j]
            # For fixed k, the valid j range is [max(0, k-n+1), min(n-1, k)]
            # a[k-j] for j in that range
            indices = np.arange(j_start, j_end + 1)
            A[k, indices] = a[k - indices]
    return A


def solve_lp_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_alternate(a_init, max_iter=200, verbose=False):
    """LP alternating optimization with adaptive step size."""
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    # Adaptive step size schedule
    for it in range(max_iter):
        conv_a = np.convolve(a, a)
        M = float(np.max(conv_a))
        
        if M < 1e-15:
            break
        
        # Solve LP for g
        g = solve_lp_response(a, M)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Also solve LP for a given g (symmetric step)
        conv_g = np.convolve(g, g)
        M_g = float(np.max(conv_g))
        a_new = solve_lp_response(g, M_g)
        if a_new is not None:
            a_new_sum = np.sum(a_new)
            if a_new_sum > 1e-10:
                a_new = a_new / a_new_sum
            # Check both g and a_new
            obj_g = compute_obj(g)
            obj_a_new = compute_obj(a_new)
            
            if obj_g < best_obj:
                best_obj = obj_g
                best_a = g.copy()
            if obj_a_new < best_obj:
                best_obj = obj_a_new
                best_a = a_new.copy()
            
            # Update: use a_new as next iterate (it's the "best response" to g)
            # Blend for stability
            t = 0.5 + 0.3 * (it / max_iter)  # increasing blend
            a = (1 - t) * a + t * a_new
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s < 1e-10:
                a = a_new
            else:
                a = a / s
        else:
            obj_g = compute_obj(g)
            if obj_g < best_obj:
                best_obj = obj_g
                best_a = g.copy()
            a = g
        
        if verbose and (it % 20 == 0):
            print(f"    LP it={it}: best={best_obj:.6f}")
    
    return best_a, best_obj


def adam_smoothed_grad(a_init, n_steps=5000, verbose=False):
    """
    Adam optimizer with log-sum-exp smoothing of the max.
    Minimize log(soft_max(conv(a,a))) - 2*log(sum(a)) subject to a>=0.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    
    # Learning rate schedule
    lr0 = 0.1
    
    for step in range(n_steps):
        # Compute convolution
        conv = np.convolve(a, a)
        max_c = float(np.max(conv))
        
        if step % 100 == 0 and verbose:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        
        # Active set (soft): use log-sum-exp for smoothing
        tau = max_c * 0.01  # smoothing parameter
        if tau < 1e-15:
            tau = 1e-15
        # soft_max = (1/tau) * log(sum(exp(tau * conv / max_c)))
        # grad of soft_max w.r.t. a_j = (1/max_c) * sum_k (exp(tau*conv_k/max_c) / sum(exp(tau*conv/max_c))) * d(conv_k)/da_j
        # Simplified: use the active set with a tolerance
        tol = max_c * 0.05
        active = conv >= max_c - tol
        n_active = max(1, int(np.sum(active)))
        
        # Compute gradient: grad[j] = (2/n_active) * sum_{k active} a[k-j] (if 0<=k-j<n)
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                grad[j_start:j_end+1] += 2.0 * a[k - (j_start:j_end+1)]
        grad /= n_active
        
        # Subtract the "sum constraint" gradient: -2/sum(a)
        sum_a = np.sum(a)
        grad -= 2.0 / sum_a
        
        # Adam update
        t = step + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        lr = lr0 / (1.0 + 0.001 * step)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new
    
    obj_final = compute_obj(a)
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = a.copy()
    
    return best_a, best_obj

So we want the autocorrelation to be as flat as possible. The question is: what non-negative sequence has the flattest autocorrelation?

This is related to "perfect autocorrelation" sequences, but in the non-negative continuous domain.

One approach: think of a as a "probability distribution" (since a >= 0 and we can normalize sum=1). Then conv(a,a) is the autocorrelation of this distribution. We want to minimize the peak of the autocorrelation.

For a uniform distribution (a_j = 1/n), the autocorrelation is triangular: conv_k = (n-|k|)/n^2 for |k| < n. Peak = 1/n at k=0. Objective = 2n * (1/n) = 2.

"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP alternating optimization, mirror descent, Adam smoothed gradient,
spectral shaping, and coordinate refinement with diverse initializations.
Key innovations over previous approach: mirror descent with exponential weights,
log-sum-exp smoothed gradients, spectral magnitude compression, and
iterated local search with random perturbations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            indices = np.arange(j_start, j_end + 1)
            A[k, indices] = a[k - indices]
    return A


def solve_lp_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_alternate(a_init, max_iter=200, verbose=False):
    """LP alternating with symmetric updates and adaptive blending."""
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for it in range(max_iter):
        conv_a = np.convolve(a, a)
        M = float(np.max(conv_a))
        if M < 1e-15:
            break

        g = solve_lp_response(a, M)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Symmetric: solve for a_new given g
        conv_g = np.convolve(g, g)
        M_g = float(np.max(conv_g))
        a_new = solve_lp_response(g, M_g)

        if a_new is not None:
            a_new_sum = np.sum(a_new)
            if a_new_sum > 1e-10:
                a_new = a_new / a_new_sum
            obj_g = compute_obj(g)
            obj_an = compute_obj(a_new)
            if obj_g < best_obj:
                best_obj = obj_g
                best_a = g.copy()
            if obj_an < best_obj:
                best_obj = obj_an
                best_a = a_new.copy()
            t = 0.4 + 0.4 * min(1.0, it / max_iter)
            a = (1 - t) * a + t * a_new
        else:
            obj_g = compute_obj(g)
            if obj_g < best_obj:
                best_obj = obj_g
                best_a = g.copy()
            a = g

        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s < 1e-10:
            a = g
        else:
            a = a / s

        if verbose and (it % 50 == 0):
            print(f"    LP it={it}: best={best_obj:.6f}")

    return best_a, best_obj


def mirror_descent(a_init, n_steps=10000, verbose=False):
    """Mirror descent (exponential weights) on the simplex."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for step in range(n_steps):
        conv = np.convolve(a, a)
        max_c = float(np.max(conv))
        if max_c < 1e-15:
            break

        tol = max_c * (0.02 + 0.03 * (step / n_steps))
        active = conv >= max_c - tol
        n_active = max(1, int(np.sum(active)))

        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                grad[j_start:j_end + 1] += 2.0 * a[k - (j_start:j_end + 1)]
        grad /= n_active

        eta = 0.5 / (1.0 + 0.005 * step)
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-15)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a = a_new / s

        if step % 1000 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a, best_obj


def adam_smoothed_grad(a_init, n_steps=5000, verbose=False):
    """Adam optimizer with active-set gradient on the simplex."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.8, 0.999
    eps = 1e-8
    lr0 = 0.05

    for step in range(n_steps):
        conv = np.convolve(a, a)
        max_c = float(np.max(conv))
        if max_c < 1e-15:
            break

        tol = max_c * (0.01 + 0.04 * (step / n_steps))
        active = conv >= max_c - tol
        n_active = max(1, int(np.sum(active)))

        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                grad[j_start:j_end + 1] += 2.0 * a[k - (j_start:j_end + 1)]
        grad /= n_active

        sum_a = np.sum(a)
        grad -= 2.0 / sum_a

        t = step + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr = lr0 / (1.0 + 0.002 * step)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a = a_new / s

        if step % 500 == 0:
            obj = compute_obj(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    obj = compute_obj(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a, best_obj


def spectral_shaping(a_init, n_iters=200, verbose=False):
    """Spectral magnitude compression to flatten the power spectrum."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for it in range(n_iters):
        A = np.fft.rfft(a)
        if n <= 1:
            break
        max_mag = np.max(np.abs(A[1:]))
        if max_mag < 1e-15:
            break

        alpha = 0.4 + 0.4 * (it / n_iters)
        norms = np.abs(A[1:]) / max_mag
        compressed = norms ** alpha
        phases = np.angle(A[1:])
        A_new = A.copy()
        A_new[1:] = compressed * max_mag * np.exp(1j * phases)

        a_new = np.fft.irfft(A_new, n=n)
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        obj = compute_obj(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new

    return best_a, best_obj


def coord_refine(a_init, n_rounds=5, verbose=False):
    """Coordinate descent with multi-point trial values."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for rnd in range(n_rounds):
        improved = False
        for j in range(n):
            old_val = a[j]
            best_x = old_val
            best_local = best_obj

            if old_val > 1e-10:
                candidates = [0.0, old_val * 0.2, old_val * 0.5, old_val * 0.7,
                              old_val, old_val * 1.3, old_val * 1.7, old_val * 2.5]
            else:
                mean_val = np.mean(a[a > 0]) if np.any(a > 0) else 1.0 / n
                candidates = [0.0, mean_val * 0.1, mean_val * 0.3, mean_val * 0.5, mean_val]

            for x in candidates:
                a[j] = x
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_x = x

            if best_x != old_val:
                a[j] = best_x
                improved = True

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        if not improved:
            break

    return best_a, best_obj


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 8))
    if choice == 0:
        a = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / max(1, n - 1)))
    elif choice == 1:
        m = np.arange(n)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * m / max(1, n - 1)) + 0.08 * np.cos(4 * np.pi * m / max(1, n - 1))
    elif choice == 2:
        m = np.arange(n)
        x = (m - (n - 1) / 2) / max(1, (n - 1) / 2)
        x = np.clip(x, -1, 1)
        a = np.i0(5 * np.sqrt(np.maximum(0, 1 - x ** 2))) / np.i0(5)
    elif choice == 3:
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0, np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
    elif choice == 4:
        mid = n // 2
        a = np.ones(n)
        a[:mid] = rng.uniform(0.3, 1.5)
        a[mid:] = rng.uniform(0.3, 1.5)
    elif choice == 5:
        period = int(rng.integers(max(2, n // 20), max(3, n // 3)))
        pattern = rng.random(period) ** rng.uniform(0.5, 3)
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 6:
        center = rng.uniform(0.2 * n, 0.8 * n)
        width = rng.uniform(n * 0.05, n * 0.3)
        a = np.exp(-((np.arange(n) - center) ** 2) / (2 * width ** 2))
    else:
        a = rng.random(n)

    s = np.sum(a)
    if s < 1e-10:
        a = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / max(1, n - 1)))
        s = np.sum(a)
    return a / s


def perturb_and_reoptimize(a_init, rng, max_iter_lp=50):
