import time

from scipy.sparse import csr_matrix

def compute_score(f):

    f = np.asarray(f, dtype=np.float64)

    if S < 0.01:
        return np.inf
    C = np.convolve(f, f)
    return 2.0 * n * float(np.max(C)) / (S * S)


def subgradient_optimize(f, deadline, verbose_prefix="", max_iters=100000):
    """Subgradient method with adaptive step size and momentum."""
    f = f.copy().astype(np.float64)
    n = len(f)
    S = float(np.sum(f))
    f = f / S  # normalize to sum=1
    
    best_f = f.copy()
    best_score = compute_score(f)
    
    # Momentum
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.99
    m = np.zeros(n)
    vv = np.zeros(n)
    
    # Adaptive step size
    eta = 1.0 / n  # initial step size
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        C = np.convolve(f, f)
        max_C = C.max()
        
        if max_C <= 1e-15:
            break
        
        # Find active lags (within 1% of max)
        threshold = max_C * 0.99
        active = C >= threshold
        
        # Compute subgradient
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if not active[k]:
                continue
            # ∂C[k]/∂f[i] for all i
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                # f[j] * f[k-j]: derivative w.r.t. f[j] is f[k-j], w.r.t. f[k-j] is f[j]
                i1 = j
                i2 = k - j
                if i1 == i2:
                    g[i1] += 2.0 * f[i1]
                else:
                    g[i1] += f[i2]
                    g[i2] += f[i1]
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        vv = beta2 * vv + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** (it + 1))
        vv_hat = vv / (1 - beta2 ** (it + 1))
        update = m_hat / (np.sqrt(vv_hat) + 1e-8)
        
        new_f = f - eta * update
        new_f = np.clip(new_f, 0.0, 1000.0)
        ns = float(np.sum(new_f))
        if ns < 1e-10:
            continue
        new_f = new_f / ns
        
        sc = compute_score(new_f)
        if sc < best_score:
            f = new_f
            best_f = f.copy()
            best_score = sc
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                eta *= 0.5
                if eta < 1e-10:
                    break
        
        if it % 5000 == 0 and it > 0:
            print(f"    {verbose_prefix}SG iter {it}: score={best_score:.6f}, eta={eta:.6f}")
    
    return best_f, best_score

Optimization of step function heights to minimize 2n*max(conv(f,f))/sum(f)^2.

    """Fast score computation (assumes valid f)."""
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return np.inf
    C = np.convolve(f, f)
    return 2.0 * n * float(np.max(C)) / (S * S)


def subgradient_step(f, active_lags, beta=1.0):
    """
    Compute subgradient of max_C(f) using active lags.
    Returns gradient vector g where g[i] = sum_{k in active} dC[k]/df[i].
    """
    n = len(f)
    # a[k] = 1 if k is active
    a = np.zeros(2 * n - 1, dtype=np.float64)
    a[active_lags] = 1.0

def compute_score_fast(f):
    """Fast internal score computation."""
    f = np.asarray(f, dtype=np.float64)
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return np.inf
    C = np.convolve(f, f)
    max_C = float(np.max(C))
    return 2.0 * n * max_C / (S * S)


def compute_gradient_maxC(f, eps_frac=0.01):
    """
    Compute subgradient of max(C[k]) using lags within eps_frac of max.
    Returns gradient vector g of length n.
    """
    n = len(f)
    C = np.convolve(f, f)
    max_C = C.max()
    if max_C <= 0:
        return np.zeros(n)
    
    # Use lags that are "close to max" with soft weighting
    # p[k] proportional to how close C[k] is to max
    threshold = max_C * (1.0 - eps_frac)
    weights = np.clip(C - threshold, 0, None) / max_C
    if weights.sum() < 1e-15:
        # Only the argmax
        weights = np.zeros(2 * n - 1)
        weights[int(np.argmax(C))] = 1.0
    else:
        weights = weights / weights.sum()
    
    # g[i] = sum_k weights[k] * (f[k-i] + f[i]) * I(0<=k-i<n)
    # = corr[i] + f[i] * count[i]
    # where corr[i] = sum_j f[j] * weights[i+j]
    # and count[i] = sum_j weights[i+j]
    
    # Cross-correlation: corr[i] = sum_{j=0}^{n-1} f[j] * a[i+j]
    # Using np.correlate(a, f, mode='full')[n-1:2*n-1]
    a = weights  # length 2n-1
    corr_full = np.correlate(a, f, mode='full')
    corr = corr_full[n - 1: 2 * n - 1]
    
    # Sliding window sum: count[i] = sum_{j=0}^{n-1} a[i+j]
    a_padded = np.concatenate([a, np.zeros(n)])
    prefix = np.concatenate([[0.0], np.cumsum(a_padded)])
    count = prefix[n:2*n] - prefix[0:n]
    
    g = corr + f * count
    return g


def compute_gradient_softmax(f, beta):
    """
    Compute gradient of log-sum-exp(beta * C) / (sum f)^2 (scale-invariant).
    Since we normalize f to sum=1, just compute gradient of soft-max(C).
    Returns (gradient, soft_max_value).

    n = len(f)
    C = np.convolve(f, f)
    max_C = C.max()
    
    # Stable log-sum-exp
    shifted = beta * (C - max_C)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-300:
        return np.zeros(n), 0.0
    p = exp_shifted / total
    
    # soft_max value (for reporting)
    soft_val = max_C + np.log(total) / beta
    
    # g[i] = sum_k p[k] * (f[k-i] + f[i]) * I(0<=k-i<n)
    # = corr[i] + f[i] * count[i]
    a = p  # length 2n-1

    prefix = np.concatenate([[0.0], np.cumsum(a_padded)])
    count = prefix[n:2*n] - prefix[0:n]
    
    g = corr + f * count
    return g, soft_val


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)

    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    return None

    corr = corr_full[n - 1: 2 * n - 1]
    
    a_padded = np.concatenate([p, np.zeros(n)])
    prefix = np.concatenate([[0.0], np.cumsum(a_padded)])
    count = prefix[n: 2 * n] - prefix[0: n]
    
    g = corr + f * count
    return g, soft_val


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f, g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((np.array(vals), (np.array(rows, dtype=np.int32),
                                         np.array(cols, dtype=np.int32))),
                      shape=(2 * n - 1, n))


def solve_lp_direction(f, M, time_limit=20):
    """Solve LP: max sum(g) s.t. g>=0, conv(f,g)_k <= M."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_refine(f, deadline, max_iters=50, step_init=0.1, verbose_prefix=""):
    """LP-based refinement with line search."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, compute_score_fast(f)
    f = f / S
    best_f = f.copy()
    best_score = compute_score_fast(f)
    step = step_init
    no_imp = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        C = np.convolve(f, f)
        M = float(C.max())
        if M <= 1e-15:
            break
        g = solve_lp_direction(f, M)
        if g is None or np.sum(g) < 1e-15:
            no_imp += 1
            if no_imp > 8:
                break
            step *= 0.7
            continue
        g = g / S
        
        best_t, best_s = step, best_score
        for t in [step, step * 1.5, step * 0.5, step * 2.0, step * 0.3]:
            nf = np.clip((1 - t) * f + t * g, 0, 1000)
            ns = nf.sum()
            if ns < 1e-10:
                continue
            nf = nf / ns
            sc = compute_score_fast(nf)
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = t
        
        if best_s < best_score - 1e-12:
            f = np.clip((1 - best_t) * f + best_t * g, 0, 1000)
            f = f / f.sum()
            best_f, best_score = f.copy(), best_s
            no_imp = 0
        else:
            no_imp += 1
            step *= 0.6
            if step < 0.003 or no_imp > 20:
                break
    
    return best_f, best_score


def adam_softmax_optimize(f, deadline, beta_init=3.0, beta_final=300.0,
                          lr_init=None, max_iters=100000, verbose_prefix=""):
    """Soft-max gradient descent with Adam optimizer."""
    f = f.copy().astype(np.float64)
    n = len(f)
    f = f / f.sum()
    if lr_init is None:
        lr_init = 2.0 / np.sqrt(n)
    
    best_f = f.copy()
    best_score = compute_score_fast(f)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    no_imp = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        frac = min(1.0, it / (max_iters * 0.75))
        beta = beta_init * (beta_final / beta_init) ** frac
        
        grad, _ = compute_grad_softmax(f, beta)
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        m_hat = m / (1 - b1 ** it)
        v_hat = v / (1 - b2 ** it)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        nf = f - lr * update
        nf = np.clip(nf, 0, 1000)
        ns = nf.sum()
        if ns < 1e-10:
            no_imp += 1
            continue
        nf = nf / ns
        
        sc = compute_score_fast(nf)
        if np.isfinite(sc) and sc < best_score:
            f = nf
            best_f, best_score = f.copy(), sc
            no_imp = 0
            if it % 20000 == 0:
                print(f"    {verbose_prefix}Adam it={it}: score={best_score:.6f}, beta={beta:.1f}")
        else:
            no_imp += 1
            if no_imp % 1000 == 0:
                lr *= 0.5
                if lr < 1e-10:
                    break
    
    return best_f, best_score


def lbfgs_refine(f, deadline, beta=50.0, max_iter=200, verbose_prefix=""):
    """L-BFGS-B on soft-max with exp parameterization."""
    f = f.copy().astype(np.float64)
    n = len(f)
    f = f / f.sum()
    
    best_f = f.copy()
    best_score = compute_score_fast(f)
    x0 = np.log(np.maximum(f, 1e-12))
    
    state = {'count': 0}
    
    def obj_and_grad(x, compute_grad=True):
        state['count'] += 1
        if time.monotonic() > deadline:
            return best_score + 1.0, np.zeros(n) if compute_grad else 0
        xs = x - x.max()
        fr = np.exp(xs)
        S = fr.sum()
        if S < 1e-10:
            return 1e10, np.zeros(n) if compute_grad else 0
        fn = fr / S
        
        C = np.convolve(fn, fn)
        maxC = C.max()
        shifted = beta * (C - maxC)
        es = np.exp(shifted)
        tot = es.sum()
        if tot < 1e-300:
            val = 0.0
        else:
            val = maxC + np.log(tot) / beta
        
        if not compute_grad:
            return val
        
        p = es / tot
        corr_full = np.correlate(p, fn, mode='full')
        corr = corr_full[n - 1: 2 * n - 1]
        ap = np.concatenate([p, np.zeros(n)])
        pref = np.concatenate([[0.0], np.cumsum(ap)])
        cnt = pref[n: 2 * n] - pref[0: n]
        gf = corr + fn * cnt
        gdot = np.dot(gf, fn)
        gx = fn * (gf - gdot)
        return val, gx
    
    try:
        def callback(xk):
            return time.monotonic() >= deadline
        
        res = minimize(lambda x: obj_and_grad(x, False), x0,
                       jac=lambda x: obj_and_grad(x, True)[1],
                       method='L-BFGS-B', callback=callback,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10})
        
        if res.x is not None:
            xs = res.x - res.x.max()
            fn = np.exp(xs)
            S = fn.sum()
            if S > 1e-10:
                fn = fn / S
                sc = compute_score_fast(fn)
                if np.isfinite(sc) and sc < best_score:
                    best_f, best_score = fn.copy(), sc
    except Exception as e:
        print(f"    {verbose_prefix}L-BFGS err: {e}")
    
    return best_f, best_score


def coord_descent(f, deadline, verbose_prefix=""):
    """Simple coordinate descent with multiplicative updates."""
    f = f.copy().astype(np.float64)
    n = len(f)
    f = f / f.sum()
    best_f = f.copy()
    best_score = compute_score_fast(f)
    no_sweep = 0
    
    while time.monotonic() < deadline and no_sweep < 5:
        improved = False
        rng = np.random.default_rng(time.monotonic() * 1000 % 100000)
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline:
                break
            t = f[i]
            if t < 1e-15:
                # Try turning on
                f_try = f.copy()
                f_try[i] = best_f.min() * 0.5
                f_try = f_try / f_try.sum()
                sc = compute_score_fast(f_try)
                if np.isfinite(sc) and sc < best_score:
                    f, best_f = f_try, f_try.copy()
                    best_score = sc
                    improved = True
                continue
            
            # Try scaling f[i] by factors
            for scale in [0.2, 0.5, 1.5, 2.0, 0.0]:
                f_try = f.copy()
                f_try[i] = t * scale
                s = f_try.sum()
                if s < 1e-10:
                    continue
                f_try = f_try / s
                sc = compute_score_fast(f_try)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    f = f_try
                    best_f = f.copy()
                    best_score = sc
                    improved = True
                    break
        
        no_sweep = 0 if improved else no_sweep + 1
        if no_sweep <= 2:
            print(f"    {verbose_prefix}CD sweep {no_sweep}: {best_score:.6f}")
    
    return best_f, best_score


# --- Initialization functions ---

def init_uniform(n):
    return np.ones(n) / n


def init_exponential(n, rate=0.3, direction=1):
    """Exponentially decaying/growing sequence."""
    i = np.arange(n, dtype=np.float64)
    if direction == 1:
        f = np.exp(-rate * i / n)
    else:
        f = np.exp(-rate * (1 - i / n))
    return f / f.sum()


def init_shifted_gaussian(n, center_frac=0.3, width_frac=0.15):
    """Gaussian bump shifted from center."""
    i = np.arange(n, dtype=np.float64)
    center = center_frac * n
    width = width_frac * n
    f = np.exp(-0.5 * ((i - center) / width) ** 2)
    return f / f.sum()


def init_tukey_asymmetric(n, r=0.4, shift=0.1):
    """Tukey window with asymmetric taper."""
    f = np.ones(n)
    m1 = max(1, int(r * n * (1 - shift) / 2))
    m2 = max(1, int(r * n * (1 + shift) / 2))
    for i in range(m1):
        f[i] *= 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m1) - 1)))
    for i in range(m2):
        f[n - 1 - i] *= 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m2) - 1)))
    return f / f.sum()


def init_random_smooth(n, seed=42):
    """Random smooth positive sequence."""
    rng = np.random.default_rng(seed)
    x = rng.standard_normal(n)
    kernel_size = max(3, n // 50)
    kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (kernel_size // 3 + 1)) ** 2)
    x = np.convolve(x, kernel, mode='same')
    f = np.abs(x) + 0.1
    return f / f.sum()


def init_ramp(n, power=1.0):
    """Power-law ramp: f[i] = (i+1)^power."""
    i = np.arange(n, dtype=np.float64) + 1
    f = i ** power
    return f / f.sum()


def init_bimodal(n, pos1=0.2, pos2=0.7, width=0.08):
    """Two Gaussian bumps at different positions."""
    i = np.arange(n, dtype=np.float64)
    f = (np.exp(-0.5 * ((i - pos1 * n) / (width * n)) ** 2) +
         0.7 * np.exp(-0.5 * ((i - pos2 * n) / (width * n)) ** 2))
    return f / f.sum()


def full_pipeline(f_init, deadline, verbose_prefix=""):
    """Run full optimization pipeline on a starting point."""
    f_init = f_init.copy().astype(np.float64)
    f_init = np.clip(f_init, 0, 1000)
    S = f_init.sum()
    if S < 1e-10:
        return f_init, compute_score_fast(f_init)
    f_init = f_init / S
    
    best_f = f_init.copy()
    best_score = compute_score_fast(f_init)
    n = len(f_init)
    remaining = deadline - time.monotonic()
    
    # Phase 1: Adam soft-max (main workhorse)
    t1 = time.monotonic() + remaining * 0.45
    if time.monotonic() < t1:
        f1, s1 = adam_softmax_optimize(f_init, t1, beta_init=3.0, beta_final=200.0,
                                        lr_init=3.0 / np.sqrt(n),
                                        verbose_prefix=verbose_prefix + "P1 ")
        if s1 < best_score:
            best_f, best_score = f1, s1
    
    # Phase 2: L-BFGS polish
    t2 = time.monotonic() + remaining * 0.25
    if timemonotonic_check(deadline) and time.monotonic() < t2:
        f2, s2 = lbfgs_refine(best_f, min(t2, deadline), beta=100.0, max_iter=300,
                              verbose_prefix=verbose_prefix + "P2 ")
        if s2 < best_score:
            best_f, best_score = f2, s2
    
    # Phase 3: LP refinement
    t3 = time.monotonic() + remaining * 0.15
    if time.monotonic() < t3 and time.monotonic() < deadline - 10:
        f3, s3 = lp_refine(best_f, min(t3, deadline - 10), max_iters=30,
                          step_init=0.08, verbose_prefix=verbose_prefix + "P3 ")
        if s3 < best_score:
            best_f, best_score = f3, s3
    
    # Phase 4: Coordinate descent
    if time.monotonic() < deadline - 5:
        f4, s4 = coord_descent(best_f, deadline - 5, verbose_prefix=verbose_prefix + "P4 ")
        if s4 < best_score:
            best_f, best_score = f4, s4
    
    # Phase 5: Final Adam with higher beta
    if time.monotonic() < deadline - 2:
        f5, s5 = adam_softmax_optimize(best_f, deadline - 2, beta_init=100.0,
                                        beta_final=500.0, lr_init=1.0 / np.sqrt(n),
                                        max_iters=50000, verbose_prefix=verbose_prefix + "P5 ")
        if s5 < best_score:
            best_f, best_score = f5, s5
    
    return best_f, best_score


def time_monotonic_check(deadline):
    return time.monotonic() < deadline


def propose_candidate(budget_s=1500, **kwargs):
    """Main search function. Returns best sequence found within budget."""
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 15.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Try to load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0, 1000)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                sc = evaluate_sequence(parent.tolist())
                if np.isfinite(sc):
                    best_score = sc
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={sc:.6f}")
    except (NameError, Exception) as e:
        print(f"No parent available: {e}")
    
    # Define search plan: (n, time_fraction)
    # Allocate time based on promise
    total_time = deadline - t0
    
    # Strategy: heavily focus on n=512 (parent length) and n=1024
    plan = [
        # (length, time_fraction, description)
        (512, 0.35, "parent improvement"),
        (1024, 0.25, "longer sequence"),
        (768, 0.12, "intermediate"),
        (2048, 0.13, "very long"),
        (256, 0.05, "short check"),
    ]
    
    # First: heavily refine the parent
    if best_sequence is not None and len(best_sequence) > 0:
        n_p = len(best_sequence)
        parent_deadline = t0 + total_time * 0.35
        print(f"\n=== Refining parent (n={n_p}, budget={parent_deadline - t0:.0f}s) ===")
        try:
            bf, bs = full_pipeline(best_sequence, parent_deadline, verbose_prefix="[parent] ")
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()
                print(f"  Parent refined: {bs:.6f}")
        except Exception as e:
            print(f"  Parent refinement error: {e}")
    
    # Now try other lengths
    for n, frac, desc in plan[1:]:
        if time.monotonic() > deadline - 60:
            print("Time running out, skipping remaining lengths")
            break
        
        n_deadline = time.monotonic() + total_time * frac
        print(f"\n=== n={n} ({desc}, budget={n_deadline - time.monotonic():.0f}s) ===")
        
        # Generate diverse initializations
        inits = []
        inits.append((f'exp1', init_exponential(n, rate=1.0, direction=1)))
        inits.append((f'exp2', init_exponential(n, rate=2.0, direction=1)))
        inits.append((f'exp3', init_exponential(n, rate=3.0, direction=1)))
        inits.append((f'gauss_0.2', init_shifted_gaussian(n, 0.2, 0.12)))
        inits.append((f'gauss_0.3', init_shifted_gaussian(n, 0.3, 0.15)))
        inits.append((f'gauss_0.4', init_shifted_gaussian(n, 0.4, 0.2)))
        inits.append((f'tukey_asym', init_tukey_asymmetric(n, 0.3, 0.3)))
        inits.append((f'ramp1', init_ramp(n, 1.0)))
        inits.append((f'ramp2', init_ramp(n, 2.0)))
        inits.append((f'bimodal', init_bimodal(n)))
        inits.append((f'rand1', init_random_smooth(n, seed=7)))
        inits.append((f'rand2', init_random_smooth(n, seed=13)))
        
        # If we have a best from similar length, include it
        if best_sequence is not None:
            if len(best_sequence) == n:
                inits.insert(0, ('best', best_sequence.copy()))
            else:
                # Interpolate
                from numpy import interp
                bi = interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(best_sequence)),
                           best_sequence)
                bi = np.clip(bi, 0, None)
                if bi.sum() > 1e-10:
                    inits.insert(0, ('best_interp', bi / bi.sum()))
        
        # Evaluate all inits quickly
        init_scores = []
        for label, fi in inits:
            sc = compute_score_fast(fi)
            init_scores.append((sc, label, fi))
        init_scores.sort(key=lambda x: x[0])
        
        # Only optimize the top inits (by init score)
        n_to_optimize = min(len(init_scores), max(3, int(total_time * frac / 60)))
        
        per_init_time = (n_deadline - time.monotonic() - 5) / max(1, n_to_optimize)
        
        for idx in range(n_to_optimize):
            if time.monotonic() > n_deadline - 10:
                break
            sc_init, label, fi = init_scores[idx]
            
            # Skip if init score is already much worse than best
            if sc_init > best_score * 1.10 and idx > 0:
                continue
            
            if sc_init <
