```python
import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def recursive_flat_init(n):
    """Generate a sequence whose self-convolution is as flat as possible (first half)."""
    c = 1.0 / (2 * n - 1)
    a = np.zeros(n)
    a[0] = np.sqrt(c)
    
    for k in range(1, n):
        cross = 0.0
        # sum_{j=1}^{floor((k-1)/2)} a[j]*a[k-j]
        for j in range(1, (k - 1) // 2 + 1):
            cross += 2.0 * a[j] * a[k - j]
        if k % 2 == 0:
            cross += a[k // 2] ** 2
        a[k] = (c - cross) / (2.0 * a[0])
    
    # Check if all non-negative
    if np.any(a < 0):
        # Find where it goes negative
        neg_idx = np.where(a < 0)[0]
        if len(neg_idx) > 0:
            first_neg = neg_idx[0]
            # Option 1: truncate
            a = a[:first_neg]
            # Re-normalize for the shorter length? Or zero-pad?
            # For now, zero-pad back to length n
            full_a = np.zeros(n)
            full_a[:len(a)] = a
            a = full_a
    
    s = np.sum(a)
    if s > 0:
        a = a / s
    return a

def build_conv_matrix(a):
    """Build matrix A such that (A @ g)[k] = conv(a, g)[k] for k=0,...,2n-2."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A

def lp_solve_g(a, M):
    """Solve: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_alternate(a_init, max_iter=200, verbose=False):
    a = a_init.copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(len(a))
        s = len(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        conv = np.convolve(a, a)
        M = np.max(conv)
        
        g = lp_solve_g(a, M)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Try different blending factors
        best_t_obj = best_obj
        best_t_a = best_a
        
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_new = (1 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_t_obj:
                best_t_obj = obj_new
                best_t_a = a_new.copy()
        
        if best_t_obj < best_obj:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
        
        a = best_t_a
        
        if verbose and it % 20 == 0:
            print(f"  LP iter {it}: obj={best_obj:.6f}")
    
    return best_a, best_obj
```

```python
"""
Optimization of step function minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses recursive flat-convolution initialization combined with LP alternating
optimization and coordinate descent polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def recursive_init(n):
    """Generate sequence with flat first-half convolution (coefficients of 1/sqrt(1-x))."""
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        cross = 0.0
        # 2*sum_{j=1}^{floor((k-1)/2)} a[j]*a[k-j]
        for j in range(1, (k - 1) // 2 + 1):
            cross += 2.0 * a[j] * a[k - j]
        if k % 2 == 0:
            cross += a[k // 2] ** 2
        a[k] = (1.0 - cross) / 2.0
        if a[k] < 0:
            a[k] = 0.0
    s = np.sum(a)
    if s > 0:
        a /= s
    else:
        a = np.ones(n) / n
    return a

def build_conv_matrix(a):
    """Build matrix A such that A @ g = conv(a, g)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A

def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_alternate(a_init, max_iter=300, t_values=None, verbose=False):
    """LP alternating optimization."""
    if t_values is None:
        t_values = [0.3, 0.5, 0.7, 1.0]
    
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(len(a))
        s = len(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        g = solve_lp(a, M)
        if g is None:
            break
        g_sum = float(np.sum(g))
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        best_t_obj = best_obj
        best_t_a = best_a
        
        for t in t_values:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            obj_new = compute_obj(a_new)
            if obj_new < best_t_obj:
                best_t_obj = obj_new
                best_t_a = a_new.copy()
        
        if best_t_obj < best_obj:
            best_obj = best_t_obj
            best_a = best_t_a.copy()
        
        a = best_t_a
        
        if verbose and (it % 50 == 0):
            print(f"    LP iter {it}: obj={best_obj:.8f}")
    
    return best_a, best_obj

def coord_descent(a, n_rounds=5, verbose=False):
    """Coordinate descent polish."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    # Precompute for efficiency
    for rnd in range(n_rounds):
        improved = False
        for j in range(n):
            old_val = a[j]
            # Try several values
            candidates = [0.0, old_val * 0.5, old_val * 0.8, old_val * 1.2, old_val * 1.5, 2.0 * old_val]
            best_x = old_val
            best_local = best_obj
            
            for x in candidates:
                a[j] = x
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_x = x
            
            if best_x != old_val:
                a[j] = best_x
                improved = True
        
        obj_now = compute_obj(a)
        if obj_now < best_obj:
            best_obj = obj_now
            best_a = a.copy()
        
        if not improved:
            break
    
    return best_a, best_obj

def propose_candidate(seed=42, budget_s=1500):
    """Main search function."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0
    
    best_seq = None
    best_score = np.inf
    
    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass
    
    # N values to try
    n_values = [100, 150, 200, 250, 300, 350, 400, 500]
    
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        print(f"\nN={n}: {remaining:.0f}s remaining, current best={best_score:.6f}")
        
        # Recursive initialization
        a_init = recursive_init(n)
        obj_init = compute_obj(a_init)
        print(f"  Recursive init: obj={obj_init:.6f}")
        
        if obj_init < best_score:
            best_score = obj_init
            best_seq = a_init.copy()
        
        # Also try a few random initializations
        inits = [a_init]
        for _ in range(2):
            r_init = rng.random(n)
            r_init = r_init / np.sum(r_init)
            inits.append(r_init)
        
        for init_idx, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            
            # LP alternating
            max_iter = int(min(300, max(50, remaining / 3.0)))
            a, obj = lp_alternate(a0, max_iter=max_iter, verbose=(init_idx == 0 and n <= 200))
            
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  LP n={n} init={init_idx}: obj={obj:.6f}")
            
            # Coordinate descent
            if time.monotonic() < deadline - 30 and n <= 400:
                a2, obj2 = coord_descent(a, n_rounds=3)
                if obj2 < best_score:
                    best_score = obj2
                    best_seq = a2.copy()
                    print(f"  Coord n={n} init={init_idx}: obj={obj2:.6f}")
        
        # If the recursive init is good, do more LP iterations
        if time.monotonic() < deadline - 30:
            a3 = recursive_init(n)
            a3, obj3 = lp_alternate(a3, max_iter=200, t_values=[0.5, 0.8, 1.0])
            if obj3 < best_score:
                best_score = obj3
                best_seq = a3.copy()
                print(f"  LP-extra n={n}: obj={obj3:.6f}")
    
    # Final polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic()
        print(f"\nFinal polish: n={n_best}, obj={best_score:.6f}, {remaining:.0f}s left")
        
        # More LP
        if n_best <= 400:
            a4, obj4 = lp_alternate(best_seq, max_iter=100, t_values=[0.5, 0.7, 1.0])
            if obj4 < best_score:
                best_score = obj4
                best_seq = a4.copy()
                print(f"  Final LP: obj={obj4:.6f}")
        
        # Coordinate descent
        if n_best <= 400 and time.monotonic() < deadline - 10:

            if obj5 < best_score:
                best_score = obj5
                best_seq = a5.copy()
                print(f"  Final Coord: obj={obj5:.6f}")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)
    
    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
```
