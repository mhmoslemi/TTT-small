def build_A_sparse(a):
    """Build sparse constraint matrix for LP: (a*g)_k = sum_j a[k-j]*g[j]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            rows.extend([k] * (jh - jl + 1))
            cols.extend(range(jl, jh + 1))
            # a[k-j] for j in [jl, jh]
            idx = k - np.arange(jl, jh + 1)
            vals.extend(a[idx].tolist())
    A = lil_matrix((2 * n - 1, n))
    if rows:
        A[np.array(rows), np.array(cols)] = vals
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_lp_iterations(a, max_iters, deadline, tracker):
    """LP mirror descent with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        # Try slightly below current M to push for improvement
        for M_factor in [1.0, 0.95, 0.90]:
            M_trial = M * M_factor
            g = lp_best_response(a, M_trial)
            if g is None:
                continue
            g = np.maximum(g, 0.0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                continue
            g /= g_sum

            # Adaptive step
            base_t = max(0.03, 0.4 * (1.0 - it / float(max(1, 2 * max_iters))))
            best_a = None
            best_J = current_J
            for t in [base_t, base_t * 1.5, base_t * 0.5, 0.01]:
                a_trial = (1.0 - t) * a + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                J_trial = compute_J(a_trial)
                if J_trial < best_J - 1e-14:
                    best_J = J_trial
                    best_a = a_trial

            if best_a is not None:
                a = best_a
                current_J = best_J
                break

        else:
            # No M_factor worked, try a small random perturbation
            if it > 10:
                break

        if it % 30 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def generate_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.2, 1.0, n))

    for freq in [1, 2, 3, 5, 7, 11]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2 * np.pi * freq * t), 0.01))

    inits.append(np.maximum(1.0 + 0.3 * np.cos(4 * np.pi * t * t), 0.01))
    inits.append(np.maximum(1.0 + 0.4 * np.sin(2 * np.pi * t * 3 + rng.uniform(0, 6.28)), 0.01))
    inits.append(np.maximum(np.exp(-4.0 * (t - 0.5) ** 2) + 0.1, 0.01))

    # Triangle wave
    tri = 1.0 - 4.0 * np.abs(t - 0.5)
    inits.append(np.maximum(1.0 + 0.3 * tri, 0.01))

    # Sparse structure
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=min(n_spikes, n), replace=False)
    sparse = np.zeros(n)
    sparse[positions] = 1.0
    kernel = np.ones(5) / 5.0
    inits.append(np.maximum(np.convolve(sparse, kernel, mode='same'), 0.01))

    # Power law
    inits.append(np.maximum((t + 0.05) ** rng.uniform(-1.5, 1.5), 0.01))

    # Bimodal
    inits.append(np.maximum(np.exp(-20*(t-0.25)**2) + np.exp(-20*(t-0.75)**2) + 0.05, 0.01))

    return inits


def subgradient_polish(a, n_steps=200, deadline=None):
    """Subgradient descent on max(b_k) with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-8)
    a /= a.sum()

    step_size = 0.1
    best_J = compute_J(a)
    best_a = a.copy()
    patience = 0

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        b = fast_conv(a)
        k_star = int(np.argmax(b))

        # Subgradient of b_{k*} w.r.t. a_j
        grad = np.zeros(n)
        for j in range(n):
            km = k_star - j
            if 0 <= km < n:
                if k_star == 2 * j:
                    grad[j] = 2.0 * a[j]
                else:
                    grad[j] = 2.0 * a[km]

        # Normalize subgradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-12:
            break

        # Take step
        a_new = a - step_size * grad / g_norm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s

        J_new = compute_J(a_new)
        if J_new < best_J - 1e-12:
            a = a_new
            best_J = J_new
            best_a = a.copy()
            patience = 0
            step_size = min(step_size * 1.1, 0.5)
        else:
            patience += 1
            step_size *= 0.9
            if patience > 30:
                break

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    1. Multi-length exploration with coordinate descent + LP mirror descent
    2. Iterated local search with perturbations
    3. Subgradient polishing
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            if parent.sum() > 0.01:
                tracker.update(parent)
                print(f"Parent loaded: n={parent.size}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-length exploration ===
    lengths = [64, 128, 192, 256, 384, 512, 768, 1024]

    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1 n={n} {'='*50}", flush=True)

        inits = generate_inits(n, rng, parent)
        # Budget per init based on remaining time
        remaining = max(0, deadline - time.monotonic())
        time_per_init = max(5.0, remaining / (len(inits) * 2))

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break

            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 /= a0.sum()

            # Coordinate descent first (fast local opt)
            a1 = coordinate_descent(a0, max_sweeps=100, deadline=deadline)
            tracker.update(a1)

            # LP mirror descent (global moves)
            lp_iters = max(30, min(200, int(60000 // n)))
            a2 = run_lp_iterations(a1, lp_iters, deadline, tracker)

            # Coordinate descent again (refine after LP)
            a3 = coordinate_descent(a2, max_sweeps=150, deadline=deadline)
            tracker.update(a3)

            print(f"  init {idx}: J={compute_J(a3):.8f}", flush=True)

        tracker.report(f"Phase1 n={n} ({time.monotonic()-t0:.1f}s)")

    # === Phase 2: Iterated local search from best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Phase2: ILS n={n_best} {'='*50}", flush=True)

        temperature = 0.1
        n_restarts = 100
        best_improvement = 0.0

        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break

            a = tracker.best.copy().astype(np.float64)

            # Perturbation: random coordinate changes
            n_perturb = max(3, n_best // 10)
            pert_indices = rng.choice(n_best, size=n_perturb, replace=False)
            for pi in pert_indices:
                a[pi] = max(0.0, a[pi] * rng.uniform(0.3, 2.0))

            # Also try global noise
            if r % 5 == 0:
                a = np.maximum(a + rng.normal(0, temperature, n_best), 0.001)

            a /= a.sum()

            # Coordinate descent
            a_opt = coordinate_descent(a, max_sweeps=80, deadline=deadline)
            tracker.update(a_opt)

            # Short LP
            if r % 3 == 0:
                a_opt = run_lp_iterations(a_opt, 30, deadline, tracker)
                a_opt = coordinate_descent(a_opt, max_sweeps=50, deadline=deadline)
                tracker.update(a_opt)

            temperature *= 0.98

        tracker.report(f"Phase2 ILS ({time.monotonic()-t0:.1f}s)")

    # === Phase 3: Try different lengths from best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase3: Length exploration {'='*50}", flush=True)

        n_best = tracker.best.size
        # Try nearby lengths
        candidate_lengths = set()
        for factor in [0.75, 0.875, 1.0, 1.125, 1.25, 1.5, 2.0]:
            nl = int(n_best * factor)
            if 32 <= nl <= 2048:
                candidate_lengths.add(nl)
        candidate_lengths.discard(n_best)

        for n_trial in sorted(candidate_lengths):
            if time.monotonic() > deadline:
                break

            a = resample_seq(tracker.best, n_trial)
            a = np.maximum(a, 0.001)
            a /= a.sum()

            a = coordinate_descent(a, max_sweeps=100, deadline=deadline)
            tracker.update(a)

            a = run_lp_iterations(a, 50, deadline, tracker)
            a = coordinate_descent(a, max_sweeps=100, deadline=deadline)
            tracker.update(a)

            print(f"  n={n_trial}: J={compute_J(a):.8f}", flush=True)

        tracker.report(f"Phase3 ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Subgradient polishing ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase4: Subgradient polish {'='*50}", flush=True)

        a = tracker.best.copy()
        n_steps = min(500, int(300_000 / max(1, a.size)))
        a = subgradient_polish(a, n_steps=n_steps, deadline=deadline)
        tracker.update(a)

        # Final coordinate descent
        a = coordinate_descent(a, max_sweeps=200, deadline=deadline)
        tracker.update(a)

        tracker.report(f"Phase4 ({time.monotonic()-t0:.1f}s)")

    # === Phase 5: Final LP polish on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase5: Final LP polish {'='*50}", flush=True)
        n_final = tracker.best.size
        if n_final <= 1024:
            a = run_lp_iterations(tracker.best.copy(), 100, deadline, tracker)
            a = coordinate_descent(a, max_sweeps=200, deadline=deadline)
            tracker.update(a)
        tracker.report(f"Phase5 ({time.monotonic()-t0:.1f}s)")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
