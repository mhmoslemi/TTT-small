"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Subgradient descent with active peak tracking (fast)

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix, lil_matrix
import time

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))

def build_active_A(a, active_lags):
    """Build sparse constraint matrix for only active lags."""
    n = len(a)
    K = len(active_lags)
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(active_lags):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([i] * cnt)
            cols.extend(range(jl, jh + 1))
            # a[k-j] for j in [jl, jh]
            # when j=jl, k-j = k-jl; when j=jh, k-j = k-jh
            # so we need a[k-jh : k-jl+1] reversed
            idx = k - np.arange(jl, jh+1)
            vals.extend(a[idx].tolist())
    A = lil_matrix((K, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()

def lp_best_response_active(a, M, active_lags):
    """LP: max sum(g) s.t. (a*g)_k <= M for k in active_lags, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_active_A(a, active_lags)
    b_ub = np.full(len(active_lags), M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None

def subgradient_step(a, lr, frac_active=0.1):
    """One subgradient step targeting active peaks."""
    n = len(a)
    b = fast_conv(a)
    M = b.max()
    s = a.sum()
    
    # Find active lags (top fraction of peaks)
    threshold = M * (1 - frac_active * 0.1)  # Use lags within 10% of max
    active = np.where(b > threshold)[0]
    if len(active) == 0:
        active = [int(np.argmax(b))]
    
    # Subgradient: for each active peak k, the gradient w.r.t. a_j is a_{k-j}
    # Average over active peaks
    grad = np.zeros(n)
    for k in active:
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        idx = k - np.arange(jl, jh + 1)
        grad[jl:jh+1] += a[idx]
    grad /= len(active)
    
    # Normalize to account for sum constraint
    # dJ/da_j ∝ (2*M/s - adj_j) where adj_j is the "adjoint"
    # With s=1: direction = 2*M - grad_j (but we want to minimize, so subtract)
    direction = 2.0 * M - grad  # This should decrease J
    
    # Project and update
    a_new = a + lr * direction
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new, M

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated subgradient descent with active peak tracking
2. LP mirror descent with adaptive active-set constraints
3. L-BFGS-B smooth refinement

Key improvements: active-set LP, Nesterov acceleration, diverse inits, adaptive active frac.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_active_A(a, active_lags):
    """Build sparse constraint matrix for only active lags."""
    n = len(a)
    K = len(active_lags)
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(active_lags):
        k = int(k)
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            idx = k - np.arange(jl, jh + 1)
            cnt = jh - jl + 1
            rows.extend([i] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[idx].tolist())
    A = lil_matrix((K, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response_active(a, M, active_lags):
    """LP: max sum(g) s.t. (a*g)_k <= M for k in active_lags, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_active_A(a, active_lags)
    b_ub = np.full(len(active_lags), M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def compute_subgradient(a, frac_active=0.05):
    """Compute averaged subgradient over active peaks."""
    n = len(a)
    b = fast_conv(a)
    M = float(b.max())
    threshold = M * (1.0 - frac_active)
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        idx = k - np.arange(jl, jh + 1)
        grad[jl:jh + 1] += a[idx]
    grad /= len(active)
    return grad, M, active


def nesterov_subgradient_step(a, v, lr, k, frac_active=0.05):
    """Nesterov-accelerated subgradient step."""
    n = len(a)
    # Nesterov lookahead
    if k == 0:
        y = a.copy()
    else:
        beta = (k - 1.0) / (k + 2.0)
        y = a + beta * (a - v)
        y = np.maximum(y, 0.0)
        y /= y.sum()

    grad, M, _ = compute_subgradient(y, frac_active)
    s = y.sum()
    # Direction: we want to minimize 2*n*max(b)/s^2
    # d/da_j [max(b)/s^2] ~ (grad_j * s^2 - 2*max(b)*s) / s^4 = (grad_j - 2*max(b)/s) / s^2
    # With s=1: direction = grad_j - 2*max(b)
    # We want to decrease, so: a_new = y - lr * (grad - 2*M/s)
    direction = grad - 2.0 * M / s
    a_new = y - lr * direction
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)

    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    """Multi-phase optimization: Nesterov subgradient + LP + L-BFGS-B with multi-start."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Nesterov Subgradient (fast exploration) ===
    lengths = [128, 256, 512, 1024, 2048]

    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        remaining = deadline - time.monotonic()
        print(f"\n{'='*40} Phase 1: Nesterov n={n} (budget {remaining:.0f}s) {'='*40}", flush=True)

        inits = generate_initializations(n, rng, parent)
        n_inits = len(inits)
        max_iters = max(100, int(2000000 // n))

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            s0 = a0.sum()
            if s0 < 0.01:
                continue
            a0 /= s0
            a = run_nesterov_subgradient(a0, max_iters, deadline, tracker)

        tracker.report(f"Phase1 n={n}: {time.monotonic() - t0:.1f}s")

    # === Phase 2: LP Mirror Descent (precise) ===
    if tracker.best is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        print(f"\n{'='*40} Phase 2: LP (budget {remaining:.0f}s) {'='*40}", flush=True)

        n_best = tracker.best.size
        lp_lengths = [n_best]
        if n_best >= 256:
            lp_lengths.append(n_best // 2)
        if n_best <= 2048:
            lp_lengths.append(min(4096, n_best * 2))

        for n in lp_lengths:
            if time.monotonic() > deadline:
                break
            t0 = time.monotonic()

            inits = generate_initializations(n, rng, tracker.best)
            max_iters = max(30, int(1500000 // n))

            for idx, a0 in enumerate(inits[:8]):
                if time.monotonic() > deadline:
                    break
                a0 = np.maximum(a0.astype(np.float64), 0.001)
                s0 = a0.sum()
                if s0 < 0.01:
                    continue
                a0 /= s0
                a = run_lp_iterations(a0, max_iters, deadline, tracker, active_frac=0.05)

            tracker.report(f"Phase2 n={n}: {time.monotonic() - t0:.1f}s")

    # === Phase 3: L-BFGS-B refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        remaining = deadline - time.monotonic()
        print(f"\n{'='*40} Phase 3: L-BFGS-B n={n} (budget {remaining:.0f}s) {'='*40}", flush=True)

        bounds_list = [(0.0, 1.0)] * n
        max_fun = max(2000, min(int(remaining * 300), 40000))

        res = minimize(lbfgs_objective, tracker.best.astype(np.float64), method='L-BFGS-B',
                       bounds=bounds_list,
                       options={'maxiter': 300, 'max_fun': max_fun, 'ftol': 1e-16, 'gtol': 1e-14})
        if res.fun < 1e10:
            tracker.update(res.x)

        tracker.report(f"L-BFGS-B: {time.monotonic() - t0:.1f}s")

    # === Phase 4: Final LP polish (tight active set) ===
    if tracker.best is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        n = tracker.best.size
        if n <= 4096:
            t0 = time.monotonic()
            print(f"\n{'='*40} Phase 4: Final LP n={n} (budget {remaining:.0f}s) {'='*40}", flush=True)
            max_iters = max(50, int(remaining * 30))
            max_iters = min(max_iters, 400)
            a = run_lp_iterations(tracker.best.copy(), max_iters, deadline, tracker, active_frac=0.03)
            tracker.report(f"Final LP: {time.monotonic() - t0:.1f}s")

    # === Phase 5: Second L-BFGS-B pass ===
    if tracker.best is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t0 = time.monotonic()
            n = tracker.best.size
            print(f"\n{'='*40} Phase 5: L-BFGS-B pass 2 n={n} {'='*40}", flush=True)
            bounds_list = [(0.0, 1.0)] * n
            max_fun = max(1000, min(int(remaining * 200), 20000))
            res = minimize(lbfgs_objective, tracker.best.astype(np.float64), method='L-BFGS-B',
                           bounds=bounds_list,
                           options={'maxiter': 200, 'max_fun': max_fun, 'ftol': 1e-16, 'gtol': 1e-14})
            if res.fun < 1e10:
                tracker.update(res.x)
            tracker.report(f"Pass 2: {time.monotonic() - t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic() - tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
