```python
def peak_reduction_step(f, n_peaks=5, strength=0.1):
    R = np.convolve(f, f)
    max_R = R.max()
    # Find top n_peaks indices
    top_indices = np.argsort(R)[-n_peaks:]
    # For each i, compute its total contribution to these peaks
    contrib = np.zeros(len(f))
    for k in top_indices:
        for i in range(max(0, k-len(f)+1), min(k+1, len(f))):
            j = k - i
            if 0 <= j < len(f):
                contrib[i] += f[i] * f[j]
    # Apply reduction
    f_new = f * (1 - strength * contrib / contrib.max())
    f_new = np.clip(f_new, 0, None)
    f_new = f_new / f_new.sum()
    return f_new
```

```python
def lp_direction_reduced(f, M, n_constraints=None):
    """Solve LP with only top n_constraints constraints."""
    n = len(f)
    R = np.convolve(f, f)
    if n_constraints is None or n_constraints >= 2*n-1:
        k_indices = list(range(2*n-1))
    else:
        k_indices = np.argsort(R)[-n_constraints:].tolist()
    
    # Build constraint matrix for selected k_indices
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_indices):
        j_min = max(0, k - (n-1))
        j_max = min(k, n-1)
        for j in range(j_min, j_max+1):
            rows.append(idx)
            cols.append(j)
            vals.append(f[k-j])
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(len(k_indices), n))
    
    c = -np.ones(n)
    b_ub = np.full(len(k_indices), M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None

def lp_optimize_reduced(f, deadline, step_size=0.1, max_iters=100, n_constraints=None):

            break

        it += 1

            if no_improve > 15:

            continue

        best_t = ss

def peak_shaving(f, strength=0.05, n_peaks=5):
    """Multiplicative peak reduction."""
    R = np.convolve(f, f)
    max_R = R.max()
    if max_R < 1e-15:
        return f
    # Find top peaks
    threshold = max_R * 0.8
    hot_indices = np.where(R >= threshold)[0]
    
    contrib = np.zeros(len(f))
    for k in hot_indices:
        j_min = max(0, k - (len(f)-1))
        j_max = min(k, len(f)-1)
        for j in range(j_min, j_max+1):
            contrib[j] += f[j] * f[k-j]
    
    max_contrib = contrib.max()
    if max_contrib < 1e-15:
        return f
    
    f_new = f * (1 - strength * contrib / max_contrib)
    f_new = np.clip(f_new, 0, None)
    S = np.sum(f_new)
    if S < 1e-10:
        return f
    return f_new / S

def window_hann(n):
    return 0.5 * (1 + np.cos(2*np.pi*np.arange(n)/(n-1))) if n > 1 else np.ones(1)

def window_tukey(n, r=0.3):
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5*(1 + np.cos(np.pi*(2*i/(2*m) - 1)))
            w[n-1-i] = w[i]
    return w

def window_blackman(n):
    if n < 3:
        return np.ones(n)
    n = max(n, 3)
    w = np.ones(n)
    for i in range(n):
        w[i] = 0.42 - 0.5*np.cos(2*np.pi*i/(n-1)) + 0.08*np.cos(4*np.pi*i/(n-1))
    return np.maximum(w, 0)

def window_kaiser(n, beta=5.0):
    from scipy.special import i0
    if n < 2:
        return np.ones(n)
    w = np.ones(n)
    for i in range(n):
        alpha = (i - (n-1)/2) / ((n-1)/2)
        w[i] = i0(beta*np.sqrt(1-alpha**2)) / i0(beta)
    return w

def window_super_gauss(n, alpha=2.0):
    idx = np.arange(n, dtype=np.float64)
    center = (n-1)/2.0
    w = np.exp(-((idx - center) / (n/2.0)) ** (2*alpha))
    return w
```

```python
def compute_logsumexp_gradient(f, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*R[k])) where R=conv(f,f)."""
    n = len(f)
    R = np.convolve(f, f)
    max_R = np.max(R)
    shifted = R - max_R
    exp_R = np.exp(beta * shifted)
    total = np.sum(exp_R)
    if total < 1e-300:
        return np.zeros(n), max_R
    p = exp_R / total
    corr = np.correlate(p, f, mode='valid')
    grad = 2.0 * corr
    smooth_val = max_R + np.log(total) / beta
    return grad, smooth_val

def build_conv_matrix_sparse(f, k_indices=None):
    """Build sparse matrix A where (A@g)[idx] = convolve(f,g)[k_indices[idx]]."""
    n = len(f)
    if k_indices is None:
        k_indices = list(range(2 * n - 1))
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_indices):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(len(k_indices), n))


def lp_direction_reduced(f, M, n_constraints=None):
    """Solve LP: max sum(g) s.t. g>=0, selected conv(f,g)_k <= M."""
    n = len(f)
    R = np.convolve(f, f)
    if n_constraints is not None and n_constraints < 2 * n - 1:
        k_indices = np.argsort(R)[-n_constraints:].tolist()
    else:
        k_indices = list(range(2 * n - 1))

    A = build_conv_matrix_sparse(f, k_indices)
    c = -np.ones(n)
    b_ub = np.full(len(k_indices), M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None

def peak_shaving(f, strength=0.04, threshold_frac=0.85):
    """Multiplicative peak reduction on hot autocorrelation lags."""
    n = len(f)
    R = np.convolve(f, f)
    max_R = np.max(R)
    if max_R < 1e-15:
        return f
    threshold = max_R * threshold_frac
    hot_indices = np.where(R >= threshold)[0]

    contrib = np.zeros(n)
    for k in hot_indices:
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            contrib[j] += f[j] * f[k - j]

    max_contrib = contrib.max()
    if max_contrib < 1e-15:
        return f

    f_new = f * (1.0 - strength * contrib / max_contrib)
    f_new = np.clip(f_new, 0.0, None)
    S = np.sum(f_new)
    if S < 1e-10:
        return f
    return f_new / S


def window_hann(n):
    if n < 2:
        return np.ones(max(n, 1))
    return 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))


def window_tukey(n, r=0.3):
    w = np.ones(n)
    m = max(1, int(np.floor(r * n / 2)))
    for i in range(m):
        val = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
        w[i] = val
        w[n - 1 - i] = val
    return w


def window_blackman(n):
    if n < 3:
        return np.ones(max(n, 1))
    w = np.zeros(n)
    for i in range(n):
        w[i] = 0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) + 0.08 * np.cos(4 * np.pi * i / (n - 1))
    return np.maximum(w, 0.0)


def window_kaiser(n, beta=5.0):
    if n < 2:
        return np.ones(max(n, 1))
    w = np.zeros(n)
    for i in range(n):
        alpha = (i - (n - 1) / 2.0) / ((n - 1) / 2.0)
        if abs(alpha) < 1.0:
            w[i] = i0(beta * np.sqrt(1 - alpha * alpha)) / i0(beta)
        else:
            w[i] = 0.0
    return w


def window_super_gauss(n, alpha=2.0):
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    w = np.exp(-((idx - center) / (n / 2.0)) ** (2 * alpha))
    return w


def window_asymmetric(n, shift_frac=0.1):
    """Create an asymmetric window by shifting the center."""
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0 + shift_frac * (n - 1) / 2.0
    width = n / 3.0
    w = np.exp(-0.5 * ((idx - center) / width) ** 2)
    w = w + 0.05
    return w


def run_optimization_pipeline(f_init, opt_deadline, phase_fractions=(0.35, 0.25, 0.20, 0.20)):
    """Run multi-phase optimization: Nesterov -> LP -> PeakShave -> Polish."""
    f_init = f_init.copy().astype(np.float64)
    S = np.sum(f_init)
    if S < 1e-10:
        return f_init, float('inf')
    f_init = f_init / S

    best_f = f_init.copy()
    best_score = evaluate_sequence(best_f.tolist())
    t_start = time.monotonic()
    total = opt_deadline - t_start
    p1, p2, p3, p4 = phase_fractions

    # Phase 1: Nesterov log-sum-exp
    d1 = t_start + total * p1
    if time.monotonic() < d1 - 2:
        f1, s1 = nesterov_logsumexp_optimize(best_f, d1, beta_start=10, beta_end=300,
                                              lr_start=0.04, max_iters=3000)
        if s1 < best_score:
            best_f, best_score = f1, s1

    # Phase 2: Reduced LP
    d2 = t_start + total * (p1 + p2)
    if time.monotonic() < d2 - 2:
        n_cons = min(len(best_f), 48)
        f2, s2 = lp_optimize_reduced(best_f, d2, step_size=0.08, max_iters=40,
                                      n_constraints=n_cons)
        if s2 < best_score:
            best_f, best_score = f2, s2

    # Phase 3: Peak shaving iterations
    d3 = t_start + total * (p1 + p2 + p3)
    if time.monotonic() < d3 - 2:
        current = best_f.copy()
        for _ in range(100):
            if time.monotonic() > d3 - 2:
                break
            f_ps = peak_shaving(current, strength=0.03, threshold_frac=0.85)
            sc_ps = evaluate_sequence(f_ps.tolist())
            if np.isfinite(sc_ps) and sc_ps < best_score:
                best_score = sc_ps
                best_f = f_ps.copy()
                current = f_ps
            else:
                break

    # Phase 4: High-beta Nesterov polish
    d4 = opt_deadline
    if time.monotonic() < d4 - 2:
        f4, s4 = nesterov_logsumexp_optimize(best_f, d4, beta_start=200, beta_end=1500,
                                              lr_start=0.005, max_iters=2000, mu=0.85)
        if s4 < best_score:
            best_f, best_score = f4, s4

    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(42)

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = np.sum(parent)
            if S > 0.01:
                score = evaluate_sequence(parent.tolist())
                if np.isfinite(score):
                    best_score = score
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={score:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # Lengths and time allocation
    lengths_to_try = [512, 640, 384, 768, 1024]
    total_budget = deadline - t0
    time_fracs = {512: 0.30, 640: 0.18, 384: 0.10, 768: 0.17, 1024: 0.25}

    for n in lengths_to_try:
        if time.monotonic() > deadline - 50:
            print("Time running low, stopping length exploration")
            break

        n_budget = total_budget * time_fracs.get(n, 0.1)
        n_deadline = time.monotonic() + n_budget
        print(f"\n=== n={n} (budget {n_budget:.0f}s) ===")

        # Generate diverse initializations
        inits = []
        inits.append(('hann', window_hann(n)))
        inits.append(('tukey02', window_tukey(n, 0.2)))
        inits.append(('tukey035', window_tukey(n, 0.35)))
        inits.append(('tukey05', window_tukey(n, 0.5)))
        inits.append(('blackman', window_blackman(n)))
        inits.append(('kaiser4', window_kaiser(n, 4.0)))
        inits.append(('kaiser6', window_kaiser(n, 6.0)))
        inits.append(('sg2', window_super_gauss(n, 2.0)))
        inits.append(('sg25', window_super_gauss(n, 2.5)))
        inits.append(('asym01', window_asymmetric(n, 0.1)))
        inits.append(('asym02', window_asymmetric(n, 0.15)))

        # Add parent if available
        if best_sequence is not None:
            if len(best_sequence) == n:
                inits.insert(0, ('parent', best_sequence.copy()))
            else:
                pi = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(best_sequence)),
                               best_sequence)
                pi = np.clip(pi.astype(np.float64), 0.0, None)
                inits.insert(0, ('parent_interp', pi))

        # Score all initializations
        scored_inits = []
        for label, f_init in inits:
            f_init = np.clip(f_init.astype(np.float64), 0.0, None)
            S = np.sum(f_init)
            if S < 0.01:
                continue
            f_init = f_init / S
            sc = evaluate_sequence(f_init.tolist())
            if np.isfinite(sc):
                scored_inits.append((sc, label, f_init))

        scored_inits.sort(key=lambda x: x[0])

        # Show top scores
        for sc, label, _ in scored_inits[:5]:
            print(f"  Init [{label}]: {sc:.6f}")

        # Optimize top initializations
        n_optimize = min(4, len(scored_inits))
        per_opt_time = n_budget / max(n_optimize, 1) * 0.8

        for sc_init, label, f_init in scored_inits[:n_optimize]:
            if time.monotonic() > n_deadline - 5:
                break

            if sc_init < best_score:
                best_score = sc_init
                best_sequence = f_init.copy()

            opt_deadline = min(n_deadline, time.monotonic() + per_opt_time)
            if time.monotonic() > opt_deadline - 5:
                continue

            print(f"  Optimizing [{label}] init={sc_init:.6f} (time={per_opt_time:.0f}s)")

            try:
                bf, bs = run_optimization_pipeline(f_init, opt_deadline)
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"    -> {bs:.6f}")
                else:
                    print(f"    -> no improvement (best still {best_score:.6f})")
            except Exception as e:
                print(f"    Error: {e}")

        print(f"  After n={n}: best={best_score:.6f}")

    # Perturbation restarts on best found
    if best_sequence is not None and time.monotonic() < deadline - 40:
        print(f"\n=== Perturbation restarts (best: {best_score:.6f}) ===")
        restart_budget = deadline - time.monotonic() - 20
        n_restarts = min(8, max(3, int(restart_budget / 30)))
        per_restart = restart_budget / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break

            strength = 0.02 + 0.025 * r
            noise = 1.0 + strength * rng.standard_normal(len(best_sequence))
            f_pert = best_sequence * np.clip(noise, 0.4, 1.6)
            f_pert = np.clip(f_pert, 0.0, None)
            S = np.sum(f_pert)
            if S < 0.01:
                continue
            f_pert = f_pert / S

            r_deadline = time.monotonic() + per_restart * 0.8
            try:
                f_r, s_r = nesterov_logsumexp_optimize(f_pert, r_deadline,
                                                        beta_start=50, beta_end=500,
                                                        lr_start=0.01, max_iters=2000)
                if s_r < best_score:
                    best_score = s_r
                    best_sequence = f_r.copy()
                    print(f"  Restart {r} (str={strength:.3f}): {s_r:.6f}")
                else:
                    print(f"  Restart {r} (str={strength:.3f}): {s_r:.6f} (no improve)")
            except Exception as e:
                print(f"  Restart {r} error: {e}")

    # Final refinement
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Final refinement (n={len(best_sequence)}, score={best_score:.6f}) ===")

        # Final LP
        if time.monotonic() < deadline - 10:
            try:
                n_cons = min(len(best_sequence), 32)
                f_final, s_final = lp_optimize_reduced(best_sequence, deadline - 8,
                                                        step_size=0.03, max_iters=30,
                                                        n_constraints=n_cons)
                if s_final < best_score:
                    best_score = s_final
                    best_sequence = f_final.copy()
                    print(f"  After final LP: {best_score:.6f}")
            except Exception as e:
                print(f"  Final LP error: {e}")

        # Final high-beta Nesterov
        if time.monotonic() < deadline - 5:
            try:
                f_fp, s_fp = nesterov_logsumexp_optimize(best_sequence, deadline - 3,
                                                          beta_start=500, beta_end=3000,
                                                          lr_start=0.002, max_iters=1500,
                                                          mu=0.85)
                if s_fp < best_score:
                    best_score = s_fp
                    best_sequence = f_fp.copy()
                    print(f"  After final Nesterov: {best_score:.6f}")
            except Exception:
                pass

        # Final peak shaving
        if time.monotonic() < deadline - 3:
            current = best_sequence.copy()
            for _ in range(50):
                if time.monotonic() > deadline - 3:
                    break
                f_ps = peak_shaving(current, strength=0.015, threshold_frac=0.9)
                sc_ps = evaluate_sequence(f_ps.tolist())
                if np.isfinite(sc_ps) and sc_ps < best_score:
                    best_score = sc_ps
                    best_sequence = f_ps.copy()
                    current = f_ps
                else:
                    break

        print(f"  FINAL: n={len(best_sequence)}, score={best_score:.6f}")

    if best_sequence is None:
        best_sequence = window_hann(512)
        best_score = evaluate_sequence(best_sequence.tolist())

    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = np.sum(best_sequence)
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nDONE: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
```
