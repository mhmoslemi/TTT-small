"""
Optimized step function search using linear-ramp initialization,
smoothed subgradient descent, and LP refinement.
Key insight: monotonic (non-symmetric) sequences have lower peak-to-sum ratios
in their self-convolution than symmetric sequences.
"""
import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def smoothed_subgradient_step(a, tau, step):
    """One step of smoothed subgradient descent."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = float(np.max(conv))
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum
    # gradient of max(conv) w.r.t. a[i] is 2 * sum_k weights[k] * a[k-i] (for valid k-i)
    # This is 2 * correlate(weights, a, 'full')[i:huh]
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[:n]
    # Center for sum constraint
    grad_centered = grad - np.mean(grad)
    a_new = a - step * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s < 1e-10:
        return a
    return a_new / s


def subgradient_optimize(a, max_iter=20000, tau_start=5.0, tau_end=50.0, step_init=0.1):
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    step = step_init
    stall = 0
    patience = 300
    
    for it in range(max_iter):
        frac = it / max(1, max_iter)
        tau = tau_start + (tau_end - tau_start) * frac
        a = smoothed_subgradient_step(a, tau, step)
        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                step *= 0.8
                if step < 1e-9:
                    break
                # Add noise to escape
                rng = np.random.default_rng(it * 12345 + 67890)
                noise = rng.normal(0, 0.02, n)
                a = a + noise * step
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s < 1e-10:
                    break
                a = a / s
                continue
        if it > 100:
            step *= 0.9997
    
    return best_a, best_obj


def build_A_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def solve_lp_for_g(a, M):
    n = len(a)
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None


def lp_optimize(a, max_iter=30, t=0.3):
    n = len(a)
    a = a.copy()
    s = np.sum(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    stall = 0
    for it in range(max_iter):
        conv = compute_conv(a)
        M = float(np.max(conv))
        g, sum_g = solve_lp_for_g(a, M)
        if g is None:
            break
        if sum_g <= 1.0 + 1e-4:
            stall += 1
            if stall > 15:
                break
            continue
        stall = 0
        g = g / sum_g
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new
    return best_a, best_obj


def generate_candidates(n, rng):
    """Generate diverse initial candidates of length n."""
    candidates = []
    
    # Linear ramp (the key insight!)
    candidates.append(np.arange(1, n + 1, dtype=np.float64))
    
    # Power law ramps
    for p in [0.7, 1.0, 1.3, 1.5]:
        candidates.append((np.arange(1, n + 1, dtype=np.float64)) ** p)
    
    # Exponential ramp
    for alpha in [0.005, 0.01, 0.02, 0.05]:
        candidates.append(np.exp(alpha * np.arange(n)))
    
    # Windowed linear ramp
    i = np.arange(n, dtype=np.float64)
    for q in [0.3, 0.5, 1.0, 2.0]:
        candidates.append((i + 1) * (1 - i / n) ** q)
    
    # Cosine-modulated ramp
    for freq in [0.5, 1.0, 2.0]:
        mod = 0.5 + 0.5 * np.cos(2 * np.pi * freq * i / n)
        candidates.append((i + 1) * mod)
    
    # Random
    candidates.append(rng.random(n))
    
    # Two-level
    levels = np.ones(n)
    split = n // 2
    levels[:split] = 0.3
    candidates.append(levels)
    
    return candidates


def run_for_length(n, seed, max_time=60.0):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    best_a = None
    best_obj = np.inf
    
    candidates = generate_candidates(n, rng)
    
    # Evaluate all candidates first to find the best starting point
    for c in candidates:
        c = np.maximum(c, 0)
        s = np.sum(c)
        if s < 1e-10:
            continue
        c = c / s
        obj = compute_obj(c)
        if obj < best_obj:
            best_obj = obj
            best_a = c.copy()
    
    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = 2.0
    
    remaining = max_time - (time.monotonic() - start)
    if remaining < 5:
        return best_a, best_obj
    
    # Phase 1: Subgradient optimization from best candidate
    n_inits = min(len(candidates), max(3, int(remaining / 15)))
    candidates_sorted = sorted(candidates, key=lambda c: compute_obj(np.maximum(c, 0) / max(np.sum(np.maximum(c, 0)), 1e-10)))
    
    for init_idx in range(n_inits):
        remaining = max_time - (time.monotonic() - start)
        if remaining < 10:
            break
        c = candidates_sorted[init_idx].copy()
        c = np.maximum(c, 0)
        s = np.sum(c)
        if s < 1e-10:
            continue
        c = c / s
        
        alloc = min(remaining * 0.6, 40.0)
        sg_iters = int(alloc * 5)
        a_sg, obj_sg = subgradient_optimize(c, max_iter=sg_iters, tau_start=5.0, tau_end=40.0, step_init=0.08)
        
        # LP refinement
        remaining = max_time - (time.monotonic() - start)
        if remaining > 15 and n <= 350:
            a_lp, obj_lp = lp_optimize(a_sg, max_iter=20, t=0.3)
            if obj_lp < obj_sg:
                a_sg = a_lp
                obj_sg = obj_lp
        
        # Final polish
        remaining = max_time - (time.monotonic() - start)
        if remaining > 5:
            a_pol, obj_pol = subgradient_optimize(a_sg, max_iter=int(remaining * 2),
                                                   tau_start=20.0, tau_end=80.0, step_init=0.03)
            if obj_pol < obj_sg:
                a_sg = a_pol
                obj_sg = obj_pol
        
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    best_seq = None
    best_score = np.inf
    
    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            best_score = obj
            best_seq = parent.copy()
    except Exception:
        pass
    
    # Key lengths to explore
    n_values = [50, 80, 100, 120, 150, 200, 250, 300, 400]
    
    # If parent exists, include its length
    if best_seq is not None:
        n_parent = len(best_seq)
        n_values = sorted(set(n_values + [n_parent]))
    
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break
        
        # Time allocation based on length
        if n <= 100:
            alloc = min(30.0, remaining * 0.1)
        elif n <= 200:
            alloc = min(60.0, remaining * 0.15)
        elif n <= 300:
            alloc = min(100.0, remaining * 0.2)
        else:
            alloc = min(150.0, remaining * 0.25)
        
        alloc = max(10.0, alloc)
        print(f"N={n}: {remaining:.0f}s remaining, allocating {alloc:.0f}s")
        
        a_opt, obj = run_for_length(n, seed=seed + n * 104729, max_time=alloc)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  UPDATE: n={n}, obj={best_score:.6f}")
    
    # Final intensive polishing
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\nPolishing best sequence (n={n_best}), {remaining:.0f}s remaining")
        
        a_cur = best_seq.copy()
        round_num = 0
        
        while time.monotonic() < deadline - 5:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break
            
            sg_iters = int(min(30000, remaining * 3))
            a_cur, _ = subgradient_optimize(a_cur, max_iter=sg_iters,
                                             tau_start=30.0, tau_end=100.0,
                                             step_init=0.02)
            obj = compute_obj(a_cur)
            if obj < best_score:
                best_score = obj
                best_seq = a_cur.copy()
                print(f"  Polish {round_num} sg: obj={best_score:.6f}")
            
            remaining = deadline - time.monotonic()
            if remaining > 20 and n_best <= 350:
                a_lp, obj_lp = lp_optimize(a_cur, max_iter=30, t=0.2)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    print(f"  Polish {round_num} LP: obj={best_score:.6f}")
            
            # Try perturbation + re-optimize
            if remaining > 30:
                rng = np.random.default_rng(99999 + round_num)
                for pert_scale in [0.03, 0.05, 0.1]:
                    a_pert = a_cur * np.exp(rng.normal(0, pert_scale, n_best))
                    a_pert = np.maximum(a_pert, 0)
                    s_p = np.sum(a_pert)
                    if s_p > 1e-10:
                        a_pert = a_pert / s_p
                        obj_p = compute_obj(a_pert)
                        if
