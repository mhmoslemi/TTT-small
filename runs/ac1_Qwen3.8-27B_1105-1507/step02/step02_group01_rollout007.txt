"""
Optimized step function coefficient search using improved LP-based alternating
with per-coordinate convolution bounds, coordinate descent refinement,
correct soft-max gradients, multi-scale optimization, and adaptive scheduling.
Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute the objective function for a sequence."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_conv_matrix(a):
    """Build matrix A such that A @ b = convolve(a, b)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_per_coord(a, upper_bound=100.0):
    """LP with per-coordinate bounds: max sum(b) s.t. (a*b)_k <= (a*a)_k, b>=0."""
    n = len(a)
    conv_a = np.convolve(a, a)
    M = float(np.max(conv_a))
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = conv_a.copy()
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M


def solve_lp_uniform(a, upper_bound=100.0):
    """LP with uniform bound: max sum(b) s.t. (a*b)_k <= M, b>=0."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M


def solve_lp_mixed(a, alpha=0.7, upper_bound=100.0):
    """LP with mixed bounds: (a*a)_k * alpha + M * (1-alpha)."""
    n = len(a)
    conv_a = np.convolve(a, a)
    M = float(np.max(conv_a))
    b_ub = alpha * conv_a + (1.0 - alpha) * M
    A = build_conv_matrix(a)
    c = -np.ones(n)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x, float(-res.fun), M
    return None, None, M

def soft_max_objective_grad(a, T):
    """Compute soft-max surrogate value and its gradient w.r.t. a."""
    n = len(a)
    conv = np.convolve(a, a)
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    p = exp_vals / sum_exp
    sm_val = max_c + T * np.log(sum_exp)

    # grad[i] = (2/T) * sum_{j=0}^{n-1} p[i+j] * a[j]
    grad = np.zeros(n)
    for i in range(n):
        k_lo = i
        k_hi = min(2 * n - 1, i + n)
        grad[i] = (2.0 / T) * np.dot(p[k_lo:k_hi], a[:k_hi - k_lo])

    return sm_val, grad

            break

        if grad_norm < 1e-14:
            break

        a_norm = np.linalg.norm(a)
        if a_norm < 1e-10:

            best_obj = obj
            a = a_new

            if T > 0.001:

def block_refine(a, rng, max_time):

    if time.monotonic() - start < time_budget * 0.92:
        gd_time2 = min(8.0, time_budget * 0.08)
        a4, obj4 = gd_phase(a1, gd_time2, T_init=0.003)
        if obj4 < best_obj:
            best_a = a4.copy()
            best_obj = obj4

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization: diverse structured init -> LP alternating ->
    gradient descent -> coordinate descent -> block perturbation.
    Uses per-coordinate LP bounds for more aggressive optimization.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 8.0

    best_seq = None
    best_score = np.inf

    # Try parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # Optimize the parent if it exists and is not too large
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all() and parent.size <= 500:
            p_opt, p_obj = full_pipeline(parent.copy(), min(60.0, float(budget_s) * 0.12))
            if p_obj < best_score:
                best_score = p_obj
                best_seq = p_opt.copy()
                print(f"Parent optimized: obj={p_obj:.6f}")
            # Coordinate descent on parent
            if time.monotonic() < deadline - 50:
                cd_time = min(30.0, (deadline - time.monotonic()) * 0.3)
                p_cd, p_cd_obj = coordinate_descent(p_opt, cd_time)
                if p_cd_obj < best_score:
                    best_score = p_cd_obj
                    best_seq = p_cd.copy()
                    print(f"Parent after CD: obj={p_opt if False else p_cd_obj:.6f}")
    except Exception as e:
        print(f"Parent optimization error: {e}")

    # Main search over different n values
    n_values = [40, 60, 80, 100, 120, 150, 180, 200, 250, 300, 350, 400]
    n_styles = 4  # number of different init styles per n

    total_candidates = sum(min(n_styles, max(2, 1500 // (40 + nv // 5))) for nv in n_values)
    print(f"\nPlan: {len(n_values)} n values, ~{total_candidates} total candidates")

    candidates = []  # (obj, n, a)

    for n in n_values:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break

        # Estimate time per candidate for this n
        num_remaining = len([x for x in n_values if x >= n])
        time_per_n = remaining / max(1, num_remaining)
        time_per_n = min(time_per_n, 120.0)

        num_restarts = max(2, min(n_styles, int(time_per_n / 25)))
        time_per_restart = time_per_n / num_restarts

        print(f"\nN={n}: remaining={remaining:.0f}s, {num_restarts} restarts, {time_per_restart:.0f}s each")

        for r in range(num_restarts):
            if time.monotonic() > deadline - 60:
                break
            style = (r + n) % 7
            a_init = structured_init(n, rng, style=style)
            a_opt, obj_opt = full_pipeline(a_init, time_per_restart)

            candidates.append((obj_opt, n, a_opt))

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} r={r} style={style}: obj={obj_opt:.6f} ** NEW BEST **")
            else:
                print(f"  n={n} r={r} style={style}: obj={obj_opt:.6f}")

    # Sort candidates and do deeper optimization on top ones
    candidates.sort(key=lambda x: x[0])
    print(f"\nTop 5 candidates:")
    for i, (obj, n, a) in enumerate(candidates[:5]):
        print(f"  #{i}: n={n}, obj={obj:.6f}")

    # Deep optimization on top candidates
    top_k = min(3, len(candidates))
    for i in range(top_k):
        if time.monotonic() > deadline - 80:
            break
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break

        obj_c, n_c, a_c = candidates[i]
        print(f"\nDeep optimization on candidate #{i} (n={n_c}, obj={obj_c:.6f}), time={remaining:.0f}s")

        # Extended LP
        deep_time = min(remaining * 0.4, 80.0)
        a_deep, obj_deep = lp_phase(a_c, max(20, int(deep_time / 0.3)), deep_time,
                                     mode="per_coord", t_start=0.3, t_min=0.02)
        if obj_deep < best_score:
            best_score = obj_deep
            best_seq = a_deep.copy()
            print(f"  After deep LP: obj={obj_deep:.6f} ** NEW BEST **")

        # Coordinate descent
        if time.monotonic() < deadline - 30:
            cd_rem = deadline - time.monotonic() - 20
            cd_time = min(cd_rem * 0.5, 40.0)
            a_cd, obj_cd = coordinate_descent(a_deep, cd_time)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  After CD: obj={obj_cd:.6f} ** NEW BEST **")
            a_deep = a_cd

        # Another LP round
        if time.monotonic() < deadline - 20:
            lp_rem = min(deadline - time.monotonic() - 15, 30.0)
            a_lp2, obj_lp2 = lp_phase(a_deep, max(10, int(lp_rem / 0.3)), lp_rem,
                                       mode="mixed", alpha=0.4, t_start=0.2)
            if obj_lp2 < best_score:
                best_score = obj_lp2
                best_seq = a_lp2.copy()
                print(f"  After LP2: obj={obj_lp2:.6f} ** NEW BEST **")

        # Final coordinate descent
        if time.monotonic() < deadline - 10:
            cd2_time = min(deadline - time.monotonic() - 8, 20.0)
            a_cd2, obj_cd2 = coordinate_descent(a_deep, cd2_time)
            if obj_cd2 < best_score:
                best_score = obj_cd2
                best_seq = a_cd2.copy()
                print(f"  After CD2: obj={obj_cd2:.6f} ** NEW BEST **")

    # Final refinement on overall best
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: {remaining:.0f}s remaining")

        # GD fine-tune
        if remaining > 15:
            a_gd, obj_gd = gd_phase(best_seq, min(remaining * 0.4, 30.0), T_init=0.001)
            if obj_gd < best_score:
                best_score = obj_gd
                best_seq = a_gd.copy()
                print(f"  After final GD: obj={best_score:.6f}")

        # Block perturbation
        if time.monotonic() < deadline - 3:
            bp_time = deadline - time.monotonic() - 2.0
            a_bp, obj_bp = block_refine(best_seq, rng, bp_time)
            if obj_bp < best_score:
                best_score = obj_bp
                best_seq = a_bp.copy()
                print(f"  After BP: obj={best_score:.6f}")

        # Final coordinate descent
        if time.monotonic() < deadline - 2:
            cd_final_time = deadline - time.monotonic() - 1.0
            a_cf, obj_cf = coordinate_descent(best_seq, cd_final_time)
            if obj_cf < best_score:
                best_score = obj_cf
                best_seq = a_cf.copy()
                print(f"  After final CD: obj={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    # Validate with evaluate_sequence
    final_list = [float(x) for x in best_seq]
    try:
        eval_score = evaluate_sequence(final_list)
        print(f"\nEvaluated score: {eval_score:.6f}")
        if eval_score != best_score:
            print(f"  (computed was {best_score:.6f})")
            best_score = eval_score
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return final_list
