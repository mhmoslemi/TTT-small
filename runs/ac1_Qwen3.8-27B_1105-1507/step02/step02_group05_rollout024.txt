    p = exp_b / exp_b.sum()

    return grad, b

def lse_objective(a, tau):
    """Log-sum-exp approximation of max(b_k) * 2n / (sum a)^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    a = a / s
    b = fast_conv(a)
    b_max = b.max()
    val = b_max + np.log(b.sum()) / tau  # log-sum-exp
    # Actually: LSE = b_max + log(sum(exp(tau*(b - b_max))))/tau
    exp_b = np.exp(tau * (b - b_max))
    lse = b_max + np.log(exp_b.sum()) / tau
    return float(2 * len(a) * lse)

def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:

        score = float(evaluate_sequence(a.tolist()))

            self.best_score = score
            elapsed = time.monotonic() - self.t_start

def run_lp_iterations(a, max_iters, deadline, tracker):
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Adaptive step with more candidates
        base_t = max(0.03, 0.5 * (1.0 - it / float(2 * max_iters)))
        candidates_t = [base_t, min(1.0, base_t * 2.0), max(0.01, base_t * 0.5), base_t * 0.25]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def run_lse_gradient_descent(a, deadline, tracker, max_iters=5000):
    """Gradient descent on the log-sum-exp surrogate."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    
    # Start with moderate tau and increase
    tau = 10.0
    lr = 0.1 / n  # initial learning rate
    current_J = compute_J(a)
    tracker.update(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        grad, b = logsumexp_gradient_fft(a, tau)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Gradient step with projection
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()
        
        J_new = compute_J(a_new)
        
        if J_new < current_J:
            a = a_new
            current_J = J_new
            if it % 100 == 0:
                tracker.update(a)
            # Occasionally increase tau
            if it % 50 == 0:
                tau *= 1.05
            # Decay learning rate
            lr *= 0.999
        else:
            # Reduce learning rate
            lr *= 0.5
            if lr < 1e-12:
                break
    
    tracker.update(a)
    return a

def generate_initializations(n, rng, parent):
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))
    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))
    
    t_arr = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 4, 5, 6, 8, 12]:
        inits.append(1.0 + 0.5 * np.cos(2.0 * np.pi * freq * t_arr + rng.uniform(0, 6.28)))
    
    inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * t_arr ** 2 * 5))
    inits.append(1.0 + 0.5 * (2.0 * t_arr - 1.0) ** 3)
    inits.append(np.exp(-5.0 * (t_arr - 0.3) ** 2) + 0.05 + 0.5*np.exp(-5.0 * (t_arr - 0.7) ** 2))
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.05 + 0.3*np.abs(np.sin(2*np.pi*t_arr)))
    
    # Sparse with smoothing
    sparse = np.zeros(n)
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 32)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)
    
    # Power-law
    inits.append((t_arr + 0.1) ** rng.uniform(-2, 2))
    
    # Multi-scale
    inits.append(1.0 + 0.3*np.sin(2*np.pi*t_arr) + 0.2*np.sin(4*np.pi*t_arr) + 0.1*np.sin(8*np.pi*t_arr))
    
    return inits

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based mirror descent with constraint tightening
2. Log-sum-exp gradient descent with momentum
3. Simulated annealing with diverse restarts
4. L-BFGS-B with LSE surrogate objective
5. Hill climbing polish
Key differences from previous: LSE gradient (new), LP tightening, non-power-2 lengths, momentum.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            seg = a[jl:jh + 1]
            # For row k, col j: value is a[k-j]
            # j ranges from jl to jh, so k-j ranges from k-jh to k-jl
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    n_rows = 2 * n - 1
    A = lil_matrix((n_rows, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def lse_gradient(a, tau):
    """Compute gradient of (1/tau)*log(sum exp(tau*b_k)) w.r.t. a.
    Returns (2 * cross_correlation(p, a))[:n] and b."""
    n = len(a)
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    p = exp_b / exp_b.sum()
    # grad[j] = 2 * sum_k p[k] * a[k-j] = 2 * cross_correlation(p, a)[j]
    s = 1
    while s < 3 * n:
        s <<= 1
    fp = np.fft.rfft(p, s)
    fa = np.fft.rfft(a, s)
    corr = np.fft.irfft(fp * np.conj(fa), s)
    grad = 2.0 * corr[:n]
    return grad, b, b_max


def lse_value(a, tau):
    """Compute LSE value: (1/tau)*log(sum exp(tau*b_k))."""
    b = fast_conv(a)
    b_max = b.max()
    exp_b = np.exp(tau * (b - b_max))
    return b_max + np.log(exp_b.sum()) / tau


def run_lp_iterations(a, max_iters, deadline, tracker, tighten=0.0):
    """LP mirror descent with optional constraint tightening."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        M_target = M * (1.0 - tighten)
        g = lp_best_response(a, M_target)
        if g is None:
            g = lp_best_response(a, M)
        if g is None:
            break
        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        # Adaptive step with multiple candidates
        progress = min(1.0, it / max(1, max_iters))
        base_t = max(0.02, 0.6 * (1.0 - progress))
        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.01, base_t * 0.5)]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J
        if it % 15 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_lse_gradient_descent(a, deadline, tracker, max_iters=3000):
    """Momentum gradient descent on LSE surrogate."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)

    tau = 20.0
    lr = 0.5 / (n * tau)
    momentum = 0.9
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad, b, b_max = lse_gradient(a, tau)

        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 1e-6)
        a_new /= a_new.sum()

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-15:
            a = a_new
            current_J = J_new
            if it % 200 == 0:
                tracker.update(a)
            # Warm up: increase tau gradually
            tau = min(200.0, tau * 1.02)
            lr = 0.5 / (n * tau)
        else:
            lr *= 0.5
            v *= 0.5
            if lr < 1e-15:
                break

    tracker.update(a)
    return a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:

            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.2, 1.0, n))

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 5, 7, 11]:
        inits.append(1.0 + 0.5 * np.cos(2.0 * np.pi * freq * t + rng.uniform(0, 6.28)))

    inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * 3.7 * t))
    inits.append(1.0 + 0.5 * (2.0 * t - 1.0) ** 4)
    inits.append(0.5 * np.exp(-8.0 * (t - 0.25) ** 2) + 0.5 * np.exp(-8.0 * (t - 0.75) ** 2) + 0.05)
    inits.append(0.8 + 0.4 * np.abs(np.sin(np.pi * t)) * np.abs(np.sin(2 * np.pi * t)))
    inits.append(1.0 + 0.3 * np.sin(2 * np.pi * t) + 0.2 * np.sin(4 * np.pi * t) + 0.15 * np.sin(7 * np.pi * t))

    # Sparse with smoothing
    sparse = np.zeros(n)
    n_spikes = max(4, n // 20)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    ksize = max(3, n // 40)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Power-law
    inits.append((t + 0.05) ** rng.uniform(-1.5, 1.5))

    # Random Fourier
    coeffs = rng.normal(0, 1, size=min(n // 2 + 1, 16))
    sig = np.zeros(n)
    for i, c in enumerate(coeffs):
        sig += c * np.cos(2 * np.pi * i * t + rng.uniform(0, 6.28))
    inits.append(np.abs(sig) + 0.3)

    return inits


def run_hill_climbing(a, deadline, tracker, n_steps=2000):
    """Simple hill climbing with random perturbations."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a /= a.sum()
    n = len(a)
    current_J = compute_J(a)
    tracker.update(a)
    rng = np.random.default_rng(12345)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        # Random perturbation
        scale = 0.05 * (1.0 - step / float(n_steps)) + 0.01
        a_trial = a + rng.normal(0, scale, n)
        a_trial = np.maximum(a_trial, 1e-6)
        a_trial /= a_trial.sum()

        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-14:
            a = a_trial
            current_J = J_trial
            if step % 100 == 0:
                tracker.update(a)

    tracker.update(a)
    return a


def run_lbb_lse(a, deadline, tracker):
    """L-BFGS-B optimization using LSE surrogate with increasing tau."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a /= a.sum()

    def obj_and_grad(x, tau_ref):
        ax = np.maximum(x, 1e-10)
        s = ax.sum()
        if s < 1e-8:
            return 1e10, np.zeros(n)
        ax = ax / s
        b = fast_conv(ax)
        b_max = b.max()
        exp_b = np.exp(tau_ref * (b - b_max))
        lse = b_max + np.log(exp_b.sum()) / tau_ref
        val = 2.0 * n * lse
        # Gradient
        p = exp_b / exp_b.sum()
        s2 = 1
        while s2 < 3 * n:
            s2 <<= 1
        fp = np.fft.rfft(p, s2)
        fa = np.fft.rfft(ax, s2)
        corr = np.fft.irfft(fp * np.conj(fa), s2)
        grad = 2.0 * n * 2.0 * corr[:n] / s
        return val, grad

    # Start with moderate tau, increase
    for tau_start in [20.0, 50.0, 100.0]:
        if time.monotonic() > deadline:
            break

        class TauHolder:
            def __init__(self, t):
                self.t = t

        th = TauHolder(tau_start)

        def obj_wrapper(x):
            v, _ = obj_and_grad(x, th.t)
            return v

        def grad_wrapper(x):
            _, g = obj_and_grad(x, th.t)
            return g

        bounds_list = [(0.0, None)] * n
        res = minimize(obj_wrapper, a.astype(np.float64), method='L-BFGS-B',
                       bounds=bounds_list, jac=grad_wrapper,
                       options={'maxiter': 200, 'max_fun': 30000, 'ftol': 1e-18, 'gtol': 1e-10})
        if res.fun < 1e10:
            a = np.maximum(res.x, 1e-6)
            a /= a.sum()
            tracker.update(a)
            print(f"  L-BFGS-B tau={tau_start}: J={compute_J(a):.8f}", flush=True)

    return a
