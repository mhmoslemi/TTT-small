import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Compute conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2*n-1]

def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    conv = compute_conv(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_Toeplitz(a):
    """Build the Toeplitz matrix A where (A@g)[k] = (a*g)[k]."""
    n = len(a)
    # A is (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # We can build this efficiently
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # For fixed j, A[k, j] = a[k-j] for k = j, j+1, ..., j+n-1
        # But k goes from 0 to 2n-2
        # So for fixed j: k-j ranges from 0 to n-1, meaning k ranges from j to j+n-1
        # But k is bounded by 2n-2, so k goes from j to min(j+n-1, 2n-2)
        k_start = j
        k_end = min(j + n, 2*n-1)  # exclusive
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A

def solve_lp_g(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    A = build_Toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.status == 0:
            g = res.x
            g = np.maximum(g, 0)  # Numerical cleanup
            return g
    except:
        pass
    return None

def subgradient_step(a, tau=10.0, step=0.05):
    """One subgradient step."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        # All weight on the max
        idx = np.argmax(conv)
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum
    
    # Gradient of max(conv) with respect to a
    # d(conv[k])/da[i] = a[k-i] if 0 <= k-i < n
    # grad[i] = sum_k weights[k] * a[k-i] (for valid k-i)
    # This is the cross-correlation of weights with a
    grad = np.correlate(weights, a, mode='full')[n-1:n-1+n]
    
    # Project onto sum=1, a>=0
    grad_centered = grad - np.mean(grad)
    a_new = a - step * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new

import numpy as np

def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_toeplitz(a):
    """Build Toeplitz matrix A where (A@g)[k] = (conv(a,g))[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A[k_start:k_end, j] = a[:k_end - k_start]
    return A


def solve_lp_g(a, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns (g, sum_g) or (None, None)."""
    n = len(a)
    A = build_toeplitz(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            return g, float(np.sum(g))
    except Exception:
        pass
    return None, None


def solve_lp_minpeak(a, time_limit=20):

    Returns (g, M_opt) or (None, None)."""

    A = build_toeplitz(a)
    # Variables: g_0..g_{n-1}, M  (n+1 variables)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    # Inequality: A@g - M <= 0  ->  [A, -1] @ [g; M] <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0

    try:

                g = g / sum_g
            M_opt = None

        if g is None:
            break

        a_new, t = line_search_blend(a, g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > patience:
                # Perturb to escape
                rng = np.random.default_rng(it * 7919 + 12345)
                a = a * np.exp(rng.normal(0, 0.1, n))
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall_count = 0
                patience = 15
            else:
                # Try a small move anyway
                a = 0.95 * a + 0.05 * g
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s

    return best_a


def smoothed_subgradient(a, tau, step):
    """One smoothed subgradient step. Returns new a."""
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        weights = np.zeros(len(conv))
        weights[idx] = 1.0
    else:
        weights /= w_sum

    # grad[i] = 2 * sum_l weights[i+l] * a[l] for l=0..n-1 where i+l < 2n-1
    # This is cross-correlation
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[n - 1:n - 1 + n]

    grad_centered = grad - np.mean(grad)
    a_new = a - step * grad_centered
    a_new = np.maximum(a_new, 0)
    s = np.sum(a_new)
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def subgradient_optimize(a, max_iters=5000, time_budget=30.0, tau_start=5.0, tau_end=100.0):
    """Run smoothed subgradient optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    step = 0.08
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break

        frac = it / max(1, max_iters)
        tau = tau_start + (tau_end - tau_start) * frac

        a = smoothed_subgradient(a, tau, step)
        obj = compute_obj(a)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                step *= 0.7
                if step < 1e-9:
                    break
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.01, n) * step
                a = np.maximum(a, 0)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                stall = 0

        if it > 200:
            step *= 0.999

    return best_a


def random_init(n, seed):
    """Generate diverse random initializations."""
    rng = np.random.default_rng(seed)
    k = np.arange(n)
    choice = rng.integers(0, 8)

    if choice == 0:
        # Single cosine
        f = rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.8 * np.cos(2 * np.pi * f * k / n + phase)
    elif choice == 1:
        # Two cosines
        f1 = rng.uniform(0.3, 2.0)
        f2 = rng.uniform(2.0, 6.0)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * k / n) + 0.3 * np.cos(2 * np.pi * f2 * k / n)
    elif choice == 2:
        # Power-law decay from center
        center = n * rng.uniform(0.3, 0.7)
        alpha = rng.uniform(0.5, 3.0)
        a = 1.0 + 0.5 * np.exp(-alpha * np.abs(k - center) / (n * 0.3))
    elif choice == 3:
        # Random with smoothing
        a = rng.random(n)
        win = max(3, n // 20)
        kernel = np.ones(win) / win
        a = np.convolve(a, kernel, mode='same')
    elif choice == 4:
        # Periodic with random harmonics
        a = np.ones(n)
        for _ in range(rng.integers(1, 4)):
            f = rng.uniform(0.5, 5.0)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2 * np.pi * f * k / n + rng.uniform(0, 2 * np.pi))
    elif choice == 5:
        # Sawtooth-like
        f = rng.uniform(0.3, 2.0)
        a = 1.0 + 0.5 * np.sin(2 * np.pi * f * k / n)
    elif choice == 6:
        # Block structure
        nblocks = rng.integers(2, 6)
        a = np.ones(n)
        for b in range(nblocks):
            start = b * n // nblocks
            end = (b + 1) * n // nblocks
            a[start:end] = rng.uniform(0.3, 1.5)
    else:
        # Pure random
        a = rng.random(n) + 0.1

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def try_parent():
    """Try to load parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def optimize_n(n, seed, time_budget, verbose=False):
    """Run full optimization pipeline for a given n."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    best_a = None
    best_obj = np.inf

    n_inits = max(2, int(time_budget / 25))
    n_inits = min(n_inits, 5)

    for init_idx in range(n_inits):
        if time.monotonic() - start > time_budget:
            break

        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        a = random_init(n, seed + init_idx * 79193)

        # Phase 1: Quick subgradient to shape
        sg_time = min(10.0, remaining * 0.25)
        a = subgradient_optimize(a, max_iters=int(sg_time * 30), time_budget=sg_time,
                                 tau_start=5.0, tau_end=30.0)
        obj = compute_obj(a)
        if verbose:
            print(f"    init{init_idx} after SG: obj={obj:.6f}")

        # Phase 2: Alternating LP (main workhorse)
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 15 and n <= 800:
            lp_time = min(remaining * 0.5, 60.0)
            a = alternating_lp(a, max_iters=60, time_budget=lp_time)
            obj = compute_obj(a)
            if verbose:
                print(f"    init{init_idx} after LP: obj={obj:.6f}")

        # Phase 3: Subgradient polish
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 5:
            a = subgradient_optimize(a, max_iters=int(remaining * 20),
                                      time_budget=min(remaining * 0.3, 15.0),
                                      tau_start=20.0, tau_end=80.0)
            obj = compute_obj(a)

        # Phase 4: Final LP if time allows
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10 and n <= 500:
            a = alternating_lp(a, max_iters=30, time_budget=min(remaining * 0.5, 20.0))
            obj = compute_obj(a)

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def polish_sequence(a, time_budget, verbose=False):
    """Intensive polishing of a single sequence."""
    n = len(a)
    a = a.copy().astype(np.float64)
    start = time.monotonic()

    best_a = a.copy()
    best_obj = compute_obj(a)
    round_num = 0

    while time.monotonic() - start < time_budget - 5:
        round_num += 1
        remaining = time_budget - (time.monotonic() - start)
        if remaining < 5:
            break

        # LP step
        if n <= 600 and remaining > 20:
            lp_time = min(remaining * 0.4, 45.0)
            a_new = alternating_lp(best_a, max_iters=40, time_budget=lp_time)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} LP: {best_obj:.6f}")

        # Subgradient with varying tau
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 10:
            tau_s = 10.0 + round_num * 5
            a_new = subgradient_optimize(best_a, max_iters=int(remaining * 10),
                                          time_budget=min(remaining * 0.25, 20.0),
                                          tau_start=tau_s, tau_end=tau_s + 50)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
                if verbose:
                    print(f"    Polish r{round_num} SG: {best_obj:.6f}")

        # Min-peak LP variant
        remaining = time_budget - (time.monotonic() - start)
        if n <= 400 and remaining > 15:
            a_lp, M_opt = solve_lp_minpeak(best_a, time_limit=15)
            if a_lp is not None:
                a_new, _ = line_search_blend(best_a, a_lp)
                obj = compute_obj(a_new)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_new.copy()
                    if verbose:
                        print(f"    Polish r{round_num} MinPeak: {best_obj:.6f}")

        # Perturbation + re-optimize
        remaining = time_budget - (time.monotonic() - start)
        if remaining > 20:
            rng = np.random.default_rng(round_num * 42 + 17)
            for _ in range(3):
                scale = rng.uniform(0.03, 0.15)
                a_pert = best_a * np.exp(rng.normal(0, scale, n))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s < 1e-10:
                    continue
                a_pert = a_pert / s
                # Quick subgradient on perturbed
                a_pert = subgradient_optimize(a_pert, max_iters=500, time_budget=3.0,
                                               tau_start=10.0, tau_end=40.0)
                obj = compute_obj(a_pert)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a_pert.copy()
                    if verbose:
                        print(f"    Polish r{round_num} Perturb: {best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: explores multiple sequence lengths, uses alternating LP with
    line search, subgradient polishing, and perturbation-based escapes.
    Returns the best sequence found within the time budget.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0

    best_seq = None
    best_score = np.inf

    # Load parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()

    # N values to explore (sorted by expected difficulty)
    n_values = [100, 150, 200, 250, 300, 400, 500, 600, 800, 1000, 1200]

    # If we have a parent, prioritize its neighborhood
    if best_seq is not None:
        np_ = len(best_seq)
        extra = [np_, max(50, np_ - 50), min(1500, np_ + 100), min(1500, np_ + 200)]
        n_values = sorted(set(n_values + extra))

    print(f"Exploring n values: {n_values}")
    print(f"Initial best: {best_score:.6f}")

    # Time allocation: spend more time on promising n values
    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        # Allocate time based on n and remaining budget
        frac_of_n = n / max(n_values)
        if n <= 200:
            alloc = min(40.0, remaining * 0.12)
        elif n <= 400:
            alloc = min(70.0, remaining * 0.15)
        elif n <= 700:
            alloc = min(100.0, remaining * 0.18)
        else:
            alloc = min(120.0, remaining * 0.2)

        alloc = max(15.0, alloc)
        print(f"\nN={n}: {remaining:.0f}s remaining, alloc={alloc:.0f}s")

        a_opt, obj = optimize_n(n, seed=seed + n * 104729, time_budget=alloc,
                                 verbose=(n in [250, 400, 500]))
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  ** UPDATE: n={n}, obj={best_score:.6f} **")

    # Intensive polishing phase
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== POLISHING PHASE: n={n_best}, {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()
        for polish_round in range(5):
            if time.monotonic() > deadline - 10:
                break
            remaining = deadline - time.monotonic()
            if remaining < 15:
                break

            a_pol, obj_pol = polish_sequence(a_cur, time_budget=min(remaining * 0.5, 80.0),
                                              verbose=True)
            if obj_pol < best_score:
                best_score = obj_pol
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  After polish round {polish_round+1}: {best_score:.6f}")
            else:
                # Try perturbing and re-polishing from a slightly different point
                rng = np.random.default_rng(polish_round * 999 + 555)
                a_pert = a_cur * np.exp(rng.normal(0, 0.08, n_best))
                a_pert = np.maximum(a_pert, 0)
                s = np.sum(a_pert)
                if s > 1e-10:
                    a_pert = a_pert / s
                    a_pol2, obj_pol2 = polish_sequence(a_pert, time_budget=min(remaining * 0.3, 40.0),
                                                        verbose=False)
                    if obj_pol2 < best_score:
                        best_score = obj_pol2
                        best_seq = a_pol2.copy()
                        a_cur = a_pol2.copy()
                        print(f"  After perturb+polish round {polish_round+1}: {best_score:.6f}")

    # Also try the parent with more aggressive polishing if it's still competitive
    if parent is not None and parent_obj < best_score + 0.05 and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic()
        print(f"\n=== RETRY PARENT: {remaining:.0f}s remaining ===")
        a_p, obj_p = polish_sequence(parent, time_budget=min(remaining * 0.6, 90.0), verbose=True)
        if obj_p < best_score:
            best_score = obj_p
            best_seq = a_p.copy()
            print(f"  Parent improved to: {best_score:.6f}")

    # Final validation
    if best_seq is None:
        best_seq = np.ones(50)
        best_score = compute_obj(best_seq)

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf:
        best_score = val
    else:
        # Fallback
        result = [1.0] * 50
        best_score = 2.0

    elapsed = time.monotonic() - start_time
    print(f"\n=== FINAL: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s ===")

    return result
