"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines: (1) MiMaD-style targeted moves, (2) LP-guided optimization,
(3) smoothed subgradient with adaptive step, (4) simulated annealing,
(5) diverse initializations including structured patterns.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    """FFT-based autocorrelation."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Objective value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def normalize(a):
    """Normalize to sum=1, clip non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def mima_step(a, max_iter=1000, decay=0.999):
    """
    MiMaD-style step: find the peak in conv, reduce contributing pairs,
    increase non-contributing positions.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    step_size = 0.1
    
    for it in range(max_iter):
        conv = compute_conv(a)
        k_max = int(np.argmax(conv))
        peak_val = conv[k_max]
        
        # Find which pairs (i, k_max - i) contribute to this peak
        i_min = max(0, k_max - n + 1)
        i_max = min(n - 1, k_max)
        
        # Build a "pressure" vector: positions that contribute to the peak get negative pressure
        pressure = np.zeros(n)
        for i in range(i_min, i_max + 1):
            j = k_max - i
            if j < n:
                # Both i and j contribute
                pressure[i] -= a[j]
                pressure[j] -= a[i]
        
        # Also consider second and third peaks
        sorted_idx = np.argsort(conv)[::-1]
        for rank in range(1, min(3, len(sorted_idx))):
            k2 = sorted_idx[rank]
            if conv[k2] > 0.8 * peak_val:  # only if close to max
                i_min2 = max(0, k2 - n + 1)
                i_max2 = min(n - 1, k2)
                for i in range(i_min2, i_max2 + 1):
                    j = k2 - i
                    if j < n:
                        pressure[i] -= 0.5 * a[j]
                        pressure[j] -= 0.5 * a[i]
        
        # Center the pressure (to maintain sum)
        pressure -= pressure.mean()
        
        # Update
        a_new = a - step_size * pressure
        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        step_size *= decay
        
        if step_size < 1e-10:
            break
    
    return best_a, best_obj

def smoothed_subgradient(a, tau=20.0):
    """
    Smoothed subgradient of max(conv(a,a)) with respect to a.
    Uses log-sum-exp approximation.
    """
    n = len(a)
    conv = compute_conv(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum
    
    # Gradient: d(conv[k])/d(a[i]) = a[k-i] (if valid)
    # So d max(conv)/d(a[i]) ≈ sum_k weights[k] * a[k-i]
    # This is the cross-correlation of weights with a
    grad = np.zeros(n)
    for k in range(len(weights)):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            grad[i] += weights[k] * a[j]
    
    # Factor of 2 because conv is symmetric
    grad *= 2.0
    return grad

def subgradient_optimize(a, max_iter=5000, tau_start=5.0, tau_end=100.0, step_init=0.05):
    """Smoothed subgradient method with adaptive step."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    step = step_init
    stall = 0
    patience = 300
    
    for it in range(max_iter):
        frac = it / max(1, max_iter)
        tau = tau_start + (tau_end - tau_start) * frac
        
        conv = compute_conv(a)
        current_obj = compute_obj(a)
        
        if current_obj < best_obj - 1e-12:
            best_obj = current_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                step *= 0.8
                if step < 1e-10:
                    break
                # Add noise to escape
                rng = np.random.default_rng(it * 31 + 7)
                a = a + rng.normal(0, 0.02, n) * step
                a = normalize(a)
                continue
        
        grad = smoothed_subgradient(a, tau)
        grad_centered = grad - grad.mean()
        
        # Line search: try a few step sizes
        best_step = step
        best_a_try = None
        for factor in [1.0, 0.5, 0.2, 0.1, 2.0]:
            s_try = step * factor
            a_try = a - s_try * grad_centered
            a_try = np.maximum(a_try, 0)
            a_try = normalize(a_try)
            obj_try = compute_obj(a_try)
            if obj_try < compute_obj(a) - 1e-10:
                best_step = s_try
                best_a_try = a_try
                break
        
        if best_a_try is not None:
            a = best_a_try
        else:
            a = a - step * grad_centered
            a = np.maximum(a, 0)
            a = normalize(a)
        
        if it > 200:
            step *= 0.9998
    
    return best_a, best_obj

def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    # Build constraint matrix: (a*g)_k = sum_i a[i]*g[k-i] <= M
    # For each k, the constraint is: sum_i a[k-i] * g[i] <= M (with valid indices)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = a[j]
    
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None

def lp_optimize(a, max_iter=50, t=0.3, lp_time_limit=30):
    """Alternating LP optimization."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    stall = 0
    for it in range(max_iter):
        conv = compute_conv(a)
        M = float(np.max(conv))
        
        # Try slightly tighter M to get a more "spread out" solution
        M_tight = M * 0.95
        
        g, sum_g = solve_lp(a, M_tight)
        if g is None:
            g, sum_g = solve_lp(a, M)
        if g is None:
            break
        
        if sum_g <= 1.0 + 1e-3:
            stall += 1
            if stall > 15:
                break
            continue
        stall = 0
        
        g = normalize(g)
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj

def simulated_annealing(a, max_iter=20000, T_start=0.1, T_end=0.001):
    """Simulated annealing for escaping local minima."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(a)
    current_obj = best_obj
    
    T = T_start
    cooling = (T_end / T_start) ** (1.0 / max(1, max_iter))
    
    for it in range(max_iter):
        # Random perturbation
        rng = np.random.default_rng(it * 13 + 42)
        # Choose a subset of positions to perturb
        n_perturb = max(1, n // 10)
        idx = rng.choice(n, size=n_perturb, replace=False)
        perturbation = rng.normal(0, 0.1, n_perturb)
        
        a_new = a.copy()
        a_new[idx] = np.maximum(a_new[idx] + perturbation, 0)
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        delta = obj_new - current_obj
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_new
            current_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
        
        T *= cooling
    
    return best_a, best_obj

def diverse_init(n, rng, mode=0):
    """Generate diverse initializations."""
    if mode == 0:
        # Uniform
        return np.ones(n)
    elif mode == 1:
        # Cosine taper
        return 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    elif mode == 2:
        # Random
        return rng.random(n)
    elif mode == 3:
        # Bimodal
        center = n // 2
        return np.exp(-0.5 * (np.arange(n) / (n * 0.3))**2) + np.exp(-0.5 * ((np.arange(n) - n*0.5) / (n * 0.15))**2)
    elif mode == 4:
        # Structured: based on modular arithmetic
        p = max(2, int(n / 3))
        a = np.zeros(n)
        for i in range(n):
            a[i] = 1.0 if (i * i) % p < p // 2 else 0.3
        return a
    elif mode == 5:
        # Fibonacci-like spacing
        phi = (1 + np.sqrt(5)) / 2
        positions = [(int(i * phi) % n) for i in range(n)]
        a = np.zeros(n)
        for pos in positions:
            a[pos] += 1.0
        return a
    elif mode == 6:
        # Sine wave
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        return 0.5 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    elif mode == 7:
        # Exponential decay from center

    Alternate between:
    1. Project to make |FFT(a)| flat (in frequency domain)
    2. Project to non-negativity (in time domain)

    best_a = a.copy()
    best_obj = compute_obj(a)

        A = np.fft.rfft(a)
        mag = np.abs(A)
        # Target: flat magnitude = mean(mag)
        target_mag = np.mean(mag)
        # Project: scale each frequency component to target magnitude
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n)

        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        # Blend with previous to avoid oscillation
        a = 0.5 * a + 0.5 * a_new
    
    return best_a, best_obj

def compute_conv(a):

    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)

def compute_obj(a):

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):

def get_gradient(a, tau):

    n = len(a)

def nesterov_subgradient(a, max_iter=10000, tau_start=3.0, tau_end=60.0,
                          step_init=0.05, momentum=0.9):
    """
    Nesterov-accelerated smoothed subgradient with adaptive step.
    """
    n = len(a)
    a = normalize(a.copy())
    v = np.zeros(n)  # velocity for momentum
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    step = step_init
    stall = 0
    patience = 200
    
    for it in range(max_iter):
        frac = it / max(1, max_iter)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.7)
        
        # Nesterov lookahead
        a_nest = a - momentum * v
        a_nest = np.maximum(a_nest, 0)
        a_nest = normalize(a_nest)
        
        grad = get_gradient(a_nest, tau)
        grad_centered = grad - float(np.mean(grad))
        
        # Update velocity
        v = momentum * v + step * grad_centered
        
        # Update position
        a_new = a - v
        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new)
        
        obj_new = compute_obj(a_new)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                step *= 0.7
                v *= 0.5
                if step < 1e-10:
                    break
                # Add random perturbation
                rng = np.random.default_rng(it * 31 + 17)
                a = a + rng.normal(0, 0.03 * step, n)
                a = normalize(a)
                a_new = a.copy()
                continue
        
        a = a_new
        if it > 500:
            step *= 0.9997
            momentum = min(0.99, momentum + 0.001)
    
    return best_a, best_obj

def frequency_projection(a, max_iter=500):
    """
    Frequency-domain projection: alternate between flattening |FFT| and enforcing non-negativity.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
