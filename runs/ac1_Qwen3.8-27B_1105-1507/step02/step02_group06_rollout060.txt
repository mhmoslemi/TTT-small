def compute_objective(a):
    """Fast computation of the objective."""
    n = len(a)
    conv = np.convolve(a, a)
    M = np.max(conv)
    S = np.sum(a)

def build_conv_matrix(a):

    n = len(a)

"""
LP-based optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses alternating LP direction finding with backtracking line search,
multiple initializations, and momentum for convergence acceleration.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_objective(a):
    """Fast computation of the objective value."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))


def build_conv_matrix(a):
    """Build sparse matrix A where (A@g)[k] = convolve(a, g)[k] for k=0..2n-2."""
    n = len(a)
    # Pre-allocate
    total_nz = n * (n + 1)  # upper bound
    rows = np.empty(total_nz, dtype=np.int32)
    cols = np.empty(total_nz, dtype=np.int32)
    vals = np.empty(total_nz, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        count = j_max - j_min + 1
        rows[idx:idx + count] = k
        cols[idx:idx + count] = np.arange(j_min, j_max + 1)
        vals[idx:idx + count] = a[k - np.arange(j_min, j_max + 1)]
        idx += count
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def solve_lp_direction(a, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def subgradient_step(a, top_k=5):
    """Compute subgradient of max(conv(a,a)) and return a descent direction."""
    n = len(a)
    conv = np.convolve(a, a)
    
    # Get top-k lags
    top_indices = np.argsort(conv)[-top_k:]
    weights = np.maximum(conv[top_indices] - conv.mean(), 0)
    w_sum = weights.sum()
    if w_sum < 1e-15:
        return None
    
    weights = weights / w_sum
    
    # Compute gradient: for each active lag k, grad contribution is 2*a[k-i] (with bounds)
    grad = np.zeros(n)
    for k, w in zip(top_indices, weights):
        if w <= 0:
            continue
        # d(conv[k])/da[i] = 2*a[k-i] for valid i
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += 2.0 * w * a[k - j]
    
    return grad


def lp_optimize(a_init, deadline, verbose=False, verbose_interval=50):
    """LP-based optimization with backtracking line search."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = float(np.sum(a))
    # Normalize to sum = 1 for the optimization (objective is homogeneous)
    a = a / S
    S = 1.0
    
    best_a = a.copy()
    best_score = float(2.0 * n * np.max(np.convolve(a, a)))  # with sum=1
    # Actual score: 2n * max(conv) / sum^2 = 2n * max(conv) / 1 = 2n*max(conv)
    
    alpha = 0.3  # initial step size
    no_improve = 0
    it = 0
    momentum = np.zeros(n)
    mu = 0.3  # momentum coefficient
    
    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        
        # Solve LP
        g = solve_lp_direction(a, M, time_limit=3.0)
        if g is None:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        sum_g = float(np.sum(g))
        if sum_g < 1e-15:
            no_improve += 1
            continue
        
        # Rescale g to sum = 1
        g = g / sum_g
        
        # Apply momentum
        direction = (1 - mu) * g + mu * momentum
        momentum = direction
        direction = direction / np.sum(direction)  # renormalize
        
        # Backtracking line search
        found = False
        for a_try in [alpha, alpha * 0.7, alpha * 0.4, alpha * 0.2, 0.05]:
            candidate = (1.0 - a_try) * a + a_try * direction
            candidate = np.clip(candidate, 0.0, None)
            sum_c = float(np.sum(candidate))
            if sum_c < 1e-15:
                continue
            candidate = candidate / sum_c
            sc = float(2.0 * n * np.max(np.convolve(candidate, candidate)))
            if sc < best_score * (1 - 1e-10):
                best_score = sc
                best_a = candidate.copy()
                a = candidate
                found = True
                break
        
        it += 1
        if found:
            no_improve = 0
            alpha = min(alpha * 1.1, 0.8)
            if verbose and it % verbose_interval == 0:
                print(f"    LP iter {it}: obj={best_score/(n) * 0.5:.6f} (raw={best_score:.6f}) alpha={alpha:.3f}")
        else:
            no_improve += 1
            alpha *= 0.7
            if no_improve > 50:
                break
    
    # Convert back to original scaling
    best_score_actual = float(2.0 * n * float(np.max(np.convolve(best_a, best_a))) / (float(np.sum(best_a)) ** 2))
    return best_a, best_score_actual

"""
LP-based optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses LP direction finding with backtracking line search, momentum,
multiple initializations including parent interpolation, and subgradient
fallback. Targets n=1024 and n=2048 with diverse starting shapes.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_obj(a):
    """Fast objective: 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    S = float(np.sum(a))
    if S < 1e-10:
        return float('inf')
    return float(2.0 * n * M / (S * S))


def build_conv_matrix(a):
    """Build sparse matrix A where (A@g)[k] = sum_j a[k-j]*g[j]."""
    n = len(a)
    total_nz = n * n
    rows = np.empty(total_nz, dtype=np.int32)
    cols = np.empty(total_nz, dtype=np.int32)
    vals = np.empty(total_nz, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        cnt = j_hi - j_lo + 1
        js = np.arange(j_lo, j_hi + 1)
        rows[idx:idx + cnt] = k
        cols[idx:idx + cnt] = js
        vals[idx:idx + cnt] = a[k - js]
        idx += cnt
    return csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(2 * n - 1, n))


def solve_lp_direction(a, M, tlimit=5.0):
    """Solve: max sum(g) s.t. A@g <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': tlimit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(a_init, deadline, verbose=False, vi=30):
    """LP-based optimization with backtracking line search and momentum."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return a_init, compute_obj(a_init)
    a = a / S
    n_active = np.sum(a > 1e-10)

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(np.convolve(a, a)))
    alpha = 0.3
    momentum = np.zeros(n)
    mu = 0.25
    no_improve = 0
    it = 0
    t_start = time.monotonic()

    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        if M < 1e-15:
            break

        g = solve_lp_direction(a, M, tlimit=5.0)
        if g is None:
            no_improve += 1
            if no_improve > 40:
                break
            continue

        sg = float(np.sum(g))
        if sg < 1e-15:
            no_improve += 1
            continue

        g = g / sg

        # Blend with momentum
        direction = (1.0 - mu) * g + mu * momentum
        ds = float(np.sum(direction))
        if ds < 1e-15:
            direction = g
            ds = sg
        else:
            direction = direction / ds
        momentum = direction

        # Backtracking line search
        found = False
        candidates = [alpha, alpha * 0.6, alpha * 0.3, alpha * 0.1, 0.02]
        for a_try in candidates:
            if a_try > 1.0:
                a_try = 1.0
            cand = (1.0 - a_try) * a + a_try * direction
            cand = np.clip(cand, 0.0, None)
            cs = float(np.sum(cand))
            if cs < 1e-15:
                continue
            cand = cand / cs
            co = 2.0 * n * float(np.max(np.convolve(cand, cand)))
            if co < best_obj * (1.0 - 1e-9):
                best_obj = co
                best_a = cand.copy()
                a = cand
                found = True
                break

        it += 1
        if found:
            no_improve = 0
            alpha = min(alpha * 1.05, 0.7)
            if verbose and it % vi == 0:
                elapsed = time.monotonic() - t_start
                actual_obj = best_obj  # since sum=1, this IS the normalized objective
                print(f"    [{it}] obj={actual_obj:.6f} alpha={alpha:.3f} t={elapsed:.0f}s")
        else:
            no_improve += 1
            alpha *= 0.6
            if no_improve > 60:
                break

    # Return with actual objective (sum might not be exactly 1 due to floating point)
    sa = float(np.sum(best_a))
    if sa < 1e-10:
        return a_init, compute_obj(a_init)
    best_a = best_a / sa
    actual_obj = 2.0 * n * float(np.max(np.convolve(best_a, best_a)))
    return best_a, actual_obj


def subgrad_optimize(a_init, deadline, verbose=False, vi=200):
    """Subgradient descent with momentum for max(conv) minimization."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = float(np.sum(a))
    a = a / S

    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(np.convolve(a, a)))
    lr = 0.005
    momentum = np.zeros(n)
    mu = 0.5
    no_improve = 0
    it = 0
    top_k = 5

    while time.monotonic() < deadline:
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        if M < 1e-15:
            break

        # Compute subgradient using top-k lags
        top_idx = np.argsort(conv)[-top_k:]
        mean_conv = float(np.sum(conv)) / (2 * n - 1)
        weights = np.maximum(conv[top_idx] - mean_conv, 0.0)
        ws = float(np.sum(weights))
        if ws < 1e-15:
            weights = np.ones(len(top_idx)) / len(top_idx)
            ws = 1.0
        else:
            weights = weights / ws

        grad = np.zeros(n)
        for k, w in zip(top_idx, weights):
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            grad[j_lo:j_hi + 1] += 2.0 * w * a[k - np.arange(j_lo, j_hi + 1)]

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Momentum update
        momentum = mu * momentum - lr * grad / gn
        new_a = a + momentum
        new_a = np.clip(new_a, 0.0, None)
        ns = float(np.sum(new_a))
        if ns < 1e-15:
            lr *= 0.5
            no_improve += 1
            if no_improve > 200:
                break
            it += 1
            continue
        new_a = new_a / ns

        new_obj = 2.0 * n * float(np.max(np.convolve(new_a, new_a)))
        it += 1

        if new_obj < best_obj * (1.0 - 1e-9):
            best_obj = new_obj
            best_a = new_a.copy()
            a = new_a
            no_improve = 0
            lr = min(lr * 1.01, 0.02)
            if verbose and it % vi == 0:
                print(f"    [sub it={it}] obj={best_obj:.6f} lr={lr:.5f}")
        else:
            no_improve += 1
            lr *= 0.95
            momentum *= 0.5
            if lr < 1e-7 or no_improve > 300:
                break

    sa = float(np.sum(best_a))
    if sa < 1e-10:
        return a_init, compute_obj(a_init)
    best_a = best_a / sa
    actual_obj = 2.0 * n * float(np.max(np.convolve(best_a, best_a)))
    return best_a, actual_obj


def make_inits(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Uniform
    inits.append(('uniform', np.ones(n)))

    # Raised cosine (Hann window)
    rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * idx / (n - 1)))
    inits.append(('hann', rc))

    # Gaussian, various widths
    for sigma_frac in [0.15, 0.2, 0.25, 0.35]:
        sigma = sigma_frac * n
        gauss = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
        inits.append((f'gauss_{sigma_frac}', gauss))

    # Sine (zero at edges)
    sine = np.sin(np.pi * (idx + 0.5) / n)
    inits.append(('sine', np.maximum(sine, 0.01)))

    # Linear taper
    taper = np.minimum(idx, n - 1 - idx) / (n / 2.0) + 0.1
    inits.append(('taper', taper))

    # Bimodal
    bim = (np.exp(-0.5 * ((idx - 0.25 * n) / (0.1 * n)) ** 2) +
           np.exp(-0.5 * ((idx - 0.75 * n) / (0.1 * n)) ** 2))
    inits.append(('bimodal', bim))

    # Random (fixed seed for reproducibility)
    r = rng.uniform(0.1, 2.0, size=n)
    inits.append(('random', r))

    return inits
