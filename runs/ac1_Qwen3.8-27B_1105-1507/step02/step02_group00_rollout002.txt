"""
Multi-phase optimization combining symmetric LP alternation, log-sum-exp smoothed
gradient descent (L-BFGS-B), projected subgradient polishing, and coordinate refinement
to minimize 2n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def compute_obj(a):
    """Compute objective: 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 1e-10:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A_from_a(a):
    """Build matrix A where (A @ g)[k] = (a*g)_k for valid convolution indices."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # (a*g)_k = sum_{j=j_start}^{j_end} a[k-j] * g[j]
        idx = k - np.arange(j_start, j_end + 1)
        A[k, j_start:j_end + 1] = a[idx]
    return A


def solve_lp_given_fixed(fixed, M):
    """Solve LP: max sum(x) s.t. (fixed*x)_k <= M for all k, x >= 0."""
    n = len(fixed)
    A = build_A_from_a(fixed)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_alternate_symmetric(a_init, max_iter=500, deadline=None, verbose=False):
    """Symmetric LP alternation: alternate solving for a and g."""
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    prev_obj = best_obj
    stall_count = 0

    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Compute current M
        conv_a = np.convolve(a, a)
        M = float(np.max(conv_a))

        # Solve for g given a
        g = solve_lp_given_fixed(a, M)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Solve for a' given g
        conv_g = np.convolve(g, g)
        M_g = float(np.max(conv_g))
        a_new = solve_lp_given_fixed(g, M_g)
        if a_new is None:
            a_new = g
        a_new_sum = np.sum(a_new)
        if a_new_sum < 1e-10:
            a_new = g
        else:
            a_new = a_new / a_new_sum

        # Check objectives
        obj_g = compute_obj(g)
        obj_a_new = compute_obj(a_new)

        if obj_g < best_obj:
            best_obj = obj_g
            best_a = g.copy()
            stall_count = 0
        else:
            stall_count += 1

        if obj_a_new < best_obj:
            best_obj = obj_a_new
            best_a = a_new.copy()
            stall_count = 0

        # Update a with blending
        t_blend = 0.7
        a = (1.0 - t_blend) * a + t_blend * a_new
        a = np.maximum(a, 0.0)
        s = np.sum(a)
        if s < 1e-10:
            a = a_new
        else:
            a = a / s

        # Convergence check
        cur_obj = compute_obj(a)
        if abs(cur_obj - prev_obj) < 1e-10:
            stall_count += 10
        prev_obj = cur_obj

        if stall_count > 50:
            if verbose:
                print(f"    LP stall at iter {it}, obj={best_obj:.8f}")
            break

        if verbose and it % 100 == 0:
            print(f"    LP it={it}: best={best_obj:.8f}, cur={cur_obj:.8f}")

    return best_a, best_obj


def lse_objective_and_grad(x, t):
    """Compute log-sum-exp smoothed objective and gradient w.r.t. x (pre-softmax).
    
    a = softmax(x), objective = 2n * (1/t) * log(sum_k exp(t * (a*a)_k))
    (sum(a)=1 by construction, so obj = 2n * smoothed_max)
    """
    n = len(x)
    # Softmax
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    a = exp_x / np.sum(exp_x)

    # Convolution
    conv = np.convolve(a, a)  # length 2n-1

    # Log-sum-exp: (1/t)*log(sum exp(t*conv_k))
    max_conv = np.max(conv)
    t_conv = t * (conv - max_conv)
    # Clip for numerical stability
    t_conv = np.clip(t_conv, -700, 700)
    exp_t_conv = np.exp(t_conv)
    denom = np.sum(exp_t_conv)
    if denom < 1e-300:
        return max_conv, np.zeros(n)
    weights = exp_t_conv / denom  # softmax of t*conv

    # Smoothed value (we return the full objective 2n * smoothed_max)
    smooth_max = max_conv + np.log(denom) / t
    obj_val = 2.0 * n * smooth_max

    # Gradient w.r.t. a: grad_a_j = sum_k weights[k] * a[k-j] * I(0<=k-j<n)
    grad_a = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = k - np.arange(j_start, j_end + 1)
        grad_a[j_start:j_end + 1] += w * a[idx]

    # Chain rule: grad_x_i = sum_j grad_a_j * da_j/dx_i = sum_j grad_a_j * a_j * (delta_ji - a_i)
    # = a_i * (grad_a_i - a_i * sum_j grad_a_j)
    # Wait: da_j/dx_i = a_j * (delta_{ji} - a_i)
    # grad_x_i = sum_j grad_a_j * a_j * (delta_{ji} - a_i)
    #          = grad_a_i * a_i - a_i * sum_j grad_a_j * a_j
    grad_x = a * (grad_a - a * np.sum(grad_a * a))
    grad_x *= 2.0 * n  # Scale by 2n

    return obj_val, grad_x


def smooth_optimize(a_init, t_values, max_iter_per_t=200, deadline=None, verbose=False):
    """Optimize using log-sum-exp smoothing with increasing t."""
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)  # Ensure positive for log
    s = np.sum(a)
    a = a / s

    # Convert to log parameterization
    x = np.log(a)

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for t_val in t_values:
        if deadline is not None and time.monotonic() > deadline:
            break

        def obj_and_grad(x_curr):
            val, grad = lse_objective_and_grad(x_curr, t_val)
            return val, grad

        res = minimize(
            obj_and_grad, x,
            method='L-BFGS-B',
            options={'maxiter': max_iter_per_t, 'maxfun': max_iter_per_t * 3,
                     'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 50}
        )

        # Convert back
        x_opt = res.x
        a_opt = np.exp(x_opt - np.max(x_opt))
        a_opt = a_opt / np.sum(a_opt)
        a_opt = np.maximum(a_opt, 0.0)

        # Evaluate true objective
        true_obj = compute_obj(a_opt)
        if true_obj < best_obj:
            best_obj = true_obj
            best_a = a_opt.copy()

        # Continue from this point
        x = np.log(np.maximum(a_opt, 1e-15))

        if verbose:
            print(f"    LSE t={t_val}: true_obj={best_obj:.8f}, res_status={res.status}")

        if deadline is not None and time.monotonic() > deadline - 30:
            break

    return best_a, best_obj


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} using bisection on tau."""
    n = len(v)
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.maximum(0.0, v - mid))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    result = np.maximum(0.0, v - tau)
    # Normalize to be safe
    s = np.sum(result)
    if s > 1e-15:
        result = result / s
    return result


def subgradient_polish(a_init, n_steps=5000, deadline=None, verbose=False):
    """Projected subgradient method for polishing."""
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        conv = np.convolve(a, a)
        max_conv = np.max(conv)
        if max_conv < 1e-15:
            break

        # Active set: top fraction of convolution values
        # Use a relative threshold that tightens over time
        frac = 0.05 + 0.45 * (step / n_steps)  # Start with top 50%, end with top 5%
        threshold = np.percentile(conv, (1.0 - frac) * 100.0)
        active = conv >= threshold
        n_active = np.sum(active)
        if n_active == 0:
            active = conv >= max_conv - 1e-12
            n_active = np.sum(active)

        # Subgradient: average over active set
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = k - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += a[idx]
        grad /= n_active

        # Adaptive step size (decreasing)
        lr = 2.0 / (1.0 + 0.005 * step)

        a_new = a - lr * grad
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def coord_refine(a_init, n_rounds=5, deadline=None):
    """Coordinate descent with golden-section-like search per coordinate."""
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    s = np.sum(a)
    if s < 1e-10:
        return a, compute_obj(a)
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-15:
                # Try turning on
                a[j] = old_val + 0.01
            else:
                # Try several multipliers
                best_local = best_obj
                best_x = old_val
                multipliers = [0.0, 0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]
                for mult in multipliers:
                    x = old_val * mult
                    a[j] = x
                    obj = compute_obj(a)
                    if obj < best_local:
                        best_local = obj
                        best_x = x
                a[j] = best_x
                if best_x != old_val:
                    improved = True

        # Renormalize
        s = np.sum(a)
        if s > 1e-10:
            a = a / s
            cur_obj = compute_obj(a)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()

        if not improved:
            break

    return best_a, best_obj


def structured_init(n, rng):
    """Generate diverse initializations."""
    choice = int(rng.integers(0, 7))
    idx = np.arange(n)
    if choice == 0:
        # Alternating (two-tone)
        a = np.where(idx % 2 == 0, 1.0, rng.uniform(0.3, 0.7))
    elif choice == 1:
        # Cosine with random frequency
        freq = rng.uniform(1, 5)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * idx / n + rng.uniform(0, 2 * np.pi))
    elif choice == 2:
        # Periodic pattern
        period = int(rng.integers(max(2, n // 100), max(3, n // 10)))
        pattern = rng.random(period)
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 3:
        # Gaussian bump
        center = rng.uniform(0.2, 0.8) * n
        width = rng.uniform(n // 10, n // 3)
        a = np.exp(-0.5 * ((idx - center) / width) ** 2)
    elif choice == 4:
        # Linear ramp
        a = np.linspace(rng.uniform(0.1, 1), rng.uniform(0.1, 1), n)
    elif choice == 5:
        # Two-lobe
        mid = n // 2
        a = np.ones(n)
        a[:mid] = rng.uniform(0.5, 1.5)
        a[mid:] = rng.uniform(0.3, 0.8)
    else:
        # Pure random
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        a = rng.random(n)
        s = np.sum(a)
    return a / s

        print(f"  Coord refine: n={n_best}, {remaining:.0f}s left")
        # For large n, limit rounds
        if n_best <= 500:
            n_rounds = min(5, int(remaining / (n_best * 0.002)))
        else:
            n_rounds = min(3, int(remaining / (n_best * 0.005)))
        n_rounds = max(1, n_rounds)

        a4, obj4 = coord_refine(best_seq, n_rounds=n_rounds, deadline=deadline - 5)
        if obj4 < best_score:
            best_score = obj4
            best_seq = a4.copy()
            print(f"  After Coord: obj={obj4:.8f}")

    # Phase 5: Final LP pass on best
    print("\n=== Phase 5: Final LP ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        if n_best <= 1000 and remaining > 10:
            max_iter = int(min(300, remaining / 0.1))
            a5, obj5 = lp_alternate_symmetric(best_seq, max_iter=max_iter,
                                               deadline=deadline - 3)
            if obj5 < best_score:
                best_score = obj5
                best_seq = a5.copy()
                print(f"  Final LP: obj={obj5:.8f}")

    # Verify and return
    if best_seq is None:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)

    # Final safety: clip and ensure valid
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    s = np.sum(best_seq)
    if s < 0.01:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)

    # Verify with evaluate_sequence
    try:
        final_score = evaluate_sequence([float(x) for x in best_seq])
        if final_score < best_score:
            best_score = final_score
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.8f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
