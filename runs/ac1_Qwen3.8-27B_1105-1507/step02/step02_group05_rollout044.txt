import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, coo_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def build_A_rows(a, k_indices):
    """Build sparse rows for given k indices. A[idx, j] = a[k-j] if valid."""
    n = len(a)
    n_rows = len(k_indices)
    A_mat = lil_matrix((n_rows, n))
    for idx, k in enumerate(k_indices):
        k = int(k)
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            for j in range(jl, jh + 1):
                A_mat[idx, j] = a[k - j]
    return A_mat.tocsr()


def lp_best_response(a, M, k_indices=None):

def get_peak_indices(b, topk=30):
    """Get indices of top-k convolution coefficients, plus a margin."""
    n = len(b)

        return list(range(n))
    # Get top-k and also some neighbors for robustness

    # Expand to include neighbors
    expanded = set()
    for idx in top_idx:
        for delta in range(-3, 4):
            ni = int(idx) + delta
            if 0 <= ni < n:
                expanded.add(ni)
    return sorted(expanded)


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

        # This is a weighted convolution of a with w
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            jl = max(0, k - n + 1)
            jh = min(n - 1, k)
            for j in range(jl, jh + 1):
                grad[j] += alpha * w[k] * a[k - j]

        # Gradient step
        a_trial = a - lr * grad
        a_trial = project_simplex(a_trial)
        J_trial = compute_J(a_trial)

        if J_trial < best_J - 1e-12:
            best_J = J_trial
            best_a = a_trial.copy()
            a = a_trial
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.8
            if lr < 0.001:
                break

        if it % 10 == 0:
            tracker.update(a)

    tracker.update(best_a)
    return best_a


def coordinate_descent(a, max_iters, deadline, tracker):
    """Coordinate descent: move mass from high-peak-contribution to low-peak-contribution coords."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = a / a.sum()
    n = len(a)
    current_J = compute_J(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        k_star = int(np.argmax(b))

        # Find move that most reduces b[k_star]
        # Moving delta from i to j changes b[k_star] by:
        # -delta * a[k_star - i] + delta * a[k_star - j] + delta^2 * (corrections)
        # To first order: delta * (a[k_star - j] - a[k_star - i])
        # We want to minimize b[k_star], so move from i where a[k_star-i] is large
        # to j where a[k_star-j] is small

        def conv_at(i):
            """Get a[k_star - i] if valid, else 0."""
            idx = k_star - i
            if 0 <= idx < n:
                return a[idx]
            return 0.0

        # Find best (src, dst) pair
        best_improve = 0
        best_src = -1
        best_dst = -1
        delta = 0.05

        # Sample a subset of coordinates for efficiency
        if n <= 512:
            candidates_src = list(range(n))
            candidates_dst = list(range(n))
        else:
            candidates_src = list(range(0, n, 2))
            candidates_dst = list(range(0, n, 2))

        for i in candidates_src:
            ci = conv_at(i)
            if ci < 1e-8:
                continue
            for j in candidates_dst:
                if j == i:
                    continue
                cj = conv_at(j)
                improve = ci - cj  # positive means moving from i to j helps
                if improve > best_improve:
                    best_improve = improve
                    best_src = i
                    best_dst = j

        if best_src < 0 or best_improve < 1e-10:
            break

        # Perform the move
        a_trial = a.copy()
        move = min(delta * a[best_src], 0.3 * a[best_src])
        a_trial[best_src] -= move
        a_trial[best_dst] += move
        a_trial = np.maximum(a_trial, 0)
        a_trial = a_trial / a_trial.sum()

        J_trial = compute_J(a_trial)
        if J_trial < current_J - 1e-12:
            a = a_trial
            current_J = J_trial
            if it % 50 == 0:
                tracker.update(a)
        else:
            # Try smaller delta
            delta *= 0.5
            if delta < 1e-4:
                break

    tracker.update(a)
    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))
        inits.append(resample_seq(parent, n)[::-1])  # reversed

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    # Cosine families
    for freq in [2, 3, 4, 5, 7, 11]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr))
        inits.append(1.0 + 0.5 * np.cos(2.0 * np.pi * freq * t_arr + rng.uniform(0, 6.28)))

    # Gaussian bumps
    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        width = rng.uniform(0.05, 0.2)
        inits.append(np.exp(-0.5 * ((t_arr - center) / width) ** 2) + 0.02)

    # Double bump
    inits.append(np.exp(-5.0 * (t_arr - 0.3) ** 2) + np.exp(-5.0 * (t_arr - 0.7) ** 2) + 0.02)

    # Triangle / tent
    inits.append(1.0 - 2.0 * np.abs(t_arr - 0.5) + 0.1)

    # Power laws
    inits.append((t_arr + 0.05) ** 0.5 + (1 - t_arr + 0.05) ** 0.5 + 0.01)
    inits.append(1.0 / (t_arr + 0.3) + 1.0 / (1 - t_arr + 0.3) + 0.01)

    # Sine abs
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.1)
    inits.append(np.abs(np.sin(2 * np.pi * t_arr)) + 0.1)

    # Sparse spikes (smoothed)
    n_spikes = max(4, n // 30)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse = np.zeros(n)
    sparse[positions] = 1.0
    ksize = max(3, n // 50)
    kernel = np.ones(ksize) / ksize
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.03)

    # Random smooth (low-pass filtered noise)
    noise = rng.standard_normal(n)
    ksize = max(3, n // 20)
    kernel = np.ones(ksize) / ksize
    smooth = np.convolve(noise, kernel, mode='same')
    inits.append(np.abs(smooth) + 0.1)

    # Workshop-style: near-uniform with small perturbation
    inits.append(np.ones(n) + rng.uniform(-0.1, 0.1, n))

    return [np.maximum(x, 0.001) for x in inits]


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    1. Multi-scale LP Nesterov (n = 128 to 2048)
    2. Focused LP refinement on best n
    3. Gradient polish (smoothed max)
    4. Coordinate descent
    5. Final LP + gradient combo
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Multi-scale LP Nesterov ===
    if parent is not None:
        base_n = parent.size
    else:
        base_n = 128

    # Choose lengths to explore
    lengths_to_try = []
    for n in [64, 128, 256, 512, 1024, 2048]:
        if n >= min(64, base_n):
            lengths_to_try.append(n)
    # Make sure we include the parent length and some larger ones
    if base_n not in lengths_to_try:
        lengths_to_try.append(base_n)
    lengths_to_try = sorted(set(lengths_to_try))

    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Multi-scale LP Nesterov, lengths: {lengths_to_try}", flush=True)
    print(f"{'='*60}", flush=True)

    for n in lengths_to_try:
        if time.monotonic() > deadline - 100:
            break
        t0 = time.monotonic()
        print(f"\n--- LP Nesterov n={n} ---", flush=True)

        inits = generate_initializations(n, rng, parent)
        # More iterations for smaller n
        iters_per_init = max(50, min(200, int(60000 // n)))
        topk = min(50, 2 * n - 1)

        best_this_n = None
        best_J_this_n = float('inf')

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline - 100:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 = a0 / a0.sum()
            a = run_lp_nesterov(a0, iters_per_init, deadline - 100, tracker,
                               topk=topk, beta=0.7)
            J = compute_J(a)
            if J < best_J_this_n:
                best_J_this_n = J
                best_this_n = a.copy()

        if best_this_n is not None:
            print(f"  n={n} best: J={best_J_this_n:.8f} ({time.monotonic()-t0:.1f}s)", flush=True)
            tracker.update(best_this_n)

    tracker.report("After Phase 1")

    # === Phase 2: Focused LP refinement on best length ===
    if tracker.best is not None and time.monotonic() < deadline - 80:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: Focused LP refinement n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        # Run more LP iterations with tighter constraints
        a = run_lp_nesterov(tracker.best, 300, deadline - 80, tracker,
                           topk=60, beta=0.8)

        # Then try with full constraints for a few iterations
        a_full = tracker.best.copy()
        a_full = a_full / a_full.sum()
        for it in range(50):
            if time.monotonic() > deadline - 80:
                break
            b = fast_conv(a_full)
            M = float(b.max())
            g = lp_best_response(a_full, M, None)  # full constraints
            if g is None:
                break
            g = g / g.sum()
            a_trial = 0.8 * a_full + 0.2 * g
            a_trial = project_simplex(a_trial)
            J_trial = compute_J(a_trial)
            J_curr = compute_J(a_full)
            if J_trial < J_curr - 1e-14:
                a_full = a_trial
            else:
                break
        tracker.update(a_full)

        print(f"  Phase 2 done ({time.monotonic()-t0:.1f}s)", flush=True)
        tracker.report("After Phase 2")

    # === Phase 3: Gradient polish ===
    if tracker.best is not None and time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Gradient polish n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        # Try multiple alpha values
        for alpha in [200, 500, 1000, 2000]:
            if time.monotonic() > deadline - 60:
                break
            a = gradient_polish(tracker.best, max_iters=200,
                               deadline=deadline - 60, tracker=tracker, alpha=alpha)

        tracker.report("After Phase 3")

    # === Phase 4: Coordinate descent ===
    if tracker.best is not None and time.monotonic() < deadline - 40:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Coordinate descent n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        if n <= 1024:
            a = coordinate_descent(tracker.best, max_iters=500,
                                   deadline=deadline - 40, tracker=tracker)

        tracker.report("After Phase 4")

    # === Phase 5: Final combined polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 5: Final polish n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        # Interleave LP and gradient
        a = tracker.best.copy()
        a = a / a.sum()
        for round_idx in range(10):
            if time.monotonic() > deadline:
                break
            # LP step
            a = run_lp_nesterov(a, max_iters=30, deadline=deadline,
                               tracker=tracker, topk=40, beta=0.7)
            # Gradient step
            a = gradient_polish(a, max_iters=50, deadline=deadline,
                               tracker=tracker, alpha=1000)

        tracker.report("After Phase 5")

    # === Phase 6: Try L-BFGS-B on the best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 6: L-BFGS-B n={n}", flush=True)
        print(f"{'='*60}", flush=True)

        if n <= 2048:
            def lbb_objective(x):
                ax = np.maximum(x, 0.0)
                s = ax.sum()
                if s < 1e-8:
                    return 1e10
                ax = ax / s
                b = fast_conv(ax)
                return float(b.max())

            bounds_list = [(0.0, None)] * n
            try:
                res = minimize(lbb_objective, tracker.best.astype(np.float64),
                             method='L-BFGS-B', bounds=bounds_list,
                             options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-16})
                if res.fun < 1e10:
                    tracker.update(res.x)
            except Exception as e:
                print(f"  L-BFGS-B error: {e}", flush=True)

        tracker.report("After Phase 6")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
