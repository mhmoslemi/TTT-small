            break

            continue

            best_f = f.copy()

            break
        
        sweep += 1
        if verbose and sweep % 10 == 0:

def minimize_1d(func, a, b, tol=1e-8, max_iter=100):
    """Golden section minimization on [a, b]."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = func(d)
    x = (a + b) / 2
    return x, func(x)

Coordinate descent with exact 1D subproblem solves for minimizing
2n * max(conv(f,f)) / sum(f)^2.
"""
import numpy as np
import time

def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    max_b = np.max(conv)
    sum_a = np.sum(f)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def golden_min(func, a, b, xatol=1e-10, max_iter=200):
    """Golden section minimization (unconstrained, on [a,b])."""
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    invphi2 = invphi * invphi
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if b - a < xatol:
            break
        if fc <= fd:
            b = d; d = c; fd = fc
            c = b - invphi * (b - a)
            fc = func(c)
        else:
            a = c; c = d; fc = fd
            d = a + invphi * (b - a)
            fd = func(d)
    xopt = (a + b) / 2.0
    return xopt, func(xopt)

def solve_coord(f, i, conv_cache=None):
    """Solve 1D subproblem for coordinate i."""
    n = len(f)
    if conv_cache is None:
        conv = np.convolve(f, f)
    else:
        conv = conv_cache
    fi = f[i]
    S = np.sum(f)
    Sp = S - fi
    
    if Sp < 1e-15:
        return 0.0
    
    # a[k] = f[k-i] for valid k
    # k valid: 0 <= k-i < n => i <= k < i+n
    # k range: [0, 2n-2]
    k_start = max(0, i)
    k_end = min(2*n-1, i+n)  # exclusive
    a = np.zeros(2*n-1)
    if k_end > k_start:
        a[k_start:k_end] = f[0:k_end-k_start]
    
    # base[k] = conv[k] - 2*fi*a[k], then correct base[2i]
    base = conv - 2.0 * fi * a
    two_i = 2 * i
    if 0 <= two_i < 2*n-1:
        base[two_i] = conv[two_i] - fi * fi
    
    # Define h(x)
    def h(x):
        val = base + 2.0 * x * a
        val[two_i] = base[two_i] + x * x
        mx = np.max(val)
        return mx / (Sp + x) ** 2
    
    # Search bounds
    x_lo = 0.0
    x_hi = max(3.0 * fi + 1.0, 3.0 * Sp / n, 0.1)
    x_hi = min(x_hi, 1000.0)
    
    x_opt, h_opt = golden_min(h, x_lo, x_hi, xatol=1e-9)
    return max(0.0, x_opt)

def cd_sweep(f, verbose=False):
    """One sweep of coordinate descent. Returns updated f and improvement."""
    n = len(f)
    f_new = f.copy()
    # Forward and backward sweeps
    for i in range(n):
        f_new[i] = solve_coord(f_new, i)
    for i in range(n-1, -1, -1):
        f_new[i] = solve_coord(f_new, i)
    return f_new

def run_cd(f, deadline, max_sweeps=500, verbose_prefix=""):
    """Run coordinate descent until deadline or max_sweeps."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    
    no_improve = 0
    sweep = 0
    while time.monotonic() < deadline and sweep < max_sweeps:

            no_improve = 0

        sweep += 1
        if verbose_prefix and sweep % 20 == 0:
            print(f"  {verbose_prefix}CD sweep {sweep}: {best_score:.6f}")
    
    return best_f, best_score
