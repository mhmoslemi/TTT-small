import numpy as np
from scipy.optimize import linprog
import time

# Fast convolution
def fft_conv(a, b=None):
    if b is None:
        b = a
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    conv = fft_conv(a)
    return 2.0 * n * np.max(conv) / (s * s)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s

def grad_maxconv(a, k_peak):
    """Exact gradient of conv(a,a)[k_peak] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        idx = k_peak - j
        if 0 <= idx < n:
            g[j] = 2.0 * a[idx]
    return g

def smooth_grad(a, conv, tau):
    """Smoothed gradient using log-sum-exp."""
    n = len(a)
    max_c = np.max(conv)
    w = np.exp(np.clip(tau * (conv - max_c), -500, 0))
    w /= np.sum(w)
    # grad = 2 * sum_k w_k * a[k-j] for valid k-j
    # This is 2 * correlate(w, a, 'full')[:n]
    c = np.correlate(w, a, mode='full')
    return 2.0 * c[:n]

    A = np.zeros((2*n-1, n))

        j_max = min(n-1, k)
        if j_min <= j_max:
            A[k, j_min:j_max+1] = a[k - (j_max:j_min-1)][::-1]

    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0,None)]*n, method='highs')
    if res.status == 0:

import numpy as np

def fft_conv(a, b=None):
    """Fast full convolution using FFT."""
    if b is None:
        b = a
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-10:
        return np.ones(len(a)) / len(a)
    return a / s


def obj_value(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def smooth_gradient(a, conv, tau):
    """Smoothed gradient of max(conv) via log-sum-exp weighting."""
    n = len(a)
    max_c = np.max(conv)
    shifted = np.clip(tau * (conv - max_c), -600, 0)
    w = np.exp(shifted)
    wsum = np.sum(w)
    if wsum < 1e-300:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= wsum
    c = np.correlate(w, a, mode='full')
    return 2.0 * c[:n]


def build_constraint_matrix(a, n):
    """Build A such that A@g = conv(a, g) for g of length n."""
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def lp_direction(a, M, n, time_limit=15):
    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Returns (g, sum_g, dual_y) or (None, None, None)."""
    A = build_constraint_matrix(a, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            dual = res.ineqlin.marginals if res.ineqlin is not None else None
            return res.x, float(-res.fun), dual
    except Exception:
        pass
    return None, None, None

def dual_guided_step(a, M, n):
    """Use LP dual to construct a weighted sum objective and minimize it."""
    _, _, dual = lp_direction(a, M, n, time_limit=8)

    conv_weights = dual
    c = np.correlate(conv_weights, a, mode='full')
    grad = 2.0 * c[:n]
    grad -= np.mean(grad)
    return grad

    parent = None
    try:
        p = np.array(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0, 1000.0)
        parent = normalize(p.copy())
        o_p = obj_value(parent)
        print(f"Parent: n={len(parent)}, obj={o_p:.7f}")
        best_score = o_p
        best_seq = parent.copy()
    except Exception as e:
        print(f"No parent available: {e}")

    # Determine lengths to explore
    if parent is not None:
        n_parent = len(parent)
        n_values = sorted(set(
            [max(20, n_parent - 60), max(20, n_parent - 30), n_parent,
             min(500, n_parent + 30), min(500, n_parent + 60)] +
            [100, 150, 200, 250, 300, 350, 400]
        ))
        # Remove very large ones to keep LP feasible
        n_values = [n for n in n_values if n <= 500]
    else:
        n_values = [50, 100, 150, 200, 250, 300]

    print(f"Lengths to explore: {n_values}")

    # Time allocation: more time for parent length, less for others
    n_alloc = {}
    total_alloc = 0
    for n in n_values:
        if parent is not None and n == n_parent:
            n_alloc[n] = 0.30
        elif parent is not None and abs(n - n_parent) <= 30:
            n_alloc[n] = 0.12
        elif n <= 150:
            n_alloc[n] = 0.06
        else:
            n_alloc[n] = 0.05
        total_alloc += n_alloc[n]
    # Normalize
    for n in n_values:
        n_alloc[n] *= 0.85 * (budget_s - 60) / total_alloc  # Keep 60s for final polish

    # Sort by priority (parent length first)
    n_sorted = sorted(n_values, key=lambda x: -n_alloc[x])

    for n in n_sorted:
        if time.monotonic() > deadline:
            break
        alloc = n_alloc[n]
        remaining = deadline - time.monotonic()
        alloc = min(alloc, remaining)
        if alloc < 15:
            break

        print(f"\n--- n={n}, alloc={alloc:.0f}s ---")

        # If this is the parent length, start from parent
        if parent is not None and n == len(parent):
            a_init_list = [parent.astype(np.float64)]
        else:
            a_init_list = []

        # Generate diverse initializations
        n_inits = max(2, int(alloc / 25))
        rng = np.random.default_rng(12345 + n * 7919)
        for i in range(n_inits):
            a_init_list.append(generate_init(n, i % 7, rng))

        # Optimize each initialization
        for idx, a_init in enumerate(a_init_list[:max(3, n_inits)]):
            if time.monotonic() > deadline:
                break
            t_init_start = time.monotonic()
            remaining_for_init = deadline - t_init_start
            if remaining_for_init < 10:
                break

            t_budget = min(alloc / max(1, n_inits) * 1.5, remaining_for_init * 0.9)
            t_budget = max(10.0, t_budget)

            a_opt, o_opt = optimize_full(a_init, max_time=t_budget)
            # Final quick LP polish
            if time.monotonic() < deadline - 5:
                a_opt, o_opt2 = lp_optimize(a_opt, max_iter=20, t_init=0.2)
                if o_opt2 < o_opt:
                    o_opt = o_opt2

            if o_opt < best_score:
                best_score = o_opt
                best_seq = a_opt.copy()
                elapsed_init = time.monotonic() - t_init_start
                print(f"  n={n} init{idx}: obj={best_score:.7f} ({elapsed_init:.1f}s)")

    # Final intensive polish on the best sequence found
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== FINAL POLISH: n={n_best}, {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()
        round_num = 0

        while time.monotonic() < deadline - 5:
            round_num += 1
            remaining = deadline - time.monotonic()
            if remaining < 5:
                break

            # LP step
            if remaining > 10:
                a_lp, o_lp = lp_optimize(a_cur, max_iter=min(40, int(remaining * 0.4)),
                                         t_init=0.25, t_decay=0.96)
                if o_lp < best_score:
                    best_score = o_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy()
                    print(f"  Round {round_num} LP: {best_score:.7f}")
                else:
                    a_cur = a_lp if obj_value(a_lp) < obj_value(a_cur) else a_cur

            # Peak deflation
            if remaining > 5:
                a_pd, o_pd = peak_deflation(a_cur, n_steps=min(500, int(remaining * 2)),
                                            step=0.03)
                if o_pd < best_score:
                    best_score = o_pd
                    best_seq = a_pd.copy()
                    a_cur = a_pd.copy()
                    print(f"  Round {round_num} PeakDef: {best_score:.7f}")

            # Subgradient
            if remaining > 3:
                a_sg, o_sg = subgradient_optimize(a_cur, max_iter=int(min(3000, remaining * 3)),
                                                  tau_start=20.0, tau_end=100.0, step_init=0.015)
                if o_sg < best_score:
                    best_score = o_sg
                    best_seq = a_sg.copy()
                    a_cur = a_sg.copy()
                    print(f"  Round {round_num} SG: {best_score:.7f}")

            # Small perturbation to escape local minima
            if remaining > 8 and round_num % 3 == 0:
                rng_p = np.random.default_rng(round_num * 31)
                a_pert = a_cur * np.exp(rng_p.normal(0, 0.03, n_best))
                a_pert = normalize(a_pert)
                o_pert = obj_value(a_pert)
                if o_pert < best_score:
                    best_score = o_pert
                    best_seq = a_pert.copy()
                    a_cur = a_pert.copy()
                    print(f"  Round {round_num} Perturb: {best_score:.7f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = obj_value(best_seq)

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.7f}, time={elapsed:.1f}s")

    # Validate with evaluate_sequence
    result = [float(x) for x in best_seq]
    val = evaluate_sequence(result)
    if val < np.inf and val != best_score:
        print(f"  Validation: computed={best_score:.7f}, validated={val:.7f}")
        if val < best_score:
            best_score = val

    return result
