    if parent_valid:

        new_seq, new_score = optimize_sequence(parent_seq)
        candidate_sequences.append(new_seq)
        if new_score < best_score:
            best_seq = new_seq
            best_score = new_score
    else:
        # No parent sequence: initialize with Gaussian
        fallback_seq, _ = optimize_sequence(fallback_normalized_sequence(), config_time_fraction)
        candidate_sequences.append(fallback_seq)
        
    # Execute over time budget
    remaining_time = total_time
    
    # Optimization over multiple sequences
    if remaining_time > 30:  # Total time > 30 seconds

        # Handle multiple length searches

            for n_val in range(30, 351).__iter__():  # Aside from the parent, individually try n from 30 to 350 (ranges slightly) but cap at 1000? No, not for this method's structured fallback
                if remaining_time < 5:  # Less than 5 seconds left, break inner

                # Time alloting per inner candidate
                time_allot = min(remaining_time * config_time_fraction, 0.1 + n_val/10000)
                random_factor = 2 if random() < 0.4 else 1  # Double time on random with probability 0.4
                time_allot *= random_factor

                    candidate_a = candidate_a / norm_factor * 1000.0  # Scale to proxy for normalization
                result_obj = objective_function(candidate_a)
                if result_obj < best_score:  # locally improved
                    best_seq = candidate_a.copy()
                    best_score = result_obj
                    # Refine this candidate further
                    ref_seq, obj_val = optimize_sequence(best_seq, time_allot)
                    if obj_val < result_obj:
                        result_obj = obj_val
                        best_seq = ref_seq
                        candidate_sequences.append(best_seq)

                if result_obj - best_score < 1e-6:  # If not much better, we have found a local minima

                else:
                    candidate_sequences.append(best_seq)
            # Closeout assessments
            best_objective = min([objective_function(seq) for seq in candidate_sequences], default=None)
            if best_objective is not None:
