Optimization of step function heights to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses FFT-accelerated exponentiating gradient with adaptive softmax temperature,
followed by active-set LP polishing and coordinate descent.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve(a, b):
    """Fast full linear convolution using FFT."""
    n = len(a) + len(b) - 1
    f = 1
    while f < n:
        f *= 2
    A = np.fft.rfft(a, f)
    B = np.fft.rfft(b, f)
    return np.fft.irfft(A * B, f)[:n]


def compute_score_fast(f):
    """Compute score using FFT-based convolution."""
    n = len(f)
    conv = fft_convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))


def softmax_gradient(f, temperature):
    """Compute gradient of smooth max of convolution using FFT."""
    n = len(f)
    conv = fft_convolve(f, f)
    
    # Softmax weights (numerically stable)
    conv_shifted = conv - np.max(conv)
    weights = np.exp(temperature * conv_shifted)
    weights /= np.sum(weights)
    
    # grad_i = 2 * sum_j weights[i+j] * a[j] for j=0..n-1
    # = 2 * conv(w_rev, a)[2n-2-i] where w_rev = w[::-1]
    w_rev = weights[::-1]
    conv_wa = fft_convolve(w_rev, f)
    grad = 2.0 * conv_wa[n-1:2*n-1][::-1]
    
    return grad


def exp_gradient_optimize(f, deadline, lr=0.1, temp_init=0.1, temp_final=10.0,
                          verbose_prefix="", max_iter=100000):

    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    f = f * (S / S)  # normalize
    best_f = f.copy()
    best_score = compute_score_fast(f)
    n = len(f)
    
    it = 0
    no_improve = 0
    t = time.monotonic()
    total_time = deadline - t
    
    while time.monotonic() < deadline and it < max_iter:
        # Adaptive temperature: start low (closer to uniform), end high (closer to true max)
        progress = min(1.0, it / max(1, max_iter * 0.8))
        temp = temp_init + (temp_final - temp_init) * progress
        
        grad = softmax_gradient(f, temp)
        
        # Exponentiating gradient update
        # a_i <- a_i * exp(-lr * grad_i)
        # Normalize grad to have unit mean for stability
        grad_mean = np.mean(grad)
        grad_centered = grad - grad_mean
        
        f_new = f * np.exp(-lr * grad_centered)
        f_new = np.clip(f_new, 1e-15, None)
        
        # Renormalize to maintain sum
        sum_new = float(np.sum(f_new))
        if sum_new < 1e-15:
            lr *= 0.5
            no_improve += 1
            if no_improve > 100:

            it += 1
            continue
        f_new = f_new * (S / sum_new)

    top_indices = np.argsort(conv)[-top_k:]
    
    # Build constraint matrix: A@g <= M for the top constraints
    # But we want to use ALL constraints where conv is close to M
    threshold = M * 0.95  # Use constraints that are 95% of max
    active = np.where(conv >= threshold)[0]
    
    if len(active) == 0:
        return None
    
    # Build sparse constraint matrix for active constraints
    rows = []
    cols = []
    vals = []
    for k in active:
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    # Keep only active rows
    active_in_0_1 = np.zeros(2 * n - 1, dtype=bool)
    active_in_0_1[active] = True
    A = A[active_in_0_1, :]
    
    c = -np.ones(n)
    b_ub = np.full(len(active), M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def lp_optimize(f, deadline, top_k=5, step=0.1, verbose_prefix=""):
    """LP-based optimization."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = compute_score_fast(f)
    n = len(f)
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < 200:
        conv = fft_convolve(f, f)
        M = float(np.max(conv))
        if M <= 0:
            break
        
        g = lp_direction(f, M, top_k=top_k)
        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue
        
        sum_g = float(np.sum(g))
        if sum_g < 1e-15:
            no_improve += 1
            continue
        g = g * (S / sum_g)
        
        # Line search
        best_step_score = best_score
        best_step_f = None
        for alpha in [step, step*0.5, step*0.25, step*1.5]:
            if alpha > 1.0:
                alpha = 1.0
            candidate = (1.0 - alpha) * f + alpha * g
            candidate = np.clip(candidate, 0.0, None)
            cs = float(np.sum(candidate))
            if cs < 1e-15:
                continue
            candidate = candidate * (S / cs)
            cs_score = compute_score_fast(candidate)
            if np.isfinite(cs_score) and cs_score < best_step_score:
                best_step_score = cs_score
                best_step_f = candidate
        
        it += 1
        if best_step_f is not None:
            f = best_step_f
            best_f = f.copy()
            best_score = best_step_score
            no_improve = 0
            step *= 0.98
            if it % 10 == 0:
                print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f}")
        else:
            no_improve += 1
            step *= 0.7
            if step < 0.005 or no_improve > 30:
                break
    
    return best_f, best_score

def make_initializations(n, rng):
    """Create diverse initializations."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    
    inits.append(('uniform', np.ones(n)))
    
    cos_t = (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0
    inits.append(('cosine', cos_t))
    
    bell = np.exp(-0.5 * ((idx - (n-1)/2.0) / (n / 5.0)) ** 2)
    inits.append(('bell', bell))
    
    r = rng.uniform(0.3, 1.0, size=n)
    inits.append(('random', r))
    
    # Flat top with tapered edges
    taper = n // 10
    flat = np.ones(n)
    flat[:taper] = np.linspace(0.3, 1.0, taper)
    flat[-taper:] = np.linspace(1.0, 0.3, taper)
    inits.append(('flat_top', flat))
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.0
    alt[1::2] = 0.5
    inits.append(('alternating', alt))
    
    return inits

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve(a, b):
    """Fast full linear convolution using FFT."""
    n = len(a) + len(b) - 1
    f = 1
    while f < n:
        f *= 2
    A = np.fft.rfft(a, f)
    B = np.fft.rfft(b, f)
    return np.fft.irfft(A * B, f)[:n]


def compute_score_fast(f):
    """Compute score using FFT-based convolution."""
    n = len(f)
    conv = fft_convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))


def project_simplex(v, S):
    """Project v onto {a >= 0, sum(a) = S}. Uses sorted approach."""
    n = len(v)
    sorted_v = np.sort(v)[::-1]
    cssv = np.cumsum(sorted_v)
    rho = np.nonzero(sorted_v * np.arange(1, n + 1) > (cssv - S))[0][-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_topk(f, top_k=10):
    """Compute subgradient using top-k convolution lags with uniform weights."""
    n = len(f)
    conv = fft_convolve(f, f)
    S = float(np.sum(f))
    
    # Get top-k lags
    top_indices = np.argsort(conv)[-top_k:]
    
    # Weight by excess above median (to focus on the "problem" lags)
    median_conv = np.median(conv)
    weights = np.maximum(conv[top_indices] - median_conv, 0.0)
    w_sum = np.sum(weights)
    if w_sum < 1e-15:
        weights = np.ones(top_k)
    else:
        weights = weights / w_sum
    
    # grad_i = 2 * sum_k w_k * a_{k-i} for valid k-i
    # = 2 * sum_k w_k * f[k-i] where 0 <= k-i < n
    grad = np.zeros(n)
    for k, w in zip(top_indices, weights):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += 2.0 * w * f[k - j]
    
    return grad


def softmax_subgradient(f, temperature=1.0):
    """Compute smooth subgradient using softmax over convolution values."""
    n = len(f)
    conv = fft_convolve(f, f)
    
    conv_shifted = conv - np.max(conv)
    weights = np.exp(temperature * conv_shifted)
    weights /= np.sum(weights)
    
    # grad_i = 2 * sum_j weights[i+j] * f[j]
    # = 2 * conv(w_rev, f)[n-1:2n-1][::-1]
    w_rev = weights[::-1]
    conv_wf = fft_convolve(w_rev, f)
    grad = 2.0 * conv_wf[n - 1:2 * n - 1][::-1]
    
    return grad

def lp_optimize(f, deadline, verbose_prefix="", max_iter=100):
    """LP-based direction finding with line search."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    best_f = f.copy()
    best_score = compute_score_fast(f)
    n = len(f)
    step = 0.08
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iter:
        conv = fft_convolve(f, f)
        M = float(np.max(conv))
        if M <= 1e-15:
            break
        
        # Active constraints: lags where conv >= 0.97 * max
        threshold = M * 0.97
        active = np.where(conv >= threshold)[0]
        
        if len(active) == 0:
            break
        
        # Build constraint matrix
        rows = []
        cols = []
        vals = []
        for k in active:
            j_min = max(0, k - (n - 1))
            j_max = min(k, n - 1)
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(f[k - j])
        
        if len(rows) == 0:
            break
        
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        active_mask = np.zeros(2 * n - 1, dtype=bool)
        active_mask[active] = True
        A = A[active_mask, :]
        
        c = -np.ones(n)
        b_ub = np.full(len(active), M)
        bounds = [(0, None)] * n
        
        try:
            result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                             options={'presolve': True})
            g = result.x if result.success else None
        except Exception:
            g = None
        
        it += 1
        
        if g is not None:
            sum_g = float(np.sum(g))
            if sum_g > 1e-15:
                g = g * (S / sum_g)
                
                best_step = None
                best_ms = best_score
                for alpha in [step, step * 0.5, step * 0.25]:
                    candidate = np.maximum((1 - alpha) * f + alpha * g, 0)
                    cs = float(np.sum(candidate))
                    if cs < 1e-15:
                        continue
                    candidate = candidate * (S / cs)
                    cs_score = compute_score_fast(candidate)
                    if np.isfinite(cs_score) and cs_score < best_ms:
                        best_ms = cs_score
                        best_step = candidate
                
                if best_step is not None:
                    f = best_step
                    best_f = f.copy()
                    best_score = best_ms
                    no_improve = 0
                    step = max(step * 0.97, 0.01)
                    if it % 10 == 0:
                        print(f"    {verbose_prefix}LP iter {it}: score={best_score:.6f}")
                    continue
        
        no_improve += 1
        step *= 0.8
        if step < 0.005 or no_improve > 30:
            break
    
    return best_f, best_score

def make_initializations(n, rng, parent=None):
    """Create diverse initializations for length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    
    inits.append(('uniform', np.ones(n)))
    
    cos_t = (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0
    cos_t = np.maximum(cos_t, 0.05)
    inits.append(('cosine', cos_t))
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * idx / (n - 1)))
    rc = np.maximum(rc, 0.05)
    inits.append(('raised_cos', rc))
    
    # Bell shape
    center = (n - 1) / 2.0
    bell = np.exp(-0.5 * ((idx - center) / (n / 4.0)) ** 2)
    inits.append(('bell', bell))
    
    # Flat top with taper
    taper = max(1, n // 8)
    flat = np.ones(n)
    flat[:taper] = 0.5 + 0.5 * np.sin(np.pi * np.arange(taper) / (2 * taper))
    flat[-taper:] = 0.5 + 0.5 * np.sin(np.pi * np.arange(taper) / (2 * taper))[::-1]
    inits.append(('flat_top', flat))
    
    # Random smooth
    rough = rng.uniform(0.3, 1.0, size=n)
    from scipy.ndimage import gaussian_filter1d
    smooth = gaussian_filter1d(rough, sigma=n / 20.0)
    smooth = np.maximum(smooth, 0.05)
    inits.append(('smooth_rand', smooth))
    
    # Alternating with decay
    alt = np.ones(n)
    alt[::2] = 1.0
    alt[1::2] = 0.6
    decay = np.exp(-np.abs(idx - center) / (n / 3.0))
    inits.append(('alt_decay', alt * decay))
    
    # If parent available, interpolate
    if parent is not None and len(parent) != n:
        x_old = np.linspace(0, 1, len(parent))
        x_new = np.linspace(0, 1, n)
        parent_interp = np.interp(x_new, x_old, parent)
        inits.append(('parent_interp', np.maximum(parent_interp, 0.0)))
    elif parent is not None and len(parent) == n:
        inits.append(('parent', parent.copy()))
    
    return inits
