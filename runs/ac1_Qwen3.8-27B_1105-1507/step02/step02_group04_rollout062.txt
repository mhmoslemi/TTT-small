with adaptive temperature, multi-start initialization from parametric S-curves,
and backtracking line search. Explores sequence lengths 100-600.

import numpy as np
import time

def fft_conv(a):
    """Compute conv(a, a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def true_obj(a):
    """Compute the true objective value."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    conv = fft_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def smooth_obj_and_grad(a, tau):
    """
    Compute the smoothed (log-sum-exp) objective and its gradient w.r.t. a.
    a is assumed to be on the simplex (sum=1, a>=0).
    Returns (obj, grad) where grad is the gradient w.r.t. a (before projection).
    """
    n = len(a)
    conv = fft_conv(a)
    max_c = np.max(conv)
    shifted = tau * (conv - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    log_sum = np.log(np.sum(exp_s))
    obj = (max_c + log_sum / tau)  # This is (1/tau)*log(sum exp(tau*conv))
    
    p = exp_s / np.sum(exp_s)
    
    # Gradient of the smoothed max w.r.t. a:
    # d/da[i] (1/tau)*log(sum_k exp(tau*conv[k]))
    # = sum_k p_k * d conv[k]/da[i]
    # d conv[k]/da[i] = a[k-i] + delta(2i,k)*a[i] for valid indices
    #                  = (correlation term) + (self term)
    
    # Cross term: c[i] = sum_j a[j] * p[i+j] for j=0..n-1
    # = np.correlate(p, a, 'full')[i] for i=0..n-1
    corr = np.correlate(p, a, mode='full')[:n]
    
    # Self term: p[2i] * a[i]
    self_idx = 2 * np.arange(n)
    self_term = p[self_idx] * a
    
    grad = corr + self_term
    
    # Scale by 1/tau (derivative of the log-sum-exp)
    # Actually, (1/tau)*log(sum exp(tau*conv)) has derivative:
    # (1/tau) * sum_k tau*exp(tau*conv[k]) * dconv[k]/da[i] / sum_k exp(tau*conv[k])
    # = sum_k p_k * dconv[k]/da[i]
    # So the gradient is just sum_k p_k * dconv[k]/da[i] = grad (as computed above)
    # No extra 1/tau factor needed!
    
    return float(obj), grad

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    if np.sum(v) < 1e-15:
        return np.ones(len(v)) / len(v)
    v = np.maximum(v, 0)
    s = np.sum(v)
    return v / s

of non-negative sequences. Uses FFT-accelerated gradients, adaptive temperature

"""

import time

def fft_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def true_obj(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    conv = fft_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def smooth_grad(a, tau):
    """Compute gradient of smoothed log-sum-exp of conv(a,a)."""
    n = len(a)
    conv = fft_conv(a)
    max_c = np.max(conv)
    shifted = tau * (conv - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    p = exp_s / np.sum(exp_s)
    # Gradient = 2 * correlate(p, a, 'full')[:n]
    grad = 2.0 * np.correlate(p, a, mode='full')[:n]
    return grad

def project_simplex(v):
    if np.sum(v) < 1e-15:
        return np.ones(len(v)) / len(v)
    v = np.maximum(v, 0)
    s = np.sum(v)
    return v / s

        best_trial_a = a

            stall = 0

            a = project_simplex(a + rng_local.normal(0, 0.15, n))
            v = np.zeros(n)
            step = step0 * 0.5
            stall = 0
    
    return best_a, best_obj

Looking at this problem, I need to minimize `2*n*max(conv(a,a))/(sum(a))^2` for non-negative sequences. Key insights:

1. The objective is scale-invariant, so I can work on the simplex (sum=1, a≥0)
2. The gradient of the smoothed log-sum-exp of conv(a,a) has a clean form: `2 * correlate(p, a, 'full')[:n]` where p are the softmax weights
3. The optimal structure is likely an "S-curve" (high on one side, low on the other)
4. I should use momentum, adaptive tau, and backtracking line search

Let me derive the gradient carefully:
- `conv[k] = Σ_j a[j]·a[k-j]`
- `∂conv[k]/∂a[i] = 2·a[k-i]` (when valid)
- So `∇[smoothed max][i] = 2·Σ_j a[j]·p[i+j]`

This is efficiently computable with `np.correlate`.

"""
Multi-start smoothed subgradient optimization with FFT-accelerated gradients,
Nesterov momentum, adaptive softmax temperature, and backtracking line search.
Explores sequence lengths 80-500 with diverse S-curve and two-half initializations.
Key insight: gradient of smoothed logsumexp(conv(a,a)) = 2*correlate(softmax_weights, a)
"""

import numpy as np
import time


def fft_conv(a):
    """Compute conv(a, a) via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def true_obj(a):
    """Compute the true objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    conv = fft_conv(a)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def smooth_grad(a, tau):
    """Compute gradient of (1/tau)*log(sum_k exp(tau*conv_k)) w.r.t. a.
    
    Result: 2 * np.correlate(p, a, 'full')[:n] where p = softmax(tau*conv).
    """
    n = len(a)
    conv = fft_conv(a)
    max_c = np.max(conv)
    shifted = tau * (conv - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    p = exp_s / np.sum(exp_s)
    # grad[i] = 2 * sum_j a[j] * p[i+j] = 2 * correlate(p, a, 'full')[i]
    grad = 2.0 * np.correlate(p, a, mode='full')[:n]
    return grad


def project_simplex(v):
    """Project onto simplex {x>=0, sum(x)=1} (approximate: clip + normalize)."""
    vn = np.maximum(v, 0.0)
    s = np.sum(vn)
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return vn / s


def optimize_sequence(a_init, max_time=25.0, tau_min=5.0, tau_max=200.0,
                      step0=0.1, momentum=0.9, verbose=False):
    """Optimize on simplex using smoothed gradient + momentum + backtracking line search."""
    n = len(a_init)
    a = project_simplex(np.asarray(a_init, dtype=np.float64))
    v_mom = np.zeros(n)
    best_a = a.copy()
    best_obj = true_obj(a)
    curr_obj = best_obj

    step = step0
    start = time.monotonic()
    stall = 0
    patience = 400
    it = 0

    while time.monotonic() - start < max_time:
        it += 1
        frac = min(1.0, (time.monotonic() - start) / max_time)
        tau = tau_min + (tau_max - tau_min) * (frac ** 0.6)

        grad = smooth_grad(a, tau)
        # Project gradient to tangent space of simplex (remove mean)
        grad_c = grad - np.mean(grad)

        # Momentum
        v_mom = momentum * v_mom + grad_c

        # Backtracking line search (require decrease in true objective)
        found = False
        for bt in range(10):
            trial_step = step * (0.5 ** bt)
            a_trial = project_simplex(a - trial_step * v_mom)
            obj_trial = true_obj(a_trial)
            if obj_trial < curr_obj - 1e-12:
                a = a_trial
                curr_obj = obj_trial
                found = True
                break
        if not found:
            # All backtracks failed; take smallest step anyway
            a = project_simplex(a - step * (0.5 ** 9) * v_mom)
            curr_obj = true_obj(a)

        if curr_obj < best_obj - 1e-9:
            best_obj = curr_obj
            best_a = a.copy()
            stall = 0
            step = min(step * 1.08, 0.8)
            if verbose and it % 200 == 0:
                print(f"        it={it} tau={tau:.0f} step={step:.4f} obj={best_obj:.6f}")
        else:
            stall += 1
            step *= 0.92

        if stall > patience:
            # Escape local minimum via perturbation
            rng_esc = np.random.default_rng(it * 7919 + n * 131)
            noise = rng_esc.normal(0, 0.12, n)
            a = project_simplex(a + noise)
            v_mom = np.zeros(n)
            step = step0 * 0.6
            stall = 0
            curr_obj = true_obj(a)

    return best_a, best_obj


def init_scurve(n, sharpness, center, rng):
    """Sigmoid-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (1.0 + np.exp(-12.0 * sharpness * (center - x)))
    a *= np.exp(rng.normal(0, 0.06, n))
    return project_simplex(a)


def init_twohalf(n, gap, rng):
    """Two-level initialization (high then low)."""
    a = np.ones(n)
    a[n // 2:] = gap
    a *= np.exp(rng.normal(0, 0.04, n))
    return project_simplex(a)


def init_smoothstep(n, t0, t1, rng):
    """Smooth step from 1 to 0 between t0 and t1."""
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    mask = (x >= t0) & (x <= t1)
    t = (x[mask] - t0) / max(t1 - t0, 1e-6)
    a[mask] = 1.0 - t * t * (3.0 - 2.0 * t)  # smoothstep
    tail = x > t1
    a[tail] = 0.1
    a *= np.exp(rng.normal(0, 0.05, n))
    return project_simplex(a)


def init_cos_taper(n, decay, rng):
    """Cosine taper (high in middle, low at edges)."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * (x - 0.5)) ** 2 * decay + (1.0 - decay)
    a *= np.exp(rng.normal(0, 0.08, n))
    return project_simplex(a)


def init_linear_ramp(n, slope, rng):
    """Linear ramp from 1 to a small value."""
    x = np.linspace(0, 1, n)
    a = 1.0 - slope * x
    a = np.maximum(a, 0.05)
    a *= np.exp(rng.normal(0, 0.05, n))
    return project_simplex(a)
