def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        jl = max(0, k-n+1)
        jh = min(n-1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k]*cnt)
            cols.extend(range(jl, jh+1))
            vals.extend((a[k-jh:k-jl+1][::-1]).tolist())
    A = lil_matrix((2*n-1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return csr_matrix(A)

    if r.success and r.x is not None:

def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            seg = a[k - jh:k - jl + 1][::-1]
            vals.extend(seg.tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return csr_matrix(A)


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via cubic interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    from scipy.interpolate import CubicSpline
    cs = CubicSpline(t_old, src)
    result = cs(t_new)
    return np.maximum(result, 0.0)


def logsumexp_smooth_obj(x, T):
    """Smoothed objective: 2n * logsumexp(b/T) / (sum a)^2."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return 1e10
    ax = ax / s
    b = fast_conv(ax)
    # Log-sum-exp for numerical stability
    b_max = b.max()
    if b_max < 1e-30:
        return 1e10
    lse = b_max / T + np.log(np.sum(np.exp((b - b_max) / T)))
    n = len(ax)
    return float(2 * n * lse)


def logsumexp_smooth_grad(x, T):
    """Gradient of the smoothed objective."""
    ax = np.maximum(x, 0.0)
    s = ax.sum()
    if s < 1e-8:
        return np.zeros_like(x)
    ax_n = ax / s
    n = len(ax_n)
    b = fast_conv(ax_n)
    b_max = b.max()
    exp_b = np.exp((b - b_max) / T)
    sum_exp = exp_b.sum()
    # p_k proportional to b_k * exp(b_k/T) (Poynting-like)
    p = (b / T) * exp_b / sum_exp
    # Gradient of logsumexp(b/T) w.r.t. a[j] = sum_k p_k * a[k-j] (if valid)
    # But we also have the 1/(sum a)^2 factor and the normalization...
    # Let's just use finite differences hint via the subgradient structure
    # Actually for L-BFGS-B we don't need the gradient explicitly if we use the objective only
    return None


def raised_cosine_init(n):
    """Raised cosine window."""
    j = np.arange(n)
    return 0.5 + 0.5 * np.cos(2 * np.pi * (j + 0.5) / (n + 1))


def cos_sq_init(n):
    """Cosine squared window."""
    j = np.arange(n)
    x = np.pi * (j + 0.5) / (n + 1)
    return np.cos(x) ** 2


def dpss_like_init(n, W=0.25, iters=50):
    """Approximate DPSS via Slepian iteration (the S transform).
    Uses a simplified version: iterate between time-domain windowing
    and frequency-domain windowing."""
    a = np.ones(n) / np.sqrt(n)
    # Time window (raised cosine)
    j = np.arange(n)
    tw = 0.5 + 0.5 * np.cos(2 * np.pi * (j + 0.5) / (n + 1))
    # Frequency window (raised cosine of width 2W)
    k = np.arange(n)
    fw = 0.5 + 0.5 * np.cos(2 * np.pi * (k + 0.5) / (n + 1))
    
    for _ in range(iters):
        a = a * tw
        # Go to frequency domain
        A = np.fft.rfft(a, 2 * n)
        # Apply frequency window (low-pass)
        mask = np.ones(len(A))
        cutoff = int(W * n)
        if cutoff > 0 and cutoff < len(A):
            mask[:cutoff] = 1.0
            # Smooth transition
            transition = min(4, len(A) - cutoff)
            if transition > 1:
                mask[cutoff:cutoff + transition] = 0.5 + 0.5 * np.cos(
                    np.pi * np.arange(transition) / transition)
            mask[cutoff + transition:] = 0.0
        A_windowed = A * mask
        a = np.fft.irfft(A_windowed, 2 * n)[:n]
        a = np.maximum(a, 0.0)
    
    a = np.maximum(a, 0.001)
    a /= a.sum()
    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    j = np.arange(n)
    t = (j + 0.5) / n

    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))
    inits.append(raised_cosine_init(n))
    inits.append(cos_sq_init(n))
    inits.append(dpss_like_init(n, W=0.2, iters=30))
    inits.append(dpss_like_init(n, W=0.3, iters=30))
    inits.append(dpss_like_init(n, W=0.4, iters=30))

    # Gaussian
    inits.append(np.exp(-2.0 * (t - 0.5) ** 2) + 0.05)
    # Power law
    inits.append((t + 0.05) ** 0.5)
    inits.append(1.0 / (t + 0.1))
    # S-shaped
    inits.append(1.0 / (1.0 + np.exp(-6 * (t - 0.5))))
    # Double hump
    inits.append(np.exp(-8 * (t - 0.3) ** 2) + np.exp(-8 * (t - 0.7) ** 2) + 0.05)
    # Flat top
    inits.append(np.ones(n) * 0.5 + 0.5 * (t < 0.5).astype(float) * (t >= 0.5).astype(float))
    # Sine
    inits.append(0.5 + 0.5 * np.sin(np.pi * t))
    # Random smooth
    inits.append(np.abs(np.sin(np.pi * t * 2 + rng.uniform(0, np.pi))) + 0.1)

    return inits


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def bisection_step(a, g, current_J, max_iter=30):
    """Find optimal step size by bisection: largest t such that J((1-t)a+tg) < J(a)."""
    lo, hi = 0.0, 1.0
    best_t = 0.0
    best_J = current_J

    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        a_trial = (1.0 - mid) * a + mid * g
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            hi = mid
            continue
        a_trial = a_trial / s
        J_trial = compute_J(a_trial)
        if J_trial < best_J - 1e-14:
            best_J = J_trial
            best_t = mid
            lo = mid
        else:
            hi = mid

    if best_t > 0:
        a_new = (1.0 - best_t) * a + best_t * g
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        return a_new, best_J
    return None, current_J


def run_lp_mirror_descent(a, max_iters, deadline, tracker, use_momentum=True):
    """LP mirror descent with momentum and bisection step size."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    # Nesterov momentum
    v = np.zeros_like(a)
    beta = 0.7  # momentum parameter

    stall_count = 0
    best_local_J = current_J

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        if use_momentum:
            v = beta * v + (1.0 - beta) * (g - a)
            direction = a + v
            direction = np.maximum(direction, 0.0)
            if direction.sum() > 1e-10:
                direction = direction / direction.sum()
            else:
                direction = g
        else:
            direction = g

        # Bisection step size
        a_new, J_new = bisection_step(a, direction, current_J, max_iter=25)

        if a_new is not None:
            a = a_new
            if J_new < best_local_J - 1e-12:
                best_local_J = J_new
                stall_count = 0
            else:
                stall_count += 1
            current_J = J_new
        else:
            # Try very small step
            t_small = 0.01
            a_trial = (1.0 - t_small) * a + t_small * direction
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
                stall_count = 0
            else:
                stall_count += 1

        if stall_count > 30:
            # Reset momentum and try with smaller effective step
            v = np.zeros_like(a)
            beta = max(0.3, beta * 0.9)
            stall_count = 0

        if it % 50 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_lbb_polish(a, deadline, tracker, T_values=None):
    """Polish using L-BFGS-B on log-sum-exp smoothed objective."""
    if T_values is None:
        T_values = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2]

    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    bounds_list = [(0.0, None)] * n

    for T in T_values:
        if time.monotonic() > deadline:
            break

        def obj(x):
            return logsumexp_smooth_obj(x, T)

        max_iter = max(50, min(500, int(30.0 / T)))
        max_fun = max(200, min(5000, int(200.0 / T)))

        res = minimize(obj, a.astype(np.float64), method='L-BFGS-B',
                       bounds=bounds_list,
                       options={'maxiter': max_iter, 'max_fun': max_fun, 'ftol': 1e-18})
        if res.fun < 1e9:
            a = np.maximum(res.x, 0.0)
            s = a.sum()
            if s > 1e-10:
                a = a / s
            tracker.update(a)

    return a


def run_multi_start_lp(n, nparent, budget_s, deadline, tracker, parent):
    """Run LP mirror descent from multiple starts for given n."""
    rng = np.random.default_rng(n * 1000 + 42)
    inits = generate_initializations(n, rng, parent)
    time_per_init = budget_s / len(inits)

    best_a = None
    best_J = float('inf')

    for idx, a0 in enumerate(inits):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        a0 = np.maximum(a0.astype(np.float64), 0.001)
        a0 = a0 / a0.sum()

        # More iterations for earlier inits (they're more diverse)
        frac = min(1.0, time_per_init / max(remaining, 1.0))
        max_iters = max(30, int(200 * frac))

        a = run_lp_mirror_descent(a0, max_iters, min(deadline, time.monotonic() + time_per_init),
                                  tracker, use_momentum=True)
        J = compute_J(a)
        if J < best_J:
            best_J = J
            best_a = a.copy()

    return best_a


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    1. LP mirror descent with momentum across multiple n values and inits
    2. L-BFGS-B polishing with decreasing log-sum-exp temperature
    3. Final LP polish
    Returns best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: LP Mirror Descent across lengths ===
    # Focus on n=128 and n=256 where we expect good results
    phases = [
        (128, 0.35),  # 35% of time
        (192, 0.15),  # 15%
        (256, 0.25),  # 25%
        (384, 0.10),  # 10%
    ]

    total_budget = deadline - time.monotonic()

    for n, frac in phases:
        if time.monotonic() > deadline:
            break
        budget = frac * total_budget
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1: LP n={n} budget={budget:.0f}s {'='*50}", flush=True)

        best_n = run_multi_start_lp(n, n, budget, deadline, tracker, parent)
        if best_n is not None:
            tracker.update(best_n)
            # Do a second round on the best
            if time.monotonic() < deadline:
                remaining = deadline - time.monotonic()
                extra_budget = min(remaining * 0.3, 60.0)
                if extra_budget > 10:
                    print(f"  Second round n={n} budget={extra_budget:.0f}s", flush=True)
                    best_n2 = run_multi_start_lp(n, n, extra_budget, deadline, tracker, best_n)
                    if best_n2 is not None:
                        tracker.update(best_n2)

        tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s")

    # === Phase 2: L-BFGS-B polishing ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase2: L-BFGS-B n={n} J={tracker.best_score:.8f} {'='*50}", flush=True)

        # Polish with decreasing temperature
        a_polished = run_lbb_polish(tracker.best, deadline, tracker,
                                    T_values=[10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.8, 0.6, 0.4, 0.3, 0.2])
        tracker.update(a_polished)
        tracker.report(f"L-BFGS-B: {time.monotonic()-t0:.1f}s")

        # Also try polishing with slightly perturbed starting points
        if time.monotonic() < deadline:
            for pert in [0.02, 0.05]:
                if time.monotonic() > deadline:
                    break
                a_pert = tracker.best + rng.normal(0, pert, tracker.best.size)
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = a_pert / a_pert.sum()
                a_p2 = run_lbb_polish(a_pert, deadline, tracker,
                                      T_values=[3.0, 2.0, 1.0, 0.5, 0.3])
                tracker.update(a_p2)

    # === Phase 3: Final LP polish on the best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        if n <= 512:
            remaining = deadline - time.monotonic()
            print(f"\n{'='*50} Phase3: Final LP n={n} budget={remaining:.0f}s {'='*50}", flush=True)
            # Use the current best as parent for more LP iterations
            rng2 = np.random.default_rng(12345)
            # Single focused run
            a_final = run_lp_mirror_descent(tracker.best, 500, deadline, tracker, use_momentum=False)
            tracker.update(a_final)
        tracker.report(f"Final LP: {time.monotonic()-t0:.1f}s")

    # === Phase 4: One more L-BFGS-B pass on the final best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase4: Final polish J={tracker.best_score:.8f} {'='*50}", flush=True)
        a_final2 = run_lbb_polish(tracker.best, deadline, tracker,
                                   T_values=[2.0, 1.0, 0.5, 0.3, 0.2, 0.15, 0.1])
        tracker.update(a_final2)
        tracker.report(f"Final polish: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    # Evaluate final answer
    final_score = float(evaluate_sequence([float(x) for x in tracker.best]))
    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} (eval={final_score:.8f}) n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
