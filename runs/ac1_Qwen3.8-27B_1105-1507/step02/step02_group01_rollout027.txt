def spectral_flatten(a, n_fft=None, gamma=0.5, n_iter=1000):
    """Iterative spectral flattening using gamma correction."""
    n = len(a)
    if n_fft is None:
        n_fft = 2 * n
    
    a_work = a.copy().astype(np.float64)
    s = np.sum(a_work)
    if s < 1e-10:
        return a_work, compute_obj(a_work)
    a_work = a_work / s
    
    best_a = a_work.copy()
    best_obj = compute_obj(a_work)
    
    for it in range(n_iter):
        F = np.fft.fft(a_work, n=n_fft)
        
        # Compute magnitude and phase
        mag = np.abs(F)
        phase = np.angle(F)
        
        # Apply gamma correction to non-DC components
        mag_new = mag.copy()
        dc_val = mag[0]
        # For non-DC, compress the dynamic range
        non_dc_mask = np.arange(n_fft) != 0
        # Also handle the DC-conjugate (for real signals, F[n_fft/2] is also DC-like)
        if n_fft % 2 == 0:
            non_dc_mask[n_fft // 2] = False
        
        if mag[non_dc_mask].max() > 0:
            # Normalize non-DC magnitudes, apply gamma, renormalize
            non_dc_mag = mag[non_dc_mask]
            max_non_dc = non_dc_mag.max()
            if max_non_dc > 1e-10:
                normalized = non_dc_mag / max_non_dc
                compressed = normalized ** gamma
                # Scale back
                mag_new[non_dc_mask] = compressed * max_non_dc
        
        F_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.clip(a_new, 0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a_work = a_new
    
    return best_a, best_obj

def spectral_flatten(a, n_iter=400, gamma=0.35):
    n = len(a)
    a = a.copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a
    a = a / s
    n_fft = 1
    while n_fft < 2 * n:
        n_fft *= 2
    best_a = a.copy()
    best_obj = compute_obj(a)
    for it in range(n_iter):
        F = np.fft.rfft(a, n=n_fft)
        mag = np.abs(F)
        phase = np.angle(F)
        if len(mag) > 1:
            non_dc = mag[1:]
            max_nc = non_dc.max()
            if max_nc > 1e-10:
                mag[1:] = max_nc * (non_dc / max_nc) ** gamma
        F_new = mag * np.exp(1j * phase)
        a_new = np.fft.irfft(F_new, n=n_fft)[:n]
        a_new = np.clip(a_new, 0, None)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
    return best_a


def lp_phase(a, max_time, max_iters=400):
    n = len(a)
    if n > 900:
        return a, compute_obj(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        g = solve_lp_direction(a, M)
        if g is None:
            g = solve_lp_direction(a, M * 0.97)
        if g is None:
            stall += 1
            if stall > 25:
                break
            continue
        stall = 0
        t = 0.4 * (1.0 + np.cos(np.pi * min(1.0, it / max_iters)))
        t = max(t, 0.03)
        found = False
        for tt in [t, t * 0.6, t * 0.3, t * 1.4, t * 0.15]:
            tt = min(tt, 1.0)
            a_new = (1.0 - tt) * a + tt * g
            a_new = np.clip(a_new, 0.0, None)
            s = float(np.sum(a_new))
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_a = a_new.copy()
                best_obj = obj
                a = a_new
                found = True
                break
        if not found:
            stall += 1
            if stall > 35:
                break
    return best_a, best_obj


def subgrad_phase(a, max_time, T_init=None):
    n = len(a)
    a = a.copy()
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    if T_init is None:
        conv = np.convolve(a, a)
        T_init = 0.02 * float(np.max(conv))
    T = T_init
    lr = 0.015
    no_improve = 0
    n_fft = 1
    while n_fft < 3 * n:
        n_fft *= 2
    while time.monotonic() - start < max_time:
        conv = np.convolve(a, a)
        max_c = float(np.max(conv))
        shifted = (conv - max_c) / T
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        p = exp_vals / float(np.sum(exp_vals))
        A_rev = a[::-1]
        A_fft = np.fft.rfft(A_rev, n=n_fft)
        P_fft = np.fft.rfft(p, n=n_fft)
        g_full = np.fft.irfft(A_fft * P_fft, n=n_fft)
        grad = 2.0 * g_full[n - 1:n - 1 + n]
        obj_grad = grad - 2.0 * max_c
        a_new = a - lr * obj_grad
        a_new = np.clip(a_new, 0.0, None)
        s_new = float(np.sum(a_new))
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            no_improve = 0
            T = max(1e-6, T * 0.995)
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.82
                T = max(1e-6, T * 0.85)
            if lr < 1e-10:
                break
    return best_a, best_obj


def block_refine(a, rng, max_time):
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    T_start = 0.12
    T_end = 0.008
    while time.monotonic() - start < max_time:
        progress = min(1.0, (time.monotonic() - start) / max(-1e-10, max_time))
        T = T_start * (T_end / T_start) ** progress
        width = max(1, int(n * rng.uniform(0.02, 0.18)))
        pos = int(rng.integers(0, max(1, n - width + 1)))
        candidate = best_a.copy()
        candidate[pos:pos + width] *= np.exp(rng.normal(0, T, width))
        candidate = np.clip(candidate, 0, None)
        s = float(np.sum(candidate))
        if s < 1e-10:
            continue
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate.copy()
            best_obj = obj
        else:
            delta = obj - best_obj
            if delta < 0.1:
                log_acc = -delta / (0.02 + T * 0.05)
                if np.log(rng.random() + 1e-15) < log_acc:
                    best_a = candidate.copy()
                    best_obj = obj
    return best_a, best_obj


def diverse_init(n, rng, method=None):
    if method is None:
        method = int(rng.integers(0, 9))
    if method == 0:
        a = np.ones(n)
        ew = 1.0 + 2.0 / np.sqrt(n)
        k = max(1, n // 12)
        a[:k] *= ew
        a[-k:] *= ew
        a *= np.exp(rng.normal(0, 0.3, n))
    elif method == 1:
        a = 0.5 * (1 + np.cos(2 * np.pi * (np.arange(n) + 0.5) / n))
        a = np.maximum(a, 0.01)
        a *= np.exp(rng.normal(0, 0.2, n))
    elif method == 2:
        center = (n - 1) / 2.0
        width = n * rng.uniform(0.2, 0.5)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2) + 0.05
    elif method == 3:
        a = 1.0 / (np.arange(n) + 1) ** rng.uniform(0.4, 0.9)
    elif method == 4:
        freq = rng.uniform(0.15, 0.6)
        x = np.pi * freq * np.arange(n)
        a = np.abs(np.sin(x) / (x + 1e-10))
        a += 0.04
    elif method == 5:
        c1 = n * rng.uniform(0.15, 0.35)
        c2 = n * rng.uniform(0.65, 0.85)
        w = n * rng.uniform(0.06, 0.15)
        a = np.exp(-0.5 * ((np.arange(n) - c1) / w) ** 2) + np.exp(-0.5 * ((np.arange(n) - c2) / w)
