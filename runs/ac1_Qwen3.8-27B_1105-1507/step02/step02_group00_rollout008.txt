"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternation with FFT-accelerated convolution, subgradient polishing,
coordinate refinement, and diverse structured initializations across multiple
sequence lengths. Exploits the scale invariance of the objective and uses
adaptive step sizes for fast convergence to near-optimal solutions.

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    if n > 400:
        size = 1
        while size < 2 * n - 1:
            size *= 2
        f = np.fft.rfft(a, size)
        conv = np.fft.irfft(f * f, size)[:2 * n - 1]
    else:
        conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (sum_a * sum_a)


def compute_conv(a):
    """Compute full linear convolution efficiently."""
    n = len(a)
    if n > 400:
        size = 1
        while size < 2 * n - 1:
            size *= 2
        f = np.fft.rfft(a, size)
        conv = np.fft.irfft(f * f, size)[:2 * n - 1]
    else:
        conv = np.convolve(a, a)
    return conv


def build_A(a):
    """Build constraint matrix for LP: A[k,j] = a[k-j] if valid, else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_g(a, M_relax=1.0, timeout=30):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    conv = compute_conv(a)
    M = float(np.max(conv)) * M_relax
    if M < 1e-15:
        return None
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_alternate(a_init, max_iter=100, relax=1.0, deadline=None):
    """LP alternating optimization with adaptive step size."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        g = solve_lp_g(a, M_relax=relax)
        if g is None:
            break
        g_sum = float(np.sum(g))
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step size: larger early, smaller later
        t = 0.8 / (1.0 + 0.05 * it)

        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a = g
        else:
            a = a_new / s

        obj_new = compute_obj(a)
        if obj_new < best_obj:

def subgradient_polish(a, n_steps=1000, deadline=None):
    """Subgradient method with adaptive step size and active set tracking."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    a = a / s
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    # Precompute index ranges for each lag
    ranges = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        ranges.append((j_start, j_end))

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        h = compute_conv(a)
        max_h = float(np.max(h))
        if max_h < 1e-15:
            break

        # Active set: indices where h is close to max
        frac = 0.01 + 0.04 * (step / n_steps)
        tol = max_h * frac
        active = h >= max_h - tol
        active_idx = np.where(active)[0]
        n_active = len(active_idx)
        if n_active == 0:
            active_idx = np.array([np.argmax(h)])
            n_active = 1

        # Compute averaged subgradient
        grad = np.zeros(n)
        for k in active_idx:
            j_start, j_end = ranges[k]
            for j in range(j_start, j_end + 1):
                if 2 * j == k:
                    grad[j] += 2.0 * a[j]
                else:
                    grad[j] += 2.0 * a[k - j]
        grad /= n_active

        # Projected subgradient step
        step_size = 1.0 / (1.0 + 0.02 * step)
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def coord_refine(a, n_rounds=3, deadline=None):
    """Coordinate descent with multi-value search per coordinate."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    multipliers = [0.0, 0.3, 0.5, 0.7, 0.9, 1.0, 1.1, 1.3, 1.6, 2.0, 3.0]

    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-10:
                # Try small positive values
                candidates = [0.001, 0.01, 0.05, 0.1, 0.2, 0.5]
            else:
                candidates = [old_val * m for m in multipliers]

            best_local = best_obj
            best_x = old_val
            for x in candidates:
                x = max(0.0, float(x))
                a[j] = x
                obj = compute_obj(a)
                if obj < best_local - 1e-10:
                    best_local = obj
                    best_x = x

            a[j] = best_x
            if best_local < best_obj:
                best_obj = best_local
                best_a = a.copy()
                improved = True

        if not improved and rnd > 0:
            break

    return best_a, best_obj


def structured_init(n, rng, init_type=None):
    """Generate diverse structured initializations."""
    if init_type is None:
        init_type = int(rng.integers(0, 10))

    if init_type == 0:
        # Uniform
        a = np.ones(n)
    elif init_type == 1:
        # Linear ramp
        a = np.linspace(0.1, 1.0, n)
    elif init_type == 2:
        # Cosine window
        a = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (2 * n)))
        a = np.maximum(a, 0.01)
    elif init_type == 3:
        # Gaussian
        x = np.arange(n) / n - 0.5
        sigma = rng.uniform(0.15, 0.4)
        a = np.exp(-x ** 2 / (2 * sigma ** 2))
    elif init_type == 4:
        # Quadratic phase (chirp-like, made positive)
        x = np.arange(n) / n
        a = np.abs(np.cos(2 * np.pi * rng.uniform(1, 5) * x ** 2)) + 0.05
    elif init_type == 5:
        # Two-lobe asymmetric
        mid = n // 2
        a = np.ones(n)
        a[:mid] = rng.uniform(0.3, 1.5)
        a[mid:] = rng.uniform(0.1, 0.8)
    elif init_type == 6:
        # Pareto-like heavy tail
        a = 1.0 / (np.arange(n) + 1) ** rng.uniform(0.3, 1.0)
    elif init_type == 7:
        # Symmetric random
        half = (n + 1) // 2
        h = rng.random(half) + 0.1
        a = np.zeros(n)
        a[:half] = h
        a[half:] = h[::-1][:n - half]
    elif init_type == 8:
        # Exponential decay from center
        center = n // 2
        a = np.exp(-np.abs(np.arange(n) - center) / rng.uniform(0.1 * n, 0.5 * n))
    else:
        # Random smooth (low-pass filtered random)
        a = rng.random(n)
        # Simple smoothing
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)

    s = float(np.sum(a))
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def lp_alternate_advanced(a_init, max_iter=100, deadline=None):

        idx = min(it // 10, len(tolerances) - 1)
        relax = tolerances[idx]

        g = solve_lp_g(a, M_relax=relax)
        if g is None:
            # Try with looser constraint
            g = solve_lp_g(a, M_relax=1.0)
            if g is None:
                break

        g_sum = float(np.sum(g))
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # EMA update
        alpha = 0.4
        ema = (1.0 - alpha) * ema + alpha * g

        # Check objective of candidates
        for candidate in [g, ema]:
            cs = float(np.sum(candidate))
            if cs > 1e-10:
                cand_norm = candidate / cs
                obj_c = compute_obj(cand_norm)
                if obj_c < best_obj:
                    best_obj = obj_c
                    best_a = cand_norm.copy()

        # Update working point
        t = 0.7 / (1.0 + 0.03 * it)
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            a = g
        else:
            a = a_new / s

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search for optimal non-negative sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 8.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except NameError:
        pass
    except Exception as e:
        print(f"Parent load error: {e}")

    # Strategy: focus on a few well-chosen n values
    # The objective is 2*n*max_conv/sum^2. We want to minimize this.
    # Larger n gives more freedom but the 2*n factor grows.
    # Sweet spot seems to be around n=200-600 based on parent (n=300).
    n_values = [200, 300, 400, 500, 600, 800]

    # If parent was at n=300, prioritize nearby
    if best_seq is not None:
        n_parent = best_seq.size
        n_values = sorted(set([200, 250, n_parent, 350, 400, 500, 600, 800]))

    print(f"Trying n values: {n_values}")
    print(f"Budget: {budget_s}s, deadline in {deadline - start_time:.1f}s")

    # Phase 1: LP optimization with diverse starts
    for n in n_values:
        if time.monotonic() > deadline - 60:
            break
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break

        print(f"\n--- n={n}: {remaining:.0f}s remaining, best so far={best_score:.6f} ---")

        # Determine number of restarts based on time
        n_restarts = 3 if remaining > 300 else 2

        for restart in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break

            # Choose initialization
            if restart == 0 and best_seq is not None and len(best_seq) == n:
                a_init = best_seq.copy()
            else:
                init_type = (restart + n) % 10
                a_init = structured_init(n, rng, init_type=init_type)

            # Run LP alternation
            sub_deadline = min(deadline - 20, time.monotonic() + remaining / n_restarts / 2)
            a, obj = lp_alternate_advanced(a_init, max_iter=200, deadline=sub_deadline)

            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  LP n={n} r={restart}: obj={obj:.6f} (NEW BEST)")

            # Subgradient polish
            if time.monotonic() < deadline - 30:
                ns = int(min(2000, (deadline - time.monotonic()) / 0.01))
                if ns > 200:
                    a2, obj2 = subgradient_polish(a, n_steps=ns, deadline=deadline - 10)
                    if obj2 < best_score:
                        best_score = obj2
                        best_seq = a2.copy()
                        print(f"  SG n={n} r={restart}: obj={obj2:.6f} (NEW BEST)")

    # Phase 2: Intensive refinement of the best solution
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n--- Phase 2: Refining n={n_best}, obj={best_score:.6f}, {remaining:.0f}s left ---")

        # More aggressive LP with tighter tolerance
        if n_best <= 800:
            a3, obj3 = lp_alternate_advanced(best_seq, max_iter=200, deadline=deadline - 30)
            if obj3 < best_score:
                best_score = obj3
                best_seq = a3.copy()
                print(f"  Final LP: obj={obj3:.6f} (NEW BEST)")

        # Extended subgradient
        if time.monotonic() < deadline - 20 and n_best <= 600:
            ns = int(min(5000, (deadline - time.monotonic()) / 0.008))
            if ns > 500:
                a4, obj4 = subgradient_polish(best_seq, n_steps=ns, deadline=deadline - 10)
                if obj4 < best_score:
                    best_score = obj4
                    best_seq = a4.copy()
                    print(f"  Final SG: obj={obj4:.6f} (NEW BEST)")

        # Coordinate refinement
        if time.monotonic() < deadline - 10 and n_best <= 400:
            rounds = int(min(5, max(1, (deadline - time.monotonic()) / 30)))
            a5, obj5 = coord_refine(best_seq, n_rounds=rounds, deadline=deadline - 5)
            if obj5 < best_score:
                best_score = obj5
                best_seq = a5.copy()
                print(f"  Final Coord: obj={obj5:.6f} (NEW BEST)")

        # One more LP pass after coordinate descent
        if time.monotonic() < deadline - 15 and n_best <= 500:
            a6, obj6 = lp_alternate_advanced(best_seq, max_iter=100, deadline=deadline - 5)
            if obj6 < best_score:
                best_score = obj6
                best_seq = a6.copy()
                print(f"  Final LP2: obj={obj6:.6f} (NEW BEST)")

    # Phase 3: Try a few additional n values if time permits
    if time.monotonic() < deadline - 60:
        print("\n--- Phase 3: Additional exploration ---")
        extra_ns = [700, 1000, 1200]
        for n in extra_ns:
            if time.monotonic() > deadline - 30:
                break
            a_init = structured_init(n, rng, init_type=int(rng.integers(0, 10)))
            a, obj = lp_alternate_advanced(a_init, max_iter=80, deadline=deadline - 10)
            if obj < best_score:
                best_score = obj
                best_seq = a.copy()
                print(f"  Extra n={n}: obj={obj:.6f} (NEW BEST)")
            else:
                print(f"  Extra n={n}: obj={obj:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)

    # Ensure output is valid
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)
    best_seq = np.nan_to_num(best_seq, nan=0.0, posinf=1000.0, neginf=0.0)

    # Final evaluation
    final_score = compute_obj(best_seq)
    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: n={len(best_seq)}, score={final_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_seq]
