import numpy as np
import time

def proj_simplex(g):
    """Project g onto the probability simplex {f >= 0, sum(f) = 1}."""
    n = len(g)
    if n == 0:
        return g
    u = np.sort(g)[::-1]
    cv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > cv - 1)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1] + 1
    tau = (cv[rho-1] - 1) / rho
    return np.maximum(g - tau, 0)

def subgradient_optimize(f, deadline, momentum=0.9, lr_schedule='1_sqrt_t', soft_param=50.0):
    """Subgradient descent with momentum and simplex projection."""
    f = f.copy().astype(np.float64)

    S = np.sum(f)
    f = f / S
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    v = np.zeros(n)  # momentum
    lr = 1.0 / np.sqrt(n)  # initial learning rate
    
    for t in range(1, 100000):
        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(f, f)
        max_c = np.max(conv)

        # This is equivalent to the gradient of (1/soft_param) * log(sum(exp(soft_param * conv)))
        shifted = soft_param * (conv - max_c)
        h = np.exp(shifted)  # h[k] > 0, h at max positions = 1
        h = h / h.sum()  # normalize to be a probability distribution over lags
        
        # Gradient: grad[i] = 2 * sum_k h[k] * f[k-i] (for valid k-i)
        # = 2 * correlate(h, f, mode='valid')
        grad = 2.0 * np.correlate(h, f, mode='valid')

        v = momentum * v - lr * grad
        f_new = f + v  # NOTE: in minimization, we go in the direction of -grad
        # Wait, I need to be careful with signs.

        # The subgradient points in the direction of increase.
        # So: f_new = f - lr * grad (steepest descent)
        # With momentum: v = momentum * v - lr * grad; f_new = f + v
        # But v accumulates negative gradient, so f + v moves in the descent direction. ✓

            f = proj_simplex(f)  # ensure f is still on simplex
            # Decay learning rate
            lr = 0.999 * lr
        
        if t % 100 == 0:

def coordinate_descent(f, deadline, n_sweeps=500, verbose=False):
    f = f.copy().astype(np.float64)
    n = len(f)
    S = np.sum(f)
    f = f / S
    conv = np.convolve(f, f)
    
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    
    rng = np.random.default_rng(42)
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            old_val = f[i]
            
            # Find optimal t for f[i]
            # conv_new[k] = conv[k] + 2*(t-old_val)*f[k-i] for k in [i, i+n-1] (clipped)
            # conv_new[2i] += (t-old_val)*(t+old_val) [quadratic correction]
            
            # The objective is: max_k conv_new[k]
            # This is a convex function of t.
            
            # Use bisection on the "derivative":
            # h(t) = max(C, max_{k in affected} (conv[k] + 2*(t-old)*f[k-i]), conv[2i] + t^2 - old^2)
            
            # For efficiency, let's just use a few trial values:
            # The optimal t is roughly where the max is minimized.
            # If the max is at some k* (not 2i), then we want:
            #   conv[k*] + 2*(t-old)*f[k*-i] = current_max (or lower)
            #   => t = old - f[k*-i] / 2 (approximately, if this makes the max decrease)
            
            # Simpler: try a few candidates
            max_c = np.max(conv)
            
            # The "pressure" on coordinate i is:
            # pressure[i] = 2 * sum_k h[k] * f[k-i] (from the gradient)
            # If pressure > 0, we should decrease f[i]
            # If pressure < 0, we should increase f[i] (but f[i] >= 0)
            
            # For the 1D problem, the optimal t minimizes:
            # max(max_k (a_k + b_k * t), quadratic term)
            # where b_k = 2*f[k-i] for affected k
            # The minimum of max of linear functions occurs where two of them cross.
            
            # Practical approach: ternary search on [0, upper]
            # where upper = max(old_val * 3, 2.0/n)
            
            upper = max(old_val * 3.0, 2.0 / n, 1e-6)
            
            def h_1d(t_val):
                """Evaluate max of new convolution if f[i] = t_val."""
                d = t_val - old_val
                if abs(d) < 1e-15:
                    return np.max(conv)
                
                lo = max(0, i)
                hi = min(2*n-2, i + n - 1)
                
                # Max over unaffected positions
                # Affected: [lo, hi]
                # Unaffected: [0, lo-1] and [hi+1, 2n-2]
                unaff_max = 0.0
                if lo > 0:
                    unaff_max = max(unaff_max, np.max(conv[:lo]))
                if hi < 2*n-2:
                    unaff_max = max(unaff_max, np.max(conv[hi+1:]))
                
                # Max over affected positions
                f_idx = np.arange(lo, hi+1) - i
                new_vals = conv[lo:hi+1] + 2.0 * d * f[f_idx]
                
                # Quadratic correction at k=2i
                max_val = np.max(new_vals)
                if 2*i <= 2*n-2:
                    if lo <= 2*i <= hi:
                        # Already included in new_vals, add quadratic correction d^2
                        quad_pos = 2*i - lo
                        new_vals[quad_pos] += d * d
                        max_val = max(max_val, new_vals[quad_pos])
                    else:
                        quad_val = conv[2*i] + t_val*t_val - old_val*old_val
                        max_val = max(max_val, quad_val)
                
                return max(max_val, unaff_max)
            
            # Ternary search
            a, b = 0.0, upper
            for _ in range(30):  # 30 iterations of ternary search gives ~10^-6 precision
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                h1 = h_1d(m1)
                h2 = h_1d(m2)
                if h1 < h2:
                    b = m2
                else:
                    a = m1

                d = t_opt - old_val
                lo = max(0, i)

                improved = True

        S = np.sum(f)
        if S > 1e-10:
            f = f / S
            conv = np.convolve(f, f)  # Recompute for numerical accuracy

            best_f = f.copy()
        
        if verbose and sweep % 50 == 0:
            print(f"  CD sweep {sweep}: score={best_score:.6f}")
    
    return best_f, best_score
