**Approach: Iterative reweighting with adaptive step size**

At each step:
1. Compute conv = f*f.
2. Find the set of "hot" positions J where conv[j] is close to the max.
3. Compute the subgradient: g[i] = (1/|J|) * sum_{j in J} 2*f[j-i] (with bounds checking).
4. Take a multiplicative step: f_new[i] = f[i] * exp(-eta * g[i]).
5. Normalize.
6. If the score improved, keep the step. Otherwise, reduce eta.

This is essentially mirror descent with the negative entropy as the mirror map. It automatically handles the non-negativity constraint.

The advantage over the LP is that it's much faster per iteration (no LP solve), so we can do many more iterations. The disadvantage is that it might converge to a worse local minimum.

**Combined approach:**
1. Use mirror descent for fast initial optimization (thousands of iterations).
2. Switch to LP for careful polishing (hundreds of iterations).
3. Finish with coordinate descent.

Let me also think about the **spectral approach** more carefully.

If we want the convolution to be flat, we want the Fourier transform of f to have flat magnitude (since the Fourier transform of the convolution is the square of the Fourier transform of f). So |F(ω)|^2 should be as flat as possible, meaning |F(ω)| should be as flat as possible.

But f >= 0 and sum(f) = 1 constrain F(0) = 1 and F(ω) = conj(F(-ω)).

The problem of finding a non-negative sequence with flat spectrum is related to the "flat-top" filter design problem. There are iterative algorithms for this:
1. Compute F = FFT(f).
2. Replace |F| with the target magnitude (e.g., 1/sqrt(n_bins) for all non-DC bins, with F(0) = 1).
3. Compute f_new = IFFT(|F_target| * phase(F)).
4. Project f_new onto the non-negative simplex (clip to 0, then normalize).
5. Repeat.

This is the "cumulant action method" or "iterative amplitude and phase retrieval" (IAPR) with a non-negativity constraint. It's what the previous code's `spectral_flat` does.

The issue with this approach is that it can get stuck in local minima where the phase is "wrong." Adding random phase perturbations helps but doesn't guarantee escaping.

def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)

def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s

def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            # A[k, i] = f[k - i] for i in [i_lo, i_hi]
            A[k, i_lo:i_hi+1] = f[k - np.arange(i_lo, i_hi+1)]
    return A

    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    eta = eta0
    stall = 0
    no_improve = 0
    
    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()

        threshold = M * 0.99

            break

                    stall = 0
                else:
                    break

import numpy as np
from scipy.optimize import linprog
import time

def compute_score(f):
    """Compute 2*n*max(conv) / sum^2 for a non-negative sequence."""
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)

def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s

def build_conv_matrix(f):
    """Build matrix A where A @ b = np.convolve(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            A[k, idx] = f[k - idx]
    return A

def spectral_flat(f, n_iter=300, rng=None):
    """Spectral flattening: project onto flat magnitude spectrum + non-negativity."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()
    best_f = f.copy()
    best_score = compute_score(f)
    stall = 0
    
    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)
        n_bins = len(mag)
        energy = np.sum(f ** 2)
        target_mag = np.sqrt(energy * n_fft / n_bins)
        mag_new = np.full(n_bins, target_mag)
        mag_new[0] = 1.0
        if it % 50 == 49:
            phase = phase + rng.normal(0, 0.08, n_bins)
        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()

            best_f = f_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 150:
                break
        f = f_new
    return best_f, best_score

def multiplicative_update(f, n_iter=10000, eta0=0.05, eta_decay=0.999, eta_min=0.001, rng=None):
    """Mirror descent (multiplicative weights) for max-conv minimization."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    eta = eta0
    no_improve = 0
    
    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()
        threshold = M * 0.995
        J = np.where(conv >= threshold)[0]
        
        # Compute subgradient: grad[i] = (2/|J|) * sum_{j in J} f[j-i] (valid indices)
        grad = np.zeros(n)
        for j in J:
            i_lo = max(0, j - n + 1)
            i_hi = min(n - 1, j)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                grad[idx] += 2.0 * f[j - idx]
        grad /= len(J)
        
        if grad.max() <= 0:
            break
        
        f_new = f * np.exp(-eta * grad / grad.max())
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)
        
        score = compute_score(f_new)
        if score < best_score - 1e-13:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 200 == 0:
                eta *= eta_decay
            if no_improve > 3000:
                f_pert = f * np.exp(0.15 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    no_improve = 0
                    eta = eta0
                else:
                    break
        if eta < eta_min:
            break
    
    return best_f, best_score

def lp_refine(f, time_budget, t0=0.5, m_factor=1.0):
    """LP direction search with adaptive step size."""
    n = len(f)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    best_f = f.copy()
    best_score = compute_score(f)
    start = time.monotonic()
    t = t0
    stuck = 0
    
    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max() * m_factor
        
        A = build_conv_matrix(f)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception:
            break
        
        if not res.success or res.x is None:
            break
        
        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        
        g = b / b_sum
        
        improved = False
        for step in [min(1.0, t * 1.5), t, t * 0.5, t * 0.1, t * 0.05]:
            f_new = (1 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue

            new_score = compute_score(f_new)
            if new_score < best_score - 1e-13:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new

                improved = True
                stuck = 0
                break

            t *= 0.6

                break

def coordinate_descent(f, n_sweeps=10, rng=None):
    """Coordinate descent with geometric search."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            orig = f[i]
            if orig < 1e-12:
                continue
            best_val = orig
            best_local = best_score
            for factor in [0.1, 0.3, 0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.0/0.5, 1.0/0.3, 10.0]:
                f[i] = max(0.0, orig * factor)
                s = f.sum()
                if s < 1e-10:
                    f[i] = orig
                    continue
                conv = np.convolve(f, f)
                score = 2.0 * n * conv.max() / (s * s)
                if score < best_local:
                    best_local = score
                    best_val = f[i]
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
        f = np.clip(f, 0, None)
        f = normalize(f)
    
    return f, best_score
