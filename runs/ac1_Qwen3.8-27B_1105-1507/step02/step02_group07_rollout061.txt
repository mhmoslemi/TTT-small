"""

def build_conv_matrix(n):
    """Build sparse matrix A where (A@g)[k] = sum_j f[k-j]*g[j] (full conv structure)."""
    from scipy.sparse import csr_matrix, diags
    # A is (2n-1) x n, banded with bandwidth n
    # A[k,j] = 1 if 0 <= k-j < n, else 0
    # Actually we need A such that (A@f)*g gives convolution contributions
    # conv(f,g)[k] = sum_j f[k-j]*g[j]
    # For fixed f, this is linear in g: conv(f,g) = A(f) @ g
    # A(f)[k,j] = f[k-j] if 0<=k-j<n, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(1.0)  # placeholder, will be filled with f values
    # Actually, let's just build it directly with f values
    return rows, cols


def build_conv_matrix_f(f):
    """Build sparse matrix A(f) where (A(f)@g)[k] = conv(f,g)[k]."""
    from scipy.sparse import csr_matrix
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    if not rows:
        return csr_matrix((2 * n - 1, n))
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A=None):
    """Solve: max sum(g) s.t. g >= 0, conv(f,g)_k <= M for all k. Returns g or None."""
    n = len(f)
    if A is None:
        A = build_conv_matrix_f(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def geometric_init(n, a=1.0):
    """Initialize with geometric decay: f[i] proportional to exp(-a*i/n)."""
    i = np.arange(n, dtype=np.float64)
    f = np.exp(-a * i / n)
    f = f / f.sum()
    return f


def linear_ramp_init(n, alpha=0.5):
    """Initialize with linear ramp: high on left, low on right."""
    i = np.arange(n, dtype=np.float64)
    f = 1.0 + alpha * (1.0 - 2.0 * i / (n - 1))
    f = np.clip(f, 0.01, None)
    f = f / f.sum()
    return f


def tukey_init(n, r=0.3):
    """Tukey window initialization."""
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w / w.sum()


def psf_init(n, alpha=2.0):
    """Super-Gaussian (PSF-like) window."""
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    w = np.exp(-((idx - center) / (n / 2.0)) ** (2 * alpha))
    return w / w.sum()


def random_smooth_init(n, seed=42):
    """Random smooth initialization."""
    rng = np.random.default_rng(seed)
    x = rng.standard_normal(n)
    kernel_size = max(5, n // 20)
    kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (kernel_size / 3.0)) ** 2)
    smooth_x = np.convolve(x, kernel, mode='same')
    smooth_x = np.abs(smooth_x) + 0.1
    return smooth_x / smooth_x.sum()


def lp_optimize(f, deadline, step_init=0.15, verbose_prefix="", max_no_improve=60):
    """
    Main LP-based optimization loop.
    f is normalized to sum=1. Minimize 2n * max(conv(f,f)) / sum(f)^2.
    """
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    f = f / S
    n = len(f)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    step = step_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and no_improve < max_no_improve:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break

        A = build_conv_matrix_f(f)
        g = solve_lp_direction(f, M, A)
        it += 1

        if g is None or np.sum(g) < 1e-15:
            no_improve += 1
            continue

        g = g / np.sum(g)  # normalize to sum=1

        # Line search: try multiple step sizes
        candidates = [step, step * 0.5, step * 1.5, step * 0.7, step * 2.0,
                      0.3, 0.5, 0.7]
        candidates = [t for t in candidates if 0 < t <= 1.0]
        # Remove duplicates approximately
        unique_candidates = []
        for t in candidates:
            if not any(abs(t - u) < 0.02 for u in unique_candidates):
                unique_candidates.append(t)

        best_s = best_score
        best_t = 0
        for t in unique_candidates:
            new_f = (1 - t) * f + t * g
            new_f = np.clip(new_f, 0.0, None)
            ns = float(np.sum(new_f))
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = t

        if best_t > 0 and best_s < best_score:
            f = (1 - best_t) * f + best_t * g
            f = np.clip(f, 0.0, None)
            f = f / np.sum(f)
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            if it % 100 == 0:
                print(f"  {verbose_prefix}iter {it}: score={best_score:.6f} (t={best_t:.3f})")
        else:
            no_improve += 1
            step *= 0.7
            if step < 0.01:
                step = 0.05
        # Occasionally increase step back
        if it % 200 == 0 and no_improve < 10:
            step = max(step, 0.08)

    return best_f, best_score


def softmax_polish(f, deadline, beta_start=50.0, beta_end=500.0, lr_start=0.02,
                   max_iters=20000, verbose_prefix=""):
    """Soft-max gradient polish with Adam."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    f = f / S
    n = len(f)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = lr_start
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters and no_improve < 300:
        it_frac = min(1.0, it / (max_iters * 0.8))
        beta = beta_start * (beta_end / beta_start) ** it_frac

        # Compute soft-max gradient
        C = np.convolve(f, f)
        max_C = C.max()
        exp_C = np.exp(beta * (C - max_C))
        total = exp_C.sum()
        if total < 1e-300:
            break
        p = exp_C / total

        # grad[j] = 2 * sum_i f[i] * p[j+i] (valid correlation)
        grad = 2.0 * np.correlate(p, f, mode='valid')

        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        update = m_hat / (np.sqrt(v_hat) + eps)

        new_f = f - lr * update
        new_f = np.clip(new_f, 0.0, None)
        ns = float(np.sum(new_f))
        if ns < 1e-10:
            new_f = f
            ns = 1.0
        new_f = new_f / ns

        sc = evaluate_sequence(new_f.tolist())
        if np.isfinite(sc) and sc < best_score:
            f = new_f
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            if it % 500 == 0:
                print(f"  {verbose_prefix}SM iter {it}: score={best_score:.6f} (beta={beta:.0f})")
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.5
                if lr < 1e-7:
                    break

    return best_f, best_score


def perturb(f, rng, strength=0.05):
    """Multiplicative perturbation."""
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.05, 3.0)
    new_f = np.clip(new_f, 0.0, None)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S


def hot_cold_restart(f_best, deadline, rng, verbose_prefix=""):
    """
    Hot-cold restart: heavily perturb the best solution, then optimize.
    This helps escape local minima.
    """
    n = len(f_best)
    # Create a heavily perturbed version
    strength = 0.3 + 0.2 * rng.random()
    f_pert = perturb(f_best, rng, strength)
    # Also try mixing with a different shape
    if rng.random() < 0.5:
        a = 0.5 + rng.random() * 1.5
        f_geom = geometric_init(n, a)
        mix = 0.3 + 0.4 * rng.random()
        f_pert = (1 - mix) * f_pert + mix * f_geom
        f_pert = np.clip(f_pert, 0, None)
        f_pert = f_pert / f_pert.sum()

    return lp_optimize(f_pert, deadline, step_init=0.1, verbose_prefix=verbose_prefix)

                    best_sequence = f_init.copy()

        print(f"  Best init: {init_scores[0][1]} with score {init_scores[0][0]:.6f}")

        # Optimize the top initializations
        # Allocate time: top 1 gets 50%, top 2 gets 25%, top 3 gets 15%, rest 10%
        n_opt = min(5, len(init_scores))
        time_fracs = [0.40, 0.25, 0.15, 0.10, 0.10][:n_opt]

        for idx in range(n_opt):
            if time.monotonic() > n_deadline - 10:
                break

            sc_init, label, f_init = init_scores[idx]
            # Skip if this init is already much worse than best
            if sc_init > best_score * 1.1 and idx > 0:
                continue

            opt_time = n_budget * time_fracs[idx]
            opt_deadline = time.monotonic() + opt_time
            print(f"  Optimizing [{label}] (init_score={sc_init:.6f}, time={opt_time:.0f}s)")

            try:
                bf, bs = lp_optimize(f_init, opt_deadline, step_init=0.15,
                                     verbose_prefix=f"[n={n} {label}]")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"    Improved to {bs:.6f}")
            except Exception as e:
                print(f"    Error: {e}")

    # Hot-cold restarts
    if best_sequence is not None and time.monotonic() < deadline - 40:
        print(f"\n{'='*50}")
        print(f"Hot-cold restarts (current best: {best_score:.6f})")
        print(f"{'='*50}")
        n_best = len(best_sequence)
        restart_budget = max(0, deadline - time.monotonic() - 50)
        n_restarts = min(8, max(2, int(restart_budget / 40)))
        per_restart = restart_budget / max(n_restarts, 1)

        for r in range(n_restarts):
            if time.monotonic() > deadline - 50:
                break

            r_deadline = time.monotonic() + per_restart
            print(f"  Restart {r+1}/{n_restarts}...", end=" ", flush=True)
            try:
                bf, bs = hot_cold_restart(best_sequence, r_deadline, rng,
                                          verbose_prefix=f"[restart{r}]")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f" -> {bs:.6f} (NEW BEST)")
                else:
                    print(f" -> {bs:.6f}")
            except Exception as e:
                print(f" error: {e}")

    # Final polish with soft-max gradient
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print(f"\n{'='*50}")
        print(f"Final polish (score={best_score:.6f})")
        print(f"{'='*50}")
        polish_deadline = deadline - 3
        # First a bit more LP
        lp_time = min(30, (polish_deadline - time.monotonic()) * 0.3)
        if lp_time > 5:
            try:
                bf, bs = lp_optimize(best_sequence, time.monotonic() + lp_time,
                                     step_init=0.05, max_no_improve=30,
                                     verbose_prefix="[polish_LP]")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
            except Exception:
                pass

        # Then gradient polish
        if time.monotonic() < polish_deadline - 2:
            try:
                bf, bs = softmax_polish(best_sequence, polish_deadline,
                                        beta_start=100.0, beta_end=1000.0,
                                        lr_start=0.01, max_iters=10000,
                                        verbose_prefix="[polish_SM]")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
            except Exception:
                pass

        print(f"  Final polish score: {best_score:.6f}")

    # Final LP refinement
    if best_sequence is not None and time.monotonic() < deadline - 2:
        try:
            bf, bs = lp_optimize(best_sequence, deadline - 1, step_init=0.03,
                                 max_no_improve=20, verbose_prefix="[final]")
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()
        except Exception:
            pass

    # Ensure valid output
    if best_sequence is None:
        best_sequence = tukey_init(256, 0.3)
        best_score = evaluate_sequence(best_sequence.tolist())

    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256) / 256.0
        best_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    final_score = evaluate_sequence([float(x) for x in best_sequence])
    print(f"\n{'='*50}")
    print(f"FINAL: n={len(best_sequence)}, score={final_score:.6f}, time={elapsed:.1f}s")
    print(f"{'='*50}")
    return [float(x) for x in best_sequence]
