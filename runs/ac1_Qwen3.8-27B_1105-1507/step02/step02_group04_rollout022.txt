import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj_normalized(a):
    """Objective assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    conv = compute_conv(a)
    return 2.0 * n * float(np.max(conv))

def smooth_grad(conv, a, tau):
    """Smoothed gradient of max(conv) via log-sum-exp weighting."""
    n = len(a)
    max_val = float(np.max(conv))
    shifted = (conv - max_val) / tau
    shifted = np.clip(shifted, -40, 0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        idx = int(np.argmax(conv))
        w = np.zeros(len(conv))
        w[idx] = 1.0
    else:
        w /= w_sum
    c = np.correlate(w, a, mode='full')
    return 2.0 * c[:n]

The standard FISTA algorithm:
```

for k = 0, 1, 2, ...:

```

Where `proj` is the projection onto the constraint set (a >= 0, sum(a) = 1).

The projection onto {a : a >= 0, sum(a) = 1} is not simply `max(a, 0)` followed by normalization, because that doesn't exactly project. But as an approximation, it works well.

Actually, the exact projection onto the simplex {a >= 0, sum(a) = 1} is:
`proj(a)_i = max(a_i - theta, 0)` for some `theta` chosen so that `sum_i max(a_i - theta, 0) = 1`.

This can be computed by sorting `a` in decreasing order and finding the right `theta`. Let me implement this properly.

    result = np.maximum(v - theta, 0)
    # Due to numerical issues, renormalize
    s = np.sum(result)
    if s > 0:

    return result

                a = a_test.copy()
                improved = True
                if verbose:
                    print(f"      LP blend={bl:.2f}: obj={obj_test/(2*n):.6f}")
                break

        if not improved:
            stall += 1
            if stall > 15:
                break
        else:
            stall = 0
            # Increase blend slightly
            blend = min(0.7, blend * 1.05)

    return best_a, best_obj


def combined_optimize(a, max_time, verbose=False):
    """Combined gradient + LP optimization for a single sequence."""
    n = len(a)
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_obj = compute_obj_normalized(a)

    start = time.monotonic()
    use_lp = n <= 400

    while time.monotonic() - start < max_time:
        remaining = max_time - (time.monotonic() - start)
        if remaining < 3:
            break

        # Phase 1: Nesterov gradient
        grad_time = min(remaining * 0.6, 40.0) if not use_lp else min(remaining * 0.4, 25.0)
        grad_time = max(2.0, grad_time)
        a_g, obj_g = nesterov_optimize(a, max_time=grad_time)
        if obj_g < best_obj:
            best_obj = obj_g
            best_a = a_g.copy()
            a = a_g.copy()
            if verbose:
                print(f"      Grad: score={best_obj/(2*n):.6f}")

        # Phase 2: LP refinement
        if use_lp and (time.monotonic() - start) < max_time - 5:
            lp_time = min(10.0, (max_time - (time.monotonic() - start)) * 0.3)
            if lp_time > 2:
                a_lp, obj_lp = lp_optimize(a, max_time=lp_time)
                if obj_lp < best_obj:
                    best_obj = obj_lp
                    best_a = a_lp.copy()
                    a = a_lp.copy()
                    if verbose:
                        print(f"      LP: score={best_obj/(2*n):.6f}")

        # Phase 3: If stuck, try perturbation
        if (time.monotonic() - start) < max_time - 3:
            rng_p = np.random.default_rng(int(time.monotonic() * 1000))
            a_pert = a * (1.0 + rng_p.normal(0, 0.08, n))
            a_pert = project_simplex(a_pert)
            obj_pert = compute_obj_normalized(a_pert)
            if obj_pert < best_obj:
                best_obj = obj_pert
                best_a = a_pert.copy()
                a = a_pert.copy()

    return best_a, best_obj


def init_spectral(n, rng, method='flat'):
    """Generate diverse initializations."""
    if method == 'flat_spectrum':
        size = 1
        while size < 2 * n:
            size <<= 1
        phase = rng.uniform(0, 2 * np.pi, size // 2 + 1)
        mag = np.ones(size // 2 + 1)
        # Add some randomness to magnitude
        mag = mag * (0.5 + 0.5 * rng.random(size // 2 + 1))
        Z = mag * np.exp(1j * phase)
        Z[0] = abs(Z[0])
        if size % 2 == 0:
            Z[size // 2] = abs(Z[size // 2])
        Z_full = np.zeros(size, dtype=complex)
        Z_full[:size // 2 + 1] = Z
        for i in range(1, size // 2 + 1):
            Z_full[size - i] = np.conj(Z[i])
        a_complex = np.fft.ifft(Z_full)[:n]
        a = np.real(a_complex)
    elif method == 'gaussian_bump':
        center = rng.uniform(0.1, 0.9) * n
        width = rng.uniform(0.05, 0.35) * n
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        # Add some structure
        a *= (0.5 + 0.5 * np.cos(2 * np.pi * rng.uniform(1, 5) * np.arange(n) / n))
    elif method == 'cosine_mix':
        a = np.zeros(n)
        for _ in range(rng.integers(2, 5)):
            freq = rng.uniform(0.5, 8.0)
            phase = rng.uniform(0, 2 * np.pi)
            a += 0.5 + 0.5 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    elif method == 'bursty':
        # Concentrate in a random sub-interval
        start_idx = rng.integers(0, n // 2)
        length = rng.integers(n // 4, n // 2 + 1)
        length = min(length, n - start_idx)
        a = np.zeros(n)
        a[start_idx:start_idx + length] = rng.random(length)
        # Smooth
        if n > 2:
            kernel = np.ones(3) / 3.0
            a = np.convolve(a, kernel, mode='same')
    elif method == 'symmetric':
        half = n // 2
        h = rng.random(half)
        h = h * (0.5 + 0.5 * np.cos(np.pi * np.arange(half) / half))
        if n % 2 == 0:
            a = np.concatenate([h, h[::-1]])
        else:
            a = np.concatenate([h, [rng.random()], h[::-1]])
    else:
        a = rng.random(n)

    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def try_parent():
    """Try to load the parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            s = np.sum(parent)
            if s > 0.01:
                obj = 2.0 * len(parent) * float(np.max(compute_conv(parent))) / (s * s)
                return parent.copy(), obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: multi-n exploration with Nesterov gradient + LP + diverse inits.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    parent, parent_obj = try_parent()
    if parent is not None:
        best_seq = parent.copy()
        best_score = parent_obj
        print(f"Parent: n={len(parent)}, score={best_score:.6f}")

    # Sequence lengths to explore
    n_values = [30, 40, 50, 60, 80, 100, 120, 150, 200, 250, 300, 400, 500, 700, 1000]

    if parent is not None:
        n_p = len(parent)
        n_values = sorted(set(n_values + [n_p, max(20, n_p - 30), min(1500, n_p + 50),
                                          max(20, n_p - 60), min(2000, n_p + 100)]))

    print(f"N values: {n_values}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    methods = ['flat_spectrum', 'gaussian_bump', 'cosine_mix', 'bursty', 'symmetric']

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 20:
            break

        # Time allocation: more time for mid-range n where improvement is most likely
        if n <= 60:
            alloc = min(25.0, remaining * 0.08)
        elif n <= 150:
            alloc = min(50.0, remaining * 0.12)
        elif n <= 300:
            alloc = min(80.0, remaining * 0.15)
        elif n <= 500:
            alloc = min(120.0, remaining * 0.2)
        else:
            alloc = min(180.0, remaining * 0.25)
        alloc = max(12.0, alloc)

        print(f"\n=== n={n}: {remaining:.0f}s rem, alloc={alloc:.0f}s ===")

        # Initialize from parent if available
        if parent is not None:
            a_parent = parent / np.sum(parent)
            if abs(n - len(a_parent)) <= len(a_parent) * 0.3:
                new_x = np.linspace(0, 1, n)
                old_x = np.linspace(0, 1, len(a_parent))
                a_from_parent = np.interp(new_x, old_x, a_parent)
                a_from_parent = project_simplex(a_from_parent)
                obj_p = compute_obj_normalized(a_from_parent)
                if n <= 400:
                    t_sub = alloc * 0.4
                    a_opt, obj = combined_optimize(a_from_parent, max_time=t_sub, verbose=True)
                    score = compute_obj_normalized(a_opt)
                    if score < best_score:
                        best_score = score
                        best_seq = a_opt.copy()
                        print(f"  From parent: score={best_score:.6f}")
                alloc -= t_sub

        # Try diverse initializations
        n_inits = max(2, int(max(5, alloc) / 18))
        for init_idx in range(n_inits):
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            method = methods[init_idx % len(methods)]
            a0 = init_spectral(n, rng, method)
            init_score = compute_obj_normalized(a0)

            sub_alloc = min(remaining * 0.7, alloc - (time.monotonic() - (deadline - alloc)))
            sub_alloc = max(5.0, sub_alloc / (n_inits - init_idx))

            a_opt, obj = combined_optimize(a0, max_time=sub_alloc)
            score = compute_obj_normalized(a_opt)

            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  n={n}, init={init_idx} ({method}): score={best_score:.6f}")

        # Validate periodically
        if best_seq is not None:
            val = evaluate_sequence([float(x) for x in best_seq])
            if np.isfinite(val) and val < best_score:
                best_score = val

    # Final intensive polishing
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== Final polish: n={n_best}, {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy()
        polish_round = 0
        while time.monotonic() < deadline - 5 and polish_round < 15:
            polish_round += 1
            remaining = deadline - 5 - time.monotonic()
            if remaining < 5:
                break
            sub_time = min(30.0, remaining * 0.6)

            # Alternate between gradient and LP
            if polish_round % 2 == 1 and n_best <= 400:
                a_cur, _ = lp_optimize(a_cur, max_time=sub_time)
            else:
                a_cur, _ = nesterov_optimize(a_cur, max_time=sub_time)

            cur_score = compute_obj_normalized(a_cur)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_cur.copy()
                print(f"  Polish {polish_round}: score={best_score:.6f}")

            # Small random perturbation for diversity
            if remaining > 10:
                a_pert = a_cur * (1.0 + rng.normal(0, 0.03, n_best))
                a_pert = project_simplex(a_pert)
                p_score = compute_obj_normalized(a_pert)
                if p_score < best_score:
                    best_score = p_score
                    best_seq = a_pert.copy()
                    a_cur = a_pert.copy()

    # Fallback
    if best_seq is None:
        best_seq = np.ones(50) / 50

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_sequence(result)
    print(f"\nFINAL: n={len(result)}, score={final_score:.6f}, time={time.monotonic()-start_time:.1f}s")

    return result
