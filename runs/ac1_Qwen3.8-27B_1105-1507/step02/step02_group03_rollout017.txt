    while n_fft < need:
        n_fft *= 2
    
    F_f = np.fft.rfft(f, n_fft)
    F_h = np.fft.rfft(active_mask.astype(np.float64), n_fft)
    
    # Cross-correlation: sum_k f[k] * h[k-j] for each j
    # = ifft(fft(f) * conj(fft(h)))
    corr = np.fft.irfft(F_f * np.conj(F_h), n_fft)
    
    # corr[j] = sum_k f[k] * h[(k-j) mod n_fft]
    # For linear correlation, we need corr[j] for j = 0, ..., n-1
    # where the valid range ensures no wrap-around
    # Since n_fft >= 2n-1, there's no wrap-around for j in [0, n-1]
    g = (2.0 / n_active) * corr[:n]
    
    return g, M, conv, active_mask


def mirror_descent(f, n_iter=2000, eta0=0.5, tol_ratio=0.0, decay=0.999, rng=None):
    """Nesterov-accelerated multiplicative weight updates."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    f = f.astype(np.float64).copy()
    f = normalize(f)
    n = len(f)
    
    best_f = f.copy()
    best_score = score(f)
    
    f_prev = f.copy()
    t = 1.0
    eta = eta0
    stall = 0
    
    for it in range(n_iter):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        t = t_new
        
        # Nesterov extrapolation
        y = f + beta * (f - f_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = f.copy()
            sy = 1.0
        y = y / sy
        
        # Subgradient
        g, M, conv, active_mask = compute_subgradient(y, tol_ratio)
        
        # Multiplicative update
        f_new = y * np.exp(-eta * g)
        f_new = np.maximum(f_new, 1e-30)
        f_new = normalize(f_new)
        
        sc = score(f_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_f = f_new.copy()
            stall = 0
        else:
            stall += 1
        
        f_prev = f
        f = f_new
        eta *= decay
        
        if stall > 300:
            # Perturb to escape
            f_pert = f * (1.0 + 0.1 * rng.standard_normal(n))
            f_pert = np.maximum(f_pert, 0.0)
            f_pert = normalize(f_pert)
            sc_pert = score(f_pert)
            if sc_pert < best_score:
                f = f_pert
                f_prev = f_pert.copy()
                best_f = f.copy()
                best_score = sc_pert
                stall = 0
                eta = eta0 * 0.5
            else:
                break
    
    return best_f, best_score


def lp_refine(f, time_budget=30.0, rng=None):
    """LP-based direction finding for refinement."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(f)
    f = f.astype(np.float64).copy()
    f = normalize(f)
    
    best_f = f.copy()
    best_score = score(f)
    t = 0.5
    start = time.monotonic()
    
    # Build convolution matrix: A @ b = convolve(f, b)
    # A has shape (2n-1, n)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = f[k - j]
    
    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        
        if M < 1e-15:
            break
        
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            remaining = time_budget - (time.monotonic() - start)
            lp_time = max(1.0, min(5.0, remaining * 0.3))
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': lp_time})
        except Exception:
            break
        
        if not res.success or res.x is None:
            break
        
        b = np.maximum(res.x, 0.0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        
        g = b / b_sum  # normalized direction
        
        # Try multiple step sizes
        improved = False
        for step in [min(1.0, t * 1.3), t, t * 0.6, t * 0.3]:
            f_new = (1.0 - step) * f + step * g
            f_new = np.maximum(f_new, 0.0)
            s_new = f_new.sum()
            if s_new < 1e-10:
                continue
            f_new = f_new / s_new
            
            new_score = score(f_new)
            if new_score < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                t = min(1.0, t * 1.1)
                improved = True
                break
        
        if not improved:
            t *= 0.5
            if t < 0.01:
                # Perturb
                f_pert = f + 0.05 * rng.standard_normal(n)
                f_pert = np.maximum(f_pert, 0.0)
                f_pert = normalize(f_pert)
                if score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = score(f)
                    t = 0.3
                else:
                    break
        
        # Update A for next iteration (since f changed)
        if improved:
            A = np.zeros((2 * n - 1, n))
            for k in range(2 * n - 1):
                j_start = max(0, k - n + 1)
                j_end = min(n, k + 1)
                for j in range(j_start, j_end):
                    A[k, j] = f[k - j]
    
    return best_f, best_score
