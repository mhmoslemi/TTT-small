                continue

        weights = get_weights(conv, tau)
        grad = compute_grad_fft(a, weights)
        grad_centered = grad - grad.mean()

        log_update = -eta * grad_centered
        log_update = np.clip(log_update, -40, 40)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a = a_new / s_new

        if it > 100:
            eta *= 0.998

    return best_a, best_obj


def spectral_init(n, rng):
    """Initialize with approximately flat power spectrum (random phases)."""
    n_freqs = n // 2 + 1
    phases = rng.uniform(0, 2 * np.pi, n_freqs)
    A = np.exp(1j * phases)
    A[0] = np.abs(A[0])
    if n % 2 == 0:
        A[-1] = np.abs(A[-1])
    a = np.fft.irfft(A * np.sqrt(n), n)
    a = np.abs(a) + rng.uniform(0, 0.1, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def gaussian_init(n, rng):
    """Gaussian-shaped initialization."""
    center = n / 2.0
    width = rng.uniform(n / 8, n / 3)
    x = np.arange(n)
    a = np.exp(-(x - center) ** 2 / (2 * width ** 2))
    a += rng.uniform(0, 0.05, n)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def cosine_init(n, rng):
    """Cosine window initialization."""
    freq = rng.uniform(0.5, 3.0)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * freq * x / n + phase))
    a += rng.uniform(0, 0.02, n)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def uniform_init(n, rng):
    """Uniform with small noise."""
    a = np.ones(n) + rng.normal(0, 0.1, n)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def alternating_init(n, rng):
    """Alternating high/low pattern with noise."""
    a = np.where(np.arange(n) % 2 == 0, 1.0, 0.3)
    a += rng.normal(0, 0.1, n)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def interpolate_seq(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a)


def optimize_with_restarts(a, max_time, verbose=False, label=""):
    """Run mirror descent with periodic perturbation restarts."""
    n = len(a)
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)

    round_num = 0
    while time.monotonic() - start < max_time:
        round_num += 1
        remaining = max_time - (time.monotonic() - start)
        if remaining < 3:
            break

        frac = (time.monotonic() - start) / max_time
        pert_scale = 0.08 * (1 - frac) + 0.01

        rng_local = np.random.default_rng(abs(hash(label + str(round_num))) % (2**31))
        a_cur = best_a * np.exp(rng_local.normal(0, pert_scale, n))
        a_cur = np.maximum(a_cur, 0)
        s = a_cur.sum()
        if s < 1e-10:
            a_cur = best_a.copy()
        else:
            a_cur /= s

        iters = int(min(6000, max(400, remaining * 1.5)))
        tau_s = 3.0 + 7.0 * frac
        tau_e = 25.0 + 75.0 * frac

        a_opt, _ = mirror_descent(
            a_cur, max_iter=iters,
            tau_start=tau_s, tau_end=tau_e,
            eta_init=0.3,
            patience=180,
            pert_scale=0.02 * (1 - frac) + 0.005,
            label=label + str(round_num)
        )

        obj = compute_obj(a_opt)
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
            if verbose:
                print(f"    {label} r{round_num}: obj={best_obj:.6f}")

    return best_a, best_obj


def try_parent():
    """Try to load and evaluate the parent sequence."""
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            return parent, obj
    except Exception:
        pass
    return None, np.inf


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search function combining mirror descent with FFT gradients,
    spectral initialization, multi-length exploration, and annealing.
    Returns the best sequence found within the time budget.
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Phase 0: Evaluate parent
    parent, parent_obj = try_parent()
    if parent is not None and parent_obj < best_score:
        best_score = parent_obj
        best_seq = parent.copy()
        print(f"Initial from parent: {best_score:.6f}")

    # Phase 1: Optimize mid-range lengths (~280s)
    print("\n=== Phase 1: Mid-range optimization ===")
    p1_deadline = start_time + 280.0
    n_vals_p1 = [100, 150, 200, 250, 300, 350, 400, 500]
    if parent is not None:
        n_p = len(parent)
        n_vals_p1 = sorted(set(n_vals_p1 + [n_p, max(50, n_p - 50), min(600, n_p + 50)]))

    for n in n_vals_p1:
        if time.monotonic() > p1_deadline:
            break
        remaining = p1_deadline - time.monotonic()
        if remaining < 12:
            break

        alloc = min(50.0, remaining * 0.15)
        alloc = max(12.0, alloc)

        inits = [spectral_init(n, rng), gaussian_init(n, rng), cosine_init(n, rng)]
        if parent is not None:
            inits.append(interpolate_seq(parent, n))

        best_n = None
        best_n_obj = np.inf

        for ii, a_init in enumerate(inits):
            if time.monotonic() > p1_deadline:
                break
            sub_rem = p1_deadline - time.monotonic()
            if sub_rem < 4:
                break

            a_opt, _ = optimize_with_restarts(
                a_init, max_time=max(4.0, sub_rem * 0.25),
                label=f"P1-{n}-{ii}"
            )
            obj = compute_obj(a_opt)
            if obj < best_n_obj:
                best_n_obj = obj
                best_n = a_opt.copy()

        if best_n is not None and best_n_obj < best_score:
            best_score = best_n_obj
            best_seq = best_n.copy()
            print(f"  Phase1 n={n}: {best_score:.6f} ***")
        elif best_n is not None:
            print(f"  Phase1 n={n}: {best_n_obj:.6f}")

    # Phase 2: Explore longer sequences (~550s)
    print(f"\n=== Phase 2: Long sequences (best={best_score:.6f}) ===")
    p2_deadline = start_time + 850.0
    n_vals_p2 = [400, 500, 600, 750, 1000, 1200, 1500]

    for n in n_vals_p2:
        if time.monotonic() > p2_deadline:
            break
        remaining = p2_deadline - time.monotonic()
        if remaining < 20:
            break

        alloc = min(120.0, remaining * 0.18)
        alloc = max(20.0, alloc)
        print(f"  n={n}, alloc={alloc:.0f}s, remaining={remaining:.0f}s")

        inits = [spectral_init(n, rng), gaussian_init(n, rng), alternating_init(n, rng)]
        if parent is not None:
            inits.append(interpolate_seq(parent, n))

        best_n = None
        best_n_obj = np.inf

        for ii, a_init in enumerate(inits):
            if time.monotonic() > p2_deadline:
                break
            sub_rem = p2_deadline - time.monotonic()
            if sub_rem < 5:
                break

            a_opt, _ = optimize_with_restarts(
                a_init, max_time=max(5.0, sub_rem * 0.3),
                label=f"P2-{n}-{ii}"
            )
            obj = compute_obj(a_opt)
            if obj < best_n_obj:
                best_n_obj = obj
                best_n = a_opt.copy()
                if obj < best_score:
                    best_score = obj
                    best_seq = a_opt.copy()
                    print(f"    NEW BEST n={n}: {best_score:.6f}")

        if best_n is not None:
            print(f"  Phase2 n={n} best: {best_n_obj:.6f}")

    # Phase 3: Intensive polishing (~150s remaining)
    print(f"\n=== Phase 3: Polishing (best={best_score:.6f}, n={len(best_seq)}) ===")

    if best_seq is not None:
        n_best = len(best_seq)
        a_cur = best_seq.copy()
        polish_round = 0

        while time.monotonic() < deadline:
            polish_round += 1
            remaining = deadline - time.monotonic()
            if remaining < 4:
                break

            # High-tau mirror descent
            a_pol, _ = mirror_descent(
                a_cur, max_iter=int(min(8000, remaining * 2)),
                tau_start=15.0, tau_end=100.0,
                eta_init=0.15,
                patience=120,
                pert_scale=0.015,
                label=f"pol{polish_round}"
            )
            obj = compute_obj(a_pol)
            if obj < best_score:
                best_score = obj
                best_seq = a_pol.copy()
                a_cur = a_pol.copy()
                print(f"  Polish {polish_round}: {best_score:.6f}")

            # Perturbation escape
            if remaining > 15:
                for scale in [0.04, 0.02, 0.01]:
                    a_pert = a_cur * np.exp(rng.normal(0, scale, n_best))
                    a_pert = np.maximum(a_pert, 0)
                    s = a_pert.sum()
                    if s < 1e-10:
                        continue
                    a_pert /= s
                    obj_p = compute_obj(a_pert)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_pert.copy()
                        a_cur = a_pert.copy()
                        print(f"  Perturb s={scale}: {best_score:.6f}")

            # Try nearby lengths via interpolation
            if remaining > 25:
                for delta in [-20, -10, 10, 20, 50]:
                    n_try = n_best + delta
                    if n_try < 30 or n_try > 3000:
                        continue
                    a_interp = interpolate_seq(best_seq, n_try)
                    a_try, _ = mirror_descent(
                        a_interp, max_iter=3000,
                        tau_start=10.0, tau_end=60.0,
                        eta_init=0.2,
                        patience=100,
                        pert_scale=0.02,
                        label=f"ip{n_try}"
                    )
                    obj_try = compute_obj(a_try)
                    if obj_try < best_score:
                        best_score = obj_try
                        best_seq = a_try.copy()
                        n_best = n_try
                        a_cur = a_try.copy()
                        print(f"  Interp n={n_try}: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100) / 100
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    result = [float(max(0, min(1000.0, x))) for x in best_seq]
    return result
