import numpy as np

def lp_minimize_M(f, time_limit=5):
    """Find g minimizing M where M = max(f*g), sum(g)=1, g>=0"""
    n = len(f)
    # Variables: x = [g_0, ..., g_{n-1}, M]
    # Objective: minimize M -> c = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Constraints:
    # A*g - M <= 0 for each position k (2n-1 constraints)
    # sum(g) = 1 (1 constraint)
    # g >= 0 (bounds)
    
    A = np.zeros((n, n))  # Convolution matrix: (A*g)[k] = (f*g)[k]
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = f[j]
    
    # A_ub * x <= b_ub: A*g - M <= 0
    A_ub = np.zeros((2*n-1, n+1))

    A_ub[:, -1] = -1.0
    b_ub = np.zeros(2*n-1)

    A_eq = np.zeros((1, n+1))

    bounds = [(0, None)] * n + [(0, None)]  # g >= 0, M >= 0

    it = 0
    no_improve = 0
    max_no_improve = 30

    while time.monotonic() - start < time_budget:
        g, M_opt = lp_direct(f, time_limit=min(5.0, max(1.0, (time_budget - (time.monotonic() - start)) * 0.3)))
        if g is None:
            break

        it += 1
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        improved = False
        steps_to_try = [t, t * 0.5, t * 1.5, t * 0.25, 0.7, 0.9, 1.0]
        steps_to_try = [min(1.0, max(0.01, s)) for s in steps_to_try]

        for step in steps_to_try:
            f_new = (1 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            score = compute_score(f_new)
            if score < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = score
                f = f_new
                t = min(1.0, t * 1.15)
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            t *= 0.65
            if t < t_min:
                # Try perturbations to escape
                escaped = False
                for _ in range(5):
                    f_pert = f * (1 + 0.15 * rng.normal(0, 1, n))
                    f_pert = np.clip(f_pert, 0, None)
                    f_pert = normalize(f_pert)
                    if compute_score(f_pert) < best_score:
                        f = f_pert
                        best_f = f.copy()
                        best_score = compute_score(f)
                        t = t0 * 0.5
                        no_improve = 0
                        escaped = True
                        break
                if not escaped and no_improve > max_no_improve:
                    break

    return best_f, best_score, it


def spectral_flat(f, n_iter=200, rng=None):
    """Spectral flattening: project onto flat-magnitude spectrum + non-negativity."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    n_fft = 1 << (2 * n - 1).bit_length()

    best_f = f.copy()
    best_score = compute_score(f)
    stall_count = 0

    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        mag = np.abs(F)
        phase = np.angle(F)

        current_energy = np.sum(f ** 2)
        n_bins = len(mag)
        target_mag = np.sqrt(current_energy * n_fft / n_bins)

        mag_new = np.full_like(mag, target_mag)
        mag_new[0] = 1.0

        if it % 40 == 39:
            phase = phase + rng.normal(0, 0.03, len(phase))

        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s

        score = compute_score(f_new)
        if score < best_score - 1e-10:
            best_score = score
            best_f = f_new.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 100:
                break
        f = f_new

    return best_f, best_score


def pressure_update(f, n_iter=150, eta=0.12, rng=None):
    """Pressure-based update using subgradient at near-max positions."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)

    best_f = f.copy()
    best_score = compute_score(f)
    stall_count = 0

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()

        # Use weighted subgradient from top few peaks
        top_k = min(5, len(conv))
        idx_sorted = np.argsort(conv)[::-1][:top_k]
        weights = np.exp(-0.5 * ((conv[idx_sorted] - M) / max(M * 0.01, 1e-10)) ** 2)
        weights = weights / weights.sum()

        pressure = np.zeros(n)
        for w, j in zip(weights, idx_sorted):
            for i in range(n):
                idx = j - i
                if 0 <= idx < n:
                    pressure[i] += w * f[idx]

        if pressure.max() > 0:
            pressure = pressure / pressure.max()

        eta_eff = eta * (1.0 / (1.0 + 0.02 * it))
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        score = compute_score(f_new)
        if score < best_score - 1e-12:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 80:
                f_pert = f * (1 + 0.1 * rng.normal(0, 1, n))
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    stall_count = 0
                    eta = min(0.3, eta * 1.3)
                else:
                    break

    return best_f, best_score


def coordinate_descent(f, n_sweeps=5, rng=None):
    """Coordinate descent polishing."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            original = f[i]
            if original < 1e-10:
                continue
            best_val = original
            best_local = compute_score(f)

            for factor in [0.4, 0.6, 0.8, 0.9, 1.1, 1.25, 1.6, 2.0]:
                f[i] = max(0.0, original * factor)
                s = f.sum()
                if s < 1e-10:
                    f[i] = original
                    continue
                score = compute_score(f)
                if score < best_local:
                    best_local = score
                    best_val = f[i]

            f[i] = best_val
            if best_local < best_score:
                best_score = best_local

        f = np.clip(f, 0, None)
        f = normalize(f)

    return f, best_score


def diverse_init(n, rng):
    """Generate a diverse initial sequence."""
    r = rng.random()
    if r < 0.2:
        center = n * rng.uniform(0.3, 0.7)
        width = n * rng.uniform(0.1, 0.4)
        f = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    elif r < 0.4:
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    elif r < 0.6:
        # Two bump
        c1 = n * rng.uniform(0.2, 0.4)
        c2 = n * rng.uniform(0.6, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        f = np.exp(-0.5 * (np.arange(n) - c1) ** 2 / w ** 2) + \
            np.exp(-0.5 * (np.arange(n) - c2) ** 2 / w ** 2)
    elif r < 0.8:
        # Smooth random
        f = np.ones(n)
        for _ in range(4):
            center = int(rng.integers(0, n))
            width = int(rng.integers(3, max(4, n // 6)))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / max(1, width ** 2))
            f *= (1 + 0.4 * rng.standard_normal() * kernel)
        f = np.abs(f) + 0.05
    else:
        # Sinusoidal
        freq = rng.uniform(1, 4)
        phase = rng.uniform(0, 2 * np.pi)
        f = 0.5 + 0.5 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        f = np.clip(f, 0.01, None)

    f = np.clip(f, 0.01, None)
    f = normalize(f)
    f = f * (1 + 0.03 * rng.normal(0, 1, n))
    f = np.clip(f, 0.001, None)
    f = normalize(f)
    return f


def simulated_annealing(f, rng, n_iter=5000, T0=0.04, T_end=0.00005):
    """Simulated annealing with diverse moves."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    cur_score = best_score
    no_improve = 0

    for it in range(n_iter):
        T = T0 * (T_end / T0) ** (it / max(1, n_iter))
        f_new = f.copy()

        r = rng.random()
        if r < 0.25:
            width = max(1, int(rng.integers(1, max(2, n // 12))))
            start = int(rng.integers(0, n - width))
            f_new[start:start + width] *= np.exp(rng.normal(0, 0.5, width))
        elif r < 0.45:
            i = int(rng.integers(0, n))
            f_new[i] = max(0.0, f_new[i] + rng.normal(0, 0.06))
        elif r < 0.65:
            i, j = rng.integers(0, n, 2)
            if i != j:
                f_new[i], f_new[j] = f_new[j], f_new[i]
        elif r < 0.8:
            center = int(rng.integers(0, n))
            width = max(2, int(n // 24))
            kernel = np.exp(-0.5 * (np.arange(n) - center) ** 2 / (width ** 2))
            f_new += rng.normal(0, 0.12) * kernel
        else:
            f_new *= np.exp(rng.normal(0, 0.25, n))

        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        new_score = compute_score(f_new)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            f = f_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 3000 and it > n_iter // 2:
            break

    return best_f, best_score
