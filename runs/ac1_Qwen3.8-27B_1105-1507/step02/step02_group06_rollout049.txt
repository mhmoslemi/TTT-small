    """Compute smoothed objective and gradient using log-sum-exp."""
    n = len(a)
    conv = np.convolve(a, a)
    
    # Log-sum-exp for numerical stability
    max_c = np.max(conv)
    exp_c = np.exp(t * (conv - max_c))
    sum_exp = np.sum(exp_c)
    log_sum_exp_val = max_c + np.log(sum_exp) / t
    weight = exp_c / sum_exp
    
    # Gradient computation
    a_rev = a[::-1]
    conv_aw = np.convolve(a_rev, weight)
    cross = conv_aw[n-1:n-1+n]
    grad = 2.0 * cross

    if S < 0.01:
        return float('inf')
    return float(2.0 * n * M / (S * S))


def compute_smoothed_grad(a, t):
    """Compute smoothed max(conv) and its gradient via log-sum-exp."""
    n = len(a)
    conv = np.convolve(a, a)
    max_c = float(np.max(conv))
    exp_c = np.exp(t * (conv - max_c))
    sum_exp = float(np.sum(exp_c))
    log_sum_exp_val = max_c + np.log(sum_exp) / t
    weight = exp_c / sum_exp

    # Gradient: grad[i] = 2 * sum_k weight[k] * a[k-i] for valid k-i
    # Using the cross-correlation trick:
    # cross[i] = (a[::-1] * weight)[i + n - 1]
    a_rev = a[::-1]
    conv_aw = np.convolve(a_rev, weight)
    cross = conv_aw[n - 1:n - 1 + n]
    grad = 2.0 * cross

    return log_sum_exp_val, grad, conv, max_c


def gradient_optimize(a, deadline, t_init=5.0, t_final=100.0, lr_init=0.02,
                      beta=0.9, verbose_prefix=""):
    """Gradient descent with adaptive temperature and step size."""
    a = a.copy().astype(np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-15:
        a = np.ones(n)
        S = float(n)
    a = a * (np.sqrt(2.0 * n) / S)
    S = float(np.sum(a))

    best_a = a.copy()
    best_score = compute_score(a)
    lr = lr_init
    momentum = np.zeros(n)
    no_improve = 0
    it = 0
    total_iters = 100000

    start_time = time.monotonic()
    elapsed_start = 0.0
    duration = deadline - start_time

    while time.monotonic() < deadline:
        it += 1
        # Adaptive temperature: increase over time
        frac = min(1.0, (time.monotonic() - start_time) / max(duration, 1.0))
        t_current = t_init * (t_final / t_init) ** frac

        obj, grad, conv, max_c = compute_smoothed_grad(a, t_current)

        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break

        # Momentum update (gradient descent = subtract gradient)
        momentum = beta * momentum - lr * grad / (grad_norm + 1e-15)
        new_a = a + momentum
        new_a = np.clip(new_a, 0.0, None)
        new_sum = float(np.sum(new_a))
        if new_sum < 1e-15:
            lr *= 0.5
            no_improve += 1
            if no_improve > 300:
                break
            continue
        new_a = new_a * (S / new_sum)

        new_score = compute_score(new_a)

        if np.isfinite(new_score) and new_score < best_score:
            a = new_a
            best_a = a.copy()
            best_score = new_score
            no_improve = 0
            lr *= 0.9995
            if it % 2000 == 0:
                print(f"    {verbose_prefix}iter {it}: score={best_score:.6f} "
                      f"lr={lr:.6f} t={t_current:.1f}")
        else:
            no_improve += 1
            lr *= 0.97
            momentum *= 0.7
            if lr < 1e-9 or no_improve > 1000:
                break

    return best_a, best_score


def symmetrize(a):
    """Enforce symmetry: a[i] = a[n-1-i]."""
    n = len(a)
    s = 0.5 * (a + a[::-1])
    return s

def random_restart_optimize(a, deadline, n_restarts=5, rng=None):
    """Try random perturbations to escape local minima."""
    if rng is None:
        rng = np.random.default_rng()
    a = a.copy().astype(np.float64)
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)

    restarts_available = max(1, n_restarts)
    time_per_restart = (deadline - time.monotonic()) / restarts_available

    for r in range(restarts_available):
        if time.monotonic() > deadline - 5:
            break
        # Create perturbed version
        perturbed = best_a.copy()
        n_perturb = max(10, n // 10)
        indices = rng.choice(n, size=n_perturb, replace=False)
        perturbed[indices] *= np.exp(rng.normal(0, 0.3, size=n_perturb))
        perturbed = np.clip(perturbed, 0.0, None)
        S = float(np.sum(perturbed))
        if S < 1e-15:
            continue
        perturbed = perturbed * (np.sqrt(2.0 * n) / S)

        # Quick optimization from perturbed start
        sub_deadline = time.monotonic() + time_per_restart * 0.7
        f_opt, s_opt = gradient_optimize(perturbed, sub_deadline,
                                          t_init=5.0, t_final=30.0,
                                          lr_init=0.01,
                                          verbose_prefix=f"[restart{r}] ")
        if s_opt < best_score:
            best_score = s_opt
            best_a = f_opt.copy()

        # Fine polish
        if time.monotonic() < deadline - 5:
            sub_deadline2 = time.monotonic() + time_per_restart * 0.25
            f_opt, s_opt = gradient_optimize(f_opt, sub_deadline2,
                                              t_init=20.0, t_final=100.0,
                                              lr_init=0.005,
                                              verbose_prefix=f"[restart{r}fine] ")
            if s_opt < best_score:
                best_score = s_opt
                best_a = f_opt.copy()

    return best_a, best_score


def make_initializations(n, rng):
    """Create diverse initializations for length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Uniform
    inits.append(('uniform', np.ones(n)))

    # Cosine taper
    cos_t = (1.0 + np.cos(np.pi * idx / (n - 1))) / 2.0
    inits.append(('cosine', cos_t))

    # Gaussian centered
    gauss = np.exp(-0.5 * ((idx - center) / (n / 5.0)) ** 2)
    inits.append(('gauss', gauss))

    # Wide Gaussian
    gauss_w = np.exp(-0.5 * ((idx - center) / (n / 2.5)) ** 2)
    inits.append(('gauss_wide', gauss_w))

    # Raised cosine
    raised = 0.5 * (1 + np.cos(2 * np.pi * idx / (n - 1)))
    inits.append(('raised', raised + 0.01))

    # Inverted parabola
    x_norm = idx / (n - 1)
    par = 1.0 - (2.0 * x_norm - 1.0) ** 2
    inits.append(('parabola', np.clip(par, 0.01, None)))

    # Quartic
    quart = 1.0 - (2.0 * x_norm - 1.0) ** 4
    inits.append(('quartic', np.clip(quart, 0.01, None)))

    # Exponential decay from center
    exp_c = np.exp(-1.5 * np.abs(idx - center) / (n / 3.0))
    inits.append(('exp_center', exp_c))

    # Bimodal
    bimodal = (np.exp(-0.5 * ((idx - n * 0.2) / (n / 12.0)) ** 2) +
               np.exp(-0.5 * ((idx - n * 0.8) / (n / 12.0)) ** 2))
    inits.append(('bimodal', bimodal + 0.01))

    # Smooth random
    r = rng.uniform(0, 1, size=n)
    kernel_size = max(5, n // 15)
    k = np.arange(-kernel_size // 2, kernel_size // 2 + 1)
    kernel = np.exp(-0.5 * (k / (kernel_size / 4.0)) ** 2)
    kernel /= np.sum(kernel)
    r_smooth = np.convolve(r, kernel, mode='same')
    inits.append(('rand_smooth', r_smooth))

    # Slepian-like (low-pass filtered delta)
    freqs = np.fft.rfftfreq(n)
    bandwidth = 0.3
    spec = np.exp(-0.5 * (freqs / bandwidth) ** 2)
    slep = np.abs(np.fft.irfft(spec * n, n=n))
    inits.append(('slepian', slep + 0.01))

    # Linear ramp
    inits.append(('ramp', np.linspace(0.3, 1.0, n)))

    return inits


def propose_candidate(budget_s=1500):
    """Main entry point: search for the best sequence of length 1024."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)
    n = 1024
    rng = np.random.default_rng(2024)

    best_sequence = None
    best_score = float('inf')

    # Try to load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            if parent.size != n:
                x_old = np.linspace(0, 1, parent.size)
                x_new = np.linspace(0, 1, n)
                parent_n = np.interp(x_new, x_old, parent)
            else:
                parent_n = parent.copy()
            score = compute_score(parent_n)
            if np.isfinite(score):
                best_score = score
                best_sequence = parent_n.copy()
                print(f"Parent (n={parent.size}) -> n={n}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    # Generate initializations
    inits = make_initializations(n, rng)
    if best_sequence is not None:
        inits.insert(0, ('parent', best_sequence.copy()))

    print(f"\nInitializations for n={n}:")
    for label, f in inits:
        s = compute_score(f)
        print(f"  {label}: {s:.6f}")

    # Phase 1: Quick gradient optimization on all initializations
    print("\n=== Phase 1: Quick Optimization (all inits) ===")
    phase1_deadline = t0 + 180.0
    time_per_init = (phase1_deadline - t0) / len(inits)
    results = []

    for label, f_init in inits:
        if time.monotonic() > phase1_deadline - 5:
            break
        alloc = min(time_per_init, phase1_deadline - time.monotonic() - 2)
        if alloc < 3:
            break

        # Asymmetric
        f_opt, s_opt = gradient_optimize(f_init, time.monotonic() + alloc * 0.6,
                                          t_init=5.0, t_final=30.0,
                                          lr_init=0.02,
                                          verbose_prefix=f"[{label}] ")
        results.append((s_opt, f_opt, label + "_asym"))
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()

        if time.monotonic() < phase1_deadline - 3:
            # Symmetric version
            f_sym = symmetrize(f_init)
            f_sym_opt, s_sym = gradient_optimize_sym(f_sym, time.monotonic() + alloc * 0.4,
                                                      t_init=5.0, t_final=30.0,
                                                      lr_init=0.02,
                                                      verbose_prefix=f"[{label}_sym] ")
            results.append((s_sym, f_sym_opt, label + "_sym"))
            if s_sym < best_score:
                best_score = s_sym
                best_sequence = f_sym_opt.copy()

    results.sort(key=lambda x: x[0])
    print(f"\nTop 5 after Phase 1:")
    for sc, seq, lbl in results[:5]:
        print(f"  {lbl}: {sc:.6f}")
    print(f"Best: {best_score:.6f}")

    # Phase 2: Focused optimization on top candidates
    print("\n=== Phase 2: Focused Optimization ===")
    phase2_deadline = t0 + 1100.0
    top_k = min(4, len(results))
    time_per_top = (phase2_deadline - time.monotonic()) / max(1, top_k)

    for i in range(top_k):
        if time.monotonic() > phase2_deadline - 20:
            break
        sc, f_init, label = results[i]
        alloc = min(time_per_top, phase2_deadline - time.monotonic() - 10)
        if alloc < 10:
            break
        print(f"\n--- Top {i+1}: {label} (score={sc:.6f}), alloc={alloc:.0f}s ---")

        # Sub-phase A: Medium temperature (broad search)
        sub_dead_a = time.monotonic() + alloc * 0.35
        f_opt, s_opt = gradient_optimize(f_init, sub_dead_a,
                                          t_init=3.0, t_final=20.0,
                                          lr_init=0.01, beta=0.9,
                                          verbose_prefix=f"[{label}-A] ")
        if s_opt < best_score:
            best_score = s_opt
            best_sequence = f_opt.copy()
            print(f"  Phase A: improved to {best_score:.6f}")

        # Sub-phase B: High temperature (fine search)
        if time.monotonic() < phase2_deadline - 15:
            sub_dead_b = time.monotonic() + alloc * 0.35
            f_opt, s_opt = gradient_optimize(f_opt, sub_dead_b,
                                              t_init=20.0, t_final=80.0,
                                              lr_init=0.005, beta=0.85,
                                              verbose_prefix=f"[{label}-B] ")
            if s_opt < best_score:
                best_score = s_opt
                best_sequence = f_opt.copy()
                print(f"  Phase B: improved to {best_score:.6f}")

        # Sub-phase C: Symmetric refinement
        if time.monotonic() < phase2_deadline - 10:
            sub_dead_c = time.monotonic() + alloc * 0.2
            f_sym = symmetrize(f_opt)
            f_opt_s, s_opt_s = gradient_optimize_sym(f_sym, sub_dead_c,
                                                      t_init=10.0, t_final=80.0,
                                                      lr_init=0.004, beta=0.9,
                                                      verbose_prefix=f"[{label}-C] ")
            if s_opt_s < best_score:
                best_score = s_opt_s
                best_sequence = f_opt_s.copy()
                print(f"  Phase C (sym): improved to {best_score:.6f}")

        # Sub-phase D: Random restarts from current best
        if time.monotonic() < phase2_deadline - 5:
            sub_dead_d = time.monotonic() + alloc * 0.1
            f_opt_r, s_opt_r = random_restart_optimize(f_opt, sub_dead_d,
                                                        n_restarts=3, rng=rng)
            if s_opt_r < best_score:
                best_score = s_opt_r
                best_sequence = f_opt_r.copy()
                print(f"  Phase D (restart): improved to {best_score:.6f}")

    # Phase 3: Final fine polish
    print(f"\n=== Phase 3: Final Polish (best={best_score:.6f}) ===")
    if best_sequence is not None and time.monotonic() < deadline - 5:
        # Very fine gradient with high temperature
        f_pol, s_pol = gradient_optimize(best_sequence, deadline - 5.0,
                                          t_init=50.0, t_final=200.0,
                                          lr_init=0.002, beta=0.8,
                                          verbose_prefix="[polish] ")
        if s_pol < best_score:
            best_score = s_pol
            best_sequence = f_pol.copy()

        # Also try symmetric polish
        f_sym = symmetrize(best_sequence)
        if time.monotonic() < deadline - 5:
            f_pol, s_pol = gradient_optimize_sym(f_sym, deadline - 5.0,
                                                  t_init=50.0, t_final=200.0,
                                                  lr_init=0.001, beta=0.8,
                                                  verbose_prefix="[polish_sym] ")
            if s_pol < best_score:
                best_score = s_pol
                best_sequence = f_pol.copy()

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(1024)
        best_score = compute_score(best_sequence)

    internal_score = compute_score(best_sequence)
    print(f"\nInternal score: {internal_score:.6f}")

    official_score = evaluate_sequence([float(x) for x in best_sequence])
    if not np.isfinite(official_score):
        print("WARNING: Fallback to uniform")
        best_sequence = np.ones(512)
        official_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"FINAL: n={len(best_sequence)}, score={official_score:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_sequence]
