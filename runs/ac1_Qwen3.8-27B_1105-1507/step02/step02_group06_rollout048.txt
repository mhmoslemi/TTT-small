        top_indices = np.argsort(conv)[-top_k:]
        w = conv[top_indices].astype(np.float64)
        w = w / w.sum()
        indices = top_indices
    else:
        indices = np.array(weights[0])
        w = np.array(weights[1], dtype=np.float64)
        w = w / w.sum()
    
    grad = np.zeros(n)
    for k, wi in zip(indices, w):
        if wi <= 0:
            continue
        k = int(k)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # ∂conv_k/∂f_j = 2*f[k-j] for j in [i_min, i_max]
        grad[i_min:i_max+1] += 2.0 * wi * f[k - i_max:k - i_min + 1][::-1]
    
    return grad


def line_search(f, g, S, best_score, n_steps=20):
    """Geometric line search along direction f -> g."""
    best_t = 0.0
    best_sc = best_score
    # Geometric grid from 0.005 to 0.95
    log_ts = np.linspace(np.log(0.005), np.log(0.95), n_steps)
    for t in np.exp(log_ts):
        cand = (1.0 - t) * f + t * g
        cand = np.clip(cand, 0.0, None)
        cs = float(np.sum(cand))
        if cs < 1e-10:
            continue
        cand = cand * (S / cs)
        sc = compute_score(cand)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_t = t
    return best_t, best_sc


def lp_optimize(f, deadline, verbose=True):
    """Main LP-based optimization loop."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    it = 0
    no_improve = 0
    t0 = time.monotonic()
    
    while time.monotonic() < deadline:
        it += 1
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        
        if M < 1e-15:
            break
        
        # Solve LP
        g, dual = solve_lp(f, M)
        if g is None:
            no_improve += 1
            if no_improve > 100:
                break
            continue
        
        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            no_improve += 1
            continue
        
        g = g * (S / sum_g)
        
        # Line search
        best_t, sc = line_search(f, g, S, best_score)
        
        if best_t > 0:
            f = (1.0 - best_t) * f + best_t * g
            f = np.clip(f, 0.0, None)
            f_s = float(np.sum(f))
            f = f * (S / f_s)
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            
            if verbose and it % 100 == 0:
                elapsed = time.monotonic() - t0
                print(f"  LP iter {it}: score={best_score:.6f}, t={best_t:.4f}, "
                      f"elapsed={elapsed:.0f}s")
        else:
            no_improve += 1
            if no_improve > 200:
                # Try with a slightly different M
                if no_improve % 50 == 0:
                    g2, _ = solve_lp(f, M * 1.01)
                    if g2 is not None:
                        g2 = g2 * (S / float(np.sum(g2)))
                        best_t2, sc2 = line_search(f, g2, S, best_score)
                        if best_t2 > 0:
                            f = (1.0 - best_t2) * f + best_t2 * g2
                            best_f = f.copy()
                            best_score = sc2
                            no_improve = 0
                if no_improve > 500:
                    break
    
    return best_f, best_score


def subgradient_optimize(f, deadline, lr_init=0.005, top_k=20, verbose=True):
    """Subgradient descent with momentum."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    lr = lr_init
    momentum = np.zeros(n)
    beta = 0.9
    no_improve = 0
    it = 0
    t0 = time.monotonic()
    
    while time.monotonic() < deadline and it < 50000:
        it += 1
        grad = get_subgradient(f, top_k=top_k)
        ng = np.linalg.norm(grad)
        if ng < 1e-15:
            break
        
        grad = grad / ng
        momentum = beta * momentum - lr * grad
        f_new = f + momentum
        f_new = np.clip(f_new, 0.0, None)
        fns = float(np.sum(f_new))
        if fns < 1e-10:
            lr *= 0.5
            no_improve += 1
            if no_improve > 100:
                break
            continue
        f_new = f_new * (S / fns)
        
        sc = compute_score(f_new)
        if np.isfinite(sc) and sc < best_score:
            f = f_new
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            if verbose and it % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"  Sub iter {it}: score={best_score:.6f}, lr={lr:.6f}, "
                      f"elapsed={elapsed:.0f}s")
        else:
            f = f  # don't update
            no_improve += 1
            lr *= 0.97
            momentum *= 0.8
            if lr < 1e-8 or no_improve > 300:
                break
    
    return best_f, best_score

conv_k = sum_{i: valid} a[i] * a[k-i]

∂conv_k / ∂a[j] = a[k-j] (if 0 <= k-j < n) + a[k-j] (if 0 <= k-j < n)

Wait, I need to be more careful. Let me write out the sum explicitly.

conv_k = sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[i] * a[k-i]

When we differentiate with respect to a[j]:
- a[j] appears in the sum when i = j (giving a[j] * a[k-j]), AND when k-i = j (giving a[k-j] * a[j]).
- If j = k-j (i.e., k = 2j), both terms are the same, so we get 2 * a[j].
- Otherwise, we get a[k-j] + a[k-j] = 2*a[k-j].

But wait, we need to check that j is in the valid range for the sum. The sum goes from i=max(0,k-n+1) to i=min(k,n-1). So j must be in this range for the first term. And k-j must be in [0, n-1] for the second term (which is the same condition: 0 <= k-j < n).

So: ∂conv_k / ∂a[j] = 2 * a[k-j] if 0 <= j < n AND 0 <= k-j < n, else 0.

This can be written as: ∂conv_k / ∂a = 2 * flip_shift(a, k) where (flip_shift(a, k))[j] = a[k-j] if 0 <= k-j < n, else 0.

def conv_deriv(f, k):
    n = len(f)
    result = np.zeros(n)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    # result[i_min:i_max+1] = 2 * f[k - [i_min, ..., i_max]]
    # = 2 * f[[k-i_min, ..., k-i_max]]
    idx = k - np.arange(i_min, i_max + 1)
    result[i_min:i_max+1] = 2.0 * f[idx]
    return result
