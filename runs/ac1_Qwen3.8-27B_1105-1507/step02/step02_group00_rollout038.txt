import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    n = len(a)
    # A has shape (2n-1, n)
    # A[k, j] = a[k-j] if 0 <= k-j < n else 0
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        # For j in [j_start, j_end], A[k,j] = a[k-j]
        # k-j ranges from k-j_end to k-j_start
        A[k, j_start:j_end+1] = a[k-j_end:k-j_start+1][::-1]
    return A

def solve_lp(a):
    """Solve: min M s.t. (a*g)_k <= M, sum(g)=1, g>=0"""
    n = len(a)
    A = build_A(a)  # shape (2n-1, n)
    # Variables: g_0, ..., g_{n-1}, M
    # Total variables: n+1
    # Constraints:
    #   A @ g - M <= 0 => A @ g + (-1)*M <= 0  (2n-1 constraints)
    #   sum(g) = 1
    #   g >= 0, M >= 0 (M >= 0 is implicit since all entries of A are >= 0 and g >= 0)
    
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize M
    
    # Inequality constraints: A @ g - M <= 0
    A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    
    # Equality constraint: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # g >= 0, M >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n], res.x[-1]
    return None, None

def lp_alternate_fast(a_init, max_iter=5000, verbose=False):
    """Fast LP alternation with adaptive damping."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    prev_obj = best_obj
    stall_count = 0
    damping = 1.0  # Start with full step
    
    for it in range(max_iter):
        g, M = solve_lp(a)
        if g is None:
            break
        
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Objective of the LP solution
        obj_g = compute_obj(g)
        
        # Adaptive damping: if oscillating, increase damping
        if it > 0 and obj_g > prev_obj + best_obj * 0.001:
            damping = min(0.9, damping + 0.05)
        else:
            damping = max(0.5, damping - 0.01)
        
        a_new = (1.0 - damping) * a + damping * g
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s
        
        obj_new = compute_obj(a_new)
        prev_obj = obj_new
        
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        a = a_new
        
        if stall_count > 500:
            break
        
        if verbose and it % 100 == 0:
            print(f"    LP it={it}: obj={best_obj:.8f}, damping={damping:.2f}")
    
    return best_a, best_obj

Let me think about the active_set_polish function. The idea is:
1. Compute $h = a*a$

def project_simplex(v):
    """Project v onto the simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

This is the standard algorithm from "Euclidean projection on the simplex" (Duchi et al., 2012).

```python
"""
LP-alternation optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive damping, multi-length search, active-set polynomial, and coordinate polish.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build the (2n-1) x n constraint matrix for convolution with a."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je + 1] = a[k - je:k - js + 1][::-1]
    return A


def solve_lp(a):
    """Solve LP: min M s.t. (a*g)_k <= M, sum(g)=1, g>=0. Returns (g, M) or (None, None)."""
    n = len(a)
    A = build_A(a)
    nv = n + 1  # n components of g + M
    c = np.zeros(nv)
    c[n] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * nv
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n].copy(), float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_alternate(a_init, max_iter=20000, deadline=None, verbose=False):
    """LP alternation with adaptive damping and stagnation restart."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = float(np.sum(a))
    if s < 1e-12:
        a[:] = 1.0 / n
    else:
        a /= s

    best_a = a.copy()
    best_obj = compute_obj(a)
    
    damp = 1.0
    prev_obj = best_obj
    stall = 0
    patience = 300
    rng_local = np.random.default_rng(int(np.sum(a * np.arange(n)) * 1e6) % (2**31))

    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        g, M = solve_lp(a)
        if g is None:
            break

        gs = float(np.sum(g))
        if gs < 1e-12:
            break
        g /= gs

        a_new = (1.0 - damp) * a + damp * g
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new < 1e-12:
            break
        a_new /= s_new

        obj_new = compute_obj(a_new)

        # Adaptive damping
        if obj_new > prev_obj * 1.01 + 1e-9:
            damp = max(0.2, damp * 0.85)
        elif obj_new < prev_obj * 0.995:
            damp = min(1.0, damp * 1.1)
        else:
            damp = min(1.0, damp + 0.02)

        prev_obj = obj_new

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new

        if stall >= patience:
            # Perturb and restart
            noise = rng_local.normal(0, 0.05, n)
            a = a + noise
            a = np.maximum(a, 0)
            s_p = float(np.sum(a))
            if s_p < 1e-12:
                break
            a /= s_p
            stall = 0
            patience = max(100, patience - 30)
            if patience < 50:
                break

        if verbose and it % 200 == 0:
            print(f"    LP it={it}: obj={best_obj:.8f} damp={damp:.3f} stall={stall}")

    return best_a, best_obj


def project_simplex(v):
    """Project vector v onto the probability simplex."""
    d = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, d + 1) > (cssv - 1.0))[0]
    if len(rho_candidates) == 0:
        return np.ones(d) / d
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_polish(a, n_steps=5000, deadline=None, verbose=False):
    """Polish using projected subgradient on active set."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = float(np.sum(a))
    a /= s
    best_a = a.copy()
    best_obj = compute_obj(a)

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        h = np.convolve(a, a)
        hmax = float(np.max(h))
        if hmax < 1e-15:
            break

        # Active set with widening tolerance
        tol = hmax * (0.005 + 0.02 * (1.0 - step / max(1, n_steps)))
        active = h >= hmax - tol
        if not np.any(active):
            active = h >= hmax - 1e-10

        # Subgradient: average of active gradients
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            grad[js:je + 1] += a[k - je:k - js + 1][::-1]
        grad /= float(len(active_idx))
        # Double for bilinear term (except diagonal)
        # Actually, the gradient of (a*a)_k w.r.t. a_j is a_{k-j} + a_{j} if k-j != j
        # Let me just double the gradient (approximately)
        # More precisely: grad_k[j] = a[k-j] for j in range, plus a[j] if k-j==j
        # For simplicity, just use the one-sided gradient and multiply by 2
        grad *= 2.0

        # Step size
        eta = 2.0 / (best_obj * (step + 1) ** 0.7)
        eta = min(eta, 0.5)

        a_new = a - eta * grad
        a_new = project_simplex(a_new)

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def coord_polish(a, n_rounds=5, deadline=None):
    """Coordinate-wise golden section search."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_obj = compute_obj(a)
    best_a = a.copy()

    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            old = a[j]
            # Try a range of values
            candidates = [0.0, old * 0.5, old * 0.8, old * 1.2, old * 1.5, old * 2.0]
            best_val = old
            best_local = best_obj
            for cv in candidates:
                a[j] = max(0.0, cv)
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_val = max(0.0, cv)
            # Binary search
            lo, hi = 0.0, max(old * 3, 1e-3)
            if best_val < old:
                hi = old
            else:
                lo = old
            for _ in range(15):
                mid = (lo + hi) / 2.0
                a[j] = mid
                obj = compute_obj(a)
                if obj < best_local:
                    best_local = obj
                    best_val = mid
                    lo = mid
                else:
                    hi = mid
            a[j] = best_val
            if best_local < best_obj:
                best_obj = best_local
                best_a = a.copy()

    return best_a, best_obj


def structured_init(n, rng):
    """Generate a diverse structured initialization, normalized to sum=1."""
    choice = int(rng.integers(0, 8))
    x = np.arange(n, dtype=np.float64)
    if choice == 0:
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0.0, np.cos(2 * np.pi * freq * x / n + phase))
    elif choice == 1:
        width = rng.uniform(n * 0.1, n * 0.5)
        center = rng.uniform(0, n)
        a = np.exp(-0.5 * ((x - center) / width) ** 2)
    elif choice == 2:
        mid = n // 2
        a = np.ones(n)
        a[:mid] *= rng.uniform(0.3, 1.5)
        a[mid:] *= rng.uniform(0.3, 1.5)
    elif choice == 3:
        period = max(2, int(rng.integers(2, max(3, n // 10))))
        pattern = rng.random(period)
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 4:
        a = np.abs(np.sin(x * rng.uniform(0.1, 2.0)))
    elif choice == 5:
        a = np.exp(-rng.uniform(0.5, 3.0) * (x / n - 0.5) ** 2)
    elif choice == 6:
        # Symmetric
        half = n // 2
        a_half = rng.random(half + 1)
        a = np.zeros(n)
        a[:half + 1] = a_half
        a[half + 1:] = a_half[:max(0, n - half - 1)][::-1]
    else:
        a = rng.random(n)

    sd = float(np.sum(a))
    if sd < 1e-10:
        a = rng.random(n)
        sd = float(np.sum(a))
    return a / sd

```python

LP-alternation optimizer with adaptive damping, multi-length search,
subgradient polish, and coordinate refinement.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        A[k, js:je + 1] = a[k - je:k - js + 1][::-1]
    return A


def solve_lp(a):
    n = len(a)
    A = build_A(a)
    nv = n + 1


    c = np.zeros(nv)
    c[n] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * nv
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n].copy(), float(res.x[n])
    except Exception:
        pass
    return None, None
