"""
Optimized step function search using LP direction finding, 
soft-max gradient descent, and peak reduction strategies.
Targets score < 1.5027.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    """Solve: max sum(g) s.t. g >= 0, conv(f,g)_k <= M for all k."""
    n = len(f)
    if A_cache is None:
        A = build_conv_matrix_sparse(f)
    else:
        A = A_cache
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None

import numpy as np

from scipy.optimize import linprog

def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = sum_j f[k-j]*g[j] = convolve(f,g)[k]."""
    n = len(f)
    data = np.tile(f, n)
    i_arr = np.arange(n * n)
    rows = (i_arr // n) + (i_arr % n)
    indptr = np.arange(0, (n + 1) * n, n, dtype=np.int64)
    return csr_matrix((data.astype(np.float64), rows.astype(np.int64), indptr),

def solve_lp(f, M, A, active_set=None):
    """Solve: max sum(g) s.t. g >= 0, (A@g)[k] <= M for active k."""
    n = len(f)
    if active_set is not None:
        A_sub = A[active_set]
        b_ub = np.full(len(active_set), M)
    else:
        A_sub = A
        b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def get_active_set(f, A, M, margin=0.1):
    """Get indices where (A@f)[k] >= M*(1-margin)."""
    vals = A @ f
    threshold = M * (1.0 - margin)
    active = np.where(vals >= threshold)[0]

        if l_max >= 0:
            grad[j] = 2.0 * np.dot(f[:l_max + 1], p[j:j + l_max + 1])
    
    soft_val = max_C + np.log(total) / beta
    return grad, soft_val


def compute_softmax_gradient_fft(f, beta):
    """Compute soft-max gradient using FFT for efficiency."""
    n = len(f)
    C = np.convolve(f, f)
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros(n), 0.0
    p = exp_C / total  # length 2n-1

    # grad[j] = 2 * sum_{l=0}^{min(n-1, 2n-2-j)} f[l] * p[j+l]
    # This is the "valid" part of the cross-correlation of f and p
    # conv(f, p)[j] = sum_i f[i] * p[j-i] -- not what we want
    # We want sum_l f[l] * p[j+l]
    # = sum_l f[l] * p_reversed[(2n-2) - (j+l)]
    # Let p_rev = p[::-1] (length 2n-1)
    # sum_l f[l] * p_rev[2n-2-j-l]
    # = conv(f, p_rev)[2n-2-j] (full convolution)
    
    p_rev = p[::-1]
    # Full convolution of f (length n) and p_rev (length 2n-1) has length n + 2n-1 - 1 = 3n-2
    conv_fp = np.convolve(f, p_rev)  # length 3n-2
    # grad[j] = 2 * conv_fp[2n-2-j]
    grad = 2.0 * conv_fp[2 * n - 2 - np.arange(n)]
    
    soft_val = max_C + np.log(total) / beta
    return grad, soft_val


def gradient_optimize(f, deadline, beta_start=10.0, beta_end=500.0,
                      lr_start=0.05, max_iters=5000, verbose_prefix=""):
    """Soft-max gradient optimization with Adam."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, float('inf')
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())

    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = lr_start
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it_frac = min(1.0, it / (max_iters * 0.6))
        beta = beta_start * (beta_end / beta_start) ** it_frac

        grad, soft_val = compute_softmax_gradient_fft(f, beta)

        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        update = m_hat / (np.sqrt(v_hat) + eps)

        new_f = f - lr * update
        new_f = np.clip(new_f, 0.0, 1000.0)
        ns = float(np.sum(new_f))
        if ns < 1e-10:
            new_f = f.copy()
        else:
            new_f = new_f / ns

        sc = evaluate_sequence(new_f.tolist())
        if np.isfinite(sc) and sc < best_score:
            f = new_f
            best_f = f.copy()
            best_score = sc
            no_improve = 0
            if it % 500 == 0:
                print(f"    {verbose_prefix}Grad iter {it}: score={best_score:.6f} (beta={beta:.1f})")
        else:
            no_improve += 1
            if no_improve > 300:
                lr *= 0.4
                no_improve = 0
                if lr < 1e-7:
                    break

    return best_f, best_score


def peak_reduction(f, deadline, max_iters=100, verbose_prefix=""):
    """Targeted peak reduction: identify top peaks and do local LP on subsets."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    f = f / S
    n = len(f)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    A = build_conv_matrix_sparse(f)
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        C = np.convolve(f, f)
        M = float(np.max(C))
        if M <= 0:
            break
        it += 1

        # Get top 20% of constraints as "active"
        n_active = max(10, (2 * n - 1) // 5)
        active = np.argsort(C)[-n_active:]
        
        g = solve_lp(f, M, A, active_set=active)
        if g is None:
            # Try with more constraints
            n_active = (2 * n - 1) // 2
            active = np.argsort(C)[-n_active:]
            g = solve_lp(f, M, A, active_set=active)
        if g is None:
            if it > 30:
                break
            continue

        # Verify against ALL constraints
        conv_g = np.convolve(f, g)
        if np.max(conv_g) > M * 1.01:
            # Solution violates some constraints; scale it down
            scale = M / np.max(conv_g)
            g = g * scale * 0.9

        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            continue
        g = g / sum_g

        # Line search
        best_t = 0
        best_s = best_score
        for t in [0.1, 0.2, 0.3, 0.15, 0.05, 0.5]:
            new_f = (1 - t) * f + t * g
            new_f = np.clip(new_f, 0, 1000)
            ns = np.sum(new_f)
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = t

        if best_t > 0:
            f = (1 - best_t) * f + best_t * g
            f = np.clip(f, 0, 1000)
            f = f / np.sum(f)
            best_f = f.copy()
            best_score = best_s
            A = build_conv_matrix_sparse(f)
        else:
            if it > 20:
                break

    return best_f, best_score


def tukey_window(n, r=0.3):
    w = np.ones(n)
    m = max(1, int(np.floor(r * n / 2)))
    for i in range(m):
        w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
        w[n - 1 - i] = w[i]
    return w


def cosine_taper(n, frac=0.3):
    """Cosine taper on edges."""
    w = np.ones(n)
    m = max(1, int(n * frac))
    for i in range(m):
        w[i] = 0.5 * (1 + np.cos(np.pi * i / m))
        w[n - 1 - i] = 0.5 * (1 + np.cos(np.pi * (m - 1 - i) / m))
    return w


def power_taper(n, alpha=2.0, frac=0.5):
    """Power-law taper."""
    w = np.ones(n)
    m = max(1, int(n * frac))
    for i in range(m):
        t = i / m
        w[i] = t ** alpha
        w[n - 1 - i] = t ** alpha
    return w


def exponential_window(n, rate=3.0):
    """Exponential window (causal)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-rate * idx / n)


def ramp_window(n, strength=0.5):
    """Linear ramp (asymmetric)."""
    idx = np.arange(n, dtype=np.float64)
    return 1.0 - strength * idx / (n - 1)


def perturb(f, rng, strength=0.05):
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.1, 3.0)
    new_f = np.clip(new_f, 0.0, 1000.0)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S


def full_optimize(f_init, deadline, do_lp=True, do_grad=True, do_peak=True,
                  lp_frac=0.4, grad_frac=0.35, peak_frac=0.25,
                  verbose_prefix=""):
    """Run multi-phase optimization."""
    f_init = f_init.copy().astype(np.float64)
    S = float(np.sum(f_init))
    if S < 1e-10:
        return f_init, float('inf')
    f_init = f_init / S

    best_f = f_init.copy()
    best_score = evaluate_sequence(f_init.tolist())
    t0 = time.monotonic()
    total_budget = max(1.0, deadline - t0)

    if do_lp:
        d1 = t0 + total_budget * lp_frac
        if d1 > t0 + 5:
            f1, s1 = lp_optimize(f_init, d1, verbose_prefix=verbose_prefix + "LP1")
            if s1 < best_score:
                best_f, best_score = f1, s1

    if do_grad:
        d2 = t0 + total_budget * (lp_frac + grad_frac)
        if d2 > time.monotonic() + 5:
            f2, s2 = gradient_optimize(best_f, d2, beta_start=20.0, beta_end=400.0,
                                        lr_start=0.03, verbose_prefix=verbose_prefix + "GR1")
            if s2 < best_score:
                best_f, best_score = f2, s2

    if do_peak:
        d3 = deadline - 2
        if d3 > time.monotonic() + 5:
            f3, s3 = peak_reduction(best_f, d3, verbose_prefix=verbose_prefix + "PK")
            if s3 < best_score:
                best_f, best_score = f3, s2 if False else s3
            # Final LP polish
            if time.monotonic() < deadline - 5:
                f4, s4 = lp_optimize(best_f, deadline - 3, step_init=0.05,
                                      max_iters=30, verbose_prefix=verbose_prefix + "LP2")
                if s4 < best_score:
                    best_f, best_score = f4, s4

    return best_f, best_score


def propose_candidate(budget_s=1500):
    """Main search function."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(42)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                score = evaluate_sequence(parent.tolist())
                if np.isfinite(score):
                    best_score = score
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    # Determine which lengths to try based on quick evaluations
    lengths_to_try = [512, 768, 1024, 640]
    
    # Quick evaluation of interpolated parent at each length
    length_scores = {}
    for n in lengths_to_try:
        if best_sequence is not None and len(best_sequence) > 1:
            x_parent = np.linspace(0, 1, len(best_sequence))
            x_new = np.linspace(0, 1, n)
            f_interp = np.interp(x_new, x_parent, best_sequence)
            f_interp = np.clip(f_interp, 0, 1000)
            if f_interp.sum() > 0.01:
                sc = evaluate_sequence(f_interp.tolist())
                length_scores[n] = sc
                if np.isfinite(sc):
                    print(f"  Interp n={n}: score={sc:.6f}")
        else:
            length_scores[n] = float('inf')

    # Sort lengths by initial score (best first)
    lengths_sorted = sorted(lengths_to_try, key=lambda x: length_scores.get(x, 999))

    # Time allocation: give more time to more promising lengths
    # Total optimization time
    opt_deadline = deadline - 30  # leave 30s for final polish
    total_opt_time = max(10.0, opt_deadline - t0)

    # Allocate time: 40% to best length, 30% to second, 15% to third, 15% to fourth
    time_fracs = [0.35, 0.30, 0.20, 0.15]
    current_time = time.monotonic()

    for idx, n in enumerate(lengths_sorted):
        if current_time > opt_deadline - 60:
            break
        if idx >= len(time_fracs):
            break

        n_budget = total_opt_time * time_fracs[idx]
        n_deadline = current_time + n_budget
        
        print(f"\n=== Optimizing n={n} (budget {n_budget:.0f}s, initial score={length_scores.get(n, 'inf'):.4f}) ===")
        
        # Generate initializations for this length
        inits = []
        if best_sequence is not None:
            if len(best_sequence) == n:
                inits.append(('parent', best_sequence.copy()))
            else:
                x_parent = np.linspace(0, 1, len(best_sequence))
                x_new = np.linspace(0, 1, n)
                f_interp = np.interp(x_new, x_parent, best_sequence)
                inits.append(('parent_interp', np.clip(f_interp, 0, None)))

        # Various windows
        inits.append(('tukey_02', tukey_window(n, 0.2)))
        inits.append(('tukey_03', tukey_window(n, 0.3)))
        inits.append(('cos_taper', cosine_taper(n, 0.3)))
        inits.append(('exp_causal', exponential_window(n, 2.0)))
        inits.append(('ramp', ramp_window(n, 0.6)))
        inits.append(('power_taper', power_taper(n, 3.0, 0.4)))

        # Random smooth initialization
        r_rng = np.random.default_rng(123 + n)
        x_rand = r_rng.standard_normal(n)
        kernel = np.exp(-0.5 * (np.arange(-15, 16) / 4.0) ** 2)
        smooth_rand = np.convolve(x_rand, kernel, mode='same')
        smooth_rand = np.abs(smooth_rand) + 0.1
        inits.append(('rand_smooth', smooth_rand))

        # Asymmetric windows (important for beating symmetric bound)
        idx_arr = np.arange(n, dtype=np.float64) / n
        inits.append(('asym_cos', 0.5 + 0.5 * np.cos(2 * np.pi * idx_arr * 0.3)))
        inits.append(('asym_exp', np.exp(-1.5 * idx_arr) + 0.1))

        n_inits = len(inits)
        per_init_budget = n_budget / max(n_inits, 1) * 0.7  # 70% for inits, 30% for restarts
        min_time_per_init = max(10.0, per_init_budget)

        for label, f_init in inits:
            if current_time > n_deadline - 10:
                break
            if time.monotonic() > opt_deadline - 30:
                break

            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            S = float(np.sum(f_init))
            if S < 0.01:
                continue
            f_init = f_init / S

            init_score = evaluate_sequence(f_init.tolist())
            if not np.isfinite(init_score):
                continue

            if init_score < best_score:
                best_score = init_score
                best_sequence = f_init.copy()
                print(f"  [n={n} {label}] init score={init_score:.6f} (NEW BEST)")

            # Only invest in optimization if this init is competitive
            threshold = best_score * 1.03
            if init_score < threshold or n_inits <= 4:
                init_deadline = min(n_deadline, current_time + min_time_per_init * 2)
                if time.monotonic() < init_deadline - 5:
                    try:
                        bf, bs = full_optimize(f_init, init_deadline,
                                               lp_frac=0.45, grad_frac=0.30, peak_frac=0.25,
                                               verbose_prefix=f"[n={n} {label}] ")
                        if bs < best_score:
                            best_score = bs
                            best_sequence = bf.copy()
                            print(f"  [n={n} {label}] optimized to {bs:.6f} (NEW BEST)")
                        elif bs < threshold:
                            print(f"  [n={n} {label}] optimized to {bs:.6f}")
                    except Exception as e:
                        print(f"  [n={n} {label}] error: {e}")
            current_time = time.monotonic()

        # Restart phase for this length
        if best_sequence is not None and len(best_sequence) == n and current_time < n_deadline - 20:
            n_restarts = min(4, max(1, int((n_deadline - current_time) / 30)))
            for r in range(n_restarts):
                if current_time > n_deadline - 10:
                    break
                strength = 0.03 + 0.02 * r
                f_pert = perturb(best_sequence, rng, strength)
                r_deadline = min(n_deadline, current_time + 25)
                if time.monotonic() < r_deadline - 5:
                    try:
                        bf, bs = full_optimize(f_pert, r_deadline,
                                               lp_frac=0.5, grad_frac=0.3, peak_frac=0.2,
                                               verbose_prefix=f"[n={n} R{r}] ")
                        if bs < best_score:
                            best_score = bs
                            best_sequence = bf.copy()
                            print(f"  [n={n} R{r}] restart: {bs:.6f} (NEW BEST)")
                    except Exception as e:
                        pass
            current_time = time.monotonic()

    # Final intensive refinement on the global best
    if best_sequence is not None:
        print(f"\n=== Final refinement (n={len(best_sequence)}, score={best_score:.6f}) ===")
        final_deadline = deadline - 5
        if time.monotonic() < final_deadline - 20:
            try:
                # Intensive LP
                d = time.monotonic() + min(120, (final_deadline - time.monotonic()) * 0.4)
                bf, bs = lp_optimize(best_sequence, d, step_init=0.1,
                                     max_iters=200, verbose_prefix="[final_LP] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  Final LP: {bs:.6f}")
            except Exception as e:
                print(f"  Final LP error: {e}")

        if time.monotonic() < final_deadline - 15:
            try:
                # Gradient polish
                d = time.monotonic() + min(80, (final_deadline - time.monotonic()) * 0.4)
                bf, bs = gradient_optimize(best_sequence, d, beta_start=100.0,
                                            beta_end=1000.0, lr_start=0.01,
                                            max_iters=3000, verbose_prefix="[final_GR] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  Final Grad: {bs:.6f}")
            except Exception as e:
                print(f"  Final Grad error: {e}")

        if time.monotonic() < final_deadline - 10:
            try:
                # Peak reduction
                d = time.monotonic() + min(60, (final_deadline - time.monotonic()) * 0.5)
                bf, bs = peak_reduction(best_sequence, d, max_iters=80,
                                        verbose_prefix="[final_PK] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  Final Peak: {bs:.6f}")
            except Exception as e:
                print(f"  Final Peak error: {e}")

        if time.monotonic() < deadline - 3:
            try:
                bf, bs = lp_optimize(best_sequence, deadline - 3, step_init=0.03,
                                     max_iters=20, verbose_prefix="[final2] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
            except Exception:
                pass

    # Ensure valid output
    if best_sequence is None:
        best_sequence = tukey_window(512, 0.3)
        best_score = evaluate_sequence(best_sequence.tolist())

    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
