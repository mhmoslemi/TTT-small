```python
import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Compute conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def normalize(a):
    """Normalize a to sum to 1."""
    s = np.sum(a)
    if s < 1e-15:
        return a
    return a / s
```

```python
"""
Optimized step function search using FFT-accelerated subgradient + alternating LP,
with peak-focused gradients, adaptive mixing, multi-length exploration, and
structured initializations. Targets objective < 1.50.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_conv(a):
    """Compute conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize_one(a):
    """Normalize a to sum to 1, ensuring non-negativity."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def peak_focused_gradient(conv, a, top_frac=0.1, tau=10.0):
    """Compute gradient focusing on top peaks only."""
    n = len(a)
    L = len(conv)
    num_top = max(1, int(L * top_frac))
    # Get indices of top peaks
    top_idx = np.argpartition(conv, -num_top)[-num_top:]
    # Log-sum-exp weights on top peaks only
    top_conv = conv[top_idx]
    max_top = np.max(top_conv)
    shifted = tau * (top_conv - max_top)
    shifted = np.clip(shifted, -300, 300)
    weights_top = np.exp(shifted)
    w_sum = np.sum(weights_top)
    if w_sum < 1e-300:
        w_full = np.zeros(L)
        w_full[top_idx[0]] = 1.0
    else:
        w_full = np.zeros(L)
        w_full[top_idx] = weights_top / w_sum
    # Gradient: grad[i] = 2 * sum_k w_full[k] * a[k-i] (valid pairs)
    # This is 2 * (w_full * a) convolved appropriately
    # grad[i] = 2 * sum_k w_full[k] * a[k-i] where 0<=k-i<n
    # = 2 * (correlate(w_full, a, 'full'))[i] taking first n elements
    c = np.correlate(w_full, a, mode='full')
    grad = 2.0 * c[:n]
    return grad


def full_subgradient_gradient(conv, a, tau=10.0):
    """Compute smoothed gradient over all lags."""
    n = len(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -300, 300)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[:n]
    return grad


def subgradient_optimize(a, max_iter=5000, tau_start=5.0, tau_end=60.0,
                         step_init=0.1, top_frac=0.05, focus_peak=True):
    """Smoothed subgradient optimization with peak focus."""
    n = len(a)
    a = normalize_one(a.copy())
    best_a = a.copy()
    conv_best = compute_conv(a)
    best_max = float(np.max(conv_best))

    step = step_init
    stall = 0
    patience = max(100, max_iter // 10)

    for it in range(max_iter):
        frac = it / max(1, max_iter)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.7)

        conv = compute_conv(a)
        cur_max = float(np.max(conv))

        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > patience:
                step *= 0.6
                if step < 1e-10:
                    break
                # Noise escape
                rng_local = np.random.default_rng(it * 7 + 13)
                a = a + rng_local.normal(0, 0.02 * step, n)
                a = normalize_one(a)
                continue

        if focus_peak:
            grad = peak_focused_gradient(conv, a, top_frac=top_frac, tau=tau)
        else:
            grad = full_subgradient_gradient(conv, a, tau=tau)

        # Center for sum constraint
        grad_centered = grad - np.mean(grad)
        a_new = a - step * grad_centered
        a_new = normalize_one(a_new)
        a = a_new

        # Adaptive step
        if it % 50 == 0:
            step *= 0.997

    return best_a


def build_sparse_A(a):
    """Build constraint matrix row by row for LP. Returns sparse-friendly format."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            for j in range(j_min, j_max + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
    return rows, cols, vals


def solve_lp_for_g(a, M, time_limit=10):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    rows, cols, vals = build_sparse_A(a)
    A_sparse = (np.array(rows), np.array(cols), np.array(vals))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None


def lp_alternating(a, max_iter=30, t_start=0.4, t_end=0.15, time_limit=8):
    """Alternating LP optimization with adaptive mixing."""
    n = len(a)
    a = normalize_one(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    stall = 0
    for it in range(max_iter):
        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_for_g(a, M, time_limit=time_limit)
        if g is None:
            break

        if sum_g <= 1.0 + 1e-3:
            stall += 1
            if stall > 10:
                break
            continue
        stall = 0

        # Normalize g to sum 1
        g = normalize_one(g)

        # Adaptive mixing: smaller t when g is very different
        diff = np.linalg.norm(g - a) / (np.linalg.norm(a) + 1e-10)
        t_frac = (1.0 - it / max_iter)
        t = t_start * (t_end / t_start) ** t_frac
        t = min(t, 2.0 / (1.0 + diff))  # Cap based on difference

        a_new = normalize_one((1.0 - t) * a + t * g)
        obj_new = compute_obj(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def directed_peak_reduction(a, num_rounds=2000, top_k=5, reduce_frac=0.3):
    """Directly reduce coordinates contributing to peak lags."""
    n = len(a)
    a = normalize_one(a.copy())
    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for rnd in range(num_rounds):
        conv = compute_conv(a)
        cur_max = float(np.max(conv))
        cur_obj = 2.0 * n * cur_max  # assuming sum=1

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()

        # Find peak lags
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        peak_lags = sorted(peak_idx.tolist())

        # For each peak lag, find contributing coordinates and reduce them
        for k in peak_lags:
            # conv[k] = sum_i a[i]*a[k-i]
            # Blame for coordinate i: a[i] * a[k-i]
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_min > j_max:
                continue
            contributions = np.zeros(n)
            js = np.arange(j_min, j_max + 1)
            contributions[js] = a[js] * a[k - js]
            # Reduce high contributors
            threshold = np.percentile(contributions[contributions > 0], 70) if np.any(contributions > 0) else 0
            to_reduce = contributions > threshold
            if np.any(to_reduce):
                a[to_reduce] *= (1.0 - reduce_frac * (1.0 - rnd / num_rounds) * 0.5)

        # Renormalize
        a = normalize_one(a)

    return best_a


def generate_initialization(n, rng, method=None):
    """Generate diverse initializations."""
    if method is None:
        method = rng.integers(0, 7)

    idx = np.arange(n)

    if method == 0:
        # Cosine
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * idx / n + phase)
    elif method == 1:
        # Two-hump
        c1 = rng.uniform(0.2, 0.4) * n
        c2 = rng.uniform(0.6, 0.8) * n
        w1 = rng.uniform(0.05, 0.2) * n
        w2 = rng.uniform(0.05, 0.2) * n
        a = np.exp(-((idx - c1) / w1) ** 2) + np.exp(-((idx - c2) / w2) ** 2)
        a += 0.1
    elif method == 2:
        # Periodic with random phase
        freq = rng.integers(2, max(3, n // 10))
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * freq * idx / n + phase)
    elif method == 3:
        # Smoothed random
        a = rng.random(n)
        # Simple smoothing
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a += 0.1
    elif method == 4:
        # Flat with noise
        a = np.ones(n)
        a += rng.normal(0, 0.3, n)
        a = np.maximum(a, 0)
    elif method == 5:
        # Ghost/Golay-inspired: alternating with variation
        base = np.ones(n) * 0.5
        for i in range(0, n, 2):
            base[i] = rng.uniform(0.7, 1.0)
            if i + 1 < n:
                base[i + 1] = rng.uniform(0.1, 0.4)
        kernel = np.ones(3) / 3
        a = np.convolve(base, kernel, mode='same')
    else:
        # Curve: power law
        alpha = rng.uniform(0.3, 2.0)
        a = (idx / (n - 1 + 1e-10)) ** alpha
        a = a + (1 - a) * 0.3
        a += 0.1

    a = np.maximum(a, 0)
    return normalize_one(a)

        print(f"  Validation improved: {val:.6f}")
        best_score = val

    return result
```
