"""
Optimization for minimizing 2n*max(conv(f,f))/sum(f)^2 over non-negative sequences.
Uses softmax-smoothed gradient descent with Adam, LP refinement, and multi-start strategy.

import numpy as np
import time

from scipy.sparse import csr_matrix


def tukey_window(n, r=0.5):
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def compute_softmax_gradient(f, beta):
    """Compute gradient of soft-max of conv(f,f)."""
    n = len(f)
    C = np.convolve(f, f)
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros(n), max_C
    p = exp_C / total
    # grad[j] = 2 * sum_k p[k] * f[k-j] (with valid indices)
    # This is 2 * correlate(p, f, mode='valid')
    corr = np.correlate(p, f, mode='valid')
    grad = 2.0 * corr
    soft_max_val = max_C + np.log(total) / beta
    return grad, soft_max_val

def build_conv_matrix(f):
    """Build matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((np.array(vals), (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))),
                      shape=(2 * n - 1, n))
