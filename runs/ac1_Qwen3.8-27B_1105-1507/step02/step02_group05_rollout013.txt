"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Smoothed-max (softmax & p-norm) subgradient with Nesterov momentum and adaptive step
2. LP best-response mirror descent
3. Multi-length diverse search with annealed restarts
4. Final hard-max subgradient polish
Key innovations: smoothed gradient for better landscapes, Nesterov acceleration,
adaptive temperature schedule, and more diverse exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def softmax_gradient(a, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*b_k)) w.r.t. a[0..n-1]."""
    b = fast_conv(a)
    max_b = b.max()
    exp_b = np.exp(beta * (b - max_b))
    p = exp_b / exp_b.sum()
    n = len(a)
    cc = np.correlate(p, a, mode='full')
    grad = 2.0 * cc[:n]
    return grad, b


def pnorm_gradient(a, p):
    """Compute gradient of (sum b_k^p)^(1/p) w.r.t. a."""
    b = fast_conv(a)
    b_pos = np.maximum(b, 1e-30)
    w = b_pos ** (p - 1)
    norm_val = (b_pos ** p).sum()
    if norm_val < 1e-30:
        return np.zeros(len(a)), b
    scale = norm_val ** (1.0 / p - 1.0)
    cc = np.correlate(w, a, mode='full')
    grad = 2.0 * scale * cc[:len(a)]
    return grad, b


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def softmax_subgradient_descent(a, max_iters, deadline, tracker, beta_start=5.0, beta_end=50.0):
    """Nesterov-accelerated softmax subgradient descent with adaptive step."""
    a = np.maximum(a.astype(np.float64), 1e-6)
    a = project_simplex(a)
    current_J = compute_J(a)
    tracker.update(a)

    n = len(a)
    x_prev = a.copy()
    x_curr = a.copy()
    momentum = 0.0

    lr_schedule = 0.5 / np.sqrt(1.0)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive beta: start smooth, end sharp
        frac = it / max(float(max_iters), 1)
        beta = beta_start + (beta_end - beta_start) * frac ** 0.7

        # Compute gradient
        grad, b = softmax_gradient(x_curr, beta)

        # Adaptive learning rate based on gradient norm
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            lr = lr_schedule
        else:
            # Scale step so that the max change is reasonable
            lr = 0.3 / (gnorm + 1e-10)
            lr = min(lr, 0.5)

        # Nesterov extrapolation
        momentum = 0.9 * momentum + 0.1 * (2 * it / (it + 1 + 2 * momentum))
        x_nest = x_curr + momentum * (x_curr - x_prev)
        x_nest = np.maximum(x_nest, 0)
        x_nest = project_simplex(x_nest)

        # Gradient step
        x_next = x_nest - lr * grad
        x_next = np.maximum(x_next, 0)
        x_next = project_simplex(x_next)

        # Check improvement
        J_trial = compute_J(x_next)
        if J_trial < current_J - 1e-15:
            x_prev = x_curr.copy()
            x_curr = x_next
            current_J = J_trial
        else:
            # Try smaller step
            x_next2 = x_curr - 0.1 * lr * grad
            x_next2 = np.maximum(x_next2, 0)
            x_next2 = project_simplex(x_next2)
            J_trial2 = compute_J(x_next2)
            if J_trial2 < current_J - 1e-15:
                x_prev = x_curr.copy()
                x_curr = x_next2
                current_J = J_trial2
            else:
                # Stuck - try random perturbation
                if it % 50 == 0 and it > 0:
                    noise = np.random.default_rng(it * 12345 + n).normal(0, 0.01, n)
                    x_pert = x_curr + noise
                    x_pert = project_simplex(x_pert)
                    J_pert = compute_J(x_pert)
                    if J_pert < current_J + 0.01:
                        x_prev = x_curr.copy()
                        x_curr = x_pert
                        current_J = J_pert
                else:
                    x_prev = x_curr.copy()
                    if it % 100 == 0:
                        tracker.update(x_curr)
                    # Try to detect if we're truly stuck
                    if it > 200:
                        break

        if it % 50 == 0:
            tracker.update(x_curr)

    tracker.update(x_curr)
    return x_curr

    tracker.update(a)

    x_prev = a.copy()
    x_curr = a.copy()
    best_x = x_curr.copy()
    best_J = current_J
    stuck_count = 0

        if time.monotonic() > deadline:
            break

        b = fast_conv(x_curr)
        k_star = int(np.argmax(b))

        j_hi = min(n - 1, k_star)
        if j_lo <= j_hi:
            grad = np.zeros(n)
            grad[j_lo:j_hi + 1] = 2.0 * x_curr[k_star - j_hi:k_star - j_lo + 1][::-1]
        else:
            grad = np.zeros(n)

        # Adaptive step: Polyak-like
        avg_b = b.mean()
        gap = M - avg_b
        if gap < 1e-12:
            break
        lr = 0.5 * gap / (np.linalg.norm(grad) ** 2 + 1e-15)
        lr = min(lr, 0.1)

        # Nesterov
        mom = it / (it + 2.0)
        x_nest = x_curr + mom * (x_curr - x_prev)
        x_nest = project_simplex(x_nest)

        x_next = x_nest - lr * grad
        x_next = np.maximum(x_next, 0)
        x_next = project_simplex(x_next)

        J_next = compute_J(x_next)
        if J_next < current_J - 1e-14:
            x_prev = x_curr.copy()
            x_curr = x_next
            current_J = J_next
            stuck_count = 0
            if current_J < best_J:
                best_J = current_J
                best_x = x_curr.copy()
        else:
            stuck_count += 1
            if stuck_count > 200:
                break

        if it % 100 == 0:
            tracker.update(x_curr)

    tracker.update(best_x)
    return best_x


def run_lp_iterations(a, max_iters, deadline, tracker):
    """Run LP mirror descent iterations with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = project_simplex(a)
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = project_simplex(g)

        # Adaptive step: try multiple step sizes
        base_t = max(0.05, 0.4 * (1.0 - it / float(2 * max_iters)))
        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.02, base_t * 0.5)]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = project_simplex((1.0 - t) * a + t * g)
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        else:
            a = best_a
            current_J = best_J

        if it % 25 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n) / n)
    inits.append(rng.uniform(0.3, 1.0, n))
    inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * 2 * t_arr))
    inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * 3 * t_arr))
    inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * 5 * t_arr))
    inits.append(1.0 + 0.5 * np.sin(2.0 * np.pi * 7 * t_arr))
    inits.append(np.exp(-4.0 * (t_arr - 0.5) ** 2) + 0.05)
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.1)

    # Paired structure: a[j] = a[n-1-j]
    half = n // 2
    pair = 1.0 + 0.3 * rng.normal(0, 1, half)
    pair = np.maximum(pair, 0.1)
    pd = np.zeros(n)
    pd[:half] = pair
    pd[half:] = pair[::-1] if n % 2 == 0 else np.concatenate([pair[::-1], [pair[-1]]])
    inits.append(pd)

    # Block structure
    nblocks = max(2, n // 32)
    block_sizes = [n // nblocks] * nblocks
    block_sizes[-1] += n - sum(block_sizes)
    bs = np.zeros(n)
    pos = 0
    for i, sz in enumerate(block_sizes):
        bs[pos:pos + sz] = 1.0 + 0.3 * rng.normal(0, 1)
        pos += sz
    inits.append(np.maximum(bs, 0.05))

    # Power-law
    inits.append((t_arr + 0.1) ** rng.uniform(-1.5, 1.5))

    # Sparse + smooth
    sparse = np.zeros(n)
    n_spikes = max(4, n // 16)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Random with structure (low-pass filtered noise)
    noise = rng.normal(0, 1, n)
    freq = np.fft.rfftfreq(n)
    cutoff = 0.3
    mask = freq < cutoff
    noise_filtered = np.fft.irfft(np.fft.rfft(noise) * mask)
    inits.append(1.0 + 0.3 * noise_filtered)

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Softmax subgradient with Nesterov (diverse inits, multi-length)
    Phase 2: LP mirror descent on best candidates
    Phase 3: Hard-max subgradient polish
    Phase 4: Diversification and final LP
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 15.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Softmax subgradient descent ===
    phase1_end = tracker.t_start + 0.55 * float(budget_s)
    print(f"\n{'='*60}", flush=True)
    print(f"Phase 1: Softmax subgradient (deadline: {(phase1_end - time.monotonic()):.0f}s)", flush=True)
    print(f"{'='*60}", flush=True)

    lengths = [64, 96, 128, 160, 192, 256]
    # Prioritize lengths near parent
    if parent is not None:
        pn = parent.size
        lengths.sort(key=lambda x: abs(x - pn))

    best_per_length = {}

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        t0 = time.monotonic()
        print(f"\n  --- Softmax n={n} ---", flush=True)

        inits = generate_initializations(n, rng, parent)
        n_inits = min(len(inits), 8)

        for idx in range(n_inits):
            if time.monotonic() > phase1_end:
                break
            a0 = np.maximum(inits[idx].astype(np.float64), 1e-6)
            a0 = project_simplex(a0)

            # Run softmax subgradient
            iters = max(200, min(3000, int(300000 // n)))
            a = softmax_subgradient_descent(a0, iters, phase1_end, tracker,
                                            beta_start=3.0, beta_end=80.0)

            # Track best per length
            J = compute_J(a)
            if n not in best_per_length or J < best_per_length[n][1]:
                best_per_length[n] = (a.copy(), J)
                print(f"    n={n} init={idx}: J={J:.8f}", flush=True)

        print(f"  --- n={n} done in {time.monotonic()-t0:.1f}s ---", flush=True)
        tracker.report(f"Phase1 n={n}")

    # === Phase 2: LP mirror descent on best candidates ===
    if time.monotonic() < phase1_end + 0.3 * float(budget_s):
        phase2_end = tracker.t_start + 0.80 * float(budget_s)
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 2: LP mirror descent (deadline: {(phase2_end - time.monotonic()):.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        # Sort lengths by their best J
        sorted_lengths = sorted(best_per_length.keys(), key=lambda x: best_per_length[x][1])

        for n in sorted_lengths[:4]:
            if time.monotonic() > phase2_end:
                break
            a_best = best_per_length[n][0].copy()
            t0 = time.monotonic()
            print(f"\n  --- LP n={n} (J={best_per_length[n][1]:.6f}) ---", flush=True)

            iters = max(50, min(300, int(60000 // max(n, 64))))
            a = run_lp_iterations(a_best, iters, phase2_end, tracker)

            if time.monotonic() < phase2_end:
                # Also try from the global best resampled to this length
                if tracker.best is not None and tracker.best.size != n:
                    a2 = resample_seq(tracker.best, n)
                    a2 = project_simplex(np.maximum(a2, 1e-6))
                    a = run_lp_iterations(a2, iters // 2, phase2_end, tracker)

            print(f"  --- LP n={n} done in {time.monotonic()-t0:.1f}s ---", flush=True)
            tracker.report(f"Phase2 n={n}")

    # === Phase 3: Hard-max subgradient polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        phase3_end = tracker.t_start + 0.92 * float(budget_s)
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 3: Hard-max subgradient polish (deadline: {(phase3_end - time.monotonic()):.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        n = tracker.best.size
        a = subgradient_hardmax_descent(tracker.best.copy(), 5000, phase3_end, tracker)
        print(f"  Polish done in {time.monotonic()-t0:.1f}s", flush=True)
        tracker.report("Phase3")

    # === Phase 4: Diversification and final LP ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*60}", flush=True)
        print(f"Phase 4: Diversification (deadline: {(deadline - t0):.0f}s)", flush=True)
        print(f"{'='*60}", flush=True)

        n = tracker.best.size
        temperature = 0.05

        for r in range(30):
            if time.monotonic() > deadline:
                break
            a = tracker.best.copy().astype(np.float64)
            a = a + rng.normal(0, temperature, n)
            a = project_simplex(np.maximum(a, 1e-6))

            # Quick softmax refinement
            a = softmax_subgradient_descent(a, 300, deadline, tracker,
                                            beta_start=20.0, beta_end=100.0)

            # If improved, try LP
            if compute_J(a) < tracker.best_score + 0.001:
                a = run_lp_iterations(a, 100, deadline, tracker)

            temperature *= 0.93

        # Final LP polish
        if time.monotonic() < deadline:
            a = run_lp_iterations(tracker.best.copy(), 200, deadline, tracker)

        tracker.report("Phase4")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*60}", flush=True)
    print(f"FINAL: J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)
    print(f"{'='*60}", flush=True)

    return [float(x) for x in tracker.best]
