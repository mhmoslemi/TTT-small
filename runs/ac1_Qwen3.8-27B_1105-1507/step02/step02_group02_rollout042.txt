**Key idea 3: Frank-Wolfe / Conditional Gradient**

The problem min max_k (a*a)_k s.t. a >= 0, sum(a) = 1 can be approached with Frank-Wolfe:
- At each step, solve the linear oracle: min <grad_f(a), g> s.t. g >= 0, sum(g) = 1, where f(a) = max_k (a*a)_k
- The gradient of f at a (when the max is attained at k*) is grad_k* = 2 * M_{k*} * a (where M_{k*} is the matrix for the k*-th autocorrelation)
- The linear oracle gives g = e_{argmin_i (grad_k*)_i} (a vertex of the simplex)
- Update: a ← (1-t)*a + t*g

This is equivalent to: at each step, put all mass on the coordinate that most reduces the current peak. This is very sparse and might be slow.

But we can use a smoothed version: instead of the max, use a log-sum-exp or softmax approximation, which gives a dense gradient and hence a dense Frank-Wolfe step.

Let me code up the "min t" approach:

```python
def find_best_g(a, t_max):
    """Find g >= 0, sum(g)=1 that minimizes max(a*g)"""
    n = len(a)
    m = 2*n - 1
    A = build_conv_matrix(a)  # A @ g = a * g (full conv)
    # min t s.t. A@g - t <= 0, sum(g) = 1, g >= 0
    # Variables: [g_0, ..., g_{n-1}, t]
    # c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1
    # Inequality: A@g - t <= 0 → [A, -1] @ [g, t] <= 0
    A_ub = np.hstack([A, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    # Equality: sum(g) = 1 → [1, ..., 1, 0] @ [g, t] = 1
    A_eq = np.hstack([np.ones(n), [0]])
    b_eq = [1]
    # Bounds: g_i >= 0, t free (or t >= 0)
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None
```

def active_gradient_lp(a, active_set):
    """Find direction d that reduces all active convolution values."""
    n = len(a)
    # Compute gradients for each active lag
    grads = []
    for k in active_set:
        grad = np.zeros(n)
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                grad[j] += 2 * a[idx]
        grads.append(grad)
    grads = np.array(grads)  # shape (|S|, n)
    
    m = len(active_set)
    # LP: min t s.t. grads @ d <= t, sum(d) = 0, -1 <= d_i <= 1
    # Variables: [d_0, ..., d_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1
    A_ub = np.hstack([grads, -np.ones((m, 1))])
    b_ub = np.zeros(m)
    A_eq = np.hstack([np.ones(n), [0]])
    b_eq = [0]
    bounds = [(-1, 1)] * n + [(-np.inf, np.inf)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return None

    for n in lengths:
        if time.monotonic() > lp_deadline:
            break
        remaining_to_lp = lp_deadline - time.monotonic()
        if n <= 512 and remaining_to_lp < 30:
            continue
        if n <= 768 and remaining_to_lp < 60:
            continue
        if n > 768 and remaining_to_lp < 120:
            continue

        print(f"\n  --- n={n} ---", flush=True)
        t0 = time.monotonic()

        inits = generate_inits(n, rng, best)
        # Limit inits based on time
        est_per_init = 30.0 if n <= 512 else 60.0
        max_inits = max(1, int(remaining_to_lp * 0.7 / est_per_init))
        max_inits = min(max_inits, len(inits))

        for i, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > lp_deadline:
                break
            a0 = np.maximum(a0, 1e-6)
            a0 = a0 / a0.sum()

            remaining = lp_deadline - time.monotonic()
            est_iter_time = 0.05 if n <= 512 else 0.10
            max_iters = max(50, int(remaining * 0.6 / len(inits[:max_inits][i:]) / est_iter_time))
            max_iters = min(max_iters, 3000)

            a_result, score = lp_iteration(a0, max_iters, lp_deadline)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"    NEW BEST: J={score:.6f} (n={n}, init={i})", flush=True)

        print(f"  n={n} done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === PHASE 2: Subgradient refinement ===
    sg_budget = total_budget * 0.15
    sg_deadline = min(time.monotonic() + sg_budget, deadline)

    if best is not None and time.monotonic() < sg_deadline:
        print(f"\n=== PHASE 2: Subgradient refinement ===", flush=True)
        t0 = time.monotonic()
        a_result, score = subgradient_refine(best, 100000, sg_deadline)
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  NEW BEST (sg): J={score:.6f}", flush=True)
        print(f"  Done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === PHASE 3: Smooth refinement ===
    smooth_budget = total_budget * 0.20
    smooth_deadline = min(time.monotonic() + smooth_budget, deadline)

    if best is not None and time.monotonic() < smooth_deadline:
        print(f"\n=== PHASE 3: Smooth refinement ===", flush=True)
        t0 = time.monotonic()
        a_result, score = smooth_refine(best, smooth_deadline)
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  NEW BEST (smooth): J={score:.6f}", flush=True)

        # Try different lengths
        for scale in [0.75, 1.5, 2.0]:
            if time.monotonic() > smooth_deadline:
                break
            n2 = int(best.size * scale)
            n2 = max(128, min(2048, n2))
            if n2 == best.size:
                continue
            a2 = resample_sequence(best, n2)
            a_result2, score2 = smooth_refine(a2, smooth_deadline)
            if score2 < best_score:
                best = a_result2.copy()
                best_score = score2
                print(f"  NEW BEST (smooth n={n2}): J={score2:.6f}", flush=True)

        print(f"  Done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === PHASE 4: Perturbation + LP re-optimization ===
    if best is not None and time.monotonic() < deadline:
        print(f"\n=== PHASE 4: Perturbation search ===", flush=True)
        t0 = time.monotonic()
        n = best.size
        remaining = deadline - time.monotonic()
        n_tries = max(3, int(remaining / 12))
        noise_scales = [0.03, 0.08, 0.05, 0.12, 0.02, 0.1]

        for trial in range(n_tries):
            if time.monotonic() > deadline:
                break
            scale = noise_scales[trial % len(noise_scales)]
            noise = rng.normal(0, scale, n)
            a_pert = best + noise
            a_pert = np.maximum(a_pert, 0)
            if a_pert.sum() < 0.01:
                continue
            a_pert = a_pert / a_pert.sum()

            remaining = deadline - time.monotonic()
            est_iter_time = 0.05 if n <= 512 else 0.1
            max_iters = max(30, int(remaining / (n_tries - trial) / est_iter_time))
            max_iters = min(max_iters, 1000)

            a_result, score = lp_iteration(a_pert, max_iters, deadline)
            if score < best_score:
                best = a_result.copy()
                best_score = score
                print(f"  NEW BEST (pert {trial}): J={score:.6f}", flush=True)

        print(f"  Done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # === PHASE 5: Final LP + smooth polish ===
    if best is not None and time.monotonic() < deadline:
        print(f"\n=== PHASE 5: Final polish ===", flush=True)
        t0 = time.monotonic()
        n = best.size
        remaining = deadline - time.monotonic()
        est_iter_time = 0.05 if n <= 512 else 0.1
        max_iters = max(100, int(remaining * 0.5 / est_iter_time))
        max_iters = min(max_iters, 5000)

        a_result, score = lp_iteration(best, max_iters, deadline)
        if score < best_score:
            best = a_result.copy()
            best_score = score

        if time.monotonic() < deadline:
            a_result, score = smooth_refine(best, deadline)
            if score < best_score:
                best = a_result.copy()
                best_score = score

        print(f"  Done in {time.monotonic()-t0:.1f}s, best={best_score:.6f}", flush=True)

    # Final validation
    if best is None:
        best = np.ones(256) / 256

    final_score = float(evaluate_sequence([float(x) for x in best]))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
