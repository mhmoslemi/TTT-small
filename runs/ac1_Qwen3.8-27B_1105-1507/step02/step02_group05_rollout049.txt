def soft_max_conv(a, T):
    """Compute smooth approximation of max(conv(a,a)) and its gradient contribution."""
    n = len(a)
    s = 1
    while s < 2*n-1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    b = np.fft.irfft(fa*fa, s)[:2*n-1]
    # Soft max
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    S = exp_Tb.sum()
    p = exp_Tb / S
    F = (Tb_max + np.log(S)) / T  # This is the soft max value
    return p, F, b

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Soft-max gradient descent on the probability simplex (new approach)
2. Nesterov-accelerated gradient with adaptive T annealing
3. LP mirror descent for polishing
4. L-BFGS-B refinement
5. Multi-scale search across sequence lengths
Key innovation: direct gradient optimization of the smooth max-convolution objective
with projective gradient descent onto the probability simplex.

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix, csr_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a, n):
    """Build sparse constraint matrix for LP."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            rows.extend([k] * (jh - jl + 1))
            cols.extend(range(jl, jh + 1))
            vals.extend((a[k - np.arange(jh - jl + 1)][::-1]).tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a, n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if r.success and r.x is not None:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def soft_max_and_grad(a, T):
    """
    Compute soft-max value F and gradient of (2n/T)*log(sum exp(T*b)) 
    for the scale-invariant objective (assuming sum(a)=1).
    Returns: (F, grad) where grad_j = 4n*(c_j - F*a_j)
    """
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    S = exp_Tb.sum()
    p = exp_Tb / S
    F = (Tb_max + np.log(S)) / T

    # Compute c_j = sum_k p_k * a_{k-j} via FFT cross-correlation
    s2 = 1
    while s2 < 3 * n:
        s2 <<= 1
    fp = np.fft.rfft(p, s2)
    fa = np.fft.rfft(a, s2)
    c_full = np.fft.irfft(fp * np.conj(fa), s2)
    c = c_full[:n]

    grad = 4.0 * n * (c - F * a)
    return F, grad


def softmax_gd(a, max_iters, deadline, tracker, T_init_mult=20, T_final_mult=300):
    """
    Soft-max gradient descent with Nesterov momentum on the probability simplex.
    a should have sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s

    T = T_init_mult * n
    momentum = 0.9
    lr = 0.3
    v = np.zeros(n)
    current_J = compute_J(a)
    tracker.update(a)

    patience = 0
    best_local_J = current_J
    best_local_a = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Anneal T
        progress = min(1.0, it / max_iters)
        T = T_init_mult * n + (T_final_mult * n - T_init_mult * n) * (progress ** 0.7)

        F, grad = soft_max_and_grad(a, T)

        # Nesterov step
        v = momentum * v + grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        J_new = compute_J(a_new)

        if J_new < current_J - 1e-12:
            a = a_new
            current_J = J_new
            patience = 0
            if current_J < best_local_J - 1e-10:
                best_local_J = current_J
                best_local_a = a.copy()
            # Slowly decay lr
            lr = max(0.01, lr * 0.999)
        else:
            patience += 1
            lr = max(0.005, lr * 0.95)
            momentum = min(0.95, momentum + 0.001)
            if patience > 200:
                # Reset momentum and try from best
                if best_local_J < current_J:
                    a = best_local_a.copy()
                    current_J = best_local_J
                v = np.zeros(n)
                patience = 0
                lr = 0.2

        if it % 50 == 0:
            tracker.update(a)

    tracker.update(a)
    if best_local_J < current_J:
        tracker.update(best_local_a)
        return best_local_a
    return a


def softmax_gd_refined(a, max_iters, deadline, tracker):
    """

        else:

def resample_seq(src, dest_n):
    """Resample 1D sequence to new length."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def run_lp_iterations(a, max_iters, deadline, tracker):
    """Run LP mirror descent iterations."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step
        best_a = None
        best_J = current_J
        for t in [0.3, 0.5, 0.7, 0.2]:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        a = best_a
        current_J = best_J

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

def generate_inits(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []
    t_arr = np.linspace(0, 1, n, endpoint=False)

    if parent is not None:
        inits.append(resample_seq(parent, n))

    # Uniform
    inits.append(np.ones(n))

    # Linear ramp
    inits.append(np.linspace(1.0, 0.1, n))
    inits.append(np.linspace(0.1, 1.0, n))

    # Cosine shapes
    for freq in [1, 2, 3, 4]:
        inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t_arr))
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t_arr + rng.uniform(0, 2 * np.pi)))

    # Quadratic
    inits.append(1.0 + 0.5 * (2.0 * t_arr - 1.0) ** 2)
    inits.append(1.0 - 0.5 * (2.0 * t_arr - 1.0) ** 2)

    # Gaussian
    inits.append(np.exp(-5.0 * (t_arr - 0.5) ** 2) + 0.1)

    # Random
    inits.append(rng.uniform(0.2, 1.0, n))

    # Sparse spikes
    n_spikes = max(3, n // 16)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse = np.zeros(n)
    sparse[positions] = 1.0
    inits.append(sparse + 0.05)

    # Triangular
    inits.append(1.0 - np.abs(2.0 * t_arr - 1.0))

    # Sine power
    inits.append(np.abs(np.sin(np.pi * t_arr)) ** 2 + 0.1)

    # Power law
    inits.append((t_arr + 0.05) ** rng.uniform(-1.5, 1.5))

    return inits

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 1: Soft-max GD ===
    lengths = [32, 48, 64, 96, 128, 192, 256, 384, 512]
    # Prioritize lengths near parent and smaller lengths
    if parent is not None:
        pn = parent.size
        lengths.sort(key=lambda x: abs(x - pn) + 0.1 * x)

    all_candidates = []

    for n in lengths:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase1 softmax-GD n={n} {'='*50}", flush=True)

        inits = generate_inits(n, rng, parent)
        # Limit inits for larger n
        max_inits = min(len(inits), max(3, 12 - n // 128))

        for idx, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.0)
            s0 = a0.sum()
            if s0 < 1e-10:
                continue
            a0 = a0 / s0

            # Determine iteration budget based on time remaining and n
            time_remaining = deadline - time.monotonic()
            iters = max(50, min(500, int(30000 // n)))

            a = softmax_gd(a0, iters, deadline, tracker)
            all_candidates.append(a.copy())

            # Also try the refined version if time permits
            if time.monotonic() < deadline and time_remaining > 60:
                a2 = softmax_gd_refined(a, min(iters, 200), deadline, tracker)
                all_candidates.append(a2.copy())

        elapsed = time.monotonic() - t0
        tracker.report(f"Phase1 n={n}: {elapsed:.1f}s")

    # Sort candidates by score
    candidates_scores = []
    for a in all_candidates:
        sc = compute_J(a)
        candidates_scores.append((sc, a))
    candidates_scores.sort(key=lambda x: x[0])

    # Keep top candidates for LP polish
    top_candidates = candidates_scores[:min(8, len(candidates_scores))]

    # === Phase 2: LP Mirror Descent on top candidates ===
    if time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase2 LP polish {'='*50}", flush=True)

        for sc, a in top_candidates:
            if time.monotonic() > deadline:
                break
            time_remaining = deadline - time.monotonic()
            n = a.size
            iters = max(20, min(200, int(60000 // n)))
            a = run_lp_iterations(a, iters, deadline, tracker)
            all_candidates.append(a.copy())

        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: L-BFGS-B ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase3 L-BFGS-B n={n} {'='*50}", flush=True)

        def lbb_obj(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-8:
                return 1e10
            ax = ax / s
            b = fast_conv(ax)
            return float(b.max())

        def lbb_grad(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-8:
                return np.zeros_like(x)
            ax = ax / s
            # Use soft-max gradient with moderate T
            T = 50.0 * n
            F, grad = soft_max_and_grad(ax, T)
            # Project gradient to simplex
            grad_proj = grad - np.mean(grad)
            return grad_proj

        bounds_list = [(0.0, None)] * n
        try:
            res = minimize(lbb_obj, tracker.best.astype(np.float64),
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-18, 'gtol': 1e-12})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)

        tracker.report(f"Phase3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Final LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        n = tracker.best.size
        if n <= 512:
            t0 = time.monotonic()
            print(f"\n{'='*50} Phase4 Final LP n={n} {'='*50}", flush=True)
            a = run_lp_iterations(tracker.best, 300, deadline, tracker)
            tracker.report(f"Phase4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Try alternating GD + LP cycles ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Phase5 Alternating cycles n={n} {'='*50}", flush=True)

        a = tracker.best.copy()
        for cycle in range(10):
            if time.monotonic() > deadline:
                break
            # GD step
            a = softmax_gd(a, 50, deadline, tracker)
            # LP step
            a = run_lp_iterations(a, 30, deadline, tracker)

        tracker.report(f"Phase5: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
