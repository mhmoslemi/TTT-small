coordinate descent refinement, adaptive soft-max, and spectral initialization.
Key improvements over previous: multiplicative updates, coordinate descent,

import numpy as np
import time


def fast_score(f):
    """Compute the objective quickly."""
    n = len(f)
    S = float(np.sum(f))
    if S < 0.01:
        return np.inf
    C = np.convolve(f, f)
    return 2.0 * n * float(C.max()) / (S * S)


def soft_max_gradient(f, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*C_k)) where C=conv(f,f)."""
    n = len(f)
    C = np.convolve(f, f)
    max_C = float(C.max())
    shifted = beta * (C - max_C)
    shifted = np.clip(shifted, -700, 700)
    exp_C = np.exp(shifted)
    total = float(exp_C.sum())
    if total < 1e-300:
        return np.zeros(n), max_C
    p = exp_C / total

    # grad[l] = sum_k p[k] * dC_k/d f_l
    # dC_k/d f_l = 2*f_{k-l} (if 0<=k-l<n and k!=2l) + f_l*(if k==2l)
    # Simplified: grad[l] = 2*sum_j p[j+l]*f[j] - p[2l]*f[l] (correction for diagonal)
    # Cross-correlation via FFT
    N_fft = 1
    target = 3 * n
    while N_fft < target:
        N_fft <<= 1
    P_fft = np.fft.rfft(p, N_fft)
    F_fft = np.fft.rfft(f, N_fft)
    cross = np.fft.irfft(np.conj(F_fft) * P_fft, N_fft)

    grad = 2.0 * cross[:n].copy()
    # Diagonal correction: at k=2l, derivative is f_l not 2*f_l
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < (2 * n - 1)
    grad[valid] -= p[diag_idx[valid]] * f[valid]

    soft_val = max_C + np.log(total) / beta
    return grad, soft_val


def coordinate_descent(f, n_passes=1, n_inner=30):
    """Optimize each coordinate via ternary search. Returns improved f."""
    n = len(f)
    f = f.copy()
    S = float(np.sum(f))

    for _ in range(n_passes):
        for j in range(n):
            old_val = f[j]
            # Determine search range
            lo = 0.0
            hi = max(old_val * 3.0 + 0.1 * S / n, 0.5 * S / n)

            # Quick evaluation
            def eval_j(val):
                f[j] = val
                s2 = S - old_val + val
                if s2 < 0.01:
                    return 1e10
                C = np.convolve(f, f)
                return float(C.max()) / (s2 * s2)

            # Ternary search
            for _ in range(n_inner):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                s1 = eval_j(m1)
                s2 = eval_j(m2)
                if s1 < s2:
                    hi = m2
                else:
                    lo = m1

            best_val = (lo + hi) / 2.0
            f[j] = best_val

        # Renormalize
        S_new = float(np.sum(f))
        if S_new > 1e-10:
            f = f / S_new

    return f

def spectral_init(n, beta_flat=10.0):
    """Initialize by trying to flatten the power spectrum."""
    # Start with a raised cosine
    idx = np.arange(n, dtype=np.float64)
    f = 0.5 * (1 + np.cos(2 * np.pi * idx / (n - 1)))
    f = f + 0.1  # Ensure positive

    # Try to flatten |F|^2 by iteratively adjusting
    for _ in range(50):
        F = np.fft.rfft(f, 2 * n)
        power = np.abs(F) ** 2
        # Target: flat power spectrum
        mean_power = np.mean(power)
        # Adjust: f_new = ifft(sqrt(target_power) * phase(F))
        target_amp = np.sqrt(np.clip(power, 0, None))
        # Blended: move amplitude toward flat
        new_amp = target_amp + beta_flat * (mean_power ** 0.5 - target_amp) / max(1, np.max(target_amp))
        new_amp = np.clip(new_amp, 0, None)
        phase = F / np.where(np.abs(F) > 1e-10, np.abs(F), 1.0)
        F_new = new_amp * phase
        f = np.fft.irfft(F_new, 2 * n)[:n]
        f = np.clip(f, 0, None)
        S = float(np.sum(f))
        if S < 1e-10:
            break
        f = f / S

    return f

    inits.append(0.54 - 0.46 * np.cos(2 * np.pi * idx / (n - 1)))

def perturb_structured(f, rng, mode=0):

    elif mode == 1:

        new_f = np.roll(f, shift)
        # Taper edges to handle wrap-around
        taper = np.minimum(np.arange(n), np.arange(n - 1, -1, -1)) + 1
        taper = taper / taper.max()
        new_f = new_f * taper
    elif mode == 2:
        # Amplitude scaling of center vs edges

        scale = 1.0 + rng.uniform(-0.2, 0.2) * np.exp(-((idx - center) / (n * 0.3)) ** 2)
        new_f = f * scale
    else:
        # Random multiplicative noise
        noise = 1.0 + 0.05 * rng.standard_normal(n)
        new_f = f * np.clip(noise, 0.5, 1.5)

    new_f = np.clip(new_f, 0.0, 1000.0)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S

    time_fracs = {512: 0.28, 384: 0.15, 640: 0.15, 768: 0.12, 448: 0.10, 1024: 0.10}
    total_budget = deadline - t0

    for n in lengths:
        if time.monotonic() > deadline - 60:
            print(f"Time running out (elapsed {time.monotonic()-t0:.0f}s)")
            break

        n_budget = total_budget * time_fracs.get(n, 0.08)
        n_deadline = time.monotonic() + n_budget
        print(f"\n=== n={n} (budget {n_budget:.0f}s) ===")

        # Build initializations
        inits = []
        labels = []

        # Parent interpolation if available
        if best_sequence is not None:
            if len(best_sequence) == n:
                inits.append(best_sequence.copy())
                labels.append('parent')
            else:
                x_old = np.linspace(0, 1, len(best_sequence))
                x_new = np.linspace(0, 1, n)
                interp = np.interp(x_new, x_old, best_sequence)
                interp = np.clip(interp, 0, None)
                inits.append(interp)
                labels.append('parent_interp')

        # Diverse inits
        div_inits = diverse_init(n, seed=n)
        for i, d in enumerate(div_inits):
            inits.append(np.clip(d, 0, 1000.0))
            labels.append(f'div{i}')

        # Filter valid inits
        valid_inits = []
        valid_labels = []
        for lab, f_init in zip(labels, inits):
            f_init = np.clip(f_init, 0.0, 1000.0)
            S = float(np.sum(f_init))
            if S > 0.01:
                f_init = f_init / S
                sc = fast_score(f_init)
                if np.isfinite(sc) and sc < 5.0:  # Only useful inits
                    valid_inits.append(f_init)
                    valid_labels.append(lab)
                    if sc < best_score:
                        best_score = sc
                        best_sequence = f_init.copy()

        print(f"  Valid inits: {len(valid_inits)} / {len(inits)}, "
              f"best init score: {min(fast_score(fi) for fi in valid_inits):.6f}")

        # Time per init
        n_valid = len(valid_inits)
        per_init_time = n_budget / max(n_valid, 1) * 0.75

        for idx_init, (lab, f_init) in enumerate(zip(valid_labels, valid_inits)):
            if time.monotonic() > n_deadline - 10:
                break

            init_time = min(per_init_time, n_deadline - time.monotonic() - 5)
            if init_time < 10:
                continue

            # Strategy 1: Multiplicative gradient descent
            f1_deadline = time.monotonic() + init_time * 0.5
            try:
                bf1, bs1 = multiplicative_gradient_descent(
                    f_init, f1_deadline,
                    beta_start=3.0, beta_end=400.0,
                    lr_start=0.4, max_iters=3000, cd_every=40,
                    verbose_prefix=f"[n={n} {lab} MGD] ")
                if bs1 < best_score:
                    best_score = bs1
                    best_sequence = bf1.copy()
                    best_f = bf1
                else:
                    best_f = f_init
            except Exception as e:
                print(f"  MGD error: {e}")
                best_f = f_init

            # Strategy 2: Standard GD with Adam on the best from strategy 1
            if time.monotonic() < n_deadline - 10:
                f2_deadline = time.monotonic() + (n_deadline - time.monotonic()) * 0.5
                try:
                    bf2, bs2 = standard_gradient_descent(
                        best_f, f2_deadline,
                        beta_start=5.0, beta_end=300.0,
                        lr_start=0.04, max_iters=3000,
                        verbose_prefix=f"[n={n} {lab} GD] ")
                    if bs2 < best_score:
                        best_score = bs2
                        best_sequence = bf2.copy()
                        best_f = bf2
                except Exception as e:
                    print(f"  GD error: {e}")

            # Strategy 3: Coordinate descent polish
            if time.monotonic() < n_deadline - 5 and best_f is not None:
                cd_deadline = time.monotonic() + min(20, n_deadline - time.monotonic() - 2)
                try:
                    passes = 0
                    while time.monotonic() < cd_deadline:
                        best_f = coordinate_descent(best_f, n_passes=1, n_inner=25)
                        sc_cd = fast_score(best_f)
                        if sc_cd < best_score:
                            best_score = sc_cd
                            best_sequence = best_f.copy()
                        passes += 1
                    if passes > 0:
                        print(f"  [n={n} {lab}] CD {passes} passes: {best_score:.6f}")
                except Exception as e:
                    print(f"  CD error: {e}")

            if idx_init % 3 == 0:
                print(f"  [n={n} {lab}] current best: {best_score:.6f}")

    # === Phase 2: Focused refinement on the best sequence ===
    if best_sequence is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Focused refinement (n={len(best_sequence)}, score={best_score:.6f}) ===")
        n_best = len(best_sequence)
        ref_budget = deadline - time.monotonic() - 20

        # Multiple refinement rounds with different strategies
        rounds = 4
        per_round = ref_budget / rounds

        current_f = best_sequence.copy()
        for r in range(rounds):
            if time.monotonic() > deadline - 20:
                break

            round_deadline = time.monotonic() + per_round

            # Strategy A: Log-space GD with high beta
            if time.monotonic() < round_deadline - 5:
                sub_deadline = min(round_deadline, time.monotonic() + per_round * 0.4)
                try:
                    bf, bs = multiplicative_gradient_descent(
                        current_f, sub_deadline,
                        beta_start=50.0, beta_end=800.0,
                        lr_start=0.3, max_iters=2000, cd_every=30,
                        verbose_prefix=f"[refine {r} A] ")
                    if bs < best_score:
                        best_score = bs
                        best_sequence = bf.copy()
                        current_f = bf
                except Exception as e:
                    print(f"  Refine A error: {e}")

            # Strategy B: Coordinate descent
            if time.monotonic() < round_deadline - 3:
                cd_deadline = min(round_deadline, time.monotonic() + per_round * 0.3)
                try:
                    passes = 0
                    while time.monotonic() < cd_deadline:
                        current_f = coordinate_descent(current_f, n_passes=1, n_inner=30)
                        sc = fast_score(current_f)
                        if sc < best_score:
                            best_score = sc
                            best_sequence = current_f.copy()
                        passes += 1
                    if passes > 0:
                        print(f"  Refine {r} B: CD {passes} passes, score={best_score:.6f}")
                except Exception as e:
                    print(f"  Refine B error: {e}")

            # Strategy C: Structured perturbation + quick optimize
            if time.monotonic() < round_deadline - 3:
                mode = r % 4
                f_pert = perturb_structured(current_f, rng, mode=mode)
                pert_deadline = min(round_deadline, time.monotonic() + per_round * 0.3)
                try:
                    bf, bs = multiplicative_gradient_descent(
                        f_pert, pert_deadline,
                        beta_start=20.0, beta_end=200.0,
                        lr_start=0.3, max_iters=1000, cd_every=50,
                        verbose_prefix=f"[refine {r} C] ")
                    if bs < best_score:
                        best_score = bs
                        best_sequence = bf.copy()
                        current_f = bf
                        print(f"  Refine {r} C (pert mode {mode}): {bs:.6f}")
                except Exception as e:
                    print(f"  Refine C error: {e}")

        # Final high-precision coordinate descent
        if time.monotonic() < deadline - 10:
            print(f"\n=== Final CD polish ===")
            final_deadline = deadline - 5
            passes = 0
            try:
                while time.monotonic() < final_deadline:
                    current_f = coordinate_descent(current_f, n_passes=1, n_inner=40)
                    sc = fast_score(current_f)
                    if sc < best_score:
                        best_score = sc
                        best_sequence = current_f.copy()
                    passes += 1
                print(f"  Final CD: {passes} passes, score={best_score:.6f}")
            except Exception as e:
                print(f"  Final CD error: {e}")

    # Safety: verify final answer
    if best_sequence is None:
        best_sequence = np.ones(512)
        best_score = fast_score(best_sequence)

    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = fast_score(best_sequence)

    # Final verification with the actual evaluator
    final_check = evaluate_sequence([float(x) for x in best_sequence])
    if not np.isfinite(final_check):
        best_sequence = np.ones(512)
        final_check = evaluate_sequence([float(x) for x in best_sequence])

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={best_score:.6f}, "
          f"eval={final_check:.6f}, time={elapsed:.1f}s")

    return [float(x) for x in best_sequence]
