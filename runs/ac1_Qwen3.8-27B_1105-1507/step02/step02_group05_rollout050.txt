def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

def smoothed_objective_and_grad(a, T):
    """
    Compute f_T(a) = (1/T) * log(sum_k exp(T * (a*a)_k)) and its gradient.
    f_T approximates max_k (a*a)_k as T -> inf.
    """
    n = len(a)
    b = fast_conv(a)  # length 2n-1
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    sum_exp = exp_Tb.sum()
    weights = exp_Tb / sum_exp  # length 2n-1
    
    f_smooth = (Tb_max + np.log(sum_exp)) / T
    
    # Gradient via FFT:
    # grad[i] = 2 * sum_{j=0}^{n-1} weights[i+j] * a_j
    # = 2 * (W_rev * a)[2n-2-i] where W_rev[k] = weights[2n-2-k]
    W_rev = weights[::-1]  # length 2n-1
    # Compute convolution W_rev * a via FFT
    conv_len = (2 * n - 1) + n - 1  # = 3n - 2
    fft_size = 1
    while fft_size < conv_len:
        fft_size <<= 1
    W_r_fft = np.fft.rfft(W_rev, fft_size)
    a_fft = np.fft.rfft(a, fft_size)
    conv_result = np.fft.irfft(W_r_fft * a_fft, fft_size)[:conv_len]
    
    # grad[i] = 2 * conv_result[2n-2-i]
    grad = 2.0 * conv_result[2 * n - 2::-1]
    
    return f_smooth, grad

def gradient_descent_step(a, lr, T):
    """One step of projected gradient descent on f_T(a)."""
    f, grad = smoothed_objective_and_grad(a, T)
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a, f
    a_new /= s
    return a_new, f

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

        g = np.maximum(r.x, 0.0)

def smoothed_grad_fft(a, T):
    """
    Compute gradient of f_T(a) = (1/T)*log(sum_k exp(T*(a*a)_k)).
    Returns (f_T, grad) where grad[i] = 2*sum_k w_k * a[k-i] (valid range).
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    b = fast_conv(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    sum_exp = exp_Tb.sum()
    weights = exp_Tb / sum_exp

    f_smooth = (Tb_max + np.log(sum_exp)) / T

    # grad[i] = 2 * sum_{j=0}^{n-1} weights[i+j] * a_j
    # = 2 * (W_rev * a)[2n-2-i]  where W_rev[k] = weights[2n-2-k]
    W_rev = weights[::-1]
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1
    while fft_size < conv_len:
        fft_size <<= 1
    W_r_fft = np.fft.rfft(W_rev, fft_size)
    a_fft = np.fft.rfft(a, fft_size)
    conv_result = np.fft.irfft(W_r_fft * a_fft, fft_size)[:conv_len]

    idx = np.arange(2 * n - 2, -1, -1)
    grad = 2.0 * conv_result[idx]

    return f_smooth, grad


def focused_grad(a, K=20, T=50.0):
    """Gradient using only top-K lags of the convolution (faster for large n)."""
    n = len(a)
    b = fast_conv(a)
    top_k = np.argpartition(b, -K)[-K:]
    b_top = b[top_k]
    T_btop = T * b_top
    T_max = T_btop.max()
    exp_vals = np.exp(T_btop - T_max)
    weights = exp_vals / exp_vals.sum()

    grad = np.zeros(n)
    for idx in range(K):
        k = int(top_k[idx])
        w = weights[idx]
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += 2.0 * w * a[k - i_hi:k - i_lo + 1][::-1]
    return b.max(), grad

            if no_improve % 50 == 0:
                lr = max(0.05, lr * 0.7)

        if it % 200 == 0:
            tracker.update(a)

    tracker.update(a)
    return best_a if best_J < current_J else a


def run_lp_iterations(a, max_iters, deadline, tracker):
    """LP mirror descent with adaptive step sizes."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        base_t = max(0.05, 0.4 * (1.0 - it / float(2 * max_iters)))
        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.02, base_t * 0.5)]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            t_small = 0.02
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                break
        else:
            a = best_a
            current_J = best_J

        if it % 15 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def generate_diverse_inits(n, rng, parent):
    """Generate diverse initial sequences."""
    inits = []
    if parent is not None:
        inits.append(resample_seq(parent, n))

    inits.append(np.ones(n))
    inits.append(rng.uniform(0.3, 1.0, n))

    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [1, 2, 3, 4, 5, 7]:
        inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t))
    inits.append(np.exp(-5.0 * (t - 0.5) ** 2) + 0.05)
    inits.append(np.abs(np.sin(np.pi * t)) + 0.1)
    inits.append(1.0 + 0.2 * rng.standard_normal(n))

    # Block structure
    block = max(4, n // 32)
    n_blocks = n // block
    block_vals = rng.uniform(0.5, 1.5, n_blocks)
    sparse_init = np.repeat(block_vals, block)[:n]
    inits.append(sparse_init)

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-scale gradient + LP hybrid optimization.
    Key: FFT-based gradient for speed, progressive length increase, SA restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 25.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent: n={parent.size} J={tracker.best_score:.6f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128)
        tracker.update(parent)

    # === Phase 1: Multi-scale progressive optimization ===
    lengths = [parent.size]
    n = parent.size
    while n * 2 <= 8192:
        n *= 2
        lengths.append(n)

    for n in lengths:
        if time.monotonic() > deadline:
            break

        t0 = time.monotonic()
        print(f"\n{'='*50} Scale n={n} {'='*50}", flush=True)

        best_at_n = None
        best_J_n = float('inf')

        # Try multiple initializations
        n_inits = min(4, max(2, 10000 // n))
        inits = generate_diverse_inits(n, rng, parent)
        inits = inits[:n_inits]

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break

            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 /= a0.sum()

            # Gradient descent
            gd_iters = max(500, min(5000, 3000000 // n))
            mode = "fft" if n <= 2048 else "focused"
            a = run_gradient_descent(a0, deadline, tracker, max_iters=gd_iters, mode=mode)

            # LP refinement (only for smaller n)
            if n <= 1024 and time.monotonic() < deadline:
                remaining = deadline - time.monotonic()
                lp_iters = min(50, max(5, int(remaining / max(1.0, n / 500))))
                a = run_lp_iterations(a, lp_iters, deadline, tracker)

            J_a = compute_J(a)
            if J_a < best_J_n:
                best_J_n = J_a
                best_at_n = a.copy()

            if tracker.best is not None and tracker.best.size == n:
                if compute_J(tracker.best) < best_J_n:
                    best_J_n = compute_J(tracker.best)
                    best_at_n = tracker.best.copy()

        if best_at_n is not None:
            parent = best_at_n
            print(f"  Best at n={n}: J={best_J_n:.6f} ({time.monotonic()-t0:.1f}s)", flush=True)

    # === Phase 2: Intensive SA + gradient on best ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} SA Phase n={n} {'='*50}", flush=True)

        temps = [0.15, 0.1, 0.08, 0.05, 0.03, 0.02, 0.01, 0.005]
        restarts_per_temp = max(3, min(15, int((deadline - time.monotonic()) / max(1, n / 200))))

        for temp in temps:
            for r in range(restarts_per_temp):
                if time.monotonic() > deadline:
                    break

                a = tracker.best.copy().astype(np.float64)
                a += rng.normal(0, temp, n)
                a = np.maximum(a, 0.001)
                a /= a.sum()

                # Gradient refinement
                mode = "fft" if n <= 2048 else "focused"
                gd_iters = max(200, min(2000, 1000000 // n))
                a = run_gradient_descent(a, deadline, tracker, max_iters=gd_iters, mode=mode)

                # Short LP if feasible
                if n <= 1024 and time.monotonic() < deadline:
                    remaining = deadline - time.monotonic()
                    if remaining > 5:
                        lp_iters = min(10, max(2, int(remaining / max(1.0, n / 300))))
                        a = run_lp_iterations(a, lp_iters, deadline, tracker)

            print(f"  temp={temp:.3f} done, best={tracker.best_score:.6f}", flush=True)

        tracker.report(f"SA: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Try slightly different lengths around best ===
    if tracker.best is not None and time.monotonic() < deadline:
        n_best = tracker.best.size
        t0 = time.monotonic()
        print(f"\n{'='*50} Length exploration {'='*50}", flush=True)

        candidate_lengths = [n_best // 2, n_best, n_best * 2]
        candidate_lengths = [cl for cl in candidate_lengths if 64 <= cl <= 4096]

        for n_try in candidate_lengths:
            if time.monotonic() > deadline:
                break
            if n_try == n_best:
                continue

            a = resample_seq(tracker.best, n_try)
            a = np.maximum(a, 0.001)
            a /= a.sum()

            mode = "fft" if n_try <= 2048 else "focused"
            gd_iters = max(300, min(3000, 2000000 // n_try))
            a = run_gradient_descent(a, deadline, tracker, max_iters=gd_iters, mode=mode)

            if n_try <= 1024 and time.monotonic() < deadline:
                remaining = deadline - time.monotonic()
                if remaining > 5:
                    lp_iters = min(30, max(5, int(remaining / max(1.0, n_try / 300))))
                    a = run_lp_iterations(a, lp_iters, deadline, tracker)

        tracker.report(f"Length explore: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Final LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        n = tracker.best.size
        remaining = deadline - time.monotonic()
        if n <= 2048 and remaining > 10:
            t0 = time.monotonic()
            print(f"\n{'='*50} Final LP n={n} ({remaining:.0f}s left) {'='*50}", flush=True)
            lp_iters = min(200, max(10, int(remaining / max(1.0, n / 500))))
            a = run_lp_iterations(tracker.best, lp_iters, deadline, tracker)
            tracker.report(f"Final LP: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(256)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL RESULT {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} "
          f"time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
