def simulated_annealing_lp(a, max_time, T_init=0.1, T_min=0.001, cooling=0.995):
    """Simulated annealing with LP-guided moves."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    T = T_init
    it = 0
    
    while time.monotonic() - start < max_time and T > T_min:
        # Generate move
        move_type = np.random.choice(3)
        if move_type == 0:
            # LP-guided move
            b, M = solve_lp_direction(a)
            if b is None:
                T *= cooling
                it += 1
                continue
            a_new = a + 0.3 * (b - a)
        elif move_type == 1:
            # Random block perturbation
            width = max(1, int(n * np.random.uniform(0.02, 0.15)))
            pos = int(np.random.randint(0, n - width + 1))
            a_new = a.copy()

        else:

            idx = int(np.random.randint(0, n))
            a_new = a.copy()
            a_new[idx] = max(0, a_new[idx] * np.exp(np.random.normal(0, 0.3)))
        
        a_new = project_nonneg_sum1(a_new)
        obj_new = compute_obj(a_new)
        
        # Metropolis acceptance
        delta = obj_new - compute_obj(a)

            a = a_new
            if obj_new < best_obj:
                best_a = a_new.copy()
                best_obj = obj_new

        it += 1
    
    return best_a, best_obj

Multi-strategy optimization for step function coefficients.
Combines: spectral initialization, LP alternating, adaptive subgradient,
simulated annealing, and block refinement.
"""

import numpy as np
from scipy.optimize import linprog
import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def project_nonneg_sum1(a):
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def build_A_for_lp(a):
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end+1] = a[k - j_end:k - j_start+1][::-1]
    return A

def solve_lp(a):
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A_for_lp(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        b = res.x
        s = float(np.sum(b))
        if s > 1e-10:
            b = b / s
        return b, M
    return None, M

        C_t = np.zeros(2*n)
        C_t[:2*n-1] = C_target
        target_mag = np.sqrt(np.maximum(np.real(C_t), 0))
        # Preserve phase of A_fft, set magnitude to target
        if np.any(np.abs(A_fft) > 1e-10):
            phase = A_fft / np.abs(A_fft)
        else:
            phase = np.ones(2*n, dtype=complex)
        A_new = target_mag * phase
        a = np.real(np.fft.ifft(A_new))[:n]
        a = np.maximum(a, 1e-6)
        a = a / np.sum(a)
    
    return a

def lp_refine(a, max_time, max_iters=100):
    """LP alternating refinement."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break
        b, M = solve_lp(a)
        if b is None:
            break
        
        improved = False
        for t in [0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            a_new = (1-t)*a + t*b
            a_new = project_nonneg_sum1(a_new)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_a = a_new.copy()
                best_obj = obj
                a = a_new
                improved = True
                break
        
        if not improved:
            stall += 1
            if stall > 10:
                break
        else:
            stall = 0
    
    return best_a, best_obj

def adaptive_subgradient(a, max_time, lr_init=0.005, momentum=0.9):
    """Adaptive subgradient descent with momentum on max autocorrelation."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    lr = lr_init
    v = np.zeros(n)
    stall = 0
    it = 0
    
    while time.monotonic() - start < max_time:
        conv = np.convolve(a, a)
        max_val = float(np.max(conv))
        
        # Find all peaks within 5% of max
        tol = 0.05 * max_val
        peaks = np.where(conv >= max_val - tol)[0]
        
        # Compute subgradient (average over peaks)
        grad = np.zeros(n)
        for k in peaks:
            valid_start = max(0, k - n + 1)
            valid_end = min(n - 1, k)
            for i in range(valid_start, valid_end + 1):
                grad[i] += 2.0 * a[k - i]
        grad /= len(peaks)
        
        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = project_nonneg_sum1(a_new)
        
        obj = compute_obj(a_new)
        if obj < best_obj:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            stall = 0
        else:
            # Backtrack
            found = False
            for factor in [0.5, 0.25, 0.1]:
                a_try = a + factor * v
                a_try = project_nonneg_sum1(a_try)
                obj_try = compute_obj(a_try)
                if obj_try < best_obj:
                    best_a = a_try.copy()
                    best_obj = obj_try
                    a = a_try
                    stall = 0
                    found = True
                    break
            if not found:
                stall += 1
                lr *= 0.85
                v *= 0.5
                if stall > 200:
                    v = np.zeros(n)
                    lr = lr_init * 0.3
                    stall = 0
        
        it += 1
    
    return best_a, best_obj

def simulated_annealing(a, max_time, rng, T_init=0.01, T_min=0.0001, cooling=0.998):
    """Simulated annealing with diverse moves."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    T = T_init
    current_obj = best_obj
    
    while time.monotonic() - start < max_time and T > T_min:
        move_type = rng.random()
        
        if move_type < 0.4:
            # Block multiplicative perturbation
            width = max(1, int(n * rng.uniform(0.02, 0.12)))
            pos = int(rng.integers(0, n - width + 1))
            a_new = a.copy()
            a_new[pos:pos+width] *= np.exp(rng.normal(0, 0.05 + 0.1*T, width))
        elif move_type < 0.7:
            # LP-guided move
            b, M = solve_lp(a)
            if b is None:
                T *= cooling
                continue
            a_new = a + 0.3 * (b - a)
        elif move_type < 0.85:
            # Swap two blocks
            w = max(1, int(n * 0.05))
            p1 = int(rng.integers(0, n - w))
            p2 = int(rng.integers(0, n - w))
            a_new = a.copy()
            a_new[p1:p1+w], a_new[p2:p2+w] = a_new[p2:p2+w].copy(), a_new[p1:p1+w].copy()
        else:
            # Gaussian perturbation
            a_new = a + rng.normal(0, 0.02, n)
        
        a_new = project_nonneg_sum1(a_new)
        obj_new = compute_obj(a_new)
        
        delta = obj_new - current_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_new
            current_obj = obj_new
            if obj_new < best_obj:
                best_a = a_new.copy()
                best_obj = obj_new
        
        T *= cooling
    
    return best_a, best_obj

def fine_tune(a, max_time, rng):

                best_obj = obj_lp

        if sigma < 0.001:
            break
    
    return best_a, best_obj

"""
Multi-strategy optimization for step function coefficients.
Combines: spectral initialization, LP alternating, adaptive subgradient with
momentum, simulated annealing with diverse moves, and fine-tuning.
Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute the objective: 2*n*max(autocorr) / sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def project_nonneg_sum1(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_A_for_lp(a):
    """Build constraint matrix for LP: (a*b)[k] <= M for all k."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp(a):
    """Solve LP: maximize sum(b) s.t. a*b <= M, b >= 0. Returns normalized b."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv))
    A = build_A_for_lp(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        b = res.x
        s = float(np.sum(b))
        if s > 1e-10:
            b = b / s
        return b, M
    return None, M


def spectral_init(n, rng, num_iters=60):
    """Spectral initialization: aim for flat autocorrelation via Gerchberg-Saxton."""
    c_target = np.ones(2 * n - 1) / (2 * n - 1)
    # Compute DFT of target autocorrelation (padded to 2n)
    C_t = np.zeros(2 * n)
    C_t[:2 * n - 1] = np.fft.fft(c_target)
    target_mag = np.sqrt(np.maximum(np.real(C_t), 0))

    # Initial guess
    a = np.maximum(0.1, np.random.default_rng(123).uniform(0.5, 1.0, n))
    a = a / np.sum(a)

    for it in range(num_iters):
        # Frequency domain projection
        A_fft = np.fft.fft(a, n=2 * n)
        mag = np.abs(A_fft)
        where_nonzero = mag > 1e-10
        phase = np.zeros(2 * n, dtype=complex)
        phase[where_nonzero] = A_fft[where_nonzero] / mag[where_nonzero]
        phase[~where_nonzero] = 1.0
        A_new = target_mag * phase
        a = np.real(np.fft.ifft(A_new))[:n]
        # Time domain projection
        a = np.maximum(a, 1e-8)
        a = a / np.sum(a)

    return a


def lp_refine(a, max_time, max_iters=200):

    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break
        b, M = solve_lp(a)
        if b is None:
            break

        improved = False
        for t in [0.8, 0.6, 0.4, 0.25, 0.15, 0.08, 0.03]:
            a_new = (1 - t) * a + t * b
            a_new = project_nonneg_sum1(a_new)
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_a = a_new.copy()
                best_obj = obj
                a = a_new
                improved = True
                break

        if not improved:
            stall += 1
            if stall > 15:
                break
        else:
            stall = 0

    return best_a, best_obj
