d/dx = [2*c*(S'+x)^2 - (base_k + 2*c*x)*2*(S'+x)] / (S'+x)^4
= [2*c*(S'+x) - 2*(base_k + 2*c*x)] / (S'+x)^3
= [2*c*S' + 2*c*x - 2*base_k - 4*c*x] / (S'+x)^3
= [2*c*S' - 2*base_k - 2*c*x] / (S'+x)^3
= 2*(c*S' - base_k - c*x) / (S'+x)^3

Set to zero: c*S' - base_k - c*x = 0 → x = (c*S' - base_k) / c = S' - base_k/c

Optimized step function coefficient search using exact coordinate descent,
LP-based alternating, simulated annealing, and lopsided initializations.
Key insight: non-palindromic sequences can achieve objective < 2 by making
the anti-diagonal product sum (result[n-1]) small.

import numpy as np
from scipy.optimize import linprog, minimize_scalar
import time


def compute_obj(a):
    """Compute the objective function value."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_convolution_matrix_row(a, i):
    """
    For coordinate i, return:
    - base: array of length 2n-1, where base[k] = result[k] minus terms involving a[i]
    - k_affected: indices k where result[k] depends on a[i]
    - coeffs: for each affected k, the coefficient of x (linear term)
    - has_quad: for each affected k, whether there's a quadratic term (x^2)
    """
    n = len(a)
    result = np.convolve(a, a)
    x = a[i]
    base = result.copy()
    
    # k values where 0 <= k-i < n
    k_min = max(0, i)
    k_max = min(2*n-2, i+n-1)
    k_affected = np.arange(k_min, k_max+1)
    diffs = k_affected - i  # = 0, 1, ..., k_max - i
    
    # Remove a[i] contribution from base
    for idx, k in enumerate(k_affected):
        d = k - i
        if d == i:  # k = 2i, quadratic term
            base[k] -= x * x
        else:
            base[k] -= 2 * x * a[d]
    
    return base, k_affected, diffs


def coord_descent_one_step(a, i):

    n = len(a)
    x_old = a[i]
    S_prime = float(np.sum(a)) - x_old
    
    if S_prime < 1e-15:
        return a, compute_obj(a)
    
    result = np.convolve(a, a)
    base = result.copy()

    k_min = max(0, i)
    k_max = min(2*n-2, i+n-1)
    k_affected = np.arange(k_min, k_max+1)
    diffs = k_affected - i
    
    # Subtract a[i] contributions
    quad_mask = (diffs == i)  # where k = 2i
    lin_mask = ~quad_mask
    
    base[k_affected[lin_mask]] -= 2 * x_old * a[diffs[lin_mask]]
    base[k_affected[quad_mask]] -= x_old * x_old
    
    # Max of unaffected part
    all_indices = np.arange(2*n-1)
    mask_unaffected = np.ones(2*n-1, dtype=bool)
    mask_unaffected[k_affected] = False
    if np.any(mask_unaffected):
        max_unaff = float(np.max(base[mask_unaffected]))
    else:
        max_unaff = 0.0
    
    # Precompute coefficients for affected k
    lin_coeffs = a[diffs[lin_mask]]  # coefficient of 2*x for each linear k
    quad_k = k_affected[quad_mask]  # the k where quadratic term appears
    lin_k = k_affected[lin_mask]  # the k where linear term appears
    
    def obj_1d(x):
        # Compute max of all result_k(x) / (S_prime + x)^2
        # Unaffected: max_unaff / (S_prime + x)^2
        # Affected linear: (base[k] + 2*lin_coeff*x) / (S_prime + x)^2
        # Affected quad: (base[k] + x^2) / (S_prime + x)^2
        
        if x < 0:
            x = 0.0
        denom = (S_prime + x) ** 2
        
        max_val = max_unaff
        if len(lin_k) > 0:
            vals = base[lin_k] + 2.0 * lin_coeffs * x
            max_val = max(max_val, float(np.max(vals)))
        if len(quad_k) > 0:
            vals = base[quad_k] + x * x
            max_val = max(max_val, float(np.max(vals)))
        
        return max_val / denom
    
    # Find upper bound for x
    upper = max(3.0 * x_old, 2.0 * float(np.max(a)), 0.5)
    
    # Minimize
    res = minimize_scalar(obj_1d, bounds=(0.0, upper), method='bounded',
                          options={'xatol': 1e-12, 'maxiter': 200})
    
    a_new = a.copy()
    a_new[i] = max(0.0, res.x)
    return a_new, compute_obj(a_new)

    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def lopsided_init(n, variant, rng):
    if variant == 0:
        p = rng.uniform(0.5, 2.5)
        a = np.power(np.arange(1, n+1, dtype=np.float64), p)
    elif variant == 1:
        t = rng.uniform(0.5, 3.0) * (1 if rng.random() < 0.5 else -1)
        a = np.exp(t * np.arange(n, dtype=np.float64) / n)
    elif variant == 2:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 0.8)
        a = 1.0 + amp * np.cos(2 * np.pi * np.arange(n) / n + phase)
        a = np.maximum(a, 0.01)
    elif variant == 3:
        a = np.exp(rng.normal(0, 1, n) * np.linspace(-1, 1, n) * 0.8)
    elif variant == 4:
        split = int(n * rng.uniform(0.35, 0.65))
        a = np.zeros(n)
        a[:split] = rng.uniform(0.5, 1.5, split)
        a[split:] = rng.uniform(0.05, 0.4, n - split)
    elif variant == 5:
        freq = rng.uniform(0.5, 2.0)
        a = 1.0 + 0.7 * np.sin(2 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
    else:
        a = np.maximum(0, rng.normal(0.5, 0.3, n) + np.linspace(-0.2, 0.2, n))
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def coord_step(a, i, result_cache):
    n = len(a)
    x_old = a[i]
    S_prime = float(np.sum(a)) - x_old
    if S_prime < 1e-15:
        return a, result_cache

    k_min = max(0, i)
    k_max = min(2 * n - 2, i + n - 1)
    if k_min > k_max:
        return a, result_cache

    k_aff = np.arange(k_min, k_max + 1)
    diffs = k_aff - i
    quad_mask = (diffs == i)
    lin_mask = ~quad_mask

    base_aff = result_cache[k_aff].copy()
    if np.any(lin_mask):
        base_aff[lin_mask] -= 2.0 * x_old * a[diffs[lin_mask]]
    if np.any(quad_mask):
        base_aff[quad_mask] -= x_old * x_old

    mask_unaff = np.ones(2 * n - 1, dtype=bool)
    mask_unaff[k_min:k_max + 1] = False
    if np.any(mask_unaff):
        max_unaff = float(np.max(result_cache[mask_unaff]))
    else:
        max_unaff = 0.0

    if np.any(lin_mask):
        lin_coeff = a[diffs[lin_mask]]
        lin_base = base_aff[lin_mask]
    else:
        lin_coeff = np.array([], dtype=np.float64)
        lin_base = np.array([], dtype=np.float64)

    if np.any(quad_mask):
        quad_base = base_aff[quad_mask]
    else:
        quad_base = np.array([], dtype=np.float64)

    def obj_1d(x):
        denom = (S_prime + x) * (S_prime + x)
        max_val = max_unaff
        if len(lin_coeff) > 0:
            m = float(np.max(lin_base + 2.0 * lin_coeff * x))
            if m > max_val:
                max_val = m
        if len(quad_base) > 0:
            m = float(np.max(quad_base + x * x))
            if m > max_val:
                max_val = m
        return max_val / denom

    upper = max(3.0 * abs(x_old), 2.0 * float(np.max(a)), 0.5)
    res = minimize_scalar(obj_1d, bounds=(0.0, upper), method='bounded',
                          options={'xatol': 1e-12, 'maxiter': 80})
    x_new = max(0.0, res.x)

    if abs(x_new - x_old) > 1e-14:
        delta = x_new - x_old
        new_result = result_cache.copy()
        if np.any(lin_mask):
            new_result[k_aff[lin_mask]] += 2.0 * delta * a[diffs[lin_mask]]
        if np.any(quad_mask):
            new_result[k_aff[quad_mask]] += delta * (x_new + x_old)
        result_cache = new_result
        a[i] = x_new

    return a, result_cache


def coordinate_descent(a, max_sweeps=200, time_budget=30.0, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = a.copy()
    result_cache = np.convolve(a, a).copy()
    sum_a = float(np.sum(a))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    no_improve_count = 0

    for sweep in range(max_sweeps):
        if time.monotonic() - start > time_budget:
            break
        order = rng.permutation(n)
        sweep_best = best_obj
        for i in order:
            a, result_cache = coord_step(a, i, result_cache)
        cur_obj = compute_obj(a)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 15:
                break

    return best_a, best_obj


def build_A_mat(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        A[k, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
    return A


def lp_step(a, max_time=5.0):
    n = len(a)
    result = np.convolve(a, a)
    M = float(np.max(result))
    A = build_A_mat(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, 10.0)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': max_time})
    if res.status != 0:
        return None
    g = res.x
    g_sum = float(np.sum(g))
    if g_sum <= 1.0 + 1e-6:
        return None
    g = g / g_sum
    return g


def lp_alternate(a, max_iters=20, time_budget=30.0):
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        g = lp_step(a, max_time=2.0)
        if g is None:
            break
        stall += 1
        t_vals = [0.5, 0.3, 0.15, 0.1]
        for t in t_vals:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, None)
            s = float(np.sum(a_new))
            if s < 1e-10:
                continue
            a_new = a_new / s
            obj = compute_obj(a_new)
            if obj < best_obj:
                best_a = a_new.copy()
                best_obj = obj
                a = a_new
                stall = 0
                break
        if stall > 10:
            break

    return best_a, best_obj


def block_refine(a, rng, time_budget, start_time=None):
    if start_time is None:
        start_time = time.monotonic()
    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    result_cache = np.convolve(best_a, a).copy()

    while time.monotonic() - start_time < time_budget:
        width = max(1, int(n * rng.uniform(0.02, 0.15)))
        pos = int(rng.integers(0, n - width + 1))
        sigma = rng.uniform(0.05, 0.3)
        candidate = best_a.copy()
        candidate[pos:pos + width] *= np.exp(rng.normal(0, sigma, width))
        candidate = np.clip(candidate, 0.0, None)
        s = float(np.sum(candidate))
        if s < 1e-10:
            continue
        obj = compute_obj(candidate)
        if obj < best_obj:
            best_a = candidate
            best_obj = obj

    return best_a, best_obj


def full_pipeline(a_init, total_time, rng):
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if s < 1e-10:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: Coordinate descent
    cd_time1 = min(15.0, total_time * 0.4)
    a1, obj1 = coordinate_descent(a, max_sweeps=500, time_budget=cd_time1, rng=rng)
    if obj1 < best_obj:
        best_a, best_obj = a1, obj1

    # Phase 2: LP alternating
    if time.monotonic() - start < total_time * 0.7:
        lp_time = min(15.0, total_time * 0.25)
        a2, obj2 = lp_alternate(best_a, max_iters=30, time_budget=lp_time)
        if obj2 < best_obj:
            best_a, best_obj = a2, obj2

    # Phase 3: More coordinate descent after LP
    if time.monotonic() - start < total_time * 0.85:
        cd_time2 = min(10.0, total_time * 0.2)
        a3, obj3 = coordinate_descent(best_a, max_sweeps=500, time_budget=cd_time2, rng=rng)
        if obj3 < best_obj:
            best_a, best_obj = a3, obj3

    # Phase 4: Block perturbation
    remaining = total_time - (time.monotonic() - start)
    if remaining > 2.0:
        a4, obj4 = block_refine(best_a, rng, min(remaining, 8.0))
        if obj4 < best_obj:
            best_a, best_obj = a4, obj4

    # Phase 5: Final CD
    remaining = total_time - (time.monotonic() - start)
    if remaining > 2.0:
        a5, obj5 = coordinate_descent(best_a, max_sweeps=300, time_budget=min(remaining, 5.0), rng=rng)
        if obj5 < best_obj:
            best_a, best_obj = a5, obj5

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 8.0

    best_seq = None
    best_score = np.inf

    # Try parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception:
        pass

    # N values to explore
    n_values = [150, 200, 250, 300, 350, 400, 500, 600]

    # Time allocation: give more time to promising n values
    # Strategy: quick scan first, then deep dive on best
    for n in n_values:
        if time.monotonic() > deadline - 90:
            break
        remaining = deadline - time.monotonic()
        num_left = len([x for x in n_values if x >= n])
        time_per_n = remaining / max(1, num_left)
        time_per_n = min(time_per_n, 120.0)

        # Multiple restarts for this n
        n_restarts = max(2, min(5, int(time_per_n / 15)))
        time_per_restart = time_per_n / n_restarts

        print(f"\nN={n}: remaining={remaining:.0f}s, {n_restarts} restarts, {time_per_restart:.1f}s each")

        best_n_score = np.inf
        best_n_a = None

        for r in range(n_restarts):
            if time.monotonic() > deadline - 90:
                break
            variant = (r + n) % 7
            a_init = lopsided_init(n, variant, rng)
            a_opt, obj_opt = full_pipeline(a_init, time_per_restart, rng)
            if obj_opt < best_n_score:
                best_n_score = obj_opt
                best_n_a = a_opt
            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} r={r} v={variant}: obj={obj_opt:.6f} ** NEW BEST **")
            else:
                print(f"  n={n} r={r} v={variant}: obj={obj_opt:.6f}")

        # Deep dive on best for this n if time allows
        if best_n_a is not None and time.monotonic() < deadline - 60:
            remaining = deadline - time.monotonic()
            deep_time = min(remaining * 0.3, 30.0)
            if deep_time > 5:
                a_deep, obj_deep = full_pipeline(best_n_a, deep_time, rng)
                if obj_deep < best_score:
                    best_score = obj_deep
                    best_seq = a_deep.copy()
                    print(f"  n={n} deep: obj={obj_deep:.6f} ** NEW BEST **")

    # Final polish on overall best
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: {remaining:.0f}s")
        a_final, obj_final = full_pipeline(best_seq.copy(), min(remaining * 0.6, 40.0), rng)
        if obj_final < best_score:
            best_score = obj_final
            best_seq = a_final.copy()
            print(f"  After final GP: obj={best_score:.6f}")

        # Block refinement
        if time.monotonic() < deadline - 5:
            bp_time = deadline - time.monotonic() - 3.0
            a_bp, obj_bp = block_refine(best_seq, rng, bp_time)
            if obj_bp < best_score:
                best_score = obj_bp
                best_seq = a_bp.copy()
                print(f"  After BP: obj={best_score:.6f}")

        # One more CD pass
        if time.monotonic() < deadline - 3:
            cd_time = deadline - time.monotonic() - 2.0
            a_cd, obj_cd = coordinate_descent(best_seq, max_sweeps=1000, time_budget=cd_time, rng=rng)
            if obj_cd < best_score:
                best_score = obj_cd
                best_seq = a_cd.copy()
                print(f"  After final CD: obj={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
