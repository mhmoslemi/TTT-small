import numpy as np

def compute_objective(f):
    """Compute 2n*max(conv(f,f))/sum(f)^2 without calling evaluate_sequence."""
    n = len(f)
    c = np.convolve(f, f)
    S = np.sum(f)
    if S < 1e-10:
        return np.inf
    return 2.0 * n * np.max(c) / (S * S)


def build_conv_matrix(f):
    """Build matrix A where (A@g)[k] = (f*g)[k] for k=0,...,2n-2."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_direction(f, M, A=None):
    """Solve LP to find direction g that increases sum while keeping conv(f,g) <= M."""
    n = len(f)
    if A is None:

    c = -np.ones(n)

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except:
        pass
    return None

    # Find all indices where c[k] is close to c_max

    active = c >= threshold
    if not np.any(active):
        active = c == c.max()
    p = np.zeros_like(c)
    p[active] = 1.0 / np.sum(active)
    corr = np.correlate(p, f, mode='valid')
    grad = 2.0 * corr
    new_f = f - lr * grad
    new_f = np.clip(new_f, 0, 1000)
    S = np.sum(new_f)
    if S < 1e-10:

def tukey_window(n, r=0.3):
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2*i/(2*m) - 1)))
            w[n-1-i] = w[i]
    return w


def hann_window(n):
    n = max(n, 2)
    return 0.5 * (1 + np.cos(2*np.pi*np.arange(n) / (n-1)))


def raised_cosine(n, r=0.5):
    """Raised cosine with parameter r controlling the flat top width."""
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2*i/(2*m-1))))
            w[n-1-i] = w[i]
    return w


def super_gaussian(n, alpha=2.0):
    idx = np.arange(n, dtype=np.float64)
    center = (n-1) / 2.0
    return np.exp(-((idx - center) / (n/2.0)) ** (2*alpha))

"""
Multi-phase optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses momentum-based smoothed gradient with adaptive step size, LP direction finding,
direct subgradient steps, and aggressive high-beta polishing. Explores multiple lengths.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_obj(f):
    """Fast computation of 2n*max(conv(f,f))/sum(f)^2."""
    n = len(f)
    c = np.convolve(f, f)
    S = np.sum(f)
    if S < 1e-10:
        return np.inf
    return float(2.0 * n * np.max(c) / (S * S))


def build_conv_matrix(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_direction(f, M, A=None):
    """Solve LP: max sum(g) s.t. g>=0, conv(f,g)<=M."""
    n = len(f)
    if A is None:
        A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def softmax_grad(f, beta):
    """Gradient of (1/beta)*log(sum exp(beta*conv(f,f))) w.r.t. f."""
    c = np.convolve(f, f)
    c_max = c.max()
    safe_beta = min(beta, 700.0 / max(c_max, 1e-10))
    exp_c = np.exp(safe_beta * (c - c_max))
    total = exp_c.sum()
    if total < 1e-300:
        return np.zeros(len(f)), c_max
    p = exp_c / total
    corr = np.correlate(p, f, mode='valid')
    grad = 2.0 * corr
    soft_val = c_max + np.log(total) / safe_beta
    return grad, soft_val


def subgrad_1d(f, lr):
    """Direct subgradient step on the unsmoothed max objective."""
    c = np.convolve(f, f)
    c_max = c.max()
    tol = max(1e-12, c_max * 1e-6)
    active = c >= (c_max - tol)
    p = np.zeros_like(c)
    p[active] = 1.0 / np.sum(active)
    corr = np.correlate(p, f, mode='valid')
    grad = 2.0 * corr
    new_f = f - lr * grad
    new_f = np.clip(new_f, 0, 1000)
    S = np.sum(new_f)
    if S < 1e-10:
        return f.copy()
    return new_f / S


def lp_phase(f, deadline, max_iters=20, step_init=0.12, verbose=""):
    """LP-based direction finding."""
    f = f.copy().astype(np.float64)
    f = f / np.sum(f)
    best_f = f.copy()
    best_score = compute_obj(f)
    step = step_init
    it = 0
    no_improve = 0

    while time.monotonic() < deadline and it < max_iters:
        c = np.convolve(f, f)
        M = c.max()
        if M <= 0:
            break
        A = build_conv_matrix(f)
        g = lp_direction(f, M, A)
        it += 1
        if g is None:
            no_improve += 1
            if no_improve > 5:
                break
            continue
        gs = np.sum(g)
        if gs < 1e-15:
            no_improve += 1
            continue
        g = g / gs
        best_t = step
        best_s = best_score
        for t in [step, step * 1.5, step * 0.5, step * 2.0]:
            new_f = (1 - t) * f + t * g
            new_f = np.clip(new_f, 0, 1000)
            ns = np.sum(new_f)
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = compute_obj(new_f)
            if sc < best_s:
                best_s = sc
                best_t = t
        if best_s < best_score:
            f = (1 - best_t) * f + best_t * g
            f = np.clip(f, 0, 1000)
            f = f / np.sum(f)
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            if it % 5 == 0:
                print(f"    {verbose} LP it={it} score={best_score:.6f}")
        else:
            no_improve += 1
            step *= 0.65
            if step < 0.01 or no_improve > 8:
                break
    return best_f, best_score


def momentum_grad_opt(f, deadline, beta_start=8.0, beta_end=400.0,
                       lr_init=0.04, mu_init=0.85, max_iters=4000, verbose=""):

    f = f.copy().astype(np.float64)
    f = f / np.sum(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_obj(f)
    v = np.zeros(n)
    mu = mu_init
    lr = lr_init
    it = 0
    no_improve = 0
    prev_grad_norm = None

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        frac = min(1.0, it / (max_iters * 0.7))
        beta = beta_start * (beta_end / beta_start) ** frac

        grad, _ = softmax_grad(f, beta)
        gnorm = np.sqrt(np.sum(grad ** 2))
        if gnorm < 1e-12:
            break

        # Barzilai-Borwein inspired adaptive lr
        if prev_grad_norm is not None and prev_grad_norm > 1e-12:
            # Adjust based on progress
            pass
        base_lr = 1.0 / (gnorm * np.sqrt(n) + 1e-12)
        lr = min(base_lr * 0.5, lr_init)
        if no_improve > 100:
            lr *= 0.5

        v = mu * v + lr * grad
        new_f = f - v
        new_f = np.clip(new_f, 0, 1000)
        ns = np.sum(new_f)
        if ns < 1e-10:
            v *= 0.5
            new_f = f - v
            new_f = np.clip(new_f, 0, 1000)
            ns = np.sum(new_f)
            if ns < 1e-10:
                break
        new_f = new_f / ns

        sc = compute_obj(new_f)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = new_f.copy()
            f = new_f
            no_improve = 0
            if it % 500 == 0:
                print(f"    {verbose} MG it={it} score={best_score:.6f} beta={beta:.1f} lr={lr:.6f}")
        else:
            no_improve += 1
            if no_improve > 150:
                mu = max(0.5, mu * 0.92)
                no_improve = 0
                if mu < 0.55:
                    break

        prev_grad_norm = gnorm

    return best_f, best_score


def subgrad_phase(f, deadline, lr_init=0.015, max_iters=3000, verbose=""):

    f = f.copy().astype(np.float64)

    best_f = f.copy()

    lr = lr_init
    it = 0
    no_improve = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        new_f = subgrad_1d(f, lr)

            best_score = sc
            best_f = new_f.copy()
            f = new_f
            no_improve = 0
            if it % 500 == 0:
                print(f"    {verbose} SG it={it} score={best_score:.6f} lr={lr:.6f}")
        else:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.6
                no_improve = 0
                if lr < 1e-7:

def high_beta_polish(f, deadline, beta=1500.0, lr_init=0.002, max_iters=4000, verbose=""):
    """Aggressive polishing with very high beta and tiny steps + momentum."""
    f = f.copy().astype(np.float64)
    f = f / np.sum(f)
    n = len(f)
    best_f = f.copy()
    best_score = compute_obj(f)

                lr *= 0.5

    # Phase 3: Subgradient
    d3 = t0 + budget * 0.72
    if time.monotonic() < d3:

        if s < best_score:

def tukey_window(n, r=0.3):
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def hann_window(n):
    n = max(n, 2)
    return 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))


def super_gaussian(n, alpha=2.0):
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    return np.exp(-((idx - center) / (n / 2.0)) ** (2 * alpha))


def flat_top_cosine(n, p=0.3):
    """Flat-top with cosine taper at edges."""
    w = np.ones(n)
    m = int(np.floor(p * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i + 1) / (2 * m + 1)))
            w[n - 1 - i] = w[i]
    return w

                x_new = np.linspace(0, 1, n)
                interp = np.interp(x_new, x_old, best_sequence)
                inits.append(('parent_interp', np.clip(interp, 0, None)))

        inits.append(('tukey_03', tukey_window(n, 0.3)))
        inits.append(('tukey_04', tukey_window(n, 0.4)))
        inits.append(('tukey_05', tukey_window(n, 0.5)))
        inits.append(('hann', hann_window(n)))
        inits.append(('sg2', super_gaussian(n, 2.0)))
        inits.append(('sg3', super_gaussian(n, 3.0)))
        inits.append(('ftc', flat_top_cosine(n, 0.35)))

        # Score and sort
        scored = []
        for label, fi in inits:
            fi = np.clip(fi, 0, 1000)
            S = np.sum(fi)
            if S < 0.01:
                continue
            fi = fi / S
            sc = compute_obj(fi)
            scored.append((sc, label, fi))
            if sc < best_score:
                best_score = sc
                best_sequence = fi.copy()

        scored.sort(key=lambda x: x[0])

        # Optimize top candidates
        n_top = min(3, len(scored))
        avail = n_deadline - time.monotonic()
        per_init = avail / n_top if n_top > 0 else 0

        for i in range(n_top):
            if time.monotonic() > n_deadline:
                break
            sc, label, fi = scored[i]
            init_deadline = time.monotonic() + per_init - 1
            if time.monotonic() > init_deadline:
                break
            print(f"  Optimizing [{label}] init_score={sc:.6f}")
            try:
                bf, bs = full_pipeline(fi, init_deadline, verbose=f"[n={n}/{label}] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  [{label}] -> {bs:.6f}")
            except Exception as e:
                print(f"  [{label}] error: {e}")

    # Final polishing
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print(f"\n=== FINAL POLISH (n={len(best_sequence)}, score={best_score:.6f}) ===")
        try:
            bf, bs = full_pipeline(best_sequence, deadline - 3, verbose="[FINAL] ")
            if bs < best_score:
                best_score = bs
                best_sequence = bf.copy()
            # Extra high-beta polish
            if time.monotonic() < deadline - 1:
                bf2, bs2 = high_beta_polish(best_sequence, deadline - 1, beta=2000,
                                             lr_init=0.001, max_iters=3000, verbose="[HP2] ")
                if bs2 < best_score:
                    best_score = bs2
                    best_sequence = bf2.copy()
        except Exception as e:
            print(f"Final error: {e}")

    # Validate and return
    if best_sequence is None:
        best_sequence = tukey_window(512, 0.3)
    best_sequence = np.clip(best_sequence, 0, 1000)
    S = np.sum(best_sequence)
    if S < 0.01:
        best_sequence = tukey_window(512, 0.3)

    final_score = evaluate_sequence(best_sequence.tolist())
    if not np.isfinite(final_score):
        best_sequence = tukey_window(512, 0.3)
        final_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\nFINAL: n={len(best_sequence)}, score={final_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_sequence]
