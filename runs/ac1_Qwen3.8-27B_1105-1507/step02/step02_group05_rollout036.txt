This is related to the "packing" problem or "convolution peak" minimization. The key insight is that we want a sequence where the self-convolution is as flat as possible (uniform), because then max(convolve(a,a)) is minimized relative to sum(a)².

If the sequence were perfectly "flat" in the convolution sense, we'd have max(convolve(a,a)) ≈ sum(a)² / (2n-1), giving J ≈ 2n/(2n-1) ≈ 1 for large n. But the constraint is that a_j >= 0, which makes this impossible.

    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""

def smoothed_grad(a, tau):

def adam_step(x, grad, m, v, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Single Adam step. Returns updated x, m, v."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
    return x_new, m, v


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)

    x = np.log(a)  # log-space parameters
    n = len(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    
    best_local = a.copy()
    best_local_J = compute_J(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Anneal temperature
        progress = it / max(1, n_iters - 1)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        a = np.exp(x)
        a = np.minimum(a, 1000.0)
        
        J, grad_a = smoothed_grad(a, tau)
        
        # Gradient in log-space: dJ/dx[j] = dJ/da[j] * a[j]
        grad_x = grad_a * a
        
        # Adam step
        x, m, v = adam_step(x, grad_x, m, v, it + 1, lr=lr)
        
        # Clip to prevent overflow
        x = np.clip(x, np.log(1e-6), np.log(1000.0))
        
        if it % 50 == 0 or it == n_iters - 1:
            a_check = np.exp(x)
            J_check = compute_J(a_check)
            if J_check < best_local_J:
                best_local_J = J_check
                best_local = a_check.copy()
            tracker.update(a_check)
    
    tracker.update(best_local)
    return best_local


def multiplicative_updates(a_init, n_iters, eta, deadline, tracker):
    """Multiplicative weights update to flatten the convolution."""
    a = np.maximum(a_init, 1e-6).astype(np.float64)
    n = len(a)
    best = a.copy()
    best_J = compute_J(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        b_mean = b.sum() / len(b)
        
        # Excess
        e = b - b_mean
        
        # g[j] = sum_k e[k] * a[k-j] (valid)
        # Same as smoothed_grad but with e instead of p (and without the 1/tau)
        q = e[::-1]
        s_len = 1
        while s_len < (2 * n - 1) + n - 1:
            s_len <<= 1
        A = np.fft.rfft(a, s_len)
        Q = np.fft.rfft(q, s_len)
        conv = np.fft.irfft(A * Q, s_len)
        g = np.zeros(n)
        for j in range(n):
            idx = 2 * n - 2 - j
            if 0 <= idx < len(conv):
                g[j] = conv[idx]
        
        # Normalize eta
        eta_eff = eta * (b_mean + 1e-10) / (np.abs(g).max() + 1e-10)
        eta_eff = min(eta_eff, 0.5)
        
        a = a * np.exp(-eta_eff * g)
        a = np.clip(a, 1e-6, 1000.0)
        
        if it % 50 == 0 or it == n_iters - 1:
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best = a.copy()
            tracker.update(a)
    
    tracker.update(best)
    return best


def coordinate_descent_polish(a, n_sweeps=10, deadline=None):
    """Coordinate descent: for each j, find optimal a[j] given others."""
    a = np.maximum(a, 0.0).astype(np.float64)
    n = len(a)
    best = a.copy()
    best_J = compute_J(a)
    
    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                    break
            
            # Save current value
            a_old = a[j]
            
            # The contribution of a[j] to b[k] is: 2*a[j]*a[k-j] (if valid)
            # Plus the "background" from other terms
            # b[k] = background[k] + 2*a[j]*a[k-j] (if 0<=k-j<n)
            #      = background[k] + 2*a[j]*a[j-k] (if 0<=j-k<n)
            
            # We want to minimize max_k b[k] by choosing a[j] >= 0
            # b[k] is linear in a[j] with slope 2*a[k-j] (or 2*a[j-k])
            # Since a >= 0, slopes are non-negative, so max b is non-decreasing in a[j]
            # Optimal a[j] = 0? No, because the objective has (sum a)^2 in denominator!
            
            # With the full objective J = 2n*max(b)/(sum a)^2:
            # If we increase a[j], max(b) increases but (sum a)^2 also increases
            # The trade-off determines the optimal a[j]
            
            # For simplicity, let's just try a few values
            candidates = [0.0, a_old * 0.1, a_old * 0.5, a_old, a_old * 1.5, a_old * 3.0]
            # Also try the value that equalizes the peak
            # The peak is at some position k*. The value a[j] that would make
            # b[k*] = current_max * (1 - epsilon) is...
            
            best_val = a_old
            best_val_J = best_J  # use current J as baseline
            
            for val in candidates:
                a[j] = val
                J_cand = compute_J(a)
                if J_cand < best_val_J:
                    best_val_J = J_cand
                    best_val = val
            
            a[j] = best_val
        
        J_sweep = compute_J(a)
        if J_sweep < best_J:
            best_J = J_sweep
            best = a.copy()
    
    return best


class ScoreTracker:

def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_iterations(a, max_iters, deadline, tracker):
    """Run LP mirror descent iterations."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        base_t = max(0.05, 0.5 * (1.0 - it / float(2 * max_iters)))
        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.02, base_t * 0.5)]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        else:
            a = best_a
            current_J = best_J

        if it % 20 == 0:
            tracker.update(a)

    tracker.update(a)
    return a

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))² using:
1. Smoothed gradient descent (logsumexp smoothing of max) with Adam in log-space
2. Temperature annealing from smooth to sharp
3. Multiplicative weight updates for convolution flattening
4. LP mirror descent (reused from previous approach)
5. Perturbation restarts and multi-length exploration
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def smoothed_grad(a, tau):
    """Compute smoothed objective and gradient.
    J_smooth = 2n * tau * logsumexp(b/tau) / (sum_a)^2
    Returns (J_smooth, grad) where grad = dJ/da.
    """
    n = len(a)
    b = fast_conv(a)
    sum_a = a.sum()

    b_s = b / tau
    b_m = b_s.max()
    logse = b_m + np.log(np.sum(np.exp(b_s - b_m)))

    J = 2.0 * n * tau * logse / (sum_a ** 2)

    p = np.exp(b_s - b_m)
    p /= p.sum()

    # g[j] = sum_k p[k] * a[k-j] (valid indices)
    # Using FFT: g[j] = conv[a, reverse(p)][2n-2-j]
    q = p[::-1]
    s_len = 1
    while s_len < (2 * n - 1) + n - 1:
        s_len <<= 1
    A = np.fft.rfft(a, s_len)
    Q = np.fft.rfft(q, s_len)
    conv = np.fft.irfft(A * Q, s_len)
    g = np.zeros(n)
    idxs = 2 * n - 2 - np.arange(n)
    valid = (idxs >= 0) & (idxs < len(conv))
    g[valid] = conv[idxs[valid]]

    grad = 4.0 * n * tau * (g * sum_a - logse) / (sum_a ** 3)
    return J, grad


def adam_step(x, grad, m, v, t, lr, beta1, beta2, eps):
    """Single Adam step."""
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
    return x_new, m, v


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def run_adam_smoothed(a_init, tau_start, tau_end, n_iters, lr, deadline, tracker):
    """Run Adam optimization on log-space parameters with temperature annealing."""
    a = np.maximum(a_init, 1e-6).astype(np.float64)
    x = np.log(a)
    n = len(a)

    m = np.zeros(n)
    v = np.zeros(n)

    best_local = a.copy()
    best_local_J = compute_J(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        progress = it / max(1, n_iters - 1)
        tau = tau_start * (tau_end / tau_start) ** progress

        a = np.exp(x)
        a = np.minimum(a, 1000.0)

        if a.sum() < 1e-10:
            x = np.zeros(n)
            continue

        J, grad_a = smoothed_grad(a, tau)
        grad_x = grad_a * a

        # Adaptive learning rate: reduce if grad is exploding
        grad_norm = np.linalg.norm(grad_x)
        if grad_norm > 1e6:
            grad_x = grad_x / grad_norm * 1e6

        x, m, v = adam_step(x, grad_x, m, v, it + 1, lr, 0.9, 0.999, 1e-8)
        x = np.clip(x, np.log(1e-6), np.log(1000.0))

        if it % 30 == 0 or it == n_iters - 1:
            a_check = np.exp(x)
            a_check = np.minimum(a_check, 1000.0)
            J_check = compute_J(a_check)
            if J_check < best_local_J:
                best_local_J = J_check
                best_local = a_check.copy()
            tracker.update(a_check)

    tracker.update(best_local)
    return best_local


def multiplicative_updates(a_init, n_iters, eta, deadline, tracker):
    """Multiplicative weights update to flatten the convolution."""
    a = np.maximum(a_init, 1e-6).astype(np.float64)
    n = len(a)
    best = a.copy()
    best_J = compute_J(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        b_max = b.max()
        b_mean = b.sum() / len(b)

        e = b - b_mean

        q = e[::-1]
        s_len = 1
        while s_len < (2 * n - 1) + n - 1:
            s_len <<= 1
        A = np.fft.rfft(a, s_len)
        Q = np.fft.rfft(q, s_len)
        conv = np.fft.irfft(A * Q, s_len)
        g = np.zeros(n)
        idxs = 2 * n - 2 - np.arange(n)
        valid = (idxs >= 0) & (idxs < len(conv))
        g[valid] = conv[idxs[valid]]

        g_max = np.abs(g).max()
        if g_max < 1e-12:
            break
        eta_eff = min(eta * b_max / g_max, 0.3)

        a = a * np.exp(-eta_eff * g)
        a = np.clip(a, 1e-6, 1000.0)

        if it % 50 == 0 or it == n_iters - 1:
            J = compute_J(a)
            if J < best_J:
                best_J = J
                best = a.copy()
            tracker.update(a)

    tracker.update(best)
    return best


def build_A_sparse(a):
    """Build sparse constraint matrix for LP."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def run_lp_iterations(a, max_iters, deadline, tracker):
    """Run LP mirror descent iterations."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a /= a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g /= g_sum

        base_t = max(0.03, 0.4 * (1.0 - it / float(2 * max_iters)))
        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.01, base_t * 0.5)]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            break
        else:
            a = best_a
            current_J = best_J

        if it % 15 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


class ScoreTracker:
    """Tracks the best sequence found so far."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Minimize J(a) = 2n*max(conv(a,a))/(sum a)^2 using:
    1. Smoothed gradient descent (logsumexp) with Adam in log-space
    2. Multiplicative weight updates for flattening
    3. LP mirror descent
    4. Perturbation restarts at multiple scales
    5. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}, J={tracker.best_score:.8f}", flush=True)
    except (NameError, AttributeError):
        pass

    if parent is None:
        parent = np.ones(128)
        tracker.update(parent)

    parent_n = parent.size

    # === Phase 1: Adam smoothed gradient descent across lengths ===
    t0 = time.monotonic()
    print(f"\n{'='*50} Phase 1: Adam Smoothed {'='*50}", flush=True)

    lengths = sorted(set([parent_n, max(64, parent_n // 2), parent_n * 2]))

    for n in lengths:
        if time.monotonic() > deadline:
            break
        print(f"\n  --- Adam n={n} ---", flush=True)

        for restart in range(4):
            if time.monotonic() > deadline:
                break

            if restart == 0:
                a0 = resample_seq(parent, n)
            elif restart == 1:
                a0 = np.ones(n) * (0.5 + 0.3 * np.abs(np.sin(np.pi * np.linspace(0, 1, n))))
            elif restart == 2:
                if tracker.best is not None:
                    a0 = resample_seq(tracker.best, n)
                else:
                    a0 = np.random.default_rng(rng.integers(0, 10**6)).uniform(0.5, 1.0, n)
            else:
                a0 = rng.uniform(0.3, 1.0, n)
                t_arr = np.linspace(0, 1, n)
                a0 += 0.2 * np.cos(2 * np.pi * rng.uniform(1, 5) * t_arr)
                a0 = np.maximum(a0, 0.05)

            a0 = a0.astype(np.float64)
            a0 = np.maximum(a0, 1e-4)
            a0 /= a0.sum()

            iters = max(300, min(3000, int(80000 / n)))
            tau_start = max(0.02, 1.0 / n)
            tau_end = max(1e-4, 0.02 / n)
            lr = 0.15 / np.sqrt(n)

            run_adam_smoothed(a0, tau_start, tau_end, iters, lr, deadline, tracker)

    tracker.report(f"Phase 1: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Multiplicative updates ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 2: Multiplicative Updates {'='*50}", flush=True)
        n = tracker.best.size

        a = tracker.best.copy()
        a = a / a.sum()

        for eta in [0.5, 0.2, 0.1, 0.05]:
            if time.monotonic() > deadline:
                break
            iters = max(200, min(1000, int(50000 / n)))
            a = multiplicative_updates(a, iters, eta, deadline, tracker)

        tracker.report(f"Phase 2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: LP Mirror Descent ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 3: LP Mirror Descent {'='*50}", flush=True)
        n = tracker.best.size

        if n <= 2048:
            run_lp_iterations(tracker.best, 150, deadline, tracker)

        tracker.report(f"Phase 3: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Fine-grained Adam (lower tau) ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 4: Fine Adam {'='*50}", flush=True)
        n = tracker.best.size

        a0 = tracker.best.copy()
        iters = max(500, min(5000, int(120000 / n)))
        tau_start = max(0.01, 0.3 / n)
        tau_end = max(1e-5, 0.01 / n)

        for lr_mult in [0.1, 0.05, 0.2]:
            if time.monotonic() > deadline:
                break
            lr = lr_mult / np.sqrt(n)
            a0 = tracker.best.copy()
            run_adam_smoothed(a0, tau_start, tau_end, iters, lr, deadline, tracker)

        tracker.report(f"Phase 4: {time.monotonic()-t0:.1f}s")

    # === Phase 5: LP + Adam alternation ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 5: LP+Adam Alternate {'='*50}", flush=True)
        n = tracker.best.size

        if n <= 1024:
            a = tracker.best.copy()
            for idx in range(3):
                if time.monotonic() > deadline:
                    break
                # LP step
                a = run_lp_iterations(a, 30, deadline, tracker)
                if time.monotonic() > deadline:
                    break
                # Adam step
                iters = max(200, min(500, int(20000 / n)))
                tau = max(0.005, 0.1 / n)
                lr = 0.05 / np.sqrt(n)
                a = run_adam_smoothed(a, tau, tau * 0.5, iters, lr, deadline, tracker)

        tracker.report(f"Phase 5: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Perturbation restarts ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 6: Perturbation ''' {'='*50}", flush=True)
        n = tracker.best.size

        for noise_level in [0.15, 0.1, 0.05, 0.03]:
            if time.monotonic() > deadline:
                break
            a0 = tracker.best.copy() + noise_level * rng.standard_normal(n)
            a0 = np.maximum(a0, 1e-4)
            a0 /= a0.sum()

            iters = max(200, min(800, int(40000 / n)))
            tau = max(0.02, 0.5 / n)
            tau_end = max(1e-4, 0.02 / n)
            lr = 0.1 / np.sqrt(n)
            run_adam_smoothed(a0, tau, tau_end, iters, lr, deadline, tracker)

        tracker.report(f"Phase 6: {time.monotonic()-t0:.1f}s")

    # === Phase 7: Final LP polish ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        print(f"\n{'='*50} Phase 7: Final LP Polish {'='*50}", flush=True)
        n = tracker.best.size

        if n <= 512:
            run_lp_iterations(tracker.best, 100, deadline, tracker)
        elif n <= 1024:
            run_lp_iterations(tracker.best, 50, deadline, tracker)

        tracker.report(f"Phase 7: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = compute_J(tracker.best)

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
