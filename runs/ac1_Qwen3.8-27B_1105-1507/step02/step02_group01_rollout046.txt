Optimized step function coefficient search using LP alternating, Adam GD on soft-max,

import numpy as np
from scipy.optimize import linprog
from scipy import sparse
import time

_global_best = {'seq': None, 'obj': np.inf}

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_sparse_A(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_lp(a, M):
    n = len(a)
    A = build_sparse_A(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.status == 0:
        return res.x
    return None

def coordinate_descent(a, max_time, rng, max_iters=500000):
    n = len(a)
    a = a.copy().astype(np.float64)

    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break
        
        i = it % n  # Cycle through coordinates
        
        current = a[i]
        # Try several multipliers
        multipliers = [0.0, 0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 0.1]
        
        best_val = current
        best_cand_obj = best_obj
        
        for mult in multipliers:
            val = current * mult if mult > 0 else 0.0
            if val == current:
                continue
            a[i] = val
            s = np.sum(a)
            if s < 1e-10:
                a[i] = current
                continue
            a_norm = a / s
            obj = compute_obj(a_norm)
            if obj < best_cand_obj:
                best_cand_obj = obj
                best_val = val
        
        a[i] = current
        if best_cand_obj < best_obj - 1e-12:
            a[i] = best_val
            s = np.sum(a)
            if s > 1e-10:
                a = a / s
            best_obj = best_cand_obj
            best_a = a.copy()
            no_improve = 0
        else:

def simulated_annealing(a, max_time, rng, max_iters=200000):

        if time.monotonic() - start > max_time:
            break

        T = T_start * (T_end / T_start) ** progress

        width = max(1, int(n * rng.uniform(0.02, 0.15)))
        pos = int(rng.integers(0, n - width + 1))

def structured_init(n, rng, parent=None):
    if parent is not None and parent.size == n:
        return parent.copy()
    
    choice = int(rng.integers(0, 5))
    if choice == 0:
        # Edge-aware
        c = 1.0 / np.sqrt(2.0 * n - 1.0)
        a = np.zeros(n)
        a[0] = c
        a[1] = c * 0.8
        a[-1] = c
        a[-2] = c * 0.8
        if n > 4:
            remaining = 1.0 - sum(a)
            if remaining > 0:
                interior = np.ones(n - 4) * (remaining / max(1, n - 4))
                noise = rng.normal(0, 0.1, n - 4)
                interior = np.maximum(0, interior * np.exp(noise))
                s = np.sum(interior)
                if s > 1e-10:
                    interior = interior * remaining / s
                a[2:n-2] = interior
        s = np.sum(a)
        return a / s if s > 1e-10 else np.ones(n) / n
    elif choice == 1:
        # Cosine
        phase = rng.uniform(0, 2 * np.pi)
        freq = rng.uniform(0.1, 0.5)
        a = np.maximum(0, np.cos(2 * np.pi * freq * np.arange(n) + phase))
        a += 0.1
        s = np.sum(a)
        return a / s
    elif choice == 2:
        # Gaussian
        center = n / 2.0
        width = n * rng.uniform(0.2, 0.5)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        s = np.sum(a)
        return a / s
    elif choice == 3:
        # Random positive
        a = np.maximum(0, rng.normal(0.5, 0.3, n))
        s = np.sum(a)
        return a / s if s > 1e-10 else np.ones(n) / n
    else:
        # Rise-fall
        a = np.minimum(np.arange(n), n - np.arange(n)).astype(float)
        a = a ** rng.uniform(0.5, 2.0)
        s = np.sum(a)
        return a / s

def resize_sequence(a, new_n):
    """Resize a sequence to new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new
