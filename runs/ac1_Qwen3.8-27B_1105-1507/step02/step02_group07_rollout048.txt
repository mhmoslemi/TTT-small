    n = len(f)
    conv = np.convolve(f, f)
    
    for j in range(n):
        old = f[j]
        # Compute base values (conv with f[j] = 0)
        # base[k] = conv[k] - contribution of f[j]

                else:
                    base[k] = conv[k] - 2 * old * f[k - j]

            #   else: base_val = conv[k], no x-dependence

            #   g(x) = max over k of:
            #     conv[k] + 2*(x-old)*f[k-j]  (if 0<=k-j<n, k!=2j)
            #     conv[2j] + (x^2 - old^2)   (if k=2j, which requires 2j < 2n-1)
            #     conv[k]                    (otherwise)
            
            # Precompute the "slope" and "base" for each k
            # slope[k] = 2*f[k-j] if 0<=k-j<n and k!=2j, else 0
            # base2[k] = conv[k] - slope[k]*old  (so that conv[k] + slope[k]*(x-old) = base2[k] + slope[k]*x)
            
            slopes = np.zeros(2 * n - 1)
            k_range = np.arange(2 * n - 1)
            valid = (k_range - j >= 0) & (k_range - j < n) & (k_range != 2 * j)
            slopes[valid] = 2.0 * f[k_range[valid] - j]
            
            # At k = 2j (if valid):
            if 0 <= 2 * j < 2 * n - 1:
                # The x-dependence is x^2 - old^2
                # We handle this separately
            
            base2 = conv - slopes * old  # conv[k] + slope[k]*(x-old) = base2[k] + slope[k]*x
            
            # Ternary search
            x_max = max(2.0 * old, 3.0 / n)
            
            def g_at(x):
                vals = base2 + slopes * x
                if 0 <= 2 * j < 2 * n - 1:
                    vals[2 * j] = conv[2 * j] - old * old + x * x
                return np.max(vals)
            
            a, b = 0.0, x_max
            for _ in range(30):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                if g_at(m1) < g_at(m2):
                    b = m2
                else:
                    a = m1
            
            f[j] = (a + b) / 2.0
            
            # Update conv
            new = f[j]
            delta = new - old
            if delta != 0:
                for k in range(2 * n - 1):
                    if 0 <= k - j < n:
                        if k == 2 * j:
                            conv[k] += new * new - old * old
                        else:
                            conv[k] += 2 * delta * f[k - j]
    
    S = np.sum(f)
    if S > 0:
        f = f / S
    return f
