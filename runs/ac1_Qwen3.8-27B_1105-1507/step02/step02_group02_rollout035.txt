import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))

def build_conv_matrix(a):
    """Build matrix A where (A @ x) = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None

def mirror_descent_step(a, eta, top_k=5):
    """One mirror descent step with multiplicative updates."""
    n = len(a)
    c = fast_conv(a)
    # Find top k values
    top_indices = np.argsort(c)[-top_k:]
    # Compute weights (proportional to how close to max)
    c_max = c.max()
    weights = np.zeros(2 * n - 1)
    for k in top_indices:
        weights[k] = max(0, (c[k] - 0.5 * c_max)) ** 2
    
    if weights.sum() == 0:
        weights[top_indices[0]] = 1.0
    
    weights /= weights.sum()
    
    # Compute pressure at each position
    # grad of c_k at position j = a[k-j] if 0 <= k-j < n, plus a[j-k] if 0 <= j-k < n
    pressure = np.zeros(n)
    for k, w in zip(top_indices, weights[top_indices]):
        # a[k-j] for valid j: j in [max(0, k-n+1), min(n-1, k)]
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            # a[k-j] for j in [jl, jh] means indices k-jh to k-jl
            pressure[jl:jh+1] += w * a[k-jh:k-jl+1][::-1]
            # a[j-k] for j in [max(0,k), min(n-1, k+n-1)] = [max(0,k), n-1] since k < 2n-1
            jl2 = max(0, k)
            jh2 = min(n - 1, k + n - 1)  # j-k < n means j < k+n
            if jl2 <= jh2:
                pressure[jl2:jh2+1] += w * a[jl2-k:jh2-k+1]
    
    # Multiplicative update
    a_new = a * np.exp(-eta * (pressure - pressure.mean()))
    a_new = np.maximum(a_new, 1e-10)
    a_new = a_new / a_new.sum()
    return a_new

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Mirror descent with multiplicative updates (exploits simplex structure)
2. LP best-response for aggressive moves
3. Multi-resolution coarse-to-fine optimization
4. Simulated annealing for escaping local optima
5. Gradient-based refinement with log-sum-exp smoothing

import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2 directly."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))

def build_conv_matrix(a):
    """Build Toeplitz-like matrix A where (A @ x) = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None

def pressure_vector(a, top_k_indices, weights):
    """Compute the weighted sum of gradients of c_k for k in top_k_indices.
    
    grad c_k at position j = 2*a[k-j] if 0 <= k-j < n, else 0.
    So pressure[j] = 2 * sum_k w_k * a[k-j] (for valid k-j)
    """
    n = len(a)
    pressure = np.zeros(n)
    for k, w in zip(top_k_indices, weights):
        if w == 0:
            continue
        # a[k-j] for j in [0, n): k-j ranges from k to k-n+1
        # Valid: 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            # a[k-j] for j in [j_min, j_max] gives indices k-j_max to k-j_min
            idx_start = k - j_max
            idx_end = k - j_min
            pressure[j_min:j_max + 1] += 2.0 * w * a[idx_start:idx_end + 1][::-1]
    return pressure

def mirror_descent(a, eta, n_top=5, n_iters=1000, deadline=None):
    """Mirror descent with multiplicative updates for the simplex."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        c_max = c.max()
        
        # Select top indices
        top_idx = np.argsort(c)[-n_top:]
        
        # Weights: proportional to how "active" each constraint is
        # Use exponential weighting
        vals = c[top_idx]
        w = np.exp(10 * (vals - c_max))
        w = w / w.sum()
        
        # Compute pressure
        p = pressure_vector(a, top_idx, w)
        
        # Multiplicative update (mirror descent with entropy regularization)
        # a_new = a * exp(-eta * (p - mean(p)))
        p_centered = p - p.mean()
        a = a * np.exp(-eta * p_centered)
        a = np.maximum(a, 1e-10)
        a = a / a.sum()
        
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

def lp_iteration(a, max_iters, deadline, base_t=0.5):
    """Run LP best-response iterations with adaptive step size."""
    best_local = a.copy()
    best_score = compute_score(a)
    a = a.copy()
    a = np.clip(a, 0.001, None)
    a = a / a.sum()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()
        
        # Adaptive step size
        t = base_t * (1.0 - 0.7 * it / max_iters)
        t = max(0.02, min(0.8, t))
        
        a_new = (1.0 - t) * a + t * g
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            a = a_new
        else:
            t2 = t * 0.3
            a_new2 = (1.0 - t2) * a + t2 * g
            a_new2 = np.clip(a_new2, 0.0, None)
            s2 = a_new2.sum()
            if s2 > 1e-10:
                a_new2 = a_new2 / s2
                score2 = compute_score(a_new2)
                if score2 < best_score:
                    best_score = score2
                    best_local = a_new2.copy()
                a = a_new2
            if score >= best_score and t2 < 0.01:
                break
    
    return best_local, best_score

def smooth_objective(x, beta, n):
    """Log-sum-exp smoothed objective."""
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 0.01:
        return 1e10
    x_norm = x_pos / s
    b = fast_conv(x_norm)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    log_sum = b_max * beta + np.log(np.exp(shifted).sum())
    return float(2.0 * n * log_sum)

def smooth_refine(a, deadline):
    """Refine using smooth log-sum-exp objective with L-BFGS-B."""
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    
    betas = [20, 100, 500, 2000, 10000]
    for beta in betas:
        if time.monotonic() > deadline:
            break
        x0 = best.copy()
        x0 = np.clip(x0, 0.001, None)
        x0 = x0 / x0.sum()
        
        try:
            res = minimize(smooth_objective, x0, args=(beta, n),
                           method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': 200, 'maxfun': 3000,
                                    'ftol': 1e-18, 'gtol': 1e-18})
            if res.fun < 1e10:
                res_x = np.clip(res.x, 0, None)
                if res_x.sum() > 0.01:
                    score = compute_score(res_x)
                    if score < best_score:
                        best = res_x.copy()
                        best_score = score
        except Exception:
            pass
    
    return best, best_score

def upsample(a, new_n):
    """Upsample a sequence to a new length using interpolation."""
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    a_new = np.interp(t_new, t_old, a)
    a_new = np.maximum(a_new, 0)
    if a_new.sum() > 0.01:
        a_new = a_new / a_new.sum()
    return a_new

def simulated_annealing(a, n_tries, deadline, T0=0.1, T_end=0.001):
    """Simulated annealing to escape local optima."""
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    current = a.copy()
    current_score = best_score
    
    for trial in range(n_tries):
        if time.monotonic() > deadline:
            break
        
        T = T0 * (T_end / T0) ** (trial / max(1, n_tries))
        
        # Propose: move mass from one position to another
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        delta = np.random.uniform(0.01, 0.05) * current.sum() / n
        delta = min(delta, current[i] * 0.5)
        if delta < 1e-8:
            continue
        
        new_a = current.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_a = np.maximum(new_a, 0)
        new_a = new_a / new_a.sum()
        
        new_score = compute_score(new_a)
        delta_s = new_score - current_score
        
        if delta_s < 0 or np.random.random() < np.exp(-delta_s / T):
            current = new_a
            current_score = new_score
            if new_score < best_score:
                best = new_a.copy()
                best_score = new_score
    
    return best, best_score

def generate_initializations(n, rng, parent=None):

**Theoretical consideration:**

The problem min max_k (a*a)[k] s.t. a >= 0, sum(a) = 1 is related to the problem of finding a probability distribution on {0, ..., n-1} that minimizes the maximum "concentration" of the distribution of the sum of two independent samples.

If X and Y are i.i.d. with distribution a on {0,...,n-1}, then (a*a)[k] = P(X+Y = k). We want to minimize max_k P(X+Y = k).

This is the "flattening" problem: find a distribution on {0,...,n-1} such that the sum distribution is as flat as possible.

The flat distribution on {0, ..., 2n-2} has probability 1/(2n-1) at each point. We want to get close to this.

For a uniform distribution on {0,...,n-1}, P(X+Y=k) = (min(k+1, 2n-1-k, n))/(n^2), which is a trapezoid. The max is at k=n-1: n/n^2 = 1/n.

To get a flatter sum distribution, we need X to be non-uniform. The question is: what distribution on {0,...,n-1} gives the flattest sum distribution?

This is related to the theory of "replacive" or "non-replacive" sampling, and also to the theory of "Benford's law" type phenomena.

For the continuous analog: if X and Y are i.i.d. on [0,1], what distribution minimizes max of the sum density? The answer is: the arcsine distribution! (Or more precisely, the distribution that makes X+Y uniform on [0,2].)

The arcsine distribution has density f(x) = 1/(pi*sqrt(x*(1-x))) on (0,1). This is concentrated near 0 and 1.

For the discrete analog, the "arcsine-like" distribution on {0,...,n-1} would be concentrated near the endpoints. Let's try:

If X is concentrated near 0 and n-1, then X+Y is concentrated near 0, n-1, and 2n-2. This is actually WORSE than uniform (which gives a trapezoid). So arcsine is not the right answer here.

Actually, for the continuous case, we want f such that (f*f)(x) is as flat as possible on [0,2]. The convolution (f*f)(x) = integral f(t)f(x-t) dt.

If we want (f*f)(x) = 1/2 for all x in [0,2], then taking Fourier transforms: |F(f)|^2 = F(1/2) = 1/2 * 2 * sinc... this is getting complicated.

The key point is: for non-negative functions, it's impossible to have a perfectly flat convolution (that would require |F(f)| to be constant, which for a non-negative f means f is a delta function or uniform, neither of which gives flat convolution).

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Multi-resolution coarse-to-fine optimization
2. Nesterov-accelerated LP best-response iterations
3. Greedy pairwise-swap local search (key new ingredient for fine-tuning)
4. Smooth log-sum-exp refinement with L-BFGS-B
5. Simulated annealing for escaping local optima

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2 directly."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build matrix A where (A @ x) = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None


def nesterov_lp_iteration(a, max_iters, deadline, beta0=0.9):
    """Nesterov-accelerated LP best-response with backtracking line search."""
    best_local = a.copy()
    best_score = compute_score(a)
    a = a.copy()
    a = np.clip(a, 0.001, None)
    a = a / a.sum()
    
    a_prev = a.copy()
    beta = beta0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Nesterov extrapolation
        if it > 0:
            y = a + beta * (a - a_prev)
            y = np.clip(y, 0.001, None)
            y = y / y.sum()
        else:
            y = a
        
        a_prev = a.copy()
        
        b = fast_conv(y)
        M = float(b.max())
        g = lp_best_response(y, M)
        if g is None:
            break
        g = g / g.sum()
        
        # Backtracking line search
        best_t = 0.0
        best_a = a
        best_s = compute_score(a)
        
        for t in [0.8, 0.5, 0.3, 0.15, 0.05]:
            a_new = (1.0 - t) * y + t * g
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            score = compute_score(a_new)
            if score < best_s:
                best_s = score
                best_t = t
                best_a = a_new
        
        if best_t == 0.0:
            # No improvement with any step size; try a very small step
            a_new = (1.0 - 0.01) * y + 0.01 * g
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 1e-10:
                break
            a_new = a_new / s
            best_a = a_new
            best_s = compute_score(a_new)
            if best_s >= best_score:
                break
        
        a = best_a
        a_prev = a.copy()
        
        # Update Nesterov momentum
        beta = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        beta = min(beta, 2.0)
        
        if best_s < best_score:
            best_score = best_s
            best_local = a.copy()
    
    return best_local, best_score

    2. Compute pressure p[j] = 2*a[k*-j] (gradient of c_{k*} at position j)
    3. Find best swap: move from i* (max pressure) to j* (min pressure)
    4. Compute max safe epsilon
    5. Apply swap

    best_a = a.copy()
    best_score = compute_score(a)

    no_improve_count = 0

                best_a = a.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
                if no_improve_count > 50:
                    break

            found_swap = False
            for j_try in sorted_j[:5]:

                    if diff_t > 1e-12:

                    a[j_try] += eps_try

                for j_try in sorted_j2[:3]:
                    if j_try == i_star:
                        continue
                    # Move from j_try to i_star
                    eps_try = a[j_try] * 0.3
                    safe = True
                    for k in range(2 * n - 1):
                        val_jt = a[k - i_star] if 0 <= k - i_star < n else 0.0
                        val_it = a[k - j_try] if 0 <= k - j_try < n else 0.0
                        diff_t = val_jt - val_it

                            if slack_t < 0:

                    if safe and eps_try > 1e-12:
                        a[j_try] -= eps_try
                        a[i_star] += eps_try
                        found_swap2 = True

        else:
            a[i_star] -= eps_max
            a[j_star] += eps_max
        
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            break
        a = a / s
    
    # Final score check
    final_score = compute_score(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    return best_a, best_score


def smooth_objective(x, beta, n):
    """Log-sum-exp smoothed objective."""
    x_pos = np.maximum(x, 0.0)
    s = x_pos.sum()
    if s < 0.01:
        return 1e10
    x_norm = x_pos / s
    b = fast_conv(x_norm)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -50.0, 50.0)
    log_sum = b_max * beta + np.log(np.exp(shifted).sum())
    return float(2.0 * n * log_sum)


def smooth_refine(a, deadline):
    """Refine using smooth log-sum-exp objective with L-BFGS-B."""
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)
    
    betas = [10, 50, 200, 1000, 5000]
    for beta in betas:
        if time.monotonic() > deadline:
            break
        x0 = best.copy()
        x0 = np.clip(x0, 0.001, None)
        x0 = x0 / x0.sum()
        
        try:
            res = minimize(smooth_objective, x0, args=(beta, n),
                           method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': 300, 'maxfun': 5000,
                                    'ftol': 1e-18, 'gtol': 1e-18,
                                    'gtol_rel': 1e-12})
            if res.fun < 1e10:
                res_x = np.clip(res.x, 0, None)
                if res_x.sum() > 0.01:
                    score = compute_score(res_x)
                    if score < best_score:
                        best = res_x.copy()
                        best_score = score
        except Exception:
            pass
    
    return best, best_score


def upsample(a, new_n):
    """Upsample a sequence to a new length using interpolation."""
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    t_old = np.linspace(0, 1, old_n)
    t_new = np.linspace(0, 1, new_n)
    a_new = np.interp(t_new, t_old, a)
    a_new = np.maximum(a_new, 0)
    if a_new.sum() > 0.01:
        a_new = a_new / a_new.sum()
    return a_new


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    t_arr = np.linspace(0, 1, n)
    
    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        resampled = np.maximum(np.interp(t_arr, t_old, parent), 0)
        if resampled.sum() > 0.01:
            inits.append(resampled / resampled.sum())
    
    inits.append(np.ones(n) / n)
    
    for freq in [1, 2, 3, 4, 6]:
        c = 1.0 + 0.35 * np.cos(2 * np.pi * freq * t_arr)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())
    
    # Bimodal (arcsine-like)
    mid = (n - 1) / 2.0
    c = 1.0 / np.sqrt(np.maximum(np.abs(t_arr - 0.5), 0.01) + 0.1)
    inits.append(c / c.sum())
    
    # Quadratic
    c = 1.0 + 0.5 * (4.0 * (t_arr - 0.5) ** 2)
    inits.append(c / c.sum())
    
    # Random
    inits.append(rng.uniform(0.3, 1.0, n) / n)
    
    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Main search function combining multiple optimization strategies."""
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 30.0
    
    best = None
    best_score = float('inf')
    
    # Try parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            p = np.clip(p, 0.0, 1000.0)
            sc = compute_score(p)
            if sc < best_score:
                best = p.copy()
                best_score = sc
                print(f"Parent: J={sc:.6f} (n={p.size})", flush=True)
    except (NameError, AttributeError):
        pass
    
    # === PHASE 1: Multi-resolution (25% of time) ===
    phase1_deadline = time.monotonic() + total_budget * 0.25
    lengths = [64, 128, 256, 512]
    
    if best is not None:
        pl = best.size
        if pl not in lengths:
            lengths.append(pl)
        lengths.sort()
        lengths = [l for l in lengths if l <= 1024]
    
    current_a = None
    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        
        print(f"\n=== Multi-res: n={n} ===", flush=True)
        
        inits = generate_initializations(n, rng, current_a if current_a is not None else best)
        
        for i, a0 in enumerate(inits[:5]):
            if time.monotonic() > phase1_deadline:
                break
            a0 = np.clip(a0, 0.001, None)
            a0 = a0 / a0.sum()
            
            # Quick LP
            remaining = phase1_deadline - time.monotonic()
            est_time = 0.02 if n <= 256 else 0.05
            n_iters = min(2000, max(100, int(remaining * 0.7 / est_time)))
            a_lp, score_lp = nesterov_lp_iteration(a0, n_iters, phase1_deadline)
            
            if score_lp < best_score:
                best = a_lp.copy()
                best_score = score_lp
                current_a = a_lp.copy()
                print(f"  NEW BEST: J={score_lp:.6f} (n={n}, init={i})", flush=True)
        
        if current_a is not None and n < lengths[-1]:
            next_n = lengths[lengths.index(n) + 1] if n + 1 < len(lengths) else n * 2
            current_a = upsample(current_a, next_n)
    
    # === PHASE 2: Nesterov LP at best length (25% of time) ===
    phase2_deadline = min(time.monotonic() + total_budget * 0.25, deadline)
    
    if best is not None and time.monotonic() < phase2_deadline:
        print(f"\n=== Nesterov LP: n={best.size} ===", flush=True)
        n = best.size
        remaining = phase2_deadline - time.monotonic()
        est_time = 0.02 if n <= 256 else 0.05 if n <= 512 else 0.1
        n_iters = min(5000, max(200, int(remaining / est_time)))
        
        a_result, score = nesterov_lp_iteration(best, n_iters, phase2_deadline)
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  NEW BEST (NLP): J={score:.6f}", flush=True)
    
    # === PHASE 3: Greedy local search (30% of time) ===
    phase3_deadline = min(time.monotonic() + total_budget * 0.30, deadline)
    
    if best is not None and time.monotonic() < phase3_deadline:
        print(f"\n=== Greedy Local Search: n={best.size} ===", flush=True)
        n = best.size
        remaining = phase3_deadline - time.monotonic()
        # Each iteration is O(n), roughly 2n-1 + n operations ~ 3n flops
        # For n=512: ~1500 flops per iter, so ~10^6 iters/sec
        est_iter_time = n / 3e6  # rough estimate
        max_iters = int(remaining / max(est_iter_time, 1e-6))
        max_iters = min(max_iters, 50_000_000)
        
        a_result, score = greedy_local_search(best, max_iters, phase3_deadline)
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  NEW BEST (GLS): J={score:.6f}", flush=True)
        else:
            print(f"  GLS: no improvement (stayed at {best_score:.6f})", flush=True)
    
    # === PHASE 4: Smooth refinement + final LP (15% of time) ===
    phase4_deadline = min(time.monotonic() + total_budget * 0.12, deadline)
    
    if best is not None and time.monotonic() < phase4_deadline:
        print(f"\n=== Smooth Refinement ===", flush=True)
        a_result, score = smooth_refine(best, phase4_deadline)
        if score < best_score:
            best = a_result.copy()
            best_score = score
            print(f"  NEW BEST (smooth): J={score:.6f}", flush=True)
    
    # === PHASE 5: Final greedy + LP polish (5% of time) ===
    if best is not None and time.monotonic() < deadline:
        print(f"\n=== Final Polish ===", flush=True)
        n = best.size
        remaining = deadline - time.monotonic()
        
        # Quick greedy
        gls_iters = int(remaining * 0.5 / max(n / 3e6, 1e-6))
        gls_iters = min(gls_iters, 10_000_000)
        a_result, score = greedy_local_search(best, gls_iters, time.monotonic() + remaining * 0.5)
        if score < best_score:
            best = a_result.copy()
            best_score = score
        
        # Quick LP
        remaining = deadline - time.monotonic()
        est_time = 0.02 if n <= 256 else 0.05
        n_iters = min(2000, max(50, int(remaining / est_time)))
        a_result, score = nesterov_lp_iteration(best, n_iters, deadline)
        if score < best_score:
            best = a_result.copy()
            best_score = score
        
        # One more greedy pass
        if time.monotonic() < deadline:
            remaining = deadline - time.monotonic()
            gls_iters = int(remaining / max(n / 3e6, 1e-6))
            a_result, score = greedy_local_search(best, gls_iters, deadline)
            if score < best_score:
                best = a_result.copy()
                best_score = score
    
    # === Try different lengths if time remains ===
    if best is not None and time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining > 100:
            print(f"\n=== Try Other Lengths ===", flush=True)
            for n_try in [256, 384, 640]:
                if time.monotonic() > deadline:
                    break
                if n_try == best.size:
                    continue
                a_try = upsample(best, n_try)
                est_time = 0.02 if n_try <= 256 else 0.05
                n_iters = min(1000, max(50, int((deadline - time.monotonic()) * 0.3 / est_time)))
                a_result, score = nesterov_lp_iteration(a_try, n_iters, deadline)
                if score < best_score:
                    best = a_result.copy()
                    best_score = score
                    print(f"  NEW BEST (n={n_try}): J={score:.6f}", flush=True)
                
                # Quick greedy on new length
                if time.monotonic() < deadline:
                    remaining2 = deadline - time.monotonic()
                    gls_iters = int(remaining2 * 0.5 / max(n_try / 3e6, 1e-6))
                    a_result2, score2 = greedy_local_search(a_result, gls_iters, deadline)
                    if score2 < best_score:
                        best = a_result2.copy()
                        best_score = score2
                        print(f"  NEW BEST (gls n={n_try}): J={score2:.6f}", flush=True)
    
    # Final validation
    if best is None:
        best = np.ones(256) / 256
        best_score = compute_score(best)
    
    best = np.clip(best, 0.0, 1000.0)
    final_score = float(evaluate_sequence([float(x) for x in best]))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
    
    print(f"\nFINAL: J={best_score:.6f}, n={best.size}", flush=True)
    return [float(x) for x in best]
