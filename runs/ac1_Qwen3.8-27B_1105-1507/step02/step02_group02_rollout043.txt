    s=1
    while s<m: s <<=1

def fast_conv(a):
    n=len(a)
    s=next_fft_size(2*n-1)
    fa=np.fft.rfft(a,s)
    out=np.fft.irfft(fa*fa,s)
    out=out[:2*n-1]

    return out

def correlate_weights(p, a):
    # returns 2 * sum_l a_l p_{i+l}, length n
    n=len(a)
    s=next_fft_size(3*n-2)
    ar=np.fft.rfft(a[::-1], s)
    pr=np.fft.rfft(p, s)
    corr=np.fft.irfft(ar*pr, s)[n-1:2*n-1]

"""
Search for non-negative step-function heights minimizing
J(a) = 2 n max_k (a*a)_k / (sum a)^2, where a*a is full linear convolution.

The implementation combines:
- a strong state-of-the-art style LP best-response iteration (primal + dual)
- analytic-gradient L-BFGS minimization of a log-sum-exp smoothed objective
  in a softmax/logit parameterization
- multi-start initialization, resampling, and perturbation searches
- explicit time-budget management to always return the best sequence found

No lambda functions are used, and all helpers are top-level.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time

_FFT_CACHE = {}
_LP_OPTIONS = {
    'presolve': True,
    'dual_feasibility_tolerance': 1e-9,
    'primal_feasibility_tolerance': 1e-9,
}


def next_fft_size(m):
    if m <= 1:
        return 1
    if m in _FFT_CACHE:
        return _FFT_CACHE[m]
    s = 1
    while s < m:
        s <<= 1
    _FFT_CACHE[m] = s
    return s


def fast_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    s = next_fft_size(2 * n - 1)
    fa = np.fft.rfft(a, s)
    out = np.fft.irfft(fa * fa, s)[:2 * n - 1]
    out[out < 0.0] = 0.0
    return out


def correlate_weights(p, a):
    """
    For p of length 2n-1 and a of length n, return
    2 * (a_l p_{i+l}) summed over l, i=0..n-1, which is the gradient
    of a linear function sum_k p_k (a*a)_k with respect to a.
    """
    a = np.asarray(a, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    s = next_fft_size(3 * n - 2)
    fa = np.fft.rfft(a[::-1], s)
    fp = np.fft.rfft(p, s)
    corr = np.fft.irfft(fa * fp, s)[n - 1:2 * n - 1]
    corr[corr < 0.0] = 0.0
    return 2.0 * corr


def normalize_sequence(x):
    x = np.asarray(x, dtype=np.float64).copy()
    if x.size == 0:
        return x
    x[x < 0.0] = 0.0
    s = float(x.sum())
    if s <= 1e-15:
        return np.ones(x.size, dtype=np.float64) / float(x.size)
    return x / s


def compute_score(x):
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return float('inf')
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s < 1e-15:
        return float('inf')
    a = x / s
    b = fast_conv(a)
    if b.size == 0:
        return float('inf')
    return float(2.0 * len(a) * float(b.max()))


def resample_sequence(a, n):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    if a.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    a = a / s
    if a.size == n:
        return a.copy()
    if a.size == 1:
        return np.ones(n, dtype=np.float64) / float(n)
    t_old = np.linspace(0.0, 1.0, a.size, dtype=np.float64)
    t_new = np.linspace(0.0, 1.0, n, dtype=np.float64)
    out = np.interp(t_new, t_old, a)
    out = np.maximum(out, 0.0)
    ss = float(out.sum())
    if ss <= 1e-15:
        return np.ones(n, dtype=np.float64) / float(n)
    return out / ss


def add_pattern(inits, pat):
    pat = np.asarray(pat, dtype=np.float64)
    if pat.size == 0:
        return
    pat = np.maximum(pat, 0.0)
    s = float(pat.sum())
    if s > 1e-12 and np.all(np.isfinite(pat)):
        inits.append(pat / s)


def generate_initializations(n, rng, parent=None):
    inits = []
    if n <= 0:
        return inits

    t = np.linspace(0.0, 1.0, n, dtype=np.float64)

    add_pattern(inits, np.ones(n, dtype=np.float64))
    if parent is not None:
        add_pattern(inits, resample_sequence(parent, n))

    # Smooth cosine-like basins.
    for k in (1, 2, 3, 4, 5):
        for amp in (0.3, 0.6, 0.9):
            add_pattern(inits, 1.0 + amp * np.cos(2.0 * np.pi * k * t))

    # Edge/center polynomial shapes.
    u = 2.0 * t - 1.0
    for amp in (0.5, 1.0, 2.0):
        add_pattern(inits, 1.0 + amp * u * u)
        add_pattern(inits, 1.0 + amp * (1.0 - u * u))

    # One-sided exponential shapes.
    for lam in (1.0, 3.0, 10.0):
        add_pattern(inits, np.exp(-lam * t))
        add_pattern(inits, np.exp(-lam * (1.0 - t)))

    # Gaussian bumps.
    for c in (0.25, 0.5, 0.75):
        for sig in (0.1, 0.25):
            add_pattern(inits, 0.3 + 0.7 * np.exp(-((t - c) ** 2) / (2.0 * sig * sig)))

    # Random smooth low-frequency functions.
    for _ in range(8):
        val = 1.0
        for f in range(1, 9):
            amp = rng.uniform(0.0, 1.0) / (float(f) ** 1.3)
            phase = rng.uniform(0.0, 2.0 * np.pi)
            val = val + amp * np.cos(2.0 * np.pi * f * t + phase)
        add_pattern(inits, val)

    # Lightly smoothed random positive functions.
    for _ in range(3):
        r = rng.uniform(0.3, 1.0, n)
        kernel = np.ones(9, dtype=np.float64) / 9.0
        r = np.convolve(r, kernel, mode='same')
        add_pattern(inits, r)

    return inits


def build_conv_matrix(x):
    """
    Build the dense matrix A such that (A y)_k = (x*y)_k for full convolution.
    A[k,j] = x[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(x)
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = x
    return A


def primal_lp_solution(x):
    """
    Solve max sum(y) s.t. x_lp*y <= 1, y>=0, where x_lp = x / max(x*x).
    Returns normalized direction g=y/sum(y) and S=sum(y).
    """
    n = len(x)
    if n == 0:
        return None, 0.0
    b = fast_conv(x)
    M = float(b.max())
    if M <= 1e-15:
        return None, 0.0
    x_lp = x / M
    A = build_conv_matrix(x_lp)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.ones(2 * n - 1, dtype=np.float64)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n,
                      method='highs', options=_LP_OPTIONS)
    except Exception:
        return None, 0.0
    if not res.success:
        return None, 0.0
    y = np.asarray(res.x, dtype=np.float64)
    if y is None or not np.all(np.isfinite(y)):
        return None, 0.0
    y[y < 0.0] = 0.0
    S = float(y.sum())
    if S <= 1e-15:
        return None, 0.0
    return y, S


def dual_lp_solution(x):
    """
    Solve the dual covering LP:
      min sum(lambda) s.t. A_lp^T lambda >= 1, lambda >= 0.
    Return reciprocal direction 1/(A_lp^T lambda), normalized.
    """
    n = len(x)
    if n == 0:
        return None, 0.0
    b = fast_conv(x)
    M = float(b.max())
    if M <= 1e-15:
        return None, 0.0
    x_lp = x / M
    A = build_conv_matrix(x_lp)
    AT = A.T
    m = 2 * n - 1
    c = np.ones(m, dtype=np.float64)
    b_ub = -np.ones(n, dtype=np.float64)
    try:
        res = linprog(c, A_ub=-AT, b_ub=b_ub, bounds=[(0.0, None)] * m,
                      method='highs', options=_LP_OPTIONS)
    except Exception:
        return None, 0.0
    if not res.success:
        return None, 0.0
    lam = np.asarray(res.x, dtype=np.float64)
    if lam is None or not np.all(np.isfinite(lam)):
        return None, 0.0
    lam[lam < 0.0] = 0.0
    r = AT @ lam
    if not np.all(np.isfinite(r)):
        return None, 0.0
    r = np.where(r > 1.0, r, 1.0)
    xnew = 1.0 / r
    xnew[xnew < 0.0] = 0.0
    s = float(xnew.sum())
    if s <= 1e-15:
        return None, 0.0
    return xnew / s, float(np.sum(lam))


def line_search_weights(a, b, base_score):
    weights = (0.1, 0.25, 0.5, 0.75, 0.9, 1.0)
    best_cand = None
    best_score = base_score
    for w in weights:
        c = (1.0 - w) * a + w * b
        c[c < 0.0] = 0.0
        s = float(c.sum())
        if s <= 1e-15:
            continue
        c = c / s
        sc = compute_score(c)
        if sc < best_score - 1e-13:
            best_score = sc
            best_cand = c
    return best_cand, best_score


def primal_lp_iteration(x, deadline):
    x = normalize_sequence(x)
    best = x.copy()
    best_score = compute_score(x)
    n = len(x)
    no_improve = 0
    it = 0
    last_print = best_score

    while time.monotonic() < deadline:
        if n > 1500:
            break
        if it > 20000:
            break

        y, S = primal_lp_solution(x)
        if y is None:
            break
        g = y / S
        cand, sc = line_search_weights(x, g, best_score)
        if cand is None:
            no_improve += 1
            if no_improve >= 25:
                break
        else:
            x = cand
            best = cand
            best_score = sc
            no_improve = 0
            if best_score < last_print - 1e-5 or it % 100 == 0:
                print(f"      primal LP it={it} S={S:.4f} J={best_score:.6f}", flush=True)
                last_print = best_score
        it += 1

    return best, best_score

def lp_polish(x, deadline):
    x = normalize_sequence(x)
    best = x.copy()
    best_score = compute_score(x)
    n = len(x)
    if n == 0 or time.monotonic() >= deadline:
        return best, best_score

    rem = deadline - time.monotonic()
    if rem <= 5:
        return best, best_score

    # Primal LP first: usually gives the strongest initial descent.
    primal_end = time.monotonic() + min(rem * 0.65, 300.0 if n <= 600 else 150.0)
    xp, sp = primal_lp_iteration(x, primal_end)

    if time.monotonic() < deadline - 10:

        if sd < best_score:
            best = xd
            best_score = sd

def softmax_tail(t, n):
    """
    Softmax with y_0 fixed to 0: logits are [0, t...].
    """
    t = np.asarray(t, dtype=np.float64).ravel()
    if n <= 1:
        return np.ones(1, dtype=np.float64)
    if t.size != n - 1:
        if t.size >= n - 1:
            t = t[:n - 1]
        else:
            t = np.concatenate([t, np.zeros(n - 1 - t.size, dtype=np.float64)])
    if not np.all(np.isfinite(t)):
        t = np.zeros(n - 1, dtype=np.float64)

    if t.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)

    m = max(0.0, float(np.max(t)))
    z0 = float(np.exp(-m))
    zt = np.exp(t - m)
    s = z0 + float(zt.sum())
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)

    a = np.empty(n, dtype=np.float64)
    a[0] = z0 / s
    a[1:] = zt / s
    return a


def logit_from_x(x):
    x = normalize_sequence(x)
    n = len(x)
    if n <= 1:
        return np.empty(0, dtype=np.float64)
    a = np.maximum(x, 1e-12)
    t = np.log(a)
    t = t - t[0]
    t = np.clip(t, -50.0, 50.0)
    return t[1:].copy()


def soft_lse_info(a, beta):
    b = fast_conv(a)
    if b.size == 0:
        return 0.0, np.empty(0, dtype=np.float64), 0.0
    bmax = float(b.max())
    shifted = beta * (b - bmax)
    shifted = np.clip(shifted, -745.0, 50.0)
    exp_shift = np.exp(shifted)
    sum_exp = float(exp_shift.sum())
    if sum_exp <= 0.0 or not np.isfinite(sum_exp):
        p = np.full(b.size, 1.0 / float(b.size), dtype=np.float64)
        lse = bmax
    else:
        p = exp_shift / sum_exp
        lse = bmax + float(np.log(sum_exp)) / float(beta)
    return bmax, p, float(lse)


def smooth_obj_logit(t, beta):
    t = np.asarray(t, dtype=np.float64).ravel()
    n = t.size + 1
    a = softmax_tail(t, n)
    _, _, lse = soft_lse_info(a, beta)
    return float(2.0 * n * lse)


def smooth_grad_logit(t, beta):
    t = np.asarray(t, dtype=np.float64).ravel()
    n = t.size + 1
    a = softmax_tail(t, n)
    _, p, _ = soft_lse_info(a, beta)
    ga = correlate_weights(p, a)
    dot = float(np.dot(ga, a))
    gfull = 2.0 * n * a * (ga - a * dot)
    gfull[~np.isfinite(gfull)] = 0.0
    return gfull[1:]


def smooth_refine_logit(x, deadline, betas=None, max_iters=200):
    if betas is None:
        betas = (100.0, 1000.0, 5000.0, 20000.0)

    x = normalize_sequence(x)
    best = x.copy()
    best_score = compute_score(x)
    n = len(x)
    if n <= 1 or time.monotonic() >= deadline:
        return best, best_score

    bounds = [(-50.0, 50.0)] * (n - 1)
    t_cur = logit_from_x(best)

    for beta in betas:
        if time.monotonic() >= deadline:
            break
        for attempt in range(2):
            if time.monotonic() >= deadline:
                break
            try:
                res = minimize(
                    smooth_obj_logit,
                    t_cur.copy(),
                    args=(beta,),
                    jac=smooth_grad_logit,
                    method='L-BFGS-B',
                    bounds=bounds,
                    options={
                        'maxiter': max_iters,
                        'ftol': 1e-14,
                        'gtol': 1e-8,
                        'maxls': 20,
                    }
                )
            except Exception:
                break

            if res is not None and getattr(res, 'x', None) is not None and np.all(np.isfinite(res.x)):
                t_new = np.clip(res.x, -50.0, 50.0)
            else:
                t_new = t_cur

            cand = softmax_tail(t_new, n)
            cand = normalize_sequence(cand)
            sc = compute_score(cand)

            if sc < best_score - 1e-13:
                best = cand
                best_score = sc
            t_cur = logit_from_x(best)

            if getattr(res, 'status', 1) == 0:
                break

    return best, best_score


def fast_smooth_optimize(x, deadline, betas=(1000.0, 5000.0, 20000.0), max_iters=150):
    return smooth_refine_logit(x, deadline, betas=betas, max_iters=max_iters)

        if sf < best_score:

            best_score = sf

    return best, best_score


def update_best(best, best_score, cand, cand_score, top_list=None):
    if cand is None:
        return best, best_score
    if not np.isfinite(cand_score):

    if not np.all(np.isfinite(cand)):
        return best, best_score

    if cand_score < best_score - 1e-12:

        top_list.append((float(cand_score), len(top_list), cand.copy()))
        top_list.sort()
        if len(top_list) > 3:
            top_list.pop()

    return best, best_score

        extra = [parent_len // 2, int(1.5 * parent_len), parent_len * 2]
        for e in extra:
            if e >= 64 and e <= 1536 and e not in lengths:
                lengths.append(e)
        for e in base:
            if e <= 1536 and e not in lengths:
                lengths.append(e)
    else:
        lengths = [512, 384, 768, 1024, 256]

    out = []
    for n in lengths:
        if n not in out:
            out.append(n)
    return out[:8]


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    start = time.monotonic()
    budget = float(budget_s)
    deadline = start + max(60.0, budget - 45.0)
    rng = np.random.default_rng(seed)

    parent = load_parent()
    best = None
    best_score = float('inf')
    top_list = []

    if parent is not None:
        ps = compute_score(parent)
        best, best_score = update_best(best, best_score, parent, ps, top_list)
        print(f"Parent J={best_score:.6f} n={len(parent)}", flush=True)

    if parent is not None:
        lengths = get_lengths(parent.size)
    else:
        lengths = [512, 384, 768, 1024, 256]

    candidates = []
    counter = 0

    if parent is not None:
        candidates.append((best_score, counter, parent.size, parent.copy()))
        counter += 1

    now = time.monotonic()
    while now < deadline - 120:
        for n in lengths:
            if time.monotonic() > deadline - 120:
                break
            inits = generate_initializations(n, rng, parent)
            for a in inits:
                sc = compute_score(a)
                if np.isfinite(sc):
                    candidates.append((float(sc), counter, n, a))
                    counter += 1
        if len(candidates) >= 220:
            break

    candidates.sort()

    full_candidates = []
    count_by_length = {}
    for sc, _, n, a in candidates:
        if len(full_candidates) >= 10:
            break
        c = count_by_length.get(n, 0)
        if c < 2:
            full_candidates.append((sc, a, n))
            count_by_length[n] = c + 1

    # Main multi-basin local optimization.
    if time.monotonic() < deadline:
        work_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.70
        for idx, (sc, a, n) in enumerate(full_candidates):
            if time.monotonic() >= work_deadline:
                break
            rem = work_deadline - time.monotonic()
            if rem <= 10:
                break

            if idx == 0:
                cap = 300.0 if n <= 512 else 180.0
            else:
                cap = 120.0 if n <= 640 else 80.0

            rem_count = len(full_candidates) - idx
            alloc = min(rem / max(1, rem_count), cap)
            if alloc < 15.0 and rem > 30.0:
                alloc = 15.0
            end = time.monotonic() + min(rem, alloc)

            do_lp = (n <= 1024)
            print(f"Optimizing basin idx={idx} n={n} init_J={sc:.6f} time={alloc:.1f}s", flush=True)
            x, s = local_optimize(a, end, do_lp=do_lp)
            best, best_score = update_best(best, best_score, x, s, top_list)
            if best is not None:
                print(f"  basin done J={best_score:.6f} n={len(best)}", flush=True)

    # Perturbation search around the current top basins.
    if best is not None and time.monotonic() < deadline - 60:
        pert_deadline = time.monotonic() + min(300.0, (deadline - time.monotonic()) * 0.5)
        base_arr = top_list[0][2].copy() if top_list else best.copy()
        second_arr = top_list[1][2].copy() if len(top_list) > 1 else None
        trial = 0

        while time.monotonic() < pert_deadline:
            n_best = base_arr.size
            if n_best <= 0:
                break

            mean = float(base_arr.mean())
            if trial % 2 == 0 and mean > 0.0:
                sigma_std = [0.2, 0.5, 1.0, 2.0, 0.4, 0.8][trial % 6]
                a = base_arr + rng.normal(0.0, sigma_std * mean, n_best)
            else:
                sigma_log = [0.3, 0.6, 1.0, 0.2, 0.5][trial % 5]
                a = base_arr * np.exp(rng.normal(0.0, sigma_log, n_best))
                if second_arr is not None and trial % 3 == 2:
                    w = 0.3
                    a = (1.0 - w) * base_arr + w * second_arr

            a = normalize_sequence(a)
            if time.monotonic() >= pert_deadline:
                break

            slice_time = 20.0 if n_best <= 768 else 40.0
            end = time.monotonic() + slice_time
            if end > pert_deadline:
                end = pert_deadline
            if end - time.monotonic() < 5.0:
                break

            x, s = fast_smooth_optimize(a, end)
            best, best_score = update_best(best, best_score, x, s, top_list)
            if best is not None:
                base_arr = top_list[0][2].copy() if top_list else best.copy()
            if second_arr is not None:
                second_arr = top_list[1][2].copy() if len(top_list) > 1 else None

            if best is not None and s < best_score + 1e-10:
                print(f"  perturbation {trial} J={best_score:.6f}", flush=True)

            trial += 1
            if trial > 200:
                break

    # Final polish on the global best.
    if best is not None and time.monotonic() < deadline - 15:
        n = len(best)
        rem = deadline - time.monotonic()
        if n <= 1024 and rem > 60:
            lp_end = time.monotonic() + min(rem * 0.40, 150.0)
            x, s = lp_polish(best, lp_end)
            best, best_score = update_best(best, best_score, x, s, top_list)

        if time.monotonic() < deadline - 5:
            x, s = smooth_refine_logit(
                best,
                deadline - 2.0,
                betas=(500.0, 2500.0, 10000.0, 30000.0),
                max_iters=250
            )
            best, best_score = update_best(best, best_score, x, s, top_list)

        print(f"Pre-final J={best_score:.6f} n={len(best)}", flush=True)

    if best is None:
        best = np.ones(512, dtype=np.float64) / 512.0

    # Safety: if the parent is still better by our internal metric, keep it.
    if parent is not None:
        ps = compute_score(parent)
        bs = compute_score(best)
        if ps < bs - 1e-12:
            best = parent
            best_score = ps

    final = normalize_sequence(best)
    final = np.asarray(final, dtype=np.float64)
    if final.size == 0:
        final = np.ones(512, dtype=np.float64) / 512.0

    internal_final = compute_score(final)
    print(f"Final internal J={internal_final:.6f} n={len(final)}", flush=True)

    # Cross-check / final choice using the actual evaluator when available.
    try:
        ev_best = evaluate_sequence([float(x) for x in final])
        if np.isfinite(ev_best):
            print(f"Final evaluator J={ev_best:.6f}", flush=True)
        if parent is not None:
            ev_parent = evaluate_sequence([float(x) for x in parent])
            if np.isfinite(ev_parent):
                if (not np.isfinite(ev_best)) or ev_parent < ev_best:
                    final = normalize_sequence(parent)
                    print(f"Selected parent from evaluator J={ev_parent:.6f} n={len(final)}", flush=True)
    except Exception:
        pass

    final = normalize_sequence(final)
    return [float(x) for x in final]
