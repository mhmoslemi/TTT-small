"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Strategies:
1. LP alternation for coarse optimization
2. 1D convex coordinate descent with O(n) evaluation per step
3. Subgradient polishing
4. Multiple initializations and sequence lengths
"""

import numpy as np
from scipy.optimize import linprog
import time

# Global deadline (set in propose_candidate)
_deadline = 1500.0

def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    """Build constraint matrix: A[k,j] gives conv[k] = sum_j A[k,j]*a[j]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A

def solve_lp_g(a, M_relax=1.0):
    """Solve LP: maximize sum(g) s.t. A@conv_row(a) * g <= M, g >= 0.
    Actually: s.t. (a * g)[k] <= M for all k, where a*g means convolution.
    """
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv)) * M_relax
    # A[k, j] = a[k-j] if valid, else 0
    # Then (a*g)[k] = sum_j A[k,j] * g[j]
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_alternate(a_init, max_iter=200, relax=1.0):
    """LP alternating optimization."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s  # normalize to sum 1
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        if time.monotonic() > _deadline:
            break
        g = solve_lp_g(a, M_relax=relax)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum  # normalize
        
        # Blend
        t = 0.5
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            a = g
        else:
            a = a_new / s
        
        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj

                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
                        improved = True

            break

"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses efficient O(n) coordinate descent with ternary search, LP alternation,
and subgradient polishing. Key improvement: exact 1D minimization per coordinate
exploiting the structure of single-coordinate changes in the convolution.
"""

import numpy as np
from scipy.optimize import linprog
import time

_deadline = 1500.0


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix: (a*g)[k] = sum_j A[k,j]*g[j]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_g(a, M_relax=1.0):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    conv = np.convolve(a, a)
    M = float(np.max(conv)) * M_relax
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_alternate(a_init, max_iter=200, relax=1.0, blend=0.5):
    """LP alternating optimization with blending."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for it in range(max_iter):
        if time.monotonic() > _deadline:
            break
        g = solve_lp_g(a, M_relax=relax)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum

        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            a = g
        else:
            a = a_new / s

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def f_eval_inlined(active_conv, coeff, has_diag, diag_base, inactive_max,
                   n, sum_base, old_val, x):
    """Compute objective value for a given x (inlined to avoid closure)."""
    sn = sum_base + x
    if sn < 1e-15:
        return 1e10
    max_active = float(np.max(active_conv + 2.0 * (x - old_val) * coeff))
    if has_diag:
        d = diag_base + x * x - old_val * old_val
        if d > max_active:
            max_active = d
    if inactive_max > max_active:
        max_active = inactive_max
    return 2.0 * n * max_active / (sn * sn)


def coord_descent_ternary(a_init, n_passes=200, n_ternary=40):
    """
    Coordinate descent with ternary search on each coordinate.
    Uses O(n) evaluation per ternary step by exploiting convolution structure.
    """
    global _deadline
    a = a_init.copy().astype(np.float64)
    n = len(a)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for pass_idx in range(n_passes):
        if time.monotonic() > _deadline:
            break

        conv = np.convolve(a, a)
        pass_best_obj = compute_obj(a)
        any_improved = False

        for j in range(n):
            if time.monotonic() > _deadline:
                return best_a, best_obj

            old_val = a[j]
            sum_base = 1.0 - old_val

            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)

            # Inactive max
            if k_lo == 0 and k_hi == 2 * n - 2:
                inactive_max = -np.inf
            else:
                inactive_parts = []
                if k_lo > 0:
                    inactive_parts.append(float(np.max(conv[:k_lo])))
                if k_hi < 2 * n - 2:
                    inactive_parts.append(float(np.max(conv[k_hi + 1:])))
                inactive_max = max(inactive_parts) if inactive_parts else -np.inf

            diag_k = 2 * j
            has_diag = (k_lo <= diag_k <= k_hi)

            active_ks = np.arange(k_lo, k_hi + 1)
            active_conv = conv[active_ks].copy()
            coeff = a[active_ks - j].copy()

            if has_diag:
                diag_pos = diag_k - k_lo
                coeff[diag_pos] = 0.0
                diag_base = active_conv[diag_pos]
            else:
                diag_base = -np.inf

            # Ternary search
            x_lo = 0.0
            x_hi = max(1.0, 5.0 * old_val + 0.1)

            for _ in range(n_ternary):
                m1 = x_lo + (x_hi - x_lo) / 3.0
                m2 = x_hi - (x_hi - x_lo) / 3.0

                # Evaluate f(m1)
                sn1 = sum_base + m1
                max_act1 = float(np.max(active_conv + 2.0 * (m1 - old_val) * coeff))
                if has_diag:
                    d1 = diag_base + m1 * m1 - old_val * old_val
                    if d1 > max_act1:
                        max_act1 = d1
                if inactive_max > max_act1:
                    max_act1 = inactive_max
                f1 = 2.0 * n * max_act1 / (sn1 * sn1)

                # Evaluate f(m2)
                sn2 = sum_base + m2
                max_act2 = float(np.max(active_conv + 2.0 * (m2 - old_val) * coeff))
                if has_diag:
                    d2 = diag_base + m2 * m2 - old_val * old_val
                    if d2 > max_act2:
                        max_act2 = d2
                if inactive_max > max_act2:
                    max_act2 = inactive_max
                f2 = 2.0 * n * max_act2 / (sn2 * sn2)

                if f1 < f2:
                    x_hi = m2
                else:
                    x_lo = m1

            x_opt = (x_lo + x_hi) / 2.0

            # Check boundary x=0
            sn0 = sum_base
            if sn0 > 1e-15:
                max_act0 = float(np.max(active_conv - 2.0 * old_val * coeff))
                if has_diag:
                    d0 = diag_base - old_val * old_val
                    if d0 > max_act0:
                        max_act0 = d0
                if inactive_max > max_act0:
                    max_act0 = inactive_max
                f0 = 2.0 * n * max_act0 / (sn0 * sn0)
            else:
                f0 = 1e10

            # Evaluate at x_opt
            sn_opt = sum_base + x_opt
            if sn_opt > 1e-15:
                max_act_opt = float(np.max(active_conv + 2.0 * (x_opt - old_val) * coeff))
                if has_diag:
                    d_opt = diag_base + x_opt * x_opt - old_val * old_val
                    if d_opt > max_act_opt:
                        max_act_opt = d_opt
                if inactive_max > max_act_opt:
                    max_act_opt = inactive_max
                f_opt = 2.0 * n * max_act_opt / (sn_opt * sn_opt)
            else:
                f_opt = 1e10

            if f0 < f_opt:
                x_opt = 0.0

            if x_opt < 1e-14:
                x_opt = 0.0

            if abs(x_opt - old_val) > 1e-10:
                a[j] = x_opt
                s = np.sum(a)
                if s < 1e-10:
                    a[j] = old_val
                else:
                    a = a / s
                    obj = compute_obj(a)
                    if obj < pass_best_obj:
                        pass_best_obj = obj
                        any_improved = True
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()

        if not any_improved:
            break

    return best_a, best_obj


def subgradient_polish(a_init, n_steps=3000, momentum=0.9):
    """Subgradient method with momentum for polishing."""
    global _deadline
    a = a_init.copy().astype(np.float64)
    n = len(a)
    a = a / np.sum(a)

    best_a = a.copy()
    best_obj = compute_obj(best_a)
    v = np.zeros(n)

    for step in range(n_steps):
        if time.monotonic() > _deadline:
            break

        h = np.convolve(a, a)
        max_h = float(np.max(h))
        if max_h < 1e-15:
            break

        tol = max_h * (0.0005 + 0.005 * (step / n_steps))
        active = h >= max_h - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            active = h >= max_h - 1e-10
            n_active = max(1, int(np.sum(active)))

        # Compute subgradient
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            vals = a[k - j_end:k - j_start + 1][::-1]
            grad[j_start:j_end + 1] += 2.0 * vals
            if k % 2 == 0 and 0 <= k // 2 < n:
                grad[k // 2] -= a[k // 2]
        grad /= n_active

        # Polyak-like step size
        step_size = 1.0 / (1.0 + 0.05 * step)

        v = momentum * v - step_size * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new

    return best_a, best_obj


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 8))
    if choice == 0:
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0, np.cos(2 * np.pi * freq * np.arange(n) / n + phase))
    elif choice == 1:
        mid = n // 2
        a = np.ones(n)
        a[:mid] = rng.uniform(0.3, 2.0)
        a[mid:] = rng.uniform(0.3, 2.0)
    elif choice == 2:
        period = int(rng.integers(max(2, n // 40), max(3, n // 5)))
        pattern = np.maximum(0.01, rng.random(period))
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 3:
        a = np.exp(-rng.uniform(0.5, 3.0) * (np.arange(n) / n - 0.5) ** 2)
    elif choice == 4:
        a = np.abs(np.sin(np.arange(n) * rng.uniform(0.2, 1.5))) + 0.1
    elif choice == 5:
        # Bimodal
        mid = n // 2
        sigma = n * rng.uniform(0.05, 0.15)
        a = np.exp(-(np.arange(n) - mid * rng.uniform(0.3, 0.7))**2 / (2*sigma**2))
        a += np.exp(-(np.arange(n) - mid * rng.uniform(1.3, 1.7))**2 / (2*sigma**2))
    elif choice == 6:
        # Random with smoothing
        a = np.maximum(0.01, rng.random(n))
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    else:
        # Exponential decay from both ends
        a = np.maximum(0.01, np.exp(-rng.uniform(1, 4) * np.minimum(
            np.arange(n), n - 1 - np.arange(n)) / n))

    s = np.sum(a)
    if s < 1e-10:
        a = np.random.RandomState(0).random(n)
        s = np.sum(a)
    return a / s
