"""
Optimized step function search using spectral initialization, 
adaptive smoothed subgradient, coordinate descent, and LP polishing.

import numpy as np
from scipy.optimize import linprog
import time

def compute_conv(a):
    """Compute autocorrelation conv(a,a) using FFT."""
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]

def compute_obj(a):
    """Compute objective."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def smoothed_gradient(conv, a, tau):
    """Compute smoothed gradient of max(conv) using log-sum-exp weights."""
    n = len(a)
    max_conv = np.max(conv)
    shifted = tau * (conv - max_conv)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        weights = np.zeros(len(conv))
        weights[int(np.argmax(conv))] = 1.0
    else:
        weights /= w_sum
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[:n]
    return grad

def subgradient_optimize(a, max_iter=20000, tau_start=10.0, tau_end=100.0,

    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))
    
    step = step_init
    momentum = np.zeros(n)
    beta = 0.9
    
    for it in range(max_iter):
        frac = it / max(1, max_iter)
        tau = tau_start + (tau_end - tau_start) * frac
        
        conv = compute_conv(a)
        current_max = float(np.max(conv))
        
        if current_max < best_max - 1e-10:
            best_max = current_max
            best_a = a.copy()
        
        grad = smoothed_gradient(conv, a, tau)
        grad_centered = grad - np.mean(grad)
        
        if use_momentum:
            momentum = beta * momentum + (1 - beta) * grad_centered
            direction = momentum
        else:

        # Adaptive step: scale by 1/||grad||
        grad_norm = np.linalg.norm(direction)
        if grad_norm > 1e-15:
            direction = direction / grad_norm
        
        a_new = a - step * direction
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        a = a_new

def coordinate_descent(a, max_iter=5000):

def build_A_matrix(a):
    """Build the Toeplitz constraint matrix for the LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A

def solve_lp_for_g(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None

    s = np.sum(a)

    best_obj = compute_obj(best_a)

            break

            stall += 1
            if stall > 10:
                break
            continue

        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)

"""
Optimized step function search using parametric initialization, top-k subgradient with
line search, adaptive LP mixing, and multi-scale exploration. Minimizes 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def compute_conv(a):
    """Compute full convolution conv(a,a) using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:2 * n - 1]


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = compute_conv(a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def topk_gradient(conv, a, k):
    """Compute gradient from top-k lags of convolution. Returns (gradient, top_k_indices)."""
    n = len(a)
    k = min(k, len(conv))
    top_idx = np.argsort(conv)[-k:]
    top_idx = np.sort(top_idx)
    # Gradient of sum_{k in top} conv(a,a)_k w.r.t. a
    # conv(a,a)_l = sum_j a[j]*a[l-j], grad_i = sum_{j: valid} [a[j if l-j=i else a[l-i]]]
    # More precisely: d/d a_i [conv(a,a)_l] = sum_j a[j] * (delta_{j+i,l} + delta_{j+l,i}) / appropriate
    # Actually: conv(a,a)_l = sum_j a_j * a_{l-j}
    # d/d a_i = sum_j (delta_{j,i} * a_{l-j} + a_j * delta_{l-j,i})
    #          = a_{l-i} + a_{l-i} = 2*a_{l-i} (if 0<=l-i<n)
    #          But also if i appears as l-j, i.e., j = l-i, we get a_{l-i}
    # Wait: d/d a_i [sum_j a_j * a_{l-j}] = a_{l-i} + a_{l-i} = 2*a_{l-i} if valid, else a_{l-i} if i=l-i
    # More carefully: = sum_j (delta_{j,i} a_{l-j} + a_j delta_{l-j,i})
    # = a_{l-i} + a_{l-i} if 0<=l-i<n and l-i != i
    # = a_{l-i} + a_{i} if l-i == i (i.e., l=2i), giving 2*a_i
    # So in all cases, grad_i for lag l is: 2*a_{l-i} if 0<=l-i<n, else 0
    # (The factor of 2 accounts for both terms in the product rule)
    # Actually let me just use the correlate approach for safety:
    weights = np.zeros(len(conv))
    for idx in top_idx:
        weights[idx] = 1.0 / k
    c = np.correlate(weights, a, mode='full')
    grad = 2.0 * c[:n]
    return grad, top_idx


def subgradient_optimize(a, max_iter=10000, tau_start=10.0, tau_end=100.0,
                         step_init=0.05, use_topk=False, topk_k=5):
    """Smoothed subgradient / top-k subgradient optimization."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_max = float(np.max(compute_conv(a)))

    step = step_init
    momentum = np.zeros(n)
    beta = 0.85

    for it in range(max_iter):
        frac = it / max(1, max_iter)
        tau = tau_start + (tau_end - tau_start) * frac

        conv = compute_conv(a)
        current_max = float(np.max(conv))

        if current_max < best_max - 1e-12:
            best_max = current_max
            best_a = a.copy()

        if use_topk:
            k = max(3, int(topk_k * (1.0 + frac)))
            grad, _ = topk_gradient(conv, a, k)
        else:
            # Log-sum-exp smoothing
            max_conv = np.max(conv)
            shifted = tau * (conv - max_conv)
            shifted = np.clip(shifted, -500, 500)
            weights = np.exp(shifted)
            w_sum = np.sum(weights)
            if w_sum < 1e-300:
                weights = np.zeros(len(conv))
                weights[int(np.argmax(conv))] = 1.0
            else:
                weights /= w_sum
            c = np.correlate(weights, a, mode='full')
            grad = 2.0 * c[:n]

        grad_centered = grad - np.mean(grad)

        # Momentum
        momentum = beta * momentum + (1.0 - beta) * grad_centered

        # Normalize direction
        dir_norm = np.linalg.norm(momentum)
        if dir_norm < 1e-15:
            break
        direction = momentum / dir_norm

        # Line search (a few steps)
        best_step = step
        best_val = current_max
        for mult in [1.0, 0.5, 0.25, 2.0]:
            s_try = step * mult
            a_try = a - s_try * direction
            a_try = np.maximum(a_try, 0)
            s_new = np.sum(a_try)
            if s_new < 1e-10:
                continue
            a_try = a_try / s_new
            val = float(np.max(compute_conv(a_try)))
            if val < best_val:
                best_val = val
                best_step = s_try

        a_new = a - best_step * direction
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a = a_new / s_new

        # Decrease step
        step *= 0.9997

    return best_a, best_max


def build_A_matrix(a):
    """Build the Toeplitz constraint matrix for the LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def solve_lp_for_g(a, M):
    """Solve LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    A = build_A_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 15})
        if res.status == 0:
            return res.x, float(-res.fun)
    except Exception:
        pass
    return None, None


def lp_optimize(a, max_iter=30, t=0.3):
    """Alternating LP with adaptive mixing parameter."""
    n = len(a)
    a = a.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    stall = 0
    for it in range(max_iter):
        conv = compute_conv(a)
        M = float(np.max(conv))

        g, sum_g = solve_lp_for_g(a, M)
        if g is None:
            break

        if sum_g <= 1.0 + 1e-5:
            stall += 1
            if stall > 15:
                break
            continue
        stall = 0

        g = g / sum_g

        # Adaptive t: try a few values
        best_t = t
        best_obj_t = compute_obj((1 - t) * a + t * g)
        for t_try in [t * 0.5, t * 1.5, 0.15, 0.5]:
            t_try = np.clip(t_try, 0.05, 0.9)
            a_try = (1.0 - t_try) * a + t_try * g
            a_try = np.maximum(a_try, 0)
            s_try = np.sum(a_try)
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            obj_try = compute_obj(a_try)
            if obj_try < best_obj_t:
                best_obj_t = obj_try
                best_t = t_try

        a_new = (1.0 - best_t) * a + best_t * g
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_obj


def parametric_search(n, seed, max_time=20.0):
    """Search over parametric cosine families."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    best_a = None
    best_obj = np.inf

    for num_h in [1, 2, 3, 4]:
        if time.monotonic() - start > max_time * 0.8:
            break

        dim = 1 + 2 * num_h

        for trial in range(2):
            if time.monotonic() - start > max_time * 0.9:
                break

            x0 = np.zeros(dim)
            x0[0] = 1.0
            for k in range(num_h):
                x0[1 + 2 * k] = rng.uniform(0.1, 0.5) / (k + 1)
                x0[2 + 2 * k] = rng.uniform(0, 2 * np.pi)

            def make_obj(ph):
                def param_obj(params):
                    offset = params[0]
                    x = np.arange(n)
                    a = np.full(n, offset)
                    for k in range(ph):
                        amp = params[1 + 2 * k]
                        phase = params[2 + 2 * k]
                        a += amp * np.cos(2 * np.pi * (k + 1) * x / n + phase)
                    a = np.maximum(a, 0)
                    s = np.sum(a)
                    if s < 1e-10:
                        return 1e6
                    a = a / s
                    return compute_obj(a)
                return param_obj

            obj_fn = make_obj(num_h)
            try:
                res = minimize(obj_fn, x0, method='Nelder-Mead',
                               options={'maxiter': 500, 'xatol': 1e-7, 'fatol': 1e-9})
                if res.fun < best_obj:
                    best_obj = res.fun
                    offset = res.x[0]
                    x = np.arange(n)
                    a = np.full(n, offset)
                    for k in range(num_h):
                        amp = res.x[1 + 2 * k]
                        phase = res.x[2 + 2 * k]
                        a += amp * np.cos(2 * np.pi * (k + 1) * x / n + phase)
                    a = np.maximum(a, 0)
                    s = np.sum(a)
                    if s > 1e-10:
                        best_a = a / s
            except Exception:
                pass

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def cosine_init(n, seed, num_harmonics=3):
    """Generate cosine-modulated initialization."""
    rng = np.random.default_rng(seed)
    x = np.arange(n)
    a = np.ones(n)
    for k in range(1, num_harmonics + 1):
        amp = rng.uniform(0.1, 0.4) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * k * x / n + phase)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def u_shaped_init(n, seed):
    """Generate U-shaped initialization."""
    rng = np.random.default_rng(seed)
    x = np.arange(n)
    c = rng.uniform(0.2, 0.7)
    a = 1.0 + c * np.cos(np.pi * x / max(1, n - 1))
    a += rng.normal(0, 0.03, n)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def random_init(n, seed):
    """Random initialization with some structure."""
    rng = np.random.default_rng(seed)
    choice = rng.integers(0, 4)
    x = np.arange(n)
    if choice == 0:
        a = 0.5 * (1 + np.cos(2 * np.pi * rng.uniform(0.5, 3) * x / n + rng.uniform(0, 2 * np.pi)))
    elif choice == 1:
        a = np.exp(-rng.uniform(0.5, 3.0) * np.abs(x - (n - 1) / 2.0) / (n / 2.0))
    elif choice == 2:
        a = np.ones(n)
        mask = rng.random(n) > 0.5
        a[mask] = rng.uniform(0.3, 0.8)
    else:
        a = rng.random(n)
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def run_optimization(n, seed, max_time=40.0):
    """Run full optimization pipeline for a given n."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    best_a = None
    best_obj = np.inf

    # Generate diverse initializations
    inits = []
    inits.append(cosine_init(n, seed, 3))
    inits.append(cosine_init(n, seed + 100, 5))
    inits.append(u_shaped_init(n, seed + 200))
    inits.append(random_init(n, seed + 300))
    inits.append(np.ones(n) / n)

    # Try parent if available and matching length
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if len(parent) == n:
            parent = np.maximum(parent, 0)
            s = np.sum(parent)
            if s > 1e-10:
                inits.append(parent / s)
    except Exception:
        pass

    # Parametric search as one initialization strategy
    if max_time > 15:
        pa, p_obj = parametric_search(n, seed + 500, max_time=min(15, max_time * 0.3))
        inits.append(pa)
        if p_obj < best_obj:
            best_obj = p_obj
            best_a = pa.copy()

    # Determine number of full optimization runs
    remaining = max_time - (time.monotonic() - start)
    n_runs = max(2, min(5, int(remaining / 8)))

    for init_idx in range(min(n_runs, len(inits))):
        if time.monotonic() - start > max_time - 5:
            break

        a = inits[init_idx].copy()
        s = np.sum(a)
        if s < 1e-10:
            continue
        a = a / s

        a_cur = a
        obj_cur = compute_obj(a_cur)

        # Phase 1: Subgradient (smoothed)
        remaining = max_time - (time.monotonic() - start)
        if remaining > 3:
            sg_iters = int(min(8000, max(500, remaining * 1.5)))
            a_sg, _ = subgradient_optimize(a_cur, max_iter=sg_iters,
                                           tau_start=10.0, tau_end=60.0,
                                           step_init=0.05)
            obj_sg = compute_obj(a_sg)
            if obj_sg < obj_cur:
                a_cur = a_sg
                obj_cur = obj_sg

        # Phase 2: Top-k subgradient (sharper)
        remaining = max_time - (time.monotonic() - start)
        if remaining > 3:
            sg_iters = int(min(5000, max(300, remaining * 1.5)))
            a_sg2, _ = subgradient_optimize(a_cur, max_iter=sg_iters,
                                            tau_start=20.0, tau_end=80.0,
                                            step_init=0.03, use_topk=True, topk_k=5)
            obj_sg2 = compute_obj(a_sg2)
            if obj_sg2 < obj_cur:
                a_cur = a_sg2
                obj_cur = obj_sg2

        # Phase 3: LP polishing
        remaining = max_time - (time.monotonic() - start)
        if remaining > 10 and n <= 350:
            a_lp, obj_lp = lp_optimize(a_cur, max_iter=25, t=0.3)
            if obj_lp < obj_cur:
                a_cur = a_lp
                obj_cur = obj_lp

        # Phase 4: Final subgradient
        remaining = max_time - (time.monotonic() - start)
        if remaining > 3:
            sg_iters = int(min(5000, max(200, remaining * 2)))
            a_pol, _ = subgradient_optimize(a_cur, max_iter=sg_iters,
                                            tau_start=40.0, tau_end=150.0,
                                            step_init=0.02)
            obj_pol = compute_obj(a_pol)
            if obj_pol < obj_cur:
                a_cur = a_pol
                obj_cur = obj_pol

        if obj_cur < best_obj:
            best_obj = obj_cur
            best_a = a_cur.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_obj(best_a)

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500):
    """
    Search for best step function sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses parametric/cosine initialization, adaptive subgradient with line search,
    top-k focused gradients, and LP polishing. Explores n in [60, 450].
    """
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 25.0

    best_seq = None
    best_score = np.inf

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
            best_score = obj
            best_seq = parent.copy()
    except Exception:
        pass

    # Sequence lengths to explore
    n_values = [80, 100, 120, 150, 175, 200, 225, 250, 275, 300, 350, 400, 450]

    if best_seq is not None:
        n_parent = len(best_seq)
        extra = [n_parent]
        if n_parent > 50:
            extra.append(n_parent - 25)
        if n_parent < 500:
            extra.append(n_parent + 25)
        n_values = sorted(set(n_values + extra))

    print(f"Exploring n values: {n_values}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - start_time:.0f}s")

    for n in n_values:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 12:
            break

        # Time allocation
        if n <= 100:
            alloc = min(20.0, remaining * 0.06)
        elif n <= 150:
            alloc = min(30.0, remaining * 0.08)
        elif n <= 250:
            alloc = min(50.0, remaining * 0.1)
        elif n <= 350:
            alloc = min(60.0, remaining * 0.1)
        else:
            alloc = min(70.0, remaining * 0.1)

        alloc = max(10.0, alloc)
        print(f"N={n}: {remaining:.0f}s remaining, allocating {alloc:.0f}s")

        a_opt, obj = run_optimization(n, seed=seed + n * 104729, max_time=alloc)
        if obj < best_score:
            best_score = obj
            best_seq = a_opt.copy()
            print(f"  UPDATE: n={n}, obj={best_score:.6f}")
        else:
            print(f"  n={n}: obj={obj:.6f} (no improvement)")

    # Final intensive polishing
    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        n_best = len(best_seq)
        print(f"\n=== Polishing phase (n={n_best}), {remaining:.0f}s remaining ===")

        a_cur = best_seq.copy().astype(np.float64)
        s = np.sum(a_cur)
        if s > 1e-10:
            a_cur = a_cur / s

        polish_count = 0
        while time.monotonic() < deadline - 8:
            polish_count += 1
            remaining = deadline - time.monotonic()
            if remaining < 8:
                break

            # Pass 1: Smoothed subgradient
            sg_iters = int(min(10000, max(500, remaining * 1.5)))
            a_cur, _ = subgradient_optimize(a_cur, max_iter=sg_iters,
                                            tau_start=30.0, tau_end=120.0,
                                            step_init=0.02)
            obj = compute_obj(a_cur)
            if obj < best_score:
                best_score = obj
                best_seq = a_cur.copy()
                print(f"  Polish {polish_count} (sg): obj={best_score:.6f}")

            # Pass 2: Top-k subgradient
            remaining = deadline - time.monotonic()
            if remaining > 10:
                sg_iters = int(min(8000, max(300, remaining * 1.5)))
                a_cur, _ = subgradient_optimize(a_cur, max_iter=sg_iters,
                                                tau_start=50.0, tau_end=200.0,
                                                step_init=0.015, use_topk=True, topk_k=3)
                obj = compute_obj(a_cur)
                if obj < best_score:
                    best_score = obj
                    best_seq = a_cur.copy()
                    print(f"  Polish {polish_count} (topk): obj={best_score:.6f}")

            # Pass 3: LP
            remaining = deadline - time.monotonic()
            if remaining > 12 and n_best <= 350:
                a_lp, obj_lp = lp_optimize(a_cur, max_iter=30, t=0.25)
                if obj_lp < best_score:
                    best_score = obj_lp
                    best_seq = a_lp.copy()
                    a_cur = a_lp.copy().astype(np.float64)
                    print(f"  Polish {polish_count} (LP): obj={best_score:.6f}")

            # Pass 4: Perturbation + short optimization
            remaining = deadline - time.monotonic()
            if remaining > 15:
                rng_p = np.random.default_rng(int(time.monotonic() * 10000) + polish_count)
                for noise_scale in [0.05, 0.02, 0.1]:
                    a_pert = a_cur * np.exp(rng_p.normal(0, noise_scale, n_best))
                    a_pert = np.maximum(a_pert, 0)
                    s_p = np.sum(a_pert)
                    if s_p < 1e-10:
                        continue
                    a_pert = a_pert / s_p
                    # Quick subgradient from perturbed point
                    sg_iters = int(min(3000, max(200, (remaining * 0.3))))
                    a_from_pert, _ = subgradient_optimize(a_pert, max_iter=sg_iters,
                                                          tau_start=20.0, tau_end=80.0,
                                                          step_init=0.03)
                    obj_p = compute_obj(a_from_pert)
                    if obj_p < best_score:
                        best_score = obj_p
                        best_seq = a_from_pert.copy()
                        a_cur = a_from_pert.copy().astype(np.float64)
                        print(f"  Polish {polish_count} (perturb ns={noise_scale}): obj={best_score:.6f}")
                    remaining = deadline - time.monotonic()
                    if remaining < 10:
                        break

    # Fallback
    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    # Convert to list of floats and validate
    result = [float(x) for x in best_seq]
    validation = evaluate_sequence(result)
    if validation < np.inf:
        if abs(validation - best_score) > 1e-6:
            print(f"  Note: computed={best_score:.6f}, validated={validation:.6f}")
        best_score = validation

    return result
