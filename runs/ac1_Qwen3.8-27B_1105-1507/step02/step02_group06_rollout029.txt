"""
Minimizes 2n*max(autoconv)/sum(a)^2 using LP direction finding,
FFT-based smooth gradient descent with Nesterov acceleration,
mirror descent, and multiple structured initializations across sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_objective(f):
    """Compute 2n*max(conv)/sum^2 directly."""
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    total = 0
    for k in range(2 * n - 1):
        total += min(k, n - 1) - max(0, k - (n - 1)) + 1
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g)_k <= M for all k, g >= 0."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def smooth_max_grad_fft(f, beta):
    """FFT-based gradient of log-sum-exp(beta * autoconv) / beta."""
    n = len(f)
    conv = np.convolve(f, f)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    weights = np.exp(shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.zeros_like(conv)
        weights[conv.argmax()] = 1.0
    else:
        weights /= w_sum

    # grad[i] = 2 * sum_k weights[k] * f[k-i] for valid k-i
    # Use FFT cross-correlation
    N = 1
    while N < 4 * n:
        N *= 2
    F = np.fft.rfft(f, N)
    W = np.fft.rfft(weights, N)
    corr = np.fft.irfft(np.conj(W) * F, N)[:n]
    return 2.0 * corr


def smooth_max_value(f, beta):
    """Value of log-sum-exp(beta * autoconv) / beta."""
    conv = np.convolve(f, f)
    conv_max = conv.max()
    shifted = beta * (conv - conv_max)
    return conv_max + np.log(np.exp(shifted).sum()) / beta


def nesterov_subgradient_optimize(f, deadline, beta=50.0, lr_init=0.005,
                                   verbose_prefix=""):
    """Nesterov-accelerated smooth max gradient descent with projection to simplex."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    fn = f / S  # normalize to sum=1
    best_fn = fn.copy()
    best_score = compute_objective(best_fn)
    n = len(fn)

    x = fn.copy()
    v = np.zeros(n)
    t = 1.0
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < 200000:
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_t = (t_new - 1.0) / (t_new + 1.0)
        y = x + beta_t * (x - v if it > 0 else x)
        v = x.copy()

        grad = smooth_max_grad_fft(y, beta)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Adaptive LR based on L-smoothness estimate
        L = 2.0 * np.linalg.norm(y) * (2 * n + 1) / n + 1e-8
        step = 1.0 / (L * (t_new + 1))
        step = min(step, lr * S)

        x_new = y - step * grad
        # Project onto positive simplex (sum=1, all >= 0)
        x_new = np.maximum(x_new, 0.0)
        x_sum = x_new.sum()
        if x_sum < 1e-15:
            no_improve += 50
            break
        x_new /= x_sum

        score = compute_objective(x_new)
        it += 1
        t = t_new

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_fn = x_new.copy()
            no_improve = 0
            if it % 5000 == 0:
                print(f"    {verbose_prefix}nesterov {it}: score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 2000:
                break

    return best_fn * S, best_score


def mirror_descent_optimize(f, deadline, beta=80.0, lr_init=0.01, verbose_prefix=""):
    """Mirror descent (multiplicative weights) for smooth max optimization."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    fn = f / S
    best_fn = fn.copy()
    best_score = compute_objective(best_fn)
    n = len(fn)
    x = fn.copy()
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < 100000:
        grad = smooth_max_grad_fft(x, beta)
        # Multiplicative update: x_i * exp(-lr * grad_i)
        logits = np.log(x + 1e-30) - lr * grad
        logits -= logits.max()  # numerical stability
        x_new = np.exp(logits)
        x_new /= x_new.sum()

        score = compute_objective(x_new)
        it += 1

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_fn = x_new.copy()
            x = x_new
            no_improve = 0
            if it % 5000 == 0:
                print(f"    {verbose_prefix}mirror {it}: score={best_score:.6f}")
        else:
            no_improve += 1
            lr *= 0.95
            if no_improve > 5000 or lr < 1e-8:
                break

    return best_fn * S, best_score


def lp_optimize(f, deadline, step_init=0.1, verbose_prefix=""):
    """LP-based optimization with line search."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_objective(f)
    n = len(f)
    S = float(np.sum(f))
    step = step_init
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < 5000:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 1e-15:
            break

        g = solve_lp(f, M)
        if g is None:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g < 1e-15:
            no_improve += 1
            continue

        g = g * (S / sum_g)

        # Line search
        best_step_score = best_score
        best_step_f = None
        for alpha in [step, step * 0.6, step * 0.3, step * 0.15, step * 2.0]:
            alpha = min(alpha, 1.0)
            if alpha < 0.001:
                continue
            candidate = (1.0 - alpha) * f + alpha * g
            candidate = np.clip(candidate, 0.0, None)
            cand_sum = float(np.sum(candidate))
            if cand_sum < 1e-15:
                continue
            candidate = candidate * (S / cand_sum)
            cand_score = compute_objective(candidate)
            if np.isfinite(cand_score) and cand_score < best_step_score:
                best_step_score = cand_score
                best_step_f = candidate

        it += 1
        if best_step_f is not None:
            f = best_step_f
            best_f = f.copy()
            best_score = best_step_score
            no_improve = 0
            if it % 50 == 0:
                print(f"    {verbose_prefix}LP {it}: score={best_score:.6f} (step={step:.3f})")
            step = max(step * 0.997, 0.005)
        else:
            no_improve += 1
            step *= 0.6
            if step < 0.002 or no_improve > 50:
                break

    return best_f, best_score


def combined_optimize(f, deadline, verbose_prefix=""):
    """Combined optimization: mirror descent warmup, then LP, then polish."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_objective(f)

    # Phase A: Mirror descent (30% of time)
    t_a = time.monotonic() + (deadline - time.monotonic()) * 0.3
    f_a, s_a = mirror_descent_optimize(f, t_a, beta=60.0, lr_init=0.02,
                                        verbose_prefix=verbose_prefix + "mir ")
    if s_a < best_score:
        best_score = s_a
        best_f = f_a.copy()

    # Phase B: Nesterov subgradient (20% of time)

                                              verbose_prefix=verbose_prefix + "nes ")
    if s_b < best_score:
        best_score = s_b
        best_f = f_b.copy()

    # Phase C: LP optimization (remaining time)
    f_c, s_c = lp_optimize(f_b, deadline, step_init=0.08,
                            verbose_prefix=verbose_prefix + "LP  ")
    if s_c < best_score:
        best_score = s_c
        best_f = f_c.copy()

    return best_f, best_score


def make_initializations(n, rng, parent_seq=None):
    """Create diverse initializations for length n."""
    inits = []
    idx = np.arange(n, dtype=np.float64)

    inits.append(('uniform', np.ones(n)))

    # Cosine taper
    cos_t = (1.0 + np.cos(np.pi * idx / max(n - 1, 1))) / 2.0
    inits.append(('cosine', np.maximum(cos_t, 0.01)))

    # Inverted cosine (peaks at edges)
    inv_cos = 1.0 - cos_t
    inits.append(('inv_cosine', np.maximum(inv_cos, 0.01)))

    # Two-peak (edges)
    edge_peak = np.exp(-0.5 * (idx / (n * 0.15)) ** 2) + np.exp(-0.5 * ((n - 1 - idx) / (n * 0.15)) ** 2)
    inits.append(('edge_peak', edge_peak))

    # Center peak
    center = (n - 1) / 2.0
    center_peak = np.exp(-0.5 * ((idx - center) / (n * 0.2)) ** 2)
    inits.append(('center_peak', center_peak + 0.01))

    # V-shape (low in middle, high at edges)
    v_shape = np.abs(2.0 * idx / (n - 1) - 1.0)
    inits.append(('v_shape', v_shape + 0.1))

    # Inverted-V (high in middle, low at edges)
    inv_v = 1.0 - np.abs(2.0 * idx / (n - 1) - 1.0)
    inits.append(('inv_v', inv_v + 0.1))

    # Harmonic: 1 + 0.3*cos(2*pi*i/(n-1))
    harm1 = 1.0 + 0.3 * np.cos(2.0 * np.pi * idx / max(n - 1, 1))
    inits.append(('harm1', np.maximum(harm1, 0.01)))

    # Harmonic: 1 + 0.5*cos(4*pi*i/(n-1))
    harm2 = 1.0 + 0.5 * np.cos(4.0 * np.pi * idx / max(n - 1, 1))
    inits.append(('harm2', np.maximum(harm2, 0.01)))

    # Sine (positive half)
    sine = np.sin(np.pi * idx / (n - 1))
    inits.append(('sine', np.maximum(sine, 0.01)))

    # Random
    r = rng.uniform(0.1, 2.0, size=n)
    inits.append(('random', r))

    # Random smooth (low-pass filtered random)
    r_smooth = rng.uniform(-1, 1, size=n)
    # Simple box filter
    kernel = np.ones(5) / 5.0
    r_f = np.convolve(r_smooth, kernel, mode='same')
    r_smooth = np.abs(r_f) + 0.3
    inits.append(('rand_smooth', r_smooth))

    # Parent interpolation
    if parent_seq is not None and len(parent_seq) > 0:
        x_old = np.linspace(0, 1, len(parent_seq))
        x_new = np.linspace(0, 1, n)
        parent_interp = np.interp(x_new, x_old, parent_seq)
        inits.append(('parent', np.maximum(parent_interp, 0.001)))

    return inits
