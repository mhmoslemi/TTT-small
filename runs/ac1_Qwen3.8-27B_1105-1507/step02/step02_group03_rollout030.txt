import numpy as np
from scipy.optimize import linprog
import time

def compute_score(f):
    """Compute normalized score: 2*n*max(conv)/sum^2"""
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)

def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s

def build_conv_matrix(f):
    """Build matrix A such that A @ g = np.convolve(f, g, mode='full')"""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        # (f*g)[k] = sum_j f[k-j] * g[j] for valid j
        # j must satisfy 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    return A

def lp_step(f, M=None, time_limit=2):
    """
    LP-based step: maximize sum(g) subject to (f*g)[k] <= M, g >= 0.
    Returns the normalized g.
    """
    n = len(f)
    if M is None:
        conv = np.convolve(f, f)
        M = conv.max()
    
    A = build_conv_matrix(f)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    
    if not res.success or res.x is None:
        return None
    return np.clip(res.x, 0, None)

    f = normalize(f)

    if s < 1e-15:
        return f.copy()
    return f / s

    n = len(f)

def build_A(f):
    """Build matrix A such that A @ g = np.convolve(f, g, mode='full')."""
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # f[j] contributes to positions k = j, j+1, ..., j+n-1 with weights f[k-j] = f[0], f[1], ...
        # Actually: (f*g)[k] = sum_j f[k-j]*g[j], so A[k,j] = f[k-j] when 0<=k-j<n
        # For fixed j, k-j ranges from 0 to n-1 as k ranges from j to j+n-1
        A[j:j+n, j] = f
    return A


def lp_find_g(f, M, time_limit=2):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = build_A(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': time_limit})
    if not res.success or res.x is None:
        return None
    return np.clip(res.x, 0, None)


def lp_refine(f, time_budget, rng, m_factor=0.999, blend_init=0.5, verbose=True):
    """Iterative LP refinement with adaptive M and line search."""
    n = len(f)
    f = normalize(f.copy().astype(np.float64))
    best_f = f.copy()
    best_score = score(f)
    t = blend_init
    start = time.monotonic()
    it_count = 0

    while time.monotonic() - start < time_budget:
        conv = np.convolve(f, f)
        M = conv.max()
        M_try = M * m_factor

        g = lp_find_g(f, M_try, time_limit=1.5)
        if g is None or g.sum() < 1e-10:
            g = lp_find_g(f, M, time_limit=1.5)
        if g is None or g.sum() < 1e-10:
            break

        g_norm = g / g.sum()
        sum_ratio = g.sum()  # since f is normalized to sum=1

        # Line search for best blend
        improved = False
        candidates = sorted(set([t, t * 0.5, min(1.0, t * 1.5), 0.8, 0.5, 0.3, 0.15]),
                           reverse=True)
        for blend in candidates:
            blend = min(1.0, max(0.005, blend))
            f_new = (1 - blend) * f + blend * g_norm
            f_new = np.clip(f_new, 0, None)
            s_new = f_new.sum()
            if s_new < 1e-10:
                continue
            f_new = f_new / s_new
            sc = score(f_new)
            if sc < best_score - 1e-14:
                best_f = f_new.copy()
                best_score = sc
                f = f_new
                t = min(1.0, t * 1.03)
                # Tighten M factor slightly over time
                m_factor = max(0.995, m_factor - 0.0001)
                improved = True
                it_count += 1
                break

        if not improved:
            t *= 0.6
            if t < 0.005:
                # Try perturbation to escape
                f_pert = f + rng.normal(0, 0.005, n)
                f_pert = np.clip(f_pert, 0, None)
                f_pert = normalize(f_pert)
                if score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = score(f)
                    t = 0.3
                    m_factor = max(0.99, m_factor - 0.005)
                else:
                    if verbose and it_count > 0:
                        print(f"    LP converged after {it_count} iterations, best={best_score:.6f}")
                    break

    if verbose:
        print(f"    LP done: {it_count} iters, score={best_score:.6f}, elapsed={time.monotonic()-start:.1f}s")
    return best_f, best_score


def pressure_iterate(f, n_iter=300, eta=0.12, rng=None):
    """Pressure-based multiplicative update to reduce convolution peaks."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(f)
    f = normalize(f.copy().astype(np.float64))
    best_f = f.copy()
    best_score = score(f)
    stall = 0

    for it in range(n_iter):
        conv = np.convolve(f, f)
        M = conv.max()

        # Compute pressure from top peaks
        pressure = np.zeros(n)
        sorted_idx = np.argsort(conv)[::-1]
        n_peaks = min(5, len(sorted_idx))
        for pk in range(n_peaks):
            k_star = sorted_idx[pk]
            weight = 1.0 / (pk + 1)
            j_lo = max(0, k_star - n + 1)
            j_hi = min(n - 1, k_star)
            for j in range(j_lo, j_hi + 1):
                pressure[j] += weight * f[k_star - j]

        if pressure.max() > 0:
            pressure /= pressure.max()

        eta_eff = eta * max(0.3, 1.0 / (1 + 0.005 * it))
        f_new = f * np.exp(-eta_eff * pressure)
        f_new = np.clip(f_new, 1e-15, None)
        f_new = normalize(f_new)

        sc = score(f_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_f = f_new.copy()
            f = f_new
            stall = 0
        else:
            stall += 1
            if stall > 60:
                # Perturb
                f = f * (1.0 + 0.03 * rng.normal(0, 1, n))
                f = np.clip(f, 1e-10, None)
                f = normalize(f)
                if score(f) < best_score:
                    best_f = f.copy()
                    best_score = score(f)
                stall = 0
                if stall > 200:
                    break

    return best_f, best_score


def projected_subgradient(f, n_iter=2000, rng=None):
    """Nesterov-accelerated projected subgradient on the simplex."""
    n = len(f)
    f = normalize(f.copy().astype(np.float64))
    y = f.copy()
    best_f = f.copy()
    best_score = score(f)
    t_k = 1.0

    for it in range(n_iter):
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        alpha = 2.0 / t_next

        # Subgradient at y
        conv = np.convolve(y, y)
        k_star = np.argmax(conv)
        grad = np.zeros(n)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        for j in range(j_lo, j_hi + 1):
            grad[j] = 2.0 * y[k_star - j]

        # Also add contribution from near-peaks for smoother gradient
        sorted_idx = np.argsort(conv)[::-1]
        peak_val = conv[sorted_idx[0]]
        for pk in range(1, min(3, len(sorted_idx))):
            if conv[sorted_idx[pk]] > 0.98 * peak_val:
                ks = sorted_idx[pk]
                lo = max(0, ks - n + 1)
                hi = min(n - 1, ks)
                for j in range(lo, hi + 1):
                    grad[j] += 0.5 * y[ks - j]

        # Nesterov step
        f_new = y - alpha * grad
        # Project onto simplex {x >= 0, sum(x) = 1}
        f_new = np.clip(f_new, 0, None)
        s = f_new.sum()
        if s < 1e-10:
            f_new = f.copy()
        else:
            f_new = f_new / s

        # Update y
        y = f_new + (t_k - 1.0) / t_next * (f_new - f)
        y = np.clip(y, 0, None)
        sy = y.sum()
        if sy > 1e-10:
            y = y / sy
        else:
            y = f_new.copy()

        f = f_new
        t_k = t_next

        sc = score(f)
        if sc < best_score - 1e-14:
            best_score = sc
            best_f = f.copy()

    return best_f, best_score


def coord_descent(f, n_sweeps=5, rng=None):
    """Coordinate descent with multiple trial factors."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(f)
    f = normalize(f.copy().astype(np.float64))
    best = score(f)
    factors = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0]

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            orig = f[i]
            if orig < 1e-12:
                continue
            best_val = orig
            best_local = best
            for factor in factors:
                f[i] = max(0.0, orig * factor)
                s = f.sum()
                if s < 1e-10:
                    f[i] = orig
                    continue
                sc = 2.0 * n * np.convolve(f, f).max() / (s * s)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_val = f[i]
            f[i] = best_val
            if best_local < best:
                best = best_local
        f = np.clip(f, 0, None)
        f = normalize(f)

    return f, best


def spectral_smooth(f, rng, n_iter=100):
    """Spectral method: iteratively flatten the magnitude spectrum of F^2."""
    n = len(f)
    f = normalize(f.copy().astype(np.float64))
    n_fft = 1 << (2 * n - 1).bit_length()
    best_f = f.copy()
    best_score = score(f)

    for it in range(n_iter):
        F = np.fft.rfft(f, n_fft)
        F2 = F * F  # spectrum of linear convolution
        f_new = np.fft.irfft(F2, n_fft)[:n]
        # This gives us something proportional to f*f, not f itself
        # Instead, let's work with F directly
        # We want |F| to be flat-ish
        mag = np.abs(F)
        # Target: geometric mean of magnitudes (excluding DC)
        dc = mag[0]
        non_dc = mag[1:]
        target_mag = np.mean(non_dc) if len(non_dc) > 0 else dc
        # Blend: move magnitude toward target
        blend = 0.3
        mag_new = (1 - blend) * mag + blend * target_mag
        mag_new[0] = dc  # keep DC fixed (sum = 1)
        phase = np.angle(F)
        F_new = mag_new * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, n_fft)[:n]
        f_new = np.clip(f_new, 1e-10, None)
        f_new = normalize(f_new)

        sc = score(f_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_f = f_new.copy()
            f = f_new

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1500):
    """
    Main search: LP-based refinement + pressure updates + projected subgradient
    + coordinate descent, starting from parent sequence and exploring n=512, 1024.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0

    best_overall = None
    best_score_overall = np.inf

    # Load parent
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and len(parent) > 0 and np.isfinite(parent).all():
            parent = np.clip(parent, 0, None)
            if parent.sum() > 0.01:
                parent = normalize(parent)
                ps = score(parent)
                print(f"Parent: n={len(parent)}, score={ps:.6f}")
                best_overall = parent.copy()
                best_score_overall = ps
    except Exception:
        pass

    n_parent = len(best_overall) if best_overall is not None else 512
    print(f"Starting optimization. Budget: {budget_s}s, target: 1.5027")

    # === Phase 1: Optimize at parent length ===
    n = n_parent
    remaining = deadline - time.monotonic()
    print(f"\n--- Phase 1: n={n} (remaining: {remaining:.0f}s) ---")

    if best_overall is not None and len(best_overall) == n:
        f0 = best_overall.copy()
    else:
        x = np.arange(n)
        f0 = 0.5 * (1 + np.cos(2 * np.pi * x / max(1, n - 1)))
        f0 += 0.05 * rng.normal(0, 1, n)
        f0 = np.clip(f0, 0.01, None)
        f0 = normalize(f0)

    # 1a: Spectral smoothing
    t_spec = min(20, remaining * 0.05)
    if t_spec > 3:
        f1, s1 = spectral_smooth(f0, rng, n_iter=max(50, int(t_spec * 10)))
        print(f"  Spectral: {s1:.6f}")
        if s1 < best_score_overall - 1e-10:
            best_score_overall = s1
            best_overall = f1.copy()
        f0 = f1

    # 1b: Pressure iteration
    t_press = min(60, remaining * 0.15)
    if t_press > 5:
        f2, s2 = pressure_iterate(f0, n_iter=max(100, int(t_press * 20)), eta=0.1, rng=rng)
        print(f"  Pressure: {s2:.6f}")
        if s2 < best_score_overall - 1e-10:
            best_score_overall = s2
            best_overall = f2.copy()
        f0 = f2

    # 1c: LP refinement (main workhorse)
    remaining = deadline - time.monotonic()
    t_lp1 = min(remaining * 0.35, 300)
    if t_lp1 > 15:
        f3, s3 = lp_refine(f0, t_lp1, rng, m_factor=0.999, blend_init=0.5, verbose=True)
        if s3 < best_score_overall - 1e-10:
            best_score_overall = s3
            best_overall = f3.copy()
        f0 = f3

    # 1d: Pressure again (settle)
    remaining = deadline - time.monotonic()
    t_press2 = min(40, remaining * 0.1)
    if t_press2 > 5:
        f4, s4 = pressure_iterate(f0, n_iter=max(50, int(t_press2 * 15)), eta=0.08, rng=rng)
        if s4 < best_score_overall - 1e-10:
            best_score_overall = s4
            best_overall = f4.copy()
        f0 = f4

    # 1e: More LP with tighter factor
    remaining = deadline - time.monotonic()
    t_lp2 = min(remaining * 0.25, 180)
    if t_lp2 > 10:
        f5, s5 = lp_refine(f0, t_lp2, rng, m_factor=0.9985, blend_init=0.3, verbose=True)
        if s5 < best_score_overall - 1e-10:
            best_score_overall = s5
            best_overall = f5.copy()

    # 1f: Projected subgradient
    remaining = deadline - time.monotonic()
    t_pg = min(remaining * 0.15, 60)
    if t_pg > 5 and best_overall is not None and len(best_overall) == n:
        f6, s6 = projected_subgradient(best_overall, n_iter=max(500, int(t_pg * 50)))
        if s6 < best_score_overall - 1e-10:
            best_score_overall = s6
            best_overall = f6.copy()
            print(f"  PG: {s6:.6f}")

    # === Phase 2: Try longer sequence (n=1024 or 2*n_parent) ===
    n2 = max(1024, 2 * n)
    remaining = deadline - time.monotonic()
    if remaining > 120:
        print(f"\n--- Phase 2: n={n2} (remaining: {remaining:.0f}s) ---")
        # Upsample parent
        if best_overall is not None:
            x_old = np.linspace(0, 1, len(best_overall))
            x_new = np.linspace(0, 1, n2)
            f_up = np.interp(x_new, x_old, best_overall)
        else:
            f_up = np.ones(n2)
        f_up = np.clip(f_up, 0.01, None)
        f_up = normalize(f_up)

        # Quick pressure
        t_up_press = min(30, remaining * 0.1)
        if t_up_press > 5:
            f_up, _ = pressure_iterate(f_up, n_iter=int(t_up_press * 15), eta=0.1, rng=rng)

        # LP
        t_up_lp = min(remaining * 0.3, 120)
        if t_up_lp > 10:
            f_up2, s_up = lp_refine(f_up, t_up_lp, rng, m_factor=0.999, blend_init=0.4, verbose=True)
            if s_up < best_score_overall - 1e-10:
                best_score_overall = s_up
                best_overall = f_up2.copy()
                print(f"  n={n2} LP: {s_up:.6f}")

    # === Phase 3: Final polishing ===
    remaining = deadline - time.monotonic()
    if best_overall is not None and remaining > 15:
        print(f"\n--- Phase 3: Polishing (remaining: {remaining:.0f}s) ---")

        # Coordinate descent
        t_cd = min(remaining * 0.3, 60)
        n_sweeps = max(3, int(t_cd / 5))
        f_cd, s_cd = coord_descent(best_overall, n_sweeps=n_sweeps, rng=rng)
        if s_cd < best_score_overall - 1e-10:
            best_score_overall = s_cd
            best_overall = f_cd.copy()
            print(f"  CD: {s_cd:.6f}")

        # One more LP pass with very tight factor
        remaining = deadline - time.monotonic()
        if remaining > 20:
            f_lp, s_lp = lp_refine(best_overall, min(remaining * 0.4, 60), rng,
                                   m_factor=0.999, blend_init=0.2, verbose=True)
            if s_lp < best_score_overall - 1e-10:
                best_score_overall = s_lp
                best_overall = f_lp.copy()
                print(f"  Final LP: {s_lp:.6f}")

        # Final CD
        remaining = deadline - time.monotonic()
        if remaining > 10:
            f_cd2, s_cd2 = coord_descent(best_overall, n_sweeps=3, rng=rng)
            if s_cd2 < best_score_overall - 1e-10:
                best_score_overall = s_cd2
                best_overall = f_cd2.copy()
                print(f"  Final CD: {s_cd2:.6f}")

    # Verify and return
    if best_overall is not None:
        final = [float(max(0.0, min(1000.0, x))) for x in best_overall]
        fs = evaluate_sequence(final)
        print(f"\n=== FINAL: score={fs:.6f}, n={len(final)} ===")
        return final

    # Fallback
    return [1.0] * 256
