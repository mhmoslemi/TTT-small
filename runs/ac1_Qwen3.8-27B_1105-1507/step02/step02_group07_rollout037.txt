"""
Multi-phase optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses L2 warmup, soft-max gradient descent, LP polishing, and
direct max subgradient, with symmetry enforcement.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def compute_l2_gradient(f):
    """Compute gradient of 0.5 * ||conv(f,f)||_2^2 = sum_k h[k]^2 / 2.
    Actually let's just compute gradient of sum_k h[k]^2.
    grad[i] = 4 * sum_j h[i+j] * f[j]
    """
    n = len(f)
    h = np.convolve(f, f)  # length 2n-1
    # A[i] = sum_j h[i+j] * f[j] = correlate(h, f, 'full')[i]
    A = np.correlate(h, f, mode='full')[:n]
    return 4.0 * A


def compute_softmax_gradient(f, beta):
    """Compute soft-subgradient of max(conv(f,f)) using softmax weights.
    g[i] = 2 * sum_j f[j] * p[i+j] where p = softmax(beta * h)
    """
    n = len(f)
    h = np.convolve(f, f)  # length 2n-1
    max_h = h.max()
    exp_h = np.exp(beta * (h - max_h))
    p = exp_h / exp_h.sum()
    # g[i] = 2 * sum_j f[j] * p[i+j]
    # = 2 * correlate(p, f, 'full')[i]
    g = 2.0 * np.correlate(p, f, mode='full')[:n]
    return g, max_h


def enforce_symmetry(f):
    """Enforce f[i] = f[n-1-i]."""
    return (f + f[::-1]) / 2.0


def project_onto_simplex(f, target_sum=1.0):
    """Project f onto {x >= 0, sum(x) = target_sum}."""
    f = np.clip(f, 0.0, None)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) * target_sum / len(f)
    return f * (target_sum / s)


def lp_solve_direction(f, M, max_iters=50):
    """Solve LP: max sum(g) s.t. g >= 0, conv(f,g)_k <= M for all k."""
    n = len(f)
    # Build constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except:
        pass
    return None


def l2_warmup(f, deadline, max_iters=5000, verbose=""):
    """Minimize ||conv(f,f)||_2^2 with projected gradient + Adam."""
    f = project_onto_simplex(enforce_symmetry(f.copy()))
    n = len(f)
    best_f = f.copy()
    best_L = np.sum(np.convolve(f, f)**2)
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = 0.1
    it = 0
    
    while time.monotonic() < deadline and it < max_iters:
        grad = compute_l2_gradient(f)
        it += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*grad**2

        v_hat = v / (1 - beta2**it)

        new_f = enforce_symmetry(new_f)

        if new_L < best_L:
            f = new_f
            best_f = f.copy()
            best_L = new_L

def direct_max_subgradient(f, deadline, max_iters=5000, verbose=""):
    """Use direct subgradient of max(conv(f,f))."""
    f = project_onto_simplex(enforce_symmetry(f.copy()))
    n = len(f)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    lr = 0.01
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline and it < max_iters:
        h = np.convolve(f, f)
        M = h.max()
        # Active set: positions where h[k] > 0.999 * M
        active = h > 0.999 * M
        if active.sum() == 0:
            break
        # Subgradient: g[i] = 2 * sum_{k in active} f[k-i] / |active|
        # Using the formula: g[i] = 2/|active| * sum_j f[j] * (1/|active|) * sum_{k in active} 1_{k=i+j}
        # = 2/|active| * sum_j f[j] * p[i+j] where p[k] = 1/|active| if k in active

        p[active] = 1.0 / active.sum()

        new_f = f - lr * g
        new_f = enforce_symmetry(new_f)
        new_f = project_onto_simplex(new_f)

                no_improve = 0
                if lr < 1e-8:
                    break

def tukey_window(n, r=0.3):
    w = np.ones(n)
    m = max(1, int(np.floor(r * n / 2)))
    for i in range(m):
        w[i] = 0.5 * (1 + np.cos(np.pi * (2*i/(2*m) - 1)))
        w[n-1-i] = w[i]
    return w


def super_gaussian(n, alpha=2.0):
    idx = np.arange(n, dtype=np.float64)
    center = (n-1)/2.0
    w = np.exp(-((idx-center)/(n*0.4))**(2*alpha))
    return w
