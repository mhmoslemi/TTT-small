def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def normalized(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:

def smoothed_max_grad(b, tau):
    """Returns (smoothed_max_value, gradient_weights) where gradient_weights sum to 1."""
    b_max = b.max()
    if tau < 1e-15:
        weights = np.zeros_like(b)
        mask = (b == b_max)
        weights[mask] = 1.0 / mask.sum()
        return b_max, weights

                    current_J = J_now
                    tracker.update(a)
    
    tracker.update(a)

def coordinate_descent(a, n_passes, deadline, tracker):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a)
    a = normalized(a.copy())
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    
    for pass_idx in range(n_passes):
        if time.monotonic() > deadline:
            break
        
        # Compute current b
        b = fast_conv(a)
        S = a.sum()
        
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            # Precompute c_k = b_k - contribution of a[i]
            # b_k = sum_j a_j * a_{k-j}
            # Terms involving a[i]: a[i]*a[k-i] (from j=i) and a[k-i]*a[i] (from j=k-i)
            # If k != 2i: b_k = c_k + 2*a[i]*a[k-i]
            # If k == 2i: b_k = c_k + a[i]^2
            
            ai = a[i]
            # Compute d_k = a[k-i] for valid k
            # For k in [0, 2*n-2], k-i is valid when 0 <= k-i < n, i.e., i <= k <= i+n-1
            c = b.copy()  # Will modify
            for k in range(max(0, i), min(2*n-1, i+n)):
                km = k - i  # k-i, which is in [0, n-1]
                if k == 2 * i:
                    c[k] = b[k] - ai * ai
                else:
                    c[k] = b[k] - 2 * ai * a[km]
            
            # Now optimize: minimize 2n*max(c_k + 2t*a[k-i], c_{2i} + t^2) / (S - ai + t)^2
            # over t >= 0
            
            S_other = S - ai
            best_t = ai
            best_val = float('inf')
            
            # Grid search
            t_vals = np.concatenate([
                np.linspace(0, ai * 0.1, 5),
                np.linspace(ai * 0.1, ai * 0.5, 5),
                np.linspace(ai * 0.5, ai * 1.5, 10),
                np.linspace(ai * 1.5, ai * 3, 10),
                np.linspace(ai * 3, ai * 10, 5),
            ])
            t_vals = np.unique(np.maximum(t_vals, 0))
            
            for t in t_vals:
                # Compute new b
                new_b = c.copy()
                for k in range(max(0, i), min(2*n-1, i+n)):
                    if k == 2 * i:
                        new_b[k] = c[k] + t * t
                    else:
                        new_b[k] = c[k] + 2 * t * a[k - i]
                
                S_new = S_other + t
                if S_new < 1e-15:
                    continue
                J = 2 * n * new_b.max() / (S_new * S_new)
                if J < best_val:
                    best_val = J
                    best_t = t
            
            a[i] = best_t
        
        # Renormalize
        a = normalized(a)
        J_now = compute_J(a)
        tracker.update(a)
    
    return a

import numpy as np
import time


class ScoreTracker:
    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()
    
    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)

def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def resample_seq(src, dest_n):
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


def normalize_simplex(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def smoothed_max_and_weights(b, tau):
    b_max = b.max()
    if tau < 1e-15:
        weights = np.zeros_like(b)
        mask = (b == b_max)
        weights[mask] = 1.0 / mask.sum()
        return b_max, weights
    shifted = (b - b_max) / tau
    shifted = np.clip(shifted, -80, 0)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    weights = exp_vals / total
    return b_max + tau * np.log(total), weights


def grad_autocorr(a, weights):
    """Gradient of smoothed max of conv(a,a) w.r.t. a."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fw = np.fft.rfft(weights, size)
    fa = np.fft.rfft(a, size)
    corr = np.fft.irfft(fw * np.conj(fa), size)
    return 2.0 * corr[n - 1:n - 1 + n]


def run_gradient_descent(a, deadline, tracker, tau_ratio=0.1, n_tau=8, iters_per_tau=200, lr=0.01):
    """Adam gradient descent with annealed tau."""
    n = len(a)
    a = normalize_simplex(a.copy())
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    b = fast_conv(a)
    b_max = b.max()
    tau_start = b_max / tau_ratio
    # tau schedule: geometric from tau_start to tau_start * 1e-4
    tau_list = tau_start * (1e-4) ** (np.arange(n_tau) / max(1, n_tau - 1))
    
    current_J = compute_J(a)
    best_local_J = current_J
    
    for tau_idx, tau in enumerate(tau_list):
        if time.monotonic() > deadline:
            break
        for it in range(iters_per_tau):
            if time.monotonic() > deadline:
                break
            
            b = fast_conv(a)
            sm_val, weights = smoothed_max_and_weights(b, tau)
            grad = grad_autocorr(a, weights)
            
            t += 1
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * grad * grad
            m_hat = m / (1 - 0.9 ** t)
            v_hat = v / (1 - 0.999 ** t)
            
            a = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                break
            a = a / s
            
            if it % 100 == 0:
                J_now = compute_J(a)
                if J_now < best_local_J:
                    best_local_J = J_now
                    tracker.update(a)
    
    tracker.update(a)
    return a
