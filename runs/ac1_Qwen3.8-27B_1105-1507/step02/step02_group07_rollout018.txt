    """Efficient subgradient computation using convolution."""
    n = len(f)
    # Create the "weight" signal
    g = np.zeros(2 * n - 1)
    for idx, k in enumerate(active_indices):
        g[k] = weights[idx]
    
    # The term sum_m w_m * f[k_m - j] for j=0,...,n-1
    # This is the cross-correlation of f with g, evaluated at specific points.
    # corr[j] = sum_k g[k] * f[k-j] ... wait that's not standard cross-correlation.
    
    # Actually: sum_m w_m * f[k_m - j] = sum_k g[k] * f[k-j]
    # This is the cross-correlation (g ⋆ f)[j] = sum_k g[k] * f[k-j]
    # = (g * f_rev)[j + n - 1] where f_rev is f reversed and * is convolution
    
    f_rev = f[::-1]
    conv_result = np.convolve(g, f_rev, mode='full')

    term2 = conv_result[n-1:n-1+n]  # indices n-1, n, ..., 2n-2
    
    grad = f + term2
    return grad

def optimize_subgradient(f_init, deadline, n_top=10, lr=0.01):
    f = f_init.copy() / np.sum(f_init)
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    
    it = 0
    no_improve = 0
    while time.monotonic() < deadline:
        C = np.convolve(f, f)
        M = C.max()
        
        # Find active set
        threshold = M * 0.99
        active_all = np.where(C >= threshold)[0]
        if len(active_all) > n_top:
            # Keep top n_top
            top_idx = np.argsort(C[active_all])[::-1][:n_top]
            active = active_all[top_idx]
        else:
            active = active_all
        
        # Equal weights
        weights = np.ones(len(active)) / len(active)
        
        # Compute subgradient
        grad = compute_subgradient_fast(f, active, weights)
        
        # Gradient descent step
        f_new = f - lr * grad
        f_new = np.clip(f_new, 0, 1000)
        s_new = np.sum(f_new)
        if s_new < 1e-10:
            break
        f_new = f_new / s_new
        
        score = evaluate_sequence(f_new.tolist())
        if score < best_score:
            f = f_new
            best_f = f.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.5
            if lr < 1e-10:
                break
        
        it += 1
    
    return best_f, best_score
