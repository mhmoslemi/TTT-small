Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response with backtracking line search and periodic perturbation
2. Adam optimizer on log-sum-exp smoothed objective with beta annealing
3. Multi-length exploration (384, 512, 768, 1024)
4. Fine-grained LP polishing and perturbation search

import numpy as np
from scipy.optimize import linprog
import time

def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def build_conv_matrix(a):
    """Build matrix A where (A @ x) computes full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh+1] = a[k-jh:k-jl+1][::-1]
    return A

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)]*n, method='highs')
    if res.success and res.x is not None and res.x.sum() > 1e-10:
        return res.x
    return None

def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))

def lp_iteration_improved(a, deadline, max_iters=5000):
    """LP best-response with backtracking line search and perturbation."""
    best_local = a.copy()
    best_score = compute_score(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 0.001, None)
    a = a / a.sum()
    
    no_improve_count = 0
    perturb_interval = 200
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()
        
        # Backtracking line search: try multiple step sizes
        best_t = None
        best_score_t = best_score
        
        for t in [0.3, 0.1, 0.5, 0.2, 0.05, 0.7, 0.15, 0.02]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 1e-10:
                continue
            a_new = a_new / s
            score = compute_score(a_new)
            if score < best_score_t:
                best_score_t = score
                best_t = t
        
        if best_t is not None:
            a = (1.0 - best_t) * a + best_t * g
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            if best_score_t < best_score:
                best_score = best_score_t
                best_local = a.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
        
        # Periodic perturbation
        if it % perturb_interval == perturb_interval - 1 and no_improve_count > 50:
            noise = np.random.default_rng(it).normal(0, 0.05, len(a))
            a = a + noise
            a = np.clip(a, 0.001, None)
            a = a / a.sum()
        
        if no_improve_count > 500:
            break
    
    return best_local, best_score

def adam_smooth_refine(a, deadline, betas=None):
    """Adam optimizer on smooth objective with beta annealing."""
    n = len(a)
    if betas is None:
        betas = [50, 100, 200, 500, 1000, 2000, 5000]
    
    best = a.copy()
    best_score = compute_score(a)
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        x = best.copy()
        x = np.clip(x, 0.0, None)
        s = x.sum()
        if s < 0.01:
            continue
        x = x / s
        
        # Adam parameters
        lr = 0.01
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        m_t = np.zeros(n)
        v_t = np.zeros(n)
        t_count = 0
        
        max_steps = 20000
        last_improve = 0
        
        for step in range(max_steps):
            if time.monotonic() > deadline:
                break
            
            # Compute gradient
            grad, func_val = smooth_gradient(x, beta, n)
            if not np.isfinite(func_val) or func_val > 1e10:
                break
            
            t_count += 1
            m_t = beta1 * m_t + (1 - beta1) * grad
            v_t = beta2 * v_t + (1 - beta2) * grad**2
            m_hat = m_t / (1 - beta1**t_count)
            v_hat = v_t / (1 - beta2**t_count)
            
            x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
            x_new = np.clip(x_new, 0.0, None)
            s = x_new.sum()
            if s < 0.01:
                x_new = x
            else:
                x_new = x_new / s
            
            x = x_new
            
            if step % 500 == 0:
                score = compute_score(x)
                if score < best_score:
                    best_score = score
                    best = x.copy()
                    last_improve = step
                    lr = lr * 1.0  # keep lr
                elif step - last_improve > 2000:
                    lr = lr * 0.5
                    if lr < 1e-6:
                        break
        
        # After Adam, do a few LP steps to polish
        if time.monotonic() < deadline:
            remaining_iters = int((deadline - time.monotonic()) / 0.05)
            remaining_iters = min(remaining_iters, 500)
            a_polished, score_polished = lp_iteration_improved(best, deadline, max_iters=remaining_iters)
            if score_polished < best_score:
                best = a_polished
                best_score = score_polished
    
    return best, best_score

def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    t = np.linspace(0, 1, n)
    
    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        resampled = np.maximum(np.interp(t, t_old, parent), 0)
        if resampled.sum() > 0.01:
            inits.append(resampled / resampled.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Various cosine patterns
    for freq, amp in [(1, 0.3), (1, 0.5), (2, 0.3), (2, 0.5), (3, 0.3), (4, 0.2)]:
        c = 1.0 + amp * np.cos(2 * np.pi * freq * t)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())
    
    # Sin patterns
    for freq, amp in [(1, 0.3), (2, 0.3)]:
        c = 1.0 + amp * np.sin(2 * np.pi * freq * t)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())
    
    # Random
    inits.append(rng.uniform(0.2, 1.0, n) / n)
    
    # Bimodal
    c = 1.0 + 0.5 * np.cos(2 * np.pi * t)**2
    inits.append(c / c.sum())
    
    # Exponential
    c = np.exp(-2 * (t - 0.5)**2)
    inits.append(c / c.sum())
    
    # Quadratic
    c = 1.0 + 0.5 * (4 * (t - 0.5)**2)
    inits.append(c / c.sum())
    
    return inits
