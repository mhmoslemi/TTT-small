import numpy as np
import time
from scipy.optimize import linprog

def normalize(x):
    s = x.sum()
    if s < 1e-15:
        return x
    return x / s

def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (f.sum() ** 2)

def build_conv_matrix(f):
    """Build matrix A such that A @ b = np.convolve(f, b, mode='full')."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = f[j]
    return A

def lp_iterate(f, n_lp_iters=50, alpha=0.9, max_step=0.5, rng=None):
    """LP-based iteration to minimize max convolution."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(f)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    best_f = f.copy()
    best_score = compute_score(f)
    
    A = build_conv_matrix(f)
    
    for it in range(n_lp_iters):
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Find critical constraints
        threshold = alpha * M
        critical = np.where(conv >= threshold)[0]
        
        if len(critical) == 0:
            critical = [np.argmax(conv)]
        
        # Solve LP
        c_obj = -np.ones(n)
        A_ub = A[critical, :]
        b_ub = np.full(len(critical), M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 5})
        except Exception:
            break
        
        if not res.success or res.x is None:
            break
        
        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        
        g = b / b_sum
        
        # Line search for step size
        improved = False
        for step in [max_step, max_step * 0.5, max_step * 0.25, max_step * 0.1]:
            f_new = (1 - step) * f + step * g
            f_new = np.clip(f_new, 0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_score = compute_score(f_new)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_f = f_new.copy()
                f = f_new
                # Rebuild A since f changed
                A = build_conv_matrix(f)
                improved = True
                break
        
        if not improved:

            f_pert = np.clip(f_pert, 0, None)
            f_pert = normalize(f_pert)
            if compute_score(f_pert) < best_score:
                f = f_pert
                best_f = f.copy()
                best_score = compute_score(f)
                A = build_conv_matrix(f)
            else:
                break
    
    return best_f, best_score

LP-based optimization with gradient descent and coordinate descent polishing.
Minimizes 2*n*max(conv(f,f)) / sum(f)^2 for non-negative f.
"""

import time
from scipy.optimize import linprog

def normalize(x):
    s = x.sum()
    if s < 1e-15:
        return x
    return x / s

def get_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    return 2.0 * n * conv.max() / (f.sum() ** 2)

def build_A(f):
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        indices = np.arange(k - i_end + 1, k - i_start + 1)
        A[k, i_start:i_end] = f[indices]
    return A

    """Gradient descent on smooth max of convolution."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = get_score(f)

        p = p_start * (p_end / p_start) ** (it / n_iter)
        conv = np.convolve(f, f)
        
        # Smooth max gradient
        # weights = softmax(p * conv)
        log_w = p * conv - p * conv.max()

"""
LP-based optimization with gradient preconditioning for minimizing
2*n*max(conv(f,f))/sum(f)^2 over non-negative sequences.
Uses iterative LP linearization, smooth-max gradient descent, and coordinate descent.
"""

import numpy as np
import time
from scipy.optimize import linprog


def normalize(x):
    s = x.sum()
    if s < 1e-15:
        return x
    return x / s


def get_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    return 2.0 * n * conv.max() / (s * s)


def build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(k - i_end + 1, k - i_start + 1)
        A[k, i_start:i_end] = f[idx]
    return A


def smooth_max_grad(f, p=100.0):
    """Gradient of smooth max of self-convolution."""
    n = len(f)
    conv = np.convolve(f, f)
    # Softmax weights (numerically stable)
    log_w = p * conv - p * conv.max()
    w = np.exp(log_w)
    w /= w.sum()
    # grad[m] = 2 * sum_{k=0}^{n-1} w[m+k] * f[k]
    grad = np.zeros(n)
    for m in range(n):
        grad[m] = 2.0 * np.dot(w[m:m+n], f)
    return grad


def gradient_descent(f, n_iter=5000, p_start=30.0, p_end=300.0, eta0=0.05, rng=None):
    """Smooth-max gradient descent with projection."""
    if rng is None:
        rng = np.random.default_rng(0)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_f = f.copy()
    best_score = get_score(f)
    stall = 0

    for it in range(n_iter):
        p = p_start * (p_end / p_start) ** (it / max(1, n_iter - 1))
        grad = smooth_max_grad(f, p)
        
        eta = eta0 / (1.0 + 0.002 * it)
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.125]:
            step = eta * factor
            f_new = f - step * grad
            f_new = np.clip(f_new, 0.0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            sc = get_score(f_new)
            if sc < best_score - 1e-13:
                best_score = sc
                best_f = f_new.copy()
                f = f_new
                improved = True
                stall = 0
                break
        
        if not improved:
            stall += 1
            if stall > 200:
                # Perturb
                f_p = f * (1 + 0.05 * rng.standard_normal(n))
                f_p = np.clip(f_p, 0, None)
                f_p = normalize(f_p)
                if get_score(f_p) < best_score:
                    f = f_p
                    best_f = f.copy()
                    best_score = get_score(f)
                    stall = 0
                else:
                    break

    return best_f, best_score


def lp_refine(f, time_budget, rng=None, alpha=0.92):
    """LP-based refinement (main workhorse)."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(f)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    best_f = f.copy()
    best_score = get_score(f)

    t0 = time.monotonic()
    stall = 0
    step_size = 0.4
    it = 0

    while time.monotonic() - t0 < time_budget:
        it += 1
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 5:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        # Critical constraints
        thresh = alpha * M
        critical = np.where(conv >= thresh)[0]
        if len(critical) == 0:
            critical = np.array([np.argmax(conv)])

        A = build_A(f)
        c_obj = -np.ones(n)
        A_ub = A[critical, :]
        b_ub = np.full(len(critical), M)
        bounds = [(0, None)] * n

        # Time limit for LP: be conservative
        lp_tl = min(10.0, max(1.0, remaining / 30.0))
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': lp_tl})
        except Exception:
            break

        if not res.success or res.x is None:
            stall += 1
            if stall > 15:
                break
            continue

        b = np.clip(res.x, 0, None)
        b_sum = b.sum()
        if b_sum < 1e-12:
            break

def normalize(x):
    s = x.sum()
    if s < 1e-15:

def get_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    return 2.0 * n * conv.max() / (s * s)


def build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        idx = np.arange(k - i_end + 1, k - i_start + 1)
        A[k, i_start:i_end] = f[idx]
    return A


def smooth_max_grad(f, p=100.0):
    n = len(f)
    conv = np.convolve(f, f)
    log_w = p * conv - p * conv.max()
    w = np.exp(log_w)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(n):
        grad[m] = 2.0 * np.dot(w[m:m + n], f)
    return grad

        b_sum = b.sum()
        if b_sum < 1e-12:
            break
        g = b / b_sum

        improved = False
        steps = [step_size, step_size * 0.7, step_size * 0.4, step_size * 0.2, step_size * 0.1, 0.05]
        for step in steps:
            f_new = (1.0 - step) * f + step * g
            f_new = np.clip(f_new, 0.0, None)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            sc = get_score(f_new)
            if sc < best_score - 1e-13:
                best_score = sc
                best_f = f_new.copy()
                f = f_new
                improved = True
                stall = 0
                break

        if improved:
            step_size = min(0.7, step_size * 1.02)
            if it % 100 == 0:
                print(f"    LP it={it}, score={best_score:.8f}, step={step_size:.3f}")
        else:
            stall += 1
            step_size = max(0.02, step_size * 0.8)
            if stall > 25:
                escaped = False
                for ps in [0.1, 0.15, 0.2, 0.3]:
                    f_p = f * (1 + ps * rng.standard_normal(n))
                    f_p = np.clip(f_p, 0, None)
                    f_p = normalize(f_p)
                    if get_score(f_p) < best_score:
                        f = f_p
                        best_f = f.copy()
                        best_score = get_score(f)
                        step_size = 0.3
                        stall = 0
                        escaped = True
                        break
                if not escaped:
                    break
    if it > 0:
        print(f"    LP done: {it} iters, best={best_score:.8f}")
    return best_f, best_score


def coordinate_descent(f, n_sweeps=10, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(f)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    best_score = get_score(f)
    factors = [0.5, 0.7, 0.85, 1.15, 1.3, 0.3, 2.0]

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            orig = f[i]
            if orig < 1e-12:
                continue
            best_val = orig
            best_local = best_score
            for fac in factors:
                f[i] = orig * fac
                sc = get_score(f)
                if sc < best_local:
                    best_local = sc
                    best_val = orig * fac
            f[i] = best_val
            if best_local < best_score:
                best_score = best_local
        f = normalize(np.clip(f, 0, None))
    return f, best_score


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimizer: gradient descent -> multiplicative weight -> LP refinement
    (adaptive alpha) -> coordinate descent -> final LP polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0

    best_f = None
    best_score = np.inf

    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        if parent.ndim != 1 or len(parent) == 0 or not np.isfinite(parent).all():
            parent = None
    except Exception:
        parent = None

    if parent is not None:
        parent = np.clip(parent, 0, None)
        if parent.sum() > 0.01:
            best_f = normalize(parent)
            best_score = get_score(best_f)
            print(f"Parent: n={len(parent)}, score={best_score:.8f}")

    if best_f is None:
        n = 512
        best_f = np.ones(n) * (1 + 0.2 * rng.standard_normal(n))
        best_f = np.clip(best_f, 0.01, None)
        best_f = normalize(best_f)
        best_score = get_score(best_f)
        print(f"Random init: n={len(best_f)}, score={best_score:.8f}")

    n = len(best_f)

    # Phase 1: Gradient descent (fast)
    remaining = deadline - time.monotonic()
    if remaining > 80:
        t_gd = min(100, remaining * 0.12)
        print(f"\nPhase 1: Gradient descent ({t_gd:.0f}s budget)")
        t_start = time.monotonic()
        n_iter_gd = min(6000, int(t_gd * 400))
        f_gd, s_gd = gradient_descent(best_f, n_iter=n_iter_gd, p_start=30, p_end=300, eta0=0.05, rng=rng)
        elapsed = time.monotonic() - t_start
        print(f"  GD done in {elapsed:.1f}s, score={s_gd:.8f}")
        if s_gd < best_score:
            best_score = s_gd
            best_f = f_gd

    # Phase 2: Multiplicative weight
    remaining = deadline - time.monotonic()
    if remaining > 80:
        t_mw = min(100, remaining * 0.12)
        print(f"\nPhase 2: Multiplicative weight ({t_mw:.0f}s budget)")
        t_start = time.monotonic()
        n_iter_mw = min(4000, int(t_mw * 150))
        f_mw, s_mw = mulg_weight(best_f, n_iter=n_iter_mw, eta0=0.08, rng=rng)
        elapsed = time.monotonic() - t_start
        print(f"  MW done in {elapsed:.1f}s, score={s_mw:.8f}")
        if s_mw < best_score:
            best_score = s_mw
            best_f = f_mw

    # Phase 3: LP refinement with medium alpha
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t_lp = min(remaining * 0.4, 500)
        print(f"\nPhase 3: LP refinement alpha=0.90 ({t_lp:.0f}s)")
        f_lp, s_lp = lp_refine(best_f, time_budget=t_lp, rng=rng, alpha=0.90, max_constraints=60)
        if s_lp < best_score:
            best_score = s_lp
            best_f = f_lp
            print(f"  After LP3: {best_score:.8f}")

    # Phase 4: LP refinement with tight alpha
    remaining = deadline - time.monotonic()
    if remaining > 60:
        t_lp2 = min(remaining * 0.3, 350)
        print(f"\nPhase 4: LP refinement alpha=0.96 ({t_lp2:.0f}s)")
        f_lp2, s_lp2 = lp_refine(best_f, time_budget=t_lp2, rng=rng, alpha=0.96, max_constraints=30)
        if s_lp2 < best_score:
            best_score = s_lp2
            best_f = f_lp2
            print(f"  After LP4: {best_score:.8f}")

    # Phase 5: Coordinate descent
    remaining = deadline - time.monotonic()
    if remaining > 15:
        n_sweeps = min(8, max(3, int(remaining / 10)))
        print(f"\nPhase 5: Coordinate descent ({n_sweeps} sweeps)")
        f_cd, s_cd = coordinate_descent(best_f, n_sweeps=n_sweeps, rng=rng)
        if s_cd < best_score:
            best_score = s_cd
            best_f = f_cd
            print(f"  After CD: {best_score:.8f}")

    # Phase 6: Final LP micro-polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_final = min(remaining * 0.6, 150)
        print(f"\nPhase 6: Final LP polish alpha=0.99 ({t_final:.0f}s)")
        f_fin, s_fin = lp_refine(best_f, time_budget=t_final, rng=rng, alpha=0.99, max_constraints=20)
        if s_fin < best_score:
            best_score = s_fin
            best_f = f_fin
            print(f"  Final LP: {best_score:.8f}")

    # One last CD pass
    remaining = deadline - time.monotonic()
    if remaining > 10:
        f_last, s_last = coordinate_descent(best_f, n_sweeps=3, rng=rng)
        if s_last < best_score:
            best_score = s_last
            best_f = f_last
            print(f"  Final CD: {best_score:.8f}")

    # Verify with evaluate_sequence
    best_f = np.clip(best_f, 0, 1000)
    result = [float(x) for x in best_f]
    final_score = evaluate_sequence(result)
    print(f"\n=== FINAL: score={final_score:.8f}, n={len(result)} ===")
    return result
