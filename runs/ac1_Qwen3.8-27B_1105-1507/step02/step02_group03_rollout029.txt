```python
    n = len(f)

    shifted = conv / eps
    max_shifted = shifted.max()
    log_sum_exp = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
```

```python
"""
Optimized step function search using smoothed gradient descent with Nesterov momentum,
exact coordinate descent, and LP refinement. Multi-start across different sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time


def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return f.copy()
    return f / s


def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)


def smooth_score_and_grad(f, eps):
    """Compute smoothed score and its gradient w.r.t. f."""
    n = len(f)
    conv = np.convolve(f, f)
    shifted = conv / eps
    max_sh = shifted.max()
    exp_sh = np.exp(shifted - max_sh)
    w = exp_sh / exp_sh.sum()
    log_sum_exp = max_sh + np.log(exp_sh.sum())
    smooth_max = eps * log_sum_exp
    s = f.sum()
    
    val = 2.0 * n * smooth_max / (s * s)
    
    # Gradient: g[i] = 2 * sum_j f[j] * w[i+j]
    # = 2 * correlate(w, f, 'full')[i]
    corr = np.correlate(w, f, mode='full')
    g = 2.0 * corr[:n]
    
    # d(score)/d(f[i]) = 2n * (g[i]*s - 2*smooth_max) / s^3
    grad = 2.0 * n * (g * s - 2.0 * smooth_max) / (s ** 3)
    
    return val, grad


def projected_gradient_descent(f, eps_schedule, n_iters_per_eps, lr_schedule, rng):
    """Projected gradient descent with Nesterov momentum and annealed smoothing."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    
    momentum = np.zeros(n)
    mu = 0.9  # Momentum coefficient
    
    for eps_idx, eps in enumerate(eps_schedule):
        n_iters = n_iters_per_eps
        lr_base = lr_schedule(eps_idx)
        
        for it in range(n_iters):
            # Nesterov lookahead
            f_look = f - mu * momentum
            f_look = np.maximum(f_look, 0)
            s = f_look.sum()
            if s > 1e-15:
                f_look = f_look / s
            
            val, grad = smooth_score_and_grad(f_look, eps)
            
            # Update momentum
            momentum = mu * momentum - lr_base * grad
            
            # Update
            f_new = f - momentum
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 1e-15:
                f_new = f
            else:
                f_new = f_new / s
            
            # Check improvement (using true score)
            true_score = compute_score(f_new)
            if true_score < best_score:
                best_score = true_score
                best_f = f_new.copy()
            
            f = f_new
            
            # Reduce momentum over iterations
            mu = 0.9 * (1.0 - 0.5 * it / n_iters)
    
    return best_f, best_score


def exact_coordinate_descent(f, n_sweeps, rng):
    """Exact coordinate descent: for each coordinate, find optimal value."""
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            # Save current value
            orig = f[i]
            if orig < 1e-12:
                continue
            
            # Compute base convolution (with f[i] = 0)
            f[i] = 0.0
            base = np.convolve(f, f)
            base_sum = f.sum()
            f[i] = orig
            
            # Candidate values for f[i]:
            candidates = [orig, 0.0]
            
            # Current max position
            conv_current = base + _compute_contribution(f, i, orig)
            k_max = np.argmax(conv_current)
            
            # Candidate from optimal condition (k_max active):
            j = k_max - i
            if k_max == 2 * i:
                if base_sum > 0:
                    x_cand = base[k_max] / base_sum
                    if x_cand > 0:
                        candidates.append(x_cand)
            elif 0 <= j < n and f[j] > 1e-15:
                x_cand = base_sum - base[k_max] / f[j]
                if x_cand > 0:
                    candidates.append(x_cand)
            
            # Also try a few other k positions near the max
            conv_sorted_idx = np.argsort(base)[::-1][:5]
            for k in conv_sorted_idx:
                j2 = k - i
                if k == 2 * i:
                    if base_sum > 0:
                        x_cand = base[k] / base_sum
                        if x_cand > 0 and x_cand not in candidates:
                            candidates.append(x_cand)
                elif 0 <= j2 < n and f[j2] > 1e-15:
                    x_cand = base_sum - base[k] / f[j2]
                    if x_cand > 0 and x_cand not in candidates:
                        candidates.append(x_cand)
            
            # Evaluate each candidate
            best_x = orig
            best_local = best_score
            
            for x in candidates:

                f[i] = x
                s = f.sum()
                if s < 1e-12:
                    f[i] = orig

                    best_local = sc
                    best_x = x

            if best_local < best_score:
                best_score = best_local
                improved = True

def _compute_contribution(f, i, val):
    """Compute the contribution of f[i]=val to the convolution."""
    n = len(f)
    # f[i] contributes to conv[k] via:
    #   j=i: val * f[k-i] (if 0<=k-i<n)
    #   j=k-i: f[k-i] * val (if 0<=k-i<n and k-i != i)
    # Total: 2*val*f[k-i] (if k!=2i) or val^2 (if k==2i)
    contrib = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        j = k - i
        if 0 <= j < n:
            if k == 2 * i:
                contrib[k] = val * val
            else:
                contrib[k] = 2.0 * val * f[j]
    return contrib


def lp_refine(f, time_budget, t0=0.5, rng=None, n_lp_solves=20):
    """LP-based refinement."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    n = len(f)
    f = f.copy().astype(np.float64)
    f = normalize(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    t = t0
    start = time.monotonic()
    
    solve_count = 0
    while time.monotonic() - start < time_budget and solve_count < n_lp_solves:
        conv = np.convolve(f, f)
        M = conv.max()
        
        # Build constraint matrix A: A[k,j] = f[k-j] if 0<=k-j<n else 0
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            # f[j] * f_b[k-j] contributes to conv[k]
            # For each k, the coefficient of f_b[j] is f[k-j] if 0<=k-j<n
            k_start = j  # k-j >= 0 => k >= j
            k_end = j + n - 1  # k-j <= n-1 => k <= j+n-1
            k_end = min(k_end, 2 * n - 2)
            A[k_start:k_end+1, j] = f[0:k_end-k_start+1]
        
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        remaining = time_budget - (time.monotonic() - start)
        lp_time = max(1.0, remaining / max(1, n_lp_solves - solve_count))
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': lp_time})
        except Exception:
            break
        
        solve_count += 1
        
        if not res.success or res.x is None:
            break
        
        b = np.maximum(res.x, 0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        
        g = b / b_sum
        
        # Try blending
        improved = False
        for step in [min(1.0, t * 1.2), t, t * 0.5, t * 0.25]:
            f_new = (1 - step) * f + step * g
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_score = compute_score(f_new)
            if new_score < best_score - 1e-12:
                best_f = f_new.copy()
                best_score = new_score
                f = f_new
                t = min(1.0, t * 1.05)
                improved = True
                break
        
        if not improved:
            t *= 0.7
            if t < 0.01:
                # Perturb
                f_pert = f * (1.0 + 0.1 * rng.standard_normal(n))
                f_pert = np.maximum(f_pert, 0)
                f_pert = normalize(f_pert)
                if compute_score(f_pert) < best_score:
                    f = f_pert
                    best_f = f.copy()
                    best_score = compute_score(f)
                    t = 0.3
                else:
                    break
    
    return best_f, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    r = rng.random()
    if r < 0.2:
        # Cosine window
        f = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / n))
    elif r < 0.4:
        # Gaussian bump
        center = n * rng.uniform(0.25, 0.75)
        width = n * rng.uniform(0.1, 0.4)
        f = np.exp(-0.5 * (np.arange(n) - center) ** 2 / width ** 2)
    elif r < 0.6:
        # Triangle
        x = np.abs(np.arange(n) - n/2) / (n/2)
        f = 1 - x
    elif r < 0.8:
        # Product of smooth bumps
        f = np.ones(n)
        for _ in range(3):
            c = rng.uniform(0, n)
            w = rng.uniform(n*0.05, n*0.3)
            f *= np.exp(-0.5 * (np.arange(n) - c)**2 / w**2) + 0.1
    else:
        # Random smooth (low-pass filtered noise)
        f = rng.standard_normal(n)
        # Simple moving average
        kernel = np.ones(max(2, n//32)) / max(2, n//32)
        f = np.convolve(f, kernel, mode='same')
        f = np.abs(f) + 0.1
    
    f = np.maximum(f, 0.01)
    # Add small noise to break symmetry
    f = f * (1 + 0.05 * rng.standard_normal(n))
    f = np.maximum(f, 0.001)
    f = normalize(f)
    return f
```

Also, I want to make the coordinate descent more efficient. The current implementation recomputes the full convolution for each candidate, which is O(n^2) per coordinate per candidate. For n=512 with ~5 candidates, that's about 5 * 512 * 1023 ≈ 2.6M operations per coordinate, and 512 coordinates = 1.3B per sweep. That's too slow.

Let me optimize: instead of computing the full convolution for each candidate, I can compute the max more efficiently.

For a given candidate x = f[i], the convolution is:
conv[k] = base[k] + extra_k(x)

where extra_k(x) = 2*x*f[k-i] (if valid, k≠2i) or x^2 (if k=2i).

The max of conv[k] is the max of:
- max_k [base[k] + 2*x*f[k-i]] (for k where 0<=k-i<n and k≠2i)
- base[2i] + x^2 (if 2i < 2n-1)
- max_k [base[k]] (for k where k-i is not in [0,n) and k≠2i)

The third term is just max of base over "far" positions, which is a constant (doesn't depend on x).

For the first term: max_k [base[k] + 2*x*f[k-i]]
= 2*x * max_k [f[k-i] + base[k]/(2*x)]
Hmm, that doesn't simplify nicely.

Actually, let me just precompute:
- For each i, the vector c_i[k] = f[k-i] (for valid k)
- Then conv[k] = base[k] + 2*x*c_i[k] (approximately, ignoring the k=2i special case)
- max_k conv[k] = max_k (base[k] + 2*x*c_i[k])

This is the max of n linear functions of x (plus one quadratic). The max of linear functions is convex, so we can find the minimum of the score by checking the breakpoints.

But for a practical implementation, let me just evaluate at a few candidates:
1. x = 0
2. x = current value
3. x = optimal value assuming current argmax is active
4. x = optimal value assuming the second-highest position is active
5. A few geometrically spaced values around the current

And for evaluating the score at each x, I can compute:
max_conv = max(base + 2*x*c_i, base[2i] + x^2) where c_i is the appropriate vector

This is O(n) per candidate evaluation. With 5 candidates and n=512, that's 5*512 = 2560 operations per coordinate, and 512*2560 = 1.3M per sweep. That's fast!

```python
def exact_coordinate_descent(f, n_sweeps, rng):
    f = f.copy().astype(np.float64)
    f = normalize(f)
    n = len(f)
    best_score = compute_score(f)
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        any_improved = False
        
        for i in order:
            orig = f[i]
            if orig < 1e-14:
                continue
            
            # Compute base (conv with f[i]=0) and contribution vector
            f[i] = 0.0
            base = np.convolve(f, f)  # O(n^2) but we can do better
            base_sum = f.sum()
            f[i] = orig
            
            # Contribution vector: c[k] such that conv[k] = base[k] + 2*orig*c[k] (approx)
            # More precisely: conv[k] = base[k] + extra[k](x)
            # where extra[k](x) = 2*x*f[k-i] (if 0<=k-i<n, k!=2i)
            #                       x^2 (if k==2i)
            #                       0 (otherwise)
            
            # For the linear part (ignore x^2 for a moment):
            # Linear coefficient for x at position k: 2*f[k-i] if valid
            lin_coeff = np.zeros(2*n-1)
            for k in range(2*n-1):
                j = k - i
                if 0 <= j < n and k != 2*i:
                    lin_coeff[k] = 2.0 * f[j]
            
            # Quadratic part at k=2i: x^2
            
            # Current conv for reference
            # conv_current = base + extra(orig)
            # k_max = argmax
            
            # Candidate x values:
            candidates = [0.0, orig]
            
            # Find top-3 positions by (base + lin_coeff*orig)
            conv_est = base + lin_coeff * orig
            if 2*i < len(conv_est):
                conv_est[2*i] += orig * orig  # Add quadratic
            # Remove the contribution of f[i] from conv_est (since base already has f[i]=0)
            # Actually conv_est IS the full conv with f[i]=orig
            # Let me verify: base = conv with f[i]=0
            # conv with f[i]=orig = base + 2*orig*f[k-i] (for k!=2i) + orig^2 (for k=2i)
            # = base + lin_coeff*orig + (quadratic at 2i)
            # Yes, that's correct.
            
            top_k = np.argsort(conv_est)[::-1][:5]
            for k in top_k:
                if k == 2*i:
                    # x^2 + base[2i] is max (or part of max)
                    # Optimal: 2x*(base_sum+x) = 2*(base[2i]+x^2) => x = base[2i]/base_sum
                    if base_sum > 0:
                        xc = base[2*i] / base_sum
                        if xc > 0:
                            candidates.append(min(xc, orig * 3))
                else:
                    j = k - i
                    if 0 <= j < n and f[j] > 1e-15:
                        # 2*f[j]*x + base[k] is the active term
                        # Optimal: 2*f[j]*(base_sum+x) = 2*(base[k] + 2*f[j]*x)
                        # => f[j]*base_sum + f[j]*x = base[k] + 2*f[j]*x
                        # => x = (f[j]*base_sum - base[k]) / f[j] = base_sum - base[k]/f[j]
                        xc = base_sum - base[k] / f[j]
                        if xc > 0:
                            candidates.append(min(xc, orig * 3))
            
            # Also add a few geometric values
            candidates.append(orig * 0.5)
            candidates.append(orig * 0.75)
            candidates.append(orig * 1.5)
            
            best_x = orig
            best_local = best_score
            
            for x in candidates:
                x = max(0.0, float(x))
                # Compute max conv efficiently
                conv_x = base + lin_coeff * x
                if 2*i < len(conv_x):
                    conv_x[2*i] += x * x
                mx = conv_x.max()
                s = base_sum + x
                if s < 1e-12:
                    continue
                sc = 2.0 * n * mx / (s * s)
                if sc < best_local:
                    best_local = sc
                    best_x = x
            
            f[i] = best_x
            if best_local < best_score:
                best_score = best_local
                any_improved = True
        
        f = normalize(f)
        if not any_improved:
            break
    
    return f, best_score
```
