"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Fast active-set multiplicative updates (O(n log n) per iteration)
2. LP-based Frank-Wolfe refinement for precision
3. Subgradient polishing
4. Multi-length exploration with diverse initializations
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]

def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2 directly."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))

def compute_t(a, S):
    """Compute t_j = (1/|S|) * sum_{k in S} a[k-j] (if 0<=k-j<n) using FFT."""
    n = len(a)
    m = 2 * n - 1
    u = np.zeros(m)
    u[S] = 1.0
    v = a
    
    # Cross-correlation: t_j = sum_k u[k] * v[k-j]
    # Using FFT: ifft(fft(u) * conj(fft(v)))
    size = 1
    while size < m + n - 1:
        size <<= 1
    Fu = np.fft.rfft(u, size)
    Fv = np.fft.rfft(v, size)
    t = np.fft.irfft(Fu * np.conj(Fv), size)[:n]
    t /= max(1, len(S))
    return np.maximum(t, 1e-30)

def active_set_update(a, alpha=0.5, epsilon=0.05):
    """Direct multiplicative update based on active set."""
    b = fast_conv(a)
    b_max = b.max()
    if b_max < 1e-15:
        return a.copy()
    # Active set: positions where b_k > b_max * (1 - epsilon)
    threshold = b_max * (1.0 - epsilon)
    S = np.where(b >= threshold)[0]
    
    t = compute_t(a, S)
    
    # Multiplicative update: a_j *= (mean(t)/t_j)^alpha
    c = t.mean()
    ratio = (c / t) ** alpha
    a_new = a * ratio
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    a_new = a_new / s
    return a_new

def active_set_descent(a0, deadline, max_iters=50000):
    """Run active-set multiplicative updates."""
    a = np.maximum(a0, 1e-10)
    a = a / a.sum()
    best = a.copy()
    best_score = compute_score(a)
    n = len(a)
    
    alpha = 0.5
    epsilon = 0.05
    stall_count = 0
    best_stall = 0
    
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        
        a_new = active_set_update(a, alpha, epsilon)
        score = compute_score(a_new)
        
        if score < best_score:
            best_score = score
            best = a_new.copy()
            a = a_new
            stall_count = 0
        else:
            stall_count += 1
            # Try with different parameters
            if stall_count % 10 == 0:
                alpha *= 0.9
                epsilon = max(0.01, epsilon * 0.9)
            if stall_count > 200:
                # Reset and try larger perturbation
                alpha = 0.8
                epsilon = 0.1
                noise = 0.01 * np.random.randn(n)
                a = a + noise
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                stall_count = 0
                if best_stall > 1000:
                    break
                best_stall += 1
            elif stall_count > 50:
                best_stall += 1
    
    return best, best_score

def build_conv_matrix_sparse(a):
    """Build the convolution matrix as a list of (rows, cols, values) for sparsity."""
    n = len(a)
    m = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        for j in range(jl, jh + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return rows, cols, vals

def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(m, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None

def fw_iteration(a, deadline, max_iters=2000):
    """Frank-Wolfe with LP best response using 2/(t+2) step size."""
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    best = a.copy()
    best_score = compute_score(a)
    n = len(a)
    
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        
        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()
        
        # Standard FW step size
        gamma = 2.0 / (it + 2.0)
        d = g - a
        a_new = a + gamma * d
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best = a_new.copy()
            a = a_new
    
    return best, best_score

def subgradient_polish(a, deadline, max_iters=10000):
    """Subgradient method polishing for the active max position."""
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    best = a.copy()
    best_score = compute_score(a)
    n = len(a)
    lr = 0.01
    stale = 0
    
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        
        b = fast_conv(a)
        k = int(np.argmax(b))
        
        # Gradient of (a*a)_k wrt a
        grad = np.zeros(n)
        km = k - np.arange(n)
        valid = (km >= 0) & (km < n)
        grad[valid] = 2.0 * a[km[valid]]
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best = a_new.copy()
            a = a_new
            stale = 0
        else:
            stale += 1
            if stale % 50 == 0:
                lr *= 0.9
            if lr < 1e-10:
                break
    
    return best, best_score

def generate_initializations(n, rng, parent=None):

    inits = []

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Fast active-set multiplicative updates (O(n log n) per iteration, no LP needed)
2. LP-based Frank-Wolfe conditional gradient for precision refinement
3. Smoothed p-norm gradient descent with Adam for escaping local optima
4. Multi-length exploration with diverse structured initializations
5. Adaptive parameter tuning and perturbation restarts
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) = 2n*max(conv(a,a))/sum(a)^2 directly."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_t_fft(a, S):
    """Compute t_j = (1/|S|) * sum_{k in S} a[k-j] (if 0<=k-j<n) using FFT cross-correlation."""
    n = len(a)
    m = 2 * n - 1
    u = np.zeros(m)
    u[S] = 1.0
    size = 1
    while size < m + n - 1:
        size <<= 1
    Fu = np.fft.rfft(u, size)
    Fv = np.fft.rfft(a, size)
    t = np.fft.irfft(Fu * np.conj(Fv), size)[:n]
    t /= max(1, len(S))
    return np.maximum(t, 1e-30)


def active_set_step(a, alpha, epsilon):
    """One active-set multiplicative update. Returns updated a."""
    b = fast_conv(a)
    b_max = b.max()
    if b_max < 1e-15:
        return a.copy()
    threshold = b_max * (1.0 - epsilon)
    S = np.where(b >= threshold)[0]
    if len(S) == 0:
        S = np.array([int(np.argmax(b))])
    t = compute_t_fft(a, S)
    c = t.mean()
    ratio = (c / t) ** alpha
    a_new = a * ratio
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    a_new = a_new / s
    return a_new


def active_set_descent(a0, deadline, max_iters=200000, verbose=False):
    """Run active-set multiplicative updates with adaptive parameters."""
    a = np.maximum(a0, 1e-10)
    a = a / a.sum()
    best = a.copy()
    best_score = compute_score(a)
    n = len(a)

    alpha = 0.5
    epsilon = 0.05
    stall = 0
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        a_new = active_set_step(a, alpha, epsilon)
        score = compute_score(a_new)

        if score < best_score:
            best_score = score
            best = a_new.copy()
            a = a_new
            stall = 0
            no_improve = 0
            if verbose and it % 1000 == 0:
                print(f"    AS iter {it}: J={best_score:.6f}", flush=True)
        else:
            a = a_new
            stall += 1
            no_improve += 1

            if stall == 50:
                alpha = max(0.1, alpha * 0.85)
                epsilon = max(0.01, epsilon * 0.9)
                stall = 0
            elif stall == 150:
                alpha = 0.7
                epsilon = 0.08
                stall = 0
            elif stall >= 300:
                # Perturb and reset
                noise_scale = 0.02 + 0.01 * (no_improve % 5)
                a = best + noise_scale * np.random.randn(n)
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                alpha = 0.5
                epsilon = 0.05
                stall = 0
                no_improve += 50

            if no_improve > 5000:
                break

    return best, best_score


def build_conv_matrix(a):
    """Build Toeplitz-like matrix A where (A @ x) = full convolve(a, x)."""
    n = len(a)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh + 1] = a[k - jh:k - jl + 1][::-1]
    return A


def lp_best_response(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g or None."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None


def fw_iteration(a, deadline, max_iters=2000, verbose=False):
    """Frank-Wolfe with LP best response using standard 2/(t+2) step size."""
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    best = a.copy()
    best_score = compute_score(a)
    n = len(a)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break
        g = g / g.sum()

        gamma = 2.0 / (it + 2.0)
        d = g - a
        a_new = a + gamma * d
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best = a_new.copy()
            a = a_new
            stall = 0
            if verbose and it % 50 == 0:
                print(f"    FW iter {it}: J={best_score:.6f}", flush=True)
        else:
            # Try smaller step
            improved = False
            for factor in [0.5, 0.25, 0.1]:
                a_try = a + gamma * factor * d
                a_try = np.maximum(a_try, 0)
                s2 = a_try.sum()
                if s2 < 1e-10:
                    continue
                a_try = a_try / s2
                sc = compute_score(a_try)
                if sc < best_score:
                    best_score = sc
                    best = a_try.copy()
                    a = a_try
                    improved = True
                break
            stall += 1 if not improved else 0

            if stall > 50:
                break

    return best, best_score


def smooth_objective(x, p, n):
    """p-norm smoothed objective for normalized non-negative x."""
    x_pos = np.maximum(x, 1e-10)
    s = x_pos.sum()
    if s < 0.01:
        return 1e10
    x_norm = x_pos / s
    b = fast_conv(x_norm)
    b_max = b.max()
    if b_max < 1e-15:
        return 1e-15
    b_norm = b / b_max
    val = b_max * (np.mean(b_norm ** p)) ** (1.0 / p)
    return float(2.0 * n * val)


def smooth_refine(a, deadline, max_p=100):
    """Refine using p-norm smoothed objective with L-BFGS-B."""
    n = len(a)
    best = a.copy()
    best_score = compute_score(a)

    for p in [4, 16, 64, max_p]:
        if time.monotonic() >= deadline:
            break
        x0 = np.maximum(best, 1e-6)
        x0 = x0 / x0.sum()
        try:
            res = minimize(smooth_objective, x0, args=(p, n),
                           method='L-BFGS-B',
                           bounds=[(0, None)] * n,
                           options={'maxiter': 200, 'maxfun': 3000,
                                    'ftol': 1e-18, 'gtol': 1e-18})
            if res.fun < 1e10:
                res_x = np.maximum(res.x, 0)
                if res_x.sum() > 0.01:
                    score = compute_score(res_x)
                    if score < best_score:
                        best = res_x.copy()
                        best_score = score
        except Exception:
            pass

    return best, best_score


def subgradient_polish(a, deadline, max_iters=500000):
    """Subgradient method targeting the active max position."""
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    best = a.copy()
    best_score = compute_score(a)
    n = len(a)
    lr = 0.005
    stale = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        b = fast_conv(a)
        k = int(np.argmax(b))

        # Gradient of (a*a)_k wrt a: grad[m] = 2*a[k-m] if valid
        grad = np.zeros(n)
        m_idx = np.arange(n)
        km = k - m_idx
        valid = (km >= 0) & (km < n)
        grad[valid] = 2.0 * a[km[valid]]

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best = a_new.copy()
            a = a_new
            stale = 0
        else:
            a = a_new
            stale += 1
            if stale % 100 == 0:
                lr *= 0.95
            if lr < 1e-12:
                break

    return best, best_score


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    inits = []
    t_arr = np.linspace(0, 1, n)

    if parent is not None and parent.size > 1:
        t_old = np.linspace(0, 1, parent.size)
        resampled = np.maximum(np.interp(t_arr, t_old, parent), 0)
        if resampled.sum() > 0.01:
            inits.append(resampled / resampled.sum())

    inits.append(np.ones(n) / n)

    # Various cosines
    for freq, amp in [(1, 0.3), (2, 0.3), (3, 0.2), (1, 0.5), (2, 0.15)]:
        c = 1.0 + amp * np.cos(2 * np.pi * freq * t_arr)
        c = np.maximum(c, 0.01)
        inits.append(c / c.sum())

    # Random
    inits.append(rng.uniform(0.3, 1.0, n) / n)

    # Bimodal
    c = 1.0 + 0.4 * np.cos(2 * np.pi * t_arr) ** 2
    inits.append(c / c.sum())

    # Power-law shape
    c = 1.0 / (0.5 + t_arr)
    inits.append(c / c.sum())
    c = (1.0 - t_arr + 0.5)
    inits.append(c / c.sum())

    return inits
