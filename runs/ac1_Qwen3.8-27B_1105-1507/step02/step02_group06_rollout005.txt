"""
Optimization of step function sequences to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses power-law initializations with LP-based direction finding and subgradient descent.
Key insight: asymmetric power-law sequences (p_j ∝ j^α) achieve ~4/3 objective vs 2 for uniform.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, csc_matrix


def compute_score(f):
    n = len(f)
    conv = np.convolve(f, f)
    M = float(np.max(conv))
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf')
    return float(2 * n * M / (S * S))


def build_conv_matrix(f):
    """Build sparse matrix A where (A@g)[k] = sum_j f[k-j]*g[j] for valid j."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def subgradient_step(f, top_k=5):
    """Compute subgradient of max convolution and return step direction."""
    n = len(f)
    conv = np.convolve(f, f)
    
    # Get the top-k lags
    top_indices = np.argsort(conv)[-top_k:]
    max_val = conv[top_indices[0]] if top_indices[0] == np.argmax(conv) else conv.max()
    max_val = conv.max()
    
    # Weight by excess above 80th percentile
    thresh = np.percentile(conv, 80)
    weights = np.maximum(conv[top_indices] - thresh, 1e-10)
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for k, w in zip(top_indices, weights):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            grad[j] += w * f[k - j]
    
    return grad


def subgradient_optimize(f, deadline, lr=0.01, top_k=5, max_iter=10000):
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    momentum = np.zeros(n)
    decay = 0.9
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and it < max_iter:
        conv = np.convolve(f, f)
        M = float(np.max(conv))
        if M < 1e-15:
            break
        
        grad = subgradient_step(f, top_k)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        
        momentum = decay * momentum - lr * grad
        new_f = np.clip(f + momentum, 0, None)
        new_sum = float(np.sum(new_f))
        if new_sum < 1e-10:
            lr *= 0.5
            no_improve += 1
            if no_improve > 100:
                break
            it += 1
            continue
        new_f *= S / new_sum
        
        new_score = compute_score(new_f)
        it += 1
        
        if new_score < best_score:
            f = new_f
            best_f = f.copy()
            best_score = new_score
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.97
            momentum *= 0.5
            if lr < 1e-8 or no_improve > 500:
                break
    
    return best_f, best_score


def lp_optimize(f, deadline, step_init=0.1, max_lp_time=60):
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = compute_score(f)
    n = len(f)
    S = float(np.sum(f))
    step = step_init
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        if no_improve > 100:
            break
        
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M < 1e-15:
            break
        
        t_start = time.monotonic()
        g = solve_lp_direction(f, M)
        lp_time = time.monotonic() - t_start
        
        if g is None:
            no_improve += 5
            continue
        
        sum_g = float(np.sum(g))
        if sum_g < 1e-10:
            no_improve += 1
            continue
        
        g *= S / sum_g
        
        # Line search
        best_cand_score = best_score
        best_cand_f = None
        for alpha in [step, step*0.5, step*1.5, step*0.25, 0.9, 0.7]:
            if alpha < 0.01 or alpha > 1.0:
                continue
            cand = np.clip((1-alpha)*f + alpha*g, 0, None)
            cs = float(np.sum(cand))
            if cs < 1e-10:
                continue
            cand *= S / cs
            cs_score = compute_score(cand)
            if cs_score < best_cand_score:
                best_cand_score = cs_score
                best_cand_f = cand

            best_f = f.copy()
            best_score = best_cand_score
            no_improve = 0
            step = max(step * 0.99, 0.02)
            if it % 10 == 0:
                print(f"    LP iter {it}: score={best_score:.6f} (step={step:.3f}, time={lp_time:.1f}s)")
        else:
            no_improve += 1
            step *= 0.8
            if step < 0.01:
                break
    
    return best_f, best_score

    p /= np.sum(p)
    return p
