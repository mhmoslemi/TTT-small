    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):

            idx = k - j
            if 0 <= idx < n:

    return A

def normalize(a):

    if s < 1e-15:

    return a / s

def project_simplex(v):
    """Project v onto the simplex {x : sum(x)=1, x>=0}."""
    n = len(v)
    if np.all(v >= 0) and abs(np.sum(v) - 1) < 1e-10:
        return v
    # Standard projection algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = sum_j a[k-j] * b[j] = (a*b)[k]."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A[k, j] = a[k - j]
    return A

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

def softmax_obj_and_grad(a, beta):
    """Compute log-sum-exp objective and its gradient."""
    n = len(a)
    conv = np.convolve(a, a)
    shifted = beta * (conv - np.max(conv))
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    p = exp_vals / sum_exp
    lse = np.max(conv) + np.log(sum_exp) / beta
    
    # Gradient: sum_k p_k * 2 * a[k-i] for each i
    # grad[i] = 2 * sum_k p[k] * a[k-i] (where 0 <= k-i < n)
    grad = np.zeros(n)
    for i in range(n):
        # k ranges over valid values where 0 <= k-i < n, i.e., i <= k <= i+n-1
        k_lo = max(0, i)
        k_hi = min(2*n-2, i+n-1)
        j = k_lo - i  # j = k - i, ranges from max(0,i)-i=0 to min(2n-2,i+n-1)-i
        # Actually: grad[i] = 2 * sum_{k: 0<=k-i<n} p[k] * a[k-i]
        # Let j = k-i, then j ranges from max(0,-i) to min(2n-2-i, n-1)
        # j from 0 to min(n-1, 2n-2-i), and k = i+j ranges from i to i+min(n-1,2n-2-i)
        j_lo = 0
        j_hi = min(n-1, 2*n-2-i)
        grad[i] = 2.0 * np.dot(p[i:i+j_hi+1], a[j_lo:j_hi+1])
    
    return lse, grad

def projected_gd(a, time_budget=30.0, beta_init=10.0, lr_init=0.1):
    """Projected gradient descent with log-sum-exp smoothing on the simplex."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    start = time.monotonic()

    best_obj = compute_obj(a)
    beta = beta_init
    lr = lr_init

        _, grad = softmax_obj_and_grad(a, beta)

        else:

def simulated_annealing(a, time_budget=30.0, rng=None):
    """Simulated annealing on the simplex."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    T = compute_obj(a) * 0.01  # initial temperature
    T_min = T * 0.001
    
    while time.monotonic() - start < time_budget:
        # Generate move: randomly perturb a block
        width = max(1, int(n * rng.uniform(0.02, 0.15)))
        pos = int(rng.integers(0, n - width + 1))
        a_new = a.copy()
        a_new[pos:pos+width] *= np.exp(rng.normal(0, 0.3, width))
        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new)

            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
        
        T = max(T_min, T * 0.999)
    
    return best_a, best_obj

"""
Optimized step function coefficient search using correctly-formulated LP
alternating, projected gradient descent with log-sum-exp smoothing,
simulated annealing, symmetric optimization, and diverse multi-scale search.
Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
import time


def compute_obj(a):
    """Compute the objective: 2n * max(a*a) / (sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    """Normalize to sum to 1, handle zeros."""
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = (a*b)[k] (linear convolution)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        A[k, j_lo:j_hi + 1] = a[k - j_hi:k - j_lo + 1][::-1]
    return A


def lp_minimize_peak(a):
    """
    Solve: min t s.t. (a*b)_k <= t for all k, sum(b)=1, b>=0.
    Returns (b_optimal, t_optimal).
    """
    n = len(a)
    A = build_conv_matrix(a)
    # Variables: b_0,...,b_{n-1}, t => n+1 vars
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.status == 0:
        return np.maximum(res.x[:n], 0), float(res.x[-1])
    return None, None


def lp_alternating(a, max_iters=3000, time_budget=60.0):
    """LP alternating: iteratively solve the LP and interpolate."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() - start > time_budget:
            break
        b, t = lp_minimize_peak(a)
        if b is None:
            break
        b = normalize(b)
        obj = compute_obj(b)
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_a = b.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 50:
                # Try without interpolation
                b2 = normalize(np.maximum(b, 0))
                obj2 = compute_obj(b2)
                if obj2 < best_obj:
                    best_obj = obj2
                    best_a = b2.copy()
                    stall_count = 0

        # Adaptive interpolation
        alpha = 0.5 if it < 10 else (0.3 if it < 50 else 0.1)
        a = (1 - alpha) * a + alpha * b
        a = np.maximum(a, 0)
        a = normalize(a)

    return best_a, best_obj


def lse_gradient(a, beta):
    """
    Compute log-sum-exp smoothed objective and gradient.
    L_beta = (1/beta) * log(sum_k exp(beta * (a*a)_k))
    grad[i] = 2 * sum_{j: i+j valid} p[i+j] * a[j]
    """
    n = len(a)
    conv = np.convolve(a, a)
    max_c = np.max(conv)
    shifted = beta * (conv - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    p = exp_vals / sum_exp
    lse_val = max_c + np.log(sum_exp) / beta

    grad = np.zeros(n)
    for i in range(n):
        j_hi = min(n - 1, 2 * n - 2 - i)
        if j_hi >= 0:
            grad[i] = 2.0 * np.dot(p[i:i + j_hi + 1], a[0:j_hi + 1])

    return lse_val, grad


def projected_gd(a, time_budget=30.0, beta_init=20.0):
    """Projected gradient descent with log-sum-exp on the simplex."""
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    beta = beta_init
    lr = 0.05
    last_improve = 0

    for it in range(500000):
        if time.monotonic() - start > time_budget:
            break
        _, grad = lse_gradient(a, beta)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        # Adaptive learning rate
        step = lr / max(grad_norm, 1e-10)
        a_new = a - step * grad
        a_new = project_simplex(a_new)

        obj = compute_obj(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            last_improve = it
            lr = min(lr * 1.02, 0.5)
            beta *= 1.002
            if beta > 50:
                beta = 50
        else:
            lr *= 0.95
            if lr < 1e-10:
                break
            if it - last_improve > 500:
                beta = max(5.0, beta * 0.8)
                lr = 0.03
                last_improve = it

    return best_a, best_obj


def simulated_annealing(a, time_budget=30.0, rng=None):
    """Simulated annealing with block perturbations."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a)
    a = normalize(a.copy().astype(np.float64))
    start = time.monotonic()
    best_a = a.copy()
    best_obj = compute_obj(a)
    cur_obj = best_obj
    T = cur_obj * 0.005
    T_min = T * 0.01

    while time.monotonic() - start < time_budget:
        # Random perturbation
        style = rng.integers(0, 3)
        a_new = a.copy()
        if style == 0:
            # Block scaling
            width = max(1, int(n * rng.uniform(0.03, 0.2)))
            pos = int(rng.integers(0, n - width + 1))
            a_new[pos:pos + width] *= np.exp(rng.normal(0, 0.4, width))
        elif style == 1:
            # Single element
            idx = int(rng.integers(0, n))
            a_new[idx] *= np.exp(rng.normal(0, 0.5))
        else:
            # Gaussian bump
            center = int(rng.integers(0, n))
            width = max(2, int(n * rng.uniform(0.05, 0.2)))
            bump = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2) * rng.uniform(0.1, 0.5)
            a_new += bump * (rng.random() - 0.5)

        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new)
        obj_new = compute_obj(a_new)
        delta = obj_new - cur_obj

        if delta < 0 or rng.random() < np.exp(-max(delta, 0) / max(T, 1e-10)):
            a = a_new
            cur_obj = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()

        T = max(T_min, T * 0.9995)

    return best_a, best_obj


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    choice = int(rng.integers(0, 10))
    if choice == 0:
        a = np.ones(n)
    elif choice == 1:
        # Edge-concentrated with uniform interior
        a = np.full(n, 0.1)
        a[0:3] += 0.5
        a[-3:] += 0.5
        a += 0.02 * rng.random(n)
    elif choice == 2:
        # Center Gaussian
        center = n / 2.0
        width = n * rng.uniform(0.25, 0.5)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2) + 0.05
    elif choice == 3:
        # Cosine wave
        freq = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0, np.cos(2 * np.pi * freq * np.arange(n) / n + phase)) + 0.02
    elif choice == 4:
        # Bimodal
        a = np.zeros(n)
        p1, p2 = n // 4, 3 * n // 4
        w = max(3, n // 15)
        for p in [p1, p2]:
            for d in range(-w, w + 1):
                idx = p + d
                if 0 <= idx < n:
                    a[idx] += np.exp(-d * d / (2 * (w / 2) ** 2))
        a += 0.03
    elif choice == 5:
        # Inverse cosine (peaked at edges)
        x = np.arange(n)
        a = np.cos(np.pi * x / (n - 1)) ** 2 + 0.05
        a = np.maximum(0, a)
    elif choice == 6:
        # Periodic
        period = max(3, int(n / rng.integers(3, 8)))
        pattern = np.maximum(0.1, rng.random(period))
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 7:
        # Triangular
        center = n // 2
        a = 1.0 - np.abs(np.arange(n) - center) / center
        a = np.maximum(0.01, a)
    elif choice == 8:
        # Random smooth
        a = np.abs(rng.normal(0.5, 0.3, n))
        # Smooth with moving average
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    else:
        # Power law decay from center
        center = n // 2
        d = np.abs(np.arange(n) - center)
        alpha = rng.uniform(0.5, 2.0)
        a = 1.0 / (1.0 + d ** alpha)

    a = np.maximum(a, 0)
    return normalize(a)


def try_symmetric(a, time_budget=30.0):
    """Optimize with symmetry constraint a[i] = a[n-1-i]."""
    n = len(a)
    a_sym = normalize(0.5 * (a + a[::-1]))
    best_a = a_sym.copy()
    best_obj = compute_obj(a_sym)
    start = time.monotonic()

    for it in range(1000):
        if time.monotonic() - start > time_budget:
            break
        b, t = lp_minimize_peak(a_sym)
        if b is None:
            break
        b_sym = normalize(np.maximum(0.5 * (b + b[::-1]), 0))
        obj = compute_obj(b_sym)
        if obj < best_obj:
            best_obj = obj
            best_a = b_sym.copy()
        a_sym = normalize(0.5 * a_sym + 0.5 * b_sym)

    return best_a, best_obj


def full_pipeline(a_init, time_budget, rng=None):
    """Run the full optimization pipeline on a given initialization."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a_init)
    a = normalize(a_init.copy().astype(np.float64))
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()

    # Phase 1: LP alternating
    lp_time = min(time_budget * 0.45, 50.0)
    a1, obj1 = lp_alternating(a, max_iters=5000, time_budget=lp_time)
    if obj1 < best_obj:
        best_obj = obj1
        best_a = a1.copy()

    # Phase 2: Projected GD
    if time.monotonic() - start < time_budget * 0.8:
        gd_time = min(time_budget * 0.25, 25.0)
        a2, obj2 = projected_gd(a1, time_budget=gd_time)
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()

    # Phase 3: LP again from GD result
    if time.monotonic() - start < time_budget * 0.9:
        lp_time2 = min(time_budget * 0.15, 20.0)
        a3, obj3 = lp_alternating(a2, max_iters=3000, time_budget=lp_time2)
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()

    # Phase 4: Simulated annealing
    if time.monotonic() - start < time_budget * 0.97:
        sa_time = max(1.0, time_budget * 0.07)
        a4, obj4 = simulated_annealing(best_a, time_budget=sa_time, rng=rng)
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase optimization: LP alternating + projected GD + SA + symmetric.
    Tries parent, diverse initializations, multiple n values.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf

    # Try and optimize parent
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            parent = normalize(parent)
            obj = compute_obj(parent)
            if obj < best_score:
                best_score = obj
                best_seq = parent.copy()
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception as e:
        print(f"Parent error: {e}")

    # Optimize parent with full pipeline
    if parent is not None and parent.size <= 500:
        p_time = min(60.0, float(budget_s) * 0.12)
        p_opt, p_obj = full_pipeline(parent.copy(), p_time, rng)
        if p_obj < best_score:
            best_score = p_obj
            best_seq = p_opt.copy()
            print(f"Parent optimized: obj={p_obj:.6f}")

    # Try different n values
    n_values = [60, 80, 100, 120, 150, 180, 200, 250, 300, 350, 400, 500]

    for n in n_values:
        if time.monotonic() > deadline - 30:
            break
        remaining = deadline - time.monotonic()
        if remaining < 25:
            break

        # Adaptive budget
        num_remaining = len([x for x in n_values if x >= n])
        time_per_n = remaining / max(1, num_remaining)
        time_per_n = min(time_per_n, 70.0)

        if time_per_n < 12:
            continue

        num_restarts = max(1, min(3, int(time_per_n / 22)))
        time_per_restart = time_per_n / num_restarts

        print(f"\nN={n}: {num_restarts} restarts, ~{time_per_restart:.0f}s each")

        for r in range(num_restarts):
            if time.monotonic() > deadline - 20:
                break
            a_init = diverse_init(n, rng)
            a_opt, obj_opt = full_pipeline(a_init, time_per_restart, rng)

            if obj_opt < best_score:
                best_score = obj_opt
                best_seq = a_opt.copy()
                print(f"  n={n} r={r}: obj={obj_opt:.6f} ** NEW BEST **")
            else:
                print(f"  n={n} r={r}: obj={obj_opt:.6f}")

    # Symmetric optimization on best found
    if best_seq is not None and time.monotonic() < deadline - 15:
        t_sym = min(deadline - time.monotonic() - 5, 25.0)
        a_s, obj_s = try_symmetric(best_seq, time_budget=t_sym)
        if obj_s < best_score:
            best_score = obj_s
            best_seq = a_s.copy()
            print(f"Symmetric: obj={best_score:.6f}")

    # Final intensive LP + GD refinement
    if best_seq is not None and time.monotonic() < deadline - 8:
        remaining = deadline - time.monotonic() - 4
        if remaining > 15:
            a_f, obj_f = full_pipeline(best_seq.copy(), remaining, rng)
            if obj_f < best_score:
                best_score = obj_f
                best_seq = a_f.copy()
                print(f"Final refinement: obj={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(100)
        best_score = compute_obj(best_seq)

    # Ensure all values are non-negative and finite
    best_seq = np.maximum(best_seq, 0)
    best_seq = best_seq[np.isfinite(best_seq)]
    if len(best_seq) == 0:
        best_seq = np.ones(100)

    final_seq = [float(x) for x in best_seq]
    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(final_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return final_seq
