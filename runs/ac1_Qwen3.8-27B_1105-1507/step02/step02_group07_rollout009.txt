"""
Multi-phase optimization for minimizing 2n*max(conv(f,f))/sum(f)^2.
Uses Nesterov-accelerated softmax gradient, multiplicative weight updates,
LP direction finding, simulated annealing, and multi-length exploration.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def build_conv_matrix_sparse(f):
    """Build sparse matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    rows = np.array(rows, dtype=np.int32)
    cols = np.array(cols, dtype=np.int32)
    vals = np.array(vals, dtype=np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve: max sum(g) s.t. g >= 0, convolve(f,g)_k <= M for all k."""
    n = len(f)
    A = build_conv_matrix_sparse(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception:
        pass
    return None


def compute_softmax_gradient(f, beta):
    """Compute gradient of (1/beta)*log(sum exp(beta*C[k])) where C=conv(f,f)."""
    n = len(f)
    C = np.convolve(f, f)
    max_C = C.max()
    exp_C = np.exp(beta * (C - max_C))
    total = exp_C.sum()
    if total < 1e-300:
        return np.zeros(n), 0.0
    p = exp_C / total
    corr = np.correlate(p, f, mode='valid')
    grad = 2.0 * corr
    soft_max_val = max_C + np.log(total) / beta
    return grad, soft_max_val

def lp_iterate(f, deadline, step_size_init=0.1, max_iters=100, verbose=""):
    """LP-based iterations with bisection line search."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)
    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    it = 0
    no_improve = 0
    step_size = step_size_init

    A_cache = build_conv_matrix_sparse(f)

    while time.monotonic() < deadline and it < max_iters:
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))
        if M <= 0:
            break

        # Solve LP with cached matrix (rebuild periodically)
        if it % 10 == 0:
            A_cache = build_conv_matrix_sparse(f)
        g = solve_lp_direction(f, M)
        it += 1

        if g is None:
            no_improve += 1
            if no_improve > 15:
                break
            continue

        sum_g = float(np.sum(g))
        if sum_g <= 1e-15:
            no_improve += 1
            continue

        g = g / sum_g * S  # normalize to same sum

        # Bisection line search
        best_t = 0.0
        best_s = best_score
        lo, hi = 0.0, min(1.0, step_size * 3)

        for _ in range(8):
            mid = (lo + hi) / 2
            new_f = (1.0 - mid) * f + mid * g
            new_f = np.clip(new_f, 0.0, 1000.0)
            ns = float(np.sum(new_f))
            if ns < 1e-10:
                hi = mid
                continue
            new_f = new_f / ns
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = mid
                lo = mid
            else:
                hi = mid

        # Also try the original step_size and nearby
        for trial_t in [step_size, step_size * 1.5, step_size * 0.5]:
            if trial_t > 1.0:
                continue
            new_f = (1.0 - trial_t) * f + trial_t * g
            new_f = np.clip(new_f, 0.0, 1000.0)
            ns = float(np.sum(new_f))
            if ns < 1e-10:
                continue
            new_f = new_f / ns
            sc = evaluate_sequence(new_f.tolist())
            if np.isfinite(sc) and sc < best_s:
                best_s = sc
                best_t = trial_t

        if best_t > 0 and best_s < best_score:
            f = (1.0 - best_t) * f + best_t * g
            f = np.clip(f, 0.0, 1000.0)
            S = float(np.sum(f))
            f = f / S
            best_f = f.copy()
            best_score = best_s
            no_improve = 0
            if it % 20 == 1:
                print(f"    {verbose}LP iter {it}: score={best_score:.6f} (t={best_t:.4f})")
        else:
            no_improve += 1
            step_size *= 0.6
            if step_size < 0.003 or no_improve > 30:
                break

    return best_f, best_score


def simulated_annealing(f, deadline, T_start=0.02, T_end=0.001, max_iters=3000):
    """Simulated annealing with random perturbations."""
    f = f.copy().astype(np.float64)
    S = float(np.sum(f))
    if S < 1e-10:
        return f, evaluate_sequence(f.tolist())
    f = f / S
    n = len(f)

    best_f = f.copy()
    best_score = evaluate_sequence(f.tolist())
    current_score = best_score
    rng = np.random.default_rng(42 + n)
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        frac = min(1.0, it / max_iters)
        T = T_start * (T_end / T_start) ** frac

        # Random perturbation: modify 5-20 random elements
        n_perturb = rng.integers(5, 20)
        indices = rng.choice(n, size=n_perturb, replace=False)
        noise = 1.0 + T * rng.standard_normal(n_perturb)
        new_f = f.copy()
        new_f[indices] = np.clip(new_f[indices] * noise, 0.0, 1000.0)
        ns = float(np.sum(new_f))
        if ns < 1e-10:
            continue
        new_f = new_f / ns

        new_score = evaluate_sequence(new_f.tolist())
        if not np.isfinite(new_score):
            continue

        delta = new_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-10)):
            f = new_f
            current_score = new_score
            if new_score < best_score:
                best_f = f.copy()
                best_score = new_score
                if it % 500 == 0:
                    print(f"    SA iter {it}: best={best_score:.6f}, curr={current_score:.6f}, T={T:.5f}")
        # else: reject

    return best_f, best_score


def perturb(f, rng, strength=0.05):
    """Add multiplicative noise and renormalize."""
    noise = 1.0 + strength * rng.standard_normal(len(f))
    new_f = f * np.clip(noise, 0.1, 2.0)
    new_f = np.clip(new_f, 0.0, 1000.0)
    S = float(np.sum(new_f))
    if S < 1e-10:
        return f.copy()
    return new_f / S


def tukey_window(n, r=0.3):
    w = np.ones(n)
    m = int(np.floor(r * n / 2))
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def flat_top_window(n, gamma=0.2):
    """Flat-top window with raised cosine transitions."""
    w = np.ones(n)
    m = int(gamma * n / 2)
    if m > 0:
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (2 * m) - 1)))
            w[n - 1 - i] = w[i]
    return w


def super_gaussian_window(n, alpha=2.0):
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    w = np.exp(-((idx - center) / (n / 2.0)) ** (2 * alpha))
    return w


def full_optimize(f_init, deadline, verbose_prefix=""):
    """Run multi-phase optimization: LP + Nesterov + MW + SA."""
    f_init = f_init.copy().astype(np.float64)
    S = float(np.sum(f_init))
    if S < 1e-10:
        return f_init, evaluate_sequence(f_init.tolist())
    f_init = f_init / S

    best_f = f_init.copy()
    best_score = evaluate_sequence(f_init.tolist())
    t0 = time.monotonic()

    # Phase 1: Nesterov softmax (additive) - 35%
    p1_dead = t0 + (deadline - t0) * 0.35
    if time.monotonic() < p1_dead:
        f1, s1 = nesterov_softmax_optimize(f_init, p1_dead, beta_start=20.0,
                                            beta_end=300.0, lr_start=0.015, max_iters=6000,
                                            mode='additive')
        if s1 < best_score:
            best_f, best_score = f1, s1
            print(f"    {verbose_prefix}P1 (Nesterov add): {best_score:.6f}")

    # Phase 2: LP iterations - 20%
    p2_dead = t0 + (deadline - t0) * 0.55
    if time.monotonic() < p2_dead:
        f2, s2 = lp_iterate(best_f, p2_dead, step_size_init=0.08, max_iters=80,
                             verbose=verbose_prefix + "P2 ")
        if s2 < best_score:
            best_f, best_score = f2, s2
            print(f"    {verbose_prefix}P2 (LP): {best_score:.6f}")

    # Phase 3: Multiplicative weight - 20%
    p3_dead = t0 + (deadline - t0) * 0.75
    if time.monotonic() < p3_dead:
        f3, s3 = mp_w_optimize(best_f, p3_dead, lr_start=0.05, max_iters=4000)
        if s3 < best_score:
            best_f, best_score = f3, s3
            print(f"    {verbose_prefix}P3 (MW): {best_score:.6f}")

    # Phase 4: Nesterov softmax (multiplicative) - 15%
    p4_dead = t0 + (deadline - t0) * 0.90
    if time.monotonic() < p4_dead:
        f4, s4 = nesterov_softmax_optimize(best_f, p4_dead, beta_start=100.0,
                                            beta_end=800.0, lr_start=0.008, max_iters=4000,
                                            mode='multiplicative')
        if s4 < best_score:
            best_f, best_score = f4, s4
            print(f"    {verbose_prefix}P4 (Nesterov mult): {best_score:.6f}")

    # Phase 5: Simulated annealing - 10%
    if time.monotonic() < deadline - 2:
        f5, s5 = simulated_annealing(best_f, deadline - 2, T_start=0.01, T_end=0.0005,
                                      max_iters=2000)
        if s5 < best_score:
            best_f, best_score = f5, s5
            print(f"    {verbose_prefix}P5 (SA): {best_score:.6f}")

    return best_f, best_score


def propose_candidate(budget_s=1500, **kwargs):
    """Main search function: multi-length, multi-init, multi-phase optimization."""
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 10.0)

    best_sequence = None
    best_score = float('inf')
    rng = np.random.default_rng(12345)

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).copy()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim == 1 and parent.size > 0 and np.all(np.isfinite(parent)):
            S = float(np.sum(parent))
            if S > 0.01:
                score = evaluate_sequence(parent.tolist())
                if np.isfinite(score):
                    best_score = score
                    best_sequence = parent.copy()
                    print(f"Parent n={parent.size}: score={score:.6f}")
    except (NameError, Exception) as e:
        print(f"Could not load parent: {e}")

    # Define lengths and time budgets
    # Primary focus on n=512 (where we're closest to target), then explore others
    lengths_to_try = [512, 768, 1024, 384]
    time_budgets = {512: 0.40, 768: 0.20, 1024: 0.15, 384: 0.10}
    total_budget = deadline - t0

    for n in lengths_to_try:
        if time.monotonic() > deadline - 60:
            print("Time running out, stopping length exploration")
            break

        n_budget = total_budget * time_budgets.get(n, 0.1)
        n_deadline = time.monotonic() + n_budget
        print(f"\n{'='*60}")
        print(f"=== n={n} (budget {n_budget:.0f}s, elapsed {time.monotonic()-t0:.0f}s) ===")
        print(f"{'='*60}")

        # Generate initializations for this length
        inits = []

        # If parent has this exact length, use it
        if best_sequence is not None and len(best_sequence) == n:
            inits.append(('parent', best_sequence.copy()))

        # Tukey windows
        for r in [0.2, 0.35, 0.5]:
            inits.append((f'tukey_{r}', tukey_window(n, r)))

        # Flat-top
        inits.append(('flat_top', flat_top_window(n, 0.25)))

        # Super-Gaussian
        for alpha in [1.5, 2.5]:
            inits.append((f'sg_{alpha}', super_gaussian_window(n, alpha)))

        # Random smooth
        for seed in [7, 13]:
            r_rng = np.random.default_rng(seed * 1000 + n)
            x = r_rng.standard_normal(n)
            kernel_size = max(1, n // 20)
            kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size + 1) / (kernel_size / 2.5)) ** 2)
            kernel = kernel / kernel.sum()
            smooth_x = np.convolve(x, kernel, mode='same')
            smooth_x = np.abs(smooth_x) + 0.05
            inits.append((f'rand_smooth_{seed}', smooth_x))

        # Interpolated parent (if different length)
        if best_sequence is not None and len(best_sequence) != n:
            from numpy import interp
            parent_interp = interp(np.linspace(0, 1, n),
                                   np.linspace(0, 1, len(best_sequence)),
                                   best_sequence)
            parent_interp = np.clip(parent_interp, 0, None)
            inits.insert(0, ('parent_interp', parent_interp))

        # Optimize each init with limited time
        n_inits = len(inits)
        # Do at most 4 full optimizations, quick-check the rest
        max_full = min(4, n_inits)
        per_init_time = n_budget / max_full * 0.85

        # Sort inits by initial score (best first)
        scored_inits = []
        for label, f_init in inits:
            f_init = np.clip(f_init.astype(np.float64), 0.0, 1000.0)
            S = float(np.sum(f_init))
            if S < 0.01:
                continue
            f_init = f_init / S
            sc = evaluate_sequence(f_init.tolist())
            if np.isfinite(sc):
                scored_inits.append((sc, label, f_init))
                if sc < best_score:
                    best_score = sc
                    best_sequence = f_init.copy()

        scored_inits.sort(key=lambda x: x[0])

        for i, (init_sc, label, f_init) in enumerate(scored_inits):
            if time.monotonic() > n_deadline:
                break

            # Only do full optimization for the best few inits
            if i < max_full:
                init_deadline = min(n_deadline, time.monotonic() + per_init_time)
                if time.monotonic() < init_deadline - 10:
                    try:
                        bf, bs = full_optimize(f_init, init_deadline,
                                               verbose_prefix=f"[n={n}, {label}] ")
                        if bs < best_score:
                            best_score = bs
                            best_sequence = bf.copy()
                            print(f"  [n={n}, {label}] NEW BEST: {bs:.6f}")
                        else:
                            print(f"  [n={n}, {label}] best after opt: {bs:.6f}")
                    except Exception as e:
                        print(f"  [n={n}, {label}] error: {e}")
            else:
                # Quick optimization (just Nesterov with few iters)
                quick_dead = time.monotonic() + 15.0
                try:
                    bf, bs = nesterov_softmax_optimize(f_init, quick_dead, beta_start=30.0,
                                                         beta_end=100.0, lr_start=0.02,
                                                         max_iters=2000, mode='additive')
                    if bs < best_score:
                        best_score = bs
                        best_sequence = bf.copy()
                        print(f"  [n={n}, {label}] quick: {bs:.6f}")
                except Exception:
                    pass

    # === Perturbation restarts on the best ===
    if best_sequence is not None and time.monotonic() < deadline - 30:
        print(f"\n{'='*60}")
        print(f"=== Perturbation restarts (current best: {best_score:.6f}) ===")
        n_best = len(best_sequence)
        restart_budget = deadline - time.monotonic() - 15

        # Try multiple perturbation strengths
        strengths = [0.02, 0.04, 0.06, 0.08, 0.12, 0.15]
        n_restarts = min(len(strengths), max(3, int(restart_budget / 40)))
        per_restart = restart_budget / n_restarts

        for r in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break

            strength = strengths[r % len(strengths)]
            f_pert = perturb(best_sequence, rng, strength=strength)
            f_pert_deadline = time.monotonic() + per_restart * 0.75

            try:
                # For restarts, use a lighter optimization
                bf, bs = full_optimize(f_pert, f_pert_deadline,
                                       verbose_prefix=f"[restart {r}, str={strength}] ")
                if bs < best_score:
                    best_score = bs
                    best_sequence = bf.copy()
                    print(f"  Restart {r} (str={strength}): NEW BEST {bs:.6f}")
                else:
                    print(f"  Restart {r} (str={strength}): {bs:.6f} (no improvement)")
            except Exception as e:
                print(f"  Restart {r} error: {e}")

    # === Final refinement on absolute best ===
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print(f"\n{'='*60}")
        print(f"=== Final refinement (n={len(best_sequence)}, score={best_score:.6f}) ===")
        final_dead = deadline - 3
        try:
            # Quick LP
            f_lp, s_lp = lp_iterate(best_sequence, min(final_dead - 10, time.monotonic() + 30),
                                     step_size_init=0.03, max_iters=30, verbose="[finLP] ")
            if s_lp < best_score:
                best_score = s_lp
                best_sequence = f_lp.copy()

            # Final Nesterov with high beta
            f_ns, s_ns = nesterov_softmax_optimize(best_sequence, final_dead, beta_start=200.0,
                                                    beta_end=1000.0, lr_start=0.003,
                                                    max_iters=3000, mode='additive')
            if s_ns < best_score:
                best_score = s_ns
                best_sequence = f_ns.copy()

            # Final MW
            f_mw, s_mw = mp_w_optimize(best_sequence, min(final_dead, time.monotonic() + 15),
                                        lr_start=0.02, max_iters=2000)
            if s_mw < best_score:
                best_score = s_mw
                best_sequence = f_mw.copy()
        except Exception as e:
            print(f"  Final refinement error: {e}")

        print(f"  Final score: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = tukey_window(512, 0.3)
        best_score = evaluate_sequence(best_sequence.tolist())
        print(f"Fallback used: score={best_score:.6f}")

    # Ensure valid output
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    S = float(np.sum(best_sequence))
    if S < 0.01:
        best_sequence = np.ones(256)
        best_score = evaluate_sequence(best_sequence.tolist())

    elapsed = time.monotonic() - t0
    print(f"\n{'='*60}")
    print(f"FINAL: n={len(best_sequence)}, score={best_score:.6f}, time={elapsed:.1f}s")
    print(f"{'='*60}")
    return [float(x) for x in best_sequence]
