```python
import numpy as np

import time

def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

def build_A(a):
    n = len(a)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        A[k, j_start:j_end+1] = a[k-j_end:k-j_start+1][::-1]
    return A

def solve_lp_g(a, M):
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_alternate(a_init, max_iter=200, t=0.5, relax=1.0):
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    a = a / s
    
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    
    for it in range(max_iter):
        conv = np.convolve(a, a)
        M = float(np.max(conv)) * relax
        g = solve_lp_g(a, M)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        a_new = (1-t)*a + t*g
        a_new = np.maximum(a_new, 0)
        s = np.sum(a_new)
        if s < 1e-10:
            a = g
        else:
            a = a_new / s
        
        obj = compute_obj(a)
        if obj < best_obj:

            best_a = a.copy()
    
    return best_a, best_obj
```

```python
            #   conv_new[2j] = conv[2j] + (x^2-v^2)

                # So: conv_new = conv + 2*(x-v)*shift + (x-v)^2 * [has_self at k_self]

                if has_self:
                    conv_new[k_self] += (x-v)**2

                a[j] = res.x

                a[j] = res.x
            else:
                a[j] = res.x

                improved = True
                best_obj = new_obj
                best_a = a.copy()
        
        if not improved:
            break
    
    return best_a, best_obj
```

```python
        # Or just use a directly with constraints
        a = np.maximum(params, 0)
        s = np.sum(a)
        if s < 1e-10:
            return 1e10
        a = a / s
        conv = np.convolve(a, a)
        # Smooth max: (1/T) * log(sum(exp(T * conv)))
        # For numerical stability:
        max_c = np.max(conv)
        smooth_max = max_c + np.log(np.mean(np.exp(T*(conv - max_c))))/T
        return 2*n*smooth_max
    
    res = minimize(smooth_obj, a0, method='L-BFGS-B', 
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    
    a_opt = np.maximum(res.x, 0)
    s = np.sum(a_opt)
    if s > 1e-10:
        a_opt = a_opt / s
    else:
        a_opt = a0
    
    return a_opt, compute_obj(a_opt)
```

The problem min max(conv(a,a)) s.t. sum(a)=1, a>=0 is related to finding the minimax vector for a set of quadratic forms. The quadratic form for lag k is a^T T_k a where T_k is the Toeplitz matrix.

The minimum of max_k a^T T_k a over the probability simplex is related to the "fractional covering" or "game" interpretation. The value of the game is:
V = max_{p in simplex} min_{q in simplex} p^T (sum_k q_k T_k) p

By von Neumann's minimax theorem, this equals:
V = min_{q in simplex} max_{p in simplex} p^T (sum_k q_k T_k) p

The inner maximization is the largest eigenvalue of (sum_k q_k T_k) (restricted to non-negative vectors, but for PSD matrices it's the largest eigenvalue).

So V = min_{q: q>=0, sum(q)=1} lambda_max(sum_k q_k T_k)

This is a convex optimization problem (minimizing the maximum eigenvalue of a linear matrix function is convex)! We can solve it with SDP or with the "bisection + LMI" approach.

But this gives us the optimal value and the optimal q (the "dual" distribution over lags), not directly the optimal a. To get a, we need to find the eigenvector corresponding to lambda_max at the optimal q.

```python
"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP alternating, fast incremental coordinate descent, smoothed L-BFGS,
and subgradient polishing with diverse initializations and multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog, minimize_scalar, minimize
import time


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def build_A(a):
    """Build constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        A[k, j_start:j_end + 1] = a[k - j_end:k - j_start + 1][::-1]
    return A


def solve_lp_g(a, M):
    """Solve LP: max sum(g) s.t. (a*g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_alternate(a_init, max_iter=100, t=0.5, relax=1.0, time_limit=None):
    """LP alternating optimization with adaptive blending."""
    a = a_init.copy().astype(np.float64)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    n = len(a)
    best_a = a.copy()
    best_obj = compute_obj(best_a)
    start_t = time.monotonic()

    for it in range(max_iter):
        if time_limit is not None and time.monotonic() - start_t > time_limit:
            break

        conv = np.convolve(a, a)
        M = float(np.max(conv)) * relax
        g = solve_lp_g(a, M)
        if g is None:
            break
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive blending: decrease t as we converge
        t_eff = t * (1.0 - 0.3 * it / max_iter)
        a_new = (1.0 - t_eff) * a + t_eff * g
        a_new = np.maximum(a_new, 0.0)
        s = np.sum(a_new)
        if s < 1e-10:
            a = g / g_sum
        else:
            a = a_new / s

        obj = compute_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def coord_descent_fast(a_init, n_rounds=15, xatol=1e-8):
    """Fast coordinate descent with incremental convolution updates."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    else:
        a = a / S

    conv = np.convolve(a, a).copy()
    best_a = a.copy()
    best_obj = 2.0 * n * float(np.max(conv)) / (S * S)
    conv_len = 2 * n - 1

    for rnd in range(n_rounds):
        improved = False

        for j in range(n):
            v = a[j]

            # Precompute shift: shift[k] = a[k-j] if 0<=k-j<n else 0
            shift = np.zeros(conv_len)
            j_start = j
            j_end = j + n
            shift[j_start:j_end] = a

            k_self = 2 * j
            has_self = 0 <= k_self < conv_len

            # Define objective for x (new value of a[j])
            def f_x(x):
                s_new = S - v + x
                if s_new < 1e-10:
                    return 1e10
                # conv_new = conv + 2*(x-v)*shift + (x-v)^2 at k_self
                delta = x - v
                if abs(delta) < 1e-15:
                    max_c = float(np.max(conv))
                    return 2.0 * n * max_c / (s_new * s_new)
                conv_new = conv + 2.0 * delta * shift
                if has_self:
                    conv_new[k_self] += delta * delta
                max_c = float(np.max(conv_new))
                return 2.0 * n * max_c / (s_new * s_new)

            # Upper bound for x
            x_upper = max(v * 8.0 + 0.05, S * 0.3)
            if x_upper < 0.01:
                x_upper = 0.1

            try:
                res = minimize_scalar(f_x, bounds=(0.0, x_upper),
                                      method='bounded', options={'xatol': xatol})
                x_new = float(res.x)
            except Exception:
                x_new = v

            x_new = max(0.0, min(x_new, x_upper))

            if abs(x_new - v) > 1e-10:
                delta = x_new - v
                conv += 2.0 * delta * shift
                if has_self:
                    conv[k_self] += delta * delta
                a[j] = x_new
                S += delta

                new_obj = 2.0 * n * float(np.max(conv)) / (S * S)
                if new_obj < best_obj - 1e-12:
                    improved = True
                    best_obj = new_obj
                    best_a = a.copy()
                elif new_obj > best_obj + 1e-6:
                    # Revert if got much worse (numerical issues)
                    conv -= 2.0 * delta * shift
                    if has_self:
                        conv[k_self] -= delta * delta
                    a[j] = v
                    S -= delta

        if not improved:
            break

    return best_a, best_obj


def smoothed_optimize(a_init, T=30.0, max_iter=300):
    """Minimize smoothed max(conv(a,a)) using L-BFGS-B. Normalize sum=1."""
    n = len(a_init)
    a0 = a_init.copy().astype(np.float64)
    S = float(np.sum(a0))
    if S < 1e-10:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / S

    def smooth_obj(x):
        a = np.maximum(x, 0.0)
        s = float(np.sum(a))
        if s < 1e-10:
            return 1e10
        a = a / s
        conv = np.convolve(a, a)
        max_c = float(np.max(conv))
        # Log-sum-exp smoothing
        scaled = T * (conv - max_c)
        # Clip for numerical stability
        scaled = np.clip(scaled, -50, 50)
        smooth_max = max_c + np.log(float(np.mean(np.exp(scaled)))) / T
        return 2.0 * n * smooth_max

    try:
        res = minimize(smooth_obj, a0, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
        a_opt = np.maximum(res.x, 0.0)
        s_opt = float(np.sum(a_opt))
        if s_opt > 1e-10:
            a_opt = a_opt / s_opt
        else:
            a_opt = a0
    except Exception:
        a_opt = a0

    return a_opt, compute_obj(a_opt)


def subgradient_polish(a_init, n_steps=2000):
    """Subgradient method for polishing."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    S = float(np.sum(a))
    a = a / S

    best_a = a.copy()
    best_obj = compute_obj(best_a)

    for step in range(n_steps):
        conv = np.convolve(a, a)
        max_h = float(np.max(conv))
        if max_h < 1e-15:
            break

        # Active set with relative tolerance
        tol = max_h * (0.0005 + 0.005 * (step / n_steps))
        active = conv >= max_h - tol
        n_active = int(np.sum(active))
        if n_active == 0:
            active = conv >= max_h - 1e-10
            n_active = int(np.sum(active))

        # Compute subgradient
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k in active_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            vals = a[k - j_end:k - j_start + 1][::-1]
            grad[j_start:j_end + 1] += 2.0 * vals
            if k % 2 == 0 and 0 <= k // 2 < n:
                grad[k // 2] -= a[k // 2]
        grad /= n_active

        # Adaptive step size with decay
        step_size = 0.3 / (1.0 + 0.005 * step)

        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s = float(np.sum(a_new))
        if s < 1e-10:
            break
        a_new = a_new / s

        obj_new = compute_obj(a_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

    return best_a, best_obj


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 8))
    x = np.arange(n)
    if choice == 0:
        freq = rng.uniform(1.0, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = np.maximum(0, np.cos(2 * np.pi * freq * x / n + phase))
    elif choice == 1:
        # Two-lobe
        mid = n // 2
        a = np.ones(n)
        a[:mid] *= rng.uniform(0.3, 2.0)
        a[mid:] *= rng.uniform(0.3, 2.0)
    elif choice == 2:
        # Periodic
        period = int(rng.integers(max(2, n // 20), max(3, n // 3)))
        pattern = rng.random(period)
        a = np.tile(pattern, n // period + 1)[:n]
    elif choice == 3:
        # Gaussian window
        center = rng.uniform(0.25, 0.75) * n
        width = rng.uniform(0.1, 0.4) * n
        a = np.exp(-0.5 * ((x - center) / width) ** 2)
    elif choice == 4:
        # Chirp-like
        a = np.abs(np.sin(2 * np.pi * rng.uniform(1, 10) * x / n + 0.5 * x / n))
    elif choice == 5:
        # Triangular
        a = 1.0 - 2.0 * np.abs(x / n - 0.5)
    elif choice == 6:
        # Multi-frequency
        f1, f2 = rng.uniform(1, 8), rng.uniform(2, 15)
        a = np.maximum(0, 0.7 * np.cos(2 * np.pi * f1 * x / n) + 0.3 * np.cos(2 * np.pi * f2 * x / n))
    else:
        a = rng.random(n)

    s = float(np.sum(a))
    if s < 1e-10:
        a = rng.random(n)
        s = float(np.sum(a))
    return a / s


def resample_sequence(seq, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    old_x = np.arange(old_n, dtype=np.float64)
    new_x = np.linspace(0, old_n - 1, new_n)
    resampled = np.interp(new_x, old_x, seq)
    return np.maximum(resampled, 0.0)


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """
    Multi-phase search combining LP alternation, coordinate descent,
    smoothed optimization, and subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0

    best_seq = None
    best_score = np.inf

    def update_best(a, obj):
        nonlocal best_seq, best_score
        if obj < best_score:
            best_score = obj
            best_seq = a.copy()
            print(f"  NEW BEST: n={len(a)}, score={obj:.6f}")

    # Try parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0, 1000.0)
        if parent.size > 0 and np.isfinite(parent).all():
            obj = compute_obj(parent)
            update_best(parent, obj)
            print(f"Parent (n={parent.size}): obj={obj:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    n_parent = len(best_seq) if best_seq is not None else 300

    # Define n values to try, prioritizing near parent
    n_candidates = sorted(set([
        n_parent,
        max(50, n_parent - 20), max(50, n_parent - 10),
        n_parent + 5, n_parent + 10, n_parent + 20,
        max(50, n_parent - 50), n_parent + 50, n_parent + 100,
        200, 250, 350, 400, 500
    ]))
    n_candidates = [n for n in n_candidates if 50 <= n <= 800]

    # Sort by priority: parent n first, then nearby
    n_sorted = sorted(n_candidates, key=lambda x: abs(x - n_parent))

    print(f"Trying n values: {n_sorted[:10]}")
    print(f"Time budget: {budget_s}s, current best: {best_score:.6f}")

    for n in n_sorted:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 15:
            break

        print(f"\n=== N={n} ({remaining:.0f}s remaining, best={best_score:.6f}) ===")

        # --- Initialization set ---
        inits = []
        if best_seq is not None and len(best_seq) == n:
            inits.append(best_seq.copy())
        elif best_seq is not None and len(best_seq) != n:
            inits.append(resample_sequence(best_seq, n))
        # Add structured inits
        inits.append(structured_init(n, rng))
        if remaining > 60:
            inits.append(structured_init(n, rng))

        for init_idx, a_init in enumerate(inits):
            if time.monotonic() > deadline:
                break
            remaining = deadline - time.monotonic()
            if remaining < 10:
                break

            a_init = a_init.copy().astype(np.float64)
            s = float(np.sum(a_init))
            if s < 1e-10:
                continue
            a_init = a_init / s

            # Phase A: LP Alternating
            lp_time = min(remaining * 0.3, 120.0)
            max_iter_lp = int(min(150, max(30, lp_time / 0.5)))
            a1, obj1 = lp_alternate(a_init, max_iter=max_iter_lp, t=0.5, relax=1.0,
                                    time_limit=lp_time)
            update_best(a1, obj1)

            # Phase B: Coordinate Descent
            a2, obj2 = coord_descent_fast(a1, n_rounds=12)
            update_best(a2, obj2)

            # Phase C: Smoothed L-BFGS
            if time.monotonic() < deadline - 20:
                T_val = 30.0 if n <= 400 else 15.0
                a3, obj3 = smoothed_optimize(a2, T=T_val, max_iter=200)
                update_best(a3, obj3)

                # Phase D: More coordinate descent after smoothing
                a4, obj4 = coord_descent_fast(a3, n_rounds=10)
                update_best(a4, obj4)

            # Phase E: Subgradient polish (if time)
            if time.monotonic() < deadline - 15:
                ns = int(min(1500, (deadline - time.monotonic()) / 0.01))
                if ns > 200:
                    a5, obj5 = subgradient_polish(a4, n_steps=ns)
                    update_best(a5, obj5)
                    # Quick coordinate descent after subgradient
                    a6, obj6 = coord_descent_fast(a5, n_rounds=8)
                    update_best(a6, obj6)

    # --- Final refinement on best solution ---
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        print(f"\n=== FINAL REFINEMENT ({remaining:.0f}s left, best={best_score:.6f}) ===")

        # Extended coordinate descent
        a_f1, obj_f1 = coord_descent_fast(best_seq, n_rounds=25, xatol=1e-10)
        update_best(a_f1, obj_f1)

        # Smoothed with higher T
        if time.monotonic() < deadline - 15:
            a_f2, obj_f2 = smoothed_optimize(a_f1, T=50.0, max_iter=300)
            update_best(a_f2, obj_f2)
            a_f3, obj_f3 = coord_descent_fast(a_f2, n_rounds=15, xatol=1e-10)
            update_best(a_f3, obj_f3)

        # One more LP pass
        if time.monotonic() < deadline - 20 and len(best_seq) <= 500:
            a_f4, obj_f4 = lp_alternate(a_f3, max_iter=50, t=0.4, relax=0.99,
                                        time_limit=min(60, (deadline - time.monotonic()) - 5))
            update_best(a_f4, obj_f4)
            a_f5, obj_f5 = coord_descent_fast(a_f4, n_rounds=15, xatol=1e-10)
            update_best(a_f5, obj_f5)

        # Random perturbation + re-optimize (to escape local minima)
        if time.monotonic() < deadline - 30:
            for perturb_trial in range(3):
                if time.monotonic() > deadline - 10:
                    break
                noise = rng.normal(0, 0.02, size=len(best_seq))
                a_pert = best_seq + noise
                a_pert = np.maximum(a_pert, 0)
                s_p = float(np.sum(a_pert))
                if s_p < 1e-10:
                    continue
                a_pert = a_pert / s_p
                a_r1, obj_r1 = coord_descent_fast(a_pert, n_rounds=15)
                update_best(a_r1, obj_r1)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(300)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")

    # Ensure valid output
    result = [float(max(0, x)) for x in best_seq]
    return result
```
