import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import lil_matrix

def fast_conv(a):
    """Full self-convolution via FFT, O(n log n)."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_J(a):
    """Compute J(a) = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_conv(a)
    return float(2 * len(a) * b.max() / (s * s))


def build_A_sparse(a):
    """Build sparse constraint matrix for LP: A[k,j] = a[k-j] if valid."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            cnt = jh - jl + 1
            rows.extend([k] * cnt)
            cols.extend(range(jl, jh + 1))
            segment = a[jl:jh + 1]
            # A[k, j] = a[k - j] for j in [jl, jh]
            # k - j ranges from k-jh to k-jl
            vals.extend(a[k - jh:k - jl + 1][::-1].tolist())
    A = lil_matrix((2 * n - 1, n))
    for idx in range(len(rows)):
        A[rows[idx], cols[idx]] = vals[idx]
    return A.tocsr()


def lp_best_response(a, M):
    """LP: max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    A = build_A_sparse(a)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    r = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        if g.sum() > 1e-10:
            return g
    return None


def coordinate_descent_step(a, max_passes=3, deadline=None):
    """Coordinate descent: for each j, find best a_j via candidate evaluation."""
    n = len(a)
    a = a.copy()
    S = a.sum()
    
    for _pass in range(max_passes):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            old_val = a[j]
            S_rest = S - old_val
            
            # Candidate values for a[j]
            candidates = [0.0, old_val, old_val * 0.5, old_val * 1.5]
            
            # Find value that equalizes b_0 with current max of other b_k
            b = fast_conv(a)
            M = b.max()
            b0 = b[0]  # = sum a_i^2
            # Want a_j^2 contribution to make b_0 smaller
            # b_0 = a_j^2 + sum_{i!=j} a_i^2
            # If we set a_j = x, b_0_new = x^2 + (b_0 - old_val^2)
            # Other b_k change linearly: b_k_new = b_k + x*(a_{k-j}+a_{j-k}) - old_val*(a_{k-j}+a_{j-k})
            
            # Add candidate: value that makes b_0 = second_max
            second_max = np.partition(b, -2)[-2] if n > 1 else 0
            b0_rest = b0 - old_val * old_val
            # x^2 + b0_rest = second_max => x = sqrt(second_max - b0_rest)
            if second_max > b0_rest:
                candidates.append(np.sqrt(second_max - b0_rest))
            
            # Add candidate: current value scaled toward mean
            if S > 0:
                candidates.append(S / n)
            
            best_val = old_val
            best_J = compute_J(a)
            
            for cv in candidates:
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - old_val) < 1e-12:
                    continue
                a[j] = cv
                J_try = compute_J(a)
                if J_try < best_J - 1e-12:
                    best_J = J_try
                    best_val = cv
            
            if best_val != old_val:
                a[j] = best_val
                S = S_rest + best_val
                improved = True
            else:
                a[j] = old_val
                S = S_rest + old_val
        
        if not improved:
            break
    
    return a


def subgradient_step(a, step_size=0.01, num_averages=20):
    """Subgradient descent using active constraint gradients with averaging."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = a.sum()
    
    if S < 1e-10:
        return a
    
    # Find all active (near-max) constraints
    b = fast_conv(a)
    M = b.max()
    tolerance = 0.01 * M
    active = np.where(b >= M - tolerance)[0]
    
    # Average subgradient over active constraints
    grad = np.zeros(n)
    for k in active:
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                grad[j] += a[kj]
            jk = j - k
            if 0 <= jk < n and jk != kj:
                grad[j] += a[jk]
    
    grad /= len(active)
    
    # J = 2nM/S^2, dJ/da_j = 2n(M_aa_j*S - 2*M*grad_component)/S^3
    # Simplified: move in direction that reduces M/S^2
    # d(M/S^2)/da_j = (grad_j * S - 2*M) / S^2 (approximately, for the subgradient)
    
    descent = np.zeros(n)
    for j in range(n):
        # We want to decrease J, so move against the gradient of J
        # dJ/da_j ~ 2n * (grad_j * S - 2*M) / S^3
        # If positive, decrease a_j; if negative, increase a_j
        dj = 2.0 * n * (grad[j] * S - 2.0 * M) / (S ** 3)
        descent[j] = -dj
    
    # Normalize and apply
    d_norm = np.linalg.norm(descent)
    if d_norm > 1e-15:
        descent /= d_norm
    
    a_new = a + step_size * descent
    a_new = np.maximum(a_new, 0.0)
    
    if a_new.sum() < 1e-10:
        return a
    
    # Line search
    best_a = a
    best_J = compute_J(a)
    for alpha in [step_size, step_size * 0.5, step_size * 0.25, step_size * 2.0]:
        a_try = np.maximum(a + alpha * descent, 0.0)
        if a_try.sum() < 1e-10:
            continue
        J_try = compute_J(a_try)
        if J_try < best_J:
            best_J = J_try
            best_a = a_try
    
    return best_a


def resample_seq(src, dest_n):
    """Resample 1D sequence to new length via linear interpolation."""
    src = np.asarray(src, dtype=np.float64)
    if src.size == dest_n:
        return src.copy()
    t_old = np.linspace(0, 1, src.size)
    t_new = np.linspace(0, 1, dest_n)
    return np.maximum(np.interp(t_new, t_old, src), 0.0)


class ScoreTracker:
    """Tracks the best sequence found so far using evaluate_sequence."""

    def __init__(self):
        self.best = None
        self.best_score = float('inf')
        self.t_start = time.monotonic()

    def update(self, a):
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.size == 0 or a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < self.best_score:
            self.best = a.copy()
            self.best_score = score
            elapsed = time.monotonic() - self.t_start
            print(f"  * J={score:.8f} n={a.size} t={elapsed:.0f}s", flush=True)

    def report(self, label=""):
        if self.best is not None:
            print(f"  {label} best: J={self.best_score:.8f} n={self.best.size}", flush=True)


def run_lp_iterations(a, max_iters, deadline, tracker):
    """Run LP mirror descent iterations with adaptive step size."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        M = float(b.max())
        g = lp_best_response(a, M)
        if g is None:
            break

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Adaptive step
        base_t = max(0.05, 0.5 * (1.0 - it / float(2 * max(max_iters, 1))))
        candidates_t = [base_t, min(1.0, base_t * 1.5), max(0.02, base_t * 0.5)]

        best_a = None
        best_J = current_J

        for t in candidates_t:
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < best_J - 1e-14:
                best_J = J_trial
                best_a = a_trial

        if best_a is None:
            t_small = 0.02
            a_trial = (1.0 - t_small) * a + t_small * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                break
            a_trial = a_trial / s
            J_trial = compute_J(a_trial)
            if J_trial < current_J - 1e-14:
                a = a_trial
                current_J = J_trial
            else:
                break
        else:
            a = best_a
            current_J = best_J

        if (it + 1) % 25 == 0:
            tracker.update(a)

    tracker.update(a)
    return a


def run_subgradient_loop(a, max_iters, deadline, tracker):
    """Iterative subgradient descent with decreasing step size."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    step = 0.1
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        a = subgradient_step(a, step_size=step)
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            break
        a = a / a.sum()
        J_new = compute_J(a)
        if J_new < current_J:
            current_J = J_new
            step = max(0.01, step * 1.02)
        else:
            step = max(0.005, step * 0.8)
        
        if (it + 1) % 30 == 0:
            tracker.update(a)
    
    tracker.update(a)
    return a


def run_coordinate_descent(a, max_passes, deadline, tracker):
    """Run full coordinate descent."""
    a = np.maximum(a.astype(np.float64), 0.001)
    a = a / a.sum()
    current_J = compute_J(a)
    tracker.update(a)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        a = coordinate_descent_step(a, max_passes=1, deadline=deadline)
        J_new = compute_J(a)
        if J_new < current_J - 1e-10:
            current_J = J_new
            if (p + 1) % 5 == 0:
                tracker.update(a)
        elif p > 3:
            # Try larger perturbations to escape
            break
    
    tracker.update(a)
    return a


def generate_initializations(n, rng, parent):
    """Generate diverse initial sequences of length n."""
    inits = []

    if parent is not None:
        inits.append(resample_seq(parent, n))

    # Uniform (theoretical optimum if J>=2)
    inits.append(np.ones(n))

    inits.append(rng.uniform(0.3, 1.0, n))

    t_arr = np.linspace(0, 1, n, endpoint=False)
    for freq in [2, 3, 4, 5, 6, 8]:
        inits.append(np.maximum(1.0 + 0.4 * np.cos(2.0 * np.pi * freq * t_arr), 0.01))

    inits.append(np.maximum(1.0 + 0.3 * np.cos(2.0 * np.pi * t_arr ** 2 * 3), 0.01))
    inits.append(np.maximum(1.0 + 0.5 * (2.0 * t_arr - 1.0) ** 2, 0.01))
    inits.append(np.exp(-3.0 * (t_arr - 0.5) ** 2) + 0.05)
    inits.append(np.abs(np.sin(np.pi * t_arr)) + 0.1)
    inits.append(np.maximum(1.0 + 0.3 * np.sin(2.0 * np.pi * t_arr + rng.uniform(0, 6.28)), 0.01))

    # Sparse structure
    sparse = np.zeros(n)
    n_spikes = max(4, n // 24)
    positions = rng.choice(n, size=n_spikes, replace=False)
    sparse[positions] = 1.0
    kernel = np.ones(3) / 3.0
    inits.append(np.convolve(sparse, kernel, mode='same') + 0.05)

    # Power-law
    inits.append(np.maximum((t_arr + 0.1) ** rng.uniform(-1.5, 1.5), 0.01))

    # Triangle
    inits.append(1.0 - 2.0 * np.abs(t_arr - 0.5) + 0.1)

    # W-shaped
    inits.append(1.0 + 0.5 * np.cos(4.0 * np.pi * t_arr) * (1.0 - t_arr))

    return inits


def propose_candidate(seed=42, budget_s=1500, **kwargs):
    """Multi-strategy optimizer: LP + subgradient + coordinate descent + SA."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 30.0
    tracker = ScoreTracker()

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)):
            parent = np.maximum(p, 0.0)
            tracker.update(parent)
            print(f"Parent loaded: n={parent.size}", flush=True)
    except (NameError, AttributeError):
        pass

    # === Phase 0: Quick evaluation of structured candidates ===
    print("\n=== Phase 0: Structured candidates ===", flush=True)
    for n in [16, 32, 64, 128, 256, 512, 1024]:
        if time.monotonic() > deadline:
            break
        tracker.update(np.ones(n))
        # Triangular
        t = np.linspace(0, 1, n)
        tracker.update(1.0 - np.abs(2*t - 1))
        # Gaussian
        tracker.update(np.exp(-4*(t - 0.5)**2) + 0.01)

    tracker.report("Phase 0")

    # === Phase 1: LP Mirror Descent across lengths ===
    lengths_to_try = [128, 256, 512, 1024]
    if parent is not None and parent.size not in lengths_to_try:
        lengths_to_try.insert(0, parent.size)
    
    # Remove duplicates, keep order
    seen = set()
    unique_lengths = []
    for l in lengths_to_try:
        if l not in seen:
            seen.add(l)
            unique_lengths.append(l)
    lengths_to_try = unique_lengths

    for n in lengths_to_try:
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        print(f"\n{'='*50} LP n={n} {'='*50}", flush=True)

        inits = generate_initializations(n, rng, parent)
        iters_per_init = max(30, min(200, int(60000 // n)))

        for idx, a0 in enumerate(inits):
            if time.monotonic() > deadline:
                break
            a0 = np.maximum(a0.astype(np.float64), 0.001)
            a0 = a0 / a0.sum()
            a = run_lp_iterations(a0, iters_per_init, deadline, tracker)
            
            # Follow up with coordinate descent on the LP result
            if time.monotonic() < deadline and a.sum() > 0.01:
                a = run_coordinate_descent(a, max_passes=5, deadline=deadline, tracker=tracker)

        tracker.report(f"Phase1 n={n}: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Subgradient + Coordinate Descent alternating ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} Subgrad+Coord n={n} {'='*50}", flush=True)

        a = tracker.best.copy()
        for round_id in range(5):
            if time.monotonic() > deadline:
                break
            a = run_subgradient_loop(a, max_iters=50, deadline=deadline, tracker=tracker)
            a = run_coordinate_descent(a, max_passes=3, deadline=deadline, tracker=tracker)
        
        tracker.report(f"Phase2: {time.monotonic()-t0:.1f}s")

    # === Phase 3: Simulated Annealing with diverse restarts ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} SA n={n} {'='*50}", flush=True)

        temperature = 0.1
        n_restarts = 40

        for r in range(n_restarts):
            if time.monotonic() > deadline:
                break
            a = tracker.best.copy().astype(np.float64)
            
            # Random perturbation
            noise = rng.normal(0, temperature, n)
            a = a + noise
            a = np.maximum(a, 0.001)
            a = a / a.sum()

            # Alternate between LP and subgradient
            if r % 2 == 0:
                a = run_lp_iterations(a, max(15, 60 - r), deadline, tracker)
            else:
                a = run_subgradient_loop(a, max(10, 40 - r // 2), deadline, tracker)
            
            # Polish with coordinate descent
            if time.monotonic() < deadline:
                a = run_coordinate_descent(a, max_passes=2, deadline=deadline, tracker=tracker)

            temperature *= 0.95

        tracker.report(f"SA: {time.monotonic()-t0:.1f}s")

    # === Phase 4: L-BFGS-B refinement ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n = tracker.best.size
        print(f"\n{'='*50} L-BFGS-B n={n} {'='*50}", flush=True)

        def lbb_objective(x):
            ax = np.maximum(x, 0.0)
            s = ax.sum()
            if s < 1e-8:
                return 1e10
            ax = ax / s
            b = fast_conv(ax)
            return float(b.max())

        bounds_list = [(0.0, 10.0)] * n
        try:
            res = minimize(lbb_objective, tracker.best.astype(np.float64),
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': 300, 'max_fun': 50000, 'ftol': 1e-16})
            if res.fun < 1e10:
                tracker.update(res.x)
        except Exception as e:
            print(f"  L-BFGS-B error: {e}", flush=True)

        tracker.report(f"L-BFGS-B: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Try to extend/shrink the best sequence ===
    if tracker.best is not None and time.monotonic() < deadline:
        t0 = time.monotonic()
        n_best = tracker.best.size
        print(f"\n{'='*50} Length adjustment (base n={n_best}) {'='*50}", flush=True)
        
        for n_try in [n_best // 2, n_best * 2, n_best + 64, max(16, n_best - 64)]:
            if time.monotonic() > deadline or n_try < 8 or n_try > 4096:
                continue
            a_resampled = resample_seq(tracker.best, n_try)
            a_resampled = np.maximum(a_resampled, 0.001)
            a_resampled = a_resampled / a_resampled.sum()
            tracker.update(a_resampled)
            
            # Quick LP polish
            a_resampled = run_lp_iterations(a_resampled, 30, deadline, tracker)
        
        tracker.report(f"Length adj: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Final polish with many LP iterations ===
    if tracker.best is not None and time.monotonic() < deadline:
        n = tracker.best.size
        if n <= 1024:
            t0 = time.monotonic()
            print(f"\n{'='*50} Final LP polish n={n} {'='*50}", flush=True)
            a = run_lp_iterations(tracker.best.copy(), 200, deadline, tracker)
            a = run_coordinate_descent(a, max_passes=10, deadline=deadline, tracker=tracker)
            tracker.report(f"Final: {time.monotonic()-t0:.1f}s")

    # Fallback
    if tracker.best is None:
        tracker.best = np.ones(128)
        tracker.best_score = float(evaluate_sequence(tracker.best.tolist()))

    print(f"\n{'='*50} FINAL {'='*50}", flush=True)
    print(f"J={tracker.best_score:.8f} n={tracker.best.size} time={time.monotonic()-tracker.t_start:.0f}s", flush=True)

    return [float(x) for x in tracker.best]
