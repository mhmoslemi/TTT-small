"""
Optimized step function coefficient search using min-M LP alternating,
FFT-accelerated gradient refinement with p-norm smoothing, and adaptive
multi-scale search. Targets objective below 1.5027.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import fft, ifft
import time


def compute_obj(a):
    n = len(a)
    if n == 0:
        return np.inf
    conv = np.convolve(a, a)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def normalize(a):
    s = np.sum(a)
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_A(a):
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n), dtype=np.float64)
    for k in range(num_rows):
        if k < n:
            A[k, 0:k+1] = a[k::-1]
        else:
            A[k, k-n+1:n] = a[n-1:k-n:-1]
    return A


def solve_lp_max_sum(a, M, upper_bound=100.0):
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 10})
    if res.status == 0:
        return res.x
    return None


def solve_lp_min_M(a, upper_bound=100.0):
    n = len(a)
    A_base = build_A(a)
    num_rows = 2 * n - 1
    A_ub = np.hstack([A_base, -np.ones((num_rows, 1))])
    b_ub = np.zeros(num_rows)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, upper_bound)] * n + [(0.0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'time_limit': 10})
    if res.status == 0:
        return res.x[:n], float(res.x[n])
    return None, None


def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = fft(a, size)
    fb = fft(b, size)
    result = np.real(ifft(fa * fb))
    return result[:n]


def gradient_pnorm(a, p, T=None):
    n = len(a)
    conv = fft_convolve(a, a)
    conv = np.maximum(conv, 1e-30)
    if T is not None:
        max_log = float(np.max(np.log(conv)))
        w = np.exp((p - 1) * (np.log(conv) - max_log) - (p - 1) * T)
    else:
        max_val = float(np.max(conv))
        normalized = conv / max_val
        w = normalized ** (p - 1)
    cross = fft_convolve(w[::-1], a)
    grad = 2.0 * cross[:n]
    return grad, float(np.max(conv))


def soft_max_grad(a, T):
    n = len(a)
    conv = fft_convolve(a, a)
    max_c = float(np.max(conv))
    shifted = (conv - max_c) / T
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(np.sum(exp_vals))
    p_vec = exp_vals / sum_exp
    cross = fft_convolve(p_vec[::-1], a)
    grad = (2.0 / T) * cross[:n]
    return grad, max_c


def edge_aware_init(n, rng):
    c = 1.0 / np.sqrt(2.0 * n - 1.0)
    a = np.zeros(n)
    edges = min(4, n)
    for i in range(edges // 2):
        a[i] = c
        a[n-1-i] = c
    if n > edges:
        remaining = 1.0 - edges * c
        if remaining < 0:
            remaining = 0.0
        interior = np.ones(n - edges) * (remaining / max(1, n - edges))
        noise = rng.normal(0, 0.1, n - edges)
        interior = np.maximum(0, interior * np.exp(noise))
        s = np.sum(interior)
        if s > 1e-10:
            interior = interior * remaining / s
        a[edges//2:n - edges//2] = interior
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def structured_init(n, rng):
    choice = int(rng.integers(0, 8))
    if choice == 0:
        return edge_aware_init(n, rng)
    elif choice == 1:
        phase = rng.uniform(0, 2 * np.pi)
        freq = rng.uniform(0.03, 0.5)
        a = np.maximum(0, np.cos(2 * np.pi * freq * np.arange(n) + phase))
        s = np.sum(a)
        return a / s if s > 1e-10 else np.ones(n) / n
    elif choice == 2:
        period = int(rng.integers(max(3, n // 30), max(4, n // 3)))
        pattern = np.maximum(0, rng.random(period) - 0.15)
        a = np.tile(pattern, n // period + 1)[:n]
        a += 0.05 * rng.random(n)
        s = np.sum(a)
        return a / s if s > 1e-10 else np.ones(n) / n
    elif choice == 3:
        center = n / 2.0
        width = n * rng.uniform(0.15, 0.5)
        a = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        a += 0.2 * np.ones(n)
        s = np.sum(a)
        return a / s
    elif choice == 4:
        a = np.ones(n)
        a[0] = 2.0
        a[-1] = 2.0
        if n > 2:
            a[1] = 0.5
            a[-2] = 0.5
        s = np.sum(a)
        return a / s
    elif choice == 5:
        a = np.maximum(0, rng.normal(1.0, 0.5, n))
        s = np.sum(a)
        return a / s if s > 1e-10 else np.ones(n) / n
    elif choice == 6:
        t = np.linspace(0, 1, n)
        alpha = rng.uniform(0.5, 3.0)
        a = t**alpha + (1-t)**alpha + 0.1
        s = np.sum(a)
        return a / s
    else:
        a = np.ones(n)
        a[len(a)//4:3*len(a)//4] *= rng.uniform(0.5, 0.8)
        s = np.sum(a)
        return a / s


def lp_optimize(a, max_iters, max_time, t_values=None):
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    a = normalize(a)
    best_a = a.copy()
    best_obj = compute_obj(a)
    start = time.monotonic()
    stall = 0

    if t_values is None:
        t_values = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02]

    for it in range(max_iters):
        if time.monotonic() - start > max_time:
            break

        conv = fft_convolve(a, a)[:2*n-1]
        M = float(np.max(conv))

        b = solve_lp_max_sum(a, M)
        if b is None:
            break

        b_sum = float(np.sum(b))
        if b_sum <= 1.0 + 1e-6:
            stall += 1
            if stall > 12:
                break
            continue
        stall = 0

        b = b / b_sum
        improved = False
        for t in t_values:
            a_new = (1.0 - t) * a + t * b
            a_new = np.clip(a_new, 0.0, None)
            a_new = normalize(a_new)
            obj = compute_obj(a_new)
            if obj < best_obj - 1e-12:
                best_a = a_new.copy()
                best_obj = obj
                a = a_new
                improved = True
                break

        if not improved:
            a_new = (1.0 - 0.05) * a + 0.05 * b
            a_new = np.clip(a_new, 0.0, None)
            a_new = normalize(a_new)
            obj = compute_obj(a_new)
            if obj < best_obj + 1e-8:
                a = a_new

    return best_a, best_obj


def gd_optimize(a, max_time, T_init=None, p_init=10.0):
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.clip(a, 0.0, None)
    a = normalize(a)
    best_a = a.copy()

    start = time.monotonic()

    if T_init is None:
        conv = fft_convolve(a, a)
        T_init = 0.05 * float(np.max(conv))

    T = T_init
    p = p_init
    lr = 0.5
    last_improve = 0
    patience = 300

    for it in range(50000):
        if time.monotonic() - start > max_time:
            break

        grad, max_c = soft_max_grad(a, T)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        grad = grad / (grad_norm + 1e-15)
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0.0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s
        obj = compute_obj(a_new)

        if obj < best_obj - 1e-12:
            best_a = a_new.copy()
            best_obj = obj
            a = a_new
            last_improve = it
            T = max(0.0001, T * 0.99)
            p = min(200.0, p * 1.02)
        else:
            lr *= 0.95
            if lr < 1e-12:
                break
            if it - last_improve > patience:
                T = max(0.0001, T * 0.7)
                p = max(5.0, p * 0.8)
                lr = 0.3
                last_improve = it

    return best_a, best_obj

    if best_seq is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        print(f"\nFinal refinement: {remaining:.0f}s")

        # LP polish
        lp_polish_time = min(remaining * 0.4, 60.0)
        a_lp, obj_lp = lp_optimize(best_seq.copy(), max(20, int(lp_polish_time / 0.3)),
                                   lp_polish_time,
                                   t_values=[0.1, 0.05, 0.02, 0.01, 0.005, 0.002])
        if obj_lp < best_score:
            best_score = obj_lp
            best_seq = a_lp.copy()
            print(f"  After LP polish: obj={best_score:.6f}")

        # GD polish
        if time.monotonic() < deadline - 10:
            gd_polish_time = min((deadline - time.monotonic()) * 0.3, 30.0)
            a_gd, obj_gd = gd_optimize(best_seq.copy(), gd_polish_time, T_init=0.003)
            if obj_gd < best_score:
                best_score = obj_gd
                best_seq = a_gd.copy()
                print(f"  After GD polish: obj={best_score:.6f}")

        # Block perturbation
        if time.monotonic() < deadline - 5:
            bp_time = deadline - time.monotonic() - 3.0
            a_bp, obj_bp = block_refine(best_seq, rng, bp_time)
            if obj_bp < best_score:
                best_score = obj_bp
                best_seq = a_bp.copy()
                print(f"  After block refine: obj={best_score:.6f}")

        # Coordinate refinement
        if time.monotonic() < deadline - 3:
            coord_time = deadline - time.monotonic() - 1.0
            a_coord, obj_coord = coordinate_refine(best_seq, rng, coord_time)
            if obj_coord < best_score:
                best_score = obj_coord
                best_seq = a_coord.copy()
                print(f"  After coord refine: obj={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(200)
        best_score = compute_obj(best_seq)

    elapsed = time.monotonic() - start_time
    print(f"\nDONE: n={len(best_seq)}, score={best_score:.6f}, time={elapsed:.1f}s")
    return [float(x) for x in best_seq]
