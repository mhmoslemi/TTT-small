Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. FISTA (accelerated proximal gradient) on log-sum-exp smoothed objective
   with continuation method (increasing smoothing parameter)

import numpy as np
import time


def fast_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    s = 1
    while s < 2 * n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2 * n - 1]


def compute_score(a):
    """Compute J(a) directly."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def smooth_obj_and_grad(a, beta, n):
    """
    Compute the smoothed objective and gradient.
    obj = 2n * LSE(beta * conv(a,a)) / (sum(a))^2  -- actually let's just do:
    obj = 2n * LSE(beta * conv(a_norm, a_norm)) where a_norm = a / sum(a)
    
    For simplicity, work with normalized a directly.
    """
    a_pos = np.maximum(a, 1e-15)
    s = a_pos.sum()
    a_norm = a_pos / s
    b = fast_conv(a_norm)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -100.0, 0.0)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    weights = exp_shifted / total
    log_sum = np.log(total)
    obj = 2.0 * n * (b_max + log_sum / beta)
    
    # Gradient: d obj / da_j = 2n * sum_k weights_k * d(conv)_k/da_j / s^2 * s - ... 
    # Actually since we're working with a_norm = a/s, and we want grad w.r.t. a_norm:
    # Let's just compute grad w.r.t. a_norm directly.
    # obj = 2n * (b_max + (1/beta) * log(sum exp(beta*(b_k - b_max))))
    # where b = conv(a_norm, a_norm)
    # d obj / d a_norm_j = 2n * sum_k weights_k * db_k/da_norm_j
    # db_k/da_norm_j = 2 * a_norm_{k-j} (if 0 <= k-j < n, else 0) [by symmetry of self-conv]
    # Actually: b_k = sum_i a_norm_i * a_norm_{k-i}
    # db_k/da_norm_j = a_norm_{k-j} + a_norm_{k-j} = 2*a_norm_{k-j} (if valid)
    # Wait, more carefully:
    # db_k/da_norm_j = sum_i [delta_{ij} * a_norm_{k-i} + a_norm_i * delta_{k-i,j}]
    # = a_norm_{k-j} (if 0<=k-j<n) + a_norm_{k-j} (if 0<=k-j<n, since i=k-j)
    # = 2 * a_norm_{k-j} (if 0 <= k-j < n)
    
    # So grad_j = 2n * sum_k weights_k * 2 * a_norm_{k-j} (if 0<=k-j<n)
    # = 4n * sum_k weights_k * a_norm_{k-j} (if 0<=k-j<n)
    
    # Let m = k-j, so k = j+m. Valid range: 0 <= j+m <= 2n-2 and 0 <= m <= n-1.
    # grad_j = 4n * sum_{m: 0<=m<n, 0<=j+m<=2n-2} weights_{j+m} * a_norm_m
    
    # This is a cross-correlation: grad_j = 4n * (weights ⋆ a_norm)_j
    # where (w ⋆ a)(j) = sum_m w(j+m) * a(m)
    
    # Compute via FFT:
    # Let v = weights reversed: v(i) = weights(2n-2-i)
    # Then (w ⋆ a)(j) = sum_m a(m) * w(j+m) = sum_m a(m) * v(2n-2-j-m) = (a * v)(2n-2-j)
    # So we compute linear convolution of a_norm and v, then index.
    
    v = weights[::-1]  # reverse
    # Compute conv(a_norm, v) via FFT
    conv_len = 2 * n - 1
    s_fft = 1
    while s_fft < 2 * conv_len - 1:
        s_fft <<= 1
    fa = np.fft.rfft(a_norm, s_fft)
    fv = np.fft.rfft(v, s_fft)
    conv_av = np.fft.irfft(fa * fv, s_fft)[:2 * conv_len - 1]
    # conv_av[k] = sum_m a_norm(m) * v(k-m)
    # We want conv_av[2n-2-j] for each j
    grad = 4.0 * n * conv_av[2*n-2::-1]  # reverse to get index 2n-2-j for j=0..n-1
    
    return obj, grad


def fista_smooth(a_init, n, deadline, beta_start=10, beta_end=10000, n_beta_steps=20, iters_per_beta=30):
    """
    FISTA on the smoothed objective with continuation.
    """
    a = a_init.copy()
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    best = a.copy()
    best_score = compute_score(a)
    
    # Beta schedule: geometric
    betas = np.geomspace(beta_start, beta_end, n_beta_steps)
    
    for beta_idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        
        # FISTA iteration
        x = a.copy()
        z = a.copy()
        t_k = 1.0
        
        for it in range(iters_per_beta):
            if time.monotonic() > deadline:
                    break
            
            obj, grad = smooth_obj_and_grad(z, beta, n)
            
            # Step size: use 1/L where L is an upper bound on Lipschitz constant
            # For the smoothed objective, L ≈ 4n * beta (rough estimate)
            # Use backtracking
            step = 1.0 / (4.0 * n * beta * 2)
            # Actually let's use a simpler approach: fixed step with damping
            step = 1.0 / (2.0 * n * beta)
            
            x_new = z - step * grad
            x_new = np.maximum(x_new, 0.0)
            x_new = x_new / max(x_new.sum(), 1e-10)
            
            # Check if we should accept
            obj_new, _ = smooth_obj_and_grad(x_new, beta, n)
            if obj_new < obj + 1e-10:
                x = x_new
            else:
                # Halve step
                step *= 0.5
                x_new = z - step * grad
                x_new = np.maximum(x_new, 0.0)
                x_new = x_new / max(x_new.sum(), 1e-10)
                x = x_new
            
            # FISTA update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            z = x + ((t_k - 1.0) / t_new) * (x - z)
            z = np.maximum(z, 0.0)
            z = z / max(z.sum(), 1e-10)
            t_k = t_new
        
        a = x
        
        # Check actual score
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best = a.copy()
        
        if (beta_idx + 1) % 5 == 0:
            print(f"    FISTA beta={beta:.0f}, score={best_score:.6f}", flush=True)
    
    return best, best_score


def peak_shave(a, n, num_peaks=5, strength=0.1):

    """

    # This means we want to decrease a_j that contributes most to the peaks
    
    a_new = a.copy()
    for k in top_k:
        # Gradient of b_k w.r.t. a: grad_j = 2*a_{k-j} (if 0<=k-j<n)
        m_start = max(0, k - n + 1)
        m_end = min(n, k + 1)
        if m_start < m_end:
            grad_contrib = 2.0 * a[m_start:m_end][::-1] if k < n else 2.0 * a[k-n+1:k+1][::-1]
            # Hmm, let me think again.
            # b_k = sum_j a_j * a_{k-j} for valid j
            # db_k/da_j = a_{k-j} (from first factor) + a_{k-j} (from second factor) = 2*a_{k-j}
            # Valid j: 0 <= j < n and 0 <= k-j < n => max(0, k-n+1) <= j <= min(n-1, k)
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            if j_start < j_end:

    a_new_sum = a_new.sum()

    """Mirror descent (exponentiated gradient) on the max objective."""
    a = a_init.copy()
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    best = a.copy()
    best_score = compute_score(a)
    
    # Running average
    a_avg = np.zeros(n)
    count = 0
    
    eta = eta_start
    it = 0
    
    while time.monotonic() < deadline:
        b = fast_conv(a)
        b_max = b.max()
        score = 2.0 * n * b_max  # since sum(a) = 1
        
        if score < best_score:
            best_score = score
            best = a.copy()
        
        # Find the "active" lag (where max is achieved)
        k_star = np.argmax(b)
        
        # Subgradient of b_{k_star} w.r.t. a:
        # db_k/da_j = 2 * a_{k-j} (if 0 <= k-j < n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n, k_star + 1)
        if j_start < j_end:
            idx = k_star - np.arange(j_start, j_end)
            grad = np.zeros(n)
            grad[j_start:j_end] = 2.0 * a[idx]
        else:
            grad = np.zeros(n)
        
        # Mirror descent step
        a_new = a * np.exp(-eta * grad)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-10:
            break
        a_new = a_new / a_new_sum
        
        a = a_new
        a_avg += a
        count += 1
        eta *= eta_decay
        it += 1
        
        if it % 1000 == 0:
            avg_score = compute_score(a_avg / count)
            print(f"    MD it={it}, cur={score:.6f}, avg={avg_score:.6f}, best={best_score:.6f}", flush=True)
            if avg_score < best_score:
                best_score = avg_score
                best = a_avg / count
    
    return best, best_score

"""
Minimizes J(a) = 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP best-response with adaptive line search (primary local optimizer)
2. FISTA with log-sum-exp continuation (secondary refiner)  
3. Multi-start perturbation search (global exploration)
4. Starts from parent construction for warm start
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    n = len(a)
    s = 1
    while s < 2*n - 1:
        s <<= 1
    fa = np.fft.rfft(a, s)
    return np.fft.irfft(fa * fa, s)[:2*n - 1]


def compute_score(a):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_conv(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    n = len(a)
    m = 2*n - 1
    A = np.zeros((m, n))
    for k in range(m):
        jl = max(0, k - n + 1)
        jh = min(n - 1, k)
        if jl <= jh:
            A[k, jl:jh+1] = a[k-jh:k-jl+1][::-1]
    return A


def lp_best_response(a, M):
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x.sum() > 1e-10:
        return res.x
    return None


def smooth_obj_grad(a, beta, n):
    a_pos = np.maximum(a, 1e-15)
    s = a_pos.sum()
    if s < 0.01:
        return 1e10, np.zeros(n)
    a_n = a_pos / s
    b = fast_conv(a_n)
    b_max = b.max()
    shifted = beta * (b - b_max)
    shifted = np.clip(shifted, -100.0, 0.0)
    es = np.exp(shifted)
    total = es.sum()
    w = es / total
    obj = 2.0 * n * (b_max + np.log(total) / beta)
    
    # Gradient via FFT cross-correlation
    v = w[::-1]
    conv_len = 2*n - 1
    s_fft = 1
    while s_fft < 2*conv_len - 1:
        s_fft <<= 1
    fa = np.fft.rfft(a_n, s_fft)
    fv = np.fft.rfft(v, s_fft)
    conv_av = np.fft.irfft(fa * fv, s_fft)[:2*conv_len - 1]
    grad = 4.0 * n * conv_av[2*n-2::-1]
    
    return float(obj), grad


def fista_polish(a, n, deadline, beta_start=100, beta_end=20000, n_steps=20, iters=30):
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    best = a.copy()
    best_score = compute_score(a)
    betas = np.geomspace(beta_start, beta_end, n_steps)
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        x = a.copy()
        z = x.copy()
        t_k = 1.0
        step = 1.0 / (8.0 * n * beta)
        
        for it in range(iters):
            if time.monotonic() > deadline:
                break
            obj, grad = smooth_obj_grad(z, beta, n)
            if not np.isfinite(obj):
                break
            x_new = z - step * grad
            x_new = np.maximum(x_new, 0.0)
            x_new = x_new / max(x_new.sum(), 1e-10)
            
            # Backtracking
            obj_new, _ = smooth_obj_grad(x_new, beta, n)
            if obj_new > obj:
                step *= 0.5
                x_new = z - step * grad
                x_new = np.maximum(x_new, 0.0)
                x_new = x_new / max(x_new.sum(), 1e-10)
            
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k*t_k)) / 2.0
            z = x_new + ((t_k - 1.0) / t_new) * (x_new - z)
            z = np.maximum(z, 0.0)
            z = z / max(z.sum(), 1e-10)
            x = x_new
            t_k = t_new
        
        a = x
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best = a.copy()
    
    return best, best_score


def lp_iteration_linesearch(a, max_iters, deadline):
    """LP best-response with line search on step size."""
    n = len(a)
    best_local = a.copy()
    best_score = compute_score(a)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()

        g_sum = g.sum()
        if g_sum < 1e-10:

        g = g / g_sum
        
        # Line search over t
        best_t = 0.0
        best_s = best_score
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:

            a_try = np.maximum(a_try, 0)

            a_try = a_try / s_try
            s_val = compute_score(a_try)
            if s_val < best_s:
                best_s = s_val
                best_t = t
        
        if best_t > 0:
            a = (1-best_t)*a + best_t*g
            a = np.maximum(a, 0)
            a = a / a.sum()
            score = compute_score(a)

        else:

            score = compute_score(a_try)
            if score < best_score:
                best_score = score
                best_local = a_try.copy()
                a = a_try
            else:
                break
    
    return best_local, best_score
