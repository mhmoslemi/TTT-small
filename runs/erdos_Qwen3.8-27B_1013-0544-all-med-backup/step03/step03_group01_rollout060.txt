import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-16 * n:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-16:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-16:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    shifted = alpha * (c - c_max)
    shifted_max = shifted.max()
    exp_vals = np.exp(shifted - shifted_max)
    log_sum = shifted_max + np.log(exp_vals.sum())
    obj = c_max + log_sum / alpha
    weights = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        s = k - (n - 1)
        if s >= 0:
            idx1 = np.arange(n - s)
            grad[idx1] += (1 - h[idx1 + s]) * w
            idx2 = np.arange(s, n)
            grad[idx2] -= h[idx2 - s] * w
        else:
            sp = -s
            idx1 = np.arange(sp, n)
            grad[idx1] += (1 - h[idx1 - sp]) * w
            idx2 = np.arange(n - sp)
            grad[idx2] -= h[idx2 + sp] * w
    grad *= dx
    return obj, grad

def proj_grad_descent(h, n, alpha, deadline, lr_init=0.01, max_iter=5000):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        obj, grad = lse_obj_grad(h, n, alpha)
        
        # Backtracking line search
        t = lr
        improved = False
        for _ in range(20):
            h_new = project_feasible(h - t * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()
            if c_new < obj:  # descent condition (simplified)
                h = h_new
                improved = True
                break
            t *= 0.5
        
        if not improved:
            lr *= 0.8
            if lr < 1e-10:
                break
    
    return best_h, best_c

def coordinate_descent(h, n, deadline, max_sweeps=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            # Try a few candidate values for h[j]
            candidates = [0.0, 1.0, h[j]]
            # Add values from active constraints
            c_raw = np.correlate(h, 1 - h, mode='full')
            top_k = np.argsort(c_raw)[-5:]
            for k in top_k:
                s = k - (n - 1)
                # Breakpoint: where this constraint becomes inactive
                # F_k(t) = a_k * t + b_k
                # a_k = (1-h[j+s])*I1 - h[j-s]*I2 (coefficient of h[j])
                a_k = 0.0
                b_k = c_raw[k]
                if 0 <= j + s < n:
                    a_k += (1 - h[j + s])
                if 0 <= j - s < n:
                    a_k -= h[j - s]
                if abs(a_k) > 1e-14:
                    bp = (b_k - c_raw[np.argmax(c_raw)]) / a_k  # approximate
                    if 0 <= bp <= 1:
                        candidates.append(bp)
            
            best_t = h[j]
            best_val = best_c
            for t in candidates:

                h[j] = t
                c_val = compute_c5(h, n)
                if c_val < best_val - 1e-14:

                    best_t = t
            h[j] = best_t

                h = project_feasible(h, n)
                c_after = compute_c5(h, n)
                if c_after < best_c - 1e-14:
                    best_c = c_after
                    improved = True

    return h, best_c

def pair_local(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                for s in (1.0, -1.0):
                    hi = h[i] + s * d
                    hj = h[j] - s * d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i] = hi
                        h[j] = hj
                        c = compute_c5(h, n)
                        if c < best_c:
                            best_c = c
                        else:
                            h[i] -= s * d
                            h[j] += s * d
    return h, best_c

        for n in n_trials:
            if time.time() > deadline - 30:
                break
            x = np.linspace(0, 2, n)
            h0 = project_feasible(0.5 + 0.3 * np.sin(np.pi * x) + 0.1 * np.sin(2 * np.pi * x), n)
            if best_h is not None:
                old_x = np.linspace(0, 2, best_n)
                hinterp = project_feasible(np.interp(x, old_x, best_h), n)
                if compute_c5(hinterp, n) < compute_c5(h0, n):
                    h0 = hinterp
            dl = min(deadline - 25, time.time() + per_n)
            h, c = optimize_n(h0, n, dl, rng)
            if c < best_c:
                best_c, best_h, best_n = c, h.copy(), n

    # Final refinement
    if best_h is not None and time.time() < deadline - 8:
        h, c = pgd(best_h, best_n, 2000000, deadline - 4, lr=0.0005)
        if c < best_c:
            best_c, best_h = c, h
        if time.time() < deadline - 3:
            h, c = coord_descent(best_h, best_n, deadline - 1)
            if c < best_c:
                best_c, best_h = c, h
        if time.time() < deadline - 1:
            best_h, best_c = pair_local(best_h, best_n, deadline, [0.001, 0.0005])

    if best_h is None:
        best_h = project_feasible(np.full(50, 0.5), 50)
        best_n = 50
        best_c = compute_c5(best_h, best_n)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
