    corr = np.correlate(h, 1.0 - h, mode="full")

    else:

Now let me think about one more potential improvement: **using the fact that the problem has a specific structure**.

The correlation values at lags k and n-1-k+... hmm, I don't see an obvious symmetry.

What about the relationship between the "interior" lags and "boundary" lags? For small |s| (s = k-(n-1)), the correlation involves most of the array. For large |s|, it involves fewer elements. The max is likely to occur at small |s| (interior lags) because there are more terms contributing.

Actually, for s=0 (k=n-1): corr = sum h[i](1-h[i]) = sum h[i] - sum h[i]²
For the maximum of this, we want to maximize sum h[i] - sum h[i]². With sum h[i] = n/2 fixed, this is maximized when h[i] are as spread out as possible (since sum h[i]² is minimized when all equal). So for s=0, the max is achieved by h[i] = 0.5 for all i, giving corr = n/2 - n/4 = n/4.

For s=1: corr = sum_{i=0}^{n-2} h[i](1-h[i+1]) = (n-1)/2 - sum h[i]h[i+1]
Maximized by making h[i]h[i+1] as small as possible, i.e., alternating 0 and 1.

So different lags want different structures! The optimal h must balance all lags simultaneously.

This is the fundamental tension in the problem. The smoothmax approach handles this by focusing on the worst lag(s) while still being sensitive to all of them.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

_smx_params = {'beta': 100, 'top_k': None, 'target': 100.0}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += d / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smx_val_grad(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        idx = np.argsort(-corr)[:top_k]
    else:
        idx = np.arange(len(corr))
    c = corr[idx]
    m = c.max()
    e = np.exp(beta * (c - m))
    se = e.sum()
    w = e / se
    val = (m + np.log(se) / beta) * dx
    grad = np.zeros(n)
    for i in range(len(idx)):
        wk = w[i] * dx
        if wk < 1e-15:
            continue
        s = idx[i] - n + 1
        lo1, hi1 = max(0, -s), min(n, n - s)
        if hi1 > lo1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1+s:hi1+s])
        lo2, hi2 = max(0, s), min(n, n + s)
        if hi2 > lo2:
            grad[lo2:hi2] += wk * (-h[lo2-s:hi2-s])
    return val, grad

def _obj(x):
    return smx_val_grad(x, _smx_params['beta'], _smx_params['top_k'])[0]

def _grad(x):
    return smx_val_grad(x, _smx_params['beta'], _smx_params['top_k'])[1]

def _cons_fun(x):
    return x.sum() - _smx_params['target']

def _cons_jac(x):
    return np.ones(x.size)

def optimize_slsqp(h, beta, maxiter, top_k):
    n = len(h)
    _smx_params['beta'] = beta
    _smx_params['top_k'] = top_k
    _smx_params['target'] = n / 2.0
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=Bounds(np.zeros(n), np.ones(n)),
                       constraints={'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac},
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n / 2.0)
    except Exception:
        return project_feasible(h, n / 2.0)

def subgrad_steps(h, n_steps, lr_start, lr_end):
    n = len(h)
    t = n / 2.0
    dx = 2.0 / n
    lrs = np.geomspace(lr_start, lr_end, n_steps)
    for lr in lrs:
        corr = np.correlate(h, 1.0 - h, mode="full")
        k = np.argmax(corr)
        s = k - (n - 1)
        g = np.zeros(n)
        lo1, hi1 = max(0, -s), min(n, n - s)
        if hi1 > lo1:
            g[lo1:hi1] += (1.0 - h[lo1+s:hi1+s])
        lo2, hi2 = max(0, s), min(n, n + s)
        if hi2 > lo2:
            g[lo2:hi2] += (-h[lo2-s:hi2-s])
        h = h - lr * g * dx
        h = project_feasible(h, t)
    return h

def multi_top_subgrad(h, n_steps, lr_start, lr_end, top_m, beta):
    n = len(h)
    t = n / 2.0
    dx = 2.0 / n
    lrs = np.geomspace(lr_start, lr_end, n_steps)
    for lr in lrs:
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argsort(-corr)[:top_m]
        c = corr[idx]
        m = c.max()
        e = np.exp(beta * (c - m))
        w = e / e.sum()
        g = np.zeros(n)
        for i in range(len(idx)):
            wk = w[i]
            s = idx[i] - (n - 1)
            lo1, hi1 = max(0, -s), min(n, n - s)
            if hi1 > lo1:
                g[lo1:hi1] += wk * (1.0 - h[lo1+s:hi1+s])
            lo2, hi2 = max(0, s), min(n, n + s)
            if hi2 > lo2:
                g[lo2:hi2] += wk * (-h[lo2-s:hi2-s])
        h = h - lr * g * dx
        h = project_feasible(h, t)
    return h
