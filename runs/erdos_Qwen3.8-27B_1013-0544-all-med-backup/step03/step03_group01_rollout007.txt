        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    def con_fun(x):
        return float(np.sum(x) - n / 2.0)
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-17})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pair_local(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    h = h.copy()
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(5):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i + 1, n):
                    for s in (1.0, -1.0):
                        hi, hj = h[i] + s * d, h[j] - s * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i], h[j] = hi - s * d, hj + s * d
            if not improved:
                break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_use = len(h_init)
    except Exception:
        h_init = None
        n_use = 100
    best_h, best_c = None, float('inf')
    # Phase 1: periodic patterns
    p_deadline = t0 + 25
    periods = [p for p in range(3, min(n_use // 2 + 1, 25)) if n_use % p == 0]
    for p in [7, 9, 11, 13, 14, 16, 17, 19, 20, 25]:
        if p not in periods and p < n_use:
            periods.append(p)
    patterns = []
    for p in sorted(periods):
        if time.time() > p_deadline:
            break
        xo, _ = optimize_periodic(p, time.time() + 6)
        if xo is not None:
            reps = n_use // p
            ht = np.concatenate([np.tile(xo, reps), xo[:n_use % p]])
            ht = project_feasible(ht, n_use)
            c = compute_c5(ht, n_use)
            patterns.append((c, ht, p))
            if c < best_c:
                best_c, best_h = c, ht.copy()
    patterns.sort(key=lambda x: x[0])
    # Phase 2: SLSQP from best periodic
    if time.time() < deadline - 120:
        for _, ht, _ in patterns[:4]:
            if time.time() > deadline - 60:
                break
            h_start = ht.copy()
            for alpha in [500, 5000, 50000, 200000, 1000000]:
                rem = deadline - time.time()
                if rem < 25:
                    break
                ht2 = slsqp_full(h_start, n_use, alpha, int(min(3000, rem * 30)))
                c = compute_c5(ht2, n_use)
                if c < best_c - 1e-14:
                    best_c, best_h = c, ht2.copy()
                    h_start = ht2.copy()
    # Phase 3: from h_init
    if h_init is not None and time.time() < deadline - 80:
        h_start = h_init.copy()
        for alpha in [10000, 100000, 500000]:
            rem = deadline - time.time()
            if rem < 20:
                break
            ht2 = slsqp_full(h_start, n_use, alpha, int(min(2000, rem * 30)))
            c = compute_c5(ht2, n_use)
            if c < best_c - 1e-14:
                best_c, best_h = c, ht2.copy()
                h_start = ht2.copy()
    # Phase 4: random restarts
    if time.time() < deadline - 50 and best_h is not None:
        for _ in range(10):
            if time.time() > deadline - 30:
                break
            scale = 0.03 + 0.08 * _ / 10
            hp = project_feasible(best_h + scale * rng.randn(n_use), n_use)
            ht2 = slsqp_full(hp, n_use, 50000, 1500)
            c = compute_c5(ht2, n_use)
            if c < best_c - 1e-14:
                best_c, best_h = c, ht2.copy()
    # Phase 5: local search
    if best_h is not None and time.time() < deadline - 8:
        best_h, best_c = pair_local(best_h, n_use, deadline, [0.004, 0.002, 0.001])
    if best_h is None:
        best_h = project_feasible(h_init if h_init is not None else np.full(n_use, 0.5), n_use)
        best_c = compute_c5(best_h, n_use)
    best_h = project_feasible(best_h, n_use)
    best_c = compute_c5(best_h, n_use)
    return best_h, best_c, n_use
