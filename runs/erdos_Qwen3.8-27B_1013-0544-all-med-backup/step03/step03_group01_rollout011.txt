```python
def compute_subgradient(h, n):
    """Compute subgradient of max_l c[l] w.r.t. h"""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c5 = c.max()
    
    # Active set: lags where c is close to max
    tol = 1e-8
    active = np.where(c >= c5 - tol)[0]
    
    grad = np.zeros(n)
    for m in active:
        lag = m - (n - 1)
        # c[m] = sum_i h[i]*(1-h[i+lag])*dx (valid indices)
        # d/dh[j]:
        #   contribution from h[j] as first factor: (1-h[j+lag])*dx if 0 <= j+lag < n
        #   contribution from h[j] as second factor (i+lag = j, i = j-lag): -h[j-lag]*dx if 0 <= j-lag < n
        for j in range(n):
            g = 0.0
            jp = j + lag
            jm = j - lag
            if 0 <= jp < n:
                g += (1.0 - h[jp]) * dx
            if 0 <= jm < n:
                g -= h[jm] * dx
            grad[j] += g
    
    grad /= len(active)
    return grad, c5, active
```

```python
    dx = 2.0 / n
    c = compute_correlation(h, n)
    best_c = float(c.max())
    lags = np.arange(-(n-1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                ip = i + lags
                im = i - lags
                jp = j + lags
                jm = j - lags
                vi_p = (ip >= 0) & (ip < n)
                vi_m = (im >= 0) & (im < n)
                vj_p = (jp >= 0) & (jp < n)
                vj_m = (jm >= 0) & (jm < n)
                
                d = np.zeros(2*n - 1)
                d[vi_p] += (1 - h[ip[vi_p]]) * dx
                d[vi_m] -= h[im[vi_m]] * dx
                d[vj_p] -= (1 - h[jp[vj_p]]) * dx
                d[vj_m] += h[jm[vj_m]] * dx
                
                delta_lo = max(-h[i], h[j] - 1.0)
                delta_hi = min(1.0 - h[i], h[j])
                if delta_lo >= delta_hi - 1e-15:
                    continue
                
                pos = d > 1e-14
                neg = d < -1e-14
                zeros_c = c[~pos & ~neg]
                
                lo_C = float(zeros_c.max()) if len(zeros_c) > 0 else 0.0
                hi_C = best_c
                
                if lo_C >= hi_C:
                    continue
                
                d_lo_final = delta_lo
                d_hi_final = delta_hi
                
                for _ in range(28):
                    mid_C = (lo_C + hi_C) / 2.0
                    d_lo = delta_lo
                    d_hi = delta_hi
                    if pos.any():
                        ub = ((mid_C - c[pos]) / d[pos]).min()
                        if ub < d_hi:
                            d_hi = ub
                    if neg.any():
                        lb = ((mid_C - c[neg]) / d[neg]).max()
                        if lb > d_lo:
                            d_lo = lb
                    d_lo_final = d_lo
                    d_hi_final = d_hi
                    if d_lo <= d_hi + 1e-14:
                        hi_C = mid_C
                    else:
                        lo_C = mid_C
                
                if hi_C < best_c - 1e-14:
                    delta_opt = np.clip((d_lo_final + d_hi_final) / 2.0, delta_lo, delta_hi)
                    h[i] += delta_opt
                    h[j] -= delta_opt
                    h[i] = np.clip(h[i], 0, 1)
                    h[j] = np.clip(h[j], 0, 1)
                    c = compute_correlation(h, n)
                    best_c = float(c.max())
                    improved = True
        if not improved:
            break
    return h, best_c

def subgrad_optimize(h, n, deadline, target=0.38085):

    best_h = h.copy()

            best_h = h.copy()

        grad = np.zeros(n)
        for m in active:
            lag = m - (n - 1)
            jp = np.arange(n) + lag
            jm = np.arange(n) - lag
            vp = (jp >= 0) & (jp < n)
            vm = (jm >= 0) & (jm < n)
            g = np.zeros(n)
            g[vp] += (1 - h[jp[vp]]) * dx
            g[vm] -= h[jm[vm]] * dx
            grad += g
        grad /= len(active)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Adaptive step size
        gap = max(c5 - target, 1e-6)
        step = 0.5 * gap / gnorm
        step = min(step, 0.05)

        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)

            h = h_new

            # Try smaller step
            h_new = h - step * 0.1 * grad
            h_new = project_feasible(h_new, n)
            h = h_new
    
    return best_h, best_c
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_correlation(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13 * n:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-13:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-13:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def slsqp_step(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = (c - m) * alpha
        mx = shifted.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
        return m + np.log(np.sum(np.exp(shifted))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)

        return h
