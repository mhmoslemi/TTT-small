    """Efficient coordinate descent with incremental correlate updates."""
    n = len(h)
    h = h.copy()

        for i in range(n):
            idx1 = i - lags
            idx2 = i + lags
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)

            if np.any(valid1):
                a[valid1] += (1.0 - h[idx1[valid1]])

                a[valid2] -= h[idx2[valid2]]

                max_s = a[active].max()

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def coordinate_descent(h, target, max_sweeps=50, tol=1e-11):

            a = np.zeros(n_p)

            if np.any(valid2):
                a[valid2] -= h[idx2[valid2]]
            b = c - h[i] * a
            lo_t, hi_t = 0.0, 1.0

    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
