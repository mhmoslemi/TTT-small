import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_round(h, target):
    """Round h to binary with exact sum=target."""
    n = len(h)
    nb = (h >= 0.5).astype(float)
    deficit = target - nb.sum()
    if deficit > 0:
        # Need to set more to 1: pick from h>0.5 closest to 0.5, or from h<0.5 closest to 0.5
        zeros = np.where(nb == 0)[0]
        order = zeros[np.argsort(h[zeros])[::-1]]
        nb[order[:int(np.ceil(deficit))]] = 1.0
    elif deficit < 0:
        ones = np.where(nb == 1)[0]
        order = ones[np.argsort(h[ones])]
        nb[order[:int(np.ceil(-deficit))]] = 0.0
    # Fix exact sum
    while nb.sum() > target:
        ones = np.where(nb == 1)[0]
        idx = ones[np.argmin(h[ones])]
        nb[idx] = 0.0
    while nb.sum() < target:
        zeros = np.where(nb == 0)[0]
        idx = zeros[np.argmax(h[zeros])]
        nb[idx] = 1.0
    return nb

def active_set_gradient(h, epsilon_frac=0.02):
    """Compute gradient of sum of near-maximal correlations."""
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    threshold = max_c - epsilon_frac * max_c
    active = np.where(corr >= threshold)[0]
    
    grad = np.zeros(N)
    for k in active:
        lag = int(k) - (N - 1)
        # dc_k/dh[j] = (1-h[j-lag])*1[0<=j-lag<N] - h[j+lag]*1[0<=j+lag<N]
        if lag >= 0:
            # j-lag valid: lag <= j < N
            if lag < N:
                grad[lag:] += (1.0 - h[:N-lag])
            # j+lag valid: 0 <= j < N-lag
            if lag < N:
                grad[:N-lag] -= h[lag:]
        else:
            l = -lag  # l > 0
            # j-lag = j+l valid: 0 <= j+l < N, so 0 <= j < N-l
            if l < N:
                grad[:N-l] += (1.0 - h[l:])
            # j+lag = j-l valid: 0 <= j-l < N, so l <= j < N
            if l < N:
                grad[l:] -= h[:N-l]
    
    return grad, len(active), max_c

def s_iteration(h, target, max_iter=20000, epsilon_frac=0.02, t0=None, budget_s=None):
    """S-iteration with active set gradient descent."""
    N = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    momentum = np.zeros(N)
    stall = 0
    
    for it in range(max_iter):
        if t0 is not None and budget_s is not None:
            if time.time() - t0 > budget_s - 5:
                break
        grad, n_active, max_c = active_set_gradient(cur, epsilon_frac)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad_dir = -grad / gn
        momentum = 0.85 * momentum + 0.15 * grad_dir
        mn = np.linalg.norm(momentum)
        if mn < 1e-14:
            break
        momentum /= mn
        
        step = 0.05 / (1.0 + 0.005 * it)
        improved = False
        for _ in range(30):
            trial = project_h(cur + step * momentum, target)
            c5_t = compute_c5(trial)
            if c5_t < cur_c5 - 1e-13:
                cur = trial
                cur_c5 = c5_t
                if cur_c5 < best_c5:
                    best = cur.copy()
                    best_c5 = cur_c5
                improved = True
                stall = 0
                break
            step *= 0.5
        if not improved:
            stall += 1
            momentum *= 0.5
            if stall > 200:
                break
    
    return best, best_c5

def binary_simulated_annealing(h_init, target, t0, budget_s, n_iter_max=500000, t_start=1.0, t_end=0.01):
    """Simulated annealing on binary space."""
    N = len(h_init)
    dx = 2.0 / N
    cur = binary_round(h_init, target).copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    
    ones = np.where(cur == 1)[0]
    zeros = np.where(cur == 0)[0]
    
    n_total = n_iter_max
    for it in range(n_total):
        if time.time() - t0 > budget_s - 10:
            break
        # Temperature schedule
        frac = it / max(1, n_total)
        T = t_start * (t_end / t_start) ** frac
        
        # Pick random flip pair
        i = zeros[np.random.randint(len(zeros))]
        j = ones[np.random.randint(len(ones))]
        
        # Apply flip
        cur[i] = 1.0
        cur[j] = 0.0
        new_c5 = compute_c5(cur)
        
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            # Update ones/zeros
            zeros = np.delete(zeros, np.where(zeros == i)[0][0])
            ones = np.append(ones, i)
            ones = np.delete(ones, np.where(ones == j)[0][0])
            zeros = np.append(zeros, j)
            # Shuffle to maintain randomness
            np.random.shuffle(zeros)
            np.random.shuffle(ones)
            if new_c5 < best_c5:
                best = cur.copy()
                best_c5 = new_c5
        else:
            # Revert
            cur[i] = 0.0
            cur[j] = 1.0
            # ones/zeros unchanged
    
    return best, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_round(h, target):
    n = len(h)
    order = np.argsort(h)[::-1]
    nb = np.zeros(n)
    nb[order[:int(target)]] = 1.0
    return nb

def active_set_gradient(h, eps_frac):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    threshold = mc - eps_frac * mc
    active = np.where(corr >= threshold)[0]
    grad = np.zeros(N)
    for k in active:
        lag = int(k) - (N - 1)
        if lag >= 0:
            if lag < N:
                grad[lag:] += (1.0 - h[:N - lag])
                grad[:N - lag] -= h[lag:]
        else:
            l = -lag
            if l < N:
                grad[:N - l] += (1.0 - h[l:])
                grad[l:] -= h[:N - l]
    return grad, len(active), mc

def s_iteration(h, target, max_iter, eps_frac, t0, deadline):
    N = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    mom = np.zeros(N)
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad, na, mc = active_set_gradient(cur, eps_frac)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        gd = -grad / gn
        mom = 0.85 * mom + 0.15 * gd
        mn = np.linalg.norm(mom)
        if mn < 1e-14:
            break
        mom /= mn
        step = 0.05 / (1.0 + 0.003 * it)
        improved = False
        for _ in range(30):
            trial = project_h(cur + step * mom, target)
            c5t = compute_c5(trial)
            if c5t < cur_c5 - 1e-13:
                cur = trial
                cur_c5 = c5t
                if cur_c5 < best_c5:
                    best = cur.copy()
                    best_c5 = cur_c5
                improved = True
                stall = 0
                break
            step *= 0.5
        if not improved:
            stall += 1
            mom *= 0.5
            if stall > 300:
                break
    return best, best_c5

def binary_local_search(hb, target, t0, deadline, max_rounds=50):
    """Hill climbing: try all pair swaps, keep best."""
    N = len(hb)
    cur = hb.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    ones = np.where(cur == 1)[0]
    zeros = np.where(cur == 0)[0]
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        best_gain = 0
        best_i = -1
        best_j = -1
        # Sample a subset of swaps for speed
        n_try = min(len(ones) * len(zeros), 5000)
        idx_o = np.random.choice(len(ones), size=min(len(ones), 100), replace=False)
        idx_z = np.random.choice(len(zeros), size=min(len(zeros), 100), replace=False)
        for io in idx_o:
            if time.time() > deadline:
                break
            for iz in idx_z:
                j = ones[io]
                i = zeros[iz]
                cur[j] = 0.0
                cur[i] = 1.0
                c5t = compute_c5(cur)
                gain = cur_c5 - c5t
                if gain > best_gain:
                    best_gain = gain
                    best_i = i
                    best_j = j
                cur[j] = 1.0
                cur[i] = 0.0
        if best_i >= 0:
            cur[best_j] = 0.0
            cur[best_i] = 1.0
            cur_c5 -= best_gain
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
            if cur_c5 < best_c5:
                best = cur.copy()
                best_c5 = cur_c5
            improved = True
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:

        nr = int(n_base * mult)
        if nr % 2 == 0 and 50 <= nr <= 800:
            resolutions.append(nr)
    
    for n_pts in resolutions:
        if time.time() > deadline:
            break
        target = n_pts / 2.0
        
        # Interpolate parent to this resolution
        x_old = np.linspace(0, 2, len(h_parent))
        x_new = np.linspace(0, 2, n_pts)
        h_start = project_h(np.interp(x_new, x_old, h_parent), target)
        
        # S-iteration with decreasing epsilon
        h_cur = h_start.copy()
        c5_cur = compute_c5(h_cur)
        for eps in [0.1, 0.05, 0.02, 0.01, 0.005]:
            if time.time() > deadline:
                break
            h_s, c5_s = s_iteration(h_cur, target, max_iter=15000,
                                     eps_frac=eps, t0=t0, deadline=deadline)
            if c5_s < c5_cur:
                h_cur = h_s
                c5_cur = c5_s
            if c5_cur < best_c5:
                best_h = h_cur.copy()
                best_c5 = c5_cur
        
        # Binary refinement
        if time.time() > deadline:
            break
        hb = binary_round(h_cur, target)

            best_c5 = c5_fin
    
    return best_h, best_c5, int(len(best_h))

def power_norm_obj(h, p, target):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    val = (np.sum(corr**p))**(1.0/p) * dx
    return val

def power_norm_grad(h, p, target):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # grad of (sum corr^p)^(1/p)
    # = (sum corr^p)^(1/p - 1) * sum_i corr[i]^(p-1) * dc_i/dh[j]
    base = (np.sum(corr**p))**(1.0/p - 1.0)
    weights = corr**(p-1)  # shape (2N-1,)
    grad = np.zeros(N)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-10:
            continue
        lag = k - (N - 1)
        if lag >= 0:
            if lag < N:
                grad[lag:] += w * (1.0 - h[:N-lag])
                grad[:N-lag] -= w * h[lag:]
        else:
            l = -lag
            if l < N:
                grad[:N-l] += w * (1.0 - h[l:])
                grad[l:] -= w * h[:N-l]
    return base * grad

So: `dc_idx/dh[j] = (1-h[j-l]) * 1[0≤j-l<N] - h[j+l] * 1[0≤j+l<N]`

This matches what I had. Good.

Now for the code, when l ≥ 0:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_round(h, target):
    n = len(h)
    order = np.argsort(h)[::-1]
    nb = np.zeros(n)
    nb[order[:int(target)]] = 1.0
    return nb

def active_set_gradient(h, eps_frac):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    threshold = mc - eps_frac * mc
    active = np.where(corr >= threshold)[0]
    grad = np.zeros(N)
    for k in active:
        lag = int(k) - (N - 1)
        if lag >= 0:
            if lag < N:
                grad[lag:] += (1.0 - h[:N - lag])
                grad[:N - lag] -= h[lag:]
        else:
            m = -lag
            if m < N:
                grad[:N - m] += (1.0 - h[m:])
                grad[m:] -= h[:N - m]
    return grad, len(active), mc

def s_iteration(h, target, max_iter, eps_frac, t0, deadline):
    N = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    mom = np.zeros(N)
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad, na, mc = active_set_gradient(cur, eps_frac)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        gd = -grad / gn
        mom = 0.85 * mom + 0.15 * gd
        mn = np.linalg.norm(mom)
        if mn < 1e-14:
            break
        mom /= mn
        step = 0.05 / (1.0 + 0.003 * it)
        improved = False
        for _ in range(30):
            trial = project_h(cur + step * mom, target)
            c5t = compute_c5(trial)
            if c5t < cur_c5 - 1e-13:
                cur = trial
                cur_c5 = c5t
                if cur_c5 < best_c5:
                    best = cur.copy()
                    best_c5 = cur_c5
                improved = True
                stall = 0
                break
            step *= 0.5
        if not improved:
            stall += 1
            mom *= 0.5
            if stall > 300:
                break
    return best, best_c5

def binary_local_search(hb, target, t0, deadline, max_rounds=50, sample_size=80):
    N = len(hb)
    cur = hb.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        n_o = min(len(ones), sample_size)
        n_z = min(len(zeros), sample_size)
        io_sel = np.random.choice(len(ones), size=n_o, replace=False)
        iz_sel = np.random.choice(len(zeros), size=n_z, replace=False)
        best_gain = 0.0
        best_i = -1
        best_j = -1
        for io in io_sel:
            if time.time() > deadline:
                break
            j = ones[io]
            for iz in iz_sel:
                i = zeros[iz]
                cur[j] = 0.0
                cur[i] = 1.0
                c5t = compute_c5(cur)
                gain = cur_c5 - c5t
                if gain > best_gain:
                    best_gain = gain
                    best_i = i
                    best_j = j
                cur[j] = 1.0
                cur[i] = 0.0
        if best_i >= 0:
            cur[best_j] = 0.0
            cur[best_i] = 1.0
            cur_c5 -= best_gain
            if cur_c5 < best_c5:
                best = cur.copy()
                best_c5 = cur_c5
        else:
            break
    return best, best_c5

def smooth_max_obj(x, target, beta):
    N = len(x)
    dx = 2.0 / N
    h = project_h(x, target)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    val = m + np.log(np.sum(np.exp(shifted)) / len(corr)) / beta
    return val * dx

def slsqp_refine(h, target, t0, deadline):
    N = len(h)
    dx = 2.0 / N
    best = h.copy()
    best_c5 = compute_c5(best)
    bounds = [(0.0, 1.0)] * N

    def _con_fun(x):
        return x.sum() - target

    def _con_jac(x):
        return np.ones(N)

    constr = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]

    for beta in [100, 500, 2000]:
        if time.time() > deadline:
            break

        def _obj(x):
            return smooth_max_obj(x, target, beta)

        try:
            res = minimize(_obj, best, method='SLSQP', bounds=bounds,
                           constraints=constr, options={'maxiter': 2000, 'ftol': 1e-15})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5
