def find_optimal_coord(A, B, x_min=0.0, x_max=1.0):
    """Find x in [x_min, x_max] that minimizes max_i (A[i]*x + B[i])"""
    m = len(A)
    if m == 0:
        return x_min
    
    # Sort by slope
    order = np.argsort(A)
    A_s = A[order]
    B_s = B[order]
    
    # Build upper envelope using stack
    # Each entry: (slope, intercept, x_start)
    # x_start is where this line becomes the envelope
    lines = []  # list of (a, b, x_start)
    
    for a, b in zip(A_s, B_s):
        # Skip if this line is never on the upper envelope
        # A new line (a, b) makes the last line obsolete if
        # the intersection of new with last is ≤ x_start of last
        while lines:
            la, lb, lx = lines[-1]
            if a == la:
                # Same slope: keep the one with higher intercept
                if b <= lb:
                    break  # new line is below, skip it
                else:
                    lines.pop()  # remove old line, continue
                continue
            # Intersection of (a,b) and (la,lb): x = (lb - b) / (a - la)
            x_int = (lb - b) / (a - la)
            if x_int <= lx + 1e-15:
                lines.pop()
            else:
                break
        if lines:
            la, lb, lx = lines[-1]
            if a == la:
                if b > lb:
                    lines[-1] = (a, b, lx)
                # else skip
                continue
            x_int = (lb - b) / (a - la)
            lines.append((a, b, max(x_int, -1e18)))
        else:
            lines.append((a, b, -1e18))
    
    # Now find the minimum on [x_min, x_max]
    # The envelope is a sequence of lines, each active on [x_start_k, x_start_{k+1})
    best_x = x_min
    best_val = A_s[0] * x_min + B_s[0]  # will be overwritten
    
    for k, (a, b, x_start) in enumerate(lines):
        x_end = lines[k+1][2] if k+1 < len(lines) else 1e18
        # Active interval: [x_start, x_end)
        x_lo = max(x_start, x_min)
        x_hi = min(x_end, x_max)
        if x_lo >= x_hi:
            continue
        # On this interval, f(x) = a*x + b
        # Minimum of a*x+b on [x_lo, x_hi]:
        if a < 0:
            x_best = x_hi
        elif a > 0:
            x_best = x_lo
        else:
            x_best = x_lo
        val = a * x_best + b
        if val < best_val:
            best_val = val
            best_x = x_best
    
    return best_x

def coord_optimize_1d(A, B):

    # Quick check: evaluate at endpoints
    f0 = np.max(A * 0.0 + B)  # = max(B)
    f1 = np.max(A * 1.0 + B)  # = max(A+B)
    
    # Check if minimum is at endpoint
    # At x=0, the subgradient is {A[i] for i in argmax B[i]}
    # At x=1, the subgradient is {A[i] for i in argmax (A[i]+B[i])}
    
    # Bisection approach
    lo, hi = 0.0, 1.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        vals = A * mid + B
        idx = np.argmax(vals)
        val = vals[idx]
        # Find all active lines (within tolerance)
        active = np.where(np.abs(vals - val) < 1e-10)[0]
        slopes = A[active]
        if slopes.max() <= 0:
            # f is non-increasing at mid, minimum is to the right
            lo = mid
        elif slopes.min() >= 0:
            # f is non-decreasing at mid, minimum is to the left
            hi = mid
        else:
            # 0 is in subdifferential, mid is optimal
            break
    
    # Final evaluation
    candidates = [0.0, 1.0, (lo+hi)/2.0]
    best_x = 0.0
    best_f = np.inf
    for x in candidates:
        f = np.max(A * x + B)
        if f < best_f:
            best_f = f
            best_x = x
    return best_x
