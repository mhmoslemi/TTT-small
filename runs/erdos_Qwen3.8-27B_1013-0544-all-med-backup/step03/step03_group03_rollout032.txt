delta_c[k] = (1-h[j-k]) * [j >= k] - h[j+k] * [j+k < n]

And when flipping from 1 to 0, it's the negative.

This means I CAN compute the delta efficiently! For each flip, I can compute the new c[k] = c[k] + delta_c[k] for all k, and check if max(c) decreases.

But I still need O(n) to update all c values. So it's the same as recomputing. Unless I only check the lags that are currently at the maximum or near it.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        if abs(s - target_sum) < 1e-15:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            idx = np.arange(m)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        if m > 0:
            idx = np.arange(m)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, target_sum, n):
    return h.sum() - target_sum

def con_jac(h, target_sum, n):
    return np.ones(n)

def binary_local_search(h_bin, n, max_iter, time_limit, t0, seed):
    rng = np.random.RandomState(seed)
    c_cur = compute_c5(h_bin)
    stagnation = 0
    for it in range(max_iter):
        if time.time() - t0 > budget_s_global - time_limit:
            break
        if stagnation > 2000:
            break
        corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
        worst_lag = np.argmax(corr)
        lag = worst_lag - (n - 1)
        cands = []
        if lag >= 0:
            for i in range(lag, n):
                if h_bin[i] == 1 and h_bin[i - lag] == 0:
                    cands.append(i)
                if h_bin[i] == 1 and i - lag >= 0 and h_bin[i - lag] == 0:
                    cands.append(i - lag)
        else:
            la = -lag
            for i in range(0, n - la):
                if h_bin[i + la] == 1 and h_bin[i] == 0:
                    cands.append(i + la)
                if h_bin[i] == 1 and h_bin[i + la] == 0:
                    cands.append(i)
        cands = list(set(cands))
        if len(cands) > 40:
            cands = list(rng.choice(cands, 40, replace=False))
        rng.shuffle(cands)
        improved = False
        for j in cands:
            h_bin[j] = 1.0 - h_bin[j]
            c = compute_c5(h_bin)
            if c < c_cur - 1e-13:
                c_cur = c
                improved = True
                stagnation = 0
                break
            h_bin[j] = 1.0 - h_bin[j]
        if not improved:
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                i = int(rng.choice(ones))
                j = int(rng.choice(zeros))
                h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
                c = compute_c5(h_bin)
                if c < c_cur - 1e-13:
                    c_cur = c
                    improved = True
                    stagnation = 0
                else:
                    h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        if not improved:
            stagnation += 1
    return h_bin, c_cur

budget_s_global = 1100.0

def run(seed=42, budget_s=1100, **kwargs):
    global budget_s_global
    budget_s_global = budget_s
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(210, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]

    best_h = project_feasible(h_init.copy(), target_sum)
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP warm-up
    if tl() > 50:
        for beta in [50, 200, 1000]:
            if tl() < 20:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 2: Binary local search (aggressive)
    if tl() > 40:
        h_bin = (best_h >= 0.5).astype(float)
        while h_bin.sum() > n // 2:
            one_idx = np.where(h_bin == 1)[0]
            j = one_idx[np.argmin(best_h[one_idx])]
            h_bin[j] = 0
        while h_bin.sum() < n // 2:
            zero_idx = np.where(h_bin == 0)[0]
            j = zero_idx[np.argmax(best_h[zero_idx])]
            h_bin[j] = 1
        h_bin, c_bin = binary_local_search(h_bin.copy(), n, 15000, tl() - 30, t0, seed)
        if c_bin < best_c5:
            best_h = h_bin.copy()
            best_c5 = c_bin

    # Phase 3: SLSQP un-rounding
    if tl() > 40:
        for beta in [500, 5000, 50000]:
            if tl() < 15:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 12000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 4: Binary search again on refined solution
    if tl() > 30:
        h_bin = (best_h >= 0.5).astype(float)
        while h_bin.sum() > n // 2:
            one_idx = np.where(h_bin == 1)[0]
            j = one_idx[np.argmin(best_h[one_idx])]
            h_bin[j] = 0
        while h_bin.sum() < n // 2:
            zero_idx = np.where(h_bin == 0)[0]
            j = zero_idx[np.argmax(best_h[zero_idx])]
            h_bin[j] = 1
        h_bin, c_bin = binary_local_search(h_bin.copy(), n, 10000, tl() - 15, t0, seed + 1)
        if c_bin < best_c5:
            best_h = h_bin.copy()
            best_c5 = c_bin

    # Phase 5: Subgradient with adaptive top-k
    if tl() > 30:
        h = best_h.copy()
        velocity = np.zeros(n)
        for it in range(40000):
            if tl() < 15:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_top = min(6, 2 + it // 3000)
            top_idx = np.argsort(corr)[-k_top:]
            w = np.exp(12.0 * (corr[top_idx] - corr[top_idx].max()))
            w /= w.sum()
            grad = np.zeros(n)
            for j, k in enumerate(top_idx):
                lag = k - (n - 1)
                grad += w[j] * subgrad_for_lag(h, lag, n)
            lr = 0.004 / (1.0 + 0.0002 * it)
            velocity = 0.85 * velocity + 0.15 * (-lr * grad)
            h_new = project_feasible(h + velocity, target_sum)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                h = h_new
                best_h = h_new.copy()
                best_c5 = c_new
            else:
                velocity *= 0.5

    # Phase 6: Resolution exploration
    if tl() > 30:
        n_cands = sorted(set([n - 10, n + 10, n + 20, n + 40, 2 * n, max(60, n - 30)]))
        for n_try in n_cands:
            if n_try < 50 or n_try > 600 or tl() < 15:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_t = np.interp(x_new, x_old, best_h)
            h_t = project_feasible(h_t, ts_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_t, n_try)}]
            for beta in [300, 3000, 30000]:
                if tl() < 8:
                    break
                try:
                    res = minimize(obj_smooth, h_t, args=(dx_t, beta), method='SLSQP',
                                   bounds=bnds_t, constraints=cons_t,
                                   options={'maxiter': 8000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                except Exception:
                    pass

    # Phase 7: Final high-precision SLSQP
    if tl() > 10:
        n_final = len(best_h)
        dx_f = 2.0 / n_final
        ts_f = n_final / 2.0
        bnds_f = Bounds(np.zeros(n_final), np.ones(n_final))
        cons_f = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_f, n_final)}]
        for beta in [100000, 500000]:
            if tl() < 5:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx_f, beta), method='SLSQP',
                               bounds=bnds_f, constraints=cons_f,
                               options={'maxiter': 20000, 'ftol': 1e-17})
                h_new = project_feasible(res.x, ts_f)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
