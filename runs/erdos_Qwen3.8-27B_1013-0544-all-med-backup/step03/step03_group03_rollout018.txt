import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    exp_s = np.exp(shifted)
    val = m + np.log(exp_s.sum()) / beta
    w = exp_s / exp_s.sum()
    return val, w

def grad_smooth_max(h, w, dx):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        idx = k - np.arange(i_lo, i_hi + 1)
        grad[i_lo:i_hi + 1] += wk * (1.0 - 2.0 * h[idx])
    return grad * dx

def obj_pen(x, beta, target_sum, lam):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    val, _ = smooth_max_val_weights(corr, beta)
    return val * dx + lam * (x.sum() - target_sum) ** 2

def grad_pen(x, beta, target_sum, lam):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    _, w = smooth_max_val_weights(corr, beta)
    g = grad_smooth_max(x, w, dx)
    g += 2.0 * lam * (x.sum() - target_sum)
    return g

def sharpen(h, p):
    """Push values toward 0 and 1 while preserving sum after projection."""
    target = h.sum()
    h_sharpened = np.where(h > 0, h ** p, 0.0)
    return project_feasible(h_sharpened, target)

def lfgsb_optimize(h0, target_sum, betas, lam, max_iter, tol):
    """Run L-BFGS-B with penalty at multiple betas (continuation)."""
    n = len(h0)
    best = h0.copy()
    best_c5 = compute_c5(best)
    bnds = [(0.0, 1.0)] * n
    for beta in betas:
        x0 = best.copy()
        try:
            res = minimize(obj_pen, x0, jac=lambda x: grad_pen(x, beta, target_sum, lam),
                           method='L-BFGS-B', bounds=bnds,
                           options={'maxiter': max_iter, 'ftol': tol, 'gtol': 1e-9})
            h_new = project_feasible(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best = h_new.copy()
                best_c5 = c5
        except Exception:
            pass
    return best, best_c5

def projected_gradient(h, target_sum, beta, lr0, n_iters, momentum):
    """Nesterov-accelerated projected gradient."""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    v = np.zeros(n)
    best = h_cur.copy()
    best_c5 = compute_c5(best)
    for it in range(n_iters):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        _, w = smooth_max_val_weights(corr, beta)
        g = grad_smooth_max(h_cur, w, dx)
        lr = lr0 / (1.0 + 0.002 * it)
        v = momentum * v - lr * g
        h_new = project_feasible(h_cur + v, target_sum)
        c_new = compute_c5(h_new)
        if c_new < best_c5 + 0.0005:
            h_cur = h_new
            if c_new < best_c5:
                best = h_new.copy()
                best_c5 = c_new
        else:
            v *= 0.5
    return best, best_c5

            h_try = best_h + np.random.randn(n_cur) * scale
            h_try = project_feasible(h_try, ts_cur)
            h_opt, c_opt = lfgsb_optimize(h_try, ts_cur, [100, 500], lam=20.0, max_iter=800, tol=1e-13)
            if c_opt < best_c5:
                best_h = h_opt
                best_c5 = c_opt

    # Phase 6: Final high-precision polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        n_fin = len(best_h)
        ts_fin = n_fin / 2.0
        for beta in [2000, 10000]:
            if remaining < 15:
                break
            h_fin, c_fin = lfgsb_optimize(best_h, ts_fin, [beta], lam=50.0, max_iter=5000, tol=1e-16)
            if c_fin < best_c5:
                best_h = h_fin
                best_c5 = c_fin

    # Phase 7: Micro coordinate descent
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        n_cd = len(best_h)
        ts_cd = n_cd / 2.0
        h_cd = best_h.copy()
        deltas = [0.01, 0.005, 0.001, 0.0005]
        for sweep in range(50):
            remaining = budget_s - (time.time() - t0)
            if remaining < 5:
                break
            improved = False
            step = n_cd // 50
            for i in range(sweep * step, min((sweep + 1) * step, n_cd)):
                for d in deltas:
                    for sgn in [1, -1]:
                        h_try = h_cd.copy()
                        h_try[i] += sgn * d
                        h_try = project_feasible(h_try, ts_cd)
                        c = compute_c5(h_try)
                        if c < best_c5 - 1e-10:
                            h_cd = h_try
                            best_h = h_try.copy()
                            best_c5 = c
                            improved = True
                            break
                    if improved:
                        break
            if not improved and sweep > 5:
                break

    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
