def project_h(h, target_sum):

    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:

        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = free.sum()
        if nf == 0:

            nf = free.sum()
            if nf == 0:

        h[free] += diff / nf

def corr_lag_grad(h, lag, n):
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif 0 < lag < n:
        g[:n-lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n-lag]
    elif -n < lag < 0:
        a = -lag
        g[a:] += (1.0 - h[:n-a])
        g[:n-a] -= h[a:]
    return g

def smooth_max_obj_grad(h, dx, beta):
