import numpy as np
from scipy.optimize import minimize, Bounds
from functools import partial

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    """Euclidean projection onto {0<=x<=1, sum(x)=target} via bisection on lambda."""
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[:n - lag] += (1.0 - h[lag:]) * dx
        grad[lag:] += -h[:n - lag] * dx
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n - l] * dx
    return grad

def resample_h(h_old, n_new):
    if len(h_old) == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, len(h_old))
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def subgradient_dir(h, target, dx, momentum, rate=0.75):
    n = len(h)
    center = n - 1
    cv = corr_values(h)
    max_cv = cv.max()
    beta = 40.0
    w = np.exp((cv - max_cv) * beta)
    w /= w.sum()
    grad = np.zeros(n)
    nz = np.where(w > 1e-8)[0]
    for idx in nz:
        lag = int(idx) - center
        grad += w[idx] * grad_lag(h, lag, n, dx)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-15:
        momentum[:] = 0
        return np.zeros(n), 0.0
    momentum = rate * momentum - grad / gnorm
    mnorm = np.linalg.norm(momentum)
    if mnorm < 1e-15:
        momentum = -grad / gnorm
        mnorm = 1.0
    return -momentum / mnorm, mnorm

def qp_direction(h, target, dx, gap=0.02, max_active=40):
    n = len(h)
    center = n - 1
    cv = corr_values(h)
    max_cv = cv.max()
    active = np.where(cv >= max_cv - gap)[0]
    if len(active) > max_active:
        order = np.argsort(-cv[active])[:max_active]
        active = active[order]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
    for idx in active:
        lag = int(idx) - center
        g = grad_lag(h, lag, n, dx)
        constraints.append(cv[idx] + g @ (h_var - h) >= t_var)
    prob = cp.Problem(cp.Maximize(t_var), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iter=3000, abstol=1e-8, reltol=1e-8)
        if h_var.value is not None:
            return h_var.value - h
    except Exception:
        pass
    try:
        prob.solve(solver=cp.SCS, max_iter=3000, eps=1e-7)
        if h_var.value is not None:
            return h_var.value - h
    except Exception:
        pass
    return None
