from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def weighted_corr_grad(x, w, n):
    """Compute weighted sum of correlations and its gradient."""
    dx = 2.0 / n
    corr = compute_corr(x)
    obj = np.dot(w, corr) * dx
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - x[:n-s])
            grad[:n-s] += wk * (-x[s:])
        else:
            t = -s
            grad[:n-t] += wk * (1.0 - x[t:])
            grad[t:] += wk * (-x[:n-t])
    
    grad *= dx
    return obj, grad

def solve_qp_qp(h, k_top, temp, maxiter=200):
    """Solve weighted QP for top-k lags."""
    n = len(h)
    target = n / 2.0
    corr = compute_corr(h)
    
    # Select top-k lags
    idx = np.argsort(corr)[-k_top:]
    w = np.zeros(len(corr))
    # Softmax weights among selected
    vals = corr[idx]
    w_sel = np.exp(vals / temp)
    w_sel /= w_sel.sum()
    w[idx] = w_sel
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        return weighted_corr_grad(x, w, n)[0]
    
    def grad(x):
        return weighted_corr_grad(x, w, n)[1]
    
    # Handle equality constraint via penalty or use constrained optimizer
    # Use SLSQP with linear equality constraint
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda: np.ones(n)}]
    
    res = minimize(obj, h, jac=grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(res.x, target), w

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Water-filling projection
        lo, hi = -2.0, 2.0
        for _ in range(50):
            mid = (lo + hi) / 2
            s = np.clip(h - mid, 0, 1).sum()
            if s > target_sum:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) / 2, 0, 1)
    return h

def weighted_obj_grad(x, w, n):
    dx = 2.0 / n
    corr = compute_corr(x)
    obj = np.dot(w, corr) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - x[:n-s])
            grad[:n-s] += wk * (-x[s:])
        else:
            t = -s
            grad[:n-t] += wk * (1.0 - x[t:])
            grad[t:] += wk * (-x[:n-t])
    grad *= dx
    return obj, grad

def bb_projected_gradient(h, w, n, max_iters, alpha_max=1.0):
    """BB projected gradient for weighted QP."""
    x = h.copy()
    x_prev = x.copy()
    g_prev = weighted_obj_grad(x, w, n)[1]
    best_x = x.copy()
    best_obj = compute_c5(x)
    
    for t in range(max_iters):
        obj, g = weighted_obj_grad(x, w, n)
        
        # BB step size
        d = x - x_prev
        y = g - g_prev
        dy = np.dot(d, y)
        dd = np.dot(d, d)
        if dy > 1e-10 and dd > 1e-10:
            alpha = dd / dy
            alpha = np.clip(alpha, 1e-4, alpha_max)
        else:
            alpha = 0.01

        x_new = project_feasible(x - alpha * g, n / 2.0)

        if c < best_obj:

            best_x = x_new.copy()
        
        x_prev = x.copy()
        g_prev = g.copy()
        x = x_new
    
    return best_x, best_obj

- Flipping 0→1: change = 1*(1-1) - 0*(1-0) = 0. So no change for s=0. ✓ (makes sense, h*(1-h) = 0 for binary)
- Using my formula: case (a) gives (1-h[i]) = 0, case (b) gives -h[i] = 0. Total = 0. ✓

For s≠0, the two cases are disjoint (i+s ≠ i-s unless s=0).

So the total number of corr entries affected by flipping h[i] is at most 2n-2 (all s except 0), but in practice:
- Case (a): s ranges from -i to n-1-i, giving n values
- Case (b): s ranges from i-(n-1) to i, giving n values
- Total: up to 2n-2 unique s values (with overlap at s=0 which contributes 0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        lo, hi = -2.0, 2.0
        for _ in range(50):
            mid = (lo + hi) / 2
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) / 2, 0, 1)
    return h

def weighted_obj_grad(x, w, n):
    dx = 2.0 / n
    corr = compute_corr(x)
    obj = np.dot(w, corr) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - x[:n-s])
            grad[:n-s] += wk * (-x[s:])
        else:
            t = -s
            grad[:n-t] += wk * (1.0 - x[t:])
            grad[t:] += wk * (-x[:n-t])
    grad *= dx
    return obj, grad

def bb_gradient(h, w, n, max_iters=1000):
    x = h.copy()
    x_prev = x.copy()
    _, g_prev = weighted_obj_grad(x, w, n)
    best_x = x.copy()
    best_c = compute_c5(x)
    for t in range(max_iters):
        obj, g = weighted_obj_grad(x, w, n)
        d = x - x_prev
        y = g - g_prev
        dy = np.dot(d, y)
        dd = np.dot(d, d)

        else:

        x_new = project_feasible(x - alpha * g, n // 2)
        c = compute_c5(x_new)
        if c < best_c:
            best_c = c
            best_x = x_new.copy()
        x_prev = x
        x = x_new
        g_prev = g
    return best_x, best_c

        np.random.shuffle(ones_idx)
        np.random.shuffle(zeros_idx)
        for i in ones_idx:
            for j in zeros_idx[:20]:  # try first 20 zeros for each one
                # Simulate swap: b[i] 1->0, b[j] 0->1
                # Effect on corr
                old_corr = corr.copy()
                # h[i]: 1->0
                for s in range(-i, n - i):
                    k = n - 1 + s
                    if 0 <= i + s < n:
                        corr[k] -= (1 - b[i + s])
                    if 0 <= i - s < n:
                        corr[k] += b[i - s]
                # h[j]: 0->1
                for s in range(-j, n - j):
                    k = n - 1 + s
                    if 0 <= j + s < n:
                        corr[k] += (1 - b[j + s])
                    if 0 <= j - s < n:
                        corr[k] -= b[j - s]
                
                new_max = np.max(corr)
                if new_max < cur_max - 0.5:  # need improvement of at least 0.5 (rounding)
                    b[i] = 0
                    b[j] = 1
                    cur_max = new_max
                    improved = True
                    break
                else:
                    corr = old_corr
            if improved:
                break
        if not improved:
            break
    
    return b, cur_max * (2.0 / n)

def flip_effect(b, i, val_change, n):
    """Compute corr delta when b[i] changes by val_change (+1 or -1)."""
    delta = np.zeros(2 * n - 1)
    for s in range(max(-i, -(n-1)), min(n - 1 - i, n - 1) + 1):
        k = n - 1 + s
        if 0 <= i + s < n:
            delta[k] += val_change * (1 - b[i + s])
        if s != 0 and 0 <= i - s < n:
            delta[k] += -val_change * b[i - s]
    return delta

def binary_local_search(b, max_rounds=500, temperature=None):
    n = len(b)
    b = b.copy()
    corr = compute_corr(b).copy()
    cur_max = np.max(corr)
    ones_list = list(np.where(b == 1)[0])
    zeros_list = list(np.where(b == 0)[0])

    for rnd in range(max_rounds):

                    if s != 0 and 0 <= i-s < n:
                        delta[k] += b[i-s]
                for s in range(max(-j, -(n-1)), min(n-1-j, n-1)+1):
                    k = n - 1 + s

                    if s != 0 and 0 <= j-s < n:
                        delta[k] -= b[j-s]
