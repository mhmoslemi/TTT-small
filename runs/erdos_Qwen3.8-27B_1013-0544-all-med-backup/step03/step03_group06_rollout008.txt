import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(80):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-12
        else:
            free = h > 1e-12
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def weighted_subgrad(h, lags, weights):
    n = len(h)
    grad = np.zeros(n)
    for lag, w in zip(lags, weights):
        if w < 1e-12:
            continue
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            grad[:n - m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:n - m]
    return grad


def get_active(corr, threshold_frac, beta):
    n = len(corr)
    max_c = np.max(corr)
    if max_c <= 0:
        return np.array([n // 2], dtype=int), np.array([1.0])
    w = np.exp(beta * (corr - max_c))
    mask = corr > threshold_frac * max_c
    if not mask.any():
        mask = np.zeros(n, dtype=bool)
        mask[int(np.argmax(corr))] = True
    idx = np.where(mask)[0]
    w_active = w[mask]
    lags = (idx - (n - 1)).astype(int)
    return lags, w_active


def backtrack(h, grad, target, best_c5, max_step):
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-14:
        return h, best_c5
    grad = grad / gnorm
    step = max_step
    for _ in range(14):
        h_trial = project_h(h - step * grad, target)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5 - 1e-16:
            return h_trial, c5_trial
        step *= 0.65
    return h, best_c5

    c = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(c)
    xs = np.clip(beta * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(xs)) / len(c)) / beta) * dx

    return x.sum() - target


def eq_jac(x):
    return np.ones(x.shape)
