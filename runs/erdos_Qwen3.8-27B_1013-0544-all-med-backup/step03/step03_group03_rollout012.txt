import cvxpy as cp
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_lag(h, lag, n):
    """Subgradient of the overlap at a specific lag."""
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            idx = np.arange(m)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        if m > 0:
            idx = np.arange(m)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad

def top_k_subgrad(h, corr, n, k_top, temperature=10.0):
    """Weighted subgradient from top-k lags."""
    top_idx = np.argsort(corr)[-k_top:]
    w = np.exp(temperature * (corr[top_idx] - corr[top_idx].max()))
    w /= w.sum()
    grad = np.zeros(n)
    for j, k in enumerate(top_idx):
        lag = k - (n - 1)
        grad += w[j] * subgrad_for_lag(h, lag, n)
    return grad

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

        return budget_s - (time.time() - t0)

            if h_new is not None:
                c_new = compute_c5(h_new)
                if c_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c_new
    
    # Phase 2: Subgradient with adaptive parameters
    if tl() > 150:

        velocity = np.zeros(n)
        beta_m = 0.85

            if tl() < 130:

            c_curr = np.max(corr) * dx

            temperature = 5.0 + 3.0 * (1.0 + np.sin(0.01 * it))
            grad = top_k_subgrad(h, corr, n, k_top, temperature)

            c_new = compute_c5(h_new)
            if c_new < best_c5 + 0.003:

                if c_new < best_c5:

                    best_c5 = c_new

Actually, I realize there's a subtlety. The LP/QP solution minimizes the LINEARIZED objective, which might not correspond to improving the TRUE objective. The linearization underestimates the true bilinear terms (or overestimates, depending on the situation). 

For f(a,b) = a(1-b), the linearization at (a0,b0) is:
f_lin(a,b) = a(1-b0) - a0*b + a0*b0

The true value f(a,b) = a(1-b).
Difference: f - f_lin = a(1-b) - a(1-b0) + a0*b - a0*b0 = -ab + ab0 + a0b - a0b0 = -(a-a0)(b-b0)

So f - f_lin = -(a-a0)(b-b0).

If (a-a0)(b-b0) > 0 (both moved in the same direction), then f < f_lin (linearization overestimates).
If (a-a0)(b-b0) < 0 (moved in opposite directions), then f > f_lin (linearization underestimates).

In our problem, we're trying to minimize the overlap. If the linearization overestimates (which happens when h[i] and h[i+k] both increase or both decrease from h0), then the true overlap is less than the LP predicts, which is good. If it underestimates, the true overlap is more, which is bad.

So the LP is a "lower bound" problem in some sense - it gives a solution that might not be feasible for the true (non-convex) problem. We need to verify the true C₅ after each LP solve.
