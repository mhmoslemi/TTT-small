def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_grad(x, beta, penalty, target):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    sum_dev = x.sum() - target
    val += penalty * sum_dev * sum_dev
    
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        s = k - n + 1
        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        if j_hi > j_lo:
            grad_h[j_lo:j_hi] += wk * (1.0 - x[j_lo + s:j_hi + s])
        j_lo2 = max(0, s)
        j_hi2 = min(n, n + s)
        if j_hi2 > j_lo2:
            grad_h[j_lo2:j_hi2] += wk * (-x[j_lo2 - s:j_hi2 - s])
    grad_h += 2.0 * penalty * sum_dev
    return val, grad_h

def sa_step(h, corr, i0, delta, n):
    """Update corr after changing h[i0] by delta."""
    s = np.arange(-(n-1), n)  # all lags
    # First factor contribution: delta * (1-h[i0+s]) for valid i0+s
    valid1 = (i0 + s >= 0) & (i0 + s < n)
    h_first = 1.0 - h[numpy.where(valid1, i0 + s, 0)]  # safe index
    contrib1 = numpy.where(valid1, delta * h_first, 0.0)
    # Second factor contribution: -h[i0-s] * delta for valid i0-s
    valid2 = (i0 - s >= 0) & (i0 - s < n)
    h_second = h[numpy.where(valid2, i0 - s, 0)]
    contrib2 = numpy.where(valid2, -delta * h_second, 0.0)
    corr += contrib1 + contrib2
    return corr

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def obj_grad(x, beta, penalty, target):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    sum_dev = x.sum() - target
    val += penalty * sum_dev * sum_dev
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        s = k - n + 1
        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        if j_hi > j_lo:
            grad_h[j_lo:j_hi] += wk * (1.0 - x[j_lo + s:j_hi + s])
        j_lo2 = max(0, s)
        j_hi2 = min(n, n + s)
        if j_hi2 > j_lo2:
            grad_h[j_lo2:j_hi2] += wk * (-x[j_lo2 - s:j_hi2 - s])
    grad_h += 2.0 * penalty * sum_dev
    return val, grad_h


def lbfgs_solve(h0, beta, penalty, target, maxiter=500, gtol=1e-12):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_grad, h0, args=(beta, penalty, target),
                       jac=True, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': gtol, 'maxls': 50})
        h_out = res.x
    except Exception:
        h_out = h0.copy()
    return project_feasible(h_out, target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def continuation_opt(h, n, target, betas, penalty, maxiter, t0, budget):
    h_cur = h.copy()
    for beta in betas:
        if budget - (time.time() - t0) < 50:
            break
        h_cur = lbfgs_solve(h_cur, beta, penalty, target, maxiter=maxiter)
    return h_cur


def basin_hop(h_best, best_c5, n, target, n_hops, t0, budget, seed,
              noise_start, noise_rate, betas, maxiter, penalty):
    rng = np.random.RandomState(seed)
    for hop in range(n_hops):
        if budget - (time.time() - t0) < 50:
            break
        noise = noise_start * (noise_rate ** hop)
        h_pert = h_best + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        for beta in betas:
            if budget - (time.time() - t0) < 35:
                break
            h_pert = lbfgs_solve(h_pert, beta, penalty, target, maxiter=maxiter)
        c = compute_c5(h_pert)
        if c < best_c5:
            best_c5 = c
            h_best = h_pert.copy()
    return h_best, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    budget = budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_cur = np.array(initial_h_values, dtype=float).copy()
    else:
        h_cur = np.full(200, 0.5)

    best_c5 = compute_c5(h_cur)
    h_best = h_cur.copy()

    # Phase 1: Continuation at original resolution
    n1 = len(h_cur)
    t1 = n1 / 2.0
    h_cur = continuation_opt(h_cur, n1, t1,
                             [10, 25, 50, 100, 200, 400], 500.0, 400, t0, budget)
    c = compute_c5(h_cur)
    if c < best_c5:
        best_c5 = c
        h_best = h_cur.copy()

    # Phase 2: Basin hopping at original resolution
    h_best, best_c5 = basin_hop(h_best, best_c5, n1, t1, 8, t0, budget,
                                seed + 1, 0.02, 0.8, [100, 300, 500], 300, 500.0)

    # Phase 3: Interpolate to n=400 and optimize
    n2 = 400
    t2 = n2 / 2.0
    h400 = interp_to_n(h_best, n2)
    h400 = continuation_opt(h400, n2, t2,
                            [10, 25, 50, 100, 200, 400, 600], 500.0, 350, t0, budget)
    c = compute_c5(h400)
    if c < best_c5:
        best_c5 = c
        h_best = h400.copy()

    # Phase 4: Basin hopping at n=400
    n_use = len(h_best)
    t_use = n_use / 2.0
    h_best, best_c5 = basin_hop(h_best, best_c5, n_use, t_use, 10, t0, budget,
                                seed + 10, 0.015, 0.82, [200, 500, 800], 350, 800.0)

    # Phase 5: Try a sinusoidal starting point at n=400
    if budget - (time.time() - t0) > 120:
        n3 = 400
        t3 = n3 / 2.0
        x = np.linspace(0, 2, n3, endpoint=False) + 1.0 / n3
        h_sin = 0.5 + 0.35 * np.cos(2.0 * np.pi * x)
        h_sin = project_feasible(h_sin, t3)
        h_sin = continuation_opt(h_sin, n3, t3,
                                 [20, 50, 100, 200, 400], 500.0, 300, t0, budget)
        c = compute_c5(h_sin)
        if c < best_c5:
            best_c5 = c
            h_best = h_sin.copy()
        h_best, best_c5 = basin_hop(h_best, best_c5, n_use, t_use, 5, t0, budget,
                                    seed + 50, 0.012, 0.82, [300, 600], 300, 800.0)

    # Phase 6: Fine basin hopping near the best
    h_best, best_c5 = basin_hop(h_best, best_c5, n_use, t_use, 8, t0, budget,
                                seed + 100, 0.006, 0.85, [500, 800, 1000], 300, 1000.0)

    # Phase 7: Final high-beta polish
    if budget - (time.time() - t0) > 30:
        h_fin = lbfgs_solve(h_best, 1000, 2000.0, t_use, maxiter=600, gtol=1e-14)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_c5 = c_fin
            h_best = h_fin

    # Phase 8: One more very fine hop
    if budget - (time.time() - t0) > 25:
        h_best, best_c5 = basin_hop(h_best, best_c5, n_use, t_use, 4, t0, budget,
                                    seed + 200, 0.003, 0.85, [800, 1200], 250, 2000.0)

    return h_best, best_c5, len(h_best)
