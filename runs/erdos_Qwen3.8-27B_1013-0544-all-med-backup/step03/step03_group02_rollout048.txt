from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted)) / len(corr)) / beta

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n - lag]
    else:
        la = -lag
        g[la:] += (1.0 - h[:n - la])
        g[:n - la] -= h[la:]
    return g

def grad_sub(h, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        grad += grad_lag(h, lag)
    return grad, corr

    c = np.correlate(h, 1.0 - h, mode="full")

    return x.sum() - target
