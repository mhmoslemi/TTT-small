def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-16:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag]) * dx

    else:

        grad[:n - l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n - l] * dx
    return grad

                cur_c5 = c5_trial
                improved = True
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
                break
            step *= 0.5

        if not improved:
            patience += 1
            if patience >= max_patience:
                patience = 0
                mu = 0.0
                velocity = np.zeros(n)
                T = 0.003 * max(0.1, 1.0 - it / max_iters)
                noise = np.random.normal(0, T, n)
                h_trial = project_h(cur + noise, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 + T * 2:
                    cur = h_trial
                    cur_c5 = c5_trial
                    mu = 0.4
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
    return cur, cur_c5, n


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    else:
        n_parent = 250
        h_parent = np.full(n_parent, 0.5)

    best_state = [compute_c5(h_parent), h_parent.copy(), n_parent]

    n_list = sorted(set([n_parent, 200, 300, 350, 400]))
    n_list = [n for n in n_list if 60 <= n <= 500]

    for n in n_list:
        if time_left() < 80:
            break
        target = n / 2.0
        dx = 2.0 / n
        bounds = Bounds(np.zeros(n), np.ones(n))

        # Initializations
        inits = [project_h(resample(h_parent, n), target)]
        inits.append(project_h(resample(best_state[1], n), target))
        x = np.linspace(0, 2, n)
        # Sigmoid-like shape
        h_sig = 1.0 / (1.0 + np.exp(-4.0 * (x - 1.0)))
        inits.append(project_h(h_sig, target))
        # Double sigmoid (S-shape with plateau)
        h_ds = 0.5 + 0.45 * np.tanh(2.5 * (x - 1.0))
        inits.append(project_h(h_ds, target))
        # Asymmetric bump
        h_asym = 0.5 + 0.3 * np.exp(-((x - 0.6) / 0.3) ** 2) - 0.1 * np.exp(-((x - 1.5) / 0.4) ** 2)
        inits.append(project_h(h_asym, target))

        for h_init in inits:
            if time_left() < 40:
                break
            cur = project_h(h_init.copy(), target)
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_state[0]:
                best_state[0] = cur_c5
                best_state[1] = cur.copy()
                best_state[2] = n

            # SLSQP with smooth max - increasing beta
            for beta in [80, 200, 500, 1500]:
                if time_left() < 20:
                    break
                try:
                    con_fun = lambda x: np.sum(x) - target
                    con_jac = lambda x: np.ones(n)
                    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
                    obj = partial(obj_smooth_max, dx=dx, beta=beta)
                    res = minimize(obj, cur, method='SLSQP', bounds=bounds,
                                   constraints=cons, options={'maxiter': 300, 'ftol': 1e-14})
                    h_new = project_h(res.x, target)
                    c5_new = compute_c5(h_new)
                    if c5_new < cur_c5:
                        cur = h_new
                        cur_c5 = c5_new
                    if cur_c5 < best_state[0]:
                        best_state[0] = cur_c5
                        best_state[1] = cur.copy()
                        best_state[2] = n
                except Exception:
                    pass

            # Momentum descent
            cur, cur_c5, _ = descent_momentum(
                cur, n, target, dx,
                max_iters=4000, time_fn=time_left, min_time=20,
                best_state=best_state
            )

            # After descent, do one more SLSQP round
            if time_left() > 15:
                try:
                    con_fun = lambda x: np.sum(x) - target
                    con_jac = lambda x: np.ones(n)
                    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
                    obj = partial(obj_smooth_max, dx=dx, beta=2000)
                    h_cur = project_h(cur, target)
                    res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                                   constraints=cons, options={'maxiter': 500, 'ftol': 1e-15})
                    h_new = project_h(res.x, target)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_state[0]:
                        best_state[0] = c5_new
                        best_state[1] = h_new.copy()
                        best_state[2] = n
                except Exception:
                    pass

    # Final ultra-fine polish
    if time_left() > 8:
        best_h = best_state[1].copy()
        best_c5 = best_state[0]
        best_n = best_state[2]
        target_f = best_n / 2.0
        dx_f = 2.0 / best_n
        vel = np.zeros(best_n)
        mu_f = 0.3

        for it in range(30000):
            if time_left() < 3:
                break
            corr_raw = np.correlate(best_h, 1.0 - best_h, mode="full")
            corr_val = corr_raw * dx_f
            top_idx = np.argsort(corr_val)[::-1][:4]
            vals = corr_val[top_idx]
            spread = vals[0] - vals[-1] + 1e-12
            temp = 100.0 / (1.0 + spread * 300.0)
            temp = max(50.0, min(800.0, temp))
            weights = np.exp((vals - vals[0]) * temp)
            weights /= weights.sum()

            grad = np.zeros(best_n)
            center = best_n - 1
            for t_idx, w in zip(top_idx, weights):
                lag = int(t_idx) - center
                grad += w * grad_lag(best_h, lag, best_n, dx_f)

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-17:
                break
            grad /= gnorm
            vel = mu_f * vel + grad
            vnorm = np.linalg.norm(vel)
            if vnorm < 1e-17:
                break
            direction = -vel / vnorm

            step = 0.004 / (1.0 + 0.001 * it)
            for _ in range(20):
                h_trial = project_h(best_h + step * direction, target_f)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-16:
                    best_h = h_trial
                    best_c5 = c5_trial
                    break
                step *= 0.5

    return best_h, best_c5, len(best_h)
