```python
def corr_delta_swap(i, j, h, n):
    """Compute change in full corr array when h[i]: 1->0 and h[j]: 0->1."""
    s = np.arange(-(n-1), n)  # all lags
    # h[m] = 0 for m out of [0, n-1]
    def h_ext(m):
        """h at position m, 0 if out of bounds."""
        # m is array
        result = np.zeros_like(m, dtype=float)
        valid = (m >= 0) & (m < n)
        result[valid] = h[m[valid]]
        return result
    
    A = 1.0 - h_ext(i + s)  # was contribution from h[i] as first factor
    B = h_ext(i - s)        # was contribution from h[i] as second factor  
    C = 1.0 - h_ext(j + s)
    D = h_ext(j - s)
    
    return -A + B + C - D
```

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)

        else:

            grad[:n-t] += wk * (1.0 - h[t:])
