from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_topk(h, top_k):
    """Compute subgradient of max_k corr[k] using top-k active lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for li in top_idx:
        d = int(li) - (n - 1)
        # First term: (1-h[j-d]) for valid j where 0<=j-d<n
        if d >= 0:
            j_start, j_end = d, n
        else:
            j_start, j_end = 0, n + d
        if j_start < j_end:
            grad[j_start:j_end] += (1.0 - h[j_start - d:j_end - d])
        # Second term: -h[j+d] for valid j where 0<=j+d<n
        if d >= 0:
            j_start2, j_end2 = 0, n - d
        else:
            j_start2, j_end2 = -d, n
        if j_start2 < j_end2:
            grad[j_start2:j_end2] -= h[j_start2 + d:j_end2 + d]
    return grad / top_k

def nesterov_optimize(h0, target, max_iter, time_budget, t_start):
    """Nesterov momentum projected gradient descent."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Adaptive parameters
    step = 0.1
    momentum = 0.9
    top_k = 5
    
    for it in range(max_iter):
        if time.time() - t_start > time_budget:
            break
        if it % 500 == 0 and it > 0:
            top_k = min(top_k + 2, 30)
        
        # Nesterov lookahead
        h_look = h - v * momentum  # wait, Nesterov is: y = x + momentum * v_prev
        # Actually standard Nesterov: 
        # v_new = momentum * v_old + grad(y)
        # y = h + momentum * v_old (lookahead)
        # h_new = h - step * v_new
        
        y = h + momentum * v
        y = np.clip(y, 0, 1)  # clip for evaluation
        grad = subgrad_topk(y, top_k)
        v = momentum * v + grad
        h_new = h - step * v
        h_new = project_h(h_new, target)
        
        # Backtracking line search
        c5_new = compute_c5(h_new)
        if c5_new > best_c5:

            for alpha in [0.5, 0.25, 0.1, 0.05]:

                h_trial = project_h(h_trial, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5:
                    h_new = h_trial
                    c5_new = c5_trial
                    break
            else:
                step *= 0.5
                continue
        
        h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    
    return best_h, best_c5

                    break
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(280, 0.5)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    n0 = len(h0)

    # Phase 1: Subgradient at parent resolution
    t1 = t0 + budget_s * 0.22
    h1, c1 = do_subgrad(best_h, n0 / 2.0, t1, t0)
    if c1 < best_c5:
        best_h, best_c5 = h1, c1

    # Phase 2: Higher resolution (2x)
    remaining = budget_s - (time.time() - t0)
    if remaining > 180:
        n2 = 2 * n0
        x_old = np.linspace(0, 2, n0)
        x_new = np.linspace(0, 2, n2)
        h_hi = project_h(np.interp(x_new, x_old, best_h), n2 / 2.0)
        t2 = t0 + t1 - t0 + remaining * 0.42
        h2, c2 = do_subgrad(h_hi, n2 / 2.0, t2, t0)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: SLSQP refinement at best resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 70:
        nc = len(best_h)
        tgt = nc / 2.0
        dxc = 2.0 / nc
        bnds = Bounds(np.zeros(nc), np.ones(nc))
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt,
                 'jac': lambda x: np.ones(nc)}]
        for beta in [1500, 4000]:
            if time.time() > budget_s - 40 - t0 + t0:
                break
            if time.time() - t0 > budget_s - 40:
                break
            def obj_sls(x, _b=beta):
                corr = np.correlate(x, 1.0 - x, mode="full")
                m = np.max(corr)
                xm = np.clip(_b * (corr - m), -500, 500)
                return (m + np.log(np.sum(np.exp(xm)) / (_b * len(corr)))) * dxc
            try:
                res = minimize(obj_sls, best_h, method='SLSQP', bounds=bnds,
                              constraints=cons, options={'maxiter': 4000, 'ftol': 1e-16})
                hs = project_h(res.x, tgt)
                cs = compute_c5(hs)
                if cs < best_c5:
                    best_h, best_c5 = hs, cs
            except Exception:
                pass

    # Phase 4: Ultra-fine subgradient polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 25:
        nf = len(best_h)
        tg = nf / 2.0
        hf = best_h.copy()
        cf = best_c5
        vf = np.zeros(nf)
        step = 0.001
        mom = 0.94
        for it in range(200000):
            if time.time() - t0 > budget_s - 3:
                break
            g = subgrad_topk(hf, 10)
            vf = mom * vf + (1.0 - mom) * g
            h_try = project_h(hf - step * vf, tg)
            c_try = compute_c5(h_try)
            if c_try < cf - 1e-15:
                hf = h_try
                cf = c_try
                step = min(step * 1.001, 0.02)
                if cf < best_c5:
                    best_h = hf.copy()
                    best_c5 = cf
            else:
                step *= 0.9
                vf *= 0.5
                if step < 1e-10:
                    break

    return best_h, best_c5, int(len(best_h))
