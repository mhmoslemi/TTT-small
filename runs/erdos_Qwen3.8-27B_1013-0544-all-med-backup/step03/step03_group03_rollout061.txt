def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-15:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_topk(corr, k, beta):
    # Smooth max using only top k elements
    top_k = np.argsort(corr)[-k:]
    vals = corr[top_k]
    m = np.max(vals)
    shifted = beta * (vals - m)
    return m + np.log(np.sum(np.exp(shifted))) / beta

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-15:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def subgrad_lag(h, lag):
    """Subgradient of overlap at given lag (positive or negative)."""
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        return 1.0 - 2.0 * h
    elif lag > 0:
        m = lag
        g[:n-m] += (1.0 - h[m:])
        g[m:] -= h[:n-m]
    else:
        p = -lag
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

def weighted_subgrad(h, corr, top_k, beta):
    """Weighted subgradient using top-k correlations."""
    n = len(h)
    top_idx = np.argsort(corr)[-top_k:]
    w = np.exp(beta * (corr[top_idx] - corr[top_idx].max()))
    w /= w.sum()
    grad = np.zeros(n)
    for j, k in enumerate(top_idx):
        lag = k - (n - 1)
        grad += w[j] * subgrad_lag(h, lag)
    return grad

def compute_c5(h):

    dx = 2.0 / n

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-15:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def subgrad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        return 1.0 - 2.0 * h
    elif lag > 0:
        m = lag
        g[:n-m] += (1.0 - h[m:])
        g[m:] -= h[:n-m]
    else:
        p = -lag
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

def weighted_subgrad(h, corr, top_k, beta):
    n = len(h)
    top_idx = np.argsort(corr)[-top_k:]
    w = np.exp(beta * (corr[top_idx] - corr[top_idx].max()))
    w /= w.sum()
    grad = np.zeros(n)
    for j, k in enumerate(top_idx):
        lag = k - (n - 1)
        grad += w[j] * subgrad_lag(h, lag)
    return grad
