4. Update h[i] = x*
5. Adjust for the sum constraint (if needed)

The 1D problem: minimize max_k (b_k + a_k * x) over x ∈ [0,1].
This is minimized at x where the maximum of the lines is as small as possible. The optimal x is:
- 0 if all a_k ≥ 0 (or the max at x=0 is already ≤ the max at any x > 0)
- 1 if all a_k ≤ 0
- Otherwise, at the intersection point of the two "tightest" lines.

More precisely, if we sort the lines by their slope a_k, the upper envelope is a convex piecewise linear function. The minimum is at the "valley" of this envelope, which occurs at a breakpoint (intersection of two adjacent lines in the upper envelope).

For a simpler approach: binary search on the optimal value t. For a given t, check if there exists x ∈ [0,1] such that b_k + a_k * x ≤ t for all k. This is a system of linear inequalities in x:
- For a_k > 0: x ≤ (t - b_k) / a_k
- For a_k < 0: x ≥ (t - b_k) / a_k
- For a_k = 0: need b_k ≤ t

So x must be in [max_{a_k<0} (t-b_k)/a_k, min_{a_k>0} (t-b_k)/a_k] ∩ [0,1].
If this interval is non-empty, t is feasible.

We can binary search on t to find the minimum feasible t, and then pick any x in the feasible interval.

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            break
        h[free] += (target_sum - s) / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def find_optimal_x(corr_a, corr_b):
    """Minimize max_k ((1-x)*corr_a[k] + x*corr_b[k]) for x in [0,1]."""
    diff = corr_b - corr_a
    # Binary search on t
    lo = 0.0
    hi = float(max(np.max(corr_a), np.max(corr_b)))
    
    for _ in range(50):
        t = (lo + hi) * 0.5
        pos = diff > 1e-14
        neg = diff < -1e-14
        zero = ~pos & ~neg
        if np.any(zero):
            if np.max(corr_a[zero]) > t + 1e-14:
                lo = t
                continue
        x_lo = 0.0
        x_hi = 1.0
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((t - corr_a[pos]) / diff[pos])))
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((t - corr_a[neg]) / diff[neg])))
        if x_lo > x_hi + 1e-14:
            lo = t
        else:
            hi = t
    
    # Find x at t = hi
    t = hi
    pos = diff > 1e-14
    neg = diff < -1e-14
    x_lo = 0.0
    x_hi = 1.0
    if np.any(pos):
        x_hi = min(x_hi, float(np.min((t - corr_a[pos]) / diff[pos])))
    if np.any(neg):
        x_lo = max(x_lo, float(np.max((t - corr_a[neg]) / diff[neg])))
    return float(np.clip((max(x_lo, 0.0) + min(x_hi, 1.0)) * 0.5, 0.0, 1.0))

def coord_descent(h, target_sum, n_sweeps=1, rng=None):
    n = len(h)
    h = h.copy()
    if rng is None:
        rng = np.random.default_rng()
    for _ in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            h0 = h.copy(); h0[i] = 0.0
            h1 = h.copy(); h1[i] = 1.0
            ca = np.correlate(h0, 1.0 - h0, mode="full")
            cb = np.correlate(h1, 1.0 - h1, mode="full")
            h[i] = find_optimal_x(ca, cb)
        h = project_h(h, target_sum)
    return h

    start = time.time()
    
    def tleft():
        return budget_s - (time.time() - start)
    
    best_c5 = float('inf')
    best_h = None

            c5_new = compute_c5(h)

When h[i] changes from old to new (delta = new - old):
For each lag d (index m = n-1+d in the output):
- If 0 ≤ i+d ≤ n-1: corr[m] += delta * (1 - h[i+d]) [old h[i+d]]
- If 0 ≤ i-d ≤ n-1: corr[m] -= delta * h[i-d] [old h[i-d]]
- Special case d=0: corr[m] += delta*(1-2*old_h[i]) - delta² [exact update]

Wait, but after the update, h[i] = new. And the formula uses h[i+d] and h[i-d] which are unchanged (for d≠0). So:

For d ≠ 0:
corr[n-1+d] += delta * (1-h[i+d]) [if 0≤i+d≤n-1]
corr[n-1+d] -= delta * h[i-d] [if 0≤i-d≤n-1]

For d = 0:
corr[n-1] += delta*(1-2*old_i) - delta*delta

where old_i is the old value of h[i].

    return float(np.clip((max(x_lo, 0.0) + min(x_hi, 1.0)) * 0.5, 0.0, 1.0))

    n = len(h)
    order = rng.permutation(n)
    for i in order:
        h[i] = 0.0
        ca = np.correlate(h, 1.0 - h, mode="full")
        h[i] = 1.0
        cb = np.correlate(h, 1.0 - h, mode="full")
        h[i] = find_optimal_x(ca, cb)
    return project_h(h, target_sum)

def cd_optimize(h, target_sum, rng, max_time, tol=1e-10):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    t0 = time.time()
    prev_diff = 1.0
    while time.time() - t0 < max_time:
        h_new = cd_sweep(h, target_sum, rng)
        c5_new = compute_c5(h_new)
        diff = np.max(np.abs(h_new - h))
        h = h_new
        c5 = c5_new
        if diff < 1e-9 or (prev_diff < tol and diff < tol):
            break
        prev_diff = diff
    return h, c5

def lbfgs_polish(h, target_sum, beta, dx, maxiter=300):
    n = len(h)
    def obj(x):
        hc = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hc, 1.0 - hc, mode="full")
        penalty = 100.0 * (hc.sum() - target_sum) ** 2
        return smooth_max(corr, beta) * dx + penalty
    try:
        res = minimize(obj, h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(res.x, target_sum)

        return h
