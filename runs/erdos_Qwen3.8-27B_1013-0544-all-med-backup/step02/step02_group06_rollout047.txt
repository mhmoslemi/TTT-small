def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):

        if abs(d) < 1e-14:

        m = (h > 1e-10) & (h < 1.0 - 1e-10)

        else:

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = m.sum()
        if cnt == 0:
            h = np.clip(h + d / len(h), 0.0, 1.0)
            break
        adj = d / cnt
        lo_push = 1e-12 - h[m].min() if adj < 0 else -np.inf
        hi_push = 1.0 - 1e-12 - h[m].max() if adj > 0 else np.inf
        adj_c = np.clip(adj, lo_push, hi_push)
        h[m] += adj_c
        h = np.clip(h, 0.0, 1.0)
    return h


def pair_wise_descent(h, n, max_sweeps=50, tol=1e-10):
    best_c5 = compute_c5(h)
    h = h.copy()
    for sweep in range(max_sweeps):
        improved = False
        order = np.random.permutation(n)
        for i in order:
            j = (i + 1) % n
            s = h[i] + h[j]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-10:
                continue
            saved_i, saved_j = h[i], h[j]

            def obj_1d(t):
                h[i] = t
                h[j] = s - t
                return compute_c5(h)

            res = minimize_scalar(obj_1d, bounds=(lo, hi), method='bounded',
                                   options={'xatol': 1e-10, 'maxiter': 60})
            if res.success:
                h[i] = res.x
                h[j] = s - res.x
                c_new = compute_c5(h)
                if c_new < best_c5 - tol:
                    best_c5 = c_new
                    improved = True
                else:
                    h[i] = saved_i
                    h[j] = saved_j
            else:
                h[i] = saved_i
                h[j] = saved_j
        if not improved:
            break
    return h, best_c5


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def _objective(h, n, target_sum, beta, lam):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * (2.0 / n) + lam * (h.sum() - target_sum) ** 2


def gen_binary(n, rng):
    idx = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h


def gen_sinusoidal(n, a, w, p):
    x = np.linspace(0, 2.0, n, endpoint=False) + 1.0 / n
    h = 0.5 + a * np.sin(w * x + p)
    return np.clip(h, 0.0, 1.0)


def gen_block(n, bs, rng):
    h = np.zeros(n)
    for i in range(n // bs):
        h[i * bs:(i + 1) * bs] = 1.0 if i % 2 == 0 else 0.0
    h += rng.randn(n) * 0.05
    return np.clip(h, 0.0, 1.0)
