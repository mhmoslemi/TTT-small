    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(60):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            break
        step = diff / free.sum()
        if step > 0:
            mx = (1.0 - h[free]).min()
            step = min(step, mx)
        else:
            mx = h[free].min()
            step = max(step, -mx)
        h[free] += step
    return np.clip(h, 0.0, 1.0)

                c5_t, corr_t = compute_c5(h_cur)
                top2 = np.argsort(-corr_t)[:2]
                for k_idx in top2:
                    d = int(k_idx) - (n_points - 1)
                    i_lo = max(0, -d)
                    i_hi = min(n_points, n_points - d)
                    if i_hi > i_lo:
                        i = np.random.randint(i_lo, i_hi)
                        j = i + d
                        if 0 <= j < n_points and i != j:
                            delta = np.random.uniform(0.03, 0.1)
                            if h_cur[i] >= h_cur[j]:
                                h_new[i] -= delta
                            else:
                                h_new[j] -= delta
            elif r < 0.48:
                # Block swap
                blen = np.random.randint(2, min(7, n_points // 10 + 1))
                i = np.random.randint(0, n_points - blen + 1)
                j = np.random.randint(0, n_points - blen + 1)
                tries = 0
                while abs(j - i) <= blen and tries < 30:
                    j = np.random.randint(0, n_points - blen + 1)
                    tries += 1
                if abs(j - i) > blen:
                    tmp = h_new[i:i + blen].copy()
                    h_new[i:i + blen] = h_new[j:j + blen]
                    h_new[j:j + blen] = tmp
            elif r < 0.63:
                # Gaussian subset perturbation
                frac = 0.08 + 0.25 * (T / 0.006)
                mask = np.random.random(n_points) < frac
                nm = int(mask.sum())
                if nm > 0:
                    scale = 0.02 + 0.06 * (T / 0.006)
                    h_new[mask] += np.random.randn(nm) * scale
            elif r < 0.78:
                # Block circular shift
                blen = np.random.randint(3, min(6, n_points // 8 + 1))
                i = np.random.randint(0, n_points - blen + 1)
                shift = np.random.randint(-3, 4)
                if shift != 0:
                    block = h_new[i:i + blen].copy()
                    h_new[i:i + blen] = np.roll(block, shift)
            elif r < 0.88:
                # Pair swaps
                for _ in range(np.random.randint(1, 4)):
                    i, j = np.random.choice(n_points, 2, replace=False)
                    h_new[i], h_new[j] = h_new[j], h_new[i]
            else:
                # Bimodal push
                n_push = np.random.randint(3, 8)
                idx = np.random.choice(n_points, n_push, replace=False)
                for i in idx:
                    if h_cur[i] > 0.5:
                        h_new[i] += np.random.uniform(0.04, 0.12)
                    else:
                        h_new[i] -= np.random.uniform(0.04, 0.12)

            h_new = project_h(h_new, target)
            c5_new, _ = compute_c5(h_new)
            delta = c5_new - c5_cur
            if delta < 0 or np.random.random() < np.exp(-min(400.0, delta / T)):
                h_cur = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_h = h_cur.copy()
                    best_c5 = c5_cur
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

            T = max(T_min, T * cooling)
            if no_improve > 600:
                T = max(T, 0.005)
                no_improve = 0

    # Phase 4: Final high-precision polish
    if tleft() > 30:
        for _ in range(4):
            if tleft() < 15:
                break
            c5_cur, corr_cur = compute_c5(best_h)
            for n_act in [3, 5]:
                if tleft() < 10:
                    break
                weights = active_weights(corr_cur, n_act, 0.25)
                _g_params['weights'] = weights
                _g_params['dx'] = dx
                _g_params['target'] = target
                try:
                    res = minimize(_obj_call, best_h, jac=_obj_grad_call,
                                   method='SLSQP', bounds=bounds,
                                   constraints=[{'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_jac}],
                                   options={'maxiter': 5000, 'ftol': 1e-18})
                    h_new = project_h(res.x, target)
                    c5_new, _ = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_h, best_c5 = h_new.copy(), c5_new
                except Exception:
                    pass

    # Also try n=100 if time permits
    if tleft() > 120:
        n2 = 100
        dx2 = 2.0 / n2
        target2 = n2 / 2.0
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n_points, endpoint=False), best_h)
        h2 = project_h(h2, target2)
        c5_2, _ = compute_c5(h2)
        bounds2 = Bounds(np.zeros(n2), np.ones(n2))
        best2 = h2.copy()
        best_c5_2 = c5_2

        for n_act in [4, 6, 8]:
            for _ in range(5):
                if tleft() < 60:
                    break
                c5_t, corr_t = compute_c5(best2)
                weights = active_weights(corr_t, n_act, 0.4)
                _g_params['weights'] = weights
                _g_params['dx'] = dx2
                _g_params['target'] = target2
                try:
                    res = minimize(_obj_call, best2, jac=_obj_grad_call,
                                   method='L-BFGS-B', bounds=bounds2,
                                   options={'maxiter': 500, 'gtol': 1e-14, 'ftol': 1e-16})
                    h_new = project_h(res.x, target2)
                    c5_new, _ = compute_c5(h_new)
                    if c5_new < best_c5_2:
                        best2, best_c5_2 = h_new.copy(), c5_new
                except Exception:
                    pass
            if tleft() < 40:
                break

        for _ in range(10):
            if tleft() < 30:
                break
            c5_t, corr_t = compute_c5(best2)
            weights = active_weights(corr_t, 5, 0.35)
            _g_params['weights'] = weights
            _g_params['dx'] = dx2
            _g_params['target'] = target2
            try:
                res = minimize(_obj_call, best2, jac=_obj_grad_call,
                               method='SLSQP', bounds=bounds2,
                               constraints=[{'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_jac}],
                               options={'maxiter': 2000, 'ftol': 1e-15})
                h_new = project_h(res.x, target2)
                c5_new, _ = compute_c5(h_new)
                if c5_new < best_c5_2:
                    best2, best_c5_2 = h_new.copy(), c5_new
            except Exception:
                pass

        if best_c5_2 < best_c5:
            best_h, best_c5, n_points = best2, best_c5_2, n2

    # Final projection and verification
    best_h = project_h(best_h, n_points / 2.0)
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, n_points
