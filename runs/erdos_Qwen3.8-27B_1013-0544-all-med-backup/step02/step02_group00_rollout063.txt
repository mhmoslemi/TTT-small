import numpy as np
from scipy.optimize import minimize

# Global parameters for L-BFGS-B
_g_beta = 100.0
_g_penalty = 500.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nfree = free.sum()
        if nfree == 0:
            free = np.ones(len(h), dtype=bool)
            nfree = len(h)
        h[free] += diff / nfree
        h = np.clip(h, 0.0, 1.0)
    return h

def _gobj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    xx = np.clip(_g_beta * (corr - m), -500, 500)
    sm = m + np.log(np.sum(np.exp(xx)) / len(xx)) / _g_beta
    return sm + _g_penalty * (np.sum(x) - n * 0.5) ** 2

def _ggrad(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    xx = np.clip(_g_beta * (corr - m), -500, 500)
    ex = np.exp(xx)
    w = ex / ex.sum()
    g = np.zeros(n)
    order = np.argsort(w)[::-1]
    for k in order:
        wk = w[k]
        if wk < 1e-10:
            break
        lag = int(k) - (n - 1)
        if lag == 0:
            g += wk * (1.0 - 2.0 * x)
        else:
            a = max(0, -lag)
            b = min(n, n - lag)
            if b > a:

            c = max(0, lag)
            d = min(n, n + lag)

                g[c:d] += -wk * x[c - lag:d - lag]
    g *= dx
    g += 2.0 * _g_penalty * (np.sum(x) - n * 0.5)
    return g

    _g_beta = beta
    _g_penalty = penalty
    n = len(h0)

                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'maxfun': 4000})
    return project_h(res.x, n * 0.5)

def proj_grad_optimize(h0, target, max_iter, dx_budget):
    """Projected gradient with momentum."""
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    best = h.copy()
    best_c5 = c5
    v = np.zeros(n)
    momentum = 0.9
    
    for it in range(max_iter):
        if time.time() - dx_budget > 5:
            break
        beta = 100 + 500.0 * it / max_iter
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        m = np.max(corr)
        xx = np.clip(beta * (corr - m), -500, 500)
        ex = np.exp(xx)
        w = ex / ex.sum()
        
        g = np.zeros(n)
        order = np.argsort(w)[::-1]
        for k in order:
            wk = w[k]
            if wk < 1e-10:
                break
            lag = int(k) - (n - 1)
            if lag == 0:
                g += wk * (1.0 - 2.0 * h)
            else:
                a = max(0, -lag)
                b = min(n, n - lag)

                c = max(0, lag)
                d = min(n, n + lag)
                if d > c:
                    g[c:d] += -wk * h[c - lag:d - lag]
        g *= dx

        if gn < 1e-15:
            break
        g /= gn

        h_new = project_h(h + v, target)
        c5_new = compute_c5(h_new)

            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
        else:

            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
    
    return best, best_c5

def local_search(h0, target, max_iter=5000):

    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    best = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

_g_beta = 100.0
_g_pen = 500.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf == 0:
            h += diff / len(h)
        else:
            h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def _sm_grad(h, beta):
    """Gradient of smooth max of correlate(h,1-h)*dx, no penalty."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    xx = np.clip(beta * (corr - m), -500, 500)
    ex = np.exp(xx)
    w = ex / ex.sum()
    g = np.zeros(n)
    order = np.argsort(w)[::-1]
    for k in order:
        wk = w[k]
        if wk < 1e-10:
            break
        lag = int(k) - (n - 1)
        a1 = max(0, lag)
        b1 = min(n, n + lag)
        if b1 > a1:
            g[a1:b1] += wk * (1.0 - h[a1 - lag:b1 - lag])
        a2 = max(0, -lag)
        b2 = min(n, n - lag)
        if b2 > a2:
            g[a2:b2] += -wk * h[a2 + lag:b2 + lag]
    return g * dx

def _gobj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    xx = np.clip(_g_beta * (corr - m), -500, 500)
    sm = m + np.log(np.sum(np.exp(xx)) / len(xx)) / _g_beta
    return sm + _g_pen * (np.sum(x) - n * 0.5) ** 2

def _ggrad(x):
    n = len(x)
    g = _sm_grad(x, _g_beta)
    g += 2.0 * _g_pen * (np.sum(x) - n * 0.5)
    return g

def lbfgs_run(h0, beta, pen, maxiter):
    global _g_beta, _g_pen
    _g_beta = float(beta)
    _g_pen = float(pen)
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(_gobj, h0.copy(), jac=_ggrad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'maxfun': maxiter * 3})
    return project_h(res.x, n * 0.5)

def pg_step(h, target, beta):
    """One projected gradient step with normalized direction."""
    n = len(h)
    g = _sm_grad(h, beta)
    gn = np.linalg.norm(g)
    if gn < 1e-15:
        return h.copy(), 0.0
    return g / gn, gn

def pg_optimize(h0, target, n_iter, t_start, t_budget):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    best = h.copy()
    best_c5 = c5
    v = np.zeros(n)
    mom = 0.9
    for it in range(n_iter):
        if time.time() - t_start > t_budget:
            break
        beta = 80.0 + 600.0 * it / max(1, n_iter)
        g, gn = pg_step(h, target, beta)
        if gn < 1e-15:
            break
        step = 0.06 / (1.0 + 0.015 * it)
        v = mom * v - step * g
        h_new = project_h(h + v, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
        else:
            v *= 0.5
            h = project_h(h + v, target)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
    return best, best_c5

def local_search(h0, target, n_iter, t_start, t_budget):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    best = h.copy()
    best_c5 = c5
    for it in range(n_iter):
        if time.time() - t_start > t_budget:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_k = np.argsort(corr)[::-1][:3]
        g = np.zeros(n)
        for k in top_k:
            lag = int(k) - (n - 1)
            a1 = max(0, lag)
            b1 = min(n, n + lag)
            if b1 > a1:
                g[a1:b1] += (1.0 - h[a1 - lag:b1 - lag])
            a2 = max(0, -lag)
            b2 = min(n, n - lag)
            if b2 > a2:
                g[a2:b2] += -h[a2 + lag:b2 + lag]
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        step = 0.04 / (1.0 + 0.008 * it)
        for _ in range(4):
            h_try = project_h(h - step * g, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5 - 1e-15:
                h = h_try
                c5 = c5_try
                if c5 < best_c5:
                    best_c5 = c5
                    best = h.copy()
                break
            step *= 0.4
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        hp = np.array(initial_h_values, dtype=float).copy()
    else:
        hp = np.full(140, 0.5)

    best_h = hp.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(hp)

    n_list = sorted(set([best_n, 120, 140, 160, 180, 200]))
    n_list = [n for n in n_list if 60 <= n <= 250]

    for n in n_list:
        if budget_s - (time.time() - t0) < 50:
            break
        target = n * 0.5
        xg = np.linspace(0, 2, n)

        inits = []
        if n == len(hp):
            inits.append(hp.copy())
        else:
            inits.append(project_h(np.interp(xg, np.linspace(0, 2, len(hp)), hp), target))
        if n != best_n:
            inits.append(project_h(np.interp(xg, np.linspace(0, 2, best_n), best_h), target))
        inits.append(project_h(np.random.random(n), target))
        inits.append(project_h(np.where(np.arange(n) % 2 == 0, 0.7, 0.3), target))
        inits.append(project_h(0.5 + 0.35 * np.sin(2 * np.pi * 2 * xg / 2.0), target))
        inits.append(project_h(0.5 + 0.3 * np.cos(2 * np.pi * 3 * xg / 2.0), target))

        for h0 in inits:
            if budget_s - (time.time() - t0) < 25:
                break
            h_cur = h0.copy()
            c_cur = compute_c5(h_cur)
            if c_cur < best_c5:
                best_h, best_c5, best_n = h_cur.copy(), c_cur, n

            for beta in [40, 80, 150, 300, 600]:
                if budget_s - (time.time() - t0) < 15:
                    break
                h_new = lbfgs_run(h_cur, beta, 500.0, maxiter=800)
                c_new = compute_c5(h_new)
                if c_new < best_c5:
                    best_h, best_c5, best_n = h_new.copy(), c_new, n
                if c_new <= c_cur + 1e-10:
                    h_cur = h_new.copy()
                    c_cur = c_new
                else:
                    h_cur = h_new.copy()

            if budget_s - (time.time() - t0) < 15:
                break
            h_pg, c_pg = pg_optimize(h_cur, target, 2000, t0, budget_s - 10)
            if c_pg < best_c5:
                best_h, best_c5, best_n = h_pg.copy(), c_pg, n
            h_cur = h_pg

            if budget_s - (time.time() - t0) < 10:
                break
            h_ls, c_ls = local_search(h_cur, target, 3000, t0, budget_s - 5)
            if c_ls < best_c5:
                best_h, best_c5, best_n = h_ls.copy(), c_ls, n
                h_cur = h_ls

    # Final polish with very high beta
    if budget_s - (time.time() - t0) > 20:
        nf = len(best_h)
        for beta in [800, 1500, 3000]:
            if budget_s - (time.time() - t0) < 10:
                break
            h_new = lbfgs_run(best_h, beta, 2000.0, maxiter=2000)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_h, best_c5 = h_new.copy(), c_new

    # Final local search
    if budget_s - (time.time() - t0) > 8:
        h_ls, c_ls = local_search(best_h, len(best_h) * 0.5, 10000, t0, budget_s - 2)
        if c_ls < best_c5:
            best_h, best_c5 = h_ls.copy(), c_ls

    # Iterated perturbation
    if budget_s - (time.time() - t0) > 15:
        for trial in range(5):
            if budget_s - (time.time() - t0) < 12:
                break
            noise = np.random.randn(len(best_h)) * 0.05
            h_pert = project_h(best_h + noise, len(best_h) * 0.5)
            h_new = lbfgs_run(h_pert, 500.0, 1000.0, maxiter=500)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_h, best_c5 = h_new.copy(), c_new

    return best_h, best_c5, best_n
