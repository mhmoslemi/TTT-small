    m = c.max()
    shifted = alpha * (c - m)

        b = coeffs[2*m]

    return h

    return coeffs

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15: break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15: break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def slsqp_step(h, n, alpha, maxiter=2000, ftol=1e-14):
    dx = 2.0 / n
    def obj(v):
        c = np.correlate(v, 1-v, mode='full') * dx
        m = c.max()
        s = alpha * (c - m)
        mx = s.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(s-mx))))/alpha
        return m + np.log(np.sum(np.exp(s)))/alpha
    def cons_fun(v):
        return np.sum(v) - n/2.0
    def cons_jac(v):
        return np.ones(n)
    try:
        res = minimize(obj, h, method='SLSQP',
                      bounds=[(0,1)]*n,
                      constraints=[{'type':'eq','fun':cons_fun,'jac':cons_jac}],
                      options={'maxiter':maxiter,'ftol':ftol})
        return project_feasible(res.x, n)
    except:
        return h

    s = alpha * (c - m)

    # dc_k/dh[j] = dx * (h[j-k]*(1-h[j]) ... ) - wait this is complicated

        if c < best_c5 - 1e-15:
            best_h, best_c5 = ht, c

    # Final validation
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
