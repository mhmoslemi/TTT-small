    return project_feasible(h, target_sum)


def coord_sweep_anneal(h, target_sum, n_eval=60, temperature=0.0, order=None):
    n = len(h)
    if order is None:
        order = np.random.permutation(n)
    m_arr = np.arange(2 * n - 1)
    current_c5 = np.max(np.correlate(h, 1.0 - h, mode="full"))
    for p in order:
        v = h[p]
        C = np.correlate(h, 1.0 - h, mode="full")
        g = np.zeros(2 * n - 1)
        j = p + m_arr - n + 1
        i = p - m_arr + n - 1
        mj = (j >= 0) & (j < n)
        mi = (i >= 0) & (i < n)
        g[mj] += (1.0 - h[j[mj]])
        g[mi] += (-h[i[mi]])
        deltas = np.linspace(-v, 1.0 - v, n_eval)
        C_new = C[None, :] + deltas[:, None] * g[None, :]
        C_new[:, n - 1] -= deltas ** 2
        obj = C_new.max(axis=1)
        if temperature > 0:
            # Accept with some probability even if worse (simulated annealing)
            best_idx = np.argmin(obj)
            best_d = deltas[best_idx]
            best_obj = obj[best_idx]
            if best_obj > current_c5 and np.random.rand() < np.exp(-(best_obj - current_c5) / temperature):
                h[p] = v + best_d
                current_c5 = best_obj
            elif best_obj <= current_c5:
                h[p] = v + best_d
                current_c5 = best_obj
        else:
            best_d = deltas[np.argmin(obj)]
            h[p] = v + best_d
    return h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0

    best_h = project_feasible(h_init.copy(), target_sum)
    best_c5 = compute_c5(best_h)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: Aggressive coordinate descent (60% budget)
    phase1_budget = 0.55 * budget_s
    n_eval = 50
    no_improve = 0
    sweep = 0
    while remaining() > budget_s - phase1_budget + 20 and sweep < 20000:
        sweep += 1
        h = best_h.copy()
        if sweep % 50 == 0:
            # Small perturbation to escape local optima
            idx = np.random.choice(n, size=max(3, n // 10), replace=False)
            h[idx] += np.random.randn(len(idx)) * 0.05
            h = project_feasible(h, target_sum)
        h = coord_sweep(h, target_sum, n_eval=n_eval)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200 and sweep % 100 == 0:
            # Restart with larger perturbation
            h = best_h.copy()
            h += np.random.randn(n) * 0.1
            h = project_feasible(h, target_sum)
            best_h = h
            no_improve = 0

    # Phase 2: Simulated annealing coordinate descent (15% budget)
    phase2_start = time.time() - t0
    phase2_budget = 0.15 * budget_s
    if remaining() > phase2_budget + 30:
        h = best_h.copy()
        temp = 0.02
        decay = 0.97
        for it in range(2000):
            if time.time() - t0 > budget_s - phase2_budget - 30:
                break
            h = coord_sweep_anneal(h, target_sum, n_eval=40, temperature=temp)
            h = project_feasible(h, target_sum)
            temp *= decay
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 3: SLSQP polishing (25% budget)
    if remaining() > 30:
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
        for beta in [200, 1000, 5000, 20000]:
            if remaining() < 15:
                break
            for trial in range(2):
                if remaining() < 10:
                    break
                x0 = best_h.copy()
                if trial > 0:
                    x0 += np.random.randn(n) * 0.003
                    x0 = project_feasible(x0, target_sum)
                try:
                    res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 5000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_new.copy()
                except Exception:
                    pass

    # Phase 4: Final coordinate descent polish
    if remaining() > 10:
        h = best_h.copy()
        for _ in range(200):
            if remaining() < 5:
                break
            h = coord_sweep(h, target_sum, n_eval=100)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 5: Try nearby resolutions
    if remaining() > 60:
        for n_try in [n - 5, n + 5, n + 10, n - 10, n + 15, n - 15]:
            if n_try < 30 or remaining() < 20:
                continue
            ts = n_try / 2.0
            dx_t = 2.0 / n_try
            h_t = project_feasible(resample_h(best_h, n_try), ts)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                n = n_try
                target_sum = ts
                dx = dx_t
                continue
            # Quick coord descent on new resolution
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
            for _ in range(50):
                if remaining() < 10:
                    break
                h_t = coord_sweep(h_t, ts, n_eval=50)
                c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                n = n_try
                target_sum = ts
                dx = dx_t
            if remaining() < 15:
                break
            # SLSQP on new resolution
            try:
                res = minimize(obj_smooth, h_t, args=(dx_t, 2000), method='SLSQP',
                               bounds=bnds_t, constraints=cons_t,
                               options={'maxiter': 3000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, ts)
                c5_t = compute_c5(h_new)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h_new.copy()
                    n = n_try
                    target_sum = ts
                    dx = dx_t
            except Exception:
                pass

    # Final check
    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
