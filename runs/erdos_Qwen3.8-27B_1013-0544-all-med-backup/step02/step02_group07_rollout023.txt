    """Compute gradient of smooth_max(correlate(h, 1-h), beta)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Weights for smooth max
    m = np.max(corr)
    weights = np.exp(beta * (corr - m)) / np.sum(np.exp(beta * (corr - m)))
    
    n = n_points
    grad = np.zeros(n)
    
    for lag_idx in range(2*n - 1):
        w = weights[lag_idx]
        if w < 1e-15:

        d = lag_idx - (n - 1)

                grad[j+d] += w * (-h[j])

            for j in range(ad, n):
                grad[j] += w * (1 - h[j-ad])

def smooth_max_val(x, beta):
    m = np.max(x)
    mx = x - m
    return m + np.log(np.sum(np.exp(beta * mx))) / beta

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)

_g_target = 35.0
_g_n = 70
_g_beta = 100

def _obj(x):
    h = project_h(x, _g_target)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max_val(corr, _g_beta) * (2.0 / _g_n)

def _obj_grad(x):
    h = project_h(x, _g_target)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    ex = np.exp(_g_beta * (corr - m))
    w = ex / np.sum(ex)
    n = _g_n
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag_idx in range(2 * n - 1):
        wi = w[lag_idx] * dx
        if wi < 1e-16:
            continue
        d = lag_idx - (n - 1)
        if d >= 0:
            grad[:n - d] += wi * (1.0 - h[d:])
            grad[d:] += wi * (-h[:n - d])
        else:
            ad = -d
            grad[ad:] += wi * (1.0 - h[:n - ad])
            grad[:n - ad] += wi * (-h[ad:])
    grad -= np.mean(grad)
    return grad

def coord_descent_pairwise(h, n, target, n_sweeps=200, top_k=5, rng=None):
    """Pairwise coordinate descent focusing on active lags."""
    if rng is None:
        rng = np.random
    best = compute_c5(h)
    h = h.copy()
    
    for sweep in range(n_sweeps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[-top_k:][::-1]
        
        improved = False
        for _ in range(n):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            s_pair = h[i] + h[j]
            lo = max(0.0, s_pair - 1.0)
            hi = min(1.0, s_pair)
            if hi - lo < 1e-12:
                continue
            # Try several t values in [lo, hi]
            for t in rng.uniform(lo, hi, size=5):
                h_try = h.copy()
                h_try[i] = t
                h_try[j] = s_pair - t
                h_try = project_h(h_try, target)
                c5_try = compute_c5(h_try)
                if c5_try < best - 1e-12:
                    h = h_try
                    best = c5_try
                    improved = True
                    break
        if not improved:
            # Perturb to escape
            mask = rng.random(n) < 0.15
            h[mask] += rng.randn(int(mask.sum())) * 0.04
            h = project_h(h, target)
            best = compute_c5(h)
    return h, best

def run(seed, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        n_points = len(initial_h_values)
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        n_points = 70
        h_init = np.full(n_points, 0.5)
    
    target = n_points / 2.0
    rng = np.random.RandomState(seed)
    
    best_h = project_h(h_init, target)
    best_c5 = compute_c5(best_h)
    
    def time_left():
        return budget_s - (time.time() - start)
    
    # Phase 1: L-BFGS-B with increasing beta
    global _g_target, _g_n, _g_beta
    _g_target = target
    _g_n = n_points
    
    for beta in [30, 60, 120, 250, 500, 1000, 2000]:
        if time_left() < 40:
            break
        _g_beta = beta
        try:
            res = minimize(_obj, best_h, jac=_obj_grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n_points,
                          options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-14})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass
    
    # Phase 2: Pairwise coordinate descent
    if time_left() > 120:
        h_cur = best_h.copy()
        n_sweeps = int(min(400, time_left() * 0.3))
        h_cur, c5_cur = coord_descent_pairwise(h_cur, n_points, target,
                                               n_sweeps=n_sweeps, top_k=5, rng=rng)
        if c5_cur < best_c5:
            best_h = h_cur
            best_c5 = c5_cur
    
    # Phase 3: More L-BFGS-B rounds from improved point
    if time_left() > 60:
        for beta in [500, 1000, 2000, 4000]:
            if time_left() < 20:
                break
            _g_beta = beta
            try:
                res = minimize(_obj, best_h, jac=_obj_grad, method='L-BFGS-B',
                              bounds=[(0.0, 1.0)] * n_points,
                              options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-15})
                h_new = project_h(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass
    
    # Phase 4: Targeted simulated annealing
    if time_left() > 80:
        h_cur = best_h.copy()
        c5_cur = best_c5
        T = 0.004
        T_min = 0.0001
        for it in range(50000):
            if time_left() < 20:
                break
            h_try = h_cur.copy()
            corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
            top_idx = np.argsort(corr)[-3:][::-1]
            
            for lag_idx in top_idx:
                d = int(lag_idx - (n_points - 1))
                if d == 0:
                    idx = rng.choice(n_points, 3, replace=False)
                    for i in idx:
                        if h_try[i] > 0.5:
                            h_try[i] -= 0.03 + 0.03 * rng.random()
                        else:
                            h_try[i] += 0.03 + 0.03 * rng.random()
                else:
                    ad = abs(d)
                    n_pairs = n_points - ad
                    if n_pairs <= 0:
                        continue
                    idx = rng.choice(n_pairs, min(4, n_pairs), replace=False)
                    for ii in idx:
                        if d > 0:
                            a, b = ii + ad, ii
                        else:
                            a, b = ii, ii + ad
                        if h_try[a] > 0.4:
                            h_try[a] -= 0.02 + 0.02 * rng.random()
                        if h_try[b] < 0.6:
                            h_try[b] += 0.02 + 0.02 * rng.random()
            
            h_try = project_h(h_try, target)
            c5_try = compute_c5(h_try)
            delta = c5_try - c5_cur
            if delta < 0 or rng.random() < np.exp(-min(200.0, delta / T)):
                h_cur = h_try
                c5_cur = c5_try
                if c5_cur < best_c5:
                    best_h = h_cur.copy()
                    best_c5 = c5_cur
            
            T = max(T * 0.9992, T_min)
    
    # Phase 5: Final coordinate descent + L-BFGS-B polish
    if time_left() > 30:
        h_cur = best_h.copy()
        h_cur, c5_cur = coord_descent_pairwise(h_cur, n_points, target,
                                               n_sweeps=100, top_k=3, rng=rng)
        if c5_cur < best_c5:
            best_h = h_cur
            best_c5 = c5_cur
        
        if time_left() > 15:
            _g_beta = 3000
            try:
                res = minimize(_obj, best_h, jac=_obj_grad, method='L-BFGS-B',
                              bounds=[(0.0, 1.0)] * n_points,
                              options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-16})
                h_new = project_h(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass
    
    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
