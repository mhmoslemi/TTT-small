                                           method='SLSQP', bounds=bounds, constraints=con,

                            h_new = project_h(res.x, target_sum)
                            c5 = compute_c5(h_new)
                            if c5 < curr_c5:
                                curr_c5, curr_h = c5, h_new.copy()
                                h0 = h_new.copy()
                        except Exception:
                            pass

            # Phase C: Penalty feasibility approach
            if curr_h is not None and check_time():
                target_c = curr_c5 - 0.001
                for pw in [50.0, 200.0]:
                    if not check_time():
                        break
                    try:
                        res = minimize(obj_penalty_feasibility, h0, args=(dx, target_c, pw, 200.0),
                                       method='SLSQP', bounds=bounds, constraints=con,
                                       options={'maxiter': 200, 'ftol': 1e-13})
                        h_new = project_h(res.x, target_sum)
                        c5 = compute_c5(h_new)
                        if c5 < curr_c5:
                            curr_c5, curr_h = c5, h_new.copy()
                            h0 = h_new.copy()
                    except Exception:
                        pass

        # Phase D: Dual Annealing for global search
        if check_time():
            h_da = curr_h if curr_h is not None else starts[0]
            h_da = project_h(h_da, target_sum)
            try:
                res = minimize(obj_smooth_max, h_da, args=(dx, 80.0),
                               method='dual_annealing', bounds=bounds,
                               options={'maxiter': 200, 'initial_temp': 0.5, 'restarts': 2, 'polish': False})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < curr_c5:
                    curr_c5, curr_h = c5, h_new.copy()
            except Exception:
                pass

        # Phase E: Stochastic local search
        if curr_h is not None and check_time():
            base = curr_h.copy()
            for it in range(15):
                if not check_time():
                    break
                scale = 0.015 + 0.01 * it
                h0 = base + np.random.randn(n_points) * scale
                h0 = project_h(h0, target_sum)
                try:
                    res = minimize(obj_smooth_max, h0, args=(dx, 150.0),
                                   method='SLSQP', bounds=bounds, constraints=con,
                                   options={'maxiter': 120, 'ftol': 1e-12})
                    h_new = project_h(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < curr_c5:
                        curr_c5, curr_h = c5, h_new.copy()
                        base = h_new.copy()
                    else:
                        break
                except Exception:
                    break

        if curr_c5 < best_c5:
            update_best(curr_h, curr_c5, n_points)

    # Final high-precision polish
    if best_h is not None and time.time() - t_start < budget_s - 15:
        n_f = best_n
        ts = n_f / 2.0
        dx_f = 2.0 / n_f
        bounds_f = Bounds(np.zeros(n_f), np.ones(n_f))
        con_f = [{'type': 'eq', 'fun': con_sum, 'jac': jac_con, 'args': (ts,)}]

        for beta in [300, 500, 800]:
            if time.time() - t_start > budget_s - 8:
                break
            try:
                res = minimize(obj_smooth_max, best_h, args=(dx_f, beta),
                               method='SLSQP', bounds=bounds_f, constraints=con_f,
                               options={'maxiter': 1500, 'ftol': 1e-15})
                h_new = project_h(res.x, ts)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

        # Active lag final polish
        if time.time() - t_start < budget_s - 8:
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            top_lags = list(np.argsort(corr)[::-1][:8])
            for pb in [300, 500]:
                if time.time() - t_start > budget_s - 5:
                    break
                try:
                    res = minimize(obj_active_lag, best_h, args=(dx_f, pb, top_lags),
                                   method='SLSQP', bounds=bounds_f, constraints=con_f,
                                   options={'maxiter': 1000, 'ftol': 1e-15})
                    h_new = project_h(res.x, ts)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h, best_c5 = h_new.copy(), c5
                except Exception:
                    pass

    if best_h is None:
        n = 70
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n

    return best_h, best_c5, best_n
