        v, _ = smoothmax_val_grad(x, beta)
        return v + penalty * (x.sum() - target) ** 2

    def objg(x):
        _, g = smoothmax_val_grad(x, beta)
        return g + 2.0 * penalty * (x.sum() - target) * np.ones(n)

    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(obj, h, jac=objg, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8,
                                'maxls': 40})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def momentum_smoothmax(h, max_iters, beta, lr0, decay, momentum, nesterov_mu):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    v = np.zeros(n)
    best_x = x.copy()
    best_c = compute_c5(x)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = nesterov_mu if t > 2 else (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        val, grad = smoothmax_val_grad(y, beta)
        v = momentum * v - lr * grad
        x_new = project_feasible(y + v, target)
        c = compute_c5(x_new)
        if c < best_c:
            best_c = c
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_c


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def make_binary_approx(h, rng):
    n = len(h)
    target = n // 2
    h_bin = (h + rng.random(n) * 0.01 > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones > target:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])
        for idx in order:
            if h_bin.sum() == target:
                break
            h_bin[ones[idx]] = 0
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        for idx in order:
            if h_bin.sum() == target:
                break
            h_bin[zeros[idx]] = 1
    return project_feasible(h_bin, n / 2.0)


def mutate(h, sigma, rng):
    n = len(h)
    h_new = h.copy() + rng.normal(0, sigma, n)
    return project_feasible(h_new, n / 2.0)


def crossover(h1, h2, rng):
    n = len(h1)
    pt = rng.randint(1, n - 1)
    child = np.concatenate([h1[:pt], h2[pt:]])
    return project_feasible(child, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)

    best_c5 = compute_c5(h_best)

    # Phase 1: Adaptive smooth max at parent resolution via L-BFGS-B
    betas = [15, 40, 100, 250, 500, 1000, 2000]
    for beta in betas:
        if remaining() < 50:
            break
        h_new = lbfgs_refine(h_best, maxiter=300, beta=beta, penalty=50.0)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best = h_new
            best_c5 = c

    # Phase 2: Momentum-based smooth max (Nesterov) for additional refinement
    if remaining() > 100:
        h_m, c_m = momentum_smoothmax(h_best, 3000, beta=500, lr0=0.05,
                                      decay=0.9995, momentum=0.9, nesterov_mu=0.5)
        if c_m < best_c5:
            h_best = h_m
            best_c5 = c_m

    # Phase 3: Resolution sweep
    for n_try in [150, 200, 250, 300, 350, 500]:
        if remaining() < 80:
            break
        if n_try == len(h_best):
            continue
        h_interp = interp_to_n(h_best, n_try)
        for beta in [80, 200, 600]:
            if remaining() < 50:
                break
            h_new = lbfgs_refine(h_interp, maxiter=200, beta=beta, penalty=50.0)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best = h_new
                best_c5 = c
                h_interp = h_new.copy()
        if remaining() > 40:
            h_m, c_m = momentum_smoothmax(h_interp, 1500, beta=800, lr0=0.03,
                                          decay=0.9997, momentum=0.85, nesterov_mu=0.4)
            if c_m < best_c5:
                h_best = h_m
                best_c5 = c_m

    # Phase 4: Population-based diversity
    if remaining() > 150:
        n_cur = len(h_best)
        pop = [h_best.copy()]
        # Generate diverse individuals
        pop.append(mutate(h_best, 0.3, rng))
        pop.append(mutate(h_best, 0.5, rng))
        pop.append(make_binary_approx(h_best, rng))
        h_sym = (h_best + h_best[::-1]) / 2.0
        pop.append(project_feasible(h_sym, n_cur / 2.0))
        pop.append(mutate(h_best, 0.8, rng))

        # Evaluate population
        pop_scores = [compute_c5(p) for p in pop]
        for i in range(len(pop)):
            if pop_scores[i] < best_c5:
                best_c5 = pop_scores[i]
                h_best = pop[i].copy()

        # GA loop with local search
        max_ga_time = min(remaining() * 0.5, 300)
        ga_start = time.time()
        while time.time() - ga_start < max_ga_time and remaining() > 60:
            # Select parents
            order = np.argsort(pop_scores)
            p1 = pop[order[rng.randint(0, 3)]]
            p2 = pop[order[rng.randint(0, 3)]]
            # Crossover + mutation
            if rng.random() < 0.7:
                child = crossover(p1, p2, rng)
            else:
                child = p1.copy()
            child = mutate(child, 0.2, rng)
            # Local polish
            child = lbfgs_refine(child, maxiter=80, beta=200, penalty=50.0)
            cs = compute_c5(child)
            # Replace worst
            worst_idx = np.argmax(pop_scores)
            pop[worst_idx] = child
            pop_scores[worst_idx] = cs
            if cs < best_c5:
                best_c5 = cs
                h_best = child.copy()
                pop[0] = child.copy()
                pop_scores[0] = cs

    # Phase 5: Final high-precision polish
    if remaining() > 30:
        for beta in [800, 1500, 3000]:
            if remaining() < 20:
                break
            h_new = lbfgs_refine(h_best, maxiter=300, beta=beta, penalty=200.0)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best = h_new
                best_c5 = c

    # Phase 6: Short momentum refinement
    if remaining() > 30:
        h_m, c_m = momentum_smoothmax(h_best, 2000, beta=2000, lr0=0.01,
                                      decay=0.9999, momentum=0.9, nesterov_mu=0.3)
        if c_m < best_c5:
            h_best = h_m
            best_c5 = c_m

    return h_best, best_c5, len(h_best)
