import numpy as np
from scipy.optimize import minimize, Bounds, differential_evolution
from scipy.interpolate import interp1d
import time


def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr_full(h)) * dx)


def smoothmax(h, beta, top_k=None):
    """Smooth max of correlation values."""
    corr = compute_corr_full(h)
    dx = 2.0 / len(h)
    if top_k is not None and top_k < len(corr):
        # Use top-k smooth max
        top_vals = np.sort(corr)[-top_k:]
        m = top_vals[0]
        exps = np.exp(beta * (top_vals - m))
        sw = np.sum(exps)
        val = (m + np.log(sw) / beta) * dx
        return val, top_vals, m
    else:
        m = np.max(corr)
        exps = np.exp(beta * (corr - m))
        sw = np.sum(exps)
        val = (m + np.log(sw) / beta) * dx
        return val, corr, m


def smoothmax_grad(h, beta, top_k=None):
    """Gradient of smooth max wrt h."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    j = np.arange(n)
    
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(corr)[-top_k:]
        top_vals = corr[top_idx]
        m = np.min(top_vals)
        exps = np.exp(beta * (top_vals - m))
        w = exps / np.sum(exps)
    else:
        top_idx = np.argsort(corr)[-max(10, n//10):]
        # Use all with weights
        m = np.max(corr)
        exps = np.exp(beta * (corr - m))
        w_full = exps / np.sum(exps)
        top_idx = np.arange(len(corr))
        w = w_full
    
    grad = np.zeros(n)
    for idx, wk in zip(top_idx, w):
        if wk < 1e-12:
            continue
        s = idx - (n - 1)
        # h[i] appears as h(x) at position i, and as h(x+k) at position i-s
        js = j - s  # positions where h[i] is in the "h(x)" slot
        jp = j + s  # positions where h[i] is in the "1-h(x+k)" slot (i.e., h[i-s])
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return grad


def project_feasible(h, target_sum):
    """Project h to [0,1] with sum = target_sum."""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference among free elements
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
            # Second pass
            diff2 = target_sum - h.sum()
            free2 = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free2.sum() > 0 and abs(diff2) > 1e-10:
                h[free2] += diff2 / free2.sum()
                h = np.clip(h, 0.0, 1.0)
    return h


def lbfgsb_smooth_max(h, beta, top_k, maxiter=500):
    """Optimize h using L-BFGS-B with smooth max objective."""
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        h_p = project_feasible(x, target)
        v, _, _ = smoothmax(h_p, beta, top_k)
        return v
    
    def objg(x):
        h_p = project_feasible(x, target)
        g = smoothmax_grad(h_p, beta, top_k)
        return g
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    res = minimize(obj, h, jac=objg, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
    h_out = project_feasible(res.x, target)
    return h_out


def slsqp_refine(h, maxiter=300, beta=200, top_k=10):
    """Refine h using SLSQP with equality constraint."""
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        v, _, _ = smoothmax(x, beta, top_k)
        return v
    
    def objg(x):
        return smoothmax_grad(x, beta, top_k)
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def interp_to_n(h, n_new, kind='cubic'):
    """Interpolate h to new resolution."""
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind=kind, fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def fourier_project(h, p):
    """Extract Fourier coefficients and reproject."""
    n = len(h)
    i = np.arange(n)
    # Compute DCT/STFT coefficients
    coeffs = np.zeros(2*p)
    for k in range(1, p+1):
        coeffs[2*(k-1)] = 2.0/n * np.sum(h * np.cos(2*np.pi*k*i/n))
        coeffs[2*(k-1)+1] = 2.0/n * np.sum(h * np.sin(2*np.pi*k*i/n))
    # Reconstruct
    h_new = np.full(n, np.mean(h))
    for k in range(1, p+1):
        h_new += coeffs[2*(k-1)] * np.cos(2*np.pi*k*i/n)
        h_new += coeffs[2*(k-1)+1] * np.sin(2*np.pi*k*i/n)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n/2.0)

def binary_projection(h, threshold=0.5):
    """Project towards binary and fix sum."""
    n = len(h)
    target = n / 2.0
    h_bin = (h >= threshold).astype(float)
    # Adjust to get exactly n/2 ones
    n_ones = h_bin.sum()
    if n_ones > target:
        # Turn off some 1s (those closest to threshold from above)
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])  # closest to threshold first
        n_to_flip = int(n_ones - target)
        for idx in order[:n_to_flip]:
            h_bin[idx] = 0
    elif n_ones < target:
        # Turn on some 0s
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])  # closest to threshold first
        n_to_flip = int(target - n_ones)
        for idx in order[:n_to_flip]:
            h_bin[idx] = 1
    return h_bin

        if remaining_time(t0, budget_s) > 30:
            h_try = lbfgsb_run(h_best, 300, 15, 400)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                h_best, best_c5 = h_try, c_try

    # Phase 6: Very high beta final polish
    if remaining_time(t0, budget_s) > 20:
        h_try = lbfgsb_run(h_best, 800, 20, 500)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            h_best, best_c5 = h_try, c_try

    # Phase 7: Iterate between resolutions
    if remaining_time(t0, budget_s) > 50:
        n_hi = len(h_best) + 100
        h_up = interp_to_n(h_best, n_hi)
        h_up = lbfgsb_run(h_up, 100, 10, 300)
        c_up = compute_c5(h_up)
        if c_up < best_c5:
            h_best, best_c5 = h_up, c_up
            n_hi = len(h_best)
        else:
            n_hi = len(h_best)
        h_dn = interp_to_n(h_best, max(100, n_hi // 2))
        h_dn = lbfgsb_run(h_dn, 80, 8, 200)
        c_dn = compute_c5(h_dn)
        if c_dn < best_c5:
            h_best, best_c5 = h_dn, c_dn
            h_best = interp_to_n(h_best, n_hi)
            h_best = lbfgsb_run(h_best, 200, 12, 300)
            c_re = compute_c5(h_best)
            if c_re < best_c5:
                best_c5 = c_re

    return h_best, best_c5, len(h_best)
