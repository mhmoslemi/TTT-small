    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break

            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)

def coordinate_descent(h, n_iter, target_sum, beta=100, n_active=5):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_val = smooth_max(np.correlate(best, 1-best, mode='full'), beta) * dx

        corr = np.correlate(best, 1-best, mode='full')
        wk = np.exp(beta * (corr - corr.max()))
        wk /= wk.sum()

        # For each lag, grad contribution to h[i]
        # corr[k] = sum_i h[i]*(1-h[i+k]) for valid i

                continue
            lag = lag_idx - (n - 1)

                grad += w * (1 - 2*best)
            elif lag > 0:

                        contrib += (1.0 - best[j + lag])
                    if j - lag >= 0:
                        contrib -= best[j - lag]

                    if j - ll >= 0:
                        contrib += (1.0 - best[j - ll])
                    if j + ll < n:
                        contrib -= best[j + ll]
                    grad[j] += w * contrib
        
        # Scale by dx
        grad *= dx
        
        # Find active indices (those with largest gradient)
        # Do coordinate descent: for each active coordinate, find optimal value
        active_idx = np.argsort(np.abs(grad))[::-1][:n_active]
        
        improved = False
        for idx in active_idx:
            if grad[idx] == 0:
                continue

            # subject to sum constraint
            old_val = best[idx]

            # Actually, let's do a 1D search along the direction that changes h[idx]
            # and compensates in a "free" coordinate.

            if free_mask.sum() == 0:
                continue
            comp_idx = np.argmax(free_mask * np.random.rand(n))

            for delta in np.linspace(-0.1, 0.1, 21):

                new_h[idx] += delta
                new_h[comp_idx] -= delta
                new_h = project_h(new_h, target_sum)
                obj = smooth_max(np.correlate(new_h, 1-new_h, mode='full'), beta) * dx
                if obj < best_obj - 1e-14:

                    best_delta = delta
            if best_delta != 0:
                best[idx] += best_delta
                best[comp_idx] -= best_delta
                best = project_h(best, target_sum)
                best_val = best_obj
                improved = True
        
        if not improved and it > 50:
            break
    
    return best, best_val

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            # Need to move a 0 to 1 or 1 to 0
            # Add to the closest to 1 from below, or closest to 0 from above
            if diff > 0:
                idx = np.argmin(h)
                h[idx] += min(diff, 1.0 - h[idx])
            else:
                idx = np.argmax(h)
                h[idx] += max(diff, -(h[idx]))
            h = np.clip(h, 0.0, 1.0)
            continue
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_smooth_max(h, beta):
    """Compute gradient of smooth_max(correlate(h, 1-h, 'full'), beta) * dx w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Weights for each lag
    wk = np.exp(beta * (corr - corr.max()))
    wk /= wk.sum()
    
    grad = np.zeros(n)
    for lag_idx in range(len(corr)):
        w = wk[lag_idx]
        if w < 1e-12:
            continue
        lag = lag_idx - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            # corr = sum_{i=0}^{n-lag-1} h[i] * (1-h[i+lag])
            # d/dh[j] = (1-h[j+lag]) for j < n-lag
            #            -h[j-lag] for j >= lag
            # Combined: for j in [0, n-1]
            #   if j < n-lag: +(1-h[j+lag])
            #   if j >= lag:  -h[j-lag]
            # Use vectorized form
            # Part 1: j in [0, n-lag-1], contribution (1-h[j+lag])
            grad[:n-lag] += w * (1.0 - h[lag:])
            # Part 2: j in [lag, n-1], contribution -h[j-lag]
            grad[lag:] -= w * h[:n-lag]
        else:
            ll = -lag
            # corr = sum_{i=ll}^{n-1} h[i] * (1-h[i-ll])
            #       = sum_{i=0}^{n-ll-1} h[i+ll] * (1-h[i])
            # d/dh[j]:
            #   if j >= ll: (1-h[j-ll]) [from h[j]*(1-h[j-ll])]
            #   if j < n-ll: -h[j+ll] [from h[j+ll]*(1-h[j])]
            grad[ll:] += w * (1.0 - h[:n-ll])
            grad[:n-ll] -= w * h[ll:]
    
    grad *= dx
    return grad

def subgradient_descent(h, target_sum, n_iter, beta=100, lr0=0.5, decay=0.999, momentum=0.5):
    n = len(h)
    cur = h.copy()
    best = h.copy()
    best_val = compute_c5(best)
    vel = np.zeros(n)
    lr = lr0
    
    for it in range(n_iter):
        grad = gradient_smooth_max(cur, beta)
        # Subgradient step with momentum
        vel = momentum * vel - lr * grad
        # Project velocity to be orthogonal to the sum constraint
        vel -= np.mean(vel)  # This makes sum(vel) = 0, preserving the sum
        cur = cur + vel
        cur = project_h(cur, target_sum)
        
        val = compute_c5(cur)
        if val < best_val:
            best = cur.copy()
            best_val = val
        
        lr *= decay
        if lr < 1e-6:
            break
    
    return best, best_val

def randomized_restart(h, target_sum, n_perturbations, sigma, beta):
    """Try random perturbations and keep the best"""
    n = len(h)
    best = h.copy()
    best_val = compute_c5(best)
    for _ in range(n_perturbations):
        pert = sigma * np.random.randn(n)
        pert -= pert.mean()  # Preserve sum
        trial = best + pert
        trial = project_h(trial, target_sum)
        val = compute_c5(trial)
        if val < best_val:
            best = trial.copy()
            best_val = val
    return best, best_val

def slsqp_optimize(h, target_sum, beta, maxiter=1000):
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(h_in):
        return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), beta) * dx
    
    def con_fun(x_in):
        return np.sum(x_in) - target_sum
    
    def con_jac(x_in):
        return np.ones(x_in.shape)
    
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = project_h(res.x, target_sum)
        return h_new
    except:
        return h

def coordinate_search_1d(h, idx, target_sum, beta):
    """Find optimal h[idx] by 1D search, adjusting another coordinate for the sum constraint."""
    n = len(h)
    dx = 2.0 / n
    best_val = np.inf
    best_v = h[idx]
    
    # We'll fix all other coordinates and optimize h[idx]
    # But we need to maintain the sum constraint, so we'll adjust another coordinate
    #Strategy: for each candidate h[idx], adjust the sum by spreading the difference
    #over all other free coordinates
    
    for v in np.linspace(0, 1, 50):
        h_trial = h.copy()
        h_trial[idx] = v
        # Adjust sum
        diff = target_sum - h_trial.sum()
        # Spread over free coordinates (excluding idx)
        free = np.ones(n, dtype=bool)
        free[idx] = False
        h_free = h_trial[free].copy()
        h_free += diff / free.sum()
        h_free = np.clip(h_free, 0, 1)
        h_trial[free] = h_free
        # Check if sum is satisfied
        if abs(h_trial.sum() - target_sum) > 1e-6:
            continue
        val = smooth_max(np.correlate(h_trial, 1-h_trial, 'full'), beta) * dx
        if val < best_val:
            best_val = val
            best_v = v
    return best_v, best_val

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            if diff > 0:
                idx = np.argmin(h)
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + diff)
            h = np.clip(h, 0.0, 1.0)
            continue
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    wk = np.exp(beta * (corr - corr.max()))
    wk /= wk.sum()
    grad = np.zeros(n)
    for lag_idx in range(len(corr)):
        w = wk[lag_idx]
        if w < 1e-12:
            continue
        lag = lag_idx - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n - lag]
        else:
            ll = -lag
            grad[ll:] += w * (1.0 - h[:n - ll])
            grad[:n - ll] -= w * h[ll:]
    return grad * dx


def nesterov_subgrad(h, target_sum, n_iter, beta=200, lr0=0.4, decay=0.9998, mu=0.9):
    n = len(h)
    best = h.copy()
    best_val = compute_c5(best)
    y = h.copy()
    t = 1.0
    lr = lr0
    for it in range(n_iter):
        grad = grad_smooth_max(y, beta)
        grad -= np.mean(grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_m = (t_new - 1.0) / t_new
        y = project_h(y + beta_m * (y - h), target_sum)
        h_new = project_h(y - lr * grad, target_sum)
        val = compute_c5(h_new)
        if val < best_val:
            best = h_new.copy()
            best_val = val
        h = h_new
        lr *= decay
        if lr < 1e-7:
            break
    return best, best_val


def slsqp_opt(h, target_sum, beta, maxiter=1500):
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj(x):
        return smooth_max(np.correlate(x, 1.0 - x, mode="full"), beta) * dx

    def cf(x):
        return np.sum(x) - target_sum

    def cj(x):
        return np.ones(x.shape)

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_h(res.x, target_sum)
    except Exception:
        return h


def sa_perturb(h, target_sum, n_trials, sigma, beta):
    n = len(h)
    best = h.copy()
    best_val = compute_c5(best)
    for _ in range(n_trials):
        p = sigma * np.random.randn(n)
        p -= p.mean()
        trial = project_h(best + p, target_sum)
        v = compute_c5(trial)
        if v < best_val:
            best = trial
            best_val = v
    return best, best_val

        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, nn)
        h_up = project_h(np.interp(x_new, x_old, best_h), tgt_n)
        h_bin = project_h((best_h > 0.5).astype(float), tgt_n) if len(best_h) != nn else project_h((best_h > 0.5).astype(float).copy(), tgt_n)
        # Resample binary from best_h
        h_bin = project_h(np.interp(x_new, x_old, (best_h > 0.5).astype(float)), tgt_n)

        # Structured init: low-frequency cosine
        h_cos = 0.5 + 0.35 * np.cos(2 * np.pi * 3 * np.arange(nn) / nn)
        h_cos = project_h(h_cos, tgt_n)

        inits = [h_up, h_bin, h_cos]

        for hi in inits:
            if tl() < 30:
                break
            hc = hi.copy()
            cc = compute_c5(hc)
            if cc < best_c5:
                best_h, best_c5 = hc.copy(), cc

            for b in [30, 60, 120, 250, 500]:
                if tl() < 20:
                    break
                hn = slsqp_opt(hc, tgt_n, b, 1000)
                c = compute_c5(hn)
                if c < cc - 1e-14:
                    hc = hn
                    cc = c
                if cc < best_c5:
                    best_h, best_c5 = hc.copy(), cc

            if tl() > 40:
                hn, c = nesterov_subgrad(hc, tgt_n, 4000, beta=150, lr0=0.15, decay=0.99985, mu=0.9)
                if c < best_c5:
                    best_h, best_c5 = hn, c

            if tl() > 20:
                hn, c = sa_perturb(hc, tgt_n, 100, 0.03, 500)
                if c < best_c5:
                    best_h, best_c5 = hn, c

    # Phase 3: final polish
    tf = len(best_h) / 2.0
    if tl() > 30:
        for b in [300, 600, 1000]:
            if tl() < 15:
                break
            hn = slsqp_opt(best_h, tf, b, 3000)
            c = compute_c5(hn)
            if c < best_c5:
                best_h, best_c5 = hn, c

    if tl() > 20:
        hn, c = nesterov_subgrad(best_h, tf, 8000, beta=800, lr0=0.03, decay=0.99995, mu=0.95)
        if c < best_c5:
            best_h, best_c5 = hn, c

    if tl() > 15:
        hn, c = sa_perturb(best_h, tf, 500, 0.005, 1000)
        if c < best_c5:
            best_h, best_c5 = hn, c

    # Final SLSQP squeeze
    if tl() > 10:
        hn = slsqp_opt(best_h, tf, 1500, 5000)
        c = compute_c5(hn)
        if c < best_c5:
            best_h, best_c5 = hn, c

    return best_h, best_c5, len(best_h)
