import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 0.0) & (h < 1.0)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def grad_lag(h, n, shift):
    """Gradient of correlation at given shift w.r.t. h."""
    g = np.zeros(n)
    # h[i] appears as h[m] with m=i: valid if 0 <= i+shift < n
    valid_m = (np.arange(n) + shift >= 0) & (np.arange(n) + shift < n)
    g[valid_m] += (1.0 - h[valid_m + shift])
    # h[i] appears as h[m+shift] with m=i-shift: valid if 0 <= i-shift < n
    valid_p = (np.arange(n) - shift >= 0) & (np.arange(n) - shift < n)
    g[valid_p] += (-h[valid_p - shift])
    return g

def objective_smooth(h, n, target_sum, beta, lam):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * (2.0 / n) + lam * (h.sum() - target_sum) ** 2

    best_c5 = compute_c5(best_h)

    for beta, lam in [(20, 10), (50, 30), (100, 60), (300, 150), (600, 300)]:

                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': 500, 'ftol': 1e-15})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:

    # Phase 2: Direct gradient descent on worst lag(s)
    # Use multi-lag weighting

            if c_try < best_c5:
                best_h = c_try  # BUG: should be h_try

        a[vb] += -h[i - shifts[vb]]
        B = c_cur - h[i] * a
        tg = np.linspace(0.0, 1.0, num_grid)
        matrix = B[:, None] + a[:, None] * tg[None, :]
        vals = np.max(matrix, axis=0)
        bi = int(np.argmin(vals))
        best_t = tg[bi]
        lo = tg[max(0, bi - 1)]
        hi = tg[min(num_grid - 1, bi + 1)]
        tf = np.linspace(lo, hi, 50)
        vf = np.max(B[:, None] + a[:, None] * tf[None, :], axis=0)
        fi = int(np.argmin(vf))
        if vf[fi] <= vals[bi]:
            best_t = tf[fi]
        h[i] = best_t
    h = project_box_sum(h, ts)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    else:
        n = 70
        h_init = np.full(n, 0.5)

    ts = n / 2.0
    bnds = [(0.0, 1.0)] * n
    best_h = project_box_sum(h_init, ts)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with smooth max, increasing beta
    for beta, lam in [(20, 10), (60, 40), (150, 80), (400, 200), (800, 400)]:
        if budget_s - (time.time() - t0) < 40:
            break
        try:
            r = minimize(obj_fun, best_h, args=(n, ts, beta, lam),
                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': 600, 'ftol': 1e-15})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # Phase 2: Coordinate descent
    for _ in range(40):
        if budget_s - (time.time() - t0) < 30:
            break
        h_new = coord_sweep(best_h, n, ts)
        c = compute_c5(h_new)
        if c < best_c5:
            best_h, best_c5 = h_new, c

    # Phase 3: Alternating L-BFGS-B + coord descent with perturbations
    it = 0
    while budget_s - (time.time() - t0) > 100:
        it += 1
        # Coordinate descent
        h_try = coord_sweep(best_h, n, ts)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            best_h, best_c5 = h_try, c_try

        # L-BFGS-B polish
        if budget_s - (time.time() - t0) > 50:
            beta = 400 if it < 15 else 1000
            try:
                r = minimize(obj_fun, best_h, args=(n, ts, beta, 400.0),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 400, 'ftol': 1e-15})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

        # Periodic perturbation with simulated annealing
        if it % 4 == 0 and budget_s - (time.time() - t0) > 120:
            frac = (budget_s - (time.time() - t0)) / budget_s
            scale = 0.04 * max(0.15, frac)
            h_pert = project_box_sum(best_h + np.random.randn(n) * scale, ts)
            c_pert = compute_c5(h_pert)
            # Accept if better, or with small probability (annealing)
            if c_pert < best_c5:
                best_h, best_c5 = h_pert, c_pert
            elif np.random.rand() < 0.05 * frac:
                best_h = h_pert
            # Try coord descent from perturbed point
            h_cd = coord_sweep(h_pert, n, ts)
            c_cd = compute_c5(h_cd)
            if c_cd < best_c5:
                best_h, best_c5 = h_cd, c_cd

        # Structured perturbation: target worst lag
        if it % 7 == 0 and budget_s - (time.time() - t0) > 150:
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            k_star = int(np.argmax(corr)) - (n - 1)
            h_sg = best_h.copy()
            for j in range(n):
                gval = 0.0
                if 0 <= j + k_star < n:
                    gval += 1.0 - h_sg[j + k_star]
                if 0 <= j - k_star < n:
                    gval += -h_sg[j - k_star]
                h_sg[j] -= 0.02 * gval
            h_sg = project_box_sum(h_sg, ts)
            h_cd = coord_sweep(h_sg, n, ts)
            c_cd = compute_c5(h_cd)
            if c_cd < best_c5:
                best_h, best_c5 = h_cd, c_cd

    # Phase 4: Final high-precision polish
    if budget_s - (time.time() - t0) > 25:
        for beta in [1500, 3000]:
            if budget_s - (time.time() - t0) < 14:
                break
            try:
                r = minimize(obj_fun, best_h, args=(n, ts, beta, 800.0),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 5000, 'ftol': 1e-17})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    # Final coordinate descent
    h_new = coord_sweep(best_h, n, ts)
    c = compute_c5(h_new)
    if c < best_c5:
        best_h, best_c5 = h_new, c

    best_h = project_box_sum(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
