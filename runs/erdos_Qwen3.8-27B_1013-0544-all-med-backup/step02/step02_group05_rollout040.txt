        # Compute base and coeff for each lag
        base = np.empty(2*n-1)
        coeff = np.empty(2*n-1)
        
        for k in range(2*n-1):
            shift = k - (n-1)
            i_lo = max(0, shift)
            i_hi = min(n-1, n-1+shift)
            
            c = 0.0

- Total: O(n^2) + n*O(n) = O(n^2)

That's very efficient!

    dx = 2.0 / n
    corr = compute_corr(h).copy()

            coeff = np.zeros(2*n-1)
            for k in range(2*n-1):
                shift = k - (n-1)
                i_lo = max(0, shift)
                i_hi = min(n-1, n-1+shift)
                
                c = 0.0
                # h[j] as first factor (i=j in the sum):
                if i_lo <= j <= i_hi:
                    if 0 <= j-shift <= n-1:
                        c += (1.0 - h[j-shift])
                    else:
                        c += 1.0
                
                # h[j] as second factor (i=j+shift):
                i2 = j + shift
                if 0 <= i2 <= n-1 and i_lo <= i2 <= i_hi:
                    c -= h[i2]
                
                coeff[k] = c
            
            # base = corr - h[j] * coeff
            base = corr - h[j] * coeff
            
            # Minimize max_k (base[k] + coeff[k] * x) for x in [0,1]
            # Use 21-point grid + refinement
            x_grid = np.linspace(0, 1, 21)
            obj_vals = np.max(base + coeff * x_grid[:, None], axis=1)
            x_best = x_grid[np.argmin(obj_vals)]
            
            # Refine
            lo = max(0.0, x_best - 0.05)
            hi = min(1.0, x_best + 0.05)
            x_fine = np.linspace(lo, hi, 21)
            obj_fine = np.max(base + coeff * x_fine[:, None], axis=1)
            x_best = x_fine[np.argmin(obj_fine)]
            
            # Update
            delta = x_best - h[j]
            if abs(delta) > 1e-12:
                h[j] = x_best
                corr += delta * coeff
        
        # Project to maintain sum constraint
        h = project_h(h, target_sum)
        # Recompute corr after projection
        corr = compute_corr(h)
    
    return h

    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

                    c += 1.0

import numpy as np
from scipy.optimize import minimize, Bounds
import time

_TARGET_SUM = 0.0
_N_POINTS = 0
_BETA_OBJ = 100.0
_DX_OBJ = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def sum_eq_fun(x):
    return x.sum() - _TARGET_SUM

def sum_eq_jac(x):
    return np.ones(_N_POINTS)

def obj_smooth(x):
    c = compute_corr(x)
    return smooth_max(c, _BETA_OBJ) * _DX_OBJ

def cd_sweep(h, corr, order=None):
    n = len(h)
    if order is None:
        order = list(range(n))
    for j in order:
        coeff = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            shift = k - (n - 1)
            i_lo = max(0, shift)
            i_hi = min(n - 1, n - 1 + shift)
            c = 0.0
            if i_lo <= j <= i_hi:
                if 0 <= j - shift <= n - 1:
                    c += (1.0 - h[j - shift])
                else:
                    c += 1.0
            i2 = j + shift
            if 0 <= i2 <= n - 1 and i_lo <= i2 <= i_hi:
                c -= h[i2]
            coeff[k] = c
        base = corr - h[j] * coeff
        xg = np.linspace(0, 1, 33)
        vals = np.max(base[None, :] + coeff[None, :] * xg[:, None], axis=1)
        xb = xg[np.argmin(vals)]
        lo = max(0.0, xb - 0.03)
        hi = min(1.0, xb + 0.03)
        xf = np.linspace(lo, hi, 33)
        vals_f = np.max(base[None, :] + coeff[None, :] * xf[:, None], axis=1)
        xb = xf[np.argmin(vals_f)]
        delta = xb - h[j]
        if abs(delta) > 1e-14:
            h[j] = xb
            corr += delta * coeff
    return h, corr

    _TARGET_SUM = target
    _N_POINTS = n
    _BETA_OBJ = float(beta)
    _DX_OBJ = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    try:
        res = minimize(obj_smooth, h, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(res.x, target)
    except Exception:
        return h
