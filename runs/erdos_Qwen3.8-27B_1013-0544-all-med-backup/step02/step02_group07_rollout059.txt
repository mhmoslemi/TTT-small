def project_h(h, target_sum):

def coord_descent_pass(h, n, n_passes=1, lam=0.0):

            valid_minus = (i - d_vals >= 0) & (i - d_vals < n)

            a[valid_minus] -= h[i - d_vals[valid_minus]]

    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()

        if abs(diff) < 1e-14:
            break

        if free.sum() > 0:

        else:

    target_sum = n / 2.0
    for _ in range(n_passes):
        corr = np.correlate(h, 1.0 - h, mode="full")
        for i in range(n):
            valid_plus = (i + d_vals >= 0) & (i + d_vals < n)
            valid_minus = (i - d_vals >= 0) & (i - d_vals < n)
            a = np.zeros(2 * n - 1)

            a[valid_minus] -= h[i - d_vals[valid_minus]]

            M_high = np.max(b + np.clip(a, 0, 1))
