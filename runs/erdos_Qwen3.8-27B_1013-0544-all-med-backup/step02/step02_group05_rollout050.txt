import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_binary(h, target_sum):
    h = np.round(np.clip(h, 0, 1))
    s = int(h.sum())
    if s > target_sum:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:s - target_sum]] = 0
    elif s < target_sum:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target_sum - s]] = 1
    return h

def binary_sa(h, n, target_sum, max_iters, seed=42, T0=0.004, T1=1e-7, alpha=0.9999):
    rng = np.random.RandomState(seed)
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_curr = corr.max() * dx
    c5_best = c5_curr
    h_best = h.copy()
    T = T0
    idx_1 = list(np.where(h == 1)[0])
    idx_0 = list(np.where(h == 0)[0])
    for it in range(max_iters):
        if T < T1:
            break
        i = idx_1[rng.randint(len(idx_1))]
        j = idx_0[rng.randint(len(idx_0))]
        if i == j:
            continue
        h[i] = 0
        h[j] = 1
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5_new = corr.max() * dx
        delta = c5_new - c5_curr
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c5_curr = c5_new
            idx_1.remove(i)
            idx_1.append(j)
            idx_0.remove(j)
            idx_0.append(i)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
        else:
            h[i] = 1
            h[j] = 0
        T *= alpha
    return h_best, c5_best

def slsqp_refine(h0, n, target_sum, beta, maxiter=800, ftol=1e-14):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max(corr, float(beta)) * dx
    def con_fun(x):
        return np.sum(x) - target_sum
    def con_jac(x):
        return np.ones(n)
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': maxiter, 'ftol': ftol})
    h_new = np.clip(res.x, 0, 1)
    h_new += (target_sum - h_new.sum()) / n
    h_new = np.clip(h_new, 0, 1)
    return h_new

                h_try = np.clip(h_try, 0, 1)
                try:
                    h_new = slsqp_refine(h_try, n_points, float(target_sum), 150, maxiter=400)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h, best_c5, best_n = h_new.copy(), c5, n_points
                        h_cont = h_new
                except Exception:
                    pass

    # Final high-precision polish
    if best_h is not None and time.time() - start_time < budget_s - 8:
        n_f = best_n
        ts = float(n_f / 2)
        dx_f = 2.0 / n_f
        for beta in [600, 900, 1200]:
            if time.time() - start_time > budget_s - 4:
                break
            try:
                h_new = slsqp_refine(best_h, n_f, ts, beta, maxiter=2000, ftol=1e-16)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    if best_h is None:
        best_h = h_parent.copy()
        best_c5 = compute_c5(best_h)
        best_n = n_parent
    return best_h, best_c5, best_n
