    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def vec_subgrad(h, top_m=4, temp=5.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[::-1][:top_m]
    w = np.exp(temp * (corr[idx] - corr[idx[0]]))
    w /= w.sum()
    grad = np.zeros(n)
    offset = n - 1
    for wi, li in zip(w, idx):
        k = int(li) - offset
        ak = abs(k)
        if ak == 0:
            grad += wi * dx * (1.0 - 2.0 * h)
        elif k > 0:
            g = np.zeros(n)
            g[:n - ak] = 1.0 - h[ak:]
            g[ak:] -= h[:n - ak]
            grad += wi * dx * g
        else:
            p = -k
            g = np.zeros(n)
            g[p:] = 1.0 - h[:n - p]
            g[:n - p] -= h[p:]
            grad += wi * dx * g
    return grad

def anneal_subgrad(h, c5, target_sum, t0, budget_s, n_steps=10000, alpha0=0.1):
    if budget_s - (time.time() - t0) < 3:
        return h, c5
    for it in range(n_steps):
        if budget_s - (time.time() - t0) < 2:
            break
        frac = it / max(n_steps, 1)
        alpha = alpha0 * (0.002 / alpha0) ** frac
        m = 5 - int(2 * frac)
        tp = 4.0 + 6.0 * (1.0 - frac)
        g = vec_subgrad(h, top_m=max(m, 2), temp=tp)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        for _ in range(10):
            ht = project_h(h - alpha * g, target_sum)
            c5t = compute_c5(ht)
            if c5t < c5:
                h, c5 = ht, c5t
                break
            alpha *= 0.5
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        hp = np.array(initial_h_values, dtype=float).copy()
        nb = len(hp)
    else:
        nb = 140
        hp = np.full(nb, 0.5)

    best_h = hp.copy()
    best_c5 = compute_c5(best_h)
    best_n = nb

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP at parent resolution with increasing beta
    ts = nb / 2.0
    dx = 2.0 / nb
    cur = hp.copy()
    cur_c5 = best_c5

    for beta in [20, 50, 100, 200, 400, 800, 1500, 3000]:
        if tl() < 15:
            break
        b_val = float(beta)
        d_val = dx
        t_val = ts
        def obj(h_in):
            return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), b_val) * d_val
        def cf(x):
            return np.sum(x) - t_val
        def cj(x):
            return np.ones(x.shape)
        try:
            r = minimize(obj, cur, method='SLSQP',
                         bounds=[(0.0, 1.0)] * nb,
                         constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                         options={'maxiter': 2000, 'ftol': 1e-15})
            hn = project_h(r.x, ts)
            c5n = compute_c5(hn)
            if c5n < cur_c5:
                cur, cur_c5 = hn, c5n
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, nb
        except Exception:
            pass

    # Phase 2: Annealed subgradient at parent resolution
    cur, cur_c5 = anneal_subgrad(cur, cur_c5, ts, t0, budget_s, n_steps=12000, alpha0=0.12)
    if cur_c5 < best_c5:
        best_h, best_c5, best_n = cur.copy(), cur_c5, nb

    # Phase 3: Try a finer resolution
    if tl() > 120:
        n2 = nb * 2
        ts2 = n2 / 2.0
        dx2 = 2.0 / n2
        xo = np.linspace(0, 2, nb, endpoint=False)
        xn = np.linspace(0, 2, n2, endpoint=False)
        h2 = project_h(np.interp(xn, xo, cur), ts2)
        c5_2 = compute_c5(h2)

        for beta in [50, 100, 200, 500, 1000]:
            if tl() < 20:
                break
            b_val = float(beta)
            d_val = dx2
            t_val = ts2
            def obj2(h_in):
                return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), b_val) * d_val
            def cf2(x):
                return np.sum(x) - t_val
            def cj2(x):
                return np.ones(x.shape)
            try:
                r = minimize(obj2, h2, method='SLSQP',
                             bounds=[(0.0, 1.0)] * n2,
                             constraints=[{'type': 'eq', 'fun': cf2, 'jac': cj2}],
                             options={'maxiter': 1000, 'ftol': 1e-14})
                hn2 = project_h(r.x, ts2)
                c5n2 = compute_c5(hn2)
                if c5n2 < c5_2:
                    h2, c5_2 = hn2, c5n2
                if c5_2 < best_c5:
                    best_h, best_c5, best_n = h2.copy(), c5_2, n2
            except Exception:
                pass

        h2, c5_2 = anneal_subgrad(h2, c5_2, ts2, t0, budget_s, n_steps=6000, alpha0=0.08)
        if c5_2 < best_c5:
            best_h, best_c5, best_n = h2.copy(), c5_2, n2

    # Phase 4: Try a lower resolution
    if tl() > 60:
        n3 = max(80, nb // 2)
        if n3 != nb and n3 != n2:
            ts3 = n3 / 2.0
            dx3 = 2.0 / n3
            xo = np.linspace(0, 2, nb, endpoint=False)
            xn = np.linspace(0, 2, n3, endpoint=False)
            h3 = project_h(np.interp(xn, xo, best_h), ts3)
            c5_3 = compute_c5(h3)
            for beta in [50, 200, 500]:
                if tl() < 15:
                    break
                b_val = float(beta)
                d_val = dx3
                t_val = ts3
                def obj3(h_in):
                    return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), b_val) * d_val
                def cf3(x):
                    return np.sum(x) - t_val
                def cj3(x):
                    return np.ones(x.shape)
                try:
                    r = minimize(obj3, h3, method='SLSQP',
                                 bounds=[(0.0, 1.0)] * n3,
                                 constraints=[{'type': 'eq', 'fun': cf3, 'jac': cj3}],
                                 options={'maxiter': 1000, 'ftol': 1e-14})
                    hn3 = project_h(r.x, ts3)
                    c5n3 = compute_c5(hn3)
                    if c5n3 < c5_3:
                        h3, c5_3 = hn3, c5n3
                    if c5_3 < best_c5:
                        best_h, best_c5, best_n = h3.copy(), c5_3, n3
                except Exception:
                    pass
            h3, c5_3 = anneal_subgrad(h3, c5_3, ts3, t0, budget_s, n_steps=4000, alpha0=0.1)
            if c5_3 < best_c5:
                best_h, best_c5, best_n = h3.copy(), c5_3, n3

    # Phase 5: Final high-precision polish on best solution
    if tl() > 10:
        nf = len(best_h)
        tf = nf / 2.0
        dxf = 2.0 / nf
        for beta in [3000, 8000]:
            if tl() < 5:
                break
            b_val = float(beta)
            d_val = dxf
            t_val = tf
            def objf(h_in):
                return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), b_val) * d_val
            def cff(x):
                return np.sum(x) - t_val
            def cjf(x):
                return np.ones(x.shape)
            try:
                r = minimize(objf, best_h, method='SLSQP',
                             bounds=[(0.0, 1.0)] * nf,
                             constraints=[{'type': 'eq', 'fun': cff, 'jac': cjf}],
                             options={'maxiter': 10000, 'ftol': 1e-16})
                hn = project_h(r.x, tf)
                c5n = compute_c5(hn)
                if c5n < best_c5:
                    best_h, best_c5 = hn, c5n
            except Exception:
                pass

        # Final subgradient pass
        best_h, best_c5 = anneal_subgrad(best_h, best_c5, tf, t0, budget_s,
                                         n_steps=8000, alpha0=0.03)

    return best_h, best_c5, best_n
