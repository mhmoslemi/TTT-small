```python
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def subgrad_for_lag(h, lag_idx, n):
    """Subgradient of overlap at a specific lag index in the full correlation."""
    # lag_idx is index in correlate output (0 to 2n-2)
    # actual lag = lag_idx - (n-1)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        # overlap = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
        # d/dh[i] for i in [0, n-1-lag]: (1-h[i+lag])
        # d/dh[i+lag] for i in [0, n-1-lag]: -h[i]
        for i in range(n - lag):
            grad[i] += (1.0 - h[i + lag])
            grad[i + lag] -= h[i]
    else:

        # overlap = sum_{i=la}^{n-1} h[i]*(1-h[i-la])

        for i in range(la, n):
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def subgrad_vec(h, lag, n):
    grad = np.zeros(n)
    k = abs(lag)
    if k >= n:
        return grad
    if lag >= 0:
        grad[k:] += (1.0 - h[:n - k])
        grad[:n - k] += -h[k:]
    else:
        grad[k:] += (1.0 - h[:n - k])
        grad[:n - k] += -h[k:]
    return grad


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def con_fun(h, target_sum):
    return h.sum() - target_sum


def con_jac(h, target_sum):
    return np.ones(len(h))


def make_anti_periodic(n, mask):
    h = np.zeros(n)
    half = n // 2
    h[:half] = mask[:half].copy()
    h[half:2 * half] = 1.0 - mask[:half]
    if n > 2 * half:
        h[2 * half:] = 0.5
    return h


def local_search(h, target_sum, n, dx, iters, remaining_time_fn):
    best = h.copy()
    best_c5 = compute_c5(best)
    for it in range(iters):
        if remaining_time_fn() < 5:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        delta = 0.1 * np.random.randn()
        h_try = best.copy()
        h_try[i] += delta
        h_try[j] -= delta
        h_try = project_feasible(h_try, target_sum)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best = h_try.copy()
            best_c5 = c5_try
    return best, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))

    best_h = project_feasible(h_init.copy(), target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with smooth max continuation
    for beta in [3, 10, 40, 150, 500, 2000]:
        if remaining() < 90:
            break
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
        for trial in range(4):
            if remaining() < 35:
                break
            x0 = best_h.copy()
            if trial > 0:
                perturb = 0.015 / trial
                x0 = project_feasible(x0 + np.random.randn(n) * perturb, target_sum)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 6000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 2: Nesterov subgradient with top-K lags
    if remaining() > 150:
        h = best_h.copy()
        v = np.zeros(n)
        mu = 0.85
        best_local = compute_c5(h)
        for it in range(80000):
            if remaining() < 50:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            K = 3 + (it // 10000)
            K = min(K, 8)
            top_k = np.argsort(corr)[-K:]
            g = np.zeros(n)
            for k_idx in top_k:
                lag = k_idx - (n - 1)
                g += subgrad_vec(h, lag, n)
            g /= K
            v = mu * v + (1.0 - mu) * g
            lr = 0.006 / (1.0 + 0.0001 * it)
            h_new = project_feasible(h - lr * v, target_sum)
            h = h_new
            if it % 2000 == 1999:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
                if c5 < best_local:
                    best_local = c5
                    mu = min(0.95, mu + 0.01)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 3: Anti-periodic exploration
    if remaining() > 100:
        half = n // 2
        for attempt in range(5):
            if remaining() < 20:
                break
            mask = np.zeros(half)
            if attempt == 0:
                mask[:half // 2] = 1.0
            elif attempt == 1:
                mask[::2] = 1.0
            else:
                mask[np.random.choice(half, half // 2, replace=False)] = 1.0
            h_ap = make_anti_periodic(n, mask)
            h_ap = project_feasible(h_ap, target_sum)
            cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
            for beta in [20, 100, 500]:
                if remaining() < 10:
                    break
                try:
                    res = minimize(obj_smooth, h_ap, args=(dx, beta), method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 3000, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                        h_ap = h_new.copy()
                except Exception:
                    pass

    # Phase 4: Multi-resolution optimization
    if remaining() > 120:
        for n_try in [n // 2, n * 3 // 2, n + 14, n - 14, n + 40]:
            if n_try < 20 or remaining() < 50:
                continue
            h_c = resample_h(best_h, n_try)
            ts = n_try / 2.0
            h_c = project_feasible(h_c, ts)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
            dx_t = 2.0 / n_try
            for beta in [30, 200, 1000]:
                if remaining() < 20:
                    break
                try:
                    res = minimize(obj_smooth, h_c, args=(dx_t, beta), method='SLSQP',
                                   bounds=bnds_t, constraints=cons_t,
                                   options={'maxiter': 4000, 'ftol': 1e-14})
                    h_back = resample_h(project_feasible(res.x, ts), n)
                    h_back = project_feasible(h_back, target_sum)
                    c5 = compute_c5(h_back)
                    if c5 < best_c5:
                        best_h = h_back.copy()
                        best_c5 = c5
                except Exception:
                    pass

    # Phase 5: Local search (coordinate swaps)
    if remaining() > 40:
        best_h, best_c5 = local_search(best_h, target_sum, n, dx, 5000, remaining)

    # Phase 6: Final high-beta polish
    if remaining() > 25:
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
        for beta in [5000, 15000, 50000]:
            if remaining() < 10:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 7: Final subgradient refine
    if remaining() > 15:
        h = best_h.copy()
        v = np.zeros(n)
        for it in range(40000):
            if remaining() < 5:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_worst = np.argmax(corr)
            lag = k_worst - (n - 1)
            g = subgrad_vec(h, lag, n)
            v = 0.92 * v + 0.08 * g
            lr = 0.002 / (1 + 0.0003 * it)
            h = project_feasible(h - lr * v, target_sum)
            if it % 3000 == 2999:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    return best_h, best_c5, len(best_h)
```
