corr[d] = sum_{n: valid(n,d)} h[n+d] * (1 - h[n])

where valid(n,d) means 0 <= n < N and 0 <= n+d < N.

Let S1 = sum_{n valid} h[n+d] and S2 = sum_{n valid} h[n+d]*h[n]

Valid if: 0 <= n < N and 0 <= n+d < N → 0 <= j-d < N and 0 <= j < N
So: ∂S1/∂h[j] = 1 if (0 <= j-d < N and 0 <= j < N), else 0.
Since 0 <= j < N is always true (j is an index), this is: 1 if 0 <= j-d < N, else 0.

∂S2/∂h[j]:
S2 = sum_n h[n+d]*h[n]
h[j] appears in two ways:
(a) h[j] = h[n+d] when n = j-d: contribution h[n] = h[j-d] (if n valid)
(b) h[j] = h[n] when n = j: contribution h[n+d] = h[j+d] (if j valid and j+d valid)

For (a): n = j-d valid means 0 <= j-d < N and 0 <= j < N → 0 <= j-d < N
Contribution: h[j-d]

For (b): n = j valid means 0 <= j < N (always) and 0 <= j+d < N
Contribution: h[j+d]

So: ∂S2/∂h[j] = h[j-d] * I(0 <= j-d < N) + h[j+d] * I(0 <= j+d < N)

Great, so for each j:
- If j-d is in [0, N): add (1 - h[j-d])
- If j+d is in [0, N): subtract h[j+d]

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta


def lag_gradient(h, d):
    n = len(h)
    grad = np.zeros(n)
    j_min = max(0, d)
    j_max = min(n - 1, n - 1 + d)
    if j_min <= j_max:
        idx = np.arange(j_min, j_max + 1)
        grad[idx] += (1.0 - h[idx - d])
    j_min2 = max(0, -d)
    j_max2 = min(n - 1, n - 1 - d)
    if j_min2 <= j_max2:
        idx2 = np.arange(j_min2, j_max2 + 1)
        grad[idx2] -= h[idx2 + d]
    return grad


def top_lags_gradient(h, k=3):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_indices = np.argsort(corr)[::-1][:k]
    grad = np.zeros(n)
    for ki in top_indices:
        d = int(ki) - (n - 1)
        grad += lag_gradient(h, d)
    return grad, corr


def subgradient_refine(h, target_sum, max_iters, t_start, t_budget, k_lags=3):
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    for it in range(max_iters):
        if time.time() - t_start > t_budget - 5:
            break
        grad, corr = top_lags_gradient(cur, k_lags)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        best_step = 0.0
        best_val = cur_c5
        for step_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            step = 0.06 * step_mult / (1.0 + 0.003 * it)
            trial = project_h(cur - step * grad, target_sum)
            c5_trial = compute_c5(trial)
            if c5_trial < best_val:
                best_val = c5_trial
                best_step = step
                cur = trial
        if best_step == 0.0:
            break
        if cur_c5 - best_val > 1e-12:
            cur_c5 = best_val
    return cur, cur_c5


def slsqp_optimize(h, target_sum, dx, n_points, t_start, t_budget, betas=None):
    if betas is None:
        betas = [30, 60, 120, 250, 500]
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    cur = h.copy()
    cur_c5 = compute_c5(cur)

    def obj(h_in):
        c = np.correlate(h_in, 1.0 - h_in, mode="full")
        return smooth_max(c, 40) * dx

    def con(x_in):
        return np.sum(x_in) - target_sum

    def con_jac(x_in):
        return np.ones(x_in.shape)

    constraints = [{'type': 'eq', 'fun': con, 'jac': con_jac}]

    for beta in betas:
        if time.time() - t_start > t_budget - 8:
            break

        def obj_b(h_in, b=beta):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            return smooth_max(c, b) * dx

        try:
            res = minimize(obj_b, cur, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 300, 'ftol': 1e-12})
            h_new = project_h(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                cur = h_new.copy()
                cur_c5 = c5_new
        except Exception:
            pass
    return cur, cur_c5


def coordinate_search(h, target_sum, t_start, t_budget, n_iter=5000):
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    rng = np.random.default_rng(12345)
    for it in range(n_iter):
        if time.time() - t_start > t_budget - 3:
            break
        j = rng.integers(n)
        base = cur[j]
        candidates = [0.0, 1.0, base + 0.05, base - 0.05, base + 0.1, base - 0.1,
                      base + 0.02, base - 0.02]
        best_val = cur_c5
        best_h = None
        for cv in candidates:
            cv = np.clip(cv, 0.0, 1.0)
            if abs(cv - base) < 1e-10:
                continue
            trial = cur.copy()
            trial[j] = cv
            trial = project_h(trial, target_sum)
            c5_t = compute_c5(trial)
            if c5_t < best_val:
                best_val = c5_t
                best_h = trial
        if best_h is not None:
            cur = best_h
            cur_c5 = best_val
    return cur, cur_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    t_end = t_start + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(100, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(h_parent)

    resolutions = []
    for n in [100, 120, 140, 160, 200, 250]:
        if n not in resolutions and 60 <= n <= 300:
            resolutions.append(n)

    for n_points in resolutions:
        if time.time() > t_end - 20:
            break
        target_sum = n_points / 2.0
        dx = 2.0 / n_points

        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_points)
        h_init = project_h(np.interp(x_new, x_old, best_h), target_sum)

        inits = [h_init]

        for (f1, a1, f2, a2) in [(1, 0.3, 3, 0.12), (2, 0.2, 5, 0.1)]:
            x = np.linspace(0, 2, n_points)
            s = 0.5 + a1 * np.sin(2 * np.pi * f1 * x / 2) + a2 * np.sin(2 * np.pi * f2 * x / 2)
            inits.append(project_h(s, target_sum))

        alt = np.where(np.arange(n_points) % 2 == 0, 0.6, 0.4)
        inits.append(project_h(alt, target_sum))

        for h_init in inits:
            if time.time() > t_end - 10:
                break

            # SLSQP phase
            cur, cur_c5 = slsqp_optimize(h_init, target_sum, dx, n_points, t_start, t_end)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_points

            # Subgradient phase
            cur2, cur2_c5 = subgradient_refine(cur, target_sum, 3000, t_start, t_end, k_lags=3)
            if cur2_c5 < best_c5:
                best_h, best_c5, best_n = cur2.copy(), cur2_c5, n_points

            # Coordinate search
            if time.time() < t_end - 10:
                cur3, cur3_c5 = coordinate_search(cur2, target_sum, t_start, t_end, n_iter=3000)
                if cur3_c5 < best_c5:
                    best_h, best_c5, best_n = cur3.copy(), cur3_c5, n_points

        # Second round of subgradient on best at this resolution
        if time.time() < t_end - 10:
            cur = best_h.copy()
            if len(cur) != n_points:
                x_b = np.linspace(0, 2, len(cur))
                x_n = np.linspace(0, 2, n_points)
                cur = project_h(np.interp(x_n, x_b, cur), target_sum)
            cur, cur_c5 = subgradient_refine(cur, target_sum, 5000, t_start, t_end, k_lags=2)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_points

    # Final high-precision SLSQP
    if time.time() < t_end - 5:
        n_f = len(best_h)
        t_f = n_f / 2.0
        dx_f = 2.0 / n_f
        best_h, best_c5 = slsqp_optimize(best_h, t_f, dx_f, n_f, t_start, t_end,
                                          betas=[200, 500, 1000])

    # Final subgradient polish
    if time.time() < t_end - 2:
        best_h, best_c5 = subgradient_refine(best_h, len(best_h) / 2.0,
                                              5000, t_start, t_end, k_lags=2)

    return best_h, best_c5, best_n
```
