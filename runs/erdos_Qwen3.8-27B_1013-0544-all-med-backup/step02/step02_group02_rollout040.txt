import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def full_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

class FastSA:
    def __init__(self, h, n):
        self.n = n
        self.h = h.copy()
        self.c = full_corr(h)
        self.dx = 2.0 / n
        self.c5 = self.c.max() * self.dx
        # Precompute pad for fast indexing
        self.h_pad = np.pad(h, (n-1, n-1))
    
    def propose_and_evaluate(self, i, j, delta):
        """Try h[i] += delta, h[j] -= delta. Return new c5 without modifying state."""
        n = self.n
        h = self.h
        lag = np.arange(-(n-1), n)  # 2n-1 values
        # Precompute indices into h_pad
        idx_i_plus = i + lag + n - 1
        idx_i_minus = i - lag + n - 1
        idx_j_plus = j + lag + n - 1
        idx_j_minus = j - lag + n - 1
        
        valid_i_first = (i + lag >= 0) & (i + lag < n)
        valid_i_second = (i - lag >= 0) & (i - lag < n) & (lag != 0)
        valid_j_first = (j + lag >= 0) & (j + lag < n)
        valid_j_second = (j - lag >= 0) & (j - lag < n) & (lag != 0)
        
        hp = self.h_pad
        dc = (delta * (1 - hp[idx_i_plus]) * valid_i_first.astype(np.float64)
             - delta * hp[idx_i_minus] * valid_i_second.astype(np.float64)
             - delta * (1 - hp[idx_j_plus]) * valid_j_first.astype(np.float64)
             + delta * hp[idx_j_minus] * valid_j_second.astype(np.float64))
        
        # Fix lag=0 (index n-1)
        dc[n-1] = delta * (2*h[j] - 2*h[i]) - 2*delta*delta
        
        new_c = self.c + dc
        return new_c.max() * self.dx, dc
    
    def accept(self, i, j, delta, dc):
        self.h[i] += delta
        self.h[j] -= delta
        self.c += dc
        self.c5 = self.c.max() * self.dx
        # Update h_pad
        self.h_pad[i + self.n - 1] = self.h[i]
        self.h_pad[j + self.n - 1] = self.h[j]
    
    def reject(self):
        pass  # nothing to do since we didn't modify state

def full_corr(h):

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

        ht = slsqp_optimize(best_h, n, alpha, 500, 1e-14)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = ht, c

    # Phase 1: Fast SA (main search)
    sa_iters = 5000000
    best_h, best_c5 = fast_sa(best_h, n, t0, budget_s, rng, sa_iters)

    # Phase 2: SLSQP polish
    for alpha in [50000, 200000, 500000]:
        if remaining() < 15:
            break
        ht = slsqp_optimize(best_h, n, alpha, 1000, 1e-16)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = ht, c

    # Phase 3: Second fast SA round (cooler)
    if remaining() > 120:
        best_h, best_c5 = fast_sa(best_h, n, t0, budget_s, rng, 3000000)
        for alpha in [200000, 500000]:
            if remaining() < 15:
                break
            ht = slsqp_optimize(best_h, n, alpha, 1000, 1e-16)
            c = compute_c5(ht, n)
            if c < best_c5 - 1e-15:
                best_h, best_c5 = ht, c

    # Phase 4: Fine 2-point optimization
    if remaining() > 15:
        best_h, best_c5 = fine_2pt(best_h, n, t0, budget_s)

    # Phase 5: Final SLSQP
    for alpha in [500000, 2000000]:
        if remaining() < 10:
            break
        ht = slsqp_optimize(best_h, n, alpha, 3000, 1e-18)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = ht, c

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
