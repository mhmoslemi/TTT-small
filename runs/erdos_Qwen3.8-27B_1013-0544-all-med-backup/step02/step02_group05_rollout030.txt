import numpy as np
from scipy.optimize import minimize, Bounds
import time

PARAMS = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def sm_gradient(corr, beta):
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -300, 300)
    ev = np.exp(sh)
    w = ev / np.sum(ev)
    val = m + np.log(np.sum(ev)) / beta
    return val, w

def obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    val, w = sm_gradient(corr, beta)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        wi = w[idx] * dx
        if abs(wi) < 1e-18:
            continue
        if k >= 0:
            grad[k:] += wi * (1.0 - h[:n - k])
            grad[:n - k] -= wi * h[k:]
        else:
            kp = -k
            grad[:n - kp] += wi * (1.0 - h[kp:])
            grad[kp:] -= wi * h[:n - kp]
    return val * dx, grad

def obj_func(h):
    return obj_grad(h, PARAMS['beta'])[0]

def grad_func(h):
    return obj_grad(h, PARAMS['beta'])[1]

def active_obj(h):
    c = compute_corr(h)
    t = c[PARAMS['top_lags']]
    m = np.max(t)
    sh = np.clip(300.0 * (t - m), -300, 300)
    ev = np.exp(sh)
    return (m + np.log(np.sum(ev)) / 300.0) * PARAMS['dx']

def constr_fun(x):
    return np.sum(x) - PARAMS['target_sum']

def constr_jac(x):
    return np.ones(len(x))

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    rng = np.random.RandomState(seed)

    n = 70
    ts = n / 2.0
    dx = 2.0 / n
    PARAMS['target_sum'] = ts
    PARAMS['dx'] = dx

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': constr_fun, 'jac': constr_jac}]

    h_init = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()

    starts = []
    if h_init is not None:
        if len(h_init) == n:
            starts.append(h_init.copy())
        else:
            xo = np.linspace(0, 2, len(h_init), endpoint=False)
            xn = np.linspace(0, 2, n, endpoint=False)
            starts.append(project_feasible(np.interp(xn, xo, h_init, period=2.0), ts))
        # Shifted version for diversity
        if len(h_init) == n:
            starts.append(np.roll(h_init.copy(), 1))
    starts.append(np.full(n, 0.5))

    best_h = None
    best_c5 = float('inf')

    for si in range(5):
        if time.time() - start > budget_s - 80:
            break
        if si < len(starts):
            h0 = starts[si].copy()
        else:
            h0 = project_feasible(0.5 + 0.3 * rng.randn(n), ts)

        h_best = h0.copy()
        c5_best = compute_c5(h0)

        # SLSQP with increasing smoothness parameter
        beta_schedule = [(25, 200), (50, 300), (100, 400), (200, 500), (400, 600)]
        for beta, maxit in beta_schedule:
            if time.time() - start > budget_s - 50:
                break
            PARAMS['beta'] = beta
            try:
                res = minimize(obj_func, h_best, method='SLSQP', bounds=bounds,
                              constraints=cons, jac=grad_func,
                              options={'maxiter': maxit, 'ftol': 1e-15})
                hn = project_feasible(res.x, ts)
                c5 = compute_c5(hn)
                if c5 < c5_best:
                    c5_best = c5
                    h_best = hn.copy()
            except Exception:
                pass

        # Active-set focused optimization
        if time.time() - start < budget_s - 25:
            corr = compute_corr(h_best)
            top_all = np.argsort(corr)[::-1]
            for tk in [3, 5, 8]:
                if time.time() - start > budget_s - 15:
                    break
                PARAMS['top_lags'] = top_all[:tk]
                try:
                    res = minimize(active_obj, h_best, method='SLSQP',
                                  bounds=bounds, constraints=cons,
                                  options={'maxiter': 400, 'ftol': 1e-14})
                    hn = project_feasible(res.x, ts)
                    c5 = compute_c5(hn)
                    if c5 < c5_best:
                        c5_best = c5
                        h_best = hn.copy()
                except Exception:
                    pass

        if c5_best < best_c5:
            best_h = h_best.copy()
            best_c5 = c5_best

    # Final high-precision polish with analytical gradient
    if best_h is not None and time.time() - start < budget_s - 5:
        for beta in [200, 400]:
            if time.time() - start > budget_s - 3:
                break
            PARAMS['beta'] = beta
            try:
                res = minimize(obj_func, best_h, method='SLSQP', bounds=bounds,
                              constraints=cons, jac=grad_func,
                              options={'maxiter': 1500, 'ftol': 1e-16})
                hf = project_feasible(res.x, ts)
                c5f = compute_c5(hf)
                if c5f < best_c5:
                    best_h = hf.copy()
                    best_c5 = c5f
            except Exception:
                pass

    return best_h, best_c5, n
