        # And derivative is -h[i+k]

        i_hi1 = min(n-1, hi + k)
        if i_lo1 <= i_hi1:
            idx = np.arange(i_lo1, i_hi1 + 1)
            grad[idx] += wk * (1.0 - h[idx - k])
        
        # Second term: i in [lo, hi]
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += wk * (-h[idx + k])
    
    grad *= dx
    return val, grad

from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(10):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = (corr - m) * beta
    # Numerically stable softmax
    shifted -= np.max(shifted)
    exps = np.exp(shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    
    # Gradient computation
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue
        # corr[k] = sum_{n: valid} h[n+k] * (1-h[n])
        # valid n: max(0,-k) <= n <= min(n-1, n-1-k)
        lo = max(0, -k)
        hi = min(n-1, n-1-k)
        if lo > hi:
            continue
        # d/dh[i] of corr[k]:
        # From h[n+k] term: i = n+k, n = i-k, valid when lo <= i-k <= hi
        #   => lo+k <= i <= hi+k, and 0 <= i <= n-1
        i1_lo = max(0, lo + k)
        i1_hi = min(n-1, hi + k)
        if i1_lo <= i1_hi:
            idx = np.arange(i1_lo, i1_hi + 1)
            grad[idx] += wk * (1.0 - h[idx - k])
        # From (1-h[n]) term: i = n, valid when lo <= i <= hi
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += wk * (-h[idx + k])
    grad *= dx
    return val, grad

    """Adaptive beta smoothmax optimization."""
    n = len(h_init)
    target = n / 2.0
    x = project_feasible(h_init.copy(), target)
    best_x = x.copy()
    best_v = compute_c5(x)
    
    for t in range(n_iter):
        progress = t / max(1, n_iter - 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = 0.05 * (1 - 0.5 * progress) / (1 + 0.1 * t)
        
        val, grad = smoothmax_val_grad(x, beta)
        x_new = x - lr * grad
        x_new = project_feasible(x_new, target)
        
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x = x_new
    
    return best_x, best_v

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    
    def objg(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

    h_new = np.clip(f(x_new), 0.0, 1.0)

        used[start:start+s] = True
        positions.append((start, start+s))
    return project_feasible(h, n/2.0)

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(10):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if d < 0:
            # Need to decrease: reduce values closest to 1
            if free.sum() > 0:
                h[free] += d / free.sum()
            else:
                h += d / len(h)
        else:
            if free.sum() > 0:
                h[free] += d / free.sum()
            else:
                h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = (corr - m) * beta
    s -= s.max()
    e = np.exp(s)
    sw = e.sum()
    w = e / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        # Term 1: d/dh[i] from h[j+lag] where j = i - lag
        # valid i: max(0, lag) <= i <= min(n-1, n-1+lag)

            grad[idx] += wk * (1.0 - h[idx - lag])

    grad *= dx
    return val, grad

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

        lag = k - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n - 1, n - 1 + lag)
        if lo1 <= hi1:
            i = np.arange(lo1, hi1 + 1)
            grad[i] += (1.0 - h[i - lag])
        lo2 = max(0, -lag)
        hi2 = min(n - 1, n - 1 - lag)
        if lo2 <= hi2:
            i = np.arange(lo2, hi2 + 1)
            grad[i] += (-h[i + lag])
    grad *= dx / top_k
    v = float(np.max(corr) * dx)
    return v, grad

    return v

def block_init(n, n_blocks, rng):
    h = np.zeros(n)
    total = n // 2

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

_GT = 0.0
_GBE = 100.0

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(10):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = (corr - m) * beta
    s -= s.max()
    e = np.exp(s)
    sw = e.sum()
    w = e / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        lo1 = max(0, lag)
        hi1 = min(n - 1, n - 1 + lag)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[idx] += wk * (1.0 - h[idx - lag])
        lo2 = max(0, -lag)
        hi2 = min(n - 1, n - 1 - lag)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            grad[idx] += wk * (-h[idx + lag])
    grad *= dx
    return val, grad

def _sf(x):
    v, _ = smoothmax_val_grad(x, _GBE)
    return v

def _sg(x):
    _, g = smoothmax_val_grad(x, _GBE)
    return g

def _sc(x):
    return x.sum() - _GT

def _scg(x):
    return np.ones(len(x))

def slsqp_polish(h, maxiter, beta):
    global _GT, _GBE
    _GT = len(h) / 2.0
    _GBE = beta
    b = Bounds(np.zeros(len(h)), np.ones(len(h)))
    c = [{'type': 'eq', 'fun': _sc, 'jac': _scg}]
    try:
        r = minimize(_sf, h, jac=_sg, method='SLSQP', bounds=b, constraints=c,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(r.x, len(h) / 2.0)
    except Exception:
        return project_feasible(h, len(h) / 2.0)

def interp_to_n(h, n_new):
    xo = (np.arange(len(h)) + 0.5) * (2.0 / len(h))
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(xo, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    return project_feasible(np.clip(f(xn), 0, 1), n_new / 2.0)

def block_init(n, nb, rng):
    h = np.zeros(n)
    total = n // 2
    raw = 0.5 + rng.random(nb)
    sizes = (raw / raw.sum() * total).astype(int)
    for i in range(nb):
        sizes[i] = max(1, sizes[i])
    diff = total - sizes.sum()
    sizes[rng.integers(nb)] += diff
    pos = 0
    for i in range(nb):
        s = min(sizes[i], n - pos)
        if s <= 0:
            break
        gap = rng.integers(0, max(1, (n - pos - s) // 2 + 1))
        st = pos + gap
        if st + s <= n:
            h[st:st + s] = 1.0
            pos = st + s
        pos += max(1, n // (2 * nb))
    return project_feasible(h, n / 2.0)

        xsv = x.copy()
        mt = rng.random()

        else:

            i = 0
            while i < n:

                    while j < n and x[j] == 1:

                    runs.append((i, j))

        else:

def grad_desc(h, n_iter, bs, be, lr0):

    target = n / 2.0
    x = project_feasible(h.copy(), target)
    bx = x.copy()
