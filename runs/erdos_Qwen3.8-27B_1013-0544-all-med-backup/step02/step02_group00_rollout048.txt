                grad[lag:] += w * (1.0 - h[:n-lag])
                if n - lag > 0:
                    grad[:n-lag] += w * (-h[lag:])
        else:
            ll = -lag
            if ll < n:
                if n - ll > 0:
                    grad[:n-ll] += w * (1.0 - h[ll:])
                    grad[ll:] += w * (-h[:n-ll])
    return grad * dx, corr

def projected_gradient_descent(h, target_sum, beta_start=50, beta_end=1000, 
                                 n_iter=5000, momentum_start=0.9, 
                                 time_func=None, time_budget=100):
    n = len(h)
    dx = 2.0 / n
    h = project_h(h.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    velocity = np.zeros(n)
    momentum = momentum_start
    
    for it in range(n_iter):
        if time_func is not None and time_func() > 0.9 * time_budget:
            break
        
        # Adaptive beta
        progress = it / max(n_iter, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        grad, corr = gradient_c5(h, beta, top_k=10)
        
        # Nesterov-style momentum
        velocity = momentum * velocity - grad
        h_new = project_h(h + velocity, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Decay momentum
        momentum = min(momentum * (1 - 0.0001 * progress), 0.99)
        h = h_new
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_c5(h, beta, top_k=12):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_indices = np.argsort(corr)[::-1][:top_k]
    m = corr[top_indices[0]]
    x_shift = np.clip(beta * (corr[top_indices] - m), -500, 500)
    exps = np.exp(x_shift)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for idx, w in zip(top_indices, weights):
        if w < 1e-8:
            continue
        lag = int(idx) - (n - 1)
        if lag >= 0:
            if 0 <= lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            ll = -lag
            if 0 < ll < n:
                grad[:n - ll] += w * (1.0 - h[ll:])
                grad[ll:] -= w * h[:n - ll]
    return grad * dx, corr

def pgd_refine(h, target_sum, budget_frac, time_func, total_budget):
    n = len(h)
    best_h = project_h(h.copy(), target_sum)
    best_c5 = compute_c5(best_h)
    velocity = np.zeros(n)
    momentum = 0.9
    step = 0.1
    alpha = 0.5
    limit = total_budget * budget_frac
    it = 0
    while time_func() < limit:
        it += 1
        progress = min(it / 5000.0, 1.0)
        beta = 30.0 * (200.0 / 30.0) ** progress
        grad, corr = gradient_c5(best_h, beta, top_k=12)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        g = grad / gnorm
        velocity = momentum * velocity + g
        vnorm = np.linalg.norm(velocity)
        if vnorm > 0:
            direction = -velocity / vnorm
        else:
            direction = -g
        lr = step / (1.0 + 0.05 * progress)
        trial = project_h(best_h + lr * direction, target_sum)
        c5_trial = compute_c5(trial)
        if c5_trial < best_c5 - 1e-15:
            best_h = trial
            best_c5 = c5_trial
            step = min(step * 1.05, 0.5)
        else:
            step *= alpha
            if step < 1e-6:
                step = 0.01
                momentum *= 0.5
    return best_h, best_c5

def local_search_pairs(h, target_sum, n_iters, time_func, limit):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    rng = np.random.default_rng(123)
    for _ in range(n_iters):
        if time_func() > limit:
            break
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        delta = rng.uniform(-0.05, 0.05)
        trial = best_h.copy()
        trial[i] += delta
        trial[j] -= delta
        trial = np.clip(trial, 0.0, 1.0)
        trial = project_h(trial, target_sum)
        c5_t = compute_c5(trial)
        if c5_t < best_c5 - 1e-14:
            best_h = trial
            best_c5 = c5_t
    return best_h, best_c5

def coordinate_descent(h, target_sum, n_rounds, time_func, limit):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for rnd in range(n_rounds):
        if time_func() > limit:
            break
        for i in range(n):
            if time_func() > limit:
                break
            old_val = best_h[i]
            # Try a few values for h[i], adjusting a random other coordinate
            j = (i + 1) % n
            for frac in [0.3, 0.5, 0.7, 0.1, 0.9]:
                new_i = 0.5 + (old_val - 0.5) * frac
                new_j = best_h[j] + (old_val - new_i)
                if new_j < 0 or new_j > 1:
                    continue
                trial = best_h.copy()
                trial[i] = new_i
                trial[j] = new_j
                trial = project_h(trial, target_sum)
                c5_t = compute_c5(trial)
                if c5_t < best_c5 - 1e-14:
                    best_h = trial
                    best_c5 = c5_t
    return best_h, best_c5

def resample_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def elapsed():
        return time.time() - start_time

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(140, 0.5)
    n0 = len(h0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    best_n = n0

    # Phase 1: PGD at parent resolution (40% budget)
    h1, c5_1 = pgd_refine(best_h, best_n / 2.0, 0.35, elapsed, budget_s)
    if c5_1 < best_c5:
        best_h, best_c5, best_n = h1, c5_1, best_n

    # Phase 2: Try different resolutions (25% budget)
    for n_try in [best_n, best_n + 20, best_n + 40, best_n + 60, best_n + 100]:
        if elapsed() > budget_s * 0.6:
            break
        if n_try < 50 or n_try > 400:
            continue
        target = n_try / 2.0
        h_init = project_h(resample_h(best_h, n_try), target)
        # Also try structured init
        if n_try % 2 == 0:
            alt = np.where(np.arange(n_try) % 2 == 0, 0.6, 0.4)
            alt = project_h(alt, target)
            c5_alt = compute_c5(alt)
            if c5_alt < compute_c5(h_init):
                h_init = alt
        h2, c5_2 = pgd_refine(h_init, target, 0.12, elapsed, budget_s)
        if c5_2 < best_c5 - 1e-15:
            best_h, best_c5, best_n = h2, c5_2, n_try

    # Phase 3: Fine-tune PGD with higher beta (20% budget)
    target = best_n / 2.0
    h3, c5_3 = pgd_refine(best_h, target, 0.15, elapsed, budget_s)
    if c5_3 < best_c5:
        best_h, best_c5 = h3, c5_3

    # Phase 4: Local search (remaining time)
    remaining = budget_s - elapsed() - 5
    if remaining > 10:
        h4, c5_4 = local_search_pairs(best_h, best_n / 2.0, int(remaining * 40),
                                       elapsed, budget_s - 5)
        if c5_4 < best_c5:
            best_h, best_c5 = h4, c5_4

        h5, c5_5 = coordinate_descent(best_h, best_n / 2.0, 30,
                                       elapsed, budget_s - 3)
        if c5_5 < best_c5:
            best_h, best_c5 = h5, c5_5

        # Final PGD polish
        h6, c5_6 = pgd_refine(best_h, best_n / 2.0, 0.03, elapsed, budget_s)
        if c5_6 < best_c5:
            best_h, best_c5 = h6, c5_6

    # Final SLSQP polish if time remains
    if elapsed() < budget_s - 10:
        nf = len(best_h)
        tf = nf / 2.0
        dxf = 2.0 / nf
        bounds = Bounds(np.zeros(nf), np.ones(nf))

        def obj_func(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            m = np.max(c)
            xs = np.clip(150.0 * (c - m), -500, 500)
            return (m + np.log(np.sum(np.exp(xs)) / len(xs)) / 150.0) * dxf

        def con_func(x_in):
            return np.sum(x_in) - tf

        constraints = [{'type': 'eq', 'fun': con_func,
                        'jac': lambda x: np.ones(x.shape)}]
        try:
            res = minimize(obj_func, best_h, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 2000, 'ftol': 1e-16})
            h_new = project_h(res.x, tf)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass

    best_h = project_h(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
