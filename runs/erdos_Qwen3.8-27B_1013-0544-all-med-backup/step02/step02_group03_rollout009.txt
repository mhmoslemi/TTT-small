            h = h + 0.0008 * (h - 0.5)
            h = project_feasible(h, ts)
            if it % 2000 == 1999:
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Phase 2: SLSQP continuation
    if tl() > 60:
        n = len(best_h)
        dx = 2.0 / n
        ts = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))
        for beta in [30, 100, 400, 1500, 6000, 20000]:
            if tl() < 25:
                break
            for trial in range(2):
                if tl() < 12:
                    break
                x0 = best_h.copy()
                if trial > 0:
                    x0 += np.random.randn(n) * (0.005 / (1 + trial))
                    x0 = project_feasible(x0, ts)
                try:
                    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': con_jac_fn, 'args': (ts,)}]
                    res = minimize(obj_fn, x0, args=(dx, beta, ts, 0.0),
                                   method='SLSQP', bounds=bounds, constraints=cons,
                                   options={'maxiter': 4000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, ts)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                except Exception:
                    pass

    # Phase 3: Coordinate descent with line search
    if tl() > 40:
        h = best_h.copy()
        n = len(h)
        dx = 2.0 / n
        ts = n / 2.0
        for sweep in range(15):
            if tl() < 20:
                break
            for i in range(n):
                cur = h[i]
                best_obj = compute_c5(h)
                best_val = cur
                for delta in [-0.15, -0.08, -0.04, -0.01, 0.01, 0.04, 0.08, 0.15]:
                    nv = cur + delta
                    if nv < 0 or nv > 1:
                        continue
                    h[i] = nv
                    diff = ts - h.sum()
                    if abs(diff) > 1e-12:
                        others = np.ones(n, dtype=bool)
                        others[i] = False
                        m = h[others] < 1.0 if diff > 0 else h[others] > 0.0
                        if m.sum() > 0:
                            h[others][m] += diff / m.sum()
                        h = np.clip(h, 0, 1)
                    c5_try = compute_c5(h)
                    if c5_try < best_obj - 1e-14:
                        best_obj = c5_try
                        best_val = nv
                h[i] = best_val
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-14:
                best_h = h.copy()
                best_c5 = c5

    # Phase 4: Multi-resolution sweep
    if tl() > 70:
        n_best = len(best_h)
        for dn in [-20, -10, -5, 5, 10, 20, 30, 40]:
            n_try = n_best + dn
            if n_try < 25 or tl() < 35:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            for beta in [300, 2000, 8000]:
                if tl() < 18:
                    break
                try:
                    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': con_jac_fn, 'args': (ts_t,)}]
                    res = minimize(obj_fn, h_t, args=(dx_t, beta, ts_t, 0.0),
                                   method='SLSQP', bounds=bnds_t, constraints=cons,
                                   options={'maxiter': 3000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                        n_best = n_try
                except Exception:
                    pass

    # Phase 5: Final high-beta SLSQP polish
    if tl() > 15:
        n = len(best_h)
        dx = 2.0 / n
        ts = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))
        for beta in [5000, 20000, 50000]:
            if tl() < 8:
                break
            try:
                cons = [{'type': 'eq', 'fun': cons_eq, 'jac': con_jac_fn, 'args': (ts,)}]
                res = minimize(obj_fn, best_h, args=(dx, beta, ts, 0.0),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 6000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, ts)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new
                    best_c5 = c5
            except Exception:
                pass

    # Phase 6: Quick coordinate polish
    if tl() > 10:
        h = best_h.copy()
        n = len(h)
        ts = n / 2.0
        for i in range(n):
            cur = h[i]
            best_obj = compute_c5(h)
            best_val = cur
            for delta in [-0.05, -0.02, -0.005, 0.005, 0.02, 0.05]:
                nv = cur + delta
                if nv < 0 or nv > 1:
                    continue
                h[i] = nv
                diff = ts - h.sum()
                if abs(diff) > 1e-12:
                    others = np.ones(n, dtype=bool)
                    others[i] = False
                    m = h[others] < 1.0 if diff > 0 else h[others] > 0.0
                    if m.sum() > 0:
                        h[others][m] += diff / m.sum()
                    h = np.clip(h, 0, 1)
                c5_try = compute_c5(h)
                if c5_try < best_obj - 1e-14:
                    best_obj = c5_try
                    best_val = nv
            h[i] = best_val
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
