        vv = compute_c5(h)
        if vv < best_val:
            best_val = vv
            best = h.copy()
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: Momentum subgradient, single lag
    if tl() > 120:
        h, c5 = momentum_subgrad(best_h, n, dx, target_sum, t0, budget_s, top_k=1)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 2: Momentum subgradient, top-3 lags
    if tl() > 100:
        h, c5 = momentum_subgrad(best_h, n, dx, target_sum, t0, budget_s, top_k=3)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 3: Coordinate descent
    if tl() > 80:
        h, c5 = coord_descent(best_h, n, dx, target_sum, t0, budget_s, sweeps=150)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 4: SLSQP continuation
    if tl() > 60:
        for beta in [30, 100, 500, 2000, 10000]:
            if tl() < 15:
                break
            for trial in range(2):
                if tl() < 10:
                    break
                x0 = best_h.copy()
                if trial == 1:
                    x0 += np.random.randn(n) * 0.004
                    x0 = project_feasible(x0, target_sum)
                try:
                    res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 3000, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h, best_c5 = h_new, c5
                except Exception:
                    pass

    # Phase 5: Try different resolutions
    if tl() > 100:
        for n_try in [50, 60, 80, 90, 120]:
            if tl() < 30:
                break
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            h_t = project_feasible(resample_h(best_h, n_try), ts_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts_t, n_try)}]
            for beta in [100, 1000]:
                if tl() < 15:
                    break
                try:
                    res = minimize(obj_smooth, h_t, args=(dx_t, beta), method='SLSQP',
                                   bounds=bnds_t, constraints=cons_t,
                                   options={'maxiter': 2000, 'ftol': 1e-13})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h, best_c5 = h_new, c5
                        n, dx, target_sum = n_try, dx_t, ts_t
                        bounds, cons = bnds_t, cons_t
                except Exception:
                    pass

    # Phase 6: Coordinate descent polish at final resolution
    n = len(best_h)
    dx = 2.0 / n
    target_sum = n / 2.0
    if tl() > 30:
        h, c5 = coord_descent(best_h, n, dx, target_sum, t0, budget_s, sweeps=80)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 7: Final SLSQP with high beta
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
    if tl() > 20:
        for beta in [20000, 80000]:
            if tl() < 8:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new, c5
            except Exception:
                pass

    # Phase 8: One more momentum pass if time remains
    if tl() > 40:
        h, c5 = momentum_subgrad(best_h, n, dx, target_sum, t0, budget_s, top_k=2)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    return best_h, best_c5, len(best_h)
