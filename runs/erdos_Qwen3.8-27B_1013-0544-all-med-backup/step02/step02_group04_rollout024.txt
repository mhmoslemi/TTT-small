    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def grad_corr_k(h, k):
    """Gradient of C(k) with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        for i in range(n - k):
            g[i] += 1.0 - h[i + k]
            if i >= k:
                g[i] -= h[i - k]
    else:
        kk = -k
        for i in range(kk, n):
            g[i] += 1.0 - h[i - kk]
            if i + kk < n:
                g[i] -= h[i + kk]
    return g

    """Return (smooth_max_value, gradient) of smooth max of corr values."""
    n = len(h)
    corr = corr_raw(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    # Avoid overflow
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    w_sum = w.sum()
    w_norm = w / w_sum
    val = (m + np.log(w_sum) / beta)

    for ki in range(len(corr)):
        if w_norm[ki] < 1e-15:
            continue
        k = ki - (n - 1)
        g += w_norm[ki] * grad_corr_k(h, k)
    g -= g.mean()
    return val, g

    return x.sum() - target_sum

import numpy as np
from scipy.optimize import minimize, Bounds
import time

_global_params = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def grad_corr_k(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        for i in range(n):
            if i < n - k:
                g[i] += 1.0 - h[i + k]
            if i >= k:
                g[i] -= h[i - k]
    else:
        kk = -k
        for i in range(n):
            if i >= kk:
                g[i] += 1.0 - h[i - kk]
            if i < n - kk:
                g[i] -= h[i + kk]
    return g

def smooth_max_val_grad(h, beta):
    n = len(h)
    corr = corr_raw(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.minimum(shifted, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    w_norm = w / w_sum
    val = m + np.log(max(w_sum, 1e-300)) / beta
    g = np.zeros(n)
    for ki in range(2 * n - 1):
        wi = w_norm[ki]
        if wi < 1e-14:
            continue
        k = ki - (n - 1)
        g += wi * grad_corr_k(h, k)
    g -= g.mean()
    return val, g

def obj_func(x):
    val, _ = smooth_max_val_grad(x, _global_params['beta'])
    return val * _global_params['dx']

def obj_grad_func(x):
    _, g = smooth_max_val_grad(x, _global_params['beta'])
    return g * _global_params['dx']

    return x.sum() - _global_params['target_sum']

def con_jac(x):
    return np.ones(len(x))
