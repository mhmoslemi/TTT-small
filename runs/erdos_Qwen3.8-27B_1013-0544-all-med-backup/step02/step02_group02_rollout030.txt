def projected_gradient_step(h, n, alpha, lr):
    """One step of projected gradient on log-sum-exp objective."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    
    # Softmax weights (numerically stable)
    c_shifted = c - c.max()
    weights = np.exp(alpha * c_shifted)
    weights /= weights.sum()
    
    # Gradient of weighted sum
    center = n - 1
    grad = np.zeros(n)
    for m in range(len(c)):
        if weights[m] < 1e-15:
            continue
        k = m - center
        w = weights[m]
        # d c[k] / d h[j] as derived above
        if k >= 0:
            # h[n+k] term: j in [k, n-1], contrib (1-h[j-k])
            j_range = np.arange(k, n)
            grad[j_range] += w * (1 - h[j_range - k]) * dx
            # (1-h[n]) term: j in [0, n-1-k], contrib -h[j+k]
            j_range = np.arange(0, n-k)
            grad[j_range] += -w * h[j_range + k] * dx
        else:
            ka = -k  # positive
            # h[n-k] = h[n+ka... wait, k is negative
            # h[n+k] where k<0: n+k = n-ka. j = n+k => n = j-k = j+ka
            # valid n: 0 <= n <= n-1 AND 0 <= n+k <= n-1 => 0 <= n-ka <= n-1 => ka <= n <= n-1+ka
            # Also n in [0, n-1], so n in [ka, n-1]
            # j = n+k = n-ka, so j in [0, n-1-ka]
            # contribution: (1-h[n]) = (1-h[j-k]) = (1-h[j+ka])
            j_range = np.arange(0, n-ka)

            # (1-h[n]) term: n in [ka, n-1] (from above)
            # j = n, contrib -h[n+k] = -h[n-ka] = -h[j-ka]
            j_range = np.arange(ka, n)
            grad[j_range] += -w * h[j_range - ka] * dx
    
    # Projected gradient step
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
