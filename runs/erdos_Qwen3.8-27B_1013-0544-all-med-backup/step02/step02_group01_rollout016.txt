import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time


def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr_full(h)) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-9) & (h < 1.0 - 1e-9)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 0.0)
    exps = np.exp(shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue
        s = k - (n - 1)
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if lo1 < hi1:

            grad[ar] += wk * (1.0 - h[ar + s])

        hi2 = min(n, n + s)
        if lo2 < hi2:
            ar = np.arange(lo2, hi2)
            grad[ar] += wk * (-h[ar - s])
    grad *= dx
    return val, grad

    t_start = time.time()

    for t in range(max_iters):
        if (t & 127) == 0 and time.time() - t_start > time_limit:
            break
        frac = min(1.0, t / max(1, int(max_iters * 0.35)))
        beta = beta_start * (beta_end / beta_start) ** frac

        v_a = b2 * v_a + (1 - b2) * grad * grad
        m_hat = m_a / (1 - b1 ** t1)
        v_hat = v_a / (1 - b2 ** t1)

        v, _ = smoothmax_val_grad(x, beta)
        return v
