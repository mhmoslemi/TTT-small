def projected_gradient_step(h, n, step_size):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    k_star = np.argmax(c)
    # Gradient of c[k_star] with respect to h
    # c[k_star] = sum_{i=0}^{n-1-k_star} h[i] * (1-h[i+k_star])
    # ∂c[k_star]/∂h[j] = (1-h[j+k_star]) if j < n-k_star
    #                   - h[j-k_star] if j >= k_star
    #                   (both if k_star <= j < n-k_star)
    
    grad = np.zeros(n)
    ks = k_star

    for j in range(n - ks):
        grad[j] += (1 - h[j + ks])
    # For j in [ks, n-1]: grad[j] += -h[j-ks]
    for j in range(ks, n):
        grad[j] -= h[j - ks]

    else:
        total_h = h_new.sum()
        if total_h > 0:

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    bounds = [(0.0, 1.0)] * n

             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except:
        return h

from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n, rng):
    h_b = (h > 0.5).astype(float)
    n_ones = int(h_b.sum())
    if n_ones > n // 2:
        idx = np.where(h_b == 1)[0]
        to_flip = rng.choice(idx, n_ones - n // 2, replace=False)
        h_b[to_flip] = 0
    elif n_ones < n // 2:
        idx = np.where(h_b == 0)[0]
        to_flip = rng.choice(idx, n // 2 - n_ones, replace=False)
        h_b[to_flip] = 1
    return h_b

def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_opt(h, n, alpha, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})

    except Exception:

            # Flip multi_bit ones to zero and multi_bit zeros to one
