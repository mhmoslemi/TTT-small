```python
import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

_global_target = 0.0
_global_beta = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    max_sh = np.max(shifted)
    exps = np.exp(shifted - max_sh)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + (max_sh + np.log(sw)) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad

def _slsqp_obj(x):
    return smoothmax_val_grad(x, _global_beta)[0]

def _slsqp_jac(x):
    return smoothmax_val_grad(x, _global_beta)[1]

def _sum_con_fun(x):
    return x.sum() - _global_target

def _sum_con_jac(x):
    return np.ones(len(x))

def _lbfgs_obj(x):
    v, g = smoothmax_val_grad(x, _global_beta)
    s = x.sum() - _global_target
    return v + 5000.0 * s * s

def _lbfgs_jac(x):
    v, g = smoothmax_val_grad(x, _global_beta)
    s = x.sum() - _global_target
    return g + 10000.0 * s * np.ones(len(x))

def slsqp_refine(h, maxiter, beta):
    global _global_target, _global_beta
    _global_target = len(h) / 2.0
    _global_beta = beta
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, _global_target)
    except Exception:
        return project_feasible(h, _global_target)

def lbfgsb_refine(h, maxiter, beta):
    global _global_target, _global_beta
    _global_target = len(h) / 2.0
    _global_beta = beta
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(_lbfgs_obj, h, jac=_lbfgs_jac, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, _global_target)
    except Exception:
        return project_feasible(h, _global_target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def nesterov_subgrad(h, max_iters, lr0, decay, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])
        grad = grad / top_k * dx
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def perturb(h, rng, sigma):
    n = len(h)
    return project_feasible(h + rng.normal(0, sigma, n), n / 2.0)

def swap_perturb(h, rng, n_swaps):
    n = len(h)
    h_new = h.copy()
    for _ in range(n_swaps):
        idx = rng.choice(n, 2, replace=False)
        h_new[idx[0]], h_new[idx[1]] = h_new[idx[1]], h_new[idx[0]]
    return project_feasible(h_new, n / 2.0)

def block_perturb(h, rng, block_size, sigma):
    n = len(h)
    h_new = h.copy()
    n_blocks = max(2, n // (2 * block_size))
    for _ in range(n_blocks):
        start = rng.randint(0, max(1, n - block_size))
        end = min(start + block_size, n)
        h_new[start:end] += rng.normal(0, sigma, end - start)
    return project_feasible(h_new, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)
    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: Escalating-beta SLSQP at current resolution
    if tl() > 150:
        for beta in [150, 400, 800]:
            if tl() < 70:
                break
            h1 = slsqp_refine(h_best, maxiter=500, beta=beta)
            c1 = compute_c5(h1)
            if c1 < best_c5:
                h_best, best_c5 = h1, c1

    # Phase 2: ILS - Gaussian perturb + SLSQP reoptimize
    if tl() > 250:
        h_work = h_best.copy()
        sigmas = [0.12, 0.08, 0.05, 0.03, 0.015]
        for sigma in sigmas:
            if tl() < 80:
                break
            for trial in range(25):
                if tl() < 50:
                    break
                h_p = perturb(h_work, rng, sigma)
                h_opt = slsqp_refine(h_p, maxiter=350, beta=500)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c5:
                    h_best, best_c5 = h_opt, c_opt
                    h_work = h_opt.copy()

    # Phase 3: Swap + block perturbations with Nesterov
    if tl() > 150:
        for trial in range(12):
            if tl() < 80:
                break
            if trial % 2 == 0:
                h_p = swap_perturb(h_best, rng, n_swaps=4)
            else:
                h_p = block_perturb(h_best, rng, block_size=12, sigma=0.08)
            h_n, c_n = nesterov_subgrad(h_p, 3000, 0.035, 0.9996, top_k=10)
            if c_n < best_c5:
                h_best, best_c5 = h_n, c_n
            if tl() > 35:
                h_s = slsqp_refine(h_best, maxiter=400, beta=600)
                c_s = compute_c5(h_s)
                if c_s < best_c5:

    if tl() > 100:

        if tl() > 40:

            if c_s < best_c5:

            if tl() < 90:

            h_interp = interp_to_n(h_best, n_try)

            if tl() < 30:

    return h_best, best_c5, len(h_best)
```
