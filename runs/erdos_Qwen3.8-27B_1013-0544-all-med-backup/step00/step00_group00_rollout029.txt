import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    # Log-sum-exp trick
    shifted = corr - m
    # For numerical stability
    max_s = np.max(shifted) * alpha
    if max_s > 500:
        penalty = beta * (np.sum(h) - n / 2.0) ** 2
        return m + penalty
    log_sum = np.log(np.sum(np.exp(alpha * shifted))) / alpha
    penalty = beta * (np.sum(h) - n / 2.0) ** 2
    return m + log_sum + penalty

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    penalty = beta * (np.sum(h) - n / 2.0) ** 2
    if alpha < 1e-6:
        return m + penalty
    shifted = corr - m
    max_exp = np.max(shifted) * alpha
    if max_exp > 500.0:
        return m + penalty
    with np.errstate(over='ignore'):
        log_sum = np.log(np.sum(np.exp(alpha * shifted))) / alpha
    return m + log_sum + penalty

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    # Bisection: find c such that sum(clip(h+c, 0, 1)) = target
    s0 = np.sum(np.clip(h, 0, 1))
    if abs(s0 - target) < 1e-10:
        return np.clip(h, 0, 1)
    lo = -2.0
    hi = 2.0
    for _ in range(100):

        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid

    m = np.max(corr)
    penalty = beta * (np.sum(h) - n / 2.0) ** 2
    shifted = corr - m
    a = alpha
    if a < 1e-3:
        return m + penalty
    max_exp = float(np.max(shifted)) * a
    if max_exp > 500.0:
        return m + penalty
    with np.errstate(over='ignore', invalid='ignore'):
        vals = a * shifted
        log_sum = np.log(np.sum(np.exp(vals))) / a
    return m + log_sum + penalty

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def get_subgrad(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = int(np.argmax(corr))
    sg = np.zeros(n)
    for j in range(n):
        jm = j - k_max
        if 0 <= jm < n:
            sg[j] += 1.0 - h[jm]
        jp = j + k_max
        if 0 <= jp < n:
            sg[j] -= h[jp]
    return sg

def local_search(h, rng, n_evals_budget):
    n = len(h)
    best = h.copy()
    best_v = compute_c5(best)
    evals = 0
    for it in range(100000):
        if evals >= n_evals_budget:
            break
        j = rng.randint(n)
        deltas = [0.01, -0.01, 0.02, -0.02, 0.005, -0.005, 0.05, -0.05]
        improved = False
        for d in deltas:
            if evals >= n_evals_budget:
                break
            h_trial = best.copy()
            h_trial[j] += d
            if h_trial[j] < 0 or h_trial[j] > 1:
                continue
            # Maintain sum constraint: adjust another coordinate
            sum_err = -d
            if sum_err > 0:
                cands = np.where((best < 1.0) & (np.arange(n) != j))[0]
            else:
                cands = np.where((best > 0.0) & (np.arange(n) != j))[0]
            if len(cands) == 0:
                continue
            # Try the candidate that gives the best result
            best_j2 = -1
            best_v2 = best_v
            for j2 in cands[:10]:  # Limit to 10 candidates for speed
                h_try2 = h_trial.copy()
                h_try2[j2] += sum_err
                if 0 <= h_try2[j2] <= 1:
                    v = compute_c5(h_try2)
                    evals += 1
                    if v < best_v2:
                        best_v2 = v
                        best_j2 = j2
            if best_j2 >= 0:
                best[best_j2] += sum_err
                best[j] += d
                best_v = best_v2
                improved = True
                break
        if not improved:
            # Try a larger random perturbation occasionally
            if rng.random() < 0.1:
                h_try = best + rng.randn(n) * 0.05
                h_try = project_feasible(h_try)
                v = compute_c5(h_try)
                evals += 1
                if v < best_v:
                    best = h_try.copy()
                    best_v = v
    return best, best_v, evals

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_obj(h, alpha, beta):
