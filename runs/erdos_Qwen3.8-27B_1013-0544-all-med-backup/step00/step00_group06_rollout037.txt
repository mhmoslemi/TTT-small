def project_hypersimplex(v, m):
    """Project v onto {x : 0 <= x <= 1, sum(x) = m}"""
    n = len(v)
    # Binary search for threshold
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > m:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, 0, 1)

```python
import numpy as np
from scipy.signal import convolve
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def project_hypersimplex(v, m):
    n = len(v)
    lo, hi = -1.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > m:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, 0, 1)

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full") * dx
    m = np.max(corr)
    exp_val = np.exp(T * (corr - m))
    z = np.sum(exp_val)
    weights = exp_val / z
    obj = m + np.log(z) / T
    
    # Gradient via convolution
    grad = dx * np.convolve(1 - 2*h, weights, mode='full')[n-1:2*n-1]
    return obj, grad

def optimize_from_start(h0, T_vals, n_iter, lr0, seed, time_limit):
    """Run projected gradient with adaptive T and learning rate."""
    np.random.seed(seed)
    h = h0.copy()
    n = len(h)
    m_target = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        T = T_vals[min(it // (n_iter // len(T_vals)), len(T_vals)-1)]
        obj, grad = smooth_obj_grad(h, T)
        
        # Backtracking line search
        lr = lr0
        for _ in range(30):
            hTrial = project_hypersimplex(h - lr * grad, m_target)
            # Check if projection actually satisfies constraint
            objTrial, _ = smooth_obj_grad(hTrial, T)
            if objTrial < obj:
                h = hTrial
                break
            lr *= 0.5
        else:
            h = project_hypersimplex(h - lr * grad, m_target)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```

    lo, hi = -2.0, 3.0

        if s > m:
            lo = mid
        else:
            hi = mid

    corr = np.correlate(h, 1-h, mode="full") * dx

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_hypersimplex(v, m):
    n = len(v)
    lo, hi = -2.0, 3.0
    for _ in range(300):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > m:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(corr)
    exp_val = np.exp(T * (corr - m))
    z = np.sum(exp_val)
    weights = exp_val / z
    obj = m + np.log(z) / T
    w2 = weights[::-1]
    conv1 = np.convolve(1.0 - h, weights, mode='full')
    conv2 = np.convolve(h, w2, mode='full')
    grad = dx * (conv1 - conv2)[n - 1:2 * n - 1]
    return obj, grad
