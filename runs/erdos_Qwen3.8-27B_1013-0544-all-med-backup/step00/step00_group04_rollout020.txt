```python
    """Generate Legendre sequence of length p-1."""
    n = p - 1
    h = np.zeros(n)
    for i in range(n):
        val = (i + 1) % p
        # Check if val is a quadratic residue mod p
        if pow(val, (p-1)//2, p) == 1:
            h[i] = 1
    return h

def random_binary(n, rng):

    ones = rng.choice(n, n//2, replace=False)
```

```python
    if best_h is not None and time.time() < deadline - 15:
        d = deadline
        h_opt, c5_opt = simulated_anneal(best_h, best_n, d, rng)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, best_n
            print(f"SA: C5={c5_opt:.6f}", flush=True)

    # Phase 6: Final polish
    if best_h is not None and time.time() < deadline - 2:
        h_opt, c5_opt = hill_climb(best_h, best_n, deadline)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, best_n

    print(f"FINAL: C5={best_c5:.6f}, n={best_n}, t={time.time()-t_start:.1f}s", flush=True)
    return best_h, best_c5, best_n
```
