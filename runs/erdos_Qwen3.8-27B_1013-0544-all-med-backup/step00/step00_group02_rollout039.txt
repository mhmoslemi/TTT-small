import numpy as np
import time

def _c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def _ls(h, rng, max_iters=500, tlim=20):
    n = len(h)
    h = h.astype(float).copy()
    t0 = time.time()
    cm = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
    for _ in range(max_iters):
        if time.time() - t0 > tlim:
            break
        ih = list(np.where(h > 0.5)[0])
        il = list(np.where(h < 0.5)[0])
        if not ih or not il:
            break
        bg = 1e-12
        bp = None
        for i in ih:
            hv = h[i]
            for j in il:
                lv = h[j]
                if abs(hv + lv - 1.0) > 1e-9:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                nm = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
                h[i] = hv
                h[j] = lv
                g = cm - nm
                if g > bg:
                    bg = g
                    bp = (i, j)
        if bp is not None:
            i, j = bp
            h[i] = 0.0
            h[j] = 1.0
            cm -= bg
        else:
            valid = [(i, j) for i in ih for j in il if abs(h[i] + h[j] - 1.0) < 1e-9]
            if not valid:
                break
            i, j = valid[rng.randint(len(valid))]
            h[i] = 0.0
            h[j] = 1.0
            cm = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
    return h, cm * (2.0 / n)

def _rb(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    t = n // 2
    h[idx[:t]] = 1.0
    if n % 2 == 1:
        h[idx[t]] = 0.5
    return h

def _make_bin_from_cont(g, n):
    order = np.argsort(-g)
    h = np.zeros(n)
    h[order[:n // 2]] = 1.0
    if n % 2 == 1:
        h[order[n // 2]] = 0.5
    return h

        hb = _make_bin_from_cont(hi, ni)

            bs = min(bs, co)
