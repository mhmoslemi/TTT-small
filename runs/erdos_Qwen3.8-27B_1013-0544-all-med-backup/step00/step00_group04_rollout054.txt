            if s_opt < best_score + 0.005 and time.time() < deadline - 5:
                t4 = min(deadline - 3, time.time() + 20)
                h_opt2, s_opt2 = pair_swap_sa(h_opt, t4, seed + n * 17)
                if s_opt2 < best_score:
                    best_score, best_h, best_n = s_opt2, h_opt2, n
    
    return best_h, best_score, best_n
