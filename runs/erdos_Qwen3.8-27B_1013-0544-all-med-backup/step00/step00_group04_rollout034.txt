def structured_start(n, seed):
    """Create a structured starting point"""
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    half = n // 2
    # Create a quasi-periodic pattern
    period = max(3, n // 8)
    for i in range(n):
        if (i // period) % 2 == 0:
            h[i] = 1.0
    # Adjust to have exactly half ones
    n_ones = int(np.sum(h))
    if n_ones > half:
        ones_idx = np.where(h == 1)[0]
        rng.shuffle(ones_idx)
        h[ones_idx[:n_ones - half]] = 0
    elif n_ones < half:
        zeros_idx = np.where(h == 0)[0]
        rng.shuffle(zeros_idx)
        h[zeros_idx[:half - n_ones]] = 1
    return h
