Actually, I can provide the analytical gradient to SLSQP to speed things up. Let me compute the gradient of the smooth max.

The smooth max is: f(h) = (1/β) log(sum_k exp(β * c_k))

where w_k = exp(β*c_k) / sum_j exp(β*c_j) is the softmax weight.

def smooth_max_and_grad(h, beta):

    # Softmax weights
    m = corrs.max()
    exps = np.exp(beta * (corrs - m))
    w = exps / exps.sum()
    f = (m + np.log(exps.sum())) / beta

    grad = np.zeros(n)

        if w[k] < 1e-10:
            continue
        L = k - (n - 1)
        grad += w[k] * gradient_at_lag(h, L, dx)
    
    return f, grad

def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_at_lag(h, L, dx):
    n = len(h)
    g = np.zeros(n)
    if L == 0:
        return dx * (1.0 - 2.0 * h)
    elif L > 0:
        g[L:] += dx * (1.0 - h[:n - L])
        g[:n - L] -= dx * h[L:]
    else:
        L2 = -L
        g[:n - L2] += dx * (1.0 - h[L2:])
        g[L2:] -= dx * h[:n - L2]
    return g
