import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum_inplace(h, n):
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return
    step = diff / n
    if np.all(h + step >= -1e-12) and np.all(h + step <= 1.0 + 1e-12):
        h[:] = np.clip(h + step, 0.0, 1.0)
        return
    for _ in range(n * 10):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idxs = np.where(h < 1.0 - 1e-14)[0]
        else:
            idxs = np.where(h > 1e-14)[0]
        if len(idxs) == 0:
            break
        for j in idxs:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                adj = min(1.0 - h[j], diff)
                h[j] += adj
                diff -= adj
            else:
                adj = min(h[j], -diff)
                h[j] -= adj
                diff += adj

def make_f1d(beta, alpha):
    def f1d(t):
        v = beta + alpha * t
        v[0] = beta[0] + t * (1.0 - t)
        return float(np.max(v))
    return f1d

def optimize_1d(beta, alpha):
    t_grid = np.linspace(0.0, 1.0, 401)
    T = t_grid[:, np.newaxis]
    vals = beta[np.newaxis, :] + alpha[np.newaxis, :] * T
    vals[:, 0] = beta[0] + t_grid * (1.0 - t_grid)
    f_vals = vals.max(axis=1)
    t_best = t_grid[np.argmin(f_vals)]
    span = 1.0 / 400.0
    lo = max(0.0, t_best - span)
    hi = min(1.0, t_best + span)
    def f1d(t):
        v = beta + alpha * t
        v[0] = beta[0] + t * (1.0 - t)
        return float(np.max(v))
    for _ in range(45):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if f1d(m1) < f1d(m2):
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-11:
            break
    return (lo + hi) * 0.5

def coordinate_descent_pass(h, n, max_time):

            return

        h[j] = 0.0

        h[j] = 1.0

        alpha = corr1 - corr0
        beta = corr0.copy()
        t_opt = optimize_1d(beta, alpha)
        h[j] = t_opt
        fix_sum_inplace(h, n)
