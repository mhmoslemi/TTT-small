import numpy as np
from scipy.optimize import minimize, linprog
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr * dx)

def smooth_max_obj(h, alpha=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    overlaps = corr * dx
    m = np.max(overlaps)
    return m + np.log(np.sum(np.exp(alpha * (overlaps - m)))) / alpha

def smooth_max_grad(h, alpha=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    overlaps = corr * dx
    m = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m))
    weights = exp_vals / np.sum(exp_vals)
    
    # lags for output index i: lag = i - (n-1)
    lags = np.arange(len(overlaps)) - (n - 1)
    
    grad = np.zeros(n)
    for i in range(len(overlaps)):
        if weights[i] < 1e-10:
            continue
        lag = lags[i]
        w = weights[i] * dx
        # d(overlaps[i])/dh[j] = (1-h[j-lag]) * [0<=j-lag<n] - h[j+lag] * [0<=j+lag<n]
        # For each j:
        # Term 1: (1-h[j-lag]) for j such that 0<=j-lag<n, i.e., lag<=j<lag+n
        # Term 2: -h[j+lag] for j such that 0<=j+lag<n, i.e., -lag<=j<n-lag
        
        # Term 1: j ranges from max(0, lag) to min(n-1, lag+n-1)
        j1_start = max(0, lag)
        j1_end = min(n, lag + n)
        if j1_start < j1_end:
            jj = np.arange(j1_start, j1_end)
            grad[jj] += w * (1 - h[jj - lag])
        
        # Term 2: j ranges from max(0, -lag) to min(n-1, n-1-lag)
        j2_start = max(0, -lag)
        j2_end = min(n, n - lag)
        if j2_start < j2_end:
            jj = np.arange(j2_start, j2_end)
            grad[jj] -= w * h[jj + lag]
    
    return grad

def project_to_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute the difference
        if diff > 0:
            # Need to increase sum; add to elements < 1
            mask = h < 1.0
            if np.sum(mask) > 0:
                add = diff / np.sum(mask)
                h[mask] = np.minimum(h[mask] + add, 1.0)
        else:
            # Need to decrease sum; subtract from elements > 0
            mask = h > 0.0
            if np.sum(mask) > 0:
                sub = -diff / np.sum(mask)
                h[mask] = np.maximum(h[mask] - sub, 0.0)
    return h

def optimize_h(h0, n, alpha=50, max_iter=1000, time_limit=60):
    """Optimize h starting from h0."""
    target_sum = n / 2.0
    
    def obj(x):
        return smooth_max_obj(x, alpha)
    
    def grad(x):
        return smooth_max_grad(x, alpha)
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    
    result = minimize(obj, h0, method='SLSQP', jac=grad,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': 1e-12})
    return result.x, result.fun

                print(f"  a={alpha}: C5={c5:.6f} [{time.time()-t_start:.0f}s]")

    # Phase 2: try different resolutions
    n_list = []
    nh = len(best_h)
    for candidate in [46, 60, 90, 120, 150, 200]:
        if candidate != nh:
            n_list.append(candidate)

    for n_target in n_list:
        if time.time() - t_start > budget_s - 60:
            break
        h_start = interp_h(best_h, len(best_h), n_target)
        h_start = project_feasible(h_start, n_target)
        for alpha in [20, 80, 300]:
            if time.time() - t_start > budget_s - 20:
                break
            h_opt = do_minimize(h_start, n_target, alpha, 1500)
            if h_opt is not None:
                c5 = evaluate_c5(h_opt)
                if c5 < best_c5:
                    best_h = h_opt
                    best_c5 = c5
                    print(f"  n={n_target} a={alpha}: C5={c5:.6f} [{time.time()-t_start:.0f}s]")

    # Phase 3: random perturbations at best resolution
    n_best = len(best_h)
    for trial in range(20):
        if time.time() - t_start > budget_s - 15:
            break
        scale = 0.03 + 0.02 * trial
        h_pert = best_h + np.random.randn(n_best) * scale
        h_pert = project_feasible(h_pert, n_best)
        h_opt = do_minimize(h_pert, n_best, 100, 800)
        if h_opt is not None:
            c5 = evaluate_c5(h_opt)
            if c5 < best_c5:
                best_h = h_opt
                best_c5 = c5
                print(f"  pert{trial}: C5={c5:.6f} [{time.time()-t_start:.0f}s]")

    # Phase 4: final polish with high alpha
    if time.time() - t_start < budget_s - 10:
        for alpha in [2000, 5000]:
            if time.time() - t_start > budget_s - 8:
                break
            h_opt = do_minimize(best_h, len(best_h), alpha, 1000)
            if h_opt is not None:
                c5 = evaluate_c5(h_opt)
                if c5 < best_c5:
                    best_h = h_opt
                    best_c5 = c5
                    print(f"  polish a={alpha}: C5={c5:.6f} [{time.time()-t_start:.0f}s]")

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    print(f"FINAL: n={len(best_h)}, C5={best_c5:.6f}, t={time.time()-t_start:.0f}s")
    return best_h, best_c5, len(best_h)
