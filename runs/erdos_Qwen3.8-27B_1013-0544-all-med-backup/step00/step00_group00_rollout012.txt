            if attempt == 0:
                x_start2 = h2.copy()
            else:
                x_start2 = best_h2 + rng.normal(0, 0.04, n2)
                x_start2 = project_feasible(x_start2, n2)

            remaining = max(30, budget_s - (time.time() - start_time) - 80)
            max_iter = max(500, int(remaining * 15))

            res2 = minimize(obj_func, x_start2, method='L-BFGS-B',
                           jac=grad_func, args=(dx2, penalty),
                           bounds=bounds2,
                           options={'maxiter': max_iter, 'factr': 1e8})
            if res2.x is not None:
                h2_new = project_feasible(res2.x, n2)
                best_h2 = h2_new.copy()
                c5_2 = compute_c5(h2_new)
                if c5_2 < best_c5:
                    best_h = h2_new.copy()
                    best_c5 = c5_2

    # Final projection and verification
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
