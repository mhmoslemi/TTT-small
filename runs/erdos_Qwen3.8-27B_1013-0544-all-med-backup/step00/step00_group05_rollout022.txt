    target = len(h) / 2.0
    for _ in range(20):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-15

            if s < 1e-15:

            h[mask] -= (diff / s) * h[mask]
        else:
            mask = h < 1.0 - 1e-15
            s = (1.0 - h[mask]).sum()
            if s < 1e-15:
                break

    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    for _ in range(30):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-15
            if not mask.any():
                break
            s = h[mask].sum()
            if s < 1e-15:
                break
            h[mask] -= (diff / s) * h[mask]
        else:
            mask = h < 1.0 - 1e-15
            if not mask.any():
                break
            s = (1.0 - h[mask]).sum()
            if s < 1e-15:
                break
            h[mask] += (-diff / s) * (1.0 - h[mask])

                res = minimize(nm_obj, h_interp, method='Nelder-Mead',
                               options={'maxiter': 5000, 'xatol': 1e-11, 'fatol': 1e-13, 'adaptive': True})
                h_opt = repair(res.x)
                c_opt = c5_value(h_opt)
                if c_opt < best_c5:
                    best_c5, best_h = c_opt, h_opt
            except Exception:
                pass

    # Phase 6: Final NM polish
    if time.time() - t0 < budget_s * 0.95:
        noise = np.random.uniform(-0.03, 0.03, len(best_h))
        h_pert = repair(best_h + noise)
        try:
            res = minimize(nm_obj, h_pert, method='Nelder-Mead',
                           options={'maxiter': 8000, 'xatol': 1e-12, 'fatol': 1e-14, 'adaptive': True})
            h_opt = repair(res.x)
            c_opt = c5_value(h_opt)
            if c_opt < best_c5:
                best_c5, best_h = c_opt, h_opt
        except Exception:
            pass

    return best_h, best_c5, len(best_h)
