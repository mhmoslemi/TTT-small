    corr = np.correlate(h, 1 - h, mode="full")
    n = len(h)
    dx = 2.0 / n
    return np.max(corr) * dx

def compute_gradient(h, s_max):
    """Compute gradient of f_{s_max} w.r.t. h"""
    n = len(h)
    k = s_max  # index into the full correlation
    s = k - (n - 1)  # the actual shift
    grad = np.zeros(n)
    for m in range(n):
        g = 0.0
        if 0 <= m + s <= n - 1:
            g += (1 - h[m + s])
        if 0 <= m - s <= n - 1:
            g -= h[m - s]
        grad[m] = g
    return grad

import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_shift(h, s, n):
    """Gradient of the correlation at shift s w.r.t. h"""
    grad = np.zeros(n)
    if s >= 0:
        grad[:n-s] += (1.0 - h[s:])
        grad[s:] -= h[:n-s]
    else:
        sp = -s
        grad[sp:] += (1.0 - h[:n-sp])
        grad[:n-sp] -= h[sp:]
    return grad

def project(v, n):
    """Project v onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # If already feasible, return as is
    if v.min() >= 0 and v.max() <= 1 and abs(np.sum(v) - target) < 1e-12:
        return v.copy()
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(v - t, 0.0, 1.0)

def optimize_from(h, n, lr, n_iters, top_k, t_end):
    """Subgradient descent from initial point h"""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    
    for it in range(n_iters):
        if time.time() > t_end:
            break
        # Compute top-k shifts
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        
        # Average gradient of top-k shifts
        grad = np.zeros(n)
        for k in top_idx:
            s = int(k - (n - 1))
            grad += grad_shift(h, s, n)
        grad /= top_k
        
        # Step
        h = project(h - lr * grad, n)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Decay learning rate
        lr *= 0.9998
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def grad_shift(h, s, n):
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] -= h[:n - s]
    else:
        sp = -s
        grad[sp:] += (1.0 - h[:n - sp])
        grad[:n - sp] -= h[sp:]
    return grad

def project(v, n):
    target = n / 2.0
    if v.min() >= -1e-12 and v.max() <= 1.0 + 1e-12 and abs(np.sum(v) - target) < 1e-10:
        return np.clip(v, 0.0, 1.0)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s_val = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s_val > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(v - t, 0.0, 1.0)

def subgrad_optimize(h, n, lr, max_iters, top_k, t_end):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)[0]
    stall = 0
    for it in range(max_iters):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k = min(top_k, len(corr))
        top_idx = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(n)
        for idx in top_idx:
            s = int(idx - (n - 1))
            grad += grad_shift(h, s, n)
        grad /= k
        h = project(h - lr * grad, n)
        c5_val = compute_c5(h, n)[0]
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        lr *= 0.9999
        if stall > 300:
            break
    return best_h, best_c5

    rng = np.random.RandomState(seed)

            best_h = h_opt.copy()
    return best_h, best_c5
