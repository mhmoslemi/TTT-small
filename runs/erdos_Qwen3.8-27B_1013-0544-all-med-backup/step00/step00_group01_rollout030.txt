import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def compute_c5_fast(h, one_mh, n):
    dx = 2.0 / n
    corr = np.correlate(h, one_mh, mode='full')
    return dx * np.max(corr)

def fix_sum(h, target_sum):
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    h = h + diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    diff2 = target_sum - h.sum()
    if abs(diff2) > 1e-14:
        if diff2 > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff2
    return h
