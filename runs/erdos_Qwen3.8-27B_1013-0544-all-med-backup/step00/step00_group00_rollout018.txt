    c = np.correlate(h, 1 - h, mode="full")
    return np.max(c) * dx


def _lp_obj(x, dx):
    return x[-1]


def _lp_ineq(x, dx):
    n = len(x) - 1
    h = x[:n]
    t = x[-1]
    c = np.correlate(h, 1 - h, mode="full")
    return t - c * dx


def _lp_eq(x, n):
    return np.sum(x[:n]) - n / 2.0


def _lp_eq_jac(x, n):
    j = np.zeros(len(x))
    j[:n] = 1.0
    return j


def _project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        free = (h > 0.005) & (h < 0.995)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - np.sum(h)
            if abs(diff2) > 1e-12:
                free2 = (h > 0.001) & (h < 0.999)
                if np.sum(free2) > 0:
                    h[free2] += diff2 / np.sum(free2)
                    h = np.clip(h, 0.0, 1.0)
    return h


def _make_starts(init_vals, n, rng):
    dx = 2.0 / n
    starts = []

    if init_vals is not None:
        x_init = np.linspace(0, 2, len(init_vals))
        x_new = np.linspace(0, 2, n)
        h0 = np.interp(x_new, x_init, init_vals)
        h0 = _project_feasible(h0, n)
        t0 = _evaluate_c5(h0, dx) + 0.05
        starts.append(np.concatenate([h0, [t0]]))

    for i in range(8):
        h_r = rng.uniform(0, 1, n)
        h_r = h_r / max(h_r.sum(), 1e-10) * (n / 2.0)
        h_r = _project_feasible(h_r, n)
        t0 = _evaluate_c5(h_r, dx) + 0.05
        starts.append(np.concatenate([h_r, [t0]]))

    # Bump starts
    for sigma in [0.15, 0.25, 0.4]:
        x = np.linspace(0, 2, n)
        h_b = np.exp(-0.5 * ((x - 1.0) / sigma) ** 2)
        h_b = h_b / max(h_b.sum(), 1e-10) * (n / 2.0)
        h_b = _project_feasible(h_b, n)
        t0 = _evaluate_c5(h_b, dx) + 0.05
        starts.append(np.concatenate([h_b, [t0]]))

    # Plateau starts
    for frac in [0.3, 0.4, 0.5]:
        h_p = np.zeros(n)
        w = max(1, int(n * frac * 0.5))
        h_p[n // 2 - w: n // 2 + w] = 1.0
        h_p = _project_feasible(h_p, n)
        t0 = _evaluate_c5(h_p, dx) + 0.05
        starts.append(np.concatenate([h_p, [t0]]))

    return starts


def _optimize(h0, n, maxiter, ftol):
    dx = 2.0 / n
    t0 = _evaluate_c5(h0, dx) + 0.01
    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {"type": "ineq", "fun": _lp_ineq, "args": (dx,)},
        {"type": "eq", "fun": _lp_eq, "jac": _lp_eq_jac, "args": (n,)},
    ]
    res = minimize(_lp_obj, x0, args=(dx,), method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={"maxiter": maxiter, "ftol": ftol})
    if hasattr(res, "x"):
        return _project_feasible(res.x[:n], n)
    return None


def _optimize_direct(h0, n, maxiter, ftol):
    dx = 2.0 / n
    target = n / 2.0

    def obj(h):
        return _evaluate_c5(h, dx) + 1e4 * (np.sum(h) - target) ** 2

    x0 = h0.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _lp_eq, "jac": _lp_eq_jac, "args": (n,)}]

    def eq_wrapper(h):
        return np.sum(h) - target

    def eq_jac(h):
        return np.ones(n)

    cons2 = [{"type": "eq", "fun": eq_wrapper, "jac": eq_jac}]
    res = minimize(obj, x0, method="SLSQP", bounds=bounds,
                   constraints=cons2, options={"maxiter": maxiter, "ftol": ftol})
    if hasattr(res, "x"):
        return _project_feasible(res.x, n)
    return None

                h_sol = _optimize_direct(h_init, n, 200, 1e-8)
            if h_sol is not None:

                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_sol.copy()
                    best_n = n

        # Refinement phase on best at this n
        if best_h is not None and best_n == n and time.time() - t_start < budget_s - 20:
            for _ in range(4):
                if time.time() - t_start > budget_s - 15:
                    break
                h_ref = _optimize(best_h, n, 500, 1e-11)
                if h_ref is None:
                    h_ref = _optimize_direct(best_h, n, 400, 1e-11)
                if h_ref is not None:
                    c5 = _evaluate_c5(h_ref, dx)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_ref.copy()
                    else:
                        break

    # Final polish with perturbation restarts
    if best_h is not None and time.time() - t_start < budget_s - 20:
        n = best_n
        dx = 2.0 / n
        for _ in range(5):
            if time.time() - t_start > budget_s - 10:
                break
            scale = 0.05 / (1.0 + _ * 0.5)
            h_pert = best_h + rng.normal(0, scale, n)
            h_pert = _project_feasible(h_pert, n)
            h_sol = _optimize(h_pert, n, 400, 1e-10)
            if h_sol is not None:

                    best_c5 = c5
                    best_h = h_sol.copy()

        best_h = _project_feasible(np.ones(n) * 0.5, n)
        best_c5 = _evaluate_c5(best_h, 2.0 / n)
        best_n = n

    return (best_h, best_c5, best_n)
