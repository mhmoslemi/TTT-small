```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def get_all_overlaps(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_constraint(h, target_sum):
    """Project onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Bisection on additive shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    return result

def soft_max_gradient(h, n, beta):
    """Compute gradient of soft-max objective."""
    corr = get_all_overlaps(h)
    # Softmax weights
    shifted = beta * corr - np.max(beta * corr)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    # Gradient is weighted sum of individual gradients
    # c_s = sum_i h[i]*(1-h[i+s]) for valid i
    # dc_s/dh[i] = (1-h[i+s])*I(0<=i+s<n) - h[i-s]*I(0<=i-s<n)
    
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        w = weights[s + n - 1]
        if w < 1e-10:
            continue
        # Add w * dc_s/dh
        if s >= 0:
            # First part: (1-h[i+s]) for i = 0..n-1-s
            if n - s > 0:
                idx = slice(0, n - s)
                grad[idx] += w * (1.0 - h[idx + s])
            # Second part: -h[i-s] for i = s..n-1
            if s < n:
                idx2 = slice(s, n)
                grad[idx2] -= w * h[idx2 - s]
        else:
            a = -s
            # First part: (1-h[i+a]) for i = a..n-1  [since i+s >= 0 means i >= a]
            if n - a > 0:
                idx = slice(a, n)
                grad[idx] += w * (1.0 - h[idx + s])  # h[i - a]
            # Second part: -h[i+a] for i = 0..n-1-a  [since i-s < n means i < n-a]
            if n - a > 0:
                idx2 = slice(0, n - a)
                grad[idx2] -= w * h[idx2 + a]
    
    return grad * (2.0 / n)  # Scale by dx

def optimize_soft_max(h, n, target_sum, betas, steps_per_beta, base_lr):
    """Optimize using soft-max with annealing."""
    h = h.copy()
    for beta in betas:
        for step in range(steps_per_beta):
            lr = base_lr / (1.0 + step * 0.01)
            grad = soft_max_gradient(h, n, beta)
            h = h - lr * grad
            h = project_to_constraint(h, target_sum)
    return h
```

```python
    return np.correlate(h, 1.0 - h, mode="full")

def project(h, target_sum):
    """Project h onto {0<=h_i<=1, sum=h=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    lo, hi = -1.0, 1.0

        if np.clip(h + mid, 0, 1).sum() < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def softmax_obj_and_grad(h, n, beta):
    """Compute soft-max objective and its gradient."""
    corr = get_correlation(h)  # length 2n-1
    # c_k is the overlap at shift (k - n + 1)
    # We want to minimize max(c_k) * 2/n
    # Soft-max: (1/beta) * log(sum exp(beta * c_k))
    max_c = corr.max()
    exp_shifted = np.exp(beta * (corr - max_c))
    log_sum = max_c + np.log(exp_shifted.sum()) / beta  # This is the soft-max value
    
    # Weights for gradient
    w = exp_shifted / exp_shifted.sum()
    
    # Gradient: d(soft_max)/d(h[i]) = (2/n) * sum_k w_k * dc_k/dh[i]
    # dc_k/dh[i] where shift s = k - n + 1:
    # = (1-h[i+s])*I(0<=i+s<n) - h[i-s]*I(0<=i-s<n)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            # (1-h[i+s]) for i in [0, n-s-1]
            if n - s > 0:
                grad[:n-s] += wk * (1.0 - h[s:])
            # -h[i-s] for i in [s, n-1]
            if s < n:
                grad[s:] -= wk * h[:n-s]
        else:
            a = -s
            # (1-h[i+a]) for i in [a, n-1]  [i+s>=0 means i>=a]
            if n - a > 0:
                grad[a:] += wk * (1.0 - h[:n-a])
            # -h[i+a] for i in [0, n-a-1]  [i-s<n means i+a<n means i<n-a]
            if n - a > 0:
                grad[:n-a] -= wk * h[a:]
    
    grad *= (2.0 / n)
    return log_sum * (2.0 / n), grad
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0, 1).sum() < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_grad(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    e = np.exp(beta * (corr - m))
    w = e / e.sum()
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                grad_h[:n - s] += wk * (1.0 - h[s:])
                grad_h[s:] -= wk * h[:n - s]
        else:
            a = -s
            if a < n:
                grad_h[a:] += wk * (1.0 - h[:n - a])
                grad_h[:n - a] -= wk * h[a:]
    return (2.0 / n) * grad_h

def optimize_soft(h_start, n, ts, betas, steps_per_beta, lr_scale, deadline):
    h = h_start.copy()
    local_best = compute_c5(h)
    local_best_h = h.copy()
    for beta in betas:
        lr = lr_scale / (1.0 + 0.05 * beta)
        for _ in range(steps_per_beta):
            if time.time() > deadline:
                return local_best_h, local_best
            g = soft_max_grad(h, n, beta)
            h_new = project(h - lr * g, ts)
            c5 = compute_c5(h_new)
            if c5 < local_best:
                local_best = c5
                local_best_h = h_new.copy()
            h = h_new
    return local_best_h, local_best

def binary_local_search(h_start, n, ts, deadline):
    """Local search on near-binary patterns via pair swaps."""
    h = np.round(h_start)
    h = np.clip(h, 0, 1)
    # Ensure exact sum
    diff = ts - h.sum()
    if abs(diff) > 1e-10:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if diff > 0:
            for idx in zeros[:int(np.ceil(diff))]:
                h[idx] = 1.0
        else:
            for idx in ones[:int(np.ceil(-diff))]:
                h[idx] = 0.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    no_improve = 0
    while time.time() < deadline and no_improve < 500:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_improve = 0
        else:
            h[i], h[j] = 1.0, 0.0
            no_improve += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(48, 0.5)

    n = len(h0)
    ts = n / 2.0
    h0 = project(h0, ts)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Soft-max gradient descent on initial grid
    dl = t0 + budget_s * 0.30
    h1, c5_1 = optimize_soft(h0, n, ts, [0.5, 1, 2, 5, 10, 20, 50, 100, 200], 400, 1.0, dl)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    # Phase 2: Fine-tune with high beta
    dl = t0 + budget_s * 0.45
    h2, c5_2 = optimize_soft(best_h, n, ts, [50, 100, 200, 500, 1000], 800, 0.3, dl)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2

    # Phase 3: Binary local search on even grid
    n_even = n if n % 2 == 0 else n + 1
    if n_even != n:
        ts_even = n_even / 2.0
        x_old = np.linspace(0, 2, n + 1)
        x_new = np.linspace(0, 2, n_even + 1)
        h_bi = np.interp(x_new[:-1], x_old, np.append(best_h, best_h[-1]))
        h_bi = np.round(h_bi)
        h_bi = np.clip(h_bi, 0, 1)
        h_bi = project(h_bi, ts_even)
    else:
        ts_even = ts
        h_bi = np.round(best_h)
        h_bi = np.clip(h_bi, 0, 1)
        h_bi = project(h_bi, ts_even)

    dl = t0 + budget_s * 0.55
    h_bi_opt, c5_bi = binary_local_search(h_bi, n_even, ts_even, dl)
    if c5_bi < best_c5:
        best_c5, best_h = c5_bi, h_bi_opt

    # Phase 4: Upsampled grids with soft-max optimization
    for factor in [2, 3, 4]:
        dl = t0 + budget_s * 0.75
        if time.time() > dl:
            break
        n_new = len(best_h) * factor
        ts_new = n_new / 2.0
        x_old = np.linspace(0, 2, len(best_h) + 1)
        x_new = np.linspace(0, 2, n_new + 1)
        h_up = np.interp(x_new[:-1], x_old, np.append(best_h, best_h[-1]))
        h_up = project(h_up, ts_new)
        h3, c5_3 = optimize_soft(h_up, n_new, ts_new, [1, 5, 10, 20, 50, 100, 200], 300, 0.5, dl)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3

    # Phase 5: Restart loop with diverse initializations
    n_b = len(best_h)
    ts_b = n_b / 2.0
    dl = t0 + budget_s * 0.96
    while time.time() < dl:
        strategy = rng.integers(0, 4)
        if strategy == 0:
            h_r = best_h + rng.standard_normal(n_b) * 0.05
        elif strategy == 1:
            h_r = best_h + rng.standard_normal(n_b) * 0.2
        elif strategy == 2:
            h_r = np.zeros(n_b)
            idx = rng.choice(n_b, size=int(round(ts_b)), replace=False)
            h_r[idx] = 1.0
        else:
            # Semi-random: mix of 0, 0.5, 1
            vals = rng.choice([0.0, 0.5, 1.0], size=n_b, p=[0.4, 0.2, 0.4])
            h_r = vals + rng.standard_normal(n_b) * 0.1

        h_r = project(h_r, ts_b)
        h4, c5_4 = optimize_soft(h_r, n_b, ts_b, [2, 10, 30, 100, 300], 400, 0.5, dl)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4

    # Final: ensure valid return
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
```
