```python
def smooth_c5_and_grad_fast(h, beta=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    max_c = np.max(corr)
    exps = np.exp(beta * (corr - max_c))
    sum_exps = np.sum(exps)
    obj = (max_c + np.log(sum_exps) / beta) * dx
    weights = exps / sum_exps
    
    # Build gradient matrix M where M[j,k] = (1-h[j+lag_k]) - h[j-lag_k]
    # lag_k = k - n + 1 for k = 0, ..., 2n-2
    lags = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    
    # For each j (0..n-1) and each lag d (-(n-1)..(n-1)):
    # M[j, d+n-1] = (1 - h[j+d]) - h[j-d]
    # where h[i]=0 if i outside [0,n-1]
    
    # Create index arrays
    j_idx = np.arange(n)[:, None]  # (n, 1)
    d_idx = lags[None, :]  # (1, 2n-1)
    
    idx1 = j_idx + d_idx  # (n, 2n-1): j+d
    idx2 = j_idx - d_idx  # (n, 2n-1): j-d
    
    # Get h values with 0 padding
    h1 = np.where((idx1 >= 0) & (idx1 < n), h[np.clip(idx1, 0, n-1)], 0.0)
    h2 = np.where((idx2 >= 0) & (idx2 < n), h[np.clip(idx2, 0, n-1)], 0.0)
    
    M = (1.0 - h1) - h2  # shape (n, 2n-1)
    
    grad = M @ (weights * dx)
    
    return obj, grad
```

```python
    exps = np.exp(beta * (corr - max_c))
    sum_exps = np.sum(exps)
    obj = (max_c + np.log(sum_exps) / beta) * dx
    weights = exps / sum_exps

    lags = np.arange(2 * n - 1) - (n - 1)
    j_idx = np.arange(n)[:, None]
    d_idx = lags[None, :]
    idx1 = j_idx + d_idx
    idx2 = j_idx - d_idx
    h1 = np.where((idx1 >= 0) & (idx1 < n), h[np.clip(idx1, 0, n - 1)], 0.0)
    h2 = np.where((idx2 >= 0) & (idx2 < n), h[np.clip(idx2, 0, n - 1)], 0.0)
    M = (1.0 - h1) - h2
    grad = M @ (weights * dx)
    return obj, grad

def optimize_projected_gradient(h0, max_time=60.0, beta=50.0):
    """Project gradient descent with adaptive step size."""
    n = len(h0)
    target = n / 2.0
    h = project_box_sum(h0.astype(float).copy(), target)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 0.05
    start = time.time()
    it = 0
    while time.time() - start < max_time:
        obj, grad = smooth_c5_and_grad(h, beta)
        h_new = project_box_sum(h - lr * grad, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            lr *= 0.85
        if lr < 1e-6:
            break
        it += 1
        if it % 200 == 0:
            beta = max(beta * 0.95, 5.0)
    return best_h, best_c5

def optimize_binary_local(h0, max_time=30.0):
    """Local search on binary solutions (flip moves)."""
    n = len(h0)
    h = np.zeros(n)
    ones = np.where(h0 >= 0.5)[0]
    h[ones] = 1.0
    if h.sum() != n / 2:
        zeros = np.where(h == 0)[0]
        need = int(n / 2) - len(ones)
        if need > 0 and len(zeros) >= need:
            h[zeros[:need]] = 1.0
        elif need < 0:
            h[ones[need:]] = 0.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    rng = np.random.default_rng(12345)
    start = time.time()
    temp = 0.01
    while time.time() - start < max_time and temp > 0.001:
        # Pick two positions: one with 1, one with 0, swap
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = ones_idx[rng.integers(len(ones_idx))]
        j = zeros_idx[rng.integers(len(zeros_idx))]
        h[i], h[j] = 0.0, 1.0
        c5_new = compute_c5(h)
        delta = c5_new - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            best_c5 = min(best_c5, c5_new)
            best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        if it % 500 == 0:
            temp *= 0.98
        it += 1
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    rng = np.random.default_rng(seed)
    t0 = time.time()
    best_c5 = 0.5
    best_h = None
    best_n = 0

    # Phase 1: Optimize from initial_h_values
    if 'initial_h_values' in dir() or True:
        try:
            h_init = np.array(initial_h_values, dtype=float)
        except:
            h_init = None
        if h_init is not None:
            n0 = len(h_init)
            remaining = budget_s - (time.time() - t0)
            h_opt, c5_opt = optimize_projected_gradient(h_init, max_time=remaining * 0.4, beta=50.0)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n0
            # Binary local search from the optimized solution
            remaining = budget_s - (time.time() - t0)
            if remaining > 30:
                h_bin, c5_bin = optimize_binary_local(h_opt, max_time=min(remaining * 0.3, 60.0))
                if c5_bin < best_c5:
                    best_c5 = c5_bin
                    best_h = h_bin.copy()
                    best_n = n0
            # Further refinement with lower beta
            remaining = budget_s - (time.time() - t0)
            if remaining > 30:
                h_opt2, c5_opt2 = optimize_projected_gradient(h_opt, max_time=remaining * 0.2, beta=20.0)
                if c5_opt2 < best_c5:
                    best_c5 = c5_opt2
                    best_h = h_opt2.copy()
                    best_n = n0

    # Phase 2: Try different n values with various initializations
    n_candidates = [64, 80, 96, 128, 160, 192, 256, 320, 384, 512]
    remaining = budget_s - (time.time() - t0)
    time_per_n = max(remaining / len(n_candidates) * 0.8, 5.0)

    for n in n_candidates:
        if time.time() - t0 > budget_s - 10:
            break
        target = n / 2.0
        # Multiple initializations
        inits = []
        # Binary random
        h_rand = np.zeros(n)
        perm = rng.permutation(n)
        h_rand[perm[:n//2]] = 1.0
        inits.append(h_rand)
        # Interpolated from best if available
        if best_h is not None:
            x_old = np.linspace(0, 2, len(best_h), endpoint=False) + 1.0/len(best_h)
            x_new = np.linspace(0, 2, n, endpoint=False) + 1.0/n
            h_interp = np.interp(x_new, x_old, best_h, period=2.0)
            inits.append(h_interp)
        # Block pattern
        h_block = np.zeros(n)
        block_size = max(1, n // 8)
        for b in range(8):
            start_b = b * (n // 8)
            h_block[start_b:start_b + block_size//2] = 1.0
        inits.append(h_block)

        for h_init in inits:
            if time.time() - t0 > budget_s - 10:
                break
            h_init_p = project_box_sum(h_init, target)
            h_opt, c5_opt = optimize_projected_gradient(h_init_p, max_time=min(time_per_n/2, 30.0), beta=50.0)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
                remaining = budget_s - (time.time() - t0)
                time_per_n = max(remaining / max(len(n_candidates) - n_candidates.index(n) - 1, 1) * 0.7, 5.0)

    # Phase 3: Final refinement on the best solution
    remaining = budget_s - (time.time() - t0)
    if best_h is not None and remaining > 30:
        h_final, c5_final = optimize_projected_gradient(best_h, max_time=remaining * 0.5, beta=30.0)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
        # One more pass with even lower beta
        remaining = budget_s - (time.time() - t0)
        if remaining > 20:
            h_final2, c5_final2 = optimize_projected_gradient(best_h, max_time=remaining * 0.5, beta=10.0)
            if c5_final2 < best_c5:
                best_c5 = c5_final2
                best_h = h_final2

    # Ensure final solution is valid
    if best_h is None:
        best_h = np.zeros(95)
        best_h[:47] = 1.0
        best_n = 95
        best_c5 = compute_c5(best_h)

    final_n = len(best_h)
    final_target = final_n / 2.0
    best_h = project_box_sum(best_h, final_target)
    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, final_n)
```
