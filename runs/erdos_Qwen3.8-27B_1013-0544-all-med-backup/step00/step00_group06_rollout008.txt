import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return float(np.max(corr)), corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if 0 <= k < n:
        g[:n-k] += 1.0 - h[k:]
        g[k:] += -h[:n-k]
    elif -n < k < 0:
        m = -k
        g[m:] += 1.0 - h[:n-m]
        g[:n-m] += -h[m:]
    return g

def optimize_subgrad(h, n, budget_s, lr0=0.005, top_m=5):
    dx = 2.0 / n
    t0 = time.time()
    best_c5, _ = evaluate_c5(h)
    best_h = h.copy()
    velocity = np.zeros(n)
    beta = 0.9
    
    while time.time() - t0 < budget_s:
        corr = np.correlate(h, 1 - h, mode="full") * dx
        top = np.argsort(corr)[-top_m:]
        g = np.zeros(n)
        for idx in top:
            k = idx - (n - 1)
            g += grad_lag(h, k)
        g *= dx
        
        lr = lr0 / (1.0 + 0.01 * (time.time() - t0))
        velocity = beta * velocity + (1 - beta) * g
        h_new = h - lr * velocity
        h_new = project_feasible(h_new, n)
        
        c5, _ = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

    return float(np.max(corr)), corr

        if cnt == 0:
            break

        h = np.clip(h, 0.0, 1.0)

    n = len(h)

    if 0 <= k < n:
        g[:n - k] += 1.0 - h[k:]
        g[k:] += -h[:n - k]
    elif -n < k < 0:

        rem = T - (time.time() - t_start)
        if rem > 30:
            h2, c5b = optimize_coords(h2, n2, rem - 15)
            if c5b < best_c5:
                best_c5, best_h = c5b, h2.copy()

    # Final: one more coordinate pass at best resolution
    rem = T - (time.time() - t_start)
    if rem > 20:
        nf = len(best_h)
        hf, cf = optimize_coords(best_h.copy(), nf, rem - 10)
        if cf < best_c5:
            best_c5, best_h = cf, hf.copy()

    best_h = project_feasible(best_h, len(best_h))
    final_c5, _ = evaluate_c5(best_h)
    return best_h, final_c5, len(best_h)
