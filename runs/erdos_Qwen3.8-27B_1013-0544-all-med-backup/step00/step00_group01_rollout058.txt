def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def fix_ones(h, n, rng):

    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones != target:

            ones_idx = np.where(h == 1)[0]

            h[to_flip] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            to_flip = rng.choice(zeros_idx, size=target - n_ones, replace=False)

    return h

    best_h = h.copy()

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_weight(h, n, rng):
    """Ensure 0-1 vector has exactly n/2 ones."""
    target = n // 2
    n_ones = int(np.sum(h == 1))
    if n_ones > target:
        ones = np.where(h == 1)[0]
        h[rng.choice(ones, n_ones - target, replace=False)] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        h[rng.choice(zeros, target - n_ones, replace=False)] = 1
    return h

def sa_optimize(h, n, rng, time_limit):
    """Simulated annealing for 0-1 swap optimization."""
    t0 = time.time()
    c5 = compute_c5(h)
    best_c5, best_h = c5, h.copy()
    T = 0.05
    cool = 0.9997
    
    while time.time() - t0 < time_limit and T > 1e-5:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i], h[j] = 0, 1
        nc5 = compute_c5(h)
        d = nc5 - c5
        if d < 0 or rng.random() < np.exp(-d / T):
            c5 = nc5
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
        else:
            h[i], h[j] = 1, 0
        T *= cool
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_weight(h, n, rng):
    target = n // 2
    n_ones = int(np.sum(h == 1))
    if n_ones > target:
        ones = np.where(h == 1)[0]
        h[rng.choice(ones, n_ones - target, replace=False)] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        h[rng.choice(zeros, target - n_ones, replace=False)] = 1
    return h

def thue_morse_init(n):
    return np.array([(bin(i).count('1') & 1) for i in range(n)], dtype=float)

def sa_optimize(h, n, rng, time_limit):
    t0 = time.time()
    c5 = compute_c5(h)
    best_c5, best_h = c5, h.copy()
    T = 0.04
    cool = 0.9996
    while time.time() - t0 < time_limit and T > 1e-5:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i], h[j] = 0, 1
        nc5 = compute_c5(h)
        d = nc5 - c5
        if d < 0 or rng.random() < np.exp(-d / T):
            c5 = nc5
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
        else:
            h[i], h[j] = 1, 0
        T *= cool
    return best_h, best_c5

def greedy_search(h, n, time_limit):
    t0 = time.time()
    c5 = compute_c5(h)
    improved = True
    while improved and time.time() - t0 < time_limit:
        improved = False
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        for i in ones:
            if time.time() - t0 > time_limit:
                return h, c5
            best_j = -1
            best_val = c5
            for j in zeros:
                if time.time() - t0 > time_limit:
                    break
                h[i], h[j] = 0, 1
                nc5 = compute_c5(h)
                h[i], h[j] = 1, 0
                if nc5 < best_val:
                    best_val = nc5
                    best_j = j
            if best_j >= 0 and best_val < c5 - 1e-15:
                h[i], h[best_j] = 0, 1
                c5 = best_val
                improved = True
                break
    return h, c5

def two_swap_search(h, n, rng, time_limit):
    t0 = time.time()
    c5 = compute_c5(h)
    best_c5, best_h = c5, h.copy()
    while time.time() - t0 < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i1 = ones[rng.integers(len(ones))]
        i2 = ones[rng.integers(len(ones))]
        j1 = zeros[rng.integers(len(zeros))]
        j2 = zeros[rng.integers(len(zeros))]
        if i1 == i2 or j1 == j2:
            continue
        h[i1], h[j1] = 0, 1
        h[i2], h[j2] = 0, 1
        nc5 = compute_c5(h)
        if nc5 < best_c5:
            best_c5, best_h = nc5, h.copy()
        else:
            h[i1], h[j1] = 1, 0
            h[i2], h[j2] = 1, 0
    return best_h, best_c5
