```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float((corr * dx).max())

def get_correlation(h):
    return np.correlate(h, 1 - h, mode="full")

def smooth_objective(h, temp, dx):
    corr = get_correlation(h) * dx
    c = corr / temp
    c_shift = c - c.max()
    return temp * (np.log(np.sum(np.exp(c_shift))) + c.max())

def smooth_gradient(h, temp, dx):
    n = len(h)
    corr = get_correlation(h)
    c = corr * dx / temp
    c_shift = c - c.max()
    exp_c = np.exp(c_shift)
    S = exp_c.sum()
    w = exp_c / S
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        i_lo = max(0, n - 1 - k)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            j = i + k - (n - 1)
            grad[j] += w[k] * (1.0 - h[i])
            grad[i] += -w[k] * h[j]
    return dx * grad

def project_h(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection on threshold
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    return np.clip(h - t, 0.0, 1.0)

    t_start = time.time()

        return time.time() - t_start

        h = np.array(initial_h_values, dtype=float)
    else:

    best_h = h.copy()
    best_c5 = evaluate(h)

        nonlocal best_h, best_c5
        h_new = project_h(h_new, n)
        c5 = evaluate(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
        return h_new
    
    # Phase 1: Log-sum-exp optimization with decreasing temperature
    h_curr = h.copy()
    for temp in [0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]:
```

- h[j] appears in the term where k-i=j, i.e., i=k-j: contributes -h[k-j] if 0≤k-j<n

When only j is valid but k-j is not: the term where i=j contributes (1-h[k-j]) but k-j is out of range, so h[k-j] doesn't exist. In this case, the term h[j]*(1-h[k-j]) doesn't exist in the sum (because k-j is out of range). So no contribution.

So: grad[j] = sum_{m=0}^{n-1} w[j+m] * (1-2*h[m])
