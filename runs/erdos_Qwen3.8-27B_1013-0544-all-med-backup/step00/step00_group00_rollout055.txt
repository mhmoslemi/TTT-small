import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_objective(h, n, penalty):
    c5 = compute_c5(h)
    sv = np.sum(h) - n / 2.0
    return c5 + penalty * sv * sv

def smooth_objective(h, n, penalty, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    logsum = m + temp * np.log(np.sum(np.exp((corr - m) / temp)))
    sv = np.sum(h) - n / 2.0
    return logsum * dx + penalty * sv * sv

def optimize_from(h0, n, penalty, temps, max_iter, bounds):
    h = h0.copy()
    for temp in temps:
        res = minimize(smooth_objective, h, args=(n, penalty, temp),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        h = np.clip(res.x, 0, 1)
    res = minimize(c5_objective, h, args=(n, penalty),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': 500, 'ftol': 1e-12})
    h = np.clip(res.x, 0, 1)
    return h

def local_search(h, n, num_passes=5):
    h = h.copy()
    current = compute_c5(h)
    for _ in range(num_passes):
        improved = False
        for i in range(n):
            for delta in [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]:
                h_try = h.copy()
                new_val = h[i] + delta
                if 0 <= new_val <= 1:
                    h_try[i] = new_val
                    c5_try = compute_c5(h_try)
                    if c5_try < current - 1e-12:
                        h = h_try
                        current = c5_try
                        improved = True
        if not improved:
            break
    return h
