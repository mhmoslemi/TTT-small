import numpy as np

def lag_gradient(h, k):
    n = len(h)
    g = np.zeros(n)
    js = max(0, -k)
    je = min(n, n - k)
    if js < je:
        g[js:je] += 1.0 - h[js + k:je + k]
    js = max(0, k)
    je = min(n, n + k)
    if js < je:
        g[js:je] -= h[js - k:je - k]
    return g

def compute_obj(h, n):
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def compute_smooth_grad(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    log_w = (corr - cmax) / tau
    log_w -= np.max(log_w)
    w = np.exp(log_w)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] > 1e-12:
            k = m - (n - 1)
            grad += w[m] * lag_gradient(h, k)
    return grad * dx

def compute_subgrad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    active = np.where(np.abs(corr - max_val) < 1e-8)[0]
    grad = np.zeros(n)
    for m in active:
        k = m - (n - 1)
        grad += lag_gradient(h, k)
    grad /= len(active)
    return grad * dx
