    mask2 = (idx2 >= 0) & (idx2 < n)

corr_k = sum_i h[i+k'] * (1-h[i]) = sum_i h[i+k'] - sum_i h[i+k'] * h[i]

If I fix the values h[i+k'] (the "shifted" version), then corr_k is linear in h.

This suggests an iterative approach:
1. Start with current h
2. Compute all corr_k values
3. For the active constraints (where corr_k ≈ max), write them as linear constraints using the current h values for the "fixed" part
4. Solve the resulting LP
5. Update h
6. Repeat

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")


def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))


def grad_corr_k(h, k_lag):
    n = len(h)
    j = np.arange(n)
    grad = np.zeros(n)
    idx1 = j - k_lag
    mask1 = (idx1 >= 0) & (idx1 < n)
    grad[mask1] += (1.0 - h[idx1[mask1]])
    idx2 = j + k_lag
    mask2 = (idx2 >= 0) & (idx2 < n)
    grad[mask2] -= h[idx2[mask2]]
    return grad


def soft_max_obj(x, temp):
    n = len(x)
    dx = 2.0 / n
    corr = compute_corr(x) * dx
    max_c = np.max(corr)
    return max_c + temp * np.log(np.sum(np.exp((corr - max_c) / temp)))


def soft_max_grad(x, temp):
    n = len(x)
    dx = 2.0 / n
    corr = compute_corr(x)
    cs = corr * dx
    max_s = np.max(cs)
    p = np.exp((cs - max_s) / temp)
    p /= np.sum(p)
    grad = np.zeros(n)
    for k_idx in range(len(cs)):
        if p[k_idx] > 1e-8:
            k_lag = k_idx - (n - 1)
            grad += p[k_idx] * grad_corr_k(x, k_lag)
    grad *= dx
    return grad


def exact_max_obj(x, dummy):
    return compute_c5(x)


def exact_max_grad(x, dummy):

    dx = 2.0 / n

    max_val = np.max(corr)

    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    for k_idx in active:
        k_lag = k_idx - (n - 1)
        grad += grad_corr_k(x, k_lag)
    grad *= dx / len(active)
    return grad


def constraint_sum(x):
    return np.sum(x) - len(x) / 2.0


def constraint_sum_jac(x):
    return np.ones(len(x))


def make_constraints():
    return [{'type': 'eq', 'fun': constraint_sum, 'jac': constraint_sum_jac}]


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        free = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
        if len(free) > 0:
            h[free[0]] += residual
            h = np.clip(h, 0.0, 1.0)
    return h

    res = minimize(obj_func, x0, args=(temp,), jac=grad_func,
                   method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return res
