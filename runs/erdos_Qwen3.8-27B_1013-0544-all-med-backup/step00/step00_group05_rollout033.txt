import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute C5 = max_k correlate(h, 1-h, 'full')[k] * dx"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_all_g(h, dx):
    """Compute all correlation values * dx"""
    corr = np.correlate(h, 1 - h, mode='full')
    return corr * dx

def project_onto_constraint(v, target_sum):
    """Project v onto {h: 0 <= h[i] <= 1, sum(h) = target_sum}"""
    # h[i] = clip(v[i] - lambda, 0, 1)
    # Find lambda such that sum(h) = target_sum
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(v - mid, 0, 1)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2, 0, 1)
    return h

def subgradient_step(h, dx, step_size):
    """One subgradient descent step"""
    n = len(h)
    g = compute_all_g(h, dx)
    k_star = np.argmax(g)
    
    # Compute gradient of g_{k_star}
    shift = k_star - (n - 1)  # actual shift
    grad = np.zeros(n)
    
    if shift >= 0:
        # g = dx * sum_{i=0}^{n-1-shift} h[i] * (1-h[i+shift])
        for i in range(n - shift):
            grad[i] += (1 - h[i + shift]) * dx
            grad[i + shift] += (-h[i]) * dx
    else:
        s = -shift
        # g = dx * sum_{i=s}^{n-1} h[i] * (1-h[i-s])
        for i in range(s, n):
            grad[i] += (1 - h[i - s]) * dx
            grad[i - s] += (-h[i]) * dx
    
    h_new = h - step_size * grad
    return project_onto_constraint(h_new, n / 2), g[k_star]
