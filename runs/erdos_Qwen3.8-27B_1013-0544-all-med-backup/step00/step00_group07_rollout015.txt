```python
import numpy as np
import time

def compute_c5_values(h):
    """Compute all correlation values."""
    return np.correlate(h, 1-h, mode="full")

def max_c5(h, dx):
    """Compute C5 bound."""
    c = compute_c5_values(h)
    return np.max(c) * dx

def round_to_binary(h, n):
    """Round h to binary with exactly n/2 ones."""
    h = h.copy()
    # Sort by value, set top n/2 to 1, rest to 0
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n//2]] = 1.0
    return h_bin

def local_search(h, n, dx, max_iters=100000, rng=None):
    """Iterated local search for binary h."""
    if rng is None:
        rng = np.random.RandomState(42)
    
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    best_c = np.max(np.correlate(h, 1-h, mode="full"))
    best_h = h.copy()
    
    for it in range(max_iters):
        # Pick random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Try the swap
        h[i] = 0
        h[j] = 1
        
        new_c = np.max(np.correlate(h, 1-h, mode="full"))
        
        if new_c <= best_c:
            best_c = new_c
            best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            # Revert
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c * dx

def simulated_annealing(h, n, dx, iterations=500000, T_start=1.0, T_end=0.01, rng=None):
    """Simulated annealing for binary h."""
    if rng is None:
        rng = np.random.RandomState(42)
    
    h = h.copy()
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    best_c = np.max(np.correlate(h, 1-h, mode="full"))
    best_h = h.copy()
    current_c = best_c
    
    for it in range(iterations):
        T = T_start * (T_end / T_start) ** (it / iterations)
        
        # Pick random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        
        new_c = np.max(np.correlate(h, 1-h, mode="full"))
        delta = new_c - current_c
        
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
            # Update sets
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
```
