def projected_step(h, grad, step_size, n):
    """Take a gradient step and project onto feasible set."""
    h_new = h - step_size * grad
    # Project onto {0 <= h <= 1, sum(h) = n/2}
    # Use bisection on lambda
    target = n / 2.0
    
    def project(lam):
        return np.clip(h_new - lam, 0, 1)

    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(project(mid))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = project((lo + hi) / 2)
    return h_proj

def subgradient(h, n):

    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_val = np.max(corr)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def make_binary(h, rng):
    n = len(h)
    h_bin = (h > 0.5).astype(np.float64)
    target = n // 2
    num_ones = int(h_bin.sum())
    if num_ones > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(ones, num_ones - target, replace=False)
        h_bin[idx] = 0
    elif num_ones < target:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(zeros, target - num_ones, replace=False)
        h_bin[idx] = 1
    return h_bin

def steepest_descent(h, max_rounds=200):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    for _ in range(max_rounds):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_delta = 0
        best_j, best_k = -1, -1
        for j in ones:
            h[j] = 0
            for k in zeros:
                h[k] = 1
                c = compute_c5(h)
                d = c - c5
                if d < best_delta - 1e-15:
                    best_delta = d
                    best_j, best_k = j, k
                h[k] = 0
            h[j] = 1
        if best_j < 0:
            break
        h[best_j] = 0
        h[best_k] = 1
        c5 += best_delta
    return h, c5

def simulated_annealing(h, rng, t_start, max_iters, time_fn):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)

    best_h = h.copy()
    temp = t_start

            break

        k = rng.integers(n)
        if j == k:
            continue
        h[j], h[k] = h[k], h[j]
        new_c5 = compute_c5(h)
        diff = new_c5 - c5
        if diff < 0 or rng.random() < np.exp(-diff / max(temp, 1e-15)):

                best_h = h.copy()

            h[j], h[k] = h[k], h[j]
