import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_grad(h, t):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = float(np.max(corr))
    w = np.exp(t * (corr - mc))
    sw = float(np.sum(w))
    if sw < 1e-300:
        return np.zeros(n), mc
    w /= sw
    grad = np.zeros(n)
    shift = n - 1
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-12:
            continue
        k = m - shift
        a1 = max(0, k)
        b1 = min(n, n + k)
        if a1 < b1:
            idx = np.arange(a1, b1)
            grad[idx] += wm * (1.0 - h[idx - k])
        a2 = max(0, -k)
        b2 = min(n, n - k)
        if a2 < b2:
            idx = np.arange(a2, b2)
            grad[idx] -= wm * h[idx + k]
    return grad, mc

def gradient_descent(h_init, seed, budget, t_init=40.0):
    n = len(h_init)

    target = float(n) / 2.0
    h = project_feasible(h_init, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    t = t_init
    lr = 0.02
    stale = 0
    while time.time() - start < budget * 0.95:
        grad, _ = smooth_grad(h, t)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-10:
            break
        alpha = lr
        improved = False
        for trial in range(12):
            h_new = project_feasible(h - alpha * grad, target)
            c5_new = compute_c5(h_new)
            if c5_new < cur_c5 - 1e-12:
                improved = True
                break
            alpha *= 0.5
        if improved:
            h = h_new
            cur_c5 = c5_new
            stale = 0
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            if trial == 0:
                lr = min(lr * 1.04, 0.15)
        else:
            stale += 1
            lr *= 0.85
            t = min(t * 1.04, 500.0)

                stale = 0
                lr = 0.025
                t = t_init
            elif stale > 600:
                break
        if lr < 5e-8:
            break
    return best_h, best_c5

def coordinate_polish(h, budget):
    n = len(h)
    target = float(n) / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()

                    best_h = h_try
                    best_c5 = c5_try
                    any_improve = True
                    break
