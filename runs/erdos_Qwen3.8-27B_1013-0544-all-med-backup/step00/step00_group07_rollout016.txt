import numpy as np
import time

def compute_c5(h):
    """Compute C5 = max_k correlate(h, 1-h, mode='full') * dx."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c.max(), c

def to_binary(h):
    """Convert to binary, keeping top n/2 values as 1."""
    n = len(h)
    target = n // 2
    idx = np.argsort(-h, kind='stable')
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

    """Interpolate h from its current length to new_n."""
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False) + 1.0/old_n  # midpoints
    x_new = np.linspace(0, 2, new_n, endpoint=False) + 1.0/new_n
    return np.interp(x_new, x_old, h)

    """Create alternating blocks of 1s and 0s."""
    # Create pattern: block_size 1s, block_size 0s, repeat
    unit = np.zeros(2 * block_size)
    unit[:block_size] = 1.0
    repeats = n // (2 * block_size) + 1
    h = np.tile(unit, repeats)[:n]
    # Fix count: ensure exactly n//2 ones
    target = n // 2

        h[ones_idx[target:]] = 0
    elif current_ones < target:
        # Add ones (from zero positions)
        zeros_idx = np.where(h == 0)[0]
        h[zeros_idx[:target - current_ones]] = 1
    return h

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c.max(), c

def to_binary(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(-h, kind="stable")
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def block_pattern(n, block_size):
    if block_size * 2 > n:
        block_size = max(1, n // 2 - 1)
    unit = np.zeros(2 * block_size)
    unit[:block_size] = 1.0
    h = np.tile(unit, n // (2 * block_size) + 1)[:n]
    target = n // 2
    cur = int(h.sum())
    if cur > target:
        idx = np.where(h == 1)[0]
        h[idx[target:]] = 0
    elif cur < target:
        idx = np.where(h == 0)[0]
        h[idx[:target - cur]] = 1
    return h

def local_search_binary(h, rng, time_limit=30):

    h = h.copy()

        zeros = np.where(h == 0)[0]

            for j in zeros:
                h[i], h[j] = h[j], h[i]
                c5, _ = compute_c5(h)
                h[i], h[j] = h[j], h[i]
