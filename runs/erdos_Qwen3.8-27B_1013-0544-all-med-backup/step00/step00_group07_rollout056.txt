import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 0.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return h

def smooth_logsumexp(corr, dx, T):
    x = corr * dx / T
    m = np.max(x)
    return T * (m + np.log(np.sum(np.exp(x - m))))

def compute_grad(h, dx, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    x = corr * dx / T
    m = np.max(x)
    w = np.exp(x - m)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    # For each j, compute gradient
    # grad[j] = dx * [sum_s w[s+N-1] * (1-h[j-s]) * 1{0<=j-s<N} 
    #                  - sum_s w[s+N-1] * h[j+s] * 1{0<=j+s<N}]
    
    # Efficient computation using the relationship:
    # First term for j: sum_{t=0}^{n-1} w[j-t+n-1] * (1-h[t]) where 0<=j-t+n-1<=2n-2
    # Second term for j: sum_{t=0}^{n-1} w[t-j+n-1] * h[t] where 0<=t-j+n-1<=2n-2
    
    # I'll use a direct loop for clarity (n is at most ~640)
    for j in range(n):

        # In terms of w index: w_idx = s + n - 1, so j-n+1+n-1 <= w_idx <= j+n-1
        # = j <= w_idx <= j+n-1, clipped to [0, 2n-2]

        w_hi = min(2*n-2, j+n-1)
        if w_lo <= w_hi:
            # s = w_idx - (n-1), and h index = j - s = j - w_idx + n - 1
            h_idx = j + n - 1 - np.arange(w_lo, w_hi+1)
            # h_idx should be in [0, n-1]
            valid = (h_idx >= 0) & (h_idx < n)
            grad[j] += np.sum(w[w_lo:w_hi+1][valid] * (1 - h[h_idx[valid]]))
        
        # Second term: sum over s where 0<=j+s<n, i.e., -j <= s <= n-1-j

        w_hi2 = min(2*n-2, 2*n-2-j)

            # h_idx2 should be in [0, n-1]

            grad[j] -= np.sum(w[w_lo2:w_hi2+1][valid2] * h[h_idx2[valid2]])
    
    return grad * dx

import numpy as np

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 0.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return h

def compute_grad(h, dx, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    x = corr * dx / T
    m = np.max(x)
    w = np.exp(x - m)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for j in range(n):
        # First term: sum_s w[s+n-1]*(1-h[j-s]) for 0<=j-s<n
        w_lo = max(0, j)
        w_hi = min(2 * n - 2, j + n - 1)
        if w_lo <= w_hi:
            h_idx = j + n - 1 - np.arange(w_lo, w_hi + 1)
            valid = (h_idx >= 0) & (h_idx < n)
            grad[j] += np.sum(w[w_lo:w_hi + 1][valid] * (1 - h[h_idx[valid]]))
        # Second term: -sum_s w[s+n-1]*h[j+s] for 0<=j+s<n
        w_lo2 = max(0, n - 1 - j)
        w_hi2 = min(2 * n - 2, 2 * n - 2 - j)
        if w_lo2 <= w_hi2:
            h_idx2 = np.arange(w_lo2, w_hi2 + 1) - (n - 1) + j
            valid2 = (h_idx2 >= 0) & (h_idx2 < n)
            grad[j] -= np.sum(w[w_lo2:w_hi2 + 1][valid2] * h[h_idx2[valid2]])
    return grad * dx

    while time.time() - start < time_limit:
        grad = compute_grad(h, dx, T)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        val = evaluate_c5(h_new)
        
        if val < best_val:
            best_val = val
            best = h_new.copy()
            stall = 0
            lr = min(1.2, lr * 1.03)
        else:
            stall += 1
            lr = max(0.05, lr * 0.93)
        
        h = h_new
        T = max(T_min, T * 0.997)
        
        if stall > 80:
            # Random restart from best with perturbation
            kick = rng.normal(0, 0.15, size=n)
            h_pert = best + kick
            h_pert = np.clip(h_pert, 0.0, 1.0)
            h_pert = fix_sum(h_pert, n)

            if val_p < best_val:

                best = h_pert.copy()
                h = h_pert.copy()
            else:
                h = best.copy()
