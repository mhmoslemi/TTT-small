from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_value(scaled, t):
    m = np.max(scaled)
    if m > 700.0 / t:  # overflow protection
        return m
    exps = np.exp(t * (scaled - m))

    def obj(x):

        scaled = corr * dx
        sm = smooth_max_value(scaled, t)
        integ = np.sum(h) * dx - 1.0
        return sm + penalty * integ * integ

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_value(scaled, t):
    m = np.max(scaled)
    if t > 1e-6:
        shifted = scaled - m
        max_shift = np.max(shifted)
        if max_shift * t > 500:
            return m
        exps = np.exp(t * shifted)
        return m + np.log(np.sum(exps)) / t
    return m

def make_objective(dx, t, penalty):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        scaled = corr * dx
        sm = smooth_max_value(scaled, t)
        integ = np.sum(h) * dx - 1.0
        return sm + penalty * integ * integ
    return obj

def fix_constraint(h, n):
    """Iteratively fix the integral constraint while maintaining [0,1] bounds."""
    h = h.copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        step = diff / n
        h += step
        h = np.clip(h, 0.0, 1.0)
    # Final check: if still off, do weighted adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def local_refine(h, dx, max_rounds=50):
    """Coordinate descent with adaptive step sizing."""
    n = len(h)
    h = h.copy()
    c5_cur = compute_c5(h, dx)
    step = 0.02
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            for d in [step, -step]:
                h_try = h.copy()
                h_try[i] = h[i] + d
                if h_try[i] < 0 or h_try[i] > 1:
                    continue
                h_try = fix_constraint(h_try, n)
                c5_try = compute_c5(h_try, dx)
                if c5_try < c5_cur - 1e-12:
                    h = h_try
                    c5_cur = c5_try
                    improved = True
                    break
        if not improved:
            step *= 0.7
            if step < 1e-5:
                break
    return h, c5_cur
