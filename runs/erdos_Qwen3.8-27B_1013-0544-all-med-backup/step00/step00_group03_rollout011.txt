import numpy as np

def msequence(n):
    """Generate an m-sequence of length 2^n - 1 using a primitive polynomial"""
    # Use GF(2) with primitive polynomial
    # For n=6: x^6 + x + 1 (primitive)
    # For n=7: x^7 + x^3 + 1
    # For n=8: x^8 + x^4 + x^3 + x^2 + 1
    polys = {
        4: [0b10011],  # x^4 + x + 1
        5: [0b100111],  # x^5 + x^2 + 1
        6: [0b1000011],  # x^6 + x + 1
        7: [0b1001001],  # x^7 + x^3 + 1
        8: [0b100110101],  # x^8 + x^4 + x^3 + x^2 + 1
    }
    
    if n not in polys:
        return None
    
    poly = polys[n]
    seq = []
    state = 1  # Start with all 1s
    
    length = (1 << n) - 1
    for _ in range(length):
        seq.append(state & 1)
        # Shift and apply feedback
        lsb = state & 1
        state >>= 1
        if lsb:
            state ^= ((poly[0] >> 1) << 1)  # XOR with polynomial (without leading bit)
    
    return np.array(seq, dtype=float)
