def project_to_sum(h, target_sum):

    diff = target_sum - h.sum()
    if diff > 0:

        idx = np.argsort(-h)  # most headroom first? No, least h first

            h[i] += add
            diff -= add
            if diff < 1e-12:
                break
    elif diff < 0:

        for i in idx:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
            if diff > -1e-12:

    corr = np.correlate(h, 1 - h, mode="full")

    vals = T * corr * dx

            j1 = n_i + lag
            j2 = n_i
            grad[j1] += p[k] * (1 - h[j2])
            grad[j2] -= p[k] * h[j1]
    grad *= dx
    return log_sum_exp / T, grad

            total = h.sum()
            if total == 0:

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj_and_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    vals = T * corr * dx
    max_val = np.max(vals)
    exp_vals = np.exp(vals - max_val)
    s = exp_vals.sum()
    p = exp_vals / s
    obj = (max_val + np.log(s)) / T
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        n_min = max(0, -lag)
        n_max = min(n - 1, n - 1 - lag)
        for ni in range(n_min, n_max + 1):
            j1 = ni + lag
            j2 = ni
            grad[j1] += pk * (1.0 - h[j2])
            grad[j2] -= pk * h[j1]
    grad *= dx
    return obj, grad
