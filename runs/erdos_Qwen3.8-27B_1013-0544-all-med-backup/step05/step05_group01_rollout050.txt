    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-15: break
            h += (diff / t) * room
        else:
            t = h.sum()
            if t < 1e-15: break
            h += (diff / t) * h

    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:

    best_h, best_c = local_refine(best_h, n, t_local2, 2)

    if time.time() < deadline - (deadline - t0_global) * 0.03:
        ht = slsqp(best_h, n, 20000000, 5000)
        c = compute_c5(ht, n)
        if c < best_c: best_c, best_h = c, ht

    best_h, best_c = local_refine(best_h, n, deadline - 1, 3)
    return project_feasible(best_h, n), best_c

t0_global = 0.0

def run(seed=42, budget_s=1100, **kwargs):
    global t0_global
    t0_global = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0_global + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)

    alloc1 = (deadline - t0_global) * 0.42
    best_h, best_c = optimize_h(best_h, n_init, t0_global + alloc1, rng)

    for n_try in [200, 300]:
        if time.time() > deadline - 70: break
        remaining = deadline - time.time()
        alloc = min(remaining * 0.18, 110)
        if alloc < 35: break
        x0 = np.linspace(0, 2, len(best_h))
        x1 = np.linspace(0, 2, n_try)
        h_interp = np.interp(x1, x0, best_h)
        h_interp = project_feasible(h_interp, n_try)
        h_opt, c_opt = optimize_h(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt

    n_best = len(best_h)
    if time.time() < deadline - 45:
        remaining = deadline - time.time() - 25
        n_restarts = max(2, int(remaining / 50))
        scales = [0.003, 0.008, 0.015, 0.025, 0.04, 0.06]
        for r in range(n_restarts):
            if time.time() > deadline - 25: break
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(50, (deadline - time.time() - 25) * 0.38)
            if alloc < 18: break
            h_opt, c_opt = optimize_h(hp, n_best, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt

    if time.time() < deadline - 3:
        n_f = len(best_h)
        if n_f <= 200:
            for d in [0.00005, 0.00002, 0.00001, 0.000005]:
                if time.time() > deadline - 1: break
                best_h, best_c = full_search(best_h, n_f, deadline - 1, d, 150)
        else:
            for d, rf in [(0.00005, 0.5), (0.00002, 0.4), (0.00001, 0.3), (0.000005, 0.2)]:
                if time.time() > deadline - 1: break
                best_h, best_c = near_search(best_h, n_f, deadline - 1, d, max(5, n_f * rf // 10), 200)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
