            if tot < 1e-14:

def compute_smoothed_max_and_grad(h, n, alpha):
    """Compute softmax-smoothed max of correlations and its gradient."""
    corr = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    m = corr.max()
    shifted = alpha * (corr - m)
    mx = shifted.max()
    exp_vals = np.exp(shifted - mx)
    weights = exp_vals / exp_vals.sum()
    smooth_val = m + np.log(exp_vals.sum()) / alpha + mx / alpha
    
    # Compute gradient
    grad = np.zeros(n)
    corr_len = 2 * n - 1
    for m_idx in range(corr_len):
        w = weights[m_idx]
        if w < 1e-8:
            continue
        s = m_idx - (n - 1)
        if s >= 0:
            grad[s:] += w * (1.0 - h[:n-s])
            grad[:n-s] -= w * h[s:]
        else:
            neg_s = -s
            grad[:n-s] += w * (1.0 - h[neg_s:])
            grad[neg_s:] -= w * h[:n-s]
    
    return smooth_val, grad, weights

    best_h = h.copy()
    v = np.zeros(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16, deadline=time.time()+1e9):

    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1 - x, mode='full') * dx
        m = corr.max()
        shifted = alpha * (corr - m)
        mx = shifted.max()
        kv = 50.0
        if mx < kv:
            return m + np.log(np.sum(np.exp(shifted))) / alpha
        else:
            return m + (mx - kv + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    
    def obj_grad(x):
        corr = np.correlate(x, 1 - x, mode='full') * dx
        m = corr.max()
        shifted = alpha * (corr - m)
        mx = shifted.max()
        exp_vals = np.exp(shifted - mx)
        weights = exp_vals / exp_vals.sum()
        
        g = np.zeros(n)
        for m_idx in range(2*n-1):
            w = weights[m_idx]
            if w < 1e-8:
                continue
            s = m_idx - (n - 1)
            if s >= 0:
                g[s:] += w * (1.0 - x[:n-s]) * dx
                g[:n-s] -= w * x[s:] * dx
            else:
                ns = -s
                g[:n-ns] += w * (1.0 - x[ns:]) * dx
                g[ns:] -= w * x[:n-ns] * dx
        return g
    
    def con_fun(x):
        return np.sum(x) - n/2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0,1)]*n
    cons = [{'type':'eq', 'fun':con_fun, 'jac':con_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter':maxiter, 'ftol':ftol})
        if res.success or res.fun < compute_c5(h, n):
            return project(res.x, n)
    except Exception:
        pass
    return h.copy()

        if c_opt < best_c:
            h_opt, c_opt = targeted_local_search(h_opt, n_try, time.time() + alloc * 0.4)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Phase 7: Final refinement
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new, c_new = targeted_local_search(best_h, n, deadline - 2, max_passes=2000)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        if time.time() < deadline - 2:
            h_new, c_new = broad_local_search(best_h, n, deadline - 1, radius=60,
                                              deltas=[0.002, 0.001, 0.0005, 0.0002])
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        best_h = project(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        best_h = project(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    
    return best_h, best_c, len(best_h)
