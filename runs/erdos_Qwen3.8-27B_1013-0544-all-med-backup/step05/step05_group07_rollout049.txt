            if 0 <= j2 < n and j2 in zeros:
                candidates.append((i, j2))
        if not candidates:
            if ones and zeros:
                candidates = [(list(ones)[rng.randint(len(ones))], list(zeros)[rng.randint(len(zeros))])]
            else:
                break
        rng.shuffle(candidates)
        found = False
        for (i, j) in candidates[:6]:
            if time.time() > deadline or evals >= max_evals:
                break
            if i not in ones or j not in zeros:
                continue
            best_h[i], best_h[j] = 0.0, 1.0
            ones.discard(i); ones.add(j)
            zeros.discard(j); zeros.add(i)
            evals += 1
            c_new = compute_c5(best_h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                found = True
            else:
                best_h[i], best_h[j] = 1.0, 0.0
                ones.discard(j); ones.add(i)
                zeros.discard(i); zeros.add(j)
        if not found:
            stuck += 1
            if stuck > 50 and ones and zeros:
                i = list(ones)[rng.randint(len(ones))]
                j = list(zeros)[rng.randint(len(zeros))]
                best_h[i], best_h[j] = 0.0, 1.0
                ones.discard(i); ones.add(j)
                zeros.discard(j); zeros.add(i)
                evals += 1
                c_new = compute_c5(best_h, n)
                if c_new >= best_c:
                    best_h[i], best_h[j] = 1.0, 0.0
                    ones.discard(j); ones.add(i)
                    zeros.discard(i); zeros.add(j)
                stuck = 0
    return best_h, best_c

def binarize_and_refine(h, n, deadline, rng):
    h_bin = np.zeros(n)
    h_bin[np.argsort(h)[::-1][:n // 2]] = 1.0
    return directed_binary(h_bin, n, deadline, rng)

def multi_restart(h, n, deadline, rng, n_restarts=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    t_per = (deadline - time.time()) / n_restarts
    for r in range(n_restarts):
        t_end = time.time() + t_per
        noise_scale = 0.15 * (r + 1) / n_restarts
        h_try = project_feasible(best_h + noise_scale * rng.randn(n), n)
        h_b = (h_try > 0.5).astype(float)
        h_b[np.argsort(h_try)[::-1][:n // 2]] = 1.0
        h_b[np.argsort(h_try)[:n // 2]] = 0.0
        hb, cb = directed_binary(h_b, n, min(t_end, time.time() + t_per * 0.5), rng)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
        ht, ct = subgradient_optimize(best_h.copy(), n, t_end, rng)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    h_bin = np.zeros(n)
    h_bin[np.argsort(h0)[::-1][:n // 2]] = 1.0
    c_bin = compute_c5(h_bin, n)
    if c_bin < best_c:
        best_c, best_h = c_bin, h_bin.copy()

    p1 = t0 + 0.30 * (budget_s - 3)
    h1, c1 = directed_binary(best_h.copy(), n, p1, rng)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()

    p2 = t0 + 0.50 * (budget_s - 3)
    h2, c2 = subgradient_optimize(best_h.copy(), n, p2, rng)
    if c2 < best_c:
        best_c, best_h = c2, h2.copy()

    p3 = t0 + 0.65 * (budget_s - 3)
    h3, c3 = pair_descent(best_h.copy(), n, p3, rng, max_evals=150000)
    if c3 < best_c:
        best_c, best_h = c3, h3.copy()

    p4 = t0 + 0.80 * (budget_s - 3)
    h4, c4 = subgradient_optimize(best_h.copy(), n, p4, rng)
    if c4 < best_c:
        best_c, best_h = c4, h4.copy()

    p5 = t0 + 0.90 * (budget_s - 3)
    h5, c5 = pair_descent(best_h.copy(), n, p5, rng, max_evals=100000)
    if c5 < best_c:
        best_c, best_h = c5, h5.copy()

    if time.time() < deadline - 60:
        h6, c6 = multi_restart(best_h.copy(), n, deadline - 10, rng, n_restarts=2)
        if c6 < best_c:
            best_c, best_h = c6, h6.copy()

    final_h = project_feasible(best_h, n)
    final_c = compute_c5(final_h, n)
    return final_h, final_c, n
