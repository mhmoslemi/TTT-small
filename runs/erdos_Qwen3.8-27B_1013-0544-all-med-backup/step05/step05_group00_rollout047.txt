```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, n, step_size, n_crit=5):
    """One step of subgradient descent for the max correlation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    # Find critical shifts (top n_crit)
    threshold = corr.max() * 0.95
    critical = np.where(corr >= threshold)[0]
    if len(critical) == 0:
        critical = np.array([np.argmax(corr)])
    
    # Compute subgradient
    g = np.zeros(n)
    for k_idx in critical:
        k = k_idx - (n - 1)  # Convert to actual lag
        for j in range(n):
            # ∂f_k/∂h[j] = (1-h[j+k]) - h[j-k] (with boundary handling)
            jk = j + k
            jmk = j - k
            term = 0.0
            if 0 <= jk < n:
                term += (1 - h[jk])
            if 0 <= jmk < n:
                term -= h[jmk]
            g[j] += term
    g /= len(critical)
    
    # Update
    h_new = h - step_size * g
    h_new = project_feasible(h_new, n)
    return h_new, corr.max() * dx

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

                    break

                break

                break

            best_c = cur_c
            best_h = h.copy()
            current_h = h.copy()
```

For each zero i, compute a quick score for each one j:
score(j) = -(corr + P[i] - Q[j] - 2δ(n-1) + δ(n-1+j-i) + δ(n-1+i-j)).max()

To compute this quickly, I can approximate:
- The new max is at most corr.max() (the old max)
- The main reduction comes from the shifts where corr is currently high
- So: approximate_score(j) = max over critical shifts k of (corr[k] + P[i][k] - Q[j][k])

This is O(|K|) per candidate j, where |K| is the number of critical shifts. If |K| is small (say 5-20), this is much faster than O(n).

```python
def fast_candidate_score(i, j, P_i, Q_j, corr, critical_shifts, n):
    """Quick approximate score for swap (i,j)."""
    # Only check critical shifts
    vals = np.zeros(len(critical_shifts))
    for idx, k in enumerate(critical_shifts):
        vals[idx] = corr[k] + P_i[k] - Q_j[k]
    # Add the point modifications
    k1 = n-1+j-i
    k2 = n-1+i-j
    for idx, k in enumerate(critical_shifts):
        if k == n-1:
            vals[idx] -= 2.0
        if k == k1:
            vals[idx] += 1.0
        if k == k2:
            vals[idx] += 1.0
    return vals.max()
```

```python
def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def grad(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        exp_shifted = np.exp(shifted - sm)
        weights = exp_shifted / exp_shifted.sum()
        # Gradient of sum_k weights[k] * g[k]
        # g[k] = dx * sum_i x[i] * (1-x[i+k-(n-1)])
        # dg[k]/dx[j] = dx * (1-x[j+k-(n-1)]) - dx * x[j-k+(n-1)] (with bounds)
        grad_x = np.zeros(n)
        m_range = np.arange(-(n-1), n)
        for k_idx in range(2*n-1):
            if weights[k_idx] < 1e-10:
                continue
            k = k_idx - (n-1)
            # dg[k]/dx[j] = dx * [(1-x[j+k]) - x[j-k]] (for in-range)
            jk = np.arange(n) + k
            jmk = np.arange(n) - k
            mask_jk = (jk >= 0) & (jk < n)
            mask_jmk = (jmk >= 0) & (jmk < n)
            term1 = np.zeros(n)
            term1[mask_jk] = 1 - x[jk[mask_jk]]
            term2 = np.zeros(n)
            term2[mask_jmk] = x[jmk[mask_jmk]]
            grad_x += weights[k_idx] * dx * (term1 - term2)
        return grad_x
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
```

```python
def binary_sa_search(h_init, n, deadline, seed=42, n_restarts=5, 
                     max_sweeps=5000, m_candidates=10, n_crit=10):
    """Binary local search with simulated annealing."""
    target = n // 2
    rng = np.random.RandomState(seed)
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()
    m_range = np.arange(-(n-1), n)
    
    # Precompute contribution vectors
    def compute_contrib(h_curr):
        """Compute C[i] = contribution vector for position i going 0->1."""
        C = np.zeros((n, 2*n-1))
        for i in range(n):
            im = i + m_range
            msk = (im >= 0) & (im < n)
            C[i, msk] = 1.0 - h_curr[im[msk]]
            imn = i - m_range
            msk = (imn >= 0) & (imn < n)
            C[i, msk] -= h_curr[imn[msk]]
        return C
    
    C = compute_contrib(h)
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    def update_C_after_swap(i, j, C):
        """Update C after swapping h[i]:0->1, h[j]:1->0."""
        # h[i] became 1, h[j] became 0
        # C[p][k] = (1-h[p+k-(n-1)]) - h[p-k+(n-1)]
        # Changing h[i] from 0 to 1:
        #   Affects C[p] at k where p+k-(n-1)=i → k=i-p+n-1, term (1-h) increases by... wait
        #   (1-h[p+k-(n-1)]) changes when p+k-(n-1)=i, i.e., the value 1-h[i] goes from 1 to 0
        #   So C[p][i-p+n-1] decreases by 1 for all p where i-p+n-1 is in [0, 2n-2]
        #   Also h[p-k+(n-1)] changes when p-k+(n-1)=i, i.e., k=p-i+n-1
        #   So C[p][p-i+n-1] increases by 1 (since we subtract h, and h went up)
        
        # Similarly for h[j] going from 1 to 0:
        #   (1-h[p+k-(n-1)]) changes: 1-h[j] goes from 0 to 1
        #   C[p][j-p+n-1] increases by 1
        #   h[p-k+(n-1)] changes: h[j] goes from 1 to 0
        #   C[p][p-j+n-1] decreases by 1
        
        # This is O(n) per position change, so O(n) total for the swap
        # But updating all C[p] for all p is O(n²) in the worst case
        
        # Actually, for each p:
        # C[p] changes at at most 4 indices: i-p+n-1, p-i+n-1, j-p+n-1, p-j+n-1
        # So total updates are O(n) (4 per p, n values of p)
        for p in range(n):
            k1 = i - p + n - 1  # from (1-h) term for h[i]: 0->1, so (1-h) goes 1->0, C decreases
            k2 = p - i + n - 1  # from -h term for h[i]: h goes 0->1, so -h goes 0->-1, C decreases
            k3 = j - p + n - 1  # from (1-h) term for h[j]: 1->0, so (1-h) goes 0->1, C increases
            k4 = p - j + n - 1  # from -h term for h[j]: h goes 1->0, so -h goes -1->0, C increases
            
            if 0 <= k1 < 2*n-1:
                C[p, k1] -= 1.0
            if 0 <= k2 < 2*n-1:
                C[p, k2] -= 1.0
            if 0 <= k3 < 2*n-1:
                C[p, k3] += 1.0
            if 0 <= k4 < 2*n-1:
                C[p, k4] += 1.0
    
    current_h = h.copy()
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            n_flips = rng.randint(5, 20)
            for _ in range(n_flips):
                oi = list(ones)
                zi = list(zeros)
                if oi and zi:
                    a = rng.choice(oi)
                    b = rng.choice(zi)
                    current_h[a], current_h[b] = 0.0, 1.0
                    ones.discard(a); ones.add(b)
                    zeros.discard(b); zeros.add(a)
        
        h = current_h.copy()
        C = compute_contrib(h)
        corr = np.correlate(h, 1 - h, mode='full').copy()
        cur_c = corr.max() * dx
        
        # Simulated annealing temperature
        T = 0.005
        no_improve_count = 0
        
        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            
            # Decrease temperature
            T *= 0.999
            T = max(T, 0.0001)
            
            improved = False
            z_list = sorted(zeros)
            rng.shuffle(z_list)
            
            for i in z_list:
                if time.time() > deadline:
                    break
                if i not in zeros:
                    continue
                
                # Quick scoring: for each one j, estimate new max
                # new_corr ≈ corr + C[i] - C[j] + point_mods
                # Quick: just check the top critical shifts
                crit_indices = np.argsort(corr)[-n_crit:]
                
                best_score = np.inf
                best_j = -1
                second_best_score = np.inf
                second_best_j = -1
                
                for j in ones:
                    # Quick approximation: check only critical shifts
                    # delta at critical shift k ≈ C[i][k] - C[j][k] + point_mods
                    # For speed, just use C[i][k] - C[j][k] (ignore point mods for ranking)
                    diff = C[i, crit_indices] - C[j, crit_indices]
                    # The point mods: -2 at n-1, +1 at n-1+j-i, +1 at n-1+i-j
                    # These only affect specific indices, so for the critical indices:
                    temp = corr[crit_indices] + diff
                    if n-1 in crit_indices:
                        idx_nm1 = np.where(crit_indices == n-1)[0]
                        temp[idx_nm1] -= 2.0
                    k1 = n-1+j-i
                    if k1 in crit_indices:
                        idx_k1 = np.where(crit_indices == k1)[0]
                        temp[idx_k1] += 1.0
                    k2 = n-1+i-j
                    if k2 in crit_indices:
                        idx_k2 = np.where(crit_indices == k2)[0]
                        temp[idx_k2] += 1.0
                    score = temp.max()
                    if score < best_score:
                        second_best_score = best_score
                        second_best_j = best_j
                        best_score = score
                        best_j = j
                    elif score < second_best_score:
                        second_best_score = score
                        second_best_j = j
                
                if best_j < 0:
                    continue
                
                # Full evaluation of best candidate
                delta = C[i] - C[best_j]
                delta[n-1] -= 2.0
```

C[p][k] = (1 - h[p + k - (n-1)]) (if in range) - h[p - k + (n-1)] (if in range)

When h[i] changes from 0 to 1:
- First term (1-h[...]): changes when p+k-(n-1) = i, i.e., k = i-p+n-1
  (1-h[i]) goes from 1 to 0, so C[p][i-p+n-1] decreases by 1
- Second term -h[...]: changes when p-k+(n-1) = i, i.e., k = p-i+n-1
  -h[i] goes from 0 to -1, so C[p][p-i+n-1] decreases by 1

When h[j] changes from 1 to 0:
- First term: changes when p+k-(n-1) = j, i.e., k = j-p+n-1
  (1-h[j]) goes from 0 to 1, so C[p][j-p+n-1] increases by 1
- Second term: changes when p-k+(n-1) = j, i.e., k = p-j+n-1
  -h[j] goes from -1 to 0, so C[p][p-j+n-1] increases by 1

So:
For each p in range(n):
  C[p][i-p+n-1] -= 1 (if in range)
  C[p][p-i+n-1] -= 1 (if in range)
  C[p][j-p+n-1] += 1 (if in range)
  C[p][p-j+n-1] += 1 (if in range)

That's 4n index updates total (one for each p, 4 indices each). For n=250, that's 1000 updates. This is O(n) which is fine.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def compute_contrib(h, n):
    m_range = np.arange(-(n - 1), n)
    C = np.zeros((n, 2 * n - 1))
    for i in range(n):
        im = i + m_range
        msk = (im >= 0) & (im < n)
        C[i, msk] = 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        C[i, msk] -= h[imn[msk]]
    return C

def update_contrib(C, i, j, n):
    for p in range(n):
        k1 = i - p + n - 1
        if 0 <= k1 < 2 * n - 1:
            C[p, k1] -= 1.0
        k2 = p - i + n - 1
        if 0 <= k2 < 2 * n - 1:
            C[p, k2] -= 1.0
        k3 = j - p + n - 1
        if 0 <= k3 < 2 * n - 1:
            C[p, k3] += 1.0
        k4 = p - j + n - 1
        if 0 <= k4 < 2 * n - 1:
            C[p, k4] += 1.0

def make_delta(C, i, j, n):
    delta = C[i] - C[j]
    delta[n - 1] -= 2.0
    k1 = n - 1 + j - i
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + i - j
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def binary_sa_search(h_init, n, deadline, seed=42, n_restarts=5,
                     max_sweeps=8000, n_crit=12):
    target = n // 2
    rng = np.random.RandomState(seed)
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    best_c = np.inf
    best_h = h.copy()
    current_h = h.copy()

    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        if restart > 0:
            n_flips = min(5 + 3 * restart, 30)
            for _ in range(n_flips):
                oi = np.where(current_h == 1)[0]
                zi = np.where(current_h == 0)[0]
                if len(oi) > 0 and len(zi) > 0:
                    a = rng.choice(oi)
                    b = rng.choice(zi)
                    current_h[a], current_h[b] = 0.0, 1.0
        h = current_h.copy()
        C = compute_contrib(h, n)
        corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
        cur_c = corr.max() * dx
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        T = 0.003
        stall = 0

        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            T = max(T * 0.9995, 0.00005)
            improved = False
            z_order = zeros[:]
            rng.shuffle(z_order)
            for i in z_order:
                if time.time() > deadline:
                    break
                if h[i] != 0:
                    continue
                crit = np.argsort(corr)[-n_crit:]
                crit_set = set(crit.tolist())
                best_sc = np.inf
                best_j = -1
                for j in ones:
                    if h[j] != 1:
                        continue
                    diff = C[i, crit] - C[j, crit]
                    temp = corr[crit] + diff
                    if n - 1 in crit_set:
                        m = crit == n - 1
                        temp[m] -= 2.0
                    k1 = n - 1 + j - i
                    if k1 in crit_set:
                        m = crit == k1
                        temp[m] += 1.0
                    k2 = n - 1 + i - j
                    if k2 in crit_set:
                        m = crit == k2
                        temp[m] += 1.0
                    sc = temp.max()
                    if sc < best_sc:
                        best_sc = sc
                        best_j = j
                if best_j < 0:
                    continue
                delta = make_delta(C, i, best_j, n)
                new_c = (corr + delta).max() * dx
                accept = False
                if new_c < cur_c - 1e-14:
                    accept = True
                elif T > 0.0001 and rng.random() < np.exp(-(new_c - cur_c) / T):
                    accept = True
                if accept:
                    h[i], h[best_j] = 1.0, 0.0
                    update_contrib(C, i, best_j, n)
                    corr = corr + delta
                    cur_c = new_c
                    jpos = ones.index(best_j)
                    ones[jpos] = i
                    ipos = zeros.index(i)
                    zeros[ipos] = best_j
                    improved = True
                    stall = 0
                    break
            if not improved:
                stall += 1
            if stall > 300 and T < 0.0001:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            current_h = h.copy()
    return best_h, best_c
```
