import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def build_V_matrix(h, n):
    lags = np.arange(2 * n - 1) - (n - 1)
    V = np.zeros((n, 2 * n - 1))
    for m in range(n):
        a = m + lags
        b = m - lags
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        V[m, va] = 1.0 - h[a[va]]
        V[m, vb] -= h[b[vb]]
    return V

def fast_pair_search(h, n, deadline, deltas, max_passes=2000):
    h = h.copy()
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    best_c = float(np.max(c) * dx)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            V = build_V_matrix(h, n)
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if i + 1 >= n:
                    break
                V_rest = V[i + 1:, :]
                base1 = c + d * V[i]
                nm1 = np.max(base1[None, :] - d * V_rest, axis=1) * dx
                v1 = (h[i] + d <= 1.0) & (h[i + 1:] - d >= 0.0)
                vm1 = np.where(v1, nm1, np.inf)
                jo = int(np.argmin(vm1))
                bv1 = vm1[jo]
                base2 = c - d * V[i]
                nm2 = np.max(base2[None, :] + d * V_rest, axis=1) * dx
                v2 = (h[i] - d >= 0.0) & (h[i + 1:] + d <= 1.0)
                vm2 = np.where(v2, nm2, np.inf)
                jo2 = int(np.argmin(vm2))
                bv2 = vm2[jo2]
                if bv1 < best_c - 1e-14 and bv1 <= bv2:
                    h[i] += d
                    h[jo + i + 1] -= d
                    c = np.correlate(h, 1 - h, mode='full')
                    best_c = bv1
                    improved = True
                elif bv2 < best_c - 1e-14:
                    h[i] -= d
                    h[jo2 + i + 1] += d
                    c = np.correlate(h, 1 - h, mode='full')
                    best_c = bv2
                    improved = True
            if not improved:
                break
    return h, best_c

def stagnation_break(h, n, best_c, rng, n_perturb=8):
    """Make random non-improving moves to escape local optima, then re-optimize."""
    h = h.copy()
    c = compute_c5(h, n)
    for _ in range(n_perturb):
        i, j = rng.choice(n, 2, replace=False)
        d = rng.uniform(0.005, 0.04) * (1 if rng.rand() < 0.5 else -1)
        hi, hj = h[i] + d, h[j] - d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h[i], h[j] = hi, hj
    h = project_feasible(h, n)
    return h

def optimize_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # SLSQP phase
    alphas = [1000, 10000, 100000, 1000000, 10000000]
    for a in alphas:
        if time.time() > deadline - 20:
            break
        ht = slsqp_optimize(best_h, n, a, maxiter=4000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()
    # Fast pair search
    if time.time() < deadline - 5:
        deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
        best_h, best_c = fast_pair_search(best_h, n, deadline, deltas, max_passes=3000)
    # Stagnation break + re-optimize if time remains
    if time.time() < deadline - 30:
        hp = stagnation_break(best_h, n, best_c, rng)
        h_opt, c_opt = optimize_n(hp, n, deadline, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
