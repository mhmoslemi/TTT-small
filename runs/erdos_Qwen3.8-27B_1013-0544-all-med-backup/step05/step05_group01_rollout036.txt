    Compute gradient of sum_m w[m]*C[m] w.r.t. h
    where C = correlate(h, 1-h, 'full')
    w has length 2n-1
    """

            grad[kp:] -= wk * h[:n-kp]

    cm = c.max()
    shifted = alpha * (c - cm)
    mx = shifted.max()

    w = exp_shifted / exp_shifted.sum()
    smooth_val = cm + (mx + np.log(exp_shifted.sum())) / alpha

        if c_trial < best_c:
            best_c = c_trial
            best_h = h_trial.copy()
        h = h_trial
        
        # If no improvement for a while, reduce step size

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def corr_gradient(h, n, w):
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if abs(wk) < 1e-14:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            grad[k:n] += wk * (1.0 - h[0:n-k])
            grad[0:n-k] -= wk * h[k:n]
        else:
            kp = -k
            grad[0:n-kp] += wk * (1.0 - h[kp:n])
            grad[kp:n] -= wk * h[0:n-kp]
    return grad

def soft_max_w(c, alpha):
    cm = c.max()
    shifted = alpha * (c - cm)
    mx = shifted.max()
    e = np.exp(shifted - mx)
    s = e.sum()
    return e / s, cm + (mx + np.log(s)) / alpha

def grad_descent(h, n, deadline, n_steps=3000, alpha0=500, alpha_end=500000):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        alpha = alpha0 * (alpha_end / alpha0) ** (step / max(n_steps, 1))
        c = compute_corr(h, n)
        w, _ = soft_max_w(c, alpha)
        grad = corr_gradient(h, n, w) / alpha
        lr = 0.005 * (1.0 + 2.0 * (1.0 - step / n_steps))
        h_trial = project_feasible(h - lr * grad, n)
        c_trial = compute_c5(h_trial, n)
        if c_trial < best_c - 1e-15:
            best_c = c_trial
            best_h = h_trial.copy()
        h = h_trial
    return best_h, best_c

def neighbor_search(h, n, deadline, deltas, max_passes=1000):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n - 1):
                hi = h[i] + d
                hj = h[i + 1] - d
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h[i], h[i + 1] = hi, hj
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        h[i], h[i + 1] = hi - d, hj + d
                hi = h[i] - d
                hj = h[i + 1] + d
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h[i], h[i + 1] = hi, hj
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                    else:
                        h[i], h[i + 1] = hi + d, hj - d
            if not improved:
                break
    return h, best_c

def pair_search(h, n, deadline, deltas, max_passes=500):
