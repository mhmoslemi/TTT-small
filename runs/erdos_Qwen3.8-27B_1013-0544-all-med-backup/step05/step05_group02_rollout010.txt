def project_feasible(h, n):

            if tot < 1e-14:

            h = h + (diff / tot) * h

def grad_lag(h, n, k):
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        # c = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
        # d/dh[j]: (1-h[j-s]) for j>=s, minus h[j+s] for j<=n-1-s

        g[idx1] += (1.0 - h[idx1 - s])
        idx2 = np.arange(0, n - s)  # j from 0 to n-1-s
        g[idx2] -= h[idx2 + s]
    else:

        # c = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
        # d/dh[j]: (1-h[j+d]) for j<=n-1-d, minus h[j-d] for j>=d

        g[idx1] += (1.0 - h[idx1 + d])
        idx2 = np.arange(d, n)
        g[idx2] -= h[idx2 - d]
    return g

def soft_max_grad(h, n, T):
    c = get_corr(h, n)
    mx = c.max()
    if T < 1e-8:
        k_star = int(np.argmax(c))
        return grad_lag(h, n, k_star), mx
    exponents = T * (c - mx)
    weights = np.exp(exponents)
    weights /= weights.sum()
    # Only use lags with significant weight for efficiency

    t_pgd = min(deadline - 10, time.time() + (deadline - time.time()) * 0.4)
    h_pgd, c_pgd = pgd_full(best_h, n, t_pgd)
    if c_pgd < best_c:
        best_c = c_pgd
        best_h = h_pgd
    # SLSQP phase
    for T in [100, 500, 2000, 8000, 30000]:
        if time.time() > deadline - 10:
            break
        ht = slsqp_softmax(best_h, n, T, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht
    # Polish
    if time.time() < deadline - 5:
        t_polish = min(deadline - 3, time.time() + 20)
        best_h, best_c = multi_lag_polish(best_h, n, t_polish, num_lags=3)
    if time.time() < deadline - 2:
        best_h, best_c = coord_polish(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize initial solution
    alloc1 = min(300, (deadline - t0) * 0.35)
    h_opt, c_opt = optimize_full(h_init, n_init, t0 + alloc1)
    best_h, best_c = h_opt, c_opt

    # Phase 2: Try anti-periodic structure
    if time.time() < deadline - 80:
        n2 = n_init // 2
        h_half = h_init[:n2].copy()
        alloc2 = min(150, (deadline - time.time()) * 0.25)
        hh_opt, c_app = pgd_anti_periodic(h_half, n_init, time.time() + alloc2)
        if c_app < best_c:
            best_c = c_app
            best_h = anti_periodic_full(hh_opt)
        # Also polish the anti-periodic result unconstrained
        ap_full = anti_periodic_full(hh_opt)
        ap_full = project_feasible(ap_full, n_init)
        if time.time() < deadline - 40:
            h_ap, c_ap = optimize_full(ap_full, n_init, min(deadline - 30, time.time() + 80))
            if c_ap < best_c:
                best_c = c_ap
                best_h = h_ap

    # Phase 3: Try different n values
    for n_try in [150, 200, 300]:
        if n_try == n_init or time.time() > deadline - 80:
            continue
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(100, (deadline - time.time()) * 0.2)
        if alloc < 30:
            break
        h_opt, c_opt = optimize_full(h_interp, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: Random restarts with structured perturbations
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(3, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.003 + 0.04 * (r / max(1, n_restarts - 1))
            # Structured perturbation: low-freq + high-freq
            x = np.linspace(0, 2, n)
            freq = rng.randint(2, 8)
            phase = rng.uniform(0, 2 * np.pi)
            pert = scale * np.sin(freq * np.pi * x / 2 + phase) + rng.normal(0, scale * 0.5, n)
            hp = best_h + pert
            hp = project_feasible(hp, n)
            alloc = min(60, (deadline - time.time() - 30) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_full(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        best_h, best_c = coord_polish(best_h, n, deadline - 2)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
