def simulated_annealing_binary(hb, n, deadline, T0=1.0, cooling=0.999, max_sweeps=10000):
    """Simulated annealing on binary solutions with swap moves."""
    best_h = hb.copy()
    best_c = compute_c5(hb, n)
    curr_c = best_c
    T = T0
    
    ones = np.where(hb == 1)[0].tolist()
    zeros = np.where(hb == 0)[0].tolist()
    rng = np.random.RandomState(42)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        T *= cooling

        n_tries = min(len(ones) * len(zeros), 500)
        for _ in range(n_tries):

            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            if i == j: continue

def project_feasible(h, n):
