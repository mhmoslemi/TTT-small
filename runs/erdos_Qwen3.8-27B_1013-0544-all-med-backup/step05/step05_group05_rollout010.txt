            grad[k_lag:] -= dx * h[:n-k_lag]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_lag(h, n, lag):
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[:m] += dx * (1.0 - h[lag:])
            g[lag:] -= dx * h[:m]
    else:
        m = n + lag
        if m > 0:
            g[:m] -= dx * h[-lag:]
            g[-lag:] += dx * (1.0 - h[:m])
    return g

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def fw_decode(grad, n):
    """Frank-Wolfe linear minimizer: binary vector with 1s where grad is smallest."""
    idx = np.argsort(grad)[:n//2]
    v = np.zeros(n)
    v[idx] = 1.0
    return v

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_lag(h, n, lag):
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[:m] += dx * (1.0 - h[lag:])
            g[lag:] -= dx * h[:m]
    else:
        m = n + lag
        if m > 0:
            g[:m] -= dx * h[-lag:]
            g[-lag:] += dx * (1.0 - h[:m])
    return g

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def adam_opt(h0, n, deadline, max_iter=50000):
    h = h0.copy().astype(float)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t += 1
        g = compute_all_g(h, n)
        top3 = np.argsort(g)[::-1][:3]
        ag = g[top3]
        sh = ag - ag.max()
        w = np.exp(100.0 * sh)
        w /= w.sum()
        grad = np.zeros(n)
        for i in range(3):
            lag = int(top3[i] - (n - 1))
            grad += w[i] * grad_lag(h, n, lag)
        m_hat = b1 * m_hat + (1.0 - b1) * grad
        v_hat = b2 * v_hat + (1.0 - b2) * (grad * grad)
        m_c = m_hat / (1.0 - b1**t)
        v_c = v_hat / (1.0 - b2**t)
        lr = 0.05 / (1.0 + 0.001 * t)
        h = project_feasible(h - lr * m_c / (np.sqrt(v_c) + eps), n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c
