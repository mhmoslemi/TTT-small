import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_round(h, n, alpha, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        h_x = np.clip(x, 0, 1)
        c = np.correlate(h_x, 1 - h_x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def worst_lag_gradient(h, n, k_star):
    s = k_star - (n - 1)
    g = np.zeros(n)
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    idx = np.arange(lo, hi + 1)
    g[idx] += (1.0 - h[idx + s])
    lo2 = max(0, lo + s)
    hi2 = min(n - 1, hi + s)
    idx2 = np.arange(lo2, hi2 + 1)
    g[idx2] -= h[idx2 - s]
    return g

def directed_polish(h, n, deadline, max_stall=300):
    dx = 2.0 / n
    c = get_corr(h, n)
    best_c = float(c.max())
    best_h = h.copy()
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    step_idx = 0
    stall = 0
    while time.time() < deadline:
        k_star = int(np.argmax(c))
        g = worst_lag_gradient(h, n, k_star)
        pos_max = g.max()
        neg_min = g.min()
        if pos_max < 1e-12 or (-neg_min) < 1e-12:
            break
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            g_sorted = np.argsort(g)
            i_dst = int(g_sorted[0])
            i_src = int(g_sorted[-1])
            if i_src == i_dst:
                break
        d = steps[step_idx]
        d = min(d, h[i_src] * 0.99, (1.0 - h[i_dst]) * 0.99)
        if d < 1e-10:
            step_idx = min(step_idx + 1, len(steps) - 1)
            stall += 1
            if stall > max_stall:
                break
            continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = get_corr(h, n)
        new_max = float(new_c.max())
        if new_max < best_c - 1e-15:
            best_c = new_max
            best_h = h.copy()
            c = new_c
            stall = 0
            step_idx = max(0, step_idx - 1)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            if stall > max_stall:
                step_idx = min(step_idx + 1, len(steps) - 1)
                if step_idx >= len(steps) - 1:
                    break
                stall = 0
    return best_h, best_c

def binary_local_search(h, n, deadline, rng):

            idx = zeros[rng.randint(len(zeros))]
            h_bin[idx] = 1
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
    
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    
    T = 0.01

    cooling = 0.999
    
    while time.time() < deadline and T > T_min:
        # Pick a random swap
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        i_one = ones[rng.randint(len(ones))]
        i_zero = zeros[rng.randint(len(zeros))]
        
        # Try the swap
        h_bin[i_one] = 0
        h_bin[i_zero] = 1
        new_c = compute_c5(h_bin, n)

        if delta < 0 or rng.rand() < np.exp(-delta / T):
            if new_c < best_c:
                best_c = new_c
                best_h = h_bin.copy()
        else:
            h_bin[i_one] = 1
            h_bin[i_zero] = 0
        
        T *= cooling
    
    return best_h, best_c

def subgradient_method(h, n, deadline):
    """Subgradient descent for min max of quadratic functions."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def worst_lag_gradient(h, n, k_star):
    s = k_star - (n - 1)
    g = np.zeros(n)
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    idx = np.arange(lo, hi + 1)
    g[idx] += (1.0 - h[idx + s])
    if s < 0:
        lo2 = max(0, lo - s)
        hi2 = min(n - 1, hi - s)
        idx2 = np.arange(lo2, hi2 + 1)
        g[idx2] -= h[idx2 + s]
    else:
        lo2 = max(0, lo + s)
        hi2 = min(n - 1, hi + s)
        idx2 = np.arange(lo2, hi2 + 1)
        g[idx2] -= h[idx2 - s]
    return g

def multi_lag_grad(h, n, top_k=5, temperature=1.0):
    """Compute weighted gradient using top-k worst lags with softmax weighting."""
    c = get_corr(h, n)
    # Softmax weights over top lags
    top_idx = np.argsort(c)[-top_k:]
    weights = np.exp((c[top_idx] - c[top_idx].max()) / temperature)
    weights = weights / weights.sum()
    g = np.zeros(n)
    for w, k in zip(weights, top_idx):
        g += w * worst_lag_gradient(h, n, int(k))
    return g

def directed_polish_v2(h, n, deadline, max_stall=400, num_lags=3):
    """Improved directed polish with multi-lag gradient and adaptive steps."""
    dx = 2.0 / n
    c = get_corr(h, n)
    best_c = float(c.max())
    best_h = h.copy()
    steps = [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    step_idx = 0
    stall = 0
    while time.time() < deadline:
        c = get_corr(h, n)
        curr = float(c.max())
        if curr < best_c - 1e-15:
            best_c = curr
            best_h = h.copy()
        
        if num_lags > 1:
            g = multi_lag_grad(h, n, top_k=num_lags, temperature=50.0)
        else:
            k_star = int(np.argmax(c))
            g = worst_lag_gradient(h, n, k_star)
        
        if g.max() < 1e-13:
            break
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            sorted_g = np.argsort(g)
            i_src = int(sorted_g[-1])
            i_dst = int(sorted_g[0])
            if i_src == i_dst:
                break
        d = steps[step_idx]
        d = min(d, h[i_src] * 0.999, (1.0 - h[i_dst]) * 0.999)
        if d < 1e-11:
            step_idx = min(step_idx + 1, len(steps) - 1)
            stall += 1
            if stall > max_stall:
                break
            continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = float(get_corr(h, n).max())
        if new_c < curr - 1e-15:
            stall = 0
            step_idx = max(0, step_idx - 1)
            c = get_corr(h, n)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            step_idx = min(step_idx + 1, len(steps) - 1)
            if stall > max_stall:
                break
    return best_h, best_c

def binary_sa(h, n, deadline, seed):
    """Simulated annealing on binary sequences."""
    rng = np.random.RandomState(seed)
    h_bin = (h > 0.5).astype(float)
    # Fix sum
    num_ones = int(h_bin.sum())
    target_ones = n // 2
    if num_ones > target_ones:
        extra = np.where(h_bin == 1)[0]
        for i in range(num_ones - target_ones):
            idx = extra[rng.randint(len(extra))]
            h_bin[idx] = 0
    elif num_ones < target_ones:
        deficit = np.where(h_bin == 0)[0]
        for i in range(target_ones - num_ones):
            idx = deficit[rng.randint(len(deficit))]
            h_bin[idx] = 1
    
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    T = 0.005
    T_min = 1e-7
    cooling = 0.9995
    
    while time.time() < deadline and T > T_min:
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        i_o = ones[rng.randint(len(ones))]
        i_z = zeros[rng.randint(len(zeros))]
        h_bin[i_o] = 0
        h_bin[i_z] = 1
        new_c = compute_c5(h_bin, n)
        delta = new_c - best_c
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            if new_c < best_c:
                best_c = new_c
                best_h = h_bin.copy()
        else:
            h_bin[i_o] = 1
            h_bin[i_z] = 0
        T *= cooling
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_round(h, n, alpha, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        h_x = np.clip(x, 0, 1)
        c = np.correlate(h_x, 1 - h_x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-17})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_gradient(h, n, k_star):
    t = k_star - (n - 1)
    g = np.zeros(n)
    if t >= 0:
        idx1 = np.arange(t, n)
        g[idx1] += (1.0 - h[idx1 - t])
        idx2 = np.arange(0, n - t)
        g[idx2] -= h[idx2 + t]
    else:
        tp = -t
        idx1 = np.arange(0, n - tp)
        g[idx1] += (1.0 - h[idx1 + tp])
        idx2 = np.arange(tp, n)
        g[idx2] -= h[idx2 - tp]
    return g

def multi_lag_grad(h, n, top_k=5, temp=100.0):
    c = get_corr(h, n)
    top_idx = np.argsort(c)[-top_k:]
    w = np.exp((c[top_idx] - c[top_idx].max()) / temp)
    w = w / w.sum()
    g = np.zeros(n)
    for wi, ki in zip(w, top_idx):
        g += wi * lag_gradient(h, n, int(ki))
    return g

def directed_polish(h, n, deadline, num_lags=1, max_stall=400):
    dx = 2.0 / n
    c = get_corr(h, n)
    best_c = float(c.max())
    best_h = h.copy()
    steps = [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    step_idx = 0
    stall = 0
    while time.time() < deadline:
        c = get_corr(h, n)
        curr = float(c.max())
        if curr < best_c - 1e-15:
            best_c = curr
            best_h = h.copy()
        if num_lags > 1:
            g = multi_lag_grad(h, n, top_k=num_lags, temp=80.0)
        else:
            g = lag_gradient(h, n, int(np.argmax(c)))
        if g.max() - g.min() < 1e-13:
            break
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            sg = np.argsort(g)
            i_src, i_dst = int(sg[-1]), int(sg[0])
            if i_src == i_dst:
                break
        d = steps[step_idx]
        d = min(d, h[i_src] * 0.999, (1.0 - h[i_dst]) * 0.999)
        if d < 1e-12:
            step_idx = min(step_idx + 1, len(steps) - 1)
            stall += 1
            if stall > max_stall:
                break
            continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = float(get_corr(h, n).max())
        if new_c < curr - 1e-15:
            stall = 0
            step_idx = max(0, step_idx - 1)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            if stall > 80:
                step_idx = min(step_idx + 1, len(steps) - 1)
                stall = 0
            if step_idx >= len(steps) - 1 and stall > max_stall:
                break
    return best_h, best_c

def binary_sa(h, n, deadline, seed):
    rng = np.random.RandomState(seed)
    h_bin = (h > 0.5).astype(float)
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    if len(ones) > n // 2:
        for _ in range(len(ones) - n // 2):
            i = ones[rng.randint(len(ones))]
            h_bin[i] = 0
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
    elif len(ones) < n // 2:
        for _ in range(n // 2 - len(ones)):
            i = zeros[rng.randint(len(zeros))]
            h_bin[i] = 1
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    T = 0.008
    T_min = 1e-7
    cool = 0.9997
    while time.time() < deadline and T > T_min:
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        io = ones[rng.randint(len(ones))]
        iz = zeros[rng.randint(len(zeros))]
        h_bin[io] = 0
        h_bin[iz] = 1
        nc = compute_c5(h_bin, n)
        if nc < best_c or rng.rand() < np.exp(-(nc - best_c) / T):
            if nc < best_c:
                best_c = nc
                best_h = h_bin.copy()
