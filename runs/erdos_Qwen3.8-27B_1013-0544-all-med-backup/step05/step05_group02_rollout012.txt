    max_c = c.max()
    shifted = alpha * (c - max_c)
    max_shifted = shifted.max()

    weights /= weights.sum()
    
    # g_raw[j] = sum_k weights[k] * ((1-h[j+s_k]) - h[j-s_k])
    W = weights.sum()
    g_raw = np.full(n, W)
    sig = np.where(weights > 1e-12)[0]

        s = k - (n - 1)
        w = weights[k]
        if s >= 0:
            g_raw[:n-s] -= w * h[s:]
            g_raw[s:] -= w * h[:n-s]
        else:
            sn = -s
            g_raw[sn:] -= w * h[:n-sn]
            g_raw[:n-sn] -= w * h[sn:]
    
    return g_raw * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _grad_smooth(h, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1 - h, mode='full')
    c = c_raw * dx
    max_c = c.max()
    shifted = alpha * (c - max_c)
    ms = shifted.max()
    w = np.exp(shifted - ms)
    w /= w.sum()
    W = w.sum()
    g = np.full(n, W)
    sig = np.where(w > 1e-12)[0]
    for k in sig:
        s = k - (n - 1)
        wi = w[k]
        if s >= 0:
            g[:n-s] -= wi * h[s:]
            g[s:] -= wi * h[:n-s]
        else:
            sn = -s
            g[sn:] -= wi * h[:n-sn]
            g[:n-sn] -= wi * h[sn:]
    return g * dx

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    h_x = np.clip(x, 0, 1)
    c = np.correlate(h_x, 1 - h_x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    ms = sh.max()
    return m + (ms + np.log(np.sum(np.exp(sh - ms)))) / alpha

def _smooth_obj_grad(x, n, alpha):
    return _grad_smooth(np.clip(x, 0, 1), n, alpha)

def _sum_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_round(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18},
                      jac=lambda x: _smooth_obj_grad(x, n, alpha), args=(n, alpha))
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gd_polish(h, n, alpha, deadline, max_steps=20000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 2.0
    prev_c = best_c
    stall = 0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        g = _grad_smooth(h, n, alpha)
        g_proj = g - g.mean()
        norm_g = np.linalg.norm(g_proj)
        if norm_g < 1e-14:
            break
        improved = False
        for _ in range(30):
            h_new = h - lr * g_proj
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < prev_c - 1e-15:
                h = h_new
                prev_c = c_new
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    stall = 0
                improved = True
                break
            lr *= 0.5
        if not improved:
            stall += 1
            lr = min(lr * 2.0, 5.0)
            if stall > 100:
                break
        else:
            lr = min(lr * 1.05, 5.0)
    return best_h, best_c

def worst_lag_gradient(h, n, k_star):
    s = k_star - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    else:
        sn = -s
        g[sn:] += (1.0 - h[:n-sn])
        g[:n-sn] -= h[sn:]
    return g

def directed_polish(h, n, deadline, max_stall=300):
    dx = 2.0 / n
    c = get_corr(h, n)
    best_c = float(c.max())
    best_h = h.copy()
    steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    step_idx = 0
    stall = 0
    num_lags = 5
    while time.time() < deadline:
        k_star_list = np.argsort(c)[-num_lags:]
        g = np.zeros(n)
        for k in k_star_list:
            g += worst_lag_gradient(h, n, int(k))
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        d = steps[step_idx]
        d = min(d, h[i_src] * 0.99, (1.0 - h[i_dst]) * 0.99)
        if d < 1e-12:
            step_idx = min(step_idx + 1, len(steps) - 1)
            stall += 1
            if stall > max_stall:
                break
            continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = get_corr(h, n)
        new_max = float(new_c.max())
        if new_max < best_c - 1e-15:
            best_c = new_max
            best_h = h.copy()
            c = new_c
            stall = 0
            step_idx = max(0, step_idx - 1)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            step_idx = min(step_idx + 1, len(steps) - 1)
            if stall > max_stall:
                stall = 0
                step_idx = min(step_idx + 1, len(steps) - 1)
                if step_idx >= len(steps) - 1:
                    break
    return best_h, best_c
