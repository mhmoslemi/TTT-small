def grad_overlap_shift(h, n, s):
    n = len(h)
    g = np.zeros(n)
    if s > 0:
        # (1 - h[i+s]) for i in [0, n-1-s]
        g[:n-s] += (1 - h[s:])
        # -h[i-s] for i in [s, n-1]
        g[s:] -= h[:n-s]
    elif s < 0:
        p = -s
        # (1 - h[i-p]) for i in [p, n-1]
        g[p:] += (1 - h[:n-p])
        # -h[i+p] for i in [0, n-1-p]

    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for it in range(max_iters):
        if deadline and time.time() > deadline:
            break
        
        overlaps = np.correlate(h, 1-h, mode='full') * dx
        max_idx = np.argmax(overlaps)
        current_max = overlaps[max_idx]
        
        if current_max <= best_c + 1e-12:
            best_c = current_max
        
        # Get shift for the max lag
        s = max_idx - (n - 1)
        
        # Compute gradient
        g = grad_overlap_shift(h, n, s)
        
        # Also add gradients from top few lags (weighted)

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def sturmian_init(n):
    alpha = 1.0 / ((1.0 + np.sqrt(5.0)) / 2.0)
    x = np.arange(n, dtype=float)
    h = np.floor((x + 1.0) * alpha) - np.floor(x * alpha)
    return h.astype(float)

def zc_init(n):
    x = np.arange(n, dtype=float)
    return 0.5 * (1.0 + np.cos(np.pi * x * (x + 1.0) / n))

def full_pipeline(h, n, deadline, rng):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    t_g = min(15, (deadline - time.time()) * 0.15)
    h, c = gradient_search(h, n, time.time() + t_g, top_m=8,
                           seed=rng.randint(10000))
    if c < best_c:
        best_c = c
        h = h
    for a in [500, 5000, 50000, 200000]:
        if time.time() > deadline - 5:
            break
        ht = slsqp_optimize(h, n, a, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht
    if time.time() < deadline - 3:
        h, c = gradient_search(h, n, time.time() + 10, top_m=10,
                               seed=rng.randint(10000))
        if c < best_c:
            best_c = c
            h = h
    if time.time() < deadline - 2 and n <= 200:
        h, best_c = pair_search(h, n, deadline - 1,
                                [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
    elif time.time() < deadline - 2:
        h, best_c = pair_search(h, n, deadline - 1,
                                [0.01, 0.005, 0.002, 0.001])
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = sturmian_init(250)
        n_init = 250

    best_h = None
    best_c = float('inf')

    h_init = project_feasible(h_init, n_init)
    c0 = compute_c5(h_init, n_init)
    best_h, best_c = h_init.copy(), c0

    alloc1 = min(180, (deadline - t0) * 0.2)
    h_opt, c_opt = full_pipeline(h_init, n_init, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c, best_h = c_opt, h_opt

    starts = []
    starts.append(('sturmian', sturmian_init(n_init)))
    starts.append(('zc', zc_init(n_init)))
    starts.append(('sturmian2', 1.0 - sturmian_init(n_init)))
    for r in range(4):
        hp = best_h + rng.normal(0.0, 0.05 + 0.03 * r, n_init)
        starts.append(('perturb%d' % r, hp))

    for name, hs in starts:
        if time.time() > deadline - 120:
            break
        hs = project_feasible(hs, n_init)
        alloc = min(80, (deadline - time.time()) * 0.2)
        if alloc < 20:
            break
        h_opt, c_opt = full_pipeline(hs, n_init, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt

    for n_try in [100, 150, 300, 500]:
        if time.time() > deadline - 80:
            break
        if n_try == n_init:
            continue
        x_old = np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(80, (deadline - time.time()) * 0.18)
        if alloc < 20:
            break
        h_opt, c_opt = full_pipeline(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt

    for n_try in [100, 150]:
        if time.time() > deadline - 60:
            break
        h_st = sturmian_init(n_try)
        h_st = project_feasible(h_st, n_try)
        alloc = min(60, (deadline - time.time()) * 0.2)
        if alloc < 15:
            break
        h_opt, c_opt = full_pipeline(h_st, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt

    if best_h is not None and time.time() < deadline - 30:
        n = len(best_h)
        remaining = deadline - time.time() - 10
        n_restarts = max(2, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            scale = 0.02 + 0.04 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0.0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 15) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = full_pipeline(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt

    if best_h is not None and time.time() < deadline - 20:
        n = len(best_h)
        ht = slsqp_optimize(best_h, n, 1000000, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c, best_h = c, ht
        if time.time() < deadline - 5:
            h2, c2 = gradient_search(best_h, n, time.time() + 10, top_m=12,
                                     seed=rng.randint(10000))
            if c2 < best_c:
                best_c, best_h = c2, h2
        if time.time() < deadline - 3:
            if n <= 200:
                best_h, best_c = pair_search(best_h, n, deadline - 1,
                                             [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
            else:
                best_h, best_c = pair_search(best_h, n, deadline - 1,
                                             [0.005, 0.002, 0.001, 0.0005])

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
