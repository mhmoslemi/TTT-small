import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_b_for_j(h, n, j):
    """Compute b vector: derivative of g with respect to h[j] for each lag"""
    b = np.zeros(2 * n - 1)
    # For L >= 0 (correlation index n-1+L):
    for L in range(0, n):
        idx = n - 1 + L
        # h[j] as h[i+L] where i=j-L: valid if 0<=j-L and j-L<=n-1-L
        if 0 <= j - L and j - L <= n - 1 - L:
            b[idx] += (1.0 - h[j - L])
        # h[j] as h[i] where i=j: valid if j <= n-1-L (so j+L < n)
        if j <= n - 1 - L:
            b[idx] += (-h[j + L])
    # For L < 0 (lag = -L, correlation index n-1-L):
    for L in range(1, n):
        idx = n - 1 - L
        # g(-L) = sum_{i=L}^{n-1} h[i-L]*(1-h[i])
        # h[j] as h[i-L] where i-L=j, i=j+L: valid if 0<=j and j<=n-1-L
        if 0 <= j and j <= n - 1 - L:

        # h[j] as h[i] where i=j: valid if L<=j<=n-1 (so j-L >= 0)
        if L <= j and j <= n - 1:
            b[idx] += (-h[j - L])
    return b

    for _ in range(max_iter):
        if hi - lo < tol:

        mid = (lo + hi) / 2
        vals = g + b * mid

def min1d(g, b, lo, hi):

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_b_j(h, n, j):
    b = np.zeros(2 * n - 1)
    K1 = min(j, n - 1)
    if K1 >= 0:
        ps = np.arange(0, K1 + 1)
        b[n - 1 + ps] += (1.0 - h[j - ps])
    K2 = n - 1 - j
    if K2 >= 0:
        ps = np.arange(0, K2 + 1)
        b[n - 1 + ps] += (-h[j + ps])
    K3 = n - 1 - j
    if K3 >= 1:
        qs = np.arange(1, K3 + 1)
        b[n - 1 - qs] += (1.0 - h[j + qs])
    K4 = min(j, n - 1)
    if K4 >= 1:
        qs = np.arange(1, K4 + 1)
        b[n - 1 - qs] += (-h[j - qs])
    return b

def min1d(g, b, lo, hi):
    if hi - lo < 1e-15:
        return (lo + hi) / 2
    for _ in range(60):
        if hi - lo < 1e-15:
            break
        mid = (lo + hi) / 2
        vals = g + b * mid
        m = vals.max()

        if active.min() > 1e-10:

        if stagnant > perturb_every and perturb_size > 0:
            # Perturb a few random coordinates
            for _ in range(3):
                j = int(rng.randint(n))
                h[j] = float(np.clip(h[j] + rng.uniform(-perturb_size, perturb_size), 0, 1))
            h = project_feasible(h, n)
            stagnant = 0
            perturb_size *= 0.5
    return h, best_c

def smooth_max(h, n, alpha):
    g = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = g.max()
    sh = g - m
    return m + (sh.max() + np.log(np.exp(sh - sh.max()).sum())) / alpha

def slsqp_polish(h0, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    def obj(x):
        return smooth_max(x, n, alpha)
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h0
