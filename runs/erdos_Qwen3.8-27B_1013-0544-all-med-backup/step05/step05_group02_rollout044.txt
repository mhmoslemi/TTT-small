    dx = 2.0 / n
    def obj(x):
        h_x = np.clip(x, 0, 1)
        c = np.correlate(h_x, 1 - h_x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: float(np.sum(x) - n / 2.0)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def fista_step(h, n, y, v, t_val, top_k=3, lr_scale=1.0):
    """One FISTA iteration. Returns new_h, new_v, new_t."""
    c = get_corr(y, n)
    top_idx = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    for k in top_idx:
        g += worst_lag_gradient(y, n, int(k))
    g /= len(top_idx)
    gnorm = np.linalg.norm(g)
    if gnorm < 1e-14:
        return project_feasible(y, n), v, t_val
    step = lr_scale / (t_val * gnorm * 0.5)
    step = min(step, 0.5)
    h_new = y - step * g
    h_new = project_feasible(h_new, n)
    c_new = compute_c5(h_new, n)
    c_y = compute_c5(y, n)
    if c_new > c_y + step * gnorm * 0.1:
        step *= 0.5
        h_new = project_feasible(y - step * g, n)
    v_new = h_new + (t_val - 1.0) / (t_val + 1.0) * (h_new - v)
    t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
    return h_new, v_new, t_new

def fista_optimize(h_init, n, deadline, top_k=3, max_iter=5000):
    best_h = project_feasible(h_init.copy(), n)
    best_c = compute_c5(best_h, n)
    v = best_h.copy()
    t_val = 1.0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        h_new, v, t_val = fista_step(best_h, n, v, v, t_val, top_k)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new
    return best_h, best_c

def directed_polish(h, n, deadline, max_stall=500):
    c = get_corr(h, n)
    best_c = float(c.max())
    best_h = h.copy()
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6]
    step_idx = 0
    stall = 0
    while time.time() < deadline:
        k_star = int(np.argmax(c))
        g = worst_lag_gradient(h, n, k_star)
        if g.max() < 1e-13 or (-g.min()) < 1e-13:
            break
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        d = steps[step_idx]
        d = min(d, h[i_src] * 0.99, (1.0 - h[i_dst]) * 0.99)
        if d < 1e-13:
            step_idx = min(step_idx + 1, len(steps) - 1)
            stall += 1
            if stall > max_stall:
                break
            continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = get_corr(h, n)
        new_max = float(new_c.max())
        if new_max < best_c - 1e-15:
            best_c = new_max
            best_h = h.copy()
            c = new_c
            stall = 0
            step_idx = max(0, step_idx - 1)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            if stall > max_stall // 4:
                step_idx = min(step_idx + 1, len(steps) - 1)
                if step_idx >= len(steps) - 1:
                    break
                stall = 0
    return best_h, best_c

def multi_lag_polish(h, n, deadline, num_lags=4):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    step_idx = 0
    stall = 0
    while time.time() < deadline:
        c = get_corr(h, n)
        curr = float(c.max())
        if curr < best_c - 1e-15:
            best_c = curr
            best_h = h.copy()
        top_k = np.argsort(c)[-num_lags:]
        g = np.zeros(n)
        for k in top_k:
            g += worst_lag_gradient(h, n, int(k))
        if g.max() < 1e-13:
            break
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        d = steps[step_idx]
        d = min(d, h[i_src] * 0.99, (1.0 - h[i_dst]) * 0.99)
        if d < 1e-12:
            step_idx = min(step_idx + 1, len(steps) - 1)
            stall += 1
            if stall > 200:
                break
            continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = float(get_corr(h, n).max())
        if new_c < curr - 1e-15:
            stall = 0
            step_idx = max(0, step_idx - 1)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            if stall > 100:
                step_idx = min(step_idx + 1, len(steps) - 1)
                if step_idx >= len(steps) - 1:
                    break
                stall = 0
    return best_h, best_c

def optimize_full(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    alphas = [500, 2000, 10000, 50000, 200000, 1000000]
    for a in alphas:
        if time.time() > deadline - 30:
            break
        ht = slsqp_round(best_h, n, a, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht
    if time.time() < deadline - 20:
        best_h, best_c = fista_optimize(best_h, n, min(deadline - 20, time.time() + 40), top_k=4)
    if time.time() < deadline - 10:
        best_h, best_c = multi_lag_polish(best_h, n, min(deadline - 10, time.time() + 30), num_lags=4)
    if time.time() < deadline - 5:
        best_h, best_c = directed_polish(best_h, n, deadline - 3)
    if time.time() < deadline - 5:
        ht = slsqp_round(best_h, n, 5000000, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht
        if time.time() < deadline - 2:
            best_h, best_c = directed_polish(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        alloc = (deadline - t0) * 0.5
        h_opt, c_opt = optimize_full(h_init, n_init, t0 + alloc)
        best_h, best_c = h_opt, c_opt

    # Try alternative resolutions
    if best_h is not None and time.time() < deadline - 100:
        x_orig = np.linspace(0, 2, len(best_h))
        for n_try in [200, 300]:
            if n_try == n_init or time.time() > deadline - 50:
                continue
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_orig, best_h)
            h_interp = project_feasible(h_interp, n_try)
            remaining = deadline - time.time() - 30
            if remaining < 50:
                break
            alloc = min(150, remaining * 0.4)
            h_opt, c_opt = optimize_full(h_interp, n_try, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Random restarts
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(3, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.003 + 0.04 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 30) * 0.4)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_full(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final polish
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        best_h, best_c = directed_polish(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
