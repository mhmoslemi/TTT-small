import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            s = mask.sum()
            if s == 0:
                break
            h[mask] += diff / s
        else:
            mask = h > 1e-12
            s = mask.sum()
            if s == 0:
                break
            h[mask] += diff / s
    return np.clip(h, 0.0, 1.0)

def get_active_lag_subgrad(h, n):
    """Get subgradient at the active lag."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m_star = int(np.argmax(c))
    k = m_star - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:]) * dx
        g[k:] += -h[:n-k] * dx
    else:
        i = -k
        g[i:] += -h[:n-i] * dx
        g[:n-i] += (1.0 - h[i:]) * dx
    return g, c, m_star

def binary_sa(h, n, t0, t1, rng):
    """Simulated annealing for binary solutions."""
    target = n // 2
    h_cur = h.copy()
    c_cur = compute_c5(h_cur, n)
    best_h = h_cur.copy()
    best_c = c_cur
    T = 0.01
    ones = np.where(h_cur == 1)[0]
    zeros = np.where(h_cur == 0)[0]
    
    while time.time() < t1:
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h_cur[i], h_cur[j] = 1.0, 0.0
        c_new = compute_c5(h_cur, n)
        delta = c_new - c_cur
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            c_cur = c_new
            ones[ones == j] = i
            zeros[zeros == i] = j
            if c_new < best_c:
                best_c = c_new
                best_h = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = 0.0, 1.0
        T *= 0.999
        if T < 1e-8:

def soft_max_obj(x, n, alpha):
    """Soft-max approximation of max_k c[k]."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    logsum = sm + np.log(np.exp(shifted - sm).sum())
    return m + logsum / alpha

def soft_max_grad(x, n, alpha):
    """Gradient of soft-max approximation at active lag."""
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full')
    m_star = int(np.argmax(c))
    k = m_star - (n - 1)
    g = np.zeros(n)
    if k >= 0:

        g[k:] += -x[:n-k] * dx
    else:

        g[i:] += -x[:n-i] * dx
        g[:n-i] += (1.0 - x[i:]) * dx
    # Add constraint penalty gradient
    g += 2e4 * (x.sum() - n/2.0)
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def subgrad_active(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    m_star = int(np.argmax(c))
    k = m_star - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[:n - k] += (1.0 - h[k:]) * dx
        g[k:] += -h[:n - k] * dx
    else:
        i = -k
        g[i:] += (1.0 - h[:n - i]) * dx
        g[:n - i] += -h[i:] * dx
    return g, c * dx

def soft_max_lbfgs(h, n, alpha_sm, maxiter, deadline):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        s = alpha_sm * (c - m)
        sm = s.max()
        val = m + (sm + np.log(np.exp(s - sm).sum())) / alpha_sm
        val += 5e3 * (x.sum() - n / 2.0) ** 2
        return val

    def grad(x):
        g, _ = subgrad_active(x, n)
        g += 1e4 * (x.sum() - n / 2.0)
        return g

    try:
        res = minimize(obj, project_feasible(h, n), method='L-BFGS-B',
                       jac=grad, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_sa_smart(h, n, deadline, rng):
    target = n // 2
    h_cur = h.copy()
    c_cur = compute_c5(h_cur, n)
    best_h = h_cur.copy()
    best_c = c_cur
    T = 0.008
    ones = np.where(h_cur == 1)[0].tolist()
    zeros = np.where(h_cur == 0)[0].tolist()
    it = 0
    while time.time() < deadline:
        it += 1
        if not ones or not zeros:
            break
        # Smart move: focus on positions affecting the active lag
        _, c_full = subgrad_active(h_cur, n)
        k_star = int(np.argmax(c_full)) - (n - 1)
        # Prefer positions near the "boundary" of the active lag
        if k_star >= 0:
            hot_zeros = [z for z in zeros[:min(len(zeros), 60)] if z < n - k_star]
            hot_ones = [o for o in ones[:min(len(ones), 60)] if o >= k_star]
        else:
            i = -k_star
            hot_zeros = [z for z in zeros[:min(len(zeros), 60)] if z >= i]
            hot_ones = [o for o in ones[:min(len(ones), 60)] if o < n - i]
        if hot_zeros and hot_ones:
            i_move = hot_zeros[rng.randint(len(hot_zeros))]
            j_move = hot_ones[rng.randint(len(hot_ones))]
        else:
            i_move = zeros[rng.randint(len(zeros))]
            j_move = ones[rng.randint(len(ones))]
        h_cur[i_move], h_cur[j_move] = 1.0, 0.0
        c_new = compute_c5(h_cur, n)
        delta = c_new - c_cur
        if delta < -1e-14 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_cur = c_new
            ones.remove(j_move)
            ones.append(i_move)
            zeros.remove(i_move)
            zeros.append(j_move)
            if c_new < best_c:
                best_c = c_new
                best_h = h_cur.copy()
        else:
            h_cur[i_move], h_cur[j_move] = 0.0, 1.0
        T *= 0.9992
        if T < 5e-5:
            T = 5e-5
    return best_h, best_c

def pair_descent(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001]
    for outer in range(200):
        if time.time() > deadline:
            break
        g, _ = subgrad_active(best_h, n)
        g_proj = g - g.mean()
        pos = np.where(g_proj < -1e-8)[0]
        neg = np.where(g_proj > 1e-8)[0]
        if len(pos) == 0 or len(neg) == 0:
            break
        improved = False
        # Try top gradient positions
        pos_sorted = pos[np.argsort(g_proj[pos])][:30]
        neg_sorted = neg[np.argsort(-g_proj[neg])][:30]
        for i in pos_sorted:
            if time.time() > deadline:
                break
            for j in neg_sorted:
                if time.time() > deadline:
                    break
                for s in steps:
                    if best_h[i] + s > 1.0 or best_h[j] - s < 0.0:
                        continue
                    h_try = best_h.copy()
                    h_try[i] += s
                    h_try[j] -= s
                    h_try = project_feasible(h_try, n)
                    c_try = compute_c5(h_try, n)
                    if c_try < best_c - 1e-14:
                        best_c = c_try
                        best_h = h_try.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            # Random perturbation to escape
            idx = rng.choice(n, 4, replace=False)
            h_try = best_h.copy()
            signs = rng.choice([-1, 1], 4)
            for m in range(4):
                h_try[idx[m]] += signs[m] * 0.01
            h_try = project_feasible(h_try, n)
            if compute_c5(h_try, n) < best_c:
                best_h = h_try
                best_c = compute_c5(h_try, n)
            else:
                break
    return best_h, best_c

def projected_subgrad_with_overshoot(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 0.3
    stall = 0
    while time.time() < deadline:
        g, _ = subgrad_active(h, n)
        g_proj = g - g.mean()
        moved = False
        for fa in [1.0, 0.5, 0.25, 0.1]:
            h_new = project_feasible(h - fa * alpha * g_proj, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                alpha = min(alpha * 1.03, 3.0)
                stall = 0
                moved = True
                break
        if not moved:
            alpha *= 0.9
            stall += 1
            if stall > 80:
                h = project_feasible(best_h + 0.05 * rng.randn(n), n)
                alpha = 0.3
                stall = 0
        if alpha < 1e-5:
            break
    return best_h, best_c
