def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx), c * dx

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

        return np.where((idx >= 0) & (idx < n), h[idx], 0.0)
    
    # But we need to account for the swap itself
    # After swap: h[i]=0, h[j]=1
    
    # For the computation, let's use the "new" h except at positions i and j
    # Actually, let me just compute directly:
    
    i_s = i + s  # positions i+s for all s
    i_ms = i - s  # positions i-s
    j_s = j + s
    j_ms = j - s

    # Actually if i+s == i, that's s=0. h_new[i] = 0.
    # If i+s == j, h_new[i+s] = h_new[j] = 1.
    hi_s = np.where((i_s >= 0) & (i_s < n), 0.0, 0.0)
    hi_s[(i_s >= 0) & (i_s < n)] = h[i_s[(i_s >= 0) & (i_s < n)]]
    hi_s[i_s == j] = 1.0  # after swap
    hi_s[i_s == i] = 0.0  # after swap (this is s=0)
    
    hj_s = np.where((j_s >= 0) & (j_s < n), 0.0, 0.0)
    hj_s[(j_s >= 0) & (j_s < n)] = h[j_s[(j_s >= 0) & (j_s < n)]]
    hj_s[j_s == i] = 0.0  # after swap
    hj_s[j_s == j] = 1.0  # after swap (s=0)
    
    hi_ms = np.where((i_ms >= 0) & (i_ms < n), 0.0, 0.0)
    hi_ms[(i_ms >= 0) & (i_ms < n)] = h[i_ms[(i_ms >= 0) & (i_ms < n)]]
    hi_ms[i_ms == j] = 1.0
    hi_ms[i_ms == i] = 0.0
    
    hj_ms = np.where((j_ms >= 0) & (j_ms < n), 0.0, 0.0)
    hj_ms[(j_ms >= 0) & (j_ms < n)] = h[j_ms[(j_ms >= 0) & (j_ms < n)]]
    hj_ms[j_ms == i] = 0.0
    hj_ms[j_ms == j] = 1.0
    
    delta = (hi_s - 1.0) + hi_ms + (1.0 - hj_s) - hj_ms
    return corr + delta

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room

            s = h.sum()
            if s < 1e-14:

F = logsumexp(alpha * g) / alpha = (log(sum exp(alpha*g)))/alpha

dF/dx = (1/alpha) * sum_k (exp(alpha*g[k])/sum_j exp(alpha*g[j])) * alpha * dg[k]/dx
= sum_k p[k] * dg[k]/dx

where p[k] = exp(alpha*g[k]) / sum_j exp(alpha*g[j])

This is much cleaner! The gradient is just a weighted average of the gradients of the individual g[k].

So: dF/dx[i] = sum_k p[k] * dg[k]/dx[i] = sum_k p[k] * dx * ((1-x[i+s_k]) - x[i-s_k])

Let me code this up:

def softmax_obj_and_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    # Compute p[k] = exp(alpha*g[k]) / sum exp(alpha*g[j])
    ag = alpha * g
    ag_max = ag.max()
    exp_ag = np.exp(ag - ag_max)
    p = exp_ag / exp_ag.sum()
    # Objective value
    F = (ag_max + np.log(exp_ag.sum())) / alpha
    # Gradient
    s = np.arange(2*n-1) - (n-1)
    grad = np.zeros(n)
    for i in range(n):
        idx_a = i + s
        mask_a = (idx_a >= 0) & (idx_a < n)
        A_i = np.dot(p[mask_a], x[idx_a[mask_a]])
        idx_b = i - s
        mask_b = (idx_b >= 0) & (idx_b < n)
        B_i = np.dot(p[mask_b], x[idx_b[mask_b]])
        grad[i] = dx * (1.0 - A_i - B_i)
    return F, grad

This is correct and efficient. The gradient computation is O(n²) = 62500 for n=250.

Now, with this gradient, SLSQP should converge much faster.

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

            break

            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h

def softmax_obj_grad(x, alpha, n):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    ag = alpha * g
    ag_max = ag.max()

    s = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)

        ia = i + s
        ma = (ia >= 0) & (ia < n)
        A = np.dot(p[ma], x[ia[ma]])
        ib = i - s
        mb = (ib >= 0) & (ib < n)
        B = np.dot(p[mb], x[ib[mb]])
        grad[i] = dx * (1.0 - A - B)
    return F, grad
