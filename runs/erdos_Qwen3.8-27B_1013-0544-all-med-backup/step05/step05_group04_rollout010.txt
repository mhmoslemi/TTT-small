    grad = np.zeros(n)
    if k >= 0:
        # r = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        # dr/dh[j]: 
        #   from first factor: j=i, so for j in [0, n-1-k]: (1-h[j+k])
        #   from second factor: i+k=j, so j in [k, n-1]: -h[j-k]
        n_valid = n - k
        grad[0:n_valid] += (1.0 - h[k:n])       # 1-h[j+k] for j=0..n_valid-1
        grad[k:n] -= h[0:n_valid]                # -h[j-k] for j=k..n-1
    else:

    return grad, c.max()

    return grad

def obj_smooth(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_obj(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_refine(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    configs = [
        (500, 10.0), (2000, 10.0), (10000, 100.0),
        (50000, 100.0), (200000, 1000.0), (1000000, 1000.0)
    ]
    
    for alpha, mu in configs:
        if time.time() > deadline:
            break
        try:
            res = minimize(obj_smooth, h, jac=grad_obj, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-15,
                                   'gtol': 1e-12, 'maxfun': max_iters * 5})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    
    return h, best_c

def slsqp_refine(h, n, deadline, alpha=200000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha, 0.0), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def gradient_step(h, n, deadline, lr_init=0.05, temp=0.05, n_steps=5000):
    """Projected gradient with smoothed objective."""
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    alpha = 1.0 / temp
    
    for it in range(n_steps):
        if time.time() > deadline:
            break
        
        g = grad_smooth_max(h, n, alpha)
        
        # Backtracking line search
        step_lr = lr
        for _ in range(30):
            h_new = project_feasible(h - step_lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c:
                h = h_new
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                break
            step_lr *= 0.5
        else:
            lr *= 0.5
        
        if lr < 1e-10:
            break
    
    return best_h, best_c

def pair_search(h, n, deadline, deltas, radius=None):

                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break

def coordinate_descent(h, n, deadline):
    """Try moving a single point (splitting its mass to neighbors)."""
    best_c = compute_c5(h, n)
    for _ in range(100):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return h, best_c
            # Try moving mass from i to i+1
            for delta in [0.01, 0.005, 0.001, 0.0001]:
                for di, dj in [(i, i+1), (i, i-1)]:
                    if dj < 0 or dj >= n or di == dj:

                        h[di] -= delta; h[dj] += delta
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[di] += delta; h[dj] -= delta
        if not improved:
            break
    return h, best_c

    if time.time() < deadline - 50:

        best_h, best_c = pair_search(best_h, n_init, t5,
                                     [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                     radius=60)

    # Phase 6: Try other n values
    if time.time() < deadline - 80:
        for n_try in [150, 200, 300]:
            if time.time() > deadline - 50:
                break
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(60, deadline - time.time() - 30)
            if alloc < 25:
                break
            h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.6, max_iters=150)
            if time.time() < deadline - 20:
                h_t, c_t = slsqp_phase(h_t, n_try, time.time() + min(30, deadline - time.time() - 15),
                                       alpha=500000, maxiter=1000)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                n_init = n_try

    # Phase 7: Final L-BFGS + SLSQP + pair search on best
    if time.time() < deadline - 10:
        nf = len(best_h)
        if time.time() < deadline - 8:
            best_h, best_c = lbfgs_phase(best_h, nf, deadline - 8, max_iters=100)
        if time.time() < deadline - 5:
            best_h, best_c = pair_search(best_h, nf, deadline - 2,
                                         [0.001, 0.0005, 0.0002, 0.0001],
                                         radius=40)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
