Here's a faster approach:

To compute the best j for a given i:

- We want to find d and j such that max(g + d*s) is minimized
- A good heuristic: find j such that s has the most negative components at the positions where g is large

This gives: for each i, find j that maximizes g[top] @ (F[j, top] - F[i, top]) = weighted_F[j] - weighted_F[i]

Wait, actually the direction matters. If we move mass from j to i (h[i] += d, h[j] -= d), the effect on g is d * (F[i] - F[j]). We want this to reduce g at the top positions, so we want d > 0 and (F[i, top] - F[j, top]) < 0, i.e., F[j, top] > F[i, top].

So we want j with high F[j, top] and i with low F[i, top]. The "score" for the pair is:
score = g[top] @ (F[j, top] - F[i, top]) = weighted_F[j] - weighted_F[i]

We want score > 0 for a potential improvement.

So the algorithm becomes:
1. Compute weighted_F = F[:, top_idx] @ g[top_idx] (vector of length n)
2. For each i (sorted by weighted_F ascending), find the top M j's with the highest weighted_F[j] - weighted_F[i]
3. For each such pair, try the optimal delta
4. Apply the best move

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_F(h, n):
    F = np.zeros((n, 2 * n - 1))
    dm = np.arange(2 * n - 1) - (n - 1)
    for m_idx in range(2 * n - 1):
        d = dm[m_idx]
        if d >= 0:
            ln = n - d
            F[:ln, m_idx] += (1.0 - h[d:d+ln])
            F[d:d+ln, m_idx] -= h[:ln]
        else:
            ln = n + d
            F[ln:ln+ln, m_idx] += (1.0 - h[:ln])
            F[:ln, m_idx] -= h[ln:ln+ln]
    return F

def update_F(F, i, j, delta, n, dm):
    m_idx = np.arange(2 * n - 1)
    v1 = (i + dm >= 0) & (i + dm < n)
    if v1.any():
        F[i + dm[v1], m_idx[v1]] -= delta
    v2 = (i - dm >= 0) & (i - dm < n)
    if v2.any():
        F[i - dm[v2], m_idx[v2]] -= delta
    v3 = (j + dm >= 0) & (j + dm < n)
    if v3.any():
        F[j + dm[v3], m_idx[v3]] += delta
    v4 = (j - dm >= 0) & (j - dm < n)
    if v4.any():
        F[j - dm[v4], m_idx[v4]] += delta

def pair_descent(h, n, deadline, K=12, max_sweeps=300):
    dx = 2.0 / n
    dm = np.arange(2 * n - 1) - (n - 1)
    m_idx = np.arange(2 * n - 1)
    best_h = h.copy()
    g = compute_g(h, n)
    best_c = g.max()
    F = compute_F(h, n)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        top_idx = np.argsort(g)[::-1][:K]
        current_max = g.max()
        improved = 0
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                s = F[i] - F[j]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                # Quick prune: if s is ~0 at all top positions, skip
                if np.max(np.abs(s[top_idx])) < 1e-12:
                    continue
                cands = [d_lo, d_hi]
                for m1 in range(K):
                    for m2 in range(m1 + 1, K):
                        ds = s[top_idx[m2]] - s[top_idx[m1]]
                        if abs(ds) > 1e-12:
                            dc = (g[top_idx[m1]] - g[top_idx[m2]]) / ds
                            if d_lo < dc < d_hi:
                                cands.append(dc)
                cands = np.array(cands)
                cands = np.clip(cands, d_lo, d_hi)
                # Vectorized evaluation
                mat = g[np.newaxis, :] + cands[:, np.newaxis] * s[np.newaxis, :]
                vals = mat.max(axis=1)
                best_val = vals.min()
                if best_val < current_max - 1e-12:
                    best_d = float(cands[vals.argmin()])
                    h[i] += best_d
                    h[j] -= best_d
                    g += best_d * s
                    update_F(F, i, j, best_d, n, dm)
                    current_max = g.max()
                    if current_max < best_c - 1e-10:
                        best_c = current_max
                        best_h = h.copy()
                    top_idx = np.argsort(g)[::-1][:K]
                    improved += 1
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            g = compute_g(h, n)
            F = compute_F(h, n)
        if improved == 0:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0
    return best_h, best_c

def slsqp_power(h, n, p, deadline, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        gp = np.clip(g, 0, None) ** p
        return (gp.sum()) ** (1.0 / p)
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_logsumexp(h, n, alpha, deadline, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        shifted = alpha * (g - g.max())
        sm = shifted.max()
        return g.max() + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_polish(h, n, deadline):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])
    for _ in range(80):
        if time.time() > deadline:
            break
        improved = False
        for z in zeros:
            for o in ones:
                h_bin[z] = 1.0
                h_bin[o] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_c - 1e-14:
                    best_c = c
                    ones.remove(o)
                    ones.append(z)
                    zeros.remove(z)
                    zeros.append(o)
                    improved = True
                else:
                    h_bin[z] = 0.0
                    h_bin[o] = 1.0
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h_bin, best_c
