    best_c = compute_c5(hb, n)

            # Random swap
            i = zeros[np.random.randint(len(zeros))]
            j = list(ones)[np.random.randint(len(ones))]
            
            # Perform swap
            h[i], h[j] = 1.0, 0.0
            c_new = compute_c5(h, n)

                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            else:

            T *= cooling
    
    return best_h, best_c
