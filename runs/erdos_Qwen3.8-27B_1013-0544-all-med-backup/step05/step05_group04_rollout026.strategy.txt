    best_solution = None
    best_c = float('inf')
    n_final = 250
    
    # Helper function for projection
    def project_feasible(x):
        '''Adjust x to satisfy 0 <= x[i] <=1 and sum(x)=1'''    
        x = np.clip(x, 0.0, 1.0)
        sum_val = x.sum()
        target = 1.0  # since sum(dx)=1, but n_points=250 gives dx=0.008, so 1/sum_val?
        # For n_particles=250, target should be n_final/2
        if n_final % 2 == 0:
            target = n_final / 2
        # If we need to adjust for different n_values
        if abs(sum_val - target) < 1e-6:
            return x
        
        # Simple redistribution
        direction = (target - sum_val) / n_final
        return x + direction
    
    # Objective function
    def compute_c5(x):
        n = len(x)
        dx = 2.0 / n
        corr = np.correlate(x, 1-x, mode='full')
        max_corr = np.max(corr) * dx
#         time.sleep(0)  # Placeholder for expensive operation
        return float(max_corr)
        
    # Particle swarm optimization
    def pso_objective(x):
        return compute_c5(project_feasible(x))
    
    # If parent is available
    try:
        parent = np.array(kwargs['initial_h_values'])
        n_parent = len(parent)
        n_val_parent = n_values.index(n_parent) if n_parent in n_values else -1
    except KeyError:
        parent = None
        n_val_parent = -1
        
    # Determine best n_value based on time allocation
    time_per_n = budget_s / len(n_values)
    current_time = time.time()
    remaining_time = budget_s - (current_time - t0)
    
    # Main loop over different n_values
    constraint_strings = []
    for n_val in n_values:
        if time.time() > deadline:
            break
            
        n_val_points = n_val
        
        # Prepare particle positions
        if parent is not None and len(parent) == n_val_points:
            x0 = parent.copy()
        else:
            # Generate random starting point
            x0 = np.random.rand(n_val_points)
            x0 = project_feasible(x0)
        
        # Configuration parameters for this n
        n_particles_here = min(n_particles, int(remaining_time * 0.2 / 50))  # Reduce by time available
        max_iter_here = min(max_iter_pso, int(remaining_time * 0.6 / (30*n_particles_here)))
        
        # Particles and velocities initialization
        pos = np.zeros((n_particles_here, n_val_points))
        vel = np.zeros((n_particles_here, n_val_points))
        pbest = np.full((n_particles_here, n_val_points), float('inf'))  # Initialize with worst solutions
        gbest, best_fitness = None, float('inf')
        
        # Initialize particles
        for i in range(n_particles_here):
            # Random starting point
            pos[i] = np.random.rand(n_val_points)
            pos[i] = project_feasible(pos[i])
            pbest[i] = np.inf
            fitness = compute_c5(pos[i])
            pbest[i] = fitness
            
            # Update best
            if fitness < best_fitness:
                best_fitness = fitness
                gbest = pos[i].copy()
        
        # PSO parameters
        w = 0.9  # Inertia weight
        c1 = 2.0  # Cognitive parameter
        c2 = 2.0  # Social parameter
        
        # Main PSO loop
        for it in range(max_iter_here):
            if time.time() > deadline:
                break
                
            # Update velocities and positions
            r1, r2 = np.random.rand(), np.random.rand()
            for i in range(n_particles_here):
                # Standard PSO update
                r1, r2 = np.random.rand(), np.random.rand()
                vel[i] = (w * vel[i] + 
                          c1 * r1 * (pbest[i] - pos[i]) + 
                          c2 * r2 * (gbest - pos[i]))
                
                # Limit velocity magnitude
                max_vel = 0.2
                vel[i] = np.clip(vel[i], -max_vel, max_vel)
                
                # Update position
                new_pos = pos[i] + vel[i]
                pos[i] = project_feasible(new_pos)
                
                # Update particle best
                fitness = compute_c5(pos[i])
                if fitness < pbest[i]:
                    pbest[i] = fitness
                    
                    # Update global best
                    if fitness < best_fitness:
                        best_fitness = fitness
                        gbest = pos[i].copy()
            
            # Record time used
            elapsed = time.time() - t0
            time_used = elapsed
            remaining = budget_s - time_used
            
            # Check if improvement found
            if best_solution is None or best_c > best_fitness:
                best_solution = gbest.copy()
                best_c = best_fitness
        
        # Call SLSQP for refinement after PSO
        try:
            # Prepare SLSQP with constraints and bounds
            bounds = [(0.0, 1.0)] * n_val_points
            # Here, we use the correct target value based on n
            target = n_val_points / 2
            con = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
            
            # Use smoother objective if available
            # For now, use the original objective
            def objective_func(x):
                return compute_c5(x)
                
            # First refinement with SLSQP
            result = minimize(objective_func, gbest, method='SLSQP', bounds=bounds, constraints=con)
            
            # Update best if improved
            if result.success:
                final_x = result.x
                c_val = compute_c5(final_x)
                if c_val < best_c:
                    best_c = c_val
                    best_solution = final_x.copy()
        except Exception as e:
            print("SLSQP refinement failed: ", str(e))
            pass
        
        # Monitor time
        if time.time() > deadline:
            break
    
    # Final adjustments
    if best_solution is not None:
        # We will return the one with the smallest C5
        # But need to ensure we return the solution dimensions
        n_final = len(best_solution)
    
    # Fallback if unable to find solution
    if best_solution is None:
        n_final = 250
        best_solution = np.full(n_final, 0.5)
        best_c = compute_c5(best_solution)
    
    # Return results
    return best_solution, best_c, n_final
