Wait, let me be more careful. For shift k (can be positive or negative):
A(k) = #{i: 0≤i<n, 0≤i+k<n, best[i]=1, best[i+k]=0}

The contribution of position p as a "source" (best[p]=1): exists if 0≤p+k<n and best[p+k]=0.
The contribution of position p as a "target" (best[p+k']=0 where p=p'+k'): exists if p-k≥0 and best[p-k]=1. (Here p' = p-k, and we need best[p-k]=1 and best[p]=0.)

So for the swap (p: 1→0, q: 0→1):
- p loses its "source" contribution: -[0≤p+k*<n and best[p+k*]=0]
- p gains a "target" contribution: +[p-k*≥0 and 0≤p-k*<n and best[p-k*]=1]
  (Before: best[p]=1, so (1-best[p])=0, no target contribution. After: best[p]=0, so (1-best[p])=1.)
- q gains a "source" contribution: +[0≤q+k*<n and best[q+k*]=0]
- q loses its "target" contribution: -[q-k*≥0 and 0≤q-k*<n and best[q-k*]=1]
  (Before: best[q]=0, so (1-best[q])=1, target contribution exists if best[q-k*]=1. After: best[q]=1, no target contribution.)

Special cases (when p+k*=q or q+k*=p etc.):
- If p+k*=q: best[p+k*]=best[q]=0, so "p loses source" = -1. Also q-k*=p, best[q-k*]=best[p]=1, so "q loses target" = -1. Total so far: -2.
- If q+k*=p: best[q+k*]=best[p]=1, so "q gains source" = 0 (since best[q+k*]≠0). Also p-k*=q, best[p-k*]=best[q]=0, so "p gains target" = 0. Total: 0.
- If p-k*=q: best[p-k*]=best[q]=0, so "p gains target" = 0. Also q+k*=p, best[q+k*]=best[p]=1, so "q gains source" = 0. Total: 0.

So the formula handles special cases correctly if I just use the "before" values.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def delta_a(h, n, p, q, k):
    """Delta A(k) for swap h[p]:1->0, h[q]:0->1."""
    d = 0
    if 0 <= p + k < n and h[p + k] == 0:
        d -= 1
    if 0 <= p - k < n and h[p - k] == 1:
        d += 1
    if 0 <= q + k < n and h[q + k] == 0:
        d += 1
    if 0 <= q - k < n and h[q - k] == 1:
        d -= 1
    return d

def guided_ls(h, n, deadline, rng, max_iters=200000):
    """Guided local search for binary h."""
    best = h.copy()
    best_c = compute_c5(best, n)
    tabu = {}
    tabu_len = 20
    it = 0
    stuck = 0
    
    while time.time() < deadline and it < max_iters:
        it += 1
        c = np.correlate(best, 1 - best, mode='full')
        m_star = int(np.argmax(c))
        k_star = m_star - (n - 1)
        cur_max = c[m_star]
        
        # Get top 3 worst shifts
        top_idx = np.argsort(c)[::-1][:3]
        top_ks = [int(idx) - (n - 1) for idx in top_idx]
        
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        
        # Find problematic positions for k_star
        if k_star >= 0:
            prob = ones[(ones + k_star < n) & (best[np.minimum(ones + k_star, n-1)] == 0)]
        else:
            kk = -k_star
            prob = ones[(ones >= kk) & (best[ones - kk] == 0)]
        
        if len(prob) == 0:
            prob = ones
        
        # Score moves
        best_score = 0
        best_move = None
        prob_sample = prob if len(prob) <= 30 else rng.choice(prob, 30, replace=False)
        
        for p in prob_sample:
            # Try several zeros
            q_sample = rng.choice(zeros, min(15, len(zeros)), replace=False)
            for q in q_sample:
                if q == p:
                    continue
                key = (int(min(p, q)), int(max(p, q)))
                if it - tabu.get(key, -999) < tabu_len:
                    continue
                # Score: primary = delta for k_star, secondary = sum of deltas for top_ks
                score = -delta_a(best, n, p, q, k_star)  # want to decrease A(k*)
                # Check other shifts don't worsen too much
                ok = True
                for k2 in top_ks[1:]:
                    if delta_a(best, n, p, q, k2) > 1:
                        ok = False
                        break
                if not ok:
                    continue
                if score > best_score:
                    best_score = score
                    best_move = (int(p), int(q), key)
        
        if best_move is not None and best_score > 0:
            p, q, key = best_move
            best[p], best[q] = 0.0, 1.0
            tabu[key] = it
            stuck = 0
        else:
            # No improving move found, try a neutral or slight worsening move
            stuck += 1
            if stuck > 50:
                # Kick: random perturbation
                n_kick = 2 + rng.randint(3)
                for _ in range(n_kick):
                    p = int(ones[rng.randint(len(ones))])
                    q = int(zeros[rng.randint(len(zeros))])
                    best[p], best[q] = 0.0, 1.0
                    ones = np.where(best == 1)[0]
                    zeros = np.where(best == 0)[0]
                stuck = 0
    
    c_final = compute_c5(best, n)
    best_c = min(best_c, c_final)
    if c_final < best_c:
        best_c = c_final
    # Restore best if needed (we're tracking in-place, so best IS the current)
    return best, c_final

def simulated_annealing(h, n, deadline, rng, max_iters=300000):

    cur = h.copy()
    cur_c = compute_c5(cur, n)
    best = cur.copy()
    best_c = cur_c
    T = 2.0
    cooling = 0.9998
    it = 0

        c = np.correlate(cur, 1 - cur, mode='full')
        m_star = int(np.argmax(c))
        k_star = m_star - (n - 1)
        cur_max = c[m_star]
        
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]

        if k_star >= 0:
            prob = ones[(ones + k_star < n) & (cur[np.minimum(ones + k_star, n-1)] == 0)]
        else:

            prob = ones[(ones >= kk) & (cur[ones - kk] == 0)]

                if c_now < best_c:
                    best_c = c_now
                    best = cur.copy()

    c_final = compute_c5(cur, n)
    if c_final < best_c:
        best_c = c_final
        best = cur.copy()
    return best, best_c

def smooth_max(c, alpha):
    """Smooth maximum using log-sum-exp."""
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def continuous_refine(h, n, deadline):
    """Continuous refinement using L-BFGS-B."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    for alpha in [1000, 5000, 20000, 100000, 500000]:
        if time.time() > deadline - 5:
            break
        dx = 2.0 / n
        
        def obj(x):
            c = np.correlate(x, 1 - x, mode='full') * dx
            val = smooth_max(c, alpha)
            val += 1e5 * (x.sum() - n / 2.0) ** 2
            return val
        
        def grad(x):
            c = np.correlate(x, 1 - x, mode='full')
            m_star = int(np.argmax(c))
            k = m_star - (n - 1)
            g = np.zeros(n)
            if k >= 0:
                g[k:] += (1.0 - x[:n - k])
                g[:n - k] += -x[k:]
            else:
                g[:n + k] += (1.0 - x[-k:])
                g[-k:] += -x[:n + k]
            g *= dx
            g += 2e5 * (x.sum() - n / 2.0)
            return g
        
        try:
            res = minimize(obj, best_h, method='L-BFGS-B', jac=grad,
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-9})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c
