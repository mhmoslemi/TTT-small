That's reasonable. Let me also add a "smarter" 1D solver that uses the structure:

For the 1D problem, the optimal x* satisfies: there exist k1, k2 such that:
- a_k1 + b_k1 * x* = a_k2 + b_k2 * x* = max
- b_k1 < 0 < b_k2 (one line decreasing, one increasing at the optimum)

So the optimal x* = (a_k2 - a_k1) / (b_k1 - b_k2)

We can find this by:
1. For x=0, find k1 = argmax_k a_k
2. For x=1, find k2 = argmax_k (a_k + b_k)
3. If b_k1 > 0 and b_k2 < 0, the optimum is at x=0 or x=1
4. Otherwise, check the intersection of the line from k1 (at x=0) and k2 (at x=1)
5. Also check a few more candidates

This is a heuristic that often finds the exact optimum or very close to it.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    if sh.max() > 500:
        mx = sh.max()
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def coord_descent(h, n, n_sweeps=100, deadline=None):
    """Coordinate descent with 1D subproblem solving."""
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        
        r = np.correlate(h, 1 - h, mode='full') * dx
        
        for j in range(n):
            if deadline is not None and time.time() > deadline:
                return h, best_c
            
            # Compute b_k for coordinate j
            # h[j] appears in r(k) for lags k where 0<=j+k<n or 0<=j-k<n
            # In 'full' mode, lag index = k + (n-1)
            
            # b_k = (1 - h[j+k]) * [0<=j+k<n] - h[j-k] * [0<=j-k<n]
            # Only lags where b_k != 0 matter
            
            # k range: -(n-1) to (n-1)
            # mask1: 0 <= j+k < n => -j <= k < n-j
            # mask2: 0 <= j-k < n => j-n+1 <= k <= j
            
            k_min = max(-(n-1), -j, j - (n-1))
            k_max = min(n-1, n-1-j, j)
            
            if k_min > k_max:
                continue
            
            ks = np.arange(k_min, k_max + 1)
            
            # For each k, determine which masks are active
            m1 = (ks >= -j) & (ks < n - j)  # h[j+k] valid
            m2 = (ks >= j - (n-1)) & (ks <= j)  # h[j-k] valid
            
            b = np.zeros(len(ks))
            if m1.any():
                b[m1] = 1.0 - h[j + ks[m1]]
            if m2.any():
                b[m2] -= h[j - ks[m2]]
            
            # Skip if b is all zero
            if np.abs(b).max() < 1e-15:
                continue
            
            # a_k = r at lag k - h[j] * b_k
            # Lag index in full correlate: idx = k + (n-1)
            idx = ks + (n-1)
            a = r[idx] - h[j] * b
            
            # Solve: min_{x in [0,1]} max(a_k + b_k * x)
            # Use grid of points
            grid = np.linspace(0, 1, 50)
            vals = a + np.outer(grid, b)  # (50, len(ks))
            max_vals = vals.max(axis=1)
            
            # Also check the intersection-based candidate
            # Find k with max a (at x=0) and k with max (a+b) (at x=1)
            k_at_0 = np.argmax(a)
            k_at_1 = np.argmax(a + b)
            if b[k_at_0] < 0 and b[k_at_1] > 0:
                x_int = (a[k_at_0] - a[k_at_1]) / (b[k_at_1] - b[k_at_0])
                if 0 < x_int < 1:
                    grid = np.append(grid, x_int)
                    max_vals = np.append(max_vals, (a + b * x_int).max())
            
            best_idx = np.argmin(max_vals)
            new_x = grid[best_idx]
            
            if new_x != h[j]:
                h[j] = new_x
        
        # Project to feasible every sweep
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
        else:
            # If no improvement after several sweeps, we might be stuck
            # Try a perturbation
            if sweep > 0 and sweep % 20 == 0:
                h += 0.01 * np.random.randn(n)
                h = project_feasible(h, n)
    
    return h, best_c
```

Actually, for the 1D minimization of a convex piecewise-linear function, I can use the following exact algorithm:

4. If b_{k*} > 0, the minimum is to the left; if b_{k*} < 0, to the right

```python
def solve_1d(a, b, n_evals=30):
    """Solve min_{x in [0,1]} max_k (a_k + b_k * x) using bisection on subgradient."""
    lo, hi = 0.0, 1.0
    for _ in range(n_evals):
        mid = (lo + hi) / 2
        vals = a + b * mid
        k_star = np.argmax(vals)
        if b[k_star] > 0:
            hi = mid
        elif b[k_star] < 0:
            lo = mid
        else:
            break
    return (lo + hi) / 2
```
