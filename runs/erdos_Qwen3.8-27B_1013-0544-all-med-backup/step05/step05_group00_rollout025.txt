def compute_c5(h, n):

        top_lags = np.argsort(corr)[::-1][:num_hot]
        
        # Compute combined subgradient
        sg = np.zeros(n)
        for t in top_lags:
            k = int(t) - (n - 1)
            sg += compute_subgradient(h, n, k)
        sg /= num_hot
        
        # Rank candidates
        from_scored = sorted(ones, key=lambda i: -sg[i])
        to_scored = sorted(zeros, key=lambda j: sg[j])
        
        K_f = min(20, len(from_scored))
        K_t = min(20, len(to_scored))
        
        # Generate and rank candidates
        cand = []
        for i in from_scored[:K_f]:
            for j in to_scored[:K_t]:
                if i != j:
                    cand.append((sg[j] - sg[i], i, j))
        cand.sort()
        cand = cand[:max_candidates]
        
        # Evaluate
        best_swap = None
        best_new_max = cur_max + 1e-10
        
        for sc, i, j in cand:
            if time.time() > deadline:
                break
            h[i] = 0.0
            h[j] = 1.0
            new_corr = np.correlate(h, 1 - h, mode='full')
            new_max = float(new_corr.max())
            h[i] = 1.0
            h[j] = 0.0
            
            if new_max < best_new_max:
                best_new_max = new_max
                best_swap = (i, j, new_corr.copy())
        
        # Accept with annealing
        accept_thresh = cur_max
        if anneal_temp > 0 and rng.random() < 0.15:
            accept_thresh = cur_max + anneal_temp / dx
        
        if best_swap is not None and best_new_max <= accept_thresh - 1e-12:
            i, j, new_corr = best_swap
            h[i], h[j] = 0.0, 1.0
            corr = new_corr
            cur_max = best_new_max
            ones.remove(i)
            ones.append(j)
            zeros.append(i)
            zeros.remove(j)
            if cur_max * dx < best_c:
                best_c = cur_max * dx
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 300 == 0 and len(ones) > 0 and len(zeros) > 0:
                # Random kick: swap 2-3 random pairs
                n_kicks = rng.randint(1, 3)
                for _ in range(n_kicks):
                    i = int(rng.choice(ones))
                    j = int(rng.choice(zeros))
                    h[i], h[j] = 0.0, 1.0
                    ones.remove(i)
                    ones.append(j)
                    zeros.append(i)
                    zeros.remove(j)
                corr = np.correlate(h, 1 - h, mode='full')
                cur_max = float(corr.max())
                no_improve = 0
    
    return best_h, best_c
