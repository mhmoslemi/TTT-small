import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgradient(h, n, k):
    """Compute subgradient of C(k) w.r.t. h"""
    dx = 2.0 / n
    grad = np.zeros(n)
    lag = k - (n - 1)  # convert from correlation index to actual lag
    if lag >= 0:
        # C(lag) = dx * sum_{i=0}^{n-1-lag} h[i](1-h[i+lag])
        # dC/dh[j] = dx*(1-h[j+lag]) for j <= n-1-lag
        #           - dx*h[j-lag] for j >= lag
        for j in range(n - lag):
            grad[j] += dx * (1.0 - h[j + lag])
        for j in range(lag, n):
            grad[j] -= dx * h[j - lag]
    else:
        to = -lag
        for j in range(to):
            grad[j] += dx * (1.0 - h[j + to])
        for j in range(to, n):
            grad[j] -= dx * h[j - to]
    return grad

def momentum_subgrad(h0, n, deadline, max_iter=100000, alpha_list=None):
    """Nesterov-accelerated subgradient descent"""
    if alpha_list is None:
        alpha_list = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    h = h0.copy()
    v = np.zeros(n)  # momentum
    mu = 0.9  # momentum coefficient
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g_all = compute_all_g(h, n)
        m = g_all.max()
        k_star = int(np.argmax(g_all))
        
        # Subgradient

                h = h_trial.copy()
                improved = True
                break

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def weighted_grad(h, n, C, alpha, dx):
    C_max = C.max()
    w = np.exp(alpha * (C - C_max))
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    threshold = 1e-5
    active = np.where(w > threshold)[0]
    for k_idx in active:
        d = int(k_idx) - (n - 1)
        wk = w[k_idx] * dx
        if d >= 0:
            grad[:n-d] += wk * (1.0 - h[d:])
            if d > 0:
                grad[d:] -= wk * h[:n-d]
        else:
            p = -d
            grad[p:] += wk * (1.0 - h[:n-p])
            if p > 0:
                grad[:n-p] -= wk * h[p:]
    return grad

        if time.time() > deadline:
            break
        t = min((it + 1) / max_iter, 1.0)
        alpha = alpha_s * (alpha_e / alpha_s) ** t
        lr = lr_s * (lr_e / lr_s) ** t
        C = compute_all_g(h, n)
        grad = weighted_grad(h, n, C, alpha, dx)
        v = mu * v - lr * grad

        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant % 3000 == 0:
                v *= 0.3
                stagnant = 0
    return best_h, best_c

def binary_local_search(h0, n, deadline, max_sweeps=300000, seed=123):
    h = to_binary(h0, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
