    h = h.copy()

            if tot < 1e-15:

        g[s:] -= h[:n-s]
    return g

    """Projected gradient method with momentum and adaptive step."""
    target = n / 2.0
    h = project_sum(h, target)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    v = np.zeros(n)  # momentum

        if new_c <= curr + 1e-14:
            h = h_new
            step = min(step * 1.04, 0.5)
        else:
            step *= 0.55
            v *= 0.3
            if step < 1e-11:
                break
        if stall > 3000:
            break
    return best_h, best_c

def _sum_fun(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(len(x))

def slsqp_opt(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1 - hx, mode='full') * dx
        m = float(c.max())
        shifted = alpha * (c - m)
        mx = float(shifted.max())
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_sum(res.x, n / 2.0)
    except Exception:
        return h.copy()

def coord_polish(h, n, deadline, top_k=1):
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.005
    stall = 0
    while time.time() < deadline:
        c = get_corr(h, n)
        curr = float(c.max())
        if curr < best_c - 1e-15:
            best_c = curr
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        idx = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for ki in idx:
            g += corr_grad(h, n, int(ki) - (n - 1))
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        d = min(step, h[i_src] * 0.99, (1.0 - h[i_dst]) * 0.99)
        if d < 1e-13:
            step *= 0.5
            stall += 1
            if stall > 800:
                break
            continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = compute_c5(h, n)
        if new_c < curr - 1e-16:
            step = min(step * 1.3, 0.05)
            stall = 0
        else:
            h[i_src] += d
            h[i_dst] -= d
            step *= 0.5
            stall += 1
    return best_h, best_c

def make_structured_init(n, freq, amp):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amp * np.sin(np.pi * freq * x)
    return project_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    best_h = None
    best_c = float('inf')

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = None
        n0 = 250

    # Phase 1: Optimize initial solution with PGM
    if h0 is not None:
        alloc = (deadline - t0) * 0.28
        h_opt, c_opt = pgm(h0, n0, t0 + alloc, top_k=3, mu=0.85, step0=0.04)
        best_h, best_c = h_opt, c_opt
        # SLSQP polish
        if time.time() < deadline - 80:
            for a in [2000, 10000]:
                if time.time() > deadline - 80:
                    break
                ht = slsqp_opt(best_h, n0, a, maxiter=1500)
                cc = compute_c5(ht, n0)
                if cc < best_c - 1e-14:
                    best_c = cc
                    best_h = ht
        # Coordinate polish
        if time.time() < deadline - 40:
            best_h, best_c = coord_polish(best_h, n0, min(deadline - 40, time.time() + 80))

    # Phase 2: Try other n values
    other_ns = [200, 300, 350, 400]
    for n_try in other_ns:
        if time.time() > deadline - 100:
            break
        alloc = min(120, (deadline - time.time() - 50) * 0.22)
        if alloc < 35:
            break
        if best_h is not None:
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, np.linspace(0, 2, len(best_h)), best_h)
        else:
            h_start = make_structured_init(n_try, 5, 0.3)
        h_start = project_sum(h_start, n_try / 2.0)
        h_opt, c_opt = pgm(h_start, n_try, time.time() + alloc, top_k=3, mu=0.85)
        if time.time() < deadline - 30:
            ht = slsqp_opt(h_opt, n_try, 5000, maxiter=1500)
            c2 = compute_c5(ht, n_try)
            if c2 < c_opt:
                c_opt, h_opt = c2, ht
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Structured restarts at best n
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 25
        n_restarts = max(4, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            if r % 3 == 0:
                scale = 0.008 + 0.04 * (r / max(1, n_restarts - 1))
                hp = best_h + rng.normal(0, scale, n)
            elif r % 3 == 1:
                hp = make_structured_init(n, 5 + r // 3, 0.25 + 0.1 * (r % 5))
                hp = 0.5 * hp + 0.5 * best_h
            else:
                hp = best_h.copy()
                n_flip = n // 5
                idx_flip = rng.choice(n, n_flip, replace=False)
                hp[idx_flip] = 1.0 - best_h[idx_flip]
            hp = project_sum(hp, n / 2.0)
            alloc = min(60, (deadline - time.time() - 25) * 0.3)
            if alloc < 20:
                break
            h_opt, c_opt = pgm(hp, n, time.time() + alloc, top_k=3, mu=0.82)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Final fine polish
    if best_h is not None:
        n = len(best_h)
        if time.time() < deadline - 10:
            best_h, best_c = coord_polish(best_h, n, deadline - 5)
        if time.time() < deadline - 5:
            h_pg = pgm(best_h, n, deadline - 2, top_k=5, mu=0.95, step0=0.01)
            if h_pg[1] < best_c:
                best_h, best_c = h_pg
        best_h = project_sum(best_h, n / 2.0)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        n = 250
        best_h = project_sum(np.full(n, 0.5), n / 2.0)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
