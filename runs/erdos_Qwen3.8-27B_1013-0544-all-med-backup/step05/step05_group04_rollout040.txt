def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    cs = c - c_max
    ec = np.exp(alpha * cs)
    obj = c_max + np.log(ec.sum()) / alpha
    w = ec / ec.sum()

            grad[:n-t] += wk * (1.0 - h[t:])

    h = tm.astype(float)
    return project_feasible(h, n)

def block_init(n, period):
    h = np.zeros(n)
    half = period // 2
    h[:n] = np.where(np.arange(n) % period < half, 1.0, 0.0)
    return project_feasible(h, n)

def directed_pair(h, n, deadline):
    best_c = compute_c5(h, n)
    for d in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        while time.time() < deadline:
            improved = False
            c_full = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
            k_star = int(np.argmax(c_full))
            s = k_star - (n - 1)
            grad = np.zeros(n)
            if s >= 0:
                grad[s:] += (1.0 - h[:n - s])
                grad[:n - s] -= h[s:]
            else:
                t = -s
                grad[:n - t] += (1.0 - h[t:])
                grad[t:] -= h[:n - t]
            inc = np.where((grad < -0.05) & (h < 1.0 - d + 1e-9))[0]
            dec = np.where((grad > 0.05) & (h > d - 1e-9))[0]
            max_pairs = 200
            npairs = 0
            for i in inc:
                if npairs >= max_pairs or time.time() > deadline:
                    break
                for j in dec:
                    if npairs >= max_pairs or time.time() > deadline:
                        break
                    npairs += 1
                    h[i] += d
                    h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d
            if not improved:
                break
    return h, best_c

def full_pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def slsqp_polish(h, n, alpha, maxiter=2000):
    target = n / 2.0
    def obj(x):
        dx = 2.0 / n
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        cs = c - m
        mx = cs.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(cs - mx)))) / alpha
        return m + np.log(np.sum(np.exp(cs))) / alpha
    def obj_grad(x):
        o, g = smooth_obj_grad(x, n, alpha)
        return o, g
    def con_fun(x):
        return x.sum() - target
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, args=(), jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(120, 0.5)
        n_init = 120

    best_h = None
    best_c = float('inf')

    # Generate diverse initializations
    inits = [h_init.copy()]
    inits.append(thue_morse_init(n_init))
    for per in [4, 6, 8, 12, 16, 24]:
        inits.append(block_init(n_init, per))
    for _ in range(8):
        inits.append(project_feasible(rng.random(n_init), n_init))
    # Symmetric patterns
    for _ in range(4):
        half = rng.random(n_init // 2)
        sym = np.concatenate([half, half[::-1]])
        if len(sym) > n_init:
            sym = sym[:n_init]
        elif len(sym) < n_init:
            sym = np.concatenate([sym, sym[::-1]])[:n_init]
        inits.append(project_feasible(sym, n_init))

    # Quick evaluation
    scores = [compute_c5(hi, n_init) for hi in inits]
    order = np.argsort(scores)
    top_k = min(6, len(inits))
    top_inits = [inits[order[i]] for i in range(top_k)]

    # Phase 1: Gradient optimize top candidates
    remaining = deadline - time.time()
    alloc_each = min(100, remaining * 0.45 / top_k)
    for idx in range(top_k):
        if time.time() > deadline - 120:
            break
        alloc = min(alloc_each, deadline - time.time() - 120)
        if alloc < 15:
            break
        h_opt, c_opt = grad_optimize(top_inits[idx], n_init, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()

    # Phase 2: Try a different n
    if best_h is not None and time.time() < deadline - 100:
        n_alt = 200 if n_init >= 250 else 300
        x_old = np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, n_alt)
        h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_alt)
        alloc = min(120, (deadline - time.time()) * 0.25)
        if alloc > 40:
            h_opt, c_opt = grad_optimize(h_interp, n_alt, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()

    # Phase 3: Directed pair search on best
    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        best_h, best_c = directed_pair(best_h, n, deadline - 40)

    # Phase 4: SLSQP polish
    if best_h is not None and time.time() < deadline - 30:
        n = len(best_h)
        for a in [50000, 200000]:
            if time.time() > deadline - 30:
                break
            ht = slsqp_polish(best_h, n, a, maxiter=1500)
            c = compute_c5(ht, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = ht

    # Phase 5: Final pair search
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        if n <= 200:
            best_h, best_c = full_pair_search(best_h, n, deadline - 2,
                                               [0.005, 0.002, 0.001, 0.0005, 0.0002])
        else:
            best_h, best_c = directed_pair(best_h, n, deadline - 2)

    # Phase 6: Tiny final moves
    if best_h is not None and time.time() < deadline - 1:
        n = len(best_h)
        if n <= 180:
            best_h, best_c = full_pair_search(best_h, n, deadline - 0.5,
                                               [0.0005, 0.0002, 0.0001])
        else:
            best_h, best_c = directed_pair(best_h, n, deadline - 0.5)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
