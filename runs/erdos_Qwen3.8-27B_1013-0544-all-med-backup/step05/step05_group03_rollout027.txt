    idx = np.argsort(h)[::-1]

def compute_D(hb, n):
    ns = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    D = np.zeros((n, ns), dtype=np.int8)
    for p in range(n):
        fv = (p + shifts >= 0) & (p + shifts < n)
        D[p, fv] = (1 - hb[p + shifts[fv]]).astype(np.int8)
        sv = (p - shifts >= 0) & (p - shifts < n)
        D[p, sv] -= hb[p - shifts[sv]].astype(np.int8)
    return D

def update_D(D, i, j, n):
    """Update D after hb[i]: 0->1, hb[j]: 1->0"""
    ns = 2 * n - 1
    off = n - 1
    # i: 0->1 causes D to decrease at affected positions
    for p in range(max(0, i - n + 1), min(n - 1, i) + 1):
        D[p, i - p + off] -= 1
    for p in range(i, min(n - 1, i + n - 1) + 1):

        D[p, j - p + off] += 1
    for p in range(j, min(n - 1, j + n - 1) + 1):
        D[p, p - j + off] += 1
