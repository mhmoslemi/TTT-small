def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-10
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def compute_D(h, i, j, n):
    ks = np.arange(2*n-1)
    s = ks - n + 1
    D = np.zeros(2*n-1)
    m = (j+s >= 0) & (j+s < n)
    D[m] += h[j+s[m]]
    m = (j-s >= 0) & (j-s < n)
    D[m] += h[j-s[m]]
    m = (i+s >= 0) & (i+s < n)
    D[m] -= h[i+s[m]]
    m = (i-s >= 0) & (i-s < n)
    D[m] -= h[i-s[m]]
    return D

def coord_descent(h, n, deadline, rng=None, max_sweeps=200, random_pairs=False):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    best_c = float(c.max())
    it = 0
    for sweep in range(max_sweeps):

            break

            D = compute_D(h, i, j, n)

    target = n / 2.0

def compute_D(h, i, j, n):
    ks = np.arange(2 * n - 1)
