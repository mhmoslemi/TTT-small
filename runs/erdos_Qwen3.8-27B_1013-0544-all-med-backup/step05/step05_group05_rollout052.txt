import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    g = compute_all_g(h, n)
    m = g.max()
    sh = g - m
    return m + np.log(np.exp(sh - sh.max()).sum() / len(sh)) / alpha

def slsqp_opt(h0, n, alpha, maxiter, deadline):
    def obj(x):
        return smooth_max_obj(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def subgrad_polish(h0, n, deadline, max_iter=200000):
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    lr = 0.5
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(best_h, n)
        m = g.max()
        # Use top-k lags for smoother gradient
        k = int(np.argmax(g))
        grad = np.zeros(n)
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += dx * (1.0 - best_h[:n - lag])
            grad[:n - lag] -= dx * best_h[lag:]
        else:
            to = -lag
            grad[:n - to] += dx * (1.0 - best_h[to:])
            grad[to:] -= dx * best_h[:n - to]
        ng = np.linalg.norm(grad)
        if ng < 1e-15:
            break
        grad /= ng
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001, 0.0005, 0.0001]:
            step = lr * f
            ht = project_feasible(best_h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-14:
                best_h = ht
                found = True
                break
        if not found:
            lr *= 0.8
            if lr < 1e-14:
                break
        else:
            c = compute_c5(best_h, n)
            if c < best_c - 1e-15:
                best_c = c
    return best_h, best_c

def binary_local_search(h0, n, deadline, seed=123, max_sweeps=50000):
    target = n // 2
    idx = np.argsort(h0)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)

        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)

        for i in zeros:
            if time.time() > deadline:

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, params):
    n, alpha = params
    g = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = g.max()
    sh = g - m
    return m + np.log(np.exp(sh - sh.max()).sum() / len(sh)) / alpha

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    g = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = g.max()
    sh = g - m
    return m + np.log(np.exp(sh - sh.max()).sum() / len(sh)) / alpha

def sum_eq(x, n):
    return x.sum() - n / 2.0
