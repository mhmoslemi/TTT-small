    # h is binary, sum(h) = n/2
    n1 = 2*n - 1
    center = n - 1
    
    # Precompute P and Q
    # s ranges from -(n-1) to (n-1), indexed as 0 to 2n-2

    # where s = s_range[s_idx]
    P = np.zeros((n, n1))
    Q = np.zeros((n, n1))
