import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_step(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and not np.any(np.isnan(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def coord_m_d(h, i, N, d_vals):
    m_d = np.zeros(2 * N - 1)
    for di in range(2 * N - 1):
        d = d_vals[di]
        val = 0.0
        if d >= 0:
            if d <= i <= N - 1:
                val += (1.0 - h[i - d])
            if 0 <= i <= N - 1 - d:
                val -= h[i + d]
        else:
            dp = -d
            if 0 <= i <= N - 1 - dp:
                val += (1.0 - h[i + dp])
            if dp <= i <= N - 1:
                val -= h[i - dp]
        m_d[di] = val
    return m_d

def coordinate_descent(h, n, deadline, max_sweeps=2000, perturb_every=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    N = n
    d_vals = np.arange(2 * N - 1) - (N - 1)
    rng = np.random.RandomState(12345)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        g = compute_g(h, N)
        improved = False
        for i in range(N):
            if time.time() > deadline:
                break
            m_d = coord_m_d(h, i, N, d_vals)
            c_d = g - m_d * h[i]
            lo, hi = 0.0, 1.0
            for _ in range(35):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(c_d + m_d * m1)
                f2 = np.max(c_d + m_d * m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            t_new = (lo + hi) / 2.0
            delta = t_new - h[i]
            if abs(delta) > 1e-13:
                h[i] = t_new
                g += m_d * delta
                improved = True
        if not improved:
            break
        h = project_feasible(h, N)
        c = compute_c5(h, N)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:

        if stall >= perturb_every and sweep % perturb_every == 0:

            h = project_feasible(h + pert, N)
            c = compute_c5(h, N)
            if c > best_c + 0.01:

    return best_h, best_c

def pair_descent(h, n, deadline, K=15, max_sweeps=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    N = n
    d_vals = np.arange(2 * N - 1) - (N - 1)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        g = compute_g(h, N)
        cur_max = g.max()
        top = np.argsort(g)[::-1][:K]
        improved = 0
        for i in range(N):
            if time.time() > deadline:
                break
            m_i = coord_m_d(h, i, N, d_vals)
            for j in range(i + 1, N):

                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])

                lo, hi = d_lo, d_hi

                        lo = m1

                        h[j] -= delta
                        g += s * delta

                        top = np.argsort(g)[::-1][:K]
