        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_run(h, n, alpha, maxiter=200):

        g = np.correlate(x, 1 - x, mode='full') * (2.0/n)
        m = g.max()

        sm = shifted.max()

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})

        return h

import numpy as np
from scipy.optimize import minimize
import time

_slsqp_n = 250
_slsqp_alpha = 1000

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_obj(x):
    g = np.correlate(x, 1 - x, mode='full') * (2.0 / _slsqp_n)
    m = g.max()
    shifted = _slsqp_alpha * (g - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / _slsqp_alpha

    _slsqp_alpha = alpha

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
