            alpha_ls *= 0.5
        if not improved:
            step = max(step * 0.85, 1e-10)

    return best_h, best_c

def pair_polish(h, n, deadline, max_pairs=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    c = get_corr(h, n)
    k_star = int(np.argmax(c))
    g = lag_gradient(h, n, k_star)
    rng = np.random.RandomState(777)
    step = 0.005
    stall = 0

    for it in range(max_pairs):
        if time.time() > deadline:
            break
        if stall > 300:
            step *= 0.8
            stall = 0
            if step < 1e-9:
                break
            c = get_corr(h, n)
            k_star = int(np.argmax(c))
            g = lag_gradient(h, n, k_star)

        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break

        d = min(step, h[i_src] * 0.99, (1.0 - h[i_dst]) * 0.99)
        if d < 1e-10:
            stall += 10
            continue

        h[i_src] -= d
        h[i_dst] += d
        new_c = compute_c5(h, n)
        if new_c < best_c - 1e-14:
            best_c = new_c
            best_h = h.copy()
            stall = 0
            step = min(step * 1.05, 0.02)
            c = get_corr(h, n)
            k_star = int(np.argmax(c))
            g = lag_gradient(h, n, k_star)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1

    return best_h, best_c

def structural_perturb(h, n, rng):
    """Apply a structural perturbation to escape local optima."""
    h_new = h.copy()
    op = rng.randint(4)
    if op == 0:
        # Shift a segment
        seg_len = rng.randint(n // 10, n // 3)
        start = rng.randint(0, n - seg_len)
        shift = rng.randint(-n // 10, n // 10 + 1)
        segment = h_new[start:start + seg_len].copy()
        h_new = project_feasible(h_new, n)
        # Rotate within bounds
        h_new[start:start + seg_len] = np.roll(segment, shift)
    elif op == 1:
        # Add sinusoidal perturbation
        freq = rng.uniform(1, 5)
        amp = rng.uniform(0.05, 0.2)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2, n)
        h_new += amp * np.sin(freq * np.pi * x + phase)
    elif op == 2:
        # Flip a segment around 0.5
        seg_len = rng.randint(n // 8, n // 3)
        start = rng.randint(0, n - seg_len)
        h_new[start:start + seg_len] = 1.0 - h_new[start:start + seg_len]
    else:
        # Gaussian noise
        h_new += rng.normal(0, 0.1, n)
    return project_feasible(h_new, n)

def optimize_for_n(h_init, n, deadline):
    h = project_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP with increasing alpha
    alphas = [500, 2000, 10000, 50000]
    for a in alphas:
        if time.time() > deadline - 10:
            break
        ht = slsqp_smooth(best_h, n, a, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht

    # Phase 2: Subgradient with momentum
    if time.time() < deadline - 15:
        sub_deadline = min(deadline - 10, time.time() + (deadline - time.time()) * 0.4)
        ht, c = subgradient_polish(best_h, n, sub_deadline, top_k=10)
        if c < best_c:
            best_c = c
            best_h = ht

    # Phase 3: Pair polish
    if time.time() < deadline - 5:
        ht, c = pair_polish(best_h, n, deadline - 3)
        if c < best_c:
            best_c = c
            best_h = ht

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        h_parent = None
        n_parent = 250

    best_h = None
    best_c = float('inf')

    if h_parent is not None:
        x_parent = np.linspace(0, 2, n_parent)

    # Try parent n first with most time
    if h_parent is not None:
        alloc1 = min((deadline - t0) * 0.45, 400)
        h_opt, c_opt = optimize_for_n(h_parent.copy(), n_parent, t0 + alloc1)
        best_h, best_c = h_opt, c_opt

    # Try other n values
    n_candidates = [100, 150, 200, 250, 300, 400]
    n_candidates = [n for n in n_candidates if n != n_parent]
    
    remaining = deadline - time.time()
    per_n = remaining * 0.3 / max(1, len(n_candidates))
    
    for n in n_candidates:
        if time.time() > deadline - 60:
            break
        if per_n < 30:
            break
        if h_parent is not None:
            x_new = np.linspace(0, 2, n)
            h_init = np.interp(x_new, x_parent, h_parent)
        else:
            h_init = np.full(n, 0.5)
        h_init = project_feasible(h_init, n)
        alloc = min(per_n, deadline - time.time() - 50)
        if alloc < 25:
            break
        h_opt, c_opt = optimize_for_n(h_init, n, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Multi-start with structural perturbations on best solution
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(3, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            hp = structural_perturb(best_h, n, rng)
            alloc = min(50, (deadline - time.time() - 25) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_for_n(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final ultra-fine polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        ht, c = pair_polish(best_h, n, deadline - 2, max_pairs=200000)
        if c < best_c:
            best_c = c
            best_h = ht
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        n = 250
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
