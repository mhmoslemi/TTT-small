- But g_k is quadratic, so this isn't convex

This is a linear program! We can solve it with CVXPY.

This gives a "cutting plane" or "linearization" approach:
1. Start with h₀
2. At each iteration, solve the LP to get h₁
3. Evaluate the true g_k at h₁
4. If max g_k(h₁) < previous, update
5. Add new cuts (linearizations at h₁)
6. Repeat

This is essentially the **cutting plane method** for the non-smooth objective!

```python
import cvxpy as cp

def cutting_plane(h, n, deadline, max_iters=100):
    dx = 2.0 / n
    t = cp.Variable()
    h_var = cp.Variable(n)
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    
    cuts = []
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Add cuts from current point
        g = compute_all_g(best_h, n)
        for k in range(len(g)):
            # Linearize g_k at best_h
            grad_k = grad_gk(best_h, k, n, dx)
            # g_k(h) ≈ g_k(best_h) + grad_k @ (h - best_h)
            lhs = g[k] + grad_k @ (h_var - best_h)
            cuts.append(lhs <= t)
        
        # Solve LP
        obj = cp.Minimize(t)
        prob = cp.Problem(obj, constraints + cuts[-len(g):])  # Only use recent cuts
        prob.solve()
        
        if h_var.value is None:
            break
        
        new_h = np.clip(h_var.value, 0, 1)
```

```python
    """Gradient of g_k = dx * sum_i h[i]*(1-h[i+shift]) with respect to h."""
    shift = k - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        # h[i] term: (1-h[i+shift]) for i in [0, n-1-shift]
        grad[:n-shift] += (1.0 - h[shift:n])
        # (1-h[i+shift]) term: -h[i-shift] for i in [shift, n-1]
        grad[shift:] -= h[:n-shift]
    else:
        s = -shift
        # h[i] term: (1-h[i-s]) for i in [s, n-1]
        grad[s:] += (1.0 - h[:n-s])
        # (1-h[i-s]) term: -h[i+s] for i in [0, n-1-s]
        grad[:n-s] -= h[s:n]
```

    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def grad_gk_at(h, shift, n, dx):
    grad = np.zeros(n)
    if shift >= 0:
        if n - shift > 0:
            grad[:n-shift] += (1.0 - h[shift:n])
        if shift > 0:
            grad[shift:] -= h[:n-shift]
    else:
        s = -shift
        if n - s > 0:
            grad[s:] += (1.0 - h[:n-s])
        if s > 0:
            grad[:n-s] -= h[s:n]
    return dx * grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lp_cutting(h0, n, deadline, M=20):
    dx = 2.0 / n
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    for it in range(200):
        if time.time() > deadline:
            break
        g = compute_all_g(best_h, n)
        top = np.argsort(g)[-M:]
        t = cp.Variable()
        hv = cp.Variable(n)
        cons = [hv >= 0, hv <= 1, cp.sum(hv) == n/2]
        for k in top:
            gi = int(k)
            shift = gi - (n-1)
            gk = g[gi]
            gk_grad = grad_gk_at(best_h, shift, n, dx)
            cons.append(gk + gk_grad @ (hv - best_h) <= t)
        prob = cp.Problem(cp.Minimize(t), cons)
        try:
            prob.solve(solver=cp.ECOS, max_iter=10000)
        except:
            break
        if hv.value is None:
            break
        # Try full step and fractional steps
        improving = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.1]:
            nh = (1-frac) * best_h + frac * np.clip(hv.value, 0, 1)
            nh = project_feasible(nh, n)
            nc = compute_c5(nh, n)
            if nc < best_c - 1e-13:
                best_c = nc
                best_h = nh
                improving = True
                break
        if not improving:
            break
    return best_h, best_c

def subgrad_momentum(h0, n, deadline, max_iter=200000):
    dx = 2.0 / n
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    x = h0.copy()
    v = np.zeros(n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(x, n)
        m = g.max()
        ks = int(np.argmax(g))
        shift = ks - (n-1)
        grad = grad_gk_at(x, shift, n, dx)
        v = 0.8 * v + grad
        lr = 0.2 / (1 + 0.001 * it)
        x = project_feasible(x - lr * v, n)
        c = compute_c5(x, n)
        if c < best_c:
            best_c = c
            best_h = x.copy()
    return best_h, best_c

def binary_polish(h0, n, deadline):
    target = n // 2
    idx = np.argsort(h0)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    best_c = compute_c5(hb, n)
    for _ in range(5000):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        for i in zeros:
            for j in ones:
                hb[i], hb[j] = 1.0, 0.0
                c = compute_c5(hb, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                    break
                else:
                    hb[i], hb[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    return hb, best_c
