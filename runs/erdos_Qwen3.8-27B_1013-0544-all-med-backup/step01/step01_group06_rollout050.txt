**Key insight**: The problem is to maximize min_k R(k) where R(k) = autocorrelation. This is a semi-definite programming (SDP) type problem in disguise, but the bilinear nature makes it non-convex.

Minimize -t + λ * sum_k max(0, t - R(k))²

Or equivalently, minimize the "excess" above t:
Minimize sum_k max(0, R_max - R(k))²

where R_max = max_k R(k). This flattens the autocorrelation.

import time

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(get_corr(h)) * (2.0 / n))

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad(h, alpha):
    """Fast gradient using correlate"""
    n = len(h)
    corr = get_corr(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.correlate(w, 1.0 - 2.0 * h, mode='valid')
    return grad * (2.0 / n)

    lr = 0.1
    best_h = h
    best_c5 = c5_cur
    best_lr = lr
    for _ in range(max_iter):
        h_new = project_box_sum(h - lr * grad, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            best_lr = lr
        lr *= 0.5
        if lr < 1e-6:
            break
    return best_h, best_c5

    alpha = alpha_start

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break

                if c5_pert < best_c5:

                    best_c5 = c5_pert
                alpha = alpha_start
                stall = 0
    return best_h, best_c5

    n = len(best_h)

            h_try = best_h.copy()
            h_try[s1:s1+n1] -= amount
