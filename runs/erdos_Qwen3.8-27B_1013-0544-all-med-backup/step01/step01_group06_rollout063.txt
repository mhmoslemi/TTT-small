        v[valid_add] += -(1 - h[idx_add])

        idx_sub = pos - k_range[valid_sub]

        c5 = new_c5
        improved = True

import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_sum(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max(corr, alpha):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def gradient_of_smooth_max(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= np.sum(w)
    # Gradient of smooth_max with respect to h
    # d/dh[i] of corr[k] for k = k_idx - (n-1):
    # For k > 0: d/dh[i] = -(1-h[i-k]) if 0<=i-k<n, plus (1-h[i+k]) if 0<=i+k<n... 
    # Actually let me use the formula from before
    grad = np.zeros(n)
    for k_idx in range(2*n-1):
        if w[k_idx] < 1e-10:
            continue
        k = k_idx - (n-1)
        if k == 0:
            # d/dh[i] of h[i](1-h[i]) = 1-2h[i]
            grad += w[k_idx] * (1 - 2*h)
        elif k > 0:
            # corr[k_idx] = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            # d/dh[i] = (1-h[i+k]) for i in [0, n-1-k]
            #           -h[i-k] for i in [k, n-1]
            idx1 = np.arange(0, n-k)
            grad[idx1] += w[k_idx] * (1 - h[idx1 + k])
            idx2 = np.arange(k, n)
            grad[idx2] -= w[k_idx] * h[idx2 - k]
        else:
            p = -k
            # corr[k_idx] = sum_{i=p}^{n-1} h[i]*(1-h[i-p])
            idx1 = np.arange(p, n)
            grad[idx1] += w[k_idx] * (1 - h[idx1 - p])
            idx2 = np.arange(0, n-p)
            grad[idx2] -= w[k_idx] * h[idx2 + p]
    return grad * (2.0 / n)

    def compute_V(pos):
        v = np.zeros(2*n - 1)

        idx_add = pos + k_range[valid_add]
        v[valid_add] += -(1.0 - h[idx_add])
        valid_sub = (pos - k_range >= 0) & (pos - k_range < n) & (k_range != 0)
        idx_sub = pos - k_range[valid_sub]
        v[valid_sub] += h[idx_sub]
        return v
    
    def compute_W(pos):
        w = np.zeros(2*n - 1)
        valid_add = (pos + k_range >= 0) & (pos + k_range < n) & (k_range != 0)
        idx_add = pos + k_range[valid_add]
        w[valid_add] += (1.0 - h[idx_add])
        valid_sub = (pos - k_range >= 0) & (pos - k_range < n) & (k_range != 0)
        idx_sub = pos - k_range[valid_sub]
        w[valid_sub] += -h[idx_sub]
        return w
    
    improved = True
    stall = 0
    while improved and time.time() < deadline:
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        n_ones, n_zeros = len(ones), len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break
        
        V = np.array([compute_V(i) for i in ones])

            base = corr + V[ii]
            candidates = np.max(base[None, :] + W, axis=1)
            j_best = np.argmin(candidates)
            gain = max_corr - candidates[j_best]

def gradient_of_corr_max(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-12:
            continue
        k = k_idx - (n - 1)
        wk = w[k_idx]
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            # d/dh[i] = (1-h[i-k]) for k<=i<=n-1, minus h[i+k] for 0<=i<=n-1-k
            for i in range(k, n):
                grad[i] += wk * (1.0 - h[i - k])
            for i in range(0, n - k):
                grad[i] -= wk * h[i + k]
        else:
            p = -k
            # d/dh[i] = (1-h[i-p]) for p<=i<=n-1, minus h[i+p] for 0<=i<=n-1-p
            for i in range(p, n):
                grad[i] += wk * (1.0 - h[i - p])
            for i in range(0, n - p):
                grad[i] -= wk * h[i + p]
    
    return grad * (2.0 / n)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_sum(h, target):
    lo, hi = float(h.min()) - 1.0, float(h.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_smooth_max(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= np.sum(w)
    grad = np.zeros(n)
    w0 = w[n-1]
    if w0 > 1e-12:
        grad += w0 * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk_pos = w[n-1+k]
        if wk_pos > 1e-12:
            grad[k:] += wk_pos * (1.0 - h[:n-k])
            grad[:n-k] -= wk_pos * h[k:]
        wk_neg = w[n-1-k]
        if wk_neg > 1e-12:
            grad[k:] += wk_neg * (1.0 - h[:n-k])
            grad[:n-k] -= wk_neg * h[k:]
    return grad * (2.0 / n)

def binary_swap_search(h, deadline, max_iters=20000):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    c5 = max_corr * dx
    k_range = np.arange(-(n-1), n)
    
    def compute_V(pos):
        v = np.zeros(2*n - 1)
        va = (pos + k_range >= 0) & (pos + k_range < n) & (k_range != 0)
        v[va] += -(1.0 - h[pos + k_range[va]])
        vs = (pos - k_range >= 0) & (pos - k_range < n) & (k_range != 0)
        v[vs] += h[pos - k_range[vs]]
        return v
    
    def compute_W(pos):
        w = np.zeros(2*n - 1)
        va = (pos + k_range >= 0) & (pos + k_range < n) & (k_range != 0)
        w[va] += (1.0 - h[pos + k_range[va]])
        vs = (pos - k_range >= 0) & (pos - k_range < n) & (k_range != 0)
        w[vs] += -h[pos - k_range[vs]]
        return w
    
    iters = 0
    stall = 0
    while iters < max_iters and time.time() < deadline:
        iters += 1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        V = np.array([compute_V(i) for i in ones])
        W = np.array([compute_W(j) for j in zeros])
        best_gain = 1e-9
        best_i, best_j = -1, -1
        for ii in range(len(ones)):
            base = corr + V[ii]
            cands = np.max(base[None, :] + W, axis=1)
            jb = np.argmin(cands)
            gain = max_corr - cands[jb]
            if gain > best_gain:
                best_gain = gain
                best_i, best_j = ii, jb
        if best_i < 0:
            stall += 1
            if stall > 3:
                break
            continue
        stall = 0
        i_pos, j_pos = ones[best_i], zeros[best_j]
        h[i_pos] = 0.0; h[j_pos] = 1.0
        corr = corr + V[best_i] + W[best_j]
        max_corr = np.max(corr)
        c5 = max_corr * dx
    return h, c5

def grad_desc(h0, target, deadline, rng, alpha=100):
    n = len(h0)
    h = project_sum(h0.astype(float), target)
    c5 = compute_c5(h)
    lr = 0.01
    v = np.zeros(n)
    mom = 0.85
    while time.time() < deadline:
        g = grad_smooth_max(h, alpha)
        v = mom * v - lr * g
        h_new = project_sum(h + v, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h, c5 = h_new, c5_new
        else:
            v *= 0.5
            lr *= 0.95
            if lr < 1e-6:
                break
    return h, c5

def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_ext = np.concatenate([h_old, [h_old[0]]])
    x_ext = np.concatenate([x_old, [2.0 + 1.0/n_old]])
    return np.interp(x_new, x_ext, h_ext)

def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    while h_bin.sum() > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[rng.randint(len(idx))]] = 0
    while h_bin.sum() < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[rng.randint(len(idx))]] = 1
    return h_bin
