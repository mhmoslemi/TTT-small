    grad *= dx
    s = float(np.sum(x))
    grad += 400.0 * (s - ts)
    return grad

def _exact_f(x):
    return compute_c5(x, _dx)

def _eq_c(x):
    return np.sum(x) - _ts

def _grad_lag(h, L):
    n = len(h)
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[:n - L] += 1.0 - h[L:]
        g[L:] -= h[:n - L]
    else:
        L2 = -L
        g[L2:] += 1.0 - h[:n - L2]
        g[:n - L2] -= h[L2:]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    global _dx, _ts, _n, _al
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except NameError:
        h_init = None
        n_init = 70

    n_list = [n_init, 60, 80, 100, 90, 70]
    seen = set()
    n_list_f = []
    for nn in n_list:
        if nn not in seen and nn % 2 == 0:
            seen.add(nn)
            n_list_f.append(nn)

    best_h = None
    best_c5 = np.inf
    best_n = 70

    for n in n_list_f:
        if time.time() - t0 > budget_s - 50:
            break
        dx = 2.0 / n
        ts = n / 2.0
        _dx, _ts, _n = dx, ts, n
        nh = n - 1

        if h_init is not None and n == n_init:
            h = project_feasible(h_init, ts)
        else:
            xi = np.arange(n)
            h = 0.5 + 0.25 * np.sin(2.0 * np.pi * 4.0 * xi / n) + rng.normal(0, 0.08, n)
            h = project_feasible(h, ts)

        bc5 = compute_c5(h, dx)
        bh = h.copy()

        # Phase 1: L-BFGS-B with increasing alpha
        for al in [10.0, 25.0, 50.0, 100.0, 200.0]:
            if time.time() - t0 > budget_s - 30:
                break
            _al = al
            try:
                res = minimize(_smooth_f, bh, jac=_smooth_g, method='L-BFGS-B',
                              bounds=[(0.0, 1.0)] * n,
                              options={'maxiter': 2000, 'ftol': 1e-16})
                if res.x is not None:
                    h_ref = project_feasible(res.x, ts)
                    c5_ref = compute_c5(h_ref, dx)
                    if c5_ref < bc5:
                        bc5 = c5_ref
                        bh = h_ref
            except Exception:
                pass

        # Phase 2: Subgradient with adaptive active set
        if time.time() - t0 < budget_s - 25:
            h = bh.copy()
            ss = 0.004
            mom = np.zeros(n)
            stag = 0
            while time.time() - t0 < budget_s - 10:
                corr = np.correlate(h, 1.0 - h, mode='full')
                cmax = float(np.max(corr))
                tol = max(0.03, cmax * 0.003)
                active = np.where(corr >= cmax - tol)[0]
                if len(active) < 3:
                    active = np.argsort(corr)[-10:]
                elif len(active) > 12:
                    active = np.argsort(corr)[-12:]

                grad = np.zeros(n)
                for idx in active:
                    grad += _grad_lag(h, int(idx) - nh)
                grad /= float(len(active))

                mom = 0.6 * mom - ss * grad
                h_new = project_feasible(h + mom, ts)
                c5_new = compute_c5(h_new, dx)

                if c5_new < bc5:
                    bc5 = c5_new
                    bh = h_new.copy()
                    stag = 0
                else:
                    stag += 1
                    if stag % 50 == 0:
                        ss *= 0.82
                    if ss < 1e-8:
                        h = bh + rng.normal(0, 0.04, n)
                        h = project_feasible(h, ts)
                        ss = 0.003
                        mom = np.zeros(n)
                        stag = 0
                h = h_new

        # Phase 3: SLSQP polish
        if time.time() - t0 < budget_s - 12:
            try:
                res = minimize(_exact_f, bh, method='SLSQP',
                              bounds=[(0.0, 1.0)] * n,
                              constraints=[{'type': 'eq', 'fun': _eq_c}],
                              options={'maxiter': 8000, 'ftol': 1e-18})
                if res.x is not None:
                    h_ref = project_feasible(res.x, ts)
                    c5_ref = compute_c5(h_ref, dx)
                    if c5_ref < bc5:
                        bc5 = c5_ref
                        bh = h_ref
            except Exception:
                pass

        # Phase 4: Perturbed SLSQP
        if time.time() - t0 < budget_s - 10:
            h_p = bh + rng.normal(0, 0.008, n)
            h_p = project_feasible(h_p, ts)
            try:
                res = minimize(_exact_f, h_p, method='SLSQP',
                              bounds=[(0.0, 1.0)] * n,
                              constraints=[{'type': 'eq', 'fun': _eq_c}],
                              options={'maxiter': 5000, 'ftol': 1e-18})
                if res.x is not None:
                    h_ref = project_feasible(res.x, ts)
                    c5_ref = compute_c5(h_ref, dx)
                    if c5_ref < bc5:
                        bc5 = c5_ref
                        bh = h_ref
            except Exception:
                pass

        # Phase 5: Fine subgradient
        if time.time() - t0 < budget_s - 8:
            h = bh.copy()
            ss = 0.0005
            mom = np.zeros(n)
            stag = 0
            while time.time() - t0 < budget_s - 3:
                corr = np.correlate(h, 1.0 - h, mode='full')
                cmax = float(np.max(corr))
                tol = 1e-4
                active = np.where(corr >= cmax - tol)[0]
                if len(active) < 1:
                    active = np.array([int(np.argmax(corr))])
                elif len(active) > 6:
                    active = np.argsort(corr)[-6:]

                grad = np.zeros(n)
                for idx in active:
                    grad += _grad_lag(h, int(idx) - nh)
                grad /= float(len(active))

                mom = 0.4 * mom - ss * grad
                h_new = project_feasible(h + mom, ts)
                c5_new = compute_c5(h_new, dx)

                if c5_new < bc5:
                    bc5 = c5_new
                    bh = h_new.copy()
                    stag = 0
                else:
                    stag += 1
                    ss *= 0.93
                    if ss < 1e-12:
                        break
                h = h_new

        if bc5 < best_c5:
            best_c5 = bc5
            best_h = bh.copy()
            best_n = n

    if best_h is None:
        best_h = np.full(70, 0.5)
        best_c5 = compute_c5(best_h, 2.0/70)
        best_n = 70

    return (best_h, best_c5, best_n)
