```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, t, dx):
    max_c = np.max(corr)
    if t < 1e10:
        shifted = t * (corr - max_c)
        val = max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))
    else:
        val = max_c
    return val * dx

def smooth_max_obj(h, t, dx, penalty_weight, target_sum):
    corr = np.correlate(h, 1 - h, mode="full")
    sm = smooth_max_val(corr, t, dx)
    pen = penalty_weight * (np.sum(h) - target_sum) ** 2
    return sm + pen

def smooth_max_grad(h, t, dx, penalty_weight, target_sum):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    w = np.exp(shifted) / np.sum(np.exp(shifted))
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        offset = k - (n - 1)
        wk = w[k]
        if offset >= 0:
            L = n - offset
            grad[:L] += wk * (1.0 - h[offset:])
            grad[offset:] -= wk * h[:L]
        else:
            d = -offset
            L = n - d
            grad[d:] += wk * (1.0 - h[:L])
            grad[:L] -= wk * h[d:]
    
    grad *= dx / t
    
    # Penalty gradient
    sum_diff = np.sum(h) - target_sum
    grad += 2.0 * penalty_weight * sum_diff
    
    return grad

def fix_sum(h, target_sum):
    h = h.copy()
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-15:
        h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def fourier_perturb(h, scale, rng):
    n = len(h)
    spec = np.fft.rfft(h)
    pert = np.zeros_like(spec)
    n_freq = len(spec)
    for k in range(1, n_freq):
        w = 1.0 / (1.0 + 0.05 * k)
        pert[k] = complex(rng.normal(0, scale * w), rng.normal(0, scale * w))
    return np.real(np.fft.irfft(spec + pert, n=n))
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, t, dx):
    max_c = np.max(corr)
    if t < 1e10:
        shifted = t * (corr - max_c)
        val = max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))
    else:
        val = max_c
    return val * dx

def fix_sum(h, target_sum):
    h = h.copy()
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-15:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def sm_objective(x, t, dx, pw, ts):
    corr = np.correlate(x, 1 - x, mode="full")
    return smooth_max_val(corr, t, dx) + pw * (np.sum(x) - ts) ** 2

def sm_gradient(x, t, dx, pw, ts):
    n = len(x)
    corr = np.correlate(x, 1 - x, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    w = np.exp(shifted) / np.sum(np.exp(shifted))
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        offset = k - (n - 1)
        wk = w[k]
        if offset >= 0:
            L = n - offset
            grad[:L] += wk * (1.0 - x[offset:])
            grad[offset:] -= wk * x[:L]
        else:
            d = -offset
            L = n - d
            grad[d:] += wk * (1.0 - x[:L])
            grad[:L] -= wk * x[d:]
    grad *= dx / t
    grad += 2.0 * pw * (np.sum(x) - ts)
    return grad

def lbfgs_step(h, t, dx, ts, pw, maxiter, ftol):
    n = len(h)
    try:
        res = minimize(
            sm_objective, h.copy(), jac=sm_gradient,
            args=(t, dx, pw, ts),
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12,
                     'maxfun': maxiter * 5, 'iprint': -1}
        )
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h.copy()

def fourier_perturb(h, scale, rng):
    n = len(h)
    spec = np.fft.rfft(h)
    pert = np.zeros_like(spec)
    nf = len(spec)
    for k in range(1, nf):
        w = 1.0 / (1.0 + 0.08 * k)
        pert[k] = complex(rng.normal(0, scale * w), rng.normal(0, scale * w))
    return np.real(np.fft.irfft(spec + pert, n=n))

def subgrad_step(h, target_sum, step_size, dx):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    k_star = np.argmax(corr)
    offset = k_star - (n - 1)
    grad = np.zeros(n)
    if offset >= 0:
        L = n - offset
        grad[:L] += (1.0 - h[offset:])
        grad[offset:] -= h[:L]
    else:
        d = -offset
        L = n - d
        grad[d:] += (1.0 - h[:L])
        grad[:L] -= h[d:]
    grad *= dx
    h_new = h - step_size * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    return fix_sum(h_new, target_sum)
```
