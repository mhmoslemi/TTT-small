def project_feasible(h, target_sum):

        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()

def smooth_max(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)

        return smooth_max(x, beta)

def subgradient_descent(h0, target_sum, n_iter=2000, lr_init=0.05, lr_decay=0.9995):
    n = len(h0)
    h = h0.copy()
    best = h0.copy()

    lr = lr_init
    
    for it in range(n_iter):
        corr = compute_corr(h)
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)

        if lag >= 0:
            if n - lag > 0:
                grad[:n-lag] += (1.0 - h[lag:])
            if lag < n:

        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
        lr *= lr_decay
    
    return best

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def gradient_descent_true(h0, target_sum, n_iter=5000, lr=0.001):
    n = len(h0)
    h = h0.copy()
    best = h0.copy()
    best_c5 = compute_c5(h0)
    
    for it in range(n_iter):
        corr = compute_corr(h)
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        
        # Subgradient
        grad = np.zeros(n)
        if lag >= 0:
            if n - lag > 0:
                grad[:n-lag] += (1.0 - h[lag:])
            if lag < n:
                grad[lag:] -= h[:n-lag]
        else:
            al = -lag
            if n - al > 0:
                grad[al:] += (1.0 - h[:n-al])
            if al < n:
                grad[:n-al] -= h[al:]
        
        # Project gradient onto sum=0 subspace
        grad = grad - np.mean(grad)
        
        # Line search
        best_lr = lr
        best_val = compute_c5(project_feasible(h - lr * grad, target_sum))
        for trial_lr in [lr*0.1, lr*0.5, lr, lr*2, lr*5]:
            h_trial = project_feasible(h - trial_lr * grad, target_sum)
            val = compute_c5(h_trial)
            if val < best_val:
                best_val = val
                best_lr = trial_lr
        
        h = project_feasible(h - best_lr * grad, target_sum)
        
        if best_val < best_c5:
            best_c5 = best_val
            best = h.copy()

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    h = h.copy()
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.any():
            h[free] += diff / free.sum()
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def get_subgradient(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        if n - lag > 0:
            grad[:n-lag] += (1.0 - h[lag:])
        if lag < n:
            grad[lag:] -= h[:n-lag]
    else:
        al = -lag
        if n - al > 0:
            grad[al:] += (1.0 - h[:n-al])
        if al < n:
            grad[:n-al] -= h[al:]
    return grad

    n = len(h)
