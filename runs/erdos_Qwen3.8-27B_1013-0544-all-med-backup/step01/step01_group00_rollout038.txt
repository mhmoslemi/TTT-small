```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def get_corrs(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_descent(h, max_iter=500, step_scale=0.1):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    
    for it in range(max_iter):
        c = get_corrs(h_cur)
        c5 = c.max()
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cur.copy()
        
        # Active set
        active = np.where(c > c5 - 0.02)[0]
        
        # Compute subgradient (average gradient of active corr functions)
        # For lag l = m - (n-1), corr = sum_i h[i]*(1-h[i+l])*dx (for l>=0)
        # d/dh[j] = dx * [(1-h[j-l]) * 1_{0<=j-l<n} - h[j+l] * 1_{0<=j+l<n}]
        
        grad = np.zeros(n)
        for m in active:
            l = m - (n - 1)
            # h[j] appears as h[i] when i=j: contributes -h[j+l]*dx if 0<=j+l<n
            # h[j] appears as h[i+l] when i+l=j, i=j-l: contributes (1-h[j-l])*dx if 0<=j-l<n
            if l >= 0:
                # j as h[i]: valid if j < n and j+l < n, i.e., j < n-l
                #   contribution: -h[j+l]*dx
                idx = np.arange(n - l)
                grad[idx] -= h_cur[idx + l] * dx
                # j as h[i+l]: valid if j-l >= 0 and j-l < n, i.e., j >= l
                #   contribution: (1-h[j-l])*dx
                idx2 = np.arange(l, n)
                grad[idx2] += (1.0 - h_cur[idx2 - l]) * dx
            else:
                l2 = -l
                # j as h[i]: valid if j < n and j-l2 < n (always true if l2 < n) AND j >= 0
                # Actually for negative lag, corr = sum_i h[i]*(1-h[i-l2])*dx where i-l2 >= 0
                # So i >= l2
                # j as h[i]: valid if j >= l2, contribution: -h[j-l2]*dx
                idx = np.arange(l2, n)
                grad[idx] -= h_cur[idx - l2] * dx
                # j as h[i-l2]: valid if i-l2 = j, i = j+l2, valid if j+l2 < n
                #   contribution: (1-h[j+l2])*dx
                idx2 = np.arange(n - l2)
                grad[idx2] += (1.0 - h_cur[idx2 + l2]) * dx
        
        grad /= len(active)
        
        # Adaptive step size
        step = step_scale / (it + 10)
        h_new = h_cur - step * grad
        h_new = project_feasible(h_new)
        h_cur = h_new
        
        if step < 1e-6:
            break
    
    return best_h, best_c5

def smooth_objective(h, temp):
    c = get_corrs(h)
    m = c.max()
    # Log-sum-exp for numerical stability
    shifted = (c - m) / temp
    log_sum = m + temp * np.log(np.sum(np.exp(shifted)))
    return log_sum

def smooth_grad(h, temp):
    c = get_corrs(h)
    # Weight for each k (softmax-like)
    shifted = (c - c.max()) / temp
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for m_idx, w in enumerate(weights):
        if w < 1e-8:
            continue
        l = m_idx - (n - 1)
        if l >= 0:
            idx = np.arange(n - l)
            grad[idx] -= w * h[idx + l] * dx
            idx2 = np.arange(l, n)
            grad[idx2] += w * (1.0 - h[idx2 - l]) * dx
        else:
            l2 = -l
            idx = np.arange(l2, n)
            grad[idx] -= w * h[idx - l2] * dx
            idx2 = np.arange(n - l2)
            grad[idx2] += w * (1.0 - h[idx2 + l2]) * dx
    
    return grad

def projected_smooth_descent(h, temp, max_iter=200, lr=None):
    n = len(h)
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    
    if lr is None:
        lr = 0.5
    
    for it in range(max_iter):
        # Backtracking line search
        alpha = lr
        f_cur = smooth_objective(h_cur, temp)
        g = smooth_grad(h_cur, temp)
        
        for inner in range(20):
            h_try = h_cur - alpha * g
            h_try = project_feasible(h_try)
            f_try = smooth_objective(h_try, temp)
            if f_try < f_cur - 1e-4 * alpha * np.dot(g, g):
                break
            alpha *= 0.5
        
        if h_try is not h_cur:
            h_cur = h_try
            c5 = compute_c5(h_cur)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cur.copy()
    
    return best_h, best_c5

def cvxpy_slp(h, target_t, max_iter=30, delta=0.05):
    """SLP using cvxpy LP at each iteration"""
    import cvxpy as cp
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    
    for it in range(max_iter):
        c = get_corrs(h_cur)
        c5 = c.max()
        
        if c5 <= target_t:
            break
        
        # Active set
        active = np.where(c > target_t)[0]
        if len(active) == 0:
            active = np.where(c > c5 - delta)[0]
        
        # Build LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n * 0.5]
        
        # Linearize each active constraint
        for m_idx in active:
            l = m_idx - (n - 1)
            # corr_l(h) at h_cur is c[m_idx]
            # Gradient: for j, d(corr_l)/dh[j]
            # Linearized: c[m_idx] + sum_j grad[j]*(h_var[j] - h_cur[j]) <= t_var
            # => sum_j grad[j]*h_var[j] <= t_var - c[m_idx] + sum_j grad[j]*h_cur[j]
            # => sum_j grad[j]*h_var[j] <= t_var  (since corr is "homogeneous" in a sense)
            
            # Actually let me just compute: linearized_corr = c[m_idx] + grad · (h_var - h_cur)
            # = c[m_idx] + grad · h_var - grad · h_cur
            # Constraint: linearized_corr <= t_var
            # => grad · h_var <= t_var - c[m_idx] + grad · h_cur
            
            # Compute grad for this lag
            g = np.zeros(n)
            if l >= 0:
                idx = np.arange(n - l)
                g[idx] -= h_cur[idx + l] * dx
                idx2 = np.arange(l, n)
                g[idx2] += (1.0 - h_cur[idx2 - l]) * dx
            else:
                l2 = -l
                idx = np.arange(l2, n)
                g[idx] -= h_cur[idx - l2] * dx
                idx2 = np.arange(n - l2)
                g[idx2] += (1.0 - h_cur[idx2 + l2]) * dx
            
            rhs_offset = -c[m_idx] + np.dot(g, h_cur)
            constraints.append(cp.sum(g * h_var) <= t_var + rhs_offset)
        
        objective = cp.Minimize(t_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if h_var.value is not None:
                h_new = np.clip(h_var.value, 0.0, 1.0)
                h_new = project_feasible(h_new)
                h_cur = h_new
        except:
            break
    
    return h_cur

def interp_up(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new)
```

where "valid" means 0 ≤ index < n AND the index i = j (for first) or i = j+l (for second) is in the valid range of the sum.

For l > 0: the sum is over i from l to n-1 (since i-l ≥ 0 requires i ≥ l, and i < n).
- h[j] as h[i] with i=j: valid if l ≤ j < n. Contribution: (1-h[j-l])
- h[j] as h[i-l] with i=j+l: valid if l ≤ j+l < n, i.e., 0 ≤ j < n-l. Contribution: -h[j+l]

For l < 0 (let k = -l > 0): the sum is over i from 0 to n-k-1 (since i+k < n requires i < n-k).
c = sum_{i=0}^{n-k-1} h[i] * (1-h[i+k])
- h[j] as h[i] with i=j: valid if 0 ≤ j < n-k. Contribution: (1-h[j+k])
- h[j] as h[i+k] with i=j-k: valid if 0 ≤ j-k < n-k, i.e., k ≤ j < n. Contribution: -h[j-k]

```python
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def get_corrs(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_lag(h, l):
    """Gradient of corr at lag l with respect to h, times dx."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if l == 0:
        g = (1.0 - 2.0 * h) * dx
    elif l > 0:
        # contribution from h[i] term: (1-h[i-l]) for i from l to n-1
        g[l:] += (1.0 - h[:-l]) * dx
        # contribution from h[i-l] term: -h[i] for i from 0 to n-1-l
        g[:n-l] -= h[l:] * dx
    else:
        k = -l
        # contribution from h[i] term: (1-h[i+k]) for i from 0 to n-1-k
        g[:-k] += (1.0 - h[k:]) * dx
        # contribution from h[i+k] term: -h[i] for i from k to n-1
        g[k:] -= h[:-k] * dx
    return g

def smooth_max_obj(h, temp):
    c = get_corrs(h)
    m = c.max()
    # Log-sum-exp
    shifted = (c - m) / temp
    return m + temp * np.log(np.sum(np.exp(shifted)))

def smooth_max_grad(h, temp):
    c = get_corrs(h)
    m = c.max()
    shifted = (c - m) / temp
    w = np.exp(shifted)
    w /= w.sum()
    n = len(h)
    grad = np.zeros(n)
    # Only process significant weights for speed
    for idx in np.where(w > 1e-6)[0]:
        l = idx - (n - 1)
        grad += w[idx] * compute_grad_lag(h, l)
    return grad

def proj_smooth_descent(h, temp, max_iter=200, init_lr=1.0):
    """Projected gradient descent with backtracking on smooth max."""
    n = len(h)
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    lr = init_lr
    
    for it in range(max_iter):
        g = smooth_max_grad(h_cur, temp)
        f_cur = smooth_max_obj(h_cur, temp)
        
        # Backtracking line search
        alpha = lr
        improved = False
        for _ in range(30):
            h_try = project_feasible(h_cur - alpha * g)
            f_try = smooth_max_obj(h_try, temp)
            if f_try < f_cur - 1e-5 * alpha * np.dot(g, h_try - h_cur):
                h_cur = h_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            # Try smaller
            alpha = max(1e-8, alpha * 0.1)
            h_cur = project_feasible(h_cur - alpha * g)
        
        lr = alpha
        c5 = compute_c5(h_cur)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cur.copy()
    
    return best_h, best_c5

def t_var_slsqp(h, max_iter=400):
    """SLSQP with t variable."""
    n = len(h)
    c5_cur = compute_c5(h)
    v0 = np.append(h.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    def obj(v):
        return v[-1]
    
    def ineq(v):
        hh = v[:n]
        t = v[-1]
        return get_corrs(hh) - t  # want all <= 0, so ineq = t - corr >= 0
    
    def eq_fun(v):
        return np.sum(v[:n]) - n * 0.5
    
    def eq_jac(v):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    cons = [{'type': 'ineq', 'fun': lambda v: v[-1] - get_corrs(v[:n])},
            {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
    h_new = project_feasible(res.x[:n])
    return h_new, compute_c5(h_new)

def cvxpy_slp(h, target, max_iter=20, n_active=30):
    """SLP: linearize most-violated constraints, solve LP."""
    import cvxpy as cp
    n = len(h)
    h_cur = h.copy()
    
    for it in range(max_iter):
        c = get_corrs(h_cur)
        c5 = c.max()
        if c5 <= target:
            break
        
        # Get most violated constraints
        order = np.argsort(c)[::-1]
        active = order[:n_active]
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n * 0.5]
        
        for m_idx in active:
            l = m_idx - (n - 1)
            g = compute_grad_lag(h_cur, l)
            c_val = c[m_idx]
            # Linearized: c_val + g · (h_var - h_cur) <= t_var
            # g · h_var <= t_var - c_val + g · h_cur
            rhs = -c_val + np.dot(g, h_cur)
            constraints.append(cp.sum(g * h_var) <= t_var + rhs)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if h_var.value is not None:
                h_new = project_feasible(h_var.value)
                h_cur = h_new
        except:
            break
    
    return h_cur

def interp_up(h, n_new):
    n_old = len(h)
    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, np.linspace(0, 1, n_old), h)
    return project_feasible(h_new)
```

```python
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_lag(h, l):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if l == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    elif l > 0:
        g[l:] += (1.0 - h[:-l]) * dx
        g[:n - l] -= h[l:] * dx
    else:
        k = -l
        g[:n - k] += (1.0 - h[k:]) * dx
        g[k:] -= h[:-k] * dx
    return g

def smooth_max_grad(h, temp):
    c = get_corrs(h)
    m = c.max()
    shifted = (c - m) / temp
    w = np.exp(shifted)
    w /= w.sum()
    n = len(h)
    grad = np.zeros(n)
    for idx in np.where(w > 1e-7)[0]:
        lag = idx - (n - 1)
        grad += w[idx] * compute_grad_lag(h, lag)
    return grad

def smooth_max_val(h, temp):
    c = get_corrs(h)
    m = c.max()
    return m + temp * np.log(np.sum(np.exp((c - m) / temp)))

def proj_smooth_descent(h, temp, max_iter=200, init_lr=0.5):
    n = len(h)
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    lr = init_lr
    for it in range(max_iter):
        g = smooth_max_grad(h_cur, temp)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-10:
            break
        f_cur = smooth_max_val(h_cur, temp)
        alpha = lr
        improved = False
        for _ in range(25):
            h_try = project_feasible(h_cur - alpha * g)
            f_try = smooth_max_val(h_try, temp)
            if f_try < f_cur:
                h_cur = h_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            break
        lr = alpha
        c5 = compute_c5(h_cur)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cur.copy()
    return best_h, best_c5

def slp_obj(v):
    return v[-1]

def slp_ineq(v):
    n = len(v) - 1
    return v[-1] - get_corrs(v[:n])

def slp_eq(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5

def slp_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j

def t_var_slsqp(h, max_iter=400, ftol=1e-12):
    n = len(h)
    c5_cur = compute_c5(h)
    v0 = np.append(h.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': slp_ineq},
        {'type': 'eq', 'fun': slp_eq, 'jac': slp_eq_jac}
    ]
    res = minimize(slp_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = project_feasible(res.x[:n])
    return h_new, compute_c5(h_new)

def cvxpy_slp(h, target, max_iter=20, n_active=25):
    import cvxpy as cp
    n = len(h)
    h_cur = h.copy()
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    base_cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n * 0.5]
    for it in range(max_iter):
        c = get_corrs(h_cur)
        if c.max() <= target:
            break
        order = np.argsort(c)[::-1]
        active = order[:n_active]
        constraints = list(base_cons)
        for m_idx in active:
            lag = m_idx - (n - 1)
            g = compute_grad_lag(h_cur, lag)
            c_val = c[m_idx]
            rhs = -c_val + np.dot(g, h_cur)
            constraints.append(cp.sum(g * h_var) <= t_var + rhs)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                break
        if h_var.value is not None:
            h_cur = project_feasible(h_var.value)
        else:
            break
    return h_cur

def interp_up(h, n_new):
    n_old = len(h)
    h_new = np.interp(np.arange(n_new) / n_new, np.arange(n_old) / n_old, h)
    return project_feasible(h_new)

def balanced_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.full(n, 0.5)
    h += rng.randn(n) * 0.1
    return project_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float).copy()
    h = project_feasible(h)
    n = len(h)
    c5 = compute_c5(h)
    h_best = h.copy()
    c5_best = c5

    # Phase 1: SLSQP t-var
    if time_left() > 250:
        h_new, c5_new = t_var_slsqp(h, max_iter=500, ftol=1e-12)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 2: Smooth projected gradient descent (multiple temps)
    for temp in [0.05, 0.02, 0.01, 0.005, 0.002]:
        if time_left() < 100:
            break
        h_new, c5_new = proj_smooth_descent(h_best, temp, max_iter=150, init_lr=0.3)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 3: SLP with cvxpy
    if time_left() > 150:
        h_new = cvxpy_slp(h_best, target=c5_best - 0.005, max_iter=25, n_active=20)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 4: SLSQP again after refinement
    if time_left() > 200:
        h_new, c5_new = t_var_slsqp(h_best, max_iter=500, ftol=1e-13)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 5: Smooth descent at very low temp
    for temp in [0.001, 0.0005]:
        if time_left() < 80:
            break
        h_new, c5_new = proj_smooth_descent(h_best, temp, max_iter=200, init_lr=0.1)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 6: Multi-start with perturbations
    for attempt in range(8):
        if time_left() < 80:
            break
        scale = 0.015 + 0.01 * attempt
        h_pert = project_feasible(h_best + rng.randn(n) * scale)
        h_new, c5_new = proj_smooth_descent(h_pert, temp=0.005, max_iter=100, init_lr=0.3)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
        if time_left() > 60:
            h_new2, c5_new2 = t_var_slsqp(h_new, max_iter=200, ftol=1e-11)
            if c5_new2 < c5_best:
                h_best = h_new2.copy()
                c5_best = c5_new2

    # Phase 7: Resolution increase
    for n_new in [270, 360]:
        if time_left() < 200:
            break
        h_fine = interp_up(h_best, n_new)
        c5_fine = compute_c5(h_fine)
        if time_left() > 150:
            h_new, c5_new = t_var_slsqp(h_fine, max_iter=400, ftol=1e-12)
            if c5_new < c5_fine:
                h_fine = h_new.copy()
                c5_fine = c5_new
        for temp in [0.01, 0.005, 0.002]:
            if time_left() < 80:
                break
            h_new, c5_new = proj_smooth_descent(h_fine, temp, max_iter=100, init_lr=0.2)
            if c5_new < c5_fine:
                h_fine = h_new.copy()
                c5_fine = c5_new
        if time_left() > 100:
            h_new, c5_new = t_var_slsqp(h_fine, max_iter=300, ftol=1e-12)
            if c5_new < c5_fine:
                h_fine = h_new.copy()
                c5_fine = c5_new
        if c5_fine < c5_best:
            h_best = h_fine.copy()
            c5_best = c5_fine

    # Final SLP polish
    if time_left() > 60:
        h_new = cvxpy_slp(h_best, target=c5_best - 0.003, max_iter=15, n_active=20)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    h_best = project_feasible(h_best)
    return h_best, compute_c5(h_best), len(h_best)
```
