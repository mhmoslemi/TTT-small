            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                best_h = h_new
                best_c5 = c5_new
            h_cur = h_new

    # Phase 3: Subgradient exploration with periodic local polish (20% time)
    p3_end = budget_s * 0.85
    h_sub = best_h.copy()
    step = 0.04
    stalled = 0
    for it in range(2000):
        if elapsed() > p3_end:
            break
        h_sub = subgrad_step(h_sub, dx, step, tol=0.003)
        c5_sub = compute_c5(h_sub, dx)
        if c5_sub < best_c5 - 1e-14:
            best_h = h_sub
            best_c5 = c5_sub
            stalled = 0
        else:
            stalled += 1
        # Periodic local polish to escape flat regions
        if (it + 1) % 80 == 0:
            h_local = slsqp_smooth(h_sub, n, dx, 80.0, max_iter=80)
            c5_local = compute_c5(h_local, dx)
            if c5_local < c5_sub - 1e-14:
                h_sub = h_local
            if c5_local < best_c5 - 1e-14:
                best_h = h_sub.copy()
                best_c5 = c5_local
            stalled = 0
        # Adapt step size
        if stalled > 100:
            step = max(0.005, step * 0.6)
            stalled = 0
        elif it % 200 == 199 and step > 0.005:
            step *= 0.85

    # Phase 4: Final SLSQP polish on true objective (10% time)
    if elapsed() < budget_s - 20:
        for it in range(15):
            if elapsed() > budget_s - 5:
                break
            h_new = slsqp_orig(best_h, n, dx, max_iter=400)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                best_h = h_new
                best_c5 = c5_new
            # Try small perturbation + smooth then original
            if elapsed() < budget_s - 10:
                h_pert = project_feasible(best_h + rng.randn(n) * 0.015, n)
                h_sm = slsqp_smooth(h_pert, n, dx, 200.0, max_iter=100)
                c5_sm = compute_c5(h_sm, dx)
                if c5_sm < c5_new - 1e-14:
                    h_new = slsqp_orig(h_sm, n, dx, max_iter=200)
                    c5_new = compute_c5(h_new, dx)
                    if c5_new < best_c5 - 1e-14:
                        best_h = h_new
                        best_c5 = c5_new

    return (best_h, best_c5, n)
