    """Return subgradient of max_k c(k) w.r.t. h, and the max value."""

                          options={'maxiter': 5000, 'ftol': 1e-18})
            h_try = project_feasible(res.x, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c5:
                best_h = h_try
                best_c5 = c_try
        except Exception:
            pass

    # Phase 6: Final pair swaps
    if remaining() > 5:
        for _ in range(5000):
            if remaining() < 5:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d_opt, c_opt = golden_pair_swap(best_h, n, i, j, n_gauss=10)
            if c_opt < best_c5 - 1e-13:
                best_h[i] += d_opt
                best_h[j] -= d_opt
                best_c5 = c_opt

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
