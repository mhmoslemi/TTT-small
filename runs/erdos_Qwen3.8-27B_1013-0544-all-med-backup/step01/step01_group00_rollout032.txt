from scipy.optimize import minimize, dual_annealing
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def penalized_obj_and_grad(v, mu, nu):
    """v = [h (n values), t], returns (obj, grad)"""
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Penalty for f_k > t
    excess = np.maximum(0.0, corr - t)
    pen_h = np.zeros(n)
    for k in range(len(corr)):
        if excess[k] > 0:
            m = k - (n - 1)
            # grad of corr[k] w.r.t. h[j] = (1-h[j+m])*I(j+m in [0,n)) - h[j-m]*I(j-m in [0,n))
            jm = np.arange(n) + m
            jmm = np.arange(n) - m
            grad = np.zeros(n)
            valid_plus = (jm >= 0) & (jm < n)
            valid_minus = (jmm >= 0) & (jmm < n)
            grad[valid_plus] += 1.0 - h[jm[valid_plus]]
            grad[valid_minus] -= h[jmm[valid_minus]]
            pen_h += excess[k] * grad
    
    obj = t + 0.5 * mu * np.sum(excess ** 2) + 0.5 * nu * (np.sum(h) - n * 0.5) ** 2
    
    grad_h = 0.5 * 2 * mu * np.sum(excess * 1) * 0  # will compute properly
    # Actually: d/dh [0.5*mu*sum(excess^2)] = mu * sum_k excess_k * grad_k (for active k)
    # We already computed pen_h = sum_k excess_k * grad_k
    # So grad from penalty = mu * pen_h
    # grad from sum constraint = nu * (sum(h) - n/2) * ones
    
    grad_h = mu * pen_h + nu * (np.sum(h) - n * 0.5)
    grad_t = 1.0 - mu * np.sum(excess)
    
    grad = np.concatenate([grad_h, [grad_t]])
    return obj, grad

def opt_lbgfs(h0, t0, mu, nu, max_iter=200):
    n = len(h0)
    v0 = np.concatenate([h0, [t0]])
    bounds = [(0, 1)] * n + [(0, 2)]
    res = minimize(penalized_obj_and_grad, v0, args=(mu, nu),
                   method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'maxiter': max_iter, 'gtol': 1e-8})
    h_new = np.clip(res.x[:n], 0, 1)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

            h_p = make_structured_start(n_r, rng, mode=1)
        h_p = project_h(h_p, n_r * 0.5)
        h_new, c5_new = opt_smooth(h_p, [0.05, 0.02, 0.01, 0.005, 0.002],
                                   nu=1000.0, max_iter=150)
        if c5_new < c5:
            h_best = h_new
            c5 = c5_new
            if budget_s - (time.time() - t_start) > 100:
                h_new2, c5_new2 = opt_t_slsqp(h_best, max_iter=300)
                if c5_new2 < c5:
                    h_best = h_new2
                    c5 = c5_new2

    # Phase 6: Final polish at best resolution
    if budget_s - (time.time() - t_start) > 90:
        h_new, c5_new = opt_smooth(h_best, [0.002, 0.001, 0.0005, 0.0002],
                                   nu=5000.0, max_iter=200)
        if c5_new < c5:
            h_best = h_new
            c5 = c5_new

    if budget_s - (time.time() - t_start) > 60:
        h_new, c5_new = opt_t_slsqp(h_best, max_iter=500)
        if c5_new < c5:
            h_best = h_new
            c5 = c5_new

    # Final cleanup
    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
