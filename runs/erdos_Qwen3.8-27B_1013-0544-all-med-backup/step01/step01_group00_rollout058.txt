```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def lse_objective_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = (corr - m) / T
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    Z = np.sum(exp_vals)
    p = exp_vals / Z
    obj = m + T * np.log(Z)
    
    # Gradient
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)  # -(n-1), ..., n-1
    for j in range(n):
        # term1: sum_k p[k] * (1 - h[j + shifts[k]]) for valid j+shifts[k]
        idx1 = j + shifts
        valid1 = (idx1 >= 0) & (idx1 < n)
        term1 = np.sum(p[valid1] * (1.0 - h[idx1[valid1]]))
        # term2: sum_k p[k] * h[j - shifts[k]] for valid j-shifts[k]
        idx2 = j - shifts
        valid2 = (idx2 >= 0) & (idx2 < n)
        term2 = np.sum(p[valid2] * h[idx2[valid2]])
        grad[j] = dx * (term1 - term2)
    
    return obj, grad

def lse_with_penalty(h, T, lam):
    n = len(h)
    obj, grad = lse_objective_and_grad(h, T)
    s = np.sum(h) - n * 0.5
    obj += lam * s * s
    grad += 2 * lam * s * np.ones(n)
    return obj, grad

def lbfgs_step(h, T, lam, max_iter=500, gtol=1e-10):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        return lse_with_penalty(x, T, lam)[0]
    
    def jac(x):
        return lse_with_penalty(x, T, lam)[1]
    
    res = minimize(fun, h.copy(), jac=jac, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': max_iter, 'gtol': gtol,
                                           'ftol': 1e-14, 'iprint': -1})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)
```

```python
            if c5_new < c5:
                h_best = h2_new.copy()
                c5 = c5_new
                h2 = h2_new
        # Sizzle at high resolution
        if remaining() > 25:
            h_sz = sizzle_polish(h2, 100, 0.003)
            c5_sz = compute_c5(h_sz)
            if c5_sz < c5:
                h_best = h_sz.copy()
                c5 = c5_sz

    # Phase 3: Random restarts at best resolution
    h_work = h_best.copy()
    nw = len(h_work)
    for i in range(10):
        if remaining() < 45:
            break
        scale = 0.02 + 0.012 * i
        h_p = h_work + rng.randn(nw) * scale
        h_p = project_h(h_p, nw * 0.5)
        for T in [0.02, 0.005, 0.001]:
            if remaining() < 25:
                break
            tl = min(22, int(remaining()) - 8)
            if tl < 8:
                break
            h_new = lbfgs_run(h_p, T, 1000.0, 1000, tl)
            c5_new = compute_c5(h_new)
            if c5_new < c5:
                h_best = h_new.copy()
                c5 = c5_new
                h_work = h_new
                h_p = h_new

    # Phase 4: Try n=240 as intermediate (sometimes better than 360)
    if remaining() > 120:
        n3 = 240
        h3 = interp_h(h_best, n3)
        for T in [0.02, 0.005, 0.001, 0.0005]:
            if remaining() < 50:
                break
            tl = min(45, int(remaining()) - 15)
            if tl < 10:
                break
            h3_new = lbfgs_run(h3, T, 1000.0, 2000, tl)
            c5_new = compute_c5(h3_new)
            if c5_new < c5:
                h_best = h3_new.copy()
                c5 = c5_new
                h3 = h3_new

    # Phase 5: Final aggressive polish
    if remaining() > 15:
        tl = max(5, int(remaining()) - 3)
        h_new = lbfgs_run(h_best, 0.0003, 5000.0, 5000, tl)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h_best = h_new.copy()
            c5 = c5_new

    # Final sizzle
    if remaining() > 10:
        h_sz = sizzle_polish(h_best, 300, 0.003)
        c5_sz = compute_c5(h_sz)
        if c5_sz < c5:
            h_best = h_sz.copy()
            c5 = c5_sz

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_h(h_best, len(h_best) * 0.5)
    c5 = compute_c5(h_best)
    return h_best, c5, len(h_best)
```
