Hmm, I realize there's an issue with the soft-max approach. The soft-max objective is:
soft_max = max_val + (1/T) * log(sum_k exp(T * (f_k - max_val)))

This is close to max when T is large, but the gradient is:
∂soft_max/∂h_i = sum_k p_k * ∂f_k/∂h_i
where p_k = exp(T*(f_k - max_val)) / sum_j exp(T*(f_j - max_val))

So the gradient is a weighted average of the gradients of the individual f_k, weighted by their "importance" (how close they are to the max). This is actually a nice property - it gives a smooth approximation of the max that's easy to optimize.

But there's a subtlety: as T increases, the soft-max approaches the true max, and the gradient becomes more concentrated on the active lag(s). This is good for convergence but might cause issues if there are multiple nearly-active lags.

    shifted = np.clip(T * (corr - m), -700, 700)
    return m + float(np.log(np.sum(np.exp(shifted))) / T)

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:

    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_smooth(h_start, n, dx, T, max_iter=500):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target},)
    try:
        res = minimize(lambda x: smooth_c5(x, dx, T), h_start,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except:
        return project_feasible(h_start, n)
