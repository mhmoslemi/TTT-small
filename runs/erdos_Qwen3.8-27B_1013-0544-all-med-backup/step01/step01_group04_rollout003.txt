        return time.time() - start_time

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0

    best_h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: DE global exploration (30% budget)
    phase1_t = budget_s * 0.30
    pop_size = 80
    pop = [best_h.copy()]
    fit = [best_c5]
    for i in range(1, pop_size):
        if i < 6:
            h_b = (rng.rand(n) < 0.5).astype(float)
            ones = int(np.sum(h_b))
            if ones > int(n / 2):
                idx1 = np.where(h_b == 1)[0]
                rm = rng.choice(idx1, ones - int(n / 2), replace=False)
                h_b[rm] = 0.0
            elif ones < int(n / 2):
                idx0 = np.where(h_b == 0)[0]
                add = rng.choice(idx0, int(n / 2) - ones, replace=False)
                h_b[add] = 1.0
            pop.append(project_feasible(h_b, n))
        else:
            h_r = project_feasible(rng.rand(n), n)
            pop.append(h_r)
    fit = [compute_c5(p, dx) for p in pop]

    gen = 0
    f_mut = 0.85
    cr = 0.8
    while elapsed() < phase1_t:
        pop, fit = de_generation(pop, fit, n, dx, rng, f_mut, cr)
        bi = int(np.argmin(fit))
        if fit[bi] < best_c5 - 1e-14:
            best_c5 = fit[bi]
            best_h = pop[bi].copy()
        gen += 1
        f_mut = max(0.3, 0.85 * (1.0 - gen / 300.0))
        cr = max(0.4, 0.8 - 0.1 * gen / 200.0)

    # Phase 2: LSE continuation + SLSQP (45% budget)
    phase2_t = budget_s * 0.75
    s_schedule = [2, 5, 10, 25, 50, 100, 200, 500, 1000, 3000]
    h_work = best_h.copy()
    h_work_c5 = best_c5

    for s in s_schedule:
        if elapsed() > phase2_t:
            break
        n_restarts = 3 if s < 100 else 2
        for r in range(n_restarts):
            if elapsed() > phase2_t:
                break
            if r == 0:
                h_try = h_work.copy()
            elif r == 1:
                h_try = project_feasible(h_work + rng.randn(n) * 0.04, n)
            else:
                h_try = project_feasible(h_work + rng.randn(n) * 0.08, n)
            h_new = slsqp_lse(h_try, n, dx, s, target_sum, max_iter=600)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_work = h_new.copy()
                h_work_c5 = c5_new
            elif c5_new < h_work_c5:
                h_work = h_new.copy()
                h_work_c5 = c5_new

    # Phase 3: Upsample and refine (18% budget)
    if elapsed() < budget_s * 0.92:
        n2 = 2 * n
        dx2 = 2.0 / n2
        target2 = n2 / 2.0
        h_up = project_feasible(np.repeat(best_h, 2), n2)
        s_up = [5, 20, 100, 500, 2000]
        for s in s_up:
            if elapsed() > budget_s * 0.92:
                break
            h_new = slsqp_lse(h_up, n2, dx2, s, target2, max_iter=400)
            c5_new = compute_c5(h_new, dx2)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_up = h_new.copy()

    # Phase 4: Final polish with max objective (last 8%)
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    tgt_f = n_f / 2.0
    while elapsed() < budget_s - 10:
        h_try = project_feasible(best_h + rng.randn(n_f) * 0.01, n_f)
        h_new = slsqp_max(h_try, n_f, dx_f, tgt_f, max_iter=600, ftol=1e-15)
        c5_new = compute_c5(h_new, dx_f)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
