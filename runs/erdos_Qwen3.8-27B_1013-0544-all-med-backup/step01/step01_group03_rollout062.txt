def project_onto_simplex(v, target):
    # Project v onto {h: 0<=h<=1, sum(h)=target}
    lo, hi = v.min() - target, v.max()
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(v - mid, 0, 1)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, 0, 1)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def project_onto_set(v, target, n_iter=60):
    lo = v.min() - target
    hi = v.max()
    for _ in range(n_iter):
        mid = (lo + hi) * 0.5
        h_proj = np.clip(v - mid, 0.0, 1.0)
        if h_proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    # smooth max
    x = tau * corr
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    weights = exp_x / np.sum(exp_x)

import numpy as np
import time
from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_set(v, target):
    lo = float(v.min()) - float(target)
    hi = float(v.max())
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(v - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad_and_val(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = tau * corr
    x_max = float(np.max(x))
    exp_x = np.exp(x - x_max)
    w_sum = float(np.sum(exp_x))
    weights = exp_x / w_sum
    val = (x_max + np.log(w_sum)) / tau
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-10:
            continue
        p = m - (n - 1)
        if p >= 0:
            grad[:n-p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n-p]
        else:
            q = -p
            grad[q:] += w * (1.0 - h[:n-q])
            grad[:n-q] -= w * h[q:]
    grad *= dx
    return grad, val

def pgd_phase(h, tau_start, tau_end, n_iter, momentum, lr, t0, deadline):
    n = h.shape[0]
    target = n / 2.0
    h = h.copy()
    mom = np.zeros(n)
    best = h.copy()
    best_c5 = evaluate_c5(h)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        frac = (it + 1) / n_iter
        tau = tau_start * (tau_end / tau_start) ** frac
        grad, _ = compute_grad_and_val(h, tau)
        mom = momentum * mom + (1.0 - momentum) * grad
        h = project_onto_set(h - lr * mom, target)
        if (it + 1) % 3000 == 0:
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
    
    c5 = evaluate_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best = h.copy()
    return best, best_c5
