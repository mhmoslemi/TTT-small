```python
def _lbgfs_jac(x, tau, target, penalty):
    n_l = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    x_tau = tau * corr
    x_max = np.max(x_tau)
    exp_x = np.exp(x_tau - x_max)
    weights = exp_x / np.sum(exp_x)
    
    g = np.zeros(n_l)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        lag = k_idx - (n_l - 1)
        # First term: (1-h[j-lag]) for j in [max(0,lag), min(n, n+lag))
        lo1 = max(0, lag)
        hi1 = min(n_l, n_l + lag)
        if hi1 > lo1:
            g[lo1:hi1] += w * (1.0 - x[lo1-lag:hi1-lag])
        # Second term: -h[j+lag] for j in [max(0,-lag), min(n, n-lag))
        lo2 = max(0, -lag)
        hi2 = min(n_l, n_l - lag)
        if hi2 > lo2:
            g[lo2:hi2] -= w * x[lo2+lag:hi2+lag]
    
    g += 2.0 * penalty * (np.sum(x) - target)
    return g
```

```python
import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _smooth_max(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def _project_feasible(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def _lbgfs_fun(x, tau, target, penalty):
    corr = np.correlate(x, 1.0 - x, mode="full")
    x_tau = tau * corr
    x_max = np.max(x_tau)
    exp_x = np.exp(x_tau - x_max)
    f = x_max / tau + np.log(np.sum(exp_x)) / tau
    s = np.sum(x) - target
    return f + penalty * s * s

def _lbgfs_jac(x, tau, target, penalty):
    n_l = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    x_tau = tau * corr
    x_max = np.max(x_tau)
    exp_x = np.exp(x_tau - x_max)
    weights = exp_x / np.sum(exp_x)
    g = np.zeros(n_l)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        lag = k_idx - (n_l - 1)
        lo1 = max(0, lag)
        hi1 = min(n_l, n_l + lag)
        if hi1 > lo1:
            g[lo1:hi1] += w * (1.0 - x[lo1 - lag:hi1 - lag])
        lo2 = max(0, -lag)
        hi2 = min(n_l, n_l - lag)
        if hi2 > lo2:
            g[lo2:hi2] -= w * x[lo2 + lag:hi2 + lag]
    g += 2.0 * penalty * (np.sum(x) - target)
    return g

def _lbgfs_opt(h_start, tau, maxiter, penalty):
    n_pts = len(h_start)
    target = n_pts / 2.0
    bnds = [(0.0, 1.0)] * n_pts
    try:
        res = minimize(_lbgfs_fun, h_start, method='L-BFGS-B',
                      jac=_lbgfs_jac, bounds=bnds, args=(tau, target, penalty),
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        x = _project_feasible(res.x, target)
        return x, evaluate_c5(x)
    except Exception:
        return h_start, None

def _slsqp_opt(h_start, tau, maxiter):
    n_pts = len(h_start)
    target = n_pts / 2.0
    bnds = [(0.0, 1.0)] * n_pts
    cons = [{'type': 'eq', 'fun': _slsqp_sum_fun, 'jac': _slsqp_sum_jac, 'args': (target,)}]
    try:
        res = minimize(_slsqp_obj, h_start, method='SLSQP', bounds=bnds,
                      constraints=cons, args=(tau, target),
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        x = _project_feasible(res.x, target)
        return x, evaluate_c5(x)
    except Exception:
        return h_start, None

def _slsqp_obj(x, tau, target):
    corr = np.correlate(x, 1.0 - x, mode="full")
    return _smooth_max(corr, tau)

def _slsqp_sum_fun(x, target):
    return np.sum(x) - target

def _slsqp_sum_jac(x, target):
    return np.ones(len(x))

def _binary_sa(h_start, target, n_iters, t0, budget):
    n = len(h_start)
    h = (h_start > 0.5).astype(float)
    h = _project_feasible(h, target)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    T = 0.05
    for it in range(n_iters):
        if time.time() - t0 > budget - 15:
            break
        T *= 0.999
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        if h[i] == 0.0 and h[j] == 1.0:
            h[i], h[j] = 1.0, 0.0
            c5_new = evaluate_c5(h)
            delta = c5_new - best_c5
            if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
                if delta < 0:
                    best_h, best_c5 = h.copy(), c5_new
            else:
                h[i], h[j] = 0.0, 1.0
        elif h[i] == 1.0 and h[j] == 0.0:
            h[i], h[j] = 0.0, 1.0
            c5_new = evaluate_c5(h)
            delta = c5_new - best_c5
            if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
                if delta < 0:
                    best_h, best_c5 = h.copy(), c5_new
            else:
                h[i], h[j] = 1.0, 0.0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(120, 0.5)

    n0 = len(h_init)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: L-BFGS-B annealing from parent
    for tau in [30, 100, 300, 1000, 3000, 8000]:
        if rem() < 20:
            break
        h_t = best_h.copy()
        for _ in range(3):
            h_new, c5_new = _lbgfs_opt(h_t, tau, 400, 1.0)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_t = h_new

    # Phase 2: SLSQP refinement (handles equality constraint exactly)
    for tau in [1000, 3000, 8000]:
        if rem() < 20:
            break
        h_new, c5_new = _slsqp_opt(best_h, tau, 400)
        if c5_new is not None and c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 3: Binary simulated annealing
    if rem() > 120:
        n_sa = min(120, max(40, n0))
        h_bin = np.interp(np.linspace(0, 2, n_sa, endpoint=False),
                         np.linspace(0, 2, n0, endpoint=False), best_h)
        h_bin = _project_feasible(h_bin, n_sa / 2.0)
        sa_iters = min(50000, int(rem() * 30))
        h_sa, c5_sa = _binary_sa(h_bin, n_sa / 2.0, sa_iters, t0, budget_s)
        if c5_sa < best_c5:
            best_h = np.interp(np.linspace(0, 2, n0, endpoint=False),
                              np.linspace(0, 2, n_sa, endpoint=False), h_sa)
            best_h = _project_feasible(best_h, n0 / 2.0)
            best_c5 = evaluate_c5(best_h)
        elif c5_sa < best_c5 + 0.002:
            # Use as a perturbation point
            h_pert = np.interp(np.linspace(0, 2, n0, endpoint=False),
                              np.linspace(0, 2, n_sa, endpoint=False), h_sa)
            h_pert = _project_feasible(h_pert, n0 / 2.0)
            for tau in [500, 2000]:
                if rem() < 10:
                    break
                h_new, c5_new = _lbgfs_opt(h_pert, tau, 300, 1.0)
                if c5_new is not None and c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                h_pert = h_new

    # Phase 4: Multi-resolution refinement
    for tn in [80, 100, 120, 140, 160]:
        if rem() < 30:
            break
        target = tn / 2.0
        h_t = np.interp(np.linspace(0, 2, tn, endpoint=False),
                       np.linspace(0, 2, len(best_h), endpoint=False), best_h)
        h_t = _project_feasible(h_t, target)
        for tau in [200, 1000, 4000]:
            if rem() < 15:
                break
            h_new, c5_new = _lbgfs_opt(h_t, tau, 300, 1.0)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_t = h_new

    # Phase 5: Random perturbations + reoptimize
    hn = len(best_h)
    for _ in range(30):
        if rem() < 15:
            break
        scale = 0.05 + 0.2 * np.random.random()
        hp = best_h + np.random.normal(0, scale, hn)
        hp = _project_feasible(hp, hn / 2.0)
        for tau in [300, 1500]:
            if rem() < 8:
                break
            h_new, c5_new = _lbgfs_opt(hp, tau, 250, 1.0)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            hp = h_new

    # Phase 6: Differential evolution at low resolution
    if rem() > 80:
        n_de = 40
        def _de_obj(x):
            return evaluate_c5(_project_feasible(x, n_de / 2.0))
        bounds_de = [(0.0, 1.0)] * n_de
        try:
            res = differential_evolution(_de_obj, bounds_de, seed=seed + 1,
                                         maxiter=80, popsize=15, tol=1e-9,
                                         polish=False, updating='deferred')
            h_de = _project_feasible(res.x, n_de / 2.0)
            c5_de = evaluate_c5(h_de)
            if c5_de < best_c5:
                best_h = np.interp(np.linspace(0, 2, len(best_h), endpoint=False),
                                   np.linspace(0, 2, n_de, endpoint=False), h_de)
                best_h = _project_feasible(best_h, len(best_h) / 2.0)
                best_c5 = evaluate_c5(best_h)
            elif c5_de < best_c5 + 0.003:
                h_pert = np.interp(np.linspace(0, 2, len(best_h), endpoint=False),
                                   np.linspace(0, 2, n_de, endpoint=False), h_de)
                h_pert = _project_feasible(h_pert, len(best_h) / 2.0)
                for tau in [500, 2000, 5000]:
                    if rem() < 10:
                        break
                    h_new, c5_new = _lbgfs_opt(h_pert, tau, 300, 1.0)
                    if c5_new is not None and c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new
                    h_pert = h_new
        except Exception:
            pass

    # Phase 7: Final high-tau polish
    if rem() > 15:
        for tau in [5000, 15000, 40000]:
            if rem() < 10:
                break
            h_new, c5_new = _lbgfs_opt(best_h, tau, 500, 5.0)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    best_h = _project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
```
