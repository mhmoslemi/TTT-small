    if s >= 0:

        # h[j] appears as (1-h[i]) when i=j, contribution -h[j+s] for j in [0, n-1-s]

        grad[t:] -= h[:n-t]

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def corr_gradient(h, s):
    """Gradient of correlation value at shift s w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        if s < n:
            grad[s:] += 1.0 - h[:n-s]
            if s > 0 or True:
                grad[:n-s] -= h[s:]
    else:
        t = -s
        if t < n:
            grad[:n-t] += 1.0 - h[t:]
            grad[t:] -= h[:n-t]
    return grad

def project_feasible(h):
    """Project h to {0≤h≤1, sum(h)=n/2} using bisection on lambda."""
    n = len(h)
    target = n / 2.0
    # We want sum(clamp(h - lambda, 0, 1)) = target
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)
    # Fix any remaining error
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        # Distribute among non-saturated entries
        mask = (result > 0) & (result < 1)
        cnt = np.sum(mask)
        if cnt > 0:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    n_corr = 2 * n - 1
    
    while (time.time() - t_start) < time_limit:

        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c5_val = np.max(corr)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Build and solve LP

        c_vec[-1] = 1.0
        
        A_ub_list = []
        b_ub_list = []
        
        for k in range(n_corr):

            g = corr_gradient(h, s)
            row = np.zeros(n + 1)
            row[:n] = g
            row[-1] = -1.0
            A_ub_list.append(row)
            b_ub_list.append(corr[k] - g @ h)
        
        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [n / 2.0]
        
        bounds = [(0, 1)] * n + [(0, 2.0)]
        
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        
        if res.status != 0:
            break
        
        h_new = res.x[:n]
        
        if np.max(np.abs(h_new - h)) < 1e-10:
            break
        
        # Damped update to avoid oscillation
        h = 0.5 * h + 0.5 * h_new
    
    return best_h, best_c5

    return v[-1]

        h_new = project_feasible(h_new)

        c5_new = compute_c5(h_new)
        if c5_new > c5_val + 0.003:
            alpha *= 0.5
            if alpha < 0.005:
                break
        else:
            h = h_new
            if c5_new < c5_val - 1e-6:
                alpha = min(alpha * 1.08, 1.0)

    return best_h, best_c5


def momentum_subgradient(h0, time_limit, t_start):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta = 0.85
    lr = 0.05

    for it in range(500000):
        if (time.time() - t_start) > time_limit:
            break
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c5_val = float(np.max(corr))

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        k_star = int(np.argmax(corr))
        s_star = k_star - (n - 1)
        g = corr_gradient(h, s_star) * dx

        v = beta * v - lr * g
        h_new = project_feasible(h + v)

        if np.max(np.abs(h_new - h)) < 1e-12:
            lr *= 0.85
            if lr < 1e-7:
                break
        h = h_new
        if it % 200 == 0:
            lr *= 0.98

    return best_h, best_c5


def t_objective(v):
    return v[-1]


def t_ineq_all(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def sum_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def smooth_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip((corr - m) / temp, -50.0, 50.0)
    return m + temp * np.log(np.sum(np.exp(shifted)))


def smooth_eq_fun(h):
    return np.sum(h) - len(h) * 0.5


def slsqp_refine(h0, max_iter=400, ftol=1e-12):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq_all},
        {'type': 'eq', 'fun': sum_eq_fun}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_feasible(h_new)
    return h_new, compute_c5(h_new)


def smooth_refine(h0, temp, max_iter=150):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': smooth_eq_fun}]
    res = minimize(smooth_obj, h0.copy(), args=(temp,), method='L-BFGS-B',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_feasible(h_new)
    return h_new, compute_c5(h_new)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float)
    h = project_feasible(h)
    c5 = compute_c5(h)
    h_best = h.copy()

    # Phase 1: Outer Approximation
    oa_time = min(200, time_left() * 0.22)
    if oa_time > 30:
        h_ua, c5_ua = outer_approximation(h, oa_time, time.time())
        if c5_ua < c5:
            h, c5 = h_ua, c5_ua
            h_best = h.copy()

    # Phase 2: Momentum subgradient
    mg_time = min(150, time_left() * 0.15)
    if mg_time > 20:
        h_mg, c5_mg = momentum_subgradient(h, mg_time, time.time())
        if c5_mg < c5:
            h, c5 = h_mg, c5_mg
            h_best = h.copy()

    # Phase 3: SLSQP
    if time_left() > 80:
        h_s, c5_s = slsqp_refine(h, max_iter=500, ftol=1e-12)
        if c5_s < c5:
            h, c5 = h_s, c5_s
            h_best = h.copy()

    # Phase 4: Smooth refinement
    for temp in [0.03, 0.015, 0.008, 0.004, 0.002]:
        if time_left() < 50:
            break
        h_sm, c5_sm = smooth_refine(h, temp, max_iter=120)
        if c5_sm < c5:
            h, c5 = h_sm, c5_sm
            h_best = h.copy()

    # Phase 5: Second OA round
    oa_time2 = min(180, time_left() * 0.2)
    if oa_time2 > 30:
        h_ua2, c5_ua2 = outer_approximation(h, oa_time2, time.time())
        if c5_ua2 < c5:
            h, c5 = h_ua2, c5_ua2
            h_best = h.copy()

    # Phase 6: SLSQP after second OA
    if time_left() > 60:
        h_s2, c5_s2 = slsqp_refine(h, max_iter=600, ftol=1e-13)
        if c5_s2 < c5:
            h, c5 = h_s2, c5_s2
            h_best = h.copy()

    # Phase 7: Multi-start perturbations
    for i in range(10):
        if time_left() < 100:
            break
        scale = 0.015 + 0.012 * i
        h_p = h_best + rng.randn(len(h_best)) * scale
        h_p = np.clip(h_p, 0.0, 1.0)
        h_p = project_feasible(h_p)

        oa_t = min(45, time_left() * 0.25)
        if oa_t > 15:
            h_oa, c5_oa = outer_approximation(h_p, oa_t, time.time())
            if c5_oa < c5:
                h, c5 = h_oa, c5_oa
                h_best = h.copy()

        if time_left() > 45:
            h_pp, c5_pp = slsqp_refine(h, max_iter=250, ftol=1e-12)
            if c5_pp < c5:
                h, c5 = h_pp, c5_pp
                h_best = h.copy()

    # Final: smooth + SLSQP polish
    if time_left() > 40:
        h_sm, c5_sm = smooth_refine(h_best, 0.003, max_iter=100)
        if c5_sm < c5:
            h_best, c5 = h_sm, c5_sm
    if time_left() > 30:
        h_fin, c5_fin = slsqp_refine(h_best, max_iter=300, ftol=1e-13)
        if c5_fin < c5:
            h_best, c5 = h_fin, c5_fin

    h_best = project_feasible(h_best)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
