    """Update correlation vector when h[j] changes to new_val"""
    n = len(h)
    old_val = h[j]
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return corr
    
    corr_new = corr.copy()
    
    # For lag L (in numpy's full mode indexing: index = L + n - 1)
    # j is first index when: 0 <= j+L <= n-1, i.e., -j <= L <= n-1-j
    # j is second index when: 0 <= j-L <= n-1, i.e., j-n+1 <= L <= j
    
    # L=0 special case: h[j]*(1-h[j])
    # old: old_val*(1-old_val), new: new_val*(1-new_val)
    # change: (new-old) - (new^2 - old^2) = (new-old)(1-new-old)
    idx0 = n - 1  # L=0 in full mode
    corr_new[idx0] += delta * (1 - new_val - old_val)
    
    # L > 0
    # j is first index: 0 < L <= n-1-j (and L >= 0, so L from 1 to n-1-j)
    # j is second index: 0 < L <= j (and j-L >= 0)
    for L in range(1, n):
        # Check if j is first index for this L: need j+L <= n-1, i.e., L <= n-1-j
        if L <= n - 1 - j:
            # h[j] * (1 - h[j+L])
            idx = n - 1 + L  # numpy full mode index for lag L
            corr_new[idx] += delta * (1 - h[j + L])
        # Check if j is second index for this L: need j-L >= 0, i.e., L <= j
        if L <= j:
            # h[j-L] * (1 - h[j])
            idx = n - 1 + L
            corr_new[idx] += -delta * h[j - L]
    
    # L < 0: let L = -M where M > 0
    # j is first index: 0 <= j-M <= n-1, i.e., M >= 0 and j-M >= 0, so M <= j
    # Also j+L = j-M >= 0 (already checked) and j+L <= n-1 (always true since M >= 0)
    # Wait, I need: 0 <= j+L = j-M <= n-1. Since M > 0, j-M < j < n, so upper bound is fine.

    # j is second index: 0 <= j-L = j+M <= n-1. Since M > 0, j+M > j. Need j+M <= n-1, so M <= n-1-j.
