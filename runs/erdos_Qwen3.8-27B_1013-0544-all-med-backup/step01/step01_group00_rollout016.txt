def corr_gradient(h, k):
    """Compute gradient of the k-th correlation value w.r.t. h."""
    n = len(h)
    lag = k - (n-1)
    # S_k = {i : 0<=i<n and 0<=i+lag<n}
    # i range: max(0, -lag) to min(n, n-lag) - 1
    i_start = max(0, -lag)
    i_end = min(n, n - lag)  # exclusive
    
    grad = np.zeros(n)
    for i in range(i_start, i_end):
        # h[i] * (1-h[i+lag])
        # d/dh[i] = (1-h[i+lag])
        # d/dh[i+lag] = -h[i]
        grad[i] += (1 - h[i+lag])
        grad[i+lag] += (-h[i])
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    """Project h to satisfy sum(h) = target and 0 <= h <= 1."""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def binary_local_search(h_bin, max_sweeps=50):
    """Local search on binary vectors. h_bin is a 0/1 array with sum = n/2."""
    n = len(h_bin)
    h = h_bin.copy().astype(float)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        worst_k = int(np.argmax(corr))
        worst_val = corr[worst_k]
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        best_delta = worst_val  # minimum improvement threshold
        best_p, best_q = -1, -1
        
        # Try all swaps: p (0->1), q (1->0)
        # For efficiency, only try a subset
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        # Try random subset of swaps for speed
        rng = np.random.RandomState(sweep + 42)
        n_trials = min(500, n_ones * n_zeros)
        for _ in range(n_trials):
            p = zeros[rng.randint(n_zeros)]
            q = ones[rng.randint(n_ones)]
            if p == q:
                continue
            # Compute new c5 estimate (just the worst k for speed)
            # Actually, let's just compute the full thing for the candidate
            h[p] = 1.0
            h[q] = 0.0
            new_corr = np.correlate(h, 1.0 - h, mode='full') * dx
            new_c5 = np.max(new_corr)
            if new_c5 < best_delta - 1e-12:
                best_delta = new_c5
                best_p, best_q = p, q
            # Restore
            h[p] = 0.0
            h[q] = 1.0

            h[best_p] = 1.0
            h[best_q] = 0.0
            best_c5 = compute_c5(h)
            improved = True

    return h, best_c5

For a fixed p (zero position), the contribution is constant across all q. Actually no, contrib_p depends only on p, not on q. So:

For the worst lag k* with lag = k* - (n-1):
- For each zero p: contrib_p = (1-h[p+lag]) * [p+lag valid] - h[p-lag] * [p-lag valid]
- For each one q: contrib_q = -(1-h[q+lag]) * [q+lag valid] + h[q-lag] * [q-lag valid]

Then delta(p,q) = contrib_p[p] + contrib_q[q]

The best (p,q) pair minimizes contrib_p[p] + contrib_q[q], which is achieved by taking the minimum contrib_p and minimum contrib_q independently!

So we can find the best p and best q in O(n) each, for a total of O(n) per iteration!

        return v[-1] - np.correlate(v[:n], 1.0 - v[:n], mode='full') * (2.0 / n)

    def eq(v):
        return np.sum(v[:n]) - n * 0.5

    def ejac(v):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'ineq', 'fun': ineq},
                                {'type': 'eq', 'fun': eq, 'jac': ejac}],
                   options={'maxiter': max_iter, 'ftol': 1e-11})
    h_new = project_h(res.x[:n], n * 0.5)
    return h_new, compute_c5(h_new)


def opt_smooth(h0, temp, max_iter=400):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]
    res = minimize(smooth_obj, h0.copy(), args=(temp,), method='L-BFGS-B',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-13})
    h_new = project_h(res.x, n * 0.5)
    return h_new, compute_c5(h_new)


def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)


def make_structured(n, seed_val):
    rng2 = np.random.RandomState(seed_val)
    h = np.zeros(n)
    h[:n // 2] = 1.0
    idx = rng2.permutation(n)
    h = h[idx]
    return project_h(h, n * 0.5)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_h(h, n * 0.5)
    c5_best = compute_c5(h)
    h_best = h.copy()

    # Phase 1: SLSQP t-var
    if tl() > 120:
        hn, cn = opt_slsqp_t(h_best, 500)
        if cn < c5_best:
            h_best, c5_best = hn, cn

    # Phase 2: Smooth refinement
    for temp in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        if tl() < 70:
            break
        hn, cn = opt_smooth(h_best, temp, 400)
        if cn < c5_best:
            h_best, c5_best = hn, cn

    # Phase 3: Binary local search
    if tl() > 40:
        hb = np.where(h_best > 0.5, 1.0, 0.0)
        hb = project_h(hb, n * 0.5)
        hb2, cb2 = binary_local_search(hb, 600)
        if cb2 < c5_best:
            h_best, c5_best = hb2, cb2

    # Phase 4: Continuous from binary
    if tl() > 100:
        hn, cn = opt_slsqp_t(h_best, 500)
        if cn < c5_best:
            h_best, c5_best = hn, cn
        for temp in [0.01, 0.005, 0.002, 0.001]:
            if tl() < 60:
                break
            hn, cn = opt_smooth(h_best, temp, 350)
            if cn < c5_best:
                h_best, c5_best = hn, cn

    # Phase 5: Try structured starting point
    if tl() > 200:
        h_struct = make_structured(n, seed + 7)
        hs_bin = np.where(h_struct > 0.5, 1.0, 0.0)
        hs_bin = project_h(hs_bin, n * 0.5)
        hs2, cs2 = binary_local_search(hs_bin, 400)
        if cs2 < c5_best:
            h_best, c5_best = hs2, cs2
            if tl() > 100:
                hn, cn = opt_slsqp_t(h_best, 400)
                if cn < c5_best:
                    h_best, c5_best = hn, cn

    # Phase 6: Random restart binary
    for trial in range(3):
        if tl() < 80:
            break
        hr = make_structured(n, seed + 100 + trial * 17)
        hr_bin = np.where(hr > 0.5, 1.0, 0.0)
        hr_bin = project_h(hr_bin, n * 0.5)
        # Add noise to break symmetry
        noise = rng.rand(n)
        hr_bin = np.where((hr_bin > 0.5) | (noise > 0.85), 1.0, 0.0)
        hr_bin = project_h(hr_bin, n * 0.5)
        hr2, cr2 = binary_local_search(hr_bin, 400)
        if cr2 < c5_best:
            h_best, c5_best = hr2, cr2
            if tl() > 80:
                hn, cn = opt_slsqp_t(h_best, 300)
                if cn < c5_best:
                    h_best, c5_best = hn, cn

    # Phase 7: Higher resolution
    for n_new in [240, 320]:
        if tl() < 250:
            break
        hi = interp_h(h_best, n_new)
        hc = hi
        for temp in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            if tl() < 100:
                break
            hn, cn = opt_smooth(hc, temp, 300)
            if cn < c5_best:
                h_best = hn.copy()
                c5_best = cn
                hc = hn
        if tl() > 120:
            hn, cn = opt_slsqp_t(hc, 500)
            if cn < c5_best:
                h_best, c5_best = hn, cn
        # Binary at higher res
        if tl() > 50:
            hb = np.where(h_best > 0.5, 1.0, 0.0)
            hb = project_h(hb, n_new * 0.5)
            hb2, cb2 = binary_local_search(hb, 500)
            if cb2 < c5_best:
                h_best, c5_best = hb2, cb2

    # Phase 8: Final polish
    if tl() > 60:
        for temp in [0.001, 0.0005, 0.0002, 0.0001]:
            if tl() < 30:
                break
            hn, cn = opt_smooth(h_best, temp, 300)
            if cn < c5_best:
                h_best, c5_best = hn, cn

    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
