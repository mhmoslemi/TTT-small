        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_smooth(h, n, dx, target_sum, thresh=0.7, max_iter=300):
    corrs = compute_corr_vec(h, dx)
    mx = np.max(corrs)
    if mx < 1e-12:
        return h.copy()
    mask = corrs >= thresh * mx
    idx = np.where(mask)[0]
    if len(idx) == 0:
        idx = np.array([np.argmax(corrs)])

    def obj(x):
        c = compute_corr_vec(x, dx)
        return np.mean(c[idx])

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
    return project_feasible(res.x, n)

def sg_step(h, n, dx, step, ntop=5):
    corrs = compute_corr_vec(h, dx)
    top = np.argsort(corrs)[-ntop:]
    offset = n - 1
    grad = np.zeros(n)
    for k in top:
        lag = k - offset
        for j in range(n):
            g = 0.0
            jm = j - lag
            if 0 <= jm < n:
                g += (1.0 - h[jm])
            jp = j + lag
            if 0 <= jp < n:
                g -= h[jp]
            grad[j] += g * dx
    grad /= len(top)
    return project_feasible(h - step * grad, n)

def perturb(h, n, rng, scale):
    h2 = h + rng.randn(n) * scale
    return project_feasible(h2, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    tsum = n / 2.0

    best_h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(best_h, dx)

    def elapsed():
        return time.time() - t0

    patience = 0
    ss = 0.06

    while elapsed() < budget_s - 20:
        prog = elapsed() / budget_s

        if prog < 0.25:
            # Phase 1: subgradient exploration + SLSQP with broad active set
            h_try = best_h
            for _ in range(8):
                h_try = sg_step(h_try, n, dx, ss)
            h_new = slsqp_smooth(h_try, n, dx, tsum, thresh=0.6, max_iter=200)
        elif prog < 0.5:
            # Phase 2: tighter active set
            h_new = slsqp_smooth(best_h, n, dx, tsum, thresh=0.8, max_iter=400)
            if compute_c5(h_new, dx) >= best_c5:
                # Try complement
                h_comp = project_feasible(1.0 - best_h, n)
                h_alt = slsqp_smooth(h_comp, n, dx, tsum, thresh=0.8, max_iter=300)
                if compute_c5(h_alt, dx) < compute_c5(h_new, dx):
                    h_new = h_alt
        elif prog < 0.75:
            # Phase 3: mixed strategy
            r = rng.rand()
            if r < 0.4:
                h_try = best_h
                for _ in range(4):
                    h_try = sg_step(h_try, n, dx, ss * 0.5)
                h_new = slsqp_smooth(h_try, n, dx, tsum, thresh=0.9, max_iter=350)
            elif r < 0.7:
                scale = 0.02 + 0.03 * (1.0 - prog)
                h_try = perturb(best_h, n, rng, scale)
                h_new = slsqp_smooth(h_try, n, dx, tsum, thresh=0.7, max_iter=400)
            else:
                # Larger perturbation to escape
                scale = 0.05 + 0.05 * rng.rand()
                h_try = perturb(best_h, n, rng, scale)
                h_new = slsqp_smooth(h_try, n, dx, tsum, thresh=0.5, max_iter=300)
        else:
            # Phase 4: fine tuning
            scale = 0.005 + 0.01 * rng.rand()
            h_try = perturb(best_h, n, rng, scale)
            h_new = slsqp_smooth(h_try, n, dx, tsum, thresh=0.95, max_iter=500)

        c5_new = compute_c5(h_new, dx)

        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new
            patience = 0
            ss = min(0.2, ss * 1.03)
        else:
            patience += 1
            ss *= 0.96
            if patience >= 8:
                # Forced restart
                scale = 0.04 + 0.04 * rng.rand()
                h_try = perturb(best_h, n, rng, scale)
                h_new2 = slsqp_smooth(h_try, n, dx, tsum, thresh=0.5, max_iter=400)
                c5_2 = compute_c5(h_new2, dx)
                if c5_2 < best_c5:
                    best_h = h_new2
                    best_c5 = c5_2
                # Also try complement of best
                h_comp = project_feasible(1.0 - best_h, n)
                h_new3 = slsqp_smooth(h_comp, n, dx, tsum, thresh=0.7, max_iter=300)
                c5_3 = compute_c5(h_new3, dx)
                if c5_3 < best_c5:
                    best_h = h_new3
                    best_c5 = c5_3
                patience = 0

    # Final polish
    if elapsed() < budget_s - 5:
        h_final = slsqp_smooth(best_h, n, dx, tsum, thresh=0.99, max_iter=600)
        c5_final = compute_c5(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
