from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-9
    else:
        mask = h > 1e-9
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_max(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def p_norm(corr, p):
    return float(np.sum(corr**p)**(1.0/p))

def optimize_slqp(h0, n, tau, maxiter=200, ftol=1e-12):
    target = n / 2.0
    from scipy.optimize import Bounds
    bnds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(corr, tau)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    x = fix_sum(res.x, target)
    return x, evaluate_c5(x)

    corr = np.correlate(h, 1.0 - h, mode="full")

        for j in range(n):
            h_j = h[j]

            d_lo = -h_j
            d_hi = 1.0 - h_j

            idx1 = j - i_vals + n - 1
            valid1 = (idx1 >= 0) & (idx1 < n)
            coef1 = np.zeros(2*n-1)
            coef1[valid1] = 1.0 - h[idx1[valid1]]
            
            # Second term coefficients
            idx2 = j + i_vals - n + 1
            valid2 = (idx2 >= 0) & (idx2 < n)
            coef2 = np.zeros(2*n-1)
            coef2[valid2] = -h[idx2[valid2]]
            
            # Combined linear coefficient

            # Quadratic term: -1 at index n-1

            # For a given delta, new_max = max(corr + delta*lin + delta^2*quad)

            # Add 0 for reference
            best_max = np.max(corr)
            best_d = 0.0
            
            for d in deltas:
                new_corr = corr + d * lin + d * d * quad
                m = np.max(new_corr)
                if m < best_max - 1e-14:
                    best_max = m
                    best_d = d
            
            if abs(best_d) > 1e-12:
                h[j] += best_d

                corr = np.correlate(h, 1.0 - h, mode="full")
                improved = True

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-9
    else:

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-9
    else:
        mask = h > 1e-9
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(corr, tau):
    x = tau * corr
    xm = np.max(x)
    return float(xm / tau + np.log(np.sum(np.exp(x - xm))) / tau)

def coord_descent_vec(h, n_iter=10, grid_size=100):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    L = 2 * n - 1
    i_vals = np.arange(L)
    quad = np.zeros(L)
    quad[n - 1] = -1.0
    for it in range(n_iter):
        improved = False
        for j in range(n):
            hj = h[j]
            d_lo = -hj
            d_hi = 1.0 - hj
            if d_hi - d_lo < 1e-10:
                continue
            idx1 = j - i_vals + n - 1
            v1 = (idx1 >= 0) & (idx1 < n)
            lin = np.zeros(L)
            lin[v1] = 1.0 - h[idx1[v1]]
            idx2 = j + i_vals - n + 1
            v2 = (idx2 >= 0) & (idx2 < n)
            lin[v2] += -h[idx2[v2]]
            deltas = np.linspace(d_lo, d_hi, grid_size)
            new_maxs = np.max(corr[None, :] + deltas[:, None] * lin[None, :]
                              + (deltas**2)[:, None] * quad[None, :], axis=1)
            all_maxs = np.concatenate([[np.max(corr)], new_maxs])
            all_d = np.concatenate([[0.0], deltas])
            bi = np.argmin(all_maxs)
            bd = all_d[bi]
            if abs(bd) > 1e-10:
                h[j] = np.clip(hj + bd, 0.0, 1.0)
                corr = np.correlate(h, 1.0 - h, mode="full")
                improved = True
        if not improved:
            break
    h = fix_sum(h, n / 2.0)
    return h

def swap_local_search(h, n_trials=3000, n_rounds=5):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_max = np.max(corr)
    for r in range(n_rounds):
        improved = False
        for _ in range(n_trials):
            i, j = np.random.randint(0, n, 2)
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            new_corr = np.correlate(h, 1.0 - h, mode="full")
            new_max = np.max(new_corr)
            if new_max < cur_max - 1e-14:
                cur_max = new_max
                corr = new_corr
                improved = True
            else:
                h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return h

def opt_lbfbsb(h0, n, tau, lam, maxiter=400):
    target = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        sm = smooth_max_val(corr, tau)
        return sm + lam * (np.sum(x) - target) ** 2
    res = minimize(obj, h0, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
    x = fix_sum(res.x, target)
    return x, evaluate_c5(x)

def opt_slqp(h0, n, tau, maxiter=200):
    target = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max_val(corr, tau)
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    res = minimize(obj, h0, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    x = fix_sum(res.x, target)
    return x, evaluate_c5(x)

def random_binary_init(n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h
