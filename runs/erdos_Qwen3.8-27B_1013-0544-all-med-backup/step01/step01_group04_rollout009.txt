def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target):
    n = len(h)
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def soft_gradient(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx

    weights = np.exp(scaled)
    wsum = np.sum(weights)
    weights /= wsum
    grad = np.zeros(n)
