```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    return m + np.float64(np.log(np.sum(np.exp(beta * (corr - m)))) / beta)

def compute_gradient(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    w = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        for k in range(2 * n - 1):
            shift = k - (n - 1)
            if 0 <= i + shift < n:
                g += w[k] * (1.0 - h[i + shift])
            if 0 <= i - shift < n:
                g += w[k] * (-h[i - shift])
        grad[i] = g * dx
    return grad

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-10:
        # Distribute the correction
        diff = (target_sum - s) / len(h)
        h += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def projected_gradient_descent(h, target_sum, beta, n_iter, step_size, time_func):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_val = smooth_max_val(np.correlate(h, 1.0-h, 'full'), beta)
    
    for it in range(n_iter):
        if time_func() < 20:
            break
        grad = compute_gradient(h, beta)
        # Adaptive step size
        step = step_size
        for _ in range(5):
            h_new = h - step * grad
            h_new = project_h(h_new, target_sum)
            val = smooth_max_val(np.correlate(h_new, 1.0-h_new, 'full'), beta)
            if val < best_val:
                h = h_new
                if val < best_val:
                    best_val = val
                    best_h = h.copy()
                break
            step *= 0.5
        else:
            break
    return best_h, best_val
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_gradient(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    w = exps / np.sum(exps)
    T1 = np.correlate(1.0 - h, w, mode='full')[3*n-3 : 2*n-2 : -1]
    T2 = np.convolve(h, w, mode='full')[n-1 : 2*n-1]
    return dx * (T1 - T2)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = (target_sum - h.sum()) / len(h)
    if abs(diff) > 1e-12:
        h = h + diff
        h = np.clip(h, 0.0, 1.0)
    diff = (target_sum - h.sum()) / len(h)
    if abs(diff) > 1e-14:
        h = h + diff
    return h


def make_objective(n_points, dx, target_sum, beta, lam):
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        sm = m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta
        return sm * dx + lam * (np.sum(h) - target_sum) ** 2
    return obj


def make_gradient(n_points, dx, target_sum, beta, lam):
    def jac(h):
        g = compute_gradient(h, beta)
        g += 2.0 * lam * (np.sum(h) - target_sum)
        return g
    return jac


def optimize_from(h0, n_points, dx, target_sum, bounds, time_left, budget_per_phase):
    best_h = project_h(h0, target_sum)
    best_c5 = compute_c5(best_h)
    h = best_h.copy()
    lam = 500.0
    betas = [10, 30, 100, 300, 1000, 3000]
    for beta in betas:
        if time_left() < 10:
            break
        obj = make_objective(n_points, dx, target_sum, beta, lam)
        jac = make_gradient(n_points, dx, target_sum, beta, lam)
        try:
            res = minimize(obj, h, jac=jac, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 400, 'ftol': 1e-16, 'gtol': 1e-12})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
                h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5
```
