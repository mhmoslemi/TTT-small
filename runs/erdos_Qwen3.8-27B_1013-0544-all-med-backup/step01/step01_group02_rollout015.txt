def compute_grad_smooth_max(h, n, alpha):
    """Compute gradient of smooth max of correlation."""
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    shifted = shifted - shifted.max()
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    n_corr = len(c)  # 2n-1
    shift = n - 1
    grad = np.zeros(n)
    
    for k in range(n_corr):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - shift
        if lag == 0:
            grad += w * (1 - 2*h)
        elif lag > 0:
            # c_lag = sum_{m=0}^{n-1-lag} h[m]*(1-h[m+lag])
            # d/dh[j]: from h[m] term: (1-h[j+lag]) if j <= n-1-lag
            #          from h[m+lag] term: -h[j-lag] if j >= lag
            grad[:n-lag] += w * (1 - h[lag:])
            grad[lag:] += w * (-h[:n-lag])
        else:
            L = -lag
            # c_{-L} = sum_{m=L}^{n-1} h[m]*(1-h[m-L])
            # Let k'=m-L, range 0 to n-1-L
            # c_{-L} = sum_{k'=0}^{n-1-L} h[k'+L]*(1-h[k'])
            # d/dh[j]: from h[k'+L] term: (1-h[j-L]) if L <= j <= n-1
            #          from h[k'] term: -h[j+L] if 0 <= j <= n-1-L
            grad[L:] += w * (1 - h[:n-L])
            grad[:n-L] += w * (-h[L:])
    
    return grad

        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    shifted = shifted - shifted.max()
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def compute_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    shifted = shifted - shifted.max()
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    shift = n - 1
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - shift
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] += w * (-h[:n - lag])
        else:
            L = -lag
            grad[L:] += w * (1.0 - h[:n - L])
            grad[:n - L] += w * (-h[L:])
    return grad

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_try = project_feasible(res.x, n)
        return h_try, compute_c5(h_try, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def proj_grad(h, n, alpha, t0, budget_s, max_iters=3000):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    for it in range(max_iters):
        if budget_s - (time.time() - t0) < 5:
            break
        grad = compute_grad(best_h, n, alpha)
        lr = 0.5
        found = False
        for _ in range(15):
            h_new = project_feasible(best_h - lr * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c5 - 1e-15:
                best_h = h_new
                best_c5 = c_new
                found = True
                break
            lr *= 0.5
        if not found:
            break
    return best_h, best_c5

def local_swaps(h, n, t0, budget_s, max_iters=50000):
    best_c5 = compute_c5(h, n)
    h = h.copy()
    rng_local = np.random.RandomState(999)
    deltas = [0.001, 0.002, 0.003, 0.005, 0.01, 0.015, 0.02]
    for _ in range(max_iters):
        if budget_s - (time.time() - t0) < 3:
            break
        i = rng_local.randint(n)
        j = rng_local.randint(n)
        if i == j:
            continue
        best_delta = None
        best_c = best_c5
        for s in [1, -1]:
            for d in deltas:
                hi = h[i] + s * d
                hj = h[j] - s * d
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h[i] = hi
                    h[j] = hj
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_delta = (s, d)
                    h[i] -= s * d
                    h[j] += s * d
        if best_delta is not None:
            s, d = best_delta
            h[i] += s * d
            h[j] -= s * d
            best_c5 = best_c
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 80
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)

    # Phase 1: SLSQP continuation
    for alpha in [30, 100, 500, 2000, 10000, 50000]:
        if rem() < 20:
            break
        h_try, c_try = slsqp(best_h, n, alpha, maxiter=5000)
        if c_try < best_c5 - 1e-14:
            best_h, best_c5 = h_try, c_try

    # Phase 2: Diverse SLSQP restarts
    if rem() > 100:
        for restart in range(80):
            if rem() < 20:
                break
            scale = rng.choice([0.003, 0.008, 0.015, 0.03, 0.05, 0.08, 0.12, 0.2, 0.3])
            h_pert = best_h + rng.normal(0, scale, size=n)
            h_pert = project_feasible(h_pert, n)
            alpha = int(10.0 ** rng.uniform(2.0, 4.5))
            h_try, c_try = slsqp(h_pert, n, alpha, maxiter=3000)
            if c_try < best_c5 - 1e-14:
                best_h, best_c5 = h_try, c_try
                if rem() > 10:
                    h_try2, c_try2 = slsqp(best_h, n, 100000, maxiter=3000)
                    if c_try2 < best_c5 - 1e-14:
                        best_h, best_c5 = h_try2, c_try2

    # Phase 3: Projected gradient complement
    if rem() > 40:
        for alpha in [5000, 20000, 100000]:
            if rem() < 10:
                break
            h_try, c_try = proj_grad(best_h, n, alpha, t0, budget_s, max_iters=2000)
            if c_try < best_c5 - 1e-14:
                best_h, best_c5 = h_try, c_try

    # Phase 4: Fine local swaps (most critical for final push)
    if rem() > 5:
        h_try, c_try = local_swaps(best_h, n, t0, budget_s, max_iters=80000)
        if c_try < best_c5 - 1e-14:
            best_h, best_c5 = h_try, c_try

    # Phase 5: Post-swap SLSQP polish
    if rem() > 5:
        h_try, c_try = slsqp(best_h, n, 100000, maxiter=5000)
        if c_try < best_c5:
            best_h, best_c5 = h_try, c_try

    # Phase 6: Final ultra-fine swaps
    if rem() > 3:
        h_try, c_try = local_swaps(best_h, n, t0, budget_s, max_iters=30000)
        if c_try < best_c5 - 1e-14:
            best_h, best_c5 = h_try, c_try

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
