            h_init = 0.5 + 0.3 * np.cos(2 * np.pi * (1 + r % 3) * np.arange(n) / n)
        else:
            h_init = best_h + rng.normal(0, 0.03, n)

        h_init = project_feasible(h_init, target_sum)

        # Quick optimization with smooth objective
        _g_k = 5
        _g_p = 8
        res = minimize(_obj_topk, h_init, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 600, 'ftol': 1e-12})
        if res.x is not None:
            h_try = project_feasible(res.x, target_sum)
            c5_try = compute_c5(h_try, dx)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                # Polish with exact objective
                if tleft() > 25:
                    res2 = minimize(_obj_exact, h_try, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 1500, 'ftol': 1e-14})
                    if res2.x is not None:
                        h_t2 = project_feasible(res2.x, target_sum)
                        c5_t2 = compute_c5(h_t2, dx)
                        if c5_t2 < best_c5:
                            best_c5 = c5_t2
                            best_h = h_t2.copy()

    # Phase 4: Active-set gradient fine-tuning with adaptive step
    if tleft() > 8:
        h = best_h.copy()
        step = 0.001
        momentum = np.zeros(n)
        stall = 0
        for it in range(20000):
            if tleft() < 1.0:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = float(np.max(corr))
            tol = max(1e-5, c_max * 2e-4)
            active = np.where(corr >= c_max - tol)[0]
            if len(active) < 1:
                active = np.array([int(np.argmax(corr))])
            elif len(active) > 15:
                active = np.argsort(corr)[-15:]

            grad = np.zeros(n)
            for idx in active:
                L = int(idx) - n_half
                grad += _grad_lag_component(h, L, n)
            grad /= float(len(active))

            momentum = 0.5 * momentum - step * grad
            h_new = project_feasible(h + momentum, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                stall = 0
                step = min(step * 1.005, 0.003)
            else:
                stall += 1
                step *= 0.93
                if stall > 100 and step < 1e-8:
                    # Reset with small perturbation
                    h = best_h + rng.normal(0, 0.01, n)
                    h = project_feasible(h, target_sum)
                    momentum = np.zeros(n)
                    step = 0.001
                    stall = 0
            h = h_new

    # Phase 5: Final SLSQP from best
    if tleft() > 10:
        res = minimize(_obj_exact, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 2000, 'ftol': 1e-15})
        if res.x is not None:
            h_try = project_feasible(res.x, target_sum)
            c5_try = compute_c5(h_try, dx)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    return (best_h, best_c5, n)
