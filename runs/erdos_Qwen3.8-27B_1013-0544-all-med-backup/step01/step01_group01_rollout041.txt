But I think a smarter approach would be to use the **exact delta** computation. When we swap h[i] (from 1 to 0) and h[j] (from 0 to 1), the change in each correlation value can be computed in O(n):

For lag k, the correlation is corr_k = sum_l h[l] * (1 - h[l+k]).
After swapping i and j:
- If l=i: term changes from 1*(1-h[i+k]) to 0*(1-h[i+k]) = 0, so delta = -(1-h[i+k])
- If l+k=i (i.e., l=i-k): term changes from h[i-k]*(1-1) to h[i-k]*(1-0) = h[i-k], so delta = h[i-k]
- If l=j: term changes from 0*(1-h[j+k]) to 1*(1-h[j+k]) = 1-h[j+k], so delta = 1-h[j+k]
- If l+k=j (i.e., l=j-k): term changes from h[j-k]*(1-0) to h[j-k]*(1-1) = 0, so delta = -h[j-k]

So the total delta for lag k is at most 4 terms. We can compute this for all k in O(n) time.

This makes each swap evaluation O(n) instead of O(n^2), allowing us to try all O(n^2) swaps per iteration.

    """Compute the change in max correlation when swapping h[i] and h[j].

    # corr_k = sum_l h[l] * (1 - h[l+k]) for l where both l and l+k are in [0,n-1]

    # For the "full" correlation, we also have negative lags
    
    # Actually, let's just compute the full correlation array change
    # The full correlation has 2n-1 elements
    # Element at index (n-1+k) corresponds to... let me just compute directly
    
    # Easier: compute new correlation array
    # But that's O(n^2) which defeats the purpose
    
    # Let me use the formula:
    # corr_full[m] for m = 0, ..., 2n-2
    # corr_full[m] = sum_{l} h[l] * (1-h)[l + (m - (n-1))]
    #              = sum_l h[l] * (1 - h[l + m - n + 1])
    
    # Let k = m - (n-1), so m = n-1+k, k ranges from -(n-1) to (n-1)
    # corr at lag k = sum_l h[l] * (1 - h[l+k])
    
    # When we change h[i]: 1→0 and h[j]: 0→1
    # For lag k:
    #   - Term where l = i: was h[i]*(1-h[i+k]) = (1-h[i+k]), now 0. Delta = -(1-h[i+k]) if i+k in range
    #   - Term where l+k = i, i.e., l = i-k: was h[i-k]*(1-h[i]) = 0, now h[i-k]*(1-0) = h[i-k]. Delta = h[i-k] if i-k in range
    #   - Term where l = j: was h[j]*(1-h[j+k]) = 0, now (1-h[j+k]). Delta = 1-h[j+k] if j+k in range
    #   - Term where l+k = j, i.e., l = j-k: was h[j-k]*(1-h[j]) = h[j-k], now h[j-k]*0 = 0. Delta = -h[j-k] if j-k in range
    
    delta = np.zeros(2*n-1)  # indexed by m = 0..2n-2, corresponding to k = m-(n-1)
    
    for k in range(-(n-1), n):
        m = n - 1 + k
        d = 0.0
        if 0 <= i + k < n:
            d -= (1.0 - h[i+k])  # l=i term
        if 0 <= i - k < n:
            d += h[i-k]  # l=i-k term (where l+k=i)
        if 0 <= j + k < n:
            d += (1.0 - h[j+k])  # l=j term
        if 0 <= j - k < n:
            d -= h[j-k]  # l=j-k term (where l+k=j)
        delta[m] = d
    
    return delta

        if abs(s - target_sum) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += (target_sum - s) / len(h)
        else:
            h[free] += (target_sum - s) / free.sum()
    return np.clip(h, 0.0, 1.0)

def binary_steepest_descent(h_bin, n, max_iters=60, kick_every=12):
    target = n // 2
    h_bin = h_bin.astype(float).copy()
    while int(h_bin.sum()) > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[np.random.randint(len(idx))]] = 0.0
    while int(h_bin.sum()) < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[np.random.randint(len(idx))]] = 1.0
    curr_max = float(np.max(np.correlate(h_bin, 1.0 - h_bin, mode="full")))
    for it in range(max_iters):
        if it > 0 and it % kick_every == 0:
            for _ in range(4):
                i, j = np.random.choice(n, 2, replace=False)
                if h_bin[i] != h_bin[j]:
                    h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            curr_max = float(np.max(np.correlate(h_bin, 1.0 - h_bin, mode="full")))
            continue
        ones = np.where(h
_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        best_imp = 0.0
        best_swap = None
        for i in ones:
            for j in zeros:
                h_bin[i], h_bin[j] = 0.0, 1.0
                nm = float(np.max(np.correlate(h_bin, 1.0 - h_bin, mode="full")))
                h_bin[i], h_bin[j] = 1.0, 0.0
                imp = curr_max - nm
                if imp > best_imp + 1e-14:
                    best_imp = imp
                    best_swap = (int(i), int(j))
        if best_swap is None:
            break
        i, j = best_swap
        h_bin[i], h_bin[j] = 0.0, 1.0
        curr_max -= best_imp
    return h_bin, curr_max

def coordinate_descent(h, n, target, max_sweeps=80):
    h = h.copy()
    corr_all = np.correlate(h, 1.0 - h, mode="full")
    curr_max = float(np.max(corr_all))
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            a = np.zeros(2 * n - 1)
            for k in range(-(n - 1), n):
                ak = 0.0
                if 0 <= i + k < n:
                    ak += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    ak -= h[i - k]
                a[n - 1 + k] = ak
            c = corr_all - a * h[i]
            val0 = float(np.max(c))
            val1 = float(np.max(a + c))
            k0 = int(np.argmax(c))
            k1 = int(np.argmax(a + c))
            cands = [0.0, 1.0, float(h[i])]
            if k0 != k1 and abs(a[k0] - a[k1]) > 1e-15:
                ti = (c[k1] - c[k0]) / (a[k0] - a[k1])
                if 0.0 <= ti <= 1.0:
                    cands.append(ti)
            top_c = np.argsort(c)[::-1][:4]
            top_1 = np.argsort(a + c)[::-1][:4]
            for ki in top_c:
                for kj in top_1:
                    if ki != kj and abs(a[ki] - a[kj]) > 1e-15:
                        ti = (c[kj] - c[ki]) / (a[ki] - a[kj])
                        if 0.0 <= ti <= 1.0:
                            cands.append(ti)
            best_t = float(h[i])
            best_val = curr_max
            for t in cands:
                t = float(np.clip(t, 0.0, 1.0))
                val = float(np.max(a * t + c))
                if val < best_val - 1e-14:
                    best_val = val
                    best_t = t
            if abs(best_t - h[i]) > 1e-10:
                h[i] = best_t
                corrected = a * h[i] + c
                curr_max = float(np.max(corrected))
                corr_all = corrected
                improved = True
        h = project_h(h, target)
        corr_all = np.correlate(h, 1.0 - h, mode="full")
        curr_max = float(np.max(corr_all))
        if not improved:
            break
    return h

def slsqp_polish(h, n, target, beta=300, maxiter=800, ftol=1e-13):
    def obj(hv):
        corr = np.correlate(hv, 1.0 - hv, mode="full")
        return smooth_max(corr, beta) * (2.0 / n)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                  constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_h(res.x, target)

def generate_starting_points(n, target, rng):
    starts = []
    h_rand = np.zeros(n)
    h_rand[rng.choice(n, target, replace=False)] = 1.0
    starts.append(h_rand)
    h_block = np.zeros(n)
    for nb in range(2, min(8, n // 5 + 1)):
        h_b = np.zeros(n)
        bs = n // nb
        for b in range(nb):
            if b % 2 == 0:
                s = b * bs
                e = min((b + 1) * bs, n)
                h_b[s:e] = 1.0
        diff = int(h_b.sum()) - target
        if diff > 0:
            idx = np.where(h_b == 1)[0]
            h_b[idx[:diff]] = 0.0
        elif diff < 0:
            idx = np.where(h_b == 0)[0]
            h_b[idx[:(-diff)]] = 1.0
        starts.append(h_b.copy())
    h_tri = np.zeros(n)
    for i in range(n):
        if (i % 4) < 2:
            h_tri[i] = 1.0
    diff = int(h_tri.sum()) - target
    if diff > 0:
        idx = np.where(h_tri == 1)[0]
        h_tri[idx[:diff]] = 0.0
    elif diff < 0:
        idx = np.where(h_tri == 0)[0]
        h_tri[idx[:(-diff)]] = 1.0
    starts.append(h_tri.copy())
    for _ in range(3):
        h_r = np.zeros(n)
        h_r[rng.choice(n, target, replace=False)] = 1.0
        starts.append(h_r.copy())
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = 70

    n_candidates = [70]
    if 'initial_h_values' in globals() and initial_h_values is not None:
        n0 = len(initial_h_values)
        n_candidates.insert(0, n0)

    for n in n_candidates:
        if time.time() - start_time > budget_s - 60:
            break
        target = n // 2
        local_best_c5 = float('inf')
        local_best_h = None

        if n == 70 or (n not in n_candidates[1:]):
            starts = generate_starting_points(n, target, rng)
            if 'initial_h_values' in globals() and initial_h_values is not None and len(initial_h_values) == n:
                hv = np.asarray(initial_h_values, dtype=float)
                starts.insert(0, (hv > 0.5).astype(float))
        else:
            hv = np.asarray(initial_h_values, dtype=float)
            starts = [(hv > 0.5).astype(float)]

        for s, h_start in enumerate(starts):
            if time.time() - start_time > budget_s - 45:
                break
            h_binary, max_corr = binary_steepest_descent(h_start, n, max_iters=50, kick_every=10)
            c5_bin = float(max_corr) * 2.0 / n
            if c5_bin < local_best_c5:
                local_best_c5 = c5_bin
                local_best_h = h_binary.copy()

            h_cont = project_h(0.15 + 0.7 * h_binary, target)
            try:
                h_cd = coordinate_descent(h_cont, n, target, max_sweeps=60)
                c5_cd = compute_c5(h_cd)
                if c5_cd < local_best_c5:
                    local_best_c5 = c5_cd
                    local_best_h = h_cd.copy()
            except Exception:
                pass

            try:
                h_sp = slsqp_polish(h_cont, n, target, beta=200, maxiter=400)
                c5_sp = compute_c5(h_sp)
                if c5_sp < local_best_c5:
                    local_best_c5 = c5_sp
                    local_best_h = h_sp.copy()
            except Exception:
                pass

        if local_best_h is not None:
            if local_best_c5 < best_c5:
                best_c5 = local_best_c5
                best_h = local_best_h.copy()
                best_n = n

            # Final polish on best for this n
            for beta in [400, 800]:
                if time.time() - start_time > budget_s - 15:
                    break
                try:
                    h_p = slsqp_polish(best_h, n, target, beta=beta, maxiter=1000, ftol=1e-14)
                    c5_p = compute_c5(h_p)
                    if c5_p < best_c5:
                        best_c5 = c5_p
                        best_h = h_p.copy()
                except Exception:
                    pass
                try:
                    h_cd2 = coordinate_descent(best_h, n, target, max_sweeps=40)
                    c5_cd2 = compute_c5(h_cd2)
                    if c5_cd2 < best_c5:
                        best_c5 = c5_cd2
                        best_h = h_cd2.copy()
                except Exception:
                    pass

    if 'initial_h_values' in globals() and initial_h_values is not None:
        ni = len(initial_h_values)
        hi = np.asarray(initial_h_values, dtype=float)
        ti = ni // 2
        ht = project_h(hi, ti)
        try:
            hp = slsqp_polish(ht, ni, ti, beta=500, maxiter=1500, ftol=1e-14)
            cp = compute_c5(hp)
            if cp < best_c5:
                best_c5 = cp
                best_h = hp.copy()
                best_n = ni
        except Exception:
            pass

    if best_h is None:
        best_h = np.full(70, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = 70

    return best_h, best_c5, best_n
