from scipy.optimize import minimize
import time

def project_feasible(x, target_sum, n):
    """Project x onto {0<=x<=1, sum(x)=target_sum} using bisection on threshold."""
    lo = float(x.min()) - 1.5
    hi = float(x.max()) + 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr
