def smooth_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    out = np.correlate(h, g, 'full') * dx  # shape (2n-1,)
    # softmax
    max_c = np.max(out)
    exp_c = np.exp((out - max_c) / T)
    p = exp_c / np.sum(exp_c)  # shape (2n-1,)
    
    # Gradient: for each i, sum over k of p[n-1+k] * (dx/T) * d_out[n-1+k]/dh[i]
    # d_out[n-1+k]/dh[i] = (1-h[i-k])*1[0<=i-k<n] - h[i+k]*1[0<=i+k<n]
    
    grad = np.zeros(n)
    for i in range(n):
        # First term: sum over k where 0<=i-k<n, i.e., k from max(-(n-1), i-(n-1)) to min(n-1, i)
        # k = i-j where j ranges over valid j
        # Actually, k ranges such that i-k is in [0, n-1], so k in [i-(n-1), i]
        # But k itself must be in [-(n-1), n-1]
        k_min1 = max(-(n-1), i-(n-1))
        k_max1 = min(n-1, i)
        # For these k, contribution is p[n-1+k] * (dx/T) * (1-h[i-k])
        # Second term: k in [max(-(n-1), -i), min(n-1, n-1-i)]
        k_min2 = max(-(n-1), -i)
        k_max2 = min(n-1, n-1-i)
        # For these k, contribution is p[n-1+k] * (dx/T) * (-h[i+k])
        
        if k_max1 >= k_min1:
            k_range1 = np.arange(k_min1, k_max1+1)
            j_range1 = i - k_range1  # = i-k, which is in [0, n-1]
            grad[i] += np.sum(p[n-1+k_range1] * (1-h[j_range1]))
        
        if k_max2 >= k_min2:
            k_range2 = np.arange(k_min2, k_max2+1)
            j_range2 = i + k_range2  # = i+k, which is in [0, n-1]
            grad[i] -= np.sum(p[n-1+k_range2] * h[j_range2])
    
    grad *= dx / T

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(x, T, penalty):
    n = len(x)
    dx = 2.0 / n
    out = np.correlate(x, 1.0 - x, 'full') * dx
    m = np.max(out)
    exp_c = np.exp((out - m) / T)
    p = exp_c / np.sum(exp_c)
    logsumexp = m + T * np.log(np.sum(exp_c))
    s = np.sum(x) - n * 0.5
    obj = logsumexp + 0.5 * penalty * s * s
    grad = np.zeros(n)
    for i in range(n):
        k1lo = max(-(n-1), i-(n-1))
        k1hi = min(n-1, i)
        if k1hi >= k1lo:
            ks = np.arange(k1lo, k1hi + 1)
            js = i - ks
            grad[i] += np.dot(p[n-1+ks], 1.0 - x[js])
        k2lo = max(-(n-1), -i)
        k2hi = min(n-1, n-1-i)
        if k2hi >= k2lo:
            ks = np.arange(k2lo, k2hi + 1)
            js = i + ks
            grad[i] -= np.dot(p[n-1+ks], x[js])
    grad *= dx / T
    grad += penalty * s
    return obj, grad

def optimize_smooth_lbfgsb(h0, T, penalty=1e6, max_iter=400):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj_and_grad, h0.copy(), args=(T, penalty),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-9})
    h_new = project_h(res.x, n * 0.5)
    return h_new, compute_c5(h_new)

def t_objective(v):
    return v[-1]

def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5

def t_eq_jac(v):
    n = len(v)
    j = np.zeros(n)

    if time_left() > 180 and n != n_low:
        h_low = interp_h(h_best, n_low)
        for T in [0.02, 0.01, 0.005]:
            if time_left() < 60:
                break
            h_new, c5_new = optimize_smooth_lbfgsb(h_low, T, penalty=1e6, max_iter=300)
            if c5_new < c5_best:
                h_low = h_new
                h_best = h_new.copy()
                c5_best = c5_new
        if time_left() > 80:
            h_new, c5_new = optimize_t_var(h_best, max_iter=400, ftol=1e-11)
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new

    # Final: coordinate descent polish on best
    if time_left() > 30:
        h_new, c5_new = coord_descent_polish(h_best, max_sweeps=8)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
