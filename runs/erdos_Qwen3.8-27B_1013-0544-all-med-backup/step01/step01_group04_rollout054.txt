import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = int(np.sum(free))
        if nf > 0:
            h[free] -= diff / nf
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h)
            for idx in order:
                d = min(h[idx], diff)
                h[idx] -= d
                diff -= d
                if diff < 1e-15:
                    break
        else:
            order = np.argsort(h)[::-1]
            for idx in order:
                d = min(1.0 - h[idx], -diff)
                h[idx] += d
                diff += d
                if diff > -1e-15:
                    break
    return h

def de_objective(x, dx):
    return compute_c5(x, dx)

def eq_constraint(x, target_sum):
    return np.sum(x) - target_sum

def eq_constraint_jac(x, target_sum):
    return np.ones(len(x))

def de_optimize(h_init, n, dx, time_limit, pop_size, rng):
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, dx)
    pop = np.zeros((pop_size, n))
    pop_c5 = np.zeros(pop_size)
    pop[0] = best_h
    pop_c5[0] = best_c5
    for i in range(1, pop_size):
        r = rng.rand(n)
        pop[i] = project_feasible(r, n)
        pop_c5[i] = compute_c5(pop[i], dx)
        if pop_c5[i] < best_c5:
            best_c5 = pop_c5[i]
            best_h = pop[i].copy()
    gen = 0
    while time.time() < time_limit:
        gen += 1
        progress = min(1.0, gen / 3000.0)
        F = 0.9 - 0.55 * progress
        Cr = 0.8 - 0.35 * progress
        for i in range(pop_size):
            if time.time() >= time_limit:
                break
            cands = [j for j in range(pop_size) if j != i]
            a, b, c = rng.choice(cands, 3, replace=False)
            v = pop[a] + F * (pop[b] - pop[c])
            mask = rng.rand(n) < Cr
            mask[rng.randint(n)] = True
            trial = np.where(mask, v, pop[i])
            trial = project_feasible(trial, n)
            trial_c5 = compute_c5(trial, dx)
            if trial_c5 <= pop_c5[i]:
                pop[i] = trial
                pop_c5[i] = trial_c5
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
    return best_h, best_c5

def sa_optimize(h_init, n, dx, time_limit, rng):
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, dx)
    curr_h = h_init.copy()
    curr_c5 = best_c5
    T_start = 0.008
    T_end = 0.00005
    t0 = time.time()
    dur = max(1.0, time_limit - t0)
    while time.time() < time_limit:
        progress = min(1.0, (time.time() - t0) / dur)
        T = T_start * (T_end / T_start) ** progress
        n_pert = rng.randint(1, 4)
        idx = rng.choice(n, n_pert, replace=False)
        new_h = curr_h.copy()
        scale = 0.15 * (1.0 - 0.7 * progress) + 0.02
        for j in idx:
            new_h[j] += rng.uniform(-scale, scale)
        new_h = project_feasible(new_h, n)
        new_c5 = compute_c5(new_h, dx)
        delta = new_c5 - curr_c5
        if delta < 0 or (delta < 0.01 and rng.random() < np.exp(-delta / max(T, 1e-12))):
            curr_h = new_h
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
    return best_h, best_c5

def slsqp_polish(h_start, n, dx, max_iter=400):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    obj = partial(de_objective, dx=dx)
    con_fun = partial(eq_constraint, target_sum=target_sum)
    con_jac = partial(eq_constraint_jac, target_sum=target_sum)
    res = minimize(
        obj, h_start, method='SLSQP', bounds=bounds,
        constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
        options={'maxiter': max_iter, 'ftol': 1e-16}
    )
    return project_feasible(res.x, n)
