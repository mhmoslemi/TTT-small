    return project_sum(res.x, target)

def slsqp_symmetric(d_init, beta, maxiter=3000):
    """Optimize in symmetric subspace: h[i]+h[n-1-i]=1."""
    m = len(d_init)
    n = 2 * m
    dx = 2.0 / n
    bounds = [(0, 1)] * m
    
    def h_full(d):
        h = np.zeros(n)
        h[:m] = d
        h[m:] = 1.0 - d[::-1]
        return h
    
    def obj_grad(d):
        d = np.clip(d, 0, 1)
        h = h_full(d)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr) * dx
        bc = np.clip(beta * (corr - np.max(corr)), -500, 500)
        e = np.exp(bc)
        se = e.sum()
        val = (np.max(corr) + np.log(se / len(corr)) / beta) * dx
        w = e / se
        gh = np.zeros(n)
        for j in range(len(corr)):
            if w[j] < 1e-13:
                continue
            lag = j - (n - 1)
            gh += w[j] * grad_single_lag(h, lag)
        # grad w.r.t. d: grad_d[i] = gh[i] - gh[n-1-i]
        gd = gh[:m] - gh[m:][::-1]
        return val, gd * dx
    
    res = minimize(obj_grad, d_init, jac=True, method='SLSQP',
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return np.clip(res.x, 0, 1)

def subgradient_direct(h, target, n_active, max_iter, t0, budget):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(best)
    step = 0.03
    no_improve = 0
    
    for it in range(max_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        top = np.argsort(corr)[::-1][:n_active]
        g = np.zeros(n)
        for idx in top:
            lag = int(idx) - (n - 1)
            g += grad_single_lag(best, lag)
        g *= dx / n_active
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for s in [step, step*0.4, step*0.1, step*0.04, step*0.01, step*0.003, step*0.001]:
            ht = project_sum(best - s * g, target)
            c5t = compute_c5(ht)
            if c5t < best_c5:
                best = ht
                best_c5 = c5t
                step = min(s * 1.8, 0.08)
                improved = True
                break
        if not improved:
            step *= 0.4
            no_improve += 1
            if no_improve > 500 and it % 250 == 249:
                best = project_sum(best + np.random.randn(n) * 0.008, target)
                best_c5 = compute_c5(best)
                no_improve = 0
                step = 0.03
            if no_improve > 3000:
                break
        else:
            no_improve = 0
    return best, best_c5
