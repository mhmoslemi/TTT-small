    dx = 2.0 / n

def corr_vector(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:

    corr = corr_vector(h)

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_vector(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_w(h, n, num_lags, threshold_frac):
    corr = corr_vector(h)
    mc = np.max(corr)
    thr = mc - threshold_frac * (mc + 1e-10)
    idx = np.where(corr >= thr)[0]
    if len(idx) > num_lags:
        idx = idx[np.argsort(corr[idx])[::-1][:num_lags]]
    g = np.zeros(n)
    for i in idx:
        lag = int(i) - (n - 1)
        if lag >= 0:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
        else:
            la = -lag
            g[la:] += (1.0 - h[:n - la])
            g[:n - la] -= h[la:]
    return g / max(len(idx), 1), corr, mc

def smooth_max(x, beta):
    m = np.max(x)
    clipped = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(clipped)) / len(x)) / beta

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def sml_obj(h_in, beta, dx):
    return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), beta) * dx

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def nesterov_v2(h0, n, target, n_iter, step0, t0, budget, num_lags=120, thr_frac=0.008):
    h = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    step = step0
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget - 3:
            break
        g, _, _ = subgrad_w(y, n, num_lags, thr_frac)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g /= gn
        h_new = project_h(y - step * g, target)
        mu = (1.0 - np.sqrt(1.0 + 4.0 / ((it + 1) ** 2))) / 2.0
        y = h_new + mu * (h_new - h_prev)
        y = np.clip(y, 0, 1)
        h_prev, h = h, h_new
        c = compute_c5(h)
        if c < best_c5 - 1e-14:
            best_h, best_c5, stall = h.copy(), c, 0
        else:
            stall += 1
            if stall % 80 == 0:
                step *= 0.75
                if stall % 400 == 0:
                    y = h.copy()
    return best_h, best_c5

def lag_peeling(h0, n, target, n_iter, t0, budget):
    h = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    for it in range(n_iter):
        if time.time() - t0 > budget - 3:
            break
        corr = corr_vector(h)
        mc = np.max(corr)
        # Focus on top-3 worst lags
        top3 = np.argsort(corr)[::-1][:3]
        g = np.zeros(n)
        for ii in top3:
            lag = int(ii) - (n - 1)
            if lag >= 0:
                g[:n - lag] += (1.0 - h[lag:])
                g[lag:] -= h[:n - lag]
            else:
                la = -lag
                g[la:] += (1.0 - h[:n - la])
                g[:n - la] -= h[la:]
        g /= 3.0
        for alpha in (0.03, 0.015, 0.008, 0.004, 0.002):
            trial = project_h(h - alpha * g, target)
            c_trial = compute_c5(trial)
            if c_trial < best_c5 - 1e-14:
                h, best_h, best_c5 = trial, trial.copy(), c_trial
                break
    return best_h, best_c5

def anneal_moves(h0, n, target, n_moves, t0, budget, tol=0.0005):
    h = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    rng = np.random.RandomState(777)
    for it in range(n_moves):
        if time.time() - t0 > budget - 3:
            break
        # Perturb a small random block
        blen = rng.randint(3, 15)
        start = rng.randint(0, max(1, n - blen))
        delta = (rng.rand(blen) - 0.5) * 0.1
        trial = h.copy()
        trial[start:start + blen] += delta
        trial = project_h(trial, target)
        c = compute_c5(trial)
        # Accept if better or within tolerance
        if c < best_c5 + tol:
            h = trial
            if c < best_c5 - 1e-14:
                best_h, best_c5 = h.copy(), c
    return best_h, best_c5

def coord_descent_fine(cur, target, n, sweeps, t0, budget):
    best_c5 = compute_c5(cur)
    deltas = (0.008, -0.008, 0.004, -0.004, 0.002, -0.002,
              0.001, -0.001, 0.0005, -0.0005, 0.015, -0.015)
    for sw in range(sweeps):
        if time.time() - t0 > budget - 2:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > budget - 1:
                break
            for d in deltas:
                if cur[j] + d < -0.01 or cur[j] + d > 1.01:
                    continue
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved and sw > 4:
            break
    return cur, best_c5

def pair_descent(cur, target, n, n_rounds, t0, budget):
    best_c5 = compute_c5(cur)
    rng = np.random.RandomState(9999)
    for _ in range(n_rounds):
        if time.time() - t0 > budget - 2:
            break
        for _ in range(n // 3):
            if time.time() - t0 > budget - 1:
                break
            j, k = rng.choice(n, 2, replace=False)
            for d in (0.015, -0.015, 0.008, -0.008, 0.004, -0.004, 0.002, -0.002):
                trial = cur.copy()
                trial[j] += d
                trial[k] -= d
                if trial[j] < 0 or trial[j] > 1 or trial[k] < 0 or trial[k] > 1:
                    continue
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    cur, best_c5 = trial, c
                    break
    return cur, best_c5

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, len(h))
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(320, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    n = len(h_parent)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    # Phase 1: SLSQP smooth max
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        for beta in [800, 2000, 5000]:
            rem = budget_s - (time.time() - t0)
            if rem < 15:
                break
            cur = do_slsqp(best_h, n, target, beta, bounds, 6000, 1e-15)
            c = compute_c5(cur)
            if c < best_c5:
                best_h, best_c5 = cur.copy(), c

    # Phase 2: Nesterov with wide active set
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        cur, c = nesterov_v2(best_h, n, target, 12000, 0.05, t0, budget_s, num_lags=150, thr_frac=0.01)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 3: Lag peeling (new approach)
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        cur, c = lag_peeling(best_h, n, target, 3000, t0, budget_s)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 4: Simulated annealing moves
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        cur, c = anneal_moves(best_h, n, target, 3000, t0, budget_s, tol=0.0004)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 5: Nesterov again (shorter, tighter)
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        cur, c = nesterov_v2(best_h, n, target, 8000, 0.02, t0, budget_s, num_lags=200, thr_frac=0.004)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 6: Fine coordinate descent (many sweeps)
    rem = budget_s - (time.time() - t0)
    if rem > 80:
        cur, c = coord_descent_fine(best_h, target, n, 80, t0, budget_s)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 7: Pair descent
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        cur, c = pair_descent(best_h, target, n, 300, t0, budget_s)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 8: Another round of lag peeling + anneal
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        cur, c = lag_peeling(best_h, n, target, 2000, t0, budget_s)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        cur, c = anneal_moves(best_h, n, target, 2000, t0, budget_s, tol=0.0002)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 9: Try higher resolution
    for n_high in [512, 640]:
        rem = budget_s - (time.time() - t0)
        if rem < 120:
            break
        th = n_high / 2.0
        bh = Bounds(np.zeros(n_high), np.ones(n_high))
        ch = project_h(resample(best_h, n_high), th)
        cc = compute_c5(ch)
        if cc < best_c5:
            best_h, best_c5 = ch.copy(), cc
        rem = budget_s - (time.time() - t0)
        if rem > 50:
            ch = do_slsqp(ch, n_high, th, 3000, bh, 4000, 1e-15)
            cc = compute_c5(ch)
            if cc < best_c5:
                best_h, best_c5 = ch.copy(), cc
        rem = budget_s - (time.time() - t0)
        if rem > 40:
            ch, cc = nesterov_v2(ch, n_high, th, 4000, 0.03, t0, budget_s, num_lags=120, thr_frac=0.008)
            if cc < best_c5:
                best_h, best_c5 = ch.copy(), cc
        rem = budget_s - (time.time() - t0)
        if rem > 30:
            ch, cc = coord_descent_fine(ch, th, n_high, 30, t0, budget_s)
            if cc < best_c5:
                best_h, best_c5 = ch.copy(), cc

    # Final polish at best resolution
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        nf = len(best_h)
        tf = nf / 2.0
        bf = Bounds(np.zeros(nf), np.ones(nf))
        best_h = do_slsqp(best_h, nf, tf, 8000, bf, 6000, 1e-15)
        best_c5 = compute_c5(best_h)

    rem = budget_s - (time.time() - t0)
    if rem > 15:
        best_h, best_c5 = coord_descent_fine(best_h, len(best_h) / 2.0, len(best_h), 60, t0, budget_s)

    rem = budget_s - (time.time() - t0)
    if rem > 10:
        best_h, best_c5 = pair_descent(best_h, len(best_h) / 2.0, len(best_h), 150, t0, budget_s)

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
