                trial = project_h(trial, target)

                if c < best_c5 - 1e-15:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved:
            break
    return cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(320, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(h_parent)

    resolutions = sorted(set([best_n] + [r for r in [256, 320, 384, 448, 512, 576] if 200 <= r <= 600]))

    for n_pts in resolutions:
        rem = budget_s - (time.time() - t0)
        if rem < 70:
            break
        target = n_pts / 2.0
        bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
        cur = project_h(resample(best_h, n_pts), target)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5 - 1e-14:
            best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # Phase 1: SLSQP moderate beta
        rem = budget_s - (time.time() - t0)
        if rem > 50:
            cur = do_slsqp(cur, n_pts, target, 300, bounds, 2000, 1e-13)
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # Phase 2: Active-set subgradient (tight)
        rem = budget_s - (time.time() - t0)
        if rem > 40:
            cur, cur_c5 = active_subgrad(cur, n_pts, target, 0.004, 2500, t0, budget_s)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # Phase 3: SLSQP high beta
        rem = budget_s - (time.time() - t0)
        if rem > 35:
            cur = do_slsqp(cur, n_pts, target, 3000, bounds, 3000, 1e-14)
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # Phase 4: Active-set again (very tight)
        rem = budget_s - (time.time() - t0)
        if rem > 25:
            cur, cur_c5 = active_subgrad(cur, n_pts, target, 0.001, 2000, t0, budget_s)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # Phase 5: Targeted pair moves
        rem = budget_s - (time.time() - t0)
        if rem > 20:
            cur, cur_c5 = targeted_pair_moves(cur, n_pts, target, 20, t0, budget_s)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # Phase 6: Final SLSQP polish at very high beta
        rem = budget_s - (time.time() - t0)
        if rem > 20:
            cur = do_slsqp(cur, n_pts, target, 8000, bounds, 4000, 1e-15)
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

    # Final polish at best resolution
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        nf = best_n
        tf = nf / 2.0
        bf = Bounds(np.zeros(nf), np.ones(nf))
        best_h = do_slsqp(best_h, nf, tf, 10000, bf, 10000, 1e-15)
        best_c5 = compute_c5(best_h)

    rem = budget_s - (time.time() - t0)
    if rem > 25:
        best_h, best_c5 = active_subgrad(best_h, best_n, best_n / 2.0, 0.001, 3000, t0, budget_s)

    rem = budget_s - (time.time() - t0)
    if rem > 15:
        best_h, best_c5 = targeted_pair_moves(best_h, best_n, best_n / 2.0, 30, t0, budget_s)

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
