import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        logsum = mx + np.log(np.sum(np.exp(shifted - mx)))
    else:
        logsum = np.log(np.sum(np.exp(shifted)))
    return float(m + logsum / alpha)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def random_pair_search(h, n, deadline, n_iter_max=500000):
    """Fast random pair move local search."""
    best_c = compute_c5(h, n)
    h = h.copy()
    rng = np.random.RandomState(12345)
    
    deltas = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    d_idx = 0
    
    iters_per_delta = n_iter_max // len(deltas)
    
    for d_idx, d in enumerate(deltas):
        if time.time() > deadline:
            break
        for _ in range(iters_per_delta):
            if time.time() > deadline:
                return h, best_c
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            # Try h[i] += d, h[j] -= d
            if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                h[i] += d
                h[j] -= d
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                else:
                    h[i] -= d
                    h[j] += d
            # Try h[i] -= d, h[j] += d
            if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                h[i] -= d
                h[j] += d
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                else:
                    h[i] += d
                    h[j] -= d
    
    return h, best_c

Alternative: Use the subgradient information to guide the search.

**Subgradient-guided local search**:
1. Compute the active set (shifts k where c[k] is near max)
2. For each active shift k, identify which pairs of points (i, j) would most reduce c[k]
3. Try those specific pairs

This is much more targeted than random search.

For a given shift k:
c[k] = sum_i h[i](1-h[i+k])

To reduce c[k], we want to decrease h[i](1-h[i+k]) for some i. This can be done by:
- Decreasing h[i] (and increasing some h[j])
- Increasing h[i+k] (and decreasing some h[j])

The most effective move is to find i that maximizes h[i](1-h[i+k]) and j that maximizes h[j](1-h[j+k]) in a complementary way.

Actually, let me think about this differently. The subgradient tells us the direction of steepest ascent of the max. So we want to move in the negative subgradient direction.

Let me combine:
1. Subgradient computation to identify "bad" points (high contribution to active correlations)
2. Targeted moves to fix those points

Here's another idea: **Coordinate descent with compensation**

For each point i:
- Determine the optimal value of h[i] given all other points fixed
- The optimal h[i] minimizes the max correlation while maintaining the sum constraint

This is a 1D optimization problem for each coordinate (with the compensation spread over other coordinates). But it's complex because moving one coordinate affects all correlations.

Let me try a simpler but effective approach:

**Momentum-based local search**:
- Keep a "velocity" vector
- At each step, compute the gradient of the smooth approximation
- Update: h = project(h - lr * grad + momentum * velocity)
- This is essentially gradient descent with momentum on the smooth approximation

This should be fast and effective. Let me implement it:

```python
def gradient_descent_momentum(h, n, deadline, lr_init=0.01, momentum=0.9, alpha=50000):
    h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    best_c = compute_c5(h, n)
    best_h = h.copy()
    box = time.time()
    
    while time.time() < deadline:
        # Compute gradient of smooth objective
        dx = 2.0 / n
        c = np.correlate(h, 1-h, mode='full') * dx
        m = c.max()
        active = np.where(c > m - 0.01)[0]  # near-active set
        
        # Subgradient of max over active set
        g = np.zeros(n)
        for k_idx in active:
            k = k_idx - (n-1)
            i_vals = np.arange(n)
            ipk = i_vals + k
            imk = i_vals - k
            vp = (ipk >= 0) & (ipk < n)
            vm = (imk >= 0) & (imk < n)
            contrib = np.zeros(n)
            contrib[vp] = (1 - h[ipk[vp]])
            contrib[vm] -= h[imk[vm]]
            g += contrib
        g *= dx / len(active)
        
        # Momentum update
        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            # Maybe reduce learning rate slightly
            lr *= 0.999
        else:
            h = h_new  # still move (for exploration)
            lr *= 0.99
        
        # Periodically reset momentum
        if int((time.time() - box) * 100) % 10 == 0:
            v *= 0.5
    
    return best_h, best_c
```

Wait, this has an issue. The subgradient I'm computing is the gradient of the sum over active c[k], not the gradient of the smooth objective. Let me use the actual smooth objective gradient.

The smooth objective is: f(h) = m + (1/alpha) * log(sum(exp(alpha*(c[k] - m))))

The gradient is:
∇f(h) = (1/sum(exp)) * sum_k exp(alpha*(c[k]-m)) * ∇c[k]

This is a weighted average of the gradients of individual c[k], weighted by their softmax probabilities.

Let me implement this properly:

```python
def smooth_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    
    # Weighted sum of gradients of c[k]
    g = np.zeros(n)
    i_vals = np.arange(n)
    for k_idx in range(len(c)):
        if weights[k_idx] < 1e-10:
            continue
        k = k_idx - (n-1)
        ipk = i_vals + k
        imk = i_vals - k
        vp = (ipk >= 0) & (ipk < n)
        vm = (imk >= 0) & (imk < n)
        contrib = np.zeros(n)
        contrib[vp] = (1 - h[ipk[vp]])
        contrib[vm] -= h[imk[vm]]
        g += weights[k_idx] * dx * contrib
    
    return g
```

This is O(n * number_of_significant_k), which for n=200 and maybe 20-50 significant k values, is about 10,000-100,000 operations per gradient evaluation. Very fast.

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = np.max(sh)
    if mx > 500:
        lse = mx + np.log(np.sum(np.exp(sh - mx)))
    else:
        lse = np.log(np.sum(np.exp(sh)))
    return float(m + lse / alpha)

                       bounds=bounds, constraints=cons,

        contrib[vm] -= h[imk[vm]]

    # Phase 3: SLSQP again from improved point (10% time)
    t3 = t0 + (deadline - t0) * 0.50
    if time.time() < t3 - 2:
        ht = slsqp_optimize(best_h, n, 150000, maxiter=3000, ftol=1e-16)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c; best_h = ht

    # Phase 4: Systematic pair search (33% time)
    t4 = t0 + (deadline - t0) * 0.83
    if n <= 250:
        ht, c = pair_search_full(best_h, n, t4,
                                  [0.01, 0.005, 0.003, 0.002, 0.001, 0.0005], max_passes=100)
        if c < best_c:
            best_c = c; best_h = ht
    else:
        ht, c = random_pair_search(best_h, n, t4, rng, n_iters=500000)
        if c < best_c:
            best_c = c; best_h = ht

    # Phase 5: Final SLSQP (5% time)
    t5 = t0 + (deadline - t0) * 0.92
    if time.time() < t5 - 2:
        ht = slsqp_optimize(best_h, n, 500000, maxiter=4000, ftol=1e-17)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c; best_h = ht

    # Phase 6: Final fine pair search (remaining)
    if time.time() < deadline - 2:
        if n <= 250:
            ht, c = pair_search_full(best_h, n, deadline - 1,
                                      [0.002, 0.001, 0.0005, 0.0003, 0.0002], max_passes=80)
        else:
            ht, c = random_pair_search(best_h, n, deadline - 1, rng, n_iters=200000)
        if c < best_c:
            best_c = c; best_h = ht

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
