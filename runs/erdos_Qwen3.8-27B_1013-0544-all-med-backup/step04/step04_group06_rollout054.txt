def projected_nesterov(h, target, t0, budget, time_limit):

    best_c5 = compute_c5(best)

    step = 0.01

        y = project_sum(x + momentum * v, target)
        
        # Compute gradient (subgradient of max)
        corr = np.correlate(y, 1.0 - y, mode="full")
