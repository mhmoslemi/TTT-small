def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

def compute_subgrad(h, lags):

    grad = np.zeros(n)

**Alternative: Use the structure of the problem**

The correlation c[k] = sum_i h[i] * (1-h[i+k]) can be written as:
c[k] = sum_i h[i] - sum_i h[i]*h[i+k]

For the max to be minimized, we want all c[k] to be as equal as possible (by the "minimize the maximum" principle, the optimum occurs when the top few lags are balanced).

So at the optimum, the top-M lags should all have approximately the same value. This gives us M-1 equations (c[k1] = c[k2] = ... = c[kM]) plus the constraint sum(h) = n/2 and 0 ≤ h ≤ 1.

If we could identify which lags are active at the optimum, we could solve a system of equations. But identifying the active set is the hard part.

**Heuristic active set approach:**
1. Start with an initial h
2. Find the top-M lags
3. Solve the "equilibration" problem: find h such that c[k1] = c[k2] = ... = c[kM] (all equal to some t), min t, subject to constraints
4. This is a nonlinear system (M equations in n unknowns + t). We can use Newton's method.
5. After solving, check if any inactive lag now exceeds t. If so, add it to the active set.
6. Repeat until stable.

This is essentially a **constraint generation** or **cutting plane** approach. The "constraints" are c[k] ≤ t for all k, and we add them one at a time (or in batches).

    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    for _ in range(max_newton):
        if time_fn() < 5:
            break
        # Compute current values and gradients for active lags
        F = np.zeros(len(active_lags))
        J = np.zeros((len(active_lags), n))
        for idx, lag in enumerate(active_lags):
            # F[idx] = c[lag] * dx
            F[idx] = compute_lag_value(x, lag) * dx
            J[idx] = compute_lag_grad(x, lag) * dx
        # All-residual: F - t (where t = mean(F))
        t = np.mean(F)
        residual = F - t
        # We want to find dx such that J @ dx ≈ -residual
        # But we also need the constraint sum(x) = target
        # Use least squares with the constraint
        # Minimize ||J @ delta + residual||^2 subject to sum(delta) = 0
        # Lagrangian: ||J @ delta + r||^2 + lambda * sum(delta)
        # KKT: 2 J^T (J @ delta + r) + lambda * 1 = 0
        #      sum(delta) = 0
        JtJ = J.T @ J
        Jtr = J.T @ residual
        # Add constraint via block solver
        # [2 J^T J, 1; 1^T, 0] [delta; lambda] = [-2 J^T r; 0]
        A = np.zeros((n+1, n+1))
        A[:n, :n] = 2 * JtJ
        A[:n, n] = 1.0
        A[n, :n] = 1.0
        b = np.zeros(n+1)
        b[:n] = -2 * Jtr
        try:
            sol = np.linalg.solve(A + 1e-10 * np.eye(n+1), b)
            delta = sol[:n]

        for alpha in [0.5, 0.25, 0.1, 0.05, 0.01]:

            c5_new = compute_c5(x_new)
            c5_old = compute_c5(x)

                break

            break

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgrad(h, lags):
    n = len(h)
    grad = np.zeros(n)
    for lag in lags:
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:-lag])
            grad[:n - lag] -= h[lag:]
        else:
            m = -lag
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def compute_weighted_subgrad(h, top_idx, corr, n):
    grad = np.zeros(n)
    max_c = corr[top_idx[0]]
    for i in top_idx:
        c_i = corr[i]
        w = np.exp(-(max_c - c_i) / 0.01)
        lag = int(i) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            grad[:n - m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:n - m]
    return grad

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -400, 400)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_obj(h_in, dx, beta):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * dx

def slsqp_con(h_in, target):
    return h_in.sum() - target

def slsqp_jac(h_in, n):
    return np.ones(n)

def slsqp_run(h, target, dx, betas, max_iter, time_fn):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    best = h.copy()
    best_c5 = compute_c5(best)
    for beta in betas:
        if time_fn() < 12:
            break
        try:
            res = minimize(slsqp_obj, best, args=(dx, beta), method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_jac, 'args': (target,)}],
                          options={'maxiter': max_iter, 'ftol': 1e-20})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def subgrad_run(h, target, time_fn, max_iters, m_start, m_end,
                steps, momentum, perturb_every, base_pert):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    no_improve = 0
    for it in range(max_iters):
        if time_fn() < 3:
            break
        frac = it / max(1, max_iters)
        m_act = int(m_start + (m_end - m_start) * frac)
        corr = np.correlate(best, 1.0 - best, mode="full")
        top = np.argsort(corr)[::-1][:m_act]
        g = compute_weighted_subgrad(best, top, corr, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        vel = momentum * vel + (1.0 - momentum) * (g / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        d = -vel / vn
        found = False
        for step in steps:
            ht = project_h(best + step * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best = ht
                best_c5 = c5t
                no_improve = 0
                found = True
                break
        if not found:
            no_improve += 1
            if no_improve >= perturb_every and it % perturb_every == perturb_every - 1:
                scale = base_pert * (0.5 + 0.5 * np.sqrt(frac))
                p = np.random.randn(n) * scale
                ht = project_h(best + p, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    vel *= 0.4
                    no_improve = 0
            if no_improve > perturb_every * 25:
                break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)

    best_c5 = compute_c5(best_h)
    n0 = len(best_h)
    target0 = n0 / 2.0
    dx0 = 2.0 / n0

    # Phase 1: SLSQP with progressive beta
    best_h, best_c5 = slsqp_run(best_h, target0, dx0,
                                 [30, 80, 200, 500, 1200, 3000, 8000], 8000, rem)

    # Phase 2: Subgradient - growing active set, moderate momentum
    best_h, best_c5 = subgrad_run(best_h, target0, rem,
                                   max_iters=45000, m_start=4, m_end=80,
                                   steps=[0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                   momentum=0.87, perturb_every=200, base_pert=0.009)

    # Phase 3: Restart and re-optimize (escape local minima)
    if rem() > 50:
        for _ in range(5):
            if rem() < 25:
                break
            h_r = project_h(best_h + np.random.randn(n0) * 0.004, target0)
            h_r, c5_r = slsqp_run(h_r, target0, dx0, [400, 1500, 6000], 4000, rem)
            if c5_r < best_c5:
                best_h = h_r
                best_c5 = c5_r
                best_h, best_c5 = subgrad_run(best_h, target0, rem,
                                               max_iters=8000, m_start=5, m_end=40,
                                               steps=[0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002],
                                               momentum=0.9, perturb_every=200, base_pert=0.005)

    # Phase 4: Fine subgradient with high momentum and many lags
    best_h, best_c5 = subgrad_run(best_h, target0, rem,
                                   max_iters=35000, m_start=10, m_end=120,
                                   steps=[0.006, 0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                                   momentum=0.93, perturb_every=350, base_pert=0.004)

    # Phase 5: Upscale to 2x resolution
    if rem() > 90:
        n2 = n0 * 2
        target2 = n2 / 2.0
        dx2 = 2.0 / n2
        x_old = np.linspace(0, 2, n0)
        x_new = np.linspace(0, 2, n2)
        h2 = project_h(np.interp(x_new, x_old, best_h), target2)
        c5_2 = compute_c5(h2)
        if rem() > 45:
            h2, c5_2 = slsqp_run(h2, target2, dx2, [200, 800, 3000, 10000], 6000, rem)
        if rem() > 25:
            h2, c5_2 = subgrad_run(h2, target2, rem,
                                     max_iters=25000, m_start=5, m_end=70,
                                     steps=[0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                     momentum=0.91, perturb_every=250, base_pert=0.004)
        if c5_2 < best_c5:
            best_h = h2
            best_c5 = c5_2

    # Phase 6: Final ultra-fine polish
    n_f = len(best_h)
    target_f = n_f / 2.0
    best_h, best_c5 = subgrad_run(best_h, target_f, rem,
                                   max_iters=12000, m_start=15, m_end=150,
                                   steps=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001],
                                   momentum=0.95, perturb_every=500, base_pert=0.002)

    return best_h, best_c5, len(best_h)
