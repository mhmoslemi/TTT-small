from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def snap_binary(h, target):
    sorted_idx = np.argsort(-h)
    z = np.zeros(len(h))
    n_ones = int(target)
    z[sorted_idx[:n_ones]] = 1.0
    return z

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(np.clip(beta * (x - m), -500, 500))) / len(x)) / beta

def sml_obj(h_in, beta, dx):
    return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), beta) * dx

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol, t0, budget):
    rem = budget - (time.time() - t0)
    if rem < 15:
        return h0
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def sa_loop(h, n, target, rng, T_start, T_end, cooling, t_end):
    """Core SA loop on binary solution."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    T = T_start
    
    while time.time() < t_end and T > T_end:
        a = rng.randint(n)
        b = rng.randint(n)
        if a == b or h[a] == h[b]:

        h[a], h[b] = h[b], h[a]
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5

                best_c5 = new_c5
        else:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def snap_binary(h, target):
    sorted_idx = np.argsort(-h)
    z = np.zeros(len(h))
    n_ones = int(target)
    z[sorted_idx[:n_ones]] = 1.0

def resample(h, n_new):

    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def sum_eq(x, t):

    return m + np.log(np.sum(np.exp(np.clip(beta * (x - m), -500, 500))) / len(x)) / beta
