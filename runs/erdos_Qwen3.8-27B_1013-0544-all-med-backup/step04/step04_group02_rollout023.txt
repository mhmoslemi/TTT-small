def _minimize_1d_convex(A, B, dx, lam):
    c = B + lam / dx
    # f(t) <= v iff for all l: c[l]*t <= v/dx - A[l]
    # If c[l] > 0: t <= (v/dx - A[l]) / c[l]
    # If c[l] < 0: t >= (v/dx - A[l]) / c[l]
    # If c[l] = 0: need A[l] <= v/dx

    neg = c < -1e-15
    zero = ~pos & ~neg

        vd = v / dx
        if np.any(A[zero] > vd):
            return False
        t_lo = 0.0
        t_hi = 1.0
        if np.any(neg):
            t_lo = max(t_lo, float(np.max((vd - A[neg]) / c[neg])))
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((vd - A[pos]) / c[pos])))

    pos = c > 1e-14
    neg = c < -1e-14

import numpy as np
import time
from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_sm(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_f(h, ts):
    return h.sum() - ts

def con_j(h, ts):
    return np.ones_like(h)

def _min1d(A, B, dx, lam):
    """Minimize f(t)=dx*max_l(A_l+B_l*t)+lam*t over t in [0,1] via bisection on value."""
    c = B + lam / dx
    pos = c > 1e-14
    neg = c < -1e-14
    zero = ~(pos | neg)
    def feas(v):
        vd = v / dx
        if np.any(zero):
            if np.any(A[zero] > vd + 1e-15):
                return False
        tl = 0.0
        th = 1.0
        if np.any(neg):
            tl = max(tl, float(np.max((vd - A[neg]) / c[neg])))
        if np.any(pos):
            th = min(th, float(np.min((vd - A[pos]) / c[pos])))
        return tl <= th + 1e-12
    f0 = float(np.max(A)) * dx
    f1 = float(np.max(A + B)) * dx + lam
    vlo, vhi = 0.0, max(f0, f1) + 2.0 * abs(lam) / dx + 1.0
    for _ in range(35):
        vm = (vlo + vhi) * 0.5
        if feas(vm):
            vhi = vm
        else:
            vlo = vm
    vd = vhi / dx
    tl, th = 0.0, 1.0
    if np.any(neg):
        tl = max(tl, float(np.max((vd - A[neg]) / c[neg])))
    if np.any(pos):
        th = min(th, float(np.min((vd - A[pos]) / c[pos])))
    return float(np.clip((tl + th) * 0.5, 0.0, 1.0))

def coord_desc(h, deadline, lam0=0.0, seed=0):
    n = len(h)
    dx = 2.0 / n
    ts = n / 2.0
    lam = lam0
    c5 = compute_c5(h)
    lags_arr = np.arange(-(n - 1), n)
    best_c5 = c5
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for outer in range(20000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        for i in range(n):
            alpha = (i + lags_arr) >= 0
            alpha = alpha & (i + lags_arr < n)
            beta_m = (i - lags_arr) >= 0
            beta_m = beta_m & (i - lags_arr < n)
            ha = np.where(alpha, h[i + lags_arr[alpha]], 0.0)
            hb = np.where(beta_m, h[i - lags_arr[beta_m]], 0.0)
            A = corr - alpha.astype(float) * h[i] * (1.0 - ha) - beta_m.astype(float) * hb * (1.0 - h[i])
            B = alpha.astype(float) * (1.0 - ha) - beta_m.astype(float) * hb
            new_val = _min1d(A, B, dx, lam)
            if abs(new_val - h[i]) > 1e-15:
                corr += (new_val - h[i]) * B
                h[i] = new_val
        h = project_feasible(h, ts)
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h.copy()
        s = h.sum()
        lam += 0.08 * (s - ts)
        lam = float(np.clip(lam, -5.0, 5.0))
        # Periodic perturbation to escape local optima
        if outer % 500 == 499 and outer > 0:
            h2 = project_feasible(h + rng.randn(n) * 0.01, ts)
            if compute_c5(h2) < best_c5 + 0.001:
                h = h2
    return best_h, best_c5

def slsqq(h, n, deadline, seed):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j, 'args': (ts,)}]
    c5 = compute_c5(h)
    for beta in [200, 1000, 5000, 20000, 100000]:
        if time.time() > deadline:
            break
        for p in [0.0, 0.005, 0.015]:
            if time.time() > deadline:
                break
            x0 = h.copy()
            if p > 0:
                x0 = project_feasible(x0 + np.random.randn(n) * p, ts)
            try:
                res = minimize(obj_sm, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-15})
                h2 = project_feasible(res.x, ts)
                c2 = compute_c5(h2)
                if c2 < c5:
                    h, c5 = h2, c2
            except Exception:
                pass
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h, best_c5, best_n = None, float('inf'), 0

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    if parent is None:
        parent = np.full(350, 0.5)
    pn = len(parent)

    # Phase 1: Coordinate descent at parent resolution (50%)
    d1 = t0 + budget_s * 0.50
    h1, c1 = coord_desc(parent.copy(), d1, seed=seed)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1.copy(), c1, pn

    # Reversed parent
    if time.time() < d1:
        hr = parent[::-1].copy()
        hr = project_feasible(hr, pn / 2.0)
        h2, c2 = coord_desc(hr, d1, seed=seed + 7)
        if c2 < best_c5:
            best_h, best_c5, best_n = h2.copy(), c2, pn

    # Phase 2: Other resolutions (20%)
    d2 = t0 + budget_s * 0.70
    for ntrial in [250, 400, 500, 300]:
        if time.time() > d2:
            break
        if ntrial == pn:
            continue
        x_old = np.linspace(0, 2, pn, endpoint=False)
        x_new = np.linspace(0, 2, ntrial, endpoint=False)
        h0 = np.interp(x_new, x_old, parent)
        h0 = project_feasible(h0, ntrial / 2.0)
        hh, cc = coord_desc(h0, d2, seed=seed + ntrial)
        if cc < best_c5:
            best_h, best_c5, best_n = hh.copy(), cc, ntrial

    # Phase 3: SLSQP polish (15%)
    d3 = t0 + budget_s * 0.85
    if best_h is not None and time.time() < d3:
        hp, cp = slsqq(best_h.copy(), best_n, d3, seed + 100)
        if cp < best_c5:
            best_h, best_c5 = hp, cp

    # Phase 4: Final coordinate descent polish (13%)
    d4 = t0 + budget_s * 0.98
    if best_h is not None and time.time() < d4:
        best_h, best_c5 = coord_desc(best_h, d4, seed=seed + 999)

    if best_h is None:
        best_h = np.full(100, 0.5)
        best_n = 100
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
