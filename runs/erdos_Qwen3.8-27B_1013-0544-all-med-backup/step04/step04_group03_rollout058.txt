```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time
import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def binary_snap(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def smooth_max(x, beta):
    m = np.max(x)
    x_clipped = np.clip(x - m, -500.0/beta, 500.0/beta)
    return m + np.log(np.sum(np.exp(beta * x_clipped)) / len(x)) / beta

def slsqp_obj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=8000, ftol=1e-16):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(r.x, 0, 1), target)
    except Exception:
        return h0

def binary_swap_optimize(h_bin, n, t0, budget, n_iter=500000, rng=None):
    if rng is None:
        rng = np.random.default_rng(12345)
    h = h_bin.copy()
    cur_lags = compute_lags(h)
    cur_max = float(np.max(cur_lags))
    best_h = h.copy()
    best_max = cur_max
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_iter):
        if budget - (time.time() - t0) < 5:
            break
        # Pick random pair
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        
        # Compute effect of swap: h[a] 1->0, h[b] 0->1
        # This changes correlations at all lags
        # For efficiency, compute delta for the max lag
        # corr[k] changes: removing contribution at position a (was 1, now 0)
        # and adding contribution at position b (was 0, now 1)
        
        # Quick evaluation: just swap and recompute
        h[a] = 0.0
        h[b] = 1.0
        new_lags = compute_lags(h)
        new_max = float(np.max(new_lags))
        
        if new_max < cur_max - 1e-15:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
            # Accept
        elif new_max <= cur_max + 1e-15:
            # Accept (equal)
            pass
        else:
            # Reject with some probability for exploration
            frac = it / max(n_iter, 1)
            temp = 0.15 * (0.005) ** frac
            if rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-15)):
                cur_max = new_max

        h[b] = 1.0  # b was 0, now 1

        zeros = np.where(h == 0)[0]

def targeted_subgrad(h, target, n, t0, budget, n_iter=30000, step_init=0.015, tol=0.005):
    cur = h.copy()
    best_local = compute_c5(cur)
    mom = np.zeros(n)
    step = step_init
    stall = 0
    
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        lags = compute_lags(cur)
        cur_max = float(np.max(lags))
        # Active lags: those within tol of max
        active = np.where(lags >= cur_max - tol)[0]
        n_active = len(active)
        
        # Compute gradient as average over active lags
        grad = np.zeros(n)
        for idx in active:
            lag = int(idx) - (n - 1)
            if lag >= 0:
                if lag < n:
                    grad[:n - lag] += (1.0 - cur[lag:])
                    grad[lag:] -= cur[:n - lag]
            else:
                la = -lag
                grad[la:] += (1.0 - cur[:n - la])
                grad[:n - la] -= cur[la:]
        grad /= n_active
        
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        
        # Nesterov momentum
        mom = 0.9 * mom - step * grad
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        
        if c_try < best_local - 1e-15:
            cur = trial
            best_local = c_try
            stall = 0
            step = min(step * 1.03, 0.08)
        else:
            stall += 1
            step *= 0.75
            mom *= 0.3
            if stall > 200:
                # Try a smaller step
                mom = np.zeros(n)
                trial = project_h(cur - 0.005 * grad, target)
                c_try = compute_c5(trial)
                if c_try < best_local:
                    cur = trial
                    best_local = c_try
                    stall = 0
                else:
                    break
    
    return cur, best_local
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def binary_snap(h, n):
    n_ones = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -400, 400)
    return m + np.logaddexp.reduce(x_shift) - np.log(len(x)) + m * 0 + m - m + (m - m) + np.log(np.mean(np.exp(x_shift))) / beta

def smooth_max_v2(x, beta):
    m = np.max(x)
    x_shift = np.clip(x - m, -100.0/beta, 100.0/beta)
    return m + (1.0/beta) * np.log(np.mean(np.exp(beta * x_shift)))

def slsqp_obj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max_v2(c, beta) * (2.0 / n)

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=8000, ftol=1e-16):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(np.clip(r.x, 0, 1), target)
    except Exception:
        return h0

def fast_lag_delta(h, a, b, n):
    """Compute delta of correlations when h[a] 1->0 and h[b] 0->1 (binary case)."""
    lags = np.zeros(2*n-1)
    # h[a]: 1->0
    # As h[n+L]=h[a]: L in [a-(n-1), a], n=a-L, delta = -(1-h[a-L])
    L_start = max(0, a - (n - 1))
    L_end = min(n - 1, a)
    for L in range(L_start, L_end + 1):
        idx = int(L)
        lags[idx] -= (1.0 - h[a - L])
    # As (1-h[n]) with n=a: L in [-a, n-1-a], delta = +h[a+L]
    Ls2 = max(-(n - 1), -a)
    Le2 = min(n - 1, n - 1 - a)
    for L in range(Ls2, Le2 + 1):
        lags[L + (n-1) if L < 0 else L] += h[a + L] if 0 <= a + L < n else 0.0
    # h[b]: 0->1
    L_start = max(0, b - (n - 1))
    L_end = min(n - 1, b)
    for L in range(L_start, L_end + 1):
        lags[L] += (1.0 - h[b - L])
    Ls2 = max(-(n - 1), -b)
    Le2 = min(n - 1, n - 1 - b)
    for L in range(Ls2, Le2 + 1):
        lags[L + (n-1) if L < 0 else L] -= h[b + L] if 0 <= b + L < n else 0.0
    return lags

def targeted_subgrad(h, target, n, t0, budget, n_iter=30000, step_init=0.012, tol=0.004):
    cur = h.copy()
    best_local = compute_c5(cur)
    mom = np.zeros(n)
    step = step_init
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        lags = compute_lags(cur)
        cur_max = float(np.max(lags))
        active = np.where(lags >= cur_max - tol)[0]
        n_act = max(len(active), 1)
        grad = np.zeros(n)
        for idx in active:
            lag = int(idx) - (n - 1)
            if lag >= 0:
                if lag < n:
                    grad[:n - lag] += (1.0 - cur[lag:])
                    grad[lag:] -= cur[:n - lag]
            else:
                la = -lag
                grad[la:] += (1.0 - cur[:n - la])
                grad[:n - la] -= cur[la:]
        grad /= n_act
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        mom = 0.9 * mom - step * grad
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_local - 1e-15:
            cur = trial
            best_local = c_try
            stall = 0
            step = min(step * 1.04, 0.06)
        else:
            stall += 1
            step *= 0.72
            mom *= 0.3
            if stall > 150:
                mom = np.zeros(n)
                trial = project_h(cur - 0.004 * grad, target)
                c_try = compute_c5(trial)
                if c_try < best_local:
                    cur = trial
                    best_local = c_try
                    stall = 0
                else:
                    break
    return cur, best_local

def binary_anneal_fast(h_bin, n, t0, budget, n_iter=300000, rng=None):
    if rng is None:
        rng = np.random.default_rng(54321)
    h = h_bin.copy()
    cur_lags = compute_lags(h)
    cur_max = float(np.max(cur_lags))
    best_h = h.copy()
    best_max = cur_max
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    for it in range(n_iter):
        if budget - (time.time() - t0) < 5:
            break
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        new_lags = compute_lags(h)
        new_max = float(np.max(new_lags))
        if new_max < cur_max - 1e-15:
            cur_max = new_max
            cur_lags = new_lags
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        elif new_max <= cur_max + 1e-15:
            cur_lags = new_lags
        else:
            frac = it / max(n_iter, 1)
            temp = 0.12 * (0.003) ** frac
            if rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-15)):
                cur_max = new_max
                cur_lags = new_lags
            else:
                h[a] = 1.0
                h[b] = 0.0
                continue
        # Update ones/zeros arrays
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    return best_h, best_max

def coord_descent_fine(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.025, -0.025, 0.012, -0.012, 0.006, -0.006, 0.003, -0.003, 0.0015, -0.0015]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        rng = np.random.default_rng(sw * 7919 + 13)
        order = rng.permutation(n)
        for j in order:
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved:
            break
    return cur, best_c5

            if remaining < 10:
                break
            cur_hr = do_slsqp(cur_hr, n_hr, target_hr, beta, maxiter=5000)
        remaining = budget_s - (time.time() - t0)
        if remaining > 30:
            hb = binary_snap(cur_hr, n_hr)
            hb, _ = binary_anneal_fast(hb, n_hr, t0, budget_s, n_iter=300000, rng=np.random.default_rng(seed+99))
            cur_hr = hb + rng.standard_normal(n_hr) * 0.02
            cur_hr = project_h(cur_hr, target_hr)
            for beta in [1000, 5000]:
                remaining = budget_s - (time.time() - t0)
                if remaining < 8:
                    break
                cur_hr = do_slsqp(cur_hr, n_hr, target_hr, beta, maxiter=5000)
        c_try = compute_c5(cur_hr)
        if c_try < best_c5:
            best_h = cur_hr.copy()
            best_c5 = c_try
            n = n_hr
            target = target_hr
    # Phase 6: Multi-perturbation re-optimization
    remaining = budget_s - (time.time() - t0)
    for att in range(5):
        if remaining < 20:
            break
        scales = [0.02, 0.01, 0.03, 0.015, 0.025]
        hp = best_h + rng.standard_normal(n) * scales[att]
        hp = project_h(hp, target)
        for beta in [2000, 8000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 8:
                break
            hp = do_slsqp(hp, n, target, beta, maxiter=7000)
        remaining = budget_s - (time.time() - t0)
        if remaining > 10:
            hp, c_sub = targeted_subgrad(hp, target, n, t0, budget_s, n_iter=8000, step_init=0.006, tol=0.002)
        c_try = compute_c5(hp)
        if c_try < best_c5:
            best_h = hp.copy()
            best_c5 = c_try
    # Phase 7: Final coordinate descent polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        best_h, best_c5 = coord_descent_fine(best_h, target, n, 400, t0, budget_s)
    # Phase 8: Ultra-high beta SLSQP
    remaining = budget_s - (time.time() - t0)
    if remaining > 6:
        for beta in [50000, 120000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 5:
                break
            cur = do_slsqp(best_h, n, target, beta, maxiter=12000, ftol=1e-18)
            c_try = compute_c5(cur)
            if c_try < best_c5:
                best_h = cur
                best_c5 = c_try
    # Final tiny coordinate polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 4:
        best_h, best_c5 = coord_descent_fine(best_h, target, n, 200, t0, budget_s)
    return best_h, best_c5, len(best_h)
```
