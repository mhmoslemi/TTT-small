def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = free.sum()
        if nf == 0:
            free = np.ones(len(h), dtype=bool)
            nf = len(h)
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def weighted_subgradient(h, lags, weights):
    n = len(h)
    grad = np.zeros(n)
    for j in range(len(lags)):
        w = weights[j]
        if w < 1e-10:
            continue
        lag = lags[j]
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            grad[:n - m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:n - m]
    return grad
