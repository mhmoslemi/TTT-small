    if time.time() - t0 < budget_s - 60:
        cur, c = subgrad_polyak(best_h, target, 8000, 0.025, t0, budget_s)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 3: SLSQP polish
    if time.time() - t0 < budget_s - 50:
        for k_top in [8, 15]:
            if time.time() - t0 > budget_s - 40:
                break
            cur = slsqp_active(best_h, n, target, 10000, k_top, 5000, 1e-16)
            c = compute_c5(cur)
            if c < best_c5:
                best_h, best_c5 = cur.copy(), c

    # Phase 4: Pair descent (efficient for preserving sum)
    if time.time() - t0 < budget_s - 40:
        cur, c = pair_descent(best_h, target, 5000, t0, budget_s)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 5: Binary + optimise from binary
    if time.time() - t0 < budget_s - 70:
        zb = snap_binary(best_h, target)
        zb, cb = binary_hill_climb(zb, target, 5000, t0, budget_s)
        if cb < best_c5:
            best_h, best_c5 = zb.copy(), cb
        cur = slsqp_active(zb, n, target, 3000, 15, 4000, 1e-16)
        c = compute_c5(cur)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 6: Coordinate descent fine-tuning
    if time.time() - t0 < budget_s - 20:
        cur, c = coord_descent(best_h, target, 30, t0, budget_s)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 7: Perturbation restarts
    for _ in range(8):
        if time.time() - t0 > budget_s - 45:
            break
        mag = 0.008
        h_p = project_h(best_h + np.random.uniform(-mag, mag, n), target)
        cur = slsqp_active(h_p, n, target, 5000, 15, 3000, 1e-16)
        c = compute_c5(cur)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c
        if time.time() - t0 > budget_s - 30:
            break
        cur, c = pair_descent(cur, target, 2000, t0, budget_s)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    # Phase 8: Final SLSQP + coord
    if time.time() - t0 < budget_s - 20:
        cur = slsqp_active(best_h, n, target, 15000, 10, 6000, 1e-16)
        c = compute_c5(cur)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c
    if time.time() - t0 < budget_s - 10:
        cur, c = coord_descent(best_h, target, 40, t0, budget_s)
        if c < best_c5:
            best_h, best_c5 = cur.copy(), c

    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
