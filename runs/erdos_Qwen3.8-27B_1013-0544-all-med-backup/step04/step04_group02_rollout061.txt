import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def soft_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    W = np.exp(shifted)
    W /= W.sum()
    return W

def analytical_gradient(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    W = soft_weights(corr, beta)
    full1 = np.correlate(W, h, mode='full')
    full2 = np.correlate(W[::-1], h, mode='full')
    T1 = full1[n-1:2*n-1][::-1]
    T2 = full2[n-1:2*n-1][::-1]
    return dx * (1.0 - T1 - T2)

def slsqp_optimize(h, n, deadline, seed, beta_start=100, beta_end=50000, n_betas=5):
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def con_fun(h_in):
        return h_in.sum() - ts
    def con_jac(h_in):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    betas = [int(beta_start * (beta_end/beta_start)**(i/(n_betas-1))) for i in range(n_betas)]
    
    for beta in betas:
        if time.time() > deadline:
            break
        
        def obj(h_in, b=beta):
            corr = np.correlate(h_in, 1.0 - h_in, mode='full')
            return smooth_max_value(corr, b) * dx
        
        def grad(h_in, b=beta):
            return analytical_gradient(h_in, dx, b)
        
        try:
            res = minimize(obj, best_h, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-15, 'maxls': 50})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass
    
    return best_h, best_c5

def nesterov_optimize(h, n, deadline, seed, beta=5000):
    np.random.seed(seed)
    ts = n / 2.0
    dx = 2.0 / n
    c5 = compute_c5(h)
    
    x = h.copy()
    v = np.zeros(n)
    gamma = 0.9
    lr = 0.005
    best_h = h.copy()
    best_c5 = c5
    stall = 0
    
    for it in range(50000):
        if time.time() > deadline:
            break
        grad = analytical_gradient(x, dx, beta)
        v = gamma * v - lr * grad
        x_new = project_feasible(x + v, ts)
        t_curr = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0
        x = x_new + (gamma / t_curr) * (x_new - x)
        x = np.clip(x, 0.0, 1.0)
        
        c_new = compute_c5(x_new)
        if c_new < best_c5 - 1e-12:
            best_h = x_new.copy()
            best_c5 = c_new
            stall = 0
            lr = min(lr * 1.01, 0.02)
        else:
            stall += 1
            if stall % 50 == 0:
                lr *= 0.9
                v *= 0.7
                if lr < 1e-6:
                    break
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    time_per = (d3 - time.time()) / max(len(other_ns), 1)
    for n_try in other_ns:
        if time.time() > d3:

        d_res = min(time.time() + time_per, d3)
