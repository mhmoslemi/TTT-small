import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def sm_obj_grad(h, beta, n):
    """Smooth max objective and analytical gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(x)
    se = np.sum(e)
    w = e / se
    obj = (m + np.log(se / (2*n-1)) / beta) * dx
    
    grad = np.zeros(n)
    # Loop over lags with significant weight
    for k in range(-(n-1), n):
        wi = w[n-1+k]
        if wi < 1e-12:
            continue
        if k > 0:
            grad[k:] += wi * (1.0 - h[:n-k])
            grad[:n-k] -= wi * h[k:]
        elif k < 0:
            kp = -k
            grad[:n-kp] += wi * (1.0 - h[kp:])
            grad[kp:] -= wi * h[:n-kp]
        else:
            grad += wi * (1.0 - 2.0 * h)
    
    return obj, grad * dx

def sum_constraint(h, target):
    return np.sum(h) - target

def sum_jac(h, target):
    return np.ones(len(h))

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def slsqp_analytical(h0, n, target, beta, maxiter=10000):
    """SLSQP with analytical gradient."""
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    
    def obj_and_grad(x):
        o, g = sm_obj_grad(x, beta, n)
        return o, g
    
    try:
        r = minimize(obj_and_grad, h0, jac=True, method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_h(r.x, target)
    except Exception:
        return h0

def continuous_anneal(h, target, n, t0, budget, n_iter=50000, rng=None):
    """Simulated annealing with continuous perturbations."""
    if rng is None:
        rng = np.random.default_rng(123)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        frac = it / n_iter
        # Temperature schedule
        temp = 0.05 * (1.0 - frac) + 0.001
        
        # Random block perturbation
        block_size = int(rng.integers(1, max(2, n // 8)))
        start = rng.integers(0, n - block_size)
        block = cur[start:start+block_size].copy()
        cur[start:start+block_size] = np.clip(block + rng.standard_normal(block_size) * temp, 0, 1)
        cur = project_h(cur, target)
        
        new_c5 = compute_c5(cur)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5

            st = rng.integers(0, n - bsz)
            old_b = cur[st:st + bsz].copy()
            cur[st:st + bsz] = np.clip(old_b + rng.standard_normal(bsz) * temp * 2, 0, 1)
            cur = project_h(cur, target)
            new_c5 = compute_c5(cur)
            if new_c5 >= cur_c5 and rng.random() > np.exp(-(new_c5 - cur_c5) / max(temp * 0.02, 1e-15)):
                cur[st:st + bsz] = old_b
                cur = project_h(cur, target)
            else:
                cur_c5 = new_c5
        else:
            bsz = int(rng.integers(1, max(2, n // 6)))
            s1 = rng.integers(0, n - bsz)
            s2 = rng.integers(0, n - bsz)
            if abs(s1 - s2) <= bsz:
                continue
            b1 = cur[s1:s1 + bsz].copy()
            b2 = cur[s2:s2 + bsz].copy()
            cur[s1:s1 + bsz] = b2
            cur[s2:s2 + bsz] = b1
            new_c5 = compute_c5(cur)
            if new_c5 < cur_c5:
                cur_c5 = new_c5
            else:
                cur[s1:s1 + bsz] = b1
                cur[s2:s2 + bsz] = b2
                if rng.random() > np.exp(-(new_c5 - cur_c5) / max(temp * 0.03, 1e-15)):
                    cur[s1:s1 + bsz] = b1
                    cur[s2:s2 + bsz] = b1
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best = cur.copy()
    return best, best_c5

def coord_polish(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.03, -0.03]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-15:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 2:
            break
    return cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(320, 0.5)
        h_init[:160] = 1.0
        h_init[160:] = 0.0
    n = len(h_init)
    target = n / 2.0
    best_h = project_h(h_init, target)
    best_c5 = compute_c5(best_h)

    # Phase 1: Progressive SLSQP with analytical gradient
    for beta in [30, 100, 300, 1000, 3000]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 10:
            break
        best_h = slsqp_smooth(best_h, n, target, beta, maxiter=6000)
        c_try = compute_c5(best_h)
        if c_try < best_c5:
            best_c5 = c_try

    # Phase 2: Block annealing
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        rng = np.random.default_rng(seed)
        best_h, best_c5 = block_anneal(best_h, target, n, t0, budget_s,
                                       int(remaining * 8), rng)

    # Phase 3: SLSQP after annealing
    for beta in [2000, 8000, 30000]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 10:
            break
        best_h = slsqp_smooth(best_h, n, target, beta, maxiter=8000)
        c_try = compute_c5(best_h)
        if c_try < best_c5:
            best_c5 = c_try

    # Phase 4: Perturbation + re-optimize
    rng = np.random.default_rng(seed + 100)
    for att in range(10):
        remaining = budget_s - (time.time() - t0)
        if remaining < 25:
            break
        scale = 0.04 * (0.75 ** att)
        hp = best_h + rng.standard_normal(n) * scale
        hp = project_h(hp, target)
        for beta in [300, 3000, 15000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 10:
                break
            hp = slsqp_smooth(hp, n, target, beta, maxiter=5000)
            c_try = compute_c5(hp)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = hp.copy()

    # Phase 5: Try different resolution
    remaining = budget_s - (time.time() - t0)
    for n_hr in [256, 448, 512]:
        if remaining < 60:
            break
        target_hr = n_hr / 2.0
        cur_hr = project_h(resample_fn(best_h, n_hr), target_hr)
        for beta in [100, 500, 2000, 10000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 15:
                break
            cur_hr = slsqp_smooth(cur_hr, n_hr, target_hr, beta, maxiter=5000)
            c_try = compute_c5(cur_hr)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = cur_hr.copy()
                n = n_hr
                target = target_hr

    # Phase 6: Another annealing round at best resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        rng = np.random.default_rng(seed + 200)
        best_h, best_c5 = block_anneal(best_h, target, n, t0, budget_s,
                                       int(remaining * 6), rng)

    # Phase 7: Final SLSQP polish
    for beta in [50000, 100000]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 8:
            break
        best_h = slsqp_smooth(best_h, n, target, beta, maxiter=10000)
        best_c5 = compute_c5(best_h)

    # Phase 8: Coordinate polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 5:
        best_h, best_c5 = coord_polish(best_h, target, n, 200, t0, budget_s)

    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
