import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -400, 400)
    e = np.exp(sh)
    se = e.sum()
    val = (m + np.log(se / len(corr)) / beta) * dx
    w = e / se
    g = np.zeros(n)
    nc = len(corr)
    for j in range(nc):
        if w[j] < 1e-13:
            continue
        lag = j - (n - 1)
        if lag == 0:
            g += w[j] * (1.0 - 2.0 * h)
        elif lag > 0:
            g[lag:] += w[j] * (1.0 - h[:-lag])
            g[:n - lag] -= w[j] * h[lag:]
        else:
            m2 = -lag
            g[:n - m2] += w[j] * (1.0 - h[m2:])
            g[m2:] -= w[j] * h[:n - m2]
    return val, g * dx

def slsqp_step(h0, n, beta, maxiter=500):
    target = n / 2.0
    bounds = [(0, 1)] * n
    def obj(x):
        return smooth_max_grad(np.clip(x, 0, 1), beta)[0]
    def grad(x):
        return smooth_max_grad(np.clip(x, 0, 1), beta)[1]
    def eq(x):
        return np.sum(x) - target
    def eqj(x):
        return np.ones(n)
    res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': eq, 'jac': eqj}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_sum(res.x, target)

def proj_grad(h0, n, beta, max_iter, t0, budget):
    target = n / 2.0
    x = h0.copy()
    best = x.copy()
    best_f = compute_c5(best)
    step = 0.005
    stuck = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 3:
            break
        _, g = smooth_max_grad(x, beta)
        x_new = project_sum(x - step * g, target)
        f_new = compute_c5(x_new)
        if f_new < best_f - 1e-15:
            best = x_new.copy()
            best_f = f_new
            stuck = 0
        else:
            stuck += 1
        if stuck > 50:
            step *= 0.7
            if step < 1e-6:
                step = 0.003
                pert = np.random.randn(n) * 0.005
                x = project_sum(best + pert, target)
                stuck = 0
        elif it % 200 == 199 and stuck < 20:
            step = min(step * 1.05, 0.04)
        x = x_new
    return best, best_f

def subgrad_polish(h, t0, budget, n_active=6, max_iter=5000):
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_c5 = compute_c5(best)
    no_improve = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        top_idx = np.argsort(corr)[::-1][:n_active]
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag == 0:
                g += (1.0 - 2.0 * best)
            elif lag > 0:
                g[lag:] += (1.0 - best[:-lag])
                g[:n - lag] -= best[lag:]
            else:
                m2 = -lag
                g[:n - m2] += (1.0 - best[m2:])
                g[m2:] -= best[:n - m2]
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0003]:
            ht = project_sum(best - step * g, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best = ht
                best_c5 = c5t
                improved = True
                break
        if not improved:
            no_improve += 1
            if no_improve > 100 and it % 50 == 49:
                scale = 0.01 + 0.02 * (no_improve // 500)
                pert = np.random.randn(n) * scale
                ht = project_sum(best + pert, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    no_improve = 0
            if no_improve > 3000:
                break
        else:
            no_improve = 0
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(480, 0.5)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP refinement at initial resolution
    n1 = len(best_h)
    for beta in [500, 5000]:
        if time.time() - t0 > budget_s * 0.12:
            break
        h_new = slsqp_step(best_h, n1, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Quick subgrad polish at current resolution
    best_h, best_c5 = subgrad_polish(best_h, t0, budget_s, n_active=5, max_iter=2000)

    # Phase 2: Resolution increase with projected gradient
    resolutions = [800, 1200, 1600, 2000]
    for n_target in resolutions:
        if time.time() - t0 > budget_s * 0.82:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_target)
        h_interp = project_sum(np.interp(x_new, x_old, best_h), n_target / 2.0)
        h_interp = h_interp.copy()
        improved_up = False
        for beta in [200, 1000, 5000, 20000, 80000]:
            if time.time() - t0 > budget_s * 0.82:
                break
            h_new, c5_new = proj_grad(h_interp, n_target, beta,
                                      max_iter=3000, t0=t0, budget=budget_s)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
                improved_up = True
            h_interp = h_new
        if not improved_up:
            # Try a multistart at this resolution
            for attempt in range(2):
                if time.time() - t0 > budget_s * 0.82:
                    break
                h_try = project_sum(best_h + np.random.randn(n_target) * 0.03, n_target / 2.0)
                h_new, c5_new = proj_grad(h_try, n_target, 5000,
                                          max_iter=2000, t0=t0, budget=budget_s)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new, c5_new

    # Phase 3: Final intensive subgradient polishing
    best_h, best_c5 = subgrad_polish(best_h, t0, budget_s, n_active=8, max_iter=10000)

    # Phase 4: One more SLSQP if time allows (high beta)
    if time.time() - t0 < budget_s - 80 and len(best_h) <= 1600:
        n_final = len(best_h)
        h_new = slsqp_step(best_h, n_final, 50000, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Final micro-polish
    if time.time() - t0 < budget_s - 10:
        best_h, best_c5 = subgrad_polish(best_h, t0, budget_s, n_active=4, max_iter=3000)

    return best_h, best_c5, len(best_h)
