   ```python
   def compute_c5(h):
       n = len(h)
       dx = 2.0 / n
       corr = np.correlate(h, 1.0 - h, mode="full")
       return float(np.max(corr) * dx)
   
   def project_h(h, target_sum):
       h = np.clip(h, 0.0, 1.0)
       for _ in range(100):
           diff = target_sum - h.sum()
           if abs(diff) < 1e-14:
               break
           free = (h > 1e-12) & (h < 1.0 - 1e-12)
           if free.sum() == 0:
               free = np.ones(len(h), dtype=bool)
           h[free] += diff / free.sum()
           h = np.clip(h, 0.0, 1.0)
       return h
   
   def smooth_max_val_grad(h, beta):
       """Compute smooth max of correlation * dx and its gradient."""
       n = len(h)
       dx = 2.0 / n
       corr = np.correlate(h, 1.0 - h, mode="full")
       m = np.max(corr)
       # Clip for numerical stability
       x = np.clip(beta * (corr - m), -500, 500)
       exp_vals = np.exp(x)
       sum_exp = np.sum(exp_vals)
       weights = exp_vals / sum_exp
       val = (m + np.log(sum_exp) / beta) * dx
       
       # Gradient: only compute for lags with significant weight
       grad = np.zeros(n)
       threshold = 1e-8
       active = np.where(weights > threshold)[0]
       for k in active:
           w = weights[k] * dx
           L = int(k) - (n - 1)
           if L == 0:
               grad += w * (1.0 - 2.0 * h)
           elif 0 < L < n:
               grad[L:] += w * (1.0 - h[:-L])

           elif -n < L < 0:
               M = -L
               grad[:n-M] += w * (1.0 - h[M:])
               grad[M:] -= w * h[:n-M]
       return val, grad
   ```

Actually, here's an idea: **successive linearization (outer approximation method)**.

At the current point h*, the overlap at lag k is:
overlap_k(h) = dx * sum_i h[i]*(1-h[i+k])

This is bilinear, but at h* we can write:
overlap_k(h) ≈ overlap_k(h*) + sum_i (dc_k/dh_i)|_{h*} * (h[i] - h*[i])

where dc_k/dh_i = (1-h*[i+k]) - h*[i-k] (with appropriate range checks).

So we can solve a linear program:
minimize t
subject to t ≥ dx * (overlap_k(h*) + sum_i dc_k/dh_i * (h[i]-h*[i])) for all k
           0 ≤ h[i] ≤ 1
           sum h[i] = n/2

This LP gives a lower bound on the true optimum and a new iterate h_new. We can then check if the true objective at h_new is better than at h*, and iterate.

This is the "successive convex programming" or "outer approximation" method. It should converge to a local minimum.

The LP has n variables + 1 slack variable t, and 2n-1 + 1 constraints (one per lag + sum constraint) + 2n bound constraints. For n=500, that's about 1500 constraints and 501 variables. This is very manageable for a commercial LP solver.

```python
import cvxpy as cp

def solve_lp(h_star, n):
    target = n / 2.0
    dx = 2.0 / n
    h = cp.Variable(n)
    t = cp.Variable()
    
    corr_star = np.correlate(h_star, 1.0 - h_star, mode="full")
    c_k = corr_star * dx  # current overlap values
    
    # Linearization: overlap_k(h) ≈ c_k + dx * sum_i dc_k/dh_i * (h[i] - h_star[i])
    # dc_k/dh_i = (1-h_star[i+L]) * [valid] - h_star[i-L] * [valid]
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == target]
    
    for k in range(2 * n - 1):
        L = k - (n - 1)
        # Compute gradient of overlap_k w.r.t. h
        grad_k = np.zeros(n)
        if L == 0:
            grad_k[:] = 1.0 - 2.0 * h_star
        elif 0 < L < n:
            grad_k[L:] += (1.0 - h_star[:-L])
            grad_k[:n-L] -= h_star[L:]
        elif -n < L < 0:
            M = -L
            grad_k[:n-M] += (1.0 - h_star[M:])
            grad_k[M:] -= h_star[:n-M]
        
        # linearized overlap at lag k
        linear_k = c_k[k] + dx * (grad_k @ (h - h_star))
        constraints.append(linear_k <= t)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    
    # Try solvers in order
    for solver in [cp.ECOS, cp.SCS, cp.GLPK, cp.CBC]:
        try:
            prob.solve(solver=solver)
            if h.value is not None:
                return h.value, t.value
        except:
            continue
    return None, None
```

```python
def project_h(h, target_sum):
    h = h.copy()

def smooth_max_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
```
