    weights = exp_s / s
    return val, weights

def adam_optimize(h, n, deadline, beta, lr=0.05, max_stag=200, seed=42):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    t = 0
    b1, b2 = 0.85, 0.999
    c5 = compute_c5(h)
    stagnant = 0
    best_local = h.copy()
    best_local_c5 = c5
    n_full = 2 * n - 1
    while time.time() < deadline:
        t += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        val, w = smooth_max_grad(corr, beta)
        threshold = 1e-5
        active = w > threshold
        if not np.any(active):
            active_idx = np.argsort(w)[-10:]
        else:
            active_idx = np.where(active)[0]
            if len(active_idx) > 30:
                active_idx = np.argsort(w[active_idx])[-30:] + np.where(active)[0][np.argsort(w[active_idx])[-30:]]
        grad = np.zeros(n)
        for k in active_idx:
            lag = k - (n - 1)
            grad += w[k] * grad_for_lag(h, lag, n)
        grad *= dx
        m1 = b1 * m1 + (1 - b1) * grad
        m2 = b2 * m2 + (1 - b2) * grad * grad
        m1_hat = m1 / (1 - b1**t)
        m2_hat = m2 / (1 - b2**t)
        h_new = project_feasible(h - lr * m1_hat / (np.sqrt(m2_hat) + 1e-8), ts)
        c_new = compute_c5(h_new)
        if c_new < c5 - 1e-12:
            h = h_new
            c5 = c_new
            if c5 < best_local_c5:
                best_local = h.copy()
                best_local_c5 = c5
            stagnant = 0
        else:
            stagnant += 1
            if stagnant % 50 == 0:
                lr *= 0.92
            if stagnant > max_stag:
                if t > 500:
                    break
                h = project_feasible(best_local + np.random.randn(n) * 0.02, ts)
                stagnant = 0
                lr = max(lr, 0.01)
    return best_local, best_local_c5

def subgradient_polish(h, n, deadline, seed):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    velocity = np.zeros(n)
    lr = 0.003
    stagnant = 0
    for it in range(20000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_n = min(8, len(corr))
        top_idx = np.argsort(corr)[-top_n:]
        w = np.exp(15.0 * (corr[top_idx] - corr[top_idx].max()))
        w /= w.sum()
        grad = np.zeros(n)
        for j, k in enumerate(top_idx):
            lag = k - (n - 1)
            grad += w[j] * grad_for_lag(h, lag, n)
        velocity = 0.8 * velocity - lr * grad
        h_new = project_feasible(h + velocity, ts)
        c_new = compute_c5(h_new)
        if c_new < c5 - 1e-11:
            h = h_new
            c5 = c_new
            stagnant = 0
        else:
            stagnant += 1
            velocity *= 0.5
            lr *= 0.93
        if stagnant > 500 and lr < 1e-7:
            break
    return h, c5

def slsqp_optimize(h, n, deadline, beta, seed):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts, 'jac': lambda x: np.ones(n)}]
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max_grad(corr, beta)[0] * dx
    c5 = compute_c5(h)
    x0 = h.copy()
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 10000, 'ftol': 1e-14})
        h_new = project_feasible(res.x, ts)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            return h_new, c5_new
    except Exception:
        pass
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s * 0.96

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    if parent is None:
        parent = np.full(200, 0.5)
    pn = len(parent)

    best_h = parent.copy()
    best_c5 = compute_c5(parent)
    best_n = pn

    # Phase 1: Adam at parent resolution with annealing (20%)
    d1 = t0 + budget_s * 0.20
    for beta in [30, 80, 200, 800, 3000]:
        if time.time() > d1:
            break
        h_new, c_new = adam_optimize(best_h, pn, d1, beta, lr=0.04, seed=seed + beta)
        if c_new < best_c5:
            best_h, best_c5 = h_new, c_new

    # Phase 2: SLSQP polish at parent resolution (10%)
    d2 = t0 + budget_s * 0.30
    for beta in [2000, 10000]:
        if time.time() > d2:
            break
        h_new, c_new = slsqp_optimize(best_h, pn, d2, beta, seed=seed + 1000 + beta)
        if c_new < best_c5:
            best_h, best_c5 = h_new, c_new
    if time.time() < d2:
        best_h, best_c5 = subgradient_polish(best_h, pn, d2, seed + 500)

    # Phase 3: Higher resolution optimization (40%)
    d3 = t0 + budget_s * 0.72
    resolutions = [500, 700, 1050, 1400]
    for n_new in resolutions:
        if time.time() > d3:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h0 = np.interp(x_new, x_old, best_h)
        h0 = project_feasible(h0, n_new / 2.0)
        c0 = compute_c5(h0)
        for beta in [100, 500, 2000, 8000]:
            if time.time() > d3:
                break
            h_new, c_new = adam_optimize(h0, n_new, d3, beta, lr=0.015, seed=seed + n_new + beta)
            if c_new < c0:
                h0, c0 = h_new, c_new
        if c0 < best_c5:
            best_h, best_c5, best_n = h0, c0, n_new

    # Phase 4: Re-interpolate to parent resolution if higher res won (5%)
    d4 = t0 + budget_s * 0.78
    if best_n != pn and time.time() < d4:
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, pn, endpoint=False)
        h_low = np.interp(x_new, x_old, best_h)
        h_low = project_feasible(h_low, pn / 2.0)
        h_new, c_new = adam_optimize(h_low, pn, d4, 5000, lr=0.01, seed=seed + 777)
        if c_new < best_c5:
            best_h, best_c5, best_n = h_new, c_new, pn

    # Phase 5: SLSQP + subgradient on best (14%)
    d5 = t0 + budget_s * 0.92
    if time.time() < d5:
        h_new, c_new = slsqp_optimize(best_h, best_n, d5, 20000, seed=seed + 2000)
        if c_new < best_c5:
            best_h, best_c5 = h_new, c_new
        if time.time() < d5:
            best_h, best_c5 = subgradient_polish(best_h, best_n, d5, seed + 888)

    # Phase 6: Final micro-polish (4%)
    d6 = deadline
    if time.time() < d6:
        best_h, best_c5 = subgradient_polish(best_h, best_n, d6, seed + 999)

    # Try a few random restarts if time permits
    if time.time() < deadline:
        for rep in range(3):
            if time.time() > deadline:
                break
            h_r = project_feasible(best_h + np.random.randn(best_n) * 0.03, best_n / 2.0)
            h_r, c_r = adam_optimize(h_r, best_n, deadline, 2000, lr=0.02, seed=seed + 5000 + rep)
            if c_r < best_c5:
                best_h, best_c5 = h_r, c_r

    # Final verification
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
