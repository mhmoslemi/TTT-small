    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()

            grad[lag:] += p[k] * (1 - x[:n-lag])

            grad[:n-ll] += p[k] * (1 - x[ll:])
            # x[i-ll] valid for i = ll..n-1
            grad[ll:] -= p[k] * x[:n-ll]
    
    grad *= dx
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(x, n, t):
    """Compute smooth max of g and its gradient."""
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    shifted = g - m
    exps = np.exp(t * shifted)
    sm = exps.sum()
    obj = m + np.log(sm) / t
    p = exps / sm
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += pk * (1 - x[:n-lag])
            grad[:n-lag] -= pk * x[lag:]
        else:
            ll = -lag
            grad[:n-ll] += pk * (1 - x[ll:])

    return obj, grad

        _, g = smooth_max_and_grad(x, n, t)

def local_search_pairs(h, n, deadline, max_iter=10000):
    """Local search with pair transfers."""
    best_c = compute_c5(h, n)
    h = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Try random pairs with various transfer amounts
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        # Find best transfer from i to j (or vice versa)
        best_delta = 0
        best_trial_c = best_c
        for delta in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
            for s in [1.0, -1.0]:
                hi, hj = h[i] + s*delta, h[j] - s*delta
                if 0 <= hi <= 1 and 0 <= hj <= 1:
                    h[i], h[j] = hi, hj
                    c = compute_c5(h, n)
                    if c < best_trial_c:
                        best_trial_c = c
                        best_delta = s * delta
                    h[i], h[j] = h[i] - s*delta, h[j] + s*delta
        if best_delta != 0:
            h[i] += best_delta
            h[j] -= best_delta
            best_c = best_trial_c
    
    return h, best_c

            for trial in np.linspace(0, 1, 11):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_grad(x, n, t):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    sh = g - m
    e = np.exp(t * sh)
    sm = e.sum()
    obj = m + np.log(sm) / t
    p = e / sm
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14: continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += pk * (1.0 - x[:n-lag])
            grad[:n-lag] -= pk * x[lag:]
        else:
            ll = -lag
            grad[:n-ll] += pk * (1.0 - x[ll:])
            grad[ll:] -= pk * x[:n-ll]
    grad *= dx
    return obj, grad

def opt_lbfgs(h, n, t, lam, maxiter=10000):
    def f(x):
        g = np.correlate(x, 1-x, mode='full') * (2.0/n)
        m = g.max(); sh = g - m
        e = np.exp(t*sh)
        return m + np.log(e.sum())/t + lam*(x.sum()-n/2.0)**2
    def jf(x):
        g = np.correlate(x, 1-x, mode='full') * (2.0/n)
        m = g.max(); sh = g - m
        e = np.exp(t*sh); p = e/e.sum()
        dx = 2.0/n
        gr = np.zeros(n)
        for k in range(2*n-1):
            pk = p[k]
            if pk < 1e-14: continue
            lag = k-(n-1)
            if lag >= 0:
                gr[lag:] += pk*(1.0-x[:n-lag])
                gr[:n-lag] -= pk*x[lag:]
            else:
                ll = -lag
                gr[:n-ll] += pk*(1.0-x[ll:])
                gr[ll:] -= pk*x[:n-ll]
        gr *= dx
        gr += 2*lam*(x.sum()-n/2.0)
        return gr
    res = minimize(f, h, jac=jf, method='L-BFGS-B',
                   bounds=[(0.0,1.0)]*n,
                   options={'maxiter': maxiter, 'ftol':1e-15, 'gtol':1e-8})
    return project_feasible(res.x, n)

def opt_slsqp(h, n, t, maxiter=3000):
    def f(x):
        return smooth_max_grad(x, n, t)[0]
    def jf(x):
        return smooth_max_grad(x, n, t)[1]
    res = minimize(f, h, jac=jf, method='SLSQP',
                   bounds=[(0.0,1.0)]*n,
                   constraints=[{'type':'eq','fun':lambda x: x.sum()-n/2.0,
                                 'jac':lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol':1e-14})
    return project_feasible(res.x, n)

def binary_search(h, n, deadline, max_sweeps=500):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:target]] = 1.0
    best_c = compute_c5(h_b, n)
    ones = set(np.where(h_b == 1)[0].tolist())
    
    for sweep in range(max_sweeps):
        if time.time() > deadline: break
        improved = False
        for j in sorted(ones):
            if time.time() > deadline: break
            for i in range(n):
                if i in ones: continue
                h_b[i] = 1.0; h_b[j] = 0.0
                c = compute_c5(h_b, n)
                if c < best_c - 1e-14:
                    best_c = c
                    ones.discard(j); ones.add(i)
                    improved = True
                    break
                h_b[i] = 0.0; h_b[j] = 1.0
            if improved: break
        if not improved: break
    return h_b, best_c

def pair_refine(h, n, deadline, max_iter=50000, rng=None):
    if rng is None:
        rng = np.random.RandomState(123)
    best_c = compute_c5(h, n)
    h = h.copy()
    dvals = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]
    for it in range(max_iter):
        if time.time() > deadline: break
        i, j = rng.randint(0, n, 2)
        if i == j: continue
        d = dvals[rng.randint(len(dvals))]
        for s in (1.0, -1.0):
            hi, hj = h[i]+s*d, h[j]-s*d
            if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                h[i], h[j] = hi, hj
                c = compute_c5(h, n)
                if c < best_c - 1e-13:
                    best_c = c
                else:
                    h[i], h[j] = hi-s*d, hj+s*d
    return h, best_c
