from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_full(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def gradient_max_corr(h, n):
    """Compute gradient of max_k c[k] where c = correlate(h, 1-h, 'full')"""
    v = 1 - h
    c = np.correlate(h, v, 'full')
    k_max = np.argmax(c)
    d = k_max - n + 1  # the lag
    # c[k_max] = sum_i h[i] * v[i+d] (valid indices)
    # grad[j] = v[j+d] (if j+d valid) - h[j-d] (if j-d valid)
    grad = np.zeros(n)
    # First term: v[j+d] for j where 0 <= j < n and 0 <= j+d < n
    j_start = max(0, -d)
    j_end = min(n, n - d)
    if j_start < j_end:
        grad[j_start:j_end] += v[j_start + d:j_end + d]
    # Second term: -h[j-d] for j where 0 <= j < n and 0 <= j-d < n
    j_start = max(0, d)
    j_end = min(n, n + d)
    if j_start < j_end:
        grad[j_start:j_end] -= h[j_start - d:j_end - d]
    return grad, c * (2.0 / n), k_max

def subgrad_analytical(h, n, deadline, max_iter=20000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.5
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad, c_vals, k_max = gradient_max_corr(h, n)
        m = c_vals[k_max]
        ng = np.linalg.norm(grad)
        if ng < 1e-12:
            break
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:
            step = lr * f / ng
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-12:
                h = ht
                found = True
                break
        if found:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            lr = min(lr * 1.05, 2.0)
        else:
            lr *= 0.9
            if lr < 1e-10:
                break
    return best_h, best_c

            ones.append(j)
            zeros.remove(j)
            zeros.append(i)

def compute_S(h, n, d_arr):
    # S[p, k] = h[p + d_arr[k]] + h[p - d_arr[k]] (with boundaries = 0)
    # For a matrix M where M[p, k] = p + d_arr[k], we want h[M[p,k]] where valid

    S = np.zeros((n, 2 * n - 1))
    S[valid_plus] = h[M_plus[valid_plus]]
    S[valid_minus] += h[M_minus[valid_minus]]
    return S

def coordinate_descent(h, n, deadline, max_sweeps=10000):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    c = np.correlate(h, 1 - h, 'full')
    d_arr = np.arange(-(n - 1), n)
    cmax = c.max()

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        any_change = False
        for i in range(n):
            if time.time() > deadline:
                break
            idx_p = i + d_arr
            idx_m = i - d_arr
            vp = (idx_p >= 0) & (idx_p < n)
            vm = (idx_m >= 0) & (idx_m < n)
            hp = np.zeros(2 * n - 1)
            hm = np.zeros(2 * n - 1)
            hp[vp] = h[idx_p[vp]]
            hm[vm] = h[idx_m[vm]]
            R = c - h[i] * hp - hm * (1.0 - h[i])
            alpha = hp - hm
            beta = hm + R
            a, b = 0.0, 1.0
            for _ in range(35):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                if np.max(alpha * m1 + beta) <= np.max(alpha * m2 + beta):
                    b = m2
                else:
                    a = m1
            opt = (a + b) / 2.0
            if abs(opt - h[i]) > 1e-12:
                delta = opt - h[i]
                h[i] = opt
                c += delta * alpha
                cmax_new = c.max()
                if cmax_new < cmax - 1e-14:
                    cmax = cmax_new
                    any_change = True
        if any_change:
            c5 = cmax * (2.0 / n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            break
    return best_h, best_c5

def binary_swap_search(h_bin, n, deadline, max_sweeps=500):
    h = h_bin.copy()
    c = np.correlate(h, 1 - h, 'full')
    best_c = int(c.max())
    best_c5 = best_c * (2.0 / n)
    best_h = h.copy()
    S = compute_S(h, n)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        best_nm = best_c
        best_i, best_j = -1, -1
        zeros_arr = np.array(zeros)
        S_zeros = S[zeros_arr]
        for i in ones:
            if time.time() > deadline:
                break
            T = c + S[i]
            new_C = T[None, :] - S_zeros
            new_maxes = new_C.max(axis=1)
            min_idx = np.argmin(new_maxes)
            nm = int(new_maxes[min_idx])
            if nm < best_nm:
                best_nm = nm
                best_i, best_j = i, int(zeros_arr[min_idx])
        if best_i >= 0:
            i, j = best_i, best_j
            h[i], h[j] = 0.0, 1.0
            S = compute_S(h, n)
            c = np.correlate(h, 1 - h, 'full')
            best_c = int(c.max())
            best_c5 = best_c * (2.0 / n)
            best_h = h.copy()
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            improved = True
        if not improved:
            break
    return best_h, best_c5

def slsqp_smooth(h, n, deadline, alpha, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, 'full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def obj_grad(x):
        g = np.correlate(x, 1 - x, 'full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        w = np.exp(sh - sm)
        w /= w.sum()
        grad = np.zeros(n)
        d = np.arange(-(n - 1), n)
        for k in range(2 * n - 1):
            if w[k] < 1e-15:
                continue
            kk = k - (n - 1)
            ip = np.arange(n) - kk
            im = np.arange(n) + kk
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            grad[vp] += w[k] * (1 - x[ip[vp]]) * dx
            grad[vm] -= w[k] * x[im[vm]] * dx
        return grad
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    n = n0
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Coordinate descent from initial (35% budget)
    p1 = t0 + 0.35 * (budget_s - 5)
    h, c = coordinate_descent(best_h.copy(), n, p1)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 2: Binary swap search (20% budget)
    p2 = t0 + 0.55 * (budget_s - 5)
    if time.time() < p2 - 10:
        h_bin = (best_h >= 0.5).astype(float)
        target = n // 2
        if int(h_bin.sum()) != target:
            idx = np.argsort(best_h)[::-1]
            h_bin = np.zeros(n)
            h_bin[idx[:target]] = 1.0
        hb, cb = binary_swap_search(h_bin, n, p2)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 3: Coordinate descent from best (may be binary now) (20% budget)
    p3 = t0 + 0.75 * (budget_s - 5)
    h, c = coordinate_descent(best_h.copy(), n, p3)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 4: SLSQP with softmax on fractional solution (10% budget)
    if time.time() < dl - 60:
        for alpha in [5000, 20000, 100000]:
            if time.time() > dl - 40:
                break
            hs = slsqp_smooth(best_h.copy(), n, dl - 30, alpha, maxiter=800)
            cs = compute_c5(hs, n)
            if cs < best_c:
                best_c = cs
                best_h = hs.copy()

    # Phase 5: Final coordinate descent polish (remaining time)
    if time.time() < dl - 3:
        h, c = coordinate_descent(best_h.copy(), n, dl)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
