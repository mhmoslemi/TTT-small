import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_and_correlation(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c)), c

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def subgradient_at_peak(h, n, L):
    """Compute dc[L]/dh for the given lag L."""
    g = np.zeros(n)
    if L >= 0:
        if L < n:
            g[L:] += (1.0 - h[:n - L])   # first term for j=L..n-1
            g[:n - L] -= h[L:]            # second term for j=0..n-1-L
    else:
        Lp = -L
        if Lp < n:
            g[:n - Lp] += (1.0 - h[Lp:])  # first term: j-L = j+Lp, valid for j=0..n-1-Lp
            g[Lp:] -= h[:n - Lp]          # second term: j+L = j-Lp, valid for j=Lp..n-1
    return g

def subgrad_optimize(h, n, deadline, top_k=3):
    """Projected subgradient descent with momentum."""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c, c_full = compute_c5_and_correlation(h, n)
    
    # Momentum
    velocity = np.zeros(n)
    beta = 0.9
    
    # Adaptive learning rate
    lr = 0.01
    
    t_start = time.time()
    stall_count = 0
    
    while time.time() < deadline:
        c_val, c_full = compute_c5_and_correlation(h, n)
        
        if c_val < best_c - 1e-15:
            best_c = c_val
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
            # Reduce learning rate if stalled
            if stall_count % 50 == 0:
                lr *= 0.7
        
        # Find top-k peaks
        top_indices = np.argsort(c_full)[::-1][:top_k]
        
        # Weighted subgradient
        g = np.zeros(n)
        weights = np.exp(-0.1 * np.arange(top_k))  # decay weights
        for i, L_idx in enumerate(top_indices):
            L = L_idx - n + 1  # convert to lag
            g += weights[i] * subgradient_at_peak(h, n, L)
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm
        
        # Momentum step
        velocity = beta * velocity - lr * g
        h_new = h + velocity
        
        # Project
        h_new = project_feasible(h_new, n)
        
        # Check if step was good
        c_new, _ = compute_c5_and_correlation(h_new, n)
        if c_new < c_val + 1e-10:
            h = h_new
        else:
            # Revert and reduce lr
            h = best_h.copy()
            lr *= 0.5
            velocity *= 0.5
        
        # Adapt lr based on progress
        elapsed = time.time() - t_start
        if elapsed > 10:
            lr = max(0.001, min(0.05, lr))
    
    return best_h, best_c

    grad = (G * weights[None, :]).sum(axis=1) * dx
    return grad

def compute_c5_and_corr(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c)), c

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_gradient(h, n, alpha):
    """Gradient of (1/alpha)*log(sum exp(alpha*c[k])) w.r.t. h."""
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    c_scaled = c_raw * dx
    a_c = alpha * c_scaled
    a_c_shifted = a_c - a_c.max()
    weights = np.exp(a_c_shifted)
    weights = weights / weights.sum()
    
    # Build gradient matrix G (n x (2n-1))
    i_idx = np.arange(n)[:, None]
    k_idx = np.arange(2 * n - 1)[None, :]
    L = k_idx - (n - 1)
    
    idx1 = i_idx - L
    valid1 = (idx1 >= 0) & (idx1 <= n - 1)
    G = valid1.astype(float) * (1.0 - h[np.clip(idx1, 0, n-1)])
    
    idx2 = i_idx + L
    valid2 = (idx2 >= 0) & (idx2 <= n - 1)
    G -= valid2.astype(float) * h[np.clip(idx2, 0, n-1)]
    
    return (G * weights[None, :]).sum(axis=1) * dx

def lse_objective(h, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    c_scaled = c_raw * dx
    a_c = alpha * c_scaled
    a_c_shifted = a_c - a_c.max()
    return c_scaled.max() + (a_c_shifted.max() + np.log(np.sum(np.exp(a_c_shifted - a_c_shifted.max())))) / alpha

        n_steps += 1

        grad = lse_gradient(h, n, alpha)
        
        # Momentum
        velocity = beta * velocity - lr * grad
        h_new = project_feasible(h + velocity, n)
        
        c_new, _ = compute_c5_and_corr(h_new, n)
        if c_new < best_c:

            lr = min(lr * 1.05, 0.1)

            lr *= 0.5
            velocity *= 0.5

            velocity = np.zeros(n)
