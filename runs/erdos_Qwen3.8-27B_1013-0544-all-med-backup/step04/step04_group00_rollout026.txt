import numpy as np
from scipy.optimize import minimize
import time

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_g(h, n)))

def smooth_max_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    gm = g.max()
    exp_g = np.exp(alpha * (g - gm))
    s = exp_g.sum()
    fval = gm + np.log(s) / alpha
    p = exp_g / s
    
    h_rev = h[::-1]
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for j in range(n):
        idx1 = np.arange(n - 1 - j, 2 * n - 1 - j)
        t1 = np.dot(p[idx1], one_minus_h)
        idx2 = np.arange(j, n + j)
        t2 = np.dot(p[idx2], h_rev)
        grad[j] = (dx / alpha) * (t1 - t2)
    
    return fval, grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    def obj(x):
        f, _ = smooth_max_obj_and_grad(x, n, alpha)
        return f
    def obj_grad(x):
        _, g = smooth_max_obj_and_grad(x, n, alpha)
        return g
    def con(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except:
        return h

def binary_local_search(h, n, deadline, max_sweeps=100):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = set(np.where(h_bin == 0)[0].tolist())
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        g = compute_g(h_bin, n)
        k_star = int(np.argmax(g))
        for i in sorted(zeros):
            if time.time() > deadline:
                break
            for j in sorted(ones):
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                c = compute_c5(h_bin, n)
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                if c < best_c - 1e-15:
                    best_c = c
                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    zeros.add(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break

import numpy as np
from scipy.optimize import minimize
import time

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_g(h, n)))

def smooth_max_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1.0 - h, mode='full') * dx
    gm = g.max()
    exp_g = np.exp(alpha * (g - gm))
    s = exp_g.sum()
    fval = gm + np.log(s) / alpha
    p = exp_g / s
    h_rev = h[::-1]
    omh = 1.0 - h
    grad = np.zeros(n)
    for j in range(n):
        idx1 = np.arange(n - 1 - j, 2 * n - 1 - j)
        t1 = np.dot(p[idx1], omh)
        idx2 = np.arange(j, n + j)
        t2 = np.dot(p[idx2], h_rev)
        grad[j] = (dx / alpha) * (t1 - t2)
    return fval, grad

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    def obj(x):
        f, _ = smooth_max_obj_and_grad(x, n, alpha)
        return f
    def obj_jac(x):
        _, g = smooth_max_obj_and_grad(x, n, alpha)
        return g
    def con(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=obj_jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def hclimbing(h, n, deadline, n_pert=200, scale=0.05):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng_p = np.random.RandomState(12345)
    for i in range(n_pert):
        if time.time() > deadline:
            break
        h_try = project_feasible(best_h + scale * rng_p.randn(n), n)
        c = compute_c5(h_try, n)
        if c < best_c:
            best_c = c
            best_h = h_try.copy()
    return best_h, best_c

def binary_local_search(h, n, deadline, max_sweeps=300):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = set(np.where(h_bin == 0)[0].tolist())
    ones_list = sorted(ones)
    zeros_list = sorted(zeros)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_cswap = best_c
            for j in ones_list:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                c = compute_c5(h_bin, n)
                if c < best_cswap:
                    best_cswap = c
                    best_j = j
                h_bin[i] = 0.0
                h_bin[j] = 1.0
            if best_j >= 0:
                h_bin[i] = 1.0
                h_bin[best_j] = 0.0
                best_c = best_cswap
                ones.discard(best_j)
                ones.add(i)
                zeros.discard(i)
                zeros.add(best_j)
                ones_list = sorted(ones)
                zeros_list = sorted(zeros)
                improved = True
                break
        if not improved:
            break
    return h_bin, best_c
