    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    n0 = len(best_h)
    target0 = n0 / 2.0
    best_c5 = compute_c5(best_h)

    # Phase 1: Multi-start (40s)
    if rem() > 250:
        cands = [best_h.copy()]
        for scale in [0.08, 0.15, 0.25, 0.4]:
            cands.append(project_h(best_h + np.random.randn(n0)*scale, target0))
        for freq in [3, 5, 7, 11]:
            p = 0.12 * np.sin(np.linspace(0, freq*np.pi, n0))
            cands.append(project_h(best_h + p, target0))
        for hc in cands[1:]:
            c = compute_c5(hc)
            if c < best_c5:
                best_c5 = c
                best_h = hc

    # Phase 2: SLSQP at base (120s)
    if rem() > 350:
        best_h, best_c5 = slsqp_solve(best_h, target0,
            [30, 100, 400, 1500, 6000, 20000], 4000, rem)

    # Phase 3: Subgradient at base (250s)
    if rem() > 300:
        best_h, best_c5 = adapt_subgrad(
            best_h, target0, rem, 40000, 45, 0.014,
            350, 0.009, 0.82, 0.94, slsqp_every=2000)

    # Phase 4: Upgrade to 2x resolution
    if rem() > 250:
        n2 = n0 * 2
        target2 = n2 / 2.0
        x_old = np.linspace(0, 2, n0)
        x_new = np.linspace(0, 2, n2)
        h2 = project_h(np.interp(x_new, x_old, best_h), target2)
        c5_2 = compute_c5(h2)
        best_h2 = h2
        best_c5_2 = c5_2

        # Phase 5: SLSQP at 2x (150s)
        if rem() > 150:
            best_h2, best_c5_2 = slsqp_solve(
                best_h2, target2, [50, 200, 800, 3000, 12000], 3500, rem)

        # Phase 6: Subgradient at 2x (remaining)
        if rem() > 80:
            best_h2, best_c5_2 = adapt_subgrad(
                best_h2, target2, rem, 30000, 55, 0.010,
                400, 0.007, 0.86, 0.96, slsqp_every=2500)

        if best_c5_2 < best_c5:
            best_h = best_h2
            best_c5 = best_c5_2
    else:
        # Stay at base resolution, do more optimization
        if rem() > 100:
            best_h, best_c5 = slsqp_solve(best_h, target0,
                [8000, 30000], 2000, rem)
        if rem() > 50:
            best_h, best_c5 = adapt_subgrad(
                best_h, target0, rem, 15000, 50, 0.006,
                500, 0.004, 0.93, 0.97, slsqp_every=0)

    # Phase 7: Final polish
    if rem() > 15:
        nf = len(best_h)
        tf = nf / 2.0
        best_h, best_c5 = adapt_subgrad(
            best_h, tf, rem, 8000, 60, 0.003,
            600, 0.003, 0.96, 0.98, slsqp_every=1500)

    best_h = project_h(best_h, len(best_h)/2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
