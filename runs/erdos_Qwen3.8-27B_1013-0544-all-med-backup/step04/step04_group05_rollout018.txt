def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def slp_phase(h, target, dx, time_fn, max_iters=500, top_m=None, trust_rho=0.3):

from scipy.optimize import linprog, minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def slp_phase(h, target, dx, time_fn, max_iters=300, lag_frac=1.0, rho=0.35):
    n = len(h)
    n_lags = 2 * n - 1
    best = h.copy()
    best_c5 = compute_c5(best)
    cur_rho = rho
    stale = 0
    for it in range(max_iters):
        if time_fn() < 80:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        threshold = np.max(corr) * 0.7
        idx = np.where(corr >= threshold)[0]
        if len(idx) < 5:
            idx = np.argsort(corr)[::-1][:max(5, int(n_lags * lag_frac))]
        A_rows = []
        b_rows = []
        for ii in idx:
            lag = int(ii) - (n - 1)
            if lag == 0:
                grad = 1.0 - 2.0 * best
            elif lag > 0:
                grad = np.zeros(n)
                grad[:n - lag] += 1.0 - best[lag:]
                grad[lag:] -= best[:n - lag]
            else:
                m = -lag
                grad = np.zeros(n)
                grad[m:] += 1.0 - best[:n - m]
                grad[:n - m] -= best[m:]
            A_rows.append(np.concatenate([grad, [-1.0]]))
            b_rows.append(-corr[ii] + grad @ best)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

            continue

def subgrad_phase(h, target, time_fn, max_iters=40000, m_active=15, momentum=0.85):

    best = h.copy()
