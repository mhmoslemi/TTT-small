    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_box_sum(x, S):
    """Project x onto {0<=h<=1, sum(h)=S} using bisection on lambda."""
    # h = clip(x - lam, 0, 1), find lam such that sum(h) = S
    lo = x.min() - 1.0  # guarantees sum > S
    hi = x.max() + 1.0  # guarantees sum < S
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def smooth_max_val_grad(h, beta):

    weights = exp_vals / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    grad = np.zeros(n)
    active = np.where(weights > 1e-7)[0]
