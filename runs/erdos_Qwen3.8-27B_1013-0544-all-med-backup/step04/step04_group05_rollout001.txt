import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_smooth_max(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[::-1][:top_k]
    c_sel = corr[idx]
    c_max = c_sel.max()
    w = np.exp(beta * (c_sel - c_max))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, i in zip(w, idx):
        if wi < 1e-14:
            continue
        lag = int(i) - (n - 1)
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wi * (1.0 - h[:-lag])
            grad[:n - lag] -= wi * h[lag:]
        else:
            m = -lag
            grad[:n - m] += wi * (1.0 - h[m:])
            grad[m:] -= wi * h[:n - m]
    return grad * dx

    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))

        c = np.correlate(h_in, 1.0 - h_in, mode="full")
        m = c.max()
        return m + np.log(np.sum(np.exp(np.clip(beta * (c - m), -500, 500)))) / beta

    def obj_grad(h_in):
        return grad_smooth_max(h_in, beta, top_k)

        return h_in.sum() - target

    def con_jac(h_in):
        return np.ones(n)

    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con, 'jac': con_jac}],
                       options={'maxiter': max_iter, 'ftol': 1e-18})
        h_new = project_h(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0, compute_c5(h0)

    best = h0.copy()
    best_c5 = compute_c5(best)
    x = best.copy()
    y = best.copy()
    vel = np.zeros(n)
    t_n = 1.0
    no_imp = 0
    for it in range(max_iters):
        if time_fn() < 3:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_n * t_n)) / 2.0
        alpha = (t_n - 1.0) / t_next
        mom = p0 + (p1 - p0) * min(1.0, it / (0.5 * max_iters))
        y = x + alpha * (y - x + vel)
        y = project_h(y, target)

        step = step_init / (1.0 + 0.003 * it)

        c5_new = compute_c5(x)
