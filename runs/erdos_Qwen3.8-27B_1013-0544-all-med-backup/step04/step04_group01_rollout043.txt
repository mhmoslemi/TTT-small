def lag_gradient(h,n,l,dx):
    G=np.zeros(n)
    if l>=0:
       r=n-l
       if r>0:
          G[:r] += dx*(1.0-h[l:])
          G[l:] -= dx*h[:r]
    else:
       a=-l
       r=n-a
       if r>0:
          G[a:] += dx*(1.0-h[:r])
          G[:r] -= dx*h[a:]
    return G

def project_feasible(h,n):
    h=np.array(h,dtype=float,copy=True)
    h=np.clip(h,0.0,1.0)
    target=n*0.5
    diff=target-h.sum()
    if abs(diff)<1e-12:
       return h
    lo=float(h.min())-1.0
    hi=float(h.max())
    for _ in range(80):
       mid=0.5*(lo+hi)
       s=np.sum(np.clip(h-mid,0.0,1.0))
       if s>target: lo=mid
       else: hi=mid
    lam=0.5*(lo+hi)
    h=np.clip(h-lam,0.0,1.0)
    diff=target-h.sum()
    if abs(diff)>1e-13:
       if diff>0:
          mask=h < 1.0-1e-12
          cnt=float(mask.sum())
          if cnt>0:
             h[mask]+=diff/cnt
       else:
          mask=h > 1e-12
          cnt=float(mask.sum())
          if cnt>0:
             h[mask]+=diff/cnt
    h=np.clip(h,0.0,1.0)
    return h

def _cand_key(x):

def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    n = int(n)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-12:
        return h
    lo = float(h.min()) - 1.0
    hi = float(h.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = float(mask.sum())
            if cnt > 0:
                h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = float(mask.sum())
            if cnt > 0:
                h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def compute_C(h, n):
    dx = 2.0 / n

def eval_c5(h, n):

def row_D(i, h, n, dx):
    L = 2 * n - 1
    m = n - 1
    D = np.zeros(L)
    D[m - i:m + n - i] += dx * (1.0 - h)
    D[m + i - n + 1:m + i + 1] -= dx * h[::-1]
    return D


def apply_move_inplace(C, h, i, j, dd, n, m, dx):
    a = dd * dx
    oh = 1.0 - h
    hr = h[::-1]
    C[m - i:m + n - i] += a * oh
    C[m + i - n + 1:m + i + 1] -= a * hr
    C[m - j:m + n - j] -= a * oh
    C[m + j - n + 1:m + j + 1] += a * hr
    if i != j:
        d2 = dd * dd
        p = m + i - j
        q = m + j - i
        L = C.size
        if 0 <= p < L:
            C[p] += d2
        if q != p and 0 <= q < L:
            C[q] += d2


def lag_gradient(h, n, l, dx):
    G = np.zeros(n)
    if l >= 0:
        r = n - l
        if r > 0:
            G[:r] += dx * (1.0 - h[l:])
            G[l:] -= dx * h[:r]
    else:
        a = -l
        r = n - a
        if r > 0:
            G[a:] += dx * (1.0 - h[:r])
            G[:r] -= dx * h[a:]
    return G


def active_set(h, C, K, cmax):

    m = n - 1

    W = np.empty((len(idx), n), dtype=float)
    for r, k in enumerate(idx):

    return idx, W


def rank_candidates(h, C, n, cmax, K, rank_deltas, top, dist, diag):

def delta_max(C, E, delta, idx1, idx2):
    if delta == 0.0:
        return float(C.max())
    tmp = C + delta * E
    d2 = delta * delta
    L = C.size
    if 0 <= idx1 < L:
        tmp[idx1] += d2
    if idx2 != idx1 and 0 <= idx2 < L:
        tmp[idx2] += d2
    return float(tmp.max())

def random_search(h, C, n, deadline, rng):

    C = np.array(C, dtype=float, copy=True)

    best_C = C.copy()
    best_c = float(C.max())
    cmax = best_c
    T = 0.0005
    deltas = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    moves = 0
    while time.time() < deadline:
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        d = float(rng.choice(deltas))
        if h[i] + d > 1.0 + 1e-12 or h[j] - d < -1e-12:
            continue
        apply_move_inplace(C, h, i, j, d, n, m, dx)
        h[i] += d
        h[j] -= d
        cn = float(C.max())
        if cn < cmax - 1e-13:
            cmax = cn
            accept = True
        else:
            accept = (T > 1e-9 and (cn - cmax) < 2.5 * T and
                      float(rng.rand()) < math.exp(-((cn - cmax) / T)))
        if accept:
            if cn < best_c:
                best_c = cn
                best_h = h.copy()
                best_C = C.copy()
        else:
            apply_move_inplace(C, h, j, i, d, n, m, dx)
            h[i] -= d
            h[j] += d
        moves += 1
        if moves % 2000 == 0:
            T *= 0.85
            if moves % 5000 == 0:
                C = compute_C(h, n)
                cmax = float(C.max())
                if cmax < best_c:

                    best_h = h.copy()
