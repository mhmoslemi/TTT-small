        best_h, best_c5 = do_subgrad(best_h, 40000, rem, top_k=12)

    # Phase 3: Swap descent
    if rem() > 30:
        best_h, best_c5 = do_swap_descent(best_h, rem, n_rounds=4)

    # Phase 4: Multi-resolution
    n_base = len(best_h)
    for n_try in [200, 250, 300, 400, 500, 600]:
        if rem() < 70:
            break
        if n_try == n_base:
            continue
        x_old = np.linspace(0, 2, n_base)
        x_new = np.linspace(0, 2, n_try)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_h(h_try, n_try / 2.0)
        h_new, c_new = do_slsqp(h_try, [200, 2000, 15000, 60000],
                                 3000, rem)
        if rem() > 20:
            h_new, c_new = do_subgrad(h_new, 20000, rem, top_k=10)
        if c_new < best_c5:
            best_h, best_c5 = h_new, c_new
            n_base = n_try

    # Phase 5: SLSQP high-beta refinement
    if rem() > 40:
        h2, c2 = do_slsqp(best_h, [20000, 80000, 200000], 6000, rem)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 6: Final subgradient + swap
    if rem() > 15:
        best_h, best_c5 = do_subgrad(best_h, 20000, rem, top_k=6)
    if rem() > 10:
        best_h, best_c5 = do_swap_descent(best_h, rem, n_rounds=3)

    # Phase 7: Final SLSQP
    if rem() > 20:
        h2, c2 = do_slsqp(best_h, [50000, 200000], 8000, rem)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Ensure exact constraint satisfaction
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
