import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def lbfgs_optimize(x0, n, dx, ts, beta, pen, bounds, maxiter=30000):
    def obj(x, b=beta, p=pen):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        return smooth_max_val(corr, b) + p * (x.sum() - ts) ** 2
    try:
        res = minimize(obj, x0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'iprint': 0})
        return res.x
    except Exception:
        return x0

def subgrad_polish(h, n, dx, ts, deadline, seed, max_iter=30000):
    np.random.seed(seed)
    c5 = compute_c5(h)
    velocity = np.zeros(n)
    lr = 0.0015
    stagnant = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = min(5, len(corr))
        top_idx = np.argsort(corr)[-top_k:]
        w = np.exp(25.0 * (corr[top_idx] - corr[top_idx].max()))
        w /= w.sum()
        grad = np.zeros(n)
        for j, k in enumerate(top_idx):
            lag = k - (n - 1)
            if lag >= 0:
                m = n - lag
                idx = np.arange(m)
                grad[idx] += w[j] * (1.0 - h[idx + lag])
                grad[idx + lag] -= w[j] * h[idx]
            else:
                la = -lag
                m = n - la
                idx = np.arange(m)
                grad[idx + la] += w[j] * (1.0 - h[idx])
                grad[idx] -= w[j] * h[idx + la]
        velocity = 0.83 * velocity - lr * grad
        h_new = project_feasible(h + velocity, ts)
        c_new = compute_c5(h_new)
        if c_new < c5 - 1e-13:
            h = h_new
            c5 = c_new
            stagnant = 0
            lr = min(lr * 1.002, 0.004)
        else:
            stagnant += 1
            velocity *= 0.45
            lr *= 0.91
        if stagnant > 800 and lr < 3e-7:
            break
    return h, c5

def optimize_at_resolution(h0, n, deadline, seed):
    dx = 2.0 / n
    ts = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best = project_feasible(h0, ts)
    best_c5 = compute_c5(best)

    # Coarse-to-fine beta schedule with L-BFGS-B
    betas = [30, 100, 500, 2000, 10000, 50000, 200000, 1000000]
    pens = [50.0, 100.0, 500.0, 1000.0, 5000.0, 10000.0, 50000.0, 100000.0]
    n_restarts = [3, 3, 2, 2, 2, 2, 3, 3]

    for bi, beta in enumerate(betas):
        if time.time() > deadline:
            break
        for r in range(n_restarts[bi]):
            if time.time() > deadline:
                break
            x0 = best.copy()
            if r > 0:
                if r == 1:
                    x0 += np.random.randn(n) * 0.004
                else:
                    freq = np.random.randint(1, n // 2 + 1)
                    ph = np.random.uniform(0, 2 * np.pi)
                    amp = 0.008 * r
                    x0 += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
                x0 = project_feasible(x0, ts)
            mx = 50000 if beta < 5000 else 20000
            res_x = lbfgs_optimize(x0, n, dx, ts, beta, pens[bi], bounds, mx)
            h_new = project_feasible(res_x, ts)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(200, 0.5)
    pn = len(h_parent)

    best_h = project_feasible(h_parent, pn / 2.0)
    best_c5 = compute_c5(best_h)
    best_n = pn

    # Phase 1: Primary optimization at parent resolution (60% time)
    d1 = t0 + budget_s * 0.60
    h_opt, c_opt = optimize_at_resolution(h_parent.copy(), pn, d1, seed)
    if c_opt < best_c5:
        best_h, best_c5, best_n = h_opt, c_opt, pn

    # Subgradient polish at parent resolution
    if time.time() < d1:
        h_sp, c_sp = subgrad_polish(best_h, best_n, 2.0 / best_n,
                                     best_n / 2.0, d1, seed + 500, max_iter=20000)
        if c_sp < best_c5:
            best_h, best_c5 = h_sp, c_sp

    # Phase 2: Try nearby resolutions (25% time)
    d2 = t0 + budget_s * 0.85
    x_old = np.linspace(0, 2, pn, endpoint=False)
    candidate_ns = []
    for d in [-20, -10, -5, 5, 10, 20, 40]:
        nn = pn + d
        if 100 <= nn <= 600 and nn != pn:
            candidate_ns.append(nn)
    for nn in candidate_ns[:6]:
        if time.time() > d2:
            break
        x_new = np.linspace(0, 2, nn, endpoint=False)
        h_init = np.interp(x_new, x_old, h_parent)
        h_init = project_feasible(h_init, nn / 2.0)
        h_n, c_n = optimize_at_resolution(h_init, nn, d2, seed + nn)
        if c_n < best_c5:
            best_h, best_c5, best_n = h_n, c_n, nn

    # Phase 3: Final high-precision L-BFGS-B (10% time)
    d3 = t0 + budget_s * 0.95
    if best_h is not None and time.time() < d3:
        n = best_n
        dx = 2.0 / n
        ts = n / 2.0
        bounds = [(0.0, 1.0)] * n
        for beta in [500000, 2000000, 10000000]:
            if time.time() > d3:
                break
            x0 = best_h.copy()
            res_x = lbfgs_optimize(x0, n, dx, ts, beta, 200000.0, bounds, 80000)
            h_new = project_feasible(res_x, ts)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 4: Final micro-polish (5% time)
    d4 = t0 + budget_s * 0.99
    if best_h is not None and time.time() < d4:
        n = best_n
        h_sp, c_sp = subgrad_polish(best_h, n, 2.0 / n, n / 2.0,
                                     d4, seed + 9999, max_iter=50000)
        if c_sp < best_c5:
            best_h, best_c5 = h_sp, c_sp

    # Final verification
    if best_h is None:
        best_h = np.full(100, 0.5)
        best_n = 100
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
