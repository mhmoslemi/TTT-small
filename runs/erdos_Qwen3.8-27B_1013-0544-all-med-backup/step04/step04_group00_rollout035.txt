def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):

        if diff > 0:

            s = room.sum()

            s = h.sum()
            if s < 1e-14:
                break

                    b = rng.choice(zi)
                    current_h[a], current_h[b] = 0.0, 1.0
        h = current_h.copy()
        corr = np.correlate(h, 1 - h, mode='full').copy()
        cur_c = corr.max() * dx
        ones = set(np.where(h == 1)[0].tolist())
        zeros = set(np.where(h == 0)[0].tolist())

        for sweep in range(max_sweeps):
            if time.time() > deadline:
                break
            improved = False
            z_list = sorted(zeros)
            for i in z_list:
                if time.time() > deadline:
                    break
                o_list = sorted(ones)
                for j in o_list:
                    im = i + m_range
                    imn = i - m_range
                    jm = j + m_range
                    jmn = j - m_range
                    delta = np.zeros(2 * n - 1)
                    msk = (im >= 0) & (im < n)
                    delta[msk] += 1.0 - h[im[msk]]
                    msk = (imn >= 0) & (imn < n)
                    delta[msk] -= h[imn[msk]]
                    msk = (jm >= 0) & (jm < n)
                    delta[msk] -= 1.0 - h[jm[msk]]
                    msk = (jmn >= 0) & (jmn < n)
                    delta[msk] += h[jmn[msk]]
                    # Cross terms for the bilinear correction
                    delta[n - 1] -= 2.0
                    k1 = n - 1 + (j - i)
                    delta[k1] += 1.0
                    k2 = n - 1 + (i - j)
                    delta[k2] += 1.0
                    new_corr = corr + delta
                    nc = new_corr.max() * dx
                    if nc < cur_c - 1e-15:
                        h[i], h[j] = 1.0, 0.0
                        corr = new_corr
                        cur_c = nc
                        ones.discard(j)
                        ones.add(i)
                        zeros.discard(i)
                        zeros.add(j)
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            current_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: SLSQP with increasing alpha
    p1 = t0 + 0.12 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [500, 2000, 10000, 50000, 200000, 1000000, 5000000]:
        if time.time() > p1:
            break
        h = slsqp_smooth(h, n, alpha, 1500, p1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Fast binary local search with restarts
    p2 = t0 + 0.42 * (budget_s - 5)
    hb, cb = fast_binary_search(best_h, n, p2, max_sweeps=800, n_restarts=5, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 3: SLSQP polish from binary solution
    p3 = t0 + 0.52 * (budget_s - 5)
    if time.time() < p3:
        hp = project_feasible(best_h + 0.05 * rng.randn(n), n)
        for alpha in [50000, 200000, 1000000]:
            if time.time() > p3:
                break
            hp = slsqp_smooth(hp, n, alpha, 2000, p3)
            c = compute_c5(hp, n)
            if c < best_c:
                best_c = c
                best_h = hp.copy()

    # Phase 4: Multi-resolution exploration
    p4 = t0 + 0.68 * (budget_s - 5)
    for nh in [200, 300, 350]:
        if time.time() > p4 - 30:
            break
        if nh == n:
            continue
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        hh = slsqp_smooth(hh, nh, 50000, 500, p4 - 20)
        ch = compute_c5(hh, nh)
        hb2, cb2 = fast_binary_search(hh, nh, p4 - 10, max_sweeps=300, n_restarts=2, seed=seed+1)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()
            n = nh
        elif ch < best_c:
            best_c = ch
            best_h = hh.copy()
            n = nh

    # Phase 5: Extended binary search (most of remaining time)
    p5 = dl - 30
    hb, cb = fast_binary_search(best_h, n, p5, max_sweeps=3000, n_restarts=5, seed=seed+2)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 6: Final SLSQP + binary polish
    if time.time() < dl - 10:
        h = slsqp_smooth(best_h.copy() + 0.02 * rng.randn(n), n, 5000000, 3000, dl - 8)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if time.time() < dl - 3:
            hb, cb = fast_binary_search(best_h, n, dl, max_sweeps=500, n_restarts=2, seed=seed+3)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
