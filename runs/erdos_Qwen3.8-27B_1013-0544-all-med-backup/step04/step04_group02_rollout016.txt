def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def power_mean_obj(h, dx, p, mu, ts):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m <= 1e-10:
        pm = 0.0
    else:
        s = np.maximum(corr, 0.0) / m
        pm = m * (np.mean(s ** p)) ** (1.0 / p)
    pen = mu * (h.sum() - ts) ** 2
    return pm * dx + pen
