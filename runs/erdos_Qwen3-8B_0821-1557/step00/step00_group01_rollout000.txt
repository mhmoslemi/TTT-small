import numpy as np
import scipy.optimize as optimize
from scipy.signal import correlate
import time

initial_h_values = np.load('initial_h_values.npy')  # Preloaded initial solution

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    n_points = 512
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = np.interp(np.linspace(0, 2, n_points), np.sort(np.unique(np.random.rand(n_points) * 2)), h)
    h = (h * n_points / 2).astype(np.float64)

        h = x.reshape(n_points)
        overlap = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint}]
    
    result = optimize.minimize(
        objective,
        h.flatten(),
        method='SLSQP',
        constraints=constraints,
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    h_opt = result.x.reshape(n_points)
    c5_bound = result.fun
    return h_opt, c5_bound, n_points
