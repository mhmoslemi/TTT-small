import numpy as np
import random
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points, can be increased if needed
    dx = 2.0 / n_points
    
    # Initial guess from provided initial_h_values or random if not available
    if initial_h_values is not None and len(initial_h_values) == n_points:
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2)  # Ensure integral is 1
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        # Ensure integral is 1
        norm = np.sum(h) * dx
        if norm == 0:
            return np.inf
        h = h * (1 / norm) * (n_points / 2)
        # Compute overlap integral
        h_1_minus = 1 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        c5 = np.max(correlate) * dx
        return c5
    
    def evaluate(h_flat):
        return objective(h_flat)
    
    def generate_child(parent1, parent2):
        # Simple crossover and mutation
        child = np.copy(parent1)
        crossover_point = np.random.randint(1, n_points - 1)
        child[crossover_point:] = parent2[crossover_point:]
        mutation_rate = 0.1
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                child[i] = np.random.rand()
        child = np.clip(child, 0, 1)
        norm = np.sum(child) * dx
        child = child * (1 / norm) * (n_points / 2)
        return child
    
    # Genetic algorithm
    population_size = 20
    population = [np.random.rand(n_points) for _ in range(population_size)]
    population = [np.clip(p, 0, 1) for p in population]
    population = [p * (n_points / 2) / (np.sum(p) * dx) for p in population]
    
    for _ in range(100):  # Max generations
        if time.time() - start_time > budget_s:
            break
        # Evaluate population
        fitness = [evaluate(p) for p in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Generate new population
        new_population = []

            parent1, parent2 = random.choices(population, k=2)
            child1 = generate_child(parent1, parent2)
            child2 = generate_child(parent2, parent1)
            new_population.extend([child1, child2])
        
        # Replace old population
        population = new_population
    
    # Final evaluation
    best_h = np.array(population[np.argmin([evaluate(p) for p in population])])
    best_c5 = evaluate(best_h)
    return best_h, best_c5, n_points
