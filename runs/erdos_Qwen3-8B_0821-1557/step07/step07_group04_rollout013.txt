import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    # Normalize function values
    def normalize_h(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    # Initialize population
    population_size = 40
    population = [normalize_h(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    
    for _ in range(200):
        if time.time() - time.time() > budget_s:
            break
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        # Select best individual
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Local optimization on best solution
        result = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint_integral}],
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        if result.success:
            best_h = result.x
            best_c5 = result.fun
        
        # Genetic operations
        # Select parents
        parent_indices = np.argsort(fitness)[:5]
        parents = [population[i] for i in parent_indices]
        
        # Crossover
        offspring = []
        for _ in range(population_size // 2):
            parent1 = random.choice(parents)
            parent2 = random.choice(parents)
            child = np.zeros(n_points)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            offspring.append(normalize_h(child))
        
        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if random.random() < 0.1:
                    offspring[i][j] += np.random.uniform(-0.1, 0.1)
                    offspring[i][j] = np.clip(offspring[i][j], 0, 1)
        
        # Replace population
        population = offspring + [best_h]
    
    # Final evaluation
    final_fitness = [objective(ind) for ind in population]
    best_idx = np.argmin(final_fitness)
    best_h = population[best_idx]
    best_c5 = final_fitness[best_idx]
    
    return best_h.tolist(), best_c5, n_points
