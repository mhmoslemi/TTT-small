import time

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    # Initialize population
    pop_size = 50
    population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
    for h in population:
        h = normalize_h(h)
    
    def normalize_h(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    # Genetic algorithm parameters
    mutation_rate = 0.1
    elite_ratio = 0.2
    generations = 100
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        
        # Select elites
        elite_idx = np.argsort(fitness)[:int(elite_ratio * pop_size)]
        elites = [population[i] for i in elite_idx]
        
        # Tournament selection
        new_population = elites.copy()
        while len(new_population) < pop_size:
            # Select two random individuals
            a, b = np.random.choice(pop_size, 2, replace=False)
            if fitness[a] < fitness[b]:
                selected = population[a]
            else:
                selected = population[b]
            
            # Apply mutation
            mutated = selected.copy()
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    mutated[i] = np.random.uniform(0, 1)
            mutated = normalize_h(mutated)
            
            new_population.append(mutated)
        
        population = new_population
    
    # Find best solution
    best_h = min(population, key=objective)
    best_c5 = objective(best_h)
    
    return best_h.tolist(), best_c5, n_points
