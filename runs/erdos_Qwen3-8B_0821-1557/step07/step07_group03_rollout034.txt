import time

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)

        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initialize population
    population_size = 20
    population = [initial_h.copy() for _ in range(population_size)]
    population = [normalize(h) for h in population]
    fitnesses = np.array([evaluate(h) for h in population])
    
    t0 = time.time()
    
    # Optimization parameters
    mutation_rate = 0.3
    mutation_strength = 0.05
    crossover_rate = 0.7
    elite_ratio = 0.2
    temp = 1.0
    temp_decay = 0.995
    
    for _ in range(500):
        if time.time() - t0 > budget_s:
            break
        
        # Evaluate fitness
        fitnesses = np.array([evaluate(h) for h in population])
        
        # Select elites
        elite_idx = np.argsort(fitnesses)[:int(elite_ratio * population_size)]
        elites = [population[i] for i in elite_idx]
        
        # Generate new population
        new_population = elites.copy()
        
        for _ in range(population_size - len(elites)):
            # Select parents
            parent1 = np.random.choice(elites)
            parent2 = np.random.choice(elites)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = np.random.choice(elites, size=n_points)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, mutation_strength, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
            
            # Normalize
            child = normalize(child)
            new_population.append(child)
        
        population = new_population
        fitnesses = np.array([evaluate(h) for h in population])
        
        # Simulated annealing
        for i in range(len(population)):
            if np.random.rand() < 0.2:
                perturb = np.random.normal(0, 0.05, n_points)

                if candidate_c5 < fitnesses[i] or np.random.rand() < np.exp((fitnesses[i] - candidate_c5) / temp):
                    population[i] = candidate
                    fitnesses[i] = candidate_c5
        
        temp *= temp_decay
    
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx]
    best_c5 = fitnesses[best_idx]
    
    return best_h, best_c5, n_points
