    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

        return evaluate(h)

    t0 = time.time()
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    mutation_strength = 0.05
    elite_ratio = 0.2
    crossover_rate = 0.8
    
    for gen in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        # Create population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [normalize(p) for p in population]
        
        # Evaluate population
        fitness = [evaluate(p) for p in population]
        
        # Select elites
        elite_indices = np.argsort(fitness)[:int(elite_ratio * population_size)]
        elites = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elites.copy()
        
        while len(new_population) < population_size:
            # Choose parents
            parent1 = np.random.choice(elite_indices)
            parent2 = np.random.choice(elite_indices)
            parent1 = population[parent1]
            parent2 = population[parent2]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                child = np.zeros_like(parent1)
                for i in range(n_points):
                    if np.random.rand() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.normal(0, mutation_strength)
                        child[i] = np.clip(child[i], 0, 1)
            
            # Normalize
            child = normalize(child)
            new_population.append(child)
        
        # Update best solution
        for candidate in new_population:
            current_c5 = evaluate(candidate)
            if current_c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = current_c5
        
        # Update population
        population = new_population
    
    return best_h, best_c5, n_points
