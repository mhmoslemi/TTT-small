def run(seed=42, budget_s=10, **kwargs):

    # Fitness function
    def fitness(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    # Constraint enforcement
    def constraint(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    # Normalize function
    def normalize(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    # Initialize population
    population_size = 50
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    
    # Main loop
    start_time = time.time()
    best_solution = population[0]
    best_fitness = fitness(best_solution)
    
    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness_values = [fitness(ind) for ind in population]
        
        # Tournament selection
        selected = []
        for _ in range(population_size):
            candidates = random.sample(range(population_size), 5)
            best_candidate = min(candidates, key=fitness_values.__getitem__)
            selected.append(population[best_candidate])
        
        # Crossover
        new_population = []
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                parent1 = selected[i]
                parent2 = selected[i+1]
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
                child2 = np.concatenate((parent2[:crossover_point], parent1[crossover_point:]))
                new_population.extend([child1, child2])
            else:
                new_population.append(selected[i])
        
        # Mutation
        for i in range(len(new_population)):
            if np.random.rand() < 0.1:
                mutation = np.random.uniform(-0.1, 0.1, size=n_points)
                new_population[i] = normalize(new_population[i] + mutation)
        
        # Replace population
        population = new_population
        
        # Update best solution
        for ind in population:
            current_fitness = fitness(ind)
            if current_fitness < best_fitness:
                best_fitness = current_fitness
                best_solution = ind
    
    return best_solution.tolist(), best_fitness, n_points
