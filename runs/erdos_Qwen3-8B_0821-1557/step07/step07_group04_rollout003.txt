import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    # Initialize population
    population_size = 30
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    
    start_time = time.time()
    
    for generation in range(200):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Selection
        selected = np.random.choice(population_size, size=population_size, p=[f / sum(fitness) for f in fitness])
        selected = [population[i] for i in selected]
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if i+1 < population_size:
                parent1 = selected[i]
                parent2 = selected[i+1]
                child1 = parent1 * 0.5 + parent2 * 0.5
                child2 = parent1 * 0.5 + parent2 * 0.5
                offspring.extend([normalize(child1), normalize(child2)])
            else:
                offspring.append(normalize(selected[i]))
        
        # Mutation
        for i in range(len(offspring)):
            if random.random() < 0.1:
                mutation = np.random.uniform(-0.1, 0.1, size=n_points)
                offspring[i] = normalize(offspring[i] + mutation)
        
        # Replace population
        population = offspring
    
    return best_h.tolist(), best_c5, n_points
