def run(seed=42, budget_s=10, **kwargs):

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    # Initial population
    population_size = 50
    population = [np.array(initial_h, dtype=np.float64) for _ in range(population_size)]
    population = [h / (np.sum(h) * dx) * (n_points / 2) for h in population]
    
    def normalize_h(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    # Genetic algorithm parameters
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 200
    start_time = time.time()
    
    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Create new population
        new_population = []
        
        for _ in range(population_size // 2):
            # Select parents
            parent1, parent2 = random.choices(population, k=2)
            
            # Crossover
            if random.random() < crossover_rate:
                split = np.random.randint(1, n_points - 1)
                child1 = np.concatenate((parent1[:split], parent2[split:]))
                child2 = np.concatenate((parent2[:split], parent1[split:]))

                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for i in range(n_points):

            # Normalize
            child1 = normalize_h(child1)

            new_population.extend([child1, child2])
        
        # Replace population
        population = new_population[:population_size]

    return best_h.tolist(), best_c5, n_points
