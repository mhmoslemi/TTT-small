        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initial solution
    h = initial_h.copy()
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.15
    mutation_strength = 0.1
    elite_ratio = 0.2
    
    # Initialize population
    population = [normalize(np.random.rand(n_points)) for _ in range(population_size)]
    population = [np.clip(p, 0, 1) for p in population]
    
    for gen in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        # Evaluate population
        fitness = [evaluate(p) for p in population]
        
        # Find the best solution
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_c5 = fitness[best_idx]
        
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
        
        # Select elites
        elite_size = int(elite_ratio * population_size)
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elites.copy()
        
        while len(new_population) < population_size:
            # Select parents
            parent1 = np.random.choice(elite_indices)
            parent2 = np.random.choice(elite_indices)
            p1 = population[parent1]
            p2 = population[parent2]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
            
            # Mutation
            for i in range(n_points):

                    child[i] += np.random.uniform(-mutation_strength, mutation_strength)
                    child[i] = np.clip(child[i], 0, 1)
            
            new_population.append(normalize(child))
        
        population = new_population
