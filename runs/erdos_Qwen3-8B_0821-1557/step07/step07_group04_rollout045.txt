def run(seed=42, budget_s=10, **kwargs):

        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    # Initial guess

    # Function to normalize h
    def normalize_h(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    # Genetic algorithm parameters
    population_size = 50
    mutation_rate = 0.1
    generations = 100
    elite_size = 5
    best_h = h_initial.copy()
    best_c5 = objective(best_h)

    start_time = time.time()

    # Initialize population
    population = [normalize_h(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        # Select parents
        parent_indices = np.random.choice(len(population), size=population_size - elite_size, replace=False)
        parents = [population[i] for i in parent_indices]
        # Crossover
        offspring = []
        for i in range(len(parents) // 2):
            p1 = parents[2*i]
            p2 = parents[2*i + 1]
            child = np.zeros_like(p1)
            for j in range(n_points):
                if random.random() < 0.5:
                    child[j] = p1[j]
                else:
                    child[j] = p2[j]
            offspring.append(child)
        # Mutate
        for i in range(len(offspring)):
            child = offspring[i]
            for j in range(n_points):
                if random.random() < mutation_rate:
                    child[j] += np.random.uniform(-0.1, 0.1)
                    child[j] = np.clip(child[j], 0, 1)
            offspring[i] = child
        # Combine and normalize
        population = elites + offspring
        population = [normalize_h(ind) for ind in population]
        # Update best solution
        for ind in population:
            current_c5 = objective(ind)
            if current_c5 < best_c5:
                best_h = ind.copy()
                best_c5 = current_c5

    # Local search
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break

        # Check if perturbed_h is valid

            if perturbed_c5 < best_c5:

                best_c5 = perturbed_c5
