import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initial solution
    h = initial_h.copy()
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Genetic algorithm parameters
    pop_size = 50
    generations = 100
    mutation_rate = 0.1
    tournament_size = 5
    
    for gen in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        # Create population
        population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
        
        # Evaluate population
        fitness = [evaluate(h) for h in population]
        
        # Select elite
        elite_index = np.argmin(fitness)
        elite = population[elite_index].copy()
        elite_fitness = fitness[elite_index]
        
        # Tournament selection
        new_population = []
        for _ in range(pop_size):
            candidates = np.random.choice(population, tournament_size, replace=False)
            winner = min(candidates, key=lambda x: evaluate(x))
            new_population.append(winner)
        
        # Crossover and mutation
        for i in range(pop_size):
            if np.random.rand() < 0.8:
                # Crossover
                j = np.random.randint(0, pop_size)
                parent1 = new_population[i]
                parent2 = new_population[j]
                child = np.zeros_like(parent1)
                split = np.random.randint(1, n_points - 1)
                child[:split] = parent1[:split]
                child[split:] = parent2[split:]
                new_population[i] = child
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, 0.05, n_points)
                new_population[i] += mutation
                new_population[i] = np.clip(new_population[i], 0, 1)
        
        # Update best solution
        for h in new_population:
            current_c5 = evaluate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        
        # Replace population with new_population
        population = new_population
        
        # Replace with elite if better
        if elite_fitness < best_c5:
            best_h = elite.copy()
            best_c5 = elite_fitness
    
    return best_h, best_c5, n_points
