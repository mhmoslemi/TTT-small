        return np.max(corr * dx)

        return np.sum(h) - n_points / 2

    def normalize_h(h):

        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    elite_size = 2

    # Initialize population

    best_h = h_initial.copy()

        fitness = [objective(ind) for ind in population]

        elite_indices = np.argsort(fitness)[:elite_size]

        for ind in population:
