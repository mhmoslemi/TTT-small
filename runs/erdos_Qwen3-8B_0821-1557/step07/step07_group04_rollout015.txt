import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    def normalize_h(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    # Generate initial population
    population_size = 20
    population = [normalize_h(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    population.append(normalize_h(initial_h))
    population = [normalize_h(h) for h in population]
    population = list(set(population))  # Remove duplicates
    population = population[:population_size]

    start_time = time.time()
    best_h = population[0]
    best_c5 = objective(best_h)

    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Select parents
        parents = np.random.choice(population, size=4, replace=False)
        # Crossover
        child = np.zeros(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parents[0][i]
            else:
                child[i] = parents[1][i]
        # Mutate
        mutation_rate = 0.1
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                child[i] = np.clip(child[i] + np.random.uniform(-0.1, 0.1), 0, 1)
        child = normalize_h(child)
        # Evaluate
        c5 = objective(child)
        # Replace worst
        if c5 < best_c5:
            best_h = child
            best_c5 = c5
        # Update population
        population.append(child)
        population = sorted(population, key=lambda x: objective(x))[:population_size]

    return best_h.tolist(), best_c5, n_points
