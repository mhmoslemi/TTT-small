    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]

    for gen in range(generations):

        fitness = [evaluate_c5(h) for h in population]
        
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Select parents for crossover
        parent_indices = np.argsort(fitness)[:population_size // 2]
        parents = [population[i] for i in parent_indices]
        
        # Create new population
        new_population = elites.copy()
        
        # Perform crossover and mutation
        for _ in range(len(new_population) - elite_size):
            # Randomly select two parents
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
