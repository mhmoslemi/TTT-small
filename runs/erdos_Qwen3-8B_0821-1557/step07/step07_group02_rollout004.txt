import time

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)

    best_c5 = evaluate(h)

                h = np.random.uniform(0, 1, n_points)

                h = best_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)

            population.append(h)
        
        # Evaluate population
        fitness = [evaluate(h) for h in population]

        best_pop_c5 = fitness[best_idx]
        
        # Keep elite solutions
        elite_indices = np.argsort(fitness)[:elite_size]

        for i in range(len(elite_h)):
