import numpy as np
import scipy.optimize as optimize
from scipy.signal import correlate

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    
    # Initial construction based on a periodic function
    n_points = 100
    h_initial = np.zeros(n_points)
    dx = 2.0 / n_points
    for i in range(n_points):
        x = i * dx
        h_initial[i] = 0.5 + 0.5 * np.cos(2 * np.pi * x / 0.5) * 0.5
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)

        h = h / np.sum(h) * (n_points / 2)
        # Compute the overlap integral
        overlap = correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization with L-BFGS-B
    result = optimize.minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
