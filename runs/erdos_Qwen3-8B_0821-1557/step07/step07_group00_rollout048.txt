    def objective(x):

        # Normalize to satisfy the integral constraint
        h = h / np.sum(h) * (n_points / 2)
        # Compute the overlap integral
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)

    population_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    
    # Initialize population
    population = np.random.uniform(0.0, 1.0, size=(population_size, n_points))
    population = population / np.sum(population, axis=1, keepdims=True) * (n_points / 2)
    
    for generation in range(generations):
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        
        # Elitism: keep best solution
        elite_idx = np.argmin(fitness)
        if fitness[elite_idx] < best_c5:
            best_h = population[elite_idx].copy()
            best_c5 = fitness[elite_idx]
        
        # Select parents
        parents_idx = np.argsort(fitness)[:int(population_size/2)]
        parents = population[parents_idx]
        
        # Generate offspring
        offspring = []
        for _ in range(population_size - len(parents)):
            # Randomly select parents for crossover
            parent1, parent2 = np.random.choice(parents_idx, size=2, replace=False)
            p1, p2 = population[parent1], population[parent2]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate((p1[:crossover_point], p2[crossover_point:]))
            else:
                child = np.random.uniform(0.0, 1.0, size=n_points)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.uniform(-0.1, 0.1, size=n_points)
                child += mutation
            
            # Normalize
            child = child / np.sum(child) * (n_points / 2)
            child = np.clip(child, 0.0, 1.0)
            offspring.append(child)
        
        # Replace population
        population[parents_idx] = parents
        population[len(parents_idx):] = offspring
    
    return best_h, best_c5, n_points
