        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    def normalize_h(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    # Initial population
    population_size = 20
    population = [normalize_h(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]

        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]

        # Selection
        selected = np.random.choice(len(population), size=population_size, p=np.array(fitness)/np.sum(fitness))
        selected = population[selected]
        
        # Crossover
        offspring = []
        for i in range(population_size // 2):
            parent1 = selected[i]
            parent2 = selected[population_size - 1 - i]
            child1 = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
