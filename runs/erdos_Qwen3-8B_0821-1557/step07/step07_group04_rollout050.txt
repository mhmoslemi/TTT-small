import random

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    def normalize_h(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    # Initial population
    population_size = 20
    population = [normalize_h(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    
    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        
        # Select top performers
        top_indices = np.argsort(fitness)[:population_size//2]
        elite = [population[i] for i in top_indices]
        
        # Create new population
        new_population = elite.copy()
        
        # Generate offspring through crossover
        for _ in range(population_size - len(elite)):
            parent1, parent2 = random.choices(elite, k=2)
            child = (parent1 + parent2) / 2
            child = normalize_h(child + np.random.normal(0, 0.05, n_points))
            new_population.append(child)
        
        population = new_population
    
    best_h = min(population, key=objective)
    best_c5 = objective(best_h)
    
    return best_h.tolist(), best_c5, n_points
