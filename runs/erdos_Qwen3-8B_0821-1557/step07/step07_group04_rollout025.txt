import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    # Initial guess
    h_initial = np.array(initial_h, dtype=np.float64)
    h_initial = h_initial / (np.sum(h_initial) * dx) * (n_points / 2)

    bounds = [(0, 1) for _ in range(n_points)]

    # Function to normalize h
    def normalize_h(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    # First optimization with SLSQP
    start_time = time.time()
    result = optimize.minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_integral}],
        options={'ftol': 1e-8, 'maxiter': 50}
    )

    best_h = result.x
    best_c5 = result.fun

    # Genetic algorithm for global search
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    elite_size = 2

    def fitness(h):
        return objective(h)

    population = [normalize_h(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    population = [h for h in population if np.all(h >= 0) and np.all(h <= 1)]

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness_scores = [fitness(h) for h in population]
        # Select elite
        elite_indices = np.argsort(fitness_scores)[:elite_size]
        elite = [population[i] for i in elite_indices]
        # Generate offspring
        offspring = []
        for _ in range(population_size - elite_size):
            # Selection
            p1 = np.random.choice(elite)
            p2 = np.random.choice(elite)
            # Crossover
            child = np.copy(p1)
            crossover_point = np.random.randint(1, n_points - 1)
            child[crossover_point:] = p2[crossover_point:]
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = np.clip(child[i] + np.random.normal(0, 0.05), 0, 1)
            offspring.append(child)
        # Replace population
        population = elite + offspring
        population = [normalize_h(h) for h in population if np.all(h >= 0) and np.all(h <= 1)]
        # Update best solution
        current_best = min(population, key=fitness)
        if fitness(current_best) < best_c5:
            best_h = current_best
            best_c5 = fitness(current_best)

    return best_h.tolist(), best_c5, n_points
