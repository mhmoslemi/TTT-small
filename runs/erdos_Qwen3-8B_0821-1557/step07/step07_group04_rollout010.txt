    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    # Initialize population
    population_size = 30
    population = [np.array(initial_h, dtype=np.float64) for _ in range(population_size)]
    for i in range(population_size):
        population[i] = population[i] / (np.sum(population[i]) * dx) * (n_points / 2)
    
    def normalize_h(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    start_time = time.time()
    best_h = np.array(initial_h, dtype=np.float64)

        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]

        # Selection
        selected = np.random.choice(population_size, size=population_size, p=np.array(fitness)/np.sum(fitness))
        selected = [population[i] for i in selected]
        
        # Crossover
        new_population = []
        for i in range(population_size//2):
            parent1 = selected[2*i]
            parent2 = selected[2*i+1]
            child1 = np.zeros_like(parent1)
            child2 = np.zeros_like(parent2)

                    child1[j] = parent1[j]

                    child1[j] = parent2[j]

        population = new_population
