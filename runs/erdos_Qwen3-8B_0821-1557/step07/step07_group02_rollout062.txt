import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    # Initialize with given starting point
    h = initial_h.copy()
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Simulated annealing parameters
    temp = 1.0
    cooling_rate = 0.995
    max_iterations = 1000
    
    for _ in range(max_iterations):
        if time.time() - t0 > budget_s:
            break
        
        # Generate a perturbed version
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturbed = h + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = normalize(h_perturbed)
        
        # Evaluate the perturbed solution
        c5_perturbed = evaluate(h_perturbed)
        
        # Accept worse solution with some probability
        if c5_perturbed < best_c5 or np.random.rand() < np.exp((best_c5 - c5_perturbed) / temp):
            h = h_perturbed
            best_h = h.copy()
            best_c5 = c5_perturbed
        
        temp *= cooling_rate
    
    # Local refinement with SLSQP
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
    res = optimize.minimize(
        objective,
        best_h,
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    best_h = res.x
    best_c5 = evaluate(best_h)
    
    return best_h, best_c5, n_points
