import numpy as np

import time

def run(seed=42, budget_s=10, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    # Initialize population
    population_size = 30
    population = [np.array(initial_h, dtype=np.float64) for _ in range(population_size)]
    for p in population:
        p = p / (np.sum(p) * dx) * (n_points / 2)
    
    def normalize_h(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    start_time = time.time()
    best_h = np.array(initial_h, dtype=np.float64)
    best_c5 = objective(best_h)
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        fitness = [objective(p) for p in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Selection
        selected = np.random.choice(population_size, size=population_size, p=np.array(fitness)/np.sum(fitness))
        selected = [population[i] for i in selected]
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if i+1 < population_size:
                parent1 = selected[i]
                parent2 = selected[i+1]
                child1 = np.concatenate([parent1[:n_points//2], parent2[n_points//2:]])
                child2 = np.concatenate([parent2[:n_points//2], parent1[n_points//2:]])
                offspring.append(child1)
                offspring.append(child2)
            else:
                offspring.append(selected[i])
        
        # Mutation
        for i in range(len(offspring)):
            if np.random.rand() < 0.1:
                perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
                mutated = normalize_h(offspring[i] + perturbation)
                offspring[i] = mutated
        
        # Replace population
        population = offspring
    
    return best_h.tolist(), best_c5, n_points
