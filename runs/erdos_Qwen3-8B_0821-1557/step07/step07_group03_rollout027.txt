    population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
    population = [normalize(h) for h in population]
    
    for generation in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        # Evaluate population
        fitness_scores = [fitness(h) for h in population]
        best_idx = np.argmax(fitness_scores)
        best_c5 = -fitness_scores[best_idx]
        
        # Update best solution
        if best_c5 < evaluate(best_h):
            best_h = population[best_idx].copy()
        
        # Select parents
        parents = np.array(population)[np.argsort(fitness_scores)][:-elite_size]
