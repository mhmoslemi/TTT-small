def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    # Initialize population
    population_size = 20
    population = [np.array(initial_h, dtype=np.float64) for _ in range(population_size)]
    
    # Normalize initial population
    for i in range(population_size):
        population[i] = population[i] / (np.sum(population[i]) * dx) * (n_points / 2)

    def normalize_h(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        
        # Find best solution
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Create new population
        new_population = []
        
        # Elitism: keep best solution
        new_population.append(best_h)
        
        # Create offspring through crossover and mutation
        for _ in range(population_size - 1):
            # Select two parents
            parent1, parent2 = np.random.choice(population, size=2, replace=False)
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation: focus on regions with high overlap
            overlap = np.correlate(child, 1 - child, mode="full") * dx
            max_overlap = np.max(overlap)
            if max_overlap > 0:
                # Find indices with high overlap
                high_overlap_indices = np.where(overlap > 0.9 * max_overlap)[0]
                if len(high_overlap_indices) > 0:
                    # Apply mutation to these indices
                    mutation = np.random.uniform(-0.2, 0.2, size=n_points)
                    mutation[high_overlap_indices] *= 2
                    child += mutation
                    child = np.clip(child, 0, 1)
            
            # Normalize
            child = normalize_h(child)
            new_population.append(child)
        
        # Update population
        population = new_population
    
    # Return best solution found
    return best_h.tolist(), best_c5, n_points
