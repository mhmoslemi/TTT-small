import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    # Initial population
    population_size = 20
    population = [np.array(initial_h, dtype=np.float64) for _ in range(population_size)]
    for i in range(population_size):
        population[i] = population[i] / (np.sum(population[i]) * dx) * (n_points / 2)

    def normalize_h(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    start_time = time.time()
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        
        # Select parents
        parents = np.argsort(fitness)[:population_size//2]
        parents = [population[i] for i in parents]
        
        # Crossover
        new_population = []
        for _ in range(population_size//2):
            if np.random.rand() < 0.8:
                # Single-point crossover
                pt = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parents[0][:pt], parents[1][pt:]])
                child2 = np.concatenate([parents[1][:pt], parents[0][pt:]])
            else:
                # Blend crossover
                child1 = (parents[0] + np.random.rand(n_points) * parents[1]) / 2
                child2 = (parents[1] + np.random.rand(n_points) * parents[0]) / 2
            new_population.append(normalize_h(child1))
            new_population.append(normalize_h(child2))
        
        # Mutation
        for i in range(len(new_population)):
            if np.random.rand() < 0.1:
                perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
                new_population[i] = normalize_h(new_population[i] + perturbation)
        
        # Replace population
        population = new_population[:population_size]
    
    best_h = min(population, key=objective)
    best_c5 = objective(best_h)
    
    return best_h.tolist(), best_c5, n_points
