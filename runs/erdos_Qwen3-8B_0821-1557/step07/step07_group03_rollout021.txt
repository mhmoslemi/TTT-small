import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    # Initial population
    population_size = 20
    population = [initial_h.copy() for _ in range(population_size)]
    population = [normalize(h) for h in population]
    fitness = np.array([evaluate(h) for h in population])
    
    # Parameters
    generations = 100
    tournament_size = 5
    mutation_rate = 0.1
    mutation_strength = 0.05
    elite_count = 1
    
    t0 = time.time()
    
    for gen in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        # Evaluate fitness
        fitness = np.array([evaluate(h) for h in population])
        
        # Select parents using tournament selection
        parents = []
        for _ in range(population_size):
            tournament = np.random.choice(np.arange(population_size), size=tournament_size, replace=False)
            winner = tournament[np.argmin(fitness[tournament])]
            parents.append(population[winner])
        
        # Generate offspring
        offspring = []
        for i in range(population_size):
            # Crossover
            parent1 = parents[i % len(parents)]
            parent2 = parents[(i+1) % len(parents)]
            child = np.copy(parent1)
            for j in range(n_points):
                if np.random.rand() < 0.5:
                    child[j] = parent2[j]
            
            # Mutation
            if np.random.rand() < mutation_rate:
                perturb = np.random.normal(0, mutation_strength, n_points)
                child += perturb
                child = np.clip(child, 0, 1)
            
            # Normalize
            child = normalize(child)
            offspring.append(child)
        
        # Combine population and offspring
        new_population = population + offspring
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        new_population = np.array(new_population)[sorted_indices]
        new_fitness = np.array([evaluate(h) for h in new_population])
        
        # Select top performers
        elite_indices = np.argsort(new_fitness)[:elite_count]
        population = new_population[elite_indices].tolist()
        
        # Update best solution
        best_index = np.argmin(fitness)
        best_h = population[best_index]
        best_c5 = fitness[best_index]
    
    return best_h, best_c5, n_points
