        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

        h = np.array(h, dtype=np.float64)

            return np.zeros_like(h)
        return h * (n_points / 2) / total
    
    # Initial population
    population_size = 50
    population = [normalize_h(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    
    start_time = time.time()

            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Selection
        tournament_size = 5
        tournament = np.random.choice(range(population_size), tournament_size, replace=False)

        # Crossover
