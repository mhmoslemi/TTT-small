    def objective(h):

        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    def create_offspring(parent1, parent2):
        crossover_point = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        mutation_mask = np.random.rand(n_points) < 0.1
        child[mutation_mask] = np.random.uniform(0, 1, np.sum(mutation_mask))
        return normalize_h(child)

    def evaluate_population(population):
        return [objective(individual) for individual in population]

    start_time = time.time()
    population = [normalize_h(np.random.uniform(0, 1, n_points)) for _ in range(50)]
