import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initial solution
    h = initial_h.copy()
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Population parameters
    population_size = 50
    generations = 100
    crossover_rate = 0.8
    mutation_rate = 0.1
    
    # Initialize population
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    
    for gen in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        best_index = np.argmin(fitness)
        current_best = population[best_index].copy()
        current_best_c5 = fitness[best_index]
        
        # Update best solution
        if current_best_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_best_c5
        
        # Selection
        selected = np.random.choice(len(population), size=population_size, p=np.array(fitness)/np.sum(fitness))
        selected = [population[i] for i in selected]
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if np.random.rand() < crossover_rate:
                parent1 = selected[i]
                parent2 = selected[i+1]
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                offspring.extend([child1, child2])
            else:
                offspring.extend([selected[i], selected[i+1]])
        
        # Mutation
        for i in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, 0.05, n_points)
                offspring[i] = np.clip(offspring[i] + mutation, 0, 1)
        
        # Replace population
        population = [normalize(ind) for ind in offspring]
    
    return best_h, best_c5, n_points
