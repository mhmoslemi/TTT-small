import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    def normalize_h(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    # Initial population
    population_size = 50
    population = [normalize_h(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    
    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        
        # Select top performers
        top_indices = np.argsort(fitness)[:population_size//2]
        top_population = [population[i] for i in top_indices]
        
        # Create new population through crossover and mutation
        new_population = []
        for _ in range(population_size):
            # Crossover
            parent1 = random.choice(top_population)
            parent2 = random.choice(top_population)
            child = np.array([parent1[i] if random.random() < 0.5 else parent2[i] for i in range(n_points)])
            
            # Mutation
            mutation_rate = 0.1
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0, 1)
            
            new_population.append(normalize_h(child))
        
        population = new_population
    
    # Find best solution
    best_h = min(population, key=objective)
    best_c5 = objective(best_h)
    
    return best_h.tolist(), best_c5, n_points
