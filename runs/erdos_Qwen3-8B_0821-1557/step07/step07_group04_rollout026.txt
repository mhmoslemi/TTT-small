import scipy.optimize as optimize
import time

    n_points = len(initial_h)

        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    # Initial guess
    h_initial = np.array(initial_h, dtype=np.float64)
    h_initial = h_initial / (np.sum(h_initial) * dx) * (n_points / 2)

    # Function to normalize h
    def normalize_h(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    # Custom penalty function for constraint
    def penalty(h):
        h = np.array(h, dtype=np.float64)
        return 1e4 * (np.sum(h) - n_points / 2) ** 2

    # Optimization with adaptive local search
    start_time = time.time()
    result = optimize.minimize(
        fun=lambda h: objective(h) + penalty(h),
        x0=h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    best_h = result.x
    best_c5 = result.fun

    # Guided local search
    for _ in range(50):
        if time.time() - start_time > budget_s:
