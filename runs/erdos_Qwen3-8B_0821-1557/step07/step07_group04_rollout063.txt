import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    # Initial guess
    h_initial = np.array(initial_h, dtype=np.float64)
    h_initial = h_initial / (np.sum(h_initial) * dx) * (n_points / 2)

    bounds = [(0, 1) for _ in range(n_points)]

    # Function to normalize h
    def normalize_h(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    def tournament_selection(population, k=3):
        candidates = np.random.choice(population, size=k, replace=False)
        return min(candidates, key=objective)

    # Genetic algorithm
    population_size = 20
    population = [normalize_h(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    best_h = h_initial
    best_c5 = objective(h_initial)
    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        # Selection
        parents = [tournament_selection(population) for _ in range(population_size // 2)]
        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                parent1 = parents[i]
                parent2 = parents[i + 1]
                child1 = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                child2 = np.where(np.random.rand(n_points) < 0.5, parent2, parent1)
                offspring.append(normalize_h(child1))
                offspring.append(normalize_h(child2))
        # Mutation
        for i in range(len(offspring)):
            mutation = np.random.normal(0, 0.05, n_points)
            offspring[i] = normalize_h(offspring[i] + mutation)
        # Evaluate and select
        population = [normalize_h(h) for h in population + offspring]
        population = [h for h in population if np.all(h >= 0) and np.all(h <= 1)]
        if not population:
            break
        population = sorted(population, key=objective)[:population_size]
        current_c5 = objective(population[0])
        if current_c5 < best_c5:
            best_h = population[0]
            best_c5 = current_c5

    return best_h.tolist(), best_c5, n_points
