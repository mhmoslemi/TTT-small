import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    # Function to normalize h
    def normalize_h(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    # Initialize population
    population_size = 20
    population = [normalize_h(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    best_h = population[0]
    best_c5 = objective(best_h)

    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx]
        
        # Selection
        selected = np.random.choice(population_size, size=population_size, p=np.array(fitness)/np.sum(fitness))
        
        # Crossover
        new_population = []
        for i in range(population_size//2):
            parent1 = selected[2*i]
            parent2 = selected[2*i + 1]
            child1 = np.array([parent1[j] if np.random.rand() < 0.5 else parent2[j] for j in range(n_points)])
            child2 = np.array([parent2[j] if np.random.rand() < 0.5 else parent1[j] for j in range(n_points)])
            new_population.append(normalize_h(child1))
            new_population.append(normalize_h(child2))
        
        # Mutation
        for i in range(population_size):
            if np.random.rand() < 0.1:
                mutation = np.random.uniform(-0.1, 0.1, size=n_points)
                new_population[i] = normalize_h(new_population[i] + mutation)
        
        population = new_population
    
    return best_h.tolist(), best_c5, n_points
