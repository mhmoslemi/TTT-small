import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    def normalize_h(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    # Initialize population
    population_size = 50
    population = [normalize_h(np.array(initial_h, dtype=np.float64) + np.random.uniform(-0.1, 0.1, n_points)) 
                  for _ in range(population_size)]
    
    start_time = time.time()
    best_h = population[0]
    best_c5 = objective(best_h)
    
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        # Select parents using tournament selection
        parents = []
        for _ in range(population_size):
            candidates = random.sample(range(population_size), 5)
            best_idx = min(candidates, key=lambda i: fitness[i])
            parents.append(population[best_idx])
        
        # Crossover
        new_population = []
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                parent1 = parents[i]
                parent2 = parents[i+1]
                child1 = np.concatenate([parent1[:n_points//2], parent2[n_points//2:]])
                child2 = np.concatenate([parent2[:n_points//2], parent1[n_points//2:]])
                new_population.append(normalize_h(child1))
                new_population.append(normalize_h(child2))
            else:
                new_population.append(parents[i])
        
        # Mutation
        for i in range(len(new_population)):
            if random.random() < 0.1:
                perturbation = np.random.uniform(-0.1, 0.1, n_points)
                new_population[i] = normalize_h(new_population[i] + perturbation)
        
        # Update population
        population = new_population
        
        # Update best solution
        for ind in population:
            current_c5 = objective(ind)
            if current_c5 < best_c5:
                best_h = ind
                best_c5 = current_c5
    
    return best_h.tolist(), best_c5, n_points
