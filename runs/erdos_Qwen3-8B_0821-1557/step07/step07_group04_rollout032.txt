import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    # Initial guess with periodic pattern
    h_initial = np.array(initial_h, dtype=np.float64)
    h_initial = h_initial / (np.sum(h_initial) * dx) * (n_points / 2)

    bounds = [(0, 1) for _ in range(n_points)]

    # Function to normalize h
    def normalize_h(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    # Optimization with adaptive local search
    start_time = time.time()
    result = optimize.minimize(
        fun=objective,
        x0=h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_integral}],
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    best_h = result.x
    best_c5 = result.fun

    # Local search with structured perturbations
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break

        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = normalize_h(best_h + perturbation)
        # Check if perturbed_h is valid

            perturbed_c5 = objective(perturbed_h)
            if perturbed_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = perturbed_c5

    return best_h.tolist(), best_c5, n_points
