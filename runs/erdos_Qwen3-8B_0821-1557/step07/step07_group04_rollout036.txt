    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    def normalize_h(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    # Initial population
    population_size = 50
    population = [normalize_h(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    population = [h for h in population if np.all(h >= 0) and np.all(h <= 1)]

    def fitness(individual):
        return objective(individual)

    def select_parents(population, fitnesses):
        indices = np.arange(len(population))
        probabilities = fitnesses / fitnesses.sum()
        return np.random.choice(indices, size=2, p=probabilities)

    def crossover(parent1, parent2):
        point = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:point], parent2[point:]])
        return child

    def mutate(individual, mutation_rate=0.1):
        mutated = individual.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                mutated[i] = np.random.uniform(0, 1)
        return mutated

    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        fitnesses = [fitness(ind) for ind in population]
        parents_indices = [select_parents(population, fitnesses) for _ in range(population_size // 2)]
        offspring = []
        for p1, p2 in parents_indices:
            child = crossover(population[p1], population[p2])
            child = mutate(child)
            offspring.append(child)
        population += offspring
        population = np.array(population)
        population = np.vstack([population, [best_h]]).tolist()
        population = [normalize_h(h) for h in population]
        population = [h for h in population if np.all(h >= 0) and np.all(h <= 1)]

        if current_c5 < best_c5:

            best_c5 = current_c5
