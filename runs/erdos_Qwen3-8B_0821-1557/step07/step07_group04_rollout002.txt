            return np.zeros_like(h)
        return h * (n_points / 2) / total

    # Genetic algorithm parameters
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    elite_size = 5
    local_search_steps = 20
    start_time = time.time()

        h = np.random.uniform(0, 1, n_points)
        return normalize_h(h)

    for gen in range(generations):

        elite_indices = np.argsort(fitness)[:elite_size]

        new_population = []

            tournament = np.random.choice(range(len(population)), size=5, replace=False)
