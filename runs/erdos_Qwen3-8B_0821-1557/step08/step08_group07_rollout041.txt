import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    initial_h[30:60] = 1.0
    initial_h[60:90] = 0.5
    h = initial_h.copy()
    
    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    # Initial solution
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Simulated Annealing parameters
    temp = 1.0
    temp_decay = 0.995
    max_iterations = 500
    
    for _ in range(max_iterations):
        if time.time() - t0 > budget_s:
            break
        
        # Generate perturbation
        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = normalize(h_perturbed)
        
        # Evaluate new solution
        c5_perturbed = evaluate(h_perturbed)
        
        # Acceptance probability
        delta = c5_perturbed - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = h_perturbed
            best_c5 = c5_perturbed
        
        # Update best solution
        if evaluate(h) < best_c5:
            best_h = h.copy()
            best_c5 = evaluate(h)
        
        # Cool down
        temp *= temp_decay
    
    return best_h, best_c5, n_points
