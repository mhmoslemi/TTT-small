import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initial population
    population_size = 30
    population = [initial_h.copy() for _ in range(population_size)]
    population = [normalize(p) for p in population]
    fitness = np.array([evaluate(p) for p in population])
    
    t0 = time.time()
    
    for generation in range(200):
        if time.time() - t0 > budget_s:
            break
        
        # Selection
        parents = np.random.choice(population_size, size=population_size, p=1 - fitness / np.max(fitness))
        parents = [population[i] for i in parents]
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                p1 = parents[i]
                p2 = parents[i + 1]
                child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
                offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            if np.random.rand() < 0.1:
                idx = np.random.randint(n_points)
                offspring[i][idx] = np.clip(offspring[i][idx] + np.random.uniform(-0.1, 0.1), 0, 1)
        
        # Evaluate offspring
        offspring_fitness = np.array([evaluate(p) for p in offspring])
        
        # Combine population
        population = parents + offspring
        fitness = np.concatenate((fitness, offspring_fitness))
        
        # Select best
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
    
    return best_h, best_c5, n_points
