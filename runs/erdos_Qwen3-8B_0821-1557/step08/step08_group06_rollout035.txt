import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.uniform(-0.1, 0.1)
                h[i] = np.clip(h[i], 0, 1)
        return normalize(h)
    
    def crossover(h1, h2):
        h = h1.copy()
        for i in range(n_points):
            if np.random.rand() < 0.5:
                h[i] = h2[i]
        return normalize(h)
    
    # Initial solution
    h = initial_h.copy()
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Genetic algorithm
    population_size = 20
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    for _ in range(100):
        if time.time() - t0 > budget_s:
            break
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        # Select parents
        parents = np.random.choice(population, size=population_size, p=fitness/np.sum(fitness))
        # Crossover and mutation
        new_population = []
        for _ in range(population_size // 2):
            h1, h2 = np.random.choice(parents, size=2, replace=False)
            child1 = crossover(h1, h2)
            child2 = mutate(child1)
            new_population.extend([child1, child2])
        population = new_population
        # Update best solution
        for ind in population:
            val = evaluate(ind)
            if val < best_c5:
                best_c5 = val
                best_h = ind.copy()
    
    # Local optimization
    for _ in range(50):
        if time.time() - t0 > budget_s:
            break
        # Use SLSQP optimizer with bounds and constraint
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        res = optimize.minimize(
            objective,
            best_h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        best_h = res.x
        best_h = np.clip(best_h, 0, 1)
        best_h = normalize(best_h)
        if evaluate(best_h) < best_c5:
            best_c5 = evaluate(best_h)
    
    return best_h, best_c5, n_points
