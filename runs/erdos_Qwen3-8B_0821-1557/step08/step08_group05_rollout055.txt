import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initial solution
    initial_h = np.random.uniform(0, 1, n_points)
    initial_h = normalize(initial_h)
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    t0 = time.time()
    
    # Simulated annealing parameters
    temp = 1.0
    cooling_rate = 0.995
    max_iter = 1000
    
    for _ in range(max_iter):
        if time.time() - t0 > budget_s:
            break
        # Generate a random perturbation
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        new_h = np.clip(best_h + perturbation, 0, 1)
        new_h = normalize(new_h)
        new_c5 = evaluate(new_h)
        
        # Acceptance probability
        if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5) / temp):
            best_h = new_h.copy()
            best_c5 = new_c5
        
        temp *= cooling_rate
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 60
    mutation_rate = 0.15
    elite_size = 5
    
    for gen in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        # Create new population with diversity
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population[0] = best_h.copy()
        
        # Evaluate population
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        best_pop_h = population[best_idx].copy()
        best_pop_c5 = fitness[best_idx]
        
        # Keep elite solutions
        elite_indices = np.argsort(fitness)[:elite_size]
        elite_h = [population[i].copy() for i in elite_indices]
        
        # Apply local optimization to elite solutions with tighter constraints
        for i in range(len(elite_h)):
            h = elite_h[i].copy()
            h = normalize(h)
            
            # Use SLSQP optimizer with bounds and constraint
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
            res = optimize.minimize(
                evaluate,
                h,
                bounds=[(0, 1) for _ in range(n_points)],
                constraints=cons,
                method='SLSQP',
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            elite_h[i] = res.x
        
        # Combine elites and best population solution
        population = elite_h + [best_pop_h]
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
    
    return best_h, best_c5, n_points
