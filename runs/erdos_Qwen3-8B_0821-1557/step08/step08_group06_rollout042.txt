import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    # Initial solution
    h = initial_h.copy()
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Genetic algorithm parameters
    pop_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initial population
    population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
    population = [normalize(p) for p in population]
    
    for gen in range(generations):
        if time.time() - t0 > budget_s:
            break
        # Evaluate fitness
        fitness = [evaluate(p) for p in population]
        # Select best individuals
        best_idx = np.argmin(fitness)
        best_fitness = fitness[best_idx]
        # Update best solution
        if best_fitness < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = best_fitness
        # Create new population
        new_population = []
        for _ in range(pop_size // 2):
            # Select parents
            parent1 = population[np.random.choice(pop_size)]
            parent2 = population[np.random.choice(pop_size)]
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] = np.clip(child1[i] + np.random.normal(0, 0.05), 0, 1)
                    child2[i] = np.clip(child2[i] + np.random.normal(0, 0.05), 0, 1)
            # Normalize
            child1 = normalize(child1)
            child2 = normalize(child2)
            new_population.append(child1)
            new_population.append(child2)
        population = new_population
    
    # Local optimization
    for _ in range(50):
        if time.time() - t0 > budget_s:
            break
        # Use SLSQP optimizer with bounds and constraint
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        res = optimize.minimize(
            objective,
            best_h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        h = res.x
        h = np.clip(h, 0, 1)
        h = normalize(h)
        if evaluate(h) < best_c5:
            best_h = h.copy()
            best_c5 = evaluate(h)
    
    return best_h, best_c5, n_points
