def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)

        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initial population
    population_size = 20

        # Evaluate fitness
        fitness = [evaluate(indiv) for indiv in population]
        
        # Select top performers
        top_indices = np.argsort(fitness)[:population_size//2]
        top_population = [population[i] for i in top_indices]
        
        # Generate new population through mutation and crossover
        new_population = []
        while len(new_population) < population_size:
            # Randomly select two parents
            parent1, parent2 = np.random.choice(top_population, 2, replace=False)
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            mutation_rate = 0.1

                    child[i] = np.clip(child[i] + np.random.uniform(-0.1, 0.1), 0, 1)
            
            # Normalize
            child = normalize(child)
            new_population.append(child)
        
        population = new_population
    
    # Find best solution
    best_idx = np.argmin(fitness)
