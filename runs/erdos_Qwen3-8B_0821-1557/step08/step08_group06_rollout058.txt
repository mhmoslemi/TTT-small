import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    population_size = 50
    mutation_rate = 0.15
    crossover_rate = 0.8
    generations = 100
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    t0 = time.time()
    
    # Initialize population
    population = [normalize(np.random.rand(n_points)) for _ in range(population_size)]
    population = [np.clip(p, 0, 1) for p in population]
    population_fitness = [evaluate(p) for p in population]
    
    for gen in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        # Select elites
        elite_idx = np.argsort(population_fitness)[:5]
        elites = [population[i] for i in elite_idx]
        
        # Select parents
        parents = []
        for _ in range(population_size - len(elites)):
            idx1, idx2 = np.random.choice(range(population_size), 2, replace=False)
            if np.random.rand() < crossover_rate:
                # Crossover
                child = np.zeros(n_points)
                for i in range(n_points):
                    if np.random.rand() < 0.5:
                        child[i] = population[idx1][i]
                    else:
                        child[i] = population[idx2][i]
            else:
                # Random selection
                child = np.random.rand(n_points)
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = np.clip(child[i] + np.random.normal(0, 0.05), 0, 1)
            parents.append(child)
        
        # Combine elites and parents
        population = elites + parents
        population = [np.clip(p, 0, 1) for p in population]
        population_fitness = [evaluate(p) for p in population]
        
        # Update best solution
        best_idx = np.argmin(population_fitness)
        if population_fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = population_fitness[best_idx]
    
    return best_h, best_c5, n_points
