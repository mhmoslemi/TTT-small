import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    # Initial solution
    h = initial_h.copy()
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    
    # Initialize population
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    
    for gen in range(generations):
        if time.time() - t0 > budget_s:
            break
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        # Select best individuals
        parents = [population[i] for i in np.argsort(fitness)[:population_size//2]]
        # Crossover
        offspring = []
        for _ in range(population_size//2):
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            child = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            offspring.append(child)
        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    offspring[i][j] = np.clip(offspring[i][j] + np.random.uniform(-0.1, 0.1), 0, 1)
        # Replace population
        population = parents + offspring
        # Update best solution
        for ind in population:
            current_c5 = evaluate(ind)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = ind.copy()
    
    # Local optimization
    for _ in range(50):
        if time.time() - t0 > budget_s:
            break
        # Use SLSQP optimizer with bounds and constraint
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        res = optimize.minimize(
            objective,
            best_h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        best_h = res.x
        best_h = np.clip(best_h, 0, 1)
        best_h = normalize(best_h)
        if evaluate(best_h) < best_c5:
            best_c5 = evaluate(best_h)
    
    return best_h, best_c5, n_points
