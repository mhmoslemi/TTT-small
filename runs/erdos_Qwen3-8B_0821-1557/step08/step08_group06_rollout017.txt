import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):

        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    # Initial solution
    h = initial_h.copy()
    h = normalize(h)
    best_h = h.copy()

    for _ in range(200):

            break

        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        h_test = h + perturb
        h_test = np.clip(h_test, 0, 1)
        h_test = normalize(h_test)
        val = evaluate(h_test)
        if val < best_c5:
            h = h_test.copy()
            best_c5 = val
            best_h = h.copy()
    
    # Global optimization with SLSQP
    for _ in range(50):
        if time.time() - t0 > budget_s:
            break
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        res = optimize.minimize(

            h,
