def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)

    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    # Initialize population
    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    population = [normalize(p) for p in population]
    fitness = [evaluate(p) for p in population]
    
    t0 = time.time()
    
    # Genetic algorithm parameters
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 100
    
    for gen in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        # Evaluate fitness
        current_fitness = fitness.copy()
        
        # Select parents
        parents = np.random.choice(range(population_size), size=population_size, p=current_fitness/np.sum(current_fitness))
        parents = np.unique(parents)
        
        # Crossover
        for i in range(0, population_size, 2):
            if np.random.rand() < crossover_rate:
                p1, p2 = parents[i % len(parents)], parents[(i+1) % len(parents)]
                child1 = np.concatenate([population[p1][:n_points//2], population[p2][n_points//2:]])
                child2 = np.concatenate([population[p2][:n_points//2], population[p1][n_points//2:]])
                population[i], population[i+1] = child1, child2
        
        # Mutation
        for i in range(population_size):
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(0, n_points)
                population[i][idx] += np.random.uniform(-0.1, 0.1)
                population[i] = np.clip(population[i], 0, 1)
                population[i] = normalize(population[i])
        
        # Evaluate new fitness
        new_fitness = [evaluate(p) for p in population]

        best_idx = np.argmin(fitness)
