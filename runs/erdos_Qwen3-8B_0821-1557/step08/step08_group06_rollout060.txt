import time

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)

    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initial solution
    h = initial_h.copy()
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = [normalize(np.random.rand(n_points)) for _ in range(population_size)]
    population.append(h)
    population = np.array([np.clip(p, 0, 1) for p in population])
    
    for gen in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        # Evaluate fitness
        fitness = np.array([evaluate(p) for p in population])
        
        # Select best individuals
        best_idx = np.argsort(fitness)[:population_size]
        selected = population[best_idx]
        
        # Crossover
        offspring = []
        for _ in range(population_size):
            if np.random.rand() < crossover_rate:
                parent1, parent2 = np.random.choice(selected, size=2, replace=False)
                child = np.interp(np.random.rand(n_points), [0, 1], [parent1, parent2])
            else:
                child = np.random.rand(n_points)
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    offspring[i][j] = np.clip(offspring[i][j] + np.random.normal(0, 0.05), 0, 1)
        
        # Replace population
        population = np.array([np.clip(p, 0, 1) for p in offspring])
        population = np.vstack((population, selected))
        population = np.unique(population, axis=0)
        population = population[:population_size]
        
        # Update best solution
        for p in population:
            current_c5 = evaluate(p)
            if current_c5 < best_c5:
                best_h = p.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
