        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

        return evaluate(h)

    population_size = 20
    generations = 100
    mutation_rate = 0.1

    population = [normalize(np.clip(np.random.rand(n_points), 0, 1)) for _ in range(population_size)]
    
    for gen in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        
        # Find the best individual
        best_idx = np.argmin(fitness)
        best_fitness = fitness[best_idx]
        
        # Update best solution
        if best_fitness < best_c5:

            best_h = population[best_idx].copy()

        parents = np.random.choice(population_size, size=population_size, p=np.array(fitness)/np.sum(fitness))

        # Crossover
        offspring = []

                    child[j] = p1[j]
