import time

def run(seed=42, budget_s=10, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initialize population
    population_size = 20
    population = [normalize(np.random.rand(n_points)) for _ in range(population_size)]
    population = [np.clip(h, 0, 1) for h in population]
    
    t0 = time.time()
    best_h = initial_h.copy()
    best_c5 = evaluate(initial_h)
    
    for _ in range(100):
        if time.time() - t0 > budget_s:
            break
        
        # Evaluate population
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_c5 = fitness[best_idx]
        
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
        
        # Selection and crossover
        selected = np.random.choice(range(population_size), size=population_size, p=np.array(fitness)/np.sum(fitness))
        new_population = []
        
        for i in range(population_size // 2):
            parent1 = population[selected[i]]
            parent2 = population[selected[population_size - 1 - i]]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])

            new_population.append(np.clip(child1, 0, 1))
            new_population.append(np.clip(child2, 0, 1))
        
        # Replace population
        population = new_population
        population = [normalize(h) for h in population]
    
    return best_h, best_c5, n_points
