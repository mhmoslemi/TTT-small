import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reduced from 70 for better exploration
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    # Initial solution
    h = np.random.uniform(0, 1, n_points)
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Local search with adaptive perturbations
    for _ in range(200):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        current_c5 = evaluate(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        if time.time() - t0 > budget_s:
            break
        
        # Adaptive perturbation
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturbed = h + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = normalize(h_perturbed)
        perturbed_c5 = evaluate(h_perturbed)
        
        if perturbed_c5 < current_c5:
            h = h_perturbed.copy()
    
    # Genetic algorithm with diversity maintenance
    population_size = 50
    generations = 60
    mutation_rate = 0.15
    elite_size = 5
    
    for gen in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population[0] = best_h.copy()
        
        # Evaluate population
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        best_pop_h = population[best_idx].copy()
        best_pop_c5 = fitness[best_idx]
        
        # Keep elite solutions
        elite_indices = np.argsort(fitness)[:elite_size]
        elite_h = [population[i].copy() for i in elite_indices]
        
        # Local optimization for elite solutions
        for i in range(len(elite_h)):
            h = elite_h[i].copy()
            h = normalize(h)
            
            # Use SLSQP optimizer with bounds and constraint
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
            res = optimize.minimize(
                objective,
                h,
                bounds=[(0, 1) for _ in range(n_points)],
                constraints=cons,
                method='SLSQP',
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            elite_h[i] = res.x
        
        # Combine elites and best population solution
        population = elite_h + [best_pop_h]
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
    
    return best_h, best_c5, n_points
