def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)

        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initial solution
    h = initial_h.copy()

    population_size = 20
    generations = 50
    mutation_rate = 0.15
    crossover_rate = 0.8
    
    # Initialize population
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    
    for gen in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Select parents
        parents = np.random.choice(len(population), size=population_size, p=np.array(fitness)/np.sum(fitness))
        parents = np.array([population[i] for i in parents])
        
        # Crossover
        offspring = []
        for i in range(population_size//2):
            if np.random.rand() < crossover_rate:
                p1, p2 = parents[2*i], parents[2*i+1]
                child = np.concatenate((p1[:n_points//2], p2[n_points//2:]))
            else:
                child = np.random.uniform(0, 1, n_points)

        # Mutation
        for i in range(len(offspring)):

            offspring[i] = normalize(offspring[i])
        
        # Replace population
        population = np.array(offspring)
    
    return best_h, best_c5, n_points
