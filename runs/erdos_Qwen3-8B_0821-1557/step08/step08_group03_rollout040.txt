import numpy as np
import scipy.signal as signal

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = h.copy()
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    def evaluate(h):
        return objective(h)
    
    # Initialize population
    population_size = 32
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = population / np.sum(population, axis=1, keepdims=True) * (n_points / 2)
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    # Genetic algorithm parameters
    generations = int(budget_s * 2)
    mutation_rate = 0.15
    crossover_rate = 0.8
    
    for gen in range(generations):
        # Evaluate fitness
        fitness = np.array([evaluate(ind) for ind in population])
        
        # Select parents
        parents_indices = np.argsort(fitness)[:population_size//2]
        parents = population[parents_indices]
        
        # Create new population
        new_population = []
        for _ in range(population_size):
            if np.random.rand() < crossover_rate:
                # Crossover
                parent1 = np.random.choice(parents)
                parent2 = np.random.choice(parents)
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            else:
                # Mutation
                child = np.random.uniform(0, 1, n_points)
            
            # Normalize
            child = child / np.sum(child) * (n_points / 2)
            
            # Apply mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, 0.05, n_points)
                child += mutation
                child = np.clip(child, 0, 1)
                child = child / np.sum(child) * (n_points / 2)
            
            new_population.append(child)
        
        population = np.array(new_population)
        
        # Update best solution
        current_best = population[np.argmin(fitness)]
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
