import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)

    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    h = initial_h.copy()
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Adaptive local search with perturbations
    for _ in range(200):
        if time.time() - t0 > budget_s:
            break
        # Random perturbation
        perturb = np.random.uniform(-0.05, 0.05, size=n_points)
        h_test = np.clip(h + perturb, 0, 1)
        h_test = normalize(h_test)
        val = evaluate(h_test)
        if val < best_c5:
            h = h_test.copy()
            best_c5 = val
            best_h = h.copy()
        
        # Gradient-based optimization
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        res = optimize.minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=cons,
            method='L-BFGS-B',
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        h = res.x

        h = normalize(h)
        if evaluate(h) < best_c5:
            best_h = h.copy()
            best_c5 = evaluate(h)
    
    return best_h, best_c5, n_points
