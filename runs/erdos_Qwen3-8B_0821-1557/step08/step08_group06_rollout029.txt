import time

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)

        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initialize population
    population_size = 30
    mutation_rate = 0.15
    population = [normalize(np.random.rand(n_points)) for _ in range(population_size)]

    t0 = time.time()

    best_c5 = evaluate(best_h)
    
    for generation in range(500):
        if time.time() - t0 > budget_s:

        # Evaluate fitness
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Selection
            parent1 = population[np.random.choice(len(population))]
            parent2 = population[np.random.choice(len(population))]
            
            # Crossover
            child = np.zeros(n_points)

                if np.random.rand() < 0.5:

                    child[i] = parent2[i]

                    child[i] = np.clip(child[i] + np.random.uniform(-0.1, 0.1), 0, 1)
            
            # Normalize
            child = normalize(child)
            new_population.append(child)
        
        # Replace population
        population = new_population
