        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initialize population
    population_size = 30
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]

    best_solution = None
    best_c5 = float('inf')
    
    for generation in range(100):
        if time.time() - t0 > budget_s:
            break
        
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_solution = population[best_idx].copy()
        
        # Create new population
        new_population = []
        
        # Elitism: keep best solution
        new_population.append(best_solution)
        
        # Selection and crossover
        for _ in range(population_size - 1):
            # Select parents
            parent1 = population[np.random.choice(len(population))]
            parent2 = population[np.random.choice(len(population))]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            
            # Mutation
            mutation_rate = 0.1
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = np.clip(child[i] + np.random.uniform(-0.1, 0.1), 0, 1)
            
            new_population.append(child)
        
        population = new_population
