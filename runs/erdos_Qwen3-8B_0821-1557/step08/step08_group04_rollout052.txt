            best_h = population[best_idx].copy()

        if time_used >= budget_s - 0.1:
            break
        
        # Create new population
        new_population = []
        for _ in range(population_size):
            # Select parents
            parent1, parent2 = random.choices(population, k=2)

            if random.random() < crossover_rate:
