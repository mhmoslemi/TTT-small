import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)

        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    # Initial solution

    improvement_threshold = 1e-5

            break
        
        # Random perturbation
        perturb = np.random.uniform(-perturbation_size, perturbation_size, size=n_points)
        h_perturb = h + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = normalize(h_perturb)
        
        # Evaluate perturbed solution
        c5_perturb = evaluate(h_perturb)
        
        # Accept improvement
        if c5_perturb < best_c5 - improvement_threshold:

            best_h = h.copy()
            best_c5 = c5_perturb
            perturbation_size *= 1.01  # Increase perturbation size for exploration
        else:
