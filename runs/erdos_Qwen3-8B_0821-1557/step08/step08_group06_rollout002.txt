from copy import deepcopy

def run(seed=42, budget_s=10, **kwargs):

        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initialize population
    population_size = 50
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    population = [h for h in population if np.sum(h) * dx >= 0.95 and np.sum(h) * dx <= 1.05]
    
    best_h = deepcopy(initial_h)
    best_c5 = evaluate(normalize(best_h))
    
    t0 = time.time()
    
    # Genetic algorithm
    for generation in range(200):
        if time.time() - t0 > budget_s:
            break
        
        # Evaluate population
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = deepcopy(population[best_idx])
        
        # Elitism
        if best_c5 < evaluate(normalize(best_h)):
            best_h = deepcopy(population[best_idx])
        
        # Selection
        selected = []
        for _ in range(population_size // 2):
            idx1 = np.random.randint(0, population_size)
            idx2 = np.random.randint(0, population_size)
            if fitness[idx1] < fitness[idx2]:
                selected.append(population[idx1])
            else:
                selected.append(population[idx2])
        
        # Crossover
        new_population = []
        for i in range(0, population_size, 2):
            if i + 1 >= population_size:
                new_population.append(selected[i])
                continue
            parent1 = selected[i]
            parent2 = selected[i+1]
            crossover_point = np.random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            new_population.append(child1)
            new_population.append(child2)
        
        # Mutation
        for i in range(len(new_population)):

                idx = np.random.randint(n_points)

            h = normalize(h)

                population.append(h)
        
        # Keep best solution
        if evaluate(normalize(best_h)) > best_c5:
            best_h = deepcopy(population[best_idx])
    
    return best_h, best_c5, n_points
