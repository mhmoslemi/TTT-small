import time

        h = np.clip(h, 0, 1)

        return np.max(corr) * dx

        return evaluate(h)

        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = h + perturbation

        energy = evaluate(h_perturbed)

            best_c5 = energy

        temp *= cooling_rate
    
    # Gradient-based optimization
    for _ in range(50):
