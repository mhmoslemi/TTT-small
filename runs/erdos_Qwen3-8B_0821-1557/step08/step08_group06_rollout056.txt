import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initial solution
    h = initial_h.copy()
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Genetic algorithm parameters
    pop_size = 20
    generations = 50
    mutation_rate = 0.1
    elite_ratio = 0.2
    
    for _ in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        # Create population
        population = [h.copy() for _ in range(pop_size)]
        for i in range(pop_size):
            population[i] = np.random.uniform(0, 1, n_points)
            population[i] = normalize(population[i])
        
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        
        # Select elites
        elite_indices = np.argsort(fitness)[:int(elite_ratio * pop_size)]
        elites = [population[i] for i in elite_indices]
        
        # Generate next generation
        next_gen = elites.copy()
        for _ in range(pop_size - len(elites)):
            # Select parents
            parent1 = population[np.random.choice(pop_size)]
            parent2 = population[np.random.choice(pop_size)]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = np.clip(child[i] + np.random.uniform(-0.1, 0.1), 0, 1)
            
            child = normalize(child)
            next_gen.append(child)
        
        population = next_gen
    
    # Evaluate final population
    fitness = [evaluate(ind) for ind in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_c5 = fitness[best_idx]
    
    return best_h, best_c5, n_points
