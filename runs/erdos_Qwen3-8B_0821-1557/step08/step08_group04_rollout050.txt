import numpy as np
import scipy.signal
from copy import deepcopy

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    
    # Initial population
    n_points = 100
    population_size = 30
    initial_population = []
    for _ in range(population_size):
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * (n_points / 2)
        initial_population.append(h)
    
    def evaluate(h):
        dx = 2.0 / n_points
        overlap = scipy.signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    def crossover(parent1, parent2):
        h = np.zeros_like(parent1)
        for i in range(len(h)):
            if np.random.rand() < 0.5:
                h[i] = parent1[i]
            else:
                h[i] = parent2[i]
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    best_h = initial_population[0]
    best_c5 = evaluate(best_h)
    
    time_used = 0
    for _ in range(100):  # Max generations
        # Evaluate population
        fitness = [evaluate(individual) for individual in initial_population]
        # Select parents
        parents = np.array(initial_population)[np.argsort(fitness)[:10]]
        # Create new population
        new_population = []
        for _ in range(population_size):
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        initial_population = new_population
        # Update best solution
        for candidate in initial_population:
            current_c5 = evaluate(candidate)
            if current_c5 < best_c5:
                best_h = candidate
                best_c5 = current_c5
        # Check time budget
        time_used += 0.1
        if time_used >= budget_s - 0.1:
            break
    
    return best_h, best_c5, n_points
