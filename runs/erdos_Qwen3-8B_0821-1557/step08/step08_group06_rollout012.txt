import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    # Initial solution
    h = initial_h.copy()
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    mutation_strength = 0.05
    elite_fraction = 0.2
    
    # Initialize population
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    population = [h for h in population if np.sum(h) * dx >= 0.95 and np.sum(h) * dx <= 1.05]
    
    for gen in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        # Evaluate population
        fitness = [evaluate(h) for h in population]
        best_index = np.argmin(fitness)
        current_best = population[best_index]
        current_c5 = fitness[best_index]
        
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
        
        # Select elites
        elite_size = int(elite_fraction * population_size)
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elites.copy()
        
        # Crossover
        for _ in range(population_size - elite_size):
            parent1 = np.random.choice(elite_indices)
            parent2 = np.random.choice(elite_indices)
            child = (population[parent1] + population[parent2]) / 2
            child = np.clip(child, 0, 1)
            child = normalize(child)
            new_population.append(child)
        
        # Mutation
        for i in range(len(new_population)):
            if np.random.rand() < mutation_rate:
                perturbation = np.random.normal(0, mutation_strength, n_points)
                mutated = new_population[i] + perturbation
                mutated = np.clip(mutated, 0, 1)
                mutated = normalize(mutated)
                new_population[i] = mutated
        
        # Replace population
        population = new_population
    
    # Local optimization
    for _ in range(50):
        if time.time() - t0 > budget_s:
            break
        # Use SLSQP optimizer with bounds and constraint
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        res = optimize.minimize(
            objective,
            best_h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        h = res.x
        h = np.clip(h, 0, 1)
        h = normalize(h)
        if evaluate(h) < best_c5:
            best_h = h.copy()
            best_c5 = evaluate(h)
    
    return best_h, best_c5, n_points
