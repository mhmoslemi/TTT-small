import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    # Initial solution
    h = initial_h.copy()
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Adaptive local search with random perturbations
    for _ in range(100):
        if time.time() - t0 > budget_s:
            break
        # Random perturbation
        perturb = np.random.normal(0, 0.05, n_points)
        h_test = h + perturb
        h_test = np.clip(h_test, 0, 1)
        h_test = normalize(h_test)
        val = evaluate(h_test)
        if val < best_c5:
            h = h_test.copy()
            best_c5 = val
    
    # Gradient descent with adaptive learning rate
    for _ in range(50):
        if time.time() - t0 > budget_s:
            break
        # Use SLSQP optimizer with bounds and constraint
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        res = optimize.minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        h = res.x
        h = np.clip(h, 0, 1)
        h = normalize(h)
        if evaluate(h) < best_c5:
            best_h = h.copy()
            best_c5 = evaluate(h)
    
    return best_h, best_c5, n_points
