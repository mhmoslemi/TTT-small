        population = [h.copy() for _ in range(population_size)]

        fitness = [evaluate(ind) for ind in population]
        
        # Select best individuals
        elite_indices = np.argsort(fitness)[:5]

        # Crossover and mutation
        for i in range(population_size):
            if np.random.rand() < 0.8:
                # Crossover with elite
                parent = np.random.choice(elite)
                child = parent.copy()
                indices = np.random.choice(n_points, size=int(n_points/2), replace=False)
