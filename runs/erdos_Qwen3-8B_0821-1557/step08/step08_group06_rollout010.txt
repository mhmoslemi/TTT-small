    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)

    h = initial_h.copy()

    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    
    # Initialize population
    population = [normalize(np.clip(h.copy() + np.random.normal(0, 0.05, n_points), 0, 1)) for _ in range(population_size)]
    
    for _ in range(generations):
        if time.time() - t0 > budget_s:
            break
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        # Select best individuals
        best_indices = np.argsort(fitness)[:population_size//2]
        selected = [population[i] for i in best_indices]
        # Generate offspring through crossover and mutation
        offspring = []
        for _ in range(population_size//2):
            # Crossover
            parent1 = selected[np.random.choice(len(selected))]
            parent2 = selected[np.random.choice(len(selected))]

            offspring.append(normalize(child))

        population = selected + offspring
