import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 1 if i < n_points//2 else 0
    h = initial_h.copy()
    h = h / np.sum(h) * (n_points / 2)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def constraint_norm(h):
        return np.sum(np.abs(h)) - n_points
    
    def optimize_h(h):
        h = h.astype(float)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    def coordinate_descent(h, max_iter=100):
        h = h.copy()
        for _ in range(max_iter):
            for i in range(n_points):
                h[i] = optimize.minimize(lambda x: evaluate(np.clip(h.copy(), 0, 1)), x0=h[i], bounds=[(0, 1)], method='L-BFGS-B').x[0]
                h = optimize_h(h)
        return h
    
    def random_search(h, num_samples=50):
        best_h = h.copy()
        best_c5 = evaluate(h)
        for _ in range(num_samples):
            perturbed = h.copy()
            for i in range(n_points):
                perturbed[i] += np.random.uniform(-0.1, 0.1)
            perturbed = optimize_h(perturbed)
            c5 = evaluate(perturbed)
            if c5 < best_c5:
                best_h = perturbed.copy()
                best_c5 = c5
        return best_h, best_c5
    
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    for _ in range(5):
        if time.time() - t0 > budget_s:
            break
        h = coordinate_descent(h)
        h, c5 = random_search(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    return best_h, best_c5, n_points
