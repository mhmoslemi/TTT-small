import numpy as np
import scipy.signal

def run(seed=42, budget_s=10, **kwargs):

    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = scipy.signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    population_size = 20
    generations = 200
    mutation_rate = 0.15
    mutation_strength = 0.2
    
    for _ in range(generations):
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        for i in range(population_size):
            population[i] = population[i] / np.sum(population[i]) * (n_points / 2)
        
        fitness = [evaluate(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
        
        if best_c5 < 0.380876:
            break
        
        new_population = []
        for _ in range(population_size // 2):
            parent1 = np.random.choice(population, p=np.array(fitness)/np.sum(fitness))
            parent2 = np.random.choice(population, p=np.array(fitness)/np.sum(fitness))
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.uniform(-mutation_strength, mutation_strength)
                    child2[i] += np.random.uniform(-mutation_strength, mutation_strength)
            
            # Normalize
            child1 = child1 / np.sum(child1) * (n_points / 2)
            child2 = child2 / np.sum(child2) * (n_points / 2)
            
            new_population.append(child1)
            new_population.append(child2)
        
        population = new_population
    
    return best_h, best_c5, n_points
