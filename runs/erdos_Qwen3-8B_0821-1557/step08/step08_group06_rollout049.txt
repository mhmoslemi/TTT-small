import time
import random

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initialize population
    population_size = 50
    population = [normalize(np.random.rand(n_points)) for _ in range(population_size)]
    population = [h for h in population if np.sum(h) * dx == 1]
    
    def tournament_selection(population, tournament_size=5):
        selected = random.sample(population, tournament_size)
        return min(selected, key=lambda x: evaluate(x))
    
    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child
    
    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                h[i] += np.random.uniform(-0.1, 0.1)
                h[i] = np.clip(h[i], 0, 1)
        return normalize(h)
    
    t0 = time.time()
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    for _ in range(100):
        if time.time() - t0 > budget_s:
            break
        # Evaluate population
        population = [normalize(h) for h in population]
        population = [h for h in population if np.sum(h) * dx == 1]
        population_scores = [evaluate(h) for h in population]
        
        # Select parents
        parent1 = tournament_selection(population)
        parent2 = tournament_selection(population)
        
        # Crossover and mutate
        child = crossover(parent1, parent2)
        child = mutate(child)
        
        # Evaluate child
        child_score = evaluate(child)
        
        # Replace worst individual
        if child_score < max(population_scores):
            population.pop(population_scores.index(max(population_scores)))
            population.append(child)
        
        # Update best solution
        current_best = min(population, key=lambda x: evaluate(x))
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
