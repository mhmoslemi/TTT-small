import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initial population
    population_size = 20
    population = [initial_h.copy() for _ in range(population_size)]
    population = [normalize(h) for h in population]
    
    t0 = time.time()
    
    for generation in range(100):
        if time.time() - t0 > budget_s:
            break
        
        # Evaluate fitness
        fitness = [evaluate(h) for h in population]
        best_index = np.argmin(fitness)
        best_h = population[best_index]
        best_c5 = fitness[best_index]
        
        # Selection
        selected = np.random.choice(population_size, size=population_size, p=[1 - f/np.sum(f) for f in fitness])
        selected = [population[i] for i in selected]
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if i+1 < population_size:
                parent1 = selected[i]
                parent2 = selected[i+1]
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                offspring.append(child1)
                offspring.append(child2)
            else:
                offspring.append(selected[i])
        
        # Mutation
        mutated = []
        for h in offspring:
            if np.random.rand() < 0.1:
                idx = np.random.randint(n_points)
                h = h.copy()
                h[idx] = np.clip(h[idx] + np.random.uniform(-0.1, 0.1), 0, 1)
                h = normalize(h)
            mutated.append(h)
        
        # Replace population
        population = mutated
    
    best_h = best_h
    best_c5 = best_c5
    return best_h, best_c5, n_points
