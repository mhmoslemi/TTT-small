import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    # Initial solution
    h = initial_h.copy()
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 200
    mutation_rate = 0.15
    crossover_rate = 0.8
    tournament_size = 5
    
    # Generate initial population
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    
    for gen in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        
        # Select parents using tournament selection
        parents = []
        for _ in range(population_size):
            contestants = np.random.choice(range(population_size), tournament_size, replace=False)
            best_idx = np.argmin([fitness[i] for i in contestants])
            parents.append(population[contestants[best_idx]])
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if np.random.rand() < crossover_rate:
                p1 = parents[i]
                p2 = parents[i+1] if i+1 < len(parents) else parents[0]
                child1 = np.concatenate((p1[:n_points//2], p2[n_points//2:]))
                child2 = np.concatenate((p2[:n_points//2], p1[n_points//2:]))
                offspring.extend([child1, child2])
            else:
                offspring.append(parents[i])
                offspring.append(parents[i+1] if i+1 < len(parents) else parents[0])
        
        # Mutation
        for i in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                mutation = np.random.uniform(-0.1, 0.1, n_points)
                offspring[i] = np.clip(offspring[i] + mutation, 0, 1)
        
        # Replace population
        population = offspring
        
        # Track best solution
        current_best = min(population, key=evaluate)
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
