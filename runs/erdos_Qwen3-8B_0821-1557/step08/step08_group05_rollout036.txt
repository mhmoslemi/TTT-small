import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    # Initial solution
    h = initial_h_values.copy()
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Adaptive simulated annealing parameters
    temp = 1.0
    cooling_rate = 0.995
    steps = 1000
    
    for _ in range(steps):
        if time.time() - t0 > budget_s:
            break
        # Generate neighbor
        h_neighbor = h.copy()
        i = np.random.randint(n_points)
        delta = np.random.uniform(-0.1, 0.1)
        h_neighbor[i] += delta
        h_neighbor = np.clip(h_neighbor, 0, 1)
        h_neighbor = normalize(h_neighbor)
        
        # Acceptance probability
        energy = evaluate(h_neighbor)
        if energy < best_c5 or np.random.rand() < np.exp((best_c5 - energy) / temp):
            h = h_neighbor
            if energy < best_c5:
                best_h = h.copy()
                best_c5 = energy
        
        temp *= cooling_rate
    
    # Structured genetic algorithm with adaptive binning
    population_size = 50
    generations = 60
    mutation_rate = 0.15
    elite_size = 5
    
    for gen in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        # Create new population with diversity
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population[0] = best_h.copy()
        
        # Evaluate population
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        best_pop_h = population[best_idx].copy()
        best_pop_c5 = fitness[best_idx]
        
        # Keep elite solutions
        elite_indices = np.argsort(fitness)[:elite_size]
        elite_h = [population[i].copy() for i in elite_indices]
        
        # Apply adaptive mutation with binning
        for i in range(len(elite_h)):
            h = elite_h[i].copy()
            h = normalize(h)
            
            # Adaptive mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    h[j] += np.random.uniform(-0.1, 0.1)
                    h[j] = np.clip(h[j], 0, 1)
            h = normalize(h)
            
            # Local optimization with constraints
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
            res = optimize.minimize(
                objective,
                h,
                bounds=[(0, 1) for _ in range(n_points)],
                constraints=cons,
                method='SLSQP',
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            elite_h[i] = res.x
        
        # Combine elites and best population solution
        population = elite_h + [best_pop_h]
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
    
    return best_h, best_c5, n_points
