import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)

    temp = 1.0

    for _ in range(1000):
        if time.time() - t0 > budget_s:
            break
        # Perturb the solution
        perturbation = np.random.normal(0, 0.05, n_points)
        h_test = h + perturbation
        h_test = np.clip(h_test, 0, 1)
        h_test = normalize(h_test)
        # Accept with probability based on temperature
        new_c5 = evaluate(h_test)

        if delta < 0 or np.random.random() < np.exp(-delta / temp):
            h = h_test
            best_c5 = new_c5

        temp *= cool_rate
    
    # Gradient-based optimization with constraints
    for _ in range(50):
