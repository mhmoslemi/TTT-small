import numpy as np
import scipy.signal
import copy

def run(seed=42, budget_s=10, **kwargs):

    dx = 2.0 / n_points
    
    # Function to evaluate C5
    def evaluate(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = scipy.signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.15
    tournament_size = 5
    
    # Initialize population
    population = [copy.deepcopy(h_initial) for _ in range(population_size)]
    best_h = copy.deepcopy(h_initial)
    best_c5 = evaluate(best_h)
    
    for generation in range(generations):
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        
        # Find the best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = copy.deepcopy(population[best_idx])
        
        # Selection and reproduction
        new_population = []
        for _ in range(population_size):
            # Tournament selection
            contestants = np.random.choice(range(population_size), tournament_size, replace=False)
            winner = contestants[np.argmin([fitness[i] for i in contestants])]
            parent = copy.deepcopy(population[winner])
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, 0.05, size=n_points)
                parent += mutation
                parent = np.clip(parent, 0, 1)
            
            # Normalize to maintain integral constraint
            parent = parent / np.sum(parent) * (n_points / 2)
            new_population.append(parent)
        
        population = new_population
        
        # Check time budget
        if generation >= generations - 1:
            break
    
    return best_h, best_c5, n_points
