import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    elite_size = 2
    t0 = time.time()
    
    # Initialize population
    def create_individual():
        h = np.random.uniform(0, 1, n_points)
        return normalize(h)
    
    population = [create_individual() for _ in range(population_size)]
    best_h = initial_h.copy()
    best_c5 = evaluate(initial_h)
    
    for gen in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        
        # Find elite individuals
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elite.copy()
        
        # Generate offspring
        for _ in range(population_size - elite_size):
            # Select parents
            parent1 = random.choice(elite)
            parent2 = random.choice(elite)
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] = np.clip(child[i] + np.random.uniform(-0.1, 0.1), 0, 1)
            
            # Normalize
            child = normalize(child)
            new_population.append(child)
        
        population = new_population
        
        # Update best solution
        for ind in population:
            current_c5 = evaluate(ind)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = ind.copy()
    
    # Local optimization on best solution
    for _ in range(50):
        if time.time() - t0 > budget_s:
            break
        
        # Use SLSQP optimizer with bounds and constraint
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        res = optimize.minimize(
            lambda x: evaluate(x),
            best_h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        best_h = res.x
        best_h = np.clip(best_h, 0, 1)
        best_h = normalize(best_h)
        if evaluate(best_h) < best_c5:
            best_c5 = evaluate(best_h)
    
    return best_h, best_c5, n_points
