import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    # Initial solution
    h = initial_h.copy()
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Local optimization
    for _ in range(50):
        if time.time() - t0 > budget_s:
            break
        # Use SLSQP optimizer with bounds and constraint
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        res = optimize.minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        h = res.x
        h = np.clip(h, 0, 1)
        h = normalize(h)
        if evaluate(h) < best_c5:
            best_h = h.copy()
            best_c5 = evaluate(h)
    
    # Genetic algorithm for global search
    population_size = 20
    generations = 10
    mutation_rate = 0.1
    elite_size = 2
    
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    
    for _ in range(generations):
        if time.time() - t0 > budget_s:
            break
        # Evaluate fitness
        fitness = [evaluate(ind) for ind in population]
        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        # Create new population
        new_population = elites.copy()
        while len(new_population) < population_size:
            # Select parents
            parent1 = random.choice(elites)
            parent2 = random.choice(elites)
            # Crossover
            crossover_point = random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] = np.random.uniform(0, 1)
            child = normalize(child)
            new_population.append(child)
        population = new_population
    
    # Evaluate final population
    final_fitness = [evaluate(ind) for ind in population]
    best_index = np.argmin(final_fitness)
    best_h = population[best_index]
    best_c5 = final_fitness[best_index]
    
    return best_h, best_c5, n_points
