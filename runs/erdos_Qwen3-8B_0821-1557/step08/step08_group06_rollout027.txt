        h = np.clip(h, 0, 1)
        h = normalize(h)

        return np.max(corr) * dx

    population_size = 20
    population = [normalize(np.random.rand(n_points)) for _ in range(population_size)]

        fitness = np.array([evaluate(ind) for ind in population])
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]

        # Select parents
        parents = np.random.choice(population_size, size=population_size, p=fitness/np.sum(fitness))
        parents = [population[i] for i in parents]
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if i+1 < population_size:
                p1 = parents[i]
                p2 = parents[i+1]
                child1 = np.copy(p1)
                child2 = np.copy(p2)
                crossover_point = np.random.randint(1, n_points-1)
                child1[crossover_point:] = p2[crossover_point:]
                child2[crossover_point:] = p1[crossover_point:]
                offspring.append(child1)

        # Mutation

        if evaluate(best_h) < best_c5:
