import scipy.optimize as optimize
import time

        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)

    temp = 1.0
    cooling_rate = 0.995
    iterations = 1000
    perturb_size = 0.1
    
    for _ in range(iterations):
        if time.time() - t0 > budget_s:
            break
        
        # Generate a perturbed solution
        perturb = np.random.uniform(-perturb_size, perturb_size, n_points)
        h_perturb = h + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = normalize(h_perturb)
        
        # Evaluate and accept with probability
        c5_perturb = evaluate(h_perturb)
        if c5_perturb < best_c5 or np.random.rand() < np.exp((best_c5 - c5_perturb) / temp):
            h = h_perturb
            best_h = h.copy()
            best_c5 = c5_perturb

        method='SLSQP',

    )
