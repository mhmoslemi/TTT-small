import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    # Initial solution
    h = initial_h.copy()
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Local optimization
    for _ in range(50):
        if time.time() - t0 > budget_s:
            break
        # Use SLSQP optimizer with bounds and constraint
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        res = optimize.minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        h = res.x
        h = np.clip(h, 0, 1)
        h = normalize(h)
        if evaluate(h) < best_c5:
            best_h = h.copy()
            best_c5 = evaluate(h)
    
    # Particle swarm optimization
    class Particle:
        def __init__(self, n_points):
            self.position = np.random.uniform(0, 1, n_points)
            self.velocity = np.random.uniform(-1, 1, n_points)
            self.pbest_position = self.position.copy()
            self.pbest_value = evaluate(self.position)
            self.position = normalize(self.position)
        
        def update(self, gbest_position, inertia=0.5, cognitive=1.5, social=1.5):
            r1, r2 = np.random.rand(), np.random.rand()
            self.velocity = inertia * self.velocity + cognitive * r1 * (self.pbest_position - self.position) + social * r2 * (gbest_position - self.position)
            self.position += self.velocity
            self.position = np.clip(self.position, 0, 1)
            self.position = normalize(self.position)
            current_value = evaluate(self.position)
            if current_value < self.pbest_value:
                self.pbest_position = self.position.copy()
                self.pbest_value = current_value
    
    particles = [Particle(n_points) for _ in range(30)]
    gbest_position = particles[0].pbest_position
    gbest_value = particles[0].pbest_value

            if p.pbest_value < gbest_value:
                gbest_position = p.pbest_position.copy()
                gbest_value = p.pbest_value
    
    if gbest_value < best_c5:
        best_h = gbest_position.copy()
        best_c5 = gbest_value
    
    return best_h, best_c5, n_points
