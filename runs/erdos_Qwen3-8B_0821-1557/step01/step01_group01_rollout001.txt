from scipy.optimize import minimize

def run(seed=42, budget_s=10, **kwargs):

    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if not np.isclose(sum_h, n_points / 2):
            return np.inf

        c5 = np.max(corr) * dx
        return c5
