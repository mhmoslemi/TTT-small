import numpy as np
import random
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = initial_h.copy()

    def objective(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) * dx) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def perturb(h):
        h = h.copy()
        i = np.random.randint(len(h))
        h[i] += np.random.uniform(-0.1, 0.1)
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) * dx) * (n_points / 2)
        return h

    start_time = time.time()
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        h = perturb(initial_h)
        result = minimize(objective, h, method='L-BFGS-B', options={'ftol': 1e-6, 'maxiter': 50})
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
        initial_h = best_h

    return best_h, best_c5, n_points
