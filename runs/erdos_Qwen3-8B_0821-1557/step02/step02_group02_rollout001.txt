def compute_c5(h, dx):
    h_shifted = np.concatenate([h[-1:], h[:-1]])
    overlap = correlate(h, 1 - h_shifted, mode="full")
    return np.max(overlap) * dx

def normalize(h, n_points):
    total = sum(h)
    factor = (n_points / 2) / total if total != 0 else 1.0
    return [x * factor for x in h]

def mutate(h, mutation_rate, n_points):
    h = copy.copy(h)
    for i in range(n_points):
        if np.random.rand() < mutation_rate:
            h[i] = np.random.uniform(0, 1)
    return normalize(h, n_points)
