import numpy as np
import random
import scipy.signal
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Choose a reasonable number of points
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    
    def evaluate_individual(h):
        h = np.clip(h, 0, 1)
        conv = scipy.signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    def normalize_individual(h):
        return h * (n_points / 2) / np.sum(h)
    
    # Initialize population
    population = [normalize_individual(np.random.rand(n_points)) for _ in range(population_size)]
    
    best_h = initial_h_values
    best_c5 = evaluate_individual(best_h)
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [evaluate_individual(h) for h in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Selection
        selected = np.argsort(fitness)[:population_size//2]
        selected = np.random.choice(selected, size=population_size, replace=True)
        new_population = [population[i] for i in selected]
        
        # Crossover
        for i in range(population_size):
            if np.random.rand() < 0.8:
                j = np.random.randint(population_size)
                parent1 = population[selected[i]]
                parent2 = population[selected[j]]
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                new_population[i] = child
        
        # Mutation
        for i in range(population_size):
            h = new_population[i]
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    h[j] += np.random.uniform(-0.1, 0.1)
                    h[j] = np.clip(h[j], 0, 1)
            new_population[i] = normalize_individual(h)
        
        population = new_population
    
    return best_h, best_c5, n_points
