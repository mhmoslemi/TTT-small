import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = initial_h.copy()
    
    def objective(h_values):
        h = h_values.copy()
        h /= np.sum(h) * dx  # normalize to satisfy ∫h(x)dx = 1
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2

    # Optimization
    start_time = time.time()
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    # Check time and return best solution
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h, objective(h), n_points)
    else:
        return (result.x, objective(result.x), n_points)
