import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 100
    population = [np.array(initial_h, dtype=np.float64) for _ in range(population_size)]
    
    def evaluate_individual(h):
        h = h.copy()
        h /= np.sum(h) * dx
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def create_individual():
        h = np.random.rand(n_points)
        h /= np.sum(h) * dx
        return h
    
    def crossover(parent1, parent2):
        split = random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:split], parent2[split:]])
        child /= np.sum(child) * dx
        return child
    
    def mutate(h, mutation_rate=0.05):
        for i in range(n_points):
            if random.random() < mutation_rate:
                h[i] = np.random.rand()
        h /= np.sum(h) * dx
        return h
    
    best_h = np.array(initial_h, dtype=np.float64)
    best_c5 = evaluate_individual(best_h)
    
    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        fitness = [evaluate_individual(ind) for ind in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_best_c5 = fitness[best_idx]
        
        if current_best_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_best_c5
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Select parents
            parent1 = random.choices(population, weights=fitness)[0]
            parent2 = random.choices(population, weights=fitness)[0]
            
            # Crossover
            child = crossover(parent1, parent2)
            
            # Mutate
            child = mutate(child)
            
            new_population.append(child)
        
        # Replace population
        population = new_population
        
        # Add some random individuals to maintain diversity
        for _ in range(population_size // 10):
            population.append(create_individual())
    
    return best_h.tolist(), best_c5, n_points
