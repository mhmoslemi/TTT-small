import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    def normalize(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        return h * (n_points / 2) / total if total != 0 else h

    def mutate(h, mutation_rate=0.05):
        h = np.array(h, dtype=np.float64)
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        return h

    def crossover(parent1, parent2):
        h = np.array(parent1, dtype=np.float64)
        for i in range(len(h)):
            if np.random.rand() < 0.5:
                h[i] = parent2[i]
        return h

    # Initial population
    population_size = 20
    population = [normalize(mutate(initial_h.copy())) for _ in range(population_size)]
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)

    start_time = time.time()
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        # Find best individual
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_c5 = fitness[best_idx]
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
        # Select parents
        parents = np.random.choice(population, size=population_size, replace=True)
        # Generate new population
        new_population = []
        for i in range(population_size // 2):
            parent1 = parents[i]
            parent2 = parents[population_size - 1 - i]
            child1 = crossover(parent1, parent2)
            child1 = normalize(mutate(child1))
            child2 = crossover(parent2, parent1)
            child2 = normalize(mutate(child2))
            new_population.extend([child1, child2])
        population = new_population

    return best_h.tolist(), best_c5, n_points
