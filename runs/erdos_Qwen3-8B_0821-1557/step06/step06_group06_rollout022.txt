import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def fitness(h):
        return -evaluate(h)  # Minimize C5 by maximizing negative
    
    # Initial population
    population_size = 50
    population = [normalize(np.array(initial_h, dtype=np.float64) + np.random.uniform(-0.2, 0.2, n_points)) 
                  for _ in range(population_size)]
    
    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s - 0.1:
            break
        
        # Evaluate fitness
        fitness_values = [fitness(ind) for ind in population]
        
        # Select top 20%
        top_idx = np.argsort(fitness_values)[-int(0.2*len(population)):]
        top_population = [population[i] for i in top_idx]
        
        # Crossover
        new_population = []
        while len(new_population) < population_size:
            parent1, parent2 = random.choices(top_population, k=2)
            child = (parent1 + parent2) / 2
            child = np.clip(child, 0, 1)
            child = normalize(child)
            new_population.append(child)
        
        population = new_population
    
    best_h = min(population, key=evaluate)
    best_c5 = evaluate(best_h)
    return best_h.tolist(), best_c5, n_points
