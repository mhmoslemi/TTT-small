    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    # Initial guess
    h_initial = np.array(initial_h, dtype=np.float64)
    h_initial = h_initial / (np.sum(h_initial) * dx) * (n_points / 2)

    bounds = [(0, 1) for _ in range(n_points)]

    # Genetic algorithm parameters
    population_size = 20
    generations = 50
    mutation_rate = 0.15
    crossover_rate = 0.8
    elite_size = 2

    def breed(parent1, parent2):
        if np.random.rand() < crossover_rate:
            split = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:split], parent2[split:]])
        else:
            child = parent1.copy()
        return child

    def mutate(individual):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] = np.clip(np.random.rand(), 0, 1)
        return individual

    def normalize(individual):
        total = np.sum(individual)
        return individual * (n_points / 2) / total

    def evaluate_population(population):
        return [objective(normalize(ind)) for ind in population]

    # Initial population
    population = [normalize(np.random.rand(n_points)) for _ in range(population_size)]
    population[0] = h_initial

    start_time = time.time()
    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break
        fitness = evaluate_population(population)
        ranked = [x for _, x in sorted(zip(fitness, population))]
        population = ranked[:elite_size]
        
        while len(population) < population_size:
            parent1 = random.choice(ranked)
            parent2 = random.choice(ranked)
            child = breed(parent1, parent2)
            child = mutate(child)
            population.append(child)
        
        # Early stopping if we have enough time
        if time.time() - start_time > budget_s - 1:
            break

    best_h = min(population, key=objective)
