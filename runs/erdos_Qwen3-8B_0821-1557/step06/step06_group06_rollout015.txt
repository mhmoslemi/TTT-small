import random

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    # Initial population
    population_size = 30
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    
    # Evaluation function
    def fitness(individual):
        return -evaluate(individual)  # Minimize C5
    
    # Mutation function
    def mutate(individual, mutation_rate=0.1):
        individual = individual.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] = np.random.uniform(0, 1)
        return normalize(individual)
    
    # Crossover function
    def crossover(parent1, parent2):
        child = np.zeros(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return normalize(child)
    
    start_time = time.time()
    best_h = initial_h
    best_c5 = evaluate(initial_h)
    
    for generation in range(100):
        if time.time() - start_time > budget_s - 0.1:

        # Evaluate population
        fitnesses = [fitness(ind) for ind in population]
        best_idx = np.argmax(fitnesses)
        best_c5 = -fitnesses[best_idx]
        best_h = population[best_idx]
        
        # Create new population

            parent1 = population[np.random.choice(len(population))]

            # Crossover
            child = crossover(parent1, parent2)
            
            # Mutate
            mutated_child = mutate(child)
