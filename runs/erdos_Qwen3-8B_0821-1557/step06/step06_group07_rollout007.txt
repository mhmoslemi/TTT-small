import scipy.optimize as optimize
import time
import random

        if total == 0:

        return h * (n_points / 2) / total
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

            new_h = np.clip(h + perturbation, 0, 1)
            new_h = normalize(new_h)
            if objective(new_h) < objective(h):
                h = new_h
        return h
    
    t0 = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    try:
        result = optimize.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,

        )
