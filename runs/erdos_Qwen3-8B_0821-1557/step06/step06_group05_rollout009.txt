import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    def normalize(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        return h * (n_points / 2) / total if total != 0 else h

    # Initial guess
    h_initial = np.array(initial_h, dtype=np.float64)
    h_initial = normalize(h_initial)

    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization
    start_time = time.time()
    result = optimize.minimize(
        fun=objective,
        x0=h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_integral}],
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    # Adaptive local search
    best_h = result.x
    best_c5 = result.fun
    best_time = time.time() - start_time

    # Perform adaptive local search
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        perturb = np.random.normal(0, 0.05, size=n_points)
        perturb = np.clip(perturb, -0.1, 0.1)
        candidate = best_h + perturb
        candidate = normalize(candidate)
        try:
            new_c5 = objective(candidate)
            if new_c5 < best_c5:
                best_h = candidate
                best_c5 = new_c5
        except:
            pass

    return best_h.tolist(), best_c5, n_points
