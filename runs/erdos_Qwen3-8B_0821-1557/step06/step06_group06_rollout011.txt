import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        return normalize(h)
    
    def crossover(parent1, parent2):
        h = parent1.copy()
        for i in range(len(h)):
            if np.random.rand() < 0.5:
                h[i] = parent2[i]
        return normalize(h)
    
    def genetic_algorithm():
        population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(20)]
        best_h = initial_h.copy()
        best_c5 = evaluate(initial_h)
        
        start_time = time.time()
        
        while time.time() - start_time < budget_s - 0.1:
            # Evaluate fitness
            fitness = [evaluate(ind) for ind in population]
            
            # Select top performers
            top_indices = np.argsort(fitness)[:10]
            elite = [population[i] for i in top_indices]
            
            # Generate new population
            new_population = elite.copy()
            while len(new_population) < 20:
                parent1, parent2 = random.choices(elite, k=2)
                child = crossover(parent1, parent2)
                if np.random.rand() < 0.1:
                    child = mutate(child)
                new_population.append(child)
            
            population = new_population
            current_best = min(zip(fitness, population), key=lambda x: x[0])
            if current_best[0] < best_c5:
                best_c5 = current_best[0]
                best_h = current_best[1]
        
        return best_h, best_c5, n_points
    
    return genetic_algorithm()
