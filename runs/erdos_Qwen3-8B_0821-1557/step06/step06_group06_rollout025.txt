import time
import random

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def fitness(h):
        return -evaluate(h)  # Minimize the overlap integral
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    pop_size = 50
    population = [np.array(initial_h, dtype=np.float64) + np.random.normal(0, 0.05, n_points) for _ in range(pop_size)]
    population = [np.clip(p, 0, 1) for p in population]
    
    start_time = time.time()
    
    for generation in range(200):
        if time.time() - start_time > budget_s - 0.1:
            break
        
        # Evaluate fitness
        fitness_values = [fitness(p) for p in population]
        best_idx = np.argmax(fitness_values)
        best_h = population[best_idx]
        best_c5 = -fitness_values[best_idx]
        
        # Selection
        selected = np.random.choice(population, size=pop_size, p=np.array(fitness_values)/np.sum(fitness_values))
        
        # Crossover
        new_population = []
        for i in range(pop_size//2):
            p1 = selected[i]
            p2 = selected[pop_size//2 + i]
            child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
            new_population.append(child)
        
        # Mutation
        for i in range(pop_size):

                mutation = np.random.normal(0, 0.05, n_points)
                new_population[i] += mutation
                new_population[i] = np.clip(new_population[i], 0, 1)
        
        population = new_population
    
    # Final evaluation
    best_h = np.array(population[np.argmax([fitness(p) for p in population])])
    best_c5 = -fitness(best_h)
