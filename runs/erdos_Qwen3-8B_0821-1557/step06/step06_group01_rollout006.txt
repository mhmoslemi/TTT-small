import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    # Initial solution
    h = initial_h.copy()
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Local search parameters
    local_steps = 50
    local_perturb = 0.05
    
    # Genetic algorithm parameters
    population_size = 30
    generations = 50
    mutation_rate = 0.25
    elite_size = 3
    
    for gen in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        # Local search on current best
        for _ in range(local_steps):
            perturb = np.random.uniform(-local_perturb, local_perturb, n_points)
            candidate = h + perturb
            candidate = np.clip(candidate, 0, 1)
            candidate = normalize(candidate)
            if evaluate(candidate) < best_c5:
                h = candidate
                best_c5 = evaluate(h)
        
        # Create new population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        
        # Evaluate population
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        best_pop_h = population[best_idx].copy()
        best_pop_c5 = fitness[best_idx]
        
        # Keep elite solutions
        elite_indices = np.argsort(fitness)[:elite_size]
        elite_h = [population[i].copy() for i in elite_indices]
        
        # Apply local optimization to elite solutions
        for i in range(len(elite_h)):
            h = elite_h[i].copy()
            h = normalize(h)
            # Use SLSQP optimizer with bounds and constraint
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
            res = optimize.minimize(
                objective,
                h,
                bounds=[(0, 1) for _ in range(n_points)],
                constraints=cons,
                method='SLSQP'
            )
            elite_h[i] = res.x
        
        # Combine elites and best population solution
        population = elite_h + [best_pop_h]
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
    
    return best_h, best_c5, n_points
