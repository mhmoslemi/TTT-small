def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)

    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def normalize(h):
        return h * (n_points / 2) / np.sum(h)

    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        return normalize(h)

    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(len(child)):
            if np.random.rand() < 0.5:
                child[i] = parent2[i]
        return normalize(child)

    population_size = 50
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    best_h = initial_h.copy()
    best_c5 = evaluate(initial_h)

    start_time = time.time()
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        fitness = [evaluate(ind) for ind in population]
        idx = np.argsort(fitness)
        population = [population[i] for i in idx[:population_size//2]]
        
        new_population = []
        for _ in range(population_size//2):
            p1, p2 = np.random.choice(population, 2, replace=False)
