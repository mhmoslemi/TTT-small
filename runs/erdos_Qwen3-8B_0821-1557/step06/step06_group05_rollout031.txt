import numpy as np
import scipy.optimize as optimize
import time
import copy

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2
    
    def normalize(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        return h * (n_points / 2) / total if total != 0 else h
    
    def perturb(h, magnitude=0.05):
        h = np.array(h, dtype=np.float64)
        noise = np.random.normal(0, magnitude, h.shape)
        perturbed = np.clip(h + noise, 0, 1)
        return normalize(perturbed)
    
    h_initial = np.array(initial_h, dtype=np.float64)
    h_initial = normalize(h_initial)
    
    best_h = copy.deepcopy(h_initial)
    best_c5 = objective(best_h)
    best_time = 0
    
    start_time = time.time()
    
    # Initial gradient-based optimization
    bounds = [(0, 1) for _ in range(n_points)]
    result = optimize.minimize(
        fun=objective,
        x0=h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_integral}],
        options={'ftol': 1e-8, 'maxfun': 100}
    )
    
    best_h = result.x
    best_c5 = result.fun
    best_time = time.time() - start_time
    
    # Simulated annealing
    temp = 1.0
    cooling_rate = 0.995
    iterations = 1000
    
    for i in range(iterations):
        if time.time() - start_time > budget_s:
            break
        h_perturbed = perturb(best_h)
        c5_perturbed = objective(h_perturbed)
        delta = c5_perturbed - best_c5
        if delta < 0 or np.random.random() < np.exp(-delta / temp):
            best_h = h_perturbed
            best_c5 = c5_perturbed
        temp *= cooling_rate
    
    return best_h.tolist(), best_c5, n_points
