        h = h / np.sum(h) * (n_points / 2)
        # Compute the overlap integral
        overlap = correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    best_n = n_points
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 30
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    for _ in range(generations):
        # Create population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        for i in range(population_size):
            population[i] = population[i] / np.sum(population[i]) * (n_points / 2)
        
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        
        # Select best individual
        best_idx = np.argmin(fitness)
        best_pop = population[best_idx]
        best_fit = fitness[best_idx]
        
        # Update best solution
        if best_fit < best_c5:
            best_h = best_pop.copy()
            best_c5 = best_fit
            best_n = n_points
        
        # Crossover and mutation
        for i in range(population_size):
            if np.random.rand() < crossover_rate:
                j = np.random.randint(0, population_size)
                if i != j:
                    # Crossover
                    mid = np.random.randint(1, n_points - 1)
                    child = np.concatenate([population[i][:mid], population[j][mid:]])
                    child = child / np.sum(child) * (n_points / 2)
                    # Mutation
                    for k in range(n_points):
                        if np.random.rand() < mutation_rate:
                            child[k] = np.random.uniform(0, 1)
                    population[i] = child
        
        # Local optimization on best solution
        if best_fit < best_c5:
            result = optimize.minimize(
                objective,
                best_pop,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],
