import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def objective(h):
        return evaluate(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h = np.array(initial_h, dtype=np.float64)
    
    # Optimization settings
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    start_time = time.time()
    
    # Perform optimization using SLSQP
    result = optimize.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Stop early if time budget is exceeded
    if time.time() - start_time > budget_s - 0.1:
        return h.tolist(), evaluate(h), n_points
    
    # Return best solution
    best_h = result.x
    best_c5 = evaluate(best_h)
    return best_h.tolist(), best_c5, n_points
