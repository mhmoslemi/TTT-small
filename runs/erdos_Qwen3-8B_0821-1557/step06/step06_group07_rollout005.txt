import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def constraint(h):
        h = normalize(h)
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def adaptive_local_search(h):
        for _ in range(100):
            perturbation = np.random.normal(0, 0.05, n_points)
            perturbed_h = np.clip(h + perturbation, 0, 1)
            perturbed_h = normalize(perturbed_h)
            if objective(perturbed_h) < objective(h):
                h = perturbed_h
        return h
    
    def genetic_algorithm():
        population_size = 50
        generations = 50
        mutation_rate = 0.15
        elitism = 10
        
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [normalize(p) for p in population]
        
        for _ in range(generations):
            fitness = [objective(p) for p in population]
            elites = np.argsort(fitness)[:elitism]
            parents = np.argsort(fitness)[elitism:]
            
            new_population = [population[i] for i in elites]
            
            for _ in range(population_size - elitism):
                p1, p2 = random.choices(parents, k=2)
                child = p1 * np.random.rand() + p2 * (1 - np.random.rand())
                child = np.clip(child, 0, 1)
                child = normalize(child)
                if np.random.rand() < mutation_rate:
                    child += np.random.normal(0, 0.05, n_points)
                    child = np.clip(child, 0, 1)
                new_population.append(normalize(child))
            
            population = new_population
        
        best_h = min(population, key=objective)
        return best_h, objective(best_h)
    
    t0 = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    try:
        result = optimize.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 50}
        )
        best_h = result.x
        best_c5 = objective(best_h)
    except:
        pass
    
    if time.time() - t0 < budget_s:
        try:
            ga_h, ga_c5 = genetic_algorithm()
            if ga_c5 < best_c5:
                best_h = ga_h
                best_c5 = ga_c5
        except:
            pass
    
    return best_h, best_c5, n_points
