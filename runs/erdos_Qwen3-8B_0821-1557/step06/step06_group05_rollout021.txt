import numpy as np
import scipy.optimize as optimize
import time
from copy import deepcopy

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    def normalize(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * (n_points / 2) / total

    # Initial guess
    h_initial = np.array(initial_h, dtype=np.float64)
    h_initial = normalize(h_initial)

    bounds = [(0, 1) for _ in range(n_points)]

    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.15
    generations = 50
    elite_fraction = 0.2
    local_optimization_steps = 5

    # Initial population
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    population = [h for h in population if np.isclose(np.sum(h), n_points / 2, atol=1e-6)]

    best_h = deepcopy(h_initial)
    best_c5 = objective(h_initial)

    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_best_c5 = fitness[best_idx]

        # Update best solution
        if current_best_c5 < best_c5:
            best_h = deepcopy(current_best)
            best_c5 = current_best_c5

        # Elite selection
        elite_size = int(elite_fraction * population_size)
        elites = [deepcopy(population[i]) for i in np.argsort(fitness)[:elite_size]]

        # Create new population
        new_population = elites
        while len(new_population) < population_size:
            # Select parents
            parent1 = np.random.choice(elite_size, p=[f/sum(fitness[:elite_size]) for f in fitness[:elite_size]])
            parent2 = np.random.choice(elite_size, p=[f/sum(fitness[:elite_size]) for f in fitness[:elite_size]])
            parent1 = elites[parent1]
            parent2 = elites[parent2]

            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])

            # Mutation
            for i in range(n_points):
                if np.random.random() < mutation_rate:
                    child[i] = np.random.uniform(0, 1)

            # Normalize
            child = normalize(child)
            new_population.append(child)

        population = new_population

        # Local optimization on best solution
        for _ in range(local_optimization_steps):
            result = optimize.minimize(
                fun=objective,
                x0=best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_integral}],
                options={'ftol': 1e-8, 'maxiter': 10}
            )
            if result.success:
                best_h = result.x
                best_c5 = result.fun

    return best_h.tolist(), best_c5, n_points
