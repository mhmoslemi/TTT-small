def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        return normalize(h)
    
    def crossover(parent1, parent2):
        h = parent1.copy()
        for i in range(len(h)):
            if np.random.rand() < 0.5:
                h[i] = parent2[i]
        return normalize(h)
    
    def fitness(h):
        return -evaluate(h)  # Minimize the overlap integral
    
    # Initial population
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(20)]
    
    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s - 0.1:
            break
        
        # Evaluate fitness
        fitness_scores = [fitness(ind) for ind in population]
        
        # Select top 10 individuals
        top_indices = np.argsort(fitness_scores)[-10:]
        top_population = [population[i] for i in top_indices]
        
        # Create new population
        new_population = []
        while len(new_population) < len(population):
            parent1, parent2 = np.random.choice(top_population, 2, replace=False)
            child = crossover(parent1, parent2)
            if np.random.rand() < 0.1:
                child = mutate(child)
            new_population.append(child)
        
        population = new_population
    
    best_h = min(population, key=evaluate)
    best_c5 = evaluate(best_h)
