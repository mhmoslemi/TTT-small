import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    population_size = 30
    generations = 100
    mutation_rate = 0.1
    tournament_size = 5
    
    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2
    
    def normalize(h):
        total = np.sum(h)
        return h * (n_points / 2) / total if total != 0 else h
    
    # Initialize population
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    
    start_time = time.time()
    
    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        
        # Select parents using tournament selection
        parents = []
        for _ in range(population_size):
            contestants = random.sample(list(enumerate(fitness)), tournament_size)
            winner_idx = min(contestants, key=lambda x: x[1])[0]
            parents.append(population[winner_idx])
        
        # Apply crossover and mutation
        new_population = []
        for i in range(population_size // 2):
            parent1 = parents[2*i]
            parent2 = parents[2*i + 1]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child1 = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            child2 = np.concatenate((parent2[:crossover_point], parent1[crossover_point:]))
            
            # Mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] = np.random.uniform(0, 1)
                    child2[j] = np.random.uniform(0, 1)
            
            new_population.extend([normalize(child1), normalize(child2)])
        
        population = new_population
    
    # Find best solution
    best_h = min(population, key=objective)
    best_c5 = objective(best_h)
    
    return best_h.tolist(), best_c5, n_points
