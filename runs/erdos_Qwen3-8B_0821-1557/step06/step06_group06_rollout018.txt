import numpy as np
import random
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] = np.clip(random.random(), 0, 1)
        return normalize(h)
    
    def crossover(h1, h2):
        h = h1.copy()
        for i in range(len(h)):
            if random.random() < 0.5:
                h[i] = h2[i]
        return normalize(h)
    
    population_size = 20
    population = [normalize(initial_h.copy()) for _ in range(population_size)]
    
    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s - 0.1:
            break
        fitness = [evaluate(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Create new population
        new_population = [best_h.copy()]
        while len(new_population) < population_size:
            p1, p2 = random.sample(population, 2)
            child = crossover(p1, p2)
            if random.random() < 0.1:
                child = mutate(child)
            new_population.append(child)
        
        population = new_population
    
    best_h = min(population, key=evaluate)
    best_c5 = evaluate(best_h)
    return best_h.tolist(), best_c5, n_points
