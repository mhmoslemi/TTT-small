import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def constraint(h):
        h = normalize(h)
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial optimization with SLSQP
    def optimize_with_slsqp(h0):
        result = optimize.minimize(
            objective,
            h0,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 50}
        )
        return result.x, objective(result.x)
    
    # Local search with perturbations
    def local_search(h, perturbation_factor=0.05, max_iter=100):
        h = h.copy()
        for _ in range(max_iter):
            perturbed = h + perturbation_factor * np.random.randn(n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = normalize(perturbed)
            if objective(perturbed) < objective(h):
                h = perturbed
        return h
    
    t0 = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    try:
        # First optimization
        best_h, best_c5 = optimize_with_slsqp(best_h)
    except:
        pass
    
    if time.time() - t0 < budget_s:
        try:
            # Local search refinement
            best_h = local_search(best_h)
            best_c5 = objective(best_h)
        except:
            pass
    
    return best_h, best_c5, n_points
