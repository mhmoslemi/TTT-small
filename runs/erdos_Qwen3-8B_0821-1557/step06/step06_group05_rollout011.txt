import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    def normalize(h):
        h = np.array(h, dtype=np.float64)
        total = np.sum(h)
        return h * (n_points / 2) / total if total != 0 else h

    def mutate(h, mutation_rate=0.1):
        h = np.array(h, dtype=np.float64)
        for i in range(len(h)):
            if np.random.random() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        return h

    def crossover(h1, h2):
        h1 = np.array(h1, dtype=np.float64)
        h2 = np.array(h2, dtype=np.float64)
        crossover_point = np.random.randint(1, len(h1) - 1)
        child = np.concatenate((h1[:crossover_point], h2[crossover_point:]))
        return child

    population_size = 20
    population = [normalize(mutate(initial_h.copy())) for _ in range(population_size)]
    best_h = initial_h.copy()
    best_c5 = evaluate(initial_h)

    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = [evaluate(ind) for ind in population]
        # Select parents
        parents = np.argsort(fitness)[:population_size // 2]
        parents = [population[i] for i in parents]
        # Generate new population
        new_population = []
        for _ in range(population_size):
            # Crossover
            parent1, parent2 = random.sample(parents, 2)
            child = crossover(parent1, parent2)
            # Mutate
            child = mutate(child)
            # Normalize
            child = normalize(child)
            new_population.append(child)
        population = new_population
        # Update best solution
        for ind in population:
            current_c5 = evaluate(ind)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = ind.copy()
    return best_h.tolist(), best_c5, n_points
