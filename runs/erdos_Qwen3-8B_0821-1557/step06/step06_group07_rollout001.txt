import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def constraint(h):
        h = normalize(h)
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def greedy_local_search(h):
        h = h.copy()
        for _ in range(100):
            i = np.random.randint(n_points)
            perturbation = np.random.uniform(-0.1, 0.1)
            h[i] += perturbation
            h = np.clip(h, 0, 1)
            h = normalize(h)
            if objective(h) < objective(h - perturbation * np.eye(n_points)[i]):
                continue
            else:
                h -= perturbation * np.eye(n_points)[i]
        return h
    
    t0 = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    try:
        result = optimize.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 50}
        )
        best_h = result.x
        best_c5 = objective(best_h)
    except:
        pass
    
    if time.time() - t0 < budget_s:
        try:
            best_h = greedy_local_search(best_h)
            best_c5 = objective(best_h)
        except:
            pass
    
    return best_h, best_c5, n_points
