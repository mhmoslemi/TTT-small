import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2

    def normalize(h):
        h = np.array(h, dtype=np.float64)
        return h * (n_points / 2) / np.sum(h)

    # Initial guess
    h_initial = np.array(initial_h, dtype=np.float64)
    h_initial = normalize(h_initial)

    bounds = [(0, 1) for _ in range(n_points)]

    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    elite_size = 2

    # Initial population
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    population = [h for h in population if np.abs(np.sum(h) - n_points / 2) < 1e-6]

    best_h = h_initial
    best_c5 = objective(h_initial)

    start_time = time.time()

    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        fitness = np.array(fitness)
        
        # Select elite
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = [population[i] for i in elite_indices]
        
        # Select parents
        parents = np.random.choice(population, size=population_size - elite_size, replace=False)
        
        # Crossover
        offspring = []
        for _ in range(population_size - elite_size):
            parent1 = random.choice(parents)
            parent2 = random.choice(parents)
            child = np.copy(parent1)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent2[i]
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if random.random() < mutation_rate:
                    offspring[i][j] = np.clip(np.random.normal(offspring[i][j], 0.1), 0, 1)
            offspring[i] = normalize(offspring[i])
        
        # Combine populations
        population = elite + offspring
        
        # Update best solution
        for ind in population:
            current_c5 = objective(ind)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = ind
    
    # Local optimization on best solution
    if time.time() - start_time < budget_s:
        result = optimize.minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_integral}],
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        best_h = result.x
        best_c5 = result.fun

    return best_h.tolist(), best_c5, n_points
