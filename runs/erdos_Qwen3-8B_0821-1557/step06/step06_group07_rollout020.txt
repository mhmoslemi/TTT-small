import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points

        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def constraint(h):
        h = normalize(h)
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def adaptive_local_search(h):
        for _ in range(100):
            perturbation = np.random.normal(0, 0.05, n_points)
            candidate = np.clip(h + perturbation, 0, 1)
            candidate = normalize(candidate)
            if objective(candidate) < objective(h):
                h = candidate
        return h
    
    t0 = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    try:
        result = optimize.minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},

        )
        best_h = result.x
        best_c5 = objective(best_h)
