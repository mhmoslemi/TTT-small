import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] = np.random.rand()
        return normalize(h)
    
    def crossover(h1, h2):
        h = h1.copy()
        for i in range(len(h)):
            if random.random() < 0.5:
                h[i] = h2[i]
        return normalize(h)
    
    def fitness(h):
        return evaluate(h)
    
    population_size = 50
    population = [normalize(np.array(initial_h, dtype=np.float64)) for _ in range(population_size)]
    
    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s - 0.1:
            break
        
        # Evaluate fitness
        fitness_scores = [fitness(ind) for ind in population]
        
        # Select parents
        parents = np.argsort(fitness_scores)[:population_size//2]
        
        # Create new population
        new_population = []
        for _ in range(population_size//2):
            parent1 = population[parents[random.randint(0, len(parents)-1)]]
            parent2 = population[parents[random.randint(0, len(parents)-1)]]
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
    
    best_h = min(population, key=fitness)
    best_c5 = evaluate(best_h)
