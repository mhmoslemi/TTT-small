import numpy as np
import random
import time
import scipy.optimize as optimize

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def mutate(h, mutation_rate=0.05):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] = np.clip(h[i] + np.random.normal(0, 0.1), 0, 1)
        return normalize(h)
    
    def crossover(parent1, parent2):
        h = parent1.copy()
        for i in range(len(h)):
            if random.random() < 0.5:
                h[i] = parent2[i]
        return normalize(h)
    
    def genetic_algorithm():
        population = [normalize(np.array(initial_h, dtype=np.float64))]
        for _ in range(100):
            if time.time() - start_time > budget_s - 0.1:
                break
            new_population = []
            for _ in range(len(population)):
                parent1 = random.choice(population)
                parent2 = random.choice(population)
                child = crossover(parent1, parent2)
                child = mutate(child)
                new_population.append(child)
            population = new_population
        best_h = min(population, key=evaluate)
        return best_h
    
    start_time = time.time()
    best_h = genetic_algorithm()
    best_c5 = evaluate(best_h)
