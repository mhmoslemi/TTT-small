import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)

    def objective(h):

        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def constraint_integral(h):

        return np.sum(h) - n_points / 2

    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    best_time = 0

    # Local search parameters
    temp = 1.0
    cooling_rate = 0.995
    iterations = 1000

    start_time = time.time()
    for _ in range(iterations):
        if time.time() - start_time > budget_s - best_time:
            break

        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed_h = np.clip(h_initial + perturbation, 0, 1)

        perturbed_h = perturbed_h / (np.sum(perturbed_h) * dx) * (n_points / 2)
        current_c5 = objective(perturbed_h)
        # Accept with probability based on simulated annealing
        if np.random.rand() < np.exp((best_c5 - current_c5) / temp):

    result = optimize.minimize(
