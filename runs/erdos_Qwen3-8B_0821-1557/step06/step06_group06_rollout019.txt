import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        return normalize(h)
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(len(child)):
            if random.random() < 0.5:
                child[i] = parent2[i]
        return normalize(child)
    
    population_size = 50
    population = [normalize(initial_h.copy()) for _ in range(population_size)]
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    
    for generation in range(1000):
        if time.time() - start_time > budget_s - 0.1:
            break
        
        # Evaluate population
        fitness = [evaluate(ind) for ind in population]
        best_index = np.argmin(fitness)
        current_best = population[best_index]
        current_c5 = fitness[best_index]
        
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Select parents
            parent1 = random.choice(population)
            parent2 = random.choice(population)
            
            # Crossover
            child = crossover(parent1, parent2)
            
            # Mutate
            mutated_child = mutate(child)
            
            new_population.append(mutated_child)
        
        population = new_population
        
    return best_h.tolist(), best_c5, n_points
