        return np.sum(h) - n_points / 2

    pop_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 2

    # Evaluate initial population
    population = [h_initial + np.random.rand(n_points) * 0.1 for _ in range(pop_size)]
    population = [h / (np.sum(h) * dx) * (n_points / 2) for h in population]

    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_c5 = fitness[best_idx]

    start_time = time.time()

    for gen in range(generations):

            break

        # Select elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]

        # Select parents
        parent_indices = np.random.choice(pop_size, size=pop_size, replace=True)
        parents = [population[i] for i in parent_indices]

        offspring = []

            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate((p1[:split], p2[split:]))
            else:
                child = p1
            offspring.append(child)

        # Mutate
        for i in range(len(offspring)):
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    offspring[i][j] = np.clip(offspring[i][j] + np.random.normal(0, 0.05), 0, 1)

        combined = list(zip(elites + offspring, fitness[:elite_size] + offspring_fitness))
