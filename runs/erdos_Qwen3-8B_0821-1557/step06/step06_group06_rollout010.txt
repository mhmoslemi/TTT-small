import time

        return np.max(corr * dx)

        return h * (n_points / 2) / np.sum(h)

        crossover_point = np.random.randint(1, n_points - 1)

        return child
    
    def mutate(h, mutation_rate=0.05):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        return h

        return [np.random.uniform(0, 1, n_points) for _ in range(size)]
    
    def select_parents(population, fitnesses):
        tournament = random.sample(list(zip(population, fitnesses)), 5)
        return min(tournament, key=lambda x: x[1])[0]
    
    population = [normalize(np.array(initial_h, dtype=np.float64))]
