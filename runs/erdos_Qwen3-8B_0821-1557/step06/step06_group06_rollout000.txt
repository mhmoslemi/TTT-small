import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if np.random.random() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        return normalize(h)
    
    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        for i in range(len(child)):
            if np.random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return normalize(child)
    
    population_size = 50
    population = [normalize(initial_h.copy()) for _ in range(population_size)]
    
    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s - 0.1:
            break
        
        # Evaluate fitness
        fitness = [evaluate(ind) for ind in population]
        
        # Select parents
        parents = np.random.choice(population, size=population_size, p=np.array(fitness)/np.sum(fitness))
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            child1 = crossover(parent1, parent2)
            child2 = mutate(child1)
            new_population.extend([child1, child2])
        
        population = new_population
    
    best_h = min(population, key=evaluate)
    best_c5 = evaluate(best_h)
