import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def mutation(h, mutation_rate=0.05):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        return normalize(h)
    
    def crossover(h1, h2):
        h = h1.copy()
        for i in range(len(h)):
            if np.random.rand() < 0.5:
                h[i] = h2[i]
        return normalize(h)
    
    def genetic_algorithm():
        population = [normalize(h) for h in [initial_h] + [np.random.uniform(0, 1, n_points) for _ in range(5)]]
        best_h = population[0]
        best_c5 = evaluate(best_h)
        
        start_time = time.time()
        
        for _ in range(100):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Evaluate population
            fitness = [evaluate(h) for h in population]
            
            # Select parents
            parents = np.random.choice(population, size=5, p=np.array(fitness)/sum(fitness))
            
            # Create new population
            new_population = []
            for _ in range(5):
                h1, h2 = random.sample(parents, 2)
                child = crossover(h1, h2)
                child = mutation(child)
                new_population.append(child)
            
            population = parents + new_population
            population = sorted(population, key=lambda x: evaluate(x))
            
            # Update best
            current_c5 = evaluate(population[0])
            if current_c5 < best_c5:
                best_h = population[0]
                best_c5 = current_c5
        
        return best_h, best_c5, n_points
    
    return genetic_algorithm()
