import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)

    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    # Initial solution
    h = initial_h.copy()
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Simulated annealing parameters
    temp = 1.0
    cooling_rate = 0.995
    max_steps = 1000
    
    for step in range(max_steps):
        if time.time() - t0 > budget_s:
            break
        
        # Generate a random perturbation
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        candidate = h + perturb
        candidate = np.clip(candidate, 0, 1)
        candidate = normalize(candidate)
        
        # Evaluate candidate
        candidate_c5 = evaluate(candidate)
        delta = candidate_c5 - best_c5
        
        # Acceptance probability
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = candidate
            best_c5 = candidate_c5
        
        # Cool down
        temp *= cooling_rate
    
    # Local refinement with SLSQP
    def local_refine(h):
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        res = optimize.minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=cons,
            method='SLSQP'
        )
        return res.x
    
    best_h = local_refine(best_h)
    best_c5 = evaluate(best_h)
    
    return best_h, best_c5, n_points
