import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def constraint(h):
        h = normalize(h)
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def local_optimize(start_h):
        result = optimize.minimize(
            objective,
            start_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        return result.x, objective(result.x)
    
    t0 = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    # Local optimization
    try:
        best_h, best_c5 = local_optimize(best_h)
    except:
        pass
    
    # Perturb and optimize
    for _ in range(5):
        if time.time() - t0 >= budget_s:
            break
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = normalize(perturbed_h)
        
        try:
            perturbed_h, perturbed_c5 = local_optimize(perturbed_h)
            if perturbed_c5 < best_c5:
                best_h, best_c5 = perturbed_h, perturbed_c5
        except:
            pass
    
    return best_h, best_c5, n_points
