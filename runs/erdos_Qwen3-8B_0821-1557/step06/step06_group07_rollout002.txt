import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def constraint(h):
        h = normalize(h)
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def simulated_annealing():
        temp = 1.0
        cooling_rate = 0.995
        max_iter = 1000
        h = initial_h.copy()
        h = normalize(h)
        best_h = h.copy()
        best_c5 = objective(h)
        
        for _ in range(max_iter):
            if temp < 0.01:
                break
            perturbation = np.random.normal(0, 0.05, n_points)
            candidate = h + perturbation
            candidate = np.clip(candidate, 0, 1)
            candidate = normalize(candidate)
            c5_candidate = objective(candidate)
            
            if c5_candidate < best_c5 or np.random.rand() < np.exp((c5_candidate - best_c5) / temp):
                h = candidate
                if c5_candidate < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_candidate
            
            temp *= cooling_rate
        
        return best_h, best_c5
    
    t0 = time.time()
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    try:
        result = optimize.minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 50}
        )
        best_h = result.x
        best_c5 = objective(best_h)
    except:
        pass
    
    if time.time() - t0 < budget_s:
        try:
            sa_h, sa_c5 = simulated_annealing()
            if sa_c5 < best_c5:
                best_h = sa_h
                best_c5 = sa_c5
        except:
            pass
    
    return best_h, best_c5, n_points
