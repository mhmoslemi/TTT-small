import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def mutate(h, mutation_rate=0.05):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        return normalize(h)
    
    def crossover(h1, h2):
        h = h1.copy()
        for i in range(len(h)):
            if np.random.rand() < 0.5:
                h[i] = h2[i]
        return normalize(h)
    
    population_size = 50
    population = [normalize(initial_h.copy()) for _ in range(population_size)]
    
    start_time = time.time()
    
    for generation in range(1000):
        if time.time() - start_time > budget_s - 0.1:
            break
        
        # Evaluate fitness
        fitness = [evaluate(ind) for ind in population]
        
        # Select parents
        parents = np.random.choice(population, size=population_size, p=np.array(fitness)/np.sum(fitness))
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            h1 = np.random.choice(parents)
            h2 = np.random.choice(parents)
            child1 = crossover(h1, h2)
            child2 = mutate(child1)
            new_population.extend([child1, child2])
        
        population = new_population
    
    best_h = min(population, key=evaluate)
    best_c5 = evaluate(best_h)
    return best_h.tolist(), best_c5, n_points
