import random
from scipy.optimize import minimize
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=10, **kwargs):

    # Starting from the initial construction
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        h_norm = h / np.sum(h) * (n_points / 2)
        overlap = np.correlate(h_norm, 1 - h_norm, mode="full") * dx
        return np.max(overlap)
    
    # Genetic algorithm setup
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    toolbox = base.Toolbox()
    toolbox.register("attr_float", lambda: np.random.uniform(0, 1))
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    
    def eval_func(individual):
        h = np.array(individual)
        # Enforce integral constraint
        h_norm = h / np.sum(h) * (n_points / 2)
        c5 = compute_c5(h_norm)
        return (c5,)
    
    toolbox.register("evaluate", eval_func)
    toolbox.register("mate", tools.cxUniform, indpb=0.1)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.2)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Initial population
    population = toolbox.population(n=50)
    
    # Run genetic algorithm
    for gen in range(100):
        offspring = algorithms.varAnd(population, toolbox, cxpb=0.5, mutpb=0.5)
        fits = toolbox.map(toolbox.evaluate, offspring)
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        population = toolbox.select(offspring, k=len(population))
        if gen % 10 == 0:
            # Track best solution
            best_ind = tools.selBest(population, 1)[0]
            best_c5 = best_ind.fitness.values[0]
            print(f"Generation {gen}: Best C5 = {best_c5}")
            if best_c5 < 0.380876:
                break
    
    # Find best solution in population
    best_ind = tools.selBest(population, 1)[0]
    best_h = np.array(best_ind)
    best_h_norm = best_h / np.sum(best_h) * (n_points / 2)
    best_c5 = compute_c5(best_h_norm)
    
    return best_h_norm, best_c5, n_points
