import scipy.signal as signal
import time

        return np.max(overlap)

    crossover_rate = 0.8
    
    # Initialize population
    population = np.random.uniform(0, 1, (pop_size, n_points))
    population = population / np.sum(population, axis=1, keepdims=True) * (n_points / 2)
    
    best_h = h0
    best_c5 = objective(h0)
    
    start_time = time.time()
    
    for _ in range(generations):
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        
        # Select parents
        parents = population[np.argsort(fitness)[:int(pop_size/2)]]
        
        # Crossover
        offspring = []
        for _ in range(pop_size - len(parents)):
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            if np.random.rand() < crossover_rate:
                idx = np.random.randint(1, n_points-1)
                child = np.concatenate([parent1[:idx], parent2[idx:]])
            else:

        population = np.vstack([parents, offspring])
