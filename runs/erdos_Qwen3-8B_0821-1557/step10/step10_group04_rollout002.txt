import numpy as np
import scipy.signal as signal
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Initialize population
    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    
    start_time = time.time()
    
    for gen in range(200):
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Create new population
        new_population = []
        
        # Elitism: keep best solution
        new_population.append(best_h)
        
        # Select parents
        for _ in range(population_size - 1):
            # Select two parents
            parents = np.random.choice(population_size, size=2, replace=False)
            parent1 = population[parents[0]]
            parent2 = population[parents[1]]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            
            # Mutation
            mutation_rate = 0.1
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0.0, 1.0)
            
            # Normalize to satisfy integral constraint
            child = child / np.sum(child) * (n_points / 2)
            
            new_population.append(child)
        
        population = new_population
    
    # Local search refinement
    best_h = population[np.argmin([objective(ind) for ind in population])]
    best_c5 = objective(best_h)
    
    # Final local search
    for _ in range(50):
        perturb = np.random.uniform(-0.02, 0.02, size=n_points)
        h_perturb = best_h + perturb
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        h_perturb = h_perturb / np.sum(h_perturb) * (n_points / 2)
        if objective(h_perturb) < best_c5:
            best_h = h_perturb
            best_c5 = objective(h_perturb)
    
    return best_h, best_c5, n_points
