import numpy as np
import scipy.signal as signal
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Initialize population
    population_size = 50
    population = np.random.uniform(0, 1, (population_size, n_points))
    population = [h / np.sum(h) * (n_points / 2) for h in population]
    
    start_time = time.time()
    
    for _ in range(200):  # Evolution steps
        # Evaluate fitness
        fitness = np.array([objective(h) for h in population])
        
        # Select parents
        parent_indices = np.argsort(fitness)[:population_size//2]
        parents = [population[i] for i in parent_indices]
        
        # Crossover
        offspring = []
        for _ in range(population_size//2):
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            mutant = offspring[i] * (1 + np.random.uniform(-0.1, 0.1, n_points))
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant / np.sum(mutant) * (n_points / 2)
            offspring[i] = mutant
        
        # Combine population
        population = parents + offspring
        population = np.array(population)
        
        # Track best solution
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
