        overlap = np.correlate(h_norm, 1 - h_norm, mode="full") * dx
        return np.max(overlap)
    
    # Genetic algorithm setup
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0, 1)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutUniform, low=0, up=1, indpb=1.0/n_points)
    toolbox.register("select", tools.selTournament, tsize=3)
    
    def evaluate(individual):
        h = np.array(individual)
        h_norm = h / np.sum(h) * (n_points / 2)
        overlap = np.correlate(h_norm, 1 - h_norm, mode="full") * dx
        return (np.max(overlap),)
    
    toolbox.register("evaluate", evaluate)
    toolbox.register("update", tools.replace, fitst=0.8, p=0.05)
    
    # Initial population
    population = toolbox.population(n=50)
    # Evaluate the initial population
    fits = list(map(toolbox.evaluate, population))
    for fit, ind in zip(fits, population):
        ind.fitness.values = fit
    
    # Run genetic algorithm
    import time
    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        offspring = algorithms.varAnd(population, toolbox, cxpb=0.5, mutpb=0.1)

            ind.fitness.values = fit
        population = toolbox.select(population + offspring, len(population))
    
    # Find the best individual
    best_individual = min(population, key=lambda x: x.fitness.values[0])
    best_h = np.array(best_individual)
