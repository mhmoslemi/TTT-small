import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    # Use a better initial construction with n=64 samples
    initial_h = np.zeros(64)
    initial_h[0:16] = 1.0
    initial_h[32:48] = 1.0
    initial_h = initial_h / np.sum(initial_h) * (64 / 2)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h0 = initial_h.copy()
    
    # Optimization with bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = opt.minimize(
        objective, 
        h0, 
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Adaptive local search with more intelligent perturbations
    for _ in range(200):
        # Create a perturbation that preserves the integral constraint
        perturb = np.random.uniform(-0.03, 0.03, size=n_points)
        h_perturb = best_h + perturb
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        h_perturb = h_perturb / np.sum(h_perturb) * (n_points / 2)
        if objective(h_perturb) < best_c5:
            best_h = h_perturb
            best_c5 = objective(best_h)
    
    # Check time budget
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Additional refinement with smaller perturbations
        for _ in range(100):
            perturb = np.random.uniform(-0.01, 0.01, size=n_points)
            h_perturb = best_h + perturb
            h_perturb = np.clip(h_perturb, 0.0, 1.0)
            h_perturb = h_perturb / np.sum(h_perturb) * (n_points / 2)
            if objective(h_perturb) < best_c5:
                best_h = h_perturb
                best_c5 = objective(h_perturb)
    
    return best_h, best_c5, n_points
