    def compute_c5(h):
        h_norm = h / np.sum(h) * (n_points / 2)
        overlap = np.correlate(h_norm, 1 - h_norm, mode="full") * dx
        return np.max(overlap)
    
    # Initialize genetic algorithm
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    def eval_func(individual):
        h = np.array(individual)
        h_norm = h / np.sum(h) * (n_points / 2)
        overlap = np.correlate(h_norm, 1 - h_norm, mode="full") * dx
        return (np.max(overlap),)
    
    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0, 1)
    toolbox.register("individual", tools.InitRepeat, creator.Individual, n_points, key=toolbox.attr_float)
    toolbox.register("population", tools.InitRepeat, creator.Individual, 50)
    toolbox.register("evaluate", eval_func)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=1/n_points)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Initial optimization with L-BFGS-B
    def objective(h):
        return compute_c5(h)
    
    result = minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-6)
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Genetic algorithm optimization
    pop = toolbox.population()
    stats = tools.Statistics(lambda ind: ind.fitness.values)
    stats.register("avg", np.mean)
    stats.register("std", np.std)
    stats.register("min", np.min)
    stats.register("max", np.max)
    
    # Run genetic algorithm
    for gen in range(100):
        offspring = algorithms.varAnd(pop, toolbox, cxpb=0.5, mutpb=0.1)
        fits = toolbox.map(toolbox.evaluate, offspring)
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        pop = toolbox.select(offspring, k=len(pop))
        record = stats.compile(pop)
        if gen % 10 == 0:
            print(f"Gen {gen}: Best C5 = {np.min([ind.fitness.values[0] for ind in pop])}")
    
    # Find best solution from genetic algorithm
    best_genetic = min(pop, key=lambda ind: ind.fitness.values[0])
    best_c5 = best_genetic.fitness.values[0]

        perturb = np.random.uniform(-0.05 * (0.95 ** step), 0.05 * (0.95 ** step), size=n_points)
        candidate = best_h + perturb
        candidate = np.clip(candidate, 0, 1)
        candidate = candidate / np.sum(candidate) * (n_points / 2)
        c5 = compute_c5(candidate)
        if c5 < best_c5:
            best_h = candidate
            best_c5 = c5
    
    return best_h, best_c5, n_points
