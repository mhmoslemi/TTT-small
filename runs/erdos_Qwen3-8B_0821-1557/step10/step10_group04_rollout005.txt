import numpy as np
import scipy.signal as signal
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Initialize population
    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    for h in population:
        h = h / np.sum(h) * (n_points / 2)
    
    def fitness(h):
        return objective(h)
    
    # Genetic algorithm parameters
    tournament_size = 5
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 100
    
    start_time = time.time()
    for gen in range(generations):
        # Evaluate fitness
        fitness_scores = [fitness(h) for h in population]
        best_idx = np.argmin(fitness_scores)
        best_fitness = fitness_scores[best_idx]
        
        # Tournament selection
        selected = []
        for _ in range(population_size):
            contestants = np.random.choice(range(population_size), tournament_size, replace=False)
            winner = contestants[np.argmin([fitness_scores[i] for i in contestants])]
            selected.append(population[winner])
        
        # Crossover
        new_population = []
        for i in range(0, population_size, 2):
            if np.random.rand() < crossover_rate:
                parent1 = selected[i]
                parent2 = selected[i+1]
                split = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:split], parent2[split:]])
                child2 = np.concatenate([parent2[:split], parent1[split:]])
                new_population.extend([child1, child2])
            else:
                new_population.append(selected[i])
                new_population.append(selected[i+1])
        
        # Mutation
        for i in range(len(new_population)):
            h = new_population[i]
            h = h / np.sum(h) * (n_points / 2)
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    h[j] += np.random.uniform(-0.05, 0.05)
                    h[j] = np.clip(h[j], 0.0, 1.0)
            h = h / np.sum(h) * (n_points / 2)
            new_population[i] = h
        
        # Replace population
        population = new_population
        
        # Elitism
        if best_fitness < np.min([fitness(h) for h in population]):
            population[best_idx] = initial_h.copy()
            population[best_idx] = population[best_idx] / np.sum(population[best_idx]) * (n_points / 2)
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break
    
    # Final refinement
    best_idx = np.argmin([fitness(h) for h in population])
    best_h = population[best_idx]
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    best_c5 = fitness(best_h)
    
    # Additional local search
    for _ in range(50):
        perturb = np.random.uniform(-0.02, 0.02, size=n_points)
        h_perturb = best_h + perturb
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        h_perturb = h_perturb / np.sum(h_perturb) * (n_points / 2)
        if objective(h_perturb) < best_c5:
            best_h = h_perturb
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
