import numpy as np
import scipy.signal as signal
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Initialize population
    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    
    start_time = time.time()
    
    for gen in range(100):
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
        
        # Apply local search to best solution
        for _ in range(50):
            perturb = np.random.uniform(-0.02, 0.02, size=n_points)
            h_perturb = best_h + perturb
            h_perturb = np.clip(h_perturb, 0.0, 1.0)
            h_perturb = h_perturb / np.sum(h_perturb) * (n_points / 2)
            if objective(h_perturb) < best_c5:
                best_h = h_perturb
                best_c5 = objective(h_perturb)
        
        # Generate new population through mutation and crossover
        new_population = []
        for _ in range(population_size):
            # Mutation
            mutant = best_h.copy()
            for i in range(n_points):
                if random.random() < 0.1:
                    mutant[i] += np.random.uniform(-0.1, 0.1)
                    mutant[i] = np.clip(mutant[i], 0.0, 1.0)
            mutant = mutant / np.sum(mutant) * (n_points / 2)
            
            # Crossover with random parent
            parent = random.choice(population)
            child = np.zeros(n_points)
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = mutant[i]
                else:
                    child[i] = parent[i]
            child = child / np.sum(child) * (n_points / 2)
            
            new_population.append(child)
        
        population = new_population
    
    # Final evaluation
    final_fitness = [objective(ind) for ind in population]
    best_idx = np.argmin(final_fitness)
    best_h = population[best_idx]
    best_c5 = final_fitness[best_idx]
    
    return best_h, best_c5, n_points
