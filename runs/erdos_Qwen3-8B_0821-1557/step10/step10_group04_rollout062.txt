import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reduced from 47 to 100 for better optimization
    dx = 2.0 / n_points
    
    # Random initial guess
    h_initial = np.random.rand(n_points)
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    
    def objective(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization with bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = opt.minimize(
        objective, 
        h_initial, 
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Multi-start local search
    for _ in range(10):
        perturb = np.random.normal(0, 0.05, size=n_points)
        h_perturb = best_h + perturb
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        h_perturb = h_perturb / np.sum(h_perturb) * (n_points / 2)
        if objective(h_perturb) < best_c5:
            best_h = h_perturb
            best_c5 = objective(best_h)
    
    # Check time budget
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Additional refinement if time allows
        for _ in range(50):
            perturb = np.random.normal(0, 0.02, size=n_points)
            h_perturb = best_h + perturb
            h_perturb = np.clip(h_perturb, 0.0, 1.0)
            h_perturb = h_perturb / np.sum(h_perturb) * (n_points / 2)
            if objective(h_perturb) < best_c5:
                best_h = h_perturb
                best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
