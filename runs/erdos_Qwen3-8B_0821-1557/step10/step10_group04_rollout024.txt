import numpy as np
import scipy.signal as signal
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    population_size = 50
    mutation_rate = 0.1
    generations = 100
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h / np.sum(h) * (n_points / 2) for h in population]
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Select parents
        parents = np.random.choice(range(population_size), size=population_size, p=np.array(fitness)/np.sum(fitness))
        parents = [population[i] for i in parents]
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if i+1 < population_size:
                p1 = parents[i]
                p2 = parents[i+1]
                child1 = np.concatenate((p1[:n_points//2], p2[n_points//2:]))
                child2 = np.concatenate((p2[:n_points//2], p1[n_points//2:]))
                offspring.append(child1)
                offspring.append(child2)
            else:
                offspring.append(parents[i])
        
        # Mutate
        for i in range(len(offspring)):
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    offspring[i][j] = np.random.uniform(0, 1)
            offspring[i] = offspring[i] / np.sum(offspring[i]) * (n_points / 2)
        
        # Replace population
        population = offspring
    
    return best_h, best_c5, n_points
