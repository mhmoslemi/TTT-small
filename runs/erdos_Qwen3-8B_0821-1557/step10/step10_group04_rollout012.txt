import scipy.signal as signal
import time

def run(seed=42, budget_s=10, **kwargs):

    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    def evaluate_individual(individual):
        return objective(individual)
    
    # Initial population
    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    population = [h / np.sum(h) * (n_points / 2) for h in population]
    
    start_time = time.time()
    
    for generation in range(100):
        # Evaluate population
        fitness = [evaluate_individual(ind) for ind in population]
        
        # Find the best individual
        best_index = np.argmin(fitness)
        best_h = population[best_index]
        best_c5 = fitness[best_index]
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Create new population
        new_population = []
        
        # Elitism: keep the best individual
        new_population.append(best_h)
        
        # Tournament selection for the rest
        for _ in range(population_size - 1):
            # Select two random individuals
            a = np.random.randint(0, population_size)
            b = np.random.randint(0, population_size)
            if fitness[a] < fitness[b]:
                parent = population[a]
            else:
                parent = population[b]
            
            # Apply mutation
            mutated = parent.copy()
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    mutated[i] += np.random.uniform(-0.1, 0.1)
                    mutated[i] = np.clip(mutated[i], 0.0, 1.0)
            
            # Normalize to maintain integral constraint
            mutated = mutated / np.sum(mutated) * (n_points / 2)
            
            new_population.append(mutated)
        
        population = new_population
    
    # Final evaluation
    fitness = [evaluate_individual(ind) for ind in population]
    best_index = np.argmin(fitness)
    best_h = population[best_index]
    best_c5 = fitness[best_index]
