import numpy as np
import scipy.signal as signal
import random
from scipy.optimize import minimize

def run(seed=42, budget_s=10, **kwargs):

    random.seed(seed)
    
    # Initial greedy construction
    n_points = 47
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    h[n_points//2:] = 0.0
    
    # Normalize to meet integral constraint
    h = h / np.sum(h) * (n_points / 2)
    
    def compute_c5(h):
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Simulated annealing
    T = 1.0
    T_min = 1e-3
    n_iter = 1000
    n_decay = 100
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for i in range(n_iter):
        T = max(T_min, T * np.exp(-i / n_decay))
        for _ in range(10):
            # Perturb solution
            perturbed = best_h.copy()
            idx = np.random.choice(n_points)
            perturbed[idx] += np.random.uniform(-0.1, 0.1)
            perturbed = np.clip(perturbed, 0.0, 1.0)
            perturbed = perturbed / np.sum(perturbed) * (n_points / 2)
            
            # Compute cost
            current_c5 = compute_c5(best_h)
            new_c5 = compute_c5(perturbed)
            
            # Acceptance probability
            if new_c5 < current_c5 or np.random.random() < np.exp((current_c5 - new_c5)/T):
                best_h = perturbed.copy()
                best_c5 = new_c5
        
        if best_c5 < 0.380876:
            break
    
    return best_h, best_c5, n_points
