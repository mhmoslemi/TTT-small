import scipy.signal as signal
import time

def run(seed=42, budget_s=10, **kwargs):

    def objective(h):

        return np.max(overlap)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    best_h = initial_h.copy()

    start_time = time.time()

    temp = 1.0
