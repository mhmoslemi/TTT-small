import numpy as np
import scipy.signal as signal
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Initial guess
    h0 = initial_h.copy()
    h0 = h0 / np.sum(h0) * (n_points / 2)
    
    start_time = time.time()
    best_h = h0
    best_c5 = objective(best_h)
    
    # Genetic algorithm parameters
    pop_size = 20
    generations = 50
    mutation_rate = 0.15
    crossover_rate = 0.8
    
    for _ in range(generations):
        # Generate population
        population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
        population = [h / np.sum(h) * (n_points / 2) for h in population]
        population.append(best_h)  # Add current best
        
        # Evaluate population
        fitness = [objective(h) for h in population]
        
        # Select top performers
        sorted_idx = np.argsort(fitness)
        top_idx = sorted_idx[:int(pop_size/2)]
        top_h = [population[i] for i in top_idx]
        
        # Create new population through crossover and mutation
        new_population = []
        for _ in range(int(pop_size/2)):
            parent1 = np.random.choice(top_h)
            parent2 = np.random.choice(top_h)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points-1)
                child = np.concatenate([parent1[:split], parent2[split:]])
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.uniform(-0.1, 0.1, n_points)
                child += mutation
            
            # Normalize
            child = child / np.sum(child) * (n_points / 2)
            new_population.append(child)
        
        # Update population
        population = top_h + new_population
        np.random.shuffle(population)
        
        # Update best solution
        for h in population:
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Early stopping if time is up
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final local search refinement
    for _ in range(50):
        perturb = np.random.uniform(-0.02, 0.02, size=n_points)
        h_perturb = best_h + perturb
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        h_perturb = h_perturb / np.sum(h_perturb) * (n_points / 2)
        if objective(h_perturb) < best_c5:
            best_h = h_perturb
            best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
