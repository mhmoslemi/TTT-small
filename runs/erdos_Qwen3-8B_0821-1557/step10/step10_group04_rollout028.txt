import numpy as np
import scipy.signal as signal
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h_norm = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h_norm, 1 - h_norm, mode="full") * dx
        return np.max(overlap)
    
    # Initial guess
    h0 = initial_h.copy()
    h0 = h0 / np.sum(h0) * (n_points / 2)
    
    # Genetic algorithm parameters
    population_size = 20
    mutation_rate = 0.15
    generations = 50
    
    def create_individual():
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        indices = np.random.choice(n_points, size=n_points//2, replace=False)
        child[indices] = parent2[indices]
        child = child / np.sum(child) * (n_points / 2)
        return child
    
    def mutate(individual):
        for i in range(n_points):
            if random.random() < mutation_rate:
                individual[i] += np.random.uniform(-0.1, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        individual = individual / np.sum(individual) * (n_points / 2)
        return individual
    
    # Initial population
    population = [create_individual() for _ in range(population_size)]
    
    start_time = time.time()
    best_h = h0
    best_c5 = objective(h0)
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_best_c5 = fitness[best_idx]
        
        if current_best_c5 < best_c5:
            best_h = current_best
            best_c5 = current_best_c5
            
        # Select parents
        parents = np.random.choice(population_size, size=population_size, replace=False)
        
        # Create new population
        new_population = []
        for i in range(population_size//2):
            parent1 = population[parents[i]]
            parent2 = population[parents[population_size//2 + i]]
            child1 = crossover(parent1, parent2)
            child2 = crossover(parent2, parent1)
            new_population.append(mutate(child1))
            new_population.append(mutate(child2))
        
        population = new_population
    
    return best_h, best_c5, n_points
