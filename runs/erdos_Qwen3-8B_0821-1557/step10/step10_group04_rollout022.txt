import numpy as np
import scipy.signal as signal
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Initial population
    population_size = 20
    population = [initial_h.copy() for _ in range(population_size)]
    
    start_time = time.time()
    
    for gen in range(100):
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        
        # Select parents
        parents = np.argsort(fitness)[:population_size//2]
        parents = [population[i] for i in parents]
        
        # Generate offspring
        offspring = []
        for _ in range(population_size//2):
            # Crossover
            parent1 = np.random.choice(parents)
            parent2 = np.random.choice(parents)
            child = np.zeros_like(parent1)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0.0, 1.0)
            
            # Normalize
            child = child / np.sum(child) * (n_points / 2)
            offspring.append(child)
        
        # Replace population
        population = parents + offspring
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Find best solution
    best_h = min(population, key=objective)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
