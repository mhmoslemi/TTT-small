import scipy.signal as signal
import time
import random

def run(seed=42, budget_s=10, **kwargs):

        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.15
    elite_size = 2
    
    # Initial population
    def create_individual():
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    population = [create_individual() for _ in range(population_size)]
    
    start_time = time.time()
    
    for gen in range(generations):
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        
        # Select elite
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = [population[i] for i in elite_indices]
        
        # Select parents for crossover
        parent_indices = np.random.choice(len(population), size=population_size - elite_size, replace=False)
        parents = [population[i] for i in parent_indices]
        
        # Crossover
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                parent1 = parents[i]
                parent2 = parents[i + 1]
                child1 = np.zeros(n_points)
                child2 = np.zeros(n_points)
                for j in range(n_points):
                    if random.random() < 0.5:
                        child1[j] = parent1[j]
                        child2[j] = parent2[j]
                    else:
                        child1[j] = parent2[j]
                        child2[j] = parent1[j]
                offspring.extend([child1, child2])
        
        # Mutate
        for i in range(len(offspring)):
            if random.random() < mutation_rate:
                for j in range(n_points):
                    if random.random() < 0.1:
                        offspring[i][j] = np.clip(np.random.uniform(0, 1), 0, 1)
        
        # Combine population
        population = elite + offspring
        
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Local search refinement
    best_h = min(population, key=objective)
    best_c5 = objective(best_h)
