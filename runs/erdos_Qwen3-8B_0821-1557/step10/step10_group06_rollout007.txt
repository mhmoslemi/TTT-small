        return np.max(overlap)

    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutUniform, low=0, up=1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    def eval_func(individual):
        h = np.array(individual)
        h_norm = h / np.sum(h) * (n_points / 2)
        overlap = np.correlate(h_norm, 1 - h_norm, mode="full") * dx
        return (np.max(overlap),)
    
    toolbox.register("population", tools.initRepeat, creator.Individual, np.random.uniform, low=0, high=1, size=n_points)
    toolbox.register("evaluate", eval_func)
    toolbox.register("stddev", np.std)
    
    # Initial population
    pop = toolbox.population(n=50)
    for ind in pop:
        ind.fitness.values = toolbox.evaluate(ind)
    
    # Local optimization with L-BFGS-B

            return compute_c5(x)
