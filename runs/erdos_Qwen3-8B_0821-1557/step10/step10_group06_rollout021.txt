import numpy as np
import cvxopt
from scipy.optimize import minimize
import random
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start from initial construction
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        h_norm = h / np.sum(h) * (n_points / 2)
        overlap = np.correlate(h_norm, 1 - h_norm, mode="full") * dx
        return np.max(overlap)
    
    # Create a genetic algorithm
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    def eval_func(individual):
        individual = np.clip(individual, 0, 1)
        individual = individual / np.sum(individual) * (n_points / 2)
        return compute_c5(individual),
    
    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0, 1)
    toolbox.register("individual", tools.initRepeat, creator.Individual, 
                    toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("evaluate", eval_func)
    toolbox.register("mate", tools.cxTwoPoint)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Genetic algorithm parameters
    population = toolbox.population(n=20)
    for gen in range(50):
        offspring = algorithms.varAnd(population, toolbox, cxpb=0.5, mutpb=0.1)
        fits = toolbox.map(toolbox.evaluate, offspring)
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        population = toolbox.select(offspring, k=len(population))
        best_ind = tools.selBest(population, 1)[0]
        if gen % 5 == 0:
            print(f"Generation {gen}, Best C5: {best_ind.fitness.values[0]}")
    
    # Local refinement with L-BFGS-B
    best_h = np.clip(best_ind, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    result = minimize(compute_c5, best_h, method='L-BFGS-B', 
                     bounds=[(0, 1)]*n_points, tol=1e-6)
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
