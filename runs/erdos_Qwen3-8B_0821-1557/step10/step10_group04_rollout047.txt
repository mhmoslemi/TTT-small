import numpy as np
import scipy.signal as signal
import time
import random

def run(seed=42, budget_s=10, **kwargs):

    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Initialize population
    population_size = 100
    population = [initial_h.copy() for _ in range(population_size)]
    
    def evaluate_individual(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    def breed(parent1, parent2):
        # Single-point crossover
        point = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:point], parent2[point:]])
        return child
    
    def mutate(h):
        # Random mutation
        for i in range(n_points):
            if random.random() < 0.1:
                h[i] += np.random.uniform(-0.1, 0.1)
                h[i] = np.clip(h[i], 0.0, 1.0)
        return h
    
    start_time = time.time()
    
    for generation in range(100):
        # Evaluate population
        fitness = [evaluate_individual(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Local search on best solution
        for _ in range(50):
            perturb = np.random.uniform(-0.02, 0.02, size=n_points)
            h_perturb = best_h + perturb
            h_perturb = np.clip(h_perturb, 0.0, 1.0)
            h_perturb = h_perturb / np.sum(h_perturb) * (n_points / 2)
            if evaluate_individual(h_perturb) < best_c5:
                best_h = h_perturb
                best_c5 = evaluate_individual(h_perturb)
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Selection and breeding
        # Tournament selection
        tournament = np.random.choice(range(population_size), size=5, replace=False)
        tournament_fitness = [fitness[i] for i in tournament]
        winner = tournament[np.argmin(tournament_fitness)]
        parent1 = population[winner].copy()
        
        # Select second parent
        tournament = np.random.choice(range(population_size), size=5, replace=False)
        tournament_fitness = [fitness[i] for i in tournament]
        winner = tournament[np.argmin(tournament_fitness)]
        parent2 = population[winner].copy()
        
        # Crossover and mutation
        child = breed(parent1, parent2)
        child = mutate(child)
        child = child / np.sum(child) * (n_points / 2)
        
        # Replace worst individual
        worst_idx = np.argmax(fitness)
        population[worst_idx] = child
    
    return best_h, best_c5, n_points
