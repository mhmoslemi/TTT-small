            selected_idx = np.argmin([fitness[i] for i in contenders])
            selected.append(population[contenders[selected_idx]])
        
        # Crossover
        new_population = []
        for i in range(population_size):

                # Single-point crossover
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([selected[i][:split], selected[j][split:]])
