import scipy.signal as signal

import random

    def objective(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    def normalize(h):
        return h / np.sum(h) * (n_points / 2)
    
    def crossover(parent1, parent2):
        split = np.random.randint(1, n_points - 1)
        child = np.concatenate((parent1[:split], parent2[split:]))
        return child
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(n_points):
            if np.random.random() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        return normalize(h)
    
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(20)]
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Create new population
        new_population = []
        for _ in range(10):
            # Select parents
            parent1, parent2 = random.sample(population, 2)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        # Replace old population
        population = new_population
