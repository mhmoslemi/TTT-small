import numpy as np
import scipy.signal as signal
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Initial population
    population_size = 20
    population = [initial_h.copy() for _ in range(population_size)]
    
    start_time = time.time()
    
    for generation in range(500):
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
        
        # Check time budget
        if time.time() - start_time > budget_s:
            break
        
        # Selection and crossover
        parents = np.random.choice(population_size, size=population_size, p=1./population_size)
        offspring = []
        for _ in range(population_size // 2):
            parent1 = population[parents[0]]
            parent2 = population[parents[1]]
            child = (parent1 + parent2) / 2
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):
            mutation = np.random.normal(0, 0.05, size=n_points)
            offspring[i] += mutation
            offspring[i] = np.clip(offspring[i], 0.0, 1.0)
            offspring[i] = offspring[i] / np.sum(offspring[i]) * (n_points / 2)
        
        # Replace population
        population = offspring
        
        # Early stopping
        if np.abs(best_c5 - 0.380876) < 1e-5:
            break
    
    return best_h, best_c5, n_points
