import numpy as np
import scipy.signal as signal
import random
from scipy.optimize import Bounds

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial greedy construction: divide [0, 2] into segments
    n_points = 100  # Start with 100 points for good resolution
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Greedily alternate between high and low values
    for i in range(n_points):
        if i % 3 == 0:
            h[i] = 1.0
        elif i % 3 == 1:
            h[i] = 0.5
        else:
            h[i] = 0.0
    
    # Normalize to satisfy the integral constraint
    h = h / np.sum(h) * (n_points / 2)
    
    def compute_c5(h):
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    # Simulated annealing
    temp = 1.0
    cooling_rate = 0.995
    max_iter = 1000
    iter_count = 0
    
    while temp > 1e-5 and iter_count < max_iter and budget_s > 0:
        # Generate a random perturbation
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        perturbed = h + perturbation
        perturbed = np.clip(perturbed, 0.0, 1.0)

        # Compute new C5 value
        new_c5 = compute_c5(perturbed)
        
        # Acceptance probability
        if new_c5 < best_c5 or np.random.random() < np.exp((best_c5 - new_c5) / temp):

            best_c5 = new_c5
        
        # Cool down
        temp *= cooling_rate
        iter_count += 1
        budget_s -= 0.001  # Simulate time cost
    
    return best_h, best_c5, n_points
