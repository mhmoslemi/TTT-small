import scipy.signal as signal
import time
import random

def run(seed=42, budget_s=10, **kwargs):

    def objective(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Initialize population
    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]

    def fitness(h):
        return objective(h)

    def crossover(parent1, parent2):
        child = parent1.copy()

            if random.random() < 0.5:
                child[i] = parent2[i]
        return child
    
    # Mutation operator
    def mutate(h):
        h = h.copy()
        for i in range(len(h)):

                h[i] += np.random.uniform(-0.1, 0.1)
                h[i] = np.clip(h[i], 0.0, 1.0)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    start_time = time.time()

        # Create new population
