import numpy as np
import scipy.signal as signal
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h_norm = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h_norm, 1 - h_norm, mode="full") * dx
        return np.max(overlap)
    
    # Initial population
    population_size = 20
    population = [initial_h.copy() for _ in range(population_size)]
    
    start_time = time.time()
    
    for generation in range(100):
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
        
        # Create new population
        new_population = []
        
        # Elitism: keep the best solution
        new_population.append(best_h)
        
        # Tournament selection and crossover
        for _ in range(population_size - 1):
            # Tournament selection
            candidates = random.sample(range(population_size), 3)
            winner = min(candidates, key=lambda x: fitness[x])
            parent1 = population[winner].copy()
            
            # Second parent with some mutation
            parent2 = population[random.choice(range(population_size))].copy()
            mutation = np.random.uniform(-0.05, 0.05, size=n_points)
            parent2 += mutation
            parent2 = np.clip(parent2, 0.0, 1.0)
            parent2 = parent2 / np.sum(parent2) * (n_points / 2)
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            mutation = np.random.uniform(-0.02, 0.02, size=n_points)
            child += mutation
            child = np.clip(child, 0.0, 1.0)
            child = child / np.sum(child) * (n_points / 2)
            
            new_population.append(child)
        
        population = new_population
    
    # Final evaluation
    fitness = [objective(ind) for ind in population]
    best_idx = np.argmin(fitness)
    best_c5 = fitness[best_idx]
    best_h = population[best_idx].copy()
    
    return best_h, best_c5, n_points
