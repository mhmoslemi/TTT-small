import time

        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    
    start_time = time.time()
    
    for _ in range(500):  # 500 generations
        # Evaluate fitness
        fitness = []
        for h in population:
            h_normalized = h / np.sum(h) * (n_points / 2)
            overlap = signal.correlate(h_normalized, 1 - h_normalized, mode="full") * dx
            fitness.append(np.max(overlap))
        
        # Select parents using tournament selection
        tournament_size = 5
        parents = []
        for _ in range(population_size):
            contestants = np.random.choice(range(population_size), tournament_size, replace=False)
            best_idx = contestants[np.argmin(fitness[contestants])]
            parents.append(population[best_idx])
        
        # Crossover
        new_population = []
        for i in range(population_size // 2):
            parent1 = parents[2*i]
            parent2 = parents[2*i + 1]
            crossover_point = np.random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            new_population.extend([child1, child2])
        
        # Mutation
        mutation_rate = 0.1
        for i in range(len(new_population)):
            h = new_population[i]
            if np.random.rand() < mutation_rate:
                mutation = np.random.uniform(-0.1, 0.1, size=n_points)
                h += mutation
                h = np.clip(h, 0.0, 1.0)
                h = h / np.sum(h) * (n_points / 2)
            new_population[i] = h
        
        # Replace population
        population = new_population
        
        # Early stopping
        if time.time() - start_time > budget_s - 1:
            break
    
    # Find best solution
    best_h = population[0]
    best_c5 = objective(best_h)
    for h in population:
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
