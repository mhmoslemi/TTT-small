import scipy.signal as signal
import time

def run(seed=42, budget_s=10, **kwargs):

    def objective(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Initialize population
    pop_size = 50
    population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
    population = [h / np.sum(h) * (n_points / 2) for h in population]
    
    start_time = time.time()
    
    for generation in range(200):
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Local search on best solution
        for _ in range(50):
            perturb = np.random.uniform(-0.02, 0.02, size=n_points)
            h_perturb = best_h + perturb
            h_perturb = np.clip(h_perturb, 0.0, 1.0)
            h_perturb = h_perturb / np.sum(h_perturb) * (n_points / 2)
            if objective(h_perturb) < best_c5:
                best_h = h_perturb
                best_c5 = objective(best_h)
        
        # Genetic operators
        # Select parents
        parent_indices = np.argsort(fitness)[:int(pop_size/2)]
        parents = [population[i] for i in parent_indices]
        
        # Crossover
        offspring = []
        for _ in range(int(pop_size/2)):
            p1, p2 = np.random.choice(parents, 2, replace=False)
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate((p1[:crossover_point], p2[crossover_point:]))
            offspring.append(child)
        
        # Mutation
        for i in range(len(offspring)):

                if np.random.rand() < 0.1:

            offspring[i] = offspring[i] / np.sum(offspring[i]) * (n_points / 2)
        
        # Replace population
        population = offspring + [best_h] * (pop_size - len(offspring))
    
    return best_h, best_c5, n_points
