import numpy as np
import scipy.signal as signal
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initial population
    population = [initial_h.copy() for _ in range(population_size)]
    population = [h / np.sum(h) * (n_points / 2) for h in population]
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    start_time = time.time()
    
    for generation in range(generations):
        # Evaluate fitness
        fitness = [objective(h) for h in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Select parents
            parent1 = population[np.random.choice(population_size)]
            parent2 = population[np.random.choice(population_size)]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:split], parent2[split:]])
                child2 = np.concatenate([parent2[:split], parent1[split:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] = np.clip(child1[i] + np.random.uniform(-0.1, 0.1), 0.0, 1.0)
                    child2[i] = np.clip(child2[i] + np.random.uniform(-0.1, 0.1), 0.0, 1.0)
            
            # Normalize
            child1 = child1 / np.sum(child1) * (n_points / 2)
            child2 = child2 / np.sum(child2) * (n_points / 2)
            
            new_population.append(child1)
            new_population.append(child2)
        
        population = new_population
    
    return best_h, best_c5, n_points
