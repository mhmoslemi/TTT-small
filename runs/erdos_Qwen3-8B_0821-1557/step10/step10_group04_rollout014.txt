import numpy as np
import scipy.signal as signal
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Initial population
    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    
    def evaluate(individual):
        h = individual.copy()
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Initial evaluation
    fitness = [evaluate(indiv) for indiv in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    start_time = time.time()
    
    # Genetic algorithm parameters
    mutation_rate = 0.15
    crossover_rate = 0.8
    generations = 200
    
    for gen in range(generations):
        # Select parents
        parents = random.choices(population, weights=1/np.array(fitness), k=population_size)
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if random.random() < crossover_rate:
                parent1 = parents[i]
                parent2 = parents[i+1]
                crossover_point = random.randint(1, n_points-1)
                child1 = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
                child2 = np.concatenate((parent2[:crossover_point], parent1[crossover_point:]))
                offspring.extend([child1, child2])
            else:
                offspring.append(parents[i])
                offspring.append(parents[i+1])
        
        # Mutation
        for i in range(len(offspring)):
            if random.random() < mutation_rate:
                idx = random.randint(0, n_points-1)
                offspring[i][idx] = np.clip(offspring[i][idx] + np.random.uniform(-0.1, 0.1), 0.0, 1.0)
        
        # Replace population
        population = offspring
        
        # Evaluate new population
        new_fitness = [evaluate(indiv) for indiv in population]
        fitness = new_fitness
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Additional local search
    for _ in range(100):
        perturb = np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturb = best_h + perturb
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        h_perturb = h_perturb / np.sum(h_perturb) * (n_points / 2)
        if objective(h_perturb) < best_c5:
            best_h = h_perturb
            best_c5 = objective(h_perturb)
    
    return best_h, best_c5, n_points
