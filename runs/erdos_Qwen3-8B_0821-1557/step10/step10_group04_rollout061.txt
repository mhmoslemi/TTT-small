import numpy as np
import scipy.signal as signal
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to evaluate overlap
    def evaluate(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    def create_individual():
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    population = [create_individual() for _ in range(population_size)]
    population[0] = initial_h.copy() / np.sum(initial_h) * (n_points / 2)
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [evaluate(ind) for ind in population]
        
        # Select parents
        parents = np.array(population)[np.argsort(fitness)]
        
        # Create next generation
        next_generation = []
        for _ in range(population_size):
            # Select two parents
            parent1 = parents[np.random.choice(len(parents))]
            parent2 = parents[np.random.choice(len(parents))]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child[i] = np.random.uniform(0, 1)
            
            # Normalize to maintain integral constraint
            child = child / np.sum(child) * (n_points / 2)
            next_generation.append(child)
        
        population = next_generation
    
    # Evaluate final population
    fitness = [evaluate(ind) for ind in population]
    best_index = np.argmin(fitness)
    best_h = population[best_index]
    best_c5 = fitness[best_index]
    
    # Final refinement with local search
    for _ in range(50):
        perturb = np.random.uniform(-0.02, 0.02, size=n_points)
        h_perturb = best_h + perturb
        h_perturb = np.clip(h_perturb, 0.0, 1.0)
        h_perturb = h_perturb / np.sum(h_perturb) * (n_points / 2)
        if evaluate(h_perturb) < best_c5:
            best_h = h_perturb
            best_c5 = evaluate(h_perturb)
    
    elapsed = time.time() - start_time

        for _ in range(50):
            perturb = np.random.uniform(-0.01, 0.01, size=n_points)

            h_perturb = np.clip(h_perturb, 0.0, 1.0)
            h_perturb = h_perturb / np.sum(h_perturb) * (n_points / 2)
            if evaluate(h_perturb) < best_c5:
                best_h = h_perturb
                best_c5 = evaluate(h_perturb)
    
    return best_h, best_c5, n_points
