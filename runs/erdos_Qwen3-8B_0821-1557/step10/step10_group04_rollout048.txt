import numpy as np
import scipy.signal as signal
import time

def run(seed=42, budget_s=10, **kwargs):

    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Initial population
    population_size = 20
    population = [initial_h.copy() for _ in range(population_size)]
    
    start_time = time.time()
    
    for generation in range(100):
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        
        # Select parents
        parents = np.random.choice(population_size, size=population_size, p=np.array(fitness)/np.sum(fitness))
        parents = [population[i] for i in parents]
        
        # Crossover
        offspring = []
        for i in range(population_size//2):
            p1 = parents[2*i]
            p2 = parents[2*i + 1]
            child = np.zeros(n_points)
            for j in range(n_points):
                if np.random.rand() < 0.5:
                    child[j] = p1[j]
                else:
                    child[j] = p2[j]
            offspring.append(child)
        
        # Mutation
        for i in range(population_size):
            ind = offspring[i]
            for j in range(n_points):
                if np.random.rand() < 0.1:
                    ind[j] += np.random.uniform(-0.1, 0.1)
                    ind[j] = np.clip(ind[j], 0.0, 1.0)
            offspring[i] = ind
        
        # Replace population
        population = offspring
        
        # Local search on best individual
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_h = best_h / np.sum(best_h) * (n_points / 2)
        for _ in range(10):
            perturb = np.random.uniform(-0.02, 0.02, size=n_points)
            h_perturb = best_h + perturb
            h_perturb = np.clip(h_perturb, 0.0, 1.0)

            break
    
    # Find best solution
    fitness = [objective(ind) for ind in population]
