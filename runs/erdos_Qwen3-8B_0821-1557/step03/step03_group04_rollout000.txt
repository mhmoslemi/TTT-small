        elite_indices = np.argsort(fitness)[:elite_size]

        new_population = [np.copy(el) for el in elites]
        for i in range(pop_size - elite_size):

            parent1 = np.random.choice(elites)

            # Crossover
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:split], parent2[split:]])
