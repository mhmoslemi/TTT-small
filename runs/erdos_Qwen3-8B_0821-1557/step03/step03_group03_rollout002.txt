import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Start with the initial solution
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    t0 = time.time()
    
    # Genetic algorithm parameters
    population_size = 30
    generations = 100
    mutation_rate = 0.15
    elite_ratio = 0.2  # Keep top 20% as elites
    
    for gen in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        # Generate new population
        population = []
        # Add elite solutions
        elite_count = int(population_size * elite_ratio)
        elites = np.array([best_h] * elite_count)
        population.extend(elites)
        
        # Generate random solutions
        for _ in range(population_size - elite_count):
            h = np.random.uniform(0, 1, n_points)
            h = normalize(h)
            population.append(h)
        
        # Evaluate fitness
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        current_best_h = population[best_idx]
        current_best_c5 = fitness[best_idx]
        
        # Update best solution
        if current_best_c5 < best_c5:
            best_h = current_best_h.copy()
            best_c5 = current_best_c5
        
        # Apply mutation to non-elites
        for i in range(population_size - elite_count):
            h = population[i].copy()
            # Apply mutation
            if np.random.rand() < mutation_rate:
                mutation = np.random.normal(0, 0.1, n_points)
                h += mutation
                h = np.clip(h, 0, 1)
                h = normalize(h)
            population[i] = h
    
    return best_h, best_c5, n_points
