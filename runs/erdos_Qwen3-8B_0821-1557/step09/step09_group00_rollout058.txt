        alpha = np.random.rand(n_points)
        child = alpha * parent1 + (1 - alpha) * parent2

        return child
    
    def mutate(individual):
        mutation = np.random.normal(0, 0.05, n_points)
        mutated = individual + mutation

        mutated = mutated / np.sum(mutated) * (n_points / 2)
        return mutated

    def fitness(individual):
        c5 = compute_c5(individual)
        # Add small penalty for constraint violation
        penalty = 1e-3 * abs(np.sum(individual) - n_points / 2)
        return c5 + penalty
    
    # Initialize population
    population = [create_individual() for _ in range(population_size)]
