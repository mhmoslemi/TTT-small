        h_norm = h / np.sum(h) * (n_points / 2)
        # Compute the overlap integral
        overlap = signal.correlate(h_norm, 1 - h_norm, mode="full") * dx
        return np.max(overlap)

    population_size = 50
    mutation_rate = 0.1
    tournament_size = 5
    generations = 100
    elite_size = 2
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h / np.sum(h) * (n_points / 2) for h in population]
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    for gen in range(generations):
        # Evaluate population
        fitness = [evaluate(h) for h in population]
        
        # Select elite
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = [population[i] for i in elite_indices]
        
        # Tournament selection
        new_population = []
        for _ in range(population_size - elite_size):
            candidates = np.random.choice(range(population_size), tournament_size, replace=False)
            winner = candidates[np.argmin(fitness[candidates])]
            new_population.append(population[winner])
        
        # Crossover and mutation
        for i in range(len(new_population)):
            if np.random.rand() < mutation_rate:
                # Mutation: small random perturbation
                new_population[i] = np.clip(new_population[i] + np.random.normal(0, 0.05, n_points), 0, 1)
                new_population[i] = new_population[i] / np.sum(new_population[i]) * (n_points / 2)
            else:
                # Crossover with elite
                parent1 = np.random.choice(elite)
                parent2 = np.random.choice(elite)
                child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
                new_population[i] = child / np.sum(child) * (n_points / 2)
        
        population = elite + new_population
        
        # Update best solution
        for h in population:
            current_c5 = evaluate(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
