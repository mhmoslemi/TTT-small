import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)

    # Initial construction with a better periodic pattern
    n_points = 128
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if 0 <= x < 0.5:
            initial_h[i] = 1.0
        elif 0.5 <= x < 1.0:
            initial_h[i] = 0.0
        elif 1.0 <= x < 1.5:

    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    
    def compute_c5(h):
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Optimization with constraints
    h = cp.Variable(n_points)
    constraints = [
        cp.sum(h) == n_points / 2,
        0 <= h,
        h <= 1
    ]
    obj = cp.Minimize(cp.Max(cp.multiply(cp.ones(2*n_points-1), cp.convolve(h, 1 - h))))
    prob = cp.Problem(obj, constraints)
    
    # Try solving with different solvers
    try:
        prob.solve(solver=cp.SCS, eps=1e-6, max_iter=1000)
        best_h = h.value
    except:
        best_h = initial_h.copy()
    
    best_c5 = compute_c5(best_h)
    
    # Local search with adaptive perturbations
    for _ in range(200):

    return best_h, best_c5, n_points
