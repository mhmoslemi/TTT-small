from scipy.optimize import minimize

        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)

    h = h_initial.copy()
    h = h / np.sum(h) * (n_points / 2)
    best_c5 = objective(h)
    best_h = h.copy()
    
    # Genetic algorithm setup
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0.0, 1.0)
    toolbox.register("individual", tools.initRepeat, creator.Individual, 
                     toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    
    toolbox.register("evaluate", objective)
    toolbox.register("mate", tools.cxBlend, alpha=0.1)
    toolbox.register("mutate", tools.mutGaussian, mu=0.0, sigma=0.05, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Run genetic algorithm
    population = toolbox.population(n=50)
    for gen in range(100):

            ind.fitness.values = fit

            best_c5 = c5_perturb
    
    return best_h, best_c5, n_points
