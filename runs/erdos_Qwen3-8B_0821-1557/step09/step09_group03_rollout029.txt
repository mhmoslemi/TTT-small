import numpy as np
import scipy.signal as signal
from scipy.optimize import minimize
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    def objective(x):
        h = x.copy()
        # Normalize to satisfy the integral constraint
        h = h / np.sum(h) * (n_points / 2)
        # Compute the overlap integral
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Set up genetic algorithm
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", np.ndarray, fitness=creator.FitnessMin)
    
    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0.0, 1.0)
    toolbox.register("individual", tools.initRepeat, creator.Individual, 
                    toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    
    toolbox.register("evaluate", objective)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=1.0/n_points)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    population = toolbox.population(n=50)
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    start_time = np.datetime64('now')
    
    # Run genetic algorithm
    for gen in range(100):
        if np.datetime64('now') - start_time > np.timedelta64(budget_s, 's'):
            break
        
        offspring = algorithms.varAnd(population, toolbox, cxpb=0.5, mutpb=0.1)
        fits = [toolbox.evaluate(ind) for ind in offspring]
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = (fit,)
        
        population = toolbox.select(offspring, len(population))
        
        # Update best solution
        for ind in population:
            if ind.fitness.values[0] < best_c5:
                best_c5 = ind.fitness.values[0]
                best_h = ind.copy()
    
    # Local refinement with gradient descent
    def local_objective(x):
        h = x.copy()
        h = h / np.sum(h) * (n_points / 2)
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    result = minimize(local_objective, best_h, method='L-BFGS-B', bounds=bounds, tol=1e-4, options={'maxiter': 50})
    best_h = result.x
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    best_c5 = local_objective(best_h)
    
    return best_h, best_c5, n_points
