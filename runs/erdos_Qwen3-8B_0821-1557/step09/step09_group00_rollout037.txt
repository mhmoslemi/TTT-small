    mutation_rate = 0.1
    crossover_rate = 0.8

        # Evaluate population
        fitness = [compute_c5(ind) for ind in population]
        best_idx = np.argmin(fitness)
        current_best_c5 = fitness[best_idx]
        
        # Update best solution
        if current_best_c5 < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = current_best_c5
        
        # Elitism: keep the best solution
        population[0] = best_h.copy()
        fitness[0] = best_c5
        
        # Create new population
        new_population = [best_h.copy() for _ in range(population_size)]
        
        for i in range(1, population_size):
            # Select parents
            parent1 = random.choices(population, weights=fitness)[0]
            parent2 = random.choices(population, weights=fitness)[0]
            
            # Crossover
            if random.random() < crossover_rate:

                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])

                child = parent1.copy()

            child = child / np.sum(child) * (n_points / 2)

        population = new_population

        if time_per_generation * gen >= budget_s:
