import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return np.inf
        h = h * (n_points / 2) / total
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def crossover(parent1, parent2):
        split = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:split], parent2[split:]])
        return child
    
    def mutate(h):
        indices = np.random.choice(n_points, size=2, replace=False)
        h[indices[0]] = np.random.uniform(0, 1)
        h[indices[1]] = np.random.uniform(0, 1)
        return h
    
    population_size = 20
    population = [np.array(initial_h) for _ in range(population_size)]
    
    t0 = time.time()
    for generation in range(100):
        if time.time() - t0 > budget_s:
            break
        fitness = [evaluate(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            parent1, parent2 = np.random.choice(population, size=2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        # Replace worst with best
        new_population.append(best_h)
        population = new_population
    
    return best_h, best_c5, n_points
