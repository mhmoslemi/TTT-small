import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h):
        return evaluate(h)
    
    # Initial solution
    h = initial_h.copy()
    h = normalize(h)
    best_h = h.copy()
    best_c5 = evaluate(h)
    t0 = time.time()
    
    # Simulated annealing parameters
    temp = 1.0
    cooling_rate = 0.995
    max_iterations = 1000
    
    for _ in range(max_iterations):
        if time.time() - t0 > budget_s:
            break
        
        # Perturb the current solution
        perturbation = np.random.normal(0, 0.05, n_points)
        new_h = h + perturbation
        new_h = np.clip(new_h, 0, 1)
        new_h = normalize(new_h)
        
        # Evaluate new solution
        new_c5 = evaluate(new_h)
        
        # Acceptance probability
        if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5) / temp):
            h = new_h
            best_c5 = new_c5
        
        # Local optimization
        if np.random.rand() < 0.2:
            res = optimize.minimize(
                objective,
                h,
                bounds=[(0, 1) for _ in range(n_points)],
                method='L-BFGS-B'
            )
            h = res.x
        
        temp *= cooling_rate
    
    return best_h, best_c5, n_points
