import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)

        total = np.sum(h)
        if total > 0:
            h = h * (n_points / 2) / total
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]

    result = optimize.minimize(objective, h0, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint},

    if time.time() - t0 > budget_s:
        return initial_h, objective(initial_h), n_points
    
    best_h = result.x

    def simulated_annealing(h_start, T=1.0, cooling_rate=0.995, iterations=100):

        best_h = h.copy()
        best_c5 = objective(h)
        for _ in range(iterations):
            T *= cooling_rate
