import time

    n_points = len(initial_h)

    def objective(h):
        h = h.copy()
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    def normalize(h):
        h = h.copy()
        h /= np.sum(h) * dx
        return h
    
    def evaluate(h):
        return objective(normalize(h))
    
    # Initial guess
    h0 = np.array(initial_h, dtype=np.float64)
    
    # Multi-start optimization
    best_h = h0.copy()
    best_c5 = evaluate(h0)
    
    start_time = time.time()
    
    for _ in range(5):  # 5 different starting points
        # Perturb initial guess
        perturbation = np.random.normal(0, 0.05, n_points)
        h_start = normalize(h0 + perturbation)
        
        # Gradient descent with adaptive step size
        h = h_start.copy()
        step_size = 0.1
        for _ in range(1000):  # 1000 iterations
            # Compute gradient numerically
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-6
                h_minus = h.copy()
                h_minus[i] -= 1e-6

            # Project gradient to maintain constraint
            grad -= (np.sum(grad) / n_points) * np.ones(n_points)
            
            # Update
            h -= step_size * grad

            step_size *= 0.9995

            best_c5 = current_c5
