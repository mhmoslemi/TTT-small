import time

        h /= norm
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

        h = np.array(h, dtype=np.float64)
        norm = np.sum(h) * dx
        return norm - 1.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    options = {
        'maxiter': 100,
        'ftol': 1e-8,
        'gtol': 1e-8,
        'disp': False
    }
    
    start_time = time.time()
    
    # Run optimization
    result = optimize.minimize(
        objective,
        h0,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        options=options
    )
    
    # Check time budget
    if time.time() - start_time > budget_s:
