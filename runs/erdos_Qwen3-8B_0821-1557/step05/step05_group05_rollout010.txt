    constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]

    def objective_function(h_val):
        h_array = np.array(h_val)
        corr = np.correlate(h_array, 1 - h_array, mode="full")
        return np.max(corr * dx)
    
    # Optimization
    start_time = time.time()
    best_c5 = float('inf')

        obj = cp.Minimize(objective_function(h))
