import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Generate initial population
    population = []
    for _ in range(10):
        h = np.random.uniform(0, 1, n_points)
        h = normalize(h)
        population.append(h)
    
    best_h = population[0]
    best_c5 = evaluate(best_h)
    elapsed_time = 0
    
    while elapsed_time < budget_s:
        # Evaluate population
        current_evals = [evaluate(h) for h in population]
        best_idx = np.argmin(current_evals)
        best_c5 = current_evals[best_idx]
        best_h = population[best_idx]
        
        # Perturb best solution
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        h_perturbed = best_h + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = normalize(h_perturbed)
        
        # Replace worst solution with perturbed one
        worst_idx = np.argmax(current_evals)
        population[worst_idx] = h_perturbed
        
        # Check time
        elapsed_time = time.time() - time.time()
    
    return best_h, best_c5, n_points
