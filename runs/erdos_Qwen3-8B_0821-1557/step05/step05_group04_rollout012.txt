from copy import deepcopy

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Initial population
    population_size = 30
    population = [np.array(initial_h, dtype=np.float64) for _ in range(population_size)]
    
    # Normalize initial population
    for h in population:
        h /= np.sum(h) * dx
    
    def evaluate(individual):
        return objective(individual)
    
    def mutate(individual, mutation_rate=0.1):
        individual = individual.copy()
        for i in range(len(individual)):
            if np.random.rand() < mutation_rate:
                individual[i] = np.random.uniform(0, 1)
        # Enforce integral constraint
        individual /= np.sum(individual) * dx
        return individual
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(len(child)):
            if np.random.rand() < 0.5:
                child[i] = parent2[i]
        # Enforce integral constraint
        child /= np.sum(child) * dx
        return child
    
    best_h = deepcopy(initial_h)
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s:

        # Evaluate fitness
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]

        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Select two parents
            parent1, parent2 = np.random.choice(population, 2, replace=False)
            # Crossover
            child = crossover(parent1, parent2)
            # Mutate
            child = mutate(child)
            new_population.append(child)
        
        # Replace old population
        population = new_population
