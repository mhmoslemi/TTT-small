import scipy.signal as signal

    dx = 2.0 / n_points
    target_integral = n_points / 2
    
    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return np.zeros_like(h)
        return h * target_integral / total
    
    def evaluate(h):
        h = normalize(h.clip(0, 1))
        corr = signal.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    population_size = 50
    mutation_rate = 0.1
    generations = 100
    
    population = [normalize(np.random.rand(n_points)) for _ in range(population_size)]
    population = [h.clip(0, 1) for h in population]
    
    for gen in range(generations):
        if gen % 10 == 0:
            print(f"Generation {gen}, best C5: {np.min([evaluate(h) for h in population])}")
        
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Select two parents
            parent1, parent2 = random.choices(population, k=2)
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] = np.random.rand()
                    child2[i] = np.random.rand()
            
            new_population.append(normalize(child1.clip(0, 1)))
            new_population.append(normalize(child2.clip(0, 1)))
        
        population = new_population[:population_size]
    
    best_h = best_h.copy()
    best_c5 = evaluate(best_h)
