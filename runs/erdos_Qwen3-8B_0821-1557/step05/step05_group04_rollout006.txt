def run(seed=42, budget_s=10, **kwargs):

    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def evaluate(h):
        return objective(h)
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] = np.clip(h[i] + np.random.normal(0, 0.1), 0, 1)
        return normalize(h)
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(len(child)):
            if random.random() < 0.5:
                child[i] = parent2[i]
        return normalize(child)
    
    population_size = 50
    population = [normalize(np.random.rand(n_points)) for _ in range(population_size)]
    
    start_time = time.time()
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        
        fitness = [evaluate(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Select parents
        parents = np.random.choice(population, size=population_size, replace=True)
        
        # Create new population
        new_population = []
        for i in range(population_size // 2):
            parent1 = parents[i]
            parent2 = parents[population_size - 1 - i]
