        velocities = np.random.uniform(-1, 1, (n_particles, n_points))
        pbest = particles.copy()
        pbest_cost = np.array([objective(p) for p in particles])
        gbest = particles[np.argmin(pbest_cost)]
        gbest_cost = np.min(pbest_cost)
