        best_c5 = objective(best_h)

            new_population = []
            for h in population:
                # Crossover
                idx = np.random.choice(n_points)
                child = np.copy(h)

                total = np.sum(child)

                new_population.append(child)
            population = new_population
            for h in population:
                result = optimize.minimize(objective, h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint},
