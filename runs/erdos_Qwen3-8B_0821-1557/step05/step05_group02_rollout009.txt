import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        h = h * (n_points / 2) / total if total > 0 else h
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    def evaluate(solution):
        return objective(solution)
    
    def mutate(solution, mutation_rate=0.1):
        solution = solution.copy()
        for i in range(len(solution)):
            if random.random() < mutation_rate:
                solution[i] = np.random.uniform(0, 1)
        return solution
    
    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        for i in range(len(parent1)):
            if random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child
    
    population_size = 50
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * (n_points / 2) / np.sum(h) if np.sum(h) > 0 else h for h in population]
    
    best_solution = initial_h.copy()
    best_c5 = objective(best_solution)
    
    t0 = time.time()
    for generation in range(100):
        if time.time() - t0 > budget_s:
            break
        # Evaluate fitness
        fitness = [evaluate(ind) for ind in population]
        # Select parents
        parents = np.argsort(fitness)[:population_size//2]
        parents = [population[i] for i in parents]
        # Create new population
        new_population = []
        for _ in range(population_size//2):
            p1, p2 = random.sample(parents, 2)
            child = crossover(p1, p2)
            child = mutate(child)
            new_population.append(child)
        # Replace old population
        population = new_population
        # Update best solution
        for ind in population:
            current_c5 = evaluate(ind)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_solution = ind.copy()
    
    return best_solution, best_c5, n_points
