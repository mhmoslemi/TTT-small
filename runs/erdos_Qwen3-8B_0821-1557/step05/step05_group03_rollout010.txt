import scipy.optimize as optimize
import time

        return h * (n_points / 2) / np.sum(h)

        h = np.clip(h, 0, 1)
        h = normalize(h)

        return np.max(corr) * dx

        return evaluate(h)

    # Gradient-based optimization
    result = optimize.minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
