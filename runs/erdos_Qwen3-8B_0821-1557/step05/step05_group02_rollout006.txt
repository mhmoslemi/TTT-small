        return np.max(corr) * dx

        return h * (n_points / 2) / total
    
    def mutate(individual, mutation_rate=0.1):
        individual = individual.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                individual[i] = np.random.rand()
        total = np.sum(individual)
        return individual * (n_points / 2) / total
    
    def crossover(parent1, parent2):
        child = np.zeros(n_points)
        for i in range(n_points):
            if random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        total = np.sum(child)
        return child * (n_points / 2) / total
    
    population = [create_individual() for _ in range(50)]
    best_h = initial_h.copy()

    t0 = time.time()

            child = crossover(parent1, parent2)
