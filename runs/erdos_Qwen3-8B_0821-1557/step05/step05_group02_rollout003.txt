import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return float('inf')
        h = h * (n_points / 2) / total
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    h0 = initial_h.copy()
    h0 = np.clip(h0, 0, 1)
    total = np.sum(h0)
    h0 = h0 * (n_points / 2) / total if total > 0 else h0
    
    t0 = time.time()
    
    # Multi-start optimization with adaptive bounds
    results = []
    for _ in range(5):
        if time.time() - t0 > budget_s:
            break
        # Perturb initial guess
        perturb = np.random.normal(0, 0.05, size=n_points)
        h_init = np.clip(h0 + perturb, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init) if np.sum(h_init) > 0 else h_init
        
        result = optimize.minimize(
            objective, h_init, method='L-BFGS-B', bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'maxiter': 50}
        )
        
        if result.success:
            results.append((result.fun, result.x))
    
    # Select the best result
    if results:
        best_c5, best_h = min(results, key=lambda x: x[0])
    else:
        best_c5 = objective(h0)
        best_h = h0
    
    return best_h, best_c5, n_points
