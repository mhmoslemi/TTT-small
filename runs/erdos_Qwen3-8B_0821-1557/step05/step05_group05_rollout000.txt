import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = np.array(h, dtype=np.float64)
        # Normalize to maintain integral constraint
        h = h / (np.sum(h) * dx) * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    # Bounds for each parameter
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    initial_guess = np.array(initial_h, dtype=np.float64)
    
    # Optimization settings
    options = {
        'maxiter': 100,
        'ftol': 1e-8,
        'gtol': 1e-8,
        'disp': False
    }
    
    start_time = time.time()
    
    # Perform optimization
    result = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        method='L-BFGS-B',
        options=options
    )
    
    # Check if we've exceeded the time budget
    if time.time() - start_time > budget_s - 1:
        # Return the best solution found so far
        best_h = initial_guess
        best_c5 = objective(best_h)
        return best_h.tolist(), best_c5, n_points
    
    # Extract the best solution
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h.tolist(), best_c5, n_points
