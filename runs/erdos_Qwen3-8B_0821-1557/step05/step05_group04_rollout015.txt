        return objective(h)

        population = [np.array(h) for h in population]
        fitnesses = [fitness(h) for h in population]
        sorted_idx = np.argsort(fitnesses)
        elite = [population[i] for i in sorted_idx[:n_elite]]
        
        new_population = elite.copy()
        while len(new_population) < len(population):
            parent1, parent2 = np.random.choice(sorted_idx[:n_elite], 2, replace=False)
            p1 = population[parent1]
            p2 = population[parent2]
            
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate((p1[:crossover_point], p2[crossover_point:]))
            else:
                child = p1
            
            if np.random.rand() < mutation_rate:
                mutation_indices = np.random.choice(n_points, int(n_points * 0.05), replace=False)
                child[mutation_indices] = np.random.uniform(0, 1, size=len(mutation_indices))
            
            # Normalize to maintain integral constraint
            child = child / np.sum(child) * (n_points / 2)
            
            new_population.append(child)
        
        return new_population
    
    population_size = 50
