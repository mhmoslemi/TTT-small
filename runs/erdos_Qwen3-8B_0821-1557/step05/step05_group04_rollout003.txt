import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Initial population
    population_size = 30
    population = [np.array(initial_h, dtype=np.float64) for _ in range(population_size)]
    
    # Evaluate initial population
    population_fitness = [objective(ind) for ind in population]
    
    # Genetic algorithm parameters
    mutation_rate = 0.1
    crossover_rate = 0.7
    generations = 50
    
    start_time = time.time()
    
    for generation in range(generations):
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        
        # Select parents
        parents = np.random.choice(population_size, size=population_size, p=np.array(fitness)/np.sum(fitness))
        parents = [population[i] for i in parents]
        
        # Crossover
        offspring = []
        for i in range(population_size // 2):
            parent1 = parents[2*i]
            parent2 = parents[2*i + 1]
            if np.random.rand() < crossover_rate:
                split = np.random.randint(1, n_points - 1)
                child1 = np.concatenate((parent1[:split], parent2[split:]))
                child2 = np.concatenate((parent2[:split], parent1[split:]))
                offspring.extend([child1, child2])
            else:
                offspring.extend([parent1, parent2])
        
        # Mutation
        for i in range(len(offspring)):
            offspring[i] = offspring[i].copy()
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    offspring[i][j] = np.random.uniform(0, 1)
        
        # Replace population
        population = offspring
        
        # Early exit if time budget is exceeded
        if time.time() - start_time > budget_s:
            best_idx = np.argmin(population_fitness)
            best_h = population[best_idx]
            best_c5 = population_fitness[best_idx]
            return best_h.tolist(), best_c5, n_points
        
        population_fitness = [objective(ind) for ind in population]
    
    best_idx = np.argmin(population_fitness)
    best_h = population[best_idx]
    best_c5 = population_fitness[best_idx]
    return best_h.tolist(), best_c5, n_points
