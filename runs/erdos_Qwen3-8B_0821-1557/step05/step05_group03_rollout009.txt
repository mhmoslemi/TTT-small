            break
        
        # Evaluate population
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_c5 = fitness[best_idx]

            best_h = current_best.copy()

        # Select parents
        parents = np.random.choice(population_size, size=elite_size, replace=False)
