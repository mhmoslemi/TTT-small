        elite_indices = np.argsort(fitness)[:elite_size]

        parent_indices = np.argsort(fitness)[-2*pop_size//3:]
        parents = [population[i] for i in parent_indices]
        
        # Crossover
        offspring = []

            parent1, parent2 = np.random.choice(parents, 2, replace=False)
