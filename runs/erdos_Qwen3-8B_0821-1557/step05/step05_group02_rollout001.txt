        alpha = np.random.rand(n_points)
        return alpha * parent1 + (1 - alpha) * parent2
    
    def mutate(individual, mutation_rate=0.05):

                individual[i] = np.random.rand()
        total = np.sum(individual)
        if total > 0:
            individual = individual * (n_points / 2) / total
        return individual
    
    population_size = 50
    population = [generate_individual() for _ in range(population_size)]

        selected = [ind for _, ind in ranked[:population_size//2]]
