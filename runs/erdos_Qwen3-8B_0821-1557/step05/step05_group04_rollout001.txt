    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = h.copy()
        h /= np.sum(h) * dx
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    # Initialize population
    population_size = 40
    population = [np.array(initial_h, dtype=np.float64) for _ in range(population_size)]
    for h in population:
        h /= np.sum(h) * dx
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] = np.clip(np.random.uniform(0, 1), 0, 1)
        return h
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        for i in range(len(child)):
            if random.random() < 0.5:
                child[i] = parent2[i]
        return child
    
    best_h = np.array(initial_h, dtype=np.float64)
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = [evaluate(h) for h in population]
        
        # Find the best solution
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_c5 = fitness[best_idx]
        
        if current_c5 < best_c5:
            best_h = current_best

        # Create new population
        new_population = []

            parent1, parent2 = random.choices(population, k=2)
            
            # Crossover
            child1 = crossover(parent1, parent2)
            child2 = crossover(parent2, parent1)

            child1 = mutate(child1)
