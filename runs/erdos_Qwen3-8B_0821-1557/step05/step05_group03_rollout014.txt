import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Define the objective function with constraints
    def objective(h):
        return evaluate(h)
    
    def constraints(h):
        return np.array([np.sum(h) - n_points / 2])  # Enforce integral constraint
    
    # Initial guess
    h_initial = initial_h.copy()
    h_initial = normalize(h_initial)
    
    # Optimization settings
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraints}]
    
    # Gradient-based optimization
    result = optimize.minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Check if optimization completed within budget
    elapsed_time = time.time() - time.time()
    if elapsed_time > budget_s:
        return initial_h, evaluate(initial_h), n_points
    
    best_h = result.x
    best_c5 = evaluate(best_h)
    
    # Genetic algorithm for global search
    population_size = 20
    generations = 5
    mutation_rate = 0.1
    
    for _ in range(generations):
        if time.time() - time.time() > budget_s:
            break
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [normalize(h) for h in population]
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx]
        best_c5 = fitness[best_idx]
        
        # Mutation
        for i in range(population_size):
            if random.random() < mutation_rate:
                perturb = np.random.normal(0, 0.05, n_points)
                population[i] = np.clip(population[i] + perturb, 0, 1)
                population[i] = normalize(population[i])
    
    # Local search with adaptive perturbations
    for _ in range(10):
        if time.time() - time.time() > budget_s:
            break
        # Perturb the solution
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        h_perturbed = best_h + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = normalize(h_perturbed)
        current_c5 = evaluate(h_perturbed)
        if current_c5 < best_c5:
            best_h = h_perturbed
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
