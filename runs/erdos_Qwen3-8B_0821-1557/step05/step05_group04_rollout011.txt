import numpy as np
import random
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def fitness(h):
        h_norm = h / np.sum(h) * n_points / 2
        c5 = evaluate(h_norm)
        # Penalize for violating integral constraint
        penalty = 100 * abs(np.sum(h_norm) - n_points / 2)
        return c5 + penalty
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        return h
    
    def crossover(parent1, parent2):
        crossover_point = np.random.randint(1, len(parent1) - 1)
        child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        return child
    
    population_size = 50
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    
    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        # Evaluate fitness
        fitness_values = [fitness(ind) for ind in population]
        # Select top 20%
        top_indices = np.argsort(fitness_values)[:int(0.2 * population_size)]
        top_parents = [population[i] for i in top_indices]
        
        # Create new population
        new_population = []
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(top_parents, 2)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
    
    best_h = min(population, key=fitness)
    best_c5 = evaluate(best_h / np.sum(best_h) * n_points / 2)
    return best_h.tolist(), best_c5, n_points
