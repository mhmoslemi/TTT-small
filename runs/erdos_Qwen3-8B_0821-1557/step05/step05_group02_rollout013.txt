        h = h * (n_points / 2) / total if total > 0 else h
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

        total = np.sum(h)

        crossover_point = np.random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        return normalize(child)
    
    def mutate(h, mutation_rate=0.05):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        return normalize(h)
    
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(20)]
    best_solution = initial_h.copy()
    best_c5 = evaluate_solution(best_solution)
    
    t0 = time.time()

        # Evaluate population
        fitness = [evaluate_solution(ind) for ind in population]
        # Select parents
        parents = np.argsort(fitness)[:10]
        # Create new population
        new_population = []
