import numpy as np

import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initialize population
    population_size = 50
    population = [normalize(np.random.uniform(0, 1, n_points)) for _ in range(population_size)]
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [evaluate(ind) for ind in population]
        best_idx = np.argmin(fitness)
        current_best = population[best_idx]
        current_c5 = fitness[best_idx]
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
        
        # Select parents
        parents = np.random.choice(population, size=population_size, p=fitness/np.sum(fitness))
        
        # Crossover
        offspring = []
        for _ in range(population_size // 2):
            parent1 = random.choice(parents)
            parent2 = random.choice(parents)
            child = np.zeros(n_points)

                    child[i] = parent1[i]

                if random.random() < 0.1:

                    offspring[i][j] = np.clip(offspring[i][j], 0, 1)
        
        # Replace population
        population = offspring
    
    return best_h, best_c5, n_points
