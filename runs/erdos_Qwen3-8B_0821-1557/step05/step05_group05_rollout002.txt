import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        norm = np.sum(h) * dx
        h /= norm
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def fitness(h):
        return evaluate(h)
    
    # Initial population
    initial_population = [np.array(initial_h, dtype=np.float64)]

    max_iter = 100
    tol = 1e-6
    start_time = time.time()
    
    # Optimization using L-BFGS-B
    def objective(x):
        return fitness(x)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint}
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = optimize.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        tol=tol,
        options={'maxiter': max_iter, 'disp': False}
    )
    
    # Check time and return best solution
    if time.time() - start_time > budget_s:
        return initial_h.tolist(), evaluate(initial_h), n_points
