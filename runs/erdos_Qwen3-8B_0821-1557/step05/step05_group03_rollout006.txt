        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Define the objective function with constraints

        return evaluate(h)

    mutation_rate = 0.1
    generations = 50
    elite_size = 5
    
    # Initial population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [normalize(p) for p in population]
    
    start_time = time.time()
    
    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [evaluate(ind) for ind in population]
        
        # Select elite
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elite.copy()
        
        # Generate offspring
        for _ in range(population_size - elite_size):
            # Select parents
            parent1 = np.random.choice(elite)
            parent2 = np.random.choice(elite)
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.uniform(-0.1, 0.1)
                    child[i] = np.clip(child[i], 0, 1)
            
            # Normalize
            child = normalize(child)
            new_population.append(child)
        
        population = new_population
    
    # Evaluate final population
    fitness = [evaluate(ind) for ind in population]
    best_index = np.argmin(fitness)
    best_h = population[best_index]
