def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        h = h.copy()
        h /= np.sum(h) * dx
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Initialize population
    population_size = 50
    population = [np.array(initial_h, dtype=np.float64) for _ in range(population_size)]
    for p in population:
        p /= np.sum(p) * dx
    
    def evaluate(individual):
        return objective(individual)
    
    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        for i in range(n_points):
            if random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        child /= np.sum(child) * dx
        return child
    
    def mutate(individual):
        for i in range(n_points):
            if random.random() < 0.1:
                individual[i] = np.random.uniform(0, 1)
        individual /= np.sum(individual) * dx
        return individual
    
    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [evaluate(ind) for ind in population]
        
        # Select top performers
        top_indices = np.argsort(fitness)[:population_size//2]
        top_population = [population[i] for i in top_indices]
        
        # Create new population
        new_population = top_population[:]
        while len(new_population) < population_size:
            # Tournament selection
            a, b = random.choices(top_indices, k=2)
            if fitness[a] < fitness[b]:
                parent1 = population[a]
            else:
                parent1 = population[b]

                parent2 = population[a]
            else:
                parent2 = population[b]
            
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
    
    best_h = min(population, key=evaluate)
    best_c5 = evaluate(best_h)
