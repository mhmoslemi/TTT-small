        return h * (n_points / 2) / np.sum(h)

    population_size = 50
    population = [normalize(np.array(initial_h, dtype=np.float64) + np.random.normal(0, 0.05, n_points)) for _ in range(population_size)]

        # Evaluate fitness
        fitness = [evaluate(ind) for ind in population]
        
        # Select parents
        parents = np.array(population)[np.argsort(fitness)][::-1][:population_size//2]
        
        # Create next generation
        next_gen = []
        for _ in range(population_size):
            # Tournament selection
            idx1, idx2 = np.random.choice(len(parents), 2, replace=False)
            parent1, parent2 = parents[idx1], parents[idx2]
            child = parent1 * 0.5 + parent2 * 0.5

    # Find best solution
