import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def constraint(h):
        return np.sum(h) - n_points / 2

        constraints={'type': 'eq', 'fun': constraint},
        tol=1e-10,
        options={'maxiter': 100}
    )
    
    # Early exit if time budget is exceeded
    if time.time() - start_time > budget_s:
        return initial_h.tolist(), objective(h0), n_points
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h.tolist(), best_c5, n_points
