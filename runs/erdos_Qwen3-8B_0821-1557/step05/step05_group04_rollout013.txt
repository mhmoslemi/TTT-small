        return np.max(corr * dx)

        h = np.random.rand(n_points)

        return h.tolist()

        child = np.array(child, dtype=np.float64)
        child /= np.sum(child) * dx
        return child.tolist()
    
    def mutate(individual, mutation_rate=0.05):
        individual = np.array(individual, dtype=np.float64)

            if random.random() < mutation_rate:
                individual[i] = np.random.rand()
        individual /= np.sum(individual) * dx
        return individual.tolist()
    
    def selection(population, fitnesses):
        sorted_indices = np.argsort(fitnesses)
        return [population[i] for i in sorted_indices[:len(population)//2]]
