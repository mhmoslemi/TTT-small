import scipy.optimize as optimize
import time

    n_points = len(initial_h)

    def objective(h):
        h = np.clip(h, 0, 1)
        integral = np.sum(h) * dx
        penalty = 1000 * (integral - 1) ** 2
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap + penalty

        return np.sum(h) * dx - 1

    t0 = time.time()
    result = optimize.minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'maxfun': 1000, 'ftol': 1e-8}
    )
    
    if time.time() - t0 > budget_s:
        return initial_h, evaluate_erdos_solution(initial_h), n_points
