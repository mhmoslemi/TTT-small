        h = np.zeros_like(parent1)

    def tournament_selection(population, k=3):
        selected = np.random.choice(len(population), k, replace=False)
        return population[selected[np.argmin([evaluate_individual(p) for p in population[selected]])]]
    
    population_size = 50
