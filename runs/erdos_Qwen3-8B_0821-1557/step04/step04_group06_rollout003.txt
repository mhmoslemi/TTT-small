import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def local_optimize(h):
        def objective(x):
            x = np.clip(x, 0, 1)
            x = normalize(x)
            corr = np.correlate(x, 1 - x, mode="full")
            return np.max(corr) * dx
        
        result = optimize.minimize(
            objective, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)]
        )
        return result.x
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    t0 = time.time()
    
    for _ in range(100):
        if time.time() - t0 > budget_s:
            break
        population = [np.random.uniform(0, 1, n_points) for _ in range(20)]
        for i in range(len(population)):
            population[i] = normalize(population[i])
        population.append(best_h)
        fitness = [evaluate(h) for h in population]
        best_idx = np.argmin(fitness)
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
        best_h = local_optimize(best_h)
    
    return best_h, best_c5, n_points
