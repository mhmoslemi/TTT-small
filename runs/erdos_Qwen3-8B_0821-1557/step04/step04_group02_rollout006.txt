import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate_individual(h):
        h = h.copy()
        total = np.sum(h) * dx
        h = h / total * (n_points / 2)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def fitness(h):
        return -evaluate_individual(h)  # Minimize C5 by maximizing fitness
    
    # Initial population
    population_size = 50
    population = [np.array(initial_h, dtype=np.float64) for _ in range(population_size)]
    for h in population:
        h /= np.sum(h) * dx
        h = h * (n_points / 2)
    
    def mutate(individual, mutation_rate=0.1):
        individual = individual.copy()
        for i in range(len(individual)):
            if random.random() < mutation_rate:
                individual[i] = np.clip(np.random.uniform(0, 1), 0, 1)
        return individual
    
    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        for i in range(len(child)):
            if random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child
    
    best_h = np.array(initial_h, dtype=np.float64)
    best_c5 = evaluate_individual(best_h)
    
    start_time = time.time()
    for generation in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate fitness
        fitnesses = [fitness(h) for h in population]
        population = [population[i] for i in np.argsort(fitnesses)[::-1][:population_size//2]]
        
        # Create new population
        new_population = []
        for _ in range(population_size):
            parent1, parent2 = random.choices(population, k=2)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
        
        # Update best solution
        for h in population:
            current_c5 = evaluate_individual(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
    
    return best_h.tolist(), best_c5, n_points
