    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate_individual(h):
        h = h.copy()
        h /= np.sum(h) * dx
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def create_individual():
        h = np.random.rand(n_points)
        h /= np.sum(h) * dx
        return h
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.rand()
        h /= np.sum(h) * dx
        return h
    
    def crossover(h1, h2):
        h = np.copy(h1)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                h[i] = h2[i]
        h /= np.sum(h) * dx
        return h
    
    def diversity(h1, h2):
        return np.sum(np.abs(h1 - h2))
    
    # Initial population
    population_size = 50
    population = [create_individual() for _ in range(population_size)]
    best_h = initial_h.copy()
    best_c5 = evaluate_individual(best_h)
    
    start_time = time.time()
    
    for generation in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate population
        fitness = [evaluate_individual(h) for h in population]
        best_idx = np.argmin(fitness)

        current_c5 = fitness[best_idx]
        
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
        
        # Select parents
        parents = np.random.choice(population_size, size=population_size, p=[f / sum(fitness) for f in fitness])
