import time

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Normalize initial_h to satisfy the integral constraint
    if not np.isclose(sum_h, n_points / 2):
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        if not np.isclose(np.sum(h), n_points / 2):
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def fitness(h):
        return objective(h)
    
    # Genetic algorithm parameters
    pop_size = 50
    n_gen = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = [np.clip(np.random.rand(n_points), 0, 1) for _ in range(pop_size)]
    population = [h * (n_points / 2) / np.sum(h) for h in population]
    
    best_h = initial_h.copy()
    best_c5 = objective(initial_h)
    t0 = time.time()
    
    for gen in range(n_gen):
        if time.time() - t0 > budget_s:
            break
        
        # Evaluate population
        fitness_values = [fitness(h) for h in population]
        best_idx = np.argmin(fitness_values)
        current_best = population[best_idx]
        current_c5 = fitness_values[best_idx]
        
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
        
        # Selection
        population = np.array(population)
        fitness_values = np.array(fitness_values)
        probabilities = fitness_values / np.sum(fitness_values)
        selected = np.random.choice(pop_size, size=pop_size, p=probabilities)
        selected = population[selected]
        
        # Crossover
        for i in range(pop_size):
            if np.random.rand() < crossover_rate:
                j = np.random.randint(pop_size)
                parent1 = selected[i]
                parent2 = selected[j]
                child = np.copy(parent1)
                for k in range(n_points):
                    if np.random.rand() < 0.5:
                        child[k] = parent2[k]
                selected[i] = child
        
        # Mutation
        for i in range(pop_size):
            for k in range(n_points):
                if np.random.rand() < mutation_rate:
                    selected[i][k] += np.random.uniform(-0.1, 0.1)
                    selected[i][k] = np.clip(selected[i][k], 0, 1)
        
        # Normalize
        for i in range(pop_size):
            selected[i] = selected[i] * (n_points / 2) / np.sum(selected[i])
        
        population = selected
    
    return best_h, best_c5, n_points
