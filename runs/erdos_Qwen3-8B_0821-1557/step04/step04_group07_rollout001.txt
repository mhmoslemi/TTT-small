    if not np.isclose(sum_h, n_points / 2):
        initial_h = np.full(n_points, 0.5)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        if not np.isclose(np.sum(h), n_points / 2):
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    population_size = 50
    mutation_rate = 0.1
    generations = 100
    
    population = [np.copy(initial_h) for _ in range(population_size)]
    for _ in range(generations):
        if time.time() - time.time() > budget_s:
            break
        
        # Evaluate fitness
        fitness = [objective(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_fitness = fitness[best_idx]
        
        # Selection
        selected = []
        for _ in range(population_size // 2):
            idx1 = np.random.randint(population_size)
            idx2 = np.random.randint(population_size)
            if fitness[idx1] < fitness[idx2]:
                selected.append(population[idx1])
            else:
                selected.append(population[idx2])
        
        # Crossover
        offspring = []
        for i in range(0, population_size, 2):
            if i + 1 < population_size:
                parent1 = selected[i]
                parent2 = selected[i + 1]
                child1 = np.copy(parent1)
                child2 = np.copy(parent2)
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        child1[j] = parent2[j]
                        child2[j] = parent1[j]
                offspring.append(child1)
                offspring.append(child2)
        
        # Mutation
        for i in range(len(offspring)):
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    offspring[i][j] = np.clip(offspring[i][j] + np.random.uniform(-0.1, 0.1), 0, 1)
        
        # Replace population
        population = [np.copy(ind) for ind in offspring]
        
        # Normalize to maintain integral constraint
        for i in range(len(population)):

            sum_pop = np.sum(population[i])
            if not np.isclose(sum_pop, n_points / 2):
                population[i] = population[i] / sum_pop * (n_points / 2)
    
    best_h = population[np.argmin([objective(ind) for ind in population])]
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
