This approach uses a genetic algorithm to evolve better approximations of the optimal step function. It starts from the initial construction and evolves better solutions by mutation and crossover, while enforcing the integral constraint. This avoids local optima and allows for more global exploration of the solution space.

import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate_individual(h):
        h = h.copy()
        h /= np.sum(h) * dx
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def generate_random_h():
        h = np.random.rand(n_points)
        h /= np.sum(h) * dx
        return h
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.rand()
        h /= np.sum(h) * dx
        return h
    
    def crossover(parent1, parent2):
        h = np.copy(parent1)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                h[i] = parent2[i]
        h /= np.sum(h) * dx
        return h
    
    population_size = 50
    population = [generate_random_h() for _ in range(population_size)]
    
    start_time = time.time()
    for generation in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate fitness
        fitness = [evaluate_individual(h) for h in population]
        
        # Select parents
        parents = np.argsort(fitness)[:population_size//2]
        parents = [population[i] for i in parents]
        
        # Create new population
        new_population = []
        for _ in range(population_size):
            parent1 = random.choice(parents)
            parent2 = random.choice(parents)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
    
    best_h = min(population, key=evaluate_individual)
    best_c5 = evaluate_individual(best_h)
    
    return best_h.tolist(), best_c5, n_points
