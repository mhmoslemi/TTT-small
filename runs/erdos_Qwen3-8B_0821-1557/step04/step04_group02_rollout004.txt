import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate_individual(h):
        h = h.copy()
        total = np.sum(h)
        h = h / (total * dx)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def crossover(parent1, parent2):
        child = np.copy(parent1)
        idx = np.random.choice(n_points)
        child[idx:] = parent2[idx:]
        return child
    
    def mutate(h):
        idx = np.random.choice(n_points)
        h[idx] = np.clip(h[idx] + np.random.uniform(-0.2, 0.2), 0, 1)
        total = np.sum(h)
        h = h / (total * dx)
        return h
    
    def normalize(h):
        total = np.sum(h)
        return h / (total * dx)
    
    population_size = 50
    population = [normalize(np.array(initial_h, dtype=np.float64)) for _ in range(population_size)]
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for generation in range(200):
        if time.time() - start_time > budget_s - 1:
            break
        # Evaluate fitness
        fitness = [evaluate_individual(ind) for ind in population]
        # Find best individual
        best_idx = np.argmin(fitness)
        best_c5 = fitness[best_idx]
        best_h = population[best_idx]
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Select parents
            parent1, parent2 = np.random.choice(population, size=2, replace=False)
            # Crossover
            child = crossover(parent1, parent2)
            # Mutate
            child = mutate(child)
            new_population.append(child)
        # Replace old population
        population = new_population
    
    return best_h.tolist(), best_c5, n_points
