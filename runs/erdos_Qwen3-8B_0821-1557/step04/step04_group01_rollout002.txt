import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Smaller n_points for faster computation
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    initial_h[:int(n_points/2)] = 1.0  # Simple initial construction
    
    def objective(h):
        h = h.copy()
        # Normalize to ensure integral constraint
        h /= np.sum(h) * dx
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h0 = initial_h.copy()
    h0 /= np.sum(h0) * dx
    
    # Optimization parameters
    max_iter = 100
    temp = 1.0
    cool_rate = 0.995
    
    start_time = time.time()
    best_h = h0.copy()
    best_c5 = objective(best_h)
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        # Generate a random perturbation
        perturb = np.random.normal(0, 0.05, n_points)
        h = best_h + perturb
        h = np.clip(h, 0, 1)  # Ensure values are within [0, 1]
        h /= np.sum(h) * dx  # Normalize
        
        # Evaluate new solution
        current_c5 = objective(h)
        
        # Accept with probability based on simulated annealing
        if np.random.rand() < np.exp((best_c5 - current_c5) / temp):
            best_h = h
            best_c5 = current_c5
        
        temp *= cool_rate
    
    return best_h.tolist(), best_c5, n_points
