import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        norm = np.sum(h) * dx
        h /= norm
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def fitness(h):
        return evaluate(h)
    
    # Initial population
    population_size = 20
    population = [np.array(initial_h, dtype=np.float64) for _ in range(population_size)]
    
    # Evaluate initial population
    fitness_values = [fitness(ind) for ind in population]
    
    # Genetic algorithm parameters
    mutation_rate = 0.1
    crossover_rate = 0.8
    generations = 100
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness_values = [fitness(ind) for ind in population]
        
        # Select parents
        parents = np.random.choice(population_size, size=population_size, p=np.array(fitness_values)/np.sum(fitness_values))
        parents = [population[i] for i in parents]
        
        # Crossover
        offspring = []
        for i in range(population_size//2):
            if np.random.rand() < crossover_rate:
                parent1 = parents[2*i]
                parent2 = parents[2*i+1]
                child = np.zeros(n_points)
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        child[j] = parent1[j]
                    else:
                        child[j] = parent2[j]
            else:
                child = parents[i]
            offspring.append(child)
        
        # Mutation
        for i in range(population_size):
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(n_points)
                population[i][idx] = np.random.uniform(0, 1)
        
        # Replace population
        population = offspring + [np.array(initial_h, dtype=np.float64) for _ in range(population_size - len(offspring))]
    
    # Find best solution
    best_index = np.argmin(fitness_values)
    best_h = population[best_index]
    best_c5 = fitness_values[best_index]
