import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reduced from 70 for better search space exploration
    dx = 2.0 / n_points
    initial_h = np.full(n_points, 0.5)
    initial_h = np.clip(initial_h, 0, 1)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        if not np.isclose(np.sum(h), n_points / 2):
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def evaluate_individual(h):
        return objective(h)
    
    def crossover(parent1, parent2):
        alpha = np.random.rand(n_points)
        return alpha * parent1 + (1 - alpha) * parent2
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h[i] = np.random.rand()
        return h
    
    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return np.full(n_points, 0.5)
        return h * (n_points / 2) / total
    
    def select_parents(population, fitnesses):
        tournament_size = 5
        tournament = np.random.choice(len(population), tournament_size)
        best_idx = tournament[np.argmin(fitnesses[tournament])]
        return population[best_idx].copy()
    
    population = [normalize(np.random.rand(n_points)) for _ in range(20)]
    population = [np.clip(h, 0, 1) for h in population]
    
    t0 = time.time()
    generations = 0
    
    while time.time() - t0 < budget_s:
        generations += 1
        
        fitnesses = [evaluate_individual(h) for h in population]
        best_idx = np.argmin(fitnesses)
        best_h = population[best_idx].copy()
        best_c5 = fitnesses[best_idx]
        
        # Create new population
        new_population = []
        for _ in range(len(population)):
            parent1 = select_parents(population, fitnesses)
            parent2 = select_parents(population, fitnesses)
            child = crossover(parent1, parent2)
            child = mutate(child)
            child = normalize(child)
            new_population.append(child)
        
        population = new_population
    
    return best_h, best_c5, n_points
