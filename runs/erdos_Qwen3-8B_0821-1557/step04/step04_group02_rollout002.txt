def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate_individual(h):
        h = h.copy()
        h /= np.sum(h) * dx
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if np.random.random() < mutation_rate:
                h[i] = np.random.uniform(0, 1)
        h /= np.sum(h) * dx
        return h
    
    def crossover(h1, h2):
        h = h1.copy()
        for i in range(len(h)):
            if np.random.random() < 0.5:
                h[i] = h2[i]
        h /= np.sum(h) * dx
        return h
    
    population_size = 50
    population = [mutate(np.array(initial_h, dtype=np.float64)) for _ in range(population_size)]
    best_h = population[0]
    best_c5 = evaluate_individual(best_h)
    
    start_time = time.time()
    while time.time() - start_time < budget_s - 1:
        # Evaluate all individuals
        fitness = [evaluate_individual(ind) for ind in population]
        
        # Select parents
        parents = np.random.choice(population, size=population_size, p=np.array(fitness)/np.sum(fitness))
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            p1 = np.random.choice(parents)
