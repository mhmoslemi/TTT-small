import time

def run(seed=42, budget_s=10, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def normalize(h):
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    t0 = time.time()
    
    # Genetic algorithm parameters
    population_size = 50
    mutation_rate = 0.2
    generations = 150
    elite_fraction = 0.2
    
    for _ in range(generations):
        if time.time() - t0 > budget_s:
            break
        
        # Generate new population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [normalize(h) for h in population]
        population.append(best_h)  # Add best solution as elite
        
        # Evaluate fitness
        fitness = [evaluate(h) for h in population]
        
        # Select elites
        elite_idx = np.argsort(fitness)[:int(elite_fraction * population_size)]
        elite_h = [population[i] for i in elite_idx]
        
        # Create new population through crossover and mutation
        new_population = []
        for h in elite_h:
            new_h = h.copy()
            # Apply mutation
            mutate_idx = np.random.rand(n_points) < mutation_rate
            new_h[mutate_idx] = np.random.uniform(0, 1, sum(mutate_idx))
            new_h = normalize(new_h)
            new_population.append(new_h)
        
        # Replace population
        population = new_population
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
    
    return best_h, best_c5, n_points
