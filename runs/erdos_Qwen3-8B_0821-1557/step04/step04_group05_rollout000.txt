        return h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Initialize population
    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]

    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    t0 = time.time()
    
    for _ in range(200):
        if time.time() - t0 > budget_s:
            break

        current_fitness = [evaluate(h) for h in population]
        
        # Select parents
        parent_indices = np.argsort(current_fitness)[:int(population_size/2)]
        parents = [population[i].copy() for i in parent_indices]
        
        # Create new population
        new_population = []
        for _ in range(int(population_size/2)):
            # Select two parents
            p1, p2 = np.random.choice(parent_indices, size=2, replace=False)
            parent1 = parents[p1]
            parent2 = parents[p2]
            
            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])

                    child[i] = np.random.uniform(0, 1)

            child = normalize(child)

        population = new_population + [best_h.copy()]
