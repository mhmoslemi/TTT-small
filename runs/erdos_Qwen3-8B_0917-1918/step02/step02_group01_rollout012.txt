import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from functools import partial
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 2 - 1
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)

    def objective(x):
        nonlocal h, dx
        h = x.copy()
        idx = np.where(h < 0)[0]
        h[idx] = 0
        idx = np.where(h > 1)[0]
        h[idx] = 1
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    constraint_func = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
    result = minimize(objective, h, bounds=bounds, constraints=constraint_func, method='L-BFGS-B', options={"ftol": 1e-8, "maxiter": 500})
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, len(h_values)
