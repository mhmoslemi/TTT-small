    np.random.seed(seed)

    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h, dtype=np.float64)

    def objective(x):
        nonlocal h, dx
        h = x.copy()
        h[h < 0] = 0
        h[h > 1] = 1
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)

    def evaluate(x):
        return objective(x)

    result = minimize(evaluate, h, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method='L-BFGS-B', options={"gtol": 1e-5, "maxiter": 100})

    def surrogate(x):
        return evaluate(x)

    kernel = ConstantKernel(1.0) * RBF(length_scale=1.0)
    gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=2)
    samples = np.random.uniform(0, 1, (100, n_points))
    samples = np.clip(samples, 0, 1)
    targets = np.array([evaluate(x) for x in samples])
    gp.fit(samples, targets)

    best_h = result.x
    best_c5 = evaluate(best_h)

    for _ in range(10):
        sample = np.random.uniform(0, 1, n_points)
        mean, std = gp.predict(sample.reshape(1, -1), return_std=True)
        perturbation = np.random.normal(loc=mean[0], scale=std[0], size=n_points)
        perturbation = np.clip(perturbation, 0, 1)
        perturbation = perturbation / np.max(perturbation) * np.max(best_h)
        h = np.clip(best_h + 0.1 * perturbation, 0, 1)
        h = h / np.sum(h) * n_points / 2
        result = minimize(evaluate, h, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method='L-BFGS-B', options={"gtol": 1e-5, "maxiter": 50})
        if evaluate(result.x) < best_c5:
            best_h = result.x
            best_c5 = evaluate(result.x)

    h_values = best_h
    c5_bound = best_c5
    return h_values, c5_bound, len(h_values)
