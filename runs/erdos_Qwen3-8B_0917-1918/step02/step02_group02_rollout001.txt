from scipy.optimize import Bounds, minimize
from functools import partial
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel

    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h, dtype=np.float64)

    def objective(x):
        nonlocal h, dx
        h = x.copy()
        idx = np.where(h < 0)[0]
        h[idx] = 0
        idx = np.where(h > 1)[0]
        h[idx] = 1
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)

    result = minimize(objective, h, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method='L-BFGS-B', options={"ftol": 1e-9, "maxiter": 100})
    h_values = result.x

    def surrogate_objective(x):
        return objective(x)

    gp = GaussianProcessRegressor(kernel=ConstantKernel(1.0) * RBF(length_scale=1.0), n_restarts_optimizer=5)
    X = np.reshape(np.linspace(0, 2, n_points), (-1, 1))
    y = np.array([surrogate_objective(x) for x in [h_values]])
    gp.fit(X, y)
    h_values = np.clip(gp.predict(X), 0, 1)
    h_values = np.clip(h_values, 0, 1)

    result = minimize(objective, h_values, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method='L-BFGS-B', options={"ftol": 1e-9, "maxiter": 50})
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, len(h_values)
