import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from functools import partial
from scipy.optimize import Bounds, minimize, LinearConstraint

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 128
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    initial_h[:n_points//2] = 1
    initial_h = np.concatenate((initial_h[:n_points//2], np.zeros(n_points//2)))
    h = np.array(initial_h, dtype=np.float64)

    def objective(x):
        nonlocal h, dx
        h = x.copy()
        idx = np.where(h < 0)[0]
        h[idx] = 0
        idx = np.where(h > 1)[0]
        h[idx] = 1
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), lb=0, ub=1)
    result = minimize(objective, h, bounds=bounds, constraints=cons, method='SLSQP', options={"ftol": 1e-9, "maxiter": 300})
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, len(h_values)
