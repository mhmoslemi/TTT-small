import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from functools import partial
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.random.rand(n_points)
    h = np.sort(h)
    h = np.searchsorted(np.linspace(0, 2, n_points), h) / n_points
    h = np.clip(h, 0, 1)
    h = np.linalg.lstsq(np.concatenate([np.ones((n_points, 1)), np.reshape(np.linspace(0, 2, n_points), (n_points, 1))]), 
                       np.reshape(h, (n_points, 1)), rcond=None)[0]
    h = np.clip(h, 0, 1)
    h = np.concatenate([h[-n_points//2:], h[:n_points//2]])
    h = np.interp(np.linspace(0, 2, n_points), np.sort(np.random.rand(n_points)), h)
    
    def objective(x):
        nonlocal h, dx
        h = x.copy()
        idx = np.where(h < 0)[0]
        h[idx] = 0
        idx = np.where(h > 1)[0]
        h[idx] = 1
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    constraint_func = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
    result = minimize(objective, h, bounds=bounds, constraints=constraint_func, method='SLSQP', options={"ftol": 1e-8, "maxiter": 150})
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, len(h_values)
