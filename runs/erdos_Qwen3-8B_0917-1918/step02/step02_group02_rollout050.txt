        idx = np.where(h < 0)[0]

        h[idx] = 1

        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)

        best_c5 = float('inf')
        best_h = np.zeros(n_points)

            cluster_centers = kmeans.cluster_centers_.flatten()
            cluster_assignments = kmeans.labels_
            h = np.zeros(n_points)
            for i in range(k):
                cluster_idx = np.where(cluster_assignments == i)[0]
                h[cluster_idx] = np.random.uniform(0, 1, len(cluster_idx))
