import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from functools import partial
from scipy.interpolate import interp1d
from sklearn.cluster import KMeans

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    initial_h = np.random.uniform(0, 1, n_points)
    initial_h = np.sort(initial_h)
    initial_h = np.interp(np.linspace(0, 1, n_points), np.sort(initial_h), np.linspace(0, 1, n_points))
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points), initial_h)
    initial_h = np.array([initial_h[i] for i in range(n_points)])
    
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = Bounds(0, 1)
    result = minimize(objective, initial_h, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method='L-BFGS-B', options={"ftol": 1e-9, "maxiter": 100})
    h_values = result.x
    c5_bound = objective(h_values)
    
    if c5_bound > 0.381:
        kmeans = KMeans(n_clusters=4).fit(np.reshape(h_values, (-1, 1)))
        h_values = np.sort(np.unique(kmeans.labels_))
        h_values = np.interp(np.linspace(0, 2, n_points), np.sort(h_values), np.linspace(0, 1, n_points))
        h_values = np.clip(h_values, 0, 1)
        h_values = np.array([h_values[i] for i in range(n_points)])
        result = minimize(objective, h_values, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method='L-BFGS-B', options={"ftol": 1e-9, "maxiter": 100})
        h_values = result.x
        c5_bound = objective(h_values)
    
    return h_values, c5_bound, len(h_values)
