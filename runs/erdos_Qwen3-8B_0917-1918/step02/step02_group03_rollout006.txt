import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(x):
        h_new = x.copy()
        h_new = np.clip(h_new, 0, 1)
        correl = np.correlate(h_new, 1 - h_new, mode="full")
        c5 = np.max(correl) * dx
        return c5
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    x0 = h.copy()
    
    bounds = opt.Bounds([0]*n_points, [1]*n_points)
    cons = opt.LinearConstraint(np.ones(n_points), [n_points/2], [n_points/2])
    
    start_time = time.time()
    result = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10})
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, len(best_h)
