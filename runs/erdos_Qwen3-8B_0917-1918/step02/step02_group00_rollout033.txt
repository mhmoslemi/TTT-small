import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from functools import partial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 128
    dx = 2.0 / n_points
    h = np.full(n_points, 0.5)
    for i in range(n_points):
        h[i] = 0.5 + 0.5 * np.cos(2 * np.pi * i / n_points)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    h *= (n_points / 2) / sum_h

    def objective(x):
        nonlocal h, dx
        h = x.copy()
        idx = np.where(h < 0)[0]
        h[idx] = 0
        idx = np.where(h > 1)[0]
        h[idx] = 1
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    result = minimize(objective, h, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method='L-BFGS-B', options={"ftol": 1e-8, "gtol": 1e-8, "maxiter": 300})
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, len(h_values)
