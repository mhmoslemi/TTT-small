```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from functools import partial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 128
    dx = 2.0 / n_points
    initial_h = np.full(n_points, 0.5)
    for i in range(n_points):
        initial_h[i] = 0.5 + 0.5 * np.cos(2 * np.pi * i / n_points)
    initial_h = np.clip(initial_h, 0, 1)
    sum_initial = np.sum(initial_h)
    initial_h *= (n_points / 2) / sum_initial

    def objective(x):
        h = x.copy()
        h[h < 0] = 0
        h[h > 1] = 1
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    result = minimize(objective, initial_h, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method='SLSQP', options={"ftol": 1e-8, "maxiter": 200})
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, len(h_values)
```
