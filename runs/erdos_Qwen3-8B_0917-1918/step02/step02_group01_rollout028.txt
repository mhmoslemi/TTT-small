import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize, differential_evolution
from functools import partial
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 1 if i % 4 == 0 else 0.5 if i % 2 == 0 else 0.25
    initial_h = np.tile(initial_h, 2)[:n_points]
    initial_h /= np.sum(initial_h) * dx
    initial_h = np.clip(initial_h, 0, 1)

    def objective(x):
        nonlocal dx
        h = x.copy()
        idx = np.where(h < 0)[0]
        h[idx] = 0
        idx = np.where(h > 1)[0]
        h[idx] = 1
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    constraint_func = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
    result = minimize(objective, initial_h, bounds=bounds, constraints=constraint_func, method='L-BFGS-B', options={"ftol": 1e-8, "maxiter": 100})
    result = minimize(objective, result.x, bounds=bounds, constraints=constraint_func, method='Nelder-Mead', options={"maxiter": 200})
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, len(h_values)
