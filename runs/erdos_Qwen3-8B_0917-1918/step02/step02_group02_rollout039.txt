import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from functools import partial
from collections import deque
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 0.5 if i % 3 == 0 else 0.3 if i % 3 == 1 else 0.7
    initial_h = initial_h + np.sin(np.linspace(0, 2*np.pi, n_points)) * 0.1
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h + np.random.normal(0, 0.05, n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    h = initial_h.copy()

    def objective(x):
        nonlocal h
        h = x.copy()
        idx = np.where(h < 0)[0]
        h[idx] = 0
        idx = np.where(h > 1)[0]
        h[idx] = 1
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * (2.0 / n_points))
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    result = minimize(objective, h, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method='L-BFGS-B', options={"ftol": 1e-9, "maxiter": 100})
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, len(h_values)
