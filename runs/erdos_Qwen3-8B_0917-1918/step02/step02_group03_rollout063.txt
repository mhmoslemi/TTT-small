import numpy as np
import random
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h, dtype=np.float64)
    
    def evaluate(solution):
        h_new = solution.copy()
        h_new[h_new < 0] = 0
        h_new[h_new > 1] = 1
        correl = np.correlate(h_new, 1 - h_new, mode="full")
        c5 = np.max(correl) * dx
        return c5
    
    def fitness(solution):
        return evaluate(solution)
    
    def create_individual():
        return np.random.uniform(0, 1, n_points)
    
    def crossover(parent1, parent2):
        split = random.randint(1, n_points - 1)
        child = np.concatenate([parent1[:split], parent2[split:]])
        return child
    
    def mutate(solution, mutation_rate=0.1):
        for i in range(n_points):
            if random.random() < mutation_rate:
                solution[i] = np.random.uniform(0, 1)
        return solution
    
    def tournament_selection(population, fitnesses, size=3):
        winners = random.sample(list(enumerate(population)), size)
        winners.sort(key=lambda x: fitnesses[x[0]])
        return winners[0][1]
    
    population_size = 50
    population = [create_individual() for _ in range(population_size)]
    fitnesses = [fitness(sol) for sol in population]
    
    start_time = time.time()
    best_solution = population[fitnesses.index(min(fitnesses))]
    best_c5 = min(fitnesses)
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        
        new_population = []
        for _ in range(population_size // 2):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
        fitnesses = [fitness(sol) for sol in population]
        current_best = population[fitnesses.index(min(fitnesses))]
        if fitness(current_best) < best_c5:
            best_solution = current_best
            best_c5 = fitness(current_best)
    
    best_solution[best_solution < 0] = 0
    best_solution[best_solution > 1] = 1
    return best_solution, best_c5, len(best_solution)
