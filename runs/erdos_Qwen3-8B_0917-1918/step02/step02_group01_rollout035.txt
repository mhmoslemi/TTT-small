import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from functools import partial
from scipy.optimize import LinearConstraint
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            initial_h[i] = 1
        else:
            initial_h[i] = 0
    h = np.array(initial_h, dtype=np.float64)

    def objective(x):
        nonlocal h
        h = x.copy()
        idx = np.where(h < 0)[0]
        h[idx] = 0
        idx = np.where(h > 1)[0]
        h[idx] = 1
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * (2.0 / n_points))
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    constraint_func = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
    result = minimize(objective, h, bounds=bounds, constraints=constraint_func, method='L-BFGS-B', options={"ftol": 1e-8, "maxiter": 200})
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, len(h_values)
