```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from functools import partial
from sklearn.gaussian_process import Lasso
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    initial_h = np.zeros(n_points)
    initial_h[:n_points//2] = 1.0
    initial_h[n_points//2:] = 0.0
    h = np.array(initial_h, dtype=np.float64)

    def objective(x):
        nonlocal h
        h = x.copy()
        idx = np.where(h < 0)[0]
        h[idx] = 0
        idx = np.where(h > 1)[0]
        h[idx] = 1
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * 2.0 / n_points)
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    result = minimize(objective, h, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method='L-BFGS-B', options={"ftol": 1e-9, "maxiter": 300})
    h_values = result.x

    def eval_objective(x):
        nonlocal h
        h = x.copy()
        idx = np.where(h < 0)[0]
        h[idx] = 0
        idx = np.where(h > 1)[0]
        h[idx] = 1
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * 2.0 / n_points)
        return c5

    def perturb(h, strength=0.1):
        noise = np.random.normal(0, strength, h.shape)
        return np.clip(h + noise, 0, 1)

    best_h = h_values
    best_c5 = eval_objective(best_h)

    for _ in range(30):
        h_perturbed = perturb(best_h, strength=0.05)
        result = minimize(eval_objective, h_perturbed, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method='L-BFGS-B', options={"ftol": 1e-9, "maxiter": 200})
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    return best_h, best_c5, n_points
```
