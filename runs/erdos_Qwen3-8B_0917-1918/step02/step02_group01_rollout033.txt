    def objective(x):
        nonlocal h, dx
        h = x.copy()
        idx = np.where(h < 0)[0]
        h[idx] = 0
        idx = np.where(h > 1)[0]
        h[idx] = 1
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    constraint_func = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)

        population_size = 30
        generations = 50
        mutation_rate = 0.1
        best_fitness = float('inf')
        best_solution = None

        def create_individual():
            return np.random.uniform(0, 1, n_points)

        def crossover(parent1, parent2):
            child = np.copy(parent1)
            for i in range(n_points):
                if np.random.rand() < 0.5:
                    child[i] = parent2[i]
            return child

        def mutate(individual):
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    individual[i] = np.clip(np.random.uniform(0, 1), 0, 1)
            return individual

        def fitness(individual):
            return objective(individual)

        population = [create_individual() for _ in range(population_size)]
        for _ in range(generations):
            fitnesses = [fitness(ind) for ind in population]
            best_idx = np.argmin(fitnesses)
            best_fitness = fitnesses[best_idx]
            best_solution = population[best_idx]
            new_population = []
            for _ in range(population_size // 2):
                parent1 = np.random.choice(population)
                parent2 = np.random.choice(population)
                child = crossover(parent1, parent2)
                child = mutate(child)
                new_population.append(child)
            population = new_population
        return best_solution

    result = minimize(objective, h, bounds=bounds, constraints=constraint_func, method='SLSQP', options={"ftol": 1e-8, "maxiter": 150})
    h_values = result.x
    c5_bound = objective(h_values)

    genetic_sol = genetic_optimizer()
    final_h = np.clip(genetic_sol, 0, 1)
