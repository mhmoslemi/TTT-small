import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 128
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    initial_h[:n_points//2] = 1
    initial_h = initial_h + np.sin(np.pi * np.arange(n_points) / n_points) / 4
    initial_h = np.clip(initial_h, 0, 1)
    
    def objective(x):
        h_new = x.copy()
        h_new[h_new < 0] = 0
        h_new[h_new > 1] = 1
        correl = np.correlate(h_new, 1 - h_new, mode="full")
        c5 = np.max(correl) * dx
        return c5
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = Bounds([0]*n_points, [1]*n_points)
    cons = LinearConstraint(np.ones(n_points), [n_points/2], [n_points/2])
    
    start_time = time.time()
    result = opt.minimize(objective, initial_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10})
    
    best_h = result.x
    best_h[best_h < 0] = 0
    best_h[best_h > 1] = 1
    best_c5 = objective(best_h)
    
    return best_h, best_c5, len(best_h)
