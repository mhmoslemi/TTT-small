    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h, dtype=np.float64)

    def objective(x):
        nonlocal h, dx
        h = x.copy()
        idx = np.where(h < 0)[0]
        h[idx] = 0
        idx = np.where(h > 1)[0]
        h[idx] = 1
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)

    def perturb(x, scale=0.05):
        return np.clip(x + np.random.normal(0, scale, x.shape), 0, 1)

    def fitness(x):
        return objective(x)

    def neighborhood(x):
        return [perturb(x, scale=0.05), perturb(x, scale=0.1), perturb(x, scale=0.2)]

    def random_start():
        return np.clip(np.random.rand(n_points), 0, 1) * (n_points / 2) / n_points

    def evaluate(candidate):
        return fitness(candidate)

    def simulate_optimality():
        best = random_start()
        best_val = evaluate(best)
        for i in range(1000):
            neighbors = neighborhood(best)
            vals = [evaluate(x) for x in neighbors]
            best = neighbors[np.argmin(vals)]
            best_val = vals[np.argmin(vals)]
        return best, best_val

    def hybrid_optimize():
        best = random_start()
        best_val = evaluate(best)
        for i in range(100):
            neighbors = neighborhood(best)
            vals = [evaluate(x) for x in neighbors]
            idx = np.argmin(vals)
            best = neighbors[idx]
            best_val = vals[idx]
            if i % 10 == 0:
                print(f"Iter {i}: Best C5 = {best_val}")
        return best, best_val

    def bayesian_optimize():
        kernel = RBF(length_scale=1.0, length_scale_bounds=(1e-2, 1e2))
        gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=10)
        X = np.array([random_start() for _ in range(50)])
        y = np.array([evaluate(x) for x in X])
        gp.fit(X, y)
        X_next = np.array([random_start() for _ in range(10)])
        X_next = np.vstack((X, X_next))
        y_next = np.array([evaluate(x) for x in X_next])
        y = np.concatenate((y, y_next))
        return X_next[np.argmin(y)], y[np.argmin(y)]

    best, best_val = simulate_optimality()
    best, best_val = hybrid_optimize()
    best, best_val = bayesian_optimize()

    h_values = best
    c5_bound = best_val
    return h_values, c5_bound, len(h_values)
