import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 0.5 if i % 4 == 0 else 0.3 if i % 4 == 1 else 0.7 if i % 4 == 2 else 0.9
    h = np.array(initial_h, dtype=np.float64)
    
    def objective(x):
        h_new = x.copy()
        h_new[h_new < 0] = 0
        h_new[h_new > 1] = 1
        correl = np.correlate(h_new, 1 - h_new, mode="full")
        c5 = np.max(correl) * dx
        return c5
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    x0 = h.copy()
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    start_time = time.time()
    result = opt.minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
    
    best_h = result.x
    best_h[best_h < 0] = 0
    best_h[best_h > 1] = 1
    best_c5 = objective(best_h)
    
    return best_h, best_c5, len(best_h)
