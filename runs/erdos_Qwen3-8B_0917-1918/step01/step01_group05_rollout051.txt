```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import linear_sum_assignment
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points//4] = 1
    h[n_points//4:n_points//2] = 0.5
    h[n_points//2:3*n_points//4] = 0.25
    h[3*n_points//4:] = 0.125
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(x):
        h_new = x.copy()
        h_new[h_new < 0] = 0
        h_new[h_new > 1] = 1
        correl = np.correlate(h_new, 1 - h_new, mode="full")
        c5 = np.max(correl) * dx
        return c5
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    x0 = h.copy()
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    start_time = time*time.time()
    result = opt.minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 100})
    
    best_h = result.x
    best_h[best_h < 0] = 0
    best_h[best_h > 1] = 1
    best_c5 = objective(best_h)
    
    return best_h, best_c5, len(best_h)
```
