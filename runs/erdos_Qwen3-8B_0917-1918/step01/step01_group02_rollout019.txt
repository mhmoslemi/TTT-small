import numpy as np
import random
from scipy.optimize import Bounds, minimize
from functools import partial

initial_h_values = np.array([0.5] * 42)

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    population_size = 50
    generations = 100
    mutation_rate = 0.2
    crossover_rate = 0.7
    best_fitness = float('inf')
    best_h = None

    def fitness(h):
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    def create_individual():
        return np.random.rand(n_points) * 2 - 1

    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        for i in range(n_points):
            if random.random() < crossover_rate:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child

    def mutate(individual):
        for i in range(n_points):
            if random.random() < mutation_rate:
                individual[i] = np.random.rand() * 2 - 1
        return individual

    population = [create_individual() for _ in range(population_size)]
    for gen in range(generations):
        fitnesses = [fitness(ind) for ind in population]
        best_index = np.argmax(fitnesses)
        best_fitness = fitnesses[best_index]
        best_h = population[best_index]
        if best_fitness < 0.380876:
            return best_h, best_fitness, n_points
        new_population = []
        for _ in range(population_size // 2):
            parent1 = random.choice(population)
            parent2 = random.choice(population)
            child1 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = crossover(parent1, parent2)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        population = new_population
    return best_h, best_fitness, n_points
