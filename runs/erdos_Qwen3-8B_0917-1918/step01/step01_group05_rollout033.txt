```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h / np.sum(h) * n_points / 2

    def objective(x):
        h_new = x.copy()
        h_new[h_new < 0] = 0
        h_new[h_new > 1] = 1
        correl = np.correlate(h_new, 1 - h_new, mode="full")
        c5 = np.max(correl) * dx
        return c5

    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), -np.inf, n_points / 2)

    start_time = time.time()
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 1000})
    
    best_h = result.x
    best_h[best_h < 0] = 0
    best_h[best_h > 1] = 1
    best_c5 = objective(best_h)
    
    return best_h, best_c5, len(best_h)
```
