    def evaluate(h):
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    def mutate(individual, mutation_rate=0.1):
        new_individual = individual.copy()
        for i in range(len(new_individual)):
            if random.random() < mutation_rate:
                new_individual[i] = random.uniform(0, 1)
        return new_individual

    def crossover(parent1, parent2):
        size = len(parent1)
        half_size = size // 2
        child = np.concatenate([parent1[:half_size], parent2[half_size:]])
        return child

    population_size = 50
    population = [np.random.rand(n_points) for _ in range(population_size)]
    best_individual = population[0]
    best_score = evaluate(best_individual)

    for gen in range(100):
        if gen > 0:
            population = [mutate(ind) for ind in population]
        fitness = [evaluate(ind) for ind in population]
        population = [np.array(p) for p, f in zip(population, fitness) if f < best_score]
