import numpy as np
import random
from scipy.optimize import Bounds, minimize
from functools import partial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        initial_h_values[i] = 0.5 + 0.5 * np.sin(np.pi * i / n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    
    def evaluate(h):
        h = np.clip(h, 0, 1)
        dx = 2.0 / n_points
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    def mutate(individual, mutation_rate=0.1):
        new_individual = individual.copy()
        for i in range(len(new_individual)):
            if random.random() < mutation_rate:
                new_individual[i] = np.random.uniform(0, 1)
        return new_individual
    
    def crossover(parent1, parent2):
        n = len(parent1)
        half = n // 2
        child = np.concatenate([parent1[:half], parent2[half:]])
        return child
    
    population_size = 50
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [h * (n_points / 2) / np.sum(h) for h in population]
    best_solution = population[0]
    best_c5 = evaluate(best_solution)
    
    for generation in range(100):
        fitness = [evaluate(ind) for ind in population]
        best_idx = np.argmin(fitness)
        best_solution = population[best_idx]
        best_c5 = fitness[best_idx]
        
        if generation % 10 == 0:
            print(f"Generation {generation}, Best C5: {best_c5}")
        
        if best_c5 < 0.380876:
            break
        
        new_population = []
        for _ in range(population_size // 2):
            parent1 = random.choice(population)
            parent2 = random.choice(population)
            child1 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            new_population.append(child1)
        
        population = new_population + [best_solution]
    
    return best_solution, best_c5, n_points
