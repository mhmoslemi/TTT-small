import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from functools import partial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points)
    h = np.sort(h)
    h = np.interp(np.linspace(0, 2, n_points), np.sort(h), np.linspace(0, 1, n_points))
    h = np.clip(h, 0, 1)
    h = h - np.min(h)
    h = h / np.max(h)
    h = np.concatenate((h, np.flipud(h[1:])))
    h = h[:n_points]
    h = (h - np.min(h)) / (np.max(h) - np.min(h))
    h = np.clip(h, 0, 1)
    h = h - np.min(h)
    h = h / np.max(h)
    h = np.concatenate((h, np.flipud(h[1:])))
    h = h[:n_points]
    h = (h - np.min(h)) / (np.max(h) - np.min(h))
    h = np.clip(h, 0, 1)
    h = h - np.min(h)
    h = h / np.max(h)
    h = np.concatenate((h, np.flipud(h[1:])))
    h = h[:n_points]
    h = (h - np.min(h)) / (np.max(h) - np.min(h))
    h = np.clip(h, 0, 1)
    h = h - np.min(h)
    h = h / np.max(h)

    def objective(x):
        nonlocal h, dx
        h = x.copy()
        idx = np.where(h < 0)[0]
        h[idx] = 0
        idx = np.where(h > 1)[0]
        h[idx] = 1
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    result = minimize(objective, h, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method='SLSQP', options={"ftol": 1e-8, "maxiter": 100})
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, len(h_values)
