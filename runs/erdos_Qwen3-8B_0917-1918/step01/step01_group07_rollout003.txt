```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            initial_h[i] = 1
        else:
            initial_h[i] = 0
    
    def compute_C5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def objective(x):
        return compute_C5(x)
    
    def constraints(x):
        return np.sum(x) - n_points / 2
    
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints={'type': 'eq', 'fun': constraints},
        options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    best_h = result.x
    best_C5 = compute_C5(best_h)
    
    return best_h, best_C5, n_points
```
