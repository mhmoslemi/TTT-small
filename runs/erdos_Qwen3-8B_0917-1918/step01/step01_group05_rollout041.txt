import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h, dtype=np.float64)
    
    def objective(x):
        h_new = x.copy()
        h_new[h_new < 0] = 0
        h_new[h_new > 1] = 1
        correl = np.correlate(h_new, 1 - h_new, mode="full")
        c5 = np.max(correl) * dx
        return c5
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    x0 = h.copy()
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = x0.copy()
    best_c5 = objective(x0)
    
    start_time = time.time()
    
    for _ in range(20):
        if time.time() - start_time > budget_s - 10:
            break
        x0 = np.random.uniform(0, 1, n_points)
        x0 = np.clip(x0, 0, 1)
        
        result = opt.minimize(
            objective, 
            x0, 
            method='L-BFGS-B', 
            bounds=bounds, 
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        current_h = result.x
        current_h[current_h < 0] = 0
        current_h[current_h > 1] = 1
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return best_h, best_c5, len(best_h)
