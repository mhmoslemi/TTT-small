import numpy as np
import scipy.optimize as optimize
import random
from deap import base, creator, algorithms, tools

def run(seed=42, budget_s=600):
    random.seed(seed)
    np.random.seed(seed)
    
    # Initial construction
    initial_h = initial_h_values
    n_points = len(initial_h)
    
    # Define evolutionary algorithm parameters
    POP_SIZE = 50
    CXPB = 0.9
    MUT_PB = 0.1
    MAX_GEN = 100
    
    # Create the fitness function
    def compute_c5(h):
        n = len(h)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n)
        return c5
    
    # Create the toolbox
    creator.create("FitnessMin", float, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    toolbox = base.Toolbox()
    toolbox.register("attr_float", lambda: np.random.uniform(0, 1, n_points))
    toolbox.register("individual", tools.initIterate, creator.Individual, toolbox.attr_float)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    
    def eval_func(individual):
        h = np.array(individual)
        # Enforce constraint sum(h) == n_points / 2
        h = h / np.sum(h) * (n_points / 2)
        c5 = compute_c5(h)
        return (c5,)
    
    toolbox.register("evaluate", eval_func)
    toolbox.register("select", tools.selTournament, tournsize=3)
    toolbox.register("mate", tools.cxTwoPoint)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
    
    # Create population
    population = toolbox.population(n=POP_SIZE)
    
    # Evaluate the entire population
    fitnesses = list(map(toolbox.evaluate, population))
    for ind, fit in zip(population, fitnesses):
        ind.fitness = fit
    
    # Begin the evolution
    for gen in range(MAX_GEN):
        offspring = algorithms.varAnd(population, toolbox, CXPB, MUT_PB)
        fits = list(map(toolbox.evaluate, offspring))
        for fit, ind in zip(fits, offspring):
            ind.fitness = fit
        population = tools.selBest(population + offspring, POP_SIZE)
        if gen % 10 == 0:
            print(f"Generation {gen}, Best C5: {min([ind.fitness for ind in population])}")
    
    best_ind = min(population, key=lambda x: x.fitness)
    best_h = best_ind
    best_c5 = best_ind.fitness
    return best_h, best_c5, n_points
