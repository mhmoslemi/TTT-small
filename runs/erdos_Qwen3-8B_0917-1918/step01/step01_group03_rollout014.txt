import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    initial = np.load("initial_h_values.npy")
    h = initial.copy()
    
    def objective(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        scaling = (n_points / 2) / total
        x_scaled = x * scaling
        conv = np.correlate(x_scaled, 1 - x_scaled, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    def constraint(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        scaling = (n_points / 2) / total
        return total - n_points / 2
    
    x0 = h.copy()
    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
    result = opt.minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
