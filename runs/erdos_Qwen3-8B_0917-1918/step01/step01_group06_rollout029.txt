import numpy as np
import scipy.optimize as optimize
import random

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 1 if i < n_points / 2 else 0
    initial_pop = [initial_h + np.random.uniform(-0.1, 0.1, n_points) for _ in range(10)]
    
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        n = len(h)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def fitness(h):
        return compute_c5(h)
    
    def mutate(individual, mutation_rate=0.1):
        h = np.copy(individual)
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] = np.clip(h[i] + np.random.uniform(-0.1, 0.1), 0, 1)
        return h
    
    def crossover(parent1, parent2):
        h = np.copy(parent1)
        for i in range(len(h)):
            if random.random() < 0.5:
                h[i] = parent2[i]
        return h
    
    best_h = initial_pop[0]
    best_c5 = fitness(best_h)
    
    for gen in range(100):
        population = []
        for _ in range(10):
            parent1 = random.choice(initial_pop)
            parent2 = random.choice(initial_pop)
            child = crossover(parent1, parent2)
            child = mutate(child)
            population.append(child)
        for i in range(len(population)):
            population[i] = np.clip(population[i], 0, 1)
            if np.sum(population[i]) * dx != 1:
                population[i] = np.array(initial_h, dtype=np.float64)
        fitness_values = [fitness(h) for h in population]
        best_index = np.argmin(fitness_values)
        best_h = population[best_index]
        best_c5 = fitness_values[best_index]
        if best_c5 < 0.380876:
            break
    
    return best_h.tolist(), best_c5, n_points
