import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    initial_h = np.load('initial_h_values.npy')
    
    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        conv = np.correlate(x, 1 - x, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    def constraint(x):
        x = np.clip(x, 0, 1)
        return np.sum(x) - n_points / 2
    
    def optimize_h(initial_guess):
        bounds = Bounds(0, 1)
        cons = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
        result = opt.minimize(objective, initial_guess, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
        return result.x, objective(result.x)
    
    h_values, c5_bound = optimize_h(initial_h)
    return h_values, c5_bound, n_points
