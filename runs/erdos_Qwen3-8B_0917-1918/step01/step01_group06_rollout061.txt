```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 256
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 1 if i % 5 == 0 else 0.5
    initial_h = np.repeat(initial_h, 2)
    initial_h = initial_h[:n_points]
    h = np.array(initial_h, dtype=np.float64)
    
    def compute_c5(h):
        n = len(h)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n)
        return c5
    
    def objective(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = optimize.minimize(
        objective,
        h,
        method='SLSQP',
        constraints=cons,
        bounds=[(0, 1) for _ in range(n_points)],
        options={'ftol': 1e-8, 'maxiter': 200}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    return best_h.tolist(), best_c5, n_points
```
