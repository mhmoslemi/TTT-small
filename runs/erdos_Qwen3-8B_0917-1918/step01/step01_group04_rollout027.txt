import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0, 1)
    
    def local_search(h0):
        result = minimize(objective, h0, method='L-BFGS-B', bounds=bounds, 
                         constraints={'type': 'eq', 'fun': constraint}, 
                         options={'ftol': 1e-8, 'maxiter': 100})
        return result.x
    
    h0 = initial_h_values.copy()
    h0 = local_search(h0)
    
    h_values = h0
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
