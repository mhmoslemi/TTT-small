        idx = np.where(h < 0)[0]

        return np.sum(x) - n_points / 2

        final_h = result.x

        final_h = h

        nonlocal final_h, dx
        final_h = x.copy()
        idx = np.where(final_h < 0)[0]
        final_h[idx] = 0
        idx = np.where(final_h > 1)[0]
        final_h[idx] = 1
        overlap = np.correlate(final_h, 1 - final_h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    result = differential_evolution(eval_func, bounds=[(0, 1) for _ in range(n_points)], strategy='best1bin', popsize=20, maxiter=100, tol=1e-8, mutation=(0.5, 1), recombination=0.9, disp=False)
