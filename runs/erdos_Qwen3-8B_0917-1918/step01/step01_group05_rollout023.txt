```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    h = np.random.rand(n_points)
    h = np.minimum(1, np.maximum(0, (h - 0.5) * 2))
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(x):
        x = np.clip(x, 0, 1)
        correl = np.correlate(x, 1 - x, mode="full")
        c5 = np.max(correl) * dx
        return c5
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = Bounds(0, 1)
    cons = {'type': 'eq', 'fun': constraint}
    
    start_time = time.time()
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 1000})
    
    best_h = result.x
    best_h[best_h < 0] = 0
    best_h[best_h > 1] = 1
    best_c5 = objective(best_h)
    
    return best_h, best_c5, len(best_h)
```
