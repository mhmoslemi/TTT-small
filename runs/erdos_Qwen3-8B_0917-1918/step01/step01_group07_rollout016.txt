```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    initial_h = np.zeros(n_points)
    initial_h[:n_points//2] = 1.0
    h = np.array(initial_h, dtype=float)
    
    def compute_C5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * (2.0 / n_points))
    
    def objective(x):
        h = x.copy()
        return compute_C5(h)
    
    def constraints(x):
        return np.sum(x) - n_points / 2
    
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints={'type': 'eq', 'fun': constraints},
        options={'maxiter': 1000, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_C5 = compute_C5(best_h)
    
    return best_h, best_C5, n_points
```
