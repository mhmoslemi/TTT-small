```python
import numpy as np
import scipy.optimize as optimize
import copy

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 512
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    h[n_points//2:] = 0.0
    h = np.roll(h, n_points//4)
    
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        n = len(h)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def objective(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = optimize.minimize(
        objective,
        h,
        method='SLSQP',
        constraints=cons,
        bounds=[(0, 1) for _ in range(n_points)],
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    return best_h.tolist(), best_c5, n_points
```
