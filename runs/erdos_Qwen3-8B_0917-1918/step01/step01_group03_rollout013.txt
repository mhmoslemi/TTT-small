import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 1 if i < n_points // 2 else 0
    initial_h = np.concatenate([initial_h, initial_h[::-1]])
    initial_h = initial_h[:n_points]
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        conv = np.correlate(x, 1 - x, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    def constraint(x):
        x = np.clip(x, 0, 1)
        return np.sum(x) - n_points / 2
    
    x0 = initial_h.copy()
    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
    result = opt.minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 1000})
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
