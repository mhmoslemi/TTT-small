```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 128
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 1.0 if i < n_points // 2 else 0.0
    initial_h = np.array(initial_h, dtype=np.float64)
    
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5
    
    def objective(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = optimize.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=Bounds(0, 1),
        constraints=cons,
        options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 200}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    return best_h.tolist(), best_c5, n_points
```
