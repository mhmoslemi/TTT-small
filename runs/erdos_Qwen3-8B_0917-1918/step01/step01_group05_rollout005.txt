```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    h_initial = np.zeros(n_points)
    h_initial[:n_points//4] = 1
    h_initial[n_points//2 - n_points//8 : n_points//2 + n_points//8] = 1
    h_initial[-n_points//4:] = 1
    h_initial = np.clip(h_initial, 0, 1)
    
    def objective(x):
        h_new = x.copy()
        h_new = np.clip(h_new, 0, 1)
        correl = np.correlate(h_new, 1 - h_new, mode="full")
        c5 = np.max(correl) * dx
        return c5
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = Bounds([0]*n_points, [1]*n_points)
    cons = {'type': 'eq', 'fun': constraint}
    
    start_time = time.time()
    result = opt.minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 1000})
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_c5 = objective(best_h)
    
    return best_h, best_c5, len(best_h)
```
