```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points//3] = 1.0
    h[n_points//3:n_points//1.5] = 0.5
    h[n_points//1.5:] = 0.0
    h = np.concatenate((h, h[::-1]))[:n_points]
    h = h / np.sum(h) * n_points / 2
    
    def compute_C5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def objective(x):
        return compute_C5(x)
    
    def constraints(x):
        return np.sum(x) - n_points / 2
    
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints={'type': 'eq', 'fun': constraints},
        options={'ftol': 1e-8, 'maxiter': 500}
    )
    
    best_h = result.x
    best_C5 = compute_C5(best_h)
    
    return best_h, best_C5, n_points
```
