import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    initial_h = np.load('initial_h_values.npy')
    h = np.array(initial_h[:n_points], dtype=np.float64)
    sum_h = np.sum(h)
    if not np.isclose(sum_h, n_points / 2):
        h = np.array([0.5 if i % 2 == 0 else 0.5 for i in range(n_points)])
    def objective(x):
        h_vals = x
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        return np.max(corr) * dx
    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        return np.max(corr) * dx
    def get_bounds(n_points):
        return [(0.0, 1.0) for _ in range(n_points)]
    def optimize_with_bounds(n_points, initial_guess):
        bounds = get_bounds(n_points)
        return minimize(objective, initial_guess, bounds=bounds)
    start_time = time.time()
    best_result = None
    for _ in range(50):
        if time.time() - start_time > budget_s - 10:
            break
        initial_guess = np.random.uniform(0, 1, n_points)
        result = optimize_with_bounds(n_points, initial_guess)
        if result.success:
            if best_result is None or result.fun < best_result.fun:
                best_result = result
    best_h = best_result.x
    best_c5 = best_result.fun
    return best_h, best_c5, n_points
