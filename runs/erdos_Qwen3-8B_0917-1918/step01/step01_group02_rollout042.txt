```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from functools import partial
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.random.rand(n_points)
    initial_h = np.minimum(1, np.maximum(0, initial_h * n_points / 2))
    
    def objective(h):
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0, 1)
    
    result = differential_evolution(
        objective,
        bounds,
        constraints={"type": "eq", "fun": constraint},
        strategy='best1bin',
        popsize=20,
        maxiter=50,
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-8,
        polish=False,
        vectorized=True,
        seed=seed
    )
    
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, len(h_values)
```
