        h = np.array(h, dtype=np.float64)
        n = len(h)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def objective(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 150}
    )
    
    best_h = result.x
    best_c_h = compute_c5(best_h)
    
    for _ in range(10):
        perturb = np.random.normal(0, 0.05, n_points)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = new_h / np.sum(new_h) * n_points / 2
        current_c5 = compute_c5(new_h)
        if current_c5 < best_c_h:
            best_h = new_h
            best_c_h = current_c5
    
    return best_h.tolist(), best_c_h, n_points
