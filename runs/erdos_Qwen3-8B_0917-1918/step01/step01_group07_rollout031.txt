import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 1 / (1 + np.exp(-10 * (i - n_points / 2)))
    h = np.array(initial_h, dtype=float)
    
    def compute_C5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def objective(x):
        return compute_C5(x)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'ftol': 1e-8, 'maxiter': 1000}
    )
    
    best_h = result.x
    best_C5 = compute_C5(best_h)
    
    return best_h, best_C5, n_points
