import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    
    def initial_guess():
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h[i] = 0.5
            else:
                h[i] = 0.5 + 0.5 * np.sin(np.pi * (i - n_points // 2) / (n_points // 2))
        return h
    
    h = initial_guess()
    sum_h = np.sum(h)
    if sum_h != n_points / 2:
        h = h * (n_points / 2) / sum_h
    
    def compute_C5(h_values):
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def objective(x):
        return compute_C5(x)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints={'type': 'eq', 'fun': constraint},
        options={'ftol': 1e-8, 'maxiter': 200, 'disp': False}
    )
    
    best_h = result.x
    best_C5 = compute_C5(best_h)
    
    return best_h, best_C5, n_points
