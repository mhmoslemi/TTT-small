import numpy as np

import random

def run(seed=42, budget_s=600):
    random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def fitness(h):
        return compute_c5(h)
    
    def mutate(h, mutation_rate=0.05):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] = np.clip(random.random(), 0, 1)
        return h
    
    def crossover(parent1, parent2):
        h = np.zeros(len(parent1))
        for i in range(len(h)):
            if random.random() < 0.5:
                h[i] = parent1[i]
            else:
                h[i] = parent2[i]
        return h
    
    population_size = 50
    population = [np.array(initial_h, dtype=np.float64) for _ in range(population_size)]
    best_fitness = float('inf')
    best_h = np.array(initial_h, dtype=np.float64)
    time_used = 0
    
    while time_used < budget_s:
        fitnesses = [fitness(h) for h in population]
        population = [np.array(h) for h in population]
        indices = np.argsort(fitnesses)
        top_parents = [population[i] for i in indices[:10]]
        
        new_population = []
        for _ in range(population_size // 2):
            parent1 = random.choice(top_parents)
            parent2 = random.choice(top_parents)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
        current_best = min(fitnesses)
        if current_best < best_fitness:
            best_fitness = current_best
            best_h = population[fitnesses.index(current_best)]

        if time_used > budget_s:
