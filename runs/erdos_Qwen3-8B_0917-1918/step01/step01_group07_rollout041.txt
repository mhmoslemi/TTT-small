import time

    def compute_C5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def objective(x):
        return compute_C5(x)
    
    def constraints(x):
        return np.sum(x) - n_points / 2
    
    best_h = h.copy()

    elapsed = 0
    
    while elapsed < budget_s:

        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',

        )
        h = result.x
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:

            best_h = h.copy()
        
        result = opt.minimize(
            objective,
            h,
            method='SLSQP',
