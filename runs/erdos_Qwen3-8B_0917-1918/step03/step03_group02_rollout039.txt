```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from functools import partial
from scipy.optimize import LinearConstraint
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.linspace(0, 2, n_points)
    h = np.sin(np.pi * h / 2) / 2 + 0.5
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)

    def objective(x):
        h_clipped = np.clip(x, 0, 1)
        overlap = np.correlate(h_clipped, 1 - h_clipped, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    constraint_func = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
    
    def run_optimization():
        result = minimize(objective, h, bounds=bounds, constraints=constraint_func, method='SLSQP', options={"ftol": 1e-8, "maxiter": 150})
        best_h = result.x
        best_c5 = objective(best_h)
        return best_h, best_c5

    best_h, best_c5 = run_optimization()

    def perturb(x, mutation_rate=0.1):
        new_x = x.copy()
        for i in range(len(new_x)):
            if random.random() < mutation_rate:
                new_x[i] = np.clip(new_x[i] + random.uniform(-0.1, 0.1), 0, 1)
        return new_x

    for _ in range(10):
        perturbed = perturb(best_h)
        result = minimize(objective, perturbed, bounds=bounds, constraints=constraint_func, method='SLSQP', options={"ftol": 1e-8, "maxiter": 150})
        current_h = result.x
        current_c5 = objective(current_h)
        if current_c5 < best_c5:
            best_h, best_c5 = current_h, current_c5

    return best_h, best_c5, len(best_h)
```
