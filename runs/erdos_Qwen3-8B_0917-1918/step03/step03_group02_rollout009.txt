```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from functools import partial
from scipy.optimize import LinearConstraint
import random
from scipy.spatial.distance import cosine

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    h = np.clip(h, 0, 1)
    total = np.sum(h) * dx
    h = np.clip(h * (1 / total) * 2, 0, 1)

    def objective(x):
        nonlocal h, dx
        h = x.copy()
        idx = np.where(h < 0)[0]
        h[idx] = 0
        idx = np.where(h > 1)[0]
        h[idx] = 1
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    constraint_func = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)

    def local_search(h_init):
        result = minimize(objective, h_init, bounds=bounds, constraints=constraint_func, method='L-BFGS-B', options={"ftol": 1e-8, "maxiter": 150})
        return result.x

    def perturb(x, mutation_rate=0.1):
        new_x = x.copy()
        for i in range(len(new_x)):
            if random.random() < mutation_rate:
                new_x[i] = np.clip(new_x[i] + random.uniform(-0.1, 0.1), 0, 1)
        return new_x

    best_h = h
    best_c5 = objective(h)

    for _ in range(20):
        perturbed = perturb(best_h)
        local_h = local_search(perturbed)
        current_c5 = objective(local_h)
        if current_c5 < best_c5:
            best_h, best_c5 = local_h, current_c5

    return best_h, best_c5, len(best_h)
```
