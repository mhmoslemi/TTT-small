import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from functools import partial
from sklearn.cluster import KMeans

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.full(n_points, 0.5)
    for i in range(n_points):
        h[i] = np.random.rand()
    h = np.sort(h)
    
    def objective(x):
        nonlocal h, dx
        h = x.copy()
        idx = np.where(h < 0)[0]
        h[idx] = 0
        idx = np.where(h > 1)[0]
        h[idx] = 1
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)

    best_h = h.copy()
    best_c5 = objective(h)

    for _ in range(10):
        h = np.random.permutation(h)
        result = minimize(objective, h, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method='SLSQP', options={"ftol": 1e-9, "maxiter": 200})
        h = result.x
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    h_values = best_h
    c5_bound = best_c5
    return h_values, c5_bound, len(h_values)
