import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from functools import partial
from sklearn.cluster import KMeans

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.linspace(0, 2, n_points)
    h = np.sin(np.pi * h / 2) / 2 + 0.5
    h = np.clip(h, 0, 1)
    h = np.random.permutation(h)

    def objective(x):
        nonlocal h, dx
        h = x.copy()
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)

    result = minimize(objective, h, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method='SLSQP', options={"ftol": 1e-9, "maxiter": 100})
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, len(h_values)
