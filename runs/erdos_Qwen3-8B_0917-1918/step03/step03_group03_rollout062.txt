```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 300
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 1 if i % 5 == 0 else 0.5 if i % 2 == 0 else 0.25
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        conv = np.correlate(x, 1 - x, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    def constraint(x):
        x = np.clip(x, 0, 1)
        return np.sum(x) - n_points / 2
    
    def optimize_local(start):
        x0 = start.copy()
        bounds = Bounds(0, 1)
        cons = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
        result = opt.minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        return result.x
    
    best_h = optimize_local(initial_h)
    for _ in range(10):
        perturb = np.random.normal(0, 0.05, n_points)
        h = best_h + perturb
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        current_h = optimize_local(h)
        if objective(current_h) < objective(best_h):
            best_h = current_h
    
    h_values = best_h
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
```
