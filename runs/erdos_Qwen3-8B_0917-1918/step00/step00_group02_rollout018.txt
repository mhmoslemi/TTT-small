```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.random.uniform(0, 1, n_points)
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    
    def objective(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def eval_erdos(h):
        return objective(h)
    
    def optimize_h(h0):
        result = minimize(eval_erdos, h0, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        return result.x
    
    best_h = initial_h
    best_c5 = objective(best_h)
    
    for _ in range(5):
        h = optimize_h(best_h)
        c5 = objective(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    return best_h, best_c5, n_points
```
