```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    h = np.append(h, np.zeros(n_points//2))
    h = np.random.permutation(h)
    
    def objective(x):
        h_val = x.reshape(n_points)
        h_val = np.clip(h_val, 0, 1)
        conv = np.correlate(h_val, 1 - h_val, mode="full")
        c5 = np.max(conv * dx)
        return c5
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = Bounds(0, 1)
    x0 = h.copy()
    
    start_time = time.time()
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'ftol': 1e-8, 'maxiter': 100})
    
    if time.time() - start_time > budget_s:
        return (h, 0.381, n_points)
    
    best_h = result.x
    best_c5 = result.fun
    return (best_h, best_c5, n_points)
```
