```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    best_c5 = np.inf
    best_h = h.copy()
    
    def objective(h):
        nonlocal best_c5, best_h
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0, 1)
    
    for _ in range(10):
        start_time = time.time()
        res = minimize(objective, h, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}], method='L-BFGS-B', options={'maxiter': 50})
        if time.time() - start_time > budget_s / 2:
            break
    
    return best_h, best_c5, len(best_h)
```
