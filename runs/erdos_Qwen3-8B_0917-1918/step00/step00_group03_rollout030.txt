import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    h[n_points//2:] = 0.0
    best_c5 = float('inf')
    best_h = h.copy()
    
    for _ in range(50):
        h = best_h.copy()
        h += np.random.uniform(-0.1, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h[-1] = 0.0
        h[0] = 0.0
        total = np.sum(h) * dx
        h = h / (total / dx) if total != 0 else h
        
        cor = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cor) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5, n_points
