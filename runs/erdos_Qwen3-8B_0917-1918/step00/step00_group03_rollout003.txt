```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    # Initial guess: spread out with peaks
    for i in range(n_points):
        h[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(x):
        h_current = np.zeros(n_points)
        for i in range(n_points):
            h_current[i] = x[i]
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
        corr = np.correlate(h_current, 1 - h_current, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    for _ in range(100):
        start_time = time.time()
        result = opt.minimize(objective, h, method='L-BFGS-B', bounds=(0, 1), options={'ftol': 1e-8, 'max_iter': 100})
        elapsed = time.time() - start_time
        if elapsed > budget_s / 2:
            break
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    return best_h, best_c5, n_points
```
