```python
import numpy as np
import scipy.optimize as opt
import copy
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * n_points / 2.0
    total_time = 0

    def compute_c5(h_values):
        h = np.array(h_values)
        h = np.concatenate((h, h))
        corr = np.correlate(h, 1 - h, mode="full")
        corr = corr[n_points // 2 : n_points // 2 + n_points]
        c5 = np.max(corr) * dx
        return c5

    def objective(x):
        nonlocal total_time
        start_time = time.time()
        h = x.copy()
        h = np.clip(h, 0, 1)
        c5 = compute_c5(h)
        total_time += time.time() - start_time
        return c5

    for _ in range(10):
        if total_time >= budget_s:
            break
        initial = np.random.uniform(0, 1, n_points) * n_points / 2.0
        result = opt.minimize(objective, initial, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        best_h = result.x
        best_c5 = compute_c5(best_h)
        print(f"Best C5: {best_c5}")
        h = best_h

    return h, best_c5, n_points
```
