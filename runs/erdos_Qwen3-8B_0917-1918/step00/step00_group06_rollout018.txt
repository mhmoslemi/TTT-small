import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 0.5 if i % 10 < 5 else 0.5
    initial_h = initial_h + np.random.normal(0, 0.1, n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h - np.mean(initial_h) + 0.5
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    
    bounds = Bounds(0, 1)
    constraint = LinearConstraint(np.ones(n_points), 1, 1)
    
    def objective(x):
        h = x
        correlations = np.correlate(h, 1 - h, mode="full")
        return np.max(correlations) * dx
    
    result = opt.minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, constraints=constraint, options={'ftol': 1e-8, 'maxiter': 100})
    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
