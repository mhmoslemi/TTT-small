import numpy as np
import scipy.optimize as opt
from scipy.optimize import minimize
from scipy.integrate import simps
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5
    h = np.minimum(h, 1.0)
    h = np.maximum(h, 0.0)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(x):
        h_new = np.zeros(n_points)
        for i in range(n_points):
            h_new[i] = x[i]
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        overlap = np.correlate(h_new, 1 - h_new, mode="full")
        return np.max(overlap * dx)
    
    result = minimize(objective, h, method='L-BFGS-B', options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8})
    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
