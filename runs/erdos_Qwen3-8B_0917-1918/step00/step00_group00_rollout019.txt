```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 2 - 1
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(x):
        nonlocal n_points, dx
        h = x.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    x0 = h
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint})
    
    result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-10, options={'maxiter': 100})
    h_best = result.x
    c5_best = result.fun
    return h_best, c5_best, n_points
```
