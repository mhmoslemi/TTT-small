import numpy as np
import scipy.optimize as opt
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 0.5 if i % 5 == 0 else 0.3
    initial_h = initial_h + np.random.uniform(-0.1, 0.1, n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h / np.mean(initial_h) * (n_points / 2) / dx
    initial_h = np.clip(initial_h, 0, 1)

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.mean(h) * (n_points / 2) / dx
        h = np.clip(h, 0, 1)
        c5 = 0
        for k in range(1, int(2 / dx) + 1):
            overlap = np.sum(h[:n_points - k] * (1 - h[k:])) * dx
            if overlap > c5:
                c5 = overlap
        return c5

    result = minimize(objective, initial_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, options={'maxiter': 100})
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.mean(best_h) * (n_points / 2) / dx
    best_h = np.clip(best_h, 0, 1)
    c5 = objective(best_h)
    return best_h, c5, n_points
