import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
from scipy.optimize import Bounds, minimize
import time

def run(seed=42, budget_s=600):
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    
    # Initial guess from initial_h_values
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values[:n_points])
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h / (dx * n_points) * (n_points / 2)
        initial_h = np.clip(initial_h, 0, 1)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / (dx * n_points) * (n_points / 2)
        h = np.clip(h, 0, 1)
        corr = signal.correlate(h, 1 - h, mode="full")
        c5_val = np.max(corr) * dx
        return c5_val
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0, 1)
    result = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, tol=1e-8, options={'maxiter': 1000})
    
    h_values = np.clip(result.x, 0, 1)
    h_values = h_values / (dx * n_points) * (n_points / 2)
    h_values = np.clip(h_values, 0, 1)
    
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
