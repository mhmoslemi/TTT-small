```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(np.abs(corr)) * dx
        return c5
    
    def objective(h):
        return evaluate(h)
    
    def initial_guess():
        n = 80
        h = np.zeros(n)
        for i in range(n):
            if i < n // 2:
                h[i] = 0.5
            else:
                h[i] = 0.5
        return h
    
    def perturb(h, delta=0.05):
        n = len(h)
        h_perturbed = h.copy()
        for i in range(n):
            if np.random.rand() < 0.1:
                h_perturbed[i] = np.clip(h[i] + np.random.normal(0, delta), 0, 1)
        return h_perturbed
    
    best_h = initial_guess()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        h = perturb(best_h)
        h = np.clip(h, 0, 1)
        current_c5 = evaluate(h)
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5
    
    return best_h, best_c5, len(best_h)
```
