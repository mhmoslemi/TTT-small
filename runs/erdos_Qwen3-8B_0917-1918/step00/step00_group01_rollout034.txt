```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * n_points / 2 / dx
    best_c5 = float('inf')
    
    for _ in range(100):
        start_time = time.time()
        if time.time() - start_time > budget_s:
            break
        h = np.random.uniform(0, 1, n_points)
        h = h * n_points / 2 / dx
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    return best_h, best_c5, n_points
```
