```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)

    def objective(h):
        h = h.reshape(-1)
        max_overlap = 0
        for k in range(1, n_points):
            corr = np.correlate(h, 1 - h, mode="full")
            max_overlap = max(max_overlap, np.max(corr) * dx)
        return max_overlap

    def callback(x):
        nonlocal best_c5
        current_c5 = objective(x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = x.copy()

    best_c5 = float('inf')
    best_h = h.copy()

    for _ in range(100):
        start_time = time.time()
        result = opt.minimize(objective, h, method='L-BFGS-B', bounds=(0, 1), callback=callback)
        elapsed = time.time() - start_time
        if elapsed > budget_s / 2:
            break

    return best_h, best_c5, n_points
```
