    initial_h = initial_h * n_points / 2
    initial_h = np.round(initial_h).astype(float)

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * n_points / 2
        h = np.round(h).astype(float)
        correlation = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlation * dx)
        return c5

    def constraint(h):
        return np.sum(h) - n_points / 2

    h0 = initial_h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(objective, h0, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="L-BFGS-B")

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * n_points / 2
    best_h = np.round(best_h).astype(float)

    best_c5 = objective(best_h)
    return best_h.tolist(), best_c5, n_points
