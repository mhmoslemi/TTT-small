```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = np.random.uniform(0, 1)
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    
    def objective(h):
        h = np.array(h, dtype=np.float64)
        h = np.concatenate([h, h[:n_points//2]])
        h = h[:n_points*2]
        corr = np.correlate(h, 1 - h[:n_points], mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def update_h(h):
        h = np.array(h, dtype=np.float64)
        h = np.concatenate([h, h[:n_points//2]])
        h = h[:n_points*2]
        return h

    def transform_h(h):
        h = np.array(h, dtype=np.float64)
        h = h[:n_points]
        return h

    def evaluate_h(h):
        h = np.array(h, dtype=np.float64)
        h = h[:n_points]
        return h

    def cost(h):
        h = np.array(h, dtype=np.float64)
        h = h[:n_points]
        if np.sum(h) * dx != 1:
            return float('inf')
        return objective(h)

    result = minimize(
        cost,
        x0=initial_h,
        method='L-BFGS-B',
        bounds=Bounds(0, 1),
        options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-8}
    )

    best_h = result.x
    best_c5 = result.fun

    return best_h, best_c5, n_points
```
