import numpy as np
import random
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def initial_h_values(n_points):
    return np.array([0.5]*n_points)

def evaluate_erdos_solution(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    initial_h = initial_h_values(n_points)
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    h_flat = np.random.uniform(0, 1, n_points).flatten()
    result = minimize(objective, h_flat, method='L-BFGS-B', bounds=[(0,1)]*n_points, constraints=[{'type': 'eq', 'fun': constraint_sum}])
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return best_h, best_c5, n_points
