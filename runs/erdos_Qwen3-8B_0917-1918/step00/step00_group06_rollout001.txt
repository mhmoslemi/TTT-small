```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    h = np.random.rand(n_points)
    h = h / np.sum(h) * (n_points / 2)
    
    def compute_c5(h):
        h = np.concatenate([h, h])
        corr = np.correlate(h, 1 - h[:n_points], mode='full')
        corr = corr[n_points - 1:-n_points + 1]
        return np.max(corr) * dx
    
    def objective(x):
        return compute_c5(x)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    best_h = result.x
    c5 = compute_c5(best_h)
    
    return best_h, c5, n_points
```
