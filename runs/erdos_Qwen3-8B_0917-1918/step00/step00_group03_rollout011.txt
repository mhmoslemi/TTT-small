import numpy as np
import scipy.optimize as opt
from scipy.optimize import linprog
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.full(n_points, 0.5)
    for i in range(n_points):
        h[i] = 0.5 + np.random.rand() * 0.5 - 0.25
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def eval_c5(h_vals):
        h = np.array(h_vals)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def objective(x):
        return eval_c5(x)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    def bounds(x):
        return np.array([0.0, 1.0])
    
    lb = np.zeros(n_points)
    ub = np.ones(n_points)
    x0 = h
    
    result = opt.minimize(objective, x0, bounds=zip(lb, ub), method='L-BFGS-B', options={'ftol': 1e-8, 'maxiter': 1000})
    best_h = result.x
    best_c5 = eval_c5(best_h)
    return best_h, best_c5, n_points
