```python
import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    h[n_points//2:] = 0.0
    h = h + np.random.normal(0, 0.1, n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)

    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    result = optimize.minimize(objective, h, bounds=bounds, method='L-BFGS-B', options={'disp': False, 'maxiter': 100})
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx
    return best_h, c5, n_points
```
