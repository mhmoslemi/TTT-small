import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    def compute_C5(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)

    def objective(h):
        return compute_C5(h)

    def constraints(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]

    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraints},
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    best_h = result.x
    best_C5 = compute_C5(best_h)
    return best_h, best_C5, n_points
