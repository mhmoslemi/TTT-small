```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * n_points / 2.0 / dx
    h = h / np.sum(h) * n_points / 2.0 / dx

    for _ in range(10):
        start_time = time.time()
        if time.time() - start_time > budget_s:
            break
        def objective(x):
            h = np.array(x)
            h = h / np.sum(h) * n_points / 2.0 / dx
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr * dx)
            return c5

        bounds = Bounds(0, 1)
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
        h = result.x

    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr * dx)
    return h, c5, n_points
```
