import numpy as np
import scipy.optimize as optimize
import copy

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h)
    
    def objective(h):
        h = np.array(h)
        correlations = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlations) * dx
        return c5
    
    def constraints(h):
        return np.sum(h) - n_points / 2
    
    def optimize_h(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraints}]
        )
        return result.x
    
    for _ in range(5):
        h_opt = optimize_h(h)
        h = np.clip(h_opt, 0, 1)
    
    c5 = objective(h)
    return h.tolist(), c5, n_points
