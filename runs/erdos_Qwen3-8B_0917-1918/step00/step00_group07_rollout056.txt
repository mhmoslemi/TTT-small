```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 2 - 1
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def optimize_h(h_initial):
        result = optimize.minimize(objective, h_initial, method='L-BFGS-B', options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6})
        return result
    
    start_time = time.time()
    best_h = h
    best_c5 = objective(h)
    
    for _ in range(3):
        result = optimize_h(best_h)
        best_h = result.x
        best_c5 = objective(best_h)
        if time.time() - start_time > budget_s - 10:
            break
    
    return best_h, best_c5, n_points
```
