import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * n_points / 2 / dx
    h = h / np.sum(h)
    h = np.clip(h, 0, 1)
    
    def compute_c5(h):
        h = np.concatenate([h, h[:n_points//2]])
        c5 = np.correlate(h, 1 - h[:n_points], mode="full")
        c5 = np.max(c5 * dx)
        return c5
    
    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * n_points / 2 / dx
        x = x / np.sum(x)
        return compute_c5(x)
    
    def constraint(x):
        x = np.clip(x, 0, 1)
        x = x * n_points / 2 / dx
        x = x / np.sum(x)
        return np.sum(x) - n_points / 2
    
    bounds = Bounds(0, 1)
    
    result = opt.minimize(objective, h, bounds=bounds, method='L-BFGS-B', tol=1e-8, options={'ftol': 1e-8, 'gtol': 1e-8})
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * n_points / 2 / dx
    best_h = best_h / np.sum(best_h)
    
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
