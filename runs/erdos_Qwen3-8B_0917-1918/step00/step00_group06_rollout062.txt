```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 1000
    h = np.random.rand(n_points)
    h = h / np.sum(h) * n_points / 2
    best_c5 = np.inf
    time_start = time.time()
    
    def obj_func(h):
        nonlocal best_c5
        h = np.clip(h, 0, 1)
        h = np.concatenate((h, h))[:n_points]
        dx = 2.0 / n_points
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        if c5 < best_c5:
            best_c5 = c5
            return c5
        return c5
    
    for _ in range(100):
        if time.time() - time_start > budget_s - 10:
            break
        h = np.random.rand(n_points)
        h = h / np.sum(h) * n_points / 2
        res = opt.minimize(lambda x: obj_func(x), h, method='L-BFGS-B', bounds=[(0, 1)] * n_points)
        h = res.x
    
    return (np.clip(h, 0, 1), best_c5, n_points)
```
