```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        initial_h_values[i] = 1.0 if i % 5 == 0 else 0.0
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    
    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def objective(h):
        return evaluate(h)
    
    def bounds(h):
        return [(0, 1) for _ in h]
    
    result = minimize(objective, x0=initial_h_values, method='L-BFGS-B', bounds=bounds, options={'ftol': 1e-8, 'maxiter': 100})
    
    best_h = result.x
    best_c5 = evaluate(best_h)
    return best_h, best_c5, n_points
```
