import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    
    def compute_c5(h):
        h = np.array(h)
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    def optimize_step(h, max_iter=100):
        nonlocal best_c5, best_h
        for _ in range(max_iter):
            start_time = time.time()
            if time.time() - start_time > budget_s:
                break
            h = h.copy()
            for i in range(n_points):
                if h[i] < 1:
                    h[i] += np.random.uniform(-0.05, 0.05)
                if h[i] > 1:
                    h[i] = 1
                if h[i] < 0:
                    h[i] = 0
            h_sum = np.sum(h)
            if h_sum != n_points / 2:
                h = h / (h_sum / (n_points / 2))
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    optimize_step(h)
    return best_h, best_c5, n_points
