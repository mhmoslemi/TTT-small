import numpy as np
import scipy.optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h, dtype=np.float64)
    
    def objective(x):
        h_values = np.array(x, dtype=np.float64)
        h_values = np.clip(h_values, 0, 1)
        shift = np.roll(h_values, 1)
        overlap = np.sum(h_values * (1 - shift)) * dx
        return overlap
    
    def evaluate_c5(h_values):
        h = np.array(h_values, dtype=np.float64)
        overlaps = []
        for k in range(1, len(h) - 1):
            shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted)) * dx
            overlaps.append(overlap)
        return max(overlaps)
    
    def fitness(x):
        return evaluate_c5(x)
    
    def simple_search():
        h_start = initial_h.copy()
        h_start = np.clip(h_start, 0, 1)
        result = minimize(fitness, h_start, bounds=[(0, 1) for _ in range(n_points)], method='L-BFGS-B', options={'maxiter': 500})
        return result.x
    
    def random_search():
        h_start = np.random.rand(n_points) * 2 - 1
        h_start = np.clip(h_start, 0, 1)
        result = minimize(fitness, h_start, bounds=[(0, 1) for _ in range(n_points)], method='L-BFGS-B', options={'maxiter': 500})
        return result.x
    
    def objective_with_bounds(x):
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(fitness, x, bounds=bounds, method='L-BFGS-B', options={'maxiter': 200})
        return result.fun
    
    def optimize():
        t0 = time.time()
        h_best = initial_h.copy()
        best_c5 = evaluate_c5(h_best)
        for _ in range(5):
            h_candidate = np.random.rand(n_points) * 2 - 1
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate[0] = 0
            h_candidate[1] = 0
            h_candidate[-1] = 0
            h_candidate[-2] = 0
            h_candidate = (h_candidate - h_candidate.mean()) * 0.5 + 0.5
            result = minimize(fitness, h_candidate, bounds=[(0, 1) for _ in range(n_points)], method='L-BFGS-B', options={'maxiter': 300})
            current_c5 = result.fun
            if current_c5 < best_c5:
                h_best = result.x
                best_c5 = current_c5
            if time.time() - t0 > budget_s - 10:
                break
        return h_best, best_c5
    
    best_h, best_c5 = optimize()
    return best_h, best_c5, len(best_h)
