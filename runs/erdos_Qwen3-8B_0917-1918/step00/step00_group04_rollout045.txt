```python
import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as opt
from scipy.optimize import minimize
from scipy.interpolate import interp1d
from collections import deque

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    def evaluate_erdos_solution(h_vals):
        n = len(h_vals)
        dx = 2.0 / n
        h = np.array(h_vals)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def objective_function(x):
        return evaluate_erdos_solution(x)
    
    def constraint_sum(x):
        return np.sum(x) - (len(x) / 2)
    
    def bounds_func(x):
        return np.all(0 <= x) and np.all(x <= 1)
    
    def initial_h_values_func(n):
        return np.array([0.5 for _ in range(n)])
    
    n_points = 256
    initial_guess = initial_h_values_func(n_points)
    
    result = minimize(objective_function, initial_guess, bounds=[(0, 1) for _ in range(n_points)], constraints=[{'type': 'eq', 'fun': constraint_sum}])
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, best_c5, n_points
```
