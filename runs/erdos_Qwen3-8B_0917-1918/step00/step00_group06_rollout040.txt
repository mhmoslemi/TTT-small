```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 2 - 1
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    best_c5 = float('inf')
    best_h = h.copy()
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    for _ in range(100):
        t0 = time.time()
        result = minimize(lambda x: objective(x), h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        if time.time() - t0 > budget_s / 2:
            break
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()
    
    return best_h, best_c5, n_points
```
