```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * n_points / 2 / dx
    h = np.clip(h, 0, n_points)

    def compute_C5(h):
        h = h / (n_points / 2)
        h = np.concatenate([h, h])
        corr = np.convolve(h, 1 - h, mode='full')
        corr = corr[n_points:]
        corr = corr * dx
        return np.max(corr)

    def objective(x):
        return compute_C5(x)

    def constraints(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(objective, h, bounds=bounds, constraints={"type": "eq", "fun": constraints}, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    h_values = result.x
    c5_bound = compute_C5(h_values)
    return h_values, c5_bound, n_points
```
