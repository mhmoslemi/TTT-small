```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points)
    h = h / np.sum(h) * (n_points / 2)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = np.concatenate([h, h])
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def fitness(h):
        return objective(h)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    result = minimize(fitness, h, method='L-BFGS-B', constraints=[{'type': 'eq', 'fun': constraint}])
    h_opt = result.x
    c5_bound = objective(h_opt)
    return h_opt, c5_bound, n_points
```
