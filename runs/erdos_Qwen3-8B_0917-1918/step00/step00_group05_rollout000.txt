import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = initial_h_values[:n_points]
    h = np.array(initial_h, dtype=np.float64)
    h = h / np.sum(h) * n_points / 2

    def objective_loss(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def objective_loss_with_grad(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        grad = np.zeros_like(h)
        for i in range(n_points):
            for k in range(n_points):
                if i + k < n_points:
                    corr_val = corr[i + k]
                    grad[i] += corr_val * (1 - h[k]) * dx
                    grad[k] -= corr_val * h[i] * dx
        return c5, grad.flatten()

    def objective(h):
        return objective_loss(h)

    def objective_with_grad(h):
        return objective_loss_with_grad(h)

    def evaluate(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    best_c5 = float('inf')
    best_h = h.copy()

    start_time = time.time()

    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        result = optimize.minimize(objective, h, method='L-BFGS-B', jac=lambda x: objective_with_grad(x)[1])
        if result.success:
            h = result.x
            current_c5 = evaluate(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            break

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    return best_h, best_c5, n_points
