import numpy as np
from scipy.optimize import minimize
from scipy.optimize.linesearch import line_search_wolfeproj
import time

def run(seed=42, budget_s=600):
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 0.5 if i < n_points // 2 else 1.0
    initial_h = np.array(initial_h, dtype=np.float64)
    
    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    h0 = initial_h
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(
        objective,
        h0,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'disp': False, 'maxiter': 1000}
    )
    
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
