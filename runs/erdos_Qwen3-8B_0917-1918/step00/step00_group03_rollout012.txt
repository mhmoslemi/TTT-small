import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h, dtype=np.float64)
    
    def objective(h):
        h = h.reshape(-1)
        overlaps = []
        for k in range(1, n_points):
            shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted)) * dx
            overlaps.append(overlap)
        return np.max(overlaps)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = opt.minimize(objective, h, bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, method='SLSQP', options={'maxiter': 100})
    
    h_opt = result.x
    c5_bound = objective(h_opt)
    
    return (h_opt.tolist(), c5_bound, n_points)
