```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 1 if i % 3 == 0 else 0.5 if i % 3 == 1 else 0
    h = h / (np.sum(h) / 2) * 2.0
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    while time.time() - start_time < budget_s:
        h = np.random.uniform(0, 1, n_points)
        h = h / (np.sum(h) / 2) * 2.0
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5, n_points
```
