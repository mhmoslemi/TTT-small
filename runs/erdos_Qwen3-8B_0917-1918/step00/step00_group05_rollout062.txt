        h_new = np.clip(x, 0, 1)

        c5 = np.max(overlap * dx)
        return c5
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    result = scipy.optimize.minimize(
        objective,
        h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'ftol': 1e-8, 'maxiter': 100}
    )
