```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points//2] = 1
    h = h / (n_points / 2)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    result = minimize(objective, x0=h, method='SLSQP', constraints=cons, bounds=bounds, options={'ftol': 1e-10, 'maxiter': 100})
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
```
