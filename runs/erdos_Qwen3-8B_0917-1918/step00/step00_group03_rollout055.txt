```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(x):
        h = np.copy(x)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    result = optimize.minimize(objective, h, method='L-BFGS-B', options={'maxiter': 1000, 'ftol': 1e-8})
    best_h = result.x
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx
    return best_h, c5, n_points
```
