```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * dx * 2
    h = h / np.sum(h) * (n_points / 2)

    def objective(x):
        h_val = x.reshape(n_points)
        corr = np.correlate(h_val, 1 - h_val, mode="full")
        c5 = np.max(corr) * dx
        return c5

    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(objective, h, bounds=bounds, method='L-BFGS-B', options={'maxiter': 1000, 'ftol': 1e-8})
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
```
