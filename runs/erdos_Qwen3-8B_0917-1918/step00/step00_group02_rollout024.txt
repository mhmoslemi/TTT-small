```python
import numpy as np
import scipy.optimize as optimize
import scipy.interpolate as interpolate
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Initial guess: alternating high and low values
    for i in range(n_points):
        if i % 3 == 0:
            h[i] = 1.0
        elif i % 3 == 1:
            h[i] = 0.5
        else:
            h[i] = 0.0
    
    # Ensure the integral constraint is satisfied
    total = np.sum(h) * dx
    scale = (1.0 / total) if total != 0 else 1.0
    h *= scale
    
    def compute_c5(h):
        h_func = interpolate.interp1d(np.linspace(0, 2, n_points), h, kind='linear', bounds_error=False)
        c5_values = []
        for k in range(1, 100):
            overlap = np.zeros_like(h)
            for i in range(n_points):
                x = i * dx
                if x + k <= 2:
                    overlap[i] = h_func(x) * (1.0 - h_func(x + k))
            c5_values.append(np.sum(overlap) * dx)
        return np.max(c5_values)
    
    def objective(x):
        h = x.reshape(n_points)
        total = np.sum(h) * dx
        if total == 0:
            return np.inf
        h = h / (total / 1.0)
        return compute_c5(h)
    
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n_points)
```
