import numpy as np
import scipy.optimize as optimize
import scipy.interpolate as interpolate
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * n_points / 2 / dx
    h = h / np.sum(h) * n_points / 2 / dx
    
    def objective(x):
        h_new = np.zeros(n_points)
        idx = np.argmin(np.abs(x - np.linspace(0, 2, n_points)), axis=1)
        for i in range(n_points):
            h_new[idx[i]] += x[i]
        h_new /= np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        conv = np.correlate(h_new, 1 - h_new, mode='full')
        c5 = np.max(conv) * dx
        return c5
    
    result = optimize.minimize(objective, h, method='L-BFGS-B', options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8})
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2 / dx
    
    conv = np.correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(conv) * dx
    return (best_h, c5_bound, n_points)
