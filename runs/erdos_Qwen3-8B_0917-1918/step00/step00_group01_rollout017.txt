import numpy as np
import scipy.optimize as opt
from scipy.interpolate import interp1d
from scipy.optimize import Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        initial_h_values[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        corr = corr[n_points - 1:]
        c5 = np.max(corr) * dx
        return c5
    
    def optimize_h():
        bounds = Bounds(0, 1)
        result = opt.minimize(objective, initial_h_values, method='L-BFGS-B', bounds=bounds, options={'ftol': 1e-10, 'gtol': 1e-10})
        return result.x
    
    h_values = optimize_h()
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
