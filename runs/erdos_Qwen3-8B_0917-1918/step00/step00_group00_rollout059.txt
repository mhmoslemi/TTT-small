import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n = len(initial_h)
    dx = 2.0 / n
    h = np.array(initial_h)
    best_c5 = np.inf
    best_h = h.copy()
    
    def objective(x):
        nonlocal best_c5
        h = x.reshape(n)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n)])
        h = result.x
        if result.success:
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5, n
