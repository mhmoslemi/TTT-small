import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = np.sin(2 * np.pi * initial_h[i] / 2) + 1
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(x):
        h_val = x.reshape(n_points)
        corr = np.correlate(h_val, 1 - h_val, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint(x):
        return np.sum(x) - n_points / 2

    x0 = h.copy()
    bounds = Bounds(0, 1)
    result = minimize(objective, x0, bounds=bounds, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
