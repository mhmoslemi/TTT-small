import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 1000
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points, endpoint=True)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def initial_guess():
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                h[i] = 1.0
        return h
    
    h0 = initial_guess()
    h0 = np.clip(h0, 0, 1)
    
    def bounds_func(x):
        return np.array([0.0, 1.0])
    
    bounds = np.array([[0.0, 1.0] for _ in range(n_points)])
    
    def constraint_func(x):
        return np.sum(x) - n_points / 2
    
    cons = LinearConstraint(bounds, -np.inf, np.inf)
    cons.fun = constraint_func
    
    result = optimize.minimize(objective, h0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
