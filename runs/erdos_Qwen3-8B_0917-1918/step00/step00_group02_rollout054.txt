```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    initial_h_values = np.random.uniform(0, 1, n_points)
    initial_h_values = initial_h_values * n_points / 2
    best_c5 = float('inf')
    best_h = initial_h_values

    def objective(h):
        nonlocal best_c5
        t0 = time.time()
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    for _ in range(100):
        h = best_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        if time.time() - t0 > budget_s:
            break

    return best_h, best_c5, n_points
```
