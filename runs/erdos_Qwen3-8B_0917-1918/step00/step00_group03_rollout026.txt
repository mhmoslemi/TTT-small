```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[::2] = 1.0
    h = np.array(h, dtype=np.float64)
    h = h / np.sum(h) * (n_points / 2)
    h = np.clip(h, 0, 1)
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    while time.time() - start_time < budget_s:
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5, n_points
```
