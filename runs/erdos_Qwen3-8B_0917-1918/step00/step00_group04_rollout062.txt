import time

def run(seed=42, budget_s=600):

    n_points = len(initial_h)
    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.array(h)
        h = np.concatenate((h, h[:n_points//2]))
        result = np.correlate(h, 1 - h[:n_points], mode="full")
        c5 = np.max(result) * dx
        return c5

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = Bounds(0, 1)
    h0 = initial_h.copy()

    start_time = time.time()
    result = minimize(objective, h0, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}], method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    elapsed = time.time() - start_time

    if elapsed > budget_s:
        print("Time exceeded budget")
        return initial_h, compute_c5(initial_h), n_points

    best_h = result.x
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
