import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 1 if i < n_points / 2 else 0
    initial_h = np.roll(initial_h, int(n_points / 4))
    initial_h = initial_h / np.sum(initial_h) * n_points / 2

    def objective(h):
        h = np.array(h, dtype=np.float64)
        h = np.roll(h, int(n_points / 4))
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    result = minimize(objective, initial_h, method='L-BFGS-B',
                      bounds=[(0, 1) for _ in range(n_points)],
                      constraints=[{'type': 'eq', 'fun': constraint_sum}])
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
