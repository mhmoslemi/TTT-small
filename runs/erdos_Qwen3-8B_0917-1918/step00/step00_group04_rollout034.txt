```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        h = np.concatenate([h, h])
        shifted = np.zeros_like(h)
        for k in range(1, 5):
            shifted[k:] = h[:-k]
            corr = np.correlate(h, 1 - h, mode="full")
            current_c5 = np.max(corr * dx)
            shifted[k:] = 0
        return current_c5

    def objective(x):
        return compute_c5(x)

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), 0, n_points / 2)

    result = opt.minimize(objective, h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    best_h = result.x
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
```
