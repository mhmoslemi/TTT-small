import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h, dtype=np.float64)
    
    def objective(x):
        nonlocal n_points, dx
        h = np.copy(x)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint})
    
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, len(h_values)
