import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * n_points / 2 / dx
    h = h / np.sum(h) * n_points / 2 / dx
    h = np.clip(h, 0, 1)
    best_c5 = np.inf
    best_h = h.copy()
    start_time = time.time()
    while time.time() - start_time < budget_s:
        h = np.random.uniform(0, 1, n_points)
        h = h * n_points / 2 / dx
        h = h / np.sum(h) * n_points / 2 / dx
        h = np.clip(h, 0, 1)
        correlate = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(np.abs(correlate)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5, n_points
