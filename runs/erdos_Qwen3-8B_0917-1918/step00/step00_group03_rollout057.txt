from scipy.optimize import Bounds
import time

    def evaluate(h):

        method='L-BFGS-B',
        bounds=bounds,
        tol=1e-8,

    )
    
    best_h = result.x
    best_c5 = evaluate(best_h)
    
    if time.time() - start_time < budget_s - 1:

        def optimize_with_bounds(h_init):
            result = minimize(
                minimize_objective,
                x0=h_init,
                method='L-BFGS-B',
                bounds=bounds,
                tol=1e-8,
                options={'ftol': 1e-9, 'gtol': 1e9}
            )
            return result.x
        
        for _ in range(5):
            new_h = optimize_with_bounds(best_h)
            new_c5 = evaluate(new_h)
            if new_c5 < best_c5:
