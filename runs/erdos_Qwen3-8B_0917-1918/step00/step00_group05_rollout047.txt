    dx = 2.0 / n_points
    h = np.array(initial_h, dtype=np.float64)
    
    def objective(h):
        h = h.reshape(-1)
        shifts = [0, 1, 2, 3, 4]
        overlaps = []
        for k in shifts:
            shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted)) * dx
            overlaps.append(overlap)
        return max(overlaps)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0, 1)
    
    for _ in range(10):
        start_time = time.time()
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}])
        if time.time() - start_time > budget_s / 2:
            break
        h = result.x
    
    final_h = h
    c5_bound = objective(final_h)
    return final_h.tolist(), c5_bound, len(final_h)
