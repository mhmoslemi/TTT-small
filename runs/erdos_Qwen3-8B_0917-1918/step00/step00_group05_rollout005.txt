import numpy as np
import scipy.optimize as opt
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    h_initial = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            h_initial[i] = 1.0
        elif i % 3 == 0:
            h_initial[i] = 0.5
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h_func = interp1d(x, h, kind='linear', bounds_error=False, fill_value=0)
        corr = np.correlate(h, 1 - h, mode="full")
        corr = corr * dx
        return np.max(corr)
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint_func})
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
