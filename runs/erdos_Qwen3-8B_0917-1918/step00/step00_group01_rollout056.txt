import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
from scipy.sparse import random as sparse_random
from scipy.sparse import csr_matrix
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)

    def objective(x):
        x = np.array(x, dtype=np.float64)
        x = x * (n_points / 2) / np.sum(x)
        corr = np.correlate(x, 1 - x, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def evaluate(x):
        x = np.array(x, dtype=np.float64)
        x = x * (n_points / 2) / np.sum(x)
        corr = np.correlate(x, 1 - x, mode="full")
        return np.max(corr) * dx

    result = minimize(evaluate, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)], options={'maxiter': 100})
    best_h = result.x
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    c5 = evaluate(best_h)
    return best_h, c5, n_points
