import scipy.optimize
import time

    initial_h = np.interp(np.linspace(0, 2, n_points), np.sort(np.unique(np.concatenate((initial_h, np.random.rand(n_points)))), axis=None), initial_h)

    best_c5 = float('inf')

    for _ in range(50):
        t0 = time.time()
        def objective(x):
            nonlocal best_c5
            h = np.zeros(n_points)
            h[:len(x)] = x
            h = np.interp(np.linspace(0, 2, n_points), np.sort(np.unique(np.concatenate((h, np.random.rand(n_points)))), axis=None), h)

            h = h * (n_points / 2) / np.sum(h)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
            return c5
        x0 = np.random.rand(n_points)
        result = scipy.optimize.minimize(objective, x0, method='L-BFGS-B', options={'maxiter': 100})

            break
