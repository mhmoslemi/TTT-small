import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h, dtype=np.float64)
    
    def objective(x):
        h = x.reshape(n_points)
        overlaps = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlaps) * dx
        return c5
    
    def constraint(x):
        h = x.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0, 1)
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints={"type": "eq", "fun": constraint})
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
