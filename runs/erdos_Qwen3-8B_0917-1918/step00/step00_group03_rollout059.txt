import numpy as np
import scipy.optimize as opt
import scipy.interpolate as interp
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points, endpoint=True)
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    h[n_points//2:] = 0.0
    h = h + np.random.normal(0, 0.05, n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    def constraints(h):
        return np.sum(h) - n_points / 2
    
    result = opt.minimize(objective, h, method='L-BFGS-B', constraints=[{'type': 'eq', 'fun': constraints}])
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
