```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    h[n_points//2:] = 0.0

    def objective(h):
        h = h.reshape(n_points)
        h = np.concatenate([h, h])
        c5 = 0
        for k in range(1, n_points):
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = max(c5, np.max(corr * dx))
        return c5

    def constrain(h):
        h = h.reshape(n_points)
        return np.sum(h) - n_points / 2

    bounds = Bounds(0, 1)
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constrain}, options={'ftol': 1e-8, 'maxiter': 100})
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
```
