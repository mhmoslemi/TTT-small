```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5
    h = np.minimum(h, 1)
    h = np.maximum(h, 0)
    h = h / np.sum(h) * (n_points / 2)
    
    def objective(x):
        h = x.reshape((n_points,))
        hh = np.correlate(h, 1 - h, mode="full")
        return np.max(hh * dx)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    x0 = h
    result = optimize.minimize(objective, x0, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="SLSQP", options={"ftol": 1e-8, "maxiter": 100})
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
```
