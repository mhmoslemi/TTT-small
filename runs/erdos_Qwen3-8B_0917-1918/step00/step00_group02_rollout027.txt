```python
import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5 * 2
    h = h / np.sum(h) * (n_points / 2)
    
    def compute_c5(h):
        h = np.concatenate((h, h[:n_points//2]))
        corr = np.correlate(h, 1 - h[:n_points], mode="full")
        return np.max(corr) * dx
    
    def objective(x):
        h = x.reshape(n_points)
        return compute_c5(h)
    
    def constraint(x):
        h = x.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    x0 = h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    result = optimize.minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 100})
    best_h = result.x
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
```
