import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    def optimize_h(h0):
        result = optimize.minimize(objective, h0, method='L-BFGS-B', bounds=[(0, 1)]*n_points, options={'maxiter': 100, 'ftol': 1e-8})
        return result.x
    
    best_h = initial_h
    best_c5 = objective(best_h)
    start_time = time.time()
    
    for _ in range(5):
        h = optimize_h(best_h)
        c5 = objective(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        if time.time() - start_time > budget_s - 10:
            break
    
    return best_h, best_c5, n_points
