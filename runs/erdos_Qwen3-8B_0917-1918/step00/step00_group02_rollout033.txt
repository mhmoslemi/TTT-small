```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds
from scipy.integrate import simps

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 0.5 if np.random.rand() < 0.5 else 0.0
    initial_h = initial_h / dx
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / dx
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    result = optimize.minimize(objective, initial_h, bounds=bounds, method='L-BFGS-B', options={'maxiter': 100})
    best_h = result.x
    c5_bound = objective(best_h)
    return (best_h, c5_bound, n_points)
```
