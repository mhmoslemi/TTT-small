```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(h):
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    bounds = Bounds(0, 1)
    
    result = opt.minimize(objective, h, bounds=bounds, method='L-BFGS-B', options={'maxiter': 200})
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
```
