```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * n_points / 2.0 / dx
    
    def cost_function(h):
        h = np.array(h, dtype=np.float64)
        h = np.concatenate((h, h[:n_points//2]))
        correlate = np.correlate(h, 1 - h[:n_points], mode='full')
        correlate = correlate[n_points:]
        c5 = np.max(correlate * dx)
        return c5
    
    def objective(h):
        return cost_function(h)
    
    result = opt.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, options={'maxiter': 100, 'disp': False})
    best_h = result.x
    best_c5 = cost_function(best_h)
    return best_h, best_c5, n_points
```
