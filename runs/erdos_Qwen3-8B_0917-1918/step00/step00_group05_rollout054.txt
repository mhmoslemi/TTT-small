import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint
from scipy.integrate import trapz
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(h):
        h_wrap = np.concatenate((h, h))
        corr = np.correlate(h_wrap, 1 - h_wrap[:n_points], mode='full')
        return np.max(corr * dx)
    
    def update_h(h):
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        return h
    
    h = update_h(h)
    
    def optimize():
        res = opt.minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        h = update_h(res.x)
        return h, objective(h)
    
    start_time = time.time()
    best_h, best_c5 = h, objective(h)
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        h_opt, c5 = optimize()
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    return best_h, best_c5, n_points
