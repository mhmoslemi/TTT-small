```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 1 if i % 3 == 0 else 0.5
    h = h / np.sum(h) * n_points / 2
    
    def compute_c5(h):
        h = np.concatenate([h, h])
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def objective(x):
        h = x.reshape(n_points)
        return compute_c5(h)
    
    def constraint(x):
        h = x.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    bounds = Bounds([0.0] * n_points, [1.0] * n_points)
    
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'ftol': 1e-10, 'maxiter': 100})
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
```
