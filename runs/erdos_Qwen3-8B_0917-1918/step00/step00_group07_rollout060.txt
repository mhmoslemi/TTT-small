        h = x.reshape(-1)
        integral = np.zeros_like(h)
        for k in range(1, n_points):
            integral += np.sum(h[:n_points - k] * (1 - h[k:]))

    def constraint(x):
        return np.sum(x) - (n_points / 2)
    
    bounds = [(0, 1) for _ in range(n_points)]
    x0 = h
    
    start_time = time.time()
    result = minimize(objective, x0, args=(dx,), bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="L-BFGS-B", options={"maxiter": 1000, "ftol": 1e-8})
    
    best_h = result.x
    best_c5 = objective(best_h, dx)
    
    if time.time() - start_time < budget_s:
        for _ in range(5):
            perturbation = np.random.normal(0, 0.1, n_points)
            perturbation = np.clip(perturbation, -0.5, 0.5)
            trial = np.clip(best_h + perturbation, 0, 1)
            trial = trial * (n_points / 2) / np.sum(trial)
            trial_c5 = objective(trial, dx)
            if trial_c5 < best_c5:
                best_h = trial
                best_c5 = trial_c5
