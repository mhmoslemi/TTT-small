```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[0:n_points//2] = 1.0
    h = h / (n_points / 2)
    
    def compute_c5(h_values):
        h = h_values
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def objective(x):
        h = np.copy(x)
        h = np.clip(h, 0, 1)
        return compute_c5(h)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    x0 = np.copy(h)
    cons = [{'type': 'eq', 'fun': constraint}]
    result = optimize.minimize(objective, x0, method='SLSQP', bounds=[(0, 1) for _ in range(n_points)], constraints=cons, options={'maxiter': 100, 'ftol': 1e-8})
    
    best_h = np.copy(result.x)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
```
