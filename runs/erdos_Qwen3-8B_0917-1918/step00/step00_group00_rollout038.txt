import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    def objective(h):
        h = np.array(h, dtype=np.float64)
        h = np.concatenate((h, h[:n_points//2]))
        corr = np.correlate(h, 1 - h[:n_points], mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h):
        h = np.array(h, dtype=np.float64)
        return np.sum(h) - n_points / 2
    
    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        h = np.concatenate((h, h[:n_points//2]))
        corr = np.correlate(h, 1 - h[:n_points], mode="full")
        return np.max(corr) * dx
    
    def optimize_solution(h0):
        res = optimize.minimize(evaluate, h0, method='L-BFGS-B', bounds=[(0, 1)]*n_points, options={'maxiter': 500})
        return res.x, res.fun
    
    start_time = time.time()
    best_h = initial_h
    best_c5 = evaluate(initial_h)
    
    try:
        res = optimize.minimize(objective, initial_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, options={'maxiter': 100})
        best_h = res.x
        best_c5 = res.fun
    except:
        pass
    
    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        res = optimize.minimize(objective, best_h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, options={'maxiter': 100})
        if res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
    
    return best_h, best_c5, n_points
