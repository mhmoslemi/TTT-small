```python
import numpy as np
import scipy.optimize as optimize
import matplotlib.pyplot as plt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(h):
        h = np.concatenate((h, h[:n_points//2]))
        corr = np.correlate(h, 1 - h[:n_points], mode="full")
        return np.max(corr) * dx
    
    result = optimize.minimize(objective, h, bounds=[(0, 1) for _ in range(n_points)], method='L-BFGS-B', options={'maxiter': 100})
    best_h = result.x
    best_h = np.concatenate((best_h, best_h[:n_points//2]))
    corr = np.correlate(best_h, 1 - best_h[:n_points], mode="full")
    c5 = np.max(corr) * dx
    return best_h[:n_points], c5, n_points
```
