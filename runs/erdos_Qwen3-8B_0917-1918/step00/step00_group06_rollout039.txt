import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        h_pad = np.concatenate([h, h])
        corr = np.correlate(h_pad, 1 - h, mode='full')
        corr = corr[n_points - 1: 2 * n_points - 1]
        return np.max(corr) * dx
    
    def obj_func(x):
        nonlocal n_points, dx
        h = x.copy()
        h = h * (n_points / 2) / np.sum(h)
        c5 = compute_c5(h)
        return c5
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = optimize.minimize(
        obj_func,
        x0=h,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    best_h = result.x
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n_points)
