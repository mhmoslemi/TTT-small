```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:int(n_points/2)] = 1.0
    h = h + np.random.normal(0, 0.1, n_points)
    h = np.clip(h, 0, 1)
    h = h * dx * n_points

    def objective(x):
        nonlocal n_points, dx
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1 - x, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def optimize_step(h):
        def f(x):
            return objective(x)
        x0 = h.copy()
        result = optimize.minimize(f, x0, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        return result.x

    start_time = time.time()
    best_h = h
    best_c5 = objective(h)
    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        h = optimize_step(best_h)
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
    return best_h, best_c5, n_points
```
