import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5
    h = h / dx
    h = h / np.sum(h)
    h = np.minimum(1.0, np.maximum(0.0, h))
    
    def evaluate(h):
        h = np.roll(h, 1)
        h = np.concatenate((h[1:], h[0]))
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def objective(x):
        x = x / dx
        x = np.minimum(1.0, np.maximum(0.0, x))
        return evaluate(x)
    
    def constraint(x):
        x = x / dx
        x = np.minimum(1.0, np.maximum(0.0, x))
        return np.sum(x) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    x0 = np.random.rand(n_points) * 0.5
    x0 = x0 / dx
    x0 = x0 / np.sum(x0)
    x0 = np.minimum(1.0, np.maximum(0.0, x0))
    
    result = opt.minimize(objective, x0, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}], method='SLSQP')
    h_values = result.x
    c5_bound = evaluate(h_values)
    return h_values, c5_bound, n_points
