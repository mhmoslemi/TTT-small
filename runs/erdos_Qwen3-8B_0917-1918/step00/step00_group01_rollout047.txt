import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h, dtype=np.float64)
    
    def compute_c5(h):
        h = np.concatenate([h, h[:n_points//2]])
        corr = np.correlate(h, 1 - h[:n_points], mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        h = np.copy(x)
        h = np.clip(h, 0, 1)
        return compute_c5(h)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    start_time = time.time()
    result = optimize.minimize(
        objective, h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-8
    )
    
    if time.time() - start_time > budget_s:
        return (initial_h, compute_c5(initial_h), n_points)
    
    best_h = np.clip(result.x, 0, 1)
    best_c5 = compute_c5(best_h)
    return (best_h.tolist(), best_c5, n_points)
