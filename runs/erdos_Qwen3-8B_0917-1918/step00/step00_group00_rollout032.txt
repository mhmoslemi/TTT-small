import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = initial_h.copy()
    
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def obj_func(x):
        nonlocal h
        h = x
        return compute_c5(h)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    try:
        result = optimize.minimize(
            obj_func,
            h,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        best_h = result.x
        c5 = compute_c5(best_h)
        return (best_h, c5, n_points)
    except:
        return (initial_h, compute_c5(initial_h), n_points)
