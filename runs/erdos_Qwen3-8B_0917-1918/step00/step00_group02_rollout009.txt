```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        h = np.concatenate([h, h])
        corr = np.correlate(h, 1 - h[:n_points], mode="full")
        corr = corr[n_points - 1:-n_points + 1]
        return np.max(corr * dx)
    
    def objective(x):
        h = x.reshape(n_points)
        return compute_c5(h)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 100})
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n_points)
```
