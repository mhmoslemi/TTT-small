import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as splinalg
from scipy.optimize import Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Initial construction with some spread
    h[0:n_points//3] = 1.0
    h[n_points//3:2*n_points//3] = 0.5
    h[2*n_points//3:] = 0.0
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        h = h.reshape(-1)
        shifted = np.zeros_like(h)
        for i in range(len(h)):
            shifted[i] = h[i]
        for i in range(len(h)):
            shifted[i + len(h)] = h[i]
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def objective(x):
        x = x.reshape(-1)
        return compute_c5(x)
    
    def constraints(x):
        x = x.reshape(-1)
        return np.sum(x) - n_points / 2
    
    bounds = Bounds(0, 1)
    
    result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
