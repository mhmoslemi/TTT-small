```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(h):
        h = np.concatenate((h, h))
        corr = np.correlate(h, 1 - h[:n_points], mode="full")
        return np.max(corr) * dx
    
    for _ in range(100):
        start_time = time.time()
        if time.time() - start_time > budget_s:
            break
        result = opt.minimize(objective, h, method='L-BFGS-B', bounds=(0, 1))
        h = result.x
        h = h * (n_points / 2) / np.sum(h)
    
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * 2.0 / n_points
    return h, c5, n_points
```
