import scipy.optimize as optimize
import time

            break
        def objective(x):
            h[x[0]] = x[1]
            return 1.0

        cons = [{'type': 'eq', 'fun': constraint1}]
        bounds = [(0, n_points-1), (0.0, 1.0)]
        result = optimize.minimize(objective, x0=[0, 0.5], bounds=bounds, constraints=cons, method='SLSQP')
        h[result.x[0]] = result.x[1]
    def compute_c5(h):
        h = np.array(h)
        c5 = 0.0
        for k in range(1, len(h)-1):
            overlap = np.sum(h * (1 - np.roll(h, k)))
            c5 = max(c5, overlap * dx)
        return c5
    c5_bound = compute_c5(h)
    return h.tolist(), c5_bound, n_points
