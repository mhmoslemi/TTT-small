```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(h):
        h = np.array(h, dtype=np.float64)
        h = h / np.sum(h) * n_points / 2
        cor = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cor) * dx
        return c5
    
    def bounds(n_points):
        return [(0, 1) for _ in range(n_points)]
    
    result = minimize(objective, h, bounds=bounds(n_points), method='L-BFGS-B', options={'ftol': 1e-10, 'maxiter': 100})
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
```
