```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    def initial_h():
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h[i] = 1.0
            else:
                h[i] = 0.0
        return h
    
    initial = initial_h()
    best_h = initial
    best_c5 = evaluate(best_h)
    
    def optimize():
        nonlocal best_h, best_c5
    
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
    
        def obj(x):
            return evaluate(x)
        
        def grad(x):
            return np.zeros_like(x)
        
        result = opt.minimize(obj, h, method='L-BFGS-B', jac=grad, bounds=Bounds([0, 1], [0, 1]), options={'maxiter': 100})
        best_h = result.x
        best_c5 = evaluate(best_h)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        optimize()
    
    return best_h, best_c5, n_points
```
