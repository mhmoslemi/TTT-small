```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / len(h))
        return c5
    
    def evaluate(h):
        return objective(h)
    
    def perturb(h, t=0.1):
        h = np.array(h, dtype=np.float64)
        h = np.clip(h + t * (np.random.rand(len(h)) - 0.5), 0, 1)
        return h
    
    def optimize():
        n_points = 100
        h = np.linspace(0, 2, n_points, endpoint=False)
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = np.clip(h, 0, 1)
        h = np.random.permutation(h)
        h = np.sort(h)
        h = np.clip(h, 0, 1)
        best_h = h
        best_c5 = evaluate(h)
        best_iter = 0
        
        for _ in range(100):
            h = perturb(best_h)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * n_points / 2
            c5 = evaluate(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
                best_iter += 1
            if best_iter > 50:
                break
        
        return best_h, best_c5, len(best_h)
    
    start_time = time.time()
    h, c5, n = optimize()
    elapsed = time.time() - start_time
    
    if elapsed > budget_s:
        return np.array([0.5] * 100), 0.38, 100
    
    return h, c5, n
```
