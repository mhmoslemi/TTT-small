import numpy as np
import scipy.optimize as opt
from scipy.optimize import LinearConstraint, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * n_points / 2 / dx
    h = h / np.sum(h) * n_points / 2 / dx
    
    def objective(x):
        h_val = x.reshape(n_points)
        h_val = np.clip(h_val, 0, 1)
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    x0 = h.copy()
    
    result = opt.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=Bounds(0, 1),
        constraints=[LinearConstraint(lambda x: np.sum(x), -np.inf, n_points / 2)]
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
