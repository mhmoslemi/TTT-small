```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds
from scipy.optimize import minimize
from functools import lru_cache
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    def objective(h_values):
        h = h_values.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        corr = corr[n_points - 1:]
        return np.max(corr) * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    def initial_h():
        if 'initial_h_values' in globals():
            return initial_h_values[:n_points]
        return np.random.rand(n_points) * 2 - 1
    
    h0 = initial_h()
    h0[h0 < 0] = 0
    h0[h0 > 1] = 1
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    res = minimize(objective, h0, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)], options={'ftol': 1e-8, 'maxiter': 100})
    
    best_h = res.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
```
