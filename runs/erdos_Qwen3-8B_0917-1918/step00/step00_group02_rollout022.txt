import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        h = np.concatenate([h, h[:n_points//2]])
        corr = np.correlate(h, 1 - h[:n_points], mode="full")
        return np.max(corr * dx)
    
    def objective(x):
        h = x.copy()
        h = h * (n_points / 2) / np.sum(h)
        return compute_c5(h)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    x0 = h
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    start_time = time.time()
    result = optimize.minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    h_values = result.x
    c5_bound = compute_c5(h_values)
    n_points = len(h_values)
    return h_values, c5_bound, n_points
