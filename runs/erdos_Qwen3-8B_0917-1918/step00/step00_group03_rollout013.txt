```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5
    h = h / np.sum(h) * (n_points / 2)
    best_c5 = float('inf')
    best_h = h.copy()
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h = np.clip(h, 0, 1)
        lags = np.arange(-n_points + 1, n_points)
        overlaps = []
        for k in lags:
            if k < 0:
                shifted = np.roll(h, -k)
            else:
                shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted)) * dx
            overlaps.append(overlap)
        return np.max(overlaps)
    
    for _ in range(100):
        t0 = time.time()
        result = optimize.minimize(
            objective,
            x0=best_h.flatten(),
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        if time.time() - t0 > budget_s / 2:
            break
        if result.success:
            h_flat = result.x
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            c5 = objective(h_flat)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5, n_points
```
