```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import minimize
from scipy.stats import norm
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.full(n_points, 0.5)
    
    def objective(params):
        h_values = params.reshape(n_points)
        h_values = np.clip(h_values, 0, 1)
        corrs = np.correlate(h_values, 1 - h_values, mode="full")
        c5 = np.max(corrs) * dx
        return c5
    
    def constraint(params):
        return np.sum(params) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}])
    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
```
