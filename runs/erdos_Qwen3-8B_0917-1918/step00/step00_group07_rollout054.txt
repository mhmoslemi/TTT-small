import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    # Initial approximation
    h[:n_points//2] = 1.0
    h[n_points//2:] = 0.0
    # Refine with piecewise constant updates
    for _ in range(10):
        c5_values = []
        # Compute all overlaps
        for k in range(1, n_points):
            overlap = np.zeros(n_points)
            for i in range(n_points):
                start = i - k
                end = i - k + 1
                if start < 0:
                    start = 0
                if end >= n_points:
                    end = n_points
                overlap[i] = np.sum(h[start:end] * (1 - h[i - k:i - k + 1]))
            c5_values.append(np.max(overlap) * dx)
        c5_values.append(np.max(h) * dx)
        max_c5 = np.max(c5_values)
        # Update h to minimize max_c5
        h[overlap > 0.5] = 0.5
        h[overlap <= 0.5] = 0.5
    # Final evaluation
    c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    return h, c5, n_points
