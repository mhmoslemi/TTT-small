```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * n_points / 2.0 / dx
    
    def compute_C5(h):
        h = np.concatenate([h, h[:n_points//2]])
        corr = np.correlate(h, 1 - h[:n_points], mode="full")
        return np.max(corr) * dx
    
    def objective(x):
        nonlocal n_points, dx
        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h * n_points / 2.0 / dx
        return compute_C5(h)
    
    def constraint(x):
        return np.sum(x) - n_points / 2.0
    
    x0 = np.ones(n_points) * (n_points / 2.0) / n_points
    cons = {'type': 'eq', 'fun': constraint}
    result = opt.minimize(objective, x0, constraints=cons, method='SLSQP', options={'ftol': 1e-6})
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * n_points / 2.0 / dx
    c5 = compute_C5(best_h)
    
    return best_h, c5, n_points
```
