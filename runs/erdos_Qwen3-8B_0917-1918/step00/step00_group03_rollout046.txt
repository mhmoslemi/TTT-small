        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def evaluate(h):
        return objective(h)
    
    def grad_objective(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        mask = np.zeros_like(corr)
        mask[np.abs(corr - max_corr) < 1e-8] = 1
        grad = np.zeros_like(h)
        for i in range(len(h)):
            for j in range(i, len(h)):
                if i + j < len(corr):
                    grad[i] += mask[i + j] * (1 - h[j])
                if i - j >= 0:
                    grad[i] += mask[i - j] * h[j]
        return grad * dx
    
    def cost_func(x):
        return evaluate(x)
    
    def grad_func(x):
        return grad_objective(x)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = optimize.minimize(
        cost_func,
        initial_h,
        bounds=bounds,
        method='L-BFGS-B',
        jac=grad_func,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_c5 = evaluate(best_h)
    
    return (best_h, best_c5, n_points)
