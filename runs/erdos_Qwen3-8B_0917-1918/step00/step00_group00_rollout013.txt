```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 1.0 if i % 10 < 5 else 0.0
    initial_h = initial_h / (np.sum(initial_h) / n_points) * (n_points / 2)

    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    def objective(x):
        return compute_c5(x)

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    x0 = initial_h

    start_time = time.time()
    result = minimize(objective, x0, bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (initial_h, compute_c5(initial_h), n_points)

    best_h = result['x']
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
```
