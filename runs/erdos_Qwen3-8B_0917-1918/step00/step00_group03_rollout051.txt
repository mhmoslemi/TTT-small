```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(h):
        h = np.reshape(h, (n_points,))
        h = np.concatenate((h, h[:n_points//2]))
        corr = np.correlate(h, 1 - h[:n_points], mode='full')
        return np.max(corr) * dx
    
    bounds = Bounds(0, 1)
    
    start_time = time.time()
    result = opt.minimize(objective, h, bounds=bounds, method='L-BFGS-B', options={'ftol': 1e-6, 'maxiter': 100})
    elapsed = time.time() - start_time
    
    if elapsed > budget_s:
        return (h, 0.381, n_points)
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
```
