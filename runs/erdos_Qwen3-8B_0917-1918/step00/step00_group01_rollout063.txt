import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def objective(h):
        return compute_c5(h)
    
    def bounds(h):
        return [(0.0, 1.0) for _ in h]
    
    def constraints(h):
        return [(np.sum(h) - n_points / 2)]
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        np.random.shuffle(h)
        h = np.sort(h)
        h = np.clip(h, 0, 1)
        np.sum(h) = n_points / 2
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    return best_h, best_c5, n_points
