import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        h = np.concatenate([h, h])
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def optimize_h(h, budget):
        n = len(h)
        x = cp.Variable(n)
        obj = cp.Minimize(cp.max(cp.sum(cp.multiply(x, 1 - x[1:]))) * dx)
        constraints = [
            cp.sum(x) == n / 2,
            x >= 0,
            x <= 1
        ]
        prob = cp.Problem(obj, constraints)
        start = time.time()
        prob.solve(solver=cp.CVXPY_MI, max_time=budget - (time.time() - start))
        if prob.status == 'optimal':
            return x.value
        return h
    
    best_h = h
    best_c5 = compute_c5(h)
    
    for _ in range(5):
        h = optimize_h(h, budget_s / 5)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
    
    return best_h, best_c5, n_points
