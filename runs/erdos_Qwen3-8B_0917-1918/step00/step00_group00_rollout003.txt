import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(h):
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def evaluate(h):
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def update_h(h, step):
        h_new = h + step * np.random.randn(h.size)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        h_new = update_h(best_h, 0.01)
        c5_new = evaluate(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    return best_h, best_c5, n_points
