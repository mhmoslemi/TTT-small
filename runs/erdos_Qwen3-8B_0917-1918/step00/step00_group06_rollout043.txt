import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = initial_h.copy()
    
    def compute_c5(h):
        h = np.array(h)
        conv = np.convolve(h, 1 - h, mode='full')
        c5 = np.max(conv) * dx
        return c5
    
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        return compute_c5(h)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 1000, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
