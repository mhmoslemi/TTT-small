    def objective(h):

        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def bounds_func(x):
        return [(0, 1) for _ in x]

    h_initial = initial_h.copy()
    result = minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds_func,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 1000, 'ftol': 1e-8}
    )

    best_c5 = objective(best_h)
