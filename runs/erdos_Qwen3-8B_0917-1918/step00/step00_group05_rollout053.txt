```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5
    h = np.minimum(h, 1.0)
    h = np.maximum(h, 0.0)
    h = h * (n_points / 2) / (np.sum(h) if np.sum(h) != 0 else 1)
    
    def objective(x):
        h_vals = np.copy(x)
        h_vals = np.minimum(h_vals, 1.0)
        h_vals = np.maximum(h_vals, 0.0)
        h_vals = h_vals * (n_points / 2) / (np.sum(h_vals) if np.sum(h_vals) != 0 else 1)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def fitness(x):
        return objective(x)
    
    result = opt.minimize(fitness, h, method='L-BFGS-B', options={'maxiter': 100, 'ftol': 1e-8})
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
```
