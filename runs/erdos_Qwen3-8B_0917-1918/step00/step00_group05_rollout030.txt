import numpy as np
import scipy.optimize as optimize
from scipy.optimize import linprog
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_C5(h):
        h = np.concatenate([h, h])
        shifted = np.zeros((2 * n_points, 2 * n_points))
        for k in range(1, n_points + 1):
            shifted[:, k] = h[k:]
        overlap = np.zeros(n_points)
        for k in range(1, n_points + 1):
            overlap[k - 1] = np.dot(h, 1 - h[k:])
        return np.max(overlap * dx)
    
    def optimize_h():
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        for _ in range(100):
            best_overlap = compute_C5(h)
            best_h = h.copy()
            for i in range(n_points):
                for delta in [-0.1, 0.1]:
                    h_test = h.copy()
                    h_test[i] += delta
                    h_test = np.clip(h_test, 0, 1)
                    h_test = h_test * (n_points / 2) / np.sum(h_test)
                    curr_overlap = compute_C5(h_test)
                    if curr_overlap < best_overlap:
                        best_overlap = curr_overlap
                        best_h = h_test
            h = best_h
        return best_h, best_overlap
    
    h_opt, c5_opt = optimize_h()
    return h_opt, c5_opt, n_points
