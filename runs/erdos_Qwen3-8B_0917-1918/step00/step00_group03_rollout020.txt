import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points, endpoint=True)
    h = np.random.rand(n_points) * 0.5
    h = h / np.sum(h) * n_points / 2

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def gradient(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n_points / 2
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        dl = np.zeros_like(h)
        for i in range(n_points):
            for j in range(n_points):
                if (i - j + n_points) % n_points == max_idx:
                    dl[i] += 1 * dx
                    dl[j] -= 1 * dx
        return dl

    result = optimize.minimize(
        fun=objective,
        x0=h,
        jac=gradient,
        bounds=[(0, 1) for _ in range(n_points)],
        method='L-BFGS-B',
        options={'gtol': 1e-5, 'maxiter': 100}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * n_points / 2
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
