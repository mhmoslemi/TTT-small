```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 2 - 1
    h[h < 0] = 0
    h[h > 1] = 1
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(h):
        h = np.concatenate([h, h[:n_points//2]])
        corr = np.correlate(h, 1 - h[:n_points], mode='full')
        return np.max(corr * dx)
    
    result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)]*n_points, options={'maxiter': 1000, 'gtol': 1e-6})
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
```
