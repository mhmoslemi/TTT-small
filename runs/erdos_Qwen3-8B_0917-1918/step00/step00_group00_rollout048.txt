```python
import numpy as np
import scipy.optimize as optimize
import scipy.interpolate as interpolate
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * n_points / 2 / dx
    max_iter = 100
    for _ in range(max_iter):
        start_time = time.time()
        shifts = np.linspace(-1.5, 1.5, 50)
        corrs = []
        for k in shifts:
            corr = np.correlate(h, 1 - h, mode='full')
            corrs.append(corr)
        corrs = np.array(corrs)
        corr = np.mean(corrs, axis=0)
        idx = np.argmax(corr)
        max_corr = corr[idx]
        if max_corr < 0.380876:
            break
        h = h * (1 - (max_corr - 0.380876) / (max_corr - 0.380876)) + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * n_points / 2 / dx
        if time.time() - start_time > budget_s / 2:
            break
    return h, max_corr, n_points
```
