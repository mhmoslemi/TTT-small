import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 1 if i % 10 < 5 else 0
    for _ in range(100):
        start_time = time.time()
        if time.time() - start_time > budget_s:
            break
        def objective(x):
            h = np.zeros(n_points)
            h[x.astype(int)] = 1
            integrals = np.zeros(n_points)
            for k in range(n_points):
                overlap = np.sum(h * (1 - np.roll(h, k)))
                integrals[k] = overlap * dx
            return np.max(integrals)
        result = opt.minimize(objective, x0=np.random.randint(0, n_points, size=n_points), method='L-BFGS-B')
        h = np.zeros(n_points)
        h[result.x.astype(int)] = 1
    integrals = np.zeros(n_points)
    for k in range(n_points):
        overlap = np.sum(h * (1 - np.roll(h, k)))
        integrals[k] = overlap * dx
    c5_bound = np.max(integrals)
    return h, c5_bound, n_points
