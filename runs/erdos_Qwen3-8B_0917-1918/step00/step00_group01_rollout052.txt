```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(h):
        h = np.concatenate([h, h])
        corr = np.correlate(h, 1 - h[:n_points], mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    def optimize_h():
        result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        return result.x
    
    start_time = time.time()
    best_h = optimize_h()
    best_c5 = objective(best_h)
    
    while time.time() - start_time < budget_s - 10:
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        h_opt = optimize.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])['x']
        curr_c5 = objective(h_opt)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h_opt
    
    return best_h, best_c5, n_points
```
