```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    h = h / (n_points / 2)
    
    def compute_c5(h):
        h = np.concatenate([h, h])
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def fitness(x):
        h = np.zeros(n_points)
        h[:x.shape[0]] = x
        return compute_c5(h)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = optimize.minimize(fitness, h, bounds=bounds, method='L-BFGS-B', options={'disp': False})
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
```
