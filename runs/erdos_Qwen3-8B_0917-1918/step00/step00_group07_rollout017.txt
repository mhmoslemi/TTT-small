import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h, dtype=np.float64)
    
    def objective(x):
        h_new = np.roll(x, 1)
        corr = np.correlate(h_new, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    x0 = h.copy()
    
    start_time = time.time()
    result = optimize.minimize(
        objective, 
        x0, 
        bounds=bounds, 
        constraints={'type': 'eq', 'fun': constraint},
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    if time.time() - start_time > budget_s - 10:
        return (h, np.max(np.correlate(np.roll(h, 1), 1 - h, mode='full')) * dx, n_points)
    
    best_h = result.x
    best_c5 = np.max(np.correlate(np.roll(best_h, 1), 1 - best_h, mode='full')) * dx
    return (best_h, best_c5, len(best_h))
