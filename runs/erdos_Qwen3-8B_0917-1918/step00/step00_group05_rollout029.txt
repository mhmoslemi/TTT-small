import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    initial_h = initial_h_values
    h = np.array(initial_h[:n_points])
    best_c5 = np.inf
    time_start = time.time()
    
    def objective(x):
        nonlocal best_c5
        h_vals = np.clip(x, 0, 1)
        conv = np.correlate(h_vals, 1 - h_vals, mode="full")
        c5 = np.max(conv) * (2.0 / n_points)
        if c5 < best_c5:
            best_c5 = c5
        return c5
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    result = opt.minimize(objective, h, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'maxiter': 100})
    
    if time.time() - time_start > budget_s - 10:
        return (initial_h[:n_points], best_c5, n_points)
    
    return (result.x, best_c5, n_points)
