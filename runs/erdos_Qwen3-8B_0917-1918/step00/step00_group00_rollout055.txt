import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 0.5 + 0.5 * np.sin(2 * np.pi * (x[i] - 0.5))
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    def objective(h):
        h = h.reshape(n_points)
        shift = np.arange(-n_points + 1, n_points)
        corr = np.zeros_like(shift)
        for k in range(len(shift)):
            corr[k] = np.sum(h[:n_points - k] * (1 - h[k:n_points]))
        return np.max(corr)
    
    def constraint(h):
        h = h.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), 0, 0, lb=0, ub=1)
    
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 200})
    
    h_values = result.x
    c5_bound = result.fun
    
    return h_values, c5_bound, n_points
