import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h_values = np.array(initial_h_values)
    h = initial_h_values.copy()
    h = np.interp(h, (h.min(), h.max()), (0, 1))
    h = np.round(h * n_points) / n_points
    h = np.clip(h, 0, 1)
    h = h[:n_points]
    h = (h * n_points).astype(int)
    h = h / n_points
    h = np.interp(h, (h.min(), h.max()), (0, 1))
    h = np.clip(h, 0, 1)
    
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    x0 = h.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint}
    
    start_time = time.time()
    result = opt.minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = np.interp(best_h, (best_h.min(), best_h.max()), (0, 1))
    best_h = np.round(best_h * n_points) / n_points
    best_h = np.clip(best_h, 0, 1)
    best_h = np.interp(best_h, (best_h.min(), best_h.max()), (0, 1))
    best_h = np.clip(best_h, 0, 1)
    best_h = np.interp(best_h, (best_h.min(), best_h.max()), (0, 1))
    best_h = np.clip(best_h, 0, 1)
    
    corr = np.correlate(best_h, 1 - best_h, mode="full")
    c5_bound = np.max(corr) * dx
    return (best_h, c5_bound, n_points)
