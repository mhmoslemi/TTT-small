```python
import numpy as np
import scipy.optimize
import scipy.integrate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(h):
        h = h.reshape(-1)
        h_ext = np.concatenate([h, h[:n_points//2]])
        corr = np.correlate(h_ext, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}], options={'maxiter': 100})
    
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
```
