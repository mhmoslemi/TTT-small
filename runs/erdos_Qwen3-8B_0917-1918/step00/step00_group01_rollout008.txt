def run(seed=42, budget_s=600):

        x = np.linspace(0, 2, n, endpoint=False)
        h = np.zeros(n)

            if 0.5 <= x[i] < 1.5:
                h[i] = 1
            elif 0.25 <= x[i] < 0.75:
                h[i] = 0.5
            else:
                h[i] = 0
        return h

        n = len(h)
        dx = 2.0 / n

        return np.max(overlap * dx)

        def obj(h):
            return compute_overlap(h)
        
        def cons(h):
            return constraint_sum(h, 2.0 / n_points)
        
        return x, h, obj, cons

        start = time()
        result = opt.minimize(

        )

            return h_initial, compute_overlap(h_initial)
        return result.x, compute_overlap(result.x)

        x = np.linspace(0, 2, n_points, endpoint=False)
        dx = 2.0 / n_points

        return np.max(overlap * dx), np.sum(h) * dx
