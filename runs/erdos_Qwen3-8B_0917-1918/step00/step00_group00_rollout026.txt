```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 2 - 1
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        h = np.concatenate([h, h])
        corr = np.correlate(h, 1 - h, mode='full')
        corr = corr[n_points//2 : n_points//2 + n_points]
        return np.max(corr * dx)
    
    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        return compute_c5(x)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint})
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-8, 'maxiter': 50})
    h_opt = result.x
    c5 = compute_c5(h_opt)
    
    return h_opt, c5, n_points
```
