```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(h):
        h = np.array(h, dtype=np.float64)
        h = np.roll(h, 1)
        h = h[:n_points]
        h = np.roll(h, 1)
        h = h[:n_points]
        correlate = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlate) * dx
        return c5
    
    for _ in range(50):
        start_time = time.time()
        result = minimize(objective, h, method='L-BFGS-B', bounds=[(0,1) for _ in range(n_points)])
        h = result.x
        if time.time() - start_time > budget_s / 2:
            break
    
    correlate = np.correlate(h, 1 - h, mode="full")
    c5_bound = np.max(correlate) * dx
    return h.tolist(), c5_bound, n_points
```
