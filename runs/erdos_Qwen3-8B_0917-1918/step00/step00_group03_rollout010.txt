import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points)
    h = h * 2 - 1
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        h = np.repeat(h, 2)
        k_values = np.arange(-n_points, n_points)
        max_overlap = 0
        for k in k_values:
            if k == 0:
                continue
            overlap = np.sum(h[:n_points] * (1 - h[k:n_points + k]))
            max_overlap = max(max_overlap, overlap)
        return max_overlap * dx
    
    def objective(x):
        h = x.reshape(n_points)
        return compute_c5(h)
    
    def constraint(x):
        h = x.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), -np.inf, n_points / 2)
    
    result = optimize.minimize(objective, h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    
    best_h = result.x
    c5 = compute_c5(best_h)
    
    return (best_h, c5, n_points)
