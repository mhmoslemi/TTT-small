```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import LinearConstraint, Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        h_pad = np.concatenate([h, h[:n_points//2]])
        h_shift = np.array([1 - h[i] for i in range(n_points)])
        corr = np.correlate(h_pad, h_shift, mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        return compute_c5(x)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = LinearConstraint(np.ones(n_points), 0, 0, lb=0, ub=1)
    bounds = Bounds(0, 1)
    
    result = optimize.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-8})
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
```
