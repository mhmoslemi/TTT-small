import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points, endpoint=False)
    
    # Initial guess based on a simple periodic function
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 0.5 + 0.5 * np.sin(10 * x[i] - np.pi / 2)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h
    
    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = LinearConstraint([1]*n_points, [0]*n_points, [n_points/2])
    bounds = Bounds([0]*n_points, [1]*n_points)
    
    result = optimize.minimize(
        objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-8, 'maxiter': 50}
    )
    
    best_h = result.x
    best_c5 = evaluate_c5(best_h)
    
    return best_h, best_c5, n_points
