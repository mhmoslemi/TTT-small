import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 2 - 1
    h = np.clip(h, 0, 1)
    h = h - np.min(h)
    h = h / np.max(h)
    h = np.minimum(h, 1)
    h = np.maximum(h, 0)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(x):
        nonlocal n_points, dx
        h_new = np.copy(x)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new - np.min(h_new)
        h_new = h_new / np.max(h_new)
        h_new = np.minimum(h_new, 1)
        h_new = np.maximum(h_new, 0)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        corr = np.correlate(h_new, 1 - h_new, mode="full")
        corr = corr * dx
        return np.max(corr)
    
    result = optimize.minimize(objective, h, method='L-BFGS-B', options={'maxiter': 1000, 'ftol': 1e-8})
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h - np.min(best_h)
    best_h = best_h / np.max(best_h)
    best_h = np.minimum(best_h, 1)
    best_h = np.maximum(best_h, 0)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    corr = np.correlate(best_h, 1 - best_h, mode="full")
    c5_bound = np.max(corr * dx)
    
    return (best_h, c5_bound, n_points)
