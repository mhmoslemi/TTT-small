```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5
    h = np.maximum(0, np.minimum(1, h))
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = np.inf

    def compute_c5(h):
        h = np.maximum(0, np.minimum(1, h))
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    for _ in range(100):
        t0 = time.time()
        if time.time() - t0 > budget_s:
            break
        h = best_h.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                h[i] = np.random.rand()
        h = np.maximum(0, np.minimum(1, h))
        h = h * (n_points / 2) / np.sum(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5, n_points
```
