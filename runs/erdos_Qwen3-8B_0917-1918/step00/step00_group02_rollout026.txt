```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 1 if np.random.rand() < 0.5 else 0
    h = h * dx * n_points / np.sum(h)

    def objective(h):
        c5 = 0
        for k in range(1, 10):
            integral = np.sum(h[:n_points - k] * (1 - h[k:])) * dx
            c5 = max(c5, integral)
        return c5

    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(objective, h, bounds=bounds, method='L-BFGS-B', tol=1e-8, options={'maxiter': 100})
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
```
