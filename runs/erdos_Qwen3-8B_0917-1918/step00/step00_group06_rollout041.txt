import numpy as np
import scipy.optimize as opt
from scipy.optimize import minimize
from scipy.integrate import simps
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    initial_h = np.roll(initial_h, n_points // 4)
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    
    def objective(h):
        h = np.array(h, dtype=np.float64)
        c5 = 0
        for k in range(1, 6):
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = max(c5, np.max(corr) * dx)
        return c5
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    def bounds(h):
        return [(0, 1) for _ in h]
    
    result = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 100, 'disp': False})
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
