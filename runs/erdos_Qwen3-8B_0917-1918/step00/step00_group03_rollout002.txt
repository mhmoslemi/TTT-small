import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * dx * 2
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        h = np.concatenate([h, h])
        corr = np.correlate(h, 1 - h, mode="full")
        corr = corr[n_points - 1:]
        return np.max(corr) * dx
    
    def objective(x):
        x = x.reshape(n_points)
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        return compute_c5(x)
    
    def grad(x):
        x = x.reshape(n_points)
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        corr = np.correlate(x, 1 - x, mode="full")
        corr = corr[n_points - 1:]
        grad = np.zeros(n_points)
        for i in range(n_points):
            x[i] = 1.0
            corr_i = np.correlate(x, 1 - x, mode="full")
            corr_i = corr_i[n_points - 1:]
            grad[i] = (corr_i - corr) * dx
            x[i] = 0.0
        return grad.flatten()
    
    result = opt.minimize(objective, h, method='L-BFGS-B', jac=grad, options={'maxiter': 100, 'gtol': 1e-6})
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    c5_bound = compute_c5(best_h)
    return (best_h, c5_bound, n_points)
