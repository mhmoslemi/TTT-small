import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        conv = np.correlate(h, 1 - h, mode='full')
        conv = conv[conv.size // 2 : conv.size // 2 + n_points]
        return np.max(conv * dx)
    
    def update_h(h):
        h = np.clip(h, 0, 1)
        idx = np.argsort(h)[::-1]
        for i in idx:
            if h[i] < 1:
                h[i] = np.clip(h[i] + 0.05, 0, 1)
                break
        return h
    
    initial_h = np.random.rand(n_points)
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    
    best_h = initial_h.copy()
    best_c5 = objective(best_h)
    
    for _ in range(100):
        start_time = time.time()
        if time.time() - start_time > budget_s:
            break
        h = best_h.copy()
        h = update_h(h)
        c5 = objective(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    return best_h, best_c5, n_points
