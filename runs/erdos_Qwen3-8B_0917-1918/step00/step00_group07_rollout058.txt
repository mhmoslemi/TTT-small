```python
import numpy as np
import scipy.optimize as optimize
from scipy.ndimage import convolve
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        h = np.concatenate((h, h))
        corr = convolve(h, 1 - h[:n_points], mode='full')
        return np.max(corr) * dx
    
    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        return compute_c5(x)
    
    def constraint(x):
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        return np.sum(x) - n_points / 2
    
    result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)], constraints=[{'type': 'eq', 'fun': constraint}])
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
```
