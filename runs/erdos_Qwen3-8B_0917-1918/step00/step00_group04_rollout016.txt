```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 2 - 1
    h = np.clip(h, 0, 1)
    h = h * dx * n_points
    h = h / np.sum(h)
    
    def obj(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    for _ in range(100):
        start_time = time.time()
        result = optimize.minimize(obj, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        if time.time() - start_time > budget_s / 2:
            break
        h = result.x
    
    return h, obj(h), n_points
```
