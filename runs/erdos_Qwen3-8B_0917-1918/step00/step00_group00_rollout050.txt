import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    h = h / (n_points / 2)

    def compute_C5(h):
        h = np.concatenate([h, h])
        c5 = 0.0
        for k in range(1, 10):
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = max(c5, np.max(corr) * dx)
        return c5

    def objective(x):
        x = x.reshape(n_points)
        x = np.clip(x, 0, 1)
        return compute_C5(x)

    def constraint(x):
        x = x.reshape(n_points)
        x = np.clip(x, 0, 1)
        return np.sum(x) - n_points / 2

    x0 = h.copy()
    bounds = [(0, 1) for _ in range(n_points)]

    result = optimize.minimize(objective, x0, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="SLSQP", options={"ftol": 1e-8, "maxiter": 100})
    best_h = result.x
    best_c5 = compute_C5(best_h)
    return best_h, best_c5, n_points
