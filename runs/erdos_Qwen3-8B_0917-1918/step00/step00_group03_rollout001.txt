import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = initial_h_values[:n_points]
    h = initial_h.copy()
    best_c5 = np.inf
    best_h = h.copy()
    
    def objective(h):
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    t0 = time.time()
    while time.time() - t0 < budget_s:
        res = optimize.minimize(objective, h, bounds=bounds, constraints=cons, method='SLSQP')
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()
        else:
            break
    
    return best_h, best_c5, n_points
