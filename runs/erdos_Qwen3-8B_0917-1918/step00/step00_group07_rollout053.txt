import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * dx * n_points / 2
    h[h > 1] = 1
    h[h < 0] = 0
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h

    def objective(x):
        nonlocal n_points, dx
        h_new = x.reshape(n_points)
        h_new[h_new > 1] = 1
        h_new[h_new < 0] = 0
        sum_h_new = np.sum(h_new)
        h_new = h_new * (n_points / 2) / sum_h_new
        c5 = 0
        for k in range(1, n_points - 1):
            corr = np.correlate(h_new, 1 - h_new, mode="full")
            c5 = max(c5, np.max(corr) * dx)
        return c5

    def constraint(x):
        nonlocal n_points
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]

    result = optimize.minimize(objective, h, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="SLSQP", options={"ftol": 1e-8, "maxiter": 1000})

    best_h = result.x
    best_h[best_h > 1] = 1
    best_h[best_h < 0] = 0
    sum_h_best = np.sum(best_h)
    best_h = best_h * (n_points / 2) / sum_h_best

    c5 = 0
    for k in range(1, n_points - 1):
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        c5 = max(c5, np.max(corr) * 2 / n_points)

    return best_h, c5, n_points
