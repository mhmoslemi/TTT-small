```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_C5(h):
        h_reshaped = np.concatenate([h, h[:n_points//2]])
        result = np.correlate(h_reshaped, 1 - h_reshaped[:n_points], mode="full")
        max_overlap = np.max(result[:n_points] * dx)
        return max_overlap
    
    def objective(x):
        return compute_C5(x)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    bounds = Bounds(0, 1)
    
    result = optimize.minimize(objective, h, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method='SLSQP')
    
    best_h = result.x
    best_C5 = compute_C5(best_h)
    
    return best_h, best_C5, n_points
```
