```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    def evaluate(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def objective(h):
        return evaluate(h)
    
    def constraint_sum(h):
        return np.sum(h) - len(h) / 2
    
    n_points = 100
    h = np.random.rand(n_points)
    h = h / np.sum(h) * (n_points / 2)
    
    result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, constraints={'type': 'eq', 'fun': constraint_sum})
    best_h = result.x
    best_c5 = evaluate(best_h)
    
    return best_h, best_c5, n_points
```
