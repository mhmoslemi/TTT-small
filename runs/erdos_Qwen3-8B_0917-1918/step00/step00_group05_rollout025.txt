```python
import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    best_c5 = np.inf
    for _ in range(100):
        start_time = time.time()
        if time.time() - start_time > budget_s:
            break
        h = np.random.permutation(h)
        h = h * (n_points / 2) / np.sum(h)
        cor = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(cor) * (2.0 / n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    return best_h, best_c5, n_points
```
