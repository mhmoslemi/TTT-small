    def objective(x):
        nonlocal best_c5, best_h
        x = np.clip(x, 0, 1)
        x = x.reshape(n_points)
        sum_x = np.sum(x) * dx
        if np.abs(sum_x - 1) > 1e-6:
            return np.inf
        cor = np.correlate(x, 1 - x, mode="full")
        c5 = np.max(cor * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()
        return c5
    
    def callback(x):
        nonlocal best_c5
        if best_c5 > np.max(np.correlate(x, 1 - x, mode="full") * dx):
            best_c5 = np.max(np.correlate(x, 1 - x, mode="full") * dx)
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        x0 = np.copy(best_h)
        res = minimize(objective, x0, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)], callback=callback)
        if res.success:
            best_h = res.x
    
    return best_h, best_c5, n_points
