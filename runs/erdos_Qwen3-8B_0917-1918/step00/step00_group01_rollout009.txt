import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    h[n_points//2:] = 0.0
    for _ in range(100):
        t0 = time.time()
        def objective(x):
            h_ = np.zeros(n_points)
            h_[np.floor(x).astype(int)] = 1.0
            h_[np.floor(x).astype(int) + 1] = 0.5
            h_ = np.clip(h_, 0, 1)
            h_ = np.interp(np.arange(n_points), np.sort(x), h_)
            h_ = np.interp(np.arange(n_points), np.sort(x), h_)
            h_ = np.interp(np.arange(n_points), np.sort(x), h_)
            h_ = np.clip(h_, 0, 1)
            kernel = 1 - np.roll(h_, 1)
            corr = np.correlate(h_, kernel, mode="full")
            c5 = np.max(corr * (2.0 / n_points))
            return c5

        def constraint(x):
            return np.sum(np.interp(np.arange(n_points), np.sort(x), np.ones(n_points))) - n_points / 2

        x0 = np.linspace(0, 2, n_points)
        res = optimize.minimize(objective, x0, method='L-BFGS-B', bounds=[(0, 2) for _ in range(n_points)])
        h_values = np.interp(np.arange(n_points), np.sort(res.x), np.ones(n_points))
        h_values = np.clip(h_values, 0, 1)
        c5_bound = np.max(np.correlate(h_values, 1 - np.roll(h_values, 1), mode="full") * (2.0 / n_points))
        if time.time() - t0 > budget_s:
            break
    return h_values, c5_bound, n_points
