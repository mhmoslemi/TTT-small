```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h_values = np.random.uniform(0, 1, n_points)
    h_values = h_values * n_points / 2.0
    best_c5 = float('inf')
    
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(np.abs(corr))
        return c5 * dx
    
    for _ in range(100):
        start = time.time()
        if time.time() - start > budget_s * 0.8:
            break
        bounds = [(0, 1) for _ in range(n_points)]
        res = optimize.minimize(compute_c5, h_values, bounds=bounds, method='L-BFGS-B', options={'maxiter': 100})
        h_values = res.x
        current_c5 = compute_c5(h_values)
        if current_c5 < best_c5:
            best_c5 = current_c5
            print(f"New best C5: {best_c5}")
    
    return h_values, best_c5, n_points
```
