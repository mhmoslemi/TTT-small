```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    h = np.append(h, np.zeros(n_points//2))
    
    def objective(x):
        h_values = x.reshape(n_points)
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr * dx)
    
    bounds = Bounds(0, 1)
    constraint = LinearConstraint(np.ones(n_points), 0, n_points / 2)
    
    result = opt.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
```
