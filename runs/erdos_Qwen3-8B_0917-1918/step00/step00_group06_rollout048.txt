import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    initial_h = initial_h_values
    h = np.array(initial_h, dtype=np.float64)
    best_c5 = np.inf
    start_time = time.time()
    
    def objective(x):
        nonlocal best_c5
        h_cur = np.clip(x, 0, 1)
        corr = np.correlate(h_cur, 1 - h_cur, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        if c5 < best_c5:
            best_c5 = c5
        return c5
    
    def fitness(x):
        return objective(x)
    
    def constraints(x):
        return np.sum(x) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraints}]
    
    result = optimize.minimize(fitness, h, method='SLSQP', constraints=cons, options={'ftol': 1e-8, 'maxiter': 100})
    best_h = result.x
    best_c5 = objective(best_h)
    
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        for _ in range(10):
            perturb = np.random.normal(0, 0.05, n_points)
            h_perturb = np.clip(best_h + perturb, 0, 1)
            h_perturb = h_perturb / np.sum(h_perturb) * n_points / 2
            c5 = objective(h_perturb)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_perturb
    
    return best_h, best_c5, n_points
