```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(x):
        nonlocal h
        h = x
        conv = np.correlate(h, 1 - h, mode="full")
        return np.max(conv * dx)
    
    def constraints(x):
        return np.sum(x) - n_points / 2
    
    constraints = LinearConstraint(np.ones(n_points), lb=0, ub=1)
    bounds = Bounds(lb=0, ub=1)
    
    result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-8, 'maxiter': 100})
    
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
```
