```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 2 - 1
    h -= np.mean(h)
    h *= 2 / n_points
    h = np.clip(h, 0, 1)
    def objective(h):
        t = time.time()
        h = np.tile(h, (2, 1))
        h = np.concatenate([h[0], h[1]])
        cor = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(cor) * dx
        return c5
    for _ in range(100):
        if time.time() - t > budget_s:
            break
        h = np.random.rand(n_points) * 2 - 1
        h -= np.mean(h)
        h *= 2 / n_points
        h = np.clip(h, 0, 1)
    t = time.time()
    h = np.random.rand(n_points) * 2 - 1
    h -= np.mean(h)
    h *= 2 / n_points
    h = np.clip(h, 0, 1)
    best_c5 = objective(h)
    return h, best_c5, n_points
```
