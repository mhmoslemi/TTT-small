import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    initial_h = initial_h_values
    h = np.array(initial_h)
    best_c5 = float('inf')
    best_h = h.copy()
    
    def cost(h):
        nonlocal best_c5
        h = h.reshape(-1)
        conv = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(conv) * (2.0 / n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        h = np.random.rand(n_points)
        h = (h * 2 - 1) * 0.5 + 0.5
        h = h / np.sum(h) * n_points / 2
        h = h / np.sum(h) * n_points / 2
        result = opt.minimize(lambda x: cost(x), x0=h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        if result.success:
            h = result.x
        if time.time() - start_time > budget_s - 10:
            break
    
    return best_h, best_c5, n_points
