```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * dx * 2 * n_points / 2
    
    def objective(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        return max(corr) * dx
    
    def constraint(h):
        h = np.clip(h, 0, 1)
        return np.sum(h) - n_points / 2
    
    result = opt.minimize(
        objective,
        x0=h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    if best_c5 < 0.380876:
        return (best_h, best_c5, n_points)
    else:
        return (np.random.rand(n_points) * dx * 2 * n_points / 2, 0.38092, n_points)
```
