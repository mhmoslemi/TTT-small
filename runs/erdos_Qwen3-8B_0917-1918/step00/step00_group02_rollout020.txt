import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = initial_h.copy()
    
    def objective(x):
        h_eval = np.clip(x, 0, 1)
        corr = correlate(h_eval, 1 - h_eval, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = minimize(objective, h, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'maxiter': 500})
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
