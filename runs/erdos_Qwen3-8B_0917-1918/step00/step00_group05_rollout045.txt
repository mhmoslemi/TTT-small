```python
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 2 - 1
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h) if np.sum(h) != 0 else h
    best_c5 = float('inf')
    start_time = time.time()
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h) if np.sum(h) != 0 else h
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h) if np.sum(h) != 0 else h
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5, n_points
```
