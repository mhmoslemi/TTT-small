```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 2 - 1
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(x):
        h_new = x.reshape(n_points)
        corr = np.correlate(h_new, 1 - h_new, mode="full")
        return np.max(corr) * dx
    
    result = opt.minimize(objective, h, method='L-BFGS-B', options={'gtol': 1e-5, 'maxiter': 100})
    h_best = result.x
    c5 = np.max(np.correlate(h_best, 1 - h_best, mode="full")) * dx
    return h_best, c5, n_points
```
