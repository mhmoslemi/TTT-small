import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5
    h = np.minimum(np.maximum(h, 0.0), 1.0)
    total = np.sum(h) * dx
    h = np.array([h[i] for i in range(n_points)])
    h = h * (1.0 / (total / dx))
    
    def objective(h):
        h = np.array(h)
        integrals = []
        for k in range(1, n_points):
            shifted = np.roll(h, k)
            integral = np.sum(h * (1 - shifted)) * dx
            integrals.append(integral)
        return np.max(integrals)
    
    result = optimize.minimize(objective, h, method='L-BFGS-B', options={'maxiter': 100, 'gtol': 1e-4})
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
