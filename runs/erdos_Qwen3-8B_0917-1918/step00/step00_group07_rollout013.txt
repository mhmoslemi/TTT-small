import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points, endpoint=False)
    
    # Initial guess
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    h[n_points//2:] = 0.5
    h = np.random.permutation(h)
    
    def objective(h_values):
        integral = np.correlate(h_values, 1 - h_values, mode="full")
        c5 = np.max(integral) * dx
        return c5
    
    def constraints(h_values):
        sum_h = np.sum(h_values)
        return sum_h - n_points / 2
    
    def evaluate_and_display(h_values):
        integral = np.correlate(h_values, 1 - h_values, mode="full")
        c5 = np.max(integral) * dx
        print(f"Current C₅: {c5}")
        return c5
    
    # Optimization
    result = minimize(lambda h: objective(h), x0=h, bounds=Bounds(0, 1), constraints=LinearConstraint(lambda h: np.sum(h), 0, n_points / 2))
    best_h = result.x
    best_c5 = evaluate_and_display(best_h)
    
    return best_h, best_c5, n_points
