```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.full(n_points, 0.5)
    h[0:n_points//2] = 0
    h[n_points//2:] = 1
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5_val = np.max(corr) * dx
        return c5_val
    
    bounds = Bounds(0, 1)
    result = optimize.minimize(
        objective, h, method='L-BFGS-B', bounds=bounds, options={'gtol': 1e-5, 'maxiter': 100}
    )
    best_h = result.x
    c5_bound = objective(best_h)
    return (best_h, c5_bound, n_points)
```
