```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 1000
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    h[n_points//2:] = 0.0
    for _ in range(10):
        t0 = time.time()
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5
        bounds = Bounds(0, 1)
        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
        h = res.x
        if time.time() - t0 > budget_s:
            break
    c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    return h, c5, n_points
```
