import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.mean(h)
    def objective(x):
        h = x
        conv = np.correlate(h, 1 - h, mode="full")
        return np.max(conv)
    result = opt.minimize(objective, h, method='L-BFGS-B', options={'maxiter': 1000, 'ftol': 1e-6})
    best_h = result.x
    conv = np.correlate(best_h, 1 - best_h, mode="full")
    c5 = np.max(conv) * (2.0 / n_points)
    return best_h, c5, n_points
