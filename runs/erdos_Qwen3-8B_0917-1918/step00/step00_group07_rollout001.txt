```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 1.0
    h = h / np.mean(h) * (n_points / 2)
    best_c5 = float('inf')
    best_h = h.copy()
    
    def compute_c5(h_vals):
        h = h_vals / np.mean(h_vals) * (n_points / 2)
        h = np.concatenate([h, h])
        shifts = np.arange(-n_points, n_points)
        overlaps = []
        for k in shifts:
            if k >= n_points:
                break
            if k < 0:
                continue
            overlap = np.sum(h[:n_points] * (1 - h[k:n_points + k]))
            overlaps.append(overlap)
        c5 = np.max(overlaps) * dx
        return c5
    
    for _ in range(100):
        start_time = time.time()
        if time.time() - start_time > budget_s * 0.6:
            break
        x = cp.Variable(n_points)
        obj = cp.Maximize(cp.norm(x - x[0], 1))
        constraints = [
            cp.sum(x) == n_points / 2,
            x >= 0,
            x <= 1
        ]
        prob = cp.Problem(obj, constraints)
        prob.solve(solver='ECOS', abstol=1e-8, reltol=1e-8)
        if prob.status == 'optimal':
            h = x.value
        else:
            h = np.random.rand(n_points) * 1.0
            h = h / np.mean(h) * (n_points / 2)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5, n_points
```
