        return c5

    def objective(x):
        x = x.reshape(-1)
        x /= np.sum(x) * dx
        x *= n_points / 2
        return compute_c5(x)

        x = x.reshape(-1)
        x /= np.sum(x) * dx
        x *= n_points / 2
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    start_time = time.time()
    result = opt.minimize(objective, h, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="SLSQP", tol=1e-8, options={"maxiter": 100})
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        return (h, compute_c5(h), n_points)

    best_h = result.x
    best_h = best_h.reshape(-1)
    best_h /= np.sum(best_h) * dx
    best_h *= n_points / 2
    c5 = compute_c5(best_h)
    return (best_h, c5, n_points)
