```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5
    h *= 2
    h -= h.min()
    h /= h.max()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)

    def objective(x):
        h = x.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    bounds = Bounds(0, 1)
    result = opt.minimize(
        objective,
        x0=h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    h_values = result.x
    c5_bound = result.fun
    return h_values, c5_bound, n_points
```
