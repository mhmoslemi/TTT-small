import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5]*n_points)
    for i in range(n_points):
        initial_h_values[i] = np.random.uniform(0, 1)
    initial_h_values = initial_h_values * n_points / 2
    
    h = initial_h_values.copy()
    for _ in range(100):
        for shift in range(1, 6):
            tmp = np.roll(h, shift)
            overlap = np.sum(h * (1 - tmp)) * dx
            if overlap > 0.38092:
                h = np.roll(h, 1)
                break
        else:
            break
    
    def objective(x):
        nonlocal h
        h = x.copy()
        overlaps = []
        for shift in range(1, 6):
            tmp = np.roll(h, shift)
            overlap = np.sum(h * (1 - tmp)) * dx
            overlaps.append(overlap)
        return max(overlaps)

    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), lb=0, ub=1)
    result = optimize.minimize(objective, h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-6, 'maxiter': 100})
    
    best_h = result.x
    best_c5 = np.max([np.sum(best_h * (1 - np.roll(best_h, shift))) * dx for shift in range(1, 6)])
    
    return (best_h, best_c5, n_points)
