import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    initial_h[initial_h == 0] = 0.5
    initial_h = initial_h + np.random.normal(0, 0.1, n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    h = initial_h.copy()
    def objective(x):
        nonlocal h
        h = x
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    def constraint(x):
        return np.sum(x) - n_points / 2
    bounds = Bounds(0, 1)
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint})
    return result.x, result.fun, n_points
