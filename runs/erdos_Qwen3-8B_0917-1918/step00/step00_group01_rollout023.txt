```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5
    h = h / np.sum(h) * n_points / 2
    h = np.clip(h, 0, 1)
    
    def objective(x):
        x = x.reshape(n_points)
        cor = np.correlate(x, 1 - x, mode="full")
        return np.max(cor) * dx
    
    def constraint(x):
        x = x.reshape(n_points)
        return np.sum(x) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint}
    bounds = Bounds(0, 1)
    
    result = opt.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-8, 'maxiter': 100})
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
```
