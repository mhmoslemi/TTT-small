```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, fmin_l_bfgs_b
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    h[n_points//2:] = 0.0
    
    def objective(x):
        nonlocal n_points, dx
        h = x.copy()
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    x0 = h.copy()
    bounds = Bounds(0, 1)
    
    start_time = time.time()
    result = optimize.minimize(objective, x0, bounds=bounds, constraints={"type": "eq", "fun": constraint, "jac": lambda x: np.ones_like(x)})
    
    if time.time() - start_time > budget_s:
        return (h, 0.38092, n_points)
    else:
        best_h = result.x
        best_c5 = objective(best_h)
        return (best_h, best_c5, n_points)
```
