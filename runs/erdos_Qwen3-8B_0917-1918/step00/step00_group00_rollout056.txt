```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * dx * 2
    h = np.minimum(np.maximum(h, 0), 1)
    h = h * n_points / 2  # Ensure integral is 1

    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def optimize_h(h, seed, budget_s):
        start_time = time.time()
        h_opt = h.copy()
        for _ in range(100):
            if time.time() - start_time > budget_s - 10:
                break
            h_opt = np.random.rand(n_points) * 2
            h_opt = np.minimum(np.maximum(h_opt, 0), 1)
            h_opt = h_opt * n_points / 2
            if np.max(np.correlate(h_opt, 1 - h_opt, mode="full")) * dx < 0.3808:
                return h_opt
        return h_opt

    best_h = optimize_h(h, seed, budget_s)
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
```
