```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 1 if i % 5 == 0 else 0.5
    
    def evaluate_h(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        integral = corr * dx
        return np.max(integral)
    
    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), lb=0, ub=n_points / 2)
    
    result = opt.minimize(
        fun=evaluate_h,
        x0=initial_h,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    best_h = result.x
    c5_bound = evaluate_h(best_h)
    return best_h, c5_bound, n_points
```
