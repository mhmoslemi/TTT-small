```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import LinearConstraint, Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5
    h = h / np.sum(h) * (n_points / 2)
    
    def objective(x):
        nonlocal n_points, dx
        h = x.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraints(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), 0, 0, name="sum_constraint")
    
    x0 = h
    
    start_time = time.time()
    result = opt.minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-6, 'maxiter': 100})
    
    if time.time() - start_time > budget_s - 10:
        return (h, 0.38092, n_points)
    
    best_h = result.x
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
```
