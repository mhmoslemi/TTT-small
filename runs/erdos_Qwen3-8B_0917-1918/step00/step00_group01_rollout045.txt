import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h, dtype=float)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = opt.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, len(best_h)
