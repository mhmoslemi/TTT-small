```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 2 - 1
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_C5(h):
        h = np.concatenate((h, h))
        conv = np.correlate(h, 1 - h[:n_points], mode="full")
        return np.max(conv) * dx
    
    best_C5 = compute_C5(h)
    best_h = h.copy()
    
    def objective(x):
        nonlocal best_C5, best_h
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        C5 = compute_C5(x)
        if C5 < best_C5:
            best_C5 = C5
            best_h = x.copy()
        return C5
    
    bounds = Bounds(0, 1)
    constraints = LinearConstraint(np.ones(n_points), 0, n_points / 2)
    
    start_time = time.time()
    result = opt.minimize(objective, h, bounds=bounds, constraints=constraints, method='SLSQP', options={'maxiter': 100})
    
    return best_h, best_C5, n_points
```
