        nonlocal best_c5, best_h

        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()

    start_time = time.time()
    while time.time() - start_time < budget_s:
        x0 = best_h.copy() + np.random.normal(0, 0.05, n_points)
        x0 = np.clip(x0, 0, 1)
        result = opt.minimize(objective, x0, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        if result.success:
            best_h = result.x
    
    return best_h, best_c5, n_points
