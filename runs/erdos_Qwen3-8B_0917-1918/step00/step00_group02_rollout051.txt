```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    h = h / np.mean(h) * (n_points / 2) / np.mean(h)

    def objective(x):
        h_vals = np.array(x)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        c5 = np.max(np.abs(corr)) * dx
        return c5

    def constraints(x):
        return np.sum(x) - n_points / 2

    x0 = h
    res = opt.minimize(objective, x0, method='L-BFGS-B', bounds=(0, 1), constraints={"type": "eq", "fun": constraints})
    h_opt = res.x
    c5_bound = objective(h_opt)
    return h_opt, c5_bound, n_points
```
