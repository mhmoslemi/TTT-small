import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        h_fft = np.fft.fft(h)
        shifted = np.roll(h, 1)
        overlap = np.fft.ifft(h_fft * np.conj(np.fft.fft(shifted))).real
        return np.max(overlap * dx)

    def objective(x):
        return compute_c5(x)

    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    x = cp.Variable(n_points)
    constraints = [cp.sum(x) == n_points / 2]
    obj = cp.Minimize(cp.norm(x - h, 2))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS, max_iter=1000)
    h_opt = x.value

    if h_opt is None:
        h_opt = h

    return h_opt, compute_c5(h_opt), n_points
