import numpy as np
import scipy.optimize as opt
import scipy.interpolate as interpolate
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    h[n_points//2:] = 0.0
    h = h + np.random.normal(0, 0.1, n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        h = np.roll(h, 1)
        c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        return c5
    
    for _ in range(100):
        start_time = time.time()
        if time.time() - start_time > budget_s:
            break
        def objective(x):
            h = np.copy(x)
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            return compute_c5(h)
        
        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(objective, h, bounds=bounds, method='L-BFGS-B', options={'maxiter': 10})
        h = result.x
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    
    h = np.clip(h, 0, 1)
    c5 = compute_c5(h)
    return h.tolist(), c5, n_points
