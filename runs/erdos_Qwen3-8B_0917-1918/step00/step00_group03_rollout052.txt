import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import time

def run(seed=42, budget_s=600, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * n_points / 2 / dx
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h

    def objective(x):
        h = x.reshape(n_points)
        shifted = np.roll(h, shift=1)
        overlap = np.sum(h * (1 - shifted))
        return overlap

    def constraint(x):
        h = x.reshape(n_points)
        return np.sum(h) - n_points / 2

    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), lb=0, ub=n_points / 2)

    result = opt.minimize(objective, h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    h_opt = result.x
    c5_val = np.max([np.sum(h_opt[i:] * (1 - h_opt[:-i])) for i in range(1, n_points)])
    c5_val *= dx

    return h_opt, c5_val, n_points
