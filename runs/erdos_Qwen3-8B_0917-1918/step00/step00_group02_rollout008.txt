import numpy as np
import scipy.optimize as optim
import scipy.signal as signal
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h, dtype=np.float64)
    time_start = time.time()
    
    def compute_C5(h):
        h = np.array(h)
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def objective(x):
        return compute_C5(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    result = optim.minimize(
        objective,
        h,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = compute_C5(best_h)
    
    if time.time() - time_start > budget_s - 10:
        return (best_h.tolist(), best_c5, n_points)
    
    return (best_h.tolist(), best_c5, n_points)
