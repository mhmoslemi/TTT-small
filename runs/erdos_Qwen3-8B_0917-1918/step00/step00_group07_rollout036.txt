```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from scipy.interpolate import interp1d
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[0:n_points//2] = 1.0
    h = h / (n_points / 2)
    
    def objective_function(h_flat):
        h = h_flat.reshape(n_points)
        shifted_h = np.roll(h, shift=1)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5

    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    bounds = Bounds(np.zeros(n_points), np.ones(n_points))

    result = minimize(objective_function, h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, options={'maxiter': 1000, 'disp': False})
    h_result = result.x
    c5_bound = result.fun
    return h_result, c5_bound, n_points
```
