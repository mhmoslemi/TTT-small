```python
import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    h[n_points//2:] = 0.0
    best_c5 = float('inf')
    
    for _ in range(100):
        t0 = time.time()
        def objective(x):
            h[0] = x[0]
            for i in range(1, n_points):
                h[i] = (x[i] + np.random.normal(0, 1e-3)) * (1 - 1e-3)
            h = np.clip(h, 0, 1)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * (2.0 / n_points)
            return c5
        
        def constraint(x):
            return np.sum(x) - n_points / 2
        
        x0 = np.random.uniform(0, 1, n_points)
        res = scipy.optimize.minimize(objective, x0, method='L-BFGS-B',
                                      bounds=[(0, 1) for _ in range(n_points)],
                                      constraints=[{'type': 'eq', 'fun': constraint}])
        
        if res.success:
            h_eval = np.clip(res.x, 0, 1)
            corr = np.correlate(h_eval, 1 - h_eval, mode="full")
            c5 = np.max(corr) * (2.0 / n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_eval
            if time.time() - t0 > budget_s:
                break
    
    return best_h, best_c5, n_points
```
