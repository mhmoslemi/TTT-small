import numpy as np
import scipy.optimize as optimize
from scipy.sparse import dia_matrix
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 150
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    initial_h[0:30] = 1.0
    initial_h[30:60] = 0.5
    initial_h[60:90] = 0.0
    initial_h[90:] = 0.5
    initial_h = initial_h + np.random.normal(0, 0.05, n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        n = len(h)
        dx = 2.0 / n
        h_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(h_corr) * dx
        return c5
    
    res = minimize(objective, initial_h, method='L-BFGS-B', options={'maxiter': 100, 'gtol': 1e-5, 'ftol': 1e-5})
    h_values = res.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
