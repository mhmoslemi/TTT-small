import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import Bounds

def run(seed=42, budget_s=600):
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[45:90] = 1.0
    h = h + np.random.normal(0, 0.01, n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def integral(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def cost(x):
        x = x.reshape(n_points)
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        return integral(x)
    
    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), 0, n_points / 2)
    
    opt_res = minimize(cost, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6, options={'maxiter': 100})
    
    h_values = opt_res.x
    c5_bound = integral(h_values)
    return h_values, c5_bound, n_points
