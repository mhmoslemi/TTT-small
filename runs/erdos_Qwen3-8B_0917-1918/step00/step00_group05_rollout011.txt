```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * n_points / 2.0 / dx
    h = np.clip(h, 0, n_points / 2.0)
    
    def objective(h):
        h = h / dx
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def optimize_h(h):
        result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        return result.x
    
    start_time = time.time()
    h_opt = optimize_h(h)
    elapsed = time.time() - start_time
    
    if elapsed < budget_s:
        h_opt = optimize_h(h_opt)
    
    c5_bound = objective(h_opt)
    return h_opt, c5_bound, n_points
```
