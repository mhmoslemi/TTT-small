```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    def compute_c5(h):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def objective(x):
        return compute_c5(x)
    
    def constraint_sum(x):
        return np.sum(x) - (len(x) / 2)
    
    n_points = 100
    x0 = np.random.uniform(0, 1, n_points)
    cons = {'type': 'eq', 'fun': constraint_sum}
    result = opt.minimize(objective, x0, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
```
