```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 2 - 1
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(x):
        h = x.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), lb=0, ub=n_points / 2)
    
    result = optimize.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-8, 'maxiter': 100})
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
```
