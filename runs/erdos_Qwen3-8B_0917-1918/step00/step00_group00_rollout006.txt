```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        h = np.concatenate([h, h])
        corr = np.correlate(h, 1 - h[:n_points], mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def objective(x):
        h = np.array(x, dtype=np.float64)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate(h)
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    x0 = h.copy()
    result = optimize.minimize(objective, x0, method='SLSQP', constraints=[{'type': 'eq', 'fun': constraint}])
    best_h = result.x
    best_c5 = evaluate(best_h)
    return (best_h, best_c5, n_points)
```
