import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = initial_h_values * 2 / initial_h_values.max() if hasattr(initial_h_values, '__len__') else np.random.rand(n_points)
    initial_h = np.clip(initial_h, 0, 1)
    total = np.sum(initial_h) * dx
    initial_h = np.clip(initial_h * (1 / (total / dx)), 0, 1)

    def objective(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h) * dx
        h = np.clip(h * (1 / (total / dx)), 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def evaluate(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h) * dx
        h = np.clip(h * (1 / (total / dx)), 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def optimize_solution():
        x0 = initial_h
        result = optimize.minimize(objective, x0, method='L-BFGS-B', bounds=[(0, 1)] * n_points, options={'ftol': 1e-8, 'maxiter': 100})
        return result.x

    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    for _ in range(10):
        h = optimize_solution()
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
        if time.time() - start_time > budget_s - 10:
            break
    return best_h, best_c5, n_points
