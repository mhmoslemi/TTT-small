```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 128
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5
    h = h * 2.0 - 1.0
    h = np.clip(h, 0.0, 1.0)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(x):
        nonlocal n_points, dx
        h = x.reshape(n_points)
        shifted = np.zeros_like(h)
        for k in range(1, n_points):
            shifted[k:] = h[:n_points - k]
        overlap = np.sum(h * (1 - shifted))
        return np.max(overlap * dx)
    
    result = optimize.minimize(objective, h, method='L-BFGS-B', options={'maxiter': 100})
    best_h = result.x
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx
    return best_h, c5, n_points
```
