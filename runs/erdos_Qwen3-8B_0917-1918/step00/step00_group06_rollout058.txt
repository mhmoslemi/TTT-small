import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 128
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(x):
        nonlocal n_points, dx
        h = x
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def bounds(x):
        return np.array([0.0, 1.0])
    
    result = opt.minimize(
        objective,
        h,
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        method="L-BFGS-B",
        options={"maxiter": 100, "ftol": 1e-8, "gtol": 1e-8}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
