import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from scipy.interpolate import interp1d
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 128
    dx = 2.0 / n_points
    h = np.random.rand(n_points)
    h = h * 2 - 1
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(x):
        x = np.clip(x, 0, 1)
        x = x * (n_points / 2) / np.sum(x)
        corr = np.correlate(x, 1 - x, mode="full")
        c5 = np.max(corr) * dx
        return c5

    bounds = Bounds(0, 1)
    
    start_time = time.time()
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, tol=1e-8, options={'maxiter': 1000})
    best_h = result.x
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
