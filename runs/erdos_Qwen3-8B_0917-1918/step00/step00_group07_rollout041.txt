import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from scipy.sparse import diags
from sklearn.linear_model import Lasso
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Start with initial construction
    if 'initial_h_values' in globals():
        h = initial_h_values[:n_points]
    else:
        # Start with a simple step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # Normalize to satisfy the integral constraint
    total = np.sum(h) * dx
    h *= 1.0 / total
    
    def overlap_integral(h):
        h_reshaped = np.concatenate((h, h))
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x) * dx
        if total == 0:
            return float('inf')
        x /= total
        return overlap_integral(x)
    
    def bounds_func(x):
        return np.clip(x, 0, 1)
    
    # Setup for optimization
    def optimize():
        result = minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            method='L-BFGS-B',
            options={'maxiter': 100, 'disp': False}
        )
        if result.success:
            return result.x
        return h
    
    start_time = time.time()
    h_opt = optimize()
    elapsed = time.time() - start_time
    
    # Refine with L1 regularization
    def l1_objective(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x) * dx
        if total == 0:
            return float('inf')
        x /= total
        return overlap_integral(x)
    
    def l1_penalty(x):
        return np.sum(np.abs(x))
    
    result = minimize(
        lambda x: l1_objective(x) + 0.1 * l1_penalty(x),
        h_opt,
        bounds=[(0, 1) for _ in range(n_points)],
        method='L-BFGS-B',
        options={'maxiter': 50, 'disp': False}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h) * dx
    best_h /= total
    c5 = overlap_integral(best_h)
    
    return best_h.tolist(), c5, n_points
