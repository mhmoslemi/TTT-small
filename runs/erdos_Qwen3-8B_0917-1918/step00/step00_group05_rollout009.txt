import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as optimize
from scipy.optimize import LinearConstraint, Bounds
from scipy.sparse import eye, csr_matrix
from scipy.sparse.linalg import lsqr
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points//2] = 1
    h = h / (n_points / 2)
    
    def compute_overlap(h):
        h_reshaped = np.resize(h, (n_points, 1))
        h_shifted = np.resize(h, (1, n_points))
        h_convolved = np.convolve(h, 1 - h, mode='full')
        c5 = np.max(h_convolved * dx)
        return c5
    
    def objective(x):
        h = x.reshape(n_points)
        return compute_overlap(h)
    
    def constraint(x):
        h = x.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    cons = LinearConstraint(np.ones(n_points), 0, 0, name='sum')
    bounds = Bounds(0, 1, name='bounds')
    
    x0 = h.copy()
    result = optimize.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
    
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    return (best_h, best_c5, n_points)
