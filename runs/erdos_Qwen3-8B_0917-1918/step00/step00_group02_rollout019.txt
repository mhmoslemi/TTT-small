```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / len(h))
    
    def update_h(h, step=0.01):
        h_new = h.copy()
        for i in range(len(h)):
            if h[i] > 0:
                h_new[i] = np.clip(h[i] - step, 0, 1)
            elif h[i] < 1:
                h_new[i] = np.clip(h[i] + step, 0, 1)
        return h_new
    
    n_points = 200
    h = np.full(n_points, 0.5)
    total = h.sum()
    h = np.clip(h * (n_points / 2) / total, 0, 1)
    
    best_h = h.copy()
    best_c5 = objective(h)
    
    start_time = time.time()
    
    for _ in range(10000):
        h = update_h(h)
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
```
