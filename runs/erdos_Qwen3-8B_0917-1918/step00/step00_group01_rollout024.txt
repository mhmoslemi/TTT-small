import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    h = np.random.rand(n_points) * 2 - 1
    h = np.clip(h, 0, 1)
    h = np.roll(h, int(np.random.rand() * n_points))
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h if sum_h != 0 else h
    
    def compute_c5(h):
        h_pad = np.pad(h, (0, n_points), mode='constant')
        corr = np.correlate(h_pad, 1 - h, mode='full')
        corr = corr[n_points - 1 : -n_points + 1]
        return np.max(corr) * (2.0 / n_points)
    
    def objective(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        x = x * (n_points / 2) / sum_x if sum_x != 0 else x
        return compute_c5(x)
    
    def constraint(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        x = x * (n_points / 2) / sum_x if sum_x != 0 else x
        return np.sum(x) - n_points / 2
    
    result = scipy.optimize.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, constraints=[{'type': 'eq', 'fun': constraint}])
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    sum_best = np.sum(best_h)
    best_h = best_h * (n_points / 2) / sum_best if sum_best != 0 else best_h
    c5_bound = compute_c5(best_h)
    return best_h, c5_bound, n_points
