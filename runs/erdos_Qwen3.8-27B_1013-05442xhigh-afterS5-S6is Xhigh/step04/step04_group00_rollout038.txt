Another approach: **Alternating minimization**. Fix which lags are active, optimize h to equalize the active lags, then update the active set. This is similar to the "bisection" or "parametric" approach.

Specifically:
- Given a target value τ, check if there exists h such that g[k] ≤ τ for all k. This is a feasibility problem.
- The feasibility problem can be written as:
  - 0 ≤ h[i] ≤ 1
  - Σ h[i] = n/2
  - g[k] ≤ τ for all k (these are bilinear constraints, so not convex)

Since the constraints are bilinear, this isn't directly a convex feasibility problem. However, if we linearize around the current solution, we get a convex subproblem.

**Sequential Linear Programming (SLP) approach**:
1. Start with current h.
2. Linearize g[k] ≈ g[k]_current + ∇g[k]_current · (h - h_current) for all k.
3. Solve the LP: minimize max_k (linearized g[k]) subject to 0≤h≤1, Σh=n/2.
4. This is a convex (LP) problem! Can be solved with CVXPY.
5. Update h, repeat.

This should converge to a local optimum (or at least make progress). The key advantage is that each subproblem is a convex LP that can be solved reliably.

```python
import numpy as np
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, 'full') * dx))

def compute_grads(h, n):
    """Compute gradient of all overlap values w.r.t. h. Returns shape (2n-1, n)."""
    dx = 2.0 / n
    grads = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        s = k - (n-1)
        if s >= 0:
            # valid j for first term: 0 to n-1-s
            # grad[j] = dx * (1 - h[j+s]) for j in [0, n-1-s]
            # grad[j] -= dx * h[j-s] for j in [s, n-1]
            j = np.arange(0, n-s)
            grads[k, j] += dx * (1 - h[j+s])
            j2 = np.arange(s, n)
            grads[k, j2] -= dx * h[j2-s]
        else:
            sp = -s
            j = np.arange(sp, n)
            grads[k, j] += dx * (1 - h[j-sp])
            j2 = np.arange(0, n-sp)
            grads[k, j2] -= dx * h[j2+sp]
    return grads

def slp_step(h, n, max_iter=10):
    """One SLP iteration: linearize and solve LP."""
    dx = 2.0 / n
    g = np.correlate(h, 1-h, 'full') * dx
    grads = compute_grads(h, n)
    
    h_var = cp.Variable(n)
    t = cp.Variable()
    g_lin = g + grads @ (h_var - h)
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n/2,
        t >= g_lin
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        prob.solve(solver=cp.GLOP, max_time=30)
    except:
        try:
            prob.solve(solver=cp.CBC, max_time=30)
        except:
            return h, g.max()
    
    if h_var.value is None:
        return h, g.max()
    
    h_new = np.clip(h_var.value, 0, 1)
    c = compute_c5(h_new, n)
    return h_new, c
```

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, 'full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break

            s = room.sum()
            if s < 1e-14:

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_matrix(h, n):
    """Returns matrix G where G[k,j] = d g[k] / d h[j]. Shape (2n-1, n)."""
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    ones = np.ones(n)
    for k in range(2*n-1):
        s = k - (n-1)
        if s >= 0:
            # d/dh[j] of dx * sum_{i=0}^{n-1-s} h[i](1-h[i+s])
            # = dx * (1 - h[j+s]) for j in [0, n-1-s]
            # - dx * h[j-s] for j in [s, n-1]
            G[k, :n-s] += dx * (1 - h[s:n])
            G[k, s:] -= dx * h[:n-s]
        else:
            sp = -s
            G[k, sp:] += dx * (1 - h[:n-sp])
            G[k, :n-sp] -= dx * h[sp:]
    return G

def slp_iteration(h, n, solver=None):
    """One SLP step."""
    dx = 2.0 / n
    g = np.correlate(h, 1-h, 'full') * dx
    G = compute_grad_matrix(h, n)
    
    h_var = cp.Variable(n)
    t = cp.Variable()
    g_lin = g + G @ (h_var - h)
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n/2,
        t >= g_lin
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        if solver is None:
            prob.solve(solver=cp.GLOP)
        else:
            prob.solve(solver=solver)
    except:
        try:
            prob.solve(solver=cp.CBC)
        except:
            return h, None
    
    if h_var.value is None:
        return h, None
    return np.clip(h_var.value, 0, 1), None

import numpy as np
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, 'full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_matrix(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        s = k - (n - 1)
        if s >= 0:
            G[k, :n - s] += dx * (1 - h[s:])
            G[k, s:] -= dx * h[:n - s]
        else:
            sp = -s
            G[k, sp:] += dx * (1 - h[:n - sp])
            G[k, :n - sp] -= dx * h[sp:]
    return G

def slp_solve(h, n, solver=None):
    g = np.correlate(h, 1 - h, 'full') * (2.0 / n)
    G = compute_grad_matrix(h, n)
    h_var = cp.Variable(n)
    t = cp.Variable()
    g_lin = g + G @ (h_var - h)
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0, t >= g_lin]
    prob = cp.Problem(cp.Minimize(t), cons)
    solved = False
    for solver_try in (solver, cp.GLOP, cp.CBC, cp.GLPK):
        if solver_try is None:
            continue
        try:
            prob.solve(solver=solver_try, max_time=60)
            if h_var.value is not None:
                return np.clip(h_var.value, 0, 1)
        except Exception:
            continue
    return None

def slp_optimize(h, n, deadline, max_iter=300, alpha_init=0.7):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = alpha_init
    for it in range(max_iter):
        if time.time() > deadline:
            break
        h_lp = slp_solve(h, n)
        if h_lp is None:
            break
        h_new = (1 - alpha) * h + alpha * h_lp
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h_new.copy()
        h = h_new
        if it % 20 == 19:
            alpha = max(0.2, alpha * 0.9)
    return best_h, best_c

def subgrad_optimize(h, n, deadline, max_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.05
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = np.correlate(h, 1 - h, 'full') * (2.0 / n)
        m = g.max()
        active = np.where(g >= m - 1e-8)[0]
        G = compute_grad_matrix(h, n)
        grad = G[active].mean(axis=0)
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            step = lr * f / ng
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-11:
                h = ht
                found = True
                break
        if not found:
            lr *= 0.92
            if lr < 1e-13:
                break
        else:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            if it % 500 == 499:
                lr = min(lr * 1.1, 0.1)
    return best_h, best_c

def pair_descent(h, n, deadline, max_sweeps=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        G = compute_grad_matrix(h, n)
        g = np.correlate(h, 1 - h, 'full') * (2.0 / n)
        m = g.max()
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] <= 1e-10 or h[i] >= 1 - 1e-10:
                continue
            for j in range(n - 1, i, -1):
                if h[j] <= 1e-10 or h[j] >= 1 - 1e-10:
                    continue
                dg = G[:, i] - G[:, j]
                eps_lo = max(-h[i] + 1e-12, 0)
                eps_hi = min(1 - h[i] - 1e-12, h[j] - 1e-12)
                if eps_lo >= eps_hi:
                    continue
                best_eps = 0.0
                best_val = m
                npts = 12
                for eps in np.linspace(0, eps_hi, npts):
                    val = (g + eps * dg).max()
                    if val < best_val:
                        best_val = val
                        best_eps = eps
                for eps in np.linspace(eps_lo, 0, npts):
                    val = (g + eps * dg).max()
                    if val < best_val:
                        best_val = val
                        best_eps = eps
                if best_val < m - 1e-13:
                    h[i] += best_eps
                    h[j] -= best_eps
                    g = g + best_eps * dg
                    m = g.max()
                    improved = True
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def ap_project(h, n):
    if n % 2 != 0:
        return h
    hh = (h[:n // 2] + (1 - h[n // 2:])) / 2.0
    out = np.zeros(n)
    out[:n // 2] = hh
    out[n // 2:] = 1 - hh
    return project_feasible(out, n)
