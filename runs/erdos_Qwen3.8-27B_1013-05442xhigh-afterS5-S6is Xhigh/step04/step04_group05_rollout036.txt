```python
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = beta * (corr - m)
    x = np.clip(x, -400, 400)
    exp_x = np.exp(x)
    w = exp_x / exp_x.sum()  # weights for each lag
    
    # Gradient: for each lag k (index in corr array, offset from n-1)
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if w[idx] < 1e-15:
            continue
        lag = idx - (n - 1)  # actual shift
        # ∂c_idx/∂h_i = (1-h[i+lag]) - h[i-lag] (with bounds)
        if lag == 0:

        elif lag > 0:

            grad[:n-lag] -= w[idx] * h[lag:]

            grad[:n-m_abs] += w[idx] * (1.0 - h[m_abs:])
            grad[m_abs:] -= w[idx] * h[:n-m_abs]
    
    return grad * dx
```

```python
def project_h_exact(v, target_sum):
    # Project v onto {0≤h≤1, ∑h = target_sum}
    # Using bisection on the shift parameter
    lo = v.min() - target_sum  # if all at 0, we need to add
    hi = v.max() + 1.0        # if all at 1, we need to subtract
    # Actually: h_i = clamp(v_i - lambda, 0, 1), find lambda such that sum = target
    # f(lambda) = sum(clamp(v_i - lambda, 0, 1)) is decreasing in lambda
    # f(-inf) = n, f(+inf) = 0
    
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(v - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2, 0, 1)
    # Final correction
    diff = target_sum - h.sum()
    if abs(diff) > 1e-15:
        free = (h > 0) & (h < 1)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0, 1)
    return h
```

```python
        if mask.any():

            h = np.clip(h, 0.0, 1.0)

def smooth_max_grad(h, beta, dx):

    corr = np.correlate(h, 1.0 - h, mode="full")

    e = np.exp(x)

        wi = w[idx]
        if wi < 1e-15:
            continue

            grad += wi * (1.0 - 2.0 * h)

        else:
            ma = -lag
            grad[:n-ma] += wi * (1.0 - h[ma:])
            grad[ma:] -= wi * h[:n-ma]
    return grad * dx
```
