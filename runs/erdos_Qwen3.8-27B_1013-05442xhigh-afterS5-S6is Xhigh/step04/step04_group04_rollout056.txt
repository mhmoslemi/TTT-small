import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(x_shift)
    sum_exp = np.sum(exp_vals)
    return (m + np.log(sum_exp) / beta) * dx

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(x_shift)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for k in active:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            m2 = -lag
            if m2 < n:
                grad[:n - m2] += w * (1.0 - h[m2:])
                grad[m2:] -= w * h[:n - m2]
    return grad

def slsqp_round(h, beta, maxiter, time_fn):
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                    'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        return smooth_max_val(x, beta)
    def grad(x):
        return smooth_max_grad(x, beta)
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, jac=grad,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_h(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h):
            return h_new
    except Exception:
        pass
    return h

def projected_gradient(h, target, n_iter, time_fn, beta, 
                        lr0=0.01, momentum=0.9, perturb_every=500, perturb_scale=0.005):
    n = len(h)
    v = np.zeros(n)  # momentum
    c5_cur = compute_c5(h)
    no_imp = 0
    
    for it in range(n_iter):
        if time_fn() < 3:
            break
        
        grad = smooth_max_grad(h, beta)
        
        # Nesterov momentum
        v = momentum * v - lr0 * grad
        h_new = project_h(h + v, target)
        
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-15:
            h = h_new
            c5_cur = c5_new
            no_imp = 0
        else:
            no_imp += 1
            # Revert
            v = momentum * v  # keep momentum direction
        
        # Decay learning rate
        lr0 *= 0.9995
        
        # Periodic perturbation
        if no_imp > 0 and no_imp % perturb_every == perturb_every - 1:

            c5_pert = compute_c5(h_pert)
            if c5_pert < c5_cur:
                h = h_pert
                c5_cur = c5_pert
                no_imp = 0
                v = np.zeros(n)
                lr0 = 0.01  # reset lr
        
        if no_imp > 10 * perturb_every:
            break
    
    return h, c5_cur

import numpy as np
from scipy.optimize import minimize
import time

_deadline = [0.0]
_beta_ref = [100.0]
_target_ref = [250.0]

def time_left():
    return _deadline[0] - time.time()

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(x_shift)
    sum_exp = np.sum(exp_vals)
    return (m + np.log(sum_exp) / beta) * dx

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(x_shift)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for k in active:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            m2 = -lag
            if m2 < n:
                grad[:n - m2] += w * (1.0 - h[m2:])
                grad[m2:] -= w * h[:n - m2]
    return grad

def _sobj(x):
    return smooth_max_val(x, _beta_ref[0])

def _sgrad(x):
    return smooth_max_grad(x, _beta_ref[0])

def _scon(x):
    return x.sum() - _target_ref[0]

def _scon_jac(x):
    return np.ones(len(x))

def slsqp_round(h, beta, maxiter):
    n = len(h)
    target = n / 2.0
    _beta_ref[0] = beta
    _target_ref[0] = target
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _scon, 'jac': _scon_jac}]
    c5_cur = compute_c5(h)
    try:
        res = minimize(_sobj, h, method='SLSQP', bounds=bounds,
                       jac=_sgrad, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_h(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-15:
            return h_new, c5_new
    except Exception:
        pass
    return h, c5_cur

def momentum_gradient(h, target, n_iter, beta, lr0, momentum,
                      perturb_every, perturb_scale):
    n = len(h)
    v = np.zeros(n)
    lr = lr0
    c5_cur = compute_c5(h)
    no_imp = 0
    for it in range(n_iter):
        if time_left() < 3:
            break
        grad = smooth_max_grad(h, beta)
        v = momentum * v - lr * grad
        h_new = project_h(h + v, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-15:
            h = h_new
            c5_cur = c5_new
            no_imp = 0
        else:
            no_imp += 1
        if it % 200 == 199:
            lr *= 0.97
        if perturb_every > 0 and no_imp > 0 and no_imp % perturb_every == perturb_every - 1:
            scale = perturb_scale * (1.0 + 0.15 * (no_imp // perturb_every))
            h_pert = project_h(h + np.random.randn(n) * scale, target)
            c5_pert = compute_c5(h_pert)
            if c5_pert < c5_cur:
                h = h_pert
                c5_cur = c5_pert
                no_imp = 0
                v = np.zeros(n)
                lr = lr0
        if perturb_every > 0 and no_imp > 25 * perturb_every and lr < lr0 * 0.05:
            break
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    global _deadline
    np.random.seed(seed)
    start_time = time.time()
    _deadline[0] = start_time + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(500, 0.5)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    target = n / 2.0

    # Phase 1: SLSQP moderate beta
    for beta in [30, 100, 500, 2000]:
        if time_left() < 50:
            break
        h_new, c5_new = slsqp_round(best_h, beta, maxiter=400)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Momentum gradient main phase
    if time_left() > 80:
        best_h, best_c5 = momentum_gradient(best_h, target,
                                             n_iter=200000, beta=5000,
                                             lr0=0.006, momentum=0.9,
                                             perturb_every=350,
                                             perturb_scale=0.004)

    # Phase 3: SLSQP high beta
    if time_left() > 40:
        for beta in [10000, 50000]:
            if time_left() < 25:
                break
            h_new, c5_new = slsqp_round(best_h, beta, maxiter=600)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 4: Fine momentum gradient
    if time_left() > 50:
        best_h, best_c5 = momentum_gradient(best_h, target,
                                             n_iter=150000, beta=30000,
                                             lr0=0.002, momentum=0.95,
                                             perturb_every=500,
                                             perturb_scale=0.002)

    # Phase 5: SLSQP very high beta
    if time_left() > 25:
        h_new, c5_new = slsqp_round(best_h, 100000, maxiter=800)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 6: Ultra-fine gradient
    if time_left() > 15:
        best_h, best_c5 = momentum_gradient(best_h, target,
                                             n_iter=80000, beta=100000,
                                             lr0=0.0008, momentum=0.97,
                                             perturb_every=800,
                                             perturb_scale=0.001)

    # Phase 7: Final SLSQP polish
    if time_left() > 12:
        h_new, c5_new = slsqp_round(best_h, 200000, maxiter=600)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Final cleanup
    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
