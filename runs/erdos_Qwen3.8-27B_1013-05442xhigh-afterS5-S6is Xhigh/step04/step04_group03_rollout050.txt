from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def compute_lags_full(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample_fn(h, n_new):
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def binary_snap(h, n):
    n_ones = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

    n = len(h_bin)
    cur = h_bin.copy()
    lags = compute_lags_full(cur)
    cur_max = float(np.max(lags))
    best_max = cur_max
    best_h = cur.copy()
    rng = np.random.default_rng(rng_seed)
    stall = 0

        if budget - (time.time() - t0) < 5:
            break

def slsqp_smooth(h, beta, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    x_shift = np.clip(beta * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(x_shift)) / len(c)) / beta) * (2.0 / n)

def sum_constraint(x, target):
    return np.sum(x) - target

def sum_jac(x, target):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, maxiter=10000):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_smooth, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(r.x, target)
    except Exception:
        return h0

**Key efficiency improvement**: Instead of computing the full correlation after each swap, compute the CHANGE in the correlation analytically.

When we swap positions a (1→0) and b (0→1):
For lag k ≥ 0: O(k) = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
Change in O(k) due to swap:
- Position a appears as h[i+k] when i+k=a, i.e., i=a-k (if 0≤a-k≤n-1-k)
  And as (1-h[i]) when i=a.
- Position b appears as h[i+k] when i=b-k (if 0≤b-k≤n-1-k)
  And as (1-h[i]) when i=b.

So the change in O(k) is:
δO(k) = [contribution of b as h[i+k] at i=b-k] - [contribution of a as h[i+k] at i=a-k]
       - [contribution of a in (1-h[i]) at i=a] + [contribution of b in (1-h[i]) at i=b]

More precisely:
- Before swap: h[a]=1, h[b]=0
- After swap: h'[a]=0, h'[b]=1
