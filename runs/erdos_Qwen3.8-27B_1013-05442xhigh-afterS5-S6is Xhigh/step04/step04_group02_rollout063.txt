import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_grad(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(shifted)
    ex_sum = ex.sum()
    w = ex / ex_sum
    grad = np.zeros(n)
    # Only process significant lags
    thresh = 1e-8
    for k in range(len(corr)):
        wk = w[k]
        if wk < thresh:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wk * (1.0 - h[lag:])
            grad[lag:] -= wk * h[:n - lag]
        else:
            la = -lag
            grad[la:] += wk * (1.0 - h[:n - la])
            grad[:n - la] -= wk * h[la:]
    obj = m + np.log(ex_sum) / beta
    return obj, grad

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - ts) < 1e-14:
            break
        d = ts - s
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        c = mask.sum()
        if c == 0:
            break
        h[mask] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def projected_momentum(h, n, deadline, seed, b0=20, b1=50000, niter=15000):
    np.random.seed(seed)
    ts = n / 2.0
    h = project_feasible(h.copy(), ts)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    vel = np.zeros(n)
    lr = 0.015
    mom = 0.8
    stall = 0
    for it in range(niter):
        if time.time() > deadline:
            break
        frac = min(it / niter, 1.0)
        beta = b0 * (b1 / b0) ** frac
        obj, grad = smooth_max_grad(h, n, beta)
        vel = mom * vel - lr * grad
        h_new = project_feasible(h + vel, ts)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5
            h = h_new
            stall = 0
            lr = min(lr * 1.003, 0.05)
        else:
            stall += 1
            vel *= 0.5
            lr *= 0.96
        if stall > 500 and lr < 1e-5:
            break
    return best_h, best_c5

def slsqp_refine(h, n, deadline, beta=100000):
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        return (m + np.log(np.exp(sh).sum()) / beta) * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 5000, 'ftol': 1e-15})
        h_new = project_feasible(res.x, ts)
        c5 = compute_c5(h_new)
        return h_new, c5
    except:
        return h, compute_c5(h)

        h0 = project_feasible(h0, n_try / 2.0)

    return best_h, best_c5, best_n

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_grad(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(shifted)
    ex_sum = ex.sum()
    w = ex / ex_sum
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wk * (1.0 - h[lag:])
            grad[lag:] -= wk * h[:n - lag]
        else:
            la = -lag
            grad[la:] += wk * (1.0 - h[:n - la])
            grad[:n - la] -= wk * h[la:]
    obj = m + np.log(ex_sum) / beta
    return obj, grad

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - ts) < 1e-14:
            break
        d = ts - s
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        c = mask.sum()
        if c == 0:
            break
        h[mask] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_max_grad(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(shifted)
    ex_sum = ex.sum()
    w = ex / ex_sum
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wk * (1.0 - h[lag:])
            grad[lag:] -= wk * h[:n - lag]
        else:
            la = -lag
            grad[la:] += wk * (1.0 - h[:n - la])
            grad[:n - la] -= wk * h[la:]
    obj = m + np.log(ex_sum) / beta
    return obj, grad
