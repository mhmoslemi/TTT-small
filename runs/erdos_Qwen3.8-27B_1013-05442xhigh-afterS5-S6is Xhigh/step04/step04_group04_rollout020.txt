```python
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum} using bisection."""
    h = np.clip(h, 0.0, 1.0)
    # Bisection on the threshold
    left, right = -2.0, 2.0
    for _ in range(100):
        mid = (left + right) / 2
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            left = mid
        else:
            right = mid
    projected = np.clip(h - (left+right)/2, 0.0, 1.0)
    # Fine-tune
    diff = target_sum - projected.sum()
    if abs(diff) > 1e-14:
        free = (projected > 1e-12) & (projected < 1.0 - 1e-12)
        if free.sum() > 0:
            projected[free] += diff / free.sum()
            projected = np.clip(projected, 0.0, 1.0)
    return projected

def subgradient_c5(h, top_k=None):
    """Compute C5 value and subgradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    c5 = float(m * dx)
    
    if top_k is None:
        top_k = 5
    top_idx = np.argsort(corr)[::-1][:top_k]
    
    grad = np.zeros(n)

        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            grad[:n - m2] += (1.0 - h[m2:])
            grad[m2:] -= h[:n - m2]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-13:
        free = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if free.sum() > 0:
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

_state = {'beta': 100.0, 'target': 100.0}

def _smf(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    beta = _state['beta']
    m = np.max(corr)
    xs = np.clip(beta * (corr - m), -500, 500)
    ev = np.exp(xs)
    se = ev.sum()
    val = (m + np.log(se) / beta) * dx
    w = ev / se
    grad = np.zeros(n)
    active = np.where(w > 1e-9)[0]
    for k in active:
        wk = w[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * x)
        elif 0 < lag < n:
            grad[lag:] += wk * (1.0 - x[:-lag])
            grad[:n-lag] -= wk * x[lag:]
        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            grad[:n-m2] += wk * (1.0 - x[m2:])
            grad[m2:] -= wk * x[:n-m2]
    return val

def _smg(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    beta = _state['beta']
    m = np.max(corr)
    xs = np.clip(beta * (corr - m), -500, 500)
    ev = np.exp(xs)
    se = ev.sum()
    w = ev / se
    grad = np.zeros(n)
    active = np.where(w > 1e-9)[0]
    for k in active:
        wk = w[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * x)
        elif 0 < lag < n:
            grad[lag:] += wk * (1.0 - x[:-lag])
            grad[:n-lag] -= wk * x[lag:]
        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            grad[:n-m2] += wk * (1.0 - x[m2:])
            grad[m2:] -= wk * x[:n-m2]
    return grad

def _ceq(x):
    return x.sum() - _state['target']

def _ceqj(x):
    return np.ones(len(x))

def do_slsqp(h0, betas, maxiter, tl):
    n = len(h0)
    target = n / 2.0
    _state['target'] = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _ceq, 'jac': _ceqj}]
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    for beta in betas:
        if tl() < 20:
            break
        _state['beta'] = beta
        try:
            res = minimize(_smf, h_cur, method='SLSQP', bounds=bounds,
                          jac=_smg, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-16:
                h_cur, c5_cur = h_new, c5_new
        except Exception:
            pass
    return h_cur, c5_cur

def subgrad_opt(h, target, n_iter, tl, top_k, steps):
    n = len(h)
    c5_cur = compute_c5(h)
    no_imp = 0
    for it in range(n_iter):
        if tl() < 3:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        nk = min(top_k, n)
        top_idx = np.argsort(corr)[::-1][:nk]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * h)
            elif 0 < lag < n:
                grad[lag:] += (1.0 - h[:-lag])
                grad[:n-lag] -= h[lag:]
            elif lag < 0 and abs(lag) < n:
                m2 = -lag
                grad[:n-m2] += (1.0 - h[m2:])
                grad[m2:] -= h[:n-m2]
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        found = False
        for st in steps:
            ht = project_h(h - st * grad, target)
            c5t = compute_c5(ht)
            if c5t < c5_cur - 1e-16:
                h, c5_cur = ht, c5t
                found = True
                no_imp = 0
                break
        if not found:
            no_imp += 1
            if no_imp % 300 == 299:
                sc = 0.003 * min(5, no_imp // 300 + 1)
                ht = project_h(h + np.random.randn(n) * sc, target)
                if compute_c5(ht) < c5_cur:
                    h, c5_cur, no_imp = ht, compute_c5(ht), 0
            if no_imp > n_iter // 8:
                break
    return h, c5_cur

def local_search(h, target, n_trials, tl, scale):
    """Random pair-swap local search."""
    n = len(h)
    c5_cur = compute_c5(h)
    rng = np.random.default_rng(99)
    for _ in range(n_trials):
        if tl() < 2:
            break
        i = rng.integers(n)
        j = (i + 1 + rng.integers(n - 1)) % n
        delta = rng.uniform(-scale, scale)
        hi_new = h[i] + delta
        hj_new = h[j] - delta
        if 0 <= hi_new <= 1 and 0 <= hj_new <= 1:
            h[i], h[j] = hi_new, hj_new
            if compute_c5(h) < c5_cur - 1e-15:
                c5_cur = compute_c5(h)
            else:
                h[i], h[j] = hi_new - delta, hj_new + delta
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)
    n_best = len(best_h)

    # Phase 1: SLSQP with annealing
    h1, c5_1 = do_slsqp(best_h, [200, 1000, 5000, 20000, 50000, 100000], 3000, tl)
    if c5_1 < best_c5:
        best_h, best_c5, n_best = h1, c5_1, len(h1)

    # Phase 2: Subgradient
    tgt = n_best / 2.0
    h2, c5_2 = subgrad_opt(best_h, tgt, 50000, tl, 5,
                           [0.03, 0.015, 0.008, 0.004, 0.002, 0.001])
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_2
        tgt = n_best / 2.0

    # Phase 3: Local search
    h3, c5_3 = local_search(best_h, tgt, 200000, tl, 0.02)
    if c5_3 < best_c5:
        best_h, best_c5 = h3, c5_3

    # Phase 4: Multi-resolution
    for n_res in [200, 300, 400, 600, 800]:
        if tl() < 60:
            break
        if n_res == n_best:
            continue
        x_old = np.linspace(0, 2, n_best)
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-13:
        free = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if free.sum() > 0:
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

_state = {'beta': 100.0, 'target': 100.0}

def _corr_and_weights(x):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode="full")
    beta = _state['beta']
    m = np.max(corr)
    xs = np.clip(beta * (corr - m), -500, 500)
    ev = np.exp(xs)
    se = ev.sum()
    return corr, m, ev / se

def _grad_from_weights(x, w):
    n = len(x)
    dx = 2.0 / n
    grad = np.zeros(n)
    active = np.where(w > 1e-9)[0]
    for k in active:
        wk = w[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * x)
        elif 0 < lag < n:
            grad[lag:] += wk * (1.0 - x[:-lag])
            grad[:n - lag] -= wk * x[lag:]
        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            grad[:n - m2] += wk * (1.0 - x[m2:])
            grad[m2:] -= wk * x[:n - m2]
    return grad

    return _grad_from_weights(x, w)

def _ceq(x):
    return x.sum() - _state['target']

def _ceqj(x):
    return np.ones(len(x))

def do_slsqp(h0, betas, maxiter, tl):

    target = n / 2.0
    _state['target'] = target

    cons = [{'type': 'eq', 'fun': _ceq, 'jac': _ceqj}]
    h_cur = h0.copy()

            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-16:

        except Exception:
            pass
    return h_cur, c5_cur
```
