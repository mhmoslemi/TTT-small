            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                jl = max(0, i - radius)
                jh = min(n, i + radius + 1)
                for j in range(jl, jh):
                    if i == j:
                        continue
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c

def optimize_one(h0, n, deadline, rng):
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # SLSQP phase with increasing alpha
    for a in [500, 2000, 10000, 50000, 200000]:
        if time.time() > deadline - 10:
            break
        ht = slsqp_opt(best_h, n, a, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c, best_h = c, ht

    # L-BFGS-B phase
    if time.time() < deadline - 15:
        for a, pen in [(50000, 100.0), (200000, 500.0), (500000, 2000.0)]:
            if time.time() > deadline - 10:
                break
            ht = lbgfb_opt(best_h, n, a, pen, maxiter=300)
            c = compute_c5(ht, n)
            if c < best_c - 1e-14:
                best_c, best_h = c, ht

    # Pair search
    if n <= 120:
        if time.time() < deadline - 5:
            best_h, best_c = pair_search(best_h, n, deadline - 3,
                [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002], max_passes=200)
    else:
        if time.time() < deadline - 5:
            best_h, best_c = near_pair(best_h, n, deadline - 3,
                [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005], radius=50)

    # Final SLSQP polish
    if time.time() < deadline - 2:
        ht = slsqp_opt(best_h, n, 1000000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c, best_h = c, ht

    if n <= 120 and time.time() < deadline - 1:
        best_h, best_c = pair_search(best_h, n, deadline - 0.3,
            [0.001, 0.0005, 0.0002, 0.0001], max_passes=100)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100

    best_h = None
    best_c = float('inf')
    x_init = np.linspace(0, 2, n_init)

    # Phase 1: Optimize at native resolution (40% of time)
    alloc1 = (deadline - t0) * 0.35
    h1, c1 = optimize_one(h_init, n_init, t0 + alloc1, rng)
    best_h, best_c = h1, c1

    # Phase 2: Try other resolutions (40% of time)
    n_list = [80, 100, 120, 150]
    n_list = [x for x in n_list if x != n_init]
    n_list.sort()  # try smaller first (faster pair search)
    phase2_start = time.time()
    phase2_end = deadline - 40
    total_p2 = phase2_end - phase2_start
    per_n = total_p2 / max(1, len(n_list))

    for idx, n_try in enumerate(n_list):
        if time.time() > phase2_end - 5:
            break
        alloc = per_n
        end_t = time.time() + alloc
        h_interp = np.interp(np.linspace(0, 2, n_try), x_init, h_init)
        h_interp = project_feasible(h_interp, n_try)
        ht, ct = optimize_one(h_interp, n_try, end_t, rng)
        if ct < best_c:
            best_c, best_h = ct, ht

    # Phase 3: Random restarts (remaining time minus buffer)
    remaining = deadline - time.time() - 20
    if remaining > 15 and best_h is not None:
        n_best = len(best_h)
        n_restarts = max(2, int(remaining / 30))
        scales = [0.01, 0.03, 0.05, 0.08, 0.12, 0.02, 0.04, 0.06]
        for r in range(n_restarts):
            if time.time() > deadline - 18:
                break
            sc = scales[r % len(scales)]
            hp = best_h + rng.normal(0, sc, n_best)
            hp = project_feasible(hp, n_best)
            alloc = min(35, (deadline - time.time() - 15) * 0.4)
            if alloc < 8:
                break
            ht, ct = optimize_one(hp, n_best, time.time() + alloc, rng)
            if ct < best_c:
                best_c, best_h = ct, ht

    # Phase 4: Final polish
    if best_h is not None and time.time() < deadline - 1:
        n_f = len(best_h)
        if n_f <= 120:
            best_h, best_c = pair_search(best_h, n_f, deadline - 0.2,
                [0.0005, 0.0002, 0.0001, 0.00005], max_passes=200)
        else:
            best_h, best_c = near_pair(best_h, n_f, deadline - 0.2,
                [0.0005, 0.0002, 0.0001], radius=30)
        best_h = project_feasible(best_h, n_f)
        best_c = compute_c5(best_h, n_f)

    if best_h is None:
        best_h = project_feasible(np.full(100, 0.5), 100)
        best_c = compute_c5(best_h, 100)

    return best_h, best_c, len(best_h)
