import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, dx, beta):
    c = corr * dx
    m = np.max(c)
    return m + np.log(np.sum(np.exp(np.clip(beta*(c-m), -500, 500)))) / beta

def compute_grad(h, beta, dx, n_lags=30):
    """Compute gradient of smooth max of correlations."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Get top lags
    idx = np.argsort(corr)[::-1][:n_lags]
    m = corr[idx[0]]
    w = np.exp(np.clip(beta * (corr[idx] - m), -500, 500))
    w /= w.sum()
    
    grad = np.zeros(n)
    for k_idx, wt in zip(idx, w):
        lag = int(k_idx) - (n - 1)
        if lag == 0:
            grad += wt * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wt * (1.0 - h[:n-lag])
            grad[:n-lag] -= wt * h[lag:]
        else:
            m2 = -lag
            grad[:n-m2] += wt * (1.0 - h[m2:])
            grad[m2:] -= wt * h[:n-m2]
    return grad * dx

def projected_grad_descent(h, target, dx, t0, budget_s, 
                            beta_start=50, beta_end=5000, 
                            n_iter=5000, momentum=0.9,
                            step_init=0.01, n_lags=30):
    """Project gradient descent with momentum for the smooth max objective."""
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    v = np.zeros(n)
    
    for it in range(n_iter):
        elapsed = time.time() - t0
        if elapsed > budget_s - 5:
            break
        
        beta = beta_start * (beta_end / beta_start) ** (it / max(1, n_iter-1))
        
        # Compute gradient
        g = compute_grad(best, beta, dx, n_lags)
        
        # Momentum update
        v = momentum * v - step_init * g
        d = v / (np.linalg.norm(v) + 1e-15)
        
        # Line search (backtracking)
        alpha = step_init
        improved = False
        for _ in range(20):
            h_new = project_h(best + alpha * d, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
                improved = True
                break
            alpha *= 0.5
        if not improved:
            # Try smaller step or different direction
            break
        
        # Decay step size
        step_init *= 0.999
    
    return best, best_c5

def project_h(h, target):
    """Project h to [0,1]^n with sum = target."""
