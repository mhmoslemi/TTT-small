    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, len(h))
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def smooth_max_obj(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    # Use log-sum-exp for numerical stability
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    sm = m + np.log(np.sum(np.exp(shifted)) / len(corr)) / beta
    return sm * dx

def smooth_max_grad(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)

    n = len(h)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def smooth_max(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(shifted)) / len(corr)) / beta) * dx

def sum_constraint(x, target):
    return np.sum(x) - target

def slsqp_opt(h0, n, target, beta, maxiter, ftol, t0, budget):
    dx = 2.0 / n
    def obj(x):
        return smooth_max(x, beta, dx)
    cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (target,)}]
    bounds = Bounds(np.zeros(n), np.ones(n))
    rem = budget - (time.time() - t0)
    if rem < 10:
        return h0
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def coord_descent(cur, target, n, t0, budget, max_sweeps=50):
    best_c5 = compute_c5(cur)
    for sw in range(max_sweeps):
        if time.time() - t0 > budget - 2:
            break
        improved = False
        # Adaptive step sizes
        if sw < 10:
            steps = [0.05, -0.05, 0.02, -0.02]
        elif sw < 30:
            steps = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
        else:
            steps = [0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
        for j in range(n):
            if time.time() - t0 > budget - 1:
                break
            hv = cur[j]
            for d in steps:
                nv = hv + d
                if nv < 0 or nv > 1:
                    continue
                trial = cur.copy()
                trial[j] = nv
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved and sw > 8:
            break
    return cur, best_c5

def pair_descent(cur, target, n, t0, budget):
    best_c5 = compute_c5(cur)
    rng = np.random.RandomState(777)
    for _ in range(200):
        if time.time() - t0 > budget - 2:
            break
        improved = False
        n_pairs = max(n // 3, 50)
        for _ in range(n_pairs):
            if time.time() - t0 > budget - 1:
                break
            j, k = rng.choice(n, 2, replace=False)
            for d in [0.03, -0.03, 0.01, -0.01, 0.005, -0.005]:

                if c < best_c5 - 1e-15:

    return cur, best_c5

def snap_binary(h, target):
    z = (h > 0.5).astype(float)
    diff = target - z.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(z == 0)[0]

            ones = np.where(z == 1)[0]

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def smooth_max_val(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    return (m + np.log(np.sum(np.exp(shifted)) / len(corr)) / beta) * dx

def sum_eq(x, target):
    return np.sum(x) - target

def slsqp_opt(h0, n, target, beta, maxiter, ftol, t0, budget):
    dx = 2.0 / n
    def obj(x):
        return smooth_max_val(x, beta, dx)
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (target,)}]
    bounds = Bounds(np.zeros(n), np.ones(n))
    if budget - (time.time() - t0) < 10:
        return h0
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def coord_descent(cur, target, n, t0, budget, max_sweeps=60):
    best_c5 = compute_c5(cur)
    for sw in range(max_sweeps):
        if time.time() - t0 > budget - 2:
            break
        improved = False
        if sw < 8:
            steps = [0.05, -0.05, 0.02, -0.02]
        elif sw < 25:
            steps = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
        else:
            steps = [0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001]
        for j in range(n):
            if time.time() - t0 > budget - 1:
                break
            hv = cur[j]
            for d in steps:
                nv = hv + d
                if nv < 0 or nv > 1:
                    continue
                trial = cur.copy()
                trial[j] = nv
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    cur, best_c5, improved = trial, c, True
                    break
        if not improved and sw > 10:
            break
    return cur, best_c5

def pair_descent(cur, target, n, t0, budget):
    best_c5 = compute_c5(cur)
    rng = np.random.RandomState(555)
    stalled = 0
    for _ in range(300):
        if time.time() - t0 > budget - 2:
            break
        improved = False
        n_pairs = max(n // 4, 30)
        for _ in range(n_pairs):
            if time.time() - t0 > budget - 1:
                break
            j, k = rng.choice(n, 2, replace=False)
            for d in [0.03, -0.03, 0.015, -0.015, 0.01, -0.01, 0.005, -0.005]:
                trial = cur.copy()
                trial[j] += d
                trial[k] -= d
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    cur, best_c5, improved = trial, c, True
                    break
        if improved:
            stalled = 0
        else:
            stalled += 1
            if stalled > 8:
                break
    return cur, best_c5

def snap_binary(h, target):
    z = (h > 0.5).astype(float)
    diff = target - z.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            zeros = np.where(z == 0)[0]
            if len(zeros) > 0:
                order = np.argsort(h[zeros])[::-1]
                n_add = min(int(np.ceil(diff)), len(zeros))
                z[zeros[order[:n_add]]] = 1.0
        else:
            ones = np.where(z == 1)[0]
            if len(ones) > 0:
                order = np.argsort(h[ones])
                n_rem = min(int(np.ceil(-diff)), len(ones))
                z[ones[order[:n_rem]]] = 0.0
    return project_h(z, target)

def anneal_step(cur, target, n, T, rng, t0, budget):
    """Simulated annealing: perturb a random subset and accept/reject."""
    n_perturb = max(1, n // (10 + int(T * 50)))
    idx = rng.choice(n, size=n_perturb, replace=False)
    trial = cur.copy()
    trial[idx] += rng.randn(n_perturb) * 0.05
    trial = project_h(trial, target)
    c_trial = compute_c5(trial)
    c_cur = compute_c5(cur)
    if c_trial <= c_cur:
        return trial, c_trial, True
    if T < 1e-8:
        return cur, c_cur, False
    if rng.random() < np.exp(-(c_trial - c_cur) / max(T, 1e-10)):
        return trial, c_trial, True
    return cur, c_cur, False

def run(seed=42, budget_s=1100, **kwargs):

    resolutions = sorted(set([base_n] + [r for r in [320, 384, 448, 512, 576]
                                        if 280 <= r <= 640]))

    for n_pts in resolutions:

        target = n_pts / 2.0

        for beta in [200, 500, 1200, 3000]:

            cur = slsqp_opt(cur, n_pts, target, 1500, 3000, 1e-15, t0, budget_s)
