import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def full_subgrad(h, n, threshold_frac):
    """Compute subgradient using all lags within threshold_fract of max."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    thr = mc - threshold_frac * (mc + 1e-10)
    idx = np.where(corr >= thr)[0]
    weights = np.maximum(corr[idx] - thr, 0) + 1e-10
    weights = weights / weights.sum()

    # Actually, we need the gradient with respect to h
    # d/dh[j] of C[k] where C[k] = sum_i h[i+k-(n-1)] * (1-h[i])
    # For a given lag k (in the full correlate output, index k means lag = k-(n-1)):
    # If lag >= 0: C = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
    #   dC/dh[j] = (1-h[j-lag]) for j in [lag, n-1] (from h[i+lag] term, i=j-lag)
    #              - h[j+lag] for j in [0, n-1-lag] (from (1-h[i]) term, i=j)
    # If lag < 0: let la = -lag
    #   C = sum_{i=0}^{n-1-la} h[i]*(1-h[i+la])
    #   dC/dh[j] = (1-h[j+la]) for j in [0, n-1-la] (from h[i] term, i=j)
    #              - h[j-la] for j in [la, n-1] (from (1-h[i]) term, i=j-la)
    
    g = np.zeros(n)

        else:
            la = -lag
            g[:n - la] += w * (1.0 - h[la:])
            g[la:] -= w * h[:n - la]
    return g, mc, corr

def nesterov_full(h0, n, target, n_iter, step0, t0, budget, threshold_frac):

    h_prev = h0.copy()
    y = h0.copy()

    step = step0

        h_new = project_h(y - step * g, target)
        mu = (1.0 - np.sqrt(1.0 + 4.0 / ((it + 1) ** 2))) / 2.0

def sml_obj(h_in, beta, dx):
    corr = np.correlate(h_in, 1.0 - h_in, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(np.clip(beta * (corr - m), -500, 500))) / len(corr)) / beta) * dx

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]

                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def subgrad_loop(h0, n, target, t0, budget, n_iter, step, temp):
    """Subgradient with fixed params, returns best found."""
    h = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    for it in range(n_iter):
        if time.time() - t0 > budget - 1:

        g, mc = weighted_subgrad(h, n, temp, 1e-6)
        gn = np.linalg.norm(g)
        if gn < 1e-14:

        h_new = project_h(h - step * g / gn, target)
        c = compute_c5(h_new)
        if c < best_c5:
            best_h, best_c5 = h_new.copy(), c
        h = h_new

def nesterov_loop(h0, n, target, t0, budget, n_iter, step0, temp):
    h = h0.copy()
    h_prev = h.copy()
    y = h.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    step = step0
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget - 1:
            break
        g, mc = weighted_subgrad(y, n, temp, 1e-6)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        h_new = project_h(y - step * g / gn, target)
        mu = (1.0 - np.sqrt(1.0 + 4.0 / ((it + 1) ** 2))) / 2.0
        y = np.clip(h_new + mu * (h_new - h_prev), 0, 1)
        h_prev, h = h, h_new
        c = compute_c5(h)
        if c < best_c5 - 1e-16:
            best_h, best_c5, stall = h.copy(), c, 0
        else:
            stall += 1
            if stall % 50 == 0:
                step *= 0.85
    return best_h, best_c5

def coord_desc(cur, target, n, t0, budget):
    best_c5 = compute_c5(cur)
    deltas = (0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001)

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new, endpoint=False),
                     np.linspace(0, 2, len(h), endpoint=False), h)
