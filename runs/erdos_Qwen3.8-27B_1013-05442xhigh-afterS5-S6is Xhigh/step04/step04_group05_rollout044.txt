    idx = np.argsort(corr)[::-1][:k]
    lags = idx - (n - 1)
    grad = np.zeros(n)
    for lag in lags:
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
        else:
            m = -lag
            grad[m:] += (1.0 - h[:n - m])
            grad[:n - m] -= h[m:]
    return grad / k

def smooth_max_obj(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -400, 400)
    return (m + np.log(np.sum(np.exp(s)) / len(corr)) / beta) * dx

def constraint_sum_diff(x, target):
    return x.sum() - target

def jac_ones(x):
    return np.ones_like(x)

def slsqp_round(h, target, n, beta, max_iter):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    res = minimize(
        smooth_max_obj, h, args=(beta, dx),
        method='SLSQP', bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum_diff,
                      'jac': jac_ones, 'args': (target,)}],
        options={'maxiter': max_iter, 'ftol': 1e-18}
    )
    return res.x

def subgrad_refined(h, target, time_fn, max_iters, k, momentum_base=0.85):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    no_improve = 0
    steps = [0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    for it in range(max_iters):
        if time_fn() < 3:
            break
        g = compute_subgrad_topk(best, k)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        mom = momentum_base + 0.05 * min(1.0, it / 5000.0)
        vel = mom * vel + (1.0 - mom) * (g / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        d = -vel / vn

        improved = False
        for step in steps:
            ht = project_h(best + step * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best = ht
                best_c5 = c5t
                no_improve = 0
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve % 150 == 0:
                scale = 0.004 * (1.0 + 0.5 * min(no_improve / 2000.0, 3.0))
                p = np.random.randn(n) * scale
                ht = project_h(best + p, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    vel *= 0.3
                    no_improve = 0
                elif no_improve > 4000:
                    break
            if no_improve > 6000:
                break
    return best, best_c5

def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    x_src = np.linspace(0, 2.0, n_src)
    x_dst = np.linspace(0, 2.0, n_dst)
    return np.interp(x_dst, x_src, h_src)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(400, 0.5)

    n0 = len(best_h)
    target0 = n0 / 2.0
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP at original resolution with increasing beta
    for beta in [30, 80, 250, 800, 2500, 6000, 15000]:
        if rem() < 25:
            break
        try:
            h_new = slsqp_round(best_h, target0, n0, beta, 4000)
            h_new = project_h(h_new, target0)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Phase 2: Subgradient with adaptive k at original resolution
    for k_val in [8, 12, 18, 25]:
        if rem() < 30:
            break
        best_h, best_c5 = subgrad_refined(best_h, target0, rem, max_iters=30000, k=k_val)

    # Phase 3: Try n=600
    if rem() > 120:
        n3 = 600
        target3 = n3 / 2.0
        h3 = project_h(interpolate_h(best_h, n3), target3)
        c5_3 = compute_c5(h3)
        for beta in [40, 150, 600, 2000, 6000]:
            if rem() < 25:
                break
            try:
                h3_new = slsqp_round(h3, target3, n3, beta, 3500)
                h3 = project_h(h3_new, target3)
                c5_3 = compute_c5(h3)
            except Exception:
                pass
        if rem() > 30:
            for k_val in [10, 18, 28]:
                if rem() < 20:
                    break
                h3, c5_3 = subgrad_refined(h3, target3, rem, max_iters=20000, k=k_val)
        if c5_3 < best_c5:
            best_h = h3
            best_c5 = c5_3

    # Phase 4: Try n=800 if time allows
    if rem() > 150:
        n4 = 800
        target4 = n4 / 2.0
        h4 = project_h(interpolate_h(best_h, n4), target4)
        c5_4 = compute_c5(h4)
        for beta in [40, 150, 600, 2000]:
            if rem() < 25:
                break
            try:
                h4_new = slsqp_round(h4, target4, n4, beta, 3000)
                h4 = project_h(h4_new, target4)
                c5_4 = compute_c5(h4)
            except Exception:
                pass
        if rem() > 30:
            for k_val in [12, 20, 30]:
                if rem() < 20:
                    break
                h4, c5_4 = subgrad_refined(h4, target4, rem, max_iters=15000, k=k_val)
        if c5_4 < best_c5:
            best_h = h4
            best_c5 = c5_4

    # Phase 5: Final polish on best resolution
    n_f = len(best_h)
    target_f = n_f / 2.0
    if rem() > 20:
        for beta in [10000, 20000]:
            if rem() < 15:
                break
            try:
                h_new = slsqp_round(best_h, target_f, n_f, beta, 8000)
                h_new = project_h(h_new, target_f)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass

    # Final subgradient sweep
    for k_val in [6, 10, 20]:
        if rem() < 5:
            break
        best_h, best_c5 = subgrad_refined(best_h, target_f, rem, max_iters=8000, k=k_val)

    best_h = project_h(best_h, target_f)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_f
