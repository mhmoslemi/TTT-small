import numpy as np
from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(cvals, beta):
    m = np.max(cvals)
    s = np.clip(beta * (cvals - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / beta

def obj_and_grad(h, beta, top_m):
    """Compute smooth max objective and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    fval = smooth_max_val(corr, beta)
    
    # Top-m indices with softmax weights
    m = min(top_m, len(corr))
    top_idx = np.argsort(corr)[::-1][:m]
    top_vals = corr[top_idx]
    w = np.exp(beta * (top_vals - top_vals[0]))
    w /= w.sum()
    
    grad = np.zeros(n)
    for i in range(m):
        wi = w[i]
        if wi < 1e-14:
            continue
        k = int(top_idx[i]) - (n - 1)
        if k >= 0:
            a = np.arange(k, n)
            grad[a] += wi * (1.0 - h[a - k])
            a2 = np.arange(0, n - k)
            grad[a2] -= wi * h[a2 + k]
        else:
            mm = -k
            a = np.arange(0, n - mm)
            grad[a] += wi * (1.0 - h[a + mm])
            a2 = np.arange(mm, n)
            grad[a2] -= wi * h[a2 - mm]
    
    return fval, grad, corr

def slsqp_with_grad(h_init, target, beta, top_m, max_iter, rem_fn):
    n = len(h_init)
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def f(h_in):
        return obj_and_grad(h_in, beta, top_m)[0]
    
    def fprime(h_in):
        return obj_and_grad(h_in, beta, top_m)[1]
    
    def cfun(h_in):
        return h_in.sum() - target
    def cjac(h_in):
        return np.ones(n)
    
    try:
        res = minimize(f, h_init, jac=fprime, method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                      options={'maxiter': max_iter, 'ftol': 1e-16})
        h_new = project_h(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h_init, compute_c5(h_init)

def projected_grad_phase(h_init, target, beta, top_m, max_iters, rem_fn,
                         step_list, mom=0.0, perturb_every=300, perturb_scale=0.005):
    """Projected gradient with optional momentum."""
    n = len(h_init)
    x = h_init.copy()
    if mom > 0:
        v = np.zeros(n)
    else:
        v = None
    best = h_init.copy()
    best_c5 = compute_c5(best)
    no_improve = 0
    
    for it in range(max_iters):
        if rem_fn() < 2:
            break
        
        _, grad, _ = obj_and_grad(x, beta, top_m)
        gn = np.linalg.norm(grad)
        if gn < 1e-16:
            break
        
        if mom > 0:
            v = mom * v + (1 - mom) * (-grad / gn)
            vn = np.linalg.norm(v)
            if vn < 1e-16:
                break
            d = v / vn
        else:
            d = -grad / gn
        
        found = False
        for step in step_list:
            xt = project_h(x + step * d, target)
            c5t = compute_c5(xt)
            if c5t < best_c5 - 1e-16:
                x = xt
                best = xt.copy()
                best_c5 = c5t
                no_improve = 0
                found = True
                break
        
        if not found:
            no_improve += 1
            if no_improve >= perturb_every and it % perturb_every == perturb_every - 1:
                p = np.random.randn(n) * perturb_scale
                xt = project_h(x + p, target)
                c5t = compute_c5(xt)
                if c5t < best_c5:
                    x = xt
                    best = xt.copy()
                    best_c5 = c5t
                    no_improve = 0
                    if mom > 0:
                        v *= 0.5
            if no_improve > perturb_every * 8:
                break
    
    return best, best_c5

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(cvals, beta):
    m = np.max(cvals)
    s = np.clip(beta * (cvals - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / beta

def obj_and_grad(h, beta, top_m):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    fval = smooth_max_val(corr, beta)
    m = min(top_m, len(corr))
    top_idx = np.argsort(corr)[::-1][:m]
    top_vals = corr[top_idx]
    w = np.exp(beta * (top_vals - top_vals[0]))
    w /= w.sum()
    grad = np.zeros(n)
    for i in range(m):
        wi = w[i]
        if wi < 1e-14:
            continue
        k = int(top_idx[i]) - (n - 1)
        if k >= 0:
            a = np.arange(k, n)
            grad[a] += wi * (1.0 - h[a - k])
            a2 = np.arange(0, n - k)
            grad[a2] -= wi * h[a2 + k]
        else:
            mm = -k
            a = np.arange(0, n - mm)
            grad[a] += wi * (1.0 - h[a + mm])
            a2 = np.arange(mm, n)
            grad[a2] -= wi * h[a2 - mm]
    return fval, grad, corr

def slsqp_with_grad(h_init, target, beta, top_m, max_iter, rem_fn):
    n = len(h_init)
    bounds = Bounds(np.zeros(n), np.ones(n))
    _cache = {}
    def f(h_in):
        return obj_and_grad(h_in, beta, top_m)[0]
    def fprime(h_in):
        return obj_and_grad(h_in, beta, top_m)[1]
    def cfun(h_in):
        return h_in.sum() - target
    def cjac(h_in):
        return np.ones(n)
    try:
        res = minimize(f, h_init, jac=fprime, method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                      options={'maxiter': max_iter, 'ftol': 1e-16})
        h_new = project_h(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h_init, compute_c5(h_init)

def projected_grad_phase(h_init, target, beta, top_m, max_iters, rem_fn,
                         step_list, mom=0.0, perturb_every=300, perturb_scale=0.005):
    n = len(h_init)
    x = h_init.copy()
    v = np.zeros(n) if mom > 0 else None
    best = h_init.copy()
    best_c5 = compute_c5(best)
    no_improve = 0
    for it in range(max_iters):
        if rem_fn() < 2:
            break
        _, grad, _ = obj_and_grad(x, beta, top_m)
        gn = np.linalg.norm(grad)
        if gn < 1e-16:
            break
        if mom > 0:
            v = mom * v + (1.0 - mom) * (-grad / gn)
            vn = np.linalg.norm(v)
            if vn < 1e-16:
                break
            d = v / vn
        else:
            d = -grad / gn
        found = False
        for step in step_list:
            xt = project_h(x + step * d, target)
            c5t = compute_c5(xt)
            if c5t < best_c5 - 1e-16:
                x = xt
                best = xt.copy()
                best_c5 = c5t
                no_improve = 0
                found = True
                break
        if not found:
            no_improve += 1
            if no_improve >= perturb_every and it % perturb_every == perturb_every - 1:
                p = np.random.randn(n) * perturb_scale
                xt = project_h(x + p, target)
                c5t = compute_c5(xt)
                if c5t < best_c5:
                    x = xt
                    best = xt.copy()
                    best_c5 = c5t
                    no_improve = 0
                    if mom > 0:
                        v *= 0.5
            if no_improve > perturb_every * 8:
                break
    return best, best_c5

def fista_phase(h_init, target, beta, top_m, max_iters, rem_fn):
    n = len(h_init)
    x = h_init.copy()
    y = h_init.copy()
    t = 1.0
    best = h_init.copy()
    best_c5 = compute_c5(best)
    step = 1.0
    for it in range(max_iters):
        if rem_fn() < 2:
            break
        f_y, grad_y, corr_y = obj_and_grad(y, beta, top_m)
        z = project_h(y - step * grad_y, target)
        f_z, _, _ = obj_and_grad(z, beta, top_m)
        quad = 0.5 / step * np.linalg.norm(z - y) ** 2
        if f_z > f_y + grad_y @ (z - y) + quad:
            step *= 0.5
            continue
        x_new = z
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = project_h(y, target)
        x = x_new
        t = t_new
        step = min(step * 1.1, 10.0)
        c5_val = compute_c5(x)
        if c5_val < best_c5:
            best = x.copy()
            best_c5 = c5_val
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(400, 0.5)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    target = n / 2.0

    # Phase 1: SLSQP with analytical gradient, increasing beta
    slsqp_params = [
        (30, 15, 8000),
        (100, 12, 8000),
        (300, 10, 10000),
        (1000, 8, 10000),
        (3000, 6, 12000),
    ]
    for beta, top_m, max_it in slsqp_params:
        if rem() < 90:
            break
        h_new, c5_new = slsqp_with_grad(best_h, target, beta, top_m, max_it, rem)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: FISTA with moderate beta
    if rem() > 60:
        h_new, c5_new = fista_phase(best_h, target, 200, 10, 100000, rem)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 3: Projected gradient with momentum
    best_h, best_c5 = projected_grad_phase(
        best_h, target, beta=300, top_m=10, max_iters=50000, rem_fn=rem,
        step_list=[0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002],
        mom=0.8, perturb_every=300, perturb_scale=0.006
    )

    # Phase 4: Fine SLSQP polish
    if rem() > 80:
        h_new, c5_new = slsqp_with_grad(best_h, target, 5000, 5, 15000, rem)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 5: Try higher resolution
    if rem() > 250:
        n2 = n + 200
        if n2 % 2 == 1:
            n2 += 1
        target2 = n2 / 2.0
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n2)
        h2 = project_h(np.interp(x_new, x_old, best_h), target2)
        c5_2 = compute_c5(h2)
        for beta, top_m, max_it in [(100, 12, 6000), (500, 8, 6000), (3000, 5, 8000)]:
            if rem() < 50:
                break
            h2, c5_2 = slsqp_with_grad(h2, target2, beta, top_m, max_it, rem)
        if rem() > 30:
            h2, c5_2 = projected_grad_phase(
                h2, target2, beta=500, top_m=8, max_iters=20000, rem_fn=rem,
                step_list=[0.02, 0.01, 0.005, 0.002, 0.001, 0.0005],
                mom=0.85, perturb_every=400, perturb_scale=0.004
            )
        if c5_2 < best_c5:
            best_h = h2
            best_c5 = c5_2
            n = n2
            target = target2

    # Phase 6: Final polish
    if rem() > 60:
        h_new, c5_new = slsqp_with_grad(best_h, target, 8000, 4, 15000, rem)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    best_h, best_c5 = projected_grad_phase(
        best_h, target, beta=500, top_m=6, max_iters=30000, rem_fn=rem,
        step_list=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
        mom=0.9, perturb_every=500, perturb_scale=0.003
    )

    return best_h, best_c5, len(best_h)
