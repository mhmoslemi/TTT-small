    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * room, 0, 1)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h * (1 + diff / tot), 0, 1)
    return h

def logsumexp_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def s_con(x, n):
    return np.sum(x) - n / 2.0

def s_con_jac(x, n):
    return np.ones(n)

def slsqp(h0, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': s_con, 'jac': s_con_jac, 'args': (n,)}]
    try:
        res = minimize(logsumexp_obj, h0, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except:
        return h0.copy()

def greedy_pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(200):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    # h[i] + d, h[j] - d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    # h[i] - d, h[j] + d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                break
    return h, best_c

def thue_morse(n):
    """Generate Thue-Morse sequence of length n."""
    t = np.zeros(n, dtype=float)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

    best_c = compute_c5(h, n)
    dx = 2.0 / n
    num_lags = 2 * n - 1
    
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        
        c_vec = np.correlate(h, 1-h, mode='full') * dx
        c_max = c_vec.max()
        
        # Weight active lags
        w = np.exp(alpha * (c_vec - c_max))
        w = w / w.sum()
        
        # Compute S[i] = weighted sum of "effect" of increasing h[i]
        # S[i] = sum_k w_k * d(c_k)/d(h[i])
        # d(c_k)/d(h[i]) for lag k = k - (n-1) in the correlation array
        # For lag l:
        #   d(c_l)/d(h[i]) = (1-h[i+l])*dx (if i+l < n) + (-h[i-l])*dx (if i-l >= 0)
        #                   for l >= 0
        #   d(c_l)/d(h[i]) = (1-h[i+|l|])*dx (if i+|l| < n) + (-h[i-|l|])*dx (if i-|l| >= 0)
        #   for l < 0 (by symmetry, same formula with |l|)

                continue

                # d/dc_0/dh[i] = (1-2*h[i])*dx

                    grad_k[i] += (-h[i - alag]) * dx

                best_c = new_c
            else:
                h[i_inc] += d
                h[j_dec] -= d
