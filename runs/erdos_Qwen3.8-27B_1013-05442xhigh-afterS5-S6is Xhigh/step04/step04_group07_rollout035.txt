        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(np.clip(beta * (x - m), -500, 500))) / len(x)) / beta

def sml_obj(h, beta, dx):
    return smooth_max(np.correlate(h, 1.0 - h, mode="full"), beta) * dx

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def snap_binary(h, target):
    z = (h > 0.5).astype(float)
    diff = int(round(target)) - int(z.sum())
    if diff > 0:
        zeros = np.where(z == 0)[0]
        order = np.argsort(-h[zeros])
        z[zeros[order[:diff]]] = 1.0
    elif diff < 0:
        ones = np.where(z == 1)[0]
        order = np.argsort(h[ones])
        z[ones[order[:(-diff)]]] = 0.0
    return project_h(z, target)

def binary_sa(h0_bin, n, t0, deadline, n_iter=30000, seed=7):
    h = h0_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    T = 0.015
    T_min = 0.0005
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Choose neighborhood size
        r = rng.random()
        if r < 0.6:
            # Single swap
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            new_c5 = compute_c5(h)
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
            elif rng.random() > np.exp(-(new_c5 - best_c5) / T):
                h[i], h[j] = h[j], h[i]
        else:
            # Block swap (2-4 adjacent positions)
            blen = rng.randint(2, 5)
            i = rng.randint(0, n - blen)
            j = rng.randint(0, n - blen)
            if abs(i - j) < blen:
                continue
            # Count ones in each block to ensure sum is preserved
            n1_block = int(h[i:i+blen].sum())
            n2_block = int(h[j:j+blen].sum())
            if n1_block != n2_block:
                continue
            h[i:i+blen], h[j:j+blen] = h[j:j+blen].copy(), h[i:i+blen].copy()
            new_c5 = compute_c5(h)
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
            elif rng.random() > np.exp(-(new_c5 - best_c5) / T):
                h[i:i+blen], h[j:j+blen] = h[j:j+blen].copy(), h[i:i+blen].copy()
        
        # Cool
        T = max(T_min, T * 0.9997)
    
    return best_h, best_c5

def fine_local_search(h0, target, n, t0, deadline, max_sweeps=60):
    h = h0.copy()
    best_c5 = compute_c5(h)
    
    steps = [0.002, -0.002, 0.001, -0.001, 0.0005, -0.0005]
    
    for sw in range(max_sweeps):
        if time.time() > deadline - 2:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline - 1:
                break
            for d in steps:
                trial = h.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    h = trial
                    best_c5 = c
                    improved = True
                    break
        if not improved and sw > 5:
            break
    
    # Pair moves
    rng = np.random.RandomState(42)
    for sw in range(max_sweeps):
        if time.time() > deadline - 2:
            break
        improved = False
        for _ in range(n):
            if time.time() > deadline - 1:
                break
            j, k = rng.choice(n, 2, replace=False)
            for d in [0.003, -0.003, 0.001, -0.001]:
                trial = h.copy()
                trial[j] += d
                trial[k] -= d
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    h = trial
                    best_c5 = c
                    improved = True
                    break
        if not improved and sw > 5:
            break
    
    return h, best_c5

def nesterov_loop(h0, n, target, t0, deadline, n_iter=5000, step0=0.02):
    h = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    step = step0
    stall = 0
    
    for it in range(n_iter):
        if time.time() > deadline - 3:
            break
        corr = np.correlate(y, 1.0 - y, mode="full")
        mc = np.max(corr)
        thr = mc - 0.004 * (mc + 1e-10)
        idx = np.where(corr >= thr)[0]
        g = np.zeros(n)
        for i in idx:
            lag = int(i) - (n - 1)
            if lag >= 0:
                g[:n - lag] += (1.0 - y[lag:])
                g[lag:] -= y[:n - lag]
            else:
                la = -lag
                g[la:] += (1.0 - y[:n - la])
                g[:n - la] -= y[la:]
        g /= max(len(idx), 1)
        h_new = project_h(y - step * g, target)
        mu = (1.0 - np.sqrt(1.0 + 4.0 / ((it + 1) ** 2))) / 2.0
        y = np.clip(h_new + mu * (h_new - h_prev), 0, 1)
        h_prev, h = h, h_new
        c = compute_c5(h)
        if c < best_c5 - 1e-15:
            best_h, best_c5, stall = h.copy(), c, 0
        else:
            stall += 1
            if stall % 200 == 0:
                step *= 0.7
    
    return best_h, best_c5

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(np.clip(beta * (x - m), -500, 500))) / len(x)) / beta

def sml_obj(h, beta, dx):
    return smooth_max(np.correlate(h, 1.0 - h, mode="full"), beta) * dx

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter=2000, ftol=1e-14):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def snap_binary(h, target):
    z = (h > 0.5).astype(float)
    diff = int(round(target)) - int(z.sum())
    if diff > 0:
        zeros = np.where(z == 0)[0]
        order = np.argsort(-h[zeros])
        z[zeros[order[:diff]]] = 1.0
    elif diff < 0:
        ones = np.where(z == 1)[0]
        order = np.argsort(h[ones])
        z[ones[order[:(-diff)]]] = 0.0
    return project_h(z, target)

def thue_morse(n):

    t = np.zeros(n, dtype=int)
    for b in range(10):
        t += (idx >> b) & 1
    return (t % 2).astype(float)

        else:

        T = max(T_min, T * 0.9998)

    h = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    step = step0
    stall = 0
    for it in range(n_iter):
        if time.time() > deadline - 3:
            break
        corr = np.correlate(y, 1.0 - y, mode="full")
        mc = np.max(corr)
        thr = mc - 0.004 * (mc + 1e-10)
        idx = np.where(corr >= thr)[0]
        g = np.zeros(n)
        for i in idx:
            lag = int(i) - (n - 1)
            if lag >= 0:
                g[:n-lag] += (1.0 - y[lag:])
                g[lag:] -= y[:n-lag]
            else:
                la = -lag
                g[la:] += (1.0 - y[:n-la])
                g[:n-la] -= y[la:]
        g /= max(len(idx), 1)
        h_new = project_h(y - step * g, target)
        mu = (1.0 - np.sqrt(1.0 + 4.0 / ((it+1)**2))) / 2.0
        y = np.clip(h_new + mu * (h_new - h_prev), 0, 1)
        h_prev, h = h, h_new
        c = compute_c5(h)
        if c < best_c5 - 1e-15:
            best_h, best_c5, stall = h.copy(), c, 0

            stall += 1
