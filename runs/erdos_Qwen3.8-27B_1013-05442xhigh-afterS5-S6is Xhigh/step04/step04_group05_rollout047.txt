            except Exception:
                pass

        best_h, target, rem,

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_step(h, top_m, dx):
    """Compute subgradient using top_m active lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_act = min(top_m, len(corr))
    top = np.argsort(corr)[::-1][:m_act]
    grad = np.zeros(n)
    for idx in top:
        lag = int(idx) - (n - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:-lag])
            grad[:n - lag] -= h[lag:]
        else:
            m = -lag
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad, corr

def subgradient_nesterov(h0, target, time_fn, max_iters, step_init, decay, mu, m_schedule, perturb_every):
    n = len(h0)
    dx = 2.0 / n
    best = h0.copy()
    best_c5 = compute_c5(best)
    x = h0.copy()
    v = np.zeros(n)
    step = step_init
    rng = np.random.RandomState(2024)
    no_improve = 0
    for it in range(max_iters):
        if time_fn() < 3:
            break
        # Cycle through m values
        m_idx = (it * 3) % len(m_schedule)
        top_m = m_schedule[m_idx]
        # Nesterov lookahead
        y = x + mu * v
        y = np.clip(y, 0.0, 1.0)
        g, _ = subgrad_step(y, top_m, dx)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g = g / gnorm
        v = mu * v - step * g
        x_new = project_h(x + v, target)
        c5_new = compute_c5(x_new)
        if c5_new < best_c5 - 1e-14:
            best = x_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        x = x_new
        step *= decay
        if no_improve >= perturb_every:
            scale = 0.01 + 0.02 * (no_improve % 5) / 5.0
            p = rng.randn(n) * scale
            xp = project_h(x + p, target)
            cp = compute_c5(xp)
            if cp < best_c5:
                best = xp.copy()
                best_c5 = cp
                x = xp
                v = np.zeros(n)
                step = step_init
                no_improve = 0
            elif no_improve >= perturb_every * 8:
                x = best.copy()
                v = np.zeros(n)
                step = step_init * 1.5
                no_improve = 0
    return best, best_c5

def binary_sa(h_start, target, time_fn, max_iters, t_init, cooling):
    n = len(h_start)
    dx = 2.0 / n
    curr = np.zeros(n)
    top_idx = np.argsort(h_start)[::-1][:int(target)]
    curr[top_idx] = 1.0
    corr = np.correlate(curr, 1.0 - curr, mode="full")
    best_c5 = float(np.max(corr) * dx)
    best = curr.copy()
    curr_c5 = best_c5
    ones = np.where(curr > 0.5)[0].copy()
    zeros = np.where(curr < 0.5)[0].copy()
    t = t_init
    rng = np.random.RandomState(777)
    no_improve = 0
    for it in range(max_iters):
        if time_fn() < 2:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        curr[i] = 0.0
        curr[j] = 1.0
        corr_new = np.correlate(curr, 1.0 - curr, mode="full")
        new_c5 = float(np.max(corr_new) * dx)
        delta = new_c5 - curr_c5
        accept = delta <= 0 or (t > 1e-6 and rng.random() < np.exp(-delta / t))
        if accept:
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = curr.copy()
                no_improve = 0
            else:
                no_improve += 1
            oi = np.where(ones == i)[0][0]
            ones = np.delete(ones, oi)
            oz = np.where(zeros == j)[0][0]
            zeros = np.delete(zeros, oz)
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
        else:
            curr[i] = 1.0
            curr[j] = 0.0
            no_improve += 1
        t = max(t * cooling, 5e-5)
        if no_improve > 200000:
            curr = best.copy()
            corr = np.correlate(curr, 1.0 - curr, mode="full")
            curr_c5 = float(np.max(corr) * dx)
            ones = np.where(curr > 0.5)[0].copy()
            zeros = np.where(curr < 0.5)[0].copy()
            t = t_init * 0.3
            no_improve = 0
    return best, best_c5

def smooth_binary_to_cont(h_bin, sigma):
    n = len(h_bin)
    r = int(3 * sigma)
    kernel = np.exp(-0.5 * (np.arange(-r, r + 1) / sigma) ** 2)
    kernel /= kernel.sum()
    return np.convolve(h_bin, kernel, mode='same')

def slsqq_polish(h, target, dx, time_fn, beta, maxiter):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(h_in):
        c = np.correlate(h_in, 1.0 - h_in, mode="full")
        m = np.max(c)
        sh = np.clip(beta * (c - m), -400, 400)
        return (m + np.log(np.sum(np.exp(sh)) / len(c)) / beta) * dx
    def con(h_in):
        return h_in.sum() - target
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con, 'jac': lambda x: np.ones(len(x))}],
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_h(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h):
            return h_new, c5_new
    except Exception:
        pass
    return h, compute_c5(h)
