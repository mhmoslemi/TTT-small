import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = free.sum()
        if nf == 0:
            free = np.ones(len(h), dtype=bool)
            nf = len(h)
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_h(h, lags):
    n = len(h)
    grad = np.zeros(n)
    for lag in lags:
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:-lag])
            grad[:n - lag] -= h[lag:]
        else:
            m = -lag
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + (logsumexp(xs) - np.log(len(x))) / beta

class SlsqpObj:
    def __init__(self, beta, dx, target):
        self.beta = beta
        self.dx = dx
        self.target = target
    def obj(self, h):
        c = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max(c, self.beta) * self.dx
    def con(self, h):
        return h.sum() - self.target
    def jac_con(self, h):
        return np.ones(len(h))

def phase_slsqp(h, target, dx, betas, max_iter, t0, budget_s):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    best = h.copy()
    best_c5 = compute_c5(best)
    for beta in betas:
        if budget_s - (time.time() - t0) < 25:
            break
        w = SlsqpObj(beta, dx, target)
        try:
            res = minimize(w.obj, best, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': w.con, 'jac': w.jac_con}],
                          options={'maxiter': max_iter, 'ftol': 1e-18})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def phase_subgrad(h, target, t0, budget_s, max_iters, band_frac,
                  step_init, step_min, momentum, mom_decay, perturb_every, perturb_scale):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    no_improve = 0
    step = step_init
    for it in range(max_iters):
        if budget_s - (time.time() - t0) < 3:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        max_c = np.max(corr)
        thresh = max_c * (1.0 - band_frac)
        active = np.where(corr >= thresh)[0]
        lags = [int(i) - (n - 1) for i in active]
        g = subgrad_h(best, lags)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        vel = momentum * vel + (1.0 - momentum) * (g / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        d = -vel / vn
        lo, hi = 0.0, step
        found = False
        for _ in range(14):
            mid = (lo + hi) / 2.0
            ht = project_h(best + mid * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                lo = mid
                found = True
            else:
                hi = mid
        if found:
            best = project_h(best + lo * d, target)
            best_c5 = compute_c5(best)
            no_improve = 0
            step = max(step_min, min(lo * 1.4, 0.15))
        else:
            no_improve += 1
            step = max(step_min, step * 0.65)
            if no_improve % perturb_every == 0:

                ht = project_h(best + p, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    vel *= 0.3
                    no_improve = 0
                    step = step_init
            if no_improve > perturb_every * 15:
                break
        momentum = max(0.5, momentum * mom_decay)
    return best, best_c5
