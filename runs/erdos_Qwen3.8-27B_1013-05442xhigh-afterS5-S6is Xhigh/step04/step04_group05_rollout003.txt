                          constraints=[{'type': 'eq', 'fun': con_fn, 'jac': jac_fn}],
                          options={'maxiter': max_iter, 'ftol': 1e-20})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(400, 0.5)

    best_c5 = compute_c5(best_h)
    n0 = len(best_h)
    target0 = n0 / 2.0

    # Phase 1: Adam with increasing beta at original resolution
    best_h, best_c5 = adam_smooth_max(
        best_h, target0, [50, 200, 800, 3000, 10000], 800, 0.02, rem
    )

    # Phase 2: Subgradient active set
    best_h, best_c5 = subgrad_active(
        best_h, target0, rem, max_iters=30000, top_k=20,
        steps=[0.012, 0.006, 0.003, 0.001, 0.0005, 0.0002],
        momentum=0.85, perturb_every=250, perturb_scale=0.01
    )

    # Phase 3: SLSQP polish
    best_h, best_c5 = slsqp_polish(best_h, target0, rem, max_iter=8000)

    # Phase 4: Try higher resolution (n=800)
    if rem() > 120:
        n2 = min(n0 * 2, 800)
        if n2 % 2 == 0 and n2 > n0:
            target2 = n2 / 2.0
            x_old = np.linspace(0, 2, n0 + 1)[:-1]
            x_new = np.linspace(0, 2, n2 + 1)[:-1]
            h2 = project_h(np.interp(x_new, x_old, best_h), target2)
            c5_2 = compute_c5(h2)
            h2, c5_2 = adam_smooth_max(
                h2, target2, [100, 500, 2000, 8000], 600, 0.015, rem
            )
            h2, c5_2 = subgrad_active(
                h2, target2, rem, max_iters=20000, top_k=25,
                steps=[0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002],
                momentum=0.88, perturb_every=300, perturb_scale=0.006
            )
            if c5_2 < best_c5:
                best_h = h2
                best_c5 = c5_2

    # Phase 5: Try n=600 as alternative
    n_cur = len(best_h)
    if n_cur != 600 and rem() > 80:
        n3 = 600
        target3 = n3 / 2.0
        x_old3 = np.linspace(0, 2, n_cur + 1)[:-1]
        x_new3 = np.linspace(0, 2, n3 + 1)[:-1]
        h3 = project_h(np.interp(x_new3, x_old3, best_h), target3)
        c5_3 = compute_c5(h3)
        h3, c5_3 = adam_smooth_max(
            h3, target3, [100, 500, 2000, 8000], 500, 0.015, rem
        )
        h3, c5_3 = subgrad_active(
            h3, target3, rem, max_iters=15000, top_k=20,
            steps=[0.008, 0.004, 0.002, 0.001, 0.0005],
            momentum=0.87, perturb_every=250, perturb_scale=0.007
        )
        if c5_3 < best_c5:
            best_h = h3
            best_c5 = c5_3

    # Phase 6: Final Adam polish with very high beta
    n_f = len(best_h)
    target_f = n_f / 2.0
    if rem() > 20:
        best_h, best_c5 = adam_smooth_max(
            best_h, target_f, [20000, 50000], 200, 0.005, rem
        )

    # Phase 7: Final subgradient micro-steps
    best_h, best_c5 = subgrad_active(
        best_h, target_f, rem, max_iters=20000, top_k=15,
        steps=[0.003, 0.001, 0.0005, 0.0002, 0.0001],
        momentum=0.9, perturb_every=400, perturb_scale=0.003
    )

    return best_h, best_c5, len(best_h)
