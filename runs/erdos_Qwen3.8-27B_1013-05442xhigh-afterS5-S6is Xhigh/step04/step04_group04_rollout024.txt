import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nfree = free.sum()
        if nfree == 0:
            free = np.ones(len(h), dtype=bool)
            nfree = len(h)
        h[free] += diff / nfree
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_of_lag(h, lag, dx):
    """Gradient of f_lag(h) = sum_i h_i (1-h_{i+lag}) * dx"""
    n = len(h)
    grad = np.zeros(n)
    if lag == 0:
        grad = dx * (1.0 - 2.0 * h)
    elif 0 < lag < n:
        grad[lag:] = dx * (1.0 - h[:-lag])
        grad[:n-lag] -= dx * h[lag:]
    elif lag < 0:
        m = -lag
        if m < n:
            grad[:n-m] = dx * (1.0 - h[m:])
            grad[m:] -= dx * h[:n-m]
    return grad

def smooth_max_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -400, 400)
    exp_vals = np.exp(x_shift)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    grad = np.zeros(n)
    active = np.where(weights > 1e-5)[0]
    for k in active:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n-lag] -= w * h[lag:]
        elif lag < 0:
            m2 = -lag
            if m2 < n:
                grad[:n-m2] += w * (1.0 - h[m2:])
                grad[m2:] -= w * h[:n-m2]
    return val, grad

    n = len(h0)
    h = h0.copy()
    best_h = h.copy()

    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.75, 0.999, 1e-8
    
    for it in range(n_iter):
        if time_fn() < 2:
            break
        t += 1
        lr = lr_init / (1.0 + 0.005 * t)
        val, grad = smooth_max_val_grad(h, beta)
        m_v = b1 * m_v + (1.0-b1) * grad
        v_v = b2 * v_v + (1.0-b2) * grad**2
        m_hat = m_v / (1.0 - b1**t)
        v_hat = v_v / (1.0 - b2**t)
        h_new = project_h(h - lr * m_hat / (np.sqrt(v_hat) + eps), target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
        # Periodic restart

            if compute_c5(h_restart) < best_c5:
                h = h_restart
                best_h = h_restart.copy()
                best_c5 = compute_c5(h_restart)
    
    return best_h, best_c5

def slsqp_smooth(h0, beta, maxiter, target, time_fn):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        v, _ = smooth_max_val_grad(x, beta)
        return v
    
    def grad(x):
        _, g = smooth_max_val_grad(x, beta)
        return g
    
    def cons(x):
        return x.sum() - target
    
    cons_jac = np.ones(n)
    constraints = [{'type': 'eq', 'fun': cons, 'jac': cons_jac}]

                      jac=grad, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_h(res.x, target)
        return h_new, compute_c5(h_new)
    except:
        return h0, compute_c5(h0)

                      jac=grad_fn, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_h(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0, compute_c5(h0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)

    best_c5 = compute_c5(best_h)
    n_best = len(best_h)
    target = n_best / 2.0

    # Phase 1: Adam at current resolution with beta annealing
    betas1 = [30, 80, 200, 500, 1500, 5000, 15000, 40000, 100000]
    for beta in betas1:
        if time_left() < 30:
            break
        lr = 0.08 / np.sqrt(n_best)
        h_new, c5_new = adam_optimize(best_h, beta, 25000, lr, target, time_left)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Subgradient with backtracking
    if time_left() > 40:
        h_new, c5_new = subgrad_backtrack(best_h, target, 30000, time_left, init_step=0.02)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 3: Multi-resolution exploration
    for n_res in [400, 500, 650, 800, 1000, 1300, 1700, 2200]:
        if time_left() < 50:
            break
        if n_res <= n_best:
            continue
        x_old = np.linspace(0, 2, n_best)
        x_new = np.linspace(0, 2, n_res)
        h_res = np.interp(x_new, x_old, best_h)
        h_res = project_h(h_res, n_res / 2.0)
        t_res = n_res / 2.0
        for beta in [200, 1000, 5000, 20000]:
            if time_left() < 30:
                break
            lr = 0.08 / np.sqrt(n_res)
            h_new, c5_new = adam_optimize(h_res, beta, 12000, lr, t_res, time_left)
            if c5_new < best_c5:
                best_h, best_c5, n_best = h_new, c5_new, n_res
                target = n_res / 2.0
        if time_left() > 30:
            h_new, c5_new = subgrad_backtrack(best_h, target, 10000, time_left, init_step=0.01)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 4: SLSQP polish at high beta
    if time_left() > 30:
        for beta in [30000, 80000, 200000]:
            if time_left() < 20:
                break
            h_new, c5_new = slsqp_smooth(best_h, beta, 5000, target, time_left)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 5: Final subgradient refinement
    if time_left() > 15:
        h_new, c5_new = subgrad_backtrack(best_h, target, 25000, time_left, init_step=0.005)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 6: Final Adam with very high beta
    if time_left() > 10:
        h_new, c5_new = adam_optimize(best_h, 300000, 10000, 0.005 / np.sqrt(n_best), target, time_left)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 7: Very final SLSQP
    if time_left() > 10:
        h_new, c5_new = slsqp_smooth(best_h, 500000, 3000, target, time_left)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    return best_h, best_c5, len(best_h)
