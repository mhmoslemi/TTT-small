def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):

        if abs(diff) < 1e-14:

            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:

            if tot < 1e-14:
                break
            h += (diff / tot) * h

def obj_smooth(x, n, alpha):

    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
