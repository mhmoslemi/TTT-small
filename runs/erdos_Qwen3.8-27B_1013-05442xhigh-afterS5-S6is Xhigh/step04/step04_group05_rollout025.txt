import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = h.astype(float).copy()
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_c5_lag(h, lag, dx):
    """Gradient of correlation at a specific lag"""
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        g[lag:] += (1.0 - h[:-lag]) * dx
        g[:n - lag] -= h[lag:] * dx
    else:
        m = -lag
        g[:n - m] += (1.0 - h[m:]) * dx
        g[m:] -= h[:n - m] * dx
    return g

def nesterov_subgrad(h, target, dx, n_steps, time_fn, m_list, step_init, step_decay, perturb_every, perturb_scale, momentum):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    x = best.copy()
    v = np.zeros(n)
    alpha = step_init
    no_improve = 0
    
    for it in range(n_steps):
        if time_fn() < 5:
            break
        
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = m_list[it % len(m_list)]
        m_act = min(m, 2 * (n - 1))
        top_idx = np.argsort(corr)[::-1][:m_act]
        
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            g += grad_c5_lag(x, lag, dx)
        g /= m_act
        
        # Nesterov-style update
        v = momentum * v + g
        x_new = x - alpha * v
        x_new = project_h(x_new, target)
        
        c5_new = compute_c5(x_new)
        if c5_new < best_c5:
            best = x_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= perturb_every:
                p = np.random.randn(n) * perturb_scale
                x_try = project_h(best + p, target)
                c5_try = compute_c5(x_try)
                if c5_try < best_c5:
                    best = x_try
                    best_c5 = c5_try
                    x = best.copy()
                    v = np.zeros(n)
                    no_improve = 0
                elif no_improve > perturb_every * 5:
                    break
        
        x = x_new if c5_new < best_c5 else x
        alpha *= step_decay
    
    return best, best_c5

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def project_h(h, target_sum):
    h = h.astype(float).copy()
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_lag(h, lag, dx):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        g[lag:] += (1.0 - h[:-lag]) * dx
        g[:n-lag] -= h[lag:] * dx
    else:
        m = -lag
        g[:n-m] += (1.0 - h[m:]) * dx
        g[m:] -= h[:n-m] * dx
    return g

def subgrad(h, target, dx, max_iters, time_fn, n_active, step_init, step_decay, perturb_every, perturb_scale, momentum, tol=0.01):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)[0]
    x = best.copy()
    vel = np.zeros(n)
    step = step_init
    no_improve = 0
    
    for it in range(max_iters):
        if time_fn() < 5:
            break
        c5_x, corr = compute_c5(x)
        # Adaptive active set: lags within tol of max
        threshold = corr.max() * (1.0 - tol)
        active_mask = corr >= threshold
        n_act = max(3, int(active_mask.sum()))
        n_act = min(n_act, n_active, 2*n-1)
        top_idx = np.argsort(corr)[::-1][:n_act]
        
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n-1)
            g += grad_lag(x, lag, dx)
        g /= n_act
        
        vel = momentum * vel + g
        x_try = project_h(x - step * vel, target)
        c5_try = compute_c5(x_try)[0]
        
        if c5_try < best_c5 - 1e-15:
            best = x_try
            best_c5 = c5_try
            x = x_try
            no_improve = 0
            step = min(step * 1.03, 0.2)
        else:
            no_improve += 1
            step *= step_decay
            if no_improve >= perturb_every:
                p = np.random.randn(n) * perturb_scale
                x_try = project_h(best + p, target)
                c5_try = compute_c5(x_try)[0]
                if c5_try < best_c5:
                    best = x_try
                    best_c5 = c5_try
                    x = best.copy()
                    vel = np.zeros(n)
                    step = step_init * 0.5
                    no_improve = 0
                elif no_improve > perturb_every * 5:
                    break
    return best, best_c5

def slsqp(h, target, dx, beta, max_iters, time_fn):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(h_in):
        c = np.correlate(h_in, 1.0 - h_in, mode="full")
        m = np.max(c)
        cs = np.clip(beta * (c - m), -500, 500)
        return (m + np.log(np.sum(np.exp(cs)) / len(cs)) / beta) * dx
    def con(h_in):
        return h_in.sum() - target
    def jac(h_in):
        return np.ones(n)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con, 'jac': jac}],
                      options={'maxiter': max_iters, 'ftol': 1e-15})
        h_new = project_h(res.x, target)
        c5_new = compute_c5(h_new)[0]
        if c5_new < compute_c5(h)[0]:
            return h_new, c5_new
        return h, compute_c5(h)[0]
    except:
        return h, compute_c5(h)[0]

def make_smooth_obj(h_init, target, dx, beta):
    n = len(h_init)
    def obj(h_in):
        c = np.correlate(h_in, 1.0 - h_in, mode="full")
        m = np.max(c)
        cs = np.clip(beta * (c - m), -500, 500)
        return (m + np.log(np.sum(np.exp(cs)) / len(cs)) / beta) * dx
    return obj

def make_con(target):
    def con(h_in):
        return h_in.sum() - target
    return con

def ones_jac(n):
    def jac(h_in):
        return np.ones(n)
    return jac
