    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

        if compute_c5(snap_binary(h_per, n_r)) < compute_c5(snap_binary(h_init, n_r)):
            h_init = h_per

        # Phase 1: Binary search (multiple passes)
        h_bin = snap_binary(h_init, n_r)
        for s in [seed + n_r, seed + n_r + 1]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 40:
                break
            h_bin, _ = binary_search(h_bin, n_r, t0, budget_s, seed=s, n_iter=200000, use_blocks=True)

        # Phase 2: Continuous refinement
        h_cont = h_bin + 0.005 * np.random.RandomState(seed + n_r).randn(n_r)
        h_cont = project_h(h_cont, target)
        for beta in [300, 1200, 4000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 12:
                break
            h_cont = slsqp(h_cont, n_r, target, beta, maxiter=5000)
            c_try = compute_c5(h_cont)
            if c_try < best_c5:
                best_h = h_cont.copy()
                best_c5 = c_try
                best_n = n_r

        # Phase 3: Back to binary, refine, then continuous again
        remaining = budget_s - (time.time() - t0)
        if remaining > 50:
            h_bin2 = snap_binary(h_cont, n_r)
            h_bin2, _ = binary_search(h_bin2, n_r, t0, budget_s, seed=seed + n_r + 20, n_iter=150000, use_blocks=True)
            h_cont2 = h_bin2 + 0.003 * np.random.RandomState(seed + n_r + 30).randn(n_r)
            h_cont2 = project_h(h_cont2, target)
            for beta in [5000, 15000]:
                remaining = budget_s - (time.time() - t0)
                if remaining < 10:
                    break
                h_cont2 = slsqp(h_cont2, n_r, target, beta, maxiter=5000)
                c_try = compute_c5(h_cont2)
                if c_try < best_c5:
                    best_h = h_cont2.copy()
                    best_c5 = c_try
                    best_n = n_r

    # Final polish on best solution
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        target = best_n / 2.0
        for beta in [30000, 80000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 8:
                break
            best_h = slsqp(best_h, best_n, target, beta, maxiter=10000)
            best_c5 = compute_c5(best_h)

    # Try trust-constr as alternative on best
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        target = best_n / 2.0
        h_tc = trust_opt(best_h, best_n, target, 10000, maxiter=1500)
        c_try = compute_c5(h_tc)
        if c_try < best_c5:
            best_h = h_tc.copy()
            best_c5 = c_try

    # One more binary-continuous cycle on the best
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        target = best_n / 2.0
        h_b = snap_binary(best_h, best_n)
        h_b, _ = binary_search(h_b, best_n, t0, budget_s, seed=seed + 9999, n_iter=150000, use_blocks=True)
        h_c = h_b + 0.002 * np.random.RandomState(seed + 8888).randn(best_n)
        h_c = project_h(h_c, target)
        for beta in [8000, 25000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 8:
                break
            h_c = slsqp(h_c, best_n, target, beta, maxiter=6000)
            c_try = compute_c5(h_c)
            if c_try < best_c5:
                best_h = h_c.copy()
                best_c5 = c_try

    return best_h, best_c5, best_n
