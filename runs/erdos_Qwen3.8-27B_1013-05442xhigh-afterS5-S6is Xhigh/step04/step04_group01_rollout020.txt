    dx = 2.0 / n

    m = c.max()

    return float(c.max()), c

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:

        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:

            h += (diff / tot) * h

    c = np.correlate(x, 1.0 - x, mode='full') * dx

    s = alpha * (c - m)
    s_max = s.max()

```python
def pair_delta_exact(h, n, i, j, d):
    """Exact O(n) computation of correlation change for h[i]+=d, h[j]-=d."""
    n_full = 2 * n - 1
    k = np.arange(n_full)
    dx = 2.0 / n
    
    # h'[i] = h[i]+d, h'[j] = h[j]-d
    hp_i = h[i] + d
    hp_j = h[j] - d
    
    # Delta from first sum: (h'[t]-h[t])*(1-h'[t+k-n+1])
    # Nonzero at t=i: d*(1-h'[s1]) where s1=i+k-(n-1)
    # Nonzero at t=j: -d*(1-h'[s2]) where s2=j+k-(n-1)
    
    s1 = i + k - (n - 1)
    s2 = j + k - (n - 1)
    
    # h'[s] = h[s] except at s=i (h[i]+d) and s=j (h[j]-d)
    h_s1 = h[s1[(s1 >= 0) & (s1 < n)]].copy()
    h_s1[s1[(s1 >= 0) & (s1 < n)] == i] = hp_i
    h_s1[s1[(s1 >= 0) & (s1 < n)] == j] = hp_j
    
    h_s2 = h[s2[(s2 >= 0) & (s2 < n)]].copy()
    h_s2[s2[(s2 >= 0) & (s2 < n)] == i] = hp_i
    h_s2[s2[(s2 >= 0) & (s2 < n)] == j] = hp_j
    
    c1 = np.zeros(n_full)
    m1 = (s1 >= 0) & (s1 < n)
    c1[m1] = d * (1.0 - h_s1)
    
    c3 = np.zeros(n_full)
    m3 = (s2 >= 0) & (s2 < n)
    c3[m3] = -d * (1.0 - h_s2)
    
    # Delta from second sum: h[t]*((1-h'[s])-(1-h[s])) = -h[t]*(h'[s]-h[s])
    # Nonzero at s=i (t=i-k+n-1): -h[t]*d
    # Nonzero at s=j (t=j-k+n-1): -h[t]*(-d) = h[t]*d
    
    t1 = i - k + (n - 1)
    t2 = j - k + (n - 1)
    
    c2 = np.zeros(n_full)
    m2 = (t1 >= 0) & (t1 < n)
    c2[m2] = -d * h[t1[m2]]
    
    c4 = np.zeros(n_full)
    m4 = (t2 >= 0) & (t2 < n)
    c4[m4] = d * h[t2[m4]]
    
    return (c1 + c2 + c3 + c4) * dx
```

This is the exact formula and is still O(n). The extra work for handling the special cases is negligible.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_full(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(c.max()), c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_lse(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    if sm > 400:
        return float(m + (sm + np.log(np.exp(s - sm).sum())) / alpha)
    return float(m + np.log(np.exp(s).sum()) / alpha)

def _eq_con(x, n):
    return np.sum(x) - n / 2.0

def _eq_jac(x, n):
    return np.ones(n)

def slsqp(h0, n, alpha, maxiter=4000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_jac, 'args': (n,)}]
    try:
        r = minimize(obj_lse, h0, args=(n, alpha), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(r.x, n)
    except Exception:
        return h0.copy()

def pair_delta(h, n, i, j, d):
    n_full = 2 * n - 1
    k = np.arange(n_full)
    dx = 2.0 / n
    hp_i = h[i] + d
    hp_j = h[j] - d
    
    s1 = i + k - (n - 1)
    m1 = (s1 >= 0) & (s1 < n)
    c1 = np.zeros(n_full)
    if m1.any():
        vals = 1.0 - h[s1[m1]].copy()
        idx = s1[m1]
        vals[idx == i] = 1.0 - hp_i
        vals[idx == j] = 1.0 - hp_j
        c1[m1] = d * vals
    
    s2 = j + k - (n - 1)
    m3 = (s2 >= 0) & (s2 < n)
    c3 = np.zeros(n_full)
    if m3.any():
        vals = 1.0 - h[s2[m3]].copy()
        idx = s2[m3]
        vals[idx == i] = 1.0 - hp_i
        vals[idx == j] = 1.0 - hp_j
        c3[m3] = -d * vals
    
    t1 = i - k + (n - 1)
    m2 = (t1 >= 0) & (t1 < n)
    c2 = np.zeros(n_full)
    if m2.any():
        c2[m2] = -d * h[t1[m2]]
    
    t2 = j - k + (n - 1)
    m4 = (t2 >= 0) & (t2 < n)
    c4 = np.zeros(n_full)
    if m4.any():
        c4[m4] = d * h[t2[m4]]
    
    return (c1 + c2 + c3 + c4) * dx

def fast_pair_search(h, n, c_arr, deadline, deltas):
    best_c = c_arr.max()
    n_full = 2 * n - 1
    for d in deltas:
        if time.time() > deadline:
            return h, best_c
        for _pass in range(5000):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 + 1e-12 and h[j] - d >= -1e-12:
                        delta = pair_delta(h, n, i, j, d)
                        new_max = (c_arr + delta).max()
                        if new_max < best_c - 1e-14:
                            h[i] += d
                            h[j] -= d
                            c_arr += delta
                            best_c = new_max

                    if h[i] - d >= -1e-12 and h[j] + d <= 1.0 + 1e-12:
                        delta = pair_delta(h, n, i, j, -d)

                            h[i] -= d
                            h[j] += d
                            c_arr += delta
                            best_c = new_max
                            improved = True
            if not improved:
                break
    return h, best_c
