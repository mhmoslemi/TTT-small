def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    """Project h onto {0 ≤ h ≤ 1, sum(h) = target_sum}"""
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 0) & (h < 1.0)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgrad(h, lags):
    """Compute subgradient of max correlation at given lags"""
    n = len(h)
    grad = np.zeros(n)
    for lag in lags:
        if lag == 0:
            # c[0] = Σ h[i](1-h[i]) = Σ h[i] - Σ h[i]²
            # dc[0]/dh[i] = 1 - 2h[i]
            grad += (1.0 - 2.0 * h)
        elif 0 < lag < n:
            # c[lag] = Σ_{i=0}^{n-1-lag} h[i](1-h[i+lag])
            # dc/dh[i] for i in [lag, n-1]: from term h[i-lag](1-h[i]) → -(1-h[i-lag])... 
            # Wait, let me be more careful.
            # c[lag] = Σ_{i=0}^{n-1-lag} h[i] * (1 - h[i+lag])
            # dc/dh[j] = sum over terms involving h[j]
            # If j < n-lag: term h[j](1-h[j+lag]) contributes (1-h[j+lag])
            # If j >= lag: term h[j-lag](1-h[j]) contributes -h[j-lag]
            # So: dc/dh[j] = (1-h[j+lag]) for j < n-lag, minus h[j-lag] for j >= lag
            # Combined: dc/dh[j] = (1-h[j+lag]) * [j < n-lag] - h[j-lag] * [j >= lag]
            
            # For j in [0, n-1-lag): contributes (1 - h[j+lag])
            grad[:n-lag] += (1.0 - h[lag:])
            # For j in [lag, n-1]: contributes -h[j-lag]
            grad[lag:] -= h[:n-lag]
        elif -n < lag < 0:
            m = -lag
            # Same as positive lag m by symmetry of the correlation
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]
    return grad

    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -300, 300)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 0) & (h < 1.0)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgrad(h, lags):
    n = len(h)
    grad = np.zeros(n)
    for lag in lags:
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
        elif -n < lag < 0:
            m = -lag
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -300, 300)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

_slp_beta = 1000
_slp_n = 400
_slp_target = 200.0

def _obj_fn(h_in):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, _slp_beta) * (2.0 / _slp_n)

def _con_fn(h_in):
    return h_in.sum() - _slp_target

def _jac_fn(h_in):
    return np.ones(_slp_n)

def slsqp_call(h, target, beta, maxiter):
    global _slp_beta, _slp_n, _slp_target
    _slp_beta = beta
    _slp_n = len(h)
    _slp_target = target
    bounds = Bounds(np.zeros(_slp_n), np.ones(_slp_n))
    try:
        res = minimize(_obj_fn, h, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': _con_fn, 'jac': _jac_fn}],
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_h(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def subgrad_loop(h, target, t0, budget, max_iters, m_lags, alpha0,
                 pert_every, pert_scale, momentum):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    no_improve = 0
    alpha = alpha0
    steps = [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.125, alpha * 0.0625]

    for it in range(max_iters):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        m_act = min(m_lags, n // 5)
        top_idx = np.argsort(corr)[::-1][:m_act]
        lags = [int(i) - (n - 1) for i in top_idx]
        g = compute_subgrad(best, lags)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        vel = momentum * vel + (1.0 - momentum) * g
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        d = -vel / vn
        found = False
        for step in steps:
            ht = project_h(best + step * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-14:
                best = ht
                best_c5 = c5t
                no_improve = 0
                found = True
                if step <= alpha * 0.1:
                    alpha *= 0.7
                    steps = [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.125, alpha * 0.0625]
                break
        if not found:
            no_improve += 1
            if no_improve % 200 == 0:
                alpha *= 0.8
                steps = [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.125, alpha * 0.0625]
        if no_improve >= pert_every and it % pert_every == pert_every - 1:
            scale = pert_scale * (1.0 + 0.4 * (no_improve // pert_every))
            p = np.random.randn(n) * scale
            ht = project_h(best + p, target)
            c5t = compute_c5(ht)
            if c5t < best_c5:
                best = ht
                best_c5 = c5t
            vel *= 0.3
            no_improve = 0
            alpha = max(alpha * 1.3, 0.0005)
            steps = [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.125, alpha * 0.0625]
        if no_improve > pert_every * 30:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(400, 0.5)

    n = len(best_h)
    target = n / 2.0
    best_c5 = compute_c5(best_h)

    def rem_time():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP warmup with increasing smoothness
    if rem_time() > 90:
        for beta in [150, 600, 2500]:
            if rem_time() < 25:
                break
            h_new, c5_new = slsqp_call(best_h, target, beta, 2500)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 2: Main subgradient - long run, moderate active set
    if rem_time() > 150:

            best_h, target, t0, budget_s,
            max_iters=200000, m_lags=35, alpha0=0.003,
            pert_every=350, pert_scale=0.003, momentum=0.89
        )

    # Phase 3: SLSQP with high beta for precise local optimization
    if rem_time() > 50:
        h_new, c5_new = slsqp_call(best_h, target, 8000, 4000)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 4: Fine subgradient with larger active set and smaller steps
    if rem_time() > 40:

    # Phase 5: Final SLSQP polish
