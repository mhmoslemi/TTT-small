If we have a (v, v/2, λ) difference set, then for each nonzero lag, the number of pairs (i, i+k) both in the set is λ. But we want the number of pairs where i is in the set and i+k is NOT in the set.

If the set S has size v/2, then:
- |S ∩ (S-k)| = number of i where both i and i+k are in S
- |S \ (S-k)| = number of i where i is in S but i+k is not = |S| - |S ∩ (S-k)| = v/2 - |S ∩ (S-k)|

So we want to minimize max_k (v/2 - |S ∩ (S-k)|), which is equivalent to maximizing min_k |S ∩ (S-k)|.

For a balanced difference set, |S ∩ (S-k)| = λ for all nonzero k. If v = 4m+1 or v = 4m+3 (Singer difference sets), then k = v/2... hmm, this requires v to be odd for a standard difference set.

For even v, we might look at "near difference sets" or other structures.

Anyway, this combinatorial approach suggests that the optimal solution might have a quasi-periodic structure. Let me not go down this path and focus on numerical optimization.

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_logsumexp(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    e = np.exp(shifted)
    return m + np.log(np.sum(e)) / beta

def obj_lse(h, dx, beta, lam, mu):
    corr = np.correlate(h, 1.0 - h, mode='full')
    sm = smooth_max_logsumexp(corr, beta) * dx
    c = h.sum() - len(h) / 2.0
    return sm + lam * c + 0.5 * mu * c * c

def grad_lse(h, dx, beta, lam, mu):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(shifted)
    se = np.sum(e)
    weights = e / se  # d logsumexp / d corr_k
    
    offset = n - 1
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-10:
            continue
        l = k - offset
        if l >= 0:
            i_range = range(n - l)
            for i in i_range:
                grad[i + l] += w * (1.0 - h[i])
                grad[i] -= w * h[i + l]
        else:
            la = -l
            for i in range(n - la):
                grad[i] += w * (1.0 - h[i + la])
                grad[i + la] -= w * h[i]
    
    c = h.sum() - n / 2.0
    grad += (lam + mu * c)
    return grad * dx

def lgbfgs_optimize(h, n, deadline, seed):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    c5 = compute_c5(h)
    
    for beta in [50, 200, 1000, 5000, 20000]:
        if time.time() > deadline:
            break
        lam = 0.0
        mu = 1.0
        for aug_iter in range(20):
            if time.time() > deadline:
                break
            res = minimize(obj_lse, h, args=(dx, beta, lam, mu),
                          jac=grad_lse, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 200, 'ftol': 1e-14})
            h_new = project_feasible(res.x, ts)
            c_new = compute_c5(h_new)
            c = h_new.sum() - ts
            lam = lam + mu * c
            if abs(c) < 1e-8:
                mu = max(mu * 0.5, 0.1)
            else:
                mu = min(mu * 2.0, 1e4)
            if c_new < c5:
                h = h_new
                c5 = c_new
            if abs(c) < 1e-10 and abs(lam) < 1e-8:
                break
    return h, c5

def subgrad_topk(h, n, deadline, seed, k_top=3):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    velocity = np.zeros(n)
    lr = 0.003
    stagnant = 0
    
    for it in range(5000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-k_top:]
        w = np.exp(20.0 * (corr[top_idx] - corr[top_idx].max()))
        w /= w.sum()
        
        offset = n - 1
        grad = np.zeros(n)
        for j, k in enumerate(top_idx):
            l = k - offset
            if l >= 0:
                for i in range(n - l):
                    grad[i + l] += w[j] * (1.0 - h[i])
                    grad[i] -= w[j] * h[i + l]
            else:
                la = -l
                for i in range(n - la):
                    grad[i] += w[j] * (1.0 - h[i + la])
                    grad[i + la] -= w[j] * h[i]
        
        velocity = 0.85 * velocity - lr * grad
        h_new = project_feasible(h + velocity, ts)
        c_new = compute_c5(h_new)
        
        if c_new < c5 - 1e-12:
            h = h_new
            c5 = c_new
            stagnant = 0
            lr = min(lr * 1.01, 0.006)
        else:
            stagnant += 1
            velocity *= 0.5
            lr *= 0.9
        if stagnant > 200 and lr < 1e-5:
            break
    return h, c5

def snap_quantize(h, n, targets=None):
    if targets is None:
        targets = [0.0, 0.3, 0.5, 0.7, 1.0]
    h_snap = h.copy()
    for t in targets:
        mask = np.abs(h_snap - t) < 0.15
        if mask.sum() > 0:
            h_snap[mask] = t
    h_snap = project_feasible(h_snap, n / 2.0)
    return h_snap

def basin_hopping(h, n, deadline, seed, c5_current):
    np.random.seed(seed)
    ts = n / 2.0
    best = h.copy()
    best_c5 = c5_current
    for it in range(50):
        if time.time() > deadline:
            break
        scale = [0.01, 0.03, 0.05, 0.08][it % 4]
        h_pert = h.copy()
        n_pert = np.random.randint(n // 4, n // 2)
        idx = np.random.choice(n, n_pert, replace=False)
        h_pert[idx] += np.random.randn(n_pert) * scale
        h_pert = project_feasible(h_pert, ts)
        c_new = compute_c5(h_pert)
        # Local refine
        if c_new < best_c5 + 0.01:
            dx = 2.0 / n
            for beta in [500, 5000]:
                if time.time() > deadline:
                    break
                try:
                    res = minimize(obj_lse, h_pert, args=(dx, beta, 0.0, 1.0),
                                  jac=grad_lse, method='L-BFGS-B',
                                  bounds=Bounds(np.zeros(n), np.ones(n)),
                                  options={'maxiter': 100, 'ftol': 1e-13})
                    h_pert = project_feasible(res.x, ts)
                    c_new = compute_c5(h_pert)
                except:
                    pass
        if c_new < best_c5:
            best = h_pert.copy()
            best_c5 = c_new
            h = best.copy()
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:

    if parent is None:

    h2, c2 = subgrad_topk(best_h.copy() if best_h is not None else parent, best_n, d2, seed + 1)
    if c2 < best_c5:

    # Phase 3: Basin hopping (25% time)
    d3 = t0 + budget_s * 0.80
    h3, c3 = basin_hopping(best_h.copy(), best_n, d3, seed + 100, best_c5)
    if c3 < best_c5:
        best_h, best_c5, best_n = h3.copy(), c3, best_n
    
    # Phase 4: Try other resolutions (10% time)
    d4 = t0 + budget_s * 0.90
    for n_try in [pn - 10, pn + 10, pn - 20, pn + 20, 300, 400]:
        if time.time() > d4:
            break
        if n_try < 100 or n_try > 600 or n_try == best_n:
            continue
        x_old = np.linspace(0, 2, len(parent), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h0 = np.interp(x_new, x_old, parent)
        h0 = project_feasible(h0, n_try / 2.0)

    # Phase 5: Final high-precision polish (10% time)
    d5 = t0 + budget_s * 0.98
    if best_h is not None:
        h5, c5_final = lgbfgs_optimize(best_h.copy(), best_n, d5, seed + 500)
        if c5_final < best_c5:

        h5, c5_final = subgrad_topk(best_h.copy(), best_n, d5, seed + 600, k_top=2)
        if c5_final < best_c5:
            best_h, best_c5 = h5, c5_final
    
    if best_h is None:

    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
```

Wait, I have a concern: the gradient computation in `grad_lse` is O(n^2) which for n=350 is about 122,500 operations. That's fine for L-BFGS-B which typically does O(n) function evaluations.

But actually, the nested loop in the gradient is the bottleneck. Let me think about whether I can vectorize it.

For a given lag l >= 0:
- grad[i+l] += w * (1-h[i]) for i in range(n-l)
- grad[i] -= w * h[i+l] for i in range(n-l)

This can be vectorized:
- grad[l:nl] += w * (1 - h[0:nl-l])  where nl = n
- grad[0:nl-l] -= w * h[l:]

Let me vectorize:

```python
def grad_lse(h, dx, beta, lam, mu):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    e = np.exp(shifted)
    se = np.sum(e)
    weights = e / se
    
    offset = n - 1
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-10:
            continue
        l = k - offset
        if l >= 0:
            grad[l:] += w * (1.0 - h[:n-l])
            grad[:n-l] -= w * h[l:]
        else:
            la = -l
            grad[:n-la] += w * (1.0 - h[la:])
            grad[la:] -= w * h[:n-la]
    
    c = h.sum() - n / 2.0
    grad += (lam + mu * c)
    return grad * dx
```
