    momentum = 0.82
    stall = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 5:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if 0 <= lag < n:
                grad[:n-lag] -= (1.0 - cur[lag:])
                grad[lag:] += cur[:n-lag]
            elif -n < lag < 0:
                la = -lag
                grad[la:] -= (1.0 - cur[:n-la])
                grad[:n-la] += cur[la:]
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        mom = momentum * mom - step * grad
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-15:
            cur = trial
            best_c5 = c_try
            best_h = cur.copy()
            stall = 0
            step = min(step * 1.02, 0.12)
        else:
            stall += 1
            step *= 0.87
            mom *= 0.4
            if stall > 30:
                mom = np.zeros(n)
                trial = project_h(cur - step * grad, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-15:
                    cur = trial
                    best_c5 = c_try
                    best_h = cur.copy()
                    stall = 0
                else:
                    step *= 0.8
                    if step < 0.001:
                        break
    return best_h, best_c5

def binary_anneal(h, n, t0, budget, max_iter=200000, seed=99):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    cur = np.zeros(n)
    cur[idx[:n_ones]] = 1.0
    best_c5 = compute_c5(cur)
    best_h = cur.copy()
    rng = np.random.default_rng(seed)
    for it in range(max_iter):
        if budget - (time.time() - t0) < 5:
            break
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        a = ones[rng.integers(len(ones))]
        b = zeros[rng.integers(len(zeros))]
        cur[a] = 0.0
        cur[b] = 1.0
        new_c5 = compute_c5(cur)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = cur.copy()
        else:
            frac = it / max_iter
            temp = 0.015 * (1e-4) ** frac
            if rng.random() < np.exp(-(new_c5 - best_c5) / max(temp, 1e-15)):
                pass
            else:
                cur[a] = 1.0
                cur[b] = 0.0
    return best_h, best_c5

def resample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    old_x = np.linspace(0, 2, n_old, endpoint=False)
    new_x = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def multi_start_optimize(h_best, n, target, t0, budget, seed, n_starts=4):
    rng = np.random.default_rng(seed)
    best_h = h_best.copy()
    best_c5 = compute_c5(best_h)
    for s in range(n_starts):
        if budget - (time.time() - t0) < 30:
            break
        scale = 0.03 * (0.6 ** s)
        h_try = best_h + rng.standard_normal(n) * scale
        h_try = project_h(h_try, target)
        for beta in [800, 4000]:
            if budget - (time.time() - t0) < 15:
                break
            h_try = slsqp_optimize(h_try, n, target, beta, maxiter=6000)
        h_try, c_try = subgrad_optimize(h_try, target, n, t0, budget, top_k=16, max_iter=10000, step_init=0.025)
        if c_try < best_c5:
            best_h = h_try
            best_c5 = c_try
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 384
        h_init = np.full(n_init, 0.5)
        h_init[:n_init//2] = 0.8
        h_init[n_init//2:] = 0.2
        h_init = project_h(h_init, n_init / 2.0)

    n = len(h_init)
    target = n / 2.0
    best_h = project_h(h_init, target)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick SLSQP cleanup
    for beta in [300, 1000]:
        if budget_s - (time.time() - t0) < 15:
            break
        h_try = slsqp_optimize(best_h, n, target, beta, maxiter=6000)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            best_h = h_try
            best_c5 = c_try

    # Phase 2: Binary anneal (two attempts with different seeds)
    for bseed in [77, 444]:
        if budget_s - (time.time() - t0) < 60:
            break
        h_bin, c_bin = binary_anneal(best_h, n, t0, budget_s, max_iter=120000, seed=bseed)
        # Refine from binary
        h_try = h_bin + 0.01 * np.random.randn(n)
        h_try = project_h(h_try, target)
        for beta in [1000, 5000]:
            if budget_s - (time.time() - t0) < 15:
                break
            h_try = slsqp_optimize(h_try, n, target, beta, maxiter=8000)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            best_h = h_try
            best_c5 = c_try

    # Phase 3: Subgradient with varying top_k
    for tk in [12, 24, 40]:
        if budget_s - (time.time() - t0) < 25:
            break
        h_try, c_try = subgrad_optimize(best_h, target, n, t0, budget_s,
                                         top_k=tk, max_iter=20000, step_init=0.03)
        if c_try < best_c5:
            best_h = h_try
            best_c5 = c_try

    # Phase 4: High-beta SLSQP (progressive sharpening)
    for beta in [3000, 10000, 30000, 100000]:
        if budget_s - (time.time() - t0) < 12:
            break
        h_try = slsqp_optimize(best_h, n, target, beta, maxiter=12000, ftol=1e-17)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            best_h = h_try
            best_c5 = c_try

    # Phase 5: Multi-start perturbation
    if budget_s - (time.time() - t0) > 60:
        best_h, best_c5 = multi_start_optimize(best_h, n, target, t0, budget_s, seed+55, n_starts=4)

    # Phase 6: Resolution lift to 512
    if budget_s - (time.time() - t0) > 80:
        n_hr = 512
        target_hr = n_hr / 2.0
        h_hr = project_h(resample(best_h, n_hr), target_hr)
        for beta in [500, 2000, 8000]:
            if budget_s - (time.time() - t0) < 20:
                break
            h_hr = slsqp_optimize(h_hr, n_hr, target_hr, beta, maxiter=6000)
        h_hr, c_try = subgrad_optimize(h_hr, target_hr, n_hr, t0, budget_s,
                                        top_k=20, max_iter=15000, step_init=0.025)
        c_try = compute_c5(h_hr)
        if c_try < best_c5:
            best_h = h_hr
            best_c5 = c_try
            n = n_hr
            target = target_hr

    # Phase 7: Continue subgradient + SLSQP at current resolution
    if budget_s - (time.time() - t0) > 30:
        h_try, c_try = subgrad_optimize(best_h, target, n, t0, budget_s,
                                         top_k=20, max_iter=20000, step_init=0.02)
        if c_try < best_c5:
            best_h = h_try
            best_c5 = c_try

    # Phase 8: Final high-beta SLSQP
    for beta in [50000, 200000]:
        if budget_s - (time.time() - t0) < 10:
            break
        h_try = slsqp_optimize(best_h, n, target, beta, maxiter=15000, ftol=1e-18)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            best_h = h_try
            best_c5 = c_try

    # Phase 9: Final subgradient polish
    if budget_s - (time.time() - t0) > 10:
        h_try, c_try = subgrad_optimize(best_h, target, n, t0, budget_s,
                                         top_k=32, max_iter=15000, step_init=0.01)
        if c_try < best_c5:
            best_h = h_try
            best_c5 = c_try

    return best_h, best_c5, n
