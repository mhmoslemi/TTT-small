import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def analytic_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sh -= sh.max()
    w = np.exp(sh)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w /= wsum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - n + 1
        if s >= 0:
            # (1-h[j+s]) for j in [0, n-s-1]
            if n - s > 0:
                grad[0:n-s] += wk * (1.0 - h[s:n])
            # -h[j-s] for j in [s, n-1]
            if s < n:
                grad[s:n] -= wk * h[0:n-s]
        else:
            ss = -s  # positive
            # (1-h[j-s]) for j in [ss, n-1]
            grad[ss:n] += wk * (1.0 - h[0:n-ss])
            # -h[j+s] for j in [0, n-ss-1]
            if n - ss > 0:
                grad[0:n-ss] -= wk * h[ss:n]
    return grad * dx

def nesterov_grad(h, n, deadline, alpha=100000, max_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    lr = 0.01
    beta = 0.9
    for it in range(max_iter):
        if time.time() > deadline:
            break
        y = h + beta * v
        y = np.clip(y, 0.0, 1.0)
        gy = analytic_grad(y, n, alpha)
        v = beta * v - lr * gy
        h_new = h - v  # wait, Nesterov: h_new = h - lr * grad(y) + momentum
        # Standard Nesterov: y = h + beta*v; grad_y = grad(y); v_new = beta*v - lr*grad_y; h_new = h + v_new
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.95
            h_new = h  # reject step
            if lr < 1e-14:
                break
        h = h_new
    return best_h, best_c

def slsqp_step(h, n, alpha, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def con_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def analytic_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sh -= sh.max()
    w = np.exp(sh)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w /= wsum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - n + 1
        if s >= 0:
            if n - s > 0:
                grad[0:n-s] += wk * (1.0 - h[s:n])
            if s < n:
                grad[s:n] -= wk * h[0:n-s]
        else:
            ss = -s
            grad[ss:n] += wk * (1.0 - h[0:n-ss])
            if n - ss > 0:
                grad[0:n-ss] -= wk * h[ss:n]
    return grad * dx

def projected_grad(h, n, deadline, alpha=100000, max_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.5
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = analytic_grad(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = lr / gn
        found = False
        for _ in range(40):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                h = h_new
                found = True
                break
            step *= 0.5
        if not found:
            lr *= 0.85
            stall += 1
            if lr < 1e-14 or stall > 200:
                break
        else:
            stall = 0
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            lr = min(lr * 1.03, 50.0)
    return best_h, best_c

def slsqp_step(h, n, alpha, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def con_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def binary_search(h, n, deadline, max_sweeps=400):
    hb = to_binary(h, n)
    best_c = compute_c5(hb, n)
    ones = set(np.where(hb == 1)[0].tolist())
    zeros = set(np.where(hb == 0)[0].tolist())
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in sorted(zeros):
            if time.time() > deadline:
                break
            for j in sorted(ones):
                hb[i], hb[j] = 1.0, 0.0
                c = compute_c5(hb, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    ones.discard(j)
                    ones.add(i)
                    zeros.discard(i)
                    zeros.add(j)
                    break
                else:
                    hb[i], hb[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    return hb, best_c

def nesterov_step(h, n, deadline, alpha=200000, max_iter=30000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    lr = 0.01
    beta = 0.9
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        y = h + beta * v
        y = np.clip(y, 0.0, 1.0)
        gy = analytic_grad(y, n, alpha)
        v = beta * v - lr * gy
        h_new = h + v
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            h = h_new
            stall = 0
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            lr = min(lr * 1.01, 5.0)
        else:
            lr *= 0.9
            stall += 1
            if lr < 1e-15 or stall > 100:
                break
    return best_h, best_c
