```
def erdos_c5(h,n):

def project_box_sum(x,target):
    x=np.asarray(x,dtype=float)
    n=x.size
    y=np.clip(x,0.0,1.0)
    if abs(float(y.sum()-target))<1e-14:
        return y
    lo=float(np.min(x))-1.0
    hi=float(np.max(x))+1.0
    for _ in range(80):
       mid=0.5*(lo+hi)
       y=np.clip(x-mid,0.0,1.0)
       if float(y.sum())>target:
          lo=mid
       else:
          hi=mid
    y=np.clip(x-0.5*(lo+hi),0.0,1.0)
    diff=float(target-y.sum())
    if abs(diff)>1e-12:
       mask=(y>1e-14)&(y<1.0-1e-14)
       cnt=int(np.count_nonzero(mask))
       if cnt>0:
          y[mask]+=diff/cnt
       else:
```

```python
import time
import numpy as np
import scipy.optimize as so

LS = {'n': 1, 'dx': 1.0, 'target': 0.5, 'beta': 1.0, 'rho': 1e5, 'lam': 0.0}
SS = {'n': 1, 'dx': 1.0, 'target': 0.5, 'beta': 1.0, 'cache_x': None, 'cache_g': None}


def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(3):
        if abs(diff) <= 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y


def softmax_values(c, beta):
    maxc = float(np.max(c))
    z = beta * (c - maxc)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)
    ws = float(ew.sum())
    f = maxc + np.log(ws) / beta
    if ws <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ew / ws
    return f, w


def smooth_grad_from_w(h, w, n, dx):
    mid = n - 1
    if n == 1:
        return (1.0 - 2.0 * h) * w[mid] * dx
    w_pos = np.zeros(n)
    w_neg = np.zeros(n)
    w_pos[1:] = w[mid + 1:]
    w_neg[1:] = w[:mid][::-1]
    s = w_pos + w_neg
    pref_p = np.cumsum(w_pos)
    pref_n = np.cumsum(w_neg)
    idx = np.arange(n)
    consts = pref_p + pref_n[n - 1 - idx]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx


def l_fg(x):
    st = LS
    n = int(st['n'])
    dx = float(st['dx'])
    target = float(st['target'])
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, w = softmax_values(c, float(st['beta']))
    g = smooth_grad_from_w(x, w, n, dx)
    res = float(x.sum() - target)
    rho = float(st['rho'])
    lam = float(st['lam'])
    if rho != 0.0 or lam != 0.0:
        f += lam * res + 0.5 * rho * res * res
        g = g + (lam + rho * res)
    return float(f), g


def refine_lbfgs(h, beta, rho, maxiter, deadline):
    h = np.asarray(h, dtype=float)
    n = len(h)
    target = 0.5 * n
    rem = deadline - time.time()
    if rem < 3.0:
        return h, erdos_c5(h, n)
    LS['n'] = n
    LS['dx'] = 2.0 / n
    LS['target'] = target
    LS['beta'] = float(beta)
    LS['rho'] = float(rho)
    LS['lam'] = 0.0
    if n > 600:
        maxiter = max(15, maxiter // 2)
    mx = int(min(maxiter, max(18, int(rem / 3.0))))
    bounds = [(0.0, 1.0)] * n
    try:
        res = so.minimize(l_fg, h, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': mx, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxls': 40, 'disp': False})
        cand = project_box_sum(res.x, target)
    except Exception:
        cand = project_box_sum(h, target)
    return cand, erdos_c5(cand, n)


def s_compute(x):
    st = SS
    n = int(st['n'])
    dx = float(st['dx'])
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, w = softmax_values(c, float(st['beta']))
    g = smooth_grad_from_w(x, w, n, dx)
    st['cache_x'] = x.copy()
    st['cache_g'] = g
    return f, g


def s_obj(x):
    f, g = s_compute(x)
    return float(f)


def s_jac(x):
    st = SS
    cx = st.get('cache_x')
    if cx is not None and np.array_equal(x, cx):
        return st.get('cache_g').copy()
    f, g = s_compute(x)
    return g


def s_con_fun(x):
    return float(x.sum() - SS['target'])


def s_con_jac(x):
    return np.ones(int(SS['n']), dtype=float)


def refine_slsqp(h, beta, maxiter, deadline):
    h = np.asarray(h, dtype=float)
    n = len(h)
    target = 0.5 * n
    rem = deadline - time.time()
    if rem < 6.0:
        return h, erdos_c5(h, n)
    SS['n'] = n
    SS['dx'] = 2.0 / n
    SS['target'] = target
    SS['beta'] = float(beta)
    SS['cache_x'] = None
    SS['cache_g'] = None
    if n > 600:
        maxiter = max(8, maxiter // 2)
    mx = int(min(maxiter, max(8, int(rem / 8.0))))
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': s_con_fun, 'jac': s_con_jac}]
    try:
        res = so.minimize(s_obj, h, method='SLSQP', jac=s_jac,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': mx, 'iprint': 0, 'ftol': 1e-14})
        cand = project_box_sum(res.x, target)
    except Exception:
        cand = project_box_sum(h, target)
    return cand, erdos_c5(cand, n)


def pair_obj(delta, h, i, j, n):
    a = h[i] + delta
    b = h[j] - delta
    if a < -1e-12 or a > 1.0 + 1e-12 or b < -1e-12 or b > 1.0 + 1e-12:
        return 1e100
    hh = h.copy()
    hh[i] = a
    hh[j] = b
    return erdos_c5(hh, int(n))


def pair_line(h, n, i, j, cur, deadline):
    hi0 = h[i]
    hj0 = h[j]
    lo = max(-hi0, hj0 - 1.0)
    hi = min(1.0 - hi0, hj0)
    if hi - lo < 1e-13:
        return h, cur
    best_d = 0.0
    best_c = cur
    if time.time() > deadline:
        return h, cur
    c0 = pair_obj(lo, h, i, j, n)
    if c0 < best_c:
        best_c = c0
        best_d = lo
    c1 = pair_obj(hi, h, i, j, n)
    if c1 < best_c:
        best_c = c1
        best_d = hi
    try:
        if time.time() < deadline - 0.3:
            rs = so.minimize_scalar(pair_obj, args=(h, i, j, n),
                                    bounds=(lo, hi), method='bounded',
                                    options={'maxiter': 25, 'xatol': 1e-10})
            if rs.x is not None:
                c2 = pair_obj(float(rs.x), h, i, j, n)
                if c2 < best_c:
                    best_c = c2
                    best_d = float(rs.x)
    except Exception:
        pass
    if best_c < cur - 1e-14:
        hh = h.copy()
        hh[i] += best_d
        hh[j] -= best_d
        hh = np.clip(hh, 0.0, 1.0)
        return hh, best_c
    return h, cur


def pair_polish(h, n, deadline, seed):
    rng = np.random.RandomState(int(seed))
    h = np.asarray(h, dtype=float)
    cur = erdos_c5(h, n)
    m = 25 if n <= 500 else 15
    frac = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
    if frac.size:
        score = np.minimum(h[frac], 1.0 - h[frac])
        order = frac[np.argsort(score)[::-1]]
    else:
        order = np.array([], dtype=int)
    if order.size >= m:
        top = order[:m]
    else:
        need = m - int(order.size)
        if need > n - int(order.size):
            need = max(0, n - int(order.size))
        if need > 0:
            extra = rng.choice(n, size=need, replace=False)
            top = np.unique(np.concatenate((order, extra)))
        else:
            top = order
    if top.size < 2:
        return h, cur
    for _ in range(3):
        improved = False
        pairs = []
        tlen = len(top)
        for a in range(tlen):
            for b in range(a + 1, tlen):
                pairs.append((int(top[a]), int(top[b])))
        if not pairs:
            break
        rng.shuffle(pairs)
        for i, j in pairs:
            if time.time() > deadline - 4.0:
                return h, erdos_c5(h, n)
            hh, cc = pair_line(h, n, i, j, cur, deadline - 4.0)
            if cc < cur - 1e-14:
                h = hh
                cur = cc
                improved = True
        if not improved:
            break
    return h, cur


def resample_n(h, n, nh):
    x = np.linspace(0.0, 2.0, int(n))
    y = np.linspace(0.0, 2.0, int(nh))
    hh = np.interp(y, x, np.asarray(h, dtype=float))
    return project_box_sum(hh, 0.5 * nh)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(10.0, float(budget_s) - 5.0)
    deadline = t0 + total
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h))
    except Exception:
        n = 400
        h = np.full(n, 0.5, dtype=float)
    if n < 2:
        n = 400
        h = np.full(n, 0.5, dtype=float)
    if n > 800:
        nold = n
        n = 800
        h = np.interp(np.linspace(0.0, 2.0, n), np.linspace(0.0, 2.0, nold), h)
    h = project_box_sum(h, 0.5 * n)
    best_h = h.copy()
    best_n = n
    best_c = erdos_c5(best_h, best_n)

    end1 = t0 + 0.36 * total
    end2 = t0 + 0.62 * total
    end3 = t0 + 0.84 * total
    end4 = deadline - 3.0

    lb_betas = [50.0, 200.0, 800.0, 3000.0, 12000.0, 60000.0, 250000.0, 1200000.0]
    lb_iters = [120, 110, 110, 100, 90, 80, 70, 55]
    for beta, mx in zip(lb_betas, lb_iters):
        if time.time() > end1 - 8.0:
            break
        cand, c = refine_lbfgs(best_h, beta, 1e5, mx, deadline)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c

    sl_betas = [300.0, 2000.0, 10000.0, 50000.0, 250000.0, 1000000.0, 4000000.0]
    sl_iters = [70, 70, 60, 55, 45, 35, 28]
    for beta, mx in zip(sl_betas, sl_iters):
        if time.time() > end2 - 8.0:
            break
        cand, c = refine_slsqp(best_h, beta, mx, deadline)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c

    for nh in (500, 300):
        if time.time() > end3 - 18.0 or nh == best_n:
            continue
        hh = resample_n(best_h, best_n, nh)
        hh, c = refine_lbfgs(hh, 800.0, 1e5, 70, deadline)
        if c < best_c + 2e-3:
            hh, c = refine_lbfgs(hh, 80000.0, 1e5, 60, deadline)
            if c < best_c - 1e-15:
                best_h = hh.copy()
                best_n = nh
                best_c = c
                if time.time() < end3 - 8.0:
                    hh, c = refine_slsqp(hh, 200000.0, 35, deadline)
                    if c < best_c - 1e-15:
                        best_h = hh.copy()
                        best_c = c

    if best_n <= 800 and time.time() < end4 - 25.0:
        best_h, best_c = pair_polish(best_h, best_n, end4 - 18.0, seed)

    if time.time() < end4 - 12.0:
        cand, c = refine_lbfgs(best_h, 1000000.0, 1e5, 35, deadline)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c

    if time.time() < end4 - 8.0:
        cand, c = refine_slsqp(best_h, 500000.0, 28, deadline)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c

    rng = np.random.RandomState(int(seed) + 17)
    for sig, sd in ((0.02, 101), (0.05, 202), (0.10, 303)):
        if time.time() > end4 - 8.0:
            break
        hh = project_box_sum(best_h + sig * rng.randn(best_n), 0.5 * best_n)
        hh, c = refine_lbfgs(hh, 40000.0, 1e5, 35, deadline)
        if c < best_c - 1e-15:
            best_h = hh.copy()
            best_c = c

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = erdos_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
```
