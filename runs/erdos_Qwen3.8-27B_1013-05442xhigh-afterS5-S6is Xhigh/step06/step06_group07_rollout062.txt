    dx = 2.0 / n

    cmax = c.max()
    shifted = c - cmax
    exp_c = np.exp(alpha * shifted)
    sum_exp = exp_c.sum()
    val = cmax + np.log(sum_exp) / alpha
    
    p = exp_c / sum_exp
    arange = np.arange(n)

    m2 = arange[:, None] - arange[None, :] + n - 1

    T2 = p[m2] @ x

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_value_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    cmax = c.max()
    shifted = c - cmax
    exp_c = np.exp(alpha * shifted)
    sum_exp = exp_c.sum()
    val = cmax + np.log(sum_exp) / alpha
    p = exp_c / sum_exp
    ar = np.arange(n)
    m1 = ar[None, :] - ar[:, None] + n - 1
    m2 = ar[:, None] - ar[None, :] + n - 1
    T1 = p[m1] @ (1.0 - x)
    T2 = p[m2] @ x
    grad = dx * (T1 - T2)
    return val, grad

def lbfgs_opt(h, n, alpha, maxiter=5000, pen=500.0):
    target = n / 2.0
    def obj(x):
        v, _ = lse_value_grad(x, alpha)
        return v + pen * (x.sum() - target) ** 2
    def grad(x):
        _, g = lse_value_grad(x, alpha)
        return g + 2 * pen * (x.sum() - target) * np.ones(n)
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except:
        return h

def slsqp_opt(h, n, alpha, maxiter=2000):
    def obj(x):
        v, _ = lse_value_grad(x, alpha)
        return v
    def grad(x):
        _, g = lse_value_grad(x, alpha)
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=grad,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except:
        return h

def grad_descent(h, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_h = h.copy()
    cur_c = best_c
    lr = 0.005
    alpha_ls = 200
    no_improve = 0
    while time.time() < deadline:
        _, grad = lse_value_grad(cur_h, alpha_ls)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        direction = -grad / gnorm
        improved = False
        for step in [lr, lr * 0.5, lr * 0.1, lr * 0.01]:
            h_new = project_feasible(cur_h + step * direction, n)
            c_new = compute_c5(h_new, n)
            if c_new < cur_c - 1e-15:
                cur_h = h_new
                cur_c = c_new
                improved = True
                break
        if improved:
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
                no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Reheat
                alpha_ls = int(rng.choice([50, 100, 300, 500, 1000]))
                cur_h = project_feasible(cur_h + 0.02 * rng.randn(n), n)
                cur_c = compute_c5(cur_h, n)
                lr = 0.005
                no_improve = 0
            else:
                lr *= 0.95
    return best_h, best_c
