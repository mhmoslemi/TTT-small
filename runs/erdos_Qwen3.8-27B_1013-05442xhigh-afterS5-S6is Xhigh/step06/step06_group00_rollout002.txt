import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13: break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_objective_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    e = np.exp(alpha * (c - cmax))
    w = e / e.sum()
    obj = (cmax + np.log(e.sum()) / alpha)  # LSE approximation to max
    # Gradient
    grad = np.zeros(n)
    nmoins = n - 1
    for s_idx in range(2 * n - 1):
        ws = w[s_idx]
        if ws < 1e-14: continue
        s = s_idx - nmoins
        if s >= 0:
            for j in range(n - s):
                g = 1.0 - h[j + s]
                if j - s >= 0:
                    g -= h[j - s]
                grad[j] += ws * g
        else:
            for j in range(-s, n):
                g = 1.0 - h[j + s]
                if j - s >= 0:
                    g -= h[j - s]
                grad[j] += ws * g
    grad *= dx
    return obj, grad

    h = h.copy()
    best_h = h.copy()
    best_obj = compute_c5(h, n)

        h = h_new
