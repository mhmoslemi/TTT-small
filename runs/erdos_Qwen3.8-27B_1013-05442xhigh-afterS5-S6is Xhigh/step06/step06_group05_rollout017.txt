from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    """Convert continuous h to binary with exactly n/2 ones."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def targeted_local_search(h_init, n, deadline, n_restarts=5, seed=42):

    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)
    best_corr_max = best_c / dx
    rng = np.random.RandomState(seed)

            break

                if len(ones) > 0 and len(zeros) > 0:
                    cur[rng.choice(ones)] = 0.0
                    cur[rng.choice(zeros)] = 1.0
        
        cur_corr = np.correlate(cur, 1.0 - cur, mode='full').astype(np.float64)
        cur_max = cur_corr.max()

        T = 3.0
        T_min = 0.05
        cooling = 0.9995
        max_iter = 50000

        while iter_count < max_iter and time.time() < deadline:
            iter_count += 1
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break

            # Strategy: look at critical lags and find good swaps
            top_k = 3
            crit_lags = np.argsort(cur_corr)[-top_k:]

            best_delta = 0  # we want negative delta (decrease in max)

    """Compute the change in correlation vector when flipping i:0->1 and j:1->0."""

        if time.time() > deadline:
            break
        c = np.correlate(cur, 1.0 - cur, mode='full')
        c_max = c.max()
        if c_max * dx < best_c:
            best_c = c_max * dx
            best_h = cur.copy()
        # Subgradient from top-k lags
        top_idx = np.argsort(c)[-top_k:]
        grad = np.zeros(n)
        idx_n = np.arange(n)
        for k_idx in top_idx:
            lag = lag_arr[k_idx]
            gk = np.ones(n)
            ip = idx_n + lag
            im = idx_n - lag
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            gk[vp] -= cur[ip[vp]]
            gk[vm] -= cur[im[vm]]
            grad += gk
        grad /= top_k
        cur = cur - lr * grad
        cur = project_feasible(cur, n)
        lr = max(lr * 0.998, 0.01)
    return best_h, best_c

def targeted_sa(h_init, n, deadline, seed=42):
    """Fast targeted simulated annealing for binary sequences."""
    dx = 2.0 / n
    full_lags = 2 * n - 1
    lag_arr = np.arange(full_lags) - (n - 1)
    best_h = h_init.copy()
    best_curve = np.correlate(best_h, 1.0 - best_h, mode='full').astype(np.float64)
    best_max = best_curve.max()
    rng = np.random.RandomState(seed)
    T = 4.0
    T_min = 0.05
    cooling = 0.9995
    cur = h_init.copy()
    cur_corr = np.correlate(cur, 1.0 - cur, mode='full').astype(np.float64)
    cur_max = cur_corr.max()
    stuck = 0
    max_stuck = 3000
    while time.time() < deadline:
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Targeted: find i (zero) where cur[i+lag_max]=1, j (one) where cur[j+lag_max]=0
        max_idx = np.argmax(cur_corr)
        max_lag = lag_arr[max_idx]
        # Secondary lags too
        # Shuffle candidates for diversity
        sz = min(len(zeros), 80)
        so = min(len(ones), 80)
        z_sample = zeros[rng.permutation(len(zeros))[:sz]]
        o_sample = ones[rng.permutation(len(ones))[:so]]
        i_cands = [i for i in z_sample if 0 <= i + max_lag < n and cur[i + max_lag] == 1]
        j_cands = [j for j in o_sample if 0 <= j + max_lag < n and cur[j + max_lag] == 0]
        if i_cands and j_cands:
            i = i_cands[rng.randint(len(i_cands))]
            j = j_cands[rng.randint(len(j_cands))]
        else:
            i = rng.choice(zeros)
            j = rng.choice(ones)
        # Compute delta vectorized
        jp = j + lag_arr
        ip = i + lag_arr
        vj = (jp >= 0) & (jp < n)
        vi = (ip >= 0) & (ip < n)
        delta = np.zeros(full_lags)
        delta[vj] += cur[jp[vj]]
        delta[vi] -= cur[ip[vi]]
        new_vals = cur_corr + delta
        new_max = new_vals.max()
        dend = new_max - cur_max
        if dend < 0 or (T > T_min and rng.rand() < np.exp(-dend / T)):
            cur[i] = 1.0
            cur[j] = 0.0
            cur_corr = new_vals
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = cur.copy()
                best_curve = cur_corr.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            stuck += 1
        # If stuck too long, perturb
        if stuck > max_stuck:
            for _ in range(rng.randint(2, 8)):
                on2 = np.where(cur == 1)[0]
                zo2 = np.where(cur == 0)[0]
                if len(on2) > 0 and len(zo2) > 0:
                    cur[rng.choice(on2)] = 0.0
                    cur[rng.choice(zo2)] = 1.0
            cur_corr = np.correlate(cur, 1.0 - cur, mode='full').astype(np.float64)
            cur_max = cur_corr.max()
            stuck = 0
        if cur_max > best_max + 5:
            # Reset to best
            cur = best_h.copy()
            cur_corr = best_curve.copy()
            cur_max = best_max
            stuck = 0
        T = max(T * cooling, T_min)
    return best_h, float(best_max * dx)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    full_lags = 2 * n - 1
    lag_arr = np.arange(full_lags) - (n - 1)
    idx_n = np.arange(n)
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full')
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return (m + (sm + np.log(np.exp(sh - sm).sum())) / alpha) * dx
    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full')
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        w = np.exp(sh - sm)
        w /= w.sum()
        grad = np.zeros(n)
        # Skip negligible weights
        for k_idx in range(full_lags):
            wk = w[k_idx]
            if wk < 1e-6:
                continue
            lag = lag_arr[k_idx]
            gk = np.ones(n)
            ip = idx_n + lag
            im = idx_n - lag
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            gk[vp] -= x[ip[vp]]
            gk[vm] -= x[im[vm]]
            grad += wk * gk
        # Also add subgradient of m (concentrated on argmax)
        kstar = np.argmax(g)
        lag_s = lag_arr[kstar]
        gs = np.ones(n)
        ips = idx_n + lag_s
        ims = idx_n - lag_s
        vps = (ips >= 0) & (ips < n)
        vms = (ims >= 0) & (ims < n)
        gs[vps] -= x[ips[vps]]
        gs[vms] -= x[ims[vms]]
        grad += gs
        return grad * dx
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    safety = budget_s - 12
    dl = t0 + safety
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.zeros(n)
        h0[::2] = 1.0
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: Subgradient continuous (20%)
    p1 = t0 + 0.20 * safety
    if time.time() < p1:
        h1, c1 = subgradient_optimize(best_h.copy(), n, p1, max_iter=15000)
        if c1 < best_c:
            best_c, best_h = c1, h1
    # Phase 2: Binarize + targeted SA (45%)
    p2 = t0 + 0.65 * safety
    if time.time() < p2:
        h_bin = binarize(best_h, n)
        hb, cb = targeted_sa(h_bin, n, p2, seed=seed)
        if cb < best_c:
            best_c, best_h = cb, hb
    # Phase 3: SLSQP polish continuous (20%)
    p3 = t0 + 0.85 * safety
    if time.time() < p3:
        h3 = slsqp_optimize(best_h.copy(), n, alpha=20000, maxiter=2000, deadline=p3)
        c3 = compute_c5(h3, n)
        if c3 < best_c:
            best_c, best_h = c3, h3
    # Phase 4: Re-binarize + SA (remaining)
    if time.time() < dl - 3:
        h_bin2 = binarize(best_h, n)
        hb2, cb2 = targeted_sa(h_bin2, n, dl - 3, seed=seed + 100)
        if cb2 < best_c:
            best_c, best_h = cb2, hb2
    # Final SLSQP if time
    if time.time() < dl - 2:
        h4 = slsqp_optimize(best_h.copy(), n, alpha=50000, maxiter=1000, deadline=dl - 2)
        c4 = compute_c5(h4, n)
        if c4 < best_c:
            best_c, best_h = c4, h4
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
