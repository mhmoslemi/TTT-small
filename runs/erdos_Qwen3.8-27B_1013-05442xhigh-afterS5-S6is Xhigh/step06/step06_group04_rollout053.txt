def two_swap_delta(h, i1, i2, j1, j2, n):

    changing = [i1, i2, j1, j2]
    new_vals = {i1: 1, i2: 1, j1: 0, j2: 0}
    
    # Find affected lags
    affected = set()
    for p in changing:
        for m in range(max(0, n-1-p), min(2*n-2, n-1+p)+1):
            affected.add(m)

    for p in changing:
        for m in range(max(0, n-1-p), min(2*n-2, n-1+p)+1):
            # phi(j,m) = p means j = p - m + (n-1)
            j = p - m + (n-1)
            if 0 <= j < n and j not in new_vals:
                affected.add(m)
    
    delta = np.zeros(2*n - 1)
    for m in affected:
        d = 0.0
        # Category 1: contributions from changing positions as h[j]
        for p in changing:
            phi_p = p + m - (n-1)
            if 0 <= phi_p < n:

                old_phiv = h[p]
                new_phiv = new_vals[p]
                d += h[j] * (old_phiv - new_phiv)
        delta[m] = d
    return delta
