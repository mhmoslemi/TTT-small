def fast_greedy(h0, n, deadline, max_sweeps=100):
    """Greedy local search - only accept improving swaps."""
    rng = np.random.RandomState(42)
    h = h0.copy()
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    corr = np.correlate(h, 1.0-h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        # Random order to avoid bias
        order_i = zeros[rng.permutation(len(zeros))]
        order_j = ones[rng.permutation(len(ones))]
        
        for ii in order_i:
            if time.time() > deadline:
                break
            for jj in order_j:
                d = make_delta(h, n, ii, jj, m_range)
                nc = (corr + d).max() * dx
                if nc < cur_c - 1e-15:
                    h[ii] = 1.0
                    h[jj] = 0.0
                    corr = corr + d
                    cur_c = nc
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c
