    best_c = compute_c5(h, n)

    if time.time() < deadline - 50:
        h1 = lbfgsb_opt(h, n, alpha=3000, maxiter=2000)
        c1 = compute_c5(h1, n)
        if c1 < best_c:
            best_c, best_h = c1, h1

    if time.time() < deadline - 35:
        h2 = lbfgsb_opt(best_h, n, alpha=30000, maxiter=3000)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c, best_h = c2, h2

    if time.time() < deadline - 25:
        h3 = lbfgsb_opt(best_h, n, alpha=300000, maxiter=3000)
        c3 = compute_c5(h3, n)
        if c3 < best_c:
            best_c, best_h = c3, h3

    if time.time() < deadline - 15:
        h4, c4 = worst_lag_refine(best_h, n, deadline - 10)
        if c4 < best_c:
            best_c, best_h = c4, h4

    if time.time() < deadline - 8:
        h5, c5 = active_refine(best_h, n, deadline - 5)
        if c5 < best_c:
            best_c, best_h = c5, h5

    if time.time() < deadline - 3:
        h6, c6 = worst_lag_refine(best_h, n, deadline - 1)
        if c6 < best_c:
            best_c, best_h = c6, h4 if c4 < c6 else h6

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        alloc = min(180, (deadline - t0) * 0.25)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc)
        best_h, best_c = h_opt, c_opt

    n_candidates = [200, 250, 350, 400, 500, 600]
    for n_try in n_candidates:
        if time.time() > deadline - 60:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 30
        if remaining < 50:
            break
        alloc = min(100, remaining * 0.3)
        if alloc < 40:
            break
        if best_h is not None:
            x_b = np.linspace(0, 2, len(best_h))
            x_n = np.linspace(0, 2, n_try)
            h_start = np.interp(x_n, x_b, best_h)
        elif h_init is not None:
            x_i = np.linspace(0, 2, n_init)
            x_n = np.linspace(0, 2, n_try)
            h_start = np.interp(x_n, x_i, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_opt, c_opt = optimize_at_n(project_feasible(h_start, n_try), n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt

    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        n_restarts = max(1, int((deadline - time.time() - 40) / 70))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = 0.015 * (1.0 + 0.4 * r)
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            alloc = min(60, (deadline - time.time() - 25) * 0.5)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt

    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        if time.time() < deadline - 12:
            h_f = lbfgsb_opt(best_h, n, alpha=500000, mu=1e8, maxiter=2000)
            c_f = compute_c5(h_f, n)
            if c_f < best_c:
                best_c, best_h = c_f, h_f
        if time.time() < deadline - 5:
            h_r, c_r = worst_lag_refine(best_h, n, deadline - 2)
            if c_r < best_c:
                best_c, best_h = c_r, h_r
        if time.time() < deadline - 2:
            h_a, c_a = active_refine(best_h, n, deadline - 1)
            if c_a < best_c:
                best_c, best_h = c_a, h_a

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
