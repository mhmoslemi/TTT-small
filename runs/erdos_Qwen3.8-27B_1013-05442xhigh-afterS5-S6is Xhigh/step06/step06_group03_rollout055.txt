        nt = int(mask.sum())
        if nt < 2:
            break
        grad_t = np.zeros(n)
        grad_t[mask] = grad[mask] - grad[mask].mean()
        
        h_new = _project_feasible(h_curr - lr * grad_t, n)
        c_new = _compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h_curr = h_new
        elif c_new < _compute_c5(h_curr, n) + 1e-15:
            h_curr = h_new
    
    return best_h, best_c

def _local_search(h, n, deadline, rng, max_iters):
    h = h.copy()
    best_c = _compute_c5(h, n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c = _compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        m = abs(lag)
        
        # Gradient of critical lag
        if lag > 0:
            grad = np.zeros(n)
            grad[:n - m] = 1.0 - h[m:]
            grad[m:] -= h[:n - m]
        else:
            grad = np.zeros(n)
            grad[m:] = 1.0 - h[:n - m]
            grad[:n - m] -= h[m:]
        
        # Try greedy swap: move mass from high-grad to low-grad
        # Among positions where the move is feasible
        sorted_idx = np.argsort(grad)
        
        # Try several candidate swaps
        tried = 0
        for trial in range(8):
            if tried >= 4:
                break
            if trial < 4:
                i = int(sorted_idx[rng.randint(0, n // 3)])
                j = int(sorted_idx[n - 1 - rng.randint(0, n // 3)])
            else:
                i = rng.randint(n)
                j = rng.randint(n)
            
            if i == j or grad[i] >= grad[j]:
                # Ensure we're moving from high to low grad
                if trial < 4:
                    continue
                gi, gj = grad[i], grad[j]
                if gi >= gj:
                    i, j = j, i
                    gi, gj = gj, gi
                    if gi >= gj:
                        continue
            
            if h[j] <= 1e-8 or h[i] >= 1.0 - 1e-8:
                continue
            
            delta = min(0.01, h[j] * 0.2, (1.0 - h[i]) * 0.2)
            if delta < 1e-9:
                continue
            
            h[i] += delta
            h[j] -= delta
            tried += 1
            c_new = _compute_c5(h, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
            else:
                h[i] -= delta
                h[j] += delta
    
    return h, best_c

def _slsqp(h, n, alpha, maxiter):
    dx = 2.0 / n
    
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        return float(m + (sm + np.log(np.sum(np.exp(shifted - sm)))) / alpha)
    
    def _obj_jac(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        exp_s = np.exp(shifted - sm)
        w = exp_s / exp_s.sum()
        
        grad = np.zeros(n)
        for k in range(len(c)):
            if w[k] < 1e-10:
                continue
            lag = k - (n - 1)
            if lag == 0:
                grad += w[k] * (1.0 - 2.0 * x)
            elif lag > 0:
                grad[:n - lag] += w[k] * (1.0 - x[lag:])
                grad[lag:] -= w[k] * x[:n - lag]
            else:
                m2 = -lag
                grad[m2:] += w[k] * (1.0 - x[:n - m2])
                grad[:n - m2] -= w[k] * x[m2:]
        return grad * dx  # Wait, c already has dx in it, so dc/dx already includes dx
    
    # Actually, c = dx * corr, so dc/dx = dx * d(corr)/dx
    # My gradient formula computes d(corr)/dx, so I need to multiply by dx
    # But wait, in _obj I use c directly (which has dx), and the gradient of the
    # smooth max of c with respect to x is: sum w_k * dc_k/dx = sum w_k * dx * d(corr_k)/dx
    # My loop computes sum w_k * d(corr_k)/dx, so I need to multiply by dx
    
    def _obj_jac2(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        exp_s = np.exp(shifted - sm)
        w = exp_s / exp_s.sum()
        
        grad = np.zeros(n)
        for k in range(len(c)):
            if w[k] < 1e-10:
                continue
            lag = k - (n - 1)
            if lag == 0:
                grad += w[k] * (1.0 - 2.0 * x)
            elif lag > 0:
                grad[:n - lag] += w[k] * (1.0 - x[lag:])
                grad[lag:] -= w[k] * x[:n - lag]
            else:
                m2 = -lag
                grad[m2:] += w[k] * (1.0 - x[:n - m2])
                grad[:n - m2] -= w[k] * x[m2:]
        return grad * dx
    
    def _cons_fun(x):
        return np.sum(x) - n / 2.0
    
    def _cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    
    try:
        res = minimize(_obj, h, jac=_obj_jac2, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _project_feasible(res.x, n)
    except Exception:
        return h.copy()
