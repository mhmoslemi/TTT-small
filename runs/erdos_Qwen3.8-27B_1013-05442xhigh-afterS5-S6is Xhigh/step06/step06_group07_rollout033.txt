def greedy_targeted(h_bin, n, deadline, seed=42):
    """Greedy that focuses on reducing the maximum overlap"""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n-1), n)
    h = h_bin.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_max = corr.max() * dx
    best_h = h.copy()
    best_max = cur_max
    fails = 0
    
    while time.time() < deadline:
        if fails > 5000:
            # Perturb and restart
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            for _ in range(2 + rng.randint(3)):
                ii = zeros[rng.randint(len(zeros))]
                jj = ones[rng.randint(len(ones))]
                h[ii], h[jj] = 1.0, 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_max = corr.max() * dx
            fails = 0
            continue
        
        # Find the max correlation value
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        # Find all positions achieving near-max
        near_max = corr >= max_val - 0.5
        max_positions = np.where(near_max)[0]
        
        improved = False
        for pos in max_positions:
            k = pos - (n - 1)  # actual shift
            if k == 0:
                continue
            # Find i where h[i]=1 and h[i+k]=0 (contributing to this shift)
            if k > 0:
                valid = np.arange(n - k)
            elif k < 0:
                valid = np.arange(-k, n)
            else:
                continue
            
            # Get contributing positions
            if k > 0:
                contrib_mask = (h[valid] == 1) & (h[valid + k] == 0)
            else:
                contrib_mask = (h[valid] == 1) & (h[valid + k] == 0)
            
            contrib_idx = valid[contrib_mask]
            
            if len(contrib_idx) == 0:
                continue
            
            # Try moving one of these
            for ci in contrib_idx[:5]:
                # We want to set h[ci] = 0, so need to set some h[jj] = 1
                # Prefer jj such that it doesn't create new max
                zeros = np.where(h == 0)[0]
                if len(zeros) == 0:
                    break
                # Try a few random zeros
                for _ in range(8):
                    jj = zeros[rng.randint(len(zeros))]
                    if jj == ci:
                        continue
                    delta = swap_delta(h, n, jj, ci, m_range)
                    new_corr = corr + delta
                    new_max = new_corr.max() * dx
                    if new_max < cur_max - 1e-15:
                        h[jj] = 1.0
                        h[ci] = 0.0
                        corr = new_corr
                        cur_max = new_max
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if improved:
            fails = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            fails += 1
    
    return best_h, best_max
