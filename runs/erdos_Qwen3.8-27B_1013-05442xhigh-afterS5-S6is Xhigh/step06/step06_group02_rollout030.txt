def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible_fast(h, n):
    """Project onto {0 <= h <= 1, sum(h) = n/2} using bisection."""
    target = n / 2.0
    # First clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # If sum is already target, done
    if abs(h.sum() - target) < 1e-15:
        return h
    # Bisection on water level t: sum(clamp(h_i - t, 0, 1)) = target
    lo, hi = -1.0, 1.0  # t range
    for _ in range(100):
        t = (lo + hi) / 2
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = (lo + hi) / 2
    return np.clip(h - t, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    """Compute gradient of smooth max of correlations w.r.t. h."""
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    m = c.max()

    mx = sh.max()
    if mx > 500:

        res = minimize(_slsqp_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible_fast(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def focused_pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        converged = False
        while time.time() < deadline and not converged:
            c_vec = compute_c5_vec(h, n)
            c_max = c_vec.max()
            top_k = np.argsort(c_vec)[-10:]
            active_lags = set()
            for m in top_k:
                if c_vec[m] > c_max - best_c * n * 0.01:
                    k = m - (n - 1)
                    active_lags.add(k)
                    active_lags.add(-k)
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                candidates = set()
                for k in active_lags:
                    if k == 0:
                        continue
                    j1 = i + k
                    j2 = i - k
                    if 0 <= j1 < n:
                        candidates.add(j1)
                    if 0 <= j2 < n:
                        candidates.add(j2)
                for j in sorted(candidates):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                converged = True
    return h, best_c

def full_pair_search(h, n, deadline, deltas, radius=80):
    best_c = compute_c5(h, n)
    for d in deltas:
        converged = False
        while time.time() < deadline and not converged:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
            if not improved:
                converged = True
    return h, best_c

def gradient_pair_search(h, n, deadline, step_size=0.003):
    best_c = compute_c5(h, n)
    for alpha in [100, 500, 2000, 10000, 50000]:
        conv = False
        while time.time() < deadline and not conv:
            improved = False
            g = grad_smooth_max(h, n, alpha)
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(n):
                    if i == j:
                        continue
                    diff = g[i] - g[j]
                    if abs(diff) < 1e-12:
                        continue
                    move = step_size * diff / (abs(diff) + 1e-10)
                    move = step_size if diff > 0 else -step_size
                    if h[i] + move >= 0.0 and h[i] + move <= 1.0 and \
                       h[j] - move >= 0.0 and h[j] - move <= 1.0:
                        h[i] += move
                        h[j] -= move
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= move
                            h[j] += move
            if not improved:
                conv = True
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n_init = 250

    best_h = project_feasible_fast(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS-B with increasing alpha
    t1 = min(t0 + 200, deadline - 400)
    if time.time() < t1:
        best_h, best_c = lbfgs_phase(best_h, n_init, t1, max_iters=250)

    # Phase 2: FISTA with Nesterov momentum
    if time.time() < deadline - 300:
        t2 = min(time.time() + 120, deadline - 300)
        h2, c2 = fista_phase(best_h, n_init, t2, alpha_start=100.0, alpha_end=200000.0)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: SLSQP with high alpha
    if time.time() < deadline - 200:
        t3 = min(time.time() + 100, deadline - 200)
        for a in [200000, 1000000]:
            if time.time() > t3:
                break
            h3, c3 = slsqp_phase(best_h, n_init, min(time.time() + 50, t3), alpha=a, maxiter=1500)
            if c3 < best_c:
                best_c = c3
                best_h = h3

    # Phase 4: Gradient-guided pair search
    if time.time() < deadline - 150:
        t4 = min(time.time() + 80, deadline - 150)
        best_h, best_c = gradient_pair_search(best_h, n_init, t4, step_size=0.002)

    # Phase 5: Focused pair search on active lags
    if time.time() < deadline - 80:
        t5 = min(time.time() + 60, deadline - 80)
        best_h, best_c = focused_pair_search(best_h, n_init, t5,
                                             [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])

    # Phase 6: Full pair search with small steps
    if time.time() < deadline - 30:
        t6 = min(time.time() + 40, deadline - 30)
        best_h, best_c = full_pair_search(best_h, n_init, t6,
                                          [0.001, 0.0005, 0.0002, 0.0001, 0.00005], radius=50)

    # Phase 7: Try other n values
    if time.time() < deadline - 60:
        for n_try in [200, 250, 400]:
            if time.time() > deadline - 40:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible_fast(h_interp, n_try)
            alloc = min(50, deadline - time.time() - 20)
            if alloc < 20:
                break
            h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.5, max_iters=150)
            if time.time() < deadline - 25:
                h_t, c_t = fista_phase(h_t, n_try, time.time() + min(20, deadline - time.time() - 15))
            if c_t < best_c:
                best_c = c_t
                best_h = h_t

    # Phase 8: Final focused pair polish
    if time.time() < deadline - 3:
        best_h, best_c = focused_pair_search(best_h, len(best_h), deadline - 2,
                                             [0.0005, 0.0002, 0.0001, 0.00005])

    best_h = project_feasible_fast(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
