import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    f = m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    log_w = alpha * c - (alpha * m + sm)
    w = np.exp(log_w)
    w /= w.sum()
    grad = np.zeros(n)
    nlm1 = n - 1
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        shift = k - nlm1
        ws = w[k] * dx
        js = max(0, -shift)
        je = min(n, n - shift)
        if js < je:
            grad[js:je] += ws * (1.0 - h[js + shift:je + shift])
        js2 = max(0, shift)
        je2 = min(n, n + shift)
        if js2 < je2:
            grad[js2:je2] -= ws * h[js2 - shift:je2 - shift]
    return f, grad

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    def obj_fn(x):
        f, _ = lse_obj_grad(x, n, alpha)
        return f
    def obj_grad_fn(x):
        _, g = lse_obj_grad(x, n, alpha)
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_fn, h, jac=obj_grad_fn, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def build_g(h, n):
    g = np.zeros((n, 2 * n - 1))
    nlm1 = n - 1
    for p in range(n):
        ps = max(0, nlm1 - p)
        pe = min(2 * n - 2, nlm1 - p + n - 1)
        if ps <= pe:
            shifted = p + np.arange(ps, pe + 1) - nlm1
            g[p, ps:pe + 1] += (1.0 - h[shifted])
        ps2 = max(0, p)
        pe2 = min(2 * n - 2, p + nlm1)
        if ps2 <= pe2:
            r_vals = p + nlm1 - np.arange(ps2, pe2 + 1)
            g[p, ps2:pe2 + 1] -= h[r_vals]
    return g

def do_swap_update(g, corr, i, j, n):
    p_idx = np.arange(n)
    mi = (p_idx != i) & (p_idx + i < 2 * n - 1)
    g[p_idx[mi], p_idx[mi] + i] -= 2.0
    mj = (p_idx != j) & (p_idx + j < 2 * n - 1)
    g[p_idx[mj], p_idx[mj] + j] += 2.0
    corr = corr + g[i] - g[j]
    return g, corr

def binary_optimize(h_start, n, deadline, top_k=5, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    target = n // 2
    idx = np.argsort(h_start)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    g = build_g(h, n)
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    stall = 0
    kick_count = 0

    while time.time() < deadline:
        top_idx = np.argsort(corr)[-top_k:]
        g_act = g[:, top_idx]
        nz = min(len(zeros), 40)
        no = min(len(ones), 40)
        z_sub = zeros[:nz]
        o_sub = ones[:no]
        M = g_act[z_sub]
        D = M[:, None, :] - g_act[o_sub][None, :, :]
        new_maxes = D.max(axis=2)
        flat = new_maxes.argmin()
        ii = flat // len(o_sub)
        jj = flat % len(o_sub)
        best_val = new_maxes[ii, jj]
        i = z_sub[ii]
        j = o_sub[jj]
        new_c = best_val * dx

        if new_c < cur_c - 1e-14:
            g, corr = do_swap_update(g, corr, i, j, n)
            h[i], h[j] = 1.0, 0.0
            zi = np.where(zeros == i)
            oj = np.where(ones == j)
            if len(zi) > 0 and len(oj) > 0:
                zeros = np.delete(zeros, zi[0])
                ones = np.delete(ones, oj[0])
                zeros = np.append(zeros, j)
                ones = np.append(ones, i)
            cur_c = corr.max() * dx
            stall = 0
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = h.copy()
        else:
            stall += 1
            if stall > 40 or (kick_count < 30 and stall > 15):

                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                g = build_g(h, n)
                cur_c = corr.max() * dx
                ones = np.where(h == 1)[0].copy()
                zeros = np.where(h == 0)[0].copy()
                stall = 0

    return best_h, best_c

def thue_morse_init(n, rng):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    num_ones = int(t.sum())
    if num_ones > target:
        ones = np.where(t == 1)[0]
        rng.shuffle(ones)
        t[ones[:num_ones - target]] = 0
    elif num_ones < target:
        zeros = np.where(t == 0)[0]
        rng.shuffle(zeros)
        t[zeros[:target - num_ones]] = 1
    return t.astype(float)
