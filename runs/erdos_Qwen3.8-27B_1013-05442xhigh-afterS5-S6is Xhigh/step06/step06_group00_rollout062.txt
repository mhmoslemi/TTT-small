                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick binary local search
    p1 = t0 + 0.05 * total
    hb, cb = greed_binary(h0, n, p1, max_sweeps=200)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 2: Main SLSQP with analytical gradient
    p2 = t0 + 0.50 * total
    alphas = [100, 500, 2000, 10000, 50000, 300000, 1000000, 5000000, 30000000]
    h = best_h.copy()
    for alpha in alphas:
        if time.time() > p2 - 5:
            break
        h = slsqp_opt(h, n, alpha, 4000, 1e-17)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 3: Multiple restarts with perturbations
    p3 = t0 + 0.85 * total
    for restart in range(50):
        if time.time() > p3 - 3:
            break
        noise_scale = 0.003 + 0.02 * (restart / 50.0)
        h = project_feasible(best_h + noise_scale * rng.randn(n), n)
        h = slsqp_opt(h, n, 2000000, 2000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()
        # Also try from binary base with noise
        if restart % 3 == 0:
            hb = to_binary(best_h, n)
            h = project_feasible(hb + 0.02 * rng.randn(n), n)
            h = slsqp_opt(h, n, 500000, 1500, 1e-16)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Phase 4: Try higher resolution
    p4 = dl - 30
    if time.time() < p4:
        for nh in [500, 600]:
            if time.time() > p4 - 20:
                break
            xo = np.linspace(0, 2, n)
            xn = np.linspace(0, 2, nh)
            hh = project_feasible(np.interp(xn, xo, best_h), nh)
            for alpha in [10000, 200000, 5000000]:
                if time.time() > p4 - 15:
                    break
                hh = slsqp_opt(hh, nh, alpha, 2000, 1e-16)
                ch = compute_c5(hh, nh)
                if ch < best_c:
                    best_c, best_h, n = ch, hh.copy(), nh

    # Phase 5: Final polish
    if time.time() < dl - 15:
        h = best_h.copy()
        for alpha in [50000000, 200000000, 1000000000]:
            if time.time() > dl - 8:
                break
            h = slsqp_opt(h, n, alpha, 3000, 1e-18)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
