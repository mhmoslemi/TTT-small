import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    """Project h onto {x: 0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    # Bisection on threshold
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(h - t, 0.0, 1.0)
    # Fix any tiny numerical issues
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference among non-boundary elements
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_grad(h, n, alpha):
    """Compute smooth objective and gradient."""
    R = np.correlate(h, h, mode='full')  # length 2n-1
    y = alpha * (n / 2.0 - R)
    y_max = y.max()
    obj = (y_max + np.log(np.sum(np.exp(y - y_max)))) / alpha
    w = np.exp(y - y_max)
    w /= w.sum()
    # Gradient
    grad = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if k == 0:
            wt = 2.0 * wk
        else:
            wt = 2.0 * (wk + w[n - 1 - k])  # Should be 2*wk by symmetry, but be safe
        if wt < 1e-15:
            continue
        if k == 0:
            grad -= wt * h
        else:
            grad[:n - k] -= wt * h[k:]
            grad[k:] -= wt * h[:n - k]
    return obj, grad

def pgd_optimize(h, n, deadline, alpha_schedule, step_scale=1.0):
    """Projected gradient descent with backtracking line search."""
    best_h = h.copy()
    best_obj = compute_c5(h, n)
    current_obj = compute_c5(h, n)
    
    for alpha in alpha_schedule:
        if time.time() > deadline:
            break
        # For this alpha, do several gradient steps
        step = step_scale / (alpha * n)
        for it in range(500):
            if time.time() > deadline:
                break
            obj_s, grad = obj_and_grad(h, n, alpha)
            # Backtracking line search
            t = step
            for _ in range(20):

                obj_new = compute_c5(h_new, n)
                if obj_new < current_obj:
                    h = h_new
                    current_obj = obj_new
                    if current_obj < best_obj:
                        best_obj = current_obj
                        best_h = h.copy()
                    break
                t *= 0.5
            # If no improvement found, try smaller step or break
            if t < 1e-10:
                break
    return best_h, best_obj
