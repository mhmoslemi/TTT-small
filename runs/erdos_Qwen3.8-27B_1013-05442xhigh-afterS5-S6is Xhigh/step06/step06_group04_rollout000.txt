def project_feasible(x, n):
    x = np.array(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(80):
        diff = target - x.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - x
            s = room.sum()
            if s < 1e-15:
                break
            x = x + (diff / s) * room
        else:
            s = x.sum()
            if s < 1e-15:
                break
            x = x + (diff / s) * x
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        if diff > 0.0:
            order = np.argsort(1.0 - x)[::-1]
        else:
            order = np.argsort(x)
        for idx in order:
            if diff > 0.0:
                take = min(1.0 - x[idx], diff)
                x[idx] += take
                diff -= take
            else:
                take = min(x[idx], -diff)
                x[idx] -= take
                diff += take
            if abs(diff) < 1e-15:
                break
    return np.clip(x, 0.0, 1.0)

    if dmax <= 0: return None, -1.0

import time
import numpy as np
from scipy.optimize import minimize, linprog

_G_N = 1
_G_A = 1.0


def project_feasible(x, n):
    x = np.asarray(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(80):
        diff = target - x.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - x
            s = room.sum()
            if s < 1e-15:
                break
            x = x + (diff / s) * room
        else:
            s = x.sum()
            if s < 1e-15:
                break
            x = x + (diff / s) * x
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        if diff > 0.0:
            order = np.argsort(1.0 - x)[::-1]
        else:
            order = np.argsort(x)
        for idx in order:
            if diff > 0.0:
                take = min(1.0 - x[idx], diff)
                x[idx] += take
                diff -= take
            else:
                take = min(x[idx], -diff)
                x[idx] -= take
                diff += take
            if abs(diff) < 1e-15:
                break
    return np.clip(x, 0.0, 1.0)


def evaluate(h, n):
    dx = 2.0 / n
    vals = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(vals)), vals


def grad_lag(h, k, n):
    dx = 2.0 / n
    center = n - 1
    g = np.zeros(n)
    if k == center:
        g[:] = (1.0 - 2.0 * h) * dx
    elif k > center:
        l = k - center
        if l < n:
            g[:n - l] += (1.0 - h[l:]) * dx
            g[l:] -= h[:n - l] * dx
    else:
        l = center - k
        if l < n:
            g[l:] += (1.0 - h[:n - l]) * dx
            g[:n - l] -= h[l:] * dx
    return g


def lp_direction(h, C, n, dmax):
    m = C.size
    cmax = float(C.max())
    if dmax <= 0.0:
        return None, -1.0
    if m <= 700:
        active = np.arange(m)
    else:
        active = np.argsort(C)[::-1]
        if active.size > 500:
            active = active[:500]
    q = active.size
    A = np.empty((q, n + 1))
    b = np.empty(q)
    for r, kk in enumerate(active):
        k = int(kk)
        A[r, :n] = grad_lag(h, k, n)
        A[r, n] = 1.0
        b[r] = max(0.0, cmax - C[k])
    c = np.zeros(n + 1)
    c[-1] = -1.0
    lo = np.maximum(-dmax, -h)
    hi = np.minimum(dmax, 1.0 - h)
    bounds = list(zip(lo, hi))
    bounds.append((0.0, 1.0))
    Aeq = np.ones(n + 1)
    Aeq[-1] = 0.0
    try:
        res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=0.0,
                      bounds=bounds, method='highs')
    except Exception:
        try:
            res = linprog(c, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=0.0,
                          bounds=bounds, method='simplex')
        except Exception:
            return None, -1.0
    if not res.success:
        return None, -1.0
    eps = float(res.x[-1])
    if eps <= 1e-11:
        return None, eps
    delta = res.x[:-1].copy()
    s = delta.sum()
    if abs(s) > 1e-10:
        delta = delta - s / n
    delta = np.clip(delta, lo, hi)
    return delta, eps


def line_search_lp(h, delta, c_old, n):
    lam = 1.0
    for _ in range(40):
        h2 = h + lam * delta
        h2 = project_feasible(h2, n)
        c2, C2 = evaluate(h2, n)
        if c2 < c_old - 1e-15:
            return h2, C2, c2
        lam *= 0.5

    return None


def lp_polish(h, C, c, n, deadline, limit_s=None):

def sum_eq(x):
    return x.sum() - 0.5 * _G_N


def sum_eq_jac(x):
    return np.ones(_G_N)


def weighted_grad(h, w, n):
    dx = 2.0 / n
    center = n - 1
    g = np.zeros(n)
    wc = w[center]
    if wc > 1e-18:
        g += wc * (1.0 - 2.0 * h) * dx
    for l in range(1, n):
        wp = w[center + l]
        if wp > 1e-18:
            g[:n - l] += wp * dx * (1.0 - h[l:])
            g[l:] -= wp * dx * h[:n - l]
        wm = w[center - l]
        if wm > 1e-18:
            g[l:] += wm * dx * (1.0 - h[:n - l])
            g[:n - l] -= wm * dx * h[l:]
    return g


def smooth_obj(x):
    n = _G_N
    dx = 2.0 / n
    a = _G_A
    C = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(C.max())
    y = a * (C - m)
    s = np.exp(y)
    return m + np.log(s.sum()) / a


def smooth_jac(x):
    n = _G_N
    dx = 2.0 / n
    a = _G_A
    C = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(C.max())
    y = a * (C - m)
    s = np.exp(y)
    w = s / s.sum()
    return weighted_grad(x, w, n)


def slsqp_polish(h, C, c, n, end_time, start_alpha=1.0):
    global _G_N, _G_A
    _G_N = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]
    alphas = [20.0, 100.0, 500.0, 2500.0, 10000.0, 50000.0, 200000.0]
    maxit = [30, 40, 60, 90, 120, 160, 200]
    for a, mi in zip(alphas, maxit):
        if float(a) < float(start_alpha):
            continue
        rem = end_time - time.time()
        if rem < 20.0:
            break
        _G_A = float(a)
        mi_use = int(max(20, min(mi, 10.0 + 0.5 * rem)))
        opts = {'maxiter': mi_use, 'ftol': 1e-15, 'disp': 0}
        try:
            res = minimize(smooth_obj, h, method='SLSQP', jac=smooth_jac,

                c2, C2 = evaluate(h2, n)
                if c2 < c - 1e-14 or c2 <= c + 1e-9:
                    h, C, c = h2, C2, c2
        except Exception:
            pass
    return h, C, c

        newx = np.linspace(0.0, 2.0, newn, endpoint=False)

        n = newn
    _G_N = n
    h = project_feasible(h0, n)
    c, C = evaluate(h, n)
    best_h = h.copy()
    best_c = c
    best_C = C.copy()

    lp1_limit = min(90.0, 0.09 * base)
    if deadline - time.time() > max(15.0, 0.5 * lp1_limit):
        h1, C1, c1 = lp_polish(h, C, c, n, deadline, limit_s=lp1_limit)
        if c1 < best_c:
            best_c, best_h, best_C = c1, h1.copy(), C1.copy()

    sls_main_end = min(deadline - max(50.0, 0.08 * base), t0 + 0.56 * base)
    if time.time() + 15.0 < sls_main_end:
        h1, C1, c1 = slsqp_polish(best_h, best_C, best_c, n, sls_main_end)
        if c1 < best_c:
            best_c, best_h, best_C = c1, h1.copy(), C1.copy()

    lp2_limit = min(45.0, 0.04 * base)
    if time.time() < deadline - max(35.0, 0.04 * base):
        h1, C1, c1 = lp_polish(best_h, best_C, best_c, n, deadline,
                               limit_s=lp2_limit)
        if c1 < best_c:
            best_c, best_h, best_C = c1, h1.copy(), C1.copy()

    perturb_amp = (0.012, 0.025, 0.05, 0.09, 0.15, 0.25)
    pert_end = min(deadline - 70.0, t0 + 0.87 * base)
    for amp in perturb_amp:
        if time.time() > pert_end:
            break
        sub_end = min(deadline - 15.0, time.time() + 85.0)
        if sub_end - time.time() < 25.0:
            break
        start = project_feasible(best_h + amp * rng.normal(size=n), n)
        c0, C0 = evaluate(start, n)
        if c0 < best_c + 0.05:
            h1, C1, c1 = lp_polish(start, C0, c0, n, sub_end, limit_s=8.0)
            h1, C1, c1 = slsqp_polish(h1, C1, c1, n, sub_end, start_alpha=500.0)
            if c1 < best_c:
                best_c, best_h, best_C = c1, h1.copy(), C1.copy()

    if time.time() < deadline - 120.0:
        for _ in range(2):
            sub_end = min(deadline - 15.0, time.time() + 75.0)
            if sub_end - time.time() < 25.0:
                break
            start = fourier_start(n, rng)
            c0, C0 = evaluate(start, n)
            if c0 < best_c + 0.12:
                h1, C1, c1 = lp_polish(start, C0, c0, n, sub_end, limit_s=5.0)
                h1, C1, c1 = slsqp_polish(h1, C1, c1, n, sub_end, start_alpha=100.0)
                if c1 < best_c:
                    best_c, best_h, best_C = c1, h1.copy(), C1.copy()

    if time.time() < deadline - 10.0:
        rem = deadline - time.time()
        lp3_limit = min(25.0, 0.45 * rem)
        if rem > 30.0:
            h1, C1, c1 = lp_polish(best_h, best_C, best_c, n, deadline,
                                   limit_s=lp3_limit)
            if c1 < best_c:
                best_c, best_h, best_C = c1, h1.copy(), C1.copy()
        if time.time() < deadline - 6.0:
            h1, C1, c1 = slsqp_polish(best_h, best_C, best_c, n, deadline,
                                      start_alpha=2500.0)
            if c1 < best_c:
                best_c, best_h, best_C = c1, h1.copy(), C1.copy()

    best_h = project_feasible(best_h, n)
    best_c, _ = evaluate(best_h, n)
    return best_h, float(best_c), int(n)
