from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def soft_max(c, alpha):
    m = c.max()
    return m + np.log(np.exp(alpha * (c - m)).sum()) / alpha

def soft_max_weights(c, alpha):
    m = c.max()
    e = np.exp(alpha * (c - m))
    return e / e.sum()

def c5_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    val = soft_max(c, alpha)
    w = soft_max_weights(c, alpha)
    # Gradient computation
    W = w  # length 2n-1
    A = 1.0 - h  # length n
    B = h  # length n

import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 100.0
_g_penalty = 1000.0
_g_n = 250

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def c5_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    val = m + np.log(np.exp(alpha * (c - m)).sum()) / alpha
    e = np.exp(alpha * (c - m))
    w = e / e.sum()
    rows = np.arange(n)[:, None]
    cols = np.arange(n)[None, :]
    M1 = w[rows + cols]
    M2 = w[cols + (n - 1) - rows]
    first = M1 @ (1.0 - h)[::-1]
    second = M2 @ h
    grad = dx * (first - second)
    return val, grad

def project_box_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(x):
    val, _ = c5_and_grad(x, _g_alpha)
    return val + _g_penalty * (x.sum() - _g_n / 2.0) ** 2

def grad_smooth(x):
    val, g = c5_and_grad(x, _g_alpha)
    g = g + 2.0 * _g_penalty * (x.sum() - _g_n / 2.0)
    return g

def quick_binary_swap(h, n, deadline, seed, max_iters=8000):
    rng = np.random.RandomState(seed)
    h = (h >= 0.5).astype(float)
    if h.sum() > n / 2:
        idx = np.argsort(h)
        h[idx[:int(h.sum() - n / 2)]] = 0.0
    elif h.sum() < n / 2:
        idx = np.argsort(h)
        h[idx[-int(n / 2 - h.sum())]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h[j] = 0.0
        h[i] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        new_c = new_corr.max() * dx
        if new_c <= cur_c + 0.008 * dx:
            corr = new_corr
            cur_c = new_c
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            if stall > 2000:
                break
        else:
            h[j] = 1.0
            h[i] = 0.0
            stall += 1
    return best_h, best_c
