                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def sum_constraint(x):
    return x.sum() - len(x) / 2.0

def smooth_obj(x, dx, alpha):
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        res = minimize(lambda x: smooth_obj(x, dx, alpha), h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    return delta

def binary_sa(h_bin, n, deadline, seed=42, t_start=0.02, t_end=0.00001, patience=30000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    cool_rate = (t_end / t_start) ** (1.0 / 500000.0)
    T = t_start
    use_double = False
    while time.time() < deadline:
        if use_double and T > 0.005:
            ia = zeros_idx[rng.randint(len(zeros_idx))]
            ib = zeros_idx[rng.randint(len(zeros_idx))]
            ja = ones_idx[rng.randint(len(ones_idx))]
            jb = ones_idx[rng.randint(len(ones_idx))]
            if ia == ib or ja == jb or ia == ja or ia == jb or ib == ja or ib == jb:
                use_double = False
            else:
                h[ia] = 1.0; h[ib] = 1.0; h[ja] = 0.0; h[jb] = 0.0
                new_corr = np.correlate(h, 1 - h, mode='full').astype(float)
                nc = new_corr.max() * dx
                dc = nc - cur_c
                if dc < 0 or (T > 1e-12 and rng.random() < np.exp(-dc / T)):
                    corr = new_corr
                    cur_c = nc
                    zeros_idx = np.where(h == 0)[0]
                    ones_idx = np.where(h == 1)[0]
                    if nc < best_c:
                        best_c = nc; best_h = h.copy(); no_improve = 0
                    else:
                        no_improve += 1
                else:
                    h[ia] = 0.0; h[ib] = 0.0; h[ja] = 1.0; h[jb] = 1.0
                    no_improve += 1
                use_double = False
                T *= cool_rate
                continue
        i = zeros_idx[rng.randint(len(zeros_idx))]
        j = ones_idx[rng.randint(len(ones_idx))]
        delta = swap_delta(h, n, i, j, m_range)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        dc = nc - cur_c
        if dc < 0 or (T > 1e-12 and rng.random() < np.exp(-dc / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            if nc < best_c:
                best_c = nc; best_h = h.copy(); no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= cool_rate
        use_double = (rng.random() < 0.15)
        if no_improve > patience:
            nk = 3 + rng.randint(6)
            for _ in range(nk):
                ii = zeros_idx[rng.randint(len(zeros_idx))]
                jj = ones_idx[rng.randint(len(ones_idx))]
                h[ii] = 1.0; h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            no_improve = 0
            T = min(t_start * 0.5, T * 4.0)
    return best_h, best_c

def fast_local_search(h, n, deadline, max_sweeps=500):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        for i in zeros:
            for j in ones:
                d = swap_delta(h, n, i, j, m_range)
                nm = (corr + d).max()
                if nm < cur_max - 1e-14:
                    h[i] = 1.0; h[j] = 0.0
                    corr = corr + d
                    cur_max = nm
                    improved = True
                    break
            if improved:
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                break
        if not improved:
            break
    return best_h, best_max * dx

def best_improve_local(h, n, deadline):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    for _ in range(200):
        if time.time() > deadline:
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        bi, bj = -1, -1
        bnm = cur_max
        for i in zeros:
            for j in ones:
                d = swap_delta(h, n, i, j, m_range)
                nm = (corr + d).max()
                if nm < bnm:
                    bnm = nm; bi = i; bj = j
        if bi < 0:
            break
        d = swap_delta(h, n, bi, bj, m_range)
        h[bi] = 1.0; h[bj] = 0.0
        corr = corr + d
        cur_max = bnm
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
    return best_h, best_max * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    p1 = t0 + 0.06 * (budget_s - 5)
    h = best_h.copy()
    for alpha in [2000, 10000, 50000, 200000]:
        if time.time() > p1:
            break
        h = slsqp_smooth(h, n, alpha, 800, 1e-15)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c; best_h = h.copy()
    p2 = t0 + 0.52 * (budget_s - 5)
    h_bin = round_to_binary(best_h, n)
    hc = compute_c5(h_bin, n)
    if hc < best_c:
        best_c = hc; best_h = h_bin.copy()
    for attempt in range(4):
        if time.time() > p2:
            break
        if attempt == 0:
            start_h = h_bin; s = seed; ts = 0.025
        elif attempt == 1:
            start_h = round_to_binary(best_h + 0.15 * rng.randn(n), n); s = seed + 100; ts = 0.02
        elif attempt == 2:
            start_h = round_to_binary(best_h + 0.3 * rng.randn(n), n); s = seed + 200; ts = 0.015
        else:
            start_h = round_to_binary(best_h + 0.5 * rng.randn(n), n); s = seed + 300; ts = 0.01
        phase_end = p2 if attempt < 3 else p2
        if attempt == 3:
            phase_end = p2
        hb, cb = binary_sa(start_h, n, phase_end, seed=s, t_start=ts, t_end=0.00001, patience=25000)
        if cb < best_c:
            best_c = cb; best_h = hb.copy()
    p3 = t0 + 0.72 * (budget_s - 5)
    hg, cg = fast_local_search(best_h, n, p3, max_sweeps=300)
    if cg < best_c:
        best_c = cg; best_h = hg.copy()
    p4 = t0 + 0.85 * (budget_s - 5)
    hb, cb = binary_sa(best_h, n, p4, seed=seed + 500, t_start=0.005, t_end=0.000005, patience=20000)
    if cb < best_c:
        best_c = cb; best_h = hb.copy()
    p5 = dl - 20
    hg, cg = fast_local_search(best_h, n, p5, max_sweeps=200)
    if cg < best_c:
        best_c = cg; best_h = hg.copy()
    if time.time() < dl - 10:
        hb, cb = best_improve_local(best_h, n, dl - 5)
        if cb < best_c:
            best_c = cb; best_h = hb.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
