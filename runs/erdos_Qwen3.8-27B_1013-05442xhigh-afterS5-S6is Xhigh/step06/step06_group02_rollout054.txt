            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_all_corrs(h, n)
    c_max = c.max()
    sh = alpha * (c - c_max)
    sh_max = sh.max()
    val = c_max + (sh_max + np.log(np.sum(np.exp(sh - sh_max)))) / alpha
    w = np.exp(sh - sh_max)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-11:
            continue
        wm = w[m] * dx
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return val, grad

def _penalty_obj(x, n, alpha, mu):
    v, _ = grad_smooth_max(x, n, alpha)
    return v + mu * (x.sum() - n / 2.0) ** 2

def _penalty_grad(x, n, alpha, mu):
    _, g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def topk_subgrad(h, n, topk=5):
    corrs = compute_all_corrs(h, n)
    topk_idx = np.argsort(corrs)[-topk:]
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in topk_idx:
        k = m - (n - 1)
        if k >= 0:
            nv = n - k
            grad[0:nv] += (1.0 - h[k:n])
            grad[k:n] += (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += (1.0 - h[0:nv])
            grad[0:nv] += (-h[kk:n])
    grad /= float(topk)
    return corrs.max() * dx, grad * dx

def subgrad_optimize(h, n, deadline, topk=5, lr0=0.005):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr0
    stale = 0
    while time.time() < deadline - 1:
        c_val, g = topk_subgrad(h, n, topk)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        improved = False
        for frac in [1.0, 0.5, 0.25]:
            h_new = project_feasible(h - frac * lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                stale = 0
                improved = True
                break
        if not improved:
            lr *= 0.95
            stale += 1
            if stale > 500:
                topk = min(topk + 3, 20)
                lr = min(lr * 3.0, lr0)
                stale = 0
            if lr < 1e-13:
                break
    return best_h, best_c

def momentum_pg(h, n, deadline, alpha_init=1000.0, lr_init=0.003, mom=0.9):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    alpha = alpha_init
    lr = lr_init
    stale = 0
    for _ in range(50000):
        if time.time() > deadline - 1:
            break
        _, g = grad_smooth_max(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        g = g / g_norm
        improved = False
        step_lr = lr
        for _ls in range(20):
            v_new = mom * v - step_lr * g
            h_new = project_feasible(h + v_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                v = v_new
                h = h_new
                best_c = c_new
                best_h = h.copy()
                stale = 0
                improved = True
                break
            step_lr *= 0.5
        if not improved:
            v = mom * v
            lr *= 0.65
            alpha = min(alpha * 1.4, 20000000.0)
            stale += 1
            if stale > 80:
                alpha = min(alpha * 2.5, 50000000.0)
                stale = 0
            if lr < 1e-13:
                break
    return best_h, best_c

def pair_search(h, n, deadline, deltas, radius=80):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline - 1:
            improved = False
            for i in range(n):
                if time.time() > deadline - 1:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def lbfgs_optimize(h, n, deadline, alphas, max_iters=150):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        mu = 1e6 / max(alpha, 1)
        try:
            res = minimize(_penalty_obj, h, jac=_penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxfun': max_iters * 5,
                                   'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n = 200
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS-B with increasing smoothness
    t1 = min(t0 + 200, deadline - 350)
    best_h, best_c = lbfgs_optimize(best_h, n, t1,
                                     alphas=[500, 2000, 10000, 50000, 200000, 1000000],
                                     max_iters=120)

    # Phase 2: Fast subgradient polish
    if time.time() < deadline - 200:
        t2 = min(time.time() + 120, deadline - 200)
        h2, c2 = subgrad_optimize(best_h, n, t2, topk=5, lr0=0.004)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Momentum PG with high alpha
    if time.time() < deadline - 130:
        t3 = min(time.time() + 70, deadline - 130)
        h3, c3 = momentum_pg(best_h, n, t3, alpha_init=10000, lr_init=0.002)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: L-BFGS-B very high alpha
    if time.time() < deadline - 80:
        t4 = min(time.time() + 50, deadline - 80)
        best_h, best_c = lbfgs_optimize(best_h, n, t4,
                                         alphas=[500000, 2000000, 5000000],
                                         max_iters=100)

    # Phase 5: Pair search coarse
    if time.time() < deadline - 50:
        t5 = min(time.time() + 40, deadline - 50)
        best_h, best_c = pair_search(best_h, n, t5,
                                      [0.01, 0.005, 0.002], radius=100)

    # Phase 6: Subgradient with more lags
    if time.time() < deadline - 25:
        t6 = min(time.time() + 20, deadline - 15)
        h6, c6 = subgrad_optimize(best_h, n, t6, topk=10, lr0=0.002)
        if c6 < best_c:
            best_c = c6
            best_h = h6

    # Phase 7: Try alternate n
    if time.time() < deadline - 30:
        for n_try in [200, 250, 400]:
            if time.time() > deadline - 20:
                break
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(25, deadline - time.time() - 10)
            if alloc < 10:
                break
            h_t, c_t = lbfgs_optimize(h_interp, n_try, time.time() + alloc,
                                       alphas=[5000, 50000, 500000], max_iters=80)
            if c_t < best_c:
                best_c = c_t
                best_h = h_t
                n = n_try

    # Phase 8: Final fine pair search
    nf = len(best_h)
    if time.time() < deadline - 2:
        best_h, best_c = pair_search(best_h, nf, deadline - 1,
                                      [0.001, 0.0005, 0.0002, 0.0001], radius=50)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
