def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full').astype(float) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, n, alpha):
    c = compute_c5_all(h, n)
    shifted = alpha * (c - c.max())
    m = shifted.max()
    weights = np.exp(shifted - m)
    weights /= weights.sum()
    grad = np.zeros(n)

        idx = np.arange(i_start, i_end + 1)
        grad[idx + mlag] += w * (1.0 - h[idx])
        grad[idx] -= w * h[idx + mlag]
    return grad
