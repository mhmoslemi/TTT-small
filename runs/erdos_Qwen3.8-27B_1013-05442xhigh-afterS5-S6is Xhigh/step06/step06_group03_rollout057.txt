import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag_idx):
    """Compute gradient of lag lag_idx (0-indexed in the full correlation array) w.r.t. h."""
    m = lag_idx - (n - 1)  # actual lag
    grad = np.zeros(n)
    abs_m = abs(m)
    if abs_m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if m > 0:
        # c = sum_{i=0}^{n-m-1} h[i](1-h[i+m])
        # dc/dh[i] for i < n-m: (1-h[i+m])
        # dc/dh[i+m] for i < n-m: -h[i]
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    else:
        # c = sum_{i=0}^{n+m-1} h[i+m](1-h[i])
        # dc/dh[i+m] for i < n+m: (1-h[i])
        # dc/dh[i] for i < n+m: -h[i+m]
        grad[-m:] += (1.0 - h[:n+m])
        grad[:n+m] -= h[-m:]
    return grad

def solve_lp_active_set(h_curr, n, active_lags, trust):
    """Solve LP with outer approximation constraints for active lags."""
    dx = 2.0 / n
    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.norm(h_var - h_curr, np.inf) <= trust]
    for k in active_lags:
        grad = lag_gradient(h_curr, n, k)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    # Also include a few non-active lags for safety
    all_lags = list(range(2 * n - 1))
    inactive = [k for k in all_lags if k not in active_lags]
    inactive.sort(key=lambda k: -c[k])
    for k in inactive[:20]:
        grad = lag_gradient(h_curr, n, k)
        cons.append(c[k] + dx * (grad @ (h_var - h_curr)) <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), cons)
    solver = None
    for s in [cp.GLOP, cp.CBC, cp.SCS]:
        try:
            prob.solve(solver=s)
            solver = s
            break
        except Exception:
            continue
    if prob.status not in ['optimal', 'optimal_inaccurate'] or h_var.value is None:
        return h_curr.copy()
    return np.clip(h_var.value, 0, 1)

def smooth_max_obj(h, n, alpha, top_m=10):
    """Smooth approximation of max of top_m lags."""
    c = compute_all_lags(h, n)
    top = np.sort(c)[-top_m:]
    m = top.max()
    shifted = alpha * (top - m)
    mx = shifted.max()
    return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha

        top = np.sort(c)[-top_m:]
        m = top.max()
        shifted = alpha * (top - m)
        mx = shifted.max()

def annealing_search(h, n, deadline, rng, T0=0.05, cooling=0.97, n_steps=500):
    """Simulated annealing to escape local minima."""
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    T = T0
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Random perturbation
        scale = 0.02 * (T / T0 + 0.1)
        h_pert = h_curr + rng.normal(0, scale, n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)
        
        if c_pert < best_c:
            best_c = c_pert
            best_h = h_pert.copy()
            h_curr = h_pert
        elif rng.random() < np.exp(-(c_pert - best_c) / T):
            h_curr = h_pert
        
        T *= cooling
    
    return best_h, best_c

        c = compute_all_lags(h_curr, n)
        top_k = max(5, min(25, n // 8))
        active = list(np.argsort(c)[-top_k:])
        h_new = lp_solve(h_curr, n, active, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-10:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-10:
            break
        h_curr = h_new
        trust = max(0.005, trust * 0.93)
    return best_h, best_c

def smooth_obj(x, n, alpha, top_m):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    top = np.sort(c)[-top_m:]
    m = top.max()
    shifted = alpha * (top - m)
    mx = shifted.max()
    if mx > 400:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_obj_grad(x, n, alpha, top_m):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    top_idx = np.argsort(c)[-top_m:]
    m = c[top_idx].max()
    shifted = alpha * (c[top_idx] - m)
    mx = shifted.max()
    exps = np.exp(shifted - mx)
    weights = alpha * exps / np.sum(exps)
    grad = np.zeros(n)
    for j in range(len(top_idx)):
        grad += weights[j] * lag_gradient(x, n, top_idx[j]) * dx
    return grad

def sum_con_fun(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(len(x))

def slsqp_step(h_curr, n, alpha, top_m, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons_def = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h_curr, method='SLSQP',
                      jac=smooth_obj_grad,
                      args=(n, alpha, top_m),
                      bounds=bounds, constraints=cons_def,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h_curr.copy()

def local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            stride = max(1, m // 50)
            if lag > 0:
                for i in range(0, n - m, stride):
                    if time.time() > deadline:
                        break
                    if h[i] > h[i + m]:
                        delta = min(d, (h[i] - h[i+m]) * 0.4)
                        if delta < 1e-7:
                            continue
                        h[i] -= delta; h[i+m] += delta
                        cn = compute_c5(h, n)
                        if cn < best_c - 1e-12:
                            best_c = cn; improved = True
                        else:
                            h[i] += delta; h[i+m] -= delta
                    elif h[i+m] > h[i]:
                        delta = min(d, (h[i+m] - h[i]) * 0.4)
                        if delta < 1e-7:
                            continue
                        h[i] += delta; h[i+m] -= delta
                        cn = compute_c5(h, n)
                        if cn < best_c - 1e-12:
                            best_c = cn; improved = True
                        else:
                            h[i] -= delta; h[i+m] += delta
            else:
                mm = -lag
                for i in range(0, n - mm, stride):
                    if time.time() > deadline:
                        break
                    if h[i+mm] > h[i]:
                        delta = min(d, (h[i+mm] - h[i]) * 0.4)
                        if delta < 1e-7:
                            continue
                        h[i+mm] -= delta; h[i] += delta
                        cn = compute_c5(h, n)
                        if cn < best_c - 1e-12:
                            best_c = cn; improved = True
                        else:
                            h[i+mm] += delta; h[i] -= delta
                    elif h[i] > h[i+mm]:
                        delta = min(d, (h[i] - h[i+mm]) * 0.4)
                        if delta < 1e-7:
                            continue
                        h[i+mm] += delta; h[i] -= delta
                        cn = compute_c5(h, n)
                        if cn < best_c - 1e-12:
                            best_c = cn; improved = True
                        else:
                            h[i+mm] -= delta; h[i] += delta
    return h, best_c

def anneal(h, n, deadline, rng, T0=0.025, cooling=0.985):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    T = T0
    while time.time() < deadline:
        scale = 0.012 * (1.0 + 4.0 * T / T0)
        h_p = project_feasible(h_curr + rng.normal(0, scale, n), n)
        c_p = compute_c5(h_p, n)
        if c_p < best_c:
            best_c = c_p; best_h = h_p.copy(); h_curr = h_p
        elif rng.random() < np.exp(-(c_p - best_c) / max(T, 1e-8)):
            h_curr = h_p
        T *= cooling
        if T < 0.0005:
            T = 0.0005
    return best_h, best_c

def optimize_n(h_start, n, deadline):
    h = project_feasible(h_start, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if time.time() < deadline - 20:
        h2, c2 = active_set_loop(h, n, min(deadline - 10, time.time() + 60), max_iters=150)
        if c2 < best_c:
            best_c = c2; best_h = h2
    if time.time() < deadline - 10:
        for alpha, tm in [(10000, 6), (50000, 10), (200000, 15)]:
            if time.time() > deadline - 8:
                break
            h3 = slsqp_step(best_h, n, alpha, tm, maxiter=2000)
            c3 = compute_c5(h3, n)
            if c3 < best_c:
                best_c = c3; best_h = h3
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    best_h = project_feasible(h_init, n_init).copy()
    best_c = compute_c5(best_h, n_init)
    # Phase 1: strong active set LP on initial n
    if time.time() < deadline - 80:
        end1 = time.time() + min(180, (deadline - t0) * 0.28)
        best_h, best_c = active_set_loop(best_h, n_init, end1, max_iters=400)
    # Phase 2: SLSQP passes
    if time.time() < deadline - 50:
        for alpha, tm in [(5000, 5), (20000, 8), (100000, 12), (500000, 18)]:
            if time.time() > deadline - 50:
                break
            h_new = slsqp_step(best_h, n_init, alpha, tm, maxiter=3000)
            c_new = compute_c5(h_new, n_init)
            if c_new < best_c:
                best_c = c_new; best_h = h_new
    # Phase 3: try other grid sizes
    for n_try in [200, 250, 350, 400, 450, 500]:
        if n_try == n_init or time.time() > deadline - 80:
            continue
        rem = deadline - time.time() - 60
        if rem < 30:
            break
        x_b = np.linspace(0, 2, len(best_h))
        x_n = np.linspace(0, 2, n_try)
        h_s = project_feasible(np.interp(x_n, x_b, best_h), n_try)
        alloc = min(50, rem * 0.4)
        h_opt, c_opt = optimize_n(h_s, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt; best_h = h_opt
    # Phase 4: local search + annealing
    if time.time() < deadline - 30:
        n = len(best_h)
        best_h, best_c = local_search(best_h, n, min(deadline - 20, time.time() + 50))
    if time.time() < deadline - 20:
        n = len(best_h)
        end_an = min(deadline - 8, time.time() + 55)
        best_h, best_c = anneal(best_h, n, end_an, rng, T0=0.02, cooling=0.987)
        if time.time() < deadline - 5:
            best_h, best_c = local_search(best_h, n, deadline - 3)
    # Phase 5: final SLSQP polish
    if time.time() < deadline - 4:
        n = len(best_h)
        for alpha in [500000, 2000000]:
            if time.time() > deadline - 3:
                break
            h_new = slsqp_step(best_h, n, alpha, 20, maxiter=3000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new; best_h = h_new
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if time.time() < deadline - 1:
        best_h, best_c = local_search(best_h, len(best_h), deadline)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
