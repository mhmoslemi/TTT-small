def project_box_sum(x, target):
    x = np.asarray(x, dtype=float).copy()
    n=x.size
    if n==0: return x
    if target <= 0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(60):
       tau=(lo+hi)*0.5
       y=np.clip(x-tau,0,1)
       s=y.sum()
       if s > target:
          lo=tau
       else:
          hi=tau
    y=np.clip(x-(lo+hi)*0.5,0,1)
    diff=target-y.sum()
    if abs(diff) > 1e-15:
       mask=(y>1e-9)&(y<1-1e-9)
       if mask.any():
          y[mask] += diff / mask.count? Need mask.sum() as float
          y=np.clip(y,0,1)
       else:
          # distribute to all with room
          if diff>0:
             mask=(y<1-1e-12); if any, y[mask]+=diff/...
          else: mask=(y>1e-12)
    # final sum correction
    s=y.sum()
    if abs(target-s)>1e-12:
       # adjust largest margin variable
       if target>s:
          idx=np.argmin(y); amount=min(1-y[idx], target-s); y[idx]+=amount

def weighted_gradient(x, n, idx, w):
    dx = 2.0 / n
    g = np.zeros(n)
    n1 = n - 1
    for wt, q in zip(w, idx):
        wt = float(wt)
        if wt <= 1e-14: continue
        lag = int(q) - n1
        if lag == 0:
            g += wt * (1.0 - 2.0 * x)
        elif lag > 0:
            m = lag
            g[m:] += wt * (1.0 - x[:-m])
            g[:-m] -= wt * x[m:]
        else:
            m = -lag
            g[:n-m] += wt * (1.0 - x[m:])
            g[m:] -= wt * x[:n-m]
    return dx * g

import numpy as np
import time
from scipy.optimize import minimize


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.0, posinf=1.0, neginf=0.0)
    n = x.size
    if n == 0:
        return x
    if target <= 0.0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(70):
        tau = 0.5 * (lo + hi)
        y = np.clip(x - tau, 0.0, 1.0)
        s = float(np.sum(y))
        if s > target:
            lo = tau
        else:
            hi = tau
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(np.sum(y))
    if abs(diff) > 1e-14:
        y = y.copy()
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if np.any(mask):
            y[mask] += diff / float(np.sum(mask))
        else:
            if diff > 0.0:
                mask = y < 1.0 - 1e-12
            else:
                mask = y > 1e-12
            if np.any(mask):
                y[mask] += diff / float(np.sum(mask))
    y = np.clip(y, 0.0, 1.0)
    s = float(np.sum(y))
    if abs(target - s) > 1e-12:
        diff = target - s
        if diff > 0.0:
            ii = int(np.argmin(y))
            y[ii] = min(1.0, y[ii] + diff)
        else:
            ii = int(np.argmax(y))
            y[ii] = max(0.0, y[ii] + diff)
    return y


def all_c(h, n):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def compute_c5(h, n):
    return float(np.max(all_c(h, n)))


def soft_value_weights(c, n, alpha, idx):
    vals = c[idx]
    m = float(np.max(vals))
    s = alpha * (vals - m)
    e = np.exp(s)
    ss = float(np.sum(e))
    val = m + float(np.log(ss)) / alpha
    return val, e / ss


def weighted_gradient(x, n, idx, w):
    dx = 2.0 / n
    g = np.zeros(n)
    n1 = n - 1
    for wt, q in zip(w, idx):
        wt = float(wt)
        if wt <= 1e-15:
            continue
        lag = int(q) - n1
        if lag == 0:
            g += wt * (1.0 - 2.0 * x)
        elif lag > 0:
            m = lag
            g[m:] += wt * (1.0 - x[:-m])
            g[:-m] -= wt * x[m:]
        else:
            m = -lag
            g[:n - m] += wt * (1.0 - x[m:])
            g[m:] -= wt * x[:n - m]
    return dx * g


def _soft_obj(x, n, alpha, idx):
    y = np.clip(x, 0.0, 1.0)
    c = all_c(y, n)
    val, _ = soft_value_weights(c, n, alpha, idx)
    return val


def _soft_jac(x, n, alpha, idx):
    y = np.clip(x, 0.0, 1.0)
    c = all_c(y, n)
    _, w = soft_value_weights(c, n, alpha, idx)
    return weighted_gradient(y, n, idx, w)


def _sum_obj(x, n):
    return float(np.sum(x)) - n / 2.0


def _sum_jac(x, n):
    return np.ones(n)


def slsqp_polish(h, n, deadline, alpha, top_m, maxiter=80):
    h = project_box_sum(h, n / 2.0)
    c0 = all_c(h, n)
    top_m = max(1, min(top_m, 2 * n - 1))
    idx = np.argsort(c0)[-top_m:]
    try:
        res = minimize(
            _soft_obj,
            h,
            args=(n, alpha, idx),
            method="SLSQP",
            jac=_soft_jac,
            bounds=[(0.0, 1.0)] * n,
            constraints=[{"type": "eq", "fun": _sum_obj, "jac": _sum_jac, "args": (n,)}],
            options={"maxiter": int(maxiter), "ftol": 1e-12},
        )
    except Exception:
        return h, compute_c5(h, n)
    x = res.x
    if x is None:
        return h, compute_c5(h, n)
    x = project_box_sum(x, n / 2.0)
    cv = compute_c5(x, n)
    return x, cv


def projected_gradient(h, n, deadline, alpha=200.0, top_m=64, iters=500):
    h = project_box_sum(h, n / 2.0)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_h = h.copy()
    top_m = max(1, min(top_m, 2 * n - 1))
    lr = 0.02
    for _ in range(int(iters)):
        if time.time() > deadline - 2:
            break
        c = all_c(cur_h, n)
        idx = np.argsort(c)[-top_m:]
        val, w = soft_value_weights(c, n, alpha, idx)
        g = weighted_gradient(cur_h, n, idx, w)
        g = g - np.mean(g)
        scale = float(np.max(np.abs(g)))
        if scale < 1e-13:
            break
        step = lr / max(scale, 1e-13)
        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.125):
            trial = project_box_sum(cur_h - step * mult * g, n / 2.0)
            tc = all_c(trial, n)
            tv, _ = soft_value_weights(tc, n, alpha, idx)
            tmax = float(np.max(tc))
            if tv <= val - 1e-14 or tmax < best_c - 1e-12:
                if tmax < best_c + 1e-7:
                    cur_h = trial
                    if tmax < best_c - 1e-14:
                        best_c = tmax
                        best_h = trial.copy()
                accepted = True
                break
        if accepted:
            lr = min(0.25, lr * 1.08)
        else:
            lr = max(1e-5, lr * 0.5)
    return best_h, best_c


def local_polish(h, n, deadline, rng, trials=5000):
    h = project_box_sum(h, n / 2.0)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_h = h.copy()
    cur_c = best_c
    T = 2e-6
    n1 = n - 1
    for d in (0.004, 0.002, 0.001, 0.0005, 0.0002):
        if time.time() > deadline - 3:
            break
        improved = False
        for _ in range(int(trials)):
            if time.time() > deadline - 3:
                break
            c = all_c(cur_h, n)
            idx = np.argsort(c)[-8:]
            q = int(idx[int(rng.randint(len(idx)))])
            lag = q - n1
            if lag == 0:
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                if cur_h[i] > cur_h[j] + 1e-12:
                    dst, src = i, j
                elif cur_h[j] > cur_h[i] + 1e-12:
                    dst, src = j, i
                else:
                    continue
                delta = min(d, 0.5 * (cur_h[dst] - cur_h[src]), cur_h[src], 1.0 - cur_h[dst])
            else:
                m = abs(lag)
                if m >= n:
                    continue
                i = int(rng.randrange(n - m))
                if lag > 0:
                    src, dst = i + m, i
                else:
                    src, dst = i, i + m
                if cur_h[src] < 1e-11 or cur_h[dst] > 1.0 - 1e-11:
                    continue
                delta = min(d, 0.5 * cur_h[src], 1.0 - cur_h[dst])
            if delta < 1e-10:
                continue
            cur_h[src] -= delta
            cur_h[dst] += delta
            cc = float(np.max(all_c(cur_h, n)))
            if cc <= cur_c + T:
                cur_c = cc
                if cc < best_c - 1e-14:
                    best_c = cc
                    best_h = cur_h.copy()
                    T = min(T, 4e-7)
                improved = True
                T = max(4e-7, T * 0.997)
            else:
                cur_h[src] += delta
                cur_h[dst] -= delta
        if not improved:
            break
    best_h = project_box_sum(best_h, n / 2.0)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def add_candidate(lst, item_c, item_h, item_n, limit=4):
    if not np.isfinite(item_c):
        return
    while len(lst) >= limit:
        if item_c >= lst[-1][0]:
            return
        lst.pop()
    lst.append((item_c, item_h, item_n))
    for i in range(len(lst) - 1, 0, -1):
        if lst[i][0] < lst[i - 1][0]:
            lst[i], lst[i - 1] = lst[i - 1], lst[i]


def make_fallback_h(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.clip(1.0 - np.abs(x - 1.0), 0.0, 1.0)
    return project_box_sum(h, n / 2.0)
