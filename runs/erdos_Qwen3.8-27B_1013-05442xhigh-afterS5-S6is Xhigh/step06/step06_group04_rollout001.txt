import numpy as np
from scipy.optimize import minimize
import time
import math


def compute_q(h, n):
    dx = 2.0 / n
    return dx * np.correlate(1.0 - h, h, mode='full')


def exact_c5(h, n):
    return float(np.max(compute_q(h, n)))


def sigmoid(z):
    z = np.clip(np.asarray(z, dtype=float), -60.0, 60.0)
    out = np.empty_like(z)
    pos = z >= 0.0
    out[pos] = 1.0 / (1.0 + np.exp(-z[pos]))
    e = np.exp(z[~pos])
    out[~pos] = e / (1.0 + e)
    return out


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = 0.5 * n
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h + mid, 0.0, 1.0).sum() < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if diff > 1e-13:
        order = np.argsort(1.0 - h)[::-1]
        for k in order:
            take = min(diff, 1.0 - h[k])
            if take <= 0.0:
                break
            h[k] += take
            diff -= take
            if abs(diff) < 1e-15:
                break
    elif diff < -1e-13:
        order = np.argsort(h)[::-1]
        for k in order:
            take = min(-diff, h[k])
            if take <= 0.0:
                break
            h[k] -= take
            diff += take
            if abs(diff) < 1e-15:
                break
    return h


def binarize(h, n):
    b = np.zeros(n, dtype=float)
    if n > 0:
        idx = np.argsort(-h)
        b[idx[:n // 2]] = 1.0
    return b


def make_start(n):
    if n % 2:
        n += 1
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.5 * np.sin(np.pi * x)
    return project_feasible(h, n), n


def interpolate_start(h, n):
    old = np.asarray(h, dtype=float).ravel()
    if old.size == n:
        return project_feasible(old, n)
    x0 = np.linspace(0.0, 2.0, old.size, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n, endpoint=False)
    return project_feasible(np.interp(x1, x0, old), n)


def get_h_from_z(z, n):
    target = 0.5 * n
    lo, hi = -100.0, 100.0
    for _ in range(38):
        mid = 0.5 * (lo + hi)
        if sigmoid(z + mid).sum() < target:
            lo = mid
        else:
            hi = mid
    return sigmoid(z + 0.5 * (lo + hi))


def gradient_q(h, p, n):
    c = n - 1
    pr = p[::-1]
    a = np.convolve(np.ones(n, dtype=float), pr, mode='full')[c:c + n]
    b = np.convolve(h, pr, mode='full')[c:c + n]
    d = np.convolve(h, p, mode='full')[c:c + n]
    return (2.0 / n) * (a - b - d)


def loss_grad_z(z, alpha, n, state=None):
    h = get_h_from_z(z, n)
    q = compute_q(h, n)
    m = float(np.max(q))
    if not np.isfinite(m):
        return 1e100, np.zeros_like(z)
    if alpha <= 1.0:
        p = np.full(q.shape, 1.0 / q.size, dtype=float)
        F = m
    else:
        s = alpha * (q - m)
        sm = float(np.max(s))
        e = np.exp(s - sm)
        Z = float(np.sum(e))
        if Z <= 0.0 or not np.isfinite(Z):
            p = np.zeros_like(q)
            p[int(np.argmax(q))] = 1.0
            F = m
        else:
            p = e / Z
            F = m + (sm + math.log(Z)) / alpha
    G = gradient_q(h, p, n)
    srit = h * (1.0 - h)
    S = float(np.sum(srit))
    if not np.isfinite(S) or S < 1e-14:
        gz = np.zeros_like(z)
    else:
        mg = float(np.dot(G, srit)) / S
        gz = srit * (G - mg)
    if not np.isfinite(F):
        return 1e100, np.zeros_like(z)
    return float(F), gz


def minimize_callback(xk, *args):
    for a in args:
        try:
            if isinstance(a, (list, tuple)) and len(a) > 0:
                t = float(a[0])
                if time.time() > t:
                    return True
        except Exception:
            pass
    return False


def continuous_optimize(h0, n, limit, alphas, maxiter=200):
    best_h = project_feasible(h0, n)
    best_c = exact_c5(best_h, n)
    hc = np.clip(best_h, 1e-12, 1.0 - 1e-12)
    z = np.clip(np.log(hc / (1.0 - hc)), -40.0, 40.0)
    bounds = [(-40.0, 40.0)] * n
    opts = {'maxiter': int(maxiter), 'maxcor': 30, 'ftol': 1e-14, 'gtol': 1e-9, 'maxls': 30}
    for alpha in alphas:
        if time.time() >= limit:
            break
        state = [float(limit)]
        res = None
        try:
            res = minimize(loss_grad_z, z, args=(float(alpha), n, state),
                           method='L-BFGS-B', jac=True, bounds=bounds,
                           options=opts, callback=minimize_callback)
        except Exception:
            try:
                res = minimize(loss_grad_z, z, args=(float(alpha), n),
                               method='L-BFGS-B', jac=True, bounds=bounds, options=opts)
            except Exception:
                res = None
        if res is not None and getattr(res, 'x', None) is not None:
            rx = res.x
            if rx.size == n and np.all(np.isfinite(rx)):
                z = np.clip(rx, -40.0, 40.0)
                h = project_feasible(get_h_from_z(z, n), n)
                c = exact_c5(h, n)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
    return best_h, best_c


def binary_swap_delta(h, i, j, n, idx):
    c = n - 1
    delta = np.zeros(2 * n - 1, dtype=float)
    delta[(idx - i) + c] += 1.0 - h
    delta[(i - idx) + c] -= h
    delta[(idx - j) + c] += h - 1.0
    delta[(j - idx) + c] += h
    delta[(j - i) + c] += 1.0
    delta[(i - j) + c] += 1.0
    delta[c] = 0.0
    return delta


def binary_local_search(h0, n, limit, seed=42):
    if n % 2:
        return h0, exact_c5(h0, n)
    h = binarize(h0, n)
    dx = 2.0 / n
    raw = np.correlate(1.0 - h, h, mode='full').astype(float)
    cur = float(np.max(raw)) * dx
    best_c = cur
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    idx = np.arange(n)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    stalls = 0
    while time.time() < limit - 2 and stalls < 6:
        best_nc = cur + 1.0
        bi = bj = -1
        bd = None
        nz = zeros.size
        no = ones.size
        for _ in range(120):
            if nz == 0 or no == 0:
                break
            i = int(zeros[rng.randint(nz)])
            j = int(ones[rng.randint(no)])
            d = binary_swap_delta(h, i, j, n, idx)
            nc = float(np.max(raw + d)) * dx
            if nc < best_nc - 1e-13:
                best_nc = nc
                bi, bj, bd = i, j, d
        if bi >= 0:
            h[bi] = 1.0
            h[bj] = 0.0
            raw = raw + bd
            cur = best_nc
            if cur < best_c:
                best_c = cur
                best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            stalls = 0
        else:
            stalls += 1
            T = 0.012
            for _ in range(5000):
                if time.time() > limit - 3:
                    break
                nz = zeros.size
                no = ones.size
                if nz == 0 or no == 0:
                    break
                i = int(zeros[rng.randint(nz)])
                j = int(ones[rng.randint(no)])
                d = binary_swap_delta(h, i, j, n, idx)
                nc = float(np.max(raw + d)) * dx
                accept = False
                if nc < cur:
                    accept = True
                else:
                    r = (nc - cur) / max(T, 1e-12)
                    if r < 700.0 and rng.random() < math.exp(-r):
                        accept = True
                if accept:
                    h[i] = 1.0
                    h[j] = 0.0
                    raw = raw + d
                    cur = nc
                    if cur < best_c:
                        best_c = cur
                        best_h = h.copy()
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                T = max(0.0003, T * 0.999)
            raw = np.correlate(1.0 - h, h, mode='full').astype(float)
            cur = float(np.max(raw)) * dx
            if cur < best_c:
                best_c = cur
                best_h = h.copy()
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    B = max(10.0, float(budget_s))
    hard_deadline = t0 + B - 2.0

    try:
        arr = np.asarray(initial_h_values, dtype=float).ravel()
        if arr.size > 0:
            best_n = int(arr.size)
            best_h = project_feasible(arr, best_n)
        else:
            best_h, best_n = make_start(250)
    except Exception:
        best_h, best_n = make_start(250)

    if best_n > 600:
        best_h = interpolate_start(best_h, 250)
        best_n = 250

    best_c = exact_c5(best_h, best_n)
    rng = np.random.RandomState(seed)

    p1 = t0 + 0.42 * B
    if time.time() < p1:
        h0, c0 = continuous_optimize(best_h, best_n, p1, [1000, 5000, 20000, 100000], 170)
        if c0 < best_c:
            best_c, best_h, best_n = c0, h0.copy(), best_n

    p2 = t0 + 0.63 * B
    if time.time() < p2:
        h0, c0 = continuous_optimize(best_h, best_n, min(p2, time.time() + 90), [10000, 50000], 120)
        if c0 < best_c:
            best_c, best_h, best_n = c0, h0.copy(), best_n
        for _ in range(7):
            if time.time() > p2 - 2:
                break
            if rng.random() < 0.5:
                hp = best_h + 0.02 * rng.randn(best_n)
            else:
                hp = 0.8 * project_feasible(best_h, best_n) + 0.2 * rng.rand(best_n)
            hp = project_feasible(hp, best_n)
            h0, c0 = continuous_optimize(hp, best_n, min(p2 - 1, time.time() + 35),
                                         [5000, 20000, 100000], 80)
            if c0 < best_c:
                best_c, best_h, best_n = c0, h0.copy(), best_n

    p3 = t0 + 0.80 * B
    for n2 in (200, 150, 100, 50):
        if n2 == best_n or time.time() > p3 - 6:
            continue
        h2 = interpolate_start(best_h, n2)
        h0, c0 = continuous_optimize(h2, n2, min(p3 - 2, time.time() + 25), [2000, 10000], 80)
        if c0 < best_c:
            best_c, best_h, best_n = c0, h0.copy(), n2

    p4 = t0 + 0.91 * B
    if best_n % 2 == 0 and time.time() < p4:
        hb, cb = binary_local_search(best_h, best_n, p4, seed + 1)
        if cb < best_c:
            best_c, best_h, best_n = cb, hb.copy(), best_n

    if time.time() < hard_deadline - 4:
        h0, c0 = continuous_optimize(best_h, best_n, hard_deadline - 3,
                                     [50000, 200000, 1000000], 90)
        if c0 < best_c:
            best_c, best_h, best_n = c0, h0.copy(), best_n

    if best_n % 2 == 0 and time.time() < hard_deadline - 4:
        hb, cb = binary_local_search(best_h, best_n, hard_deadline - 1, seed + 99)
        if cb < best_c:
            best_c, best_h, best_n = cb, hb.copy(), best_n

    best_h = project_feasible(best_h, best_n)
    best_c = exact_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
