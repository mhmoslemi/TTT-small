            i_end = min(n, n - d) if d > 0 else n
            if i_start < i_end:
                grad[i_start:i_end] += wi * (1 - h[i_start + d:i_end + d])
            # Second term: -h[i-d] for i-d in [0,n)
            #   i ranges from max(0,d) to min(n-1, n-1+d)
            i_start2 = max(0, d)
            i_end2 = min(n, n + d) if d < 0 else n
            if i_start2 < i_end2:
                grad[i_start2:i_end2] -= wi * h[i_start2 - d:i_end2 - d]
        
        # Projected gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        elif c_new > best_c + 0.01:
            # Step was too big, reduce lr
            lr *= 0.5
            h_new = h - lr * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
        
        h = h_new
        
        # Decay learning rate
        lr = max(lr * 0.999, 0.001)
        
        # Periodic restart with perturbation if stuck
        if it % 5000 == 0 and it > 0:
            if c_new >= best_c - 1e-10:
                h = best_h + 0.01 * rng.randn(n)
                h = project_feasible(h, n)
                lr = max(lr, 0.01)
    
    return best_h, best_c

def slsqp_refine(h, n, deadline, alphas=None):
    """SLSQP refinement with log-sum-exp"""
    dx = 2.0 / n
    if alphas is None:
        alphas = [500, 2000, 10000, 50000, 200000, 500000]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        
        def obj(x):
            g = np.correlate(x, 1 - x, mode='full') * dx
            m = g.max()
            sh = alpha * (g - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=coswipe: 5000, 'ftol': 1e-16})
            if res.x is not None and not np.any(np.isnan(res.x)):
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c
