def softmax_max(g, alpha):
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def slsqp_smooth(h, n, alpha, maxiter=600, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        return softmax_max(g, alpha)
    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        w = np.exp(sh - sm) / np.exp(sh - sm).sum()
        grad = -2.0 * alpha * w
        c = np.correlate(np.ones(n), x, mode='full')
        c2 = np.correlate(x, np.ones(n), mode='full')
        dg = (c - c2) * dx
        return grad * dg
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n

        if c < best_c:
            best_c, best_h = c, hr
    best_h = project_feasible(best_h, n)
    best_c, _ = compute_c5(best_h, n)
    return best_h, best_c, n
