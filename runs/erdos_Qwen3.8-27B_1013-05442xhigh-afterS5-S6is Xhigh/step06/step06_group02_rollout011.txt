    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()

                break
            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full') * (2.0/n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    n_corr = 2 * n - 1

            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

    sh = alpha * (c - m)

            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _obj(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _eq_obj(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _eq_con(x, n):
    return float(x.sum() - n / 2.0)

def _eq_jac(x, n):
    return np.ones(n)

def lbfgs_phase(h, n, deadline, alpha_list=None, max_iters=300):
    if alpha_list is None:
        alpha_list = [50, 200, 1000, 5000, 20000, 100000, 500000]
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_list:
        if time.time() > deadline - 5:
            break
        for mu in [0.01, 0.1]:
            if time.time() > deadline - 5:
                break
            try:
                res = minimize(_obj, h, jac=_grad, args=(n, alpha, mu),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': max_iters, 'ftol': 1e-18,
                                       'gtol': 1e-15, 'maxfun': max_iters * 20,
                                       'maxls': 50, 'maxcor': 30})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    h = h_new
            except Exception:
                pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=100000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_eq_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def binary_push(h, n, deadline):
    best_c = compute_c5(h, n)
    improved_any = True
    while improved_any and time.time() < deadline:
        improved_any = False
        for i in range(n):
            if time.time() > deadline:
                break
            if h[i] > 0.005:
                rest = h.copy()
                rest[i] = 0.0
                deficit = h[i]
                room = 1.0 - rest
                room[i] = 0
                tot = room.sum()
                if tot > 1e-10:
                    rest += (deficit / tot) * room
                    rest = project_feasible(rest, n)
                    c_new = compute_c5(rest, n)
                    if c_new < best_c - 1e-13:
                        best_c = c_new
                        h = rest
                        improved_any = True
            if h[i] < 0.995:
                rest = h.copy()
                rest[i] = 1.0
                surplus = 1.0 - h[i]
                avail = rest.copy()
                avail[i] = 0
                tot = avail.sum()
                if tot > 1e-10:
                    rest -= (surplus / tot) * avail
                    rest = project_feasible(rest, n)
                    c_new = compute_c5(rest, n)
                    if c_new < best_c - 1e-13:
                        best_c = c_new
                        h = rest
                        improved_any = True
    return h, best_c

def pair_search(h, n, deadline, radius=80):
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def multi_restart(h, n, deadline, rng):
    best_c = compute_c5(h, n)
    attempts = 3
    per_attempt = max(20, (deadline - time.time()) / (attempts + 1) - 10)
    for _ in range(attempts):
        if time.time() > deadline:
            break
        noise = rng.normal(0, 0.05, n)
        h_pert = project_feasible(h + noise, n)
        t_end = time.time() + per_attempt
        h_new, c_new = lbfgs_phase(h_pert, n, t_end,
                                   alpha_list=[500, 5000, 50000], max_iters=150)
        if time.time() < deadline - 10:
            h_new, c_new = binary_push(h_new, n, time.time() + 20)
            if time.time() < deadline - 5:
                h_new, c_new = pair_search(h_new, n, time.time() + 10, radius=50)
        if c_new < best_c:
            best_c = c_new
            h = h_new
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 12

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 300
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Extended L-BFGS continuation
    t1 = min(t0 + 280, deadline - 250)
    best_h, best_c = lbfgs_phase(best_h, n, t1, max_iters=400)

    # Phase 2: SLSQP with the equality constraint
    if time.time() < deadline - 180:
        for a in [50000, 200000, 1000000]:
            if time.time() > deadline - 150:
                break
            t2 = min(time.time() + 50, deadline - 130)
            h2, c2 = slsqp_phase(best_h, n, t2, alpha=a, maxiter=1500)
            if c2 < best_c:
                best_c = c2; best_h = h2

    # Phase 3: Binary push to push toward vertices
    if time.time() < deadline - 120:
        t3 = min(time.time() + 80, deadline - 120)
        h3, c3 = binary_push(best_h, n, t3)
        if c3 < best_c:
            best_c = c3; best_h = h3

    # Phase 4: L-BFGS refinement after binary push
    if time.time() < deadline - 80:
        t4 = min(time.time() + 60, deadline - 80)
        h4, c4 = lbfgs_phase(best_h, n, t4,
                            alpha_list=[2000, 20000, 200000], max_iters=200)
        if c4 < best_c:
            best_c = c4; best_h = h4

    # Phase 5: Pair search with fine deltas
    if time.time() < deadline - 40:
        t5 = min(time.time() + 100, deadline - 40)
        best_h, best_c = pair_search(best_h, n, t5, radius=60)

    # Phase 6: Multi-restart to escape local optima
    if time.time() < deadline - 60:
        t6 = deadline - 30
        best_h, best_c = multi_restart(best_h, n, t6, rng)

    # Phase 7: Final refinement
    if time.time() < deadline - 10:
        nf = len(best_h)
        h7, c7 = lbfgs_phase(best_h, nf, deadline - 8,
                            alpha_list=[50000, 500000], max_iters=100)
        if c7 < best_c:
            best_c = c7; best_h = h7
    if time.time() < deadline - 3:
        best_h, best_c = pair_search(best_h, len(best_h), deadline - 2, radius=40)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
