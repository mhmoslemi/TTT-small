Actually, np.correlate(a, v) computes sum_n a[n] * conj(v[n + k - (M-1)]).

Let me just use a different approach. I'll pad h and w and use the correlation.

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj_grad(h, n, alpha):
    """Compute soft-max of correlate and its gradient analytically."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    ms = shifted.max()
    exps = np.exp(shifted - ms)
    w = exps / exps.sum()
    obj = m + ms / alpha
    
    # Gradient computation
    # corr[k] = sum_i h[i] * (1-h[i+m]) where m = k-(n-1)
    # dcorr[k]/dh[j] = 1 - h[j+m]*1[j+m valid] - h[j-m]*1[j-m valid]
    # grad[j] = sum_k w[k] * dcorr[k]/dh[j]
    #         = sum_k w[k] * (1 - h[j+m_k] - h[j-m_k])
    #         = sum(w) - sum_k w[k]*h[j+m_k] - sum_k w[k]*h[j-m_k]
    
    # sum_k w[k] = 1 (weights are normalized)
    # sum_k w[k]*h[j+m_k]: this is like a weighted sum of h at shifted positions
    # = (w * h_crosscorr)[j] where w is treated as a kernel centered at 0
    
    # Let's define: for r from -(n-1) to (n-1), w_r = w[r + (n-1)]
    # term2[j] = sum_r w_r * h[j+r] (h out of range = 0)
    # term3[j] = sum_r w_r * h[j-r] (h out of range = 0)
    
    # Use FFT for efficiency? For n=250, direct computation is fine.
    # term2 = cross-correlation of h with w_kernel
    # where w_kernel[r] = w[r+(n-1)] for r in [-(n-1), n-1]
    
    # Actually, let's just do it the simple way for n<=500
    grad = np.ones(n)  # starts with sum(w) = 1
    js = np.arange(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-15:
            continue
        r = k - (n-1)
        # h[j+r] for valid j
        valid = (js + r >= 0) & (js + r < n)
        grad[valid] -= wk * h[js[valid] + r]
        # h[j-r] for valid j  
        valid2 = (js - r >= 0) & (js - r < n)
        grad[valid2] -= wk * h[js[valid2] - r]
    
    return obj, grad

def pgd_optimize(h0, n, deadline, alphas=None):
    """Projected gradient descent with soft-max"""
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    h = h0.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        # Phase of PGD at this alpha
        # Use decreasing step size
        for it in range(2000):
            if time.time() > deadline:
                break
            obj, grad = softmax_obj_grad(h, n, alpha)
            # Project gradient (remove component in direction of constraint)
            grad_proj = grad - grad.mean()
            # Step size (simplified: use fixed or backtracking)
            step = 1.0 / (alpha * n)  # heuristic
            # Try a few step sizes
            for factor in [1.0, 0.5, 0.25, 0.1, 0.05]:

                h_new = project_feasible(h_new, n)

                    h = h_new
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                    break
            else:
                # No step size worked
                step *= 0.5
                if step < 1e-10:
                    break
        # Refine with smaller steps
        for it in range(500):
            if time.time() > deadline:
                break
            obj, grad = softmax_obj_grad(h, n, alpha)
            grad_proj = grad - grad.mean()
            step = 0.5 / (alpha * n)
            h_new = h - step * grad_proj
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new <= obj + 1e-15:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                # Try smaller step
                step *= 0.1
                h_new = h - step * grad_proj
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                h = h_new
    
    return best_h, best_c

def swap_delta_fast(h, n, i, j):
    """Compute delta in correlation when swapping h[i]=0->1 and h[j]=1->0"""
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    # h[i] changes from 0 to 1: effect on c[k] = sum h[a]*(1-h[a+m_k])
    # For positions where a=i: h[i] was 0, now 1, so c[k] increases by (1-h[i+m_k])
    # For positions where a+m_k=i (a=i-m_k): c[k] decreases by h[i-m_k]... 
    # Wait, let me use the formula: c[k] = sum_a h[a]*(1-h[a+m_k])
    # When h[0->1] at position i:
    #   Term a=i: was 0*(1-h[i+m]) = 0, now 1*(1-h[i+m]) = 1-h[i+m]
    #     -> delta = 1-h[i+m_k] if i+m_k in [0,n)
    #   Term a where a+m_k=i (a=i-m_k): was h[a]*(1-0) = h[a], now h[a]*(1-1) = 0
    #     -> delta = -h[i-m_k] if i-m_k in [0,n)
    # When h[1->0] at position j:
    #   Term a=j: was 1*(1-h[j+m]) = 1-h[j+m], now 0*(1-h[j+m]) = 0
    #     -> delta = -(1-h[j+m_k]) if j+m_k in [0,n)
    #   Term a where a+m_k=j (a=j-m_k): was 0*h[j-m_k]... 
    #     Wait, h[j-m_k] is the value at position j-m_k, which is still the same.
    #     The term is h[a]*(1-h[a+m_k]) where a+m_k=j.
    #     Before: h[j-m_k]*(1-1) = 0
    #     After: h[j-m_k]*(1-0) = h[j-m_k]
    #     -> delta = +h[j-m_k] if j-m_k in [0,n)
    
    js = np.arange(n)
    for k in range(2*n-1):
        mk = k - (n-1)
        d = 0.0
        # h[i]: 0->1
        if 0 <= i+mk < n: d += 1.0 - h[i+mk]
        if 0 <= i-mk < n: d -= h[i-mk]
        # h[j]: 1->0
        if 0 <= j+mk < n: d -= (1.0 - h[j+mk])
        if 0 <= j-mk < n: d += h[j-mk]
        delta[k] = d * dx
    return delta

def binary_sa_fast(h_bin, n, deadline, seed=42, t_start=0.03, t_end=0.00001, patience=50000):
    """Fast binary SA with O(n) swap evaluation"""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()

            i = zeros_idx[rng.randint(len(zeros_idx))]

            new_corr_max = (corr + delta).max() * dx

                h[i] = 1.0

                corr = corr + delta
                cur_c = new_corr_max
                zeros_idx = np.where(h == 0)[0]
                ones_idx = np.where(h == 1)[0]
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        T *= cool_rate
        if no_improve > patience:
            # Restart with perturbation
            nk = 3 + rng.randint(5)
            for _ in range(nk):
                ii = zeros_idx[rng.randint(len(zeros_idx))]
                jj = ones_idx[rng.randint(len(ones_idx))]
                h[ii] = 1.0
                h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            no_improve = 0
            T = min(t_start, T * 5.0)
    return best_h, best_c

def swap_delta_vec(h, n, i, j):
    """Vectorized swap delta computation"""
    kp = np.arange(2*n-1) - (n-1)  # m_k values: -(n-1) to (n-1)
    delta = np.zeros(2*n-1)
    # h[i]: 0->1
    vi = i + kp
    valid_i = (vi >= 0) & (vi < n)
    delta[valid_i] += 1.0 - h[vi[valid_i]]
    vi2 = i - kp
    valid_i2 = (vi2 >= 0) & (vi2 < n)
    delta[valid_i2] -= h[vi2[valid_i2]]
    # h[j]: 1->0
    vj = j + kp
    valid_j = (vj >= 0) & (vj < n)
    delta[valid_j] -= 1.0 - h[vj[valid_j]]
    vj2 = j - kp
    valid_j2 = (vj2 >= 0) & (vj2 < n)
    delta[valid_j2] += h[vj2[valid_j2]]
    return delta

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def softmax_obj_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    ms = shifted.max()

    w = exps / exps.sum()
    obj = m + ms / alpha
    
    grad = np.ones(n)
    js = np.arange(n)
    kp = np.arange(2*n-1) - (n-1)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-14:
            continue
        r = kp[k]
        vi = js + r
        valid = (vi >= 0) & (vi < n)
        grad[valid] -= wk * h[vi[valid]]
        vi2 = js - r
        valid2 = (vi2 >= 0) & (vi2 < n)
        grad[valid2] -= wk * h[vi2[valid2]]
    return obj, grad

def pgd_step(h, n, alpha, max_iters=5000):
    """One phase of projected gradient descent"""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 1.0 / (alpha * 0.1)
    
    for it in range(max_iters):
        obj, grad = softmax_obj_grad(h, n, alpha)
        grad_proj = grad - grad.mean()
        gnorm = np.linalg.norm(grad_proj)
        if gnorm < 1e-12:
            break
        # Line search
        for factor in [1.0, 0.5, 0.25, 0.1]:
            h_new = h - step * factor * grad_proj
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            obj_new, _ = softmax_obj_grad(h_new, n, alpha)
            if obj_new < obj:
                h = h_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                break
        else:
            step *= 0.5
            if step < 1e-12:
                break
    return best_h, best_c

def greedy_improve(h_bin, n, deadline, seed=42, max_sweeps=200):

def topk_obj_grad(h, n, k=5):
    """Objective and gradient using top-k averaging"""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # Find top-k
    top_k = np.argsort(corr)[::-1][:k]
    obj = corr[top_k].mean()  # or max
    # Actually for min-max, we want to minimize the max.
    # Using average of top-k is a smooth approximation.
    
    # Weights: uniform over top-k
    w = np.zeros(len(corr))
    w[top_k] = 1.0 / k
    
    # Gradient
    grad = np.ones(n)
    js = np.arange(n)
    for idx in top_k:
        r = idx - (n-1)
        vi = js + r
        valid = (vi >= 0) & (vi < n)
        grad[valid] -= (1.0/k) * h[vi[valid]]
        vi2 = js - r
        valid2 = (vi2 >= 0) & (vi2 < n)
        grad[valid2] -= (1.0/k) * h[vi2[valid2]]
    return obj, grad

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def softmax_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    ms = sh.max()
    return m + (ms + np.log(np.exp(sh - ms).sum())) / alpha

def slsqp_smooth(h, n, alpha, maxiter, t_end):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        ms = sh.max()
        return m + (ms + np.log(np.exp(sh - ms).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def swap_delta_vec(h, n, i, j, kp):
    delta = np.zeros(2 * n - 1)
    vi = i + kp
    v = (vi >= 0) & (vi < n)
    delta[v] += 1.0 - h[vi[v]]
    vi2 = i - kp
    v2 = (vi2 >= 0) & (vi2 < n)
    delta[v2] -= h[vi2[v2]]
    vj = j + kp
    vj_v = (vj >= 0) & (vj < n)
    delta[vj_v] -= 1.0 - h[vj[vj_v]]
    vj2 = j - kp
    vj2_v = (vj2 >= 0) & (vj2 < n)
    delta[vj2_v] += h[vj2[vj2_v]]
    return delta

def binary_sa(h_bin, n, deadline, seed=42, t_start=0.025, t_end=0.00001, patience=35000, batch=30):

    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    kp = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    total_steps = 400000
    cool_rate = (t_end / t_start) ** (1.0 / total_steps)
    T = t_start
    while time.time() < deadline:
        for _ in range(batch):
            if time.time() > deadline:
                return best_h, best_c
            i = zeros_idx[rng.randint(len(zeros_idx))]

            nc = (corr + delta).max() * dx
            dc = nc - cur_c

                h[i] = 1.0
                h[j] = 0.0
                corr = corr + delta
