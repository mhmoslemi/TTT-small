def swap_candidate_max(h, q, a, b, D, work, n):
    work[:, :] = D
    # slices
    st = n - 1 - a
    work[st:st+n] -= q
    st = n - 1 - b
    work[st:st+n] += q
    hrev = h[::-1]  # view; but each call overhead small
    work[a:a+n] += hrev
    work[b:b+n] -= hrev
    work[n - 1] -= 2
    work[n - 1 + (b - a)] += 1
    work[n - 1 + (a - b)] += 1

    y=np.clip(y,0,1)

def compute_c5(h, n):
    h = np.asarray(h, dtype=float)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_box_sum(y, target):
    y = np.asarray(y, dtype=float)
    y = np.clip(y, 0.0, 1.0)
    if abs(float(y.sum()) - float(target)) < 1e-12:
        return y
    lo = float(y.min())
    hi = float(y.max())
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > float(target):
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(y - lam, 0.0, 1.0)
    diff = float(target) - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0.0:
            room = 1.0 - h
            mask = room > 1e-10
            if mask.any():
                h = h.copy()
                h[mask] += diff * room[mask] / float(room[mask].sum())
        else:
            mask = h > 1e-10
            if mask.any():
                h = h.copy()
                h[mask] += diff * h[mask] / float(h[mask].sum())
    return np.clip(h, 0.0, 1.0)


def upsample_values(f, n):
    f = np.asarray(f, dtype=float)
    m = len(f)
    if m == n:
        return f.copy()
    pos = (np.arange(n) + 0.5) * float(m) / float(n)
    idx = np.minimum(pos.astype(np.int64), m - 1)
    return f[idx]

def init_full_binary(h):
    h32 = h.astype(np.int32)
    q32 = (1 - h).astype(np.int32)
    c = np.correlate(h32, q32, mode="full")
    return c[::-1].astype(np.int32)


def swap_candidate(h, q, a, b, D, work, n):
    work[:, :] = D
    st = n - 1 - a
    work[st:st + n] -= q
    st = n - 1 - b
    work[st:st + n] += q
    hrev = h[::-1]
    work[a:a + n] += hrev
    work[b:b + n] -= hrev
    work[n - 1] -= 2
    work[n - 1 + (b - a)] += 1
    work[n - 1 + (a - b)] += 1
    return int(work.max()), int(np.dot(work, work))

def compute_linear_delta(h, q, a, b, work, n):
    work[:] = 0.0
    st = n - 1 - a
    work[st:st + n] -= q
    st = n - 1 - b
    work[st:st + n] += q
    hrev = h[::-1]
    work[a:a + n] += hrev
    work[b:b + n] -= hrev


def eval_pair_max(D, L, t, lin, n, i1, i2):
    np.multiply(L, t, out=lin)
    lin += D
    m = float(lin.max())
    tt = t * t
    v = lin[n - 1] - 2.0 * tt
    if v > m:
        m = v
    v = lin[i1] + tt
    if v > m:
        m = v
    v = lin[i2] + tt
    if v > m:
        m = v
    return m


def refine_pair(h, D, a, b, n, work, lin):
    q = 1.0 - h
    compute_linear_delta(h, q, a, b, work, n)
    i1 = n - 1 + (b - a)
    i2 = n - 1 + (a - b)
    best_t = 0.0
    best_m = float(D.max())
    for k in range(401):
        t = k * 0.0025
        m = eval_pair_max(D, work, t, lin, n, i1, i2)
        if m < best_m - 1e-12:
            best_m = m
            best_t = t
    lo = max(0.0, best_t - 0.01)
    hi = min(1.0, best_t + 0.01)
    for k in range(101):
        t = lo + (hi - lo) * (k * 0.01)
        m = eval_pair_max(D, work, t, lin, n, i1, i2)
        if m < best_m - 1e-12:
            best_m = m
            best_t = t
    return best_t, best_m


def fractional_sequence(h, D, n, deadline, target_C, max_rounds=10):
    dx = 2.0 / float(n)
    h = np.asarray(h, dtype=float).copy()
    D = np.asarray(D, dtype=float).copy()
    L = 2 * n - 1
    work = np.zeros(L, dtype=float)
    lin = np.empty(L, dtype=float)
    best_h = h.copy()
    best_D = D.copy()
    best_c = float(D.max()) * dx
    idx0 = n - 1

    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        ones_idx = np.nonzero(h >= 1.0 - 1e-12)[0]
        zeros_idx = np.nonzero(h <= 1e-12)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        cur_max = float(D.max())
        q = 1.0 - h
        best_m = cur_max
        best_pair = None

        for ii in range(len(ones_idx)):
            a = int(ones_idx[ii])
            if time.time() > deadline:
                break
            for jj in range(len(zeros_idx)):
                b = int(zeros_idx[jj])
                compute_linear_delta(h, q, a, b, work, n)
                i1 = n - 1 + (b - a)
                i2 = n - 1 + (a - b)
                for t in T_LIST:
                    m = eval_pair_max(D, work, t, lin, n, i1, i2)
                    if m < best_m - 1e-12:
                        best_m = m
                        best_pair = (a, b)
                if (jj & 63) == 0 and time.time() > deadline:
                    break

        if best_pair is None:
            break
        a, b = best_pair
        t_ref, m_ref = refine_pair(h, D, a, b, n, work, lin)
        if m_ref < cur_max - 1e-12:
            h[a] = 1.0 - t_ref
            h[b] = t_ref
            D = D + t_ref * work
            tt = t_ref * t_ref
            D[idx0] -= 2.0 * tt
            D[n - 1 + (b - a)] += tt
            D[n - 1 + (a - b)] += tt
            np.maximum(D, 0.0, out=D)
            c = float(D.max()) * dx
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                best_D = D.copy()
                if best_c < target_C - 1e-9:
                    break
        else:
            break

    return best_h, best_c, n


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(1.0, float(budget_s) - 10.0)
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
    except Exception:
        h0 = None

    best_h = None
    best_c = 1e9
    best_n = 0

    tasks = [
        (256, 0, 120, 150),
        (250, 0, 80, 100),
        (254, 0, 60, 80),
        (256, 1, 80, 80),
        (250, 1, 60, 70),
        (256, 2, 60, 60),
        (250, 2, 50, 50),
        (264, 0, 60, 50),
    ]

    for task_i, (n, variant, bsec, fsec) in enumerate(tasks):
        if time.time() > deadline - 20.0:
            break
        avail = deadline - time.time()
        btime = min(float(bsec), max(5.0, avail * 0.5))
        ftime = min(float(fsec), max(0.0, avail - btime - 5.0))
        if ftime < 5.0 and btime < 5.0:
            continue

        ss = seed + 101 * task_i + 17 * variant + 3 * n
        hstart = make_start(n, variant, ss, h0)
        b_dead = min(deadline, time.time() + btime)
        hb, Db, bm = binary_anneal(hstart, n, b_dead, ss)

        cbin = float(bm) * (2.0 / float(n))
        cbin_actual = compute_c5(hb.astype(float), n)
        cbin = min(cbin, cbin_actual)
        if cbin < best_c:
            best_c = cbin_actual
            best_h = hb.astype(float)
            best_n = n

        f_dead = min(deadline, time.time() + ftime)
        if f_dead > time.time() + 5.0:
            hf, cf, _ = fractional_sequence(hb.astype(float), Db, n, f_dead, TARGET_C, max_rounds=10)
            hf = project_box_sum(hf, 0.5 * n)
            cf = compute_c5(hf, n)
            if cf < best_c:
                best_c = cf
                best_h = hf
                best_n = n

    if best_h is None:
        n = 256
        h = make_start(n, 1, seed, h0).astype(float)
        h = project_box_sum(h, 0.5 * n)
        c = compute_c5(h, n)
        return h, float(c), int(n)

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
