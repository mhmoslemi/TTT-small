                continue
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv <= 0:
                continue
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def obj_smooth(h, n, alpha):
    c = c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    return m + np.log(np.sum(np.exp(sh))) / alpha

def grad_active(h, n, act):
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in act:
        k = int(mi) - (n - 1)
        if k >= 0:
            nv = n - k
            if nv <= 0:
                continue
            grad[:nv] += (1.0 - h[k:])
            grad[k:] += (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            if nv <= 0:
                continue
            grad[kk:] += (1.0 - h[:nv])
            grad[:nv] += (-h[kk:])
    return grad * dx / max(len(act), 1)

def lbfgs_multi(h, n, t_end, alpha_seq, iters_each=150):
    h = project(h, n)
    best_c = c5_val(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_seq:
        if time.time() > t_end:
            break
        try:
            res = minimize(obj_smooth, h, jac=grad_smooth, args=(n, alpha),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': iters_each, 'ftol': 1e-16, 'gtol': 1e-13})
            hn = project(res.x, n)
            cn = c5_val(hn, n)
            if cn < best_c:
                best_c = cn
                h = hn
        except Exception:
            pass
    return h, best_c

def focused_gd(h, n, t_end, init_lr=0.05, tol=2e-3):
    h = project(h, n)
    best_c = c5_val(h, n)
    best_h = h.copy()
    lr = init_lr
    stall = 0
    for _ in range(200000):
        if time.time() > t_end:
            break
        c = c5_vec(h, n)
        m = c.max()
        act = np.where(c >= m - tol)[0]
        g = grad_active(h, n, act)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        step = lr
        improved = False
        for _ in range(45):
            hn = project(h - step * g, n)
            cn = c5_val(hn, n)
            if cn < best_c - 1e-16:
                h = hn
                best_c = cn
                best_h = h.copy()
                improved = True
                stall = 0
                break
            step *= 0.5
        if improved:
            lr = min(lr * 1.08, 0.1)
        else:
            lr *= 0.5
            stall += 1
            if stall > 20:
                break
            if lr < 1e-14:
                break
    return best_h, best_c

def pair_search(h, n, t_end, deltas, radius=None):
    h = project(h, n)
    best_c = c5_val(h, n)
    for d in deltas:
        any_imp = True
        while any_imp and time.time() < t_end:
            any_imp = False
            for i in range(n):
                if time.time() > t_end:
                    return h, best_c
                jlo = max(0, i - radius) if radius else 0
                jhi = min(n, i + radius + 1) if radius else n
                for j in range(jlo, jhi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        cn = c5_val(h, n)
                        if cn < best_c - 1e-14:
                            best_c = cn
                            any_imp = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        cn = c5_val(h, n)
                        if cn < best_c - 1e-14:
                            best_c = cn
                            any_imp = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def grad_guided_pairs(h, n, t_end, step_sizes):
    h = project(h, n)
    best_c = c5_val(h, n)
    for ss in step_sizes:
        any_imp = True
        while any_imp and time.time() < t_end:
            any_imp = False
            act = np.where(c5_vec(h, n) >= c5_vec(h, n).max() - 5e-3)[0]
            g = grad_active(h, n, act)
            inc = np.where(g < -1e-10)[0]
            dec = np.where(g > 1e-10)[0]
            for i in inc:
                if time.time() > t_end:
                    break
                for j in dec:
                    if i == j:
                        continue
                    mv = ss * 0.5
                    if h[i] + mv <= 1.0 and h[j] - mv >= 0.0:
                        h[i] += mv
                        h[j] -= mv
                        cn = c5_val(h, n)
                        if cn < best_c - 1e-14:
                            best_c = cn
                            any_imp = True
                        else:
                            h[i] -= mv
                            h[j] += mv
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 300
        h_init = np.full(n, 0.5)
    best_h = project(h_init, n)
    best_c = c5_val(best_h, n)

    # Phase 1: L-BFGS-B multi-alpha
    t1 = min(t0 + 280, deadline - 380)
    best_h, best_c = lbfgs_multi(best_h, n, t1, [500, 2000, 10000, 50000, 200000], 150)

    # Phase 2: Focused gradient descent
    if time.time() < deadline - 250:
        t2 = min(time.time() + 120, deadline - 250)
        h2, c2 = focused_gd(best_h, n, t2, init_lr=0.05, tol=3e-3)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: High-alpha L-BFGS
    if time.time() < deadline - 180:
        t3 = min(time.time() + 100, deadline - 180)
        h3, c3 = lbfgs_multi(best_h, n, t3, [100000, 500000, 2000000], 120)
        if c3 < best_c:
            best_c = c3
            best_h = h3

    # Phase 4: Focused GD again (tighter)
    if time.time() < deadline - 130:
        t4 = min(time.time() + 60, deadline - 130)
        h4, c4 = focused_gd(best_h, n, t4, init_lr=0.03, tol=1e-3)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Gradient-guided pair search
    if time.time() < deadline - 100:
        t5 = min(time.time() + 60, deadline - 100)
        h5, c5 = grad_guided_pairs(best_h, n, t5, [0.01, 0.005, 0.002, 0.001])
        if c5 < best_c:
            best_c = c5
            best_h = h5

    # Phase 6: Full pair search (coarse to fine)
    if time.time() < deadline - 50:
        t6 = min(time.time() + 120, deadline - 50)
        best_h, best_c = pair_search(best_h, n, t6,
                                      [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])

    # Phase 7: Try other n values
    if time.time() < deadline - 70:
        for n2 in [200, 400]:
            if time.time() > deadline - 40:
                break
            xo = np.arange(n) * 2.0 / n + 1.0 / n
            xn = np.arange(n2) * 2.0 / n2 + 1.0 / n2
            hi = np.interp(xn, xo, best_h)
            hi = project(hi, n2)
            alloc = min(50, deadline - time.time() - 25)
            if alloc < 25:
                break
            ht, ct = lbfgs_multi(hi, n2, time.time() + alloc * 0.55, [5000, 50000, 200000], 100)
            if time.time() < deadline - 15:
                ht, ct = focused_gd(ht, n2, min(time.time() + 15, deadline - 10), init_lr=0.03, tol=2e-3)
            if time.time() < deadline - 8:
                ht, ct = pair_search(ht, n2, deadline - 5, [0.002, 0.001, 0.0005, 0.0002])
            if ct < best_c:
                best_c = ct
                best_h = ht
                n = n2

    # Phase 8: Final refinement
    if time.time() < deadline - 3:
        nf = len(best_h)
        best_h, best_c = focused_gd(best_h, nf, deadline - 1, init_lr=0.02, tol=5e-4)
    if time.time() < deadline - 1:
        nf = len(best_h)
        best_h, best_c = pair_search(best_h, nf, deadline, [0.0005, 0.0002, 0.0001])

    best_h = project(best_h, len(best_h))
    best_c = c5_val(best_h, len(best_h))
    return best_h, best_c, len(best_h)
