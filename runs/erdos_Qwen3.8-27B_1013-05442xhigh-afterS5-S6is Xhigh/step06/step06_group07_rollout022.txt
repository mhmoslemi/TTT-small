def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def sa_search(h_bin, n, deadline, seed=42, t_start=0.03, t_end=0.0001, patience=20000):

    best_c = cur_c
    best_h = h.copy()
    
    T = t_start
    no_improve = 0
    n_restarts = 0
    max_restarts = 20
    cool_steps = patience  # steps per "epoch"
    
    while time.time() < deadline and n_restarts <= max_restarts:
        # Cooling schedule for this epoch
        cool_rate = (t_end / T) ** (1.0 / cool_steps)
        
        for step in range(cool_steps):
            if step % 5000 == 0 and time.time() > deadline:
                break
            
            # Pick random i (zero) and j (one)
            while True:
                i = rng.randint(n)
                if h[i] == 0: break
            while True:
                j = rng.randint(n)
                if h[j] == 1: break
            
            # Try swap
            h[i] = 1.0
            h[j] = 0.0
            new_corr = np.correlate(h, 1-h, mode='full').astype(float)
            nc = new_corr.max() * dx
            dc = nc - cur_c
            
            if dc < 0 or (T > 1e-12 and rng.random() < np.exp(-dc / T)):
                cur_corr = new_corr
                cur_c = nc
                if nc < best_c:

                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                # Reject: undo swap
                h[i] = 0.0
                h[j] = 1.0
                no_improve += 1
            
            T *= cool_rate

            h[ii] = 1.0

        cur_c = cur_corr.max() * dx
        T = t_start * (0.8 + 0.4 * rng.random())  # randomize restart temperature
        no_improve = 0

            best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def sa_search(h_bin, n, deadline, seed=42, t_start=0.025, t_end=0.00005, patience=30000, max_restarts=30):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    omh = 1.0 - h
    cur_corr = np.correlate(h, omh, mode='full').astype(float)
    cur_c = cur_corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    no_improve = 0
    n_restarts = 0
    cool_rate = (t_end / t_start) ** (1.0 / patience)
    T = t_start

    while time.time() < deadline and n_restarts < max_restarts:
        for step in range(patience):
            if (step & 4095) == 0 and time.time() > deadline:
                break
            while True:
                i = rng.randint(n)
                if h[i] == 0.0:
                    break
            while True:
                j = rng.randint(n)
                if h[j] == 1.0:
                    break
            h[i] = 1.0; omh[i] = 0.0
            h[j] = 0.0; omh[j] = 1.0
            new_corr = np.correlate(h, omh, mode='full').astype(float)
            nc = new_corr.max() * dx
            dc = nc - cur_c
            if dc < 0.0 or (T > 1e-12 and rng.random() < np.exp(-dc / T)):
                cur_corr = new_corr
                cur_c = nc
                if nc < best_c:
                    best_c = nc
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i] = 0.0; omh[i] = 1.0
                h[j] = 1.0; omh[j] = 0.0
                no_improve += 1
            T *= cool_rate
        if time.time() > deadline:
            break
        n_restarts += 1
        nk = 3 + rng.randint(8)
        for _ in range(nk):
            while True:
                ii = rng.randint(n)
                if h[ii] == 0.0:
                    break
            while True:
                jj = rng.randint(n)
                if h[jj] == 1.0:
                    break
            h[ii] = 1.0; omh[ii] = 0.0
            h[jj] = 0.0; omh[jj] = 1.0
        cur_corr = np.correlate(h, omh, mode='full').astype(float)
        cur_c = cur_corr.max() * dx
        T = t_start * (0.7 + 0.6 * rng.random())
        no_improve = 0
    return best_h, best_c

def greedy_polish(h, n, deadline, max_passes=50):
    dx = 2.0 / n
    h = h.copy()
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0.0)[0]
        ones = np.where(h == 1.0)[0]
        for i in zeros:
            if time.time() > deadline:
                return best_h, best_c
            for j in ones:
                h[i] = 1.0; omh[i] = 0.0
                h[j] = 0.0; omh[j] = 1.0
                nc = np.correlate(h, omh, mode='full').max() * dx
                if nc < cur_c - 1e-15:
                    cur_c = nc
                    corr = np.correlate(h, omh, mode='full').astype(float)
                    improved = True
                    break
                else:
                    h[i] = 0.0; omh[i] = 1.0
                    h[j] = 1.0; omh[j] = 0.0
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def slsqp_phase(h, n, deadline, alphas):
    dx = 2.0 / n
    for alpha in alphas:
        if time.time() > deadline:
            break
        def obj(x):
            g = np.correlate(x, 1 - x, mode='full') * dx
            m = g.max()
            sh = alpha * (g - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 800, 'ftol': 1e-12})
            h = project_feasible(res.x, n)
        except Exception:
            pass
    return h

def make_structured_start(n, period, seed):
    h = np.zeros(n)
    rng = np.random.RandomState(seed)
    per = period
    ones_per = per // 2
    pattern = np.zeros(per)
    pattern[:ones_per] = 1.0
    tiles = n // per
    h[:tiles * per] = np.tile(pattern, tiles)
    rem = n % per
    if rem > 0:
        h[tiles * per:] = rng.choice([0.0, 1.0], rem)
    needed = n // 2 - h.sum()
    if needed > 0:
        zeros = np.where(h == 0.0)[0]
        pick = rng.choice(zeros, int(needed), replace=False)
        h[pick] = 1.0
    elif needed < 0:
        ones = np.where(h == 1.0)[0]
        pick = rng.choice(ones, int(-needed), replace=False)
        h[pick] = 0.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:

    starts.append(h_bin)

        at_end = min(p2, time.time() + time_per_sa * 1.5)

                best_c = cg3

    elif time.time() < dl - 10:
        # Same resolution, final SA
