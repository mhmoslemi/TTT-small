            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room

            s = h.sum()
            if s < 1e-14:

            corr += delta
            cur_c = nc
            if i in zeros:
                zeros[zeros.index(i)] = j
            if j in ones:
                ones[ones.index(j)] = i
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c

def systematic_binary(h_init, n, deadline, max_sweeps=20):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += (1.0 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                j = best_j
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += (1.0 - h[im[msk]])
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= (1.0 - h[jm[msk]])
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                h[i], h[j] = 1.0, 0.0
                corr += delta
                cur_c = best_nc
                ones_list.remove(j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: Fast gradient descent with moderate alpha (broad exploration)
    p1 = t0 + 0.15 * total
    h, c = momentum_gd(best_h, n, p1, alpha_start=20, alpha_end=5000, total_steps=5000, beta=0.9)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 2: Higher alpha gradient descent (sharpening)
    p2 = t0 + 0.40 * total
    h, c = momentum_gd(best_h, n, p2, alpha_start=5000, alpha_end=200000, total_steps=8000, beta=0.85)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 3: Very high alpha (near-exact max optimization)
    p3 = t0 + 0.60 * total
    h, c = momentum_gd(best_h, n, p3, alpha_start=200000, alpha_end=5000000, total_steps=6000, beta=0.8)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 4: Binary local search from continuous solution
    p4 = t0 + 0.75 * total
    hb, cb = systematic_binary(best_h, n, min(p4, t0 + 0.05 * total), max_sweeps=8)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    hb2, cb2 = binary_ls(best_h, n, p4, max_iters=300000, seed=seed)
    if cb2 < best_c:
        best_c, best_h = cb2, hb2.copy()

    # Phase 5: Re-smooth from best binary with gradient
    p5 = t0 + 0.88 * total
    h = best_h.copy()
    h, c = momentum_gd(h, n, p5, alpha_start=10000, alpha_end=5000000, total_steps=5000, beta=0.85)
    if c < best_c:
        best_c, best_h = c, h.copy()

    # Phase 6: Noise restarts for final polish
    p6 = dl - 2
    if time.time() < p6:
        for trial in range(5):
            if time.time() > p6 - 0.5:
                break
            h = best_h + 0.01 * rng.randn(n)
            h = project_feasible(h, n)
            h2, c2 = momentum_gd(h, n, min(p6 - 0.1, time.time() + 15),
                                 alpha_start=100000, alpha_end=10000000,
                                 total_steps=3000, beta=0.8)
            if c2 < best_c:
                best_c, best_h = c2, h2.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
