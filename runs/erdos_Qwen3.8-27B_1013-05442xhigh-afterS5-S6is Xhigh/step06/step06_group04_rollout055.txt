    delta = np.zeros(2 * n - 1)
    ks = np.arange(-(n-1), n)
    # h[i: 0->1]: changes c_k by (1-h[i-k]) - h[i+k] for valid positions
    # h[j: 1->0]: changes c_k by h[j+k] - (1-h[j-k]) = h[j+k] + h[j-k] - 1

    # Δc_k = [contribution from i going 0->1] + [contribution from j going 1->0]

    #   - As the "1-h[i]" factor: term h[i+k]*(1-h[i]) changes from h[i+k]*1 to h[i+k]*0, delta = -h[i+k]
    #   - As the "h[i'+k]" factor when i'=i-k: term h[i]*(1-h[i-k]) changes from 0*(1-h[i-k]) to 1*(1-h[i-k]), delta = 1-h[i-k]

    #   - As the "h[j'+k]" factor when j'=j-k: term h[j]*(1-h[j-k]) changes from 1*(1-h[j-k]) to 0*(1-h[j-k]), delta = -(1-h[j-k])
    # Total: Δc_k = -h[i+k] + (1-h[i-k]) + h[j+k] - (1-h[j-k])
