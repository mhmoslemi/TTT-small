            break

    # Phase 2: SLSQP with multi-scale alpha
    t_phase2_end = t0 + (deadline - t0) * 0.55
    h = best_h.copy()
    for alpha in [2000, 8000, 30000, 100000, 400000]:
        if time.time() > t_phase2_end:
            break
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-10:]
        h_new = slsqp_smooth(h, n, top_idx, alpha, maxiter=4000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = best_h.copy()

    # Phase 3: Subgradient with momentum
    t_phase3_end = t0 + (deadline - t0) * 0.80
    h_sub = best_h.copy()
    h_sub, c_sub = subgrad_descent(h_sub, n, t_phase3_end)
    if c_sub < best_c:
        best_c = c_sub
        best_h = h_sub.copy()

    # Phase 3b: Restart subgradient from perturbed best
    if time.time() < t_phase3_end:
        for restart in range(3):
            if time.time() > t_phase3_end:
                break
            hp = best_h + rng.normal(0, 0.01, n)
            hp = project_feasible(hp, n)
            h_s, c_s = subgrad_descent(hp, n, min(t_phase3_end, time.time() + 60))
            if c_s < best_c:
                best_c = c_s
                best_h = h_s.copy()

    # Phase 4: Final SLSQP polish
    if time.time() < deadline - 15:
        h = best_h.copy()
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-12:]
        h_new = slsqp_smooth(h, n, top_idx, 1000000, maxiter=5000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

    # Phase 5: Coordinate descent
    if time.time() < deadline - 3:
        best_h, best_c = coord_descent(best_h, n, deadline - 1, max_rounds=200)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
