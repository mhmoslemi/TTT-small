def compute_c5(h, n):
    dx = 2.0 / n

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:

            h += (diff / tot) * h

def obj_smooth(x, n, alpha):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float((m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * dx)
    return float((m + np.log(np.sum(np.exp(sh))) / alpha) * dx)

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha=100000, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def coord_descent(h, n, max_passes=200, n_grid=40, deadline=None, penalty=3.0):
    h = h.copy().astype(float)
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = float(np.max(c))
    sumh = float(h.sum())
    target = n / 2.0
    idx = np.arange(n)

    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        p_start = np.random.permutation(n) if _pass > 0 else idx
        improved = False
        for p in p_start:
            hp = h[p]
            d_lo = -hp
            d_hi = 1.0 - hp
            if d_hi - d_lo < 1e-14:
                continue
            k_lo = max(0, n - 1 - p)
            k_hi = min(2 * n - 2, n - 1 + p)
            um = 0.0
            if k_lo > 0:
                um = max(um, float(np.max(c[:k_lo])))
            if k_hi < 2 * n - 2:
                um = max(um, float(np.max(c[k_hi + 1:])))
            ks = np.arange(k_lo, k_hi + 1)
            q_arr = ks - (n - 1)
            s = np.zeros(len(ks))
            m0 = (q_arr == 0)
            s[m0] = 1.0 - 2.0 * hp
            mp1 = (q_arr > 0) & (p + q_arr < n)
            s[mp1] += 1.0 - h[p + q_arr[mp1]]
            mp2 = (q_arr > 0) & (p - q_arr >= 0)
            s[mp2] -= h[p - q_arr[mp2]]
            mn1 = (q_arr < 0) & (p + q_arr >= 0)
            s[mn1] += 1.0 - h[p + q_arr[mn1]]
            mn2 = (q_arr < 0) & (p - q_arr < n)
            s[mn2] -= h[p - q_arr[mn2]]
            deltas = np.linspace(d_lo, d_hi, n_grid)
            best_d = 0.0
            best_val = current_max + penalty * (sumh - target) ** 2
            m0_ks = (ks == (n - 1))
            for d in deltas:
                if abs(d) < 1e-14:
                    continue
                nv = c[ks] + d * s
                if np.any(m0_ks):
                    nv[m0_ks] -= d * d
                val = max(um, float(np.max(nv)))
                if penalty > 0:
                    val += penalty * (sumh + d - target) ** 2
                if val < best_val - 1e-12:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-10:

                c[ks] += actual_d * s

                sumh += actual_d
                current_max = max(um, float(np.max(c[ks])))
                improved = True
        if not improved:
            break
    return h, current_max * (2.0 / n)

        hp[flip_idx[to_ones]] = 1.0
        hp[flip_idx[~to_ones]] = 0.0
    hp += rng.normal(0, scale, n)
    return project_feasible(hp, n)
