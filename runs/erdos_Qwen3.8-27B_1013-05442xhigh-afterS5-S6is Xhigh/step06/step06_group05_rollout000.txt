GLOBAL_TOP = None


def full_overlap_vector(h, n):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))


def compute_c5(h, n):

    return float(np.max(g))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    if h.size != n:
        h = np.resize(h, (n,))
    np.clip(h, 0.0, 1.0, out=h)

    target = 0.5 * float(n)
    s = float(h.sum())
    if abs(s - target) <= 1e-13:
        return h

    lo = -float(h.max())
    hi = 1.0 - float(h.min())
    for _ in range(80):
        t = 0.5 * (lo + hi)
        v = h + t
        np.clip(v, 0.0, 1.0, out=v)
        sv = float(v.sum())
        if sv > target:
            hi = t
        else:
            lo = t

    if abs(diff) > 1e-12:
        for _ in range(30):
            diff = target - float(h.sum())
            if abs(diff) < 1e-14:
                break
            if diff > 0.0:
                idx = np.where(h < 1.0)[0]
            else:
                idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            inc = diff / float(idx.size)
            h[idx] = np.clip(h[idx] + inc, 0.0, 1.0)
    return h


def sum_f(x):
    return float(x.sum() - 0.5 * GLOBAL_N)


def sum_jac(x):
    return np.ones((1, GLOBAL_N), dtype=float)


def smooth_f(x):
    g = full_overlap_vector(x, GLOBAL_N)
    if GLOBAL_TOP is None:
        vals = g
    else:
        vals = g[GLOBAL_TOP]

    m = float(np.max(vals))
    q = GLOBAL_ALPHA * (vals - m)
    qmax = float(np.max(q))
    e = np.exp(q - qmax)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return m
    return m + (qmax + float(np.log(s))) / GLOBAL_ALPHA


def smooth_grad(x):
    h = np.asarray(x, dtype=float)
    n = GLOBAL_N
    g = full_overlap_vector(h, n)

    if GLOBAL_TOP is None:
        vals = g
    else:
        vals = g[GLOBAL_TOP]

    m = float(np.max(vals))
    q = GLOBAL_ALPHA * (vals - m)
    qmax = float(np.max(q))
    e = np.exp(q - qmax)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        p = np.ones(vals.shape, dtype=float) / float(vals.size)
    else:
        p = e / s

    W = np.zeros(len(g), dtype=float)
    if GLOBAL_TOP is None:
        W[:] = p * GLOBAL_DX
    else:
        W[GLOBAL_TOP] = p * GLOBAL_DX

    mlag = n - 1
    A = np.convolve(W, np.ones(n, dtype=float), mode='valid')
    C1 = np.convolve(W, h, mode='full')[mlag:mlag + n]
    C2 = np.convolve(W[::-1], h, mode='full')[mlag:mlag + n]
    return A - C1 - C2


def slsqp_polish(h, n, alpha, top_mask, maxiter):
    global GLOBAL_N, GLOBAL_ALPHA, GLOBAL_DX, GLOBAL_TOP

    h = project_feasible(h, n)
    GLOBAL_N = int(n)
    GLOBAL_ALPHA = float(alpha)
    GLOBAL_DX = 2.0 / float(n)
    GLOBAL_TOP = top_mask

    t_start = time.time()
    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = {'type': 'eq', 'fun': sum_f, 'jac': sum_jac}
    opts = {'maxiter': int(maxiter), 'ftol': 1e-14, 'disp': False}
    opts['maxls'] = 500 if n <= 300 else 300

    try:
        res = minimize(
            smooth_f,
            h,
            method='SLSQP',
            jac=smooth_grad,
            bounds=bounds,
            constraints=[cons],
            options=opts
        )
        if getattr(res, 'x', None) is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n), time.time() - t_start
    except Exception:
        pass

    return h, time.time() - t_start


def select_top_mask(h, n, k=10, tol=1e-5, cap=20):
    g = full_overlap_vector(h, n)
    m = float(np.max(g))
    near = np.where(g >= m - tol)[0]
    topk = np.argsort(g)[-k:]
    idx = np.union1d(near, topk)
    if idx.size > cap:
        idx = np.argsort(g)[-cap:]
    if idx.size == 0:
        idx = np.array([np.argmax(g)])
    mask = np.zeros(len(g), dtype=bool)
    mask[idx] = True
    return mask


def upsample_linear(h, m):
    n = len(h)
    if n <= 0:
        return project_feasible(np.full(m, 0.5), m)
    x_old = np.linspace(0.0, float(n), n, endpoint=False)
    x_new = np.linspace(0.0, float(n), m, endpoint=False)
    out = np.interp(x_new, x_old, h)
    return project_feasible(out, m)


def make_fallback(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    phase = float(rng.rand()) * 6.28
    h = 0.5 + 0.25 * np.sin(2.0 * np.pi * x / 0.63 + phase)
    h = h + 0.12 * np.sin(2.0 * np.pi * x / 0.27 + 0.7)
    return project_feasible(h, n)


def optimize_h(h, n, deadline):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    if n <= 300:
        base_iter = 60
    elif n <= 600:
        base_iter = 35
    else:
        base_iter = 25
    factor = 1.0

    full_alphas = (200.0, 800.0, 3000.0, 10000.0, 30000.0, 100000.0)
    full_margin = 80.0 if n <= 300 else 120.0

    for alpha in full_alphas:
        if time.time() > deadline - full_margin:
            break
        mi = max(20, int(base_iter * factor))
        hp, el = slsqp_polish(h, n, alpha, None, mi)
        hc = compute_c5(hp, n)
        if hc < best_c:
            best_h = hp.copy()
            best_c = hc
        h = best_h.copy()
        if el > 60.0:
            factor *= 0.4
        elif el > 30.0:
            factor *= 0.6
        elif el > 15.0:
            factor *= 0.8

    alpha = 20000.0
    top_margin = 40.0 if n <= 300 else 60.0
    for _ in range(8):
        if time.time() > deadline - top_margin:
            break
        mask = select_top_mask(best_h, n, k=12, tol=1e-5, cap=24)
        mi = max(20, int(0.8 * base_iter * factor))
        hp, el = slsqp_polish(best_h, n, alpha, mask, mi)
        hc = compute_c5(hp, n)
        if hc < best_c - 1e-15:
            best_h = hp.copy()
            best_c = hc
            alpha *= 1.2
        else:
            if alpha > 5000.0:
                alpha *= 0.7
            else:
                break
        if el > 30.0:
            factor *= 0.75

    if time.time() < deadline - 60.0:
        mi = max(20, int(base_iter * factor))
        hp, _ = slsqp_polish(best_h, n, 300000.0, None, mi)
        hc = compute_c5(hp, n)
        if hc < best_c:
            best_h = hp.copy()
            best_c = hc

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total_budget = max(60.0, float(budget_s))
    total_deadline = t0 + total_budget - 3.0

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(h0.size)
    except Exception:
        n0 = 250
        h0 = make_fallback(n0, int(seed))

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(h0, n0)
    best_n = n0

    phase1_deadline = t0 + 0.68 * total_budget
    if time.time() < phase1_deadline:
        h1, c1 = optimize_h(best_h, best_n, phase1_deadline)
        if c1 < best_c:
            best_h = h1
            best_c = c1
            best_n = n0

    if best_n < 500 and time.time() < total_deadline - 200.0:
        m = 500
        h_up = upsample_linear(best_h, m)
        c_up = compute_c5(h_up, m)
        if c_up < best_c + 0.005:
            h2, c2 = optimize_h(h_up, m, total_deadline - 80.0)
            if c2 < best_c:
                best_h = h2
                best_c = c2
                best_n = m

    if time.time() < total_deadline - 80.0:
        mi = 60 if best_n <= 300 else 30
        hp, _ = slsqp_polish(best_h, best_n, 100000.0, None, mi)
        hc = compute_c5(hp, best_n)
        if hc < best_c:
            best_h = hp
            best_c = hc

    if time.time() < total_deadline - 40.0:
        mask = select_top_mask(best_h, best_n, k=10, tol=1e-5, cap=20)
        mi = 40 if best_n <= 300 else 22
        hp, _ = slsqp_polish(best_h, best_n, 300000.0, mask, mi)
        hc = compute_c5(hp, best_n)
        if hc < best_c:
            best_h = hp
            best_c = hc

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
