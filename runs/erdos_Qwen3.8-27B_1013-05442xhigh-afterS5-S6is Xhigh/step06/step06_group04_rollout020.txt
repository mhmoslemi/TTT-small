    cons = [{'type': 'eq', 'fun': cons_f}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def run(seed=42, budget_s=1100, **kwargs):
    global n_global
    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(250, 0.5)
    n = len(h0)
    n_global = n
    
    h = binarize(h0, n)
    best_c = compute_c5_val(h, n)
    best_h = h.copy()
    
    # Phase 1: Greedy LS (25% time)
    p1_dl = t0 + 0.25 * (budget_s - 10)
    h, c = greedy_ls(h, n, p1_dl, max_sweeps=20000, seed=seed)
    if c < best_c:
        best_c, best_h = c, h.copy()
    
    # Phase 2: Best-improvement LS on best (10% time)
    p2_dl = t0 + 0.35 * (budget_s - 10)
    if time.time() < p2_dl:
        h, c = best_improvement_ls(best_h, n, p2_dl, max_sweeps=100)
        if c < best_c:
            best_c, best_h = c, h.copy()
    
    # Phase 3: ILS with kicks (45% time)
    p3_dl = t0 + 0.80 * (budget_s - 10)
    kick_count = 0
    while time.time() < p3_dl:
        kick_count += 1
        rem = p3_dl - time.time()
        if rem < 10:
            break
        kp = rng.randint(3, 9)
        h_k = make_kick(best_h, n, rng, n_pairs=kp)
        ls_time = min(80, rem * 0.7)
        h_l, c_l = greedy_ls(h_k, n, time.time() + ls_time, max_sweeps=10000, seed=seed + kick_count * 211)
        if c_l >= best_c:
            h_l, c_l = best_improvement_ls(h_l, n, time.time() + 20, max_sweeps=30)
        if c_l < best_c:
            best_c, best_h = c_l, h_l.copy()
    
    # Phase 4: SLSQP continuous refinement (15% time)
    p4_dl = deadline - 15
    if time.time() < p4_dl:
        for alpha in [3000, 15000, 80000]:
            if time.time() > p4_dl - 30:
                break
            h_cont = slsqp_refine(best_h, n, alpha, maxiter=250)
            h_bin = binarize(h_cont, n)
            h_l, c_l = greedy_ls(h_bin, n, min(p4_dl - 5, time.time() + 45),
                                  max_sweeps=8000, seed=seed + 55555 + alpha)
            if c_l < best_c:
                best_c, best_h = c_l, h_l.copy()
    
    # Phase 5: Final greedy polish
    if time.time() < deadline - 3:
        h, c = greedy_ls(best_h, n, deadline - 2, max_sweeps=30000, seed=seed + 12345)
        if c < best_c:
            best_c, best_h = c, h.copy()
    
    return best_h, best_c, n
