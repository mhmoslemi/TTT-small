```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m < n:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] += h[:n-m]
    return grad

def smooth_max_obj(x, n, alpha, top_idx=None):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c = c[top_idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sc(x, n):
    return float(np.sum(x) - n / 2.0)

def _scj(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, top_idx=None, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha, top_idx),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def cutting_plane(h0, n, deadline, max_iters=80, trust_init=0.12):
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    trust = trust_init
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-20:]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
                cp.norm(h_var - h, 1) <= trust]
        for idx in top_idx:
            lag = idx - (n - 1)
            grad = _lag_grad(h, n, lag)
            cons.append(c[idx] + grad @ (h_var - h) <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        solved = False
        for solver in [cp.GLOP, cp.CBC]:
            try:
                prob.solve(solver=solver)
                solved = True
                break
            except Exception:
                continue
        if not solved or h_var.value is None:
            break
        h_new = np.clip(h_var.value, 0, 1)
        h_new = project_feasible(h_new, n)
        step = np.max(np.abs(h_new - h))
        if step < 1e-9:
            break
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        trust = max(0.008, trust * 0.88)
    return best_h, best_c

def coord_descent_active(h, n, deadline, max_sweeps=30):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        if c.max() < best_c - 1e-12:
            best_c = c.max()
            best_h = h.copy()
        top_idx = np.argsort(c)[-4:]
        improved = False
        for idx in top_idx:
            lag = idx - (n - 1)
            m = abs(lag)
            if m == 0 or m >= n:
                continue
            stride = max(1, (n - m) // 300)
            for i in range(0, n - m, stride):
                j = i + m
                s = h[i] + h[j]
                if s >= 1:
                    u_star = (s - 1) / 2.0
                else:
                    u_star = 0.0
                u_star = float(np.clip(u_star, max(0.0, s - 1.0), min(1.0, s)))
                if abs(h[i] - u_star) > 1e-6:
                    h[i] = u_star
                    h[j] = s - u_star
                    improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    c_final = compute_c5(h, n)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c

def local_search_sa(h, n, deadline, n_iter=3000):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = 0.003
    rng = np.random.RandomState(123)
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        m = abs(lag)
        if m == 0 or m >= n:
            i = rng.randint(n - 1)
            j = i + 1
            m_use = 1
        else:
            i = rng.randint(0, n - m)
            j = i + m
            m_use = m
        if h[i] >= h[j]:
            delta = min(0.008, h[i] * 0.4)
        else:
            delta = -min(0.008, (1.0 - h[j]) * 0.4)
        if abs(delta) < 1e-9:
            continue
        h_save = h.copy()
        h[i] += delta
        h[j] -= delta
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h.copy()
        elif c_new > best_c and rng.random() > np.exp(-(c_new - best_c) / T):
            h = h_save
        T *= 0.9985
    return best_h, best_c
```
