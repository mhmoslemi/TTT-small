def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    """Gradient of c_k with respect to h."""
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad

        return grad

        grad[:n-m] += (1.0 - h[m:])

        grad += lag_gradient(h, n, lag)

    if gn < 1e-14:
        return h.copy()
    grad /= gn

    h_new = h - step_size * grad
    return project_feasible(h_new, n)

def outer_approx_step(h_curr, n, trust):

    c = compute_all_lags(h_curr, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()

    if prob.status not in ['optimal', 'optimal_inaccurate']:

    h_new = h_var.value

def smooth_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    s = np.sum(np.exp(shifted))
    return float(m + np.log(s) / alpha) if s > 0 else float(m)

    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

        h_oo, c_oo = outer_approx(best_h, n, oo_end)
        if c_oo < best_c:
            best_c = c_oo
            best_h = h_oo

    if time.time() < deadline - 40:
        sub_end = time.time() + min(35, (deadline - time.time()) * 0.3)
        h_sub, c_sub = subgradient_method(best_h, n, sub_end, top_m=15)
        if c_sub < best_c:
            best_c = c_sub
            best_h = h_sub

    if time.time() < deadline - 35:
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-12:]
        for alpha in [5000, 20000, 50000, 100000]:
            if time.time() > deadline - 35:
                break
            h_new = slsqp(best_h, n, alpha, top_idx=top_idx, maxiter=2500)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
                c = compute_all_lags(best_h, n)
                top_idx = np.argsort(c)[-12:]

    if time.time() < deadline - 15:
        h_new, c_new = fine_local_search(best_h, n, deadline - 3)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    if time.time() < deadline - 10:
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-8:]
        h_new = slsqp(best_h, n, 200000, top_idx=top_idx, maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        alloc = min(400, (deadline - t0) * 0.5)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt

    n_candidates = [200, 250, 350, 400, 500]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 30
        if remaining < 60:
            break
        alloc = min(85, remaining * 0.28)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    if best_h is not None and time.time() < deadline - 70:
        n = len(best_h)
        remaining = deadline - time.time() - 35
        n_restarts = max(1, int(remaining / 55))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            if r < n_restarts // 2:
                scale = 0.012 * (1.0 + 0.3 * r)
                hp = best_h + rng.normal(0, scale, n)
            else:
                # structured perturbation: add a sinusoid
                freq = rng.uniform(1, 5)
                phase = rng.uniform(0, 2 * np.pi)
                amp = 0.02 * (1 + 0.2 * r)
                x = np.linspace(0, 2, n)
                hp = best_h + amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 35) * 0.4)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final polish on best solution
    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        c = compute_all_lags(best_h, n)
        top_idx = np.argsort(c)[-8:]
        h_new = slsqp(best_h, n, 300000, top_idx=top_idx, maxiter=4000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        if time.time() < deadline - 3:
            best_h, best_c = fine_local_search(best_h, n, deadline - 1)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    return best_h, best_c, len(best_h)
