        if time.time() > deadline:
            break

            c = np.correlate(np.clip(x, 0, 1), 1 - np.clip(x, 0, 1), mode='full') * dx
            scaled = alpha * c
            m = scaled.max()
            return m + np.log(np.exp(scaled - m).sum()) / alpha

            c = np.correlate(x_clip, 1 - x_clip, mode='full') * dx

                # As second factor

                idx2 = n - 1 + k_vals2

                grad_c[j] -= np.sum(weights[idx2[valid2]] * x[j + k_vals2[valid2]])
            return dx * grad_c * weights.sum() / w_sum  # Actually need to be more careful
        # This gradient is expensive. Let me just use numerical gradient or SLSQP's internal one.

def c5_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx  # shape (2n-1,)
    # c[k] corresponds to lag t = k - (n-1)
    s = alpha * c
    m = s.max()
    w = np.exp(s - m)  # shape (2n-1,)
    wsum = w.sum()
    obj = (m + np.log(wsum)) / alpha
    
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        # x[j] as first factor h[i+t]: need i+t=j, i valid for lag t
        # t = j - i, where i in [0, n-1] and the term exists
        # For t >= 0 (i <= j): term h[j]*(1-h[i]) exists in C[t] if 0<=i<=n-1-t
        #   t = j-i >= 0 => i <= j. Also i <= n-1-t = n-1-(j-i) = n-j+i
        #   So i <= min(j, n-j+i) => i <= j (since n-j+i >= i, and i <= j means n-j+i >= n-j >= 0 if j <= n)
        #   Wait, I need i <= n-1-t = n-1-(j-i). So i <= n-1-j+i => 0 <= n-1-j => j <= n-1. Always true.
        #   And i >= 0. So for i in [0, j]: term exists if j-i >= 0 (true) and i <= n-1-(j-i).
        #   i <= n-1-j+i => 0 <= n-1-j. True for j <= n-1.
        #   So for i in [0, j]: contribution is (1-h[i]), corr index = n-1+t = n-1+j-i
        for i in range(0, j + 1):
            idx = n - 1 + j - i
            val += w[idx] * (1.0 - x[i])
        # For t < 0 (i > j): t = j-i < 0, |t| = i-j
        #   C[-|t|] = sum_{i'=|t|}^{n-1} h[i'-|t|]*(1-h[i'])
        #   h[j] appears as h[i'-|t|] when i'-|t|=j, i'=j+|t|=i
        #   Valid if |t| <= i' <= n-1, i.e., i-j <= i <= n-1 => i <= n-1 (true) and i >= i (true)
        #   Wait: i' = i, |t| = i-j. Need |t| <= i' => i-j <= i => -j <= 0 (true).
        #   And i' <= n-1 => i <= n-1 (true).
        #   So for i in [j+1, n-1]: contribution is (1-h[i]), corr index = n-1+t = n-1+(j-i)
        for i in range(j + 1, n):
            idx = n - 1 + j - i
            val += w[idx] * (1.0 - x[i])
        # x[j] as second factor (1-h[j]): need i=j, term h[j+t]*(1-h[j]) exists in C[t]
        # For t >= 0: i=0 to n-1-t. Need i=j, so j <= n-1-t => t <= n-1-j => j+t <= n-1
        #   Contribution: -h[j+t], corr index = n-1+t
        #   t ranges from 0 to n-1-j
        for t in range(0, n - j):
            idx = n - 1 + t
            val += w[idx] * (-x[j + t])
        # For t < 0: i=|t| to n-1. Need i=j, so |t| <= j <= n-1 => |t| <= j => -t <= j => t >= -j
        #   t ranges from -j to -1
        #   Contribution: -h[j+t], corr index = n-1+t
        for t in range(-j, 0):
            idx = n - 1 + t
            val += w[idx] * (-x[j + t])
        grad[j] = (dx / wsum) * val
    return obj, grad
