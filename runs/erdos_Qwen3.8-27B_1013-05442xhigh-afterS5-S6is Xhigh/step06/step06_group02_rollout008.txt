from scipy.optimize import minimize
import time

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_exact(v, n):
    target = n / 2.0
    lo, hi = -3.0, 4.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if diff > 1e-14:
        mask = h < 1.0
        if mask.sum() > 0:
            h[mask] = np.minimum(h[mask] + diff / mask.sum(), 1.0)
    elif diff < -1e-14:
        mask = h > 0.0
        if mask.sum() > 0:
            h[mask] = np.maximum(h[mask] + diff / mask.sum(), 0.0)
    return h

def _obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def _grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for kp in range(2 * n - 1):

        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return grad

def _con_grad(h, n, alpha, mu):
    g = _grad(h, n, alpha)
    g += 2.0 * mu * (h.sum() - n / 2.0) * np.ones(n)
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_exact(v, n):
    target = n / 2.0
    lo, hi = -4.0, 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if diff > 1e-14:
        mask = h < 1.0
        if mask.sum() > 0:
            h[mask] = np.minimum(h[mask] + diff / mask.sum(), 1.0)
    elif diff < -1e-14:
        mask = h > 0.0
        if mask.sum() > 0:
            h[mask] = np.maximum(h[mask] + diff / mask.sum(), 0.0)
    return h

def _penalty_obj(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha + mu * (h.sum() - n / 2.0) ** 2)

def _penalty_grad(h, n, alpha, mu):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for kp in range(2 * n - 1):
        if w[kp] < 1e-14:
            continue
        k = kp - (n - 1)
        wm = w[kp] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    grad += 2.0 * mu * (h.sum() - n / 2.0)
    return grad

    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

        h2, c2 = slsqp_optimize(h, n, t2, alpha=200000, maxiter=3000)
        if c2 < best_c:
            best_c = c2; best_h = h2; h = h2

    # Phase 3: Aggressive pair search (350s)
    if time.time() < deadline - 100:
        t3 = min(time.time() + 350, deadline - 100)
        h, best_c = pair_search(h, n, t3,
                                deltas=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
                                radius=100)
        best_h = h.copy()

    # Phase 4: Try alternative n values (150s)
    if time.time() < deadline - 60:
        for n_try in [200, 250]:
            if time.time() > deadline - 40:
                break
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_exact(h_interp, n_try)
            alloc = min(60, deadline - time.time() - 30)
            if alloc < 25:
                break
            h_t, c_t = lbfgs_optimize(h_interp, n_try, time.time() + alloc * 0.5,
                                       alphas=[200, 2000, 20000, 200000], iters=200)
            if time.time() < deadline - 25:
                h_t, c_t = pair_search(h_t, n_try, time.time() + min(20, deadline - time.time() - 10),
                                       deltas=[0.001, 0.0005, 0.0002], radius=50)
            if c_t < best_c:
                best_c = c_t; best_h = h_t

    # Phase 5: Try a random restart (50s)
    if time.time() < deadline - 20:
        h_rand = np.clip(0.5 + 0.3 * rng.randn(n), 0, 1)
        h_rand = project_exact(h_rand, n)
        h_r, c_r = lbfgs_optimize(h_rand, n, time.time() + 40,
                                   alphas=[100, 1000, 10000], iters=150)
        if c_r < best_c:
            best_c = c_r; best_h = h_r

    # Phase 6: Final polish
    if time.time() < deadline - 3:
        nf = len(best_h)
        hf, cf = lbfgs_optimize(best_h, nf, deadline - 8, alphas=[100000, 500000], iters=100)
        if cf < best_c:
            best_c = cf; best_h = hf
        if time.time() < deadline - 2:
            best_h, best_c = pair_search(best_h, nf, deadline - 1,
                                         deltas=[0.001, 0.0005, 0.0002, 0.0001], radius=50)

    best_h = project_exact(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
