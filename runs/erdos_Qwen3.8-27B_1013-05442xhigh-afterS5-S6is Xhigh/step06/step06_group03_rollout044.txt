    h_new = h_var.value
    if h_new is None:
        return h.copy()
    return project_feasible(h_new, n)

def full_optimize(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: LP outer approximation
    if time.time() < deadline - 25:
        trust = 0.2
        for it in range(30):
            if time.time() > deadline - 25:
                break
            h_new = lp_outer_approx(h, n, trust)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new.copy()
            step = np.max(np.abs(h_new - h))
            if step < 1e-9:
                break
            h = h_new
            trust = max(0.003, trust * 0.85)
    # Phase 2: SLSQP
    if time.time() < deadline - 15:
        for alpha in [5000, 50000, 200000]:
            if time.time() > deadline - 15:
                break
            h_new = slsqp_optimize(h, n, alpha, maxiter=2000, ftol=1e-14)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
    # Phase 3: Active set SLSQP
    if time.time() < deadline - 12:
        c = compute_all_lags(h, n)
        top_idx = np.argsort(c)[-12:]
        for alpha in [50000, 200000, 500000]:
            if time.time() > deadline - 12:
                break
            h_new = slsqp_optimize(h, n, alpha, maxiter=2000, ftol=1e-15, top_idx=top_idx)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
    # Phase 4: Fine local search
    if time.time() < deadline - 3:
        h_new, c_new = fine_local_search(best_h, n, deadline - 1)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    # Phase 5: Binary snap + reoptimize
    if time.time() < deadline - 8:
        for thresh in [0.08, 0.15, 0.25]:
            if time.time() > deadline - 5:
                break
            h_snap = binary_snap(best_h, n, thresh)
            c_snap = compute_c5(h_snap, n)
            if time.time() < deadline - 4:
                h_new = slsqp_optimize(h_snap, n, 100000, maxiter=1500, ftol=1e-15)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
            elif c_snap < best_c:
                best_c = c_snap
                best_h = h_snap
    return project_feasible(best_h, n), compute_c5(best_h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    best_h = None
    best_c = float('inf')
    if h_init is not None:
        alloc = min(250, (deadline - t0) * 0.3)
        h_opt, c_opt = full_optimize(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    # Try other grid sizes
    n_candidates = [200, 250, 350, 400, 500]
    for n_try in n_candidates:
        if time.time() > deadline - 50:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 30
        if remaining < 35:
            break
        alloc = min(70, remaining * 0.25)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = full_optimize(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Random restarts
    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        n_restarts = max(1, int((deadline - time.time() - 20) / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.01 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(40, (deadline - time.time() - 20) * 0.35)
            if alloc < 15:
                break
            h_opt, c_opt = full_optimize(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = slsqp_optimize(best_h, n, 500000, maxiter=3000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h_final, c_final = fine_local_search(best_h, n, deadline - 0.5)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
