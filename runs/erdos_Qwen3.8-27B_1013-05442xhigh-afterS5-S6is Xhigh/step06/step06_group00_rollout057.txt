from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgrad_step(h, n, eta, k_top):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    top_idx = np.argsort(corr)[-k_top:]
    lags_arr = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for idx in top_idx:
        L = int(lags_arr[idx])
        if L == 0:
            grad -= (2.0 * h - 1.0)
        elif L > 0:
            grad[:n - L] += (1.0 - h[L:])
            grad[L:] -= h[:n - L]
        else:
            La = -L
            grad[La:] += (1.0 - h[:n - La])
            grad[:n - La] -= h[La:]
    h_new = h - eta * dx * grad / k_top
    return project_feasible(h_new, n)

def lse_values(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    logsum = np.log(np.exp(shifted - sm).sum()) + sm
    obj = m + logsum / alpha
    probs = np.exp(shifted - sm)
    probs /= probs.sum()
    return obj, probs, corr

    dx = 2.0 / n
    lags_arr = np.arange(-(n - 1), n)

        elif L > 0:
            grad[:n - L] += p * dx * (1.0 - h[L:])
            grad[L:] -= p * dx * h[:n - L]
        else:
            La = -L
            grad[La:] += p * dx * (1.0 - h[:n - La])
            grad[:n - La] -= p * dx * h[La:]
    return obj, grad

    def obj_pen(x):
        hp = np.clip(x, 0, 1)
        o, _, _ = lse_values(hp, n, alpha)
        return o + 0.5 * rho * (hp.sum() - n / 2.0) ** 2
    def grad_pen(x):

        o, g = lse_gradient(hp, n, alpha, topk=60)
        g = g + rho * (hp.sum() - n / 2.0)
        return g
    bounds = [(0.0, 1.0)] * n
    try:

                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_optimize(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        hp = np.clip(x, 0, 1)
        c = np.correlate(hp, 1 - hp, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
