def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:

            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def corr_grad(h, n, weights):
    """Compute gradient of weighted sum of correlations.
    weights: array of length 2n-1, indexed as weights[n-1+k] for shift k
    Returns: gradient vector of length n
    """
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(-(n-1), n):
        w = weights[n-1+k]
        if abs(w) < 1e-18:
            continue
        if k >= 0:
            g[k:] += w * dx * (1.0 - h[:n-k])
            if n-k > 0:
                g[:n-k] -= w * dx * h[k:]
        else:
            kk = -k
            g[:n-kk] += w * dx * (1.0 - h[kk:])
            g[kk:] -= w * dx * h[:n-kk]
    return g

def lgb_optimize(h, n, alpha, mu=1e4, maxiter=300):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = np.clip(alpha * (c - m), -400, 0)
        val = m + np.log(np.exp(shifted).sum()) / alpha
        val += mu * (x.sum() - target) ** 2
        return val
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = np.clip(alpha * (c - m), -400, 0)
        exps = np.exp(shifted)
        total = exps.sum()
        weights = exps / total
        g = corr_grad(x, n, weights / alpha)
        g += 2.0 * mu * (x.sum() - target)
        return g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
    return project_feasible(res.x, n)

def proj_subgrad(h, n, maxiter=5000, eta0=0.05, deadline=None):
    """Projected subgradient method for the max problem."""
    dx = 2.0 / n
    target = n / 2.0
    h = project_feasible(h, n).copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c = get_corr(h, n)
    t = c.max()
    
    for it in range(maxiter):
        if deadline is not None and time.time() > deadline:
            break
        # Find active shifts (within 5% of max)
        threshold = t - 0.05 * t
        active = c >= threshold
        weights = np.zeros(len(c))
        weights[active] = (c[active] - (t - 0.05*t)) ** 2  # weight by excess
        
        if weights.sum() < 1e-20:
            break
        
        weights = weights / weights.sum()
        g = corr_grad(h, n, weights)
        
        # Normalized step
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        eta = eta0 / (1.0 + 0.01 * it)
        step = eta * g / gn
        
        # Try the step with backtracking
        for _ in range(10):
            h_new = project_feasible(h - step, n)
            c_new = get_corr(h_new, n)
            t_new = c_new.max()
            if t_new < t:
                break
            step *= 0.5
        else:
            break
        
        if t_new < best_c:
            best_c = t_new
            best_h = h_new.copy()
        
        h = h_new
        c = c_new
        t = c.max()
    
    return best_h, best_c

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n//2]] = 1.0
    return h_b
