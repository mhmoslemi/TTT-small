```python
    """Stable logsumexp of alpha * c."""
    m = c.max()
    a = alpha * (c - m)
    a_max = a.max()
    return m + (a_max + np.log(np.exp(a - a_max).sum())) / alpha

    """Softmax weights for the logsumexp."""
    m = c.max()
    a = alpha * (c - m)
    a_max = a.max()
    w = np.exp(a - a_max)
    w /= w.sum()
    return w

    """Subgradient of max_k c[k]*dx where c = correlate(h, 1-h, 'full')."""
    dx = 2.0 / n
    c = correlate_full(h) * dx
    max_c = c.max()
    active = np.where(c >= max_c - active_tol)[0]
    if len(active) == 0:

    grad = np.zeros(n)

        i_hi = min(n - 1, n - 1 - m)

        i_lo2 = max(0, m)
        i_hi2 = min(n - 1, n - 1 + m)
        if i_lo2 <= i_hi2:

    grad *= dx / len(active)

    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        c = correlate_full(h) * (2.0 / n)
        # Compute gradient of logsumexp
        w = logsumexp_weights(c, alpha)
        grad = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-10:

            m = k - (n - 1)
            i_lo = max(0, -m)
            i_hi = min(n - 1, n - 1 - m)

            i_lo2 = max(0, m)

            if i_lo2 <= i_hi2:
                grad[i_lo2:i_hi2+1] -= w[k] * h[i_lo2 - m : i_hi2 + 1 - m]

            best_h = h_new.copy()
        h = h_new
        # Adapt learning rate
        if step % 500 == 499:
            lr *= 0.9
    return best_h, best_c
```

```python
            if s < 1e-14:

    b[idx[:target]] = 1.0

    """Returns (c5_value, softmax_weights, correlation)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    c = corr * dx
    m = c.max()
    a = alpha * (c - m)
    a_max = a.max()
    w = np.exp(a - a_max)
    w /= w.sum()
    return m, w, corr

fft(h, n, w):
    """O(n log n) gradient of logsumexp-smoothed max correlation."""
    dx = 2.0 / n
    N = 4 * n
    h1 = 1.0 - h
    H1 = np.fft.fft(h1, N)
    H = np.fft.fft(h, N)
    W = np.fft.fft(w, N)
    t1 = np.fft.ifft(H1 * np.conj(W)).real
    t2 = np.fft.ifft(H * W).real
    grad_first = t1[n-1::-1]
    grad_second = t2[n-1:2*n-1]
    return dx * (grad_first - grad_second)

def pgd_optimize(h, n, deadline, alpha_init=100, alpha_end=1e6, n_total=20000):
    """Projected gradient descent with annealing alpha."""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.1
    n_steps = n_total
    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Anneal alpha
        frac = step / max(1, n_steps - 1)
        alpha = alpha_init * (alpha_end / alpha_init) ** frac

        _, w, _ = get_c5_and_weights(h, n, alpha)
        grad = smooth_gradient_fft(h, n, w)
        
        # Backtracking line search

                best_h = h_new.copy()
            if c_new < compute_c5(h, n) + 1e-8:  # sufficient decrease
                h = h_new
                break
            lr *= 0.5
        else:
            h = project_feasible(h - lr * grad, n)
        
        if step % 1000 == 0:
            lr = max(0.01, 0.1 * (1.0 - frac))
    
    return best_h, best_c
```

Let me also reconsider. The log-sum-exp gradient approach requires computing the FFT at each step, which is O(n log n). For n=250 and 20000 steps, that's 20000 * 250 * 8 ≈ 40M operations. Should be fine.

But the projected step (project_feasible) with 200 iterations of the bisection-like projection is O(n * 200) per step, which is 250 * 200 * 20000 = 10^9. That's too slow!

I need a faster projection. The constraint is:

- sum(h) = n/2

The Euclidean projection onto this set can be done in O(n) using bisection on a threshold:
1. Find λ such that sum(clip(h - λ, 0, 1)) = n/2
2. Projected h = clip(h - λ, 0, 1)

This is a 1D root-finding problem. We can do it with bisection in O(log(1/ε) * n) time.

```python
def project_fast(v, n):
    """Project v onto {h : 0<=h<=1, sum(h)=n/2} in O(n log(1/eps))."""
    target = n / 2.0
    # Binary search for lambda
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):  # ~2^-60 precision
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
```

This is much faster: O(n * 60) per projection vs O(n * 200) before, and with vectorized numpy it's very fast.

Let me also think about whether the projected gradient method will work well here. The objective is non-convex (max of bilinear forms), so projected gradient might get stuck in local optima. But with the log-sum-exp smoothing and annealing, we effectively solve a sequence of problems:
- For small α: minimize a smooth approximation (close to the mean of all overlaps)
- For large α: minimize a sharper approximation (close to the max overlap)

This is similar to homotopy methods and should work well.

Now, let me also think about the **2-swap local search** for binary solutions. A 2-swap means: given positions i,j (currently 1) and k,l (currently 0), we consider swapping: set i,k to 0 and j,l to 1 (or some other combination). Actually, a 2-swap is: swap the values at (i,k) with (j,l), i.e., h[i]↔h[j] and h[k]↔h[l].

The delta in the correlation for a 2-swap can be computed, but it's more complex than for a single swap. Let me skip this and focus on the non-binary optimization.

1. Start from current h
2. Propose a random perturbation: pick 2-3 random positions, add random noise

    h1 = 1.0 - h
    H1 = np.fft.fft(h1, N)
    H = np.fft.fft(h, N)
    W = np.fft.fft(w, N)
    t1 = np.fft.ifft(H1 * np.conj(W)).real
    t2 = np.fft.ifft(H * W).real
    return dx * (t1[n-1::-1] - t2[n-1:2*n-1])

def pgd_anneal(h, n, deadline, alpha_start=50.0, alpha_end=1e7, lr_init=0.5):
    h = project_fast(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    lr = lr_init
    n_steps = int((deadline - time.time()) * 500)  # ~500 steps per second
    n_steps = max(1000, min(n_steps, 50000))
    for step in range(n_steps):
        if time.time() > deadline:
            break
        frac = step / max(1, n_steps - 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
        m = c.max()
        a = alpha * (c - m)
        a_max = a.max()
        w = np.exp(a - a_max)
        w /= w.sum()
        grad = smooth_grad_fft(h, n, w)
        h_new = project_fast(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if c_new <= cur_c + 1e-10:
            h = h_new
            cur_c = c_new
        else:
            lr *= 0.5
            if lr < 1e-6:
                lr = lr_init * 0.1
        if step % 2000 == 0:
            lr = lr_init * (1.0 - 0.8 * frac)
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        a = alpha * (c - m)
        a_max = a.max()
        return m + (a_max + np.log(np.exp(a - a_max).sum())) / alpha
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_fast(res.x, n)

def binary_swap_search(h, n, deadline, seed=42, max_sweeps=2000):
    h = binarize(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                # Compute delta for swap (i: 0->1, j: 1->0)
                # delta[k] = change in corr[k]
                # corr[k] = sum_pp h[pp] * (1-h[pp+k-(n-1)])
                # After swap: h[i]=1, h[j]=0
                # delta[k] = (1-h[i+k-(n-1)]) - h[j+k-(n-1)] - (1-h[j+k-(n-1)]) + h[i+k-(n-1)]
                #         = 2*h[i+k-(n-1)] - 2*h[j+k-(n-1)] ... 
                # Wait, let me be more careful.
                # Before: corr[k] includes h[i]*(1-h[i+m_k]) + h[j]*(1-h[j+m_k]) where m_k = k-(n-1)
                #   = 0*(1-h[i+m_k]) + 1*(1-h[j+m_k]) = (1-h[j+m_k])
                # After: h[i]=1, h[j]=0
                #   = 1*(1-h[i+m_k]) + 0*(1-h[j+m_k]) = (1-h[i+m_k])
                # delta[k] = (1-h[i+m_k]) - (1-h[j+m_k]) = h[j+m_k] - h[i+m_k]
                # Also, the (1-h[...]) terms change:
                # For position i: was 1-h[i+m_k], now h[i]=1 so 1-h[i+m_k] stays (h[i+m_k] unchanged)
                # Wait, I need to be more careful. h changes at positions i and j.
                # corr[k] = sum_pp h[pp] * (1-h[pp+m_k])
                # The terms involving h[i] or h[j] (either as the first or second factor):
                # As first factor at pp=i: h[i]*(1-h[i+m_k]), changes from 0 to (1-h[i+m_k])
                # As first factor at pp=j: h[j]*(1-h[j+m_k]), changes from (1-h[j+m_k]) to 0
                # As second factor: h[pp]*(1-h[pp+m_k]) where pp+m_k = i, i.e., pp = i-m_k
                #   h[i-m_k]*(1-h[i]), changes from h[i-m_k]*1 to h[i-m_k]*0 = -h[i-m_k]
                # As second factor: pp+m_k = j, i.e., pp = j-m_k
                #   h[j-m_k]*(1-h[j]), changes from h[j-m_k]*0 to h[j-m_k]*1 = +h[j-m_k]
                # So: delta[k] = (1-h[i+m_k]) - (1-h[j+m_k]) - h[i-m_k] + h[j-m_k]
                #            = h[j+m_k] - h[i+m_k] - h[i-m_k] + h[j-m_k]
                # where each term is only included if the index is valid [0, n-1]
                
                m_k = np.arange(2*n-1) - (n-1)  # m values for each k
                # For each k (index in corr), m = k-(n-1)
                # delta[k] = h[j+m] - h[i+m] - h[i-m] + h[j-m] (valid indices only)
                
                # Vectorized:
                delta = np.zeros(2*n-1)
                mv = m_k  # shape (2n-1,)
                # h[j + mv] where 0 <= j+mv < n
                mask1 = (j + mv >= 0) & (j + mv < n)
                delta[mask1] += h[j + mv[mask1]]
                # -h[i + mv] where 0 <= i+mv < n
                mask2 = (i + mv >= 0) & (i + mv < n)
                delta[mask2] -= h[i + mv[mask2]]
                # -h[i - mv] where 0 <= i-mv < n
                mask3 = (i - mv >= 0) & (i - mv < n)
                delta[mask3] -= h[i - mv[mask3]]
                # +h[j - mv] where 0 <= j-mv < n
                mask4 = (j - mv >= 0) & (j - mv < n)
                delta[mask4] += h[j - mv[mask4]]
                
                nc = float((corr + delta).max()) * dx
                if nc < cur_c - 1e-12:
                    h[i], h[j] = 1.0, 0.0
                    corr = corr + delta
                    cur_c = nc
                    if nc < best_c:
                        best_c = nc
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c
