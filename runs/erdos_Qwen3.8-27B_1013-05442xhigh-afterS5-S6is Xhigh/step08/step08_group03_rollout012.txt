import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def cd_1d_solve(c0, A, D, n):
    """Solve min_t max_k(c0[k]+t*A[k]-t^2*D[k]) for t in [0,1]."""
    grid_pts = 50
    grid = np.linspace(0.0, 1.0, grid_pts)
    fvals = np.array([float(np.max(c0 + t * A - t * t * D)) for t in grid])
    best_idx = int(np.argmin(fvals))
    lo = grid[max(0, best_idx - 1)]
    hi = grid[min(grid_pts - 1, best_idx + 1)]
    f0 = float(np.max(c0))
    f1 = float(np.max(c0 + A - D))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    if hi - lo > 1e-13:
        c1 = b - gr * (b - a)
        d1 = a + gr * (b - a)
        fc1 = float(np.max(c0 + c1 * A - c1 * c1 * D))
        fd1 = float(np.max(c0 + d1 * A - d1 * d1 * D))
        for _ in range(50):
            if fc1 < fd1:
                b = d1; d1 = c1; fd1 = fc1
                c1 = b - gr * (b - a)
                fc1 = float(np.max(c0 + c1 * A - c1 * c1 * D))
            else:
                a = c1; c1 = d1; fc1 = fd1
                d1 = a + gr * (b - a)
                fd1 = float(np.max(c0 + d1 * A - d1 * d1 * D))

def cd_pass(h, n, deadline=None, smart_order=False):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_max = float(c.max())
    order = list(range(n))

        for k in range(2 * n - 1):
            a_ks = ind + (n - 1) - ind  # this is just n-1, wrong approach
        # Skip smart ordering for simplicity, use natural order
        pass
    improved = False
    for p in order:
        if deadline is not None and time.time() > deadline:
            break
        h0 = h[p]
        if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
            continue
        # Compute A[k]: coefficient of t in c_k change
        A = np.zeros(2 * n - 1)
        j_arr = np.arange(n)
        a_ks = p + (n - 1) - j_arr
        a_valid = (a_ks >= 0) & (a_ks < 2 * n - 1)

            A[a_ks[a_valid]] += (1.0 - h[j_arr[a_valid]])
        i_arr = np.arange(n)
        b_ks = (n - 1) + i_arr - p
        b_valid = (b_ks >= 0) & (b_ks < 2 * n - 1)
        if np.any(b_valid):

        D = np.zeros(2 * n - 1)

        c0 = c - h0 * A + h0 * h0 * D

                c += (t_new - h0) * A - (t_new * t_new - h0 * h0) * D
                h[p] = t_new
                cur_max = f_new
                improved = True
    if improved:
        h = project_feasible(h, n)
    return h, improved

def exact_cd(h, n, max_passes=200, deadline=None):
    h = h.copy()
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        h, improved = cd_pass(h, n, deadline)
        if not improved:
            break
    return h, compute_c5(h, n)

def slsqp_opt(h, n, p, maxiter=400, deadline=None):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return float(np.sum(c ** p)) ** (1.0 / p) * dx
    def sum_con(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)

    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
