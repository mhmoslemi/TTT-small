def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=500):

    best_c = compute_c5(h, n)
    for alpha in alphas:

        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,

    """Find exact minimum of max of linear + one quadratic term."""
    # Coarse grid to identify active terms
    grid = np.linspace(lo, hi, 50)
    best_grid_val = float('inf')
    best_grid_d = lo
    best_grid_max_terms = None
    for d in grid:
        nv = c_aff + d * a_coeff
        if diag_local is not None:
            nv[diag_local] -= d * d
        val = max(float(nv.max()), unaff_max)
        if val < best_grid_val:
            best_grid_val = val
            best_grid_d = d
            # Identify terms within 0.005 of max
            if val > unaff_max:
                thresh = val - 0.005
                best_grid_max_terms = np.where(nv > thresh)[0]
    
    if best_grid_max_terms is None or len(best_grid_max_terms) == 0:
        return best_grid_d, best_grid_val
    
    # Get candidate indices (top terms)
    candidates = set(best_grid_max_terms.tolist())
    if diag_local is not None:
        candidates.add(diag_local)
    
    # Compute break points among candidates
    break_points = [lo, hi]
    
    # Add crossovers between linear terms
    cand_list = list(candidates)
    for i in range(len(cand_list)):
        for j in range(i+1, len(cand_list)):
            ci, cj = cand_list[i], cand_list[j]
            di, dj = diag_local == ci, diag_local == cj
            if di and dj:
                continue  # both quadratic, unlikely
            elif di:
                # quadratic vs linear: d^2 term
                # c_aff[ci] + d*a[ci] - d^2 = c_aff[cj] + d*a[cj]
                # -d^2 + d*(a[ci]-a[cj]) + (c_aff[ci]-c_aff[cj]) = 0
                # d^2 - d*(a[ci]-a[cj]) - (c_aff[ci]-c_aff[cj]) = 0
                A = 1.0
                B = -(a_coeff[ci] - a_coeff[cj])
                C = -(c_aff[ci] - c_aff[cj])
                disc = B*B - 4*A*C
                if disc >= 0:
                    d1 = (-B + np.sqrt(disc)) / (2*A)
                    d2 = (-B - np.sqrt(disc)) / (2*A)
                    if lo <= d1 <= hi: break_points.append(d1)
                    if lo <= d2 <= hi: break_points.append(d2)
            elif dj:
                # linear vs quadratic (symmetric)
                A = 1.0
                B = -(a_coeff[cj] - a_coeff[ci])
                C = -(c_aff[cj] - c_aff[ci])
                disc = B*B - 4*A*C
                if disc >= 0:
                    d1 = (-B + np.sqrt(disc)) / (2*A)
                    d2 = (-B - np.sqrt(disc)) / (2*A)
                    if lo <= d1 <= hi: break_points.append(d1)
                    if lo <= d2 <= hi: break_points.append(d2)
            else:

                if abs(da) > 1e-15:
                    d_pt = (c_aff[ci] - c_aff[cj]) / da
                    if lo <= d_pt <= hi:

    # Also add the vertex of the quadratic (if in range)
    if diag_local is not None:
        d_vertex = a_coeff[diag_local] / 2.0
        if lo <= d_vertex <= hi:
            break_points.append(d_vertex)
    
    # Evaluate at all break points
    min_val = float('inf')
    min_d = lo
    for d_pt in break_points:

        nv = c_aff + d_pt * a_coeff

        val = max(float(nv.max()), unaff_max)
        if val < min_val:
            min_val = val
            min_d = d_pt
    
    return min_d, min_val

    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))

from scipy.optimize import minimize, golden

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0
