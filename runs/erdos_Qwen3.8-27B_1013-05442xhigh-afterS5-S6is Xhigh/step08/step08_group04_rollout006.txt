import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = float(cap.sum())
        h += (diff / tot) * cap if tot > 0 else 0.0
    else:
        tot = float(h.sum())
        h += (diff / tot) * h if tot > 0 else 0.0
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if res > 1e-12:
        idx = np.where(h < 1.0 - 1e-14)[0]
        if idx.size:
            h[idx[0]] = min(1.0, h[idx[0]] + res)
    elif res < -1e-12:
        idx = np.where(h > 1e-14)[0]
        if idx.size:
            h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h

def compute_c(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def triangle_h(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(1.0 - np.abs(x - 1.0), n)

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old.copy(), n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    z = alpha * c
    zm = float(z.max())
    return ((zm + np.log(np.sum(np.exp(z - zm)))) / alpha) * (2.0 / n)

def smooth_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    center = n - 1
    z = alpha * c
    z = z - z.max()
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm <= 1e-14:
            continue
        s = center - m
        if s == 0:
            g += wm * (1.0 - 2.0 * x)
        elif s > 0:
            g[:n - s] += wm * (1.0 - x[s:])
            g[s:] -= wm * x[:n - s]
        else:
            r = -s
            g[r:] += wm * (1.0 - x[:n - r])
            g[:n - r] -= wm * x[r:]
    return (2.0 / n) * g

def sum_con(x, n):
    return float(x.sum()) - 0.5 * n

def sum_jac(x, n):
    return np.ones(n)

def slsqp_step(h, n, alphas, maxiter, deadline):
    best_h, best_c = project_feasible(h, n), compute_c(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_jac, bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
            x = project_feasible(res.x, n)
            c = compute_c(x, n)
            if c < best_c:
                best_h, best_c = x, c
        except Exception:
            pass
    return best_h, best_c

def coord_eval(x, hp, cseg, a, qconst, unaff):
    if a.size:
        vmax = float(np.max(cseg + a * (x - hp)))
    else:
        vmax = 0.0
    val = max(unaff, qconst + x * (1.0 - x))
    return max(val, vmax)

def optimize_coordinate(hp, cseg, a, qconst, unaff, iters=15):
    m = a.size
    b = cseg - a * hp
    pos = a > 1e-12
    neg = a < -1e-12
    zero = ~(pos | neg)
    zmax = float(np.max(b[zero])) if zero.any() else -1e100
    low = max(0.0, unaff, qconst)
    if zmax > low:
        low = zmax
    high = max(low + 1e-11, unaff, qconst, qconst + hp*(1.0-hp))
    if zmax > high:
        high = zmax
    if m:
        high = max(high, float(np.max(cseg)))
    high += 1e-8
    a_pos, a_neg = a[pos], a[neg]
    b_pos, b_neg = b[pos], b[neg]
    pc, nc = a_pos.size, a_neg.size
    for _ in range(iters):
        mid = 0.5 * (low + high)
        if mid >= unaff - 1e-12 and mid >= qconst - 1e-12 and mid >= zmax - 1e-12:
            ll, lh = 0.0, 1.0
            if nc:
                v = float(np.max((mid - b_neg) / a_neg))
                if v > ll:
                    ll = v
            if pc:
                v = float(np.min((mid - b_pos) / a_pos))
                if v < lh:
                    lh = v
            if ll <= 1.0 + 1e-12 and lh >= -1e-12:
                g = mid - qconst
                if g > 0.25:
                    feas = max(0.0, ll) <= min(1.0, lh) + 1e-12
                elif g > -1e-12:
                    if g < 0:
                        g = 0.0
                    r1 = 0.5 - 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                    r2 = 0.5 + 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                    feas = (max(0.0, ll) <= min(r1, lh) + 1e-12 or
                            max(r2, ll) <= min(1.0, lh) + 1e-12)
                else:
                    feas = False
                if feas:
                    high = mid
                    continue
        low = mid
    t = high
    cands = [0.0, 1.0, hp, 0.5]
    if t >= unaff - 1e-12 and t >= qconst - 1e-12 and t >= zmax - 1e-12:
        ll, lh = 0.0, 1.0
        if nc:
            v = float(np.max((t - b_neg) / a_neg))
            if v > ll:
                ll = v
        if pc:
            v = float(np.min((t - b_pos) / a_pos))
            if v < lh:
                lh = v
        if ll <= 1.0 + 1e-12 and lh >= -1e-12:
            g = t - qconst
            if g > 0.25:
                branches = [(max(0.0, ll), min(1.0, lh))]
            elif g >= -1e-12:
                if g < 0:
                    g = 0.0
                r1 = 0.5 - 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                r2 = 0.5 + 0.5 * np.sqrt(max(0.0, 1.0 - 4.0 * g))
                branches = [(0.0, r1), (r2, 1.0)]
            else:
                branches = []
            for Lq, Uq in branches:
                L = max(Lq, ll, 0.0)
                U = min(Uq, lh, 1.0)
                if L <= U + 1e-12:
                    cands.extend([L, U, 0.5 * (L + U)])
    best_x, best_f = hp, coord_eval(hp, hp, cseg, a, qconst, unaff)
    for x in cands:
        if 0.0 <= x <= 1.0:
            f = coord_eval(x, hp, cseg, a, qconst, unaff)
            if f < best_f - 1e-13:
                best_f, best_x = f, x
    if best_f > high + 1e-5 and m:
        xg = np.linspace(0.0, 1.0, 40)
        d = xg - hp
        vals = np.maximum(np.maximum(cseg[None, :] + d[:, None] * a[None, :],
                                      qconst + xg * (1.0 - xg)), unaff)
        idx = int(np.argmin(vals))
        if vals[idx] < best_f:
            best_x, best_f = float(xg[idx]), float(vals[idx])
            l, r = max(0.0, best_x - 0.1), min(1.0, best_x + 0.1)
            if r > l:
                xg2 = np.linspace(l, r, 30)
                d2 = xg2 - hp
                vals2 = np.maximum(np.maximum(cseg[None, :] + d2[:, None] * a[None, :],
                                               qconst + xg2 * (1.0 - xg2)), unaff)
                idx2 = int(np.argmin(vals2))
                if vals2[idx2] < best_f:
                    best_x, best_f = float(xg2[idx2]), float(vals2[idx2])
    return best_x, best_f

def cd_optimize(h, n, max_passes, deadline, project_every=8, coord_iters=15):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cur = float(c.max())
    best_h, best_c = h.copy(), (2.0 / n) * cur
    center = n - 1
    L = 2 * n - 1
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            hp = float(h[p])
            lo = min(p, center - p)
            hi = max(center + p, 2 * center - p)
            s = center - np.arange(lo, hi + 1, dtype=int)
            j = p + s
            i = p - s
            a = np.zeros(hi - lo + 1, dtype=float)
            vj = (j >= 0) & (j < n)
            vi = (i >= 0) & (i < n)
            if np.any(vj):
                a[vj] += 1.0 - h[j[vj]]
            if np.any(vi):
                a[vi] -= h[i[vi]]
            off = center - lo
            seg = hi - lo + 1
            cseg = np.empty(seg - 1, dtype=float)
            asg = np.empty(seg - 1, dtype=float)
            if off:
                cseg[:off] = c[lo:center]
                asg[:off] = a[:off]
            if hi > center:
                cseg[off:] = c[center + 1:hi + 1]
                asg[off:] = a[off + 1:]
            qconst = c[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0:
                unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < L:
                unaff = max(unaff, float(c[hi + 1:].max()))
            bx, bf = optimize_coordinate(hp, cseg, asg, qconst, unaff, coord_iters)
            if bf < cur - 1e-11:
                dxv = bx - hp
                if off:
                    c[lo:center] += asg[:off] * dxv
                if hi > center:
                    c[center + 1:hi + 1] += asg[off:] * dxv
                c[center] = qconst + bx * (1.0 - bx)
                h[p] = bx
                cur = bf
                improved = True
            if (p + 1) % project_every == 0:
                h = project_feasible(h, n)
                c = np.correlate(h, 1.0 - h, mode='full')
                cur = float(c.max())
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        cval = (2.0 / n) * cur
        if cval < best_c - 1e-14:
            best_c = cval
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def binary_from_h(h):
    n = len(h)
    k = n // 2
    idx = np.argpartition(h, n - k)[n - k:]
    h2 = np.zeros(n)
    h2[idx] = 1.0
    return h2
