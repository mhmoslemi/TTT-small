import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + diff * room / tot, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + diff * h / tot, 0.0, 1.0)
    return h

def smooth_max_obj(x, n, alpha, penalty):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = (m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha) * (2.0 / n)
    s = x.sum() - n / 2.0
    return lse + penalty * s * s

def lbfgsb_run(h, n, alphas, penalties, maxiter, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        for pen in penalties:
            if time.time() > deadline:
                return best_h, best_c
            try:
                res = minimize(smooth_max_obj, best_h, args=(n, alpha, pen),
                             method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
            except Exception:
                pass
    return best_h, best_c
