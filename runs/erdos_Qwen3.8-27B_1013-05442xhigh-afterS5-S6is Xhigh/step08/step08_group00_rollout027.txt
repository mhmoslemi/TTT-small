def lbfgsb_smooth(h, n, eta_schedule, deadline, maxiter_per_eta=300):
    """Run L-BFGS-B with different smoothness parameters."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for eta in eta_schedule:
        if time.time() > deadline - 10:
            break
        bounds = [(0.0, 1.0)] * n
        result = minimize(
            lambda x, e=eta: smooth_obj_and_grad(project_feasible(x, n), n, e)[0],
            best_h,
            jac=lambda x, e=eta: smooth_obj_and_grad(project_feasible(x, n), n, e)[1],
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': maxiter_per_eta, 'ftol': 1e-18, 'gtol': 1e-14}
        )
        new_h = project_feasible(result.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_h, best_c = new_h, new_c
    return best_h, best_c

import numpy as np
from scipy.special import softmax
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def anti_periodic_cd(g, m, max_passes, deadline):
    n = 2 * m
    g = g.copy()
    h = np.concatenate([g, 1.0 - g])
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        h = np.concatenate([g, 1.0 - g])
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(m):
            h0 = g[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_lo = min(max(0, (n-1)-p), max(0, p-(n-1)), max(0, (n-1)-p+m), max(0, p+m-(n-1)))
            m_hi = max(min(2*n-2, 2*(n-1)-p), min(2*n-2, p), min(2*n-2, 2*(n-1)-p+m), min(2*n-2, p+m))
            ms = np.arange(m_lo, m_hi + 1)
            c_aff = c[ms]
            d1 = h.copy()
            d1[p] = 1.0; d1[p+m] = 0.0
            coef = np.correlate(h, 1.0 - d1, mode='full')[ms] + np.correlate(d1, 1.0 - h, mode='full')[ms]
            unaff_max = 0.0

                x_hi = min(1.0, float(np.min((L - c_aff[pos]) / coef[pos]))) if np.any(pos) else 1.0
                ok = x_lo <= x_hi + 1e-15
                if np.any(np.abs(coef) <= 1e-14) and np.any(c_aff[np.abs(coef) <= 1e-14] > L):
                    ok = False

                    L_lo = L

    h = np.concatenate([g, 1.0 - g])
    return g, compute_c5(h, n)

def lbfgsb_smooth(h, n, etas, deadline, maxiter=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for eta in etas:
        if time.time() > deadline:
            break
        def obj_fn(x, e=eta):
            c = np.correlate(x, 1.0 - x, mode='full')
            dx = 2.0 / n
            return float(np.log(np.sum(np.exp(e * c))) / e * dx)
        def grad_fn(x, e=eta):
            c = np.correlate(x, 1.0 - x, mode='full')
            w = softmax(e * c)
            dx_val = 2.0 / n
            grad = np.zeros(n)
            for k in range(2 * n - 1):
                wk = w[k]
                if wk < 1e-15:
                    continue
                p_lo = max(0, (n-1)-k)
                p_hi = min(n-1, 2*(n-1)-k)
                if p_lo <= p_hi:
                    idx = np.arange(p_lo, p_hi+1) + k - (n-1)
                    grad[p_lo:p_hi+1] += wk * (1.0 - x[idx])
                p_lo2 = max(0, k-(n-1))
                p_hi2 = min(n-1, k+(n-1))
                if p_lo2 <= p_hi2:
                    idx2 = np.arange(p_lo2, p_hi2+1) - k + (n-1)
                    grad[p_lo2:p_hi2+1] -= wk * x[idx2]
            return grad * dx_val
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj_fn, best_h, jac=grad_fn, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        except Exception:
            continue
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_h, best_c = new_h, new_c
    return best_h, best_c

def full_cd(h, n, max_passes, deadline):
    g = h[:n//2].copy()
    m = n // 2
    g, c_val = anti_periodic_cd(g, m, max_passes, deadline)
    h_new = np.concatenate([g, 1.0 - g])
    return h_new, c_val
