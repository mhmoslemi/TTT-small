    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)

        if s > target:

        else:

    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)

    mask = y < 1.0 if res > 0 else y > 0.0

    e = np.exp(z - z.max())
    s = float(e.sum())
    return (m + np.log(s) / alpha) * (2.0 / n)

    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(c.max())
    # Build constraint matrix for active lags
    # Only use lags with c(k) >= cmax - 0.05*cmax to keep LP small
    tol = 0.05 * cmax
    off = n - 1
    active = np.flatnonzero(c >= cmax - tol)
    if active.size == 0:
        active = np.array([off])
    
    # Build A_ub and b_ub for the LP
    # Constraint: grad_k · h - t <= grad_k · h0 - c(k)
    # i.e., grad_k · h + (-1)*t <= grad_k · h0 - c(k)
    A_rows = []
    b_vals = []
    for k in active:
        d = k - off
        g = np.zeros(n, dtype=float)
        if d >= 0:
            m_ = n - d
            if m_ > 0:
                g[:m_] = 1.0 - h[d:d+m_]
            if d > 0:
                g[d:] -= h[:m_]
        else:
            e_ = -d
            m_ = n - e_
            if m_ > 0:
                g[e_:] = 1.0 - h[:m_]
                g[:m_] -= h[e_:]
        # Append -1 for t variable
        row = np.append(g, -1.0)
        rhs = float(g @ h) - c[k]
        A_rows.append(row)
        b_vals.append(rhs)
    
    A_ub = np.array(A_rows, dtype=float)
    b_ub = np.array(b_vals, dtype=float)
    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.5 * n])
    c_obj = np.zeros(n + 1, dtype=float)
    c_obj[n] = 1.0  # minimize t
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

                break
    return h, c

def slsqp_stage(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_val, h, args=(n, alpha), method='SLSQP',
                       jac=smooth_grad, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        x = project_feasible(np.asarray(res.x, dtype=float), n)
        return x, compute_c5(x, n)
    except Exception:
        return h, compute_c5(h, n)

def lbfgs_stage(h, n, alpha, mu, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(pen_val, h, args=(n, alpha, mu), method='L-BFGS-B',
                       jac=pen_grad, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 30, 'disp': False})
        x = project_feasible(np.asarray(res.x, dtype=float), n)
        return x, compute_c5(x, n)
    except Exception:
        return h, compute_c5(h, n)

def pair_descent(h, n, max_iters, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    off = n - 1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        cmax = float(c.max())
        tol = max(0.01, 0.003 * cmax)
        active = np.flatnonzero(c >= cmax - tol)
        if active.size == 0:
            active = np.array([off])
        k = int(active[int(rng.randint(active.size))])
        d = k - off
        if d >= 0:
            m_ = n - d
            if m_ <= 0:
                continue
            i = int(rng.randint(m_))
            j = i + d
        else:
            e_ = -d
            m_ = n - e_
            if m_ <= 0:
                continue
            j = int(rng.randint(m_))
            i = j + e_
        s = h[i] + h[j]
        lo = max(0.0, s - 1.0)
        hi = min(1.0, s)
        if hi - lo < 1e-10:
            continue
        best = cmax
        best_v = h[i]
        # Try several values
        for t in np.linspace(lo + 0.01 * (hi - lo), hi - 0.01 * (hi - lo), 10):
            h[i] = t
            h[j] = s - t
            cn = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
            if cn < best:
                best = cn
                best_v = t
        h[i] = best_v
        h[j] = s - best_v
        c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def symmetrize(h, n):
    c = compute_c5(h, n)
    best_h, best_c = h, c
    for transform in ['rev', 'comp', 'rev_comp']:
        if transform == 'rev':
            t = project_feasible(0.5 * (h + h[::-1]), n)
        elif transform == 'comp':
            t = 1.0 - h
        else:
            t = project_feasible(0.5 * (h + 1.0 - h[::-1]), n)
        ct = compute_c5(t, n)
        if ct < best_c:
            best_h, best_c = t, ct
    return best_h, best_c

def optimize_full(h, n, deadline, rng):
    """Full optimization pipeline for one resolution."""
    h, c = symmetrize(h, n)
    # L-BFGS-B with penalty
    if time.time() < deadline - 20:
        h, c = lbfgs_stage(h, n, 50, 1e4, 50, deadline)
    # SLSQP stages
    for alpha, mx in [(100, 30), (500, 30), (2000, 25), (10000, 20)]:
        if time.time() > deadline - 10:
            break
        h2, c2 = slsqp_stage(h, n, alpha, mx, deadline)
        if c2 < c:
            h, c = h2, c2
    # LP phase
    if time.time() < deadline - 10:
        h, c = lp_phase(h, n, 30, deadline)
    # More SLSQP after LP
    if time.time() < deadline - 10:
        h2, c2 = slsqp_stage(h, n, 5000, 30, deadline)
        if c2 < c:
            h, c = h2, c2
    # Pair descent
    if time.time() < deadline - 3:
        iters = int(min(800, max(100, (deadline - time.time()) * 8)))
        h, c = pair_descent(h, n, iters, deadline, rng)
    return project_feasible(h, n), compute_c5(h, n)

import numpy as np
import time
from scipy.optimize import minimize, linprog

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    z = np.asarray(z, dtype=float)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.size == 0:
        return z
    if z.min() >= -1e-12 and z.max() <= 1.0 + 1e-12 and abs(z.sum() - target) <= 1e-9 * max(1, n):
        return np.clip(z, 0.0, 1.0)
    low = float(z.min()) - 2.0
    high = float(z.max()) + 2.0
    for _ in range(55):
        tau = 0.5 * (low + high)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(z - 0.5 * (low + high), 0.0, 1.0)
    res = target - y.sum()
    mask = y < 1.0 if res > 0 else y > 0.0
    if mask.any() and abs(res) > 1e-13:
        room = (1.0 - y) if res > 0 else y
        tot = room.sum()
        if tot > 1e-12:
            y[mask] += (res / tot) * room[mask]
    return np.clip(y, 0.0, 1.0)

def smooth_val(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    e = np.exp(z - z.max())
    s = float(e.sum())
    return (m + np.log(s) / alpha) * (2.0 / n)

def smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    e = np.exp(z - z.max())
    p = e / float(e.sum())
    grad = np.zeros(n, dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-13:
            continue
        d = k - off
        if d >= 0:
            m_ = n - d
            if m_ > 0:
                grad[:m_] += w * (1.0 - x[d:d + m_])
            if d > 0:
                grad[d:] -= w * x[:m_]
        else:
            e_ = -d
            m_ = n - e_
            if m_ > 0:
                grad[e_:] += w * (1.0 - x[:m_])
                grad[:m_] -= w * x[e_:]
    return (2.0 / n) * grad

def pen_val(x, n, alpha, mu):
    v = smooth_val(x, n, alpha)
    dev = float(x.sum() - 0.5 * n)
    return v + 0.5 * mu * dev * dev

def pen_grad(x, n, alpha, mu):
    g = smooth_grad(x, n, alpha)
    dev = float(x.sum() - 0.5 * n)
    g += mu * dev
    return g

def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac_fn(x, n):
    return np.ones_like(x, dtype=float)

def do_lp_step(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(c.max())
    off = n - 1
    tol = 0.08 * cmax
    active = np.flatnonzero(c >= cmax - tol)
    if active.size == 0:
        active = np.array([off])
    A_rows = []
    b_vals = []
    for k in active:
        d = k - off
        g = np.zeros(n, dtype=float)
        if d >= 0:
            m_ = n - d
            if m_ > 0:
                g[:m_] = 1.0 - h[d:d + m_]
            if d > 0:
                g[d:] -= h[:m_]
        else:
            e_ = -d
            m_ = n - e_
            if m_ > 0:
                g[e_:] = 1.0 - h[:m_]
                g[:m_] -= h[e_:]
        A_rows.append(np.append(g, -1.0))
        b_vals.append(float(g @ h) - c[k])
    A_ub = np.array(A_rows, dtype=float)
    b_ub = np.array(b_vals, dtype=float)
    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.5 * n])
    c_obj = np.zeros(n + 1, dtype=float)
    c_obj[n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return project_feasible(np.clip(res.x[:n], 0.0, 1.0), n)
    except Exception:
        pass
    return h.copy()

def lp_phase(h, n, max_iters, deadline):
    c = compute_c5(h, n)
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        h_new = do_lp_step(h, n)
        c_new = compute_c5(h_new, n)
        if c_new < c - 1e-12:
            h, c = h_new, c_new
        else:
            h_half = project_feasible(0.5 * (h + h_new), n)
            c_half = compute_c5(h_half, n)
            if c_half < c - 1e-12:
                h, c = h_half, c_half
            else:
                break
    return h, c

def slsqp_stage(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac_fn, 'args': (n,)}]
    try:
        res = minimize(smooth_val, h, args=(n, alpha), method='SLSQP',
                       jac=smooth_grad, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False})
        x = project_feasible(np.asarray(res.x, dtype=float), n)
        return x, compute_c5(x, n)
    except Exception:
        return h, compute_c5(h, n)

def lbfgs_stage(h, n, alpha, mu, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(pen_val, h, args=(n, alpha, mu), method='L-BFGS-B',
                       jac=pen_grad, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 30, 'disp': False})
        x = project_feasible(np.asarray(res.x, dtype=float), n)
        return x, compute_c5(x, n)
    except Exception:
        return h, compute_c5(h, n)

def pair_descent(h, n, max_iters, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    off = n - 1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        cmax = float(c.max())
        tol = max(0.005, 0.003 * cmax)
        active = np.flatnonzero(c >= cmax - tol)
        if active.size == 0:
            active = np.array([off])
        k = int(active[int(rng.randint(active.size))])
        d = k - off
        if d >= 0:
            m_ = n - d
            if m_ <= 0:
                continue
            i = int(rng.randint(m_))
            j = i + d
        else:
            e_ = -d
            m_ = n - e_
            if m_ <= 0:
                continue
            j = int(rng.randint(m_))
            i = j + e_
        s_val = h[i] + h[j]
        lo = max(0.0, s_val - 1.0)
        hi = min(1.0, s_val)
        if hi - lo < 1e-10:
            continue
        best = cmax
        best_v = h[i]
        for t in np.linspace(lo + 0.02 * (hi - lo), hi - 0.02 * (hi - lo), 8):
            h[i] = t
            h[j] = s_val - t
            cn = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
            if cn < best:
                best = cn
                best_v = t
        h[i] = best_v
        h[j] = s_val - best_v
        c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def symmetrize(h, n):
    c = compute_c5(h, n)
    best_h, best_c = h.copy(), c
    t = project_feasible(0.5 * (h + h[::-1]), n)
    ct = compute_c5(t, n)
    if ct < best_c:
        best_h, best_c = t, ct
    t2 = 1.0 - h
    ct2 = compute_c5(t2, n)
    if ct2 < best_c:
        best_h, best_c = t2.copy(), ct2
    t3 = project_feasible(0.5 * (best_h + 1.0 - best_h[::-1]), n)
    ct3 = compute_c5(t3, n)
    if ct3 < best_c:
        best_h, best_c = t3, ct3
    return best_h, best_c

def optimize_full(h, n, deadline, rng):
    h, c = symmetrize(h, n)
    if time.time() < deadline - 25:
        h2, c2 = lbfgs_stage(h, n, 50, 1e4, 50, deadline)
        if c2 < c:
            h, c = h2, c2
    for alpha, mx in [(100, 30), (500, 30), (2000, 25), (8000, 20)]:
        if time.time() > deadline - 15:
            break
        h2, c2 = slsqp_stage(h, n, alpha, mx, deadline)
        if c2 < c:
            h, c = h2, c2
    if time.time() < deadline - 10:
        h, c = lp_phase(h, n, 40, deadline)
    if time.time() < deadline - 8:
        h2, c2 = slsqp_stage(h, n, 5000, 30, deadline)
        if c2 < c:
            h, c = h2, c2
    if time.time() < deadline - 5:
        h, c = lp_phase(h, n, 20, deadline)
    if time.time() < deadline - 2:
        iters = int(min(600, max(80, (deadline - time.time()) * 10)))
        h, c = pair_descent(h, n, iters, deadline, rng)
    return project_feasible(h, n), compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.size < 2 or not np.all(np.isfinite(h0)):
            raise ValueError("bad")
    except Exception:
        n_init = 500
        x = np.linspace(0.0, 2.0, n_init, endpoint=False)
        h0 = np.full(n_init, 0.5)
        for m in range(1, 8):
            h0 += 0.12 * np.sin(np.pi * m * x + 0.3 * m)
        h0 = project_feasible(np.clip(h0, 0.0, 1.0), n_init)
    n0 = len(h0)
    best_h, best_c = symmetrize(h0, n0)
    main_dl = hard - max(150.0, 0.22 * budget_s)
    if main_dl < t0 + 30:
        main_dl = hard
    best_h, best_c = optimize_full(best_h, n0, main_dl, rng)
    cands = [300, 400, 500, 600, 700, 800]
    for m in cands:
        if time.time() > hard - 30:
            break
        if m == n0:
            continue
        rem = hard - time.time()
        dl = time.time() + min(55.0, 0.15 * rem)
        if dl < time.time() + 10:
            continue
        h_m = interp_h(best_h, m)
        h_m, c_m = optimize_full(h_m, m, dl, rng)
        if c_m < best_c:
            best_h, best_c = h_m, c_m
    if time.time() < hard - 50:
        n_cur = len(best_h)
        for s in [0.04, 0.08, 0.15]:
            if time.time() > hard - 25:
                break
            hp = project_feasible(best_h + rng.normal(0.0, s, n_cur), n_cur)
            dl = min(hard - 10, time.time() + 30)
            ht, ct = optimize_full(hp, n_cur, dl, rng)
            if ct < best_c:
                best_h, best_c = ht, ct
    if time.time() < hard - 8:
        n_cur = len(best_h)
        h, c = lp_phase(best_h, n_cur, 25, hard - 8)
        if c < best_c:
            best_h, best_c = h, c
        h2, c2 = slsqp_stage(best_h, n_cur, 15000, 30, hard - 4)
        if c2 < best_c:
            best_h, best_c = h2, c2
        if time.time() < hard - 2:
            h3, c3 = pair_descent(best_h, len(best_h), 400, hard - 1, rng)
            if c3 < best_c:
                best_h, best_c = h3, c3
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
