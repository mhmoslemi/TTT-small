    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi:
        return h0
    return (x_lo + x_hi) / 2.0


def gauss_seidel_cd(h, n, max_passes, deadline):
    """Gauss-Seidel coordinate descent with incremental correlation updates."""
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(c.max())
    
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        max_delta = 0.0
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            
            # Compute affected lags
            k_lo = n - 1 - p
            k_hi = n - 1 + p
            ks = np.arange(k_lo, k_hi + 1)
            j_a = p + ks - (n - 1)
            i_b = p - ks + (n - 1)
            valid_a = (j_a >= 0) & (j_a < n)
            valid_b = (i_b >= 0) & (i_b < n)
            
            a_k = np.zeros(len(ks))
            if np.any(valid_a):
                a_k[valid_a] += (1.0 - h[j_a[valid_a]])
            if np.any(valid_b):
                a_k[valid_b] -= h[i_b[valid_b]]
            
            c_k = c[ks].copy()
            
            # Find diagonal index
            diag_pos = np.where(ks == (n - 1))[0]
            diag_idx = diag_pos[0] if len(diag_pos) > 0 else None
            
            # Find unaff_max
            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(np.max(c[:k_lo])))
            if k_hi < 2*n - 2:
                unaff_max = max(unaff_max, float(np.max(c[k_hi+1:])))
            
            # We need to incorporate unaff_max into the optimization
            # The max we minimize is max(affected_c(x), unaff_max)
            # We can handle this by adding a "virtual" constraint: virtual_c = unaff_max
            # (constant, so it just raises the floor)
            
            # Simple approach: just use L_lo = max(0, unaff_max) as the lower bound
            # and the bisection will naturally account for it
            
            # Actually, let me just pass unaff_max and handle it in the bisection
            # by setting L_lo = unaff_max initially
            
            x_new = _find_best_x_with_floor(c_k, a_k, h0, unaff_max, diag_idx)
            
            if abs(x_new - h0) > 1e-11:
                # Update c incrementally
                delta = x_new - h0
                dc = np.zeros(len(ks))
                if np.any(valid_a):
                    dc[valid_a] += delta * (1.0 - h[j_a[valid_a]])
                if np.any(valid_b):
                    dc[valid_b] -= delta * h[i_b[valid_b]]
                if diag_idx is not None:
                    dc[diag_idx] -= delta * delta
                c[k_lo:k_hi+1] += dc
                h[p] = x_new
                max_delta = max(max_delta, abs(delta))
                improved = True
        
        if time.time() > deadline:
            break
        h = project_feasible(h, n)
        # After projection, c might be stale; recompute
        if improved and max_delta > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        if cur_max < best_c:
            best_c = cur_max
        if not improved or max_delta < 1e-12:
            break
    
    return h, compute_c5(h, n)
