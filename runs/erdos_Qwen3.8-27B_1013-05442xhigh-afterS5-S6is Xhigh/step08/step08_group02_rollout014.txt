def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(z, n):
    """Project z onto {x: 0<=x<=1, sum(x)=n/2}"""
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)
    target = 0.5 * n
    if z.min() >= -1e-12 and z.max() <= 1.0 + 1e-12 and abs(z.sum() - target) <= 1e-9:
        return np.clip(z, 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        tau = 0.5 * (lo + hi)
        s = np.clip(z - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    y = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    res = target - y.sum()
    if abs(res) > 1e-13:
        if res > 0:
            mask = y < 1.0
            room = 1.0 - y
        else:
            mask = y > 0.0
            room = y
        if room[mask].sum() > 1e-12:
            y[mask] += (res / room[mask].sum()) * room[mask]
    return np.clip(y, 0.0, 1.0)

def c5_smooth_val(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    z = alpha * (c - m)
    zm = z.max()
    e = np.exp(z - zm)
    return (m + np.log(e.sum()) / alpha) * (2.0 / n)

    z = alpha * (c - m)
    zm = z.max()

    p = e / e.sum()
    g = grad_c_k(h, n, p)
    dev = h.sum() - 0.5 * n
    g = g + (lam + mu * dev)
    return g

def optimize_al(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    lam = 0.0
    mu = 80.0
    alphas = [30, 80, 200, 500, 1000, 2500, 5000]
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        for _ in range(4):
            if time.time() > deadline - 3:
                break
            try:
                res = minimize(smooth_func, best_h, args=(n, alpha, lam, mu),
                             jac=smooth_grad, method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': 400, 'ftol': 1e-18, 'maxls': 50})
                x = np.asarray(res.x, dtype=float)
                if not np.all(np.isfinite(x)):
                    continue
                x = project_feasible(x, n)
                c = compute_c5(x, n)
                if c < best_c:
                    best_h, best_c = x, c
            except Exception:
                pass
            dev = best_h.sum() - 0.5 * n
            lam = float(np.clip(lam + mu * dev, -100.0, 100.0))
    return best_h, best_c

def subgrad_refine(h, n, n_iter, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    hc = best_h.copy()
    step = 0.2
    no_improve = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        c = np.correlate(hc, 1.0 - hc, mode='full') * (2.0 / n)
        cmax = c.max()
        tol = max(0.005, 0.006 * cmax)
        active = np.flatnonzero(c >= cmax - tol)
        p = np.zeros(len(c))
        p[active] = 1.0 / len(active)
        g = grad_c_k(hc, n, p)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        direction = -g / gnorm
        h_new = project_feasible(hc + step * direction, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_h, best_c = h_new, c_new
            hc = h_new
            step = min(step * 1.05, 0.5)
            no_improve = 0
        else:
            step *= 0.8
            no_improve += 1
            if no_improve > 80:
                hc = project_feasible(hc + rng.normal(0, 0.02, n), n)
                step = 0.2
                no_improve = 0
        if step < 1e-7:
            break
    return best_h, best_c

def interp_resample(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0, 2, n, endpoint=False)
    xn = np.linspace(0, 2, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def sym_candidate(h, n, mode):
    if mode == 0:
        return project_feasible(h.copy(), n)
    elif mode == 1:
        return project_feasible(h[::-1].copy(), n)
    elif mode == 2:
        return project_feasible(1.0 - h, n)
    elif mode == 3:
        return project_feasible(0.5 * (h + h[::-1]), n)
    else:
        return project_feasible(0.5 * (h + 1.0 - h[::-1]), n)

def sym_improve(h, n):
    best, bc = h.copy(), compute_c5(h, n)
    for mode in range(5):
        c = sym_candidate(h, n, mode)
        cv = compute_c5(c, n)
        if cv < bc:
            best, bc = c, cv
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.size < 2 or not np.all(np.isfinite(h0)):
            raise ValueError
    except Exception:
        n = 200
        x = np.linspace(0, 2, n, endpoint=False)
        h0 = np.clip(0.5 + 0.25 * np.sin(np.pi * x) + 0.1 * np.sin(2 * np.pi * x), 0, 1)
        h0 = project_feasible(h0, n)
    n0 = len(h0)
    best_h, best_c = sym_improve(h0, n0)
    # Phase 1: AL at original resolution
    if hard - t0 > 40:
        dl = t0 + min(0.35 * budget_s, hard - t0 - 80)
        best_h, best_c = optimize_al(best_h, n0, dl, rng)
    # Phase 2: Multi-res
    for m in [250, 400, 600, 800]:
        if m == n0 or time.time() > hard - 45:
            continue
        rem = hard - time.time()
        if rem < 25:
            break
        dl = time.time() + min(0.22 * rem, 55.0)
        hm = interp_resample(best_h, m)
        hm, cm = sym_improve(hm, m)
        hm, cm = optimize_al(hm, m, dl, rng)
        if cm < best_c:
            best_h, best_c = hm, cm
    # Phase 3: Subgrad at best resolution
    if time.time() < hard - 20:
        n = len(best_h)
        dl = min(hard - 8, time.time() + 0.35 * (hard - time.time()))
        best_h, best_c = subgrad_refine(best_h, n, 15000, dl, rng)
    # Phase 4: Final AL polish
    if time.time() < hard - 10:
        n = len(best_h)
        best_h, best_c = optimize_al(best_h, n, hard - 4, rng)
    # Phase 5: Random restarts
    if time.time() < hard - 18:
        n = len(best_h)
        for s in [0.04, 0.1]:
            if time.time() > hard - 10:
                break
            hp = project_feasible(best_h + rng.normal(0, s, n), n)
            dl = min(hard - 6, time.time() + 18)
            hp, cp = optimize_al(hp, n, dl, rng)
            if cp < best_c:
                best_h, best_c = hp, cp
                if time.time() < hard - 10:
                    best_h, best_c = subgrad_refine(best_h, n, 3000, hard - 6, rng)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
