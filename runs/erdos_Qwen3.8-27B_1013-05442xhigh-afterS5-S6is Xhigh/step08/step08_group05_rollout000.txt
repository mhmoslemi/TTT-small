import numpy as np
from scipy.optimize import minimize

def project_simplex(h, n):
    h = np.asarray(h, dtype=float).clip(0.0, 1.0)
    t = 0.5 * n
    d = t - h.sum()
    if abs(d) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        m = 0.5 * (lo + hi)
        if np.clip(h + m, 0.0, 1.0).sum() < t:
            lo = m
        else:
            hi = m
    y = np.clip(h + hi, 0.0, 1.0)
    d = t - y.sum()
    if abs(d) > 1e-12:
        mask = (y > 1e-9) & (y < 1.0 - 1e-9)
        if mask.any():
            y[mask] += d / mask.sum()
        else:
            idx = int(np.argmax(1.0 - y)) if d > 0 else int(np.argmax(y))
            y[idx] = np.clip(y[idx] + d, 0.0, 1.0)
    return np.clip(y, 0.0, 1.0)

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / n)

def softmax_vg(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    r = alpha * (c - m)
    r -= r.max()
    ee = np.exp(r)
    z = float(ee.sum())
    dx = 2.0 / n
    w = ee / z
    A = np.convolve(1.0 - x, w, mode="full")
    B = np.convolve(x, w[::-1], mode="full")
    s = n - 1
    return (m + np.log(z) / alpha) * dx, (A - B)[s:s + n] * dx

def smooth_obj(x, n, alpha):
    return softmax_vg(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    return softmax_vg(x, n, alpha)[1]

def sum_fun(x, n):
    return x.sum() - 0.5 * n

def sum_jac(x, n):
    return np.ones_like(x, dtype=float)

def adam_opt(h, n, alphas, iters, lr, deadline):
    h = project_simplex(h, n).copy()
    best = h.copy()
    bc = compute_c5(best, n)
    for a in alphas:
        if time.time() > deadline:
            break
        mh = np.zeros(n)
        vh = np.zeros(n)
        t = 0
        b1, b2, eps = 0.8, 0.999, 1e-8
        for _ in range(iters):
            if time.time() > deadline:
                break
            t += 1
            _, g = softmax_vg(h, n, a)
            g = g - g.mean()
            mh = b1 * mh + (1.0 - b1) * g
            vh = b2 * vh + (1.0 - b2) * g * g
            mc = mh / (1.0 - b1 ** t)
            vc = vh / (1.0 - b2 ** t)
            step = lr * mc / (np.sqrt(vc) + eps)
            h = project_simplex(h - step, n)
            c = compute_c5(h, n)
            if c < bc:
                best = h.copy()
                bc = c
    return best, bc

def slsqp_polish(h, n, alphas, maxiter, deadline):
    best = project_simplex(h, n)
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_fun, "jac": sum_jac, "args": (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best, args=(n, a),
                          jac=smooth_grad, method="SLSQP",
                          bounds=bounds, constraints=cons,
                          options={"maxiter": maxiter, "ftol": 1e-14})
        except Exception:
            continue
        nh = project_simplex(res.x, n)
        nc = compute_c5(nh, n)
        if nc < bc - 1e-14:
            best = nh
            bc = nc
    return best, bc
