def sum_con(x, n):
    return float(x.sum() - 0.5 * n)

def sum_jac(x, n):
    return np.ones_like(x, dtype=float)

def softmax_max(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    zm = float(z.max())
    e = np.exp(z - zm)
    s = float(e.sum())
    p = e / s
    smooth = (m + np.log(s) / alpha) * (2.0 / n)
    return smooth, p, c

def compute_A(h, n):
    A = np.zeros((n, 2 * n - 1), dtype=float)
    off = n - 1
    for k in range(2 * n - 1):
        d = k - off
        if d >= 0:
            A[d:, k] += 1.0 - h[:n - d]
            A[:n - d, k] -= h[d:]
        else:
            e = -d
            A[:n - e, k] += 1.0 - h[e:]
            A[e:, k] -= h[:n - e]
    return A

def smooth_val(x, n, alpha):
    return softmax_max(x, n, alpha)[0]

def smooth_grad(x, n, alpha):
    _, p, _ = softmax_max(x, n, alpha)
    A = compute_A(x, n)
    return (2.0 / n) * (A.T @ p)

def pen_val(x, n, alpha, mu):

    dev = x.sum() - 0.5 * n
    return val + 0.5 * mu * dev * dev

        h0 = random_smooth_start(200, rng)
    n0 = len(h0)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    best_h, best_c = transform_improve(best_h, n0)
    main_dl = min(hard, t0 + 0.45 * float(budget_s))
    if main_dl < t0 + 20.0:
        main_dl = hard
    best_h, best_c = optimize_at_res(best_h, n0, main_dl, rng, False)
    cands = [160, 200, 256, 320, 400, 500, 640]
    for m in cands:
        if time.time() > hard - 35.0:
            break
        if m == n0:
            continue
        rem = hard - time.time()
        if m > 400 and rem < 55.0:
            continue
        dl = time.time() + min(40.0, 0.12 * rem)
        if dl < time.time() + 6.0:
            continue
        ht, ct = optimize_at_res(interp_h(best_h, m), m, dl, rng, True)
        if ct < best_c:
            best_h, best_c = ht, ct
    n = len(best_h)
    for s in [0.015, 0.04, 0.08, 0.15]:
        if time.time() > hard - 25.0:
            break
        hp = project_feasible(best_h + rng.normal(0.0, s, n), n)
        rem = hard - time.time()
        dl = time.time() + min(35.0, 0.08 * rem)
        if dl < time.time() + 5.0:
            continue
        ht, ct = optimize_at_res(hp, n, dl, rng, True)
        if ct < best_c:
            best_h, best_c = ht, ct
    if time.time() < hard - 4.0:
        n = len(best_h)
        rem = hard - time.time() - 2.0
        mt = int(min(6000, max(800, rem * 25)))
        best_h, best_c = pair_search(best_h, n, mt, 61, hard - 2.0, rng)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
