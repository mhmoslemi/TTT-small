def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = cap.sum()
        if tot > 1e-12:
            h += (diff / tot) * cap
    else:
        tot = h.sum()
        if tot > 1e-12:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    # Final fix
    res = target - h.sum()
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]

    val = ((zm + np.log(np.sum(np.exp(z - zm)))) / alpha) * (2.0 / n)
    return val

def smooth_jac(x, n, alpha):

    center = n - 1
    z = alpha * c
    z = z - z.max()
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)

        wm = w[m]

        if s == 0:

        else:

            g[:n-r] -= wm * x[r:]
    return (2.0 / n) * g

import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(h.sum())
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        cap = 1.0 - h
        tot = cap.sum()
        if tot > 1e-12:
            h += (diff / tot) * cap
    else:
        tot = h.sum()
        if tot > 1e-12:
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    res = target - float(h.sum())
    if abs(res) > 1e-10:
        if res > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if idx.size:
            h[idx[0]] = np.clip(h[idx[0]] + res, 0, 1)
    return h

def compute_c(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * (2.0 / n)

def binary_from_h(h):
    n = len(h)
    k = n // 2
    idx = np.argpartition(h, n - k)[n - k:]
    h2 = np.zeros(n)
    h2[idx] = 1.0
    return h2

def resample_h(old, n):
    n0 = len(old)
    if n0 == n:
        return project_feasible(old.copy(), n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    return project_feasible(np.interp(x1, x0, old), n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    z = alpha * c
    zm = float(z.max())
    return ((zm + np.log(np.sum(np.exp(z - zm)))) / alpha) * (2.0 / n)

def smooth_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    center = n - 1
    z = alpha * c
    z -= z.max()
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        s = center - m
        if s == 0:
            g += wm * (1.0 - 2.0 * x)
        elif s > 0:
            g[:n - s] += wm * (1.0 - x[s:])
            g[s:] -= wm * x[:n - s]
        else:
            r = -s
            g[r:] += wm * (1.0 - x[:n - r])
            g[:n - r] -= wm * x[r:]
    return (2.0 / n) * g

def sum_con(x, n):
    return float(x.sum()) - 0.5 * n

def sum_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h = project_feasible(h, n)
    best_c = compute_c(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_jac, bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
            x = project_feasible(np.asarray(res.x), n)
            c = compute_c(x, n)
            if c < best_c:
                best_h, best_c = x, c
        except Exception:
            pass
    return best_h, best_c

def cd_pass(h, n, max_iter, deadline, proj_every=6):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    L = 2 * n - 1
    best_c = float(c.max()) * (2.0 / n)
    best_h = h.copy()
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            hp = h[p]
            sp = max(p, n - 1 - p)
            lo = max(0, center - sp)
            hi = min(L - 1, center + sp)
            seg_len = hi - lo + 1
            s_arr = center - np.arange(lo, hi + 1, dtype=int)
            c_seg = c[lo:hi + 1].copy()
            a = np.zeros(seg_len)
            j1 = p - s_arr
            j2 = p + s_arr
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            a[v1] += (1.0 - h[j1[v1]])
            a[v2] -= h[j2[v2]]
            unaff = 0.0
            if lo > 0:
                unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < L:
                unaff = max(unaff, float(c[hi + 1:].max()))
            lo_x, hi_x = 0.0, 1.0
            for _ in range(45):
                mid = 0.5 * (lo_x + hi_x)
                vals = c_seg + a * (mid - hp)
                vm = float(vals.max())
                if vm < unaff - 1e-13:
                    slope = 0.0
                else:
                    am = vals >= (vm if vm > unaff else unaff) - 1e-12
                    slope = a[am].sum()
                if slope > 1e-10:
                    lo_x = mid
                elif slope < -1e-10:
                    hi_x = mid
                else:
                    lo_x = hi_x = mid
                    break
            x_opt = 0.5 * (lo_x + hi_x)
            new_max = max(float(np.max(c_seg + a * (x_opt - hp))), unaff)
            old_max = float(c.max())
            if new_max < old_max - 1e-12:
                c[lo:hi + 1] += a * (x_opt - hp)
                h[p] = x_opt
                improved = True
            if (p + 1) % proj_every == 0:
                h = project_feasible(h, n)
                c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max()) * (2.0 / n)
        if cur < best_c:
            best_c = cur
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def directed_perturb(h, n, rng, mode):
    h = h.copy()
    c_f = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    k_max = int(np.argmax(c_f))
    s = k_max - center
    if mode == 0:
        for step in range(80):
            alpha_j = 0.04 * (1.0 - step / 80.0)
            g = np.zeros(n)
            j1 = np.arange(n) - s

            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            g[v1] += (1.0 - h[j1[v1]])
            g[v2] -= h[j2[v2]]
            h = project_feasible(h - alpha_j * g, n)
    elif mode == 1:

        idx = rng.choice(n, flip_n, replace=False)
        h[idx] = 1.0 - h[idx]

    elif mode == 2:

    else:
        h = project_feasible(h + rng.normal(0, 0.04, n), n)
    return h
