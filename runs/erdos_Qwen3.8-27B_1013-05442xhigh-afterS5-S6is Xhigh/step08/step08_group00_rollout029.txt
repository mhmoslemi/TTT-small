import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def build_M(h, n):
    """Build matrix M where M[p,i] = (1-h[i-p+n-1]) if valid, else 0.
    c = M @ h' gives the correlation when we fix the '1-h' part."""
    M = np.zeros((2*n-1, n))
    for p in range(2*n-1):
        i_lo = max(0, p - n + 1)
        i_hi = min(n - 1, p)
        if i_lo <= i_hi:
            M[p, i_lo:i_hi+1] = 1.0 - h[i_lo - p + (n-1) : i_hi - p + (n-1) + 1]
    return M

def build_M2(h, n):
    """Build matrix M2 where M2[p,j] = h[j+p-(n-1)] if valid, else 0.
    For the second LP: c[p] = row_sum[p] - M2[p,:] @ h'"""
    M2 = np.zeros((2*n-1, n))
    row_sums = np.zeros(2*n-1)
    for p in range(2*n-1):
        j_lo = max(0, n - 1 - p)
        j_hi = min(n - 1, p)
        if j_lo <= j_hi:
            M2[p, j_lo:j_hi+1] = h[j_lo + p - (n-1) : j_hi + p - (n-1) + 1]
            row_sums[p] = h[j_lo + p - (n-1) : j_hi + p - (n-1) + 1].sum()
    return M2, row_sums

def lp_alternate(h, n, max_iters, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # LP 1: fix g = 1-h, optimize h'
        g = 1.0 - h
        M = build_M(h, n)  # M @ h' = c (correlation values)
        hp = cp.Variable(n)
        t = cp.Variable()
        cons = [M @ hp <= t, cp.sum(hp) == n/2.0, hp >= 0, hp <= 1]
        prob = cp.Problem(cp.Minimize(t), cons)
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
            if hp.value is not None:
                h_new = project_feasible(np.clip(hp.value, 0, 1), n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    h = h_new
                    best_c = c_new
        except Exception:
            pass
        
        # LP 2: fix h, optimize h' in the (1-h') part
        M2, row_sums = build_M2(h, n)
        hp2 = cp.Variable(n)
        t2 = cp.Variable()
        cons2 = [-M2 @ hp2 + t2 <= -row_sums, cp.sum(hp2) == n/2.0, hp2 >= 0, hp2 <= 1]
        prob2 = cp.Problem(cp.Minimize(t2), cons2)
        try:
            prob2.solve(solver=cp.CLARABEL, verbose=False)
            if hp2.value is not None:
                h_new = project_feasible(np.clip(hp2.value, 0, 1), n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    h = h_new
                    best_c = c_new
        except Exception:

    return h, best_c

def exact_cd_1d(c, h, p, n):
    """Exact 1D coordinate descent: find optimal h[p] given others fixed."""
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:

    # - h[p] is in the second factor: i-p_out+n-1 = p, so p_out = i-p+n-1
