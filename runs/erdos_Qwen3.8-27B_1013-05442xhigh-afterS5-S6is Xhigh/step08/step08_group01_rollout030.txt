def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return (m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * (2.0 / n)

def lse_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    exps = np.exp(sh - mx)
    w = exps / exps.sum()
    t1 = fftconvolve(w, 1.0 - x, mode='full')[:n]
    t2 = np.correlate(x, w, mode='full')[:n]
    return (t1 - t2) * (2.0 / n)

        return lse_grad(x, n, alpha) + 2*pen*(x.sum()-n/2.0)*np.ones(n)

                   bounds=[(0,1)]*n, options={'maxiter':maxiter,'ftol':1e-18,'gtol':1e-14})

import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return (m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * (2.0 / n)

def lse_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    exps = np.exp(sh - mx)
    w = exps / exps.sum()
    g = np.zeros(n)
    for p in range(n):
        klo = max(0, p - (n - 1))
        khi = min(2 * n - 2, p + n - 2)
        if klo > khi:
            continue
        ks = np.arange(klo, khi + 1)
        f_idx = p + ks - n + 1
        b_idx = p - ks + n - 1
        dc = np.zeros(len(ks))
        vf = (f_idx >= 0) & (f_idx < n)
        if np.any(vf):
            dc[vf] += 1.0 - x[f_idx[vf]]
        vb = (b_idx >= 0) & (b_idx < n)
        if np.any(vb):
            dc[vb] -= x[b_idx[vb]]
        g[p] = float(np.dot(w[klo:khi + 1], dc))
    return g * (2.0 / n)

def lbfgsb_pass(h, n, alpha, maxiter=800, pen=200.0):
    def f(x):
        return lse_obj(x, n, alpha) + pen * (x.sum() - n / 2.0) ** 2
    def g(x):
        s = x.sum() - n / 2.0
        return lse_grad(x, n, alpha) + 2.0 * pen * s * np.ones(n)
    res = minimize(f, h, jac=g, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
    return project_feasible(res.x, n)

def coord_cd(h, n, max_passes=200, deadline=None, grid=50):
    h = h.copy()
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cm = float(c.max())
        imp = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            klo = max(0, p - (n - 1))
            khi = min(2 * n - 2, max(p, n - 1 - p))
            if klo > khi:
                continue
            ks = np.arange(klo, khi + 1)
            f_idx = p + ks - n + 1
            b_idx = p - ks + n - 1
            dc = np.zeros(len(ks))
            vf = (f_idx >= 0) & (f_idx < n)
            if np.any(vf):
                dc[vf] += 1.0 - h[f_idx[vf]]
            vb = (b_idx >= 0) & (b_idx < n)
            if np.any(vb):
                dc[vb] -= h[b_idx[vb]]
            ca = c[ks]
            um = 0.0
            if klo > 0:
                um = max(um, float(c[:klo].max()))
            if khi < 2 * n - 2:
                um = max(um, float(c[khi + 1:].max()))

            def o(d):
                return max(float(np.max(ca + d * dc)), um)
            lo, hi = -h0, 1.0 - h0
            dg = np.linspace(lo, hi, grid)

        if not imp:

def subgradient_polish(h, n, max_iter=400, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.02
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        max_val = c.max()
        top_k = np.where(c >= max_val - 1e-10)[0]
        sg = np.zeros(n)
        for k in top_k:
            for i in range(n):
                f = i + k - n + 1
                b = i - k + n - 1
                if 0 <= f < n:
                    sg[i] += (1.0 - h[f])
                if 0 <= b < n:
                    sg[i] -= h[b]
        sg /= len(top_k)
        h_new = project_feasible(h - lr * sg, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            h = h_new
            best_c = c_new
            stall = 0
        else:
            lr *= 0.8
            stall += 1
            if stall > 40 or lr < 1e-12:
                break
    return h, best_c

    if time.time() < deadline - 15:
        h5, c5 = coord_cd(best_h, n_cur, 300, deadline, grid=100)
        if c5 < best_c:
            best_h, best_c = h5, c5
    if time.time() < deadline - 5:
        h6, c6 = subgradient_polish(best_h, n_cur, 200, deadline)
        if c6 < best_c:
            best_h, best_c = h6, c6

    # Final 1-h check
    h_fin = project_feasible(1.0 - best_h, n_cur)
    c_fin = compute_c5(h_fin, n_cur)
    if c_fin < best_c:
        best_h, best_c = h_fin, c_fin

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
