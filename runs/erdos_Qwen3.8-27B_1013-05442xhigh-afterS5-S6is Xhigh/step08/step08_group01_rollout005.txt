    w = np.exp(alpha * (c - cm))

    for mi in range(len(w)):

                grad[sh:] -= wi * h[:n-sh]
        else:

            if s2 < n:
                grad[s2:] += wi * (1.0 - h[:n-s2])
                grad[:n-s2] -= wi * h[s2:]
    return grad * (2.0/n)

            else: break
            h = h_new
            c = compute_c5(h, n)
            if c < best: best = c
            lr = min(lr * 1.05, 0.1)

import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0, 1)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0, 1)
    return h

def _obj_1d(d, c_aff, a_coeff, diag_idx, unaff):
    nv = c_aff + d * a_coeff
    if diag_idx is not None:
        nv = nv.copy()
        nv[diag_idx] -= d * d
    return max(float(nv.max()), unaff)

def _slsqp_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    return (m + np.log(np.sum(np.exp(sh))) / alpha) * (2.0 / n)

def _sum_con(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_refine(h, n, alphas, maxiter=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
        try:
            res = minimize(_slsqp_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-20})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def coord_descent(h, n, passes=200, deadline=None, grid=120):

    for _ in range(passes):

            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            mfl = max(0, n - 1 - p)
            mfh = min(2 * n - 2, 2 * n - 2 - p)
            mbl = max(0, p)
            mbh = min(2 * n - 2, p + n - 1)
            m_lo = min(mfl, mbl)
            m_hi = max(mfh, mbh)
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a[vf] += 1.0 - h[jf[vf]]
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            ca = c[ms]
            un = 0.0
            if m_lo > 0:

            if m_hi < 2 * n - 2:

                nv[:, dl] -= d_grid ** 2
            vals = nv.max(axis=1)

        delta = c_new - c_cur
        if delta < 0 or rng.rand() < np.exp(-delta / max(t, 1e-8)):
            h = h_trial
        if c_new < best_c:
            best_c = c_new
            best_h = h_trial.copy()
    return best_h, best_c
