def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_correlation(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def active_set_subgrad(h, n, active_frac=0.1, lr=0.02, max_iter=500, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = compute_correlation(h, n)
        m_val = c.max()
        # Find active set: lags within active_frac of max
        threshold = m_val - active_frac * m_val
        active = np.where(c >= threshold)[0]
        # Compute aggregate subgradient
        sg = np.zeros(n)
        for m in active:
            # For lag m in full correlation output:
            # c[m] = sum over valid (i,j) pairs of h[i]*(1-h[j]) where j-i = m-(n-1)
            # d(c[m])/dh[p] = sum of (1-h[j]) for p=i, minus sum of h[i] for p=j
            lag = m - (n - 1)
            if lag >= 0:
                # c[m] = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
                for i in range(n - lag):

                    sg[i] += (1.0 - h[j])
                    sg[j] -= h[i]
            else:
                lag_abs = -lag
                # c[m] = sum_{i=lag_abs}^{n-1} h[i]*(1-h[i-lag_abs])
                for i in range(lag_abs, n):
                    j = i - lag_abs
                    sg[i] += (1.0 - h[j])
                    sg[j] -= h[i]
        # Normalize
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        sg /= sg_norm
        # Line search (backtracking)
        alpha = lr
        for _ in range(10):
            h_trial = project_feasible(h - alpha * sg, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c:
                h = h_trial
                best_c = c_trial
                stall = 0
                break
            alpha *= 0.5
        else:
            stall += 1
            if stall > 50:
                break
    return h, best_c

def get_active_subgrad(h, n, n_active=10):
    """Compute subgradient from the n_active largest lags."""
    c = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(c)[-n_active:]
    sg = np.zeros(n)
    for m in top_idx:
        lag = m - (n - 1)

            j = i + lag
            sg[i] += (1.0 - h[j])
            sg[j] -= h[i]
        else:
            la = -lag
            i = np.arange(la, n)
            j = i - la
            sg[i] += (1.0 - h[j])
            sg[j] -= h[i]
    return sg, c.max()

                        a_coeff[mi] -= h[p + la]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            if m_lo <= (n - 1) <= m_hi:
                diag_local = (n - 1) - m_lo
            lo = -h0
            hi = 1.0 - h0
            d_grid = np.linspace(lo, hi, grid_size)
            vals = _eval_1d_vec(d_grid, c_aff, a_coeff, diag_local, unaff_max)
            best_idx = int(np.argmin(vals))
            best_d = d_grid[best_idx]
            best_val = float(vals[best_idx])
            if 0 < best_idx < grid_size - 1:
                gl = d_grid[best_idx - 1]
                gh = d_grid[best_idx + 1]
                invphi = (np.sqrt(5.0) - 1.0) / 2.0
                x1 = gh - invphi * (gh - gl)
                x2 = gl + invphi * (gh - gl)
                f1 = _eval_1d_scalar(x1, c_aff, a_coeff, diag_local, unaff_max)
                f2 = _eval_1d_scalar(x2, c_aff, a_coeff, diag_local, unaff_max)
                for _ in range(80):
                    if f1 < f2:
                        gh = x2; x2 = x1; f2 = f1
                        x1 = gh - invphi * (gh - gl)
                        f1 = _eval_1d_scalar(x1, c_aff, a_coeff, diag_local, unaff_max)
                    else:
                        gl = x1; x1 = x2; f1 = f2
                        x2 = gl + invphi * (gh - gl)
                        f2 = _eval_1d_scalar(x2, c_aff, a_coeff, diag_local, unaff_max)
                bx = (gl + gh) / 2.0
                bf = _eval_1d_scalar(bx, c_aff, a_coeff, diag_local, unaff_max)
                if bf < best_val:
                    best_d, best_val = bx, bf
            if best_val < cur_max - 1e-14:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                cur_max = min(cur_max, best_val)
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def multi_subgrad(h, n, n_active=10, max_iter=2000, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.025
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(c)[-n_active:]
        sg = np.zeros(n)
        for m in top_idx:
            lag = m - (n - 1)
            if lag >= 0:
                i = np.arange(n - lag)
                j = i + lag
                sg[i] += (1.0 - h[j])
                sg[j] -= h[i]
            else:
                la = -lag
                i = np.arange(la, n)
                j = i - la
                sg[i] += (1.0 - h[j])
                sg[j] -= h[i]
        sn = np.linalg.norm(sg)
        if sn < 1e-15:
            break
        sg /= sn
        alpha = lr
        improved = False
        for _ in range(10):
            h_t = project_feasible(h - alpha * sg, n)
            c_t = compute_c5(h_t, n)
            if c_t < best_c - 1e-14:
                h, best_c = h_t, c_t
                improved = True
                break
            alpha *= 0.4
        if improved:
            stall = 0
            if it % 300 == 299:
                lr = min(lr * 1.3, 0.12)
        else:
            stall += 1
            lr *= 0.5
            if stall > 120 or lr < 1e-9:
                break
    return h, best_c

def pair_cd(h, n, max_passes=30, deadline=None, n_tries=200):
    h = h.copy()
    rng = np.random.RandomState(77)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        improved = False
        for _ in range(n_tries):
            if deadline is not None and time.time() > deadline:
                break
            p = rng.randint(n)
            q = rng.randint(n)
            if p == q:
                continue
            hp, hq = h[p], h[q]
            if hp < 1e-12 or hp > 1 - 1e-12 or hq < 1e-12 or hq > 1 - 1e-12:
                continue
            best_v = cur_max
            best_dp = best_dq = 0.0
            ng = 5
            dps = np.linspace(-hp, 1 - hp, ng)
            dqs = np.linspace(-hq, 1 - hq, ng)
            for dp in dps:
                h[p] = hp + dp
                for dq in dqs:
                    h[q] = hq + dq
                    v = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                    if v < best_v - 1e-14:
                        best_v, best_dp, best_dq = v, dp, dq
            h[p] = hp + best_dp
            h[q] = hq + best_dq
            if best_dp != 0.0 or best_dq != 0.0:
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Multi-lag subgradient
    t1 = t0 + 0.18 * budget_s
    if time.time() < t1:
        h1, c1 = multi_subgrad(best_h, n, 10, 3000, t1)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Fine coordinate descent
    t2 = t0 + 0.50 * budget_s
    if time.time() < t2:
        h2, c2 = coord_cd_fine(best_h, n, 400, t2, grid_size=80)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Pair coordinate descent
    t3 = t0 + 0.65 * budget_s
    if time.time() < t3:
        h3, c3 = pair_cd(best_h, n, 40, t3, n_tries=150)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Subgradient + coord again
    t4 = t0 + 0.78 * budget_s
    if time.time() < t4:
        h4, c4 = multi_subgrad(best_h, n, 15, 1500, t4)
        if c4 < best_c:
            best_h, best_c = h4, c4
        if time.time() < t4:
            h4b, c4b = coord_cd_fine(best_h, n, 200, t4, grid_size=60)
            if c4b < best_c:
                best_h, best_c = h4b, c4b

    # Phase 5: Multi-n exploration
    t5 = t0 + 0.88 * budget_s
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [500, 700, 800]:
        if time.time() > t5 - 20:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = coord_cd_fine(h_int, n_try, 150, t5, grid_size=50)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new

    # Phase 6: Final polishing
    if time.time() < deadline:
        n_cur = len(best_h)
        h6, c6 = coord_cd_fine(best_h, n_cur, 400, deadline, grid_size=100)
        if c6 < best_c:
            best_h, best_c = h6, c6

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
