def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def subgradient_step(h, n, eta):
    """One projected subgradient step."""
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(c))
    # Compute subgradient of c[k_star] w.r.t. h
    m = k_star
    n1 = n - 1
    grad = np.zeros(n)
    for p in range(n):
        # Term where h[p] is the "first" factor: h[p]*(1-h[p+m-n1])
        j1 = p + m - n1
        if 0 <= j1 < n:
            grad[p] += (1.0 - h[j1])
        # Term where h[p] is the "second" factor: h[p-m+n1]*(1-h[p])
        j2 = p - m + n1
        if 0 <= j2 < n:
            grad[p] -= h[j2]
        # For m = n-1, both j1 and j2 equal p, so we get (1-h[p]) - h[p] = 1-2h[p]
        # which is correct for d/dh[p] of h[p]*(1-h[p])
    return h - eta * grad, k_star

def smooth_objective(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx) / np.sum(np.exp(shifted - mx))
    grad = np.zeros(n)
    for k in range(len(c)):

        m_val = k
        n1 = n - 1
        for p in range(n):
            j1 = p + m_val - n1
            if 0 <= j1 < n:
                grad[p] += weights[k] * (1.0 - h[j1])
            j2 = p - m_val + n1
            if 0 <= j2 < n:
                grad[p] -= weights[k] * h[j2]
    return grad * (2.0 / n)
