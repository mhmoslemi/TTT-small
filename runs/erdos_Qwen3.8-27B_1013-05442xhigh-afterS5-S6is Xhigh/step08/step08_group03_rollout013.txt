            if tot < 1e-15:
                break
            h += (diff / tot) * h

def grad_smoothed(h, n, lam):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    w = np.exp(lam * (c - c_max))
    w_sum = w.sum()

    w_pad = np.zeros(size)

    S2 = fftconvolve(w_rev_pad, h_pad, mode='full')[2*n-1:3*n-1]
    grad = dx * (1.0 - S1 - S2)
    return grad, f, c

    """Gradient using only the active (max) constraint."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(c))
    lag = k_star - (n - 1)
    f = float(c[k_star] * dx)
    if lag >= 0:
        S1 = np.zeros(n); S1[lag:] = h[:n-lag]
        S2 = np.zeros(n); S2[:n-lag] = h[lag:]
    else:
        l = -lag
        S1 = np.zeros(n); S1[:n-l] = h[l:]
        S2 = np.zeros(n); S2[l:] = h[:n-l]
    grad = dx * (1.0 - S1 - S2)
    return grad, f, c

    best_h = h.copy()
    best_c = compute_c5(h, n)
    lam = 5.0

    momentum = 0.85
    v = np.zeros(n)

            break
        n_iter += 1

import numpy as np
from scipy.signal import fftconvolve

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smoothed(h, n, lam):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    w = np.exp(lam * (c - c_max))
    w_sum = w.sum()
    f = c_max + np.log(w_sum) / lam
    w = w / w_sum
    size = 4 * n
    h_pad = np.zeros(size)
    h_pad[n:2*n] = h
    w_pad = np.zeros(size)
    w_pad[n:n+2*n-1] = w
    w_rev_pad = np.zeros(size)
    w_rev_pad[n:n+2*n-1] = w[::-1]
    S1 = fftconvolve(w_pad, h_pad, mode='full')[2*n-1:3*n-1]
    S2 = fftconvolve(w_rev_pad, h_pad, mode='full')[2*n-1:3*n-1]
    grad = dx * (1.0 - S1 - S2)
    return grad, f, c

def grad_active(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(c))
    lag = k_star - (n - 1)
    f = float(c[k_star] * dx)

        S2 = np.zeros(n); S2[:n-lag] = h[lag:]

        S1 = np.zeros(n); S1[:n-l] = h[l:]
        S2 = np.zeros(n); S2[l:] = h[:n-l]
    grad = dx * (1.0 - S1 - S2)
    return grad, f

def exact_cd(h, n, max_passes=300, deadline=None, grid_pts=30, gs_iters=28):

        pm = p - lag

        vm = (pm >= 0) & (pm < n)

        idx_cache.append((pm[vm], vm, pp[vp], vp))

        if deadline is not None and time.time() > deadline:

            c0 = c - h0 * A + h0*h0 * D
            grid = np.linspace(0.0, 1.0, grid_pts)
