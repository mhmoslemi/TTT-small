import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline and time.time() > deadline - 30:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

            m_hi = max(m_hi_f, m_hi_b)
            ms = np.arange(m_lo, m_hi + 1)

            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))

        h = project_feasible(h, n)

                # Split: left part [d_lo, r1] and right part [r2, d_hi]
                # Pick the one with smaller |d|
                if abs(d_lo) <= abs(r2) and d_lo <= r1:
                    d_hi = min(d_hi, r1)  # use left part
                else:
                    d_lo = max(d_lo, r2)  # use right part
    
    d_lo = max(d_lo, -h0)
    d_hi = min(d_hi, 1.0 - h0)
    
    if d_lo > d_hi:
        return h0
    
    # Pick d in [d_lo, d_hi] that minimizes the actual max (should be ≈ L_star)
    # Use the midpoint for stability
    d = (d_lo + d_hi) / 2.0
    return float(np.clip(h0 + d, 0.0, 1.0))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=500, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline and time.time() > deadline - 30:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def solve_1d(c, h, p, n):
    h0 = h[p]
    m_lo = max(0, n - 1 - p, p)
    m_hi = min(2 * n - 2, n - 1 + p, p + n - 1)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    valid_f = (jf >= 0) & (jf < n)
    valid_b = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(valid_f):
        a[valid_f] += (1.0 - h[jf[valid_f]])
    if np.any(valid_b):
        a[valid_b] -= h[jb[valid_b]]
    c_aff = c[ms]
    diag_pos = np.where(ms == (n - 1))[0]
    has_diag = len(diag_pos) > 0
    unaff_max = 0.0
    if m_lo > 0:
        unaff_max = max(unaff_max, float(c[:m_lo].max()))
    if m_hi < 2 * n - 2:
        unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
    cur_max = float(c.max())
    L_lo = unaff_max
    L_hi = cur_max
    for _ in range(80):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        d_lo_v = -h0
        d_hi_v = 1.0 - h0
        ok = True
        pos = a > 1e-15
        neg = a < -1e-15
        zer = ~pos & ~neg
        if np.any(zer) and np.any(c_aff[zer] > L + 1e-15):
            ok = False
        if ok and np.any(pos):
            d_hi_v = min(d_hi_v, float(np.max((L - c_aff[pos]) / a[pos])))
        if ok and np.any(neg):
            d_lo_v = max(d_lo_v, float(np.max((L - c_aff[neg]) / a[neg])))
        if ok and d_lo_v > d_hi_v + 1e-14:
            ok = False
        if ok and has_diag:
            cd = c_aff[diag_pos[0]]
            ad = a[diag_pos[0]]
            disc = ad * ad - 4.0 * (L - cd)
            if disc >= 0:
                r1 = (ad - np.sqrt(disc)) / 2.0
                r2 = (ad + np.sqrt(disc)) / 2.0
                if d_lo_v > r1 and d_hi_v < r2:
                    ok = False
        if ok:
            L_hi = L
        else:
            L_lo = L
    L_star = L_hi
    # Recover d
    d_lo_v = -h0
    d_hi_v = 1.0 - h0
    pos = a > 1e-15
    neg = a < -1e-15
    if np.any(pos):
        d_hi_v = min(d_hi_v, float(np.max((L_star - c_aff[pos]) / a[pos])))
    if np.any(neg):
        d_lo_v = max(d_lo_v, float(np.max((L_star - c_aff[neg]) / a[neg])))
    d_lo_v = max(d_lo_v, -h0)
    d_hi_v = min(d_hi_v, 1.0 - h0)
    if has_diag:
        cd = c_aff[diag_pos[0]]
        ad = a[diag_pos[0]]
        disc = ad * ad - 4.0 * (L_star - cd)
        if disc >= 0:
            r1 = (ad - np.sqrt(disc)) / 2.0
            r2 = (ad + np.sqrt(disc)) / 2.0
            if not (d_hi_v <= r1 or d_lo_v >= r2):
                # Split: pick side closer to 0
                left_ok = d_lo_v <= r1
                right_ok = d_hi_v >= r2
                if left_ok and not right_ok:
                    d_hi_v = r1
                elif right_ok and not left_ok:
                    d_lo_v = r2
                else:
                    # Both sides available, pick one
                    if r1 >= 0 and d_hi_v > r1:
                        # Prefer left side (contains 0 if r1 >= 0)
                        if d_lo_v <= 0 <= r1:
                            d_hi_v = r1
                        elif 0 < d_lo_v:
                            d_hi_v = r1
                        else:
                            d_lo_v = r2
                    else:
                        d_lo_v = r2
    if d_lo_v > d_hi_v:
        return h0
    d = (d_lo_v + d_hi_v) / 2.0
    return float(np.clip(h0 + d, 0.0, 1.0))
