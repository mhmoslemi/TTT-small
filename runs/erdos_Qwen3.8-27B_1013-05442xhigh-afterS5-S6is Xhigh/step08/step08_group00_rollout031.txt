    m_lo = max(0, min(ko - p, p))

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-15:
                break
            h = np.clip(h + d / t * r, 0.0, 1.0)
        else:
            t = h.sum()
            if t < 1e-15:
                break
            h = np.clip(h + d / t * h, 0.0, 1.0)
    return h

def solve_coord(c, h, n, p):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    ko = n - 1
    m_lo = max(0, min(ko - p, p))
    m_hi = min(2 * n - 2, max(2 * ko - p, p + ko))
    ms = np.arange(m_lo, m_hi + 1)
    a = np.zeros(len(ms))

        k = ms[i] - ko
        if 0 <= p + k < n:
            a[i] += (1.0 - h[p + k])
        if 0 <= p - k < n:
            a[i] -= h[p - k]
    c_base = c[ms]
    diag_idx = np.where(ms == ko)[0]
    has_diag = len(diag_idx) > 0
    di = diag_idx[0] if has_diag else -1
    unaff = 0.0
    if m_lo > 0:
        unaff = max(unaff, float(c[:m_lo].max()))
    if m_hi < 2 * n - 2:
        unaff = max(unaff, float(c[m_hi + 1:].max()))
    L_lo, L_hi = unaff, float(c.max())
    neg_m = a < -1e-15
    pos_m = a > 1e-15
    zero_m = np.abs(a) <= 1e-15
    if has_diag:
        neg_m[di] = False
        pos_m[di] = False
        zero_m[di] = False
    for _ in range(52):

def cd_refine(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, 'full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        c = np.correlate(h, 1.0 - h, 'full')
        for p in range(n):
            if time.time() > deadline:
                break
            x_new = solve_coord(c, h, n, p)
            if abs(x_new - h[p]) > 1e-12:
                h[p] = x_new
                improved = True
                c = np.correlate(h, 1.0 - h, 'full')
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def slp_refine(h, n, max_iters, deadline, trust=0.15, n_active=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, 'full')
        max_c = float(c.max())
        n_act = min(n_active, len(c))
        active_idx = np.argsort(c)[-n_act:]
        A_rows = []
        b_vals = []
        for ki in active_idx:
            k = ki - (n - 1)
            g = np.zeros(n)
            for p in range(n):
                if 0 <= p + k < n:
                    g[p] += (1.0 - h[p + k])
                if 0 <= p - k < n:
                    g[p] -= h[p - k]
            rhs = -c[ki] + float(np.dot(g, h))
            row = np.zeros(n + 1)
            row[:n] = g
            row[n] = -1.0
            A_rows.append(row)
            b_vals.append(rhs)
        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        lb = np.clip(h - trust, 0.0, 1.0)
        ub = np.clip(h + trust, 0.0, 1.0)
        bounds = list(zip(lb, ub)) + [(None, None)]
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True})
        except Exception:
            break
        if res.status != 0:
            break
        h_new = project_feasible(res.x[:n], n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            h = h_new
            best_c = c_new
            if trust > 0.02:
                trust *= 0.85
        else:
            trust *= 0.7
            if trust < 0.005:
                break
    return h, best_c

def try_n_value(h_best, n_best, c_best, n_try, deadline, min_remaining):
    if n_try == n_best or time.time() > deadline - min_remaining:
        return h_best, c_best, n_best
    x_orig = np.linspace(0, 2, n_best)
    x_new = np.linspace(0, 2, n_try)
    h_int = project_feasible(np.interp(x_new, x_orig, h_best), n_try)
    h_slp, c_slp = slp_refine(h_int, n_try, 30, deadline - min_remaining, trust=0.12, n_active=150)
    h_cd, c_cd = cd_refine(h_slp, n_try, 50, deadline - min_remaining)
    if c_cd < c_best:
        return h_cd, c_cd, n_try
    return h_best, c_best, n_best
