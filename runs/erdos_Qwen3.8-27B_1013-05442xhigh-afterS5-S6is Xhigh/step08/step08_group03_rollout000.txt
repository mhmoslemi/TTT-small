            return np.ones(n)

                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def subgrad_step(h, n, step_size, deadline=None):
    """Sub-gradient step focusing on active lags."""
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cmax = c.max()
    # Active set: lags within 1% of max
    active = c > cmax - 0.01 * cmax
    # Compute negative sub-gradient: direction to decrease active correlations
    # d c[m]/d h[p] = 1 - h[p+m-(n-1)] - h[p-(m-(n-1))] (with validity)
    # We want to decrease c[m] for active m, so move h against the gradient
    grad = np.zeros(n)
    for m in np.where(active)[0]:
        shift = m - (n - 1)
        # dc[m]/dh[p] = 1 - h[p+shift] (valid) - h[p-shift] (valid)
        for p in range(n):
            g = 1.0
            i1 = p + shift
            if 0 <= i1 < n:
                g -= h[i1]
            i2 = p - shift
            if 0 <= i2 < n:
                g -= h[i2]
            grad[p] += g
    # Normalize and apply
    norm = np.linalg.norm(grad)
    if norm > 1e-15:
        h = h - step_size * grad / norm
        h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    
    # Phase 1: p-norm SLSQP with fine schedule on initial n
    h1, c1 = pnorm_slsqp(best_h, n_init, [20, 50, 100, 200, 500, 1000], 600, deadline)
    if c1 < best_c:
        best_h, best_c = h1, c1
    
    # Phase 2: Exact CD
    h2, c2 = exact_cd(best_h, n_init, 200, deadline, damping=1.0)
    if c2 < best_c:
        best_h, best_c = h2, c2
    
    # Phase 3: Multi-scale exploration
    n_list = [350, 400, 450, 500, 550, 600, 700, 800]
    if n_init in n_list:
        n_list.remove(n_init)
    for n_try in n_list:
        if time.time() > deadline - 150:
            break
        remain = deadline - time.time()
        if remain < 80:
            break
        h_int = interp_h(best_h, len(best_h), n_try)
        h_opt, c_opt = pnorm_slsqp(h_int, n_try, [100, 300, 800], 350, deadline)
        h_opt, c_opt = exact_cd(h_opt, n_try, 150, deadline, damping=1.0)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
    
    # Phase 4: Complement symmetry exploitation
    if time.time() < deadline - 80:
        n = len(best_h)
        # Try complement
        h_comp = project_feasible(1.0 - best_h, n)
        hc, cc = exact_cd(h_comp, n, 100, deadline, damping=1.0)
        if cc < best_c:
            best_h, best_c = hc, cc
        # Try average with complement
        h_avg = project_feasible(0.5 * best_h + 0.5 * (1.0 - best_h[::-1]), n)
        ha, ca = exact_cd(h_avg, n, 100, deadline, damping=1.0)
        if ca < best_c:
            best_h, best_c = ha, ca
    
    # Phase 5: Structured perturbations + CD
    if time.time() < deadline - 60:
        n = len(best_h)
        n_restarts = max(1, int((deadline - time.time() - 40) / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            if r % 3 == 0:
                # Fourier perturbation
                freqs = rng.choice(np.arange(1, n//2), size=3, replace=False)
                pert = np.zeros(n)
                for f in freqs:
                    perp = rng.normal(0, 0.03)
                    perq = rng.normal(0, 0.03)
                    pert += perp * np.cos(2*np.pi*f*np.arange(n)/n) + perq * np.sin(2*np.pi*f*np.arange(n)/n)
                hp = project_feasible(best_h + pert, n)
            elif r % 3 == 1:
                # Random perturbation
                hp = project_feasible(best_h + rng.normal(0, 0.02, n), n)
            else:
                # Block swap
                hp = best_h.copy()
                blk = n // 4
                for _ in range(2):
                    i = rng.randint(0, n - blk)
                    j = rng.randint(0, n - blk)
                    hp[i:i+blk], hp[j:j+blk] = hp[j:j+blk].copy(), hp[i:i+blk].copy()
                hp = project_feasible(hp, n)
            ho, co = pnorm_slsqp(hp, n, [200, 800], 300, deadline)
            ho, co = exact_cd(ho, n, 100, deadline, damping=1.0)
            if co < best_c:
                best_h, best_c = ho, co
    
    # Phase 6: Final high-p SLSQP + CD polish
    if time.time() < deadline - 30:
        n = len(best_h)
        hf, cf = pnorm_slsqp(best_h, n, [2000, 10000, 50000], 400, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    if time.time() < deadline - 10:
        hf, cf = exact_cd(best_h, len(best_h), 150, deadline, damping=1.0)
        if cf < best_c:
            best_h, best_c = hf, cf
    if time.time() < deadline - 3:
        hf, cf = exact_cd(best_h, len(best_h), 80, deadline, damping=0.7)
        if cf < best_c:
            best_h, best_c = hf, cf
    
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
