        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
                continue
            # Compute coefficients a[m] for affected lags
            m_lo = max(0, n-1-p)
            m_hi = min(2*n-2, n-1+p)
            ms = np.arange(m_lo, m_hi+1)
            jf = p + ms - (n-1)
            jb = p - ms + (n-1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):

                a[valid_b] -= h[jb[valid_b]]

        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=55):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n = 200
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP with fine alpha schedule
    h1, c1 = slsqp_optimize(best_h, n, [200, 500, 2000, 10000, 50000, 200000, 500000], 400)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Coarse CD
    h2, c2 = coord_descent(best_h, n, 60, deadline, gs_iters=40)
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: Try alternative n values
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [600, 500, 800]:
        if time.time() > deadline - 200:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = slsqp_optimize(h_int, n_try, [1000, 5000, 50000, 200000], 300)
        h_o, c_o = coord_descent(h_o, n_try, 50, deadline, gs_iters=50)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_orig = np.linspace(0, 2, n_try)
    # Phase 4: Multi-restart with structured perturbations
    if time.time() < deadline - 100:
        n_cur = len(best_h)
        remaining = deadline - time.time() - 80
        n_restarts = max(3, int(remaining / 55))
        for r in range(n_restarts):
            if time.time() > deadline - 80:
                break
            if r % 3 == 0:
                s = 0.01 + 0.03 * (r / max(1, n_restarts - 1))
                hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            elif r % 3 == 1:
                x = np.linspace(0, 2, n_cur)
                freq = 2.0 + 4.0 * ((r // 3) / max(1, n_restarts // 3))
                amp = 0.03 + 0.04 * (r / max(1, n_restarts - 1))
                hp = project_feasible(best_h + amp * np.sin(np.pi * freq * x), n_cur)
            else:
                x = np.linspace(0, 2, n_cur)
                freq = 5.0 + 3.0 * ((r // 3) / max(1, n_restarts // 3))
                hp = project_feasible(best_h + 0.03 * np.cos(2 * np.pi * freq * x / 2.0), n_cur)
            h_o, c_o = slsqp_optimize(hp, n_cur, [5000, 50000, 200000], 300)
            h_o, c_o = coord_descent(h_o, n_cur, 40, deadline, gs_iters=45)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Phase 5: Final high-precision CD polish
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        remaining = deadline - time.time() - 8
        max_passes = max(5, int(remaining / 3.0))
        h_f, c_f = coord_descent(best_h, n_cur, max_passes, deadline, gs_iters=65)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    # Final SLSQP polish
    if time.time() < deadline - 5:
        n_cur = len(best_h)
        h_f, c_f = slsqp_optimize(best_h, n_cur, [1000000, 5000000], 800)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h = project_feasible(best_h, n_cur)
        c_final = compute_c5(h, n_cur)
        if c_final < best_c:
            best_h, best_c = h, c_final
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
