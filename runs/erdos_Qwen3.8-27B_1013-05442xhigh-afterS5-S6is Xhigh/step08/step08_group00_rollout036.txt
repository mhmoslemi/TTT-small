    """Solve the linearized minimax LP."""
    n_lags = 2 * n - 1
    j = np.arange(n)
    A_ub = np.zeros((n_lags, n + 1))
    b_ub = np.zeros(n_lags)
    for idx in range(n_lags):
        k = idx - (n - 1)
        coeff = np.zeros(n)
        vf = (j + k >= 0) & (j + k < n)
        if np.any(vf):
            coeff[vf] += (1.0 - h0[j[vf] + k])
        vb = (j - k >= 0) & (j - k < n)
        if np.any(vb):
            coeff[vb] -= h0[j[vb] - k]
        i_lo, i_hi = max(0, -k), min(n, n - k)
        const = np.sum(h0[i_lo:i_hi] * h0[i_lo+k:i_hi+k]) if i_hi > i_lo else 0.0
        A_ub[idx, :n] = coeff
        A_ub[idx, n] = -1.0
        b_ub[idx] = -const
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    bounds = [(0, 1)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.clip(res.x[:n], 0, 1)
    return None

def lp_optimize(h, n, deadline):
    """Iterative LP-based optimization with line search."""
    h = h.copy()
    best_c = compute_c5(h, n)
    stagnant = 0
    while time.time() < deadline:
        h_lp = solve_lp(h, n)
        if h_lp is None:
            break
        h_lp = project_feasible(h_lp, n)
        # Line search
        best_alpha = 0.0
        best_c_ls = best_c
        for alpha in [0.2, 0.5, 0.8, 1.0]:
            h_try = project_feasible(h + alpha * (h_lp - h), n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c_ls:
                best_c_ls = c_try
                best_alpha = alpha
        if best_alpha == 0.0:
            stagnant += 1
            if stagnant > 5:
                break
            # Try a random perturbation
            h = project_feasible(h + 0.01 * np.random.randn(n), n)
            continue
        stagnant = 0
        h = project_feasible(h + best_alpha * (h_lp - h), n)
        best_c = compute_c5(h, n)
    return h, best_c

def solve_1d_exact(c, h, p, n):
    """Exact 1D coordinate descent for position p."""
    h0 = h[p]
    n_lags = 2 * n - 1
    c_arr = c  # current correlation array
    
    # Compute deltas
    deltas = np.zeros(n_lags)
    for idx in range(n_lags):
        k = idx - (n - 1)
        d = 0.0
        if 0 <= p + k < n:
            d += (1.0 - h[p + k])
        if 0 <= p - k < n:
            d -= h[p - k]
        deltas[idx] = d
    
    # Identify affected and unaffected
    affected = np.abs(deltas) > 1e-15
    affected[n-1] = True  # k=0 has quadratic term
    
    unaff_max = 0.0
    if np.any(~affected):
        unaff_max = float(c_arr[~affected].max())
    
    # Binary search on L
    L_lo = unaff_max
    L_hi = float(c_arr.max())
    
    for _ in range(50):
        if L_hi - L_lo < 1e-13:
            break
        L_mid = 0.5 * (L_lo + L_hi)
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        
        for idx in range(n_lags):
            if not affected[idx]:
                continue
            if idx == n - 1:
                continue  # handle quadratic separately
            d = deltas[idx]
            ck = c_arr[idx]
            if abs(d) < 1e-15:
                if ck > L_mid + 1e-12:
                    feasible = False
                    break
            elif d > 0:
                x_hi = min(x_hi, (L_mid - ck) / d + h0)
            else:
                x_lo = max(x_lo, (L_mid - ck) / d + h0)
        
        if feasible and x_lo <= x_hi:
            # Check quadratic constraint (k=0)
            C = L_mid - c_arr[n-1] + h0 - h0*h0
            disc = 1.0 - 4.0 * C
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                # Need x <= r1 or x >= r2
                # Check if [x_lo, x_hi] has any point satisfying this
                if x_hi > r1 and x_lo < r2:
                    # The interval [x_lo, x_hi] intersects (r1, r2)
                    # But we need points outside (r1, r2)
                    if x_lo >= r1 and x_hi <= r2:
                        feasible = False
                    # Otherwise, part of the interval is feasible
        
        if feasible:
            L_hi = L_mid
        else:
            L_lo = L_mid
    
    L = L_hi
    # Find feasible x
    x_lo, x_hi = 0.0, 1.0
    for idx in range(n_lags):
        if not affected[idx] or idx == n - 1:
            continue
        d = deltas[idx]
        ck = c_arr[idx]
        if abs(d) < 1e-15:
            continue
        elif d > 0:
            x_hi = min(x_hi, (L - ck) / d + h0)
        else:
            x_lo = max(x_lo, (L - ck) / d + h0)
    
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    
    # Quadratic
    C = L - c_arr[n-1] + h0 - h0*h0
    disc = 1.0 - 4.0 * C
    if disc >= 0:
        r1 = (1.0 - np.sqrt(disc)) / 2.0
        r2 = (1.0 + np.sqrt(disc)) / 2.0
        # Feasible: x in [x_lo, min(x_hi,r1)] or [max(x_lo,r2), x_hi]
        left_end = min(x_hi, r1)
        right_start = max(x_lo, r2)
        if x_lo <= left_end and left_end >= x_lo:
            # Left interval is non-empty
            x_candidate = 0.5 * (x_lo + left_end)
        elif right_start <= x_hi:
            x_candidate = 0.5 * (right_start + x_hi)
        else:
            x_candidate = h0
    else:
        if x_lo <= x_hi:
            x_candidate = 0.5 * (x_lo + x_hi)
        else:
            x_candidate = h0
    
    return float(np.clip(x_candidate, 0.0, 1.0))

def coord_descent(h, n, deadline, max_passes=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):

            h0 = h[p]

        if c_new < best_c:

        else:

                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.clip(res.x[:n], 0, 1)
    return None
