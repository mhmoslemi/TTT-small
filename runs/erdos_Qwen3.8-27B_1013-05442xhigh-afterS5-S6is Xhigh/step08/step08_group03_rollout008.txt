import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def exact_cd(h, n, max_passes=200, deadline=None, grid_pts=80, gs_iters=45, warm=None):
    h = h.copy()
    if warm is None:
        warm = h.copy()
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full').astype(float)
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-10 or h0 >= 1.0 - 1e-10:
                continue
            A = np.zeros(2 * n - 1)
            D = np.zeros(2 * n - 1)
            for i in range(n):
                j = i + p
                if j < n:
                    k = p + (n - 1) + (i - i)
                    k_idx = p + (n - 1) - i + i
                    k_idx = n - 1 + p - i
                    if 0 <= k_idx < 2 * n - 1:
                        A[k_idx] += (1.0 - h[i]) if i != p else 1.0
                        if i == p:
                            D[k_idx] += 1.0
                j2 = i - p
                if 0 <= j2 < n:
                    k_idx2 = p + (n - 1) + (j2 - i)
                    k_idx2 = n - 1 + p + j2 - i
                    if 0 <= k_idx2 < 2 * n - 1:
                        A[k_idx2] += -h[i]
                        if i == p:
                            D[k_idx2] += 1.0
            c0 = c - h0 * A + h0 * h0 * D
            grid = np.linspace(0.0, 1.0, grid_pts)
            fvals = np.array([float(np.max(c0 + t * A - t * t * D)) for t in grid])
            best_idx = int(np.argmin(fvals))
            lo = grid[max(0, best_idx - 1)]
            hi = grid[min(grid_pts - 1, best_idx + 1)]
            if hi - lo < 1e-14:
                t_star = grid[best_idx]
            else:
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a2, b2 = lo, hi
                c1 = b2 - gr * (b2 - a2)
                d1 = a2 + gr * (b2 - a2)
                fc1 = float(np.max(c0 + c1 * A - c1 * c1 * D))
                fd1 = float(np.max(c0 + d1 * A - d1 * d1 * D))
                for _ in range(gs_iters):
                    if fc1 < fd1:
                        b2 = d1; d1 = c1; fd1 = fc1
                        c1 = b2 - gr * (b2 - a2); fc1 = float(np.max(c0 + c1 * A - c1 * c1 * D))
                    else:
                        a2 = c1; c1 = d1; fc1 = fd1
                        d1 = a2 + gr * (b2 - a2); fd1 = float(np.max(c0 + d1 * A - d1 * d1 * D))
                t_star = (a2 + b2) * 0.5
            f_star = float(np.max(c0 + t_star * A - t_star * t_star * D))
            f0 = float(np.max(c0))
            f1 = float(np.max(c0 + A - D))
            best_f = min(f_star, f0, f1)
            if best_f < current_max - 1e-12:
                if f0 <= f_star and f0 <= f1:
                    t_new = 0.0
                elif f1 <= f_star and f1 <= f0:
                    t_new = 1.0
                else:
                    t_new = float(np.clip(t_star, 0.0, 1.0))
                c += (t_new - h0) * A - (t_new * t_new - h0 * h0) * D
                h[p] = t_new
                current_max = float(np.max(c))
                warm[p] = t_new
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def fast_cd(h, n, max_passes=500, deadline=None):

                continue
            j = i + p
            if j < n:
                pairs_a.append((i, n - 1 + p - i))
            else:
                j = i - p
                if 0 <= j < n:
                    pairs_a.append((i, n - 1 + p + j - i))
        idx[p] = pairs_a
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full').astype(float)
        current_max = float(c.max())
        improved = False
        order = range(n) if _ % 2 == 0 else range(n - 1, -1, -1)
        for p in order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-10 or h0 >= 1.0 - 1e-10:
                continue
            A = np.zeros(2 * n - 1)
            D = np.zeros(2 * n - 1)
            for i, k_idx in idx[p]:
                if i == p:

            A[n - 1] += 1.0

            c0 = c - h0 * A + h0 * h0 * D
            grid = np.linspace(0.0, 1.0, 60)

                fd1 = float(np.max(c0 + d1 * A - d1 * d1 * D))
                for _ in range(40):
                    if fc1 < fd1:

                    else:
                        a2 = c1; c1 = d1; fc1 = fd1

def pnorm_slsqp(h, n, p_list, maxiter_each=400, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for p in p_list:
        if deadline is not None and time.time() > deadline:
            break
        res = _slsqp_one_p(best_h, n, p, maxiter_each)
        if res is not None:
            new_h = project_feasible(res, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
    return best_h, best_c

def _slsqp_one_p(h, n, p, maxiter):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return sum(c ** p)
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        g = np.zeros_like(x)
        for i in range(n):
            g[i] = p * np.power(c, p - 1) * (1.0 - 2.0 * x[i])
        return g
    def sum_con(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return None

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def sinusoidal_perturb(h, n, rng, amp=0.05, freq_range=(2, 20)):

    x = np.arange(n)
    pert = amp * np.sin(2 * np.pi * f * x / n + phase)
    return project_feasible(h + pert, n)
