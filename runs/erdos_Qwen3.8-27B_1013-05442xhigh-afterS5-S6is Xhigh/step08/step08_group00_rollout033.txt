import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    lse = mx + np.log(np.sum(np.exp(sh - mx)))
    return (m + lse / alpha) * (2.0 / n)

def penalty_obj(x, n, L, beta):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    viol = np.maximum(0.0, c - L)
    return float(np.sum(viol ** 2))

def _sum_con(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alphas, maxiter=250):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def penalty_opt(h, n, L_start, L_end, beta, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    n_steps = 20
    Ls = np.linspace(L_start, L_end, n_steps)
    for L in Ls:
        if time.time() > deadline:
            break
        try:
            res = minimize(penalty_obj, best_h, args=(n, L, beta), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 150, 'ftol': 1e-20})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def exact_cd(h, n, max_passes, deadline, order='seq'):
    h = h.copy().astype(float)
    sn = n - 1
    two_sn = 2 * sn
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        seq = np.arange(n) if order == 'seq' else np.random.permutation(n)
        for p in seq:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
                continue
            lo = min(sn - p, p)
            hi = sn + p
            ms = np.arange(lo, hi + 1)
            jf = p + ms - sn
            jb = p - ms + sn
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            cb = c[ms].copy()
            didx = np.where(ms == sn)[0]
            has_d = len(didx) > 0
            if has_d:
                cb[didx] -= h0 * (1.0 - h0)
                a[didx] = 1.0 - 2.0 * h0
            unaff = 0.0
            if lo > 0:
                unaff = max(unaff, float(c[:lo].max()))
            if hi < two_sn:
                unaff = max(unaff, float(c[hi + 1:].max()))
            cur_max = float(c.max())
            neg = a < -1e-15
            pos = a > 1e-15
            zero = np.abs(a) <= 1e-15
            L_lo, L_hi = unaff, cur_max
            for _ in range(55):
                if L_hi - L_lo < 1e-14:
                    break
                L = (L_lo + L_hi) / 2.0
                xlo, xhi = 0.0, 1.0
                ok = True
                if np.any(zero) and np.any(cb[zero] > L + 1e-15):
                    ok = False
                if ok and np.any(neg):
                    xlo = max(xlo, float(np.max((L - cb[neg]) / a[neg])))
                if ok and np.any(pos):
                    xhi = min(xhi, float(np.min((L - cb[pos]) / a[pos])))
                if xlo > xhi + 1e-13:
                    ok = False
                if ok and has_d:
                    A = 1.0 - 2.0 * h0
                    B = L - cb[didx[0]]
                    disc = A * A - 4.0 * B
                    if disc >= 0.0:
                        sq = np.sqrt(disc)
                        r1 = (A - sq) / 2.0
                        r2 = (A + sq) / 2.0
                        if xlo > h0 + r1 and xhi < h0 + r2:
                            ok = False
                if ok:
                    L_hi = L
                else:
                    L_lo = L
            L_star = L_hi
            xlo, xhi = 0.0, 1.0
            if np.any(neg):
                xlo = max(xlo, float(np.max((L_star - cb[neg]) / a[neg])))
            if np.any(pos):
                xhi = min(xhi, float(np.min((L_star - cb[pos]) / a[pos])))
            if has_d:
                A = 1.0 - 2.0 * h0
                B = L_star - cb[didx[0]]
                disc = A * A - 4.0 * B
                if disc >= 0.0:
                    sq = np.sqrt(disc)
                    r1 = (A - sq) / 2.0
                    r2 = (A + sq) / 2.0
                    xl1, xl2 = h0 + r1, h0 + r2
                    if xlo < xl1 and xhi > xl2:
                        if h0 - xl1 <= xl2 - h0:
                            xhi = min(xhi, xl1)
                        else:
                            xlo = max(xlo, xl2)
                    elif xlo >= xl1 and xhi <= xl2:
                        if h0 - xl1 <= xl2 - h0:
                            xhi = min(xhi, xl1)
                        else:
                            xlo = max(xlo, xl2)
            xlo = max(xlo, 0.0)
            xhi = min(xhi, 1.0)
            if xlo > xhi + 1e-13:
                continue
            x_new = (xlo + xhi) / 2.0
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)

            s = 0.005 + 0.025 * (i / 5.0)

        for n_try in [500, 1000, 1400]:

            xo = np.linspace(0, 2, len(best_h), endpoint=False)

            if co < best_c:

    # Phase 6: Final high-precision passes
    n_f = len(best_h)
    if time.time() < deadline - 5:

    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
