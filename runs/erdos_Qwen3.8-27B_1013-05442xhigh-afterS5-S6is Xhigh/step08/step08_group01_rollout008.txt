    if time.time() < deadline - 120:
        h2, c2 = coord_cd_exact(best_h, n, 300, deadline, rng)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Subgradient with active set
    if time.time() < deadline - 100:
        h3, c3 = subgradient_polish(best_h, n, 400, deadline)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: More exact coordinate descent
    if time.time() < deadline - 60:
        h4, c4 = coord_cd_exact(best_h, n, 400, deadline, rng)
        if c4 < best_c:
            best_h, best_c = h4, c4
    # Phase 5: Try different resolutions
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [int(n * 0.7), int(n * 1.4)]:
        if time.time() > deadline - 80:
            break
        if abs(n_try - n) < 10:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = slsqp_run(h_int, n_try, [1000, 10000], 200, deadline)
        h_opt, c_opt = coord_cd_exact(h_opt, n_try, 150, deadline, rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = np.linspace(0, 2, n_try)
    # Phase 6: Multi-restart with perturbation
    n_cur = len(best_h)
    if time.time() < deadline - 60:
        for r in range(4):
            if time.time() > deadline - 40:
                break
            s = 0.01 + 0.03 * r
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            h_o, c_o = coord_cd_exact(hp, n_cur, 100, deadline, rng)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Phase 7: Final intensive coordinate descent
    n_cur = len(best_h)
    if time.time() < deadline - 5:
        hf, cf = coord_cd_exact(best_h, n_cur, 2000, deadline, rng)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Try 1-h symmetry
    h_flip = project_feasible(1.0 - best_h, n_cur)
    c_flip = compute_c5(h_flip, n_cur)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
        if time.time() < deadline - 3:
            hf, cf = coord_cd_exact(best_h, n_cur, 500, deadline, rng)
            if cf < best_c:
                best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
