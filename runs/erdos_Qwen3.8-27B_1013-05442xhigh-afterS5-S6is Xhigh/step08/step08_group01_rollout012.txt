def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

    h0 = h[p]

def targeted_reduction(h, n, top_k=10, max_rounds=50, deadline=None):

    h = h.copy()

            shift = m - (n - 1)

                    h_test = project_feasible(h_test, n)

                        current_max = float(c_test.max())
                        improved = True
                        break
        if not improved:

    return h, compute_c5(h, n)

    # Phase 3: Multi-resolution refinement
    for n_try in [800, 1000, 1200, 1400]:
        if time.time() > deadline - 250:
            break
        if n_try <= len(best_h):
            continue
        h_up = project_feasible(resample(best_h, len(best_h), n_try), n_try)
        h_up, c_up = coord_cd(h_up, n_try, max_passes=120, deadline=deadline - 250, grid_size=60)
        if c_up < best_c:
            best_h, best_c = h_up, c_up

    # Phase 4: SLSQP on best grid
    n_cur = len(best_h)
    if time.time() < deadline - 200:
        h4, c4 = slsqp_topk(best_h, n_cur, [500, 5000, 20000], [5, 20, 50], maxiter=400, deadline=deadline - 200)
        if c4 < best_c:
            best_h, best_c = h4, c4
            n_cur = len(h4)

    # Phase 5: Extended CD with larger grid
    if time.time() < deadline - 100:
        h5, c5 = coord_cd(best_h, n_cur, max_passes=300, deadline=deadline - 100, grid_size=80)
        if c5 < best_c:
            best_h, best_c = h5, c5
            n_cur = len(h5)

    # Phase 6: Perturbation + CD restarts
    if time.time() < deadline - 80:
        for r in range(5):
            if time.time() > deadline - 60:
                break
            sigma = 0.015 + 0.02 * r
            hp = project_feasible(best_h + rng.normal(0, sigma, n_cur), n_cur)
            h6, c6 = coord_cd(hp, n_cur, max_passes=100, deadline=deadline - 60, grid_size=60)
            if c6 < best_c:
                best_h, best_c = h6, c6
                n_cur = len(h6)

    # Phase 7: Final high-precision CD
    if time.time() < deadline - 15:
        h7, c7 = coord_cd(best_h, n_cur, max_passes=500, deadline=deadline - 5, grid_size=120)
        if c7 < best_c:
            best_h, best_c = h7, c7

    # Try 1-h symmetry
    h_flip = project_feasible(1.0 - best_h, len(best_h))
    c_flip = compute_c5(h_flip, len(best_h))
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
