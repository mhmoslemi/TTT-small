    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)

    c = compute_corr(h, n)
    m = c.max()
    z = alpha * (c - m)
    e = np.exp(z - z.max())
    s = e.sum()
    return (m + np.log(s) / alpha) * (2.0 / n)

def smooth_max_grad(h, n, alpha):
    c = compute_corr(h, n)
    m = c.max()
    z = alpha * (c - m)
    e = np.exp(z - z.max())
    p = e / e.sum()
    # Gradient of c[k] with respect to h[i]
    # c[k] = sum_j h[j] * (1-h[j+(k-(n-1))]) for valid j
    g = np.zeros(n)
    off = n - 1
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-14:
            continue
        d = k - off
        if d >= 0:
            # c[k] = sum_{j=0}^{n-1-d} h[j+d] * (1-h[j])

    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)

    e = np.exp(z - z.max())
    return (m + np.log(e.sum()) / alpha) * (2.0 / n)

    m = c.max()
    z = alpha * (c - m)
    e = np.exp(z - z.max())

    off = n - 1

    return (2.0 / n) * g

    return x.sum() - 0.5 * n

    return np.ones(x.shape)

        c_new = compute_c5(x, n)
        if c_new < compute_c5(h, n):
            return x, c_new
    except Exception:

    return h, compute_c5(h, n)

def subgrad_descent(h, n, steps, deadline):
    """Subgradient descent on the max correlation with momentum."""
    best_h = h.copy()
    best_c = compute_c5(h, n)

    lr = 0.01

        m = c.max()
