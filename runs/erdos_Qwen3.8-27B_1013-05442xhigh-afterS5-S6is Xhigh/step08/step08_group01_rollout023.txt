def project_feasible(h, n):

                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def cd_pass(h, n, c, grid_size=80, golden_iter=80, deadline=None, order=None):
    """One pass of coordinate descent. Returns updated h and c."""
    n2 = 2 * n - 1
    for p in (order if order is not None else range(n)):
        if deadline is not None and time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-12 or h0 > 1.0 - 1e-12:
            continue
        # Compute affected lags
        m_lo = max(0, n - 1 - p)
        m_hi = min(n2, n2 - p)
        ms = np.arange(m_lo, m_hi + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        a_coeff = np.zeros(len(ms))
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        a_coeff[vf] += 1.0 - h[jf[vf]]
        a_coeff[vb] -= h[jb[vb]]
        c_aff = c[ms]
        unaff_max = 0.0
        if m_lo > 0:
            unaff_max = max(unaff_max, float(c[:m_lo].max()))
        if m_hi < n2:
            unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
        # Check if diagonal is in range
        diag_idx = None
        if m_lo <= (n-1) <= m_hi:
            diag_idx = (n-1) - m_lo
        # 1D optimization
        lo = -h0
        hi = 1.0 - h0
        if hi - lo < 1e-12:
            continue
        d_grid = np.linspace(lo, hi, grid_size)
        # Vectorized evaluation
        # f(d) = max(c_aff + d*a_coeff - d^2*delta, unaff_max)
        vals = c_aff[None, :] + d_grid[:, None] * a_coeff[None, :]
        if diag_idx is not None:

        if diag_idx is not None:
            vals[:, diag_idx] -= d_grid ** 2
        row_max = vals.max(axis=1)
        vals_final = np.maximum(row_max, unaff_max)
        best_idx = int(np.argmin(vals_final))
        best_d = float(d_grid[best_idx])
        best_val = float(vals_final[best_idx])
        # Golden section refinement
        if 0 < best_idx < grid_size - 1:
            gl = float(d_grid[best_idx - 1])
            gh = float(d_grid[best_idx + 1])

                    if diag_idx is not None:

                    return max(float(nv.max()), unaff_max)
                res = golden(f1d, gl, gh, maxiter=golden_iter)
                if res.fun < best_val:
                    best_d = float(res.x)
                    best_val = float(res.fun)
            except Exception:
                pass
        h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))

import numpy as np
from scipy.optimize import golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def eval_1d(d, c_aff, a_coeff, diag_idx, unaff_max, quad_coeff=1.0):
    nv = c_aff + d * a_coeff
    if diag_idx is not None:
        nv = nv.copy()
        nv[diag_idx] -= quad_coeff * d * d
    return max(float(nv.max()), unaff_max)

def cd_pass(h, n, c, grid_size=80, g_iter=60, deadline=None, order=None):
    n2 = 2 * n - 1
    improved = False
    for p in (order if order is not None else range(n)):
        if deadline is not None and time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-10 or h0 > 1.0 - 1e-10:
            continue
        m_lo = max(0, n - 1 - p)
        m_hi = min(n2, n2 - p)
        ms = np.arange(m_lo, m_hi + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        a_coeff = np.zeros(len(ms))
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        a_coeff[vf] += 1.0 - h[jf[vf]]
        a_coeff[vb] -= h[jb[vb]]
        c_aff = c[ms]
        unaff_max = 0.0
        if m_lo > 0:
            unaff_max = max(unaff_max, float(c[:m_lo].max()))
        if m_hi < n2:
            unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
        diag_idx = None
        if m_lo <= (n-1) <= m_hi:
            diag_idx = (n-1) - m_lo
        lo = -h0
        hi = 1.0 - h0
        if hi - lo < 1e-10:
            continue
        d_grid = np.linspace(lo, hi, grid_size)
        vals = c_aff[None, :] + d_grid[:, None] * a_coeff[None, :]
        if diag_idx is not None:
            vals[:, diag_idx] -= d_grid ** 2
        row_max = vals.max(axis=1)
        finals = np.maximum(row_max, unaff_max)

        best_d = float(d_grid[best_idx])
        best_val = float(finals[best_idx])
        if 0 < best_idx < grid_size - 1:
            gl = float(d_grid[best_idx - 1])
            gh = float(d_grid[best_idx + 1])
            res = golden(eval_1d, gl, gh,
                        args=(c_aff, a_coeff, diag_idx, unaff_max, 1.0),
                        maxiter=g_iter)
            if res.fun < best_val:

                best_val = float(res.fun)
