            if tot < 1e-15:

def compute_gradient(h, n):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

def softmax_gradient(h, n, temperature):
    """Compute gradient of smoothed max using softmax weighting of lags."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    c_max = c.max()
    weights = np.exp((c - c_max) / temperature)
    weights /= weights.sum()
    
    # Gradient: for each h[j], gradient = sum over lags l where j participates
    # dc_l/dh[j] = [(1-h[j+l]) - h[j-l]] * dx (with boundary checks)
    grad = np.zeros(n)
    for j in range(n):
        total = 0.0
        for l in range(-(n-1), n):
            # j participates in c_l if:
            # (a) j+l is valid (j is first index): contributes (1-h[j+l])*dx
            # (b) j-l is valid (j-l is first index, j is second): contributes -h[j-l]*dx
            w = weights[n-1+l]

            if 0 <= j-l < n:

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_gradient(h, n, temp):
    """Compute gradient of soft-max of correlations."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    c_max = c.max()
    w = np.exp((c - c_max) / temp)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(c))
        w[n-1] = 1.0
    else:
        w /= ws
    
    # S[j] = sum of w[n-1-j : 2n-1-j]
    P = np.concatenate([[0.0], np.cumsum(w)])
    S = P[2*n-1 - np.arange(n)] - P[n-1 - np.arange(n)]
    
    # T1[j] = correlate(w, h, full)[2n-2-j]
    corr1 = np.correlate(w, h, mode='full')
    T1 = corr1[2*n-2 - np.arange(n)]
    
    # T2[j] = correlate(w, h[::-1], full)[j+n-1]
    corr2 = np.correlate(w, h[::-1], mode='full')
    T2 = corr2[np.arange(n) + n - 1]
    
    grad = dx * (S - T1 - T2)
    return grad

def adam_update(h, grad, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * grad**2
    m_hat = m / (1 - b1**t)
    v_hat = v / (1 - b2**t)
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
    return h_new, m, v
