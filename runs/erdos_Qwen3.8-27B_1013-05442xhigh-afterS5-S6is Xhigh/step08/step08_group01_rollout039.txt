def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def cd_1d_fast(h, n, max_passes=200, deadline=None):
    h = h.copy().astype(float)
    m_arr = np.arange(-(n-1), n)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = get_corr(h)
        cur_max = c.max()
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            pm = p + m_arr
            vf = (pm >= 0) & (pm < n)
            pmm = p - m_arr
            vb = (pmm >= 0) & (pmm < n)
            a = np.zeros(2 * n - 1)
            a[vf] = 1.0 - h[pm[vf]]
            a[vb] -= h[pmm[vb]]
            lo, hi = -h0, 1.0 - h0
            for _ in range(40):
                mid = 0.5 * (lo + hi)
                if hi - lo < 1e-12:
                    break
                vals = c + mid * a
                mx = vals.max()
                act = vals >= mx - 1e-14
                if a[act].min() > 1e-14:
                    hi = mid
                elif a[act].max() < -1e-14:
                    lo = mid
                else:
                    break
            d_opt = 0.5 * (lo + hi)

            f_mid = (c + d_opt * a).max()
            if f_lo < f_mid:

                d_opt = hi
            f_best = (c + d_opt * a).max()
            if f_best < cur_max - 1e-13:
                h[p] = float(np.clip(h0 + d_opt, 0.0, 1.0))
                cur_max = f_best
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h

    h = h.copy().astype(float)
    m_arr = np.arange(-(n-1), n)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = get_corr(h)
        cur_max = c.max()
        improved = False
        for start in range(0, n - block_size + 1, block_size):
            if deadline is not None and time.time() > deadline:
                break
            end = start + block_size
            hb = h[start:end]
            if np.all((hb < 1e-8) | (hb > 1.0 - 1e-8)):
                continue
            a_mat = np.zeros((2 * n - 1, block_size))
            for i in range(block_size):
                p = start + i
                pm = p + m_arr
                vf = (pm >= 0) & (pm < n)
                pmm = p - m_arr
                vb = (pmm >= 0) & (pmm < n)
                a_mat[vf, i] = 1.0 - h[pm[vf]]
                a_mat[vb, i] -= h[pmm[vb]]
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[n - 1] = False

            nc = len(c_m)
            cfun = np.zeros(block_size + 1)

                    if actual_max < cur_max - 1e-12:

                        cur_max = actual_max
                        improved = True
            except Exception:

        if not improved:

        h = project_feasible(h, n)

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def cd_1d_fast(h, n, max_passes=200, deadline=None):
    h = h.copy().astype(float)
    m_arr = np.arange(-(n - 1), n)
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = get_corr(h)
        cur_max = c.max()
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            pm = p + m_arr
            vf = (pm >= 0) & (pm < n)
            pmm = p - m_arr
            vb = (pmm >= 0) & (pmm < n)
            a = np.zeros(2 * n - 1)
            a[vf] = 1.0 - h[pm[vf]]
            a[vb] -= h[pmm[vb]]
            lo, hi = -h0, 1.0 - h0
            for _ in range(45):
                mid = 0.5 * (lo + hi)
                if hi - lo < 1e-13:
                    break
                vals = c + mid * a
                mx = vals.max()
                act = vals >= mx - 1e-15
                if a[act].min() > 1e-14:
                    hi = mid
                elif a[act].max() < -1e-14:
                    lo = mid
                else:
                    break
            d_opt = 0.5 * (lo + hi)
            f_lo = (c + lo * a).max()
            f_hi = (c + hi * a).max()

                d_opt = hi
            f_new = (c + d_opt * a).max()

                cur_max = f_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h

def block_cd_lp(h, n, block_size=10, max_passes=100, deadline=None):
    h = h.copy().astype(float)
    m_arr = np.arange(-(n - 1), n)
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[n - 1] = False
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = get_corr(h)
        cur_max = c.max()
        c_m = c[mask]
        improved = False
        for start in range(0, n - block_size + 1, block_size):
            if deadline is not None and time.time() > deadline:
                break
            end = start + block_size
            hb = h[start:end]
            if np.all((hb < 1e-9) | (hb > 1.0 - 1e-9)):
                continue
            a_mat = np.zeros((2 * n - 1, block_size))
            for i in range(block_size):
                p = start + i
                pm = p + m_arr
                vf = (pm >= 0) & (pm < n)
                pmm = p - m_arr
                vb = (pmm >= 0) & (pmm < n)
                a_mat[vf, i] = 1.0 - h[pm[vf]]
                a_mat[vb, i] -= h[pmm[vb]]
            a_m = a_mat[mask, :]
            nc = len(c_m)
            cfun = np.zeros(block_size + 1)
            cfun[-1] = 1.0
            A_ub = np.empty((nc, block_size + 1))
            A_ub[:, :block_size] = a_m
            A_ub[:, -1] = -1.0
            b_ub = -c_m
            bounds = [(-hb[i], 1.0 - hb[i]) for i in range(block_size)] + [(None, None)]
            try:
                res = linprog(cfun, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
                if res.status == 0:
                    h_new = h.copy()
                    h_new[start:end] = np.clip(hb + res.x[:block_size], 0.0, 1.0)
                    cn = get_corr(h_new)
                    actual_max = cn.max()
                    if actual_max < cur_max - 1e-12:
                        h = h_new
                        cur_max = actual_max
                        improved = True
            except Exception:
                pass
        if not improved:
            break
        h = project_feasible(h, n)
    return h
