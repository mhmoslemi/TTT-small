        best_h, best_c = optimize_at_n(best_h, n_init, deadline - 200, seed=seed)

    # Phase 2: Sweep different n values
    x_orig = np.linspace(0, 2, len(best_h))
    n_candidates = [80, 100, 120, 150, 180, 200, 250, 300, 350, 400, 500, 650, 800, 1000]
    n_candidates = [x for x in n_candidates if x != n_init]
    n_candidates.sort(key=lambda x: abs(x - n_init))

    for n_try in n_candidates:
        if time.time() > deadline - 100:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        # Also try initialized from scratch for diversity
        h_rand = project_feasible(np.full(n_try, 0.5) + rng.normal(0, 0.1, n_try), n_try)
        # Optimize both
        h_a, c_a = optimize_at_n(h_int, n_try, deadline - 80, seed=seed)
        if c_a < best_c:
            best_h, best_c = h_a, c_a
            x_orig = x_new
        h_b, c_b = optimize_at_n(h_rand, n_try, deadline - 60, seed=seed + 100)
        if c_b < best_c:
            best_h, best_c = h_b, c_b
            x_orig = np.linspace(0, 2, n_try)

    # Phase 3: Final refinement at best n
    n_best = len(best_h)
    if time.time() < deadline - 40:
        hf = cd_pass(best_h, n_best, 300, deadline - 20, iters=80, seed=seed + 500)
        cf = compute_c5(hf, n_best)
        if cf < best_c:
            best_h, best_c = hf, cf
        # SLSQP polish
        for alpha in [50000, 200000]:
            if time.time() > deadline - 20:
                break
            hf = slsqp_opt(best_h, n_best, alpha, 500)
            hf = cd_pass(hf, n_best, 100, deadline - 10, iters=80, seed=seed + 600)
            cf = compute_c5(hf, n_best)
            if cf < best_c:
                best_h, best_c = hf, cf

    # Phase 4: Try 1-h symmetry
    h_flip = project_feasible(1.0 - best_h, n_best)
    c_flip = compute_c5(h_flip, n_best)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
        # Refine the flipped version too
        if time.time() < deadline - 15:
            hf = cd_pass(best_h, n_best, 100, deadline - 5, iters=60, seed=seed + 700)
            cf = compute_c5(hf, n_best)
            if cf < best_c:
                best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
