        nv[diag_local] -= d * d

    return max(val, unaff_max)

def coord_cd(h, n, max_passes=200, deadline=None, grid_size=120):
    h = h.copy()
    for _ in range(max_passes):
        if deadline and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_f_lo = max(0, n - 1 - p)
            m_f_hi = min(2 * n - 2, 2 * n - 2 - p)
            m_b_lo = max(0, p)
            m_b_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(m_f_lo, m_b_lo)
            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            a_coeff = np.zeros(len(ms))
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += 1.0 - h[jf[vf]]
            if np.any(vb):
                a_coeff[vb] -= h[jb[vb]]
            c_aff = c[ms].copy()
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            if m_lo <= (n - 1) <= m_hi:
                diag_local = (n - 1) - m_lo
            lo = -h0
            hi = 1.0 - h0
            d_grid = np.linspace(lo, hi, grid_size)
            vals = np.array([obj_1d(d, c_aff, a_coeff, diag_local, unaff_max) for d in d_grid])
            best_idx = int(np.argmin(vals))
            best_d = d_grid[best_idx]
            best_val = float(vals[best_idx])
            if 0 < best_idx < grid_size - 1:
                gl, gh = d_grid[best_idx - 1], d_grid[best_idx + 1]
                try:
                    res = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=50)
                    if res.fun < best_val:
                        best_d = float(res.x)
                        best_val = float(res.fun)
                except Exception:
                    pass
            if best_val < current_max - 1e-15:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = min(current_max, best_val)
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

For L < 0, M = -L > 0:

- p appears in the "h[i-M]" part (as the second factor) if i-M = p, i.e., i = p+M. This is valid if M ≤ p+M ≤ n-1, i.e., 0 ≤ p ≤ n-1-M. Contributes -h[p+M] (since d/dh[p] of -h[i]*h[i-M] = -h[i-M] if i-M = p)

Wait, let me be more careful:
c = sum_{i=M}^{n-1} h[i] - h[i]*h[i-M]

∂c/∂h[p] = ∂/∂h[p] sum_{i=M}^{n-1} h[i] - ∂/∂h[p] sum_{i=M}^{n-1} h[i]*h[i-M]

First term: = 1 if M ≤ p < n, else 0

Second term:
= sum_{i=M}^{n-1} [δ(i,p)*h[i-M] + h[i]*δ(i-M,p)]
= h[p-M] * 1_{M≤p<n and p-M≥0} + h[p+M] * 1_{M≤p+M<n and p≥0}
= h[p-M] * 1_{M≤p<n} + h[p+M] * 1_{M≤p+M<n}
= h[p-M] * 1_{M≤p<n} + h[p+M] * 1_{0≤p<n-M}

So: ∂c/∂h[p] = 1_{M≤p<n} - h[p-M]*1_{M≤p<n} - h[p+M]*1_{0≤p<n-M}

import numpy as np
from scipy.optimize import minimize, golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_grad(h, n, alpha, top_m=250):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    if top_m >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argpartition(c, -top_m)[-top_m:]
    cs = c[idx]
    sh = alpha * (cs - m)
    mx = sh.max()
    es = np.exp(sh - mx)
    w = es / es.sum()
    val = (m + (mx + np.log(es.sum())) / alpha) * (2.0 / n)
    grad = np.zeros(n)
    for j in range(len(idx)):
        k = idx[j]
        L = k - (n - 1)
        wt = w[j]
        if L == 0:
            grad += wt * (1.0 - 2.0 * h)
        elif L > 0:
            g = np.ones(n)
            if n - L > 0:
                g[0:n - L] -= h[L:n]
            if L < n:
                g[L:n] -= h[0:n - L]
            grad += wt * g
        else:
            M = -L
            g = np.zeros(n)
            g[M:n] = 1.0
            g[M:n] -= h[0:n - M]
            g[0:n - M] -= h[M:n]
            grad += wt * g
    grad *= (2.0 / n)
    return val, grad

def _fg(x, n, alpha, lam):
    f, g = obj_grad(x, n, alpha)
    if lam > 0:
        s = x.sum() - n / 2.0
        f += lam * s * s
        g = g + 2.0 * lam * s
    return f, g

    for alpha in alphas:
        if deadline and time.time() > deadline - 30:
            break
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(_fg, best_h, args=(n, alpha, lam), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter_each, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 40})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    val = float(nv.max())
