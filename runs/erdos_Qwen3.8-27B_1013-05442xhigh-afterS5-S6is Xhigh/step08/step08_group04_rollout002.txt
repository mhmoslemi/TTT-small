                a[vj] += 1.0 - h[j[vj]]
            if np.any(vi):
                a[vi] -= h[i[vi]]
            off = center - lo
            cseg = np.empty(seg - 1, dtype=float)
            asg = np.empty(seg - 1, dtype=float)
            if off:
                cseg[:off] = c[lo:center]
                asg[:off] = a[:off]
            if hi > center:
                cseg[off:] = c[center + 1:hi + 1]
                asg[off:] = a[off + 1:]
            qconst = c[center] - hp * (1.0 - hp)
            unaff = 0.0
            if lo > 0:
                unaff = max(unaff, float(c[:lo].max()))
            if hi + 1 < L:
                unaff = max(unaff, float(c[hi + 1:].max()))
            bx, bf = _opt_coord(hp, cseg, asg, qconst, unaff)
            if bf < cur - 1e-13:
                dxv = bx - hp
                if off:
                    c[lo:center] += asg[:off] * dxv
                if hi > center:
                    c[center + 1:hi + 1] += asg[off:] * dxv
                c[center] = qconst + bx * (1.0 - bx)
                h[p] = bx
                cur = bf
                improved = True
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur = float(c.max())
        cval = (2.0 / n) * cur
        if cval < best_c:
            best_c, best_h = cval, h.copy()
        if not improved:
            break
    return best_h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    z = alpha * c
    zm = float(z.max())
    return ((zm + np.log(np.sum(np.exp(z - zm)))) / alpha) * (2.0 / n)

def smooth_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    center = n - 1
    z = alpha * c
    z -= z.max()
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm <= 1e-14:
            continue
        s = center - m
        if s == 0:
            g += wm * (1.0 - 2.0 * x)
        elif s > 0:
            g[:n - s] += wm * (1.0 - x[s:])
            g[s:] -= wm * x[:n - s]
        else:
            r = -s
            g[r:] += wm * (1.0 - x[:n - r])
            g[:n - r] -= wm * x[r:]
    return (2.0 / n) * g

def sum_con(x):
    return float(x.sum()) - 0.5 * len(x)

def sum_jac(x):
    return np.ones(len(x))

def slsqp_step(h, n, alphas, maxiter, deadline):
    best_h, best_c = project_feasible(h, n), compute_c(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           jac=smooth_jac, bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
            x = project_feasible(res.x, n)
            c = compute_c(x, n)
            if c < best_c:
                best_h, best_c = x, c
        except Exception:
            pass
    return best_h, best_c

def perturb(h, n, rng, typ):
    base = h.copy()
    if typ == 0:
        return project_feasible(base + rng.normal(0, 0.03, n), n)
    elif typ == 1:
        h2 = (base > 0.5).astype(float)
        h2 = project_feasible(h2, n)
        k = rng.randint(2, max(3, n // 20))
        idx = rng.choice(n, k, replace=False)
        h2[idx] = 1.0 - h2[idx]
        return project_feasible(h2, n)
    elif typ == 2:
        blk = rng.randint(5, max(6, n // 8))
        start = rng.randint(0, n - blk)
        h2 = base.copy()
        h2[start:start + blk] = 1.0 - h2[start:start + blk]
        return project_feasible(h2, n)
    elif typ == 3:
        x = (np.arange(n) + 0.5) * (2.0 / n)
        h2 = base.copy()
        for k in range(1, 6):
            amp = rng.uniform(-0.08, 0.08)
            h2 += amp * np.cos(k * np.pi * (x - 1.0))
        return project_feasible(h2, n)
    else:
        return project_feasible(base + rng.normal(0, 0.01, n), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 8.0
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n0 = 500
        x0 = np.linspace(0.1, 1.9, n0)
        h_init = project_feasible(1.0 - np.abs(x0 - 1.0), n0)
    n = len(h_init)
    if n % 2:
        n += 1
        h_init = resample_h(h_init, n)
    best_h = project_feasible(h_init, n)
    best_c = compute_c(best_h, n)

    # Phase 1: SLSQP + CD on initial
    if n <= 700 and time.time() < deadline - 120:
        h2, c2 = slsqp_step(best_h, n, (30, 300, 3000), 60, deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2
    best_h, best_c = cd_optimize(best_h, n, 12, deadline)

    # Phase 2: Multi-resolution
    for nt in [n, 256, 384, 480, 640, n]:
        if time.time() > deadline - 180:
            break
        if nt == n and nt == 256:
            continue
        h0 = resample_h(best_h, nt)
        h0, c0 = cd_optimize(h0, nt, 8, deadline)
        if c0 < best_c or nt != n:
            h_up = resample_h(h0, n)
            h_up, c_up = cd_optimize(h_up, n, 6, deadline)
            if c_up < best_c:
                best_h, best_c = h_up, c_up

    # Phase 3: Multi-start with perturbations
    n_starts = 40
    for s in range(n_starts):
        if time.time() > deadline - 60:
            break
        if best_c <= 0.38075:
            break
        typ = s % 5
        h0 = perturb(best_h, n, rng, typ)
        if s % 5 == 0 and n <= 500 and time.time() < deadline - 200:
            h0, _ = slsqp_step(h0, n, (100, 1000), 40, deadline)
        h1, c1 = cd_optimize(h0, n, 6, deadline)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 4: Final refinement
    if time.time() < deadline - 30:
        best_h, best_c = cd_optimize(best_h, n, 20, deadline)
    if time.time() < deadline - 15 and n <= 500:
        h3, c3 = slsqp_step(best_h, n, (500, 5000, 50000), 80, deadline)
        if c3 < best_c:
            best_h, best_c = h3, compute_c(h3, n)
            best_h, best_c = cd_optimize(best_h, n, 10, deadline)

    best_h = project_feasible(best_h, n)
    best_c = compute_c(best_h, n)
    return best_h, best_c, n
