import numpy as np
from scipy.optimize import golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def nesterov_momentum(h, n, max_iter=3000, deadline=None, m_top=5):
    h = h.copy().astype(float)
    h_prev = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    beta = 0.85
    lr = 0.03
    stall = 0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(c.max())
        if c_max < best_c - 1e-14:
            best_c = c_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr *= 0.7
                beta = min(0.95, beta + 0.02)
                stall = 0
                if lr < 1e-9:
                    break
        top_m_idx = np.argsort(c)[-m_top:]
        sg = np.zeros(n)
        for k in top_m_idx:
            lo1 = max(0, k - (n - 1))
            hi1 = min(n - 1, k)
            if hi1 >= lo1:
                idx = np.arange(lo1, hi1 + 1) - k + (n - 1)
                sg[lo1:hi1 + 1] += (1.0 - h[idx])
            lo2 = max(0, n - 1 - k)
            hi2 = min(n - 1, 2 * n - 2 - k)
            if hi2 >= lo2:
                idx2 = np.arange(lo2, hi2 + 1) + k - (n - 1)
                sg[lo2:hi2 + 1] -= h[idx2]
        sg /= m_top
        momentum = beta * (h - h_prev)
        h_prev = h.copy()
        h_new = project_feasible(h - lr * sg + momentum, n)
        if np.abs(h_new - h).max() < 1e-14:
            lr *= 0.5
            if lr < 1e-10:
                break
        else:
            h = h_new
        if it % 500 == 0 and c_max > best_c + 1e-6:
            lr *= 0.8
    return h, best_h, min(compute_c5(h, n), best_c)

def obj_1d(d, c_aff, a_coeff, diag_local, unaff_max):
    nv = c_aff + d * a_coeff
    if diag_local is not None:
        nv[diag_local] -= d * d
    return max(float(nv.max()), unaff_max)

def coord_descent(h, n, max_passes=200, deadline=None, grid_size=80):
    h = h.copy()
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            f_lo = max(0, n - 1 - p)
            f_hi = min(2 * n - 2, n - 1 + p)
            b_lo = max(0, p)
            b_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(f_lo, b_lo)
            m_hi = max(f_hi, b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            a_coeff = np.zeros(len(ms))
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a_coeff[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):

            c_aff = c[ms]

            lo = -h0

            vals = np.empty(grid_size)
            for gi in range(grid_size):

            best_idx = int(np.argmin(vals))
            best_d = float(d_grid[best_idx])
            best_val = float(vals[best_idx])

        if len(ones) < 2 or len(zeros) < 2:

        c_try = compute_c5(h_try, n)
        if c_try < c_max + 0.001:
            h_bin = h_try
            ones = np.where(h_bin == 1.0)[0]

    return best_h, best_c

    n_cur = len(best_h)
    if time.time() < deadline - 30:
        hf, cf = coord_descent(best_h, n_cur, 500, deadline - 10, 150)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Phase 7: One more Nesterov + coord pass
    if time.time() < deadline - 15:
        ho, bho, co = nesterov_momentum(best_h, n_cur, 1500, None, 10)
        if time.time() < deadline - 8:
            ho2, co2 = coord_descent(bho, n_cur, 100, deadline - 5, 100)
            if co2 < co:
                ho, co = ho2, co2
        if co < best_c:
            best_h, best_c = ho, co
    # Symmetry check
    n_cur = len(best_h)
    h_flip = project_feasible(1.0 - best_h, n_cur)
    c_flip = compute_c5(h_flip, n_cur)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
