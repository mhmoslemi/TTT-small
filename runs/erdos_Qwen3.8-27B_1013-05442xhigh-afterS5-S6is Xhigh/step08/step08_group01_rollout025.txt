    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def fast_coord_descent(h, n, max_passes=500, deadline=None):
    """Coordinate descent exploiting linearity of c[k] in each h[p]."""
    h = h.copy()
    dx = 2.0 / n
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            
            # Determine range of m (correlation indices) affected by h[p]
            # h[p] appears in correlation at index m when:
            #   forward: i=p, shift k=m-(n-1), need 0<=p-k<n => k in [p-n+1, p]
            #   backward: i-k=p => i=p+k, need 0<=p+k<n => k in [-p, n-p-1]
            k_lo = max(p-n+1, -p)
            k_hi = min(p, n-p-1)
            if k_lo > k_hi:
                continue
            
            m_lo = k_lo + n - 1
            m_hi = k_hi + n - 1
            ms = np.arange(m_lo, m_hi + 1)
            ks = ms - (n - 1)
            
            # For each affected m, c[m] as function of x=h[p]:
            # c[m] = x*(1-h[p-k]) + h[p+k]*(1-x) + rest  (for k≠0)
            #      = x*(1-h[p-k]-h[p+k]) + h[p+k] + rest
            # For k=0: c[m] = x*(1-x) + rest = -x²+x+rest
            # So: c[m](x) = -q*m_is_zero*x² + a*x + b
            # where m_is_zero is 1 if k=0, else 0
            
            a = np.zeros(len(ms))
            b = np.zeros(len(ms))
            is_k0 = (ks == 0)
            
            for idx in range(len(ms)):
                k = ks[idx]
                if k == 0:
                    # c[m0] = h[p]*(1-h[p]) + sum_{i≠p,valid} h[i]*(1-h[i])
                    # = h[p] - h[p]² + base
                    # base = c[m0] - h0*(1-h0)
                    b[idx] = c[m_lo + idx] - h0 * (1.0 - h0)
                    a[idx] = 0.0  # linear part (quadratic handled separately)
                else:
                    j1 = p - k
                    j2 = p + k
                    coeff = 0.0
                    if 0 <= j1 < n:
                        coeff += (1.0 - h[j1])
                    if 0 <= j2 < n:
                        coeff -= h[j2]
                    a[idx] = coeff
                    b[idx] = c[m_lo + idx] - h0 * coeff
            
            # Unaffected max
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2*n-2:
                unaff = max(unaff, float(c[m_hi+1:].max()))
            
            # Find optimal x in [0, 1] (since h[p] directly, not d)
            # c[m](x) = -is_k0[m]*x² + a[m]*x + b[m]
            # Max over m, plus unaff
            
            # Strategy: try a set of candidate x values
            # 1. Boundaries: 0, 1
            # 2. Where the k=0 term is max: x=0.5 (if present)
            # 3. Where top-2 linear terms cross
            
            candidates = [0.0, 1.0]
            if np.any(is_k0):
                candidates.append(0.5)
            
            # Find top shifts at x = h0
            vals_at_h0 = np.zeros(len(ms))
            for idx in range(len(ms)):
                if is_k0[idx]:
                    vals_at_h0[idx] = h0*(1-h0) + b[idx]
                else:
                    vals_at_h0[idx] = a[idx]*h0 + b[idx]
            
            # Get top 4
            n_top = min(4, len(ms))
            top_idx = np.argsort(vals_at_h0)[-n_top:]
            
            for i in range(len(top_idx)):
                for j in range(i+1, len(top_idx)):
                    i1, i2 = top_idx[i], top_idx[j]
                    if is_k0[i1] and is_k0[i2]:
                        continue
                    if is_k0[i1] or is_k0[i2]:
                        k0_i = i1 if is_k0[i1] else i2
                        other = i2 if is_k0[i1] else i1
                        # -x²+x+b_k0 = a_other*x + b_other
                        # x² + (a_other-1)*x + (b_other - b_k0) = 0
                        A2 = 1.0
                        B2 = a[other] - 1.0
                        C2 = b[other] - b[k0_i]
                        disc = B2*B2 - 4*C2
                        if disc >= 0:
                            s = np.sqrt(disc)
                            candidates.append((-B2 + s) / 2.0)
                            candidates.append((-B2 - s) / 2.0)
                    else:
                        da = a[i1] - a[i2]
                        if abs(da) > 1e-14:
                            candidates.append((b[i2] - b[i1]) / da)
            
            # Evaluate
            best_x = h0
            best_val = current_max
            for xc in candidates:
                xc = max(0.0, min(1.0, xc))
                # Compute max c at this x
                c_lin = a * xc + b
                if np.any(is_k0):
                    c_lin[is_k0] = xc * (1.0 - xc) + b[is_k0]
                val = max(float(c_lin.max()), unaff)
                if val < best_val - 1e-14:
                    best_val = val
                    best_x = xc
            
            if best_x != h0:
                h[p] = best_x
                improved = True
                current_max = best_val
        
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def binary_sa(h0, n, max_iter=50000, deadline=None, T0=0.01):
    """Simulated annealing on binary pattern."""
    # Start from h0 by thresholding
    h = (h0 > 0.5).astype(float)
    target = n // 2
    # Adjust to have exactly target ones
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    while len(ones) > target:
        idx = ones[np.random.randint(len(ones))]
        h[idx] = 0
        ones = np.where(h == 1)[0]
    while len(ones) < target:
        idx = zeros[np.random.randint(len(ones) + len(zeros) if False else len(zeros))]
        h[idx] = 1
        ones = np.where(h == 1)[0]
    
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(c.max())
    cur_c = best_c
    dx = 2.0 / n
    T = T0
    best_h = h.copy()
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        # Random move: flip a random position
        p = np.random.randint(n)
        old_val = h[p]
        h[p] = 1.0 - old_val
        c2 = np.correlate(h, 1.0 - h, mode='full')
        new_c = float(c2.max())
        delta = new_c - cur_c
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[p] = old_val
        T *= 0.9999
    return best_h, best_c * dx

Wait, I realize the binary SA approach might not be efficient enough. Recomputing the full correlation at each step is O(n) per step (well, O(n) for the correlate, but we only need to update the affected entries).

Let me optimize the SA by only updating the affected correlation entries.

Actually, for a flip at position p (h[p] changes by δ = ±1):
The affected correlation entries are those where p is involved.
For shift k (m = k + n - 1):
- Forward: i=p contributes h[p]*(1-h[p-k]) → changes by δ*(1-h[p-k])
- Backward: i-k=p (i=p+k) contributes h[p+k]*(1-h[p]) → changes by -δ*h[p+k]

So for each shift k, the change in c[m] is:
Δc[m] = δ*(1-h[p-k]) - δ*h[p+k] = δ*(1-h[p-k]-h[p+k]) (for k≠0)
For k=0: Δc[m] = δ*(1-2*h[p]) ... wait, let me be more careful.

Before flip: h[p] = v, c[m] includes v*(1-h[p-k]) + h[p+k]*(1-v)
After flip: h[p] = 1-v, c'[m] includes (1-v)*(1-h[p-k]) + h[p+k]*v

Difference: (1-v)*(1-h[p-k]) + h[p+k]*v - v*(1-h[p-k]) - h[p+k]*(1-v)
= (1-v)(1-h[p-k]) - v(1-h[p-k]) + h[p+k]v - h[p+k](1-v)
= (1-2v)(1-h[p-k]) + h[p+k](2v-1)
= (1-2v)[(1-h[p-k]) - h[p+k]]
= (1-2v)(1-h[p-k]-h[p+k])

For v=0: Δ = 1*(1-h[p-k]-h[p+k])
For v=1: Δ = -1*(1-h[p-k]-h[p+k]) = h[p-k]+h[p+k]-1

def smooth_slsqp(h, n, alpha, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        if alpha < 1:
            return m * dx
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / alpha) * dx
    
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        if alpha < 1:
            # subgradient
            g = np.zeros(2*n-1)
            g[c == m] = 1.0
            g /= g.sum()
            # dc/dx for each component
            grad = np.zeros(n)
            for k in range(2*n-1):
                if g[k] == 0:
                    continue
                shift = k - (n-1)
                for i in range(n):
                    i2 = i - shift
                    if 0 <= i2 < n:
                        grad[i] += g[k] * dx * (1 - x[i2])
                    i3 = i + shift
                    if 0 <= i3 < n:
                        grad[i] -= g[k] * dx * x[i3]
            return m * dx
        # smooth version gradient is complex, skip
        return None  # let SLSQP use numerical
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n), compute_c5(project_feasible(res.x, n), n)
    except:
        return h, compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_cd_fast(h, n, max_passes=1000, deadline=None, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        order = rng.permutation(n)
        for p in order:
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            k_lo = max(p - n + 1, -p)
            k_hi = min(p, n - p - 1)
            if k_lo > k_hi:
                continue
            m_lo = k_lo + n - 1
            m_hi = k_hi + n - 1
            n_m = m_hi - m_lo + 1
            ms = np.arange(m_lo, m_hi + 1)
            ks = ms - (n - 1)
            a = np.zeros(n_m)
            b = np.zeros(n_m)
            has_k0 = (k_lo <= 0 <= k_hi)
            for idx in range(n_m):
                k = ks[idx]
                mi = m_lo + idx
                if k == 0:
                    b[idx] = c[mi] - h0 * (1.0 - h0)
                else:
                    j1 = p - k
                    j2 = p + k
                    coeff = 0.0
                    if 0 <= j1 < n:
                        coeff += 1.0 - h[j1]
                    if 0 <= j2 < n:
                        coeff -= h[j2]
                    a[idx] = coeff
                    b[idx] = c[mi] - h0 * coeff
            unaff = 0.0
            if m_lo > 0:
                unaff = float(c[:m_lo].max())
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            # Build candidate x values
            cands = [0.0, 1.0]
            if has_k0:
                cands.append(0.5)
            # Find values at h0
            if has_k0:
                k0_idx = 0 - k_lo
                vals = a * h0 + b
                vals[k0_idx] = h0 * (1.0 - h0) + b[k0_idx]
                # Crossings involving k0
                for idx in range(n_m):
                    if idx == k0_idx:
                        continue
                    B2 = a[idx] - 1.0
                    C2 = b[idx] - b[k0_idx]
                    disc = B2 * B2 - 4.0 * C2
                    if disc >= 0:
                        s = np.sqrt(disc)
                        cands.append((-B2 + s) / 2.0)
                        cands.append((-B2 - s) / 2.0)
            else:
                vals = a * h0 + b
            # Top crossings among linear terms
            n_top = min(6, n_m)
            top = np.argsort(vals)[-n_top:]
            for i in range(len(top)):
                for j in range(i + 1, len(top)):
                    i1, i2 = top[i], top[j]
                    if has_k0 and (i1 == 0 - k_lo or i2 == 0 - k_lo):
                        continue
                    da = a[i1] - a[i2]
                    if abs(da) > 1e-14:
                        cands.append((b[i2] - b[i1]) / da)
            # Evaluate candidates
            best_x = h0
            best_val = current_max
            for xc in cands:
                if xc < 0.0 or xc > 1.0:
                    continue
                c_new = a * xc + b
                if has_k0:
                    ki = 0 - k_lo
                    c_new[ki] = xc * (1.0 - xc) + b[ki]
                val = max(float(np.max(c_new)), unaff)
                if val < best_val - 1e-13:
                    best_val = val
                    best_x = xc
            if best_x != h0:
                h[p] = best_x
                improved = True
                current_max = best_val
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def slsqp_smooth(h, n, alpha, maxiter=400):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / alpha) * dx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h, compute_c5(h, n)
