def nesterov_smooth(h, n, alpha_schedule, max_iter, deadline=None):
    """Projected Nesterov gradient descent on smooth max."""
    h = h.copy()
    y = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    t = 1.0
    for it in range(max_iter):

        alpha = alpha_schedule[min(it, len(alpha_schedule)-1)]
        c = np.correlate(h, 1.0 - h, mode='full')
        grad = compute_grad_smooth(h, n, alpha)
        
        # Nesterov momentum
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        beta = (t - 1) / t_new

        # Use backtracking line search
        lr = 0.5 / alpha  # initial step size (related to smoothness)
        y_new = y - lr * grad
        y_new = project_feasible(y_new, n)
        
        # Check if objective improved
        c_new = compute_c5(y_new, n)
        if c_new < best_c:
            h = y_new
            best_c = c_new
            best_h = h.copy()
        else:
            # Backtrack
            for _ in range(10):
                lr *= 0.5
                y_new = y - lr * grad

                    if c_new < best_c:
