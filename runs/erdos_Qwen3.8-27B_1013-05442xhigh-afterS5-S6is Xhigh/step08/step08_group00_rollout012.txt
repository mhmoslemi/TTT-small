        best_h, best_c, best_n = h1, c1, len(h1)
    # Phase 2: precise CD
    if time.time() < deadline - 30:
        h2, c2 = coord_cd_precise(best_h, best_n, 30, deadline)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, len(h2)
    # Phase 3: try other n values
    x_orig = np.linspace(0, 2, len(best_h), endpoint=False)
    for n_try in [100, 150, 200, 250, 300, 400, 500, 600, 800, 1000]:
        if time.time() > deadline - 100:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = coord_cd_grid(h_int, n_try, 50, deadline, grid_sz=50)
        if time.time() < deadline - 20:
            h_o, c_o = coord_cd_precise(h_o, n_try, 15, deadline)
        if c_o < best_c:
            best_h, best_c, best_n = h_o, c_o, n_try
            x_orig = np.linspace(0, 2, n_try, endpoint=False)
    # Phase 4: structured inits at various n
    for n_s in [200, 300, 400, 500]:
        if time.time() > deadline - 80:
            break
        for freq in [1, 2, 3, 4]:
            if time.time() > deadline - 60:
                break
            for amp in [0.3, 0.45, 0.49]:
                if time.time() > deadline - 40:
                    break
                hs = structured_init(n_s, freq, amp, 0.0)
                hs2, cs2 = coord_cd_grid(hs, n_s, 30, deadline, grid_sz=40)
                if time.time() < deadline - 10:
                    hs2, cs2 = coord_cd_precise(hs2, n_s, 10, deadline)
                if cs2 < best_c:
                    best_h, best_c, best_n = hs2, cs2, n_s
    # Phase 5: multi-restart from best
    if time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(1, int(remaining / 80))
        for r in range(n_restarts):
            if time.time() > deadline - 40:
                break
            s = 0.01 + 0.06 * (r + 1) / max(1, n_restarts)
            hp = project_feasible(best_h + rng.normal(0, s, n), n)
            ho, co = coord_cd_grid(hp, n, 40, deadline, grid_sz=50)
            if time.time() < deadline - 10:
                ho, co = coord_cd_precise(ho, n, 15, deadline)
            if co < best_c:
                best_h, best_c, best_n = ho, co, n
    # Phase 6: final polish
    if time.time() < deadline - 5:
        n = len(best_h)
        hf, cf = coord_cd_precise(best_h, n, 20, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
