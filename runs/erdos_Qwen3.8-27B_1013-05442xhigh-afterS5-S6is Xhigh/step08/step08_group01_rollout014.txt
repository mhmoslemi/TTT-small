import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # Soft max
    m = c.max()
    exp_c = np.exp(alpha * (c - m))
    w = exp_c / exp_c.sum()
    obj = (m + np.log(exp_c.sum())) / alpha
    # Gradient: sum_m w[m] * dc[m]/dh[p] * dx
    # dc[m]/dh[p] = (1-h[p+k]) * 1_{p+k valid} - h[p-k] * 1_{p-k valid}
    # where k = m - (n-1)
    grad = np.zeros(n)
    for m in range(2*n - 1):
        if w[m] < 1e-15:
            continue
        k = m - (n - 1)
        # Forward: p in [0,n), p+k in [0,n)
        p_lo_f = max(0, -k)
        p_hi_f = min(n - 1, n - 1 - k)
        if p_lo_f <= p_hi_f:
            ps = np.arange(p_lo_f, p_hi_f + 1)
            grad[ps] += w[m] * dx * (1.0 - h[ps + k])
        # Backward: p in [0,n), p-k in [0,n)
        p_lo_b = max(0, k)
        p_hi_b = min(n - 1, n - 1 + k)
        if p_lo_b <= p_hi_b:
            ps = np.arange(p_lo_b, p_hi_b + 1)
            grad[ps] -= w[m] * dx * h[ps - k]
    return obj, grad

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

def smooth_obj_grad(h, n, alpha, mu):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    ac = alpha * (c - m)
    e = np.exp(ac)
    se = e.sum()
    w = e / se
    obj = (m + np.log(se)) / alpha + mu * (h.sum() - n/2.0)**2
    
    grad = np.zeros(n)
    # Vectorized gradient computation
    for m_idx in range(2*n - 1):
        wm = w[m_idx]
        if wm < 1e-14:
            continue
        k = m_idx - (n - 1)
        # Forward: p in [max(0,-k), min(n-1, n-1-k)]
        p_lo = max(0, -k)
        p_hi = min(n-1, n-1-k)
        if p_lo <= p_hi:
            grad[p_lo:p_hi+1] += wm * dx * (1.0 - h[p_lo+k:p_hi+k+1])
        # Backward: p in [max(0,k), min(n-1, n-1+k)]
        p_lo_b = max(0, k)
        p_hi_b = min(n-1, n-1+k)
        if p_lo_b <= p_hi_b:
            grad[p_lo_b:p_hi_b+1] -= wm * dx * h[p_lo_b-k:p_hi_b-k+1]
    
    grad += 2.0 * mu * (h.sum() - n/2.0)
    return obj, grad

        h3, c3 = lbfgsb_phase(best_h, n, [5000, 20000, 100000], 200.0, 200, deadline - 60)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: Coordinate descent again
    if time.time() < deadline - 60:
        h4, c4 = coord_cd(best_h, n, 300, deadline - 30, grid_size=80)
        if c4 < best_c:
            best_h, best_c = h4, c4
    # Phase 5: Anti-symmetric exploration
    h_as = antisym_project(best_h, n)
    h_as = project_feasible(h_as, n)
    c_as = compute_c5(h_as, n)
    if c_as < best_c and time.time() < deadline - 80:
        h_as2, c_as2 = coord_cd(h_as, n, 150, deadline - 30, grid_size=50)
        if c_as2 < best_c:
            best_h, best_c = h_as2, c_as2
    # Phase 6: Multi-resolution
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [400, 800, 1200]:
        if time.time() > deadline - 40:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = lbfgsb_phase(h_int, n_try, [200, 2000, 10000], 100.0, 100, deadline - 30)
        h_opt, c_opt = coord_cd(h_opt, n_try, 80, deadline - 20, grid_size=40)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new
    # Phase 7: Perturbation restarts
    n_cur = len(best_h)
    for r in range(4):
        if time.time() > deadline - 20:
            break
        s = 0.01 + 0.03 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
        h_o, c_o = lbfgsb_phase(hp, n_cur, [1000, 10000], 200.0, 100, deadline - 15)
        h_o, c_o = coord_cd(h_o, n_cur, 60, deadline - 10, grid_size=40)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
    # Phase 8: Final high-precision coordinate descent
    n_cur = len(best_h)
    if time.time() < deadline - 5:
        hf, cf = coord_cd(best_h, n_cur, 400, deadline, grid_size=100)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Final: try 1-h symmetry
    h_flip = project_feasible(1.0 - best_h, n_cur)
    c_flip = compute_c5(h_flip, n_cur)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
