from scipy.optimize import golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_cd(h, n, deadline, max_passes=100):
    """Fast coordinate descent exploiting linearity of c_k in each h[p]."""
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        # Process coordinates in random order for diversity
        order = np.random.permutation(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            # Compute the linear coefficients: c_k = a_k*h[p] + b_k
            # c_k = sum_i h[i]*(1-h[i+k])
            # Term with h[p]: h[p]*(1-h[p+k]) if p+k valid, and h[p-k]*(1-h[p]) if p-k valid
            # So c_k = b_k + h[p]*a_k where:
            #   a_k = (1-h[p+k]) if p+k<n, else 0  [forward]
            #        - h[p-k] if p-k>=0, else 0    [backward]
            #   b_k = c_k - h0*a_k
            # Actually let me just compute the full linear function directly.
            
            # For correlation at lag m (0-indexed in the 'full' output):
            # c[m] = sum_i h[i] * (1-h[i+m-(n-1)])
            # where the inner index must be in [0, n-1]

            #   i = p: need p + m - (n-1) in [0,n-1], i.e., m-(n-1) in [-p, n-1-p]
            #     contributes: 1 - h[p + m - (n-1)]
            #   i + m - (n-1) = p: i = p - m + (n-1), need this in [0,n-1]
            #     contributes: h[p-m+(n-1)] * (-1) [since (1-h[p]) = 1 - h[p]]
            
            # So a[m] = (1 - h[p+m-(n-1)]) * [p+m-(n-1) in [0,n-1]]
            #           - h[p-m+(n-1)] * [p-m+(n-1) in [0,n-1]]
            # and b[m] = c[m] - h0 * a[m]
            
            # Range of m where h[p] appears:
            # Forward: p+m-(n-1) in [0,n-1] => m in [n-1-p, n-1+p]
            # Backward: p-m+(n-1) in [0,n-1] => m in [p, p+n-1]
            # Combined: m in [min(n-1-p, p), max(n-1+p, p+n-1)]
            
            m_lo = min(n - 1 - p, p)
            m_hi = max(n - 1 + p, p + n - 1)
            ms = np.arange(m_lo, m_hi + 1)
            
            a = np.zeros(len(ms))
            # Forward contribution
            jf = p + ms - (n - 1)
            vf = (jf >= 0) & (jf < n)
            a[vf] += (1.0 - h[jf[vf]])
            # Backward contribution
            jb = p - ms + (n - 1)
            vb = (jb >= 0) & (jb < n)
            a[vb] -= h[jb[vb]]
            
            b = c[ms] - h0 * a
            
            # Now minimize max(a[m]*d + b[m], max(uncorrelated))
            # where d = h[p] - h0, so h[p] = h0 + d
            # d in [-h0, 1-h0]

            if m_lo > 0:
                unaff = float(c[:m_lo].max())
            else:
                unaff = 0.0
            if m_hi < 2 * n - 2:
                unaff2 = float(c[m_hi + 1:].max())
                unaff = max(unaff, unaff2)
            
            # Find optimal d: minimize max(max_m(a[m]*d + b[m]), unaff)
            # This is a 1D convex piecewise-linear minimization
            # The optimum is where the upper envelope crosses the line y = unaff
            # or at a breakpoint of the upper envelope

            lo = -h0
            hi = 1.0 - h0

            n_grid = 40

            vals = np.max(np.c_[(a * d_grid[:, None] + b[None, :]).max(axis=1),

            best_idx = int(np.argmin(vals))

def momentum_subgrad(h, n, deadline, max_iter=500):
    """Subgradient with momentum, focusing on the worst lag."""
    h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    beta = 0.9
    lr = 0.02
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m_star = int(np.argmax(c))
        # Subgradient of c_{m_star} w.r.t. h
        sg = np.zeros(n)
        for i in range(n):
            j = i + m_star - (n - 1)
            if 0 <= j < n:
                sg[i] += (1.0 - h[j])
            ip = i - m_star + (n - 1)
            if 0 <= ip < n:
                sg[i] -= h[ip]
        v = beta * v - lr * sg
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            h = h_new
            best_c = c_new
            stall = 0
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.8
            stall += 1
            if stall > 40 or lr < 1e-10:
                break
    return h, best_c

def structured_perturbation(h, n, rng, mode):
    """Apply a structured perturbation."""
    h2 = h.copy()
    if mode == 0:
        # Block complement
        block_size = rng.randint(5, 25)

    elif mode == 1:

    elif mode == 2:

        i, j = rng.choice(n, 2, replace=False)
        h2[i], h2[j] = h2[j], h2[i]
    elif mode == 3:

        shift = rng.randint(1, 10)

        h2[mask] = 0.5 * (h2[mask] + np.roll(h2, shift)[mask])
    elif mode == 4:

    else:

        amp = rng.uniform(0.02, 0.1)
        h2 += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    return h2

import numpy as np
from scipy.optimize import golden
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _obj_1d(d, a, b, unaff):
    v = float((a * d + b).max())
    return max(v, unaff)

def coord_cd(h, n, deadline, max_passes=200, grid_size=50):
    h = h.copy()
    rng_cd = np.random.RandomState(42)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        order = rng_cd.permutation(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_lo = min(n - 1 - p, p)
            m_hi = max(n - 1 + p, p + n - 1)
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            jf = p + ms - (n - 1)
            vf = (jf >= 0) & (jf < n)
            a[vf] += (1.0 - h[jf[vf]])
            jb = p - ms + (n - 1)
            vb = (jb >= 0) & (jb < n)
            a[vb] -= h[jb[vb]]
            b = c[ms] - h0 * a
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            lo = -h0
            hi = 1.0 - h0
            d_grid = np.linspace(lo, hi, grid_size)
            vals = np.array([_obj_1d(d, a, b, unaff) for d in d_grid])
            best_idx = int(np.argmin(vals))
            best_val = float(vals[best_idx])
            if 0 < best_idx < grid_size - 1:
                gl = d_grid[best_idx - 1]
                gh = d_grid[best_idx + 1]
                try:
                    res = golden(_obj_1d, gl, gh, args=(a, b, unaff), maxiter=50)
                    if res.fun < best_val:
                        best_val = res.fun
                        best_idx = best_idx  # use improved value
                except Exception:
                    pass
            if best_val < current_max - 1e-14:
                new_hp = float(np.clip(h0 + d_grid[best_idx], 0.0, 1.0))
                if 0 < best_idx < grid_size - 1 and best_val < vals[best_idx] - 1e-12:
                    new_hp = new_hp  # golden section result already applied via best_val
                h[p] = new_hp
                improved = True
                current_max = min(current_max, best_val)
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def momentum_subgrad(h, n, deadline, max_iter=400):
    h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    beta = 0.9
    lr = 0.015
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m_star = int(np.argmax(c))
        sg = np.zeros(n)
        j_fwd = np.arange(n) + m_star - (n - 1)
        valid_fwd = (j_fwd >= 0) & (j_fwd < n)
        sg[valid_fwd] += (1.0 - h[j_fwd[valid_fwd]])
        j_bwd = np.arange(n) - m_star + (n - 1)
        valid_bwd = (j_bwd >= 0) & (j_bwd < n)
        sg[valid_bwd] -= h[j_bwd[valid_bwd]]
        v = beta * v - lr * sg
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            h = h_new
            best_c = c_new
            stall = 0
            lr = min(lr * 1.03, 0.08)
        else:
            lr *= 0.85
            stall += 1
            if stall > 50 or lr < 1e-11:
                break
    return h, best_c

def perturb_block_complement(h, n, rng, scale=1.0):
    h2 = h.copy()
    n_blocks = max(1, int(2 * scale))
    for _ in range(n_blocks):
        bs = rng.randint(3, int(20 * scale) + 3)
        if bs >= n:
            continue
        start = rng.randint(0, n - bs)
        h2[start:start + bs] = 1.0 - h2[start:start + bs]
    return h2

def perturb_swap(h, n, rng, n_swaps=3):
    h2 = h.copy()
    for _ in range(n_swaps):
        i, j = rng.choice(n, 2, replace=False)
        h2[i], h2[j] = h2[j], h2[i]
    return h2

def perturb_smooth(h, n, rng):
    h2 = h.copy()
    shift = rng.randint(1, 8)
    width = rng.randint(n // 10, n // 3)
    start = rng.randint(0, n - width)
    mask = slice(start, start + width)
    h2[mask] = 0.5 * (h2[mask] + np.roll(h2, shift)[mask])
    return h2

def perturb_freq(h, n, rng):
    h2 = h.copy()
    freq = rng.randint(1, max(2, n // 5))
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.02, 0.12)
    h2 += amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
    return h2

def perturb_gaussian(h, n, rng, scale):
    return h + rng.normal(0, scale, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    np.random.seed(seed)
    deadline = t0 + budget_s - 6

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Converge initial solution
    if time.time() < deadline - 200:
        best_h, best_c = coord_cd(best_h, n, deadline - 180, max_passes=60, grid_size=60)

    # Phase 2: Iterated local search with diverse perturbations
    pert_list = [
        lambda h2: perturb_block_complement(h2, n, rng, scale=1.0),
        lambda h2: perturb_block_complement(h2, n, rng, scale=0.5),
        lambda h2: perturb_swap(h2, n, rng, n_swaps=5),
        lambda h2: perturb_smooth(h2, n, rng),
        lambda h2: perturb_freq(h2, n, rng),
        lambda h2: perturb_gaussian(h2, n, rng, 0.05),
    ]
    # First round: larger perturbations
    if time.time() < deadline - 120:
        for attempt in range(12):
            if time.time() > deadline - 60:
                break
            h_pert = pert_list[attempt % len(pert_list)](best_h)
            h_pert = project_feasible(h_pert, n)
            h_new, c_new = coord_cd(h_pert, n, deadline - 60, max_passes=35, grid_size=50)
            if c_new < best_c:
                best_h, best_c = h_new, c_new

    # Phase 3: Momentum subgradient
    if time.time() < deadline - 40:
        h3, c3 = momentum_subgrad(best_h, n, deadline - 30)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Finer perturbations
    if time.time() < deadline - 30:
        for attempt in range(10):
            if time.time() > deadline - 15:
                break
            h_pert = pert_list[(attempt + 3) % len(pert_list)](best_h)
            h_pert = project_feasible(h_pert, n)
            h_new, c_new = coord_cd(h_pert, n, deadline - 15, max_passes=40, grid_size=60)
            if c_new < best_c:
                best_h, best_c = h_new, c_new

    # Phase 5: High-precision final coordinate descent
    if time.time() < deadline - 5:
        best_h, best_c = coord_cd(best_h, n, deadline - 2, max_passes=150, grid_size=80)

    # Try complement
    h_flip = project_feasible(1.0 - best_h, n)
    c_flip = compute_c5(h_flip, n)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
