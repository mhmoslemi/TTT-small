        if abs(diff) < 1e-15:

        if diff > 0:

            if not mask.any():

        hf, cf = coord_cd(best_h, len(best_h), max_passes=maxp, deadline=deadline - 2)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
