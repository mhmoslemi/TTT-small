            m_hi = max(m_f_hi, m_b_hi)
            ms = np.arange(m_lo, m_hi + 1)
            a_coeff = np.zeros(len(ms))
            jf_arr = p + ms - (n - 1)
            jb_arr = p - ms + (n - 1)
            vf = (jf_arr >= 0) & (jf_arr < n)
            vb = (jb_arr >= 0) & (jb_arr < n)
            if np.any(vf):
                a_coeff[vf] += 1.0 - h[jf_arr[vf]]
            if np.any(vb):
                a_coeff[vb] -= h[jb_arr[vb]]
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            diag_local = None
            if m_lo <= (n - 1) <= m_hi:
                diag_local = (n - 1) - m_lo
            lo, hi = -h0, 1.0 - h0
            d_grid = np.linspace(lo, hi, grid_size)
            vals = np.array([obj_1d(d, c_aff, a_coeff, diag_local, unaff_max) for d in d_grid])
            best_idx = int(np.argmin(vals))
            best_d = d_grid[best_idx]
            best_val = float(vals[best_idx])
            if 0 < best_idx < grid_size - 1:
                from scipy.optimize import golden
                gl, gh = d_grid[best_idx - 1], d_grid[best_idx + 1]
                try:
                    res = golden(obj_1d, gl, gh, args=(c_aff, a_coeff, diag_local, unaff_max), maxiter=50)
                    if res.fun < best_val:
                        best_d, best_val = float(res.x), float(res.fun)
                except Exception:
                    pass
            if best_val < current_max - 1e-15:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
                current_max = min(current_max, best_val)
        if not improved:
            break
        h = project_feasible(h, n / 2.0)
    return h, compute_c5(h, n)

def lbfgsb_smooth(h, n, alpha, maxiter=300):
    penalty = 1e12
    def obj(x):
        return smooth_c5_obj(x, n, alpha) + penalty * (x.sum() - n / 2.0) ** 2
    def grad(x):
        return smooth_c5_grad(x, n, alpha) + 2.0 * penalty * (x.sum() - n / 2.0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12, 'maxls': 50})
        h_new = project_feasible(res.x, n / 2.0)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n / 2.0)
    best_c = compute_c5(best_h, n)

    # Phase 1: L-BFGS-B with multiple alphas
    alphas = [100, 500, 2000, 10000, 50000]
    for a in alphas:
        if time.time() > deadline - 300:
            break
        h1, c1 = lbfgsb_smooth(best_h, n, a, maxiter=200)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Projected gradient with Nesterov momentum
    for a in [500, 5000, 50000]:
        if time.time() > deadline - 250:
            break
        h2, f2 = projected_gradient(best_h, n, a, max_iter=3000, deadline=deadline)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Coordinate descent (main workhorse)
    if time.time() < deadline - 150:
        h3, c3 = coord_cd(best_h, n, max_passes=300, deadline=deadline, grid_size=80)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Multi-n exploration
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [300, 400, 500, 700, 900, 1200]:
        if time.time() > deadline - 120:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try / 2.0)
        h_opt, c_opt = lbfgsb_smooth(h_int, n_try, 2000, maxiter=150)
        h_opt, c_opt = coord_cd(h_opt, n_try, 80, deadline, grid_size=60)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new

    # Phase 5: Multi-restart with structured perturbations
    n_cur = len(best_h)
    for r in range(8):
        if time.time() > deadline - 80:
            break
        s = 0.01 + 0.03 * r
        hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur / 2.0)
        h_o, c_o = lbfgsb_smooth(hp, n_cur, 5000, maxiter=100)
        h_o, c_o = coord_cd(h_o, n_cur, 50, deadline, grid_size=50)
        if c_o < best_c:
            best_h, best_c = h_o, c_o

    # Phase 6: Final high-precision refinement
    n_cur = len(best_h)
    if time.time() < deadline - 40:
        hf, cf = lbfgsb_smooth(best_h, n_cur, 100000, maxiter=500)
        if cf < best_c:
            best_h, best_c = hf, cf
    if time.time() < deadline - 15:
        hf, cf = coord_cd(best_h, n_cur, 400, deadline, grid_size=100)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Check 1-h symmetry
    h_flip = project_feasible(1.0 - best_h, len(best_h) / 2.0)
    c_flip = compute_c5(h_flip, len(best_h))
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
