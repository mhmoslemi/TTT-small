    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha, mx in [(100, 50), (500, 30), (2000, 20)]:
        if time.time() > deadline:
            break
        try:
            res = minimize(pen_val, best_h, args=(n, alpha, mu), method='L-BFGS-B',
                          jac=pen_grad, bounds=bounds,
                          options={'maxiter': mx, 'ftol': 1e-15, 'maxls': 30})
            x = project_feasible(np.asarray(res.x, dtype=float), n)
            c = compute_c5(x, n)
            if c < best_c:
                best_h, best_c = x, c
        except Exception:
            pass
    return best_h, best_c

def build_A(h, n):
    A = np.zeros((n, 2*n-1))
    off = n - 1
    for k in range(2*n-1):
        d = k - off
        if d >= 0:
            A[d:, k] = 1.0 - h[:n-d]
            A[:n-d, k] = -h[d:]
        else:
            e = -d
            A[:n-e, k] = 1.0 - h[e:]
            A[e:, k] = -h[:n-e]
    return A

def pair_search(h, n, max_iter, deadline, rng, n_grid=41):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    off = n - 1
    A = build_A(h, n)
    cur_max = float(c.max())
    T = 0.008 * cur_max
    accepts = 0
    for trial in range(max_iter):
        if time.time() > deadline:
            break
        cur_max = float(c.max())
        tol = max(0.08, 0.003 * cur_max)
        active = np.flatnonzero(c >= cur_max - tol)
        if len(active) > 1 and trial % 7 != 6:
            kk = int(active[int(rng.randint(len(active)))])
        else:
            i = int(rng.randint(n)); j = int(rng.randint(n))
            if i == j:
                j = (j+1) % n
                if i == j:
                    continue
            d = i - j
            kk = off + d
            if 0 <= kk < 2*n-1:
                pass
            else:
                kk = off + ((i-j) % n)
        d = kk - off
        if d >= 0:
            m = n - d
            if m <= 1:
                continue
            b = int(rng.randint(m))
            i, j = b + d, b
        else:
            e = -d
            m = n - e
            if m <= 1:
                continue
            b = e + int(rng.randint(m))
            i, j = b - e, b
        if i == j:
            continue
        B = A[i] - A[j]
        lo_u = max(-h[i], h[j] - 1.0)
        hi_u = min(1.0 - h[i], h[j])
        if hi_u - lo_u < 1e-14:
            continue
        us = np.linspace(lo_u, hi_u, n_grid)
        best_v = cur_max
        best_u = 0.0
        for u in us:
            u2 = u * u
            v = c + u * B
            v[off] -= 2.0 * u2
            qi = off + i - j
            qj = off + j - i
            if 0 <= qi < 2*n-1:
                v[qi] += u2
            if 0 <= qj < 2*n-1:
                v[qj] += u2
            vm = float(v.max())
            if vm < best_v:
                best_v = vm
                best_u = u
        accept = best_v < cur_max - 1e-10
        if not accept and best_v < cur_max + 3.0*T:
            if rng.random() < np.exp(-(best_v - cur_max) / max(T, 1e-12)):
                accept = True
        if accept:
            u = best_u
            h[i] += u
            h[j] -= u
            c = c + u * B
            u2 = u * u
            c[off] -= 2.0 * u2
            qi = off + i - j
            qj = off + j - i
            if 0 <= qi < 2*n-1:
                c[qi] += u2
            if 0 <= qj < 2*n-1:
                c[qj] += u2
            accepts += 1
            T *= 0.997
            if accepts % 40 == 0:
                A = build_A(h, n)
        else:
            T *= 0.999
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def interp_h(h, m):
    n = len(h)
    if n == m:
        return project_feasible(h.copy(), m)
    xo = np.linspace(0.0, 2.0, n, endpoint=False)
    xn = np.linspace(0.0, 2.0, m, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), m)

def random_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m_ in range(1, 10):
        a = rng.uniform(-0.4, 0.4)
        ph = rng.uniform(0, 2*np.pi)
        h += a * np.sin(0.5*np.pi*m_*x + ph)
    return project_feasible(np.clip(h, 0, 1), n)

def optimize_full(h, n, deadline, rng, quick=False):
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    hs = project_feasible(0.5*(h + h[::-1]), n)
    if compute_c5(hs, n) < c:
        h, c = hs, compute_c5(hs, n)
    hc = 1.0 - h
    if compute_c5(hc, n) < c:
        h, c = hc, compute_c5(hc, n)
    hsc = project_feasible(0.5*(h + 1.0 - h[::-1]), n)
    if compute_c5(hsc, n) < c:
        h, c = hsc, compute_c5(hsc, n)
    if not quick and time.time() < deadline - 20.0:
        dl = min(deadline - 15.0, time.time() + 40.0)
        h, c = lbfgs_opt(h, n, dl, max(1e4, 100.0*n))
    if quick:
        plans = [(50, 30), (300, 25), (2000, 20), (10000, 15)]
    else:
        plans = [(20, 60), (100, 50), (500, 40), (2000, 30), (8000, 25), (30000, 20), (100000, 15)]
    h, c = slsqp_opt(h, n, deadline, plans)
    if time.time() < deadline - 3.0:
        rem = deadline - time.time()
        n_it = min(8000, max(300, int(rem * 18)))
        h, c = pair_search(h, n, n_it, deadline, rng, n_grid=41)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard = t0 + float(budget_s) - 3.0
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        if h0.size < 2 or not np.all(np.isfinite(h0)):
            raise ValueError
    except Exception:
        h0 = random_start(200, rng)
    n0 = len(h0)
    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    if hard - t0 < 30.0:
        return best_h, compute_c5(best_h, n0), n0
    main_dl = hard - max(100.0, 0.22*float(budget_s))
    if main_dl < t0 + 20.0:
        main_dl = hard
    best_h, best_c = optimize_full(best_h, n0, main_dl, rng, quick=False)
    cands = [64, 96, 128, 160, 200, 250, 320, 400, 480, 640, 800]
    for m in cands:
        if time.time() > hard - 25.0:
            break
        if m == len(best_h):
            continue
        rem = hard - time.time()
        if m > 400 and rem < 50.0:
            continue
        if m > 640 and rem < 80.0:
            continue
        dl = time.time() + min(35.0, 0.18*rem)
        if dl < time.time() + 6.0:
            continue
        ht = interp_h(best_h, m)
        ht, ct = optimize_full(ht, m, dl, rng, quick=True)
        if ct < best_c:
            best_h, best_c = ht, ct
    # Random perturbation restarts
    for k in range(6):
        if time.time() > hard - 30.0:
            break
        n = len(best_h)
        s = [0.04, 0.08, 0.15, 0.25, 0.4, 0.6][k]
        hp = project_feasible(best_h + rng.normal(0, s, n), n)
        dl = min(hard - 10.0, time.time() + 35.0)
        if dl < time.time() + 5.0:
            continue
        ht, ct = optimize_full(hp, n, dl, rng, quick=True)
        if ct < best_c:
            best_h, best_c = ht, ct
    # Final polish
    if time.time() < hard - 8.0:
        n = len(best_h)
        dl1 = min(hard - 8.0, time.time() + 25.0)
        best_h, best_c = slsqp_opt(best_h, n, dl1, [(5000, 25), (50000, 18), (200000, 12)])
    if time.time() < hard - 4.0:
        n = len(best_h)
        rem = hard - time.time()
        n_it = max(500, int(rem * 18))
        best_h, best_c = pair_search(best_h, n, n_it, hard - 2.0, rng, n_grid=51)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
