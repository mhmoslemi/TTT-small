        d_range = np.arange(-(n-1), n, dtype=np.float64)

        mask1 = (d_range >= -j) & (d_range < n - j)
        # Term 2: -h[j-d] where 0<=j-d<n

        # Handle d=0 (index n-1)
        b[n-1] = 0.0

            else:
                lo = m1

        h[j] = t_opt
        
        if time.time() > deadline:
            break
    
    return h

                    h = best_h.copy()
                    c = np.correlate(h, 1.0 - h, mode='full')

        if ct < best_c:
            best_h, best_c = ht, ct
    h = best_h.copy()
    # Phase 1: 1D coordinate descent at native resolution
    dl1 = min(hard - 30.0, t0 + 0.5 * float(budget_s))
    h, best_c = coord_descent(h, n0, dl1, max_sweeps=400, restart_gap=12)
    # Phase 2: Pair descent polish
    if time.time() < hard - 20.0:
        rng = np.random.RandomState(seed + 1)
        dl2 = min(hard - 10.0, time.time() + 0.25 * float(budget_s))
        n_pairs = int(min(8000, max(2000, (dl2 - time.time()) * 50)))
        h, c2 = pair_descent(h, n0, dl2, n_pairs, rng)
        if c2 < best_c:
            best_h, best_c = h, c2
        else:
            h = best_h.copy()
    # Phase 3: Try higher resolution
    n_new = min(n0 * 2, 1000)
    if n_new > n0 and time.time() < hard - 40.0:
        dl3 = min(hard - 15.0, time.time() + 0.3 * float(budget_s))
        hi = interp_h(best_h, n_new)
        hi, ci = coord_descent(hi, n_new, dl3, max_sweeps=200, restart_gap=10)
        if ci < best_c:
            best_h, best_c = hi, ci
    # Phase 4: Another round at best resolution
    if time.time() < hard - 15.0:
        nf = len(best_h)
        dl4 = min(hard - 5.0, time.time() + 20.0)
        rng2 = np.random.RandomState(seed + 99)
        hp = project_feasible(best_h + rng2.normal(0.0, 0.02, nf), nf)
        hp, cp = coord_descent(hp, nf, dl4, max_sweeps=100, restart_gap=8)
        if cp < best_c:
            best_h, best_c = hp, cp
        # Final pair polish
        if time.time() < hard - 8.0:
            dl5 = hard - 4.0
            hp2, cp2 = pair_descent(best_h, nf, dl5, int(max(2000, (dl5 - time.time()) * 40)), rng2)
            if cp2 < best_c:
                best_h, best_c = hp2, cp2
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
