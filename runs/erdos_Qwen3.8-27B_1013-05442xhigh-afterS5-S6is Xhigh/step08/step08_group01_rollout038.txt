def fft_cross_correlate(a, b):
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size *= 2
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    r = np.fft.irfft(A * np.conj(B), size)
    return np.concatenate([r[-(n-1):], r[:n]])

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

            break

            tot = h.sum()
            if tot < 1e-14:
                break

    h = h.copy()
    dx = 2.0 / n

            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False

                break

            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue

            # p+k-n+1 in [0,n-1] → k in [n-1-p, 2n-2-p] ∩ [0, 2n-2]

            k_hi1 = min(2*n-2, p+n-1)
            k_lo2 = max(0, n-1-p)
            k_hi2 = min(2*n-2, 2*n-2-p)

            k_hi = max(k_hi1, k_hi2)

            lo = -h0

            best_i = int(np.argmin(vals))

            best_val = vals[best_i]

                except:

        if deadline is not None and time.time() > deadline:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

    targets = _adjust_binary(tm, n)

    starts.append(_adjust_binary(1.0 - tm, n))
    starts.append(_adjust_binary(tm[::-1], n))
    # Random balanced binary
    for _ in range(4):
        starts.append(random_balanced_binary(n, rng))
    # Parent perturbations
    for s in [0.03, 0.08, 0.15, 0.3]:
        starts.append(project_feasible(best_h + rng.normal(0, s, n), n))
    # Block shuffle of parent
    bs = best_h.copy()
    bl = n // 6
    for _ in range(3):
        i = rng.randint(0, n - 2 * bl)
        j = rng.randint(0, n - 2 * bl)
        bs[i:i+bl], bs[j:j+bl] = bs[j:j+bl].copy(), bs[i:i+bl].copy()
    starts.append(project_feasible(bs, n))
    # Hadamard-like
    starts.append(project_feasible(hadamard_seq(n), n))

    for h_s in starts:
        if time.time() > deadline - 120:
            break
        h_s = project_feasible(h_s, n)
        h_opt, c_opt = coord_descent(h_s, n, 120, deadline, grid_size=60)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 3: Simulated annealing style perturbations
    n_cur = len(best_h)
    T = 0.02
    if time.time() < deadline - 80:
        for r in range(15):
            if time.time() > deadline - 50:
                break
            s = 0.01 + 0.03 * r
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            c_p = compute_c5(hp, n_cur)
            # Accept if better or with annealing probability
            if c_p < best_c or (c_p - best_c) < T * rng.rand():
                hp, c_p = coord_descent(hp, n_cur, 100, deadline, grid_size=60)
                if c_p < best_c:
                    best_h, best_c = hp, c_p
            T *= 0.85

    # Phase 4: Targeted lag reduction
    if time.time() < deadline - 60:
        h_tgt = best_h.copy()
        for _ in range(50):
            if time.time() > deadline - 40:
                break
            c = np.correlate(h_tgt, 1.0 - h_tgt, mode='full')
            k_star = int(np.argmax(c))
            # Identify problematic pairs and flip them
            lag = k_star - (n_cur - 1)
            n_flipped = 0
            if 0 < lag < n_cur:
                for i in range(n_cur - lag):
                    if h_tgt[i] > 0.5 and h_tgt[i + lag] < 0.5 and n_flipped < 5:
                        swap = min(0.5 - h_tgt[i + lag] + 0.01, h_tgt[i] - 0.5 + 0.01)
                        swap = max(swap, 0.01)
                        h_tgt[i] -= swap
                        h_tgt[i + lag] += swap
                        n_flipped += 1
            elif -n_cur < lag < 0:
                for i in range(-lag, n_cur):
                    if h_tgt[i] > 0.5 and h_tgt[i + lag] < 0.5 and n_flipped < 5:
                        swap = min(0.5 - h_tgt[i + lag] + 0.01, h_tgt[i] - 0.5 + 0.01)
                        swap = max(swap, 0.01)
                        h_tgt[i] -= swap
                        h_tgt[i + lag] += swap
                        n_flipped += 1
            if n_flipped > 0:
                h_tgt = project_feasible(h_tgt, n_cur)
                h_tgt, c_tgt = coord_descent(h_tgt, n_cur, 80, deadline, grid_size=60)
                if c_tgt < best_c:
                    best_h, best_c = h_tgt, c_tgt
            else:
                break

    # Phase 5: Final high-precision polish
    n_cur = len(best_h)
    if time.time() < deadline - 20:
        hf, cf = coord_descent(best_h, n_cur, 500, deadline, grid_size=120)
        if cf < best_c:
            best_h, best_c = hf, cf

    # Try 1-h symmetry
    n_cur = len(best_h)
    h_flip = project_feasible(1.0 - best_h, n_cur)
    c_flip = compute_c5(h_flip, n_cur)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
        hf, cf = coord_descent(best_h, n_cur, 200, deadline, grid_size=100)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)

def _adjust_binary(h, n):
    """Adjust a near-binary array to have exactly n/2 ones."""
    h = np.array(h, dtype=float).copy()
    n_ones = int((h > 0.5).sum())
    target = n // 2
    if n_ones < target:
        zeros = np.where(h <= 0.5)[0]
        rng_tmp = np.random.RandomState(0)
        rng_tmp.shuffle(zeros)
        h[zeros[:target - n_ones]] = 1.0
    elif n_ones > target:
        ones = np.where(h > 0.5)[0]
        rng_tmp = np.random.RandomState(0)
        rng_tmp.shuffle(ones)
        h[ones[:n_ones - target]] = 0.0
    return project_feasible(h, n)
