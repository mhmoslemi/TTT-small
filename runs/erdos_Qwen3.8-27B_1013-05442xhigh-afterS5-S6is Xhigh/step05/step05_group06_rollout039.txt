    p = np.exp(alpha * (g - g_max))
    p /= p.sum()

def pair_descent(h, n, deadline, max_sweeps=500):

    best_c = compute_c5(h, n)
    nm1 = n - 1
    dm = np.arange(2 * n - 1) - nm1
    K = 10
    stall = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        F = compute_F(h, n)
        g = compute_g(h, n)
        cur_max = g.max()
        top_idx = np.argsort(g)[::-1][:K]
        improved = 0
        
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                s = F[i] - F[j]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                # Evaluate at boundaries and at points where g crosses
                cands = [d_lo, d_hi]
                for m1 in range(K):
                    for m2 in range(K):
                        if top_idx[m1] == top_idx[m2]:
                            continue
                        ds = s[top_idx[m2]] - s[top_idx[m1]]
                        if abs(ds) > 1e-15:
                            dc = (g[top_idx[m1]] - g[top_idx[m2]]) / ds
                            if d_lo < dc < d_hi:
                                cands.append(dc)

                    v = float(np.max(g + d * s))
                    if v < best_v - 1e-13:
                        best_v = v
                        best_d = d

                    h[j] -= best_d
                    g += best_d * s
                    # Update F
                    F[i] = F[i]  # Actually F changes... let me recompute
                    # For simplicity, recompute F every few moves
                    cur_max = g.max()
                    top_idx = np.argsort(g)[::-1][:K]
                    improved += 1
            # Recompute F periodically within a sweep
            if improved > 10:
                F = compute_F(h, n)
                improved = 0
        
        c = compute_c5(h, n)

            best_c = c
            best_h = h.copy()
        if improved == 0:

def binary_sa(h, n, deadline, rng):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    
    nm1 = n - 1
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    T = 0.5
    T_min = 0.01
    cooling = 0.999
    
    while time.time() < deadline:
        # Pick random 1 and 0 to swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c_new = compute_c5(h_bin, n)
        delta = c_new - best_c
        
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            # Accept
            if c_new < best_c:
                best_c = c_new
                best_h = h_bin.copy()
            # Update sets
            ones = np.delete(ones, np.where(ones == i)[0][0])
            zeros = np.delete(zeros, np.where(zeros == j)[0][0])
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
        
        T = max(T * cooling, T_min)
    
    return best_h, best_c
