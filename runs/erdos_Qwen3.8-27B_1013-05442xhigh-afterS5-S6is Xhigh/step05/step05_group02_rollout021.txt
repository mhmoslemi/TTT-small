    idx1 = i + s
    valid1 = (idx1 >= 0) & (idx1 < n)
    term1 = np.zeros(n_lags)
    term1[valid1] = (1.0 - h[idx1[valid1]]) * dx
    idx2 = i - s
    valid2 = (idx2 >= 0) & (idx2 < n)
    term2 = np.zeros(n_lags)
    term2[valid2] = h[idx2[valid2]] * dx
    return term1 - term2

def coord_descent(h, n, max_sweeps=300, time_limit=None, seed=42):
    h = h.copy()
    c = get_corr(h, n)
    best_c = float(c.max())
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() > time_limit:
            break
        order = rng.permutation(n)
        for i in order:
            grad = compute_grad_i(h, i, n)
            base = c - grad * h[i]
            t_opt = solve_1d_minimax(base, grad, 0.0, 1.0)
            delta = t_opt - h[i]
            if abs(delta) > 1e-14:
                h[i] = t_opt
                c += delta * grad
                curr = float(c.max())
                if curr < best_c:
                    best_c = curr
                    best_h = h.copy()
        h = project_feasible(h, n)
        c = get_corr(h, n)
        curr = float(c.max())
        if curr < best_c:
            best_c = curr
            best_h = h.copy()
    return best_h, best_c

def pair_coord_descent(h, n, max_sweeps=200, time_limit=None, seed=42):
    h = h.copy()
    c = get_corr(h, n)
    n_lags = 2 * n - 1
    best_c = float(c.max())
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() > time_limit:
            break
        for _ in range(n):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            grad_i = compute_grad_i(h, i, n)
            grad_j = compute_grad_i(h, j, n)
            combined = grad_i - grad_j
            d_min = max(0.0, h[j] - 1.0, -h[i])
            d_max = min(1.0 - h[i], h[j])
            if d_max - d_min < 1e-14:
                continue
            d_opt = solve_1d_minimax(c, combined, d_min, d_max)
            if abs(d_opt) > 1e-14:
                h[i] += d_opt
                h[j] -= d_opt
                c += d_opt * combined
                curr = float(c.max())
                if curr < best_c:
                    best_c = curr
                    best_h = h.copy()
    return best_h, best_c

def slsqp_refine(h, n, alpha=50000, maxiter=1000):
    dx = 2.0 / n
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1 - hx, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    def obj_grad(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1 - hx, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w = w / w.sum()
        g = np.zeros(n)
        for k in range(len(c)):
            if w[k] < 1e-10:
                continue
            s = k - (n - 1)
            lo = max(0, -s)
            hi = min(n - 1, n - 1 - s)
            idx = np.arange(lo, hi + 1)
            g[idx] += alpha * w[k] * (1.0 - h[idx + s]) * dx
            lo2 = max(0, lo + s)
            hi2 = min(n - 1, hi + s)
            idx2 = np.arange(lo2, hi2 + 1)
            g[idx2] -= alpha * w[k] * h[idx2 - s] * dx
        g -= alpha * w * 0
        return g / alpha
    def sum_fun(x):
        return float(x.sum() - n / 2.0)
    def sum_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        h_init = project_feasible(h_init, n_init)
        h_opt, c_opt = coord_descent(h_init, n_init, max_sweeps=400,
                                      time_limit=t0 + (deadline - t0) * 0.25, seed=seed)
        best_h, best_c = h_opt, c_opt

    # Try different n values with interpolation
    n_candidates = [100, 150, 200, 300, 400, 500, 750]
    for n_try in n_candidates:
        if best_h is not None and n_try == len(best_h):
            continue
        if time.time() > deadline - 80:
            break
        if best_h is not None:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_try = np.interp(x_new, x_old, best_h)
        else:
            h_try = np.full(n_try, 0.5)
        h_try = project_feasible(h_try, n_try)
        alloc = min(100, (deadline - time.time() - 40) * 0.2)
        if alloc < 30:
            break
        h_opt, c_opt = coord_descent(h_try, n_try, max_sweeps=300,
                                      time_limit=time.time() + alloc, seed=seed + 100)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Pair-based refinement on best solution
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        h_opt, c_opt = pair_coord_descent(best_h, n, max_sweeps=200,
                                           time_limit=time.time() + 60, seed=seed + 200)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # SLSQP refinement
    if best_h is not None and time.time() < deadline - 30:
        n = len(best_h)
        for alpha in [10000, 50000]:
            if time.time() > deadline - 20:
                break
            ht = slsqp_refine(best_h, n, alpha=alpha, maxiter=500)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                best_h = ht

    # Random restarts
    if best_h is not None and time.time() < deadline - 40:
        n = len(best_h)
        n_restarts = max(3, int((deadline - time.time() - 20) / 40))
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            scale = 0.003 + 0.04 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 15) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = coord_descent(hp, n, max_sweeps=200,
                                          time_limit=time.time() + alloc, seed=seed + 500 + r)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_opt, c_opt = coord_descent(best_h, n, max_sweeps=500,
                                      time_limit=deadline - 3, seed=seed + 999)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        n = 250
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
