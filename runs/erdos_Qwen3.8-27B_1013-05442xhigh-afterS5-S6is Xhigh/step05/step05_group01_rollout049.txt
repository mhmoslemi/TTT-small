    if delta_lo >= delta_hi - 1e-18:
        return delta_lo
    f0 = c_arr.max()
    flo = (c_arr + delta_lo * a).max()
    fhi = (c_arr + delta_hi * a).max()
    v_hi = min(f0, flo, fhi)
    v_lo = v_hi - np.abs(a).max() * (delta_hi - delta_lo) - 1.0
    pos = a > 1e-15
    neg = a < -1e-15
    zer = ~pos & ~neg
    for _ in range(60):
        if v_hi - v_lo < 1e-16:
            break
        v_mid = (v_lo + v_hi) / 2.0
        d_lo = delta_lo
        d_hi = delta_hi
        if zer.any() and c_arr[zer].max() > v_mid:
            v_lo = v_mid
            continue
        if pos.any():
            d_hi = min(d_hi, ((v_mid - c_arr[pos]) / a[pos]).min())
        if neg.any():
            d_lo = max(d_lo, ((v_mid - c_arr[neg]) / a[neg]).max())
        if d_lo <= d_hi:
            v_hi = v_mid
        else:
            v_lo = v_mid
    d_lo = delta_lo
    d_hi = delta_hi
    if pos.any():
        d_hi = min(d_hi, ((v_hi - c_arr[pos]) / a[pos]).min())
    if neg.any():
        d_lo = max(d_lo, ((v_hi - c_arr[neg]) / a[neg]).max())
    return (max(d_lo, delta_lo) + min(d_hi, delta_hi)) / 2.0

def coord_descent(h, n, c_arr, max_passes=200, deadline=None, rng=None):
    """In-place coordinate descent with exact 1D subproblems."""
    h = h.copy()
    best_val = c_arr.max()
    dx = 2.0 / n
    if rng is None:
        rng = np.random.RandomState(0)
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if deadline is not None and p > 0 and time.time() > deadline:
                break
            # Compute a[k] = effect on c[k] per unit delta at coordinate i
            m = 2 * n - 1
            a = np.zeros(m)
            m1_lo = max(0, n - 1 - i)
            m1_hi = min(m - 1, 2 * n - 2 - i)
            if m1_lo <= m1_hi:
                idx1 = np.arange(m1_lo, m1_hi + 1)
                g_idx = i + idx1 - (n - 1)
                a[idx1] += (1.0 - h[g_idx])
            m2_lo = max(0, i - n + 1)
            m2_hi = min(m - 1, i)
            if m2_lo <= m2_hi:
                idx2 = np.arange(m2_lo, m2_hi + 1)
                h_idx = i - idx2 + (n - 1)
                a[idx2] -= h[h_idx]
            delta_lo = max(0.0, -h[i])
            delta_hi = min(1.0, 1.0 - h[i])
            if delta_hi - delta_lo < 1e-18:
                continue
            delta = find_delta(c_arr, a, delta_lo, delta_hi)
            if abs(delta) > 1e-16:
                h[i] += delta
                c_arr = c_arr + delta * a
                nv = c_arr.max()
                if nv < best_val - 1e-14:
                    best_val = nv
                    improved = True
        if not improved:
            break
    return h, c_arr, best_val * dx

def slsqp_step(h, n, alpha, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    def sum_con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, alpha),
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(250, 0.5)
        n = 250

    h = project_feasible(h_init, n)
    c_arr = np.correlate(h, 1 - h, mode='full')
    best_h = h.copy()
    best_val = c_arr.max() * (2.0 / n)

    # Phase 1: Heavy coordinate descent
    alloc1 = min(200, (deadline - t0) * 0.4)
    h, c_arr, cv = coord_descent(h, n, c_arr, max_passes=5000, deadline=t0 + alloc1, rng=rng)
    h = project_feasible(h, n)
    cv = compute_c5(h, n)
    if cv < best_val:
        best_val = cv
        best_h = h.copy()

    # Phase 2: SLSQP refinement
    if time.time() < deadline - 60:
        for alpha in [2000, 20000, 200000]:
            if time.time() > deadline - 60:
                break
            hs = slsqp_step(h, n, alpha, maxiter=2000)
            cs = compute_c5(hs, n)
            if cs < best_val:
                best_val = cs
                best_h = hs.copy()

    # Phase 3: Another round of coordinate descent
    if time.time() < deadline - 30:
        hn = best_h.copy()
        cn = np.correlate(hn, 1 - hn, mode='full')
        hn, cn, cv = coord_descent(hn, n, cn, max_passes=3000, deadline=deadline - 25, rng=rng)
        hn = project_feasible(hn, n)
        cv = compute_c5(hn, n)
        if cv < best_val:
            best_val = cv
            best_h = hn.copy()

    # Phase 4: Randomized restarts
    if time.time() < deadline - 40:
        scales = [0.001, 0.003, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05]
        for s in scales:
            if time.time() > deadline - 30:
                break
            hp = best_h + rng.normal(0, s, n)
            hp = project_feasible(hp, n)
            cp = np.correlate(hp, 1 - hp, mode='full')
            hp, cp, cv = coord_descent(hp, n, cp, max_passes=1000, deadline=deadline - 25, rng=rng)
            hp = project_feasible(hp, n)
            cv = compute_c5(hp, n)
            if cv < best_val:
                best_val = cv
                best_h = hp.copy()

    # Phase 5: Final coordinate descent polish
    if time.time() < deadline - 3:
        hf = best_h.copy()
        cf = np.correlate(hf, 1 - hf, mode='full')
        hf, cf, cv = coord_descent(hf, n, cf, max_passes=10000, deadline=deadline - 1, rng=rng)
        hf = project_feasible(hf, n)
        cv = compute_c5(hf, n)
        if cv < best_val:
            best_val = cv
            best_h = hf

    return best_h, best_val, n
