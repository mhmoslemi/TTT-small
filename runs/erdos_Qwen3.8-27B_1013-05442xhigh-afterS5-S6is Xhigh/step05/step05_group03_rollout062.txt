            h_work[a] = 0.0
            h_work[b] = 1.0
            oi = np.where(h_work == 1)[0]
            zi = np.where(h_work == 0)[0]
        cur_h = h_work
    return best_h, best_c

def forward_construct(n, rng):
    m = 2 * n - 1
    h = np.zeros(n)
    c = np.zeros(m)
    V = np.zeros((n, m))
    for i in range(n):
        kmin = max(0, n - 1 - i)
        kmax = min(m - 1, n - 1 + i)
        V[i, kmin:kmax + 1] = 1.0
    placed = np.zeros(n, dtype=bool)
    for step in range(n // 2):
        best_i = -1
        best_max = float('inf')
        for i in range(n):
            if placed[i]:
                continue
            nm = float(np.max(c + V[i])) + 0.0002 * rng.random()
            if nm < best_max:
                best_max = nm
                best_i = i
        if best_i < 0:
            break
        h[best_i] = 1.0
        c += V[best_i]
        ip = np.arange(max(0, best_i - n + 1), min(n - 1, best_i) + 1)
        kp = best_i - ip + n - 1
        V[ip, kp] = 0.0
        placed[best_i] = True
    return h, float(c.max() * 2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h0)
    except Exception:
        h0 = None
        n_parent = 0

    best_h = None
    best_c = float('inf')
    best_n = 0

    n_main = n_parent if (h0 is not None and n_parent % 2 == 0 and n_parent >= 100) else 250
    if n_main % 2 == 1:
        n_main += 1

    total_time = dl - t0

    # Phase 1: Best known starting point (45% of time)
    if h0 is not None and len(h0) == n_main:
        start1_dl = t0 + 0.45 * total_time
        h_start = to_binary(h0, n_main)
        h1, c1 = iterated_ls(h_start, n_main, start1_dl, rng, phase_s=35.0)
        if c1 < best_c:
            best_c, best_h, best_n = c1, h1, n_main

    # Phase 2: Forward construction (25% of time)
    phase2_start = max(time.time(), t0 + 0.43 * total_time)
    phase2_dl = t0 + 0.70 * total_time
    if time.time() < phase2_dl - 10:
        h_fc, _ = forward_construct(n_main, rng)
        h2, c2 = iterated_ls(h_fc, n_main, phase2_dl, rng, phase_s=30.0)
        if c2 < best_c:
            best_c, best_h, best_n = c2, h2, n_main

    # Phase 3: Other n values (25% of time)
    phase3_start = max(time.time(), t0 + 0.68 * total_time)
    phase3_dl = t0 + 0.93 * total_time
    if time.time() < phase3_dl - 10:
        for no in [200, 300, 400]:
            if no == n_main or time.time() > phase3_dl - 20:
                continue
            t_per_n = max(20.0, (phase3_dl - time.time()) / 3.0)
            n_dl = min(time.time() + t_per_n, phase3_dl - 5)
            h_fc2, _ = forward_construct(no, rng)
            h3, c3 = iterated_ls(h_fc2, no, n_dl, rng, phase_s=25.0)
            if c3 < best_c:
                best_c, best_h, best_n = c3, h3, no

    # Phase 4: Final polish on best (remaining time)
    if best_h is not None and time.time() < dl - 5:
        h4, c4 = fast_ls(best_h.copy(), best_n, dl - 2, max_sweeps=999999, rng=rng, sa=True)
        if c4 < best_c:
            best_c, best_h = c4, h4

    if best_h is None:
        best_h = np.zeros(n_main)
        best_h[rng.permutation(n_main)[:n_main // 2]] = 1.0
        best_c = compute_c5(best_h, n_main)
        best_n = n_main

    return best_h, best_c, best_n
