            s = h.sum()
            if s < 1e-14:

    hb[idx[:target]] = 1.0
    return hb

                break

    # So this is NOT a (31,6,1) difference set. I'm confusing constructions.
    
    # Let me just use a simple approach: create a pattern with good difference distribution.
    # Use the fact that for n = m*p where p is prime, a random subset of Z_p of size ~p/2
    # tiled m times gives a reasonable starting point.
    
    # For now, just use structured random
    h = rng.binomial(1, 0.5, n).astype(float)
    target = n // 2
    while h.sum() > target:
        idx = np.where(h == 1)[0]
        h[idx[rng.randint(len(idx))]] = 0
    while h.sum() < target:
        idx = np.where(h == 0)[0]
        h[idx[rng.randint(len(idx))]] = 1
    return h

def binary_local_search(hb, n, deadline):
    """First-improvement binary local search."""
    dx = 2.0 / n
    best_c5 = compute_c5(hb, n)
    
    while time.time() < deadline:
        improved = False
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        for i in zeros:
            if time.time() > deadline:
                return hb, best_c5
            for j in ones:
                if time.time() > deadline:
                    return hb, best_c5
                hb[i], hb[j] = 1.0, 0.0
                c = float(np.max(np.correlate(hb, 1 - hb, mode='full')) * dx)
                if c < best_c5 - 1e-15:
                    best_c5 = c
                    improved = True
                    break
                hb[i], hb[j] = 0.0, 1.0
            if improved:
                break
        if not improved:
            break
    return hb, best_c5

def simulated_annealing(hb, n, deadline, T0=1.0, T1=0.01):
    """Simulated annealing on binary solutions."""
    dx = 2.0 / n
    best_c5 = compute_c5(hb, n)
    best_hb = hb.copy()
    current_c5 = best_c5
    current_hb = hb.copy()
    
    t_start = time.time()
    t_end = deadline
    total_time = t_end - t_start
    
    ones = set(np.where(current_hb == 1)[0].tolist())
    zeros = set(np.where(current_hb == 0)[0].tolist())
    
    while time.time() < t_end:
        elapsed = time.time() - t_start
        progress = min(1.0, elapsed / total_time)
        T = T0 * (T1 / T0) ** progress
        
        a = next(iter(ones))  # random from ones
        # Actually let's use proper random
        a = list(ones)[np.random.randint(len(ones))]
        b = list(zeros)[np.random.randint(len(zeros))]
        
        current_hb[a], current_hb[b] = 0.0, 1.0
        new_c5 = float(np.max(np.correlate(current_hb, 1 - current_hb, mode='full')) * dx)
        
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            ones.discard(a)
            ones.add(b)
            zeros.discard(b)
            zeros.add(a)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_hb = current_hb.copy()
        else:
            current_hb[a], current_hb[b] = 1.0, 0.0
    
    return best_hb, best_c5

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def con_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def perturb_binary(hb, n, rng, k):
    """Flip k random pairs."""
    h = hb.copy()
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for _ in range(k):
        ai = rng.randint(len(ones))
        bi = rng.randint(len(zeros))
        a, b = ones[ai], zeros[bi]
        h[a], h[b] = 0.0, 1.0
        ones[ai] = b
        zeros[bi] = a
    return h

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
