def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    """Convert continuous h to binary with exactly n//2 ones."""
    target = n // 2
    order = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b

def init_thue_morse(n, seed=0):
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    target = n // 2
    n_ones = int(tm.sum())
    rng = np.random.RandomState(seed)
    if n_ones > target:
        ones = np.where(tm == 1)[0]
        for _ in range(n_ones - target):
            tm[ones[rng.randint(len(ones))]] = 0
            ones = np.where(tm == 1)[0]
    elif n_ones < target:
        zeros = np.where(tm == 0)[0]
        for _ in range(target - n_ones):
            tm[zeros[rng.randint(len(zeros))]] = 1
            zeros = np.where(tm == 0)[0]

    dx = 2.0 / n
