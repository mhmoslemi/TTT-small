def softmax_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    z = alpha * c
    z = z - z.max()
    p = np.exp(z)
    p /= p.sum()
    
    n_full = 2 * n - 1
    ramps = np.arange(n_full) - (n - 1)  # s values: -(n-1), ..., 0, ..., (n-1)
    idx = np.arange(n)
    
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        w = p[s + n - 1]
        if w < 1e-15:
            continue
        # (1-h[j+s]) for valid j+s
        js = idx + s
        valid1 = (js >= 0) & (js < n)
        grad[valid1] += w * (1.0 - h[js[valid1]])
        # -h[j-s] for valid j-s
        jm = idx - s
        valid2 = (jm >= 0) & (jm < n)
        grad[valid2] -= w * h[jm[valid2]]
    
    grad *= dx
    return grad

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(h, n, alpha):
    dx = 2.0 / n
    c = compute_corr(h, n)
    z = alpha * (c - c.max())
    return float(c.max() + np.log(np.exp(z).sum()) / alpha)

def softmax_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_corr(h, n)
    z = alpha * (c - c.max())
    p = np.exp(z)

        jm = idx - s

        grad[v2] -= w * h[jm[v2]]
    return grad * dx

    # L-BFGS-B handles bounds but not equality constraints well
    # We'll handle the sum constraint via projection after each step

    # and set the last one to satisfy the constraint

        return softmax_obj(x, n, alpha)

        return softmax_grad(x, n, alpha)
    
    def con_fun(x):
        return x.sum() - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': tol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gradient_step(h, n, alpha, step_size=0.01):
    """One step of projected gradient descent targeting the max correlation."""
    dx = 2.0 / n
    c = compute_corr(h, n)
    z = alpha * (c - c.max())
    p = np.exp(z)
    p /= p.sum()
    
    # Gradient of softmax-smoothed max
    idx = np.arange(n)
    g = np.zeros(n)
    for s in range(-(n-1), n):
        w = p[s + n - 1]
        if w < 1e-15:
            continue
        js = idx + s
        v1 = (js >= 0) & (js < n)
        g[v1] += w * (1.0 - h[js[v1]])
        jm = idx - s
        v2 = (jm >= 0) & (jm < n)
        g[v2] -= w * h[jm[v2]]
    g *= dx
    
    # Project to sum=0 direction
    g -= g.mean()
    
    # Step
    h_new = h - step_size * g
    h_new = project_feasible(h_new, n)
    return h_new

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def softmax_obj(h, n, alpha):
    c = compute_corr(h, n)
    z = alpha * (c - c.max())
    return float(c.max() + np.log(np.exp(z).sum()) / alpha)

def softmax_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_corr(h, n)
    z = alpha * (c - c.max())
    p = np.exp(z)
    p /= p.sum()
    idx = np.arange(n)
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        w = p[s + n - 1]
        if w < 1e-15:
            continue
        js = idx + s
        v1 = (js >= 0) & (js < n)
        grad[v1] += w * (1.0 - h[js[v1]])
        jm = idx - s
        v2 = (jm >= 0) & (jm < n)
        grad[v2] -= w * h[jm[v2]]
    return grad * dx

def run_slsqp(h, n, alpha, maxiter=3000):
    def obj(x):
        return softmax_obj(x, n, alpha)
    def grad(x):
        return softmax_grad(x, n, alpha)
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return _project_feasible(h, n)

def hot_lag_search(h, n, deadline, top_k=5, step_sizes=None):
    if step_sizes is None:
        step_sizes = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    idx = np.arange(n)
    for ss in step_sizes:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = compute_corr(h, n)
            top_m = np.argsort(c)[-top_k:]
            for m in top_m:
                s = m - (n - 1)
                g_m = np.zeros(n)
                js = idx + s
                v1 = (js >= 0) & (js < n)
                g_m[v1] = (1.0 - h[js[v1]])
                jm = idx - s
                v2 = (jm >= 0) & (jm < n)
                g_m[v2] -= h[jm[v2]]
                g_m *= dx
                g_m -= g_m.mean()
                h_try = project_feasible(h - ss * g_m, n)
                c_try = compute_c5(h_try, n)
                if c_try < best_c - 1e-14:
                    best_c = c_try
                    h = h_try
                    improved = True
    return h, best_c

def pair_search_targeted(h, n, deadline, deltas=None):
    if deltas is None:
        deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    idx = np.arange(n)
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = compute_corr(h, n)
            top_m = np.argsort(c)[-3:]
            for m in top_m:
                s = m - (n - 1)
                benefit = np.zeros(n)
                js = idx + s
                v1 = (js >= 0) & (js < n)
                benefit[v1] += h[js[v1]]
                jm = idx - s
                v2 = (jm >= 0) & (jm < n)
                benefit[v2] += h[jm[v2]]
                can_add = np.where(h + d <= 1.0)[0]
                can_sub = np.where(h - d >= 0.0)[0]
                if len(can_add) == 0 or len(can_sub) == 0:
                    continue
                i_list = can_add[np.argsort(benefit[can_add])[-8:]]
                j_list = can_sub[np.argsort(benefit[can_sub])[:8]]
                for i in i_list:
                    for j in j_list:
                        if i == j:
                            continue
                        h[i] += d
                        h[j] -= d
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
    return h, best_c

def _project_feasible(h, n):
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100

    best_h = h_init.copy()
    best_c = compute_c5(best_h, n_init)

    # Phase 1: SLSQP with analytical gradient, alpha annealing
    for alpha in [500, 2000, 10000, 50000, 200000]:
        if time.time() > deadline - 80:
            break
        h_opt = run_slsqp(best_h, n_init, alpha, maxiter=3000)
        c_opt = compute_c5(h_opt, n_init)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 2: Hot lag gradient steps
    if time.time() < deadline - 40:
        h_opt, c_opt = hot_lag_search(best_h, n_init, deadline - 40, top_k=8)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: High-alpha SLSQP polish
    if time.time() < deadline - 20:
        h_opt = run_slsqp(best_h, n_init, 1000000, maxiter=5000)
        c_opt = compute_c5(h_opt, n_init)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 4: Targeted pair search
    if time.time() < deadline - 5:
        h_opt, c_opt = pair_search_targeted(best_h, n_init, deadline - 3,
                                           [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 5: Final SLSQP + pair
    if time.time() < deadline - 5:
        h_opt = run_slsqp(best_h, n_init, 2000000, maxiter=3000)
        c_opt = compute_c5(h_opt, n_init)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
        if time.time() < deadline - 2:
            h_opt, c_opt = pair_search_targeted(best_h, n_init, deadline - 1,
                                               [0.001, 0.0005, 0.0002, 0.0001])
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 6: Try n=200
    if time.time() < deadline - 50:
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, 200)
        h_200 = np.interp(x_new, x_old, best_h)
        h_200 = project_feasible(h_200, 200)
        for alpha in [5000, 50000, 500000]:
            if time.time() > deadline - 10:
                break
            h_opt = run_slsqp(h_200, 200, alpha, maxiter=2000)
            c_opt = compute_c5(h_opt, 200)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
        if time.time() < deadline - 5:
            h_opt, c_opt = hot_lag_search(best_h, len(best_h), deadline - 3, top_k=5)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 7: Multi-restart perturbation on best
    rng = np.random.RandomState(seed)
    if best_h is not None and time.time() < deadline - 30:
        n = len(best_h)
        remaining = deadline - time.time() - 10
        n_restarts = max(2, int(remaining / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 8:
                break
            scale = 0.02 + 0.06 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc_end = min(time.time() + 35, deadline - 5)
            h_opt = run_slsqp(hp, n, 50000, maxiter=2000)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
                h_opt2, c_opt2 = hot_lag_search(h_opt, n, min(time.time() + 10, deadline - 3), top_k=5)
                if c_opt2 < best_c:
                    best_c = c_opt2
                    best_h = h_opt2

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
