        h_pol, c_pol = greedy_local_search(best_h.copy(), best_n, dl - 2, rng, max_iters=50000)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol.copy()

    # Final verification
    n_final = best_n
    best_c = compute_c5(best_h, n_final)
    assert abs(best_h.sum() - n_final // 2) < 1e-10, f"Volume: {best_h.sum()} vs {n_final//2}"
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1 + 1e-10)
    best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c, n_final
