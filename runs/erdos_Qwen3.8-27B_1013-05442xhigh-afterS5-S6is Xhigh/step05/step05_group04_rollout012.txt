    for a in alphas:
        if time.time() > deadline - 50:
            break
        ht = slsqp_solve(best_h, n, a, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

    # Phase 2: Block coordinate SLSQP
    if time.time() < deadline - 40:
        alloc = min(100, (deadline - time.time()) * 0.25)
        best_h, best_c = block_slsqp(best_h, n, 20, time.time() + alloc)

    # Phase 3: SLSQP polish with very high alpha
    if time.time() < deadline - 20:
        for a in [500000, 2000000, 5000000]:
            if time.time() > deadline - 10:
                break
            ht = slsqp_solve(best_h, n, a, maxiter=3000)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                best_h = ht

    # Phase 4: Systematic near-pair search for medium adjustments
    if time.time() < deadline - 20:
        alloc = min(80, (deadline - time.time()) * 0.3)
        best_h, best_c = systematic_near_pairs(best_h, n, time.time() + alloc,
                                               [0.01, 0.005, 0.002, 0.001], radius=12)

    # Phase 5: Block SLSQP with smaller blocks
    if time.time() < deadline - 20:
        alloc = min(60, (deadline - time.time()) * 0.25)
        best_h, best_c = block_slsqp(best_h, n, 10, time.time() + alloc)

    # Phase 6: Random pair search for fine-tuning
    if time.time() < deadline - 5:
        best_h, best_c = random_pair_search(best_h, n, deadline - 3, rng,
                                            delta_range=(0.00005, 0.005))

    # Phase 7: Multi-start perturbation
    if time.time() < deadline - 30:
        for scale in [0.03, 0.06, 0.12]:
            if time.time() > deadline - 10:
                break
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            ht = slsqp_solve(hp, n, 5000, maxiter=1500)
            ht = slsqp_solve(ht, n, 100000, maxiter=1500)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                best_h = ht
                if time.time() < deadline - 5:
                    best_h, best_c = random_pair_search(best_h, n, deadline - 2,
                                                        rng, (0.0001, 0.003))

    # Phase 8: Final polish
    if time.time() < deadline - 3:
        ht = slsqp_solve(best_h, n, 5000000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
        if time.time() < deadline - 1:
            best_h, best_c = random_pair_search(best_h, n, deadline, rng,
                                                delta_range=(0.00002, 0.002))

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
