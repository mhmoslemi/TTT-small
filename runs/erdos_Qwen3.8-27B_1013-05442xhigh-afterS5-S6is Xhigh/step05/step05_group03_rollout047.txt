        j = ones[j_idx]

def slsqp_refine(hb, n, deadline, alpha=5000000):
    """Use SLSQP on continuous relaxation from binary point."""
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sh -= sh.max()
        return m + (sh.max() + np.log(np.exp(sh - sh.max()).sum())) / alpha
    def con_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    try:
        res = minimize(obj, hb.astype(float), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 2000, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except:
        return hb

Let me try a different approach: **Iterated local search (ILS)** with perturbation.

ILS works like this:
1. Start with a solution
2. Apply local search to get a local optimum
3. Perturb the local optimum (make some random changes)
4. Apply local search again
5. If the new local optimum is better, accept; otherwise, go back to the previous best
6. Repeat

This is a simple but effective metaheuristic for combinatorial optimization.

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14: break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
