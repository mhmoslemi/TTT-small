4. If h[i] = 1, we can only increase h[i] (v[i] should be ≥ 0)
5. For free i (0 < h[i] < 1), we can move in either direction
6. Take a step: h += lr * (-v) where lr is chosen by line search
7. Project back to feasible set

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    g_max = g.max()
    g_shifted = alpha * (g - g_max)
    # Clip to avoid overflow
    g_shifted = np.clip(g_shifted, -500, 0)
    w = np.exp(g_shifted)
    w_sum = w.sum()
    w = w / w_sum
    f = g_max + np.log(w_sum) / alpha
    
    dm = np.arange(2*n-1) - (n-1)
    grad = np.zeros(n)
    for m in range(2*n-1):
        d = dm[m]
        j_lo = max(0, -d)

        if j_lo < j_hi:
            grad[j_lo:j_hi] += w[m] * (1.0 - h[j_lo-d:j_hi-d])
            grad[j_lo:j_hi] -= w[m] * h[j_lo+d:j_hi+d]
    grad *= dx
    return f, grad

def slsqp_optimize(h, n, alpha, deadline, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = np.clip(alpha * (g - m), -500, 0)
        return m + np.log(np.exp(shifted).sum()) / alpha
    def obj_grad(x):
        _, grad = smooth_grad(x, n, alpha)
        return grad
    def cons_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': cons_fun}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

    best_h = h.copy()

            # Where j is chosen to maximize the improvement

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    g_max = g.max()
    g_shifted = np.clip(alpha * (g - g_max), -500, 0)
    w = np.exp(g_shifted)
    w_sum = w.sum()
    w = w / w_sum
    f = g_max + np.log(w_sum) / alpha
    dm = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        d = dm[m]
        j_lo = max(0, -d)
        j_hi = min(n, n - d)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += w[m] * (1.0 - h[j_lo - d:j_hi - d])
            grad[j_lo:j_hi] -= w[m] * h[j_lo + d:j_hi + d]
    grad *= dx
    return f, grad

def slsqp_optimize(h, n, alpha, deadline, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = np.clip(alpha * (g - m), -500, 0)
        return m + np.log(np.exp(shifted).sum()) / alpha
    def obj_grad(x):
        _, grad = smooth_grad(x, n, alpha)
        return grad
    def cons_fun(x):
        return x.sum() - n / 2.0
    cons = [{'type': 'eq', 'fun': cons_fun}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def projected_grad_descent(h, n, deadline, alpha_init=10000, alpha_end=500000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = float(alpha_init)
    lr = 0.005
    no_improve = 0
    while time.time() < deadline - 2:
        f_old, grad = smooth_grad(h, n, alpha)
        grad_proj = grad - grad.mean()
        gnorm = np.linalg.norm(grad_proj)
        if gnorm < 1e-14:
            break
        improved = False
        for _ in range(15):
            h_new = project_feasible(h - lr * grad_proj, n)
            f_new, _ = smooth_grad(h_new, n, alpha)
            if f_new < f_old - 1e-16:
                h = h_new
                improved = True
                break
            lr *= 0.5
            if lr < 1e-12:
                break
        if not improved:
            no_improve += 1
            if no_improve > 40:
                alpha = min(alpha * 1.5, float(alpha_end))
                lr = max(lr, 0.001)
                no_improve = 0
                if alpha >= alpha_end:
                    break
            continue
        no_improve = 0
        lr = min(lr * 1.15, 0.02)
        if alpha < alpha_end:
            alpha = min(alpha * 1.003, float(alpha_end))
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    c = compute_c5(h, n)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

def coord_descent(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    while time.time() < deadline - 1:
        improved_any = False
        for i in range(n):
            if time.time() > deadline:
                break
            hi = h[i]
            if hi <= 1e-10 or hi >= 1.0 - 1e-10:
                continue
            for delta in [0.02, 0.01, 0.005, -0.02, -0.01, -0.005]:
                hi_new = hi + delta
                if hi_new < 0.0 or hi_new > 1.0:
                    continue
                if delta > 0:
                    cands = np.where(h >= delta + 1e-10)[0]
                    if len(cands) == 0:
                        continue
                    j = cands[np.argmin(h[cands])]
                else:
                    room = 1.0 - h
                    cands = np.where(room >= -delta - 1e-10)[0]
                    if len(cands) == 0:
                        continue
                    j = cands[np.argmax(room[cands])]
                h[i] = hi_new
                h[j] -= delta
                h = project_feasible(h, n)
                c_new = compute_c5(h, n)
                if c_new < best_c - 1e-15:
                    best_c = c_new
                    best_h = h.copy()
                    improved_any = True
                else:
                    h[i] = hi
                    h[j] += delta
                    h = project_feasible(h, n)
        if not improved_any:
            break
    return best_h, best_c
