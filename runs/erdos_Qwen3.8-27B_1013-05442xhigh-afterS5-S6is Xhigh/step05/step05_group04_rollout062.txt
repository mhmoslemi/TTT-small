import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

Alternative: do **pair coordinate descent**. Pick pairs (i,j) and optimize the transfer amount d (h[i]+=d, h[j]-d) while maintaining the sum.

For a pair (i,j), the new c_k is:
c_k(d) = c_k(0) + d * (s_k^i - s_k^j)

where s_k^i is the sensitivity of c_k to h[i] (i.e., ∂c_k/∂h[i] at the current point).

Wait, but c_k is not linear in h (it's quadratic). However, as I showed above, for a single coordinate change, c_k IS linear in that coordinate (because the quadratic term h[i]*h[i+k] only appears once for each pair (i, i+k)).

But for a pair change (h[i]+=d, h[j]-=d), the cross term h[i]*h[j] (if i and j are k apart) would create a d² term. Let me check:

c_k = sum_m h[m]*(1-h[m+k])

If I change h[i] → h[i]+d and h[j] → h[j]-d:
- Terms with h[i] as first factor: h[i+d]*(1-h[i+k]) → adds d*(1-h[i+k])
- Terms with h[i] as second factor (m=i-k): h[i-k]*(1-h[i+d]) → adds -d*h[i-k]
- Terms with h[j] as first factor: h[j-d]*(1-h[j+k]) → adds -d*(1-h[j+k])
- Terms with h[j] as second factor (m=j-k): h[j-k]*(1-h[j-d]) → adds d*h[j-k]

But wait, if i+k = j (i.e., j = i+k), then the term h[i]*(1-h[j]) = h[i+d]*(1-h[j-d]) = (h[i]+d)(1-h[j]+d) = h[i](1-h[j]) + d(1-h[j]) + d*h[i] + d²

So there IS a d² term when j = i+k! Similarly if i = j+k.

So c_k(d) is:
- Linear in d if neither i+k=j nor j+k=i
- Quadratic in d (specifically, has a d² term with coefficient ±1) if i and j are exactly k apart

This means for most pairs, the change is linear in d, and the max of linear functions is convex. For the few pairs that are k apart, there's a quadratic term.

For the linear case, the optimal d can be found by checking where the subgradient of the max contains 0, which means finding the crossing of the lines that form the upper envelope.

**Practical approach: Adaptive pair search with binary refinement**

For each pair (i,j):
1. Evaluate c at d=0 (current)
2. Evaluate c at d=+ε and d=-ε for small ε
3. If max(c(d+), c(d-)) < c(0), the direction is good
4. Do a binary search to find the optimal d

But this requires multiple correlate calls per pair, which is expensive.

**Alternative practical approach: Subgradient method**

At each iteration:
1. Compute c = correlate(h, 1-h) * dx
2. Find the active set A = {k : c[k] ≈ max(c)}
3. Compute the subgradient: g[i] = sum over active k of the sensitivity of c_k to h[i]
4. Take a small step in the -g direction
5. Project onto the feasible set

The sensitivity of c_k to h[i]:
As derived above, c_k = ... + h[i]*(1-h[i+k]) + h[i-k]*(1-h[i]) + ...
∂c_k/∂h[i] = (1-h[i+k]) - h[i-k]  (treating other h's as fixed)

Wait, that's not quite right because h[i] also affects the (1-h[m+k]) term when m+k=i.

Let me be more careful. c_k = sum_m h[m]*(1-h[m+k]).

∂c_k/∂h[i] = (1-h[i+k]) + sum_m where m+k=i: h[m]*(-1)
= (1-h[i+k]) - h[i-k]

(where we interpret h[out_of_bounds] = 0)

So the subgradient of max_k c_k with respect to h[i] is:
g[i] = (1/|A|) * sum_{k in A} [(1-h[i+k]) - h[i-k]]

or any convex combination. A common choice is the average over active constraints.

Then the subgradient descent step is:
h[i] ← h[i] - η * g[i]

followed by projection onto {0 ≤ h ≤ 1, sum(h) = n/2}.

This is O(n²) per iteration (n active constraints × n coordinates for the gradient, plus O(n) for the correlate).

def subgradient_step(h, n, eta, tol=1e-8):
    dx = 2.0/n
    c = np.correlate(h, 1-h, mode='full') * dx
    c_max = c.max()
    # Active set: lags where c is close to max
    active = c >= c_max - tol
    
    # Compute subgradient
    g = np.zeros(n)
    for k_active in np.where(active)[0]:
        # k_active is the index in the full correlation (0 to 2n-2)
        # This corresponds to lag = k_active - (n-1)
        lag = k_active - (n - 1)
        # ∂c_k/∂h[i] = (1-h[i+lag]) - h[i-lag]
        # For each i:
        # h[i+lag] is valid when 0 <= i+lag < n
        # h[i-lag] is valid when 0 <= i-lag < n
        
        # Vectorized:
        # For term (1-h[i+lag]):
        # i ranges over [0,n-1], i+lag ranges over [lag, n-1+lag]
        # Valid when 0 <= i+lag < n, i.e., max(0,-lag) <= i < n-lag (if lag>0) 
        # or i < n+lag (if lag<0) and i >= -lag
        if lag >= 0:
            # i+lag valid when i < n-lag, i.e., i in [0, n-1-lag]
            if n - lag > 0:
                g[:n-lag] += (1 - h[lag:])  # 1 - h[i+lag] for i in [0, n-1-lag]
            # i-lag valid when i >= lag, i.e., i in [lag, n-1]
            if lag < n:
                g[lag:] += -h[:n-lag]  # -h[i-lag] for i in [lag, n-1]
        else:
            # lag < 0, let lag = -d where d > 0
            d = -lag
            # i+lag = i-d, valid when i >= d, i.e., i in [d, n-1]
            if d < n:
                g[d:] += (1 - h[:n-d])  # 1 - h[i-d] for i in [d, n-1]
            # i-lag = i+d, valid when i < n-d, i.e., i in [0, n-1-d]
            if d < n:
                g[:n-d] += -h[d:]  # -h[i+d] for i in [0, n-1-d]
    
    # Average over active constraints
    g /= active.sum()
    return g, c_max

def subgradient_descent(h, n, deadline, initial_eta=0.1):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = initial_eta
    stuck = 0
    
    while time.time() < deadline:
        g, c_max = subgradient_step(h, n, eta)
        # Step
        h_new = h - eta * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-10:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stuck = 0
        else:
            stuck += 1
            if stuck > 20:
                eta *= 0.5
                stuck = 0
                if eta < 1e-6:
                    break
        # Decrease eta slowly
        eta *= 0.999
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

    c = np.correlate(x, 1 - x, mode='full') * dx
    return float(np.sum(c**p)**(1.0/p))

                       bounds=bounds, constraints=cons,

    active = c >= c_max - tol
    n_active = int(active.sum())

        return np.zeros(n), c_max
    g = np.zeros(n)

        lag = int(k_idx) - (n - 1)

        else:

                best_h, best_c = pair_search(best_h, n, deadline - 1,
                                             [0.0005, 0.0002, 0.0001, 0.00005])
            else:
                best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                                  [0.0005, 0.0002, 0.0001], radius=40)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
