import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgrad_topk(h, n, deadline, K=5, max_iter=50000, lr=0.05):
    """Projected subgradient with top-K lags and backtracking."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    current_lr = lr
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(best_h, n)
        m = g.max()
        # Get top K lags
        top_k = np.argsort(g)[::-1][:K]
        # Compute average subgradient over top K lags
        grad = np.zeros(n)
        for k_idx in top_k:
            lag = k_idx - (n - 1)
            if lag >= 0:
                grad[lag:] += dx * (1.0 - best_h[:n-lag])
                if n - lag > 0:
                    grad[:n-lag] -= dx * best_h[lag:]
            else:
                to = -lag
                if n - to > 0:
                    grad[:n-to] += dx * (1.0 - best_h[to:])
                grad[to:] -= dx * best_h[:n-to]
        grad /= K
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        grad /= ng
        # Backtracking line search
        found = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001]:
            step = current_lr * factor
            ht = project_feasible(best_h - step * grad, n)

            if current_lr < 1e-12:
                break
        else:

    return best_h, best_c

def coordinate_descent(h, n, deadline, max_iter=100000):
    """Coordinate descent: try adjusting one element at a time."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for it in range(max_iter):

            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break

            old = h[j]
            for delta in [0.01, 0.05, 0.1, 0.2, 0.5, 1.0 - old]:
                if old + delta > 1.0 + 1e-10:
                    continue
                # Need to decrease some other element to maintain sum
                # Find the element whose decrease is least harmful
                h[j] = old + delta
                # Decrease the element with highest value (that's not j)
                # Actually, just scale down all others proportionally
                others = np.ones(n)
                others[j] = 0
                need = delta  # need to decrease by this much total
                room = h * others  # how much can we take from each
                s = room.sum()
                if s > 1e-10:
                    h = h - (need / s) * h * others
                    h[j] = old + delta  # restore j

                    c = compute_c5(h, n)

                        improved = True
                        break
                    else:

                        # redo h
                        h = h  # already restored
                        break
                else:
                    h[j] = old
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c

def project_feasible(h, n):

            s = h.sum()
            if s < 1e-14:

def compute_swap_delta(h, n, j, i):
    """Compute delta in full correlation array when swapping h[j]=1 -> 0 and h[i]=0 -> 1."""
    # j is currently 1, i is currently 0
    result = np.zeros(2*n - 1)
    idx = np.arange(2*n - 1)
    lags = idx - (n - 1)  # lags from -(n-1) to (n-1)
    
    # For each lag L, compute delta
    # Positive lags (L > 0):
    pos = lags > 0
    L = lags[pos]
    d = np.zeros(L.shape)
    
    # -[j>=L, j-L not in S]  (removing j)
    m1 = (j >= L)
    if m1.any():
        d[m1] -= (1.0 - h[j - L[m1]])
    
    # +[i>=L, i-L not in S, i-L != j]  (adding i)
    m2 = (i >= L)
    if m2.any():
        base = i - L[m2]
        d[m2] += (1.0 - h[base]) * (base != j).astype(float)
    
    # +[j+L<n, (j+L) in S, (j+L) != i]
    m3 = (j + L < n)
    if m3.any():
        idx3 = j + L[m3]
        d[m3] += h[idx3] * (idx3 != i).astype(float)
    
    # -[i+L<n, (i+L) in S]
    m4 = (i + L < n)
    if m4.any():
        idx4 = i + L[m4]
        d[m4] -= h[idx4]
    
    result[pos] = d
    
    # Negative lags (L < 0, let K = -L > 0):
    # g(-K) = sum_{t=0}^{n-1-K} h[t]*(1-h[t+K])
    neg = lags < 0
    K = -lags[neg]

    m1 = (j + K < n)

    m2 = (i + K < n)

def compute_swap_delta(h, n, j, i):
    """O(n) computation of delta in full correlation when swapping h[j]=1->0, h[i]=0->1."""
    result = np.zeros(2*n - 1)
    
    # Positive lags L = 1 to n-1 (correlation index n-1+L)
    L = np.arange(1, n)
    d = np.zeros(n-1)
    
    # Term 1: -[j>=L]*(1-h[j-L])
    m1 = L <= j
    if m1.any():
        d[m1] -= (1.0 - h[j - L[m1]])
    
    # Term 2: +[i>=L]*((1-h[i-L]) + [i-L=j])
    m2 = L <= i
    if m2.any():
        base = i - L[m2]
        d[m2] += (1.0 - h[base]) + (base == j).astype(float)
    
    # Term 3: +[j+L<n]*h[j+L]*(j+L != i)
    m3 = (j + L) < n
    if m3.any():
        idx3 = j + L[m3]
        d[m3] += h[idx3] * (idx3 != i).astype(float)

    m4 = (i + L) < n
