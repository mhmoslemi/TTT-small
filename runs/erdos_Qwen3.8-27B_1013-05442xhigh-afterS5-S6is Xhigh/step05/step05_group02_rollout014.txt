```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c * dx))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def worst_lag_gradient(h, n, k):
    """Gradient of c[k] w.r.t. h."""
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        # c[k] = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
        # dc[k]/dh[i] for i in [0, n-1-s]: -(h[i+s]) from the (1-h[i]) term
        # dc[k]/dh[i+s] for i in [0, n-1-s]: (1-h[i]) from the h[i+s] term
        idx = np.arange(n - s)
        g[idx] -= h[idx + s]       # derivative of (1-h[i]) w.r.t. h[i]
        g[idx + s] += (1 - h[idx])  # derivative of h[i+s] w.r.t. h[i+s]
    else:
        s = -s
        # c[k] = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
        idx = np.arange(n - s)
        g[idx] += (1 - h[idx + s])   # derivative of h[i]
        g[idx + s] -= h[idx]          # derivative of (1-h[i+s])
    return g

def lse_smooth(h, n, alpha):
    c = get_corr(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    lse = np.log(np.sum(np.exp(shifted - max_s)))
    return m + (max_s + lse) / alpha

def lse_gradient(h, n, alpha):
    c = get_corr(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    exp_s = np.exp(shifted - max_s)
    weights = exp_s / exp_s.sum()
    g = np.zeros(n)
    # Only compute for significant weights
    significant = np.where(weights > 1e-8)[0]
    for k in significant:
        g += weights[k] * worst_lag_gradient(h, n, int(k))
    return g, c, m

def slsqp_optimize(h, n, alpha, maxiter=3000):
    def obj(x):
        h_x = np.clip(x, 0, 1)
        return lse_smooth(h_x, n, alpha)
    
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def sum_con(x):
        return float(np.sum(x) - target)
    def sum_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def gradient_local_search(h, n, deadline, top_k=5, n_iter_max=50000):
    """Local search using LSE gradient with line search and SA."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    curr_h = h.copy()
    curr_c = best_c
    alpha = 1000.0
    rng = np.random.RandomState(777)
    T = 0.02  # SA temperature
    
    for it in range(n_iter_max):
        if time.time() > deadline:
            break
        
        g, c, m = lse_gradient(curr_h, n, alpha)
        # Find best move: decrease h at max(g), increase h at min(g)
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        
        # Line search: try several step sizes
        best_step = 0
        best_new_c = curr_c
        for d in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
```

```python
def projected_gradient_lse(h, n, deadline, alpha=10000, lr=None):
    """Projected gradient descent on LSE objective with sum constraint."""
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    curr = h.copy()
    lr = lr or 0.01/lr_decay_start
    lr_decay = 0.999
    momentum = 0.9
    v = np.zeros(n)
    
    while time.time() < deadline:
        c = get_corr(curr, n)
        m = c.max()
        shifted = alpha * (c - m)
        max_s = shifted.max()
        exp_s = np.exp(shifted - max_s)
        weights = exp_s / exp_s.sum()
        
        # Gradient
        g = np.zeros(n)
        significant = np.where(weights > 1e-6)[0]
        for k in significant:
            s = int(k) - (n - 1)
            w = weights[int(k)]
            if s >= 0:
                g[s:] += w * (1 - curr[:n-s])
                g[:n-s] -= w * curr[s:]
            else:
                s = -s
                g[:n-s] += w * (1 - curr[s:])
                g[s:] -= w * curr[:n-s]
        
        # Momentum
        v = momentum * v - lr * g
        curr = curr + v
        
        # Project: clip to [0,1] and fix sum
        curr = project_feasible(curr, n)
        
        # Update best
        c_now = compute_c5(curr, n)
        if c_now < best_c:
            best_c = c_now
            best_h = curr.copy()
        
        lr *= lr_decay
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c * dx))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def multi_grad(h, n, lags, weights):
    g = np.zeros(n)
    for k, w in zip(lags, weights):
        if w < 1e-10:
            continue
        s = k - (n - 1)
        if s >= 0:
            if s < n:
                g[s:] += w * (1.0 - h[:n - s])
                g[:n - s] -= w * h[s:]
        else:
            s2 = -s
            if s2 < n:
                g[:n - s2] += w * (1.0 - h[s2:])
                g[s2:] -= w * h[:n - s2]
    return g

def lse_weights(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    max_s = shifted.max()
    exp_s = np.exp(shifted - max_s)
    return exp_s / exp_s.sum(), m

def slsqp_lse(h, n, alpha, maxiter=2000):
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = get_corr(hx, n)
        w, m = lse_weights(c, alpha)
        return m + np.log(w.sum() * np.exp(0)) / alpha + (alpha * (c - m).max() * 0)
    def obj2(x):
        hx = np.clip(x, 0, 1)
        c = get_corr(hx, n)
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    def sfun(x):
        return float(np.sum(x) - n / 2.0)
    def sjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
    try:
        res = minimize(obj2, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lse_local_search(h, n, deadline, alpha=5000):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    curr = h.copy()
    rng = np.random.RandomState(31337)
    T = 0.015
    step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    stall = 0
    while time.time() < deadline:
        c = get_corr(curr, n)
        weights, m = lse_weights(c, alpha)
        sig = np.where(weights > 1e-7)[0]
        sig = sig[np.argsort(weights[sig])[::-1]][:30]
        g = multi_grad(curr, n, sig, weights[sig])
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        best_d = 0.0
        best_c_try = compute_c5(curr, n)
        for d in step_sizes:
            d = min(d, curr[i_src] * 0.95, (1.0 - curr[i_dst]) * 0.95)
            if d < 1e-10:
                continue
            trial = curr.copy()
            trial[i_src] -= d
            trial[i_dst] += d
            c_try = compute_c5(trial, n)
            if c_try < best_c_try:
                best_c_try = c_try
                best_d = d
        if best_d > 0:
            curr[i_src] -= best_d
            curr[i_dst] += best_d
            if best_c_try < best_c:
                best_c = best_c_try
                best_h = curr.copy()
            stall = 0
        else:
            if T > 1e-7:
                delta = best_c_try - best_c
                if rng.random() < np.exp(-delta / T):
                    d = min(step_sizes[0], curr[i_src] * 0.95, (1.0 - curr[i_dst]) * 0.95)
                    if d > 1e-10:
                        curr[i_src] -= d
                        curr[i_dst] += d
            stall += 1
            T *= 0.998
            if stall > 200:
                if rng.random() < 0.1:
                    for _ in range(5):
                        idx = rng.randint(n)
                        curr[idx] = rng.rand()
                    curr = project_feasible(curr, n)
                stall = 0
                T = 0.01
    return best_h, best_c

def pgd_lse(h, n, deadline, alpha=20000):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    curr = h.copy()
    lr = 0.005
    mom = np.zeros(n)
    beta = 0.85
    while time.time() < deadline:
        c = get_corr(curr, n)
        weights, m = lse_weights(c, alpha)
        sig = np.where(weights > 1e-6)[0]
        sig = sig[np.argsort(weights[sig])[::-1]][:40]
        g = multi_grad(curr, n, sig, weights[sig])
        mom = beta * mom - lr * g
        curr = curr + mom
        curr = project_feasible(curr, n)
        c_now = compute_c5(curr, n)
        if c_now < best_c:
            best_c = c_now
            best_h = curr.copy()
        lr *= 0.9995
        if lr < 1e-6:
            break
    return best_h, best_c

def multi_pos_search(h, n, deadline, batch=4):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    curr = h.copy()
    alpha = 10000
    while time.time() < deadline:
        c = get_corr(curr, n)
        weights, m = lse_weights(c, alpha)
        sig = np.where(weights > 1e-7)[0]
        sig = sig[np.argsort(weights[sig])[::-1]][:20]
        g = multi_grad(curr, n, sig, weights[sig])
        srcs = np.argsort(g)[-batch:]
        dsts = np.argsort(g)[:batch]
        improved = False
        for i in srcs:
            for j in dsts:
                if i == j:
                    continue
                for d in [0.005, 0.002, 0.001, 0.0005]:
                    d = min(d, curr[i] * 0.9, (1.0 - curr[j]) * 0.9)
                    if d < 1e-10:
                        continue
                    trial = curr.copy()
                    trial[i] -= d
                    trial[j] += d
                    c_try = compute_c5(trial, n)
                    if c_try < best_c:
                        curr = trial
                        best_c = c_try
                        best_h = curr.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if i != j:
                d = 0.0005
                d = min(d, curr[i] * 0.9, (1.0 - curr[j]) * 0.9)
                if d > 1e-12:
                    trial = curr.copy()
                    trial[i] -= d
                    trial[j] += d
                    if compute_c5(trial, n) < best_c + 0.0005:
                        curr = trial
    return best_h, best_c

def ilses(h, n, deadline, seed=42):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(seed)
    remaining = deadline - time.time()
    n_restarts = max(3, int(remaining / 60))
    for r in range(n_restarts):
        if time.time() > deadline - 40:
            break
        scale = 0.005 + 0.04 * (r / max(1, n_restarts - 1))
        hp = best_h + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        alloc = min(55, (deadline - time.time() - 40) * 0.5)
        if alloc < 20:
            break
        h1, c1 = lse_local_search(hp, n, time.time() + alloc * 0.6, alpha=8000)
        if time.time() < deadline - 45:
            h1, c1 = multi_pos_search(h1, n, time.time() + alloc * 0.3, batch=5)
        if c1 < best_c:
            best_c = c1
            best_h = h1
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 250
        h_init = np.full(n, 0.5)
    best_h = project_feasible(h_init.copy(), n)
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP
    for a in [500, 5000, 50000]:
        if time.time() > deadline - 200:
            break
        ht = slsqp_lse(best_h, n, a, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht
    # Phase 2: PGD
    if time.time() < deadline - 120:
        best_h, best_c = pgd_lse(best_h, n, min(deadline - 80, time.time() + 80), alpha=20000)
    # Phase 3: LSE local search
    if time.time() < deadline - 60:
        best_h, best_c = lse_local_search(best_h, n, min(deadline - 30, time.time() + 100), alpha=8000)
    # Phase 4: Multi-position search
    if time.time() < deadline - 30:
        best_h, best_c = multi_pos_search(best_h, n, deadline - 5, batch=5)
    # Phase 5: ILS
    if time.time() < deadline - 80:
        best_h, best_c = ilses(best_h, n, deadline - 30, seed=seed)
    # Phase 6: Try other resolutions
    if time.time() < deadline - 60:
        x_init = np.linspace(0, 2, len(best_h))
        for n_try in [150, 300]:
            if time.time() > deadline - 30:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, best_h)
            h_interp = project_feasible(h_interp, n_try)
            ht = slsqp_lse(h_interp, n_try, 5000, maxiter=1000)
            ht, c2 = lse_local_search(ht, n_try, min(deadline - 20, time.time() + 50), alpha=5000)
            if c2 < best_c:
                best_c = c2
                best_h = ht
    # Final polish
    if time.time() < deadline - 2:
        nf = len(best_h)
        best_h, best_c = multi_pos_search(best_h, nf, deadline - 1, batch=6)
        best_h = project_feasible(best_h, nf)
        best_c = compute_c5(best_h, nf)
    return best_h, best_c, len(best_h)
```
