def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]

    best_c = compute_c5(hb, n)
    ones_list = [int(x) for x in np.where(hb == 1)[0]]
    zeros_list = [int(x) for x in np.where(hb == 0)[0]]

        if time.time() > deadline:
            break

        rng.shuffle(zeros_list)
        rng.shuffle(ones_list)
        for i_idx in range(len(zeros_list)):
