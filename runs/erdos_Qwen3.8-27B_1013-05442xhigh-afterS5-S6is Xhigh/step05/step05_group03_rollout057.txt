Part 2: grad[1:3] -= h[0:2] → grad[1] -= h[0], grad[2] -= h[1] ✓

So grad[0] = 1-h[1], grad[1] = 1-h[2]-h[0], grad[2] = -h[1] ✓



Now let me put it all together. The full smooth gradient is:

grad = dx * Σ_{k=0}^{2n-2} p[k] * (gradient of c_{k-(n-1)})

For k from 0 to 2n-2, the lag is ℓ = k - (n-1).

Let me code this up efficiently:

```python
def c5_smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    e = np.exp(sh - sm)
    p = e / e.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = p[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:n] += wk * (1.0 - h[0:n-lag])
            grad[0:n-lag] -= wk * h[lag:n]
        else:
            m2 = -lag
            grad[0:n-m2] += wk * (1.0 - h[m2:n])
            grad[m2:n] -= wk * h[0:n-m2]
    return grad * dx
```

This is O(n^2) per call (2n-1 iterations, each O(n)). For n=250, that's about 125000 operations, which should take about 100-200 μs in Python (slower than a single C loop, but acceptable).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.clip(h[mask] + diff / cnt, 0, 1)
        else:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.clip(h[mask] + diff / cnt, 0, 1)
    return np.clip(h, 0.0, 1.0)

def c5_smooth_val(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
