def compute_gradient_vec(h, n, w):

    one_minus_h = 1 - h
    grad = np.zeros(n)
    # For each lag s (from -(n-1) to n-1):

    # For s >= 0: add w*s to grad[s:] *= one_minus_h[:n-s]
    #              sub w*s from grad[:n-s] *= h[s:]
    # For s < 0 (ss=-s > 0): add w to grad[:n-ss] *= one_minus_h[ss:]

        else:
            ss = -s
            grad[:n-ss] += wm * one_minus_h[ss:]

    grad = np.zeros(n)
    omh = 1.0 - h
    for s in range(-(n - 1), n):
        wm = w[s + n - 1]
        if wm < 1e-10:
            continue
        if s >= 0:
            grad[s:] += wm * omh[:n - s]
            grad[:n - s] -= wm * h[s:]
        else:
            ss = -s
            grad[:n - ss] += wm * omh[ss:]
            grad[ss:] -= wm * h[:n - ss]
    return grad * dx

def wgrad_optimize(h, n, deadline, beta_schedule=None):
    """Weighted gradient descent with adaptive step size."""
    h = h.copy()
    best_c = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    it = 0
    lr = 0.002
    stall = 0
    if beta_schedule is None:
        beta_schedule = [30, 50, 80, 120, 200]
    global beta  # will use module-level
    for b in beta_schedule:
        if time.time() > deadline:
            break
        _set_beta(b)
        while time.time() < deadline:
            it += 1
            grad = wgrad_direction(h, n)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            direction = -grad / gnorm
            # Try multiple step sizes
            found = False
            for t in [lr, lr * 5, lr * 20, lr * 0.5, lr * 0.1]:
                h_try = project_feasible(h + t * direction, n)
                c_try = compute_c5(h_try)
                if c_try < best_c - 1e-15:
                    best_c = c_try
                    best_h = h_try.copy()
                    h = h_try
                    found = True
                    break
            if not found:
                lr *= 0.7
                stall += 1
                if stall > 50 or lr < 1e-12:
                    break
            else:
                stall = 0
                lr = min(lr * 1.1, 0.05)
    return best_h, best_c

_beta_val = 80.0
def _set_beta(b):
    global _beta_val
    _beta_val = b

def wgrad_direction2(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_max = c.max()
    w = np.exp(beta * (c - c_max))
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(2 * n - 1)
        ws = 2 * n - 1
    w = w / ws
    grad = np.zeros(n)
    omh = 1.0 - h
    for s in range(-(n - 1), n):
        wm = w[s + n - 1]
        if wm < 1e-10:
            continue
        if s >= 0:
            grad[s:] += wm * omh[:n - s]
            grad[:n - s] -= wm * h[s:]
        else:
            ss = -s
            grad[:n - ss] += wm * omh[ss:]
            grad[ss:] -= wm * h[:n - ss]
    return grad * dx

def wgrad_optimize2(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h)
    best_h = h.copy()
    lr = 0.003
    for beta in [20, 40, 60, 100, 150, 250, 400]:
        if time.time() > deadline:
            break
        stall = 0
        while time.time() < deadline and stall < 80:
            grad = wgrad_direction2(h, n, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            direction = -grad / gnorm
            found = False
            for t in [lr, lr * 3, lr * 10, lr * 0.3, lr * 0.1]:
                h_try = project_feasible(h + t * direction, n)
                c_try = compute_c5(h_try)
                if c_try < best_c - 1e-15:
                    best_c = c_try
                    best_h = h_try.copy()
                    h = h_try
                    found = True
                    break
            if found:
                stall = 0
                lr = min(lr * 1.05, 0.05)
            else:
                stall += 1
                lr *= 0.6
                if lr < 1e-12:
                    break
    return best_h, best_c

def pair_search(h, n, deadline, deltas, max_passes=100):
    h = h.copy()
    best_c = compute_c5(h)
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(max_passes):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi, hj = h[i] + d, h[j] - d
                    if 0 <= hi <= 1 and 0 <= hj <= 1:
                        h[i], h[j] = hi, hj
                        h = project_feasible(h, n)
                        c = compute_c5(h)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                            h = project_feasible(h, n)
                    hi2, hj2 = h[i] - d, h[j] + d
                    if 0 <= hi2 <= 1 and 0 <= hj2 <= 1:
                        h[i], h[j] = hi2, hj2
                        h = project_feasible(h, n)
                        c = compute_c5(h)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi2 + d, hj2 - d
                            h = project_feasible(h, n)
            if not improved:
                break
    return h, best_c

def slsqp_round(h, n, alpha, maxiter, deadline):
    if time.time() > deadline:
        return h, compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        ht = project_feasible(res.x, n)
        return ht, compute_c5(ht)
    except Exception:
        return h, compute_c5(h)

def optimize_n(h_start, n, deadline, rng):
    h = project_feasible(h_start, n)
    best_h = h.copy()
    best_c = compute_c5(h)

    # SLSQP rounds
    for a in [500, 2000, 10000, 50000, 200000]:
        if time.time() > deadline - 20:
            break
        ht, ct = slsqp_round(best_h, n, a, 3000, deadline - 5)
        if ct < best_c - 1e-14:
            best_c, best_h = ct, ht

    # Weighted gradient descent
    if time.time() < deadline - 15:
        ht, ct = wgrad_optimize2(best_h, n, deadline - 10)
        if ct < best_c - 1e-14:
            best_c, best_h = ct, ht

    # Pair search
    if time.time() < deadline - 5:
        if n <= 250:
            best_h, best_c = pair_search(best_h, n, deadline - 2,
                                         [0.005, 0.002, 0.001, 0.0005], max_passes=80)
        else:
            best_h, best_c = pair_search(best_h, n, deadline - 2,
                                         [0.005, 0.002, 0.001], max_passes=40)

    # Final SLSQP polish
    if time.time() < deadline - 3:
        ht, ct = slsqp_round(best_h, n, 1000000, 4000, deadline - 1)
        if ct < best_c - 1e-14:
            best_c, best_h = ct, ht

    return project_feasible(best_h, n), compute_c5(best_h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 250
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h)

    # Main optimization at initial resolution
    alloc1 = min((deadline - t0) * 0.4, 300)
    best_h, best_c = optimize_n(best_h, n_init, t0 + alloc1, rng)

    # Try higher resolution by interpolation
    x_init = np.linspace(0, 2, n_init)
    for n_try in [400, 300]:
        if n_try <= n_init or time.time() > deadline - 80:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_init, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(100, deadline - time.time() - 40)
        if alloc < 30:
            break
        ht, ct = optimize_n(h_interp, n_try, time.time() + alloc, rng)
        if ct < best_c:
            best_c, best_h = ct, ht

    # Try lower resolution
    for n_try in [200, 150, 100]:
        if n_try >= n_init or time.time() > deadline - 60:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_init, best_h)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(60, deadline - time.time() - 30)
        if alloc < 20:
            break
        ht, ct = optimize_n(h_interp, n_try, time.time() + alloc, rng)
        if ct < best_c:
            best_c, best_h = ct, ht

    # Random restarts from best solution
    if time.time() < deadline - 50:
        n = len(best_h)
        remaining = deadline - time.time() - 25
        n_restarts = max(1, int(remaining / 45))
        scales = [0.003, 0.008, 0.015, 0.03, 0.05]
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(40, deadline - time.time() - 25)
            if alloc < 12:
                break
            ht, ct = optimize_n(hp, n, time.time() + alloc, rng)
            if ct < best_c:
                best_c, best_h = ct, ht

    # Final fine polish
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        if n <= 250:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.001, 0.0005, 0.0002, 0.0001], max_passes=200)
        else:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.001, 0.0005, 0.0002], max_passes=50)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h)

    return best_h, best_c, len(best_h)
