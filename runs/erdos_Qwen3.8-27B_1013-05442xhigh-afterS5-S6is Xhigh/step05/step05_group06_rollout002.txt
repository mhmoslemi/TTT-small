        best_h = h.copy()
    # Phase 3: SLSQP refinement (10% budget)
    p3 = t0 + 0.70 * (budget_s - 5)
    if time.time() < p3:
        for alpha in [50000, 500000, 5000000]:
            if time.time() > p3:
                break
            h = slsqp_optimize(best_h.copy(), n, alpha, p3, maxiter=150)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    # Phase 4: Subgradient polish (8% budget)
    p4 = t0 + 0.78 * (budget_s - 5)
    if time.time() < p4:
        h, c = subgradient_polish(best_h.copy(), n, p4)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 5: Final pair descent (14% budget)
    p5 = dl - 3
    if time.time() < p5:
        h, c = pair_descent(best_h.copy(), n, p5, K=20)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    # Phase 6: Binary polish (last bit)
    if time.time() < dl - 1:
        hb, cb = binary_search_swaps(best_h.copy(), n, dl - 1)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
