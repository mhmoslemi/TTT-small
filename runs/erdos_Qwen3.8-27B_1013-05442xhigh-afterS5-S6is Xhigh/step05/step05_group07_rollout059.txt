from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c)), c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def active_set_optimize(h, n, top_m, max_iter, deadline):

    dx = 2.0 / n
    best_c = float('inf')
    best_h = h.copy()

        active_max = c.max()
        
        # Define objective: max of active lags (use smooth max)
        alpha = 5000
        def obj(x):
            c = np.correlate(x, 1 - x, mode='full') * dx
            active = c[active_idx]
            m = active.max()
            sh = alpha * (active - m)
            sm = sh.max()
            val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
            val += 1e6 * (x.sum() - n / 2.0) ** 2
            return val
        
        def grad(x):
            # Gradient of the smooth max over active lags
            c = np.correlate(x, 1 - x, mode='full') * dx

            sh = alpha * (active - m)

                best_h = h_new.copy()
                h = h_new

    return best_h, best_c

    h_bin = np.zeros(n)
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin[idx[:target]] = 1.0
    return h_bin

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c)), c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, n, idx):
    dx = 2.0 / n
    k = idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] += -h[k:]
    else:
        g[:n + k] += (1.0 - h[-k:])
        g[-k:] += -h[:n + k]
    return g * dx

def binary_search(h_bin, n, deadline, max_rounds=200):
    best_c, _ = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        c_val, _ = compute_c5(h_bin, n)
        if c_val < best_c - 1e-14:
            best_c, best_h = c_val, h_bin.copy()
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta, best_i, best_j = -1e-15, -1, -1
        for i in zeros:
            for j in ones:
                h_bin[i] = 1.0
                h_bin[j] = 0.0

        if best_i >= 0:
            h_bin[best_i] = 1.0
            h_bin[best_j] = 0.0
        else:
            break
    return best_h, best_c
