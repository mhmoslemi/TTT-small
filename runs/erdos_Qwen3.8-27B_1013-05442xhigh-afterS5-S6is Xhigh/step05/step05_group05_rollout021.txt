import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha, penalty):
    """Compute smooth max objective and its gradient."""
    g = compute_all_g(h, n)
    m = g.max()
    shifted = g - m
    max_shift = shifted.max()
    exps = np.exp(alpha * (shifted - max_shift))
    w = exps / exps.sum()
    # Objective value
    val = m + max_shift + np.log(exps.sum()) / alpha
    # Add penalty
    s = h.sum()
    val += penalty * (s - n/2.0)**2
    
    dx = 2.0 / n
    # Compute C[j] = sum_t w[j+t] * h[t]
    C = np.array([np.dot(w[j:j+n], h) for j in range(n)])
    grad = dx * (1.0 - 2.0 * C) + 2.0 * penalty * (s - n/2.0)
    
    return val, grad

def lbfgsb_smooth(h0, n, alpha, penalty, deadline, maxiter=5000):
    """Optimize using L-BFGS-B with smooth max + penalty."""
    def fun(x):
        val, _ = smooth_obj_grad(x, n, alpha, penalty)
        return val
    def jac(x):
        _, g = smooth_obj_grad(x, n, alpha, penalty)
        return g
    bounds = [(0.0, 1.0)] * n
    best_x = h0.copy()
    best_val = fun(best_x)
    try:
        res = minimize(fun, h0, jac=jac, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10,
                      'maxls': 50, 'disp': False})
        if res.x is not None and res.fun < best_val:
            best_x = res.x
    except:
        pass
    return project_feasible(best_x, n)

def slsqp_opt(h0, n, alpha, deadline, maxiter=2000):
    """SLSQP optimization with smooth max."""
    def obj(x):
        val, _ = smooth_obj_grad(x, n, alpha, 0.0)
        return val
    def obj_grad(x):
        _, g = smooth_obj_grad(x, n, alpha, 0.0)
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    best_x = h0.copy()
    best_c = compute_c5(best_x, n)
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        hx = project_feasible(res.x, n)
        c = compute_c5(hx, n)
        if c < best_c:
            best_x = hx
    except:
        pass
    return best_x

def subgrad_polish(h0, n, deadline, max_iter=100000):
    """Subgradient method on the active constraint."""
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    lr = 0.05
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(best_h, n)
        idx = np.argsort(g)[::-1][:3]  # Top 3 active constraints
        m = g[idx[0]]
        grad = np.zeros(n)
        for ii in idx:
            k = ii
            lag = k - (n - 1)
            w = 1.0 / len(idx)
            if lag >= 0:
                grad[lag:] += w * dx * (1.0 - best_h[:n-lag])
                if n-lag > 0:
                    grad[:n-lag] -= w * dx * best_h[lag:]
            else:
                to = -lag
                if n-to > 0:
                    grad[:n-to] += w * dx * (1.0 - best_h[to:])
                grad[to:] -= w * dx * best_h[:n-to]
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        grad /= ng
        # Backtracking line search
        found = False
        for f in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.01, 0.005]:
            step = lr * f
            ht = project_feasible(best_h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-14:
                best_h = ht
                found = True
                if ct < best_c:
                    best_c = ct
                break
        if not found:
            lr *= 0.95
            if lr < 1e-12:
                break
    return best_h, best_c

    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

For a binary solution h (with sum = n/2), the overlap at lag `lag` is:
g[lag] = dx * |{i: h[i]=1, h[i+lag]=0}| = dx * (number of 1s in h that don't have a 1 at position i+lag)

To reduce the max, we want to:

- To a position where it "helps" (i.e., filling a gap)

The key insight is: if the max is at lag L, then we want to maximize the number of i where h[i]=1 AND h[i+L]=1.

So for the critical lag L:
- If we set h[i]=1 (i was 0) and h[j]=0 (j was 1), the change in the count for lag L is:
  - If h[i+L]=1 (or i+L out of range→treat as 0): we gain 1 (new pair)
  - If h[i+L]=0: no gain
  - If h[j+L]=1 (or j+L out of range): we lose 1 (broken pair)  
  - If h[j+L]=0: no loss

So the net change is: [h[i+L]=1] - [h[j+L]=1] (with boundary handling).

We want this to be positive, i.e., h[i+L]=1 and h[j+L]=0.
