    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    vals = np.exp(sh - sm)
    w = vals / vals.sum()
    obj = m + (sm + np.log(vals.sum())) / alpha
    v = 1.0 - 2.0 * h
    grad = dx * np.convolve(w, v, mode='valid')
    return obj, grad

def grad_desc(h, n, deadline, alpha=100000, max_iter=200000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.5
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        _, g = smooth_obj_grad(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        step = lr / gn
        found = False
        for _ in range(30):
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                found = True
                break
            step *= 0.5
        if found:
            lr = min(lr * 1.03, 20.0)
            stall = 0
        else:
            lr *= 0.85
            stall += 1
            if lr < 1e-13 or stall > 100:
                break
    return best_h, best_c

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def binary_ls(hb, n, deadline, max_iters=5000000):
    best_c = compute_c5(hb, n)
    best_hb = hb.copy()
    ones = list(np.where(hb == 1)[0])
    zeros = list(np.where(hb == 0)[0])
    rng = np.random.default_rng(123)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = zeros[rng.integers(len(zeros))]
        j = ones[rng.integers(len(ones))]
        hb[i] = 1.0
        hb[j] = 0.0
        c = compute_c5(hb, n)
        if c < best_c - 1e-15:
            best_c = c
            best_hb = hb.copy()
            zeros.remove(i)
            ones.remove(j)
            ones.append(i)
            zeros.append(j)
        else:
            hb[i] = 0.0
            hb[j] = 1.0
    return best_hb, best_c

def slsqp_opt(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def con(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def make_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.full(n, 0.5) + 0.4 * rng.randn(n)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 8
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = make_init(n0, seed)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)

    # Phase 1: SLSQP continuation on n0 (25%)
    p1 = t0 + 0.25 * (budget_s - 8)
    h = best_h.copy()
    for alpha in [500, 2000, 10000, 50000, 200000, 1000000, 5000000]:
        if time.time() > p1 - 5:
            break
        h = slsqp_opt(h, n0, alpha, 800, p1 - 3)
        c = compute_c5(h, n0)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Gradient descent (15%)
    p2 = t0 + 0.40 * (budget_s - 8)
    h, c = grad_desc(best_h.copy(), n0, p2, alpha=500000, max_iter=150000)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 3: Binary local search on n0 (20%)
    p3 = t0 + 0.60 * (budget_s - 8)
    hb = to_binary(best_h, n0)
    hb, cb = binary_ls(hb, n0, p3, max_iters=8000000)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Also try from perturbed
    rng = np.random.RandomState(seed + 1)
    hp = project_feasible(best_h + 0.12 * rng.randn(n0), n0)
    hb2 = to_binary(hp, n0)
    hb2, cb2 = binary_ls(hb2, n0, p3, max_iters=3000000)
    if cb2 < best_c:
        best_c = cb2
        best_h = hb2.copy()

    # Phase 4: Try larger n (30%)
    p4 = t0 + 0.90 * (budget_s - 8)
    for n_try in [300, 400, 500]:
        if time.time() > p4 - 50:
            break
        h_t = make_init(n_try, seed + n_try)
        sub_dl = min(p4 - 30, time.time() + 70)
        for alpha in [2000, 20000, 200000, 2000000]:
            if time.time() > sub_dl - 5:
                break
            h_t = slsqp_opt(h_t, n_try, alpha, 400, sub_dl - 3)
        c_t = compute_c5(h_t, n_try)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t.copy()
        # Gradient polish
        sub_dl2 = min(p4 - 20, time.time() + 20)
        h_t, c_t = grad_desc(h_t, n_try, sub_dl2, alpha=500000, max_iter=30000)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t.copy()
        # Binary
        hb_t = to_binary(h_t, n_try)
        sub_dl3 = min(p4 - 15, time.time() + 40)
        hb_t, cb_t = binary_ls(hb_t, n_try, sub_dl3, max_iters=3000000)
        if cb_t < best_c:
            best_c = cb_t
            best_h = hb_t.copy()

    # Phase 5: Final polish (10%)
    nf = len(best_h)
    if time.time() < dl - 5:
        h = slsqp_opt(best_h.copy(), nf, 20000000, 1500, dl - 3)
        c = compute_c5(h, nf)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    if time.time() < dl - 3:
        h, c = grad_desc(best_h.copy(), nf, dl - 2, alpha=1000000, max_iter=50000)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    best_h = project_feasible(best_h, nf)
    best_c = compute_c5(best_h, nf)
    return best_h, best_c, nf
