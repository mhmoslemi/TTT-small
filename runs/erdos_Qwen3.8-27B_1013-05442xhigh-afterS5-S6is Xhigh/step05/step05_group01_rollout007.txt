import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr_vec(h, n):
    """Return the full correlation vector times dx."""
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    """Project onto {x in [0,1]^n : sum(x) = n/2}."""
    target = n / 2.0
    h = h.copy()
    # Bisection for lambda: sum(clip(h - lambda, 0, 1)) = target
    lo = h.min() - 0.5
    hi = h.max() + 0.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(h - lam, 0.0, 1.0)
    # Fix any tiny numerical issues
    diff = target - proj.sum()
    if abs(diff) > 1e-14:
        free = (proj > 1e-12) & (proj < 1 - 1e-12)
        if free.sum() > 0:
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def gradient_smooth_max(h, n, alpha=5000):
    """Compute gradient of smooth max of correlation."""
    c = compute_corr_vec(h, n)
    dx = 2.0 / n
    m = c.max()
    shifted = alpha * (c - m)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    
    # Only use terms with significant weight
    active = np.where(w > 1e-8)[0]
    
    grad = np.zeros(n)
    for k in active:
        wk = w[k]
        lag = k - (n - 1)
        if lag >= 0:
            # dc/dh[m] = dx*(1-h[m+lag]) for m in [0, n-lag)
            grad[:n-lag] += wk * dx * (1 - h[lag:n])
            # dc/dh[m] = -dx*h[m-lag] for m in [lag, n)
            grad[lag:] -= wk * dx * h[:n-lag]
        else:
            l = -lag
            # dc/dh[m] = dx*(1-h[m-l]) for m in [l, n)
            grad[l:] += wk * dx * (1 - h[:n-l])
            # dc/dh[m] = -dx*h[m+l] for m in [0, n-l)
            grad[:n-l] -= wk * dx * h[l:]
    
    return grad

def gradient_descent(h, n, deadline, alpha_start=100, alpha_end=100000, n_iter=5000, lr_start=0.01):
    """Custom projected gradient descent with adaptive alpha and learning rate."""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / n_iter)
        lr = lr_start * (0.999 ** it)
        
        grad = gradient_smooth_max(h, n, alpha)
        
        # Gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        # Line search: try a few step sizes
        best_step = 0
        best_obj = compute_c5(h, n)
        for frac in [1.0, 0.5, 0.25, 0.1]:
            h_try = h - lr * frac * grad
            h_try = project_feasible(h_try, n)
            obj = compute_c5(h_try, n)
            if obj < best_obj:
                best_obj = obj
                best_step = frac
                h_new = h_try
        
        if best_step > 0:
            h = h_new
            if best_obj < best_c:
                best_c = best_obj
                best_h = h.copy()
        else:
            # No improvement, reduce learning rate
            lr *= 0.5
            if lr < 1e-8:
                break
    
    return best_h, best_c

        shifted = alpha * (c - m)
        mx = shifted.max()
        exps = np.exp(shifted - mx)
        w = exps / exps.sum()
        grad = np.zeros(n)
        active = np.where(w > 1e-8)[0]
        for k in active:
            lag = k - (n - 1)
            grad += w[k] * grad_at_lag(h, n, lag)
        improved = False
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_feasible(h - lr * frac * grad, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-13:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                improved = True
                break
        if not improved:
            lr *= 0.85
            if lr < 1e-10:
                break
    return best_h, best_c

def pair_search(h, n, deadline, deltas=None, max_passes=800):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i + 1, n):
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def multi_start(h_base, n, deadline, rng, n_starts=5):
    best_h = h_base.copy()
    best_c = compute_c5(h_base, n)
    scales = [0.01, 0.02, 0.04, 0.07, 0.1]
    alloc_each = max(30, (deadline - time.time()) / max(n_starts, 1))
    for s in range(n_starts):
        if time.time() > deadline - 10:
            break
        scale = scales[s % len(scales)]
        hp = h_base + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        t_end = time.time() + min(alloc_each, deadline - time.time() - 5)
        if t_end - time.time() < 15:
            break
        # Quick bundle + gradient
        h1, c1 = bundle_method(hp, n, t_end - 15, max_iter=100, top_m=15, eps0=0.01)
        if c1 < best_c:
            best_c = c1; best_h = h1
        if time.time() < t_end - 5:
            h2, c2 = smooth_grad_descent(h1, n, t_end, alpha_start=5000, alpha_end=50000, max_iter=1000)
            if c2 < best_c:
                best_c = c2; best_h = h2
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 250
        h_init = np.full(n, 0.5)
    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    # Phase 1: Bundle method (primary)
    t1 = min(250, (deadline - t0) * 0.28)
    h, c = bundle_method(h, n, t0 + t1, max_iter=500, top_m=30, eps0=0.005)
    if c < best_c:
        best_c = c; best_h = h.copy()
    # Phase 2: Smooth gradient (moderate alpha)
    t2 = min(180, (deadline - t0) * 0.2)
    h, c = smooth_grad_descent(best_h, n, t0 + t2, alpha_start=1000, alpha_end=100000, max_iter=4000)
    if c < best_c:
        best_c = c; best_h = h.copy()
    # Phase 3: Bundle again (tighter)
    t3 = min(200, (deadline - t0) * 0.22)
    h, c = bundle_method(best_h, n, t0 + t3, max_iter=500, top_m=40, eps0=0.002)
    if c < best_c:
        best_c = c; best_h = h.copy()
    # Phase 4: Multi-start exploration
    remaining = deadline - time.time() - 60
    if remaining > 80:
        h, c = multi_start(best_h, n, time.time() + remaining, rng, n_starts=4)
        if c < best_c:
            best_c = c; best_h = h.copy()
    # Phase 5: Fine pair search
    remaining = deadline - time.time() - 2
    if remaining > 10:
        h, c = pair_search(best_h, n, time.time() + remaining,
                           deltas=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                           max_passes=1000)
        if c < best_c:
            best_c = c; best_h = h
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
