def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_c(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_desc_step(h, i, c_full, n):
    n_pts = 2 * n - 1
    k = np.arange(n_pts)
    # L: h[i] appears as h[n+k] in result[k], i.e., i = n+k, n = i-k
    # Valid: 0 <= i-k < n => k <= i and k > i-n => max(0, i-n+1) <= k <= i
    L = (k >= 0) & (k <= i) & (i - k >= 0) & (i - k < n)
    # R: h[i] appears in (1-h[n]) where n = i, valid when 0 <= i+k < n
    R = (k >= 0) & (k <= n_pts - 1) & (i + k >= 0) & (i + k < n)
    
    idx_l = i - np.arange(n_pts)
    idx_r = i + np.arange(n_pts)
    hl = np.where(idx_l >= 0, np.where(idx_l < n, np.take(h, np.clip(idx_l, 0, n-1)), 0.0), 0.0)
    hr = np.where(idx_r >= 0, np.where(idx_r < n, np.take(h, np.clip(idx_r, 0, n-1)), 0.0), 0.0)
    
    A = L.astype(float) * (1.0 - hl) - R.astype(float) * hr
    B = c_full - L.astype(float) * h[i] * (1.0 - hl) + R.astype(float) * hr * h[i]
    
    pos_mask = A > 1e-12
    neg_mask = A < -1e-12
    zero_mask = ~(pos_mask | neg_mask)
    
    if not np.any(pos_mask):
        return 1.0
    if not np.any(neg_mask):
        return 0.0
    
    # g(t) = max_{A>0}(A*t+B) - max_{A<0}(A*t+B) is monotonically increasing
    g0 = np.max(B[pos_mask]) - np.max(B[neg_mask])
    if g0 >= 0:
        return 0.0
    g1 = np.max(A[pos_mask] + B[pos_mask]) - np.max(A[neg_mask] + B[neg_mask])
    if g1 <= 0:
        return 1.0
    
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        g_mid = np.max(A[pos_mask] * mid + B[pos_mask]) - np.max(A[neg_mask] * mid + B[neg_mask])
        if g_mid > 0:
            hi = mid
        else:
            lo = mid
    
    # Also check the zero-slope lines
    t_star = (lo + hi) / 2
    # Verify it's actually better than boundaries
    f0 = np.max(B)
    f1 = np.max(A + B)
    fm = np.max(A * t_star + B)
    candidates = [(0.0, f0), (1.0, f1), (t_star, fm)]
    best = min(candidates, key=lambda x: x[1])
    return best[0]

def coord_descent(h, n, max_sweeps=5000, deadline=None, seed=42):

    best_c = compute_c5(best_h, n)

        if deadline is not None and time.time() > deadline:
            break
        c_full = compute_all_c(best_h, n)
        prev_max_c = np.max(c_full)

                return best_h, best_c
            t = coord_desc_step(best_h, i, c_full, n)
            best_h[i] = t
        
        best_h = project_feasible(best_h, n)
        c = compute_c5(best_h, n)
        if c < best_c - 1e-15:
            best_c = c
        elif abs(c - best_c) < 1e-14:
            # Check if we've converged
            pass
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

    dx = 2.0 / n

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_desc_step(h, i, c_full, n):
    n_pts = 2 * n - 1
    k = np.arange(n_pts)
    L = k <= i
    R = k < (n - i)
    idx_l = i - k
    idx_r = i + k
    hl = np.zeros(n_pts)
    hl[L] = h[idx_l[L]]
    hr = np.zeros(n_pts)
    hr[R] = h[idx_r[R]]
    Lf = L.astype(float)
    Rf = R.astype(float)
    A = Lf * (1.0 - hl) - Rf * hr
    B = c_full - Lf * h[i] * (1.0 - hl) + Rf * hr * h[i]
    pos = A > 1e-12
    neg = A < -1e-12
    if not np.any(pos) and not np.any(neg):
        return h[i]
    if not np.any(pos):
        return 1.0
    if not np.any(neg):
        return 0.0
    g0 = np.max(B[pos]) - np.max(B[neg])
    if g0 >= 0:
        return 0.0
    g1 = np.max(A[pos] + B[pos]) - np.max(A[neg] + B[neg])

    lo, hi = 0.0, 1.0
