import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def greedy_oneswap(hb, n, deadline, max_sweeps=100):
    best_h = hb.copy()
    best_c = compute_c5(hb, n)
    cur_h = hb.copy()
    cur_c = best_c
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(cur_h == 0)[0]
        ones = np.where(cur_h == 1)[0]
        for i in zeros:
            for j in ones:
                cur_h[i], cur_h[j] = 1.0, 0.0
                c = compute_c5(cur_h, n)
                if c < cur_c - 1e-15:
                    cur_c = c
                    improved = True
                    ones = np.where(cur_h == 1)[0]
                else:
                    cur_h[i], cur_h[j] = 0.0, 1.0
        if cur_c < best_c - 1e-15:
            best_c = cur_c
            best_h = cur_h.copy()
        if not improved:
            break
    return best_h, best_c

def sa_oneswap(hb, n, t_start, t_end, deadline, rng):
    best_h = hb.copy()
    best_c = compute_c5(hb, n)
    cur_h = hb.copy()
    cur_c = best_c
    zeros = np.where(cur_h == 0)[0].tolist()
    ones = np.where(cur_h == 1)[0].tolist()
    t_start_actual = time.time()
    duration = max(deadline - t_start_actual, 1.0)
    while time.time() < deadline:
        elapsed = time.time() - t_start_actual
        frac = min(elapsed / duration, 1.0)
        t = t_start * (t_end / t_start) ** frac
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        cur_h[i] = 1.0
        cur_h[j] = 0.0
        new_c = compute_c5(cur_h, n)
        delta = new_c - cur_c
        if delta < 0 or (t > 1e-15 and rng.random() < np.exp(-delta / t)):
            cur_c = new_c
            zeros.remove(i)
            zeros.append(j)
            ones.remove(j)
            ones.append(i)
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[i] = 0.0
            cur_h[j] = 1.0
    return best_h, best_c

def sa_twoswap(hb, n, t_start, t_end, deadline, rng):
    best_h = hb.copy()
    best_c = compute_c5(hb, n)
    cur_h = hb.copy()
    cur_c = best_c
    zeros = np.where(cur_h == 0)[0].tolist()
    ones = np.where(cur_h == 1)[0].tolist()
    t_start_actual = time.time()

        j2 = ones[rng.randint(len(ones))]

        cur_h[i1] = 1.0
        cur_h[i2] = 1.0
        cur_h[j1] = 0.0
        cur_h[j2] = 0.0
