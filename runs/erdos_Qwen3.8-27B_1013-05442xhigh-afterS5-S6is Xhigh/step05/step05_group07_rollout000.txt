def lp_refine(h, n, max_iter=100, deadline=None):

    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(max_iter):
        if deadline and time.time() > deadline:
            break
        
        # Compute current overlaps
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = c.max()

        # For overlap at lag k (shift = k - (n-1)):
        # dO(k)/dh(i) = dx * [(1-h(i+k)) - h(i-k)] for valid indices

        # Variables: d (n-vector), t (scalar)
        # Minimize: t
        # Subject to: for all k: c[k] + dx * sum_i grad_k[i] * d[i] <= t
        #             sum(d) = 0
        #             -h[i] <= d[i] <= 1 - h[i]

        t = cp.Variable()
        
        constraints = [t >= m - 0.01]  # Warm start: t at least current max minus a bit

            constraints.append(c[k_idx] + grad @ d <= t)
        
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:

                prob.solve(solver=cp.CLARABEL)

        # Check if this actually helps
