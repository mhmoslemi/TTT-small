    g = np.zeros(n)
    idx = np.arange(n)
    mask_pos = (idx + s) < n
    g[mask_pos] += (1.0 - h[idx[mask_pos] + s])

    g[mask_neg] -= h[idx[mask_neg] - s]

    def obj(x):
        hx = np.clip(x, 0.0, 1.0)
        c = np.correlate(hx, 1.0 - hx, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return m + (sm + np.log(np.sum(np.exp(sh - sm)))) / alpha
    def _sc(x):
        return float(np.sum(x) - n / 2.0)

        return np.ones(n)

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, k_idx):
    s = k_idx - (n - 1)
    g = np.zeros(n)
    idx = np.arange(n)
    mpos = (idx + s) < n
    g[mpos] += (1.0 - h[idx[mpos] + s])
    mneg = (idx - s) >= 0
    g[mneg] -= h[idx[mneg] - s]
    return g

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_round(h, n, alpha, maxiter=2500):
    dx = 2.0 / n
    def obj(x, n, a):
        hx = np.clip(x, 0.0, 1.0)
        c = np.correlate(hx, 1.0 - hx, mode='full') * dx
        m = c.max()
        sh = a * (c - m)
        sm = sh.max()
        return m + (sm + np.log(np.sum(np.exp(sh - sm)))) / a
    def obj_wrapped(x):
        return obj(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except:
        return h.copy()
