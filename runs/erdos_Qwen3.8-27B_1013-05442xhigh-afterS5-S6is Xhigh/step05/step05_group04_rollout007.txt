    if rem > 5:
        ht = slsqp_grad(h, n, 2000000, maxiter=200)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht

    rem = deadline - time.time()
    if rem > 3:
        fine = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
        if n <= 160:
            h, best_c = pair_search(h, n, deadline - 1, fine)
        else:
            h, best_c = near_pair(h, n, deadline - 1, fine, radius=30)

    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize initial
    alloc1 = min(350, (deadline - t0) * 0.35)
    h_opt, c_opt = optimize_n(h_init, n_init, t0 + alloc1, rng)
    best_h, best_c = h_opt, c_opt

    # Phase 2: Try other n
    x_init = np.linspace(0, 2, n_init)
    for n_try in [150, 200, 250, 300]:
        if time.time() > deadline - 100:
            break
        if n_try == n_init:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_init, h_init)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(120, (deadline - time.time()) * 0.2)
        if alloc < 30:
            break
        h_opt, c_opt = optimize_n(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Multi-restart with structured perturbation
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(3, int(remaining / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            if r % 3 == 0:
                scale = 0.02 + 0.04 * r / max(1, n_restarts)
                hp = best_h + rng.normal(0, scale, n)
            elif r % 3 == 1:
                # Shift perturbation
                shift = rng.randint(1, max(2, n // 10))
                hp = np.roll(best_h, shift)
                hp = project_feasible(hp, n)
            else:
                # Impulse perturbation
                hp = best_h.copy()
                idx = rng.randint(0, n, size=max(1, n // 20))
                amp = rng.uniform(0.05, 0.15, len(idx)) * rng.choice([-1, 1], len(idx))
                hp[idx] += amp
                hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 20) * 0.35)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Extended pair search on best
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        if n <= 160:
            best_h, best_c = pair_search(best_h, n, deadline - 5,
                                         [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        else:
            best_h, best_c = near_pair(best_h, n, deadline - 5,
                                       [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001],
                                       radius=min(60, n // 4))

    # Phase 5: Final SLSQP + fine pair
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        ht = slsqp_grad(best_h, n, 5000000, maxiter=500)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
        rem = deadline - time.time()
        if rem > 3:
            if n <= 160:
                best_h, best_c = pair_search(best_h, n, deadline - 1,
                                             [0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
            else:
                best_h, best_c = near_pair(best_h, n, deadline - 1,
                                           [0.0005, 0.0002, 0.0001, 0.00005],
                                           radius=40)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
