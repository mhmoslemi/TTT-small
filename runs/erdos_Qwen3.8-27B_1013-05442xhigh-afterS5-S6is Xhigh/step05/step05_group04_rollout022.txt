def active_set_lp(h_old, n, dx, num_active=20):
    all_c = np.correlate(h_old, 1 - h_old, mode='full') * dx
    max_c = all_c.max()
    # Get top lags
    sorted_idx = np.argsort(all_c)[::-1]
    active_lags = sorted_idx[:num_active]
    
    # Build LP
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0
    ]
    
    for k in active_lags:
        lag = int(k) - (n - 1)
        grad = gradient_c_vec(h_old, n, lag)
        c_k = all_c[k]
        # Linearized: c_k + grad · (h - h_old) <= t
        # => grad · h - t <= c_k - grad · h_old
        constraints.append(grad @ h_var - t_var <= c_k - grad @ h_old)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    prob.solve(solver=cp.SCS, verbose=False)
    
    if h_var.value is not None:
        h_new = np.array(h_var.value).flatten()
