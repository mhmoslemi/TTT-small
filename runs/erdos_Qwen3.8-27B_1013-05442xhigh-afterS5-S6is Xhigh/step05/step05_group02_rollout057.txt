def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, deadline, max_sweeps=500):
    """Coordinate descent with optimal per-coordinate update."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c = get_c(h, n)
    sweep = 0
    stall = 0
    
    while time.time() < deadline and sweep < max_sweeps:

        improved = False

                break

            a = np.zeros(2 * n - 1)

            s_vals = np.arange(-(n-1), n)
            j1 = i + s_vals  # i+s
            j2 = i - s_vals  # i-s
            valid1 = (j1 >= 0) & (j1 < n)
            valid2 = (j2 >= 0) & (j2 < n)
            h_j1 = np.where(valid1, 1.0, 0.0)
            h_j2 = np.where(valid2, 1.0, 0.0)
            # Need h[j1] where valid, 0 otherwise

            h_j2_vals = np.zeros(2*n-1)

    bounds = [(0.0, 1.0)] * n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_fun(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def softmax_obj(x, h0, n, alpha):
    dx = 2.0 / n
    h_x = np.clip(x, 0, 1)
    c_val = np.correlate(h_x, 1 - h_x, mode='full') * dx
    m = c_val.max()
    shifted = alpha * (c_val - m)
    mx = shifted.max()
    return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha

def softmax_slsqp(h, n, alpha, maxiter=3000):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    def obj(x):
        h_x = np.clip(x, 0, 1)
        c_val = np.correlate(h_x, 1 - h_x, mode='full') * dx
        m = c_val.max()
        shifted = alpha * (c_val - m)
        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
