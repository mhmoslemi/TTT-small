from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def thue_morse(n):
    """Generate Thue-Morse sequence of length n (balanced)."""
    t = np.zeros(n)
    for i in range(n):
        t[i] = bin(i).count('1') % 2
    # Balance: ensure exactly n//2 ones
    target = n // 2
    ones = int(t.sum())
    if ones > target:
        # Flip some 1s to 0s
        idx = np.where(t == 1)[0]
        for i in range(ones - target):
            t[idx[i]] = 0
    elif ones < target:
        idx = np.where(t == 0)[0]
        for i in range(target - ones):
            t[idx[i]] = 1
    return t

def smooth_max(h, n, alpha):
    g = compute_all_g(h, n)
    m = g.max()
    sh = g - m
    return m + (sh.max() + np.log(np.exp(sh - sh.max()).sum())) / alpha

def slsqp_opt(h0, n, alpha, maxiter, deadline):
    def obj(x):
        return smooth_max(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})

    except Exception:
        return h0

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def smooth_max_obj(x, n, alpha):
    g = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = g.max()
    sh = g - m
    return m + np.log(np.exp(sh - sh.max()).sum()) / alpha

def slsqp_opt(h0, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(smooth_max_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, alpha),
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def thue_morse_binary(n, seed=0):
    t = np.zeros(n)
    for i in range(n):
        t[i] = bin(i).count('1') % 2
    target = n // 2
    ones = int(t.sum())
    rng = np.random.RandomState(seed)
    if ones > target:
        idx = np.where(t == 1)[0]
        rng.shuffle(idx)
        for i in range(ones - target):
            t[idx[i]] = 0
    elif ones < target:
        idx = np.where(t == 0)[0]
        rng.shuffle(idx)
        for i in range(target - ones):
            t[idx[i]] = 1
    return t

def binary_sa(h0, n, deadline, seed=42):
    h = make_binary(h0, n)
    cur_c = compute_c5(h, n)
    best_h = h.copy()
    best_c = cur_c
    rng = np.random.RandomState(seed)
    it = 0
    t_start, t_end = 0.05, 0.0005
    stall = 0
    while time.time() < deadline:
        it += 1
        T = t_start * (t_end / t_start) ** (it / 300000)
        if T < t_end:
            T = t_end
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i], h[j] = 0.0, 1.0
        new_c = compute_c5(h, n)
        if new_c <= cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c - 1e-15:
                best_c = new_c
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
        if stall > 3000:
            # Kick: swap 5 random pairs
            for _ in range(5):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                a = int(rng.choice(ones))
                b = int(rng.choice(zeros))
                h[a], h[b] = 0.0, 1.0
            cur_c = compute_c5(h, n)
            stall = 0
            T = max(T, t_start * 0.3)  # Reheat
    return best_h, best_c

def subgrad_polish(h0, n, deadline, max_iter=100000):
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    lr = 0.05
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(best_h, n)
        m = g.max()
        k = int(np.argmax(g))
        lag = k - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += dx * (1.0 - best_h[:n - lag])
                grad[:n - lag] -= dx * best_h[lag:]
        else:
            to = -lag
            if n - to > 0:
                grad[:n - to] += dx * (1.0 - best_h[to:])
                grad[to:] -= dx * best_h[:n - to]
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        grad /= ng
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005]:
            step = lr * f
            ht = project_feasible(best_h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-13:
                best_h = ht
                found = True
                break
        if not found:
            lr *= 0.85
            if lr < 1e-12:
                break
        else:
            c = compute_c5(best_h, n)
            if c < best_c:
                best_c = c
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)

        n0 = 250
        h0 = np.full(n0, 0.5)
    
    n = n0
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    p1 = t0 + 0.25 * (budget_s - 5)
    h = best_h.copy()

        if time.time() > p1 - 5:
