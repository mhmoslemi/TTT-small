    mid = n - 1
    if n == 1:
        return (1.0 - 2.0 * h) * w[mid] * dx
    w_pos = np.zeros(n)
    w_neg = np.zeros(n)
    w_pos[1:] = w[mid + 1:]
    w_neg[1:] = w[:mid][::-1]
    s = w_pos + w_neg
    pref_p = np.cumsum(w_pos)
    pref_n = np.cumsum(w_neg)
    idx = np.arange(n)
    consts = pref_p + pref_n[n - 1 - idx]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx

def _obj_f(x):
    n, dx = ST['n'], ST['dx']
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, w = softmax_topk(c, ST['beta'], ST['topk'])
    ST['_w'] = w
    return f

def _obj_g(x):
    w = ST.get('_w')
    if w is None:
        n, dx = ST['n'], ST['dx']
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        _, w = softmax_topk(c, ST['beta'], ST['topk'])
    n, dx = ST['n'], ST['dx']
    return grad_w(x, w, n, dx)

def lbfgs_step(h, beta, topk, maxiter, deadline):
    n = len(h)
    rem = deadline - time.time()
    if rem < 5.0:
        return h, erdos_c5(h, n)
    ST['n'] = n
    ST['dx'] = 2.0 / n
    ST['beta'] = float(beta)
    ST['topk'] = int(topk)
    ST['_w'] = None
    bounds = [(0.0, 1.0)] * n
    mx = min(maxiter, max(15, int(rem / 5.0)))
    try:
        res = so.minimize(_obj_f, h, method='L-BFGS-B', jac=_obj_g,
                          bounds=bounds,
                          options={'maxiter': mx, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxls': 40, 'disp': False})
        cand = project_box_sum(res.x, 0.5 * n)
    except Exception:
        cand = project_box_sum(h, 0.5 * n)
    return cand, erdos_c5(cand, n)

def pair_polish(h, n, deadline, seed, n_pairs=40, rounds=4):
    rng = np.random.RandomState(seed)
    h = h.copy()
    cur = erdos_c5(h, n)
    frac = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
    if frac.size:
        score = np.minimum(h[frac], 1.0 - h[frac])
        order = frac[np.argsort(score)][::-1]
    else:
        order = np.array([], dtype=int)
    if len(order) >= n_pairs:
        top = order[:n_pairs]
    else:
        need = n_pairs - len(order)
        if need > n - len(order):
            need = max(0, n - len(order))
        if need > 0:
            extra = rng.choice(n, size=need, replace=False)
            top = np.unique(np.concatenate([order, extra]))
        else:
            top = order
    if len(top) < 2:
        return h, cur
    for _ in range(rounds):
        improved = False
        pairs = [(int(top[a]), int(top[b])) 
                 for a in range(len(top)) for b in range(a + 1, len(top))]
        rng.shuffle(pairs)
        for i, j in pairs:
            if time.time() > deadline:
                return h, erdos_c5(h, n)
            hi0, hj0 = h[i], h[j]
            lo = max(-hi0, hj0 - 1.0)
            hi = min(1.0 - hi0, hj0)
            if hi - lo < 1e-13:
                continue
            best_d, best_c = 0.0, cur
            for t in np.linspace(lo, hi, 7):
                h2 = h.copy()
                h2[i], h2[j] = hi0 + t, hj0 - t
                c2 = erdos_c5(h2, n)
                if c2 < best_c:
                    best_c, best_d = c2, t
            if best_c < cur - 1e-14:
                a, b = max(lo, best_d - 0.03), min(hi, best_d + 0.03)
                for _ in range(25):
                    m1 = a + 0.382 * (b - a)
                    m2 = a + 0.618 * (b - a)
                    h1 = h.copy()
                    h1[i], h1[j] = hi0 + m1, hj0 - m1
                    h2 = h.copy()
                    h2[i], h2[j] = hi0 + m2, hj0 - m2
                    c1, c2 = erdos_c5(h1, n), erdos_c5(h2, n)
                    if c1 < c2:
                        b = m2
                    else:
                        a = m1
                    if b - a < 1e-12:
                        break
                best_d = 0.5 * (a + b)
                h[i], h[j] = hi0 + best_d, hj0 - best_d
                np.clip(h, 0.0, 1.0, out=h)
                cur = erdos_c5(h, n)
                improved = True
        if not improved:
            break
    return h, erdos_c5(h, n)

def block_swap_polish(h, n, deadline, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    cur = erdos_c5(h, n)
    for blen in [2, 3, 5, 8, 12]:
        for _ in range(3):
            improved = False
            for _ in range(50):
                if time.time() > deadline:
                    return h, erdos_c5(h, n)
                i = rng.randint(0, max(1, n - blen))
                j = rng.randint(0, max(1, n - blen))
                if abs(i - j) < blen:
                    continue
                h2 = h.copy()
                tmp = h2[j:j+blen].copy()
                h2[j:j+blen] = h2[i:i+blen]
                h2[i:i+blen] = tmp
                h2 = project_box_sum(h2, 0.5 * n)
                c2 = erdos_c5(h2, n)
                if c2 < cur - 1e-12:
                    h, cur, improved = h2, c2, True
            if not improved:
                break
    return h, erdos_c5(h, n)

def resample(h, n_old, n_new):
    x = np.linspace(0.0, 2.0, n_old)
    y = np.linspace(0.0, 2.0, n_new)
    return project_box_sum(np.interp(y, x, h), 0.5 * n_new)

def coarsen_perturb_refine(h, n, deadline, seed):
    rng = np.random.RandomState(seed)
    best_h, best_c = h.copy(), erdos_c5(h, n)
    for nc in [n // 2, n // 3]:
        if nc < 50:
            continue
        hh = resample(best_h, best_n_global, nc)
        for sig in [0.05, 0.15, 0.3]:
            if time.time() > deadline - 15:
                break
            pert = project_box_sum(hh + sig * rng.randn(nc), 0.5 * nc)
            cand, c = lbfgs_step(pert, 5000, 0, 40, deadline)
            if c < best_c - 1e-14:
                cand, c = lbfgs_step(cand, 200000, 5, 30, deadline)
                if c < best_c - 1e-14:
                    up = resample(cand, nc, n)
                    cand2, c2 = lbfgs_step(up, 50000, 5, 30, deadline)
                    if c2 < best_c - 1e-14:
                        best_h, best_c = cand2, c2
    return best_h, best_c

best_n_global = 400

def run(seed=42, budget_s=1100, **kwargs):
    global best_n_global
    t0 = time.time()
    total = max(10.0, float(budget_s) - 5.0)
    deadline = t0 + total

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    if n > 800:
        h = np.interp(np.linspace(0, 2, 800), np.linspace(0, 2, n), h).copy()
        n = 800
    if n < 2:
        n = 400
        h = np.full(n, 0.5)
    h = project_box_sum(h, 0.5 * n)
    best_n_global = n
    best_h, best_n, best_c = h.copy(), n, erdos_c5(h, n)

    # Phase 1: LBFGS with increasing beta and decreasing topk (30%)
    end1 = t0 + 0.30 * total
    schedule = [
        (100, 0, 70), (500, 0, 60), (2000, 0, 55), (10000, 0, 50),
        (50000, 0, 45), (200000, 50, 40), (1000000, 20, 35),
        (5000000, 10, 30), (20000000, 5, 25), (100000000, 3, 20),
    ]
    for beta, topk, mx in schedule:
        if time.time() > end1 - 5.0:
            break
        cand, c = lbfgs_step(best_h, beta, topk, mx, deadline)
        if c < best_c - 1e-15:
            best_h, best_c = cand, c

    # Phase 2: Pair polish (10%)
    end2 = t0 + 0.40 * total
    if time.time() < end2 - 3.0:
        best_h, best_c = pair_polish(best_h, best_n, end2 - 3.0, seed, 45, 5)
        if time.time() < end2 - 5.0:
            cand, c = lbfgs_step(best_h, 500000, 10, 40, deadline)
            if c < best_c - 1e-15:
                best_h, best_c = cand, c

    # Phase 3: Try nearby resolutions (20%)
    end3 = t0 + 0.60 * total
    for nh in [350, 450, 500, 300, 550, 600, 700, 250, 200]:
        if time.time() > end3 - 25 or abs(nh - best_n) < 15:
            continue
        hh = resample(best_h, best_n, nh)
        cand, c = lbfgs_step(hh, 5000, 0, 50, deadline)
        if c < best_c - 1e-14:
            cand, c = lbfgs_step(cand, 200000, 10, 40, deadline)
            if c < best_c - 1e-14:
                best_h, best_n, best_c = cand, nh, c
                best_n_global = nh
                if time.time() < end3 - 15:
                    best_h, best_c = pair_polish(best_h, best_n, end3 - 10, seed + nh, 30, 3)

    # Phase 4: Coarsen-perturb-refine (15%)
    end4 = t0 + 0.75 * total
    if time.time() < end4 - 10:
        best_n_global = best_n
        ch, cc = coarsen_perturb_refine(best_h, best_n, end4 - 5, seed + 50)
        if cc < best_c - 1e-14:
            best_h, best_c = ch, cc

    # Phase 5: Block swap polish (10%)
    end5 = t0 + 0.85 * total
    if time.time() < end5 - 3:
        best_h, best_c = block_swap_polish(best_h, best_n, end5 - 3, seed + 100)
        if time.time() < end5 - 5:
            cand, c = lbfgs_step(best_h, 1000000, 10, 30, deadline)
            if c < best_c - 1e-15:
                best_h, best_c = cand, c

    # Phase 6: Final polish (10%)
    end6 = deadline - 3.0
    if time.time() < end6 - 15:
        best_h, best_c = pair_polish(best_h, best_n, end6 - 12, seed + 200, 55, 5)
    if time.time() < end6 - 8:
        cand, c = lbfgs_step(best_h, 50000000, 3, 25, deadline)
        if c < best_c - 1e-15:
            best_h, best_c = cand, c
    if time.time() < end6 - 4:
        best_h, best_c = pair_polish(best_h, best_n, end6 - 1, seed + 300, 50, 3)

    # Random restarts if time remains
    rng = np.random.RandomState(seed + 999)
    for sig in [0.03, 0.08, 0.15]:
        if time.time() > end6 - 3:
            break
        hh = project_box_sum(best_h + sig * rng.randn(best_n), 0.5 * best_n)
        cand, c = lbfgs_step(hh, 100000, 10, 25, deadline)
        if c < best_c - 1e-15:
            best_h, best_c = cand, c

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = erdos_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
