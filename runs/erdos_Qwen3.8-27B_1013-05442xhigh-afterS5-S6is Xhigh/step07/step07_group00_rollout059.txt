def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

    # Phase 6: Final high-alpha SLSQP + fine CD
    if time.time() < deadline - 30:
        h6, c6 = slsqp_optimize(best_h, len(best_h), [500000, 2000000], 1000, deadline)
        if c6 < best_c:
            best_h, best_c = h6, c6
        h6, c6 = coord_cd_vec(best_h, len(best_h), 80, deadline, grid_size=300)
        if c6 < best_c:
            best_h, best_c = h6, c6

    # Phase 7: Final ultra-fine CD
    if time.time() < deadline - 3:
        h7, c7 = coord_cd_vec(best_h, len(best_h), 50, deadline, grid_size=500)
        if c7 < best_c:
            best_h, best_c = h7, c7

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
