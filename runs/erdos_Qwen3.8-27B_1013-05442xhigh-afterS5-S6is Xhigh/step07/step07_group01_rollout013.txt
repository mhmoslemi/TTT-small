    """Compute C5 for a piecewise constant function.
    boundaries: [t_0, t_1, ..., t_m] with t_0=0, t_m=2
    heights: [h_0, h_1, ..., h_{m-1}]
    """
    m = len(heights)
    n = len(boundaries) - 1
    # For each pair of pieces (i, j), compute the contribution to each lag
    # The overlap at lag l is: sum over all (i,j) pairs of h_i * (1-h_j) * length_of_overlap
    # where length_of_overlap = length of {x : x in [t_i, t_{i+1}] and x+l in [t_j, t_{j+1}]}
    # = length of [t_i, t_{i+1}] ∩ ([t_j-l, t_{j+1}-l])
    
    # For the discretized version, the lags are l = -1, 0, 1, ..., n-1 (in units of dx)
    # But for the continuous version, l can be anything in [-(m-1)*min_width, (m-1)*min_width]
    
    # Actually, for a piecewise constant function, the overlap as a function of l is piecewise linear.
    # The maximum occurs at one of the "critical" values of l, which are differences of boundaries.
    
    # Critical l values: t_j - t_i for all i, j
    critical_l = set()
    for i in range(n):
        for j in range(n):
            critical_l.add(boundaries[j] - boundaries[i])
    critical_l = sorted(critical_l)
    
    # Also include 0
    if 0 not in critical_l:
        critical_l.append(0)
    
    # For each critical l, compute the overlap
    max_overlap = 0
    for l in critical_l:
        if l < -2 or l > 2:
            continue
        overlap = 0.0
        for i in range(n):
            for j in range(n):
                # Contribution: h_i * (1 - h_j) * length([t_i, t_{i+1}] ∩ [t_j - l, t_{j+1} - l])

                overlap += heights[i] * (1 - heights[j]) * inter
        max_overlap = max(max_overlap, overlap)
    
    return max_overlap
