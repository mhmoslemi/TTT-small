            #                         + h[a+idx-n+1] * [n-1 <= idx <= n-1+n-1-a and a+idx-n+1 < n]
            #
            #   Simplify: a-idx+n-1 >= 0 means idx <= a+n-1. And a <= idx.
            #   So first term: idx in [a, min(a+n-1, 2n-2)], value = -(1-h[a-idx+n-1]) = h[a-idx+n-1]-1
            #   Second term: a+idx-n+1 < n means idx < 2n-1-a. And idx >= n-1.
            #   Also a+idx-n+1 >= 0 means idx >= n-1-a.
            #   So second term: idx in [max(n-1, n-1-a), min(n-1+n-1-a, 2n-2)]
            #   Since a >= 0: n-1-a <= n-1, so lower bound is n-1.
            #   Upper bound: n-1+n-1-a = 2n-2-a.
            #   So second term: idx in [n-1, min(2n-2-a, 2n-2)], value = h[a+idx-n+1]
            #
            # Similarly for delta_b[idx] (setting h[b] from 0 to 1):
            #   h[b] was 0, now 1.
            #   As a-component: adds (1-h[b-lag]) to c[idx] for valid b-lag.
            #   idx in [b, min(b+n-1, 2n-2)], value = (1-h[b-idx+n-1])
            #   As b-component: (1-h[b]) was 1, now 0. Removes h[b+lag] from c[idx].
            #   idx in [n-1, min(2n-2-b, 2n-2)], value = -h[b+idx-n+1]
            #
            # BUT WAIT: we need to be careful about overlap. When a-lag == b or b+lag == a etc.
            # The old_h values should be used for the "other" position.
            # Since we're computing delta from old state:
            #   In delta_a, when computing (1-h[a-idx+n-1]), if a-idx+n-1 == b, we use old_h[b]=0, so value = 1.
            #   In delta_b, when computing (1-h[b-idx+n+1]), if b-idx+n+1 == a, we use old_h[a]=1, so value = 0.
            #   In delta_a second term, h[a+idx-n+1]: if a+idx-n+1 == b, old_h[b]=0.
            #   In delta_b second term, h[b+idx-n+1]: if b+idx-n+1 == a, old_h[a]=1.
            #
            # Since we're using old_h throughout, this is consistent. Good.
            #
            # So the total delta for swap (a,b) at index idx:
            # delta[idx] = delta_a[idx] + delta_b[idx]
            #
            # And new_c[idx] = c[idx] + delta[idx]
            #
            # For efficiency, I can precompute the "pattern" vectors for each position.
            # Let PA[p] be the vector of delta_a contributions for position p (length 2n-1)
            # Let PB[p] be the vector of delta_b contributions for position p (length 2n-1)
            # Then for swap (a,b): delta = PA[a] + PB[b]
            # And new_max = max(c + PA[a] + PB[b])
            #
            # But PA and PB depend on h! So I need to recompute them whenever h changes.
            # After each accepted swap, I recompute PA and PB. That's O(n^2) per swap.
            # With O(n) swaps per sweep and O(n) sweeps, total O(n^3) = 64M for n=400. Manageable.
        
        # Let me implement this more carefully with precomputed patterns.
        # Actually, for n=400, even a simpler O(n^2) per swap approach is fine.
        # Let me just recompute the correlation from scratch after each candidate evaluation.
        # np.correlate is fast (O(n^2) or O(n log n) with FFT).
        # For n=400, it's very fast. The bottleneck is the O(n^2) candidate evaluation.
        
        # Let me try a different approach: for each zero position b, find the best one position a to swap with.
        # This reduces to: for fixed b, find a that minimizes max(c + PA[a] + PB[b])
        
        # Actually, let me just try the direct approach with numpy and see if it's fast enough.
        # For n=400: n_zeros * n_ones = 200*200 = 40000 pairs.
        # For each pair, computing new max is O(n) = O(400).
        # Total: 40000 * 400 = 16M operations per sweep. With numpy, this should be < 1 second.
        
        # Even better: vectorize over all pairs at once.
        # Build a 3D array: pair_corrs[pair_idx, corr_idx] = c[corr_idx] + delta[pair_idx, corr_idx]
        # Then take max over axis=1.
        # Memory: 40000 * 799 * 8 bytes = ~256 MB. That's a lot but might work.

    return besth, bestM * (2.0 / float(n))


def binary_optimize(h, n, deadline, max_sweeps=60):
    """Optimize binary h using greedy swaps with precomputed delta patterns."""
    if n % 2 != 0:
        hb = top_n_binary(h, n)
        return hb, compute_c5(hb, n)
    
    h = top_n_binary(h, n).copy()
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    M = float(corr.max())
    besth = h.copy()
    bestM = M
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        zeros = np.nonzero(h == 0)[0]
        ones = np.nonzero(h == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        
        # Precompute delta patterns for all positions
        # PA[p]: effect of changing h[p] from 1 to 0 on correlation
        # PB[p]: effect of changing h[p] from 0 to 1 on correlation
        ncorr = 2 * n - 1
        lags = np.arange(ncorr) - (n - 1)  # lag values
        
        # PA: for position p (h[p]=1 -> 0)
        # As a-component: -(1-h[p-lag]) for valid p-lag
        # As b-component: +h[p+lag] for valid p+lag (since 1-h[p] goes 0->1)
        # 
        # Actually let me just compute the full delta matrix for all positions.
        # For position p as "1->0" (PA):
        #   a-part: contributes -(1-h[q]) where q = p-lag, valid when 0<=q<n
        #          idx = n-1+lag = n-1+p-q
        #   b-part: contributes +h[q'] where q' = p+lag, valid when 0<=q'<n
        #          (because 1-h[p] changes from 0 to 1, adding h[p+lag] to c)
        #          idx = n-1+lag = n-1+q'-p
        
        # For position p as "0->1" (PB):
        #   a-part: contributes +(1-h[q]) where q = p-lag, valid when 0<=q<n
        #          idx = n-1+p-q
        #   b-part: contributes -h[q'] where q' = p+lag, valid when 0<=q'<n
        #          (because 1-h[p] changes from 1 to 0, removing h[p+lag] from c)
        #          idx = n-1+q'-p
        
        # Build PA and PB matrices (n x ncorr)
        PA = np.zeros((n, ncorr))
        PB = np.zeros((n, ncorr))
        
        for p in range(n):
            # a-part: q = p - lag, lag = idx - (n-1), so q = p - idx + n - 1
            # valid: 0 <= q < n => n-1-p <= idx <= n-1+p, also 0 <= idx <= 2n-2
            idx_lo = max(0, n - 1 - p)
            idx_hi = min(2 * n - 2, n - 1 + p)
            q_vals = p - np.arange(idx_lo, idx_hi + 1) + (n - 1)  # q = p - lag = p - (idx - n + 1)
            # Wait: lag = idx - (n-1), q = p - lag = p - idx + n - 1
            q_vals = p - np.arange(idx_lo, idx_hi + 1) + (n - 1)
            # For PA (1->0): subtract (1-h[q])
            PA[p, idx_lo:idx_hi+1] = -(1.0 - h[q_vals])
            # For PB (0->1): add (1-h[q])
            PB[p, idx_lo:idx_hi+1] += (1.0 - h[q_vals])
            
            # b-part: q' = p + lag = p + idx - (n-1), valid: 0 <= q' < n
            # => n-1-p <= idx <= n-1+p-1... wait
            # q' = p + idx - n + 1, valid when 0 <= p + idx - n + 1 < n
            # => n-1-p <= idx <= 2n-2-p
            idx_lo2 = max(0, n - 1 - p)
            idx_hi2 = min(2 * n - 2, 2 * n - 2 - p)
            q2_vals = p + np.arange(idx_lo2, idx_hi2 + 1) - (n - 1)
            # For PA (1->0): add h[q'] (because 1-h[p] goes from 0 to 1)
            PA[p, idx_lo2:idx_hi2+1] += h[q2_vals]
            # For PB (0->1): subtract h[q'] (because 1-h[p] goes from 1 to 0)
            PB[p, idx_lo2:idx_hi2+1] -= h[q2_vals]
        
        # Now for each pair (a in ones, b in zeros):
        # new_corr = corr + PA[a] + PB[b]
        # new_max = max(new_corr)
        
        # Vectorize: for all pairs at once (in chunks)
        zi = np.repeat(zeros, ones.size)
        oj = np.tile(ones, zeros.size)
        
        chunk = 2048
        best_delta = np.inf
        best_pair = None
        
        for start in range(0, zi.size, chunk):
            if time.time() > deadline:
                break
            end = min(start + chunk, zi.size)
            a_chunk = oj[start:end]
            b_chunk = zi[start:end]
            # deltas shape: (chunk_size, ncorr)
            deltas = PA[a_chunk] + PB[b_chunk]  # broadcasting
            new_max = (corr[None, :] + deltas).max(axis=1)
            best_local = new_max.min()
            if best_local < best_delta:
                best_delta = best_local
                loc = int(np.argmin(new_max))
                best_pair = (int(a_chunk[loc]), int(b_chunk[loc]))
        
        if best_pair is None:
            break
        a, b = best_pair
        if best_delta >= M - 1e-12:
            break  # no improvement possible
        
        h[a] = 0.0
        h[b] = 1.0
        corr = corr + PA[a] + PB[b]
        M = float(corr.max())
        if M < bestM:
            bestM = M
            besth = h.copy()
    
    return besth, bestM * dx


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 1100.0
    end = t0 + max(30.0, total_budget - 10.0)
    rng = np.random.RandomState(seed)
    
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
        if n0 < 2 or not np.all(np.isfinite(h0)):
            raise ValueError("bad initial")
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5)
    
    best_h = project_feasible(h0, n0)
    best_n = n0
    best_c = compute_c5(best_h, best_n)
    
    # Phase 1: Continuous polish
    if time.time() < end - 300.0:
        hh, cc = polish_continuous(best_h, best_n, [30.0, 100.0, 400.0, 1500.0, 5000.0], end - 250.0, 60.0)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()
    
    # Phase 2: Binary optimization
    if time.time() < end - 200.0:
        hb, bc = binary_optimize(best_h, best_n, end - 150.0, max_sweeps=80)
        if bc < best_c:
            best_c = bc
            best_h = hb.copy()
        
        # Try different starting patterns for binary
        for pattern_seed in range(3):
            if time.time() < end - 100.0:
                # Create a different starting binary pattern
                period = best_n // (4 + pattern_seed)
                if period < 4:
                    period = 4
                pat = np.zeros(best_n)
                for i in range(best_n):
                    if (i // period) % 2 == 0:
                        pat[i] = 1.0
                pat = project_feasible(pat, best_n)
                hb2, bc2 = binary_optimize(pat, best_n, end - 80.0, max_sweeps=40)
                if bc2 < best_c:
                    best_c = bc2
                    best_h = hb2.copy()
    
    # Phase 3: Final continuous polish on best binary solution
    if time.time() < end - 60.0:
        # Slightly perturb the binary solution for continuous optimization
        perturbed = best_h + 0.01 * rng.randn(best_n)
        hh, cc = polish_continuous(perturbed, best_n, [500.0, 2000.0, 8000.0], end - 40.0, 30.0)
        if cc < best_c:
            best_c = cc
            best_h = hh.copy()
            # Re-project to binary and re-optimize
            if time.time() < end - 30.0:
                hb, bc = binary_optimize(best_h, best_n, end - 20.0, max_sweeps=20)
                if bc < best_c:
                    best_c = bc
                    best_h = hb.copy()
    
    # Phase 4: Try smaller n
    if time.time() < end - 40.0:
        for nnew in (300, 200):
            if time.time() < end - 30.0:
                x0 = np.linspace(0.0, 2.0, best_n)
                x1 = np.linspace(0.0, 2.0, nnew)
