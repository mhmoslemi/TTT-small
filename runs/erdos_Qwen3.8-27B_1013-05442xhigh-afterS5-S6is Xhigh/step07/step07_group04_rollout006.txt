import numpy as np
import time
from scipy.optimize import minimize


def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float)).reshape(-1)
    if v.size != n:
        v = np.resize(v, n)
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(75):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(v - mid, 0.0, 1.0).sum())
        if s > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(80):
        d = float(tgt - v.sum())
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            room = 1.0 - v
            cap = float(room.sum())
        else:
            room = v
            cap = float(room.sum())
        if cap <= 0.0:
            break
        v += (d / cap) * room
        v = np.clip(v, 0.0, 1.0)
    return v


def build_G(h, n):
    off = n - 1
    G = np.zeros((n, 2 * n - 1), dtype=float)
    rev = (1.0 - h)[::-1]
    neg = -h
    for a in range(n):
        G[a, off + a - n + 1: off + a + 1] += rev
        G[a, off - a: off - a + n] += neg
    return G


def pair_vals(c, Lvec, deltas, off, d, dx):
    mat = c[None, :] + (dx * deltas)[:, None] * Lvec[None, :]
    q = dx * deltas * deltas
    mat[:, off] -= 2.0 * q
    mat[:, off - d] += q
    mat[:, off + d] += q
    return mat.max(axis=1)


def pair_search(G, h, c, i, j, curmax, n, off, dx):
    if i == j:
        return None
    d = abs(i - j)
    x0 = float(h[i])
    y0 = float(h[j])
    s = x0 + y0
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-13:
        return None
    Lvec = G[i] - G[j]
    xs = np.linspace(lo, hi, 32)
    x0c = float(np.clip(x0, lo, hi))
    xs = np.unique(np.clip(np.concatenate((xs, [x0c])), lo, hi))
    vals = pair_vals(c, Lvec, xs - x0, off, d, dx)
    for _ in range(2):
        k = int(np.argmin(vals))
        if len(xs) < 3:
            break
        if k == 0:
            a = xs[0]
            b = 0.5 * (xs[0] + xs[1])
        elif k == len(xs) - 1:
            a = 0.5 * (xs[-2] + xs[-1])
            b = xs[-1]
        else:
            a = xs[k - 1]
            b = xs[k + 1]
        if b - a < 1e-11:
            break
        xs = np.linspace(a, b, 64)
        vals = pair_vals(c, Lvec, xs - x0, off, d, dx)
    k = int(np.argmin(vals))
    if vals[k] < curmax - 1e-12:
        return float(np.clip(xs[k], lo, hi))
    return None


def local_pair(h, deadline, seed):
    n = len(h)
    off = n - 1
    dx = 2.0 / n
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    G = build_G(h, n)
    rows = np.arange(n)
    besth = h.copy()
    bestc = float(c.max())
    cur = bestc
    rng = np.random.RandomState(seed)
    noimp = 0
    maxno = 7000 if n > 450 else 12000
    top = np.array([], dtype=int)
    dw = None
    phi = None
    acc = 0

    while time.time() < deadline:
        if noimp > maxno:
            break

        if phi is None or acc % 25 == 0:
            cur = float(c.max())
            sh = np.maximum(40.0 * (c - cur), -50.0)
            w = np.exp(sh)
            w = w / w.sum()
            phi = dx * (G.T @ w)
            dw = np.zeros(n - 1, dtype=float)
            dw += w[off + 1:]
            dw += w[:off][::-1]
            dw += 1e-12
            dw /= dw.sum()
            ordi = np.argsort(c)[::-1]
            top = ordi[ordi != off][:16]

        cur = float(c.max())
        r = rng.random()
        if r < 0.55:
            d = int(rng.choice(n - 1, p=dw)) + 1
        elif r < 0.85 and top.size:
            d = abs(int(top[int(rng.randint(top.size))]) - off)
            if d == 0:
                d = int(rng.randint(1, n))
        else:
            d = int(rng.randint(1, n))
        d = max(1, min(n - 1, d))

        if phi is not None and r < 0.9:
            diff = np.abs(phi[:n - d] - phi[d:]) + 1e-12
            s = float(diff.sum())
            if not np.isfinite(s) or s <= 0.0:
                i = int(rng.randint(n - d))
            else:
                p = diff / s
                p = p / p.sum()
                i = int(rng.choice(n - d, p=p))
        else:
            i = int(rng.randint(n - d))

        j = i + d
        xv = pair_search(G, h, c, i, j, cur, n, off, dx)
        if xv is None:
            noimp += 1
            continue

        oi = float(h[i])
        oj = float(h[j])
        S = oi + oj
        x = float(np.clip(xv, 0.0, 1.0))
        y = S - x
        if y < 0.0:
            y = 0.0
            x = S
        elif y > 1.0:
            y = 1.0
            x = S - 1.0
        x = float(np.clip(x, 0.0, 1.0))
        y = float(np.clip(y, 0.0, 1.0))

        d1 = x - oi
        d2 = y - oj
        if abs(d1) + abs(d2) < 1e-16:
            noimp += 1
            continue

        c += dx * (d1 * G[i] + d2 * G[j])
        qself = dx * (d1 * d1 + d2 * d2)
        c[off] -= qself
        qint = dx * d1 * d2
        c[off + d] -= qint
        c[off - d] -= qint

        G[rows, off + rows - i] -= d1
        G[rows, off + rows - j] -= d2
        G[rows, off - rows + i] -= d1
        G[rows, off - rows + j] -= d2
        h[i] = x
        h[j] = y

        newm = float(c.max())
        if newm < cur - 1e-14:
            cur = newm
            if cur < bestc:
                bestc = cur
                besth = h.copy()
            noimp = 0
            acc += 1
            if acc % 100 == 0:
                h = project_box_sum(h, n)
                c = compute_c(h, n)
                G = build_G(h, n)
                cur = float(c.max())
                if cur < bestc:
                    bestc = cur
                    besth = h.copy()
        else:
            noimp += 1
            h = project_box_sum(h, n)
            c = compute_c(h, n)
            G = build_G(h, n)
            cur = float(c.max())
            if cur > bestc + 5e-5:
                h = besth.copy()
                c = compute_c(h, n)
                G = build_G(h, n)
                cur = bestc

    h = project_box_sum(besth, n)
    return h, float(compute_c(h, n).max())


def lse_obj(x, n, alpha):
    x = np.clip(np.nan_to_num(x), 0.0, 1.0)
    c = compute_c(x, n)
    m = float(c.max())
    if alpha <= 0.0:
        return m
    sh = np.maximum(alpha * (c - m), -50.0)
    e = np.exp(sh)
    s = float(e.sum())
    return m + (np.log(s) / alpha if s > 0.0 else 0.0)


def lse_jac(x, n, alpha):
    x = np.clip(np.nan_to_num(x), 0.0, 1.0)
    c = compute_c(x, n)
    m = float(c.max())
    if alpha <= 0.0:
        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
    else:
        sh = np.maximum(alpha * (c - m), -50.0)
        e = np.exp(sh)
        s = float(e.sum())
        q = e / s if s > 0.0 else np.zeros_like(c)
    return (2.0 / n) * (build_G(x, n).T @ q)


def pen_obj(x, n, alpha, mu):
    x = np.clip(np.nan_to_num(x), 0.0, 1.0)
    val = lse_obj(x, n, alpha)
    d = float(x.sum() - 0.5 * n)
    return val + 0.5 * mu * d * d


def pen_jac(x, n, alpha, mu):
    x = np.clip(np.nan_to_num(x), 0.0, 1.0)
    g = lse_jac(x, n, alpha)
    d = float(x.sum() - 0.5 * n)
    if mu:
        g = g + mu * d
    return g


def lbfgs_phase(h, n, alpha, mu, maxiter):
    h = project_box_sum(h, n)
    if maxiter <= 0:
        return h
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(pen_obj, h, args=(n, alpha, mu), method="L-BFGS-B",
                       jac=pen_jac, bounds=bounds,
                       options={"maxiter": maxiter, "maxfun": 4 * maxiter,
                                "ftol": 1e-15, "gtol": 1e-10})
        if hasattr(res, "x") and res.x is not None:
            return project_box_sum(res.x, n)
    except Exception:
        pass
    return h


def sum_fun(x, n):
    return float(np.clip(x, 0.0, 1.0).sum() - 0.5 * n)


def sum_jac(x, n):
    return np.ones(n)


def slsqp_phase(h, n, alpha, maxiter):
    h = project_box_sum(h, n)
    if maxiter <= 0:
        return h
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{"type": "eq", "fun": sum_fun, "jac": sum_jac, "args": (n,)}]
        res = minimize(lse_obj, h, args=(n, alpha), method="SLSQP",
                       jac=lse_jac, bounds=bounds, constraints=cons,
                       options={"maxiter": maxiter, "ftol": 1e-14})
        if hasattr(res, "x") and res.x is not None:
            return project_box_sum(res.x, n)
    except Exception:
        pass
    return h


def interpolate_h(h, nold, nnew):
    if nold == nnew:
        return h.copy()
    xo = (np.arange(nold) + 0.5) * (2.0 / nold)
    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)


def optimize_cand(h, n, seed, deadline):
    h = project_box_sum(h, n)
    besth = h.copy()
    bestc = float(compute_c(h, n).max())
    work = h.copy()
    workc = bestc

    rem = deadline - time.time()
    if rem > 12.0:
        dl = time.time() + 0.28 * rem
        phases = ((6.0, 2.0, 45), (25.0, 20.0, 35), (80.0, 150.0, 30), (300.0, 700.0, 20))
        for al, mu, it in phases:
            if time.time() > dl - 4.0:
                break
            h2 = lbfgs_phase(work, n, al, mu, it)
            c2 = float(compute_c(h2, n).max())
            if c2 < bestc:
                bestc = c2
                besth = h2.copy()
            if c2 <= workc + 1e-8:
                work, workc = h2, c2
            else:
                work, workc = besth.copy(), bestc

    rem = deadline - time.time()
    if rem > 15.0:
        h2, c2 = local_pair(work, time.time() + 0.45 * rem, seed)
        if c2 < bestc:
            bestc = c2
            besth = h2.copy()
        work = besth.copy()

    rem = deadline - time.time()
    if rem > 25.0:
        it = 22 if n > 450 else 35
        h2 = slsqp_phase(besth, n, 800.0, it)
        c2 = float(compute_c(h2, n).max())
        if c2 < bestc:
            bestc = c2
            besth = h2.copy()

    if time.time() < deadline - 3.0:
        h2, c2 = local_pair(besth, deadline - 2.0, seed + 29)
        if c2 < bestc:
            bestc = c2
            besth = h2.copy()

    besth = project_box_sum(besth, n)
    return besth, float(compute_c(besth, n).max())


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(5.0, float(budget_s) - 8.0)
    rng = np.random.RandomState(seed)
    cands = []

    try:
        pp = np.array(initial_h_values, dtype=float).reshape(-1)
        if pp.size >= 8:
            n0 = int(pp.size)
            p = project_box_sum(pp, n0)
            cands.append((p.copy(), n0))
            ps = 0.5 * (p + p[::-1])
            cands.append((project_box_sum(ps, n0), n0))
            pc = 1.0 - p
            cands.append((project_box_sum(pc, n0), n0))
            nf = n0 if n0 >= 512 else min(576, max(512, n0 + 96))
            if nf != n0:
                pf = project_box_sum(interpolate_h(p, n0, nf), nf)
                cands.append((pf, nf))
    except Exception:
        pass

    if not cands:
        for n, f1, f2, ph in ((256, 2, 4, 0), (288, 3, 5, 1), (320, 1, 3, 2)):
            x = (np.arange(n) + 0.5) * (2.0 / n)
            h = 0.5 + 0.42 * np.sin(f1 * np.pi * x) + 0.16 * np.sin(f2 * np.pi * x + ph)
            h = project_box_sum(h + 0.03 * rng.randn(n), n)
            cands.append((h, n))

    cs = []
    for h, n in cands:
        cs.append(float(compute_c(h, n).max()))
    order = np.argsort(np.array(cs))
    scored = []
    for idx in order[:3]:
        scored.append((cs[idx], cands[idx][0], cands[idx][1]))

    besth = scored[0][1].copy()
    bestc = scored[0][0]

    L = len(scored)
    if L == 1:
        cum = [0.90]
    elif L == 2:
        cum = [0.65, 0.92]
    else:
        cum = [0.55, 0.78, 0.92]

    total = deadline - t0
    for k, (_, h, n) in enumerate(scored):
        dl = t0 + total * cum[k] - 4.0
        if time.time() > dl - 8.0:
            break
        hopt, copt = optimize_cand(h, n, seed + 17 * k, dl)
        if copt < bestc:
            bestc = copt
            besth = hopt

    if time.time() < deadline - 4.0:
        hfin, cfin = local_pair(besth, deadline - 2.0, seed + 123)
        if cfin < bestc:
            bestc = cfin
            besth = hfin

    besth = project_box_sum(besth, len(besth))
    nfin = len(besth)
    cfin = float(compute_c(besth, nfin).max())
    return besth, cfin, nfin
