    g = np.zeros(n)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def fast_grad(h, n, lags, weights):
    g = np.zeros(n)
    for lag, w in zip(lags, weights):
        if lag == 0 or w < 1e-10:
            continue
        if lag > 0:
            g[:n-lag] += w * (1.0 - h[lag:])
            g[lag:] -= w * h[:-lag]
        else:
            j = -lag
            g[j:] += w * (1.0 - h[:-j])
            g[:j] -= w * h[j:]
    return g

    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        s = np.clip(alpha * (g - m), -50, 50)
        return m + (s.max() + np.log(np.exp(s - s.max()).sum())) / alpha
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0.0,1.0)]*n,
                       constraints=[{'type':'eq','fun':cons}],
                       options={'maxiter':maxiter, 'ftol':ftol})
        return project_feasible(res.x, n)
    except:
        return h

                       constraints=[{'type':'eq','fun':cons}],
                       options={'maxiter':maxiter, 'ftol':1e-16})

- The point where the subgradient contains 0 (if such a point exists in (0,1))
- Or at an endpoint (0 or 1)

The subgradient at h[j] is the set {a_k : k is argmax}. The minimum is where 0 ∈ conv{a_k : k is active}.

This is equivalent to: find h[j] such that the "upper envelope" of the lines has a flat point (or the minimum is at a vertex).

For practical purposes, we can:
1. Evaluate the objective at h[j]=0, h[j]=1, and at all breakpoints in (0,1)
2. Pick the best

But there are O(n) breakpoints, and we have n coordinates, so this is O(n²) per sweep. For n=400, that's 160K per sweep, which is fine.

Actually, a simpler approach: since the objective as a function of h[j] is convex piecewise-linear, we can use a **ternary search** or **binary search on the derivative** to find the minimum in O(log(1/ε) * n) time per coordinate.

Even simpler: since it's piecewise-linear with at most O(n) pieces, we can find the minimum in O(n) by checking where the slope changes from negative to positive.

Let me implement this:

```python
def coordinate_optimize(h, j, n):
    """Optimize h[j] given all other coordinates fixed."""
    dx = 2.0 / n
    # Compute a_k and b_k for each lag k
    # c[k] = a_k * h[j] + b_k
    # h is currently at some value, let's compute a and b
    
    hj = h[j]
    c = np.correlate(h, 1-h, mode='full')
    
    # For each lag k (indexed as idx = n-1+k in the correlation array):
    # c[idx] = sum_i h[i](1-h[i+k])
    # The terms involving h[j]:
    #   i=j: h[j](1-h[j+k]) = hj*(1-h[j+k])  -> coefficient of hj: (1-h[j+k])
    #   i=j-k: h[j-k](1-h[j]) = h[j-k] - h[j-k]*hj -> coefficient of hj: -h[j-k]
    # So a_k = (1-h[j+k]) - h[j-k] (with boundaries)
    # b_k = c[k] - a_k * hj
    
    a = np.zeros(2*n-1)
    for idx in range(2*n-1):
        k = idx - (n-1)
        # i=j: need j+k in [0,n)
        coeff1 = (1.0 - h[j+k]) if 0 <= j+k < n else 0.0
        # i=j-k: need j-k in [0,n), and i=j-k means the term is h[j-k](1-h[j])
        coeff2 = -h[j-k] if 0 <= j-k < n else 0.0
        a[idx] = coeff1 + coeff2
        b_idx = c[idx] - a[idx] * hj
    
    # Now minimize max_k (a_k * t + b_k) for t in [0,1]
    # This is a 1D convex optimization
    # The function is f(t) = max_k (a_k*t + b_k)
    # Minimum is where the subgradient contains 0
    
    # Find the minimum by checking critical points
    # f(t) is convex, so we can use binary search on the subgradient
    # f'(t) (subgradient) = {a_k : k in argmax}
    # Minimum is where min(argmax a_k) <= 0 <= max(argmax a_k)
    
    # Simpler: just evaluate at a few points
    # The breakpoints of f(t) occur where a_k*t + b_k = a_k'*t + b_k'
    # i.e., t = (b_k'-b_k)/(a_k-a_k')
    
    # For efficiency, let's just use a fine grid + refinement
    t_candidates = np.array([0.0, 1.0])
    # Add some heuristic points
    # The minimum is where the "knee" of the max function is
    
    # Actually, let's just use scipy minimize_scalar on the 1D function
    from scipy.optimize import minimize_scalar
    def f1d(t):
        return np.max(a * t + b_idx) * dx
    
    res = minimize_scalar(f1d, bounds=(0, 1), method='bounded', options={'xatol': 1e-10})
    return res.x
```

Hmm, this is O(n) per evaluation of f1d, and minimize_scalar does ~50 evaluations, so O(50n) per coordinate, O(50n²) per sweep. For n=400, that's 8M per sweep. Maybe 10-20 sweeps in the time budget. That could work!

But wait, I don't need scipy for the 1D problem. Since f(t) = max_k (a_k*t + b_k) is convex piecewise-linear, I can find its minimum directly:

The minimum of f(t) on [0,1] occurs at:
- t=0
- t=1
- Or at a breakpoint where two lines cross and the slope changes from negative to positive

The slope of f(t) at any point is max_k a_k for the active k at that point... no, the slope is the a_k of the k that achieves the max.

Actually, f(t) = max_k (a_k*t + b_k). The derivative (where it exists) is a_k for the k that's currently on top. As t increases, the "top" line changes at breakpoints.

To find the minimum:
1. Sort lines by intercept b_k (or by slope a_k)
2. Find the upper envelope
3. The minimum of the upper envelope on [0,1] is at a vertex or endpoint

This is the "upper envelope" problem, solvable in O(n log n) using a convex hull trick.

But for simplicity, let me just use a different approach: **iterative bisection**.

Since f is convex, I can binary search for the point where the subgradient changes sign:
- At t, compute the active set (argmax)
- If all active a_k > 0, decrease t
- If all active a_k < 0, increase t
- If some > 0 and some < 0, we're at the minimum (or very close)

This converges in O(log(1/ε)) iterations, each O(n). Total O(n log(1/ε)) per coordinate.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def fast_grad(h, n, lags, weights):
    g = np.zeros(n)
    for lag, w in zip(lags, weights):
        if lag == 0 or w < 1e-10:
            continue
        if lag > 0:
            g[:n - lag] += w * (1.0 - h[lag:])
            g[lag:] -= w * h[:-lag]
        else:
            j = -lag
            g[j:] += w * (1.0 - h[:-j])
            g[:j] -= w * h[j:]
    return g

def momentum_opt(h, n, deadline, n_top=5, temp_scale=0.05, mu_init=0.85):
    h_curr = h.copy()
    v = np.zeros(n)
    mu = mu_init
    step = 0.8
    best_c = compute_c5(h_curr, n)
    best_h = h_curr.copy()
    dx = 2.0 / n
    stall = 0
    np.random.seed(int(time.time() * 1000) % 100000)
    while time.time() < deadline:
        c = np.correlate(h_curr, 1 - h_curr, mode='full') * dx
        cmax = c.max()
        if cmax < 1e-10:
            break
        idx = np.argpartition(c, -n_top)[-n_top:]
        w = np.exp((c[idx] - cmax) / (temp_scale * cmax + 1e-12))
        w /= w.sum()
        g = fast_grad(h_curr, n, idx - (n - 1), w)
        v = mu * v - step * g
        h_new = project_feasible(h_curr + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        if stall > 8000:
            h_curr = best_h + 0.02 * np.random.randn(n)
            h_curr = project_feasible(h_curr, n)
            v = np.zeros(n)
            step = 0.4
            stall = 0
        elif c_new > best_c * 1.01:
            step *= 0.92
            v *= 0.3
        else:
            step = min(step * 1.03, 3.0)
        h_curr = h_new
    return best_h, best_c

def coord_descent(h, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        for j in range(n):
            if time.time() > deadline:
                break
            hj = h[j]
            c = np.correlate(h, 1 - h, mode='full')
            a = np.zeros(2 * n - 1)
            for idx in range(2 * n - 1):
                k = idx - (n - 1)
                c1 = (1.0 - h[j + k]) if 0 <= j + k < n else 0.0
                c2 = -h[j - k] if 0 <= j - k < n else 0.0
                a[idx] = c1 + c2
            b = c - a * hj
            def f1d(t):
                return np.max(a * t + b)
            lo, hi = 0.0, 1.0
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if f1d(m1) < f1d(m2):
                    hi = m2
                else:
                    lo = m1
            t_new = (lo + hi) / 2.0
            if abs(t_new - hj) < 1e-6:
                continue
            h[j] = t_new
        h = project_feasible(h, n)
        c_now = compute_c5(h, n)
        if c_now < best_c:
            best_c = c_now
            best_h = h.copy()
    return best_h, best_c

def lse_slsqp(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        s = np.clip(alpha * (g - m), -50, 50)
        return m + (s.max() + np.log(np.exp(s - s.max()).sum())) / alpha
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def barrier_slsqp(h, n, t, maxiter):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        return 0.5 * np.sum(np.maximum(c - t, 0) ** 2)
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def refine_full(h, n, deadline):
    c = compute_c5(h, n)
    for alpha in [50000, 500000, 5000000, 20000000, 80000000]:
        if time.time() > deadline - 5:
            break
        mi = 2500 if alpha < 1e6 else 3500
        h = lse_slsqp(h, n, alpha, mi, 1e-16)
        c = compute_c5(h, n)
    for i in range(10):
        if time.time() > deadline - 3:
            break
        t = c * (1.0 - 0.004 * (i + 1))
        h = barrier_slsqp(h, n, t, 2000)
        c = compute_c5(h, n)
    return h, c
