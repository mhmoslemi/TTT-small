def lse_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    m = c.max()
    # Compute softmax weights
    shifted = alpha * (c - m)
    p = np.exp(shifted - shifted.max())
    p /= p.sum()
    # Gradient: for each i, sum over k of p[k] * d(c_k)/d(h_i)
    # d(c_k)/d(h_i) = (1-h[i+k]) - h[i-k] (with bounds)
    grad = np.zeros(n)
    for k_offset in range(-(n-1), n):
        if abs(p[k_offset + n - 1]) < 1e-15:
            continue
        w = p[k_offset + n - 1] * dx
        # d(c_k)/d(h_i) for shift k_offset
        if k_offset == 0:
            grad += w * (1 - 2*h)
        elif k_offset > 0:
            # c_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            # d/dh_i = (1-h[i+k]) for i in [0, n-1-k)
            #          -h[i-k] for i in [k, n)

def thue_morse(n):
    """Generate Thue-Morse binary sequence of length n."""
    tm = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm

def binary_search(h_bin, n, deadline, max_sweeps=200, rng=None):

            break

grad[i] = dx * sum_k p[k] * [(1-h[i-k])*I(0≤i-k<n) - h[i+k]*I(0≤i+k<n)]

Let me define:
- q[k] = p[k] for k in [-(n-1), n-1], 0 otherwise
- a[i] = 1-h[i] for i in [0,n-1], 0 otherwise
- b[i] = h[i] for i in [0,n-1], 0 otherwise

Then:
grad[i] = dx * [sum_k q[k]*a[i-k] - sum_k q[k]*b[i+k]]

The first sum is (q * a)[i] (convolution) and the second is (q * b_rev)[i - 2(n-1)] or something. Let me be more careful.

sum_k q[k]*a[i-k] = (q * a)[i] where * is convolution: (q*a)[i] = sum_k q[k]*a[i-k]

sum_k q[k]*b[i+k]: let j = -k, then = sum_j q[-j]*b[i-j] = (q_rev * b)[i] where q_rev[j] = q[-j]

So grad = dx * ((q * a) - (q_rev * b))[i] for i in [0, n-1]

I can compute these convolutions using FFT! For n=400, the arrays are length 799, so FFT of size 2048 is fast.

Actually, for n=400, the direct O(n²) computation per gradient is 800*400 = 320,000 multiply-adds. In numpy, this is very fast (sub-millisecond). The issue is doing it 100,000 times.
