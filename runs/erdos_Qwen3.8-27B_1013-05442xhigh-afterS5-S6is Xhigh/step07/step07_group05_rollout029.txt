            alpha *= 0.95
        else:
            alpha = max(alpha * 0.8, alpha_min)
        h = x_new
        t = t_new
    return best, best_c

def pair_line(h, n, i, j, cur, deadline):
    hi0 = h[i]
    hj0 = h[j]
    lo = max(-hi0, hj0 - 1.0)
    hi = min(1.0 - hi0, hj0)
    if hi - lo < 1e-13:
        return h, cur
    best_d = 0.0
    best_c = cur
    def peval(d):
        a = h[i] + d
        b = h[j] - d
        if a < -1e-12 or a > 1.0 + 1e-12 or b < -1e-12 or b > 1.0 + 1e-12:
            return 1e100
        hh = h.copy()
        hh[i] = a
        hh[j] = b
        return erdos_c5(hh, n)
    c0 = peval(lo)
    if c0 < best_c:
        best_c = c0
        best_d = lo
    c1 = peval(hi)
    if c1 < best_c:
        best_c = c1
        best_d = hi
    try:
        if time.time() < deadline:
            rs = so.minimize_scalar(peval, bounds=(lo, hi), method='bounded',
                                    options={'maxiter': 20, 'xatol': 1e-10})
            if rs.x is not None:
                c2 = peval(float(rs.x))
                if c2 < best_c:
                    best_c = c2
                    best_d = float(rs.x)
    except Exception:
        pass
    if best_c < cur - 1e-14:
        hh = h.copy()
        hh[i] += best_d
        hh[j] -= best_d
        return hh, best_c
    return h, cur

def pair_polish(h, n, deadline, seed, n_rounds=5, n_pairs=30):
    rng = np.random.RandomState(int(seed))
    h = np.asarray(h, dtype=float)
    cur = erdos_c5(h, n)
    frac = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
    if frac.size:
        score = np.minimum(h[frac], 1.0 - h[frac])
        order = frac[np.argsort(score)[::-1]]
    else:
        order = np.arange(n)
    top = order[:min(40, len(order))]
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        pairs = []
        tl = len(top)
        for a in range(tl):
            for b in range(a + 1, tl):
                pairs.append((int(top[a]), int(top[b])))
        rng.shuffle(pairs)
        for i, j in pairs[:n_pairs]:
            if time.time() > deadline:
                break
            hh, cc = pair_line(h, n, i, j, cur, deadline)
            if cc < cur - 1e-14:
                h = hh
                cur = cc
                improved = True
        if not improved:
            break
    return h, cur

def block_polish(h, n, deadline, seed, block_size=3):
    rng = np.random.RandomState(int(seed))
    h = np.asarray(h, dtype=float).copy()
    cur = erdos_c5(h, n)
    target = 0.5 * n
    for _ in range(200):
        if time.time() > deadline:
            break
        start = rng.randint(0, n - block_size)
        vals = h[start:start + block_size].copy()
        vsum = float(vals.sum())
        def block_obj(params):
            delta = np.zeros(block_size)
            delta[:-1] = params
            delta[-1] = -float(sum(params))
            nv = vals + delta
            if np.any(nv < -1e-12) or np.any(nv > 1.0 + 1e-12):
                return 1e100
            hh = h.copy()
            hh[start:start + block_size] = nv
            return erdos_c5(hh, n)
        x0 = np.zeros(block_size - 1)
        try:
            res = so.minimize(block_obj, x0, method='Nelder-Mead',
                              options={'maxiter': 200, 'xatol': 1e-10, 'fatol': 1e-12})
            if res.fun < cur - 1e-14:
                delta = np.zeros(block_size)
                delta[:-1] = res.x
                delta[-1] = -float(sum(res.x))
                nv = vals + delta
                if np.all(nv >= -1e-12) and np.all(nv <= 1.0 + 1e-12):
                    hh = h.copy()
                    hh[start:start + block_size] = np.clip(nv, 0.0, 1.0)
                    hh = project_box_sum(hh, target)
                    c2 = erdos_c5(hh, n)
                    if c2 < cur - 1e-14:
                        h = hh
                        cur = c2
        except Exception:
            pass
    return h, cur

def cosine_init(n, A, phi):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + A * np.cos(np.pi * x + phi)
    return project_box_sum(h, 0.5 * n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(10.0, float(budget_s) - 5.0)
    deadline = t0 + total
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h))
    except Exception:
        n = 400
        h = np.full(n, 0.5, dtype=float)
    if n < 2:
        n = 400
        h = np.full(n, 0.5, dtype=float)
    if n > 800:
        nold = n
        n = 800
        h = np.interp(np.linspace(0.0, 2.0, n), np.linspace(0.0, 2.0, nold), h)
    h = project_box_sum(h, 0.5 * n)
    best_h = h.copy()
    best_n = n
    best_c = erdos_c5(best_h, best_n)

    # Phase 1: Try cosine starts and pick best
    end_p1 = t0 + 0.15 * total
    for A in (0.3, 0.4, 0.5, 0.45):
        for phi in (0.0, 0.5, 1.0):
            if time.time() > end_p1:
                break
            hc = cosine_init(n, A, phi)
            hc, cc = refine_lbfgs(hc, 800.0, 60, deadline)
            if cc < best_c - 1e-14:
                best_h = hc.copy()
                best_c = cc

    # Phase 2: LBFGS refinement with increasing beta
    end_p2 = t0 + 0.55 * total
    betas = [50.0, 300.0, 2000.0, 15000.0, 100000.0, 800000.0]
    iters = [100, 90, 80, 70, 55, 40]
    for beta, mx in zip(betas, iters):
        if time.time() > end_p2 - 5.0:
            break
        cand, c = refine_lbfgs(best_h, beta, 1e6, mx, deadline)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c

    # Phase 3: Nesterov subgradient with momentum
    end_p3 = t0 + 0.70 * total
    if time.time() < end_p3 - 20.0:
        cand, c = nesterov_subgrad(best_h, end_p3 - 15.0)
        if c < best_c - 1e-14:
            best_h = cand.copy()
            best_c = c

    # Phase 4: Pair polishing (aggressive)
    end_p4 = t0 + 0.85 * total
    if time.time() < end_p4 - 20.0:
        best_h, best_c = pair_polish(best_h, best_n, end_p4 - 10.0, seed,
                                     n_rounds=6, n_pairs=40)

    # Phase 5: Block polishing
    if time.time() < end_p4 - 15.0:
        best_h, best_c = block_polish(best_h, best_n, end_p4, seed + 7, block_size=3)

    # Phase 6: High-beta LBFGS final push
    end_final = deadline - 5.0
    if time.time() < end_final - 30.0:
        cand, c = refine_lbfgs(best_h, 2000000.0, 1e6, 30, deadline)
        if c < best_c - 1e-15:
            best_h = cand.copy()
            best_c = c

    # Phase 7: Final pair polish
    if time.time() < end_final - 10.0:
        best_h, best_c = pair_polish(best_h, best_n, end_final - 5.0, seed + 100,
                                     n_rounds=4, n_pairs=50)

    # Try random restarts
    rng = np.random.RandomState(int(seed) + 200)
    for sig, sd in ((0.01, 301), (0.03, 402), (0.06, 503)):
        if time.time() > end_final - 8.0:
            break
        hh = project_box_sum(best_h + sig * rng.randn(best_n), 0.5 * best_n)
        hh, c = refine_lbfgs(hh, 100000.0, 1e6, 25, deadline)
        if c < best_c - 1e-14:
            hh, c = pair_polish(hh, best_n, end_final - 5.0, sd, n_rounds=3, n_pairs=30)
            if c < best_c - 1e-14:
                best_h = hh.copy()
                best_c = c

    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = erdos_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
