        # i+k < n means i < n-k

            mk = -k

import numpy as np
import time

from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c)) * dx

def compute_overlaps(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, k):
    """Gradient of over_k w.r.t. h"""
    grad = np.zeros(n)
    if k >= 0:
        if n - k > 0:
            grad[:n-k] += (1.0 - h[k:])
        if k > 0:
            grad[k:] -= h[:n-k]
    else:
        mk = -k
        if n - mk > 0:
            grad[mk:] += (1.0 - h[:mk])
        if mk > 0:
            grad[:n-mk] -= h[mk:]
    return grad

def weighted_grad(h, n, weights, lags):
    """Compute weighted sum of lag gradients"""
    grad = np.zeros(n)
    for w, m in zip(weights, lags):
        if w < 1e-8:
            continue
        k = m - (n - 1)
        grad += w * lag_gradient(h, n, k)
    return grad

def smooth_optimize(h, n, alpha, maxiter=300, penalty=5000.0):

    over0 = compute_overlaps(h)

        over = compute_overlaps(x)
        a = alpha * over

        exp_a = np.exp(a - amax)
        logsumexp = amax + np.log(exp_a.sum())

        lags = idx
        g = weighted_grad(x, n, wt, lags)
        g /= alpha
        g += 2.0 * penalty * (x.sum() - n / 2.0)
        return val, g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: obj_and_grad(x)[0], h,
                       jac=lambda x: obj_and_grad(x)[1],
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def lp_step(h, n, eps, solver='CLARABEL'):
    C = compute_overlaps(h).astype(float)
    mc = float(C.max())
    # Build gradient matrix G: G[i, m] = d over_m / d h[i]
    n_over = 2 * n - 1
    G = np.zeros((n, n_over))
    for m in range(n_over):
        k = m - (n - 1)
        g = lag_gradient(h, n, k)
        G[:, m] = g
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in [solver, 'GLOP', 'CLARABEL', 'GLPK']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.8, 0.5, 0.3, 0.15]:
        h_try = h + lam * d
        h_try = project_feasible(h_try, n)
        mc_try = float(compute_overlaps(h_try).max())
        if mc_try < best_mc - 1e-14:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

def binary_init(n, seed):
    """Create a binary-like initialization with good structure"""
    rng = np.random.RandomState(seed)
    # Start with alternating blocks and randomize
    h = np.zeros(n)
    block = max(1, n // 200)

    n_ones = int(h.sum())
    if n_ones > n // 2:

        h[to_one] = 1
