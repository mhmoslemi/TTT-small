def lse_gradient(h, n, alpha):
    """Compute analytical gradient of LSE objective."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()

    sm = shifted.max()
    exp_vals = np.exp(shifted - sm)

    grad = np.zeros(n)
    for k in range(len(c)):
        if probs[k] < 1e-15:
            continue
        l = k - (n - 1)
        if l >= 0:
            # (1-h[j+l]) for j = 0..n-1-l
            grad[:n-l] += probs[k] * (1.0 - h[l:])  # wait, this isn't right
            # Let me redo this
            # dc_k/dh[j] = (1-h[j+l]) for 0<=j<n (and 0<=j+l<n means j<n-l)
            #             - h[j-l] for 0<=j<n (and 0<=j-l<n means j>=l)
            # So for l >= 0:
            #   grad[j] += (1-h[j+l]) for j in [0, n-1-l]
            #   grad[j] -= h[j-l] for j in [l, n-1]
            valid1 = np.arange(0, n - l)
            grad[valid1] += probs[k] * (1.0 - h[valid1 + l])
            valid2 = np.arange(l, n)
            grad[valid2] -= probs[k] * h[valid2 - l]
        else:
            # l < 0, let ll = -l > 0
            ll = -l
            # (1-h[j-l]) = (1-h[j+ll]) for 0<=j+ll<n, i.e., j < n-ll
            # -h[j-l] = -h[j+ll] ... wait no
            # dc_k/dh[j] = (1-h[j+l])*I(0<=j+l<n) - h[j-l]*I(0<=j-l<n)
            # With l < 0: j+l = j-|l|, valid when j >= |l|
            #             j-l = j+|l|, valid when j < n-|l|
            valid1 = np.arange(ll, n)
            grad[valid1] += probs[k] * (1.0 - h[valid1 + l])
            valid2 = np.arange(0, n - ll)
            grad[valid2] -= probs[k] * h[valid2 - l]
    
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greed_binary(h_init, n, deadline, max_sweeps=100):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            for j in ones_list:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                if msk.any():
                    delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                if msk.any():
                    delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                if msk.any():
                    delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                if msk.any():
                    delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < len(delta):
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < len(delta):
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones_list.remove(j)
                    zeros_list.remove(i)
                    ones_list.append(i)
                    zeros_list.append(j)
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def lse_slsqp(h, n, alpha, maxiter, ftol=1e-16):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        ev = np.exp(shifted - sm)
        return m + (sm + np.log(ev.sum())) / alpha
    def cons(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def lse_trust(h, n, alpha, maxiter=5000):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        ev = np.exp(shifted - sm)
        return m + (sm + np.log(ev.sum())) / alpha
    from scipy.optimize import LinearConstraint
    cons = LinearConstraint(np.ones(n), n / 2.0, n / 2.0)
    try:
        res = minimize(obj, h, method='trust-constr', bounds=bounds,
                       constraints=[cons],
                       options={'maxiter': maxiter, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

def projected_grad_lse(h, n, alpha, steps, lr_init, dl):
    """Analytical gradient projected descent on LSE."""
    dx = 2.0 / n
    x = h.copy()
    best_c = compute_c5(x, n)
    v = np.zeros(n)
    lr = lr_init
    gamma = 0.9
    for s in range(steps):
        if time.time() > dl - 2:
            break
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        ev = np.exp(shifted - sm)
        probs = ev / ev.sum()
        # Only compute gradient for significant prob lags
        significant = np.where(probs > 1e-10)[0]
        grad = np.zeros(n)
        for k in significant:
            l = k - (n - 1)
            pk = probs[k]
            if l >= 0:
                idx1 = np.arange(0, n - l)
                grad[idx1] += pk * (1.0 - x[idx1 + l])
                idx2 = np.arange(l, n)
                grad[idx2] -= pk * x[idx2 - l]
            else:
                ll = -l
                idx1 = np.arange(ll, n)
                grad[idx1] += pk * (1.0 - x[idx1 + l])
                idx2 = np.arange(0, n - ll)
                grad[idx2] -= pk * x[idx2 - l]
        grad *= dx
        v = gamma * v - lr * grad
        x_new = project_feasible(x + v, n)
        new_c = compute_c5(x_new, n)
        if new_c < best_c + 1e-16:
            x = x_new
            if new_c < best_c:
                best_c = new_c
        else:
            lr *= 0.6
            v *= 0.3
    return x, best_c

def bar_objective(x, n, t):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    excess = np.maximum(c - t, 0)
    return 0.5 * np.sum(excess ** 2)

def barrier_opt(h, n, t, maxiter=3000):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return bar_objective(x, n, t)

        return x.sum() - n / 2.0
    try:

                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h
