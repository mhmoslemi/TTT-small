    v = 1.0 - h
    R = p[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(p, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)

        for kick in range(5):
            if time.time() > end - 40.0:
                break
            hb = best_h.copy()
            mask = (hb > 0.3) & (hb < 0.7) if best_h.min() < 0.99 else np.ones(best_n, dtype=bool)
            flip_count = max(2, best_n // 50)
            ones_idx = np.nonzero(hb >= 0.5)[0]
            zeros_idx = np.nonzero(hb < 0.5)[0]
            for _ in range(flip_count):
                if ones_idx.size > 0 and zeros_idx.size > 0:
                    oi = ones_idx[rng.randint(len(ones_idx))]
                    zi = zeros_idx[rng.randint(len(zeros_idx))]
                    hb[oi], hb[zi] = 0.0, 1.0
                    ones_idx = np.delete(ones_idx, np.nonzero(ones_idx == oi)[0])
                    zeros_idx = np.delete(zeros_idx, np.nonzero(zeros_idx == zi)[0])
            hh, cc = greedy_binary(hb, best_n, min(end - 30.0, time.time() + 30.0), max_sweeps=20, chunk=2048)
            if cc < best_c:
                best_c, best_h = cc, hh.copy()
    if time.time() < end - 60.0:
        hh, cc = polish(best_h, best_n, end - 25.0, (50000.0, 200000.0))
        if cc < best_c:
            best_c, best_h = cc, hh
    for nnew in (200, 300, 500):
        if time.time() > end - 40.0:
            break
        x0 = np.linspace(0.0, 2.0, best_n)
        x1 = np.linspace(0.0, 2.0, nnew)
        hr = np.interp(x1, x0, best_h)
        hr = project_feasible(hr, nnew)
        hh, cc = greedy_binary(hr, nnew, min(end - 20.0, time.time() + 25.0), max_sweeps=15, chunk=1024)
        cr = compute_c5(hh, nnew)
        if cr < best_c:
            best_c, best_h, best_n = cr, hh, nnew
            hh2, cc2 = polish(hh, nnew, end - 15.0, (10000.0, 30000.0))
            if cc2 < best_c:
                best_c, best_h = cc2, hh2
    best_h = project_feasible(best_h, best_n)
    if best_n % 2 == 0:
        tb = to_binary(best_h, best_n)
        ct = compute_c5(tb, best_n)
        if ct < best_c:
            best_c, best_h = ct, tb
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
