import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=1000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: SLSQP with alpha schedule
    if time.time() < deadline - 200:
        h1, c1 = slsqp_run(best_h, n, [50, 500, 3000, 15000], 600)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Coordinate descent with golden section
    if time.time() < deadline - 100:
        h2, c2 = coord_cd(best_h, n, 300, deadline, grid_size=80)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Subgradient polish
    if time.time() < deadline - 80:
        h3, c3 = subgradient_polish(best_h, n, 200, deadline)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: Try other n values
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [350, 450, 600, 800]:
        if time.time() > deadline - 150:
            break
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = slsqp_run(h_int, n_try, [500, 5000, 25000], 400)
        h_opt, c_opt = coord_cd(h_opt, n_try, 100, deadline, grid_size=60)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
            x_orig = x_new
    # Phase 5: Multi-restart perturbations
    n_cur = len(best_h)
    if time.time() < deadline - 80:
        for r in range(6):
            if time.time() > deadline - 60:
                break
            s = 0.005 + 0.04 * r
            hp = project_feasible(best_h + rng.normal(0, s, n_cur), n_cur)
            h_o, c_o = slsqp_run(hp, n_cur, [2000, 20000], 300)
            h_o, c_o = coord_cd(h_o, n_cur, 60, deadline, grid_size=50)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Phase 6: Final high-precision refinement
    n_cur = len(best_h)
    if time.time() < deadline - 30:
        hf, cf = slsqp_run(best_h, n_cur, [50000, 200000, 500000], 800)
        if cf < best_c:
            best_h, best_c = hf, cf
    if time.time() < deadline - 10:
        hf, cf = coord_cd(best_h, n_cur, 500, deadline, grid_size=120)
        if cf < best_c:
            best_h, best_c = hf, cf
    # Final check: also try 1-h symmetry
    h_flip = project_feasible(1.0 - best_h, n_cur)
    c_flip = compute_c5(h_flip, n_cur)
    if c_flip < best_c:
        best_h, best_c = h_flip, c_flip
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
