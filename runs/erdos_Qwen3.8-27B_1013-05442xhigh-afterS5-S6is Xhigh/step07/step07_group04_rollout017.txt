Actually, this is a min-max of linear functions, which can be solved but is non-trivial. The golden section search is simpler and works fine.

But the key optimization is: precompute `term_diff` once, then each evaluation is just:

def pair_search_fast(h, i, j, n):
    """Fast pair search. Returns (new_h_i, new_h_j, improved) or None."""
    S = h[i] + h[j]
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-13:
        return None
    
    dx = 2.0 / n
    ks = np.arange(-(n-1), n)
    
    # Precompute term_diff
    ik = i + ks
    ik_valid = (ik >= 0) & (ik < n)
    ik2 = i - ks
    ik2_valid = (ik2 >= 0) & (ik2 < n)
    jk = j + ks
    jk_valid = (jk >= 0) & (jk < n)
    jk2 = j - ks
    jk2_valid = (jk2 >= 0) & (jk2 < n)
    
    term_i = np.zeros(2*n-1)
    term_j = np.zeros(2*n-1)
    term_i[ik_valid] = 1.0 - h[ik[ik_valid]]
    term_i[ik2_valid] -= h[ik2[ik2_valid]]
    term_j[jk_valid] = 1.0 - h[jk[jk_valid]]
    term_j[jk2_valid] -= h[jk2[jk2_valid]]
    
    term_diff = (term_i - term_j) * dx
    c_base = compute_c(h, n)  # O(n^2) but only once
    best_c = float(c_base.max())
    old_i = h[i]
    
    # Now f(t) = max(c_base + (t - old_i) * term_diff)
    # This is a convex piecewise-linear function of t (max of linear functions)
    # We can find the minimum by finding where the subgradient contains 0
    
    # Golden section search
    inv_gold = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c_ = b - inv_gold * (b - a)
    d_ = a + inv_gold * (b - a)
    f_c = float(np.max(c_base + (c_ - old_i) * term_diff))
    f_d = float(np.max(c_base + (d_ - old_i) * term_diff))

            d_ = c_
            f_d = f_c

            c_ = d_

        if b - a < 1e-12:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(40):
        d = tgt - v.sum()
        if abs(d) < 1e-14:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c > 0:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def soft_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    if alpha <= 0:
        return m
    e = np.exp(alpha * (c - m))
    return m + float(np.log(e.sum())) / alpha

def soft_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    if alpha <= 0:
        q = np.zeros(2 * n - 1)
        q[int(np.argmax(c))] = 1.0
    else:
        e = np.exp(alpha * (c - m))
        q = e / e.sum()
    dx = 2.0 / n
    off = n - 1
    g = np.zeros(n)
    nz = np.flatnonzero(q > 1e-14)
    for idx in nz:
        k = idx - off
        w = q[idx]
        if k > 0:
            g[k:] += w * (1.0 - x[:n - k])
            g[:n - k] -= w * x[k:]
        elif k < 0:
            j = -k
            g[j:] += w * (1.0 - x[:n - j])
            g[:n - j] -= w * x[j:]
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g

def projected_grad_descent(x, n, alpha, niter, step0=0.01):
    x = project_box_sum(x, n).copy()
    f_cur = soft_obj(x, n, alpha)
    step = step0
    for _ in range(niter):
        g = soft_grad(x, n, alpha)
        gn = float(np.linalg.norm(g))
        if gn < 1e-12:
            break
        for _ in range(25):
            x_new = project_box_sum(x - step * g, n)
            f_new = soft_obj(x_new, n, alpha)
            if f_new < f_cur - 1e-18:
                x = x_new
                f_cur = f_new
                break
            step *= 0.5
        else:
            break
        step = min(step * 1.5, step0)
    return x

def interpolate_h(h, nold, nnew):
    if nold == nnew:
        return h.copy()
    xo = (np.arange(nold) + 0.5) * (2.0 / nold)
    xn = (np.arange(nnew) + 0.5) * (2.0 / nnew)
    return np.interp(xn, xo, h)

def build_term_diff(h, i, j, n):
    """Precompute the term_diff vector for pair (i,j). O(n)."""
    dx = 2.0 / n
    ks = np.arange(-(n - 1), n)
    term_i = np.zeros(2 * n - 1)
    term_j = np.zeros(2 * n - 1)
    ik = i + ks
    ikv = (ik >= 0) & (ik < n)
    ik2 = i - ks
    ik2v = (ik2 >= 0) & (ik2 < n)
    term_i[ikv] = 1.0 - h[ik[ikv]]
    term_i[ik2v] -= h[ik2[ik2v]]
    jk = j + ks
    jkv = (jk >= 0) & (jk < n)
    jk2 = j - ks
    jk2v = (jk2 >= 0) & (jk2 < n)
    term_j[jkv] = 1.0 - h[jk[jkv]]
    term_j[jk2v] -= h[jk2[jk2v]]
    return (term_i - term_j) * dx

def pair_search_fast(h, c, i, j, n):
    """Fast pair search using precomputed term_diff. Returns (ti, tj) or None."""
    S = h[i] + h[j]
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-13:
        return None
    td = build_term_diff(h, i, j, n)
    old_i = h[i]
    best_c = float(c.max())
    # Golden section
    inv_g = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    cp = b - inv_g * (b - a)
    dp = a + inv_g * (b - a)
    fc = float(np.max(c + (cp - old_i) * td))
    fd = float(np.max(c + (dp - old_i) * td))
    for _ in range(45):
        if fc < fd:
            b = dp; dp = cp; fd = fc
            cp = b - inv_g * (b - a)
            fc = float(np.max(c + (cp - old_i) * td))
        else:
            a = cp; cp = dp; fc = fd
            dp = a + inv_g * (b - a)
            fd = float(np.max(c + (dp - old_i) * td))
        if b - a < 1e-12:
            break
    tb = 0.5 * (a + b)
    fb = float(np.max(c + (tb - old_i) * td))
    if fb < best_c - 1e-13:
        return float(np.clip(tb, lo, hi)), float(np.clip(S - tb, lo, hi))
    return None

def local_search(h, n, deadline, seed=1):
    """Extensive pair local search."""
    h = project_box_sum(h, n).copy()
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()
    rng = np.random.RandomState(seed)
    off = n - 1
    noimp = 0
    maxno = 5000

    while time.time() < deadline - 2:
        if noimp > maxno:
            break
        # Choose shift: biased towards worst shifts
        if rng.random() < 0.7:
            pos = c[off + 1:]
            neg = c[off - 1::-1]
            w = np.maximum(pos, neg) + 1e-12
            w /= w.sum()
            d = int(rng.choice(n - 1, p=w)) + 1
        else:
            d = int(rng.randint(1, n))
        i = int(rng.randint(n - d))
        j = i + d
        res = pair_search_fast(h, c, i, j, n)
        if res is not None:
            h[i], h[j] = res
            c = compute_c(h, n)
            v = float(c.max())
            if v < bestc:
                bestc = v
                besth = h.copy()
            noimp = 0
        else:
            noimp += 1
    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc

def gradient_phase(h, n, deadline):
    """Run projected gradient with increasing alpha."""
    h = project_box_sum(h, n).copy()
    bestc = float(compute_c(h, n).max())
    besth = h.copy()
    phases = [(20.0, 200, 0.005), (100.0, 200, 0.003), (500.0, 200, 0.002),
              (2000.0, 150, 0.001), (10000.0, 100, 0.0005)]
    for alpha, niter, step0 in phases:
        if time.time() > deadline - 3:
            break
        h2 = projected_grad_descent(h, n, alpha, niter, step0)
        h2 = project_box_sum(h2, n)
        v2 = float(compute_c(h2, n).max())
        if v2 < bestc:
            bestc = v2
            besth = h2.copy()
        h = besth
    return besth, bestc

def optimize_single(h, n, deadline, seed):
    """Full optimization pipeline for a single n."""
    h = project_box_sum(h, n)
    bestc = float(compute_c(h, n).max())
    besth = h.copy()
    # Gradient phase (60% of time)
    grad_dl = time.time() + 0.55 * (deadline - time.time())
    h1, c1 = gradient_phase(h, n, grad_dl)
    if c1 < bestc:
        bestc, besth = c1, h1
    # Local search (remaining time)
    h2, c2 = local_search(besth, n, deadline, seed)
    if c2 < bestc:

    # Final gradient + local search
    if time.time() < deadline - 5:

        if c3 < bestc:
            bestc, besth = c3, h3
        h4, c4 = local_search(besth, n, deadline, seed + 7)
        if c4 < bestc:
            bestc, besth = c4, h4
    return besth, bestc
