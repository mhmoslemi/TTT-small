                hh[i] += d; hh[j] -= d
                c = erdos_c5(hh, n)

            # Refine with scipy

                def obj(d):
                    hh = h.copy()
                    hh[i] += d; hh[j] -= d
                    return erdos_c5(hh, n)
                try:
                    rs = so.minimize_scalar(obj, bounds=(lo, hi), method='bounded',

                h = np.clip(h, 0.0, 1.0)
                cur = best_c
                improved = True
                improvements += 1
        if not improved:
            break

def binary_local_search(h, n, deadline, max_iters=100000):
    """Local search on binary (or near-binary) solutions."""
    # First, round to binary
    h_bin = (h > 0.5).astype(float)
    # Adjust to have exactly n/2 ones
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        # Remove excess ones (those closest to 0.5)
        ones = np.where(h_bin == 1)[0]
        scores = h[ones]  # ones closest to 0.5 are least "committed"
        order = np.argsort(scores)
        for idx in order:
            if h_bin.sum() > target:
                h_bin[ones[idx]] = 0
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        scores = 1 - h[zeros]  # zeros closest to 0.5
        order = np.argsort(scores)
        for idx in order:
            if h_bin.sum() < target:
                h_bin[zeros[idx]] = 1
    
    cur_c = erdos_c5(h_bin, n)
    best_h = h_bin.copy()
    best_c = cur_c
    rng = np.random.RandomState(999)
    
    for it in range(max_iters):
        if time.time() > deadline - 10:
            break
        # Pick a random 1 and a random 0
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Try the swap
        h_bin[i] = 0; h_bin[j] = 1
        c = erdos_c5(h_bin, n)
        if c <= best_c + 1e-15:  # Accept if not worse (allow equal for exploration)
            best_h = h_bin.copy(); best_c = c
        else:
            h_bin[i] = 1; h_bin[j] = 0  # Undo
    
    return best_h, best_c

def swap_delta(h, i0, j0, n):
    """Compute the change in all correlation values when swapping h[i0] (1→0) and h[j0] (0→1)."""
    # ΔC[k] for k = 0, ..., 2n-2
    # Let ℓ = k - (n-1) be the lag
    # ΔC[ℓ + n - 1] = -(1-h[i0-ℓ])*1_{0≤i0-ℓ<n} + (1-h[j0-ℓ])*1_{0≤j0-ℓ<n}
    #                   + h[i0+ℓ]*1_{0≤i0+ℓ<n} - h[j0+ℓ]*1_{0≤j0+ℓ<n}
    
    delta = np.zeros(2*n-1)
    mid = n - 1
    
    for l in range(n):  # ℓ = l (positive lag)
        k = mid + l
        # -(1-h[i0-l]) if 0 <= i0-l < n
        if 0 <= i0 - l < n:
            delta[k] -= (1.0 - h[i0-l])
        # +(1-h[j0-l]) if 0 <= j0-l < n
        if 0 <= j0 - l < n:
            delta[k] += (1.0 - h[j0-l])

        if 0 <= i0 + l < n:
            delta[k] += h[i0+l]
        # -h[j0+l] if 0 <= j0+l < n
        if 0 <= j0 + l < n:
            delta[k] -= h[j0+l]

        if 0 <= i0 + l < n:
            delta[k] -= (1.0 - h[i0+l])
        # +(1-h[j0+l]) if 0 <= j0+l < n
        if 0 <= j0 + l < n:
