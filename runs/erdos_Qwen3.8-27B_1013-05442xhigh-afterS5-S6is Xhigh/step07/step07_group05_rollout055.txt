import time
import numpy as np
import scipy.optimize as so

def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_box_sum(x, target):
    """Project x onto {y : 0<=y<=1, sum(y)=target}."""
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) < 1e-14:
        return y
    # Binary search for the right shift
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if y.sum() > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    # Fix remaining error
    diff = target - y.sum()
    mask = (y > 1e-10) & (y < 1.0 - 1e-10)
    if mask.any():
        y[mask] += diff / mask.sum()
        y = np.clip(y, 0.0, 1.0)
    return y

def grad_for_lag(h, lag, dx):
    """Gradient of c_lag(h) = sum_i h[i](1-h[i+lag])*dx wrt h."""
    n = len(h)
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    if lag >= 0:
        if lag < n:
            term1[:n-lag] = 1.0 - h[lag:]
    else:
        if -lag < n:
            term1[-lag:] = 1.0 - h[:n+lag]
    if lag >= 0:
        if lag < n:
            term2[lag:] = h[:n-lag]
    else:
        if -lag < n:
            term2[:n+lag] = h[-lag:]
    return dx * (term1 - term2)

def solve_active_lp(h, n, active_indices, dx):
    """Solve the LP: min z s.t. grad_k·h - z <= b_k for k in active, 0<=h<=1, sum(h)=n/2."""
    target = 0.5 * n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Build LP in standard form: min c^T x s.t. A_ub x <= b_ub, A_eq x = b_eq, bounds
    # x = [h_0, ..., h_{n-1}, z]
    # Objective: minimize z -> c_obj = [0,...,0,1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: grad_k · h - z <= b_k
    A_ub_list = []
    b_ub_list = []
    for idx in active_indices:
        lag = idx - (n - 1)
        g = grad_for_lag(h, lag, dx)
        b_k = g @ h - c[idx]  # grad_k · h_t - c_k(h_t)
        row = np.concatenate([g, [-1.0]])
        A_ub_list.append(row)
        b_ub_list.append(b_k)
    
    A_ub = np.array(A_ub_list) if A_ub_list else np.zeros((1, n+1))
    b_ub = np.array(b_ub_list) if b_ub_list else np.zeros(1)
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds: 0 <= h_j <= 1, z >= 0
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = so.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    return None

def active_set_refine(h, n, deadline, eps=0.005, max_iter=200):
    """Active-set cutting plane method."""
    h = np.array(h, dtype=float)
    dx = 2.0 / n
    target = 0.5 * n
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_c = erdos_c5(h, n)
    
    for it in range(max_iter):
        if time.time() > deadline - 5.0:
            break
        
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        cur_max = float(np.max(c))
        
        # Find active set
        threshold = cur_max - eps
        active = np.where(c > threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        # Limit active set size
        if len(active) > 30:
            # Take top 30
            top_idx = np.argsort(c[active])[-30:]
            active = active[top_idx]
        
        # Solve LP
        h_lp = solve_active_lp(h, n, active, dx)
        if h_lp is None:

        # Project to feasible set
        h_lp = project_box_sum(h_lp, target)
        
        # Line search
        best_alpha = 0.0
        best_obj = cur_max
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            h_trial = h + alpha * (h_lp - h)
            h_trial = project_box_sum(h_trial, target)

def grad_lag(h, lag, dx):

    t1 = np.zeros(n)
    t2 = np.zeros(n)
    if 0 <= lag < n:

    elif -n < lag < 0:
        t1[-lag:] = 1.0 - h[:n + lag]
    if 0 <= lag < n:
        t2[lag:] = h[:n - lag]
    elif -n < lag < 0:
        t2[:n + lag] = h[-lag:]
    return dx * (t1 - t2)

    A_eq = np.zeros((1, n + 1))
