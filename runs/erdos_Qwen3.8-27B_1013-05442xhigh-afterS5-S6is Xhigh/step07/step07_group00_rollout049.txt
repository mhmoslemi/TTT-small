def smooth_grad_vec(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    
    # Create index matrix
    # For each i (0..n-1) and each m (-(n-1)..n-1):
    # fwd index = i + m, valid if 0 <= i+m < n
    # bwd index = i - m, valid if 0 <= i-m < n
    
    # Build Toeplitz-like matrix
    # fwd_mat[i, m_idx] = x[i+m] if valid, else 0
    # bwd_mat[i, m_idx] = x[i-m] if valid, else 0
    
    m_vals = np.arange(-(n-1), n)  # length 2n-1
    # fwd: i + m_val in [0, n-1]
    i_vals = np.arange(n)[:, None]  # n x 1
    m_col = m_vals[None, :]  # 1 x (2n-1)
    
    fwd_idx = i_vals + m_col  # n x (2n-1)
    fwd_valid = (fwd_idx >= 0) & (fwd_idx < n)
    bwd_idx = i_vals - m_col  # n x (2n-1)
    bwd_valid = (bwd_idx >= 0) & (bwd_idx < n)
    
    # Clip indices for safe indexing
    fwd_idx_clipped = np.clip(fwd_idx, 0, n-1)
    bwd_idx_clipped = np.clip(bwd_idx, 0, n-1)
    
    x_fwd = x[fwd_idx_clipped]  # n x (2n-1)
    x_bwd = x[bwd_idx_clipped]  # n x (2n-1)
    
    A = (fwd_valid * w[None, :]).sum(axis=1)  # n
    B = (fwd_valid * w[None, :] * x_fwd).sum(axis=1)  # n
    C = (bwd_valid * w[None, :] * x_bwd).sum(axis=1)  # n
    
    return dx * (A - B - C)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_box_sum(h, target):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_func(x, n, alpha, pen):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    obj = (m + lse / alpha) * (2.0 / n)
    if pen > 0:
        obj += pen * (x.sum() - n / 2.0) ** 2
    return obj

def soft_max_grad(x, n, alpha, pen):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    m_vals = np.arange(-(n - 1), n)
    i = np.arange(n)[:, None]
    m_c = m_vals[None, :]
    fi = i + m_c
    fv = (fi >= 0) & (fi < n)
    bi = i - m_c
    bv = (bi >= 0) & (bi < n)
    fc = np.clip(fi, 0, n - 1)
    bc = np.clip(bi, 0, n - 1)
    A = (fv * w).sum(axis=1)
    B = (fv * w * x[fc]).sum(axis=1)
    C = (bv * w * x[bc]).sum(axis=1)
    grad = dx * (A - B - C)
    if pen > 0:
        grad += 2.0 * pen * (x.sum() - n / 2.0)
    return grad

def lbfgsb_opt(h, n, alpha, pen, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(soft_max_func, h, args=(n, alpha, pen),
                       jac=lambda x: soft_max_grad(x, n, alpha, pen),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return project_box_sum(res.x, n / 2.0), compute_c5(res.x, n)
    except Exception:
        return project_box_sum(h, n / 2.0), compute_c5(h, n)

def coord_cd(h, n, max_passes=100, deadline=None, grid_size=60):
    h = h.copy()
    for _ in range(max_passes):
        if deadline and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if deadline and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            m_vals = np.arange(-(n - 1), n)
            fwd_valid = (p + m_vals >= 0) & (p + m_vals < n)
            bwd_valid = (p - m_vals >= 0) & (p - m_vals < n)
            slopes = np.zeros(2 * n - 1)
            slopes[fwd_valid] += (1.0 - h[p + m_vals[fwd_valid]])
            slopes[bwd_valid] -= h[p - m_vals[bwd_valid]]
            diag_k = n - 1
            aff_mask = fwd_valid | bwd_valid
            unaff_max = float(c[~aff_mask].max()) if not aff_mask.all() else 0.0
            d_grid = np.linspace(-h0, 1.0 - h0, grid_size)
            nv = c[None, :] + d_grid[:, None] * slopes[None, :]
            nv[:, diag_k] -= d_grid ** 2[:, None]
            max_per_d = nv.max(axis=1)
            max_per_d = np.maximum(max_per_d, unaff_max)
            best_idx = int(max_per_d.argmin())
            best_d = d_grid[best_idx]
            if abs(best_d) > 1e-12:
                h[p] = h0 + best_d
                c = c + best_d * slopes
                c[diag_k] -= best_d * best_d
                improved = True
        h = project_box_sum(h, n / 2.0)
        if not improved:
            break
    return h, compute_c5(h, n)

    h = h.copy()

        improved = False

            slopes = np.zeros(2 * n - 1)
            slopes[fwd_valid] += (1.0 - h[p + m_vals[fwd_valid]])
            slopes[bwd_valid] -= h[p - m_vals[bwd_valid]]
            diag_k = n - 1
            aff_mask = fwd_valid | bwd_valid
            unaff_max = float(c[~aff_mask].max()) if not aff_mask.all() else 0.0
            # Fine grid around current best
            d_coarse = np.linspace(-h0, 1.0 - h0, 50)
            nv_c = c[None, :] + d_coarse[:, None] * slopes[None, :]
            nv_c[:, diag_k] -= d_coarse ** 2[:, None]
            mc = np.maximum(nv_c.max(axis=1), unaff_max)
            ci = int(mc.argmin())
            lo_d = d_coarse[max(0, ci - 2)]
            hi_d = d_coarse[min(len(d_coarse) - 1, ci + 2)]
            d_fine = np.linspace(lo_d, hi_d, 200)
            nv_f = c[None, :] + d_fine[:, None] * slopes[None, :]
            nv_f[:, diag_k] -= d_fine ** 2[:, None]
            mf = np.maximum(nv_f.max(axis=1), unaff_max)
            best_d = d_fine[int(mf.argmin())]
            if abs(best_d) > 1e-10:
                h[p] = h0 + best_d
                c = c + best_d * slopes
                c[diag_k] -= best_d * best_d

        h = project_box_sum(h, n / 2.0)

            break
