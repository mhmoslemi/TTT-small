import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_slsqp(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgradient_push(h, n, steps, dl, t_target=None):
    """Projected subgradient on the max-correlation with diminishing step size."""
    dx = 2.0 / n
    x = h.copy()
    n_corr = 2 * n - 1
    m_idx = n - 1
    if t_target is None:
        t_target = compute_c5(x, n) * 0.998
    for it in range(steps):
        if time.time() > dl:
            break
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        cmax = c.max()
        active = c > cmax - 1e-6
        na = active.sum()
        if na == 0:
            break
        lr = 0.02 / (1.0 + 0.01 * it)
        grad = np.zeros(n)
        for k_idx in np.where(active)[0]:
            lag = k_idx - m_idx
            g = np.zeros(n)
            g[:] = 1.0
            jk = np.arange(n) + lag
            msk = (jk >= 0) & (jk < n)
            g[~msk] -= 0.0
            g[msk] -= x[jk[msk]]
            jk2 = np.arange(n) - lag
            msk2 = (jk2 >= 0) & (jk2 < n)
            g[~msk2] -= 0.0
            g[msk2] -= x[jk2[msk2]]
            grad += g / na
        x = x - lr * grad * n
        x = np.clip(x, 0.0, 1.0)
        diff = n / 2.0 - x.sum()
        if diff > 0:
            room = 1.0 - x
            s = room.sum()
            if s > 1e-14:
                x += diff / s * room
        else:
            s = x.sum()
            if s > 1e-14:
                x += diff / s * x
        x = np.clip(x, 0.0, 1.0)
    return x

def flatness_slsqp(h, n, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        return np.sum((g - g.mean()) ** 2)
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pn_slsqp(h, n, p, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        c_pos = np.maximum(c, 1e-12)
        return (np.sum(c_pos ** p)) ** (1.0 / p)
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def barrier_push(h, n, c_cur, dl, n_iter=20, rate=0.003):
    h = h.copy()
    c = c_cur
    for i in range(n_iter):
        if time.time() > dl - 3:
            break
        t = c * (1.0 - rate * (i + 1))
        if t < 0.3:
            t = 0.3
        dx = 2.0 / n
        def obj(x):
            cc = np.correlate(x, 1.0 - x, mode='full') * dx
            excess = np.maximum(cc - t, 0)
            return 0.5 * np.sum(excess ** 2)
        def cons(x):
            return x.sum() - n / 2.0
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons}],
                           options={'maxiter': 3000, 'ftol': 1e-16})
            h = project_feasible(res.x, n)
        except Exception:
            pass
        c = compute_c5(h, n)
        if c > c_cur:
            c = c_cur
    return h, c

def greed_binary(h_init, n, deadline, max_sweeps=50):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    idx_sort = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx_sort[:n // 2]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h, best_c = h.copy(), cur_c
    ones_list = [int(x) for x in np.where(h == 1)[0]]
    zeros_list = [int(x) for x in np.where(h == 0)[0]]
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros_list:
            if time.time() > deadline:
                break
            best_j, best_nc = -1, cur_c
            for j in ones_list:
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta = np.zeros(2 * n - 1)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < len(delta):
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < len(delta):
                    delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc, best_j = nc, j
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                j = best_j
                h[i], h[j] = 1.0, 0.0
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                ones_list.remove(j)
                zeros_list.remove(i)
                ones_list.append(i)
                zeros_list.append(j)
                improved = True
                if cur_c < best_c:
                    best_c, best_h = cur_c, h.copy()
                break
        if not improved:
            break
    return best_h, best_c
