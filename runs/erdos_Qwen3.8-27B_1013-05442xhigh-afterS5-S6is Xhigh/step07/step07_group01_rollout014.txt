    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def lsbfgs_smooth(h, n, alpha, penalty, deadline, maxiter=5000):
    """L-BFGS-B with log-sum-exp smoothing and penalty for constraint."""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj_pen(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        sh = alpha * (corr - m)
        ms = sh.max()
        if ms > 500:
            ms = 500
        log_sum = ms + np.log(np.exp(sh - ms).sum())
        smooth_max = m + log_sum / alpha
        pen = penalty * (x.sum() - target) ** 2
        return smooth_max + pen
    
    def grad_pen(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        sh = alpha * (corr - m)
        ms = sh.max()
        if ms > 500:
            ms = 500
        exp_sh = np.exp(sh - ms)
        weights = exp_sh / exp_sh.sum()
        
        grad = np.zeros(n)
        for li in range(2 * n - 1):
            l = li - (n - 1)
            w = weights[li]
            if w < 1e-12:
                continue
            if l >= 0 and l < n:
                grad[:n-l] += w * dx * (1.0 - x[l:])
                grad[l:] -= w * dx * x[:n-l]
            elif l < 0 and -l < n:
                ll = -l
                grad[ll:] += w * dx * (1.0 - x[:n-ll])
                grad[:n-ll] -= w * dx * x[ll:]
        
        grad += 2 * penalty * (x.sum() - target)
        return grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h, jac=grad_pen, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15,
                       'gtol': 1e-12, 'maxfun': maxiter * 3})
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        sh = alpha * (corr - m)
        ms = sh.max()
        if ms > 500:
            ms = 500
        log_sum = ms + np.log(np.exp(sh - ms).sum())
        return m + log_sum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad_descent(h, n, deadline, mu=0.93):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = dx * 2.0
    momentum = np.zeros(n)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    stall = 0
    for step in range(3000000):
        if time.time() > deadline:
            break
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            if lag < n:
                grad[:n - lag] += (1.0 - h[lag:])
                grad[lag:] -= h[:n - lag]
        else:
            ll = -lag
            if ll < n:
                grad[ll:] += (1.0 - h[:n - ll])
                grad[:n - ll] -= h[ll:]
        grad *= dx
        momentum = mu * momentum - lr * grad
        h_new = project_feasible(h + momentum, n)
        corr = np.correlate(h_new, 1 - h_new, mode='full') * dx
        c = float(corr.max())
        if c < best_c - 1e-12:
            best_c = c
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.03, dx * 20.0)
        else:
            stall += 1
            if stall > 200:
                lr = lr * 0.95
                mu = max(mu - 0.002, 0.5)
                if lr < dx * 0.01:
                    lr = dx * 0.3
                    stall = 0
        h = h_new
    return best_h, best_c

def power_sum_smooth(h, n, p, deadline, maxiter=5000):
    """Optimize using power-sum approximation with L-BFGS-B."""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        # Power sum: (1/m * sum((corr/m)^p))^(1/p) * m ≈ max for large p
        # But we need to handle negative values... corr should be >= 0 since h in [0,1]
        corr_pos = np.maximum(corr, 0)
        cs = (np.mean(corr_pos ** p)) ** (1.0/p)
        pen = 1e6 * (x.sum() - target) ** 2
        return cs + pen
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

            no_improve = 0
            c_real = compute_c5(x_new, n)
            if c_real < best_c:
                best_c = c_real
                best_h = x_new.copy()
        else:
            no_improve += 1
            if no_improve > 300:
                lr *= 0.85
                no_improve = 0
                if lr < 1.0 / (alpha * dx * 200):
                    break
        x = x_new
        y = y_new
        t_val = t_new
    return best_h, best_c

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = corr.max()
    sh = alpha * (corr - m)
    ms = sh.max()
    if ms > 500:
        ms = 500
    log_sum = ms + np.log(np.exp(sh - ms).sum())
    return m + log_sum / alpha

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Subgradient exploration
    p1 = t0 + 60
    h1 = best_h.copy()
    h1, c1 = subgrad_descent(h1, n, p1, mu=0.94)
    if c1 < best_c:
        best_c, best_h = c1, h1.copy()

    # Phase 2: L-BFGS-B with log-sum-exp, increasing alpha
    p2 = t0 + 350
    configs = [(500, 10, 2000), (2000, 50, 3000), (10000, 500, 3000),
               (50000, 2000, 3000), (200000, 10000, 4000), (1000000, 50000, 4000)]
    for a, pen, mi in configs:
        if time.time() > p2 - 30:
            break
        h2 = best_h.copy()
        h2 = lsbfgs_smooth(h2, n, a, pen, maxiter=mi)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()

    # Phase 3: Nesterov smooth gradient
    p3 = t0 + 550
    for a in [10000, 100000]:
        if time.time() > p3 - 40:
            break
        h3 = best_h.copy()
        h3, c3 = nesterov_smooth(h3, n, a, min(t0 + 50, p3), max_steps=30000)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()

    # Phase 4: SLSQP with aggressive alpha
    p4 = t0 + 750
    for a in [50000, 500000, 5000000, 50000000]:
        if time.time() > p4 - 40:
            break
        h4 = best_h.copy()
        h4 = slsqp_smooth(h4, n, a, 3000, 1e-15)
        c4 = compute_c5(h4, n)
        if c4 < best_c:
            best_c, best_h = c4, h4.copy()

    # Phase 5: Try coarser resolution
    p5 = t0 + 900
    if n >= 200 and time.time() < p5 - 80:
        for nl in [n // 2, n // 4]:
            if nl < 30 or time.time() > p5 - 40:
                continue
            xo = np.linspace(0, 2, n)
            xl = np.linspace(0, 2, nl)
            hl = np.interp(xl, xo, best_h)
            hl = project_feasible(hl, nl)
            hl = lsbfgs_smooth(hl, nl, 5000, 100, maxiter=2000)
            hl = lsbfgs_smooth(hl, nl, 50000, 1000, maxiter=2000)
            hl = slsqp_smooth(hl, nl, 100000, 1500, 1e-14)
            c_coarse = compute_c5(hl, nl)
            if c_coarse < best_c:
                best_h = hl.copy()
                best_c = c_coarse
                n = nl
            else:
                hl_hi = np.interp(xo, xl, hl)
                hl_hi = project_feasible(hl_hi, n)
                hl_hi = lsbfgs_smooth(hl_hi, n, 50000, 1000, maxiter=2000)
                c_up = compute_c5(hl_hi, n)
                if c_up < best_c:
                    best_c, best_h = c_up, hl_hi.copy()

    # Phase 6: Noise restarts
    p6 = deadline - 60
    for trial in range(12):
        if time.time() > p6:
            break
        nl = 0.0015 * (1.0 + trial * 0.4)
        h6 = best_h + nl * rng.randn(n)
        h6 = project_feasible(h6, n)
        h6 = lsbfgs_smooth(h6, n, 10000, 100, maxiter=1500)
        h6 = lsbfgs_smooth(h6, n, 100000, 1000, maxiter=1500)
        c6 = compute_c5(h6, n)
        if c6 < best_c:
            best_c, best_h = c6, h6.copy()

    # Phase 7: Final SLSQP polish
    t_end = deadline - 3
    if time.time() < t_end - 25:
        h7 = best_h.copy()
        for a in [1000000, 10000000, 100000000]:
            if time.time() > t_end - 12:
                break
            h7 = slsqp_smooth(h7, n, a, 4000, 1e-17)
            c7 = compute_c5(h7, n)
            if c7 < best_c:
                best_c, best_h = c7, h7.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
