import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _sum_con(x, n):
    return x.sum() - n / 2.0

def _sum_con_jac(x, n):
    return np.ones(len(x))

def slsqp_optimize(h, n, alphas, maxiter_each=1000, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def _find_best_delta(h0, c_mv, a_mv, diag_mask, dm_max, delta_lo, delta_hi):
    # Find candidates: pairwise intersections + vertex + boundaries
    n_act = len(a_mv)
    candidates = [delta_lo, delta_hi, 0.0]
    for i in range(n_act):
        if diag_mask[i]:
            continue
        for j in range(i + 1, n_act):
            if diag_mask[j]:
                continue
            da = a_mv[i] - a_mv[j]
            if abs(da) > 1e-14:
                dc = c_mv[j] - c_mv[i]
                di = dc / da
                if delta_lo - 1e-9 <= di <= delta_hi + 1e-9:
                    candidates.append(di)
    if np.any(diag_mask):
        v = 0.5 - h0
        if delta_lo <= v <= delta_hi:
            candidates.append(v)
    candidates = np.array(candidates)
    candidates = np.clip(candidates, delta_lo, delta_hi)
    best_d = 0.0
    best_v = np.inf
    for d in candidates:
        vals = c_mv + d * a_mv
        vals = vals.copy()
        if np.any(diag_mask):
            vals[diag_mask] -= d * d
        val = max(float(vals.max()), dm_max)
        if val < best_v:
            best_v = val
            best_d = d
    # Refine with ternary search
    span = max((delta_hi - delta_lo) * 0.03, 0.005)
    lo = max(delta_lo, best_d - span)
    hi = min(delta_hi, best_d + span)
    for _ in range(35):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        v1 = c_mv + m1 * a_mv
        v1 = v1.copy()
        if np.any(diag_mask):
            v1[diag_mask] -= m1 * m1
        v2 = c_mv + m2 * a_mv
        v2 = v2.copy()
        if np.any(diag_mask):
            v2[diag_mask] -= m2 * m2
        f1 = max(float(v1.max()), dm_max)
        f2 = max(float(v2.max()), dm_max)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_exact(h, n, max_passes=300, deadline=None):
    h = h.copy()
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_range = np.arange(2 * n - 1)
            f_idx = p + m_range - (n - 1)
            f_valid = (f_idx >= 0) & (f_idx < n)
            b_idx = p - m_range + (n - 1)
            b_valid = (b_idx >= 0) & (b_idx < n)
            a = np.zeros(2 * n - 1)
            a[f_valid] += (1.0 - h[f_idx[f_valid]])
            a[b_valid] -= h[b_idx[b_valid]]
            m_valid = f_valid | b_valid
            mv_idx = np.where(m_valid)[0]
            c_mv = c[mv_idx]
            a_mv = a[mv_idx]
            diag_mask = (mv_idx == (n - 1))
            dm_max = float(c[~m_valid].max()) if np.any(~m_valid) else 0.0
            # Limit active set to top K for efficiency
            c_curr_max = float(c_mv.max())
            active_thresh = c_curr_max - 0.15 * max(c_curr_max, 1e-6)
            active = c_mv >= active_thresh
            # Always include diagonal
            active |= diag_mask
            # Cap at 40 active
            if active.sum() > 40:
                act_idx = np.where(active)[0]
                order = np.argsort(-c_mv[act_idx])
                keep = act_idx[order[:35]]
                active2 = np.zeros_like(active)
                active2[keep] = True
                active2[diag_mask] = True
                active = active2
            if active.sum() < 2:
                continue
            delta_lo = -h0
            delta_hi = 1.0 - h0
            delta_opt = _find_best_delta(h0, c_mv[active], a_mv[active], diag_mask[active], dm_max, delta_lo, delta_hi)
            if abs(delta_opt) > 1e-10:
                h[p] = float(np.clip(h0 + delta_opt, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def subgradient_step(h, n, step_size, deadline=None):
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(c))
    # Compute gradient of c[k_star] w.r.t. h
    m_range = np.arange(2 * n - 1)
    kg = k_star
    f_idx = m_range - (n - 1) + np.arange(n) - f_idx  # not quite right
    # Simpler: for lag kg, c[kg] = sum_i h[i]*(1-h[i+kg-(n-1)]) for valid i
    offset = kg - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        j = i + offset
        if 0 <= j < n:
            grad[i] += (1.0 - h[j])
            grad[j] += (-h[i])
    h_new = h - step_size * grad * (2.0 / n)
    h_new = project_feasible(h_new, n)
    return h_new, compute_c5(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n = 200

    h_best = project_feasible(h_init, n)
    c_best = compute_c5(h_best, n)

    # Phase 1: SLSQP
    h, c = slsqp_optimize(h_best, n, [50, 200, 1000, 5000, 50000], 2000, deadline)
    if c < c_best:
        h_best, c_best = h, c

    # Phase 2: Coordinate descent
    h, c = coord_exact(h_best, n, 300, deadline)
    if c < c_best:
        h_best, c_best = h, c

    # Phase 3: SLSQP high alpha
    h, c = slsqp_optimize(h_best, n, [50000, 200000, 1000000, 5000000], 2000, deadline)
    if c < c_best:
        h_best, c_best = h, c

    # Phase 4: More coord descent
    h, c = coord_exact(h_best, n, 300, deadline)
    if c < c_best:
        h_best, c_best = h, c

    # Phase 5: Try different n
    n_list = [300, 400, 600, 700, 800]
    for n_try in n_list:
        if time.time() > deadline - 90:
            break
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h, c = slsqp_optimize(h_int, n_try, [1000, 10000, 100000], 1500, deadline)
        h, c = coord_exact(h, n_try, 200, deadline)
        if c < c_best:
            h_best, c_best = h, c

    # Phase 6: Subgradient + coord for escape
    if time.time() < deadline - 60:
        n = len(h_best)
        for sr in range(6):
            if time.time() > deadline - 30:
                break
            h_try = h_best + rng.normal(0, 0.03 + 0.02 * sr, n)
            h_try = project_feasible(h_try, n)
            h_try, c_try = coord_exact(h_try, n, 100, deadline)
            if c_try < c_best:
                h_best, c_best = h_try, c_try

    # Phase 7: Final polish
    if time.time() < deadline - 5:
        n = len(h_best)
        h, c = slsqp_optimize(h_best, n, [5000000, 10000000], 1000, deadline)
        if c < c_best:
            h_best, c_best = h, c
        h, c = coord_exact(h_best, n, 200, deadline)
        if c < c_best:
            h_best, c_best = h, c

    h_best = project_feasible(h_best, len(h_best))
    c_best = compute_c5(h_best, len(h_best))
    return h_best, c_best, len(h_best)
