    """Find exact optimal value for coordinate p."""
    c = np.correlate(h, 1.0 - h, mode='full')
    h0 = h[p]
    
    # Compute coefficients a[k] for each lag k
    # c[k] changes by delta * a[k] - delta^2 * (k == n-1)
    k_vals = np.arange(2*n - 1)
    # Forward contribution: h[p] * (1-h[p+k-(n-1)])
    jf = p + k_vals - (n-1)
    valid_f = (jf >= 0) & (jf < n)
    # Backward contribution: -h[p-k+n-1] * h[p]
    jb = p - k_vals + (n-1)
    valid_b = (jb >= 0) & (jb < n)
    
    a = np.zeros(2*n - 1)
    a[valid_f] += (1.0 - h[jf[valid_f]])
    a[valid_b] -= h[jb[valid_b]]
    
    # Breakpoints: where c[k1](d) = c[k2](d)
    # For k1, k2 both != n-1: d = (c[k2] - c[k1]) / (a[k1] - a[k2])
    # For k != n-1 and n-1: d^2 + d*(a[k]-a[n-1]) + (c[k]-c[n-1]) = 0

    # Linear/linear breakpoints
    for k1 in range(2*n - 1):
        if k1 == n-1:

        for k2 in range(k1+1, 2*n - 1):
            if k2 == n-1:
                continue
            denom = a[k1] - a[k2]
            if abs(denom) < 1e-12:
                continue
            d = (c[k2] - c[k1]) / denom
            if -h0 <= d <= 1.0 - h0:
                candidates = np.append(candidates, d)
    
    # Linear/quadratic breakpoints (with center)
    for k in range(2*n - 1):
        if k == n-1:
            continue
        # d^2 + (a[k]-a[n-1])*d + (c[k]-c[n-1]) = 0
        A = 1.0
        B = a[k] - a[n-1]
        C = c[k] - c[n-1]
        disc = B*B - 4*A*C
        if disc < 0:
            continue
        sqrt_disc = np.sqrt(disc)
        d1 = (-B + sqrt_disc) / (2*A)
        d2 = (-B - sqrt_disc) / (2*A)
        if -h0 <= d1 <= 1.0 - h0:
            candidates = np.append(candidates, d1)
        if -h0 <= d2 <= 1.0 - h0:
            candidates = np.append(candidates, d2)
    
    # Also check where derivative of quadratic = 0: d = a[n-1]/2
    if -h0 <= a[n-1]/2 <= 1.0 - h0:

        c_new[n-1] -= d*d
        val = c_new.max()
        if val < best_val:

            best_d = d
    
    return best_d, best_val

import numpy as np
import cvxpy as cp
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_A(g, n):
    A = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        k = m - (n-1)
        lo = max(0, k)
        hi = min(n-1, k+n-1)
        if lo <= hi:
            A[m, lo:hi+1] = g[lo-k:hi-k+1]
    return A

def build_B(h, n):
    B = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        k = m - (n-1)
        lo = max(0, -k)
        hi = min(n-1, n-1-k)
        if lo <= hi:
            B[m, lo:hi+1] = h[lo+k:hi+k+1]
    return B

def solve_lp_A(g, n, solver='GLOP'):
    hv = cp.Variable(n)
    t = cp.Variable()
    A = build_A(g, n)
    cons = [A @ hv <= t, hv >= 0, hv <= 1, cp.sum(hv) == n/2.0]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except:
        prob.solve(solver=cp.ECOS, verbose=False)
    if hv.value is not None:
        return np.array(hv.value).flatten(), float(t.value)
    return None, np.inf

def solve_lp_B(h, n, solver='GLOP'):
    gv = cp.Variable(n)
    t = cp.Variable()
    B = build_B(h, n)
    cons = [B @ gv <= t, gv >= 0, gv <= 1, cp.sum(gv) == n/2.0]
    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=solver, verbose=False)
    except:
        prob.solve(solver=cp.ECOS, verbose=False)
    if gv.value is not None:
        return np.array(gv.value).flatten(), float(t.value)
    return None, np.inf

def coord_cd(h, n, max_passes=100, deadline=None, grid_size=100):
    h = h.copy()
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                return h, compute_c5(h, n)
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            # Compute effect of changing h[p] by d
            k_vals = np.arange(2*n-1)
            jf = p + k_vals - (n-1)
            jb = p - k_vals + (n-1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(2*n-1)
            if np.any(valid_f):
                a[valid_f] += 1.0 - h[jf[valid_f]]
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            
            # Find best d using grid
            d_grid = np.linspace(-h0, 1.0 - h0, grid_size)
            best_d = 0.0
            best_val = float(c.max())
            center = n - 1
            for d in d_grid:
                if abs(d) < 1e-15:
                    continue
                c_new = c + d * a
                c_new[center] -= d * d
                val = float(c_new.max())
                if val < best_val:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-12:
                h[p] = h0 + best_d
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)
