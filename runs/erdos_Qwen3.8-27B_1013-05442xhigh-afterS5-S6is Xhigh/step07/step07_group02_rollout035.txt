def smooth_gradient(h, n, alpha):

    c = np.correlate(h, 1 - h, mode='full')  # length 2n-1

        # i-k ranges over 0 to n-1, so i ranges over k to k+n-1, intersect [0,n-1]
        # i ranges over max(0,k) to min(n-1, k+n-1)
        i1_start = max(0, -k)

        if i1_end > i1_start:
            grad[i1_start:i1_end] += weight * (1.0 - h[i1_start + k:i1_end + k])

        if i2_end > i2_start:

            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_gradients(h, n):
    G = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        j1s = max(0, -k)
        j1e = min(n, n - k)
        if j1e > j1s:
            G[j1s:j1e, m] = 1.0 - h[j1s + k:j1e + k]
        j2s = max(0, k)
        j2e = min(n, n + k)
        if j2e > j2s:
            G[j2s:j2e, m] -= h[j2s - k:j2e - k]
    return G

def lp_step(h, n, eps):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    G = compute_gradients(h, n)
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in ['GLOP', 'GLPK', 'ECOS', 'CLARABEL']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()
    best_mc = mc

            best_h = h_try

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_gradients(h, n):
    G = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        j1s, j1e = max(0, -k), min(n, n - k)
        if j1e > j1s:
            G[j1s:j1e, m] = 1.0 - h[j1s + k:j1e + k]
        j2s, j2e = max(0, k), min(n, n + k)
        if j2e > j2s:
            G[j2s:j2e, m] -= h[j2s - k:j2e - k]
    return G

def lp_step(h, n, eps, top_m=None):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    G = compute_gradients(h, n)
    delta = cp.Variable(n)
    t = cp.Variable()
    if top_m is not None and top_m < 2 * n - 1:
        idx = np.argsort(C)[-top_m:]
        cons = [C[idx] + G[:, idx].T @ delta <= t]
    else:
        cons = [C + G.T @ delta <= t]
    cons += [delta >= -eps, delta <= eps, delta >= -h, delta <= 1.0 - h, cp.sum(delta) == 0]
    prob = cp.Problem(cp.Minimize(t), cons)
    for sol in ['GLOP', 'GLPK', 'CLARABEL']:
        try:
            prob.solve(solver=sol)
            break
        except Exception:
            pass
    if delta.value is None:
        return h, mc
    d = delta.value
    best_h, best_mc = h.copy(), mc
    for lam in [1.0, 0.8, 0.6, 0.4, 0.2]:
        ht = np.clip(h + lam * d, 0, 1)
        ht = project_feasible(ht, n)
        mct = float(np.max(np.correlate(ht, 1 - ht, mode='full')))
        if mct < best_mc - 1e-14:
            best_mc, best_h = mct, ht
    return best_h, best_mc

def smooth_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    cm = c - c.max()
    p = np.exp(alpha * cm)
    p /= p.sum()
    grad = np.zeros(n)
    for ko in range(2 * n - 1):
        w = p[ko]
        if w < 1e-12:
            continue
        k = ko - (n - 1)
        i1s, i1e = max(0, -k), min(n, n - k)
        if i1e > i1s:
            grad[i1s:i1e] += w * (1.0 - h[i1s + k:i1e + k])
        i2s, i2e = max(0, k), min(n, n + k)
        if i2e > i2s:
            grad[i2s:i2e] -= w * h[i2s - k:i2e - k]
    return grad

def pgd_run(h, n, alpha, lr0, max_iter, deadline):
    lr = lr0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        g = smooth_grad(h, n, alpha)
        hg = float(np.max(np.correlate(h, 1 - h, mode='full')))
        hn = project_feasible(h - lr * g, n)
        cn = float(np.max(np.correlate(hn, 1 - hn, mode='full')))
        if cn < hg:
            h, lr = hn, min(lr * 1.03, 3.0)
        else:
            lr *= 0.7
            if lr < 1e-9:
                break
    return h

def structured_noise(n, rng, scale):
    freq = rng.randint(2, max(3, n // 4 + 1))
    phase = rng.uniform(0, 2 * np.pi)
    t = np.arange(n)
    noise = scale * np.sin(2 * np.pi * freq * t / n + phase)
    if rng.rand() < 0.3:
        f2 = rng.randint(2, max(3, n // 6 + 1))
        p2 = rng.uniform(0, 2 * np.pi)
        noise += 0.5 * scale * np.sin(2 * np.pi * f2 * t / n + p2)
    return noise

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n, h = 400, np.full(400, 0.5)
    h = project_feasible(h, n)
    best_raw = float(np.max(np.correlate(h, 1 - h, mode='full')))
    best_c5 = best_raw * 2.0 / n
    best_h = h.copy()

    # Phase 1: LP with adaptive eps and structured noise
    p1_dl = t0 + 0.48 * (deadline - t0)
    h_curr = best_h.copy()
    eps = 0.08
    iters = 0
    stale = 0
    while time.time() < p1_dl:
        tm = None if iters % 4 == 0 else (15 if iters % 4 == 1 else (8 if iters % 4 == 2 else None))
        h_new, mc = lp_step(h_curr, n, eps, tm)
        if mc < best_raw - 1e-13:
            best_raw, best_c5, best_h = mc, mc * 2.0 / n, h_new.copy()
            stale = 0
        else:
            stale += 1
        h_curr = h_new
        iters += 1
        if stale > 10:
            eps = max(eps * 0.93, 0.002)
            if eps < 0.004:
                h_curr = project_feasible(h_curr + structured_noise(n, rng, 0.01), n)
                eps = 0.06
                stale = 0
        else:
            eps = min(eps * 1.01, 0.12)
        if iters % 80 == 0:
            h_curr = project_feasible(h_curr + structured_noise(n, rng, 0.006), n)

    # Phase 2: PGD multi-scale exploration
    p2_dl = t0 + 0.72 * (deadline - t0)
    hw = best_h.copy()
    for alpha in [20, 60, 150, 400, 1000]:
        if time.time() > p2_dl:
            break
        lr0 = 0.5 / (alpha + 5)
        hw = pgd_run(hw, n, alpha, lr0, 3000, p2_dl)
        cw = float(np.max(np.correlate(hw, 1 - hw, mode='full')))
        if cw < best_raw - 1e-14:
            best_raw, best_c5, best_h = cw, cw * 2.0 / n, hw.copy()

    # Phase 3: Noise restart + LP
    p3_dl = t0 + 0.90 * (deadline - t0)
    for trial in range(300):
        if time.time() > p3_dl:
            break
        if trial % 5 == 0:
            s = 0.006 + 0.012 * rng.rand()
            hw = project_feasible(best_h + s * rng.randn(n), n)
        else:
            hw = project_feasible(best_h + structured_noise(n, rng, 0.008), n)
        er = 0.05
        for _ in range(120):
            if time.time() > p3_dl:
                break
            hw, mr = lp_step(hw, n, er)
            if mr < best_raw - 1e-13:
                best_raw, best_c5, best_h = mr, mr * 2.0 / n, hw.copy()
            er = max(er * 0.995, 0.002)

    # Phase 4: High-alpha PGD polish
    if time.time() < deadline - 20:
        hw = best_h.copy()
        for alpha in [2000, 8000, 30000]:
            if time.time() > deadline - 20:
                break
            hw = pgd_run(hw, n, alpha, 0.15 / (alpha + 1), 2000, deadline - 20)
            cw = float(np.max(np.correlate(hw, 1 - hw, mode='full')))
            if cw < best_raw - 1e-14:
                best_raw, best_c5, best_h = cw, cw * 2.0 / n, hw.copy()

    # Phase 5: SLSQP final polish
    if time.time() < deadline - 5:
        dx = 2.0 / n
        def obj(x):
            g = np.correlate(x, 1 - x, mode='full') * dx
            m = g.max()
            a = 50000.0
            sh = a * (g - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / a
        def cf(x):
            return x.sum() - n / 2.0
        try:
            r = minimize(obj, best_h, method='SLSQP',
                         bounds=[(0, 1)] * n,
                         constraints=[{'type': 'eq', 'fun': cf}],
                         options={'maxiter': 3000, 'ftol': 1e-20})
            hf = project_feasible(r.x, n)
            cf_ = float(np.max(np.correlate(hf, 1 - hf, mode='full')))
            if cf_ * 2.0 / n < best_c5:
                best_c5, best_h = cf_ * 2.0 / n, hf.copy()
        except Exception:
            pass

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, int(n)
