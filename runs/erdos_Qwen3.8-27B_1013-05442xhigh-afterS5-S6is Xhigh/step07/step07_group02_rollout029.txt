def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break

            if s < 1e-14:

            s = h.sum()
            if s < 1e-14:

    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    alpha = 50  # start with moderate smoothing
    lr = 0.1  # learning rate
    t_start = time.time()
    
    while time.time() - t_start < budget:
        c = compute_correlation(h, n)
        grad = compute_gradient_sm(h, c, alpha)
        
        # Projected gradient step with line search
        # First, compute the gradient direction
        # We need to project the gradient onto the constraint Σ h = n/2
        # The projected gradient is: grad - mean(grad) * ones (to maintain the sum constraint)
        # Then clip to [0,1] and re-project
        
        grad_proj = grad - np.mean(grad)  # make zero-mean to preserve sum
        
        # Line search
        best_f = compute_c5(h, n)
        for t in [lr, lr*0.5, lr*0.25, lr*0.1, lr*0.05]:

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    """Compute smoothed C5 value and gradient."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full').astype(float)
    c_max = c.max()
    w = np.exp(alpha * (c - c_max))
    w_sum = w.sum()
    w = w / w_sum
    f = dx * (c_max + np.log(w_sum) / alpha)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-16:
            continue
        k = m - (n - 1)
        i1s = max(0, -k)
        i1e = min(n, n - k)
        if i1e > i1s:
            grad[i1s:i1e] += wm * (1.0 - h[i1s + k:i1e + k])
        i2s = max(0, k)
        i2e = min(n, n + k)
        if i2e > i2s:
            grad[i2s:i2e] -= wm * h[i2s - k:i2e - k]
    grad *= dx
    return f, grad

def gradient_descent(h, n, alpha_start, alpha_end, lr_start, lr_end, t_max, rng):
    """Run gradient descent with adaptive parameters."""
    t_start = time.time()
    h = h.copy()
    n_iter = 0
    
    while time.time() - t_start < t_max:
        # Progress from 0 to 1
        elapsed = time.time() - t_start
        progress = min(1.0, elapsed / t_max)
        
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        f, grad = smooth_obj_grad(h, n, alpha)
        
        # Project gradient to satisfy sum constraint
        grad = grad - np.mean(grad)
        
        # Line search
        best_h = h
        best_f = f
        for t in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            h_try = h - t * grad
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try, n)
            if c5_try < best_f:
                best_h = h_try
                best_f = c5_try
                break
        
        h = best_h
        n_iter += 1
        
        # Occasional noise for exploration
        if n_iter % 500 == 0 and progress < 0.8:
            noise_scale = 0.01 * (1 - progress)
            h = h + noise_scale * rng.randn(n)
            h = project_feasible(h, n)
    
    return h

    """Upsample h from n_old to n_new points."""
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = project_feasible(h_new, n_new)
    return h_new

    h = gradient_descent(h, n, 50, 5000, 0.05, 0.001, t_phase1, rng)
    c5 = compute_c5(h, n)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        best_n = n

            h_up = upsample(best_h, best_n, n_new)
            c5_up = compute_c5(h_up, n_new)
            
            t_phase = min(150, deadline - time.time() - 30)
            if t_phase < 20:

            h_ref = gradient_descent(h_up, n_new, 100, 5000, 0.05, 0.001, t_phase, rng)
            c5_ref = compute_c5(h_ref, n_new)
            
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
                best_n = n_new
    
    # Phase 3: Final polish at best resolution
    if time.time() < deadline - 10:
        t_phase = deadline - time.time() - 5
        h_pol = gradient_descent(best_h, best_n, 1000, 100000, 0.01, 0.0001, t_phase, rng)
        c5_pol = compute_c5(h_pol, best_n)
        if c5_pol < best_c5:

            best_h = h_pol.copy()
    
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    
    return best_h, best_c5, best_n

    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-16:
            continue
        k = m - (n - 1)
        i1s = max(0, -k)
        i1e = min(n, n - k)
        if i1e > i1s:
            grad[i1s:i1e] += wm * (1.0 - h[i1s + k:i1e + k])
        i2s = max(0, k)
        i2e = min(n, n + k)
        if i2e > i2s:
            grad[i2s:i2e] -= wm * h[i2s - k:i2e - k]
    return dx * grad

def gd_phase(h, n, alpha_start, alpha_end, lr_start, lr_end, t_max, rs, noise_interval=300):
    t0 = time.time()
    if n > 1000:
        mode = 'fft'
    else:
        mode = 'exact'
    h = h.copy()
    while time.time() - t0 < t_max:
        p = min(1.0, (time.time() - t0) / max(t_max, 1e-9))
        a = alpha_start * (alpha_end / alpha_start) ** p
        lr = lr_start * (lr_end / lr_start) ** p
        if mode == 'fft':
            g = grad_fft(h, n, a)
        else:
            g = grad_exact(h, n, a)
        g -= np.mean(g)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            return h
        g_dir = g / g_norm
        cur_c5 = compute_c5(h, n)
        best_fh = h
        best_fv = cur_c5
        steps = [lr, lr * 0.5, lr * 0.25, lr * 0.1]
        for t in steps:
            h_try = h - t * g_dir
            h_try = project_feasible(h_try, n)
            fv = compute_c5(h_try, n)
            if fv < best_fv - 1e-15:
                best_fv = fv
                best_fh = h_try
                break
        h = best_fh
        if noise_interval > 0 and (int((time.time() - t0) * 10) % noise_interval == 5) and p < 0.6:
            ns = 0.008 * (1.0 - p)
            h = h + ns * rs.standard_normal(n)
            h = project_feasible(h, n)
    return h

def gd_exact_polish(h, n, n_explore, t_max, rs):
    t0 = time.time()
    h = h.copy()
    cur_c5 = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = cur_c5
    phase1_end = t0 + 0.7 * t_max
    a1s, a1e = 200, 5000
    lr1s, lr1e = 0.03, 0.001
    while time.time() < phase1_end:
        p = min(1.0, (time.time() - t0) / max(0.7 * t_max, 1e-9))
        a = a1s * (a1e / a1s) ** p
        lr = lr1s * (lr1e / lr1s) ** p
        g = grad_exact(h, n, a)
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        gd = g / gn
        cv = compute_c5(h, n)
        bh = h
        bv = cv
        for t in [lr, lr * 0.5, lr * 0.25]:
            ht = project_feasible(h - t * gd, n)
            fv = compute_c5(ht, n)
            if fv < bv - 1e-15:
                bv = fv
                bh = ht
                break
        h = bh
        if compute_c5(h, n) < best_c5 - 1e-14:
            best_c5 = compute_c5(h, n)
            best_h = h.copy()
    cur_c5 = compute_c5(h, n)
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = h.copy()
    h = best_h.copy()
    hr = h.copy()
    t2_end = t0 + t_max
    a2s, a2e = 10000, 500000
    lr2s, lr2e = 0.005, 0.0001
    i = 0
    while time.time() < t2_end:
        i += 1
        p = min(1.0, (time.time() - phase1_end) / max(t_max * 0.3, 1e-9))
        a = a2s * (a2e / a2s) ** p
        lr = lr2s * (lr2e / lr2s) ** p
        g = grad_exact(h, n, a)
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-16:
            break
        gd = g / gn
        cv = compute_c5(h, n)
        bh = h
        bv = cv
        for t in [lr, lr * 0.5, lr * 0.1, lr * 0.05]:
            ht = project_feasible(h - t * gd, n)
            fv = compute_c5(ht, n)
            if fv < best_c5 - 1e-14:
                best_c5 = fv
                best_h = ht.copy()
            if fv < bv - 1e-15:
                bv = fv
                bh = ht
                break
        h = bh
        if i % n_explore == 0 and time.time() < t2_end - 2:
            h = project_feasible(hr + 0.008 * rs.standard_normal(n), n)
    if compute_c5(h, n) < best_c5:
        best_c5 = compute_c5(h, n)
        best_h = h.copy()
    return best_h, best_c5

def upsample_h(h, n_old, n_new):
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(xn, xo, h)
    return project_feasible(h_new, n_new)

def anneal_phase(h, n, t_max, rs, ng=8):
    t0 = time.time()
    h = h.copy()
    cur_c5 = compute_c5(h, n)
    for temp in [0.03, 0.02, 0.01, 0.005]:
        if time.time() - t0 > t_max * 0.8:
            break
        t_phase = t_max * 0.25
        h = gd_phase(h, n, 50, 2000, 0.05, 0.001, t_phase, rs, noise_interval=200)
        c5 = compute_c5(h, n)
        if c5 < cur_c5:
            cur_c5 = c5
    return h, cur_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rs = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_n = n
    best_c5 = compute_c5(h, n)
    if time.time() < dl - 400:
        t1 = min(120, dl - time.time() - 350)
        h_a, c5a = anneal_phase(h, n, t1, rs)
        if c5a < best_c5:
            best_c5 = c5a
            best_h = h_a.copy()
    if time.time() < dl - 200:
        t2 = min(250, dl - time.time() - 180)
        h_opt = gd_phase(best_h, best_n, 50, 5000, 0.05, 0.001, t2, rs, noise_interval=200)
        c5o = compute_c5(h_opt, best_n)
        if c5o < best_c5:
            best_c5 = c5o
            best_h = h_opt.copy()
    if time.time() < dl - 120:
        n_hi = min(best_n + 200, max(best_n + 100, 600))
        h_hi = upsample_h(best_h, best_n, n_hi)
        t3 = min(120, dl - time.time() - 60)
        if t3 > 20:
            h_hi = gd_phase(h_hi, n_hi, 50, 5000, 0.04, 0.001, t3, rs, noise_interval=200)
            c5h = compute_c5(h_hi, n_hi)
            if c5h < best_c5:
                best_c5 = c5h
                best_h = h_hi.copy()
                best_n = n_hi
    if time.time() < dl - 20:
        t4 = dl - time.time() - 5
        h_pol, c5_pol = gd_exact_polish(best_h, best_n, 300, t4, rs)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
