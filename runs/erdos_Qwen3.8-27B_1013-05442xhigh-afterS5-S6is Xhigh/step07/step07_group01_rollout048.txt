def smooth_subgrad_descent(h, n, deadline, alpha_init=100, alpha_final=100000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = dx * 5.0
    momentum = np.zeros(n)
    mu = 0.9
    alpha = alpha_init
    
    for step in range(1000000):
        if time.time() > deadline:
            break
        
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full') * dx
        
        # Compute weights (softmax)
        cmax = corr.max()
        sh = alpha * (corr - cmax)
        p = np.exp(sh)
        psum = p.sum()
        if psum < 1e-30:
            p = np.zeros(len(correlation)); p[np.argmax(corr)] = 1
        else:
            p /= psum
        
        # Compute gradient as weighted sum of lag gradients
        grad = np.zeros(n)
        # Use vectorized computation for all lags
        for k in range(2*n-1):
            w = p[k]
            if w < 1e-12:
                continue
            l = k - (n-1)
            # Add weighted gradient
            if l >= 0:
                m = n - l
                if m > 0:
                    grad[l:] += w * (1 - h[:m])
                    grad[:m] -= w * h[l:]
            else:
                ll = -l
                m = n - ll
                if m > 0:
                    grad[:m] += w * (1 - h[ll:])
                    grad[ll:] -= w * h[:m]
        grad *= dx
        
        # Gradient step with momentum
        momentum = mu * momentum - lr * grad
        h_new = project_feasible(h + momentum, n)
        
        # Evaluate
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h_new.copy()
            lr = min(lr * 1.01, dx * 50)
        else:
            # Backtrack
            lr *= 0.95
            h_new = h  # keep old
        
        h = h_new
        
        # Anneal alpha
        alpha *= 1.001
    
    return best_h, best_c

For s ≥ 0: n ranges from 0 to n-1-s.
