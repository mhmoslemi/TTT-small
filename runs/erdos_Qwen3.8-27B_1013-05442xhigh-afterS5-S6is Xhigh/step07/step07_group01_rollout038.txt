def slsqp_objective(x, n, alpha):
   g = np.correlate(x, 1.0-x, mode='full') * (2.0/n)
   m = g.max()

def lse_gradient(x,n,alpha):
    g = corr_values(x,n)
    m = g.max()
    sh = alpha * (g - m)
    # sh max maybe 0; sm = np.max(sh) for shift
    sm = np.max(sh)
    p = np.exp(sh - sm)
    s = p.sum()
    if s <= 0: p = np.full(g.shape, 1.0/g.size)
    else: p /= s
    # np order is decreasing lag; convert to increasing lag array pidx
    pidx = p[::-1].copy()
    dx = 2.0/n
    # S = sliding sum of pidx length n
    prefix = np.empty(g.size+1); prefix[0]=0.0; np.cumsum(pidx, out=prefix[1:])
    starts = (n-1) - np.arange(n)
    ends = starts + n
    S = prefix[ends] - prefix[starts]
    # Rplus = convolve(x, pidx[::-1]) slice
    rp_full = np.convolve(x, pidx[::-1], mode='full')
    Rplus = rp_full[n-1:n-1+n]
    rm_full = np.convolve(x, pidx, mode='full')
    Rminus = rm_full[n-1:n-1+n]
    return dx * (S - Rplus - Rminus)

def lse_gradient(x):
    n = int(_POLY_N); alpha=float(_POLY_ALPHA)
    g = corr_values(x,n)
    m = float(np.max(g))
    sh = alpha*(g-m)
    sm = float(np.max(sh))
    p = np.exp(sh - sm)
    s = float(p.sum())
    if s <= 0.0:
       p = np.full(g.shape, 1.0/g.size, dtype=float)
    else:
       p /= s
    pidx = p[::-1].copy()
    dx = 2.0 / n
    prefix = np.empty(g.size + 1, dtype=float)
    prefix[0]=0.0
    np.cumsum(pidx, out=prefix[1:])
    starts = (n-1) - np.arange(n)
    ends = starts + n
    S = prefix[ends] - prefix[starts]
    pidx_rev = pidx[::-1].copy()
    Rplus = np.convolve(x, pidx_rev, mode='full')[n-1:n-1+n]
    Rminus = np.convolve(x, pidx, mode='full')[n-1:n-1+n]
    return dx * (S - Rplus - Rminus)

    pidx = p[::-1].copy()
    dx = 2.0 / n

    prefix = np.empty(g.size + 1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(pidx, out=prefix[1:])
    starts = (n - 1) - np.arange(n)
    ends = starts + n
    S = prefix[ends] - prefix[starts]

    pidx_rev = pidx[::-1].copy()
    Rplus = np.convolve(x, pidx_rev, mode="full")[n - 1:n - 1 + n]
    Rminus = np.convolve(x, pidx, mode="full")[n - 1:n - 1 + n]
    return dx * (S - Rplus - Rminus)


def sum_constraint(x):
    return float(np.sum(x) - 0.5 * int(_POLY_N))


def polish_h(x, n, time_limit):
    x = project_feasible(np.asarray(x, dtype=float), n)
    best_x = x.copy()
    best_c = compute_c5_numpy(best_x)

    bounds = [(0.0, 1.0)] * int(n)
    cons = [{"type": "eq", "fun": sum_constraint}]
    global _POLY_N, _POLY_ALPHA
    _POLY_N = int(n)

    for alpha, maxit in ((50.0, 100), (150.0, 180), (600.0, 260), (2000.0, 420)):
        if time.time() > time_limit - 20:
            break
        _POLY_ALPHA = float(alpha)
        try:
            res = minimize(
                lse_objective,
                x,
                method="SLSQP",
                jac=lse_gradient,
                bounds=bounds,
                constraints=cons,
                options={"maxiter": int(maxit), "ftol": 1e-11},
            )
            nx = project_feasible(res.x, n)
            if np.all(np.isfinite(nx)):
                nc = compute_c5_numpy(nx)
                if nc < best_c:
                    best_c = nc
                    best_x = nx.copy()
                    x = nx
        except Exception:
            pass

    return best_x, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    deadline = t0 + budget - 5.0
    rng = np.random.RandomState(int(seed))

    best_h = None
    best_c = float("inf")
    best_n = 0

    try:
        h_init = np.asarray(initial_h_values, dtype=float)
        n_init = int(len(h_init))
    except Exception:
        h_init = None
        n_init = 0

    if h_init is not None and n_init > 0:
        h_init = project_feasible(h_init, n_init)
        c_init = compute_c5_numpy(h_init)
        best_h = h_init.copy()
        best_c = c_init
        best_n = n_init
    else:
        best_n = 400
        best_h = random_binary(best_n, rng).astype(float)
        best_c = compute_c5_numpy(best_h)

    target_base = 0.3808
    reserve_cont = min(240.0, 0.24 * budget)
    bin_end = min(deadline - reserve_cont, t0 + budget - 25.0)

    def use_candidate(hb, cb, nb):
        return cb < best_c - 1e-12

    if bin_end > time.time() + 10.0:
        t400 = min(t0 + 0.25 * budget, bin_end)
        if t400 > time.time() + 6.0:
            hb = binary_from_h(best_h, 400, rng)
            hb, cb = binary_search(hb, 400, rng, t400, min(target_base, best_c - 1e-10))
            if cb < best_c - 1e-12:
                best_c = cb
                best_h = np.asarray(hb, dtype=float)
                best_n = 400

        t800 = min(bin_end, t0 + 0.58 * budget)
        if t800 > time.time() + 25.0:
            hb = binary_from_h(best_h, 800, rng)
            hb, cb = binary_search(hb, 800, rng, t800, min(target_base, best_c - 1e-10))
            if cb < best_c - 1e-12:
                best_c = cb
                best_h = np.asarray(hb, dtype=float)
                best_n = 800

        if time.time() < bin_end - 6.0:
            nb = best_n if best_n in (400, 800, 1000) else 400
            if nb < 200 or nb > 1000:
                nb = 400
            hb = binary_from_h(best_h, nb, rng)
            hb, cb = binary_search(hb, nb, rng, bin_end, min(target_base, best_c - 1e-10))
            if cb < best_c - 1e-12:
                best_c = cb
                best_h = np.asarray(hb, dtype=float)
                best_n = nb

    cont_end = deadline - 3.0
    if time.time() < cont_end - 35.0:
        hp = resample_h(best_h, 400)
        hp, cpol = polish_h(hp, 400, min(cont_end - 45.0, time.time() + 160.0))
        if cpol < best_c - 1e-12:
            best_c = cpol
            best_h = hp
            best_n = 400

        if time.time() < cont_end - 25.0 and best_n >= 600:
            hp = resample_h(best_h, 800)
            hp, cpol = polish_h(hp, 800, cont_end - 10.0)
            if cpol < best_c - 1e-12:
                best_c = cpol
                best_h = hp
                best_n = 800

    final = project_feasible(np.asarray(best_h, dtype=float), int(best_n))
    final_c = compute_c5_numpy(final)
    return final, float(final_c), int(best_n)
