import time
import math
import numpy as np
import scipy.optimize as so

LS = {'n': 1, 'dx': 1.0, 'beta': 1.0}

def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / int(n)))

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(3):
        if abs(diff) <= 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y

def softmax_values(c, beta):
    maxc = float(np.max(c))
    z = beta * (c - maxc)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)
    ws = float(ew.sum())
    f = maxc + math.log(ws) / beta
    if ws <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ew / ws
    return f, w

def smooth_grad_from_w(h, w, n, dx):
    mid = n - 1
    w_pos = np.zeros(n)
    w_neg = np.zeros(n)
    w_pos[1:] = w[mid + 1:]
    w_neg[1:] = w[:mid][::-1]
    s = w_pos + w_neg
    pref_p = np.cumsum(w_pos)
    pref_n = np.cumsum(w_neg)
    idx = np.arange(n)
    consts = pref_p + pref_n[n - 1 - idx]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx

def l_fg(x):
    st = LS
    n = int(st['n'])
    dx = float(st['dx'])
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, w = softmax_values(c, float(st['beta']))
    g = smooth_grad_from_w(x, w, n, dx)
    return float(f), g

def frac_polish(h, levels, deadline):
    n = int(h.size)
    LS['n'] = n
    LS['dx'] = 2.0 / n
    cur = project_box_sum(h, 0.5 * n)
    for beta, mx in levels:
        if time.time() > deadline - 2.0:
            break
        LS['beta'] = float(beta)
        mx_eff = int(min(mx, max(8, (deadline - time.time()) / 0.3)))
        bounds = [(0.0, 1.0)] * n
        try:
            res = so.minimize(l_fg, cur, method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': mx_eff, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxls': 40, 'disp': False})
            cur = project_box_sum(res.x, 0.5 * n)
        except Exception:
            cur = project_box_sum(cur, 0.5 * n)
    return cur

def to_binary(v, n):
    n = int(n)
    v = np.asarray(v, dtype=float)
    order = np.argsort(-v, kind='stable')[:n // 2]
    s = np.zeros(n, dtype=np.int8)
    s[order] = 1
    return s

def sturmian_seed(g):
    i = np.arange(g, dtype=np.float64)
    t = np.remainder(i * 0.618033988749895, 1.0)
    return to_binary((t < 0.809016994374947).astype(float), g)

def resample_v(v, n, g):
    return np.interp(np.linspace(0.0, 2.0, int(g)), np.linspace(0.0, 2.0, int(n)), np.asarray(v, dtype=float))

def combo_move(c, s, a, b, m):
    n = m + 1
    cn = c.copy()
    cn[m - a:m - a + n] -= (1 - s)
    sn = s.copy()
    sn[a] = 0
    sn[b] = 1
    cn[m - b:m - b + n] += (1 - sn)
    cn[a:a + n] += sn[::-1]
    cn[b:b + n] -= s[::-1]
    cn[m + b - a] += 1
    cn[m + a - b] -= 1
    return cn, sn

import time
import math
import numpy as np
import scipy.optimize as so

LS = {'n': 1, 'dx': 1.0, 'beta': 1.0}


def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / int(n)))


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(3):
        if abs(diff) <= 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y


def softmax_values(c, beta):
    maxc = float(np.max(c))
    z = beta * (c - maxc)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)
    ws = float(ew.sum())
    if ws <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return maxc, w
    return float(maxc + math.log(ws) / beta), w / ws


def smooth_grad_from_w(h, w, n, dx):
    mid = n - 1
    w_pos = np.zeros(n)
    w_neg = np.zeros(n)
    w_pos[1:] = w[mid + 1:]
    w_neg[1:] = w[:mid][::-1]
    s = w_pos + w_neg
    pref_p = np.cumsum(w_pos)
    pref_n = np.cumsum(w_neg)
    idx = np.arange(n)
    consts = pref_p + pref_n[n - 1 - idx]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx


def l_fg(x):
    st = LS
    n = int(st['n'])
    dx = float(st['dx'])
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, w = softmax_values(c, float(st['beta']))
    return float(f), smooth_grad_from_w(x, w, n, dx)


def frac_polish(h, levels, deadline):
    n = int(h.size)
    LS['n'] = n
    LS['dx'] = 2.0 / n
    cur = project_box_sum(np.asarray(h, dtype=float), 0.5 * n)
    for beta, mx in levels:
        if time.time() > deadline - 2.0:
            break
        LS['beta'] = float(beta)
        maxit = int(min(mx, max(8, (deadline - time.time()) / 0.25)))
        bounds = [(0.0, 1.0)] * n
        try:
            res = so.minimize(l_fg, cur, method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxls': 40, 'disp': False})
            cur = project_box_sum(res.x, 0.5 * n)
        except Exception:
            cur = project_box_sum(cur, 0.5 * n)
    return cur


def to_binary(v, g):
    order = np.argsort(-np.asarray(v, dtype=float), kind='stable')[:max(0, g // 2)]
    s = np.zeros(int(g), dtype=np.int8)
    s[order] = 1
    return s


def sturmian_seed(g):
    i = np.arange(g, dtype=np.float64)
    t = np.remainder(i * 0.618033988749895, 1.0)
    return to_binary((t < 0.809016994374947).astype(float), g)


def resample_v(v, n, g):
    return np.interp(np.linspace(0.0, 2.0, int(g)), np.linspace(0.0, 2.0, int(n)), np.asarray(v, dtype=float))


def combo_move(c, s, a, b, m):
    n = m + 1
    cn = c.copy()
    cn[m - a:m - a + n] -= (1 - s)
    sn = s.copy()
    sn[a] = 0
    sn[b] = 1
    cn[m - b:m - b + n] += (1 - sn)
    cn[a:a + n] += sn[::-1]
    cn[b:b + n] -= s[::-1]
    cn[m + b - a] += 1
    cn[m + a - b] -= 1
    return cn, sn


def combo_search(s, n, t_end, rng, ecap):

    s = np.asarray(s).astype(np.int8).copy()

    f2 = int(np.sum(c >= f1 - 2))
    best = s.copy()
    best_f1 = f1
    best_f2 = f2
    act = np.zeros(2 * n - 1, dtype=np.float64)
    evals = 0
    stuck = 0
    boots = 0
    while time.time() < t_end and evals < ecap:
        W = max(3, f1 // 8)
        act = 0.95 * act + (c == f1).astype(np.float64) * 0.05
        w = np.maximum(0.0, c.astype(np.float64) - (f1 - W)) + act + 1e-6
        w32 = np.clip(w, 0.0, 4000.0).astype(np.int32)

import time

import numpy as np
import scipy.optimize as so

LS = {'n': 1, 'dx': 1.0, 'beta': 1.0}


def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / int(n)))


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(3):
        if abs(diff) <= 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y


def softmax_values(c, beta):
    maxc = float(np.max(c))
    z = beta * (c - maxc)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)
    ws = float(ew.sum())
    if ws <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return maxc, w
    return float(maxc + math.log(ws) / beta), w / ws


def smooth_grad_from_w(h, w, n, dx):
    mid = n - 1
    wp = np.zeros(n)
    wn = np.zeros(n)
    wp[1:] = w[mid + 1:]
    wn[1:] = w[:mid][::-1]
    s = wp + wn
    pp = np.cumsum(wp)
    pn = np.cumsum(wn)
    idx = np.arange(n)
    consts = pp + pn[n - 1 - idx]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx


def l_fg(x):
    st = LS
    n = int(st['n'])
    dx = float(st['dx'])
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, w = softmax_values(c, float(st['beta']))
    return float(f), smooth_grad_from_w(x, w, n, dx)


def frac_polish(h, levels, deadline):
    n = int(h.size)
    LS['n'] = n
    LS['dx'] = 2.0 / n
    cur = project_box_sum(np.asarray(h, dtype=float), 0.5 * n)
    for beta, mx in levels:
        if time.time() > deadline - 2.0:
            break
        LS['beta'] = float(beta)
        maxit = int(min(mx, max(8, (deadline - time.time()) / 0.25)))
        bounds = [(0.0, 1.0)] * n
        try:
            res = so.minimize(l_fg, cur, method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': maxit, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxls': 40, 'disp': False})
            cur = project_box_sum(res.x, 0.5 * n)
        except Exception:
            cur = project_box_sum(cur, 0.5 * n)
    return cur


def to_binary(v, g):
    order = np.argsort(-np.asarray(v, dtype=float), kind='stable')[:max(0, g // 2)]
    s = np.zeros(int(g), dtype=np.int8)
    s[order] = 1
    return s

def combo_move(c, s, a, b, m):

    cn = c.copy()
    cn[m - a:m - a + n] -= (1 - s)
    sn = s.copy()
    sn[a] = 0
    sn[b] = 1
    cn[m - b:m - b + n] += (1 - sn)
    cn[a:a + n] += sn[::-1]
    cn[b:b + n] -= s[::-1]
    cn[m + b - a] += 1
    cn[m + a - b] -= 1
    return cn, sn

            c = np.correlate(1 - s, s, mode='full').astype(np.int32)
            f1 = best_f1
            f2 = best_f2
            stuck = 0
            boots += 1
            if boots % 3 == 0:
                for _ in range(3):
                    x1 = int(rng.randint(n))
                    x2 = int(rng.randint(n))
                    if x1 == x2 or s[x1] == s[x2]:
                        continue
                    c, s = combo_move(c, s, x1, x2, m)
                f1 = int(c.max())
                f2 = int(np.sum(c >= f1 - 2))
                if f1 < best_f1 or (f1 == best_f1 and f2 < best_f2):
                    best = s.copy()
                    best_f1 = f1
                    best_f2 = f2
    return best, best_f1


def pair_obj(delta, h, i, j, n):
    a = h[i] + delta
    b = h[j] - delta
    if a < -1e-12 or a > 1.0 + 1e-12 or b < -1e-12 or b > 1.0 + 1e-12:
        return 1e100
    hh = h.copy()
    hh[i] = a
    hh[j] = b
    return erdos_c5(hh, n)


def pair_line_pair(h, n, i, j, cur, dline):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-13:
        return h, cur
    bd = 0.0
    bc = cur
    c0 = pair_obj(lo, h, i, j, n)
    if c0 < bc:
        bc, bd = c0, lo
    c1 = pair_obj(hi, h, i, j, n)
    if c1 < bc:
        bc, bd = c1, hi
    try:
        if time.time() < dline - 0.3:
            rs = so.minimize_scalar(pair_obj, args=(h, i, j, n), bounds=(lo, hi),
                                    method='bounded', options={'maxiter': 25, 'xatol': 1e-10})
            c2 = pair_obj(float(rs.x), h, i, j, n)
            if c2 < bc:
                bc, bd = c2, float(rs.x)
    except Exception:
        pass
    if bc < cur - 1e-14:
        hh = h.copy()
        hh[i] += bd
        hh[j] -= bd
        return np.clip(hh, 0.0, 1.0), bc
    return h, cur


def pair_polish(h, n, dline, seed):
    rng = np.random.RandomState(int(seed))
    h = np.asarray(h, dtype=float)
    cur = erdos_c5(h, n)
    msel = 16
    fr = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
    if fr.size:
        order = fr[np.argsort(np.minimum(h[fr], 1.0 - h[fr]))[::-1]][:msel]
    else:
        order = np.zeros(0, dtype=int)
    need = max(0, msel - order.size)
    top = order if need == 0 else np.unique(
        np.concatenate((order, rng.choice(n, size=need, replace=False))))
    if top.size < 2:
        return h, cur
    for _ in range(2):
        imp = False
        for a in range(top.size):
            for b2 in range(a + 1, top.size):
                if time.time() > dline - 3.0:
                    return h, erdos_c5(h, n)
                hh, cc = pair_line_pair(h, n, int(top[a]), int(top[b2]), cur, dline)
                if cc < cur - 1e-14:
                    h, cur, imp = hh, cc, True
        if not imp:
            break
    return h, cur


def roll_best(v, n):
    v = np.asarray(v, dtype=float)
    bh = v.copy()
    bc = erdos_c5(v, n)
    for r in range(1, n):
        cc = erdos_c5(np.roll(v, -r), n)
        if cc < bc - 1e-15:
            bc = cc
            bh = np.roll(v, -r)
    return bh, bc


def final_fix(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    d = float(n * 0.5 - h.sum())
    if abs(d) > 1e-12:
        fr = (h > 1e-9) & (h < 1.0 - 1e-9)
        k = int(fr.sum())
        if k > 0:
            h[fr] += d / k
        else:
            h += d / n
        h = np.clip(h, 0.0, 1.0)
    rem = float(n * 0.5 - h.sum())
    i = 0
    while abs(rem) > 1e-15 and i < n:
        if rem > 0.0 and h[i] < 1.0 - 1e-12:
            add = min(1.0 - h[i], rem)
            h[i] += add
            rem -= add
        elif rem < 0.0 and h[i] > 1e-12:
            sub = min(h[i], -rem)
            h[i] -= sub
            rem += sub
        i += 1
    return h


def pool_add(pool, cc, v, g):
    key = (g, int(round(cc * g / 2.0)))
    for ite in range(len(pool)):
        if (pool[ite][2], pool[ite][3]) == key:
            return
    if len(pool) < 6:
        pool.append((cc, v.copy(), g, key[1]))


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(30.0, float(budget_s) - 12.0)
    dl = t0 + total
    rng = np.random.RandomState(int(seed))
    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(400, 0.5, dtype=float)
    n0 = int(h0.size)
    if n0 > 700:
        h0 = np.interp(np.linspace(0.0, 2.0, 700), np.linspace(0.0, 2.0, n0), h0)
        n0 = 700
    if n0 < 100:
        h0 = np.full(400, 0.5, dtype=float)
        n0 = 400
    n0 = n0 - (n0 % 2)
    h0 = project_box_sum(h0, 0.5 * n0)
    n = n0
    bn = n
    bh = h0
    bc = erdos_c5(bh, bn)
    end1 = t0 + 0.07 * total
    p1lev = [(2.0e2, 30), (1.0e3, 35), (3.0e3, 40), (1.0e4, 40),
             (3.0e4, 30), (1.0e5, 22), (4.0e5, 14), (1.5e6, 8)]
    starts = [bh, to_binary(bh, bn).astype(float),
              project_box_sum(bh + 0.5 * rng.randn(bn), 0.5 * bn)]
    for st in starts:
        if time.time() > end1 - 15 or time.time() > dl - 90:
            break
        hh = frac_polish(st, p1lev, min(end1, dl - 60))
        cc = erdos_c5(hh, bn)
        if cc < bc - 1e-12:
            bc, bh = cc, hh
    pool = []
    pool_add(pool, bc, bh, bn)
    end2 = t0 + 0.80 * total
    gl = [bn]
    for g in (bn + 20, bn + 50, bn + 80, 500, 550):
        if 200 <= g <= 700 and g % 2 == 0 and g != bn and g not in gl:
            gl.append(g)
    for rr in range(4):
        for g in gl:
            if time.time() > end2 - 30:
                break
            vg = bh if g == bn else resample_v(bh, bn, g)
            if rr == 0:
                s = to_binary(vg, g)
            elif rr == 1:
                s = sturmian_seed(g)
            elif rr == 2:
                s = to_binary((np.random.RandomState(int(seed) + 101 * g).rand(g)
                               < 0.5).astype(float), g)
            else:
                s = to_binary(0.5 * vg + 0.5 * sturmian_seed(g).astype(float), g)
            extra = max(25.0, 0.045 * total) if rr < 2 else 26.0
            t_end = min(end2 - 10.0, time.time() + extra)
            ecap = int(2.6e6 * (400.0 / g))
            sb, f1b = combo_search(s, g, t_end, rng, ecap)
            if time.time() > dl - 60:
                break
            v = sb.astype(float)
            cc = erdos_c5(v, g)
            if cc < bc - 1e-12:
                bc, bh, bn = cc, v, g
                pool = []
            pool_add(pool, cc, v, g)
    end3 = dl - max(25.0, 0.025 * total)
    cand = [(bc, bh, bn)]
    for ite in range(min(len(pool), 8)):
        item = pool[ite]
        dup = False
        for cje in range(len(cand)):
            if cand[cje][2] == item[2]:
                dup = True
                break
        if dup:
            continue
        if len(cand) >= 4 or time.time() > end3 - 25:
            break
        cand.append((item[0], item[1], item[2]))
    for cje in range(len(cand)):
        cc0, v0, g0 = cand[cje]
        if time.time() > end3 - 20:
            break
        v0, cc0 = roll_best(v0, g0)
        if cc0 < bc - 1e-12:
            bc, bh, bn = cc0, v0, g0
            cand[cje] = (cc0, v0, g0)
        if time.time() > end3 - 15:
            break
        tfin = min(end3 - 12.0, time.time() + max(10.0, 0.03 * total))
        v2 = frac_polish(v0, [(5.0e3, 16), (2.5e4, 14), (1.2e5, 10), (5.0e5, 6)], tfin)
        v2, cc2 = roll_best(v2, g0)
        if cc2 < bc - 1e-12:
            bc, bh, bn = cc2, v2, g0
    if time.time() < min(end3 - 8.0, dl - 30.0):
        vx = project_box_sum(bh, 0.5 * bn)
        vx, ccx = pair_polish(vx, bn, min(dl - 3.0, end3 - 6.0), int(seed) + 5)
        if ccx < bc - 1e-12:
            bc, bh = ccx, vx
    if time.time() < dl - 6.0:
        bh, bc = roll_best(bh, bn)
    bh = final_fix(bh, bn)
    bc = float(erdos_c5(bh, bn))
    return bh, bc, int(bn)
