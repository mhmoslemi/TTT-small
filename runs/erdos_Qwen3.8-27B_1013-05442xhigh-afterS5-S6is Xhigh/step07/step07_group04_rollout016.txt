        for x in xs2:
            d = x - x0
            if abs(d) < 1e-14:
                continue
            delta = np.zeros(2*n-1)
            for idx, dv in ((i, d), (j, -d)):
                lo1 = n - 1 - idx
                hi1 = 2*(n-1) - idx
                delta[lo1:hi1+1] += dv * (2.0/n) * (1.0 - h)
                lo2 = idx
                hi2 = idx + n - 1
                delta[lo2:hi2+1] -= dv * (2.0/n) * h[::-1]
            c_new = c_base + delta
            v = c_new.max()
            if v < best_val:
                best_val = v
                best_x = x
        
        return best_x, S - best_x, best_val
    
    return None

Actually, let me reconsider the L-BFGS-B gradient. The sm_grad function loops over active shifts, which for high alpha might be only a few. But for low alpha (like alpha=5), many shifts have non-negligible weight, and the loop could be slow.

For low alpha, the gradient is approximately the mean of all gradients, which is:
g ≈ dx * sum_m (1-h[m-n+1+k] ...) ... this gets complicated.

Actually, let me just use a different approach for the low-alpha phases: use the sum of squares as the objective instead of the soft max.

sum_k c[k]^2 is smooth and its gradient is 2 * sum_k c[k] * (∂c[k]/∂h)

This is equivalent to minimizing the L2 norm of c, which is a different objective but might work well as a warmup.

Or I could just use a fixed number of top shifts (like top 50) with equal weight, regardless of alpha. That's simpler and faster.

def topk_grad(x, n, k):

    idx = np.argpartition(c, -k)[-k:]
    # Gradient: sum of gradients of active shifts
    dx = 2.0 / n
    off = n - 1
    g = np.zeros(n)
    for idx_m in idx:
        km = idx_m - off
        if km >= 0:
            m2 = n - km
            g[km:] += dx * (1.0 - x[:m2])
            g[:m2] -= dx * x[km:]
        else:
            k2 = -km
            m2 = n - k2
            g[:m2] += dx * (1.0 - x[k2:])
            g[k2:] -= dx * x[:m2]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1)).copy()
    tgt = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(50):
        d = tgt - v.sum()
        if abs(d) < 1e-13:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        if m.sum() > 0:
            v[m] += d / int(m.sum())
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def topk_obj(x, n, k):
    c = compute_c(x, n)
    idx = np.argpartition(c, -k)[-k:]
    return float(c[idx].sum())

def topk_grad(x, n, k):
    c = compute_c(x, n)
    idx = np.argpartition(c, -k)[-k:]
    dx = 2.0 / n
    off = n - 1
    g = np.zeros(n)
    for im in idx:
        km = im - off
        if km >= 0:
            m2 = n - km
            g[km:] += dx * (1.0 - x[:m2])
            g[:m2] -= dx * x[km:]
        else:
            k2 = -km
            m2 = n - k2
            g[:m2] += dx * (1.0 - x[k2:])
            g[k2:] -= dx * x[:m2]
    return g

def sm_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m)
    return m + float(np.log(np.exp(sh).sum())) / alpha

def sm_grad(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    sh = alpha * (c - m) - alpha * (c - m).max()
    e = np.exp(sh)
    s = float(e.sum())
    if s < 1e-300:
        return np.zeros(n)
    w = e / s
    dx = 2.0 / n
    off = n - 1
    g = np.zeros(n)
    active = np.flatnonzero(w > 1e-12)
    for im in active:
        wi = w[im]
        km = im - off
        if km >= 0:
            m2 = n - km
            g[km:] += wi * dx * (1.0 - x[:m2])
            g[:m2] -= wi * dx * x[km:]
        else:
            k2 = -km
            m2 = n - k2
            g[:m2] += wi * dx * (1.0 - x[k2:])
            g[k2:] -= wi * dx * x[:m2]
    return g

def pen_obj(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return sm_obj(x, n, alpha) + 0.5 * mu * d * d

def pen_grad(x, n, alpha, mu):
    d = float(x.sum() - 0.5 * n)
    return sm_grad(x, n, alpha) + mu * d

def topk_pen_obj(x, n, k, mu):
    d = float(x.sum() - 0.5 * n)
    return topk_obj(x, n, k) + 0.5 * mu * d * d

def topk_pen_grad(x, n, k, mu):
    d = float(x.sum() - 0.5 * n)
    return topk_grad(x, n, k) + mu * np.ones(n) * d

def lbfgsb_topk(h, n, k, mu, maxit):

                       jac=topk_pen_grad, bounds=bounds,
                       options={"maxiter": maxit, "maxfun": 3 * maxit, "ftol": 1e-16, "gtol": 1e-10, "maxls": 50})
        if hasattr(res, "x") and res.x is not None:

        pass
    return h

    L = 2 * n - 1
    term = np.zeros(L)
    lo1 = n - 1 - i; hi1 = 2 * (n - 1) - i

    lo2j = j; hi2j = j + n - 1
    term[lo2j:hi2j + 1] += dx * h[::-1]

def smart_pair_search(h, c, n, deadline, seed):
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    bestc = float(c.max())
    besth = h.copy()
    off = n - 1
    rng = np.random.RandomState(seed)
    noimp = 0
    maxno = 1500
    temp = 0.01
    while time.time() < deadline and noimp < maxno:
        # Select active shifts (top correlations)
        top_k = min(30, n - 1)
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        # Weight pair selection by contribution to top shifts
        active_shifts = (top_idx - off)
        active_shifts = active_shifts[active_shifts != 0]
        if len(active_shifts) == 0:
            break
        # Each pair at shift d contributes h[i]*(1-h[i+d]) to c[off+d]
        # Pick a shift, then a position
        sw = c[top_idx]
        sw = sw / sw.sum()
        sm_shift = int(rng.choice(active_shifts, p=sw / sw.sum() * len(active_shifts)))
        if sm_shift > 0:
            d = sm_shift
        else:
            d = -sm_shift
        i = int(rng.randint(0, n - d))
        j = i + d
        res = pair_search_vec(h, c, i, j, n)
        if res is not None:
            xn, yn, new_max = res
            h[i] = xn; h[j] = yn
            c = compute_c(h, n)
            v = float(c.max())

            else:

                if rng.random() < np.exp(-(v - bestc) / temp):
                    noimp = 0
                else:
                    h[i] = besth[i]; h[j] = besth[j]
                    c = compute_c(h, n)
                    noimp += 1
        else:
            noimp += 1
        temp *= 0.995
    besth = project_box_sum(besth, n)
    bestc = float(compute_c(besth, n).max())
    return besth, bestc
