def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_gradients(h, n):
    G = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        j1s = max(0, -k)
        j1e = min(n, n - k)
        if j1e > j1s:
            G[j1s:j1e, m] = 1.0 - h[j1s + k:j1e + k]
        j2s = max(0, k)
        j2e = min(n, n + k)
        if j2e > j2s:
            G[j2s:j2e, m] -= h[j2s - k:j2e - k]
    return G

def lp_step_active(h, n, eps, M=50, solver='GLOP'):

    mc = float(C.max())
    # Get top M active shifts
    if M < 2*n-1:
        active = np.argsort(C)[-M:]
    else:
        active = np.arange(2*n-1)
    
    C_active = C[active]
    # Compute gradients for active shifts
    G_a = np.zeros((n, len(active)))
    for idx, m in enumerate(active):
        k = m - (n - 1)
        j1s = max(0, -k)
        j1e = min(n, n - k)
        if j1e > j1s:
            G_a[j1s:j1e, idx] = 1.0 - h[j1s + k:j1e + k]
        j2s = max(0, k)
        j2e = min(n, n + k)
        if j2e > j2s:
            G_a[j2s:j2e, idx] -= h[j2s - k:j2e - k]
    
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C_active + G_a.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in [solver, 'GLPK', 'ECOS']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.8, 0.6, 0.4, 0.2]:
        h_try = np.clip(h + lam * d, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-14:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

def paired_cd_step(h, n, j, l, n_bisect=40):
    """Do one paired coordinate descent step on indices j and l."""
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    # Compute d_k = ∂g_k/∂h[j] - ∂g_k/∂h[l] for all k
    shifts = np.arange(-(n-1), n)
    # ∂g_k/∂h[j]:
    # = (1-h[j+k]) if 0<=j+k<n, else 0
    #   - h[j-k] if 0<=j-k<n, else 0
    
    def grad_component(idx):
        # Returns vector of ∂g_k/∂h[idx] for all k
        # k ranges from -(n-1) to n-1
        # term1: (1-h[idx+k]) for valid idx+k
        ik = idx + shifts  # (2n-1,)
        mask1 = (ik >= 0) & (ik < n)
        t1 = np.where(mask1, 1.0 - h[np.clip(ik, 0, n-1)], 0.0)
        # term2: -h[idx-k] for valid idx-k
        ik2 = idx - shifts
        mask2 = (ik2 >= 0) & (ik2 < n)
        t2 = np.where(mask2, -h[np.clip(ik2, 0, n-1)], 0.0)
        return t1 + t2
    
    d = grad_component(j) - grad_component(l)  # (2n-1,)
    
    # Bisection for optimal delta
    lo = max(-h[j], h[l] - 1.0)
    hi = min(1.0 - h[j], h[l])
    
    if hi - lo < 1e-15:
        return h
    
    best_fun = np.inf
    best_delta = 0.0
    for _ in range(n_bisect):
        mid = (lo + hi) / 2.0
        f_mid = C + mid * d
        fmax = f_mid.max()
        if fmax < best_fun:
            best_fun = fmax
            best_delta = mid
        # Find subgradient sign
        active_mask = np.abs(f_mid - fmax) < 1e-10
        d_active = d[active_mask]
        if d_active.max() < -1e-12:
            lo = mid  # f is decreasing, go right
        elif d_active.min() > 1e-12:
            hi = mid  # f is increasing, go left
        else:
            break  # 0 in subdifferential, at minimum
    
    h_new = h.copy()
    h_new[j] += best_delta
    h_new[l] -= best_delta
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def paired_cd_phase(h, n, n_steps, rng, n_bisect=35):

    for step in range(n_steps):

        l = rng.randint(n)

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

I need to avoid recomputing the full correlation at each CD step. Instead, I should maintain the correlation vector incrementally.

When h[j] changes by δ and h[l] changes by -δ:
- The change in g_k = Σ_i h[i](1-h[i+k]) is:
  - From h[j] changing: (1-h[j+k])*δ - h[j-k]*δ (for valid indices)
  - From h[l] changing: -(1-h[l+k])*(-δ) + h[l-k]*(-δ) = (1-h[l+k])*δ*(-1) + h[l-k]*δ... 

Let me be more careful. g_k = Σ_i h[i](1-h[i+k])

If h[j] → h[j]+δ:
- Term i=j: (h[j]+δ)(1-h[j+k]) - h[j](1-h[j+k]) = δ(1-h[j+k])
- Term i=j-k (where i+k=j): h[j-k](1-(h[j]+δ)) - h[j-k](1-h[j]) = -δ*h[j-k]
- Total change from h[j]: δ(1-h[j+k]) - δ*h[j-k]

If h[l] → h[l]-δ:
- Term i=l: (h[l]-δ)(1-h[l+k]) - h[l](1-h[l+k]) = -δ(1-h[l+k])
- Term i=l-k: h[l-k](1-(h[l]-δ)) - h[l-k](1-h[l]) = δ*h[l-k]
- Total change from h[l]: -δ(1-h[l+k]) + δ*h[l-k]

Total change in g_k: δ[(1-h[j+k]) - h[j-k] - (1-h[l+k]) + h[l-k]]

So Δg_k = δ * [(1-h[j+k]) - h[j-k] - (1-h[l+k]) + h[l-k]]
         = δ * d_k

where d_k = (1-h[j+k])(valid) - h[j-k](valid) - (1-h[l+k])(valid) + h[l-k](valid)

This is what I had before. The key is that I can maintain the C vector incrementally:
C_new = C + δ * d

So I only need to compute d (O(n) with vectorized operations) and then update C in O(n). No need to recompute the full correlation!

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lp_step_active(h, n, eps, M):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    if M >= 2 * n - 1:
        active = np.arange(2 * n - 1)
    else:
        active = np.argsort(C)[-M:]
    C_active = C[active]
    G_a = np.zeros((n, len(active)))
    for idx, m in enumerate(active):
        k = m - (n - 1)
        j1s = max(0, -k)
        j1e = min(n, n - k)
        if j1e > j1s:
            G_a[j1s:j1e, idx] = 1.0 - h[j1s + k:j1e + k]
        j2s = max(0, k)
        j2e = min(n, n + k)
        if j2e > j2s:
            G_a[j2s:j2e, idx] -= h[j2s - k:j2e - k]
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C_active + G_a.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in ['GLOP', 'GLPK', 'ECOS']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.8, 0.6, 0.4, 0.2]:
        h_try = np.clip(h + lam * d, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-14:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

def cd_update(h, C, j, l, shifts, n):
    jk = j + shifts
    jmk = j - shifts
    lk = l + shifts
    lmk = l - shifts
    d = np.zeros(2 * n - 1)
    m1 = (jk >= 0) & (jk < n)
    d[m1] += 1.0 - h[jk[m1]]
    m2 = (jmk >= 0) & (jmk < n)
    d[m2] -= h[jmk[m2]]
    m3 = (lk >= 0) & (lk < n)
    d[m3] -= (1.0 - h[lk[m3]])
    m4 = (lmk >= 0) & (lmk < n)
    d[m4] += h[lmk[m4]]
    lo = max(-h[j], h[l] - 1.0)
    hi = min(1.0 - h[j], h[l])
    if hi - lo < 1e-15:
        return h, C
    best_delta = 0.0
    best_fun = np.inf
    for _ in range(35):
        mid = (lo + hi) * 0.5
        f = C + mid * d
        fmax = f.max()
        if fmax < best_fun:
            best_fun = fmax
            best_delta = mid
        active = np.abs(f - fmax) < 1e-10
        d_act = d[active]
        if d_act.max() < -1e-12:
            lo = mid
        elif d_act.min() > 1e-12:
            hi = mid
        else:
            break
    if abs(best_delta) > 1e-15:
        h[j] += best_delta
        h[l] -= best_delta
        C += best_delta * d
    return h, C

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n

    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
