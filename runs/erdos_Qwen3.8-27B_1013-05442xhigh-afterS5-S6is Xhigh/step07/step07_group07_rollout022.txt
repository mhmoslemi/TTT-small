def refine_with_slsqp(h, n, deadline, alphas=None, label=''):
    if alphas is None:
        alphas = [10000, 50000, 200000, 1000000, 5000000, 20000000]
    c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        maxiter = 2000 if alpha < 100000 else 3000
        h = lse_slsqp(h, n, alpha, maxiter, 1e-16)
        c = compute_c5(h, n)
    return h, c
