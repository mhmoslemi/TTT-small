def project_box_sum(x, target):

        else:

    wn[1:] = w[:mid][::-1]

import time
import numpy as np
import scipy.optimize as so

GS = {}

def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum() - target)) < 1e-14:
        return y
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if float(y.sum()) > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = float(target - y.sum())
    for _ in range(3):
        if abs(diff) <= 1e-13:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            y[mask] += diff / cnt
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
        diff = float(target - y.sum())
    return y

def smooth_obj(x):
    st = GS
    n, dx, beta = st['n'], st['dx'], st['beta']
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    maxc = float(np.max(c))
    z = beta * (c - maxc)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)
    ws = float(ew.sum())
    f = maxc + np.log(ws) / beta
    st['w'] = ew / ws
    st['cx'] = x.copy()
    return f

def smooth_jac(x):
    st = GS
    if st.get('cx') is not None and np.array_equal(x, st['cx']):
        w = st['w']
    else:
        n, dx, beta = st['n'], st['dx'], st['beta']
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        maxc = float(np.max(c))
        z = beta * (c - maxc)
        np.clip(z, -745.0, 0.0, out=z)
        ew = np.exp(z)
        w = ew / float(ew.sum())
    return _grad(x, w, st['n'], st['dx'])

def _grad(h, w, n, dx):
    mid = n - 1
    wp = np.zeros(n)
    wn = np.zeros(n)
    wp[1:] = w[mid+1:]
    wn[1:] = w[:mid][::-1]
    s = wp + wn
    pp = np.cumsum(wp)
    pn = np.cumsum(wn)
    idx = np.arange(n)
    consts = pp + pn[n-1-idx]
    kern = np.zeros(2*n-1)
    kern[mid+1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid+n]
    return (w[mid]*(1.0-2.0*h) + consts - neigh) * dx

def lbfgs_refine(h, beta, maxiter, deadline):
    n = len(h)
    rem = deadline - time.time()
    if rem < 3.0:
        return h, erdos_c5(h, n)
    GS.update(n=n, dx=2.0/n, beta=float(beta))
    mx = int(min(maxiter, max(10, int(rem/2.5))))
    bounds = [(0.0, 1.0)]*n
    try:
        res = so.minimize(smooth_obj, h, method='L-BFGS-B', jac=smooth_jac,
                          bounds=bounds, options={'maxiter':mx, 'ftol':1e-18,
                          'gtol':1e-14, 'maxls':40, 'disp':False})
        cand = project_box_sum(res.x, 0.5*n)
    except Exception:
        cand = project_box_sum(h, 0.5*n)
    return cand, erdos_c5(cand, n)

def slp_refine(h, n, deadline, max_iter=80):
    """Sequential Linear Programming: directly solve LP relaxation of min-max"""
    h = np.asarray(h, dtype=float).copy()
    cur = erdos_c5(h, n)
    dx = 2.0 / n
    top_count = min(35, 2*n-1)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0-h, mode='full') * dx
        top_idx = np.argsort(c)[::-1][:top_count]
        grads = np.zeros((top_count, n))
        for i, k in enumerate(top_idx):
            lag = int(k) - (n-1)
            for j in range(n):
                d = 0.0
                if 0 <= j - lag < n:
                    d += 1.0 - h[j-lag]
                if 0 <= j + lag < n:
                    d -= h[j+lag]
                grads[i, j] = d * dx
        cv = c[top_idx]
        A = np.hstack([grads, -np.ones((top_count, 1))])
        b = -cv
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        c_obj = np.zeros(n+1)
        c_obj[n] = 1.0
        bounds = [(-h[j], 1.0-h[j]) for j in range(n)] + [(cur - 1.0, cur)]
        try:
            res = so.linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs')
        except Exception:
            break
        if not res.success:
            break
        delta = res.x[:n]
        new_c = A_eq[0, :n] @ delta
        if abs(new_c) > 1e-3:
            delta = delta / abs(new_c)
        h_new = project_box_sum(h + delta, 0.5*n)
        c_new = erdos_c5(h_new, n)
        if c_new < cur - 1e-12:
            h = h_new
            cur = c_new
        else:
            break
    return h, cur

def pair_line(h, n, i, j, cur, deadline):
    hi0, hj0 = h[i], h[j]
    lo = max(-hi0, hj0-1.0)
    hi = min(1.0-hi0, hj0)
    if hi - lo < 1e-12:
        return h, cur
    best_d, best_c = 0.0, cur
    def obj(d):
        hh = h.copy()
        hh[i], hh[j] = hi0+d, hj0-d
        return erdos_c5(hh, n)
    for d in (lo, hi, 0.5*(lo+hi)):
        c = obj(d)
        if c < best_c:
            best_c, best_d = c, d
    if time.time() < deadline:
        try:
            rs = so.minimize_scalar(obj, bounds=(lo,hi), method='bounded',
                                    options={'maxiter':30, 'xatol':1e-10})
            if rs.fun < best_c:
                best_c, best_d = rs.fun, float(rs.x)
        except Exception:
            pass
    if best_c < cur - 1e-14:
        hh = h.copy()
        hh[i], hh[j] = hi0+best_d, hj0-best_d
        return np.clip(hh, 0, 1), best_c
    return h, cur

def polish(h, n, deadline, seed):
    h = np.asarray(h, dtype=float).copy()
    cur = erdos_c5(h, n)
    rng = np.random.RandomState(int(seed))
    frac = np.where((h > 1e-9) & (h < 1.0-1e-9))[0]
    if frac.size >= 4:
        score = np.min(h[frac], 1.0-h[frac])
        order = frac[np.argsort(score)[::-1]]
        cands = order[:min(50, len(order))]
    else:
        cands = np.arange(n)
    for rnd in range(5):
        if time.time() > deadline - 5:
            break
        improved = False
        cidx = cands.copy()
        rng.shuffle(cidx)
        for a in range(len(cidx)):
            for b in range(a+1, len(cidx)):
                if time.time() > deadline - 5:
                    return h, erdos_c5(h, n)
                i, j = int(cidx[a]), int(cidx[b])
                hh, cc = pair_line(h, n, i, j, cur, deadline-5.0)
                if cc < cur - 1e-14:
                    h, cur, improved = hh, cc, True
        if not improved:
            break
    return h, cur

def resample(h, n, nh):
    x = np.linspace(0, 2, n)
    y = np.linspace(0, 2, nh)
    return project_box_sum(np.interp(y, x, h), 0.5*nh)
