    n = x.size
    y = np.clip(x, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) < 1e-14:
        return y
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - y.sum()
    for _ in range(5):
        if abs(diff) < 1e-14:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.sum() > 0:
            y[mask] += diff / mask.sum()
        else:
            y += diff / n
        y = np.clip(y, 0.0, 1.0)
    return y

def softmax_smooth_max(c, beta):
    maxc = float(np.max(c))
    z = np.clip(beta * (c - maxc), -745.0, 0.0)
    ew = np.exp(z)
    ws = float(ew.sum())
    if ws < 1e-300:
        return maxc, np.eye(1)[0] * 0  # shouldn't happen
    f = maxc + np.log(ws) / beta
    return f, ew / ws

def compute_grad(h, n, beta):
    dx = 2.0 / n
    mid = n - 1
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    _, w = softmax_smooth_max(c, beta)
    # Gradient computation (same as before)
    w_pos = np.zeros(n)
    w_neg = np.zeros(n)
    w_pos[1:] = w[mid + 1:]
    w_neg[1:] = w[:mid][::-1]
    s = w_pos + w_neg
    pref_p = np.cumsum(w_pos)
    pref_n = np.cumsum(w_neg)
    idx = np.arange(n)
    consts = pref_p + pref_n[n - 1 - idx]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    grad = (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx
    return grad

def lfg_objective(x):
    st = LS
    n = int(st['n'])
    dx = float(st['dx'])
    beta = float(st['beta'])
    target = float(st['target'])
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, w = softmax_smooth_max(c, beta)
    grad = compute_grad(x, n, beta)
    res = float(x.sum() - target)
    f += 0.5 * 1e5 * res * res
    grad = grad + 1e5 * res * np.ones(n)
    return float(f), grad

def sharpen(h, alpha):
    """Sharpen h toward 0/1 using sigmoid-like transform."""
    h = np.clip(h, 1e-10, 1 - 1e-10)
    h_new = h ** alpha / (h ** alpha + (1 - h) ** alpha)
    return project_box_sum(h_new, 0.5 * len(h))

def generate_binary(n, seed):

    rng = np.random.RandomState(seed)
    idx = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0

def generate_msequence_like(n, seed):

    rng = np.random.RandomState(seed)
    # Use a simple LCG-based bit generator
    h = np.zeros(n)
    state = seed
    for i in range(n):
        state = (state * 1103515245 + 12345) & 0x7fffffff
        h[i] = 1.0 if (state >> 16) & 1 else 0.0
    # Adjust to have exactly n/2 ones
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) > n // 2:
        excess = len(ones) - n // 2
        to_flip = rng.choice(ones, excess, replace=False)
        h[to_flip] = 0.0
    elif len(ones) < n // 2:
        deficit = n // 2 - len(ones)
        to_flip = rng.choice(zeros, deficit, replace=False)
        h[to_flip] = 1.0
    return h

def pair_polish(h, n, deadline, seed, max_rounds=5):
