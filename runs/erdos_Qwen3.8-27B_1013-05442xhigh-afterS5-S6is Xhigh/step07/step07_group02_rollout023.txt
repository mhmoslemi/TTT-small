import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h *= (target / s)
    return np.clip(h, 0.0, 1.0)

def build_G(h, n):
    G = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        if k >= 0:
            G[:n-k, m] = 1.0 - h[k:]
            G[k:, m] = G[k:, m] - h[:n-k]
        else:
            k2 = -k
            G[k2:, m] = 1.0 - h[:n-k2]
            G[:n-k2, m] = G[:n-k2, m] - h[k2:]
    return G

def lp_step(h, n, eps, solver='GLOP'):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = float(corr.max())
    G = build_G(h, n)
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        corr + G.T @ delta <= t,
        delta >= -eps * np.ones(n),
        delta <= eps * np.ones(n),
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    for sol in [solver, 'GLPK', 'ECOS']:
        try:
            prob.solve(solver=sol)
            if delta.value is not None:
                break
        except Exception:
            continue
    if delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.8, 0.5, 0.3, 0.1]:
        h_try = project_feasible(h + lam * d, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

def slsqp_polish(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def mom_subgrad(h, n, lr, mu, steps, t_limit):
    t0 = time.time()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for i in range(steps):
        if time.time() - t0 > t_limit:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        max_c = corr.max()
        thr = max_c - 0.02 * (max_c - corr.min())
        active = np.where(corr >= thr)[0] - (n - 1)
        g = np.zeros(n)
        for k in active:
            if k >= 0:
                g[:n-k] += 1.0 - h[k:]
                g[k:] -= h[:n-k]
            else:
                k2 = -k
                g[k2:] += 1.0 - h[:n-k2]
                g[:n-k2] -= h[k2:]
        g /= len(active)
        v = mu * v - lr * g
        h = project_feasible(h + v, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h *= (target / s)
    return np.clip(h, 0.0, 1.0)

def build_G_sel(h, n, idx):
    n_sel = len(idx)
    G = np.zeros((n, n_sel))
    for j, m in enumerate(idx):
        k = m - (n - 1)
        if k >= 0:
            if n - k > 0:
                G[:n-k, j] = 1.0 - h[k:]
            if k > 0:
                G[k:, j] -= h[:n-k]
        else:
            k2 = -k
            if k2 > 0:
                G[k2:, j] = 1.0 - h[:n-k2]
            if n - k2 > 0:
                G[:n-k2, j] -= h[k2:]
    return G

def lp_step(h, n, eps, top_frac, solver):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = float(corr.max())
    n_sel = max(10, int(len(corr) * top_frac))
    idx = np.argsort(corr)[-n_sel:]
    G = build_G_sel(h, n, idx)
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        corr[idx] + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    for sol in [solver, 'GLPK', 'ECOS', 'CLARABEL', 'SCS']:
        try:
            prob.solve(solver=sol)
            if delta.value is not None:
                break
        except Exception:
            continue
    if delta.value is None:
        return h, mc
    d = np.asarray(delta.value).flatten()
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
        h_try = project_feasible(h + lam * d, n)
        mc_try = float(np.max(np.correlate(h_try, 1.0 - h_try, mode='full')))
        if mc_try < best_mc:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

def slsqp_polish(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h *= (target / s)
    return np.clip(h, 0.0, 1.0)

def build_G_sel(h, n, idx):
    n_sel = len(idx)
    G = np.zeros((n, n_sel))
    for j, m in enumerate(idx):
        k = m - (n - 1)
        if k >= 0:
            if n - k > 0:
                G[:n-k, j] = 1.0 - h[k:]
            if k > 0:
                G[k:, j] -= h[:n-k]
        else:
            k2 = -k
            G[k2:, j] = 1.0 - h[:n-k2]
            G[:n-k2, j] -= h[k2:]
    return G

def lp_step(h, n, eps, top_frac, solver):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = float(corr.max())
    n_sel = max(10, int(len(corr) * top_frac))
    idx = np.argsort(corr)[-n_sel:]
    G = build_G_sel(h, n, idx)
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        corr[idx] + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    for sol in [solver, 'GLPK', 'ECOS', 'CLARABEL', 'SCS']:
        try:
            prob.solve(solver=sol)
            if delta.value is not None:
                break
        except Exception:
            continue
    if delta.value is None:
        return h, mc
    d = np.asarray(delta.value).flatten()
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
        h_try = project_feasible(h + lam * d, n)
        mc_try = float(np.max(np.correlate(h_try, 1.0 - h_try, mode='full')))
        if mc_try < best_mc:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc

def slsqp_polish(h, n, alpha, maxiter):
    dx = 2.0 / n
    ones = np.ones(n)
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    def cons_jac(x):
        return ones
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h
