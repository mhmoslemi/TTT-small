def c5(h):
    return float(cor(h).max() * (2.0 / len(h)))

def proj(v, n):
    v = np.clip(np.asarray(v, dtype=float), 0.0, 1.0)
    t = 0.5 * n
    for _ in range(7):
        d = t - float(v.sum())
        if abs(d) < 1e-12:
            break
        if d > 0:
            r = 1.0 - v
            s = float(r.sum())
            if s > 1e-14:
                v = v + d * r / s
        else:
            s = float(v.sum())
            if s > 1e-14:
                v = v + d * v / s
        v = np.clip(v, 0.0, 1.0)
    d = t - float(v.sum())
    if abs(d) > 1e-12:
        idx = np.flatnonzero(v < 1.0 if d > 0 else v > 0.0)
        if idx.size:
            v = v + (d / idx.size) * np.concatenate([np.zeros(idx[0]),
                                                     np.ones(n - idx[0])])
            v[idx] += d / idx.size
            v = np.clip(v, 0.0, 1.0)
    return v

def lse(z, beta):
    m = float(z.max())
    if beta <= 0:
        return m
    e = np.exp(beta * (z - m))
    s = float(e.sum())
    return m + (np.log(s) / beta if s > 0.0 else 0.0)

def lag_grad(g, h, k, sc=1.0):
    n = len(h)
    if k == 0:
        g += sc * (1.0 - 2.0 * h)
    elif k > 0:
        g[:n - k] -= sc * h[k:]
        g[k:] += sc * (1.0 - h[:n - k])
    else:
        k = -k
        g[:n - k] += sc * (1.0 - h[k:])
        g[k:] -= sc * h[:n - k]

    return lse(cor(x), beta)

            T = 3.0
        i = zeros[int(rng.randint(len(zeres if False else zeros)))]
        j = ones[int(rng.randint(len(ones)))]
        b[i], b[j] = 1.0, 0.0
        nv = float(cor(b).max())
        if nv < cur or (nv - cur <= 6.0 and rng.random() < np.exp(-(nv - cur) / T)):
            cur = nv
            zeros.remove(i)
            zeros.append(j)
            ones.remove(j)
            ones.append(i)
            if nv < bm:
                bm, bh, stall = nv, b.copy(), 0
            else:
                stall += 1
        else:
            b[i], b[j] = 0.0, 1.0
            stall += 1
    return bh, float(bm * (2.0 / n))

def optcand(h, n, t0, dl, seed):
    h = proj(h, n)
    bh = h.copy()
    bc = c5(h)
    dur = max(0.0, dl - t0)
    if dur < 8:
        return bh, bc
    rng = np.random.RandomState(seed)
    base = 220 if n > 350 else 320 if n > 220 else 460
    base = int(base * min(1.0, max(0.35, dur / 120.0)))
    t1 = min(dl, t0 + 0.38 * dur)
    t2 = min(dl, t0 + 0.70 * dur)
    x = bh.copy()
    for beta in (1.0, 8.0, 30.0):
        if time.time() > t1 - 2:
            break
        x = slsqp(x, n, beta, base, 1e-11, t1)
        v = c5(x)
        if v < bc:
            bc, bh = v, x.copy()
    x = bh.copy()
    for R in ([4, 8, 16, 32] if dur > 220 else [4, 8, 16]):
        if time.time() > t2 - 2:
            break
        for _ in range(2):
            ag = sel(cor(x), R)
            x = act(x, n, ag, max(50, int(base * 0.7)), 1e-11, t2)
            v = c5(x)
            if v < bc:
                bc, bh = v, x.copy()
    if dur > 150 and time.time() < dl - 30:
        for s in (0.04, 0.10, 0.18):
            if time.time() > dl - 15:
                break
            xr = proj(bh + s * rng.randn(n), n)
            ag = sel(cor(xr), 8)
            xr = act(xr, n, ag, max(50, int(base * 0.5)), 1e-11, dl)
            v = c5(xr)
            if v < bc:
                bc, bh = v, xr.copy()
    if time.time() < dl - 10:
        rem = dl - time.time()
        pgl = dl - min(rem * 0.45, 80.0) if (n % 2 == 0 and rem > 80) else dl
        x, v = pg(bh, n, pgl, 50.0, 28, 70000, 0.07)
        if v < bc:
            bc, bh = v, x.copy()
        if time.time() < dl - 8:
            x, v = pg(bh, n, dl - 1, 8.0, 60, 40000, 0.03)
            if v < bc:
                bc, bh = v, x.copy()
    if n % 2 == 0 and time.time() < dl - 5:
        rem = dl - time.time()
        frac = 0.5 if n <= 250 else 0.35 if n <= 350 else 0.22
        bl = min(dl, time.time() + frac * rem)
        xb, vb = binse(bh, n, bl, seed + 991)
        if vb < bc:
            bc, bh = vb, xb.copy()
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t = time.time()
    total = max(10.0, float(budget_s) - 6.0)
    dl = t + total
    try:
        p = np.asarray(initial_h_values, dtype=float).ravel()
        if p.size < 2 or not np.all(np.isfinite(p)):
            raise ValueError
    except Exception:
        p = None
    rng = np.random.RandomState(seed)
    if p is None:
        n0 = 300
        h0 = smooth_start(n0, rng)
    else:
        n0 = int(p.size)
        h0 = proj(p, n0)
    bh = h0
    bn = n0
    bc = c5(bh)
    if total < 30:
        bh = proj(bh, bn)
        return bh, c5(bh), bn
    sizes = [n0]
    for s in (300, 250, 200, 160, 128):
        if s < n0 and s >= 40 and s not in sizes:
            sizes.append(s)
            if len(sizes) >= 4:
                break
    if len(sizes) < 4:
        for f in (0.75, 0.55, 0.40):
            v = int(round(n0 * f / 2.0) * 2)
            if v >= 40 and v < n0 and v not in sizes:
                sizes.append(v)
                if len(sizes) >= 4:
                    break
    if len(sizes) == 4:
        fr = (0.34, 0.24, 0.22, 0.20)
    elif len(sizes) == 3:
        fr = (0.40, 0.30, 0.30)
    elif len(sizes) == 2:
        fr = (0.58, 0.42)
    else:
        fr = (1.0,)
    marks = []
    cu = 0.0
    for f in fr:
        cu += f
        marks.append(cu)
    for idx, nn in enumerate(sizes):
        d = t + marks[idx] * total
        d = min(d, dl)
        if d <= time.time():
            if idx == len(sizes) - 1:
                d = dl
            else:
                continue
        if idx == 0:
            ini = bh.copy()
        elif idx == len(sizes) - 1 and rng.rand() < 0.30:
            ini = smooth_start(nn, rng)
        else:
            ini = resample(bh, bn, nn)
        ox, oc = optcand(ini, nn, time.time(), d, seed + 101 * idx)
        if oc < bc:
            bc, bh, bn = oc, ox.copy(), nn
    if time.time() < dl - 25:
        nn = bn if bn <= 250 else 250
        ox, oc = optcand(resample(bh, bn, nn), nn, time.time(), dl, seed + 999)
        if oc < bc:
            bc, bh, bn = oc, ox.copy(), nn
    bh = proj(bh, bn)
    return bh, float(c5(bh)), int(bn)
