   d=float(h.sum()-0.5*n)

def goal_jac(x):
   n=_OPT_N; al=_OPT_ALPHA; pen=_OPT_PEN
   h=np.clip(np.asarray(x,dtype=float),0.0,1.0)
   c=score_scaled(h,n)

import numpy as np
import time
from scipy.optimize import minimize, linprog

_MAP_N = 400
_MAP_ALPHA = 1000.0
_MAP_VPEN = 10000.0
_MAP_MA = 200

def compute_c5(h, n):
    h = np.asarray(h, dtype=float)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def distribute_residual(h, r):
    if r > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        for i in idx:
            if r <= 1e-15:
                break
            add = min(r, 1.0 - h[i])
            h[i] += add
            r -= add
    else:
        r = -r
        idx = np.where(h > 1e-12)[0]
        for i in idx:
            if r <= 1e-15:
                break
            sub = min(r, h[i])
            h[i] -= sub
            r -= sub

def project_exact(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * n
    if h.min() >= -1e-9 and h.max() <= 1.0 + 1e-9:
        h = np.clip(h, 0.0, 1.0)
        distribute_residual(h, target - h.sum())
        if abs(h.sum() - target) <= 1e-10:
            return h
    y = np.clip(h, 0.0, 1.0)
    if abs(y.sum() - target) < 0.25:
        distribute_residual(y, target - y.sum())
        if abs(y.sum() - target) <= 1e-10:
            return y
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    distribute_residual(h, target - h.sum())
    return h

def score_scaled(h, n):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def top_indices(c, ma):
    ma = int(ma)
    if ma <= 0:
        return np.array([], dtype=int)
    if ma >= len(c):
        return np.arange(len(c))
    return np.argpartition(c, -ma)[-ma:]

def weighted_subgrad(h, n, idx, ws):
    g = np.zeros(n)
    lm1 = n - 1
    for idxi, w in zip(idx, ws):
        w = float(w)
        if w <= 1e-15:
            continue
        k = int(idxi - lm1)
        if k >= 0:
            wd = n - k
            g[:wd] += w * (1.0 - h[k:])
            g[k:] -= w * h[:wd]
        else:
            kk = -k
            wd = n - kk
            g[kk:] += w * (1.0 - h[:wd])
            g[:wd] -= w * h[kk:]
    return g * (2.0 / n)

def default_ma(n):
    return int(min(2 * n - 1, max(128, n // 2)))

def h_from_z(z, n):
    m = n - 1
    a = np.clip(np.asarray(z, dtype=float), 0.0, 1.0)
    h = np.empty(n, dtype=float)
    h[:m] = a
    h[m] = 0.5 * n - float(a.sum())
    return h

def map_fun(z):
    n = _MAP_N
    al = _MAP_ALPHA
    pen = _MAP_VPEN
    h = h_from_z(z, n)
    c = score_scaled(h, n)
    m = float(np.max(c))
    idx = top_indices(c, _MAP_MA)
    if idx.size:
        s = al * (c[idx] - m)
        val = m + float(np.log(np.sum(np.exp(s)))) / al
    else:
        val = m
    last = float(h[-1])
    if last > 1.0:
        val += pen * (last - 1.0) * (last - 1.0)
    elif last < 0.0:
        val += pen * last * last
    return float(val)

def map_jac(z):
    n = _MAP_N
    al = _MAP_ALPHA
    pen = _MAP_VPEN
    h = h_from_z(z, n)
    c = score_scaled(h, n)
    m = float(np.max(c))
    idx = top_indices(c, _MAP_MA)
    if idx.size:
        s = al * (c[idx] - m)
        e = np.exp(s)
        tot = float(e.sum())
        w = e / tot if tot > 0 else np.zeros(idx.size)
        gh = weighted_subgrad(h, n, idx, w)
    else:
        gh = np.zeros(n)
    last = float(h[-1])
    if last > 1.0:
        dp = 2.0 * pen * (last - 1.0)
    elif last < 0.0:
        dp = 2.0 * pen * last
    else:
        dp = 0.0
    mv = n - 1
    g = gh[:mv] - gh[-1] - dp
    za = np.asarray(z, dtype=float)
    if za.size == mv and (za.min() < -1e-9 or za.max() > 1.0 + 1e-9):
        g = ((za >= -1e-9) & (za <= 1.0 + 1e-9)) * g
    return g

def polish_map(h, n, alpha, maxiter, vpen, ma):
    global _MAP_N, _MAP_ALPHA, _MAP_VPEN, _MAP_MA
    h = project_exact(h, n)
    _MAP_N = int(n)
    _MAP_ALPHA = float(alpha)
    _MAP_VPEN = float(vpen)
    _MAP_MA = int(ma)
    z = np.clip(h[:-1], 0.0, 1.0)
    bounds = [(0.0, 1.0) for _ in range(n - 1)]
    c0 = compute_c5(h, n)
    try:
        res = minimize(map_fun, z, jac=map_jac, bounds=bounds, method='L-BFGS-B',
                       options={'maxiter': int(maxiter), 'maxfun': 2 * int(maxiter) + 10,
                                'gtol': 1e-9, 'ftol': 1e-12})
        out = h_from_z(res.x, n)
    except Exception:
        out = h
    out = project_exact(out, n)
    c1 = compute_c5(out, n)
    if c1 < c0:
        return out, c1
    return h, c0

def lp_step(h, n, eps, window, max_active):
    h = project_exact(h, n)
    c = score_scaled(h, n)
    m = float(np.max(c))
    ma = int(max_active)
    if ma >= len(c):
        act = np.arange(len(c))
    else:
        act = np.flatnonzero(c >= m - float(window))
        if act.size == 0:
            act = np.array([int(np.argmax(c))])
        if act.size > ma:
            cc = c[act]
            top = np.argpartition(cc, -ma)[-ma:]
            act = act[top]
    A = np.zeros((act.size, n + 1))
    b = np.zeros(act.size)
    dx = 2.0 / n
    for rr, idx in enumerate(act):
        k = int(idx - (n - 1))
        g = np.zeros(n)
        if k >= 0:
            wd = n - k
            g[:wd] += 1.0 - h[k:]
            g[k:] -= h[:wd]
        else:
            kk = -k
            wd = n - kk
            g[kk:] += 1.0 - h[:wd]
            g[:wd] -= h[kk:]
        g *= dx
        A[rr, :n] = g
        A[rr, n] = -1.0
        b[rr] = -c[idx]
    low = np.maximum(-h, -float(eps))
    high = np.minimum(1.0 - h, float(eps))
    bounds = list(zip(low, high))
    bounds.append((0.0, float(m) + 1.0))
    cobj = np.zeros(n + 1)
    cobj[-1] = 1.0
    Aeq = np.zeros((1, n + 1))
    Aeq[0, :n] = 1.0
    try:
        res = linprog(cobj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=[0.0], bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:
        try:
            res = linprog(cobj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=[0.0], bounds=bounds,
                          method='interior-point')
        except Exception:
            return None
    if not getattr(res, 'success', False) or res.x is None:
        return None
    p = res.x[:n]
    if np.any(~np.isfinite(p)) or float(np.linalg.norm(p)) < 1e-13:
        return None
    base = compute_c5(h, n)
    best_h = h
    best_c = base
    for mult in (1.0, 0.7, 0.4, 0.2, 0.1, 0.05):
        cand = project_exact(h + float(mult) * p, n)
        cc = compute_c5(cand, n)
        if cc < best_c - 1e-13:
            best_c = cc
            best_h = cand
    if best_c < base - 1e-13:
        return best_h, best_c
    return None

def lp_polish(h, n, deadline, max_iters=200, max_active=None):
    h = project_exact(h, n)
    best_c = compute_c5(h, n)
    if max_active is None:
        max_active = 2 * n - 1 if n <= 450 else int(min(2 * n - 1, max(80, n // 2)))
    eps = 0.1
    if int(max_active) >= 2 * n - 1:
        windows = (0.0,)
    else:
        windows = (0.02, 0.005, 0.05, 0.1, 0.2)
    stuck = 0
    for _ in range(int(max_iters)):
        if time.time() > deadline:
            break
        improved = False
        for window in windows:
            if time.time() > deadline:
                break
            res = lp_step(h, n, eps, float(window), max_active)
            if res is not None:
                h, best_c = res[0], res[1]
                improved = True
                stuck = 0
                eps = max(0.002, eps * 0.85)
                break
        if not improved:
            stuck += 1
            eps *= 0.65
            if eps < 0.0012:
                eps = 0.0012
            if stuck > 12:
                break
    return h, best_c

import numpy as np
import time
from scipy.optimize import minimize, linprog

_MAP_N = 400
_MAP_ALPHA = 1000.0
_MAP_VPEN = 10000.0
_MAP_MA = 200


def compute_c5(h, n):
    h = np.asarray(h, dtype=float)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))


def distribute_residual(h, r):
    if r > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        for i in idx:
            if r <= 1e-15:
                break
            add = min(r, 1.0 - h[i])
            h[i] += add
            r -= add
    else:
        r = -r
        idx = np.where(h > 1e-12)[0]
        for i in idx:
            if r <= 1e-15:
                break
            sub = min(r, h[i])
            h[i] -= sub
            r -= sub


def project_exact(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * n
    if h.min() >= -1e-9 and h.max() <= 1.0 + 1e-9:
        h = np.clip(h, 0.0, 1.0)
        distribute_residual(h, target - h.sum())
        if abs(h.sum() - target) <= 1e-10:
            return h
    y = np.clip(h, 0.0, 1.0)
    if abs(y.sum() - target) < 0.25:
        distribute_residual(y, target - y.sum())
        if abs(y.sum() - target) <= 1e-10:
            return y
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(y - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    distribute_residual(h, target - h.sum())
    return h


def score_scaled(h, n):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def top_indices(c, ma):
    ma = int(ma)
    if ma <= 0:
        return np.array([], dtype=int)
    if ma >= len(c):
        return np.arange(len(c))
    return np.argpartition(c, -ma)[-ma:]


def weighted_subgrad(h, n, idx, ws):
    g = np.zeros(n)
    lm1 = n - 1
    for idxi, w in zip(idx, ws):
        w = float(w)
        if w <= 1e-15:
            continue
        k = int(idxi - lm1)
        if k >= 0:
            wd = n - k
            g[:wd] += w * (1.0 - h[k:])
            g[k:] -= w * h[:wd]
        else:
            kk = -k
            wd = n - kk
            g[kk:] += w * (1.0 - h[:wd])
            g[:wd] -= w * h[kk:]
    return g * (2.0 / n)


def default_ma(n):
    return int(min(2 * n - 1, max(128, n // 2)))


def h_from_z(z, n):
    m = n - 1
    a = np.clip(np.asarray(z, dtype=float), 0.0, 1.0)
    h = np.empty(n, dtype=float)
    h[:m] = a
    h[m] = 0.5 * n - float(a.sum())
    return h


def map_fun(z):
    n = _MAP_N
    al = _MAP_ALPHA
    pen = _MAP_VPEN
    h = h_from_z(z, n)
    c = score_scaled(h, n)
    m = float(np.max(c))
    idx = top_indices(c, _MAP_MA)
    if idx.size:
        s = al * (c[idx] - m)
        val = m + float(np.log(np.sum(np.exp(s)))) / al
    else:
        val = m
    last = float(h[-1])
    if last > 1.0:
        val += pen * (last - 1.0) * (last - 1.0)
    elif last < 0.0:
        val += pen * last * last
    return float(val)


def map_jac(z):
    n = _MAP_N
    al = _MAP_ALPHA
    pen = _MAP_VPEN
    h = h_from_z(z, n)
    c = score_scaled(h, n)
    m = float(np.max(c))
    idx = top_indices(c, _MAP_MA)
    if idx.size:
        s = al * (c[idx] - m)
        e = np.exp(s)
        tot = float(e.sum())
        w = e / tot if tot > 0 else np.zeros(idx.size)
        gh = weighted_subgrad(h, n, idx, w)
    else:
        gh = np.zeros(n)
    last = float(h[-1])
    if last > 1.0:
        dp = 2.0 * pen * (last - 1.0)
    elif last < 0.0:
        dp = 2.0 * pen * last
    else:
        dp = 0.0
    mv = n - 1
    g = gh[:mv] - gh[-1] - dp
    za = np.asarray(z, dtype=float)
    if za.size == mv and (za.min() < -1e-9 or za.max() > 1.0 + 1e-9):
        g = ((za >= -1e-9) & (za <= 1.0 + 1e-9)) * g
    return g


def polish_map(h, n, alpha, maxiter, vpen, ma):
    global _MAP_N, _MAP_ALPHA, _MAP_VPEN, _MAP_MA
    h = project_exact(h, n)
    _MAP_N = int(n)
    _MAP_ALPHA = float(alpha)
    _MAP_VPEN = float(vpen)
    _MAP_MA = int(ma)
    z = np.clip(h[:-1], 0.0, 1.0)
    bounds = [(0.0, 1.0) for _ in range(n - 1)]
    c0 = compute_c5(h, n)
    try:
        res = minimize(map_fun, z, jac=map_jac, bounds=bounds, method='L-BFGS-B',
                       options={'maxiter': int(maxiter), 'maxfun': 2 * int(maxiter) + 10,
                                'gtol': 1e-9, 'ftol': 1e-12})
        out = h_from_z(res.x, n)
    except Exception:
        out = h
    out = project_exact(out, n)
    c1 = compute_c5(out, n)
    if c1 < c0:
        return out, c1
    return h, c0


def lp_step(h, n, eps, window, max_active):
    h = project_exact(h, n)
    c = score_scaled(h, n)
    m = float(np.max(c))
    ma = int(max_active)
    if ma >= len(c):
        act = np.arange(len(c))
    else:
        act = np.flatnonzero(c >= m - float(window))
        if act.size == 0:
            act = np.array([int(np.argmax(c))])
        if act.size > ma:
            cc = c[act]
            top = np.argpartition(cc, -ma)[-ma:]
            act = act[top]
    A = np.zeros((act.size, n + 1))
    b = np.zeros(act.size)
    dx = 2.0 / n
    for rr, idx in enumerate(act):
        k = int(idx - (n - 1))
        g = np.zeros(n)
        if k >= 0:
            wd = n - k
            g[:wd] += 1.0 - h[k:]
            g[k:] -= h[:wd]
        else:
            kk = -k
            wd = n - kk
            g[kk:] += 1.0 - h[:wd]
            g[:wd] -= h[kk:]
        g *= dx
        A[rr, :n] = g
        A[rr, n] = -1.0
        b[rr] = -c[idx]
    low = np.maximum(-h, -float(eps))
    high = np.minimum(1.0 - h, float(eps))
    bounds = list(zip(low, high))
    bounds.append((0.0, float(m) + 1.0))
    cobj = np.zeros(n + 1)
    cobj[-1] = 1.0
    Aeq = np.zeros((1, n + 1))
    Aeq[0, :n] = 1.0
    try:
        res = linprog(cobj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=[0.0], bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:
        try:
            res = linprog(cobj, A_ub=A, b_ub=b, A_eq=Aeq, b_eq=[0.0], bounds=bounds,
                          method='interior-point')
        except Exception:
            return None
    if not getattr(res, 'success', False) or res.x is None:
        return None
    p = res.x[:n]
    if np.any(~np.isfinite(p)) or float(np.linalg.norm(p)) < 1e-13:
        return None
    base = compute_c5(h, n)
    best_h = h
    best_c = base
    for mult in (1.0, 0.7, 0.4, 0.2, 0.1, 0.05):
        cand = project_exact(h + float(mult) * p, n)
        cc = compute_c5(cand, n)
        if cc < best_c - 1e-13:
            best_c = cc
            best_h = cand
    if best_c < base - 1e-13:
        return best_h, best_c
    return None


def lp_polish(h, n, deadline, max_iters=200, max_active=None):
    h = project_exact(h, n)
    best_c = compute_c5(h, n)
    if max_active is None:
        max_active = 2 * n - 1 if n <= 450 else int(min(2 * n - 1, max(80, n // 2)))
    eps = 0.1
    if int(max_active) >= 2 * n - 1:
        windows = (0.0,)
    else:
        windows = (0.02, 0.005, 0.05, 0.1, 0.2)
    stuck = 0
    for _ in range(int(max_iters)):
        if time.time() > deadline:
            break
        improved = False
        for window in windows:
            if time.time() > deadline:
                break
            res = lp_step(h, n, eps, float(window), max_active)
            if res is not None:
                h, best_c = res[0], res[1]
                improved = True
                stuck = 0
                eps = max(0.002, eps * 0.85)
                break
        if not improved:
            stuck += 1
            eps *= 0.65
            if eps < 0.0012:
                eps = 0.0012
            if stuck > 12:
                break
    return h, best_c
