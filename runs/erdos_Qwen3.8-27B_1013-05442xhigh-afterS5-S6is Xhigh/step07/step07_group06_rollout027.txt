import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel().copy()
    if a.size < n:
        a = np.concatenate([a, np.zeros(n - a.size)])
    elif a.size > n:
        a = a[:n]
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo, hi = float(a.min()) - 1.0, float(a.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(a - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0, 1)
    for _ in range(30):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h; s = r.sum()
            if s > 1e-15:
                h += d * r / s
        else:
            s = h.sum()
            if s > 1e-15:
                h += d * h / s
    h = np.clip(h, 0, 1)
    d = target - h.sum()
    if abs(d) > 1e-15:
        if d > 0:
            i = int(np.argmax(1 - h)); h[i] = min(1.0, h[i] + d)
        else:
            i = int(np.argmax(h)); h[i] = max(0.0, h[i] + d)
    return h

_gN = 400
_gBeta = 1.0

_gDeadline = 0.0

def _cb(*a):
    return time.time() > _gDeadline

def _obj(x):
    n = _gN; b = _gBeta; t = _gTarget
    h = np.clip(x, 0, 1)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    c = np.maximum(c, 0)
    m = float(c.max())
    y = b * (c - m)
    e = np.exp(np.minimum(y, 0))
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return m, np.zeros(n)
    p = e / s
    F = m + np.log(s) / b
    R = p[::-1]
    v = 1 - h
    A = np.convolve(R, v, mode="full")[n-1:2*n-1]
    B = np.convolve(p, h, mode="full")[n-1:2*n-1]
    g = dx * (A - B)
    if t > 0:
        cp = np.maximum(c - t, 0)
        F += 0.5 * float(cp.dot(cp))
        g += dx * (np.convolve(cp, v, mode="full")[n-1:2*n-1] -
                   np.convolve(cp, h, mode="full")[n-1:2*n-1])
    if not np.all(np.isfinite(g)):
        g = np.zeros(n)
    return float(F), g

def do_opt(h, n, beta, target, deadline, max_t=60):
    global _gN, _gBeta, _gTarget, _gDeadline
    _gN, _gBeta, _gTarget = n, float(beta), float(target)
    _gDeadline = min(deadline, time.time() + max_t)
    try:
        r = minimize(_obj, np.clip(h, 0, 1), method="L-BFGS-B", jac=True,
                     bounds=[(0.0, 1.0)] * n, callback=_cb,
                     options={"maxiter": 1000000, "maxls": 30, "ftol": 1e-15, "gtol": 1e-10})
        return project_feasible(np.asarray(r.x), n)
    except Exception:
        return project_feasible(h, n)

def subgrad_multi(h, n, step_size, n_top=8):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    top_k = np.argsort(-c)[:n_top]
    wts = np.exp(40.0 * (c[top_k] - c[top_k[0]]))
    wts /= wts.sum()
    g = np.zeros(n)
    for w, k in zip(wts, top_k):
        delta = int(k) - (n - 1)
        if delta >= 0:
            g[:n-delta] += w * (1.0 - h[delta:])
            g[delta:] -= w * h[:n-delta]
        else:
            g[-delta:] += w * (1.0 - h[:n+delta])
            g[:n+delta] -= w * h[-delta:]
    g *= dx
    return project_feasible(h - step_size * g, n)

def get_variants(h, n):
    out = [h.copy(), h[::-1].copy(), (1.0 - h).copy(), (1.0 - h[::-1]).copy()]
    out.append(0.5 * (h + h[::-1]))
    out.append(0.5 * (1.0 - h + h[::-1]))
    for s in [n // 8, n // 5, n // 4]:
        out.append(np.roll(h, s))
    return out
