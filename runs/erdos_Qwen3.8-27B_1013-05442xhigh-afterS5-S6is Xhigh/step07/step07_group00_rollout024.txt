The number of lines is O(n) (since only O(n) correlation values are affected by changing h[p]).

To find the minimum of the upper envelope:
- Sort lines by slope
- Use a deque to build the upper envelope (like the "convex hull trick")
- Find the minimum of the envelope

But we also have the quadratic term for m=n-1. So:
f(d) = max(g(d), q(d)) where g(d) = max of lines and q(d) = c[n-1] + d*(1-2h[p]) - d²

The minimum of f is at:
1. The minimum of g(d) (if at that point g > q)
2. The minimum of q(d) = (2h[p]-1)/2 = h[p] - 1/2 (if at that point q > g)
3. A point where g(d) = q(d) (intersection)
4. Boundaries

def solve_1d_subproblem(c, h, p, n):
    """Solve the 1D subproblem for coordinate p exactly (or near-exactly)."""
    h0 = h[p]
    d_lo = -h0
    d_hi = 1.0 - h0
    
    # Compute A[m] = change in c[m] per unit d, for m ≠ n-1
    # A[m] = (1-h[p+m-n+1]) if 0≤p+m-n+1<n else 0
    #       - h[p-m+n-1] if 0≤p-m+n-1<n else 0
    # For m = n-1: the change is d*(1-2h0) - d², handled separately
    
    # Only O(n) values of m are affected
    m_affected = []
    A_vals = []
    c_vals = []
    
    for m in range(2*n-1):
        a = 0.0
        valid_fwd = (0 <= p+m-n+1 < n)
        valid_bwd = (0 <= p-m+n-1 < n)
        if valid_fwd:
            a += (1.0 - h[p+m-n+1])
        if valid_bwd:
            a -= h[p-m+n-1]
        if m == n-1:
            # a should be (1-h0) - h0 = 1-2h0, but we handle the -d² separately
            continue
        if abs(a) > 1e-15:
            m_affected.append(m)
            A_vals.append(a)
            c_vals.append(c[m])
    
    A = np.array(A_vals)
    c_aff = np.array(c_vals)
    
    # Unaffected max
    mask_affected = np.zeros(2*n-1, dtype=bool)
    for m in m_affected:
        mask_affected[m] = True
    # Also mark n-1 as affected (quadratic)
    mask_affected[n-1] = True
    unaff_max = c[~mask_affected].max() if (~mask_affected).any() else -np.inf
    
    # The function is: f(d) = max(max(unaff, c_aff + d*A), c[n-1] + d*(1-2h0) - d²)
    # = max(max(unaff, c_aff + d*A), q(d))
    # where q(d) = c[n-1] + d*(1-2h0) - d²
    
    # Candidate points:
    candidates = [d_lo, d_hi, 0.0]
    
    # Minimum of q: d_q = (1-2h0)/2
    d_q = (1.0 - 2.0*h0) / 2.0
    if d_lo < d_q < d_hi:
        candidates.append(d_q)
    
    # Intersections of lines: for each pair (i,j) where A[i]≠A[j]:
    # c_aff[i] + d*A[i] = c_aff[j] + d*A[j]
    # d = (c_aff[j] - c_aff[i]) / (A[i] - A[j])
    # This is O(k²) where k = len(A). For n=500, k can be up to ~2n = 1000, so O(10^6) which is too slow.
    
    # Better: use the fact that we only need the minimum of the upper envelope.
    # Sort by slope, build upper hull.
    
    # Actually, for practical purposes, let's use a different approach:
    # Evaluate f(d) at a set of candidate points and pick the best.
    # The candidates are: boundaries, d_q, and a grid of points.
    
    # For efficiency, let's evaluate at ~50 points adaptively.
    d_grid = np.linspace(d_lo, d_hi, 51)
    f_vals = np.zeros(len(d_grid))
    for i, d in enumerate(d_grid):
        linear_max = np.max(c_aff + d * A) if len(A) > 0 else -np.inf
        q_val = c[n-1] + d*(1-2*h0) - d*d
        f_vals[i] = max(unaff_max, linear_max, q_val)
    
    # Find the best grid point
    best_i = np.argmin(f_vals)
    best_d = d_grid[best_i]
    
    # Refine around best_d with a finer grid
    d_fine = np.linspace(max(d_lo, best_d - (d_grid[1]-d_grid[0])), 
                         min(d_hi, best_d + (d_grid[1]-d_grid[0])), 101)
    f_fine = np.zeros(len(d_fine))
    for i, d in enumerate(d_fine):
        linear_max = np.max(c_aff + d * A) if len(A) > 0 else -np.inf
        q_val = c[n-1] + d*(1-2*h0) - d*d
        f_fine[i] = max(unaff_max, linear_max, q_val)
    best_i2 = np.argmin(f_fine)
    best_d = d_fine[best_i2]
    
    return best_d, float(f_fine[best_i2])
```

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0-x, mode='full')  # length 2n-1
    amax = alpha * c.max()
    exp_c = np.exp(alpha * c - amax)
    softmax = exp_c / exp_c.sum()
    
    # grad[p] = dx * sum_m softmax[m] * dc[m]/dx[p]
    # dc[m]/dx[p] = (1-x[p+m-n+1]) if 0<=p+m-n+1<n else 0
    #              - x[p-m+n-1] if 0<=p-m+n-1<n else 0
    
    grad = np.zeros(n)
    for m in range(2*n-1):
        wm = softmax[m] * dx
        if wm < 1e-15:
            continue
        # Forward: x[p] as first factor
        p_lo = max(0, n-1-m)
        p_hi = min(n-1, 2*n-2-m)
        if p_lo <= p_hi:
            idx = np.arange(p_lo, p_hi+1)
            j = idx + m - n + 1
            grad[idx] += wm * (1.0 - x[j])
        # Backward: x[p] as second factor  
        p_lo2 = max(0, m-n+1)
        p_hi2 = min(n-1, m+n-1)
        if p_lo2 <= p_hi2:
            idx2 = np.arange(p_lo2, p_hi2+1)
            j2 = idx2 - m + n - 1
            grad[idx2] -= wm * x[j2]
    return grad
```

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

        p_hi2 = min(n-1, m+n-1)
        if p_lo2 <= p_hi2:
            idx2 = np.arange(p_lo2, p_hi2+1)
            j2 = idx2 - m + n - 1
            grad[idx2] -= wm * x[j2]
    return grad

def coord_cd_exact(h, n, max_passes=200, deadline=None, refine_steps=3):
    """Coordinate descent with exact 1D subproblem solver."""
    h = h.copy()
    c5, c = compute_c5_and_corr(h, n)
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline - 2:
                break
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            d_lo = -h0
            d_hi = 1.0 - h0
            
            # Compute affected correlation values and their slopes
            m_lo = max(0, n-1-p)
            m_hi = min(2*n-2, max(2*n-2-p, p+n-1))
            
            ms = np.arange(m_lo, m_hi+1)
            A = np.zeros(len(ms))
            for i, m in enumerate(ms):
                a = 0.0
                jf = p + m - n + 1
                jb = p - m + n - 1
                if 0 <= jf < n:
                    a += (1.0 - h[jf])
                if 0 <= jb < n:
                    a -= h[jb]
                A[i] = a
            
            c_aff = c[ms]
            
            # Unaffected max
            mask = np.zeros(2*n-1, dtype=bool)
            mask[ms] = True
            mask[n-1] = True  # quadratic term
            if (~mask).any():
                unaff_max = c[~mask].max()
            else:
                unaff_max = -1e10
            
            # Solve 1D: minimize max(max(unaff, c_aff + d*A), c[n-1] + d*(1-2h0) - d²)
            # over d ∈ [d_lo, d_hi]
            
            # Coarse grid
            n_coarse = 41
            d_grid = np.linspace(d_lo, d_hi, n_coarse)
            best_d = 0.0
            best_val = c5  # current value (d=0)
            
            for d in d_grid:
                lin_max = np.max(c_aff + d * A) if len(A) > 0 else -1e10
                q_val = c[n-1] + d*(1.0-2.0*h0) - d*d
                val = max(unaff_max, lin_max, q_val)
                if val < best_val - 1e-15:
                    best_val = val
                    best_d = d
            
            if best_d == 0.0 and best_val >= c5 - 1e-15:
                continue
            
            # Refine around best_d
            span = (d_hi - d_lo) / n_coarse
            for _ in range(refine_steps):
                d_lo_r = max(d_lo, best_d - span)
                d_hi_r = min(d_hi, best_d + span)
                d_fine = np.linspace(d_lo_r, d_hi_r, 41)
                span = (d_hi_r - d_lo_r) / 41.0
                for d in d_fine:
                    lin_max = np.max(c_aff + d * A) if len(A) > 0 else -1e10
                    q_val = c[n-1] + d*(1.0-2.0*h0) - d*d
                    val = max(unaff_max, lin_max, q_val)
                    if val < best_val - 1e-15:
                        best_val = val
                        best_d = d
            
            if abs(best_d) > 1e-12:
                h[p] = h0 + best_d
                # Update correlation
                c = np.correlate(h, 1.0-h, mode='full')  # recompute (expensive but correct)
                c5 = float(c.max()) * (2.0/n)
                improved = True
        
        if not improved:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0-h, mode='full')
        c5 = float(c.max()) * (2.0/n)
    return h, c5

Wait, recomputing the full correlation after each coordinate change is O(n²) per coordinate, making the whole pass O(n³). For n=500, that's 125M operations per pass. Too slow.

I need to update c incrementally. When I change h[p] by d:
c[m] += d * A[m] for the affected m values
c[n-1] += d*(1-2*h0) - d² (the quadratic term)

So I can update c in O(n) per coordinate change:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    amax = alpha * c.max()
    return (amax + np.log(np.sum(np.exp(alpha * c - amax)))) / alpha * (2.0 / n)

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    amax = alpha * c.max()
    exp_c = np.exp(alpha * c - amax)
    sw = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for m in range(len(c)):
        wm = sw[m] * dx
        if wm < 1e-16:
            continue
        p_lo = max(0, n - 1 - m)
        p_hi = min(n - 1, 2 * n - 2 - m)
        if p_lo <= p_hi:
            idx = np.arange(p_lo, p_hi + 1)
            j = idx + m - n + 1
            grad[idx] += wm * (1.0 - x[j])
        p_lo2 = max(0, m - n + 1)
        p_hi2 = min(n - 1, m + n - 1)
        if p_lo2 <= p_hi2:
            idx2 = np.arange(p_lo2, p_hi2 + 1)
            j2 = idx2 - m + n - 1
            grad[idx2] -= wm * x[j2]
    return grad

def slsqp_run(h, n, alphas, maxiter_each=800, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n / 2.0,
                 'jac': lambda n=n: np.ones(n)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def coord_cd(h, n, max_passes=500, deadline=None):
    """Coordinate descent with incremental correlation update and exact 1D subproblem."""
    h = h.copy()
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c5_unnorm = float(c.max())
    c5 = c5_unnorm * dx

    for it in range(max_passes):
        if deadline is not None and time.time() > deadline - 3:
            break
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline - 2:
                break
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            d_lo = -h0
            d_hi = 1.0 - h0

            m_fwd_lo = max(0, n - 1 - p)
            m_fwd_hi = min(2 * n - 2, 2 * n - 2 - p)
            m_bwd_lo = 0
            m_bwd_hi = min(2 * n - 2, p + n - 1)
            m_lo = min(m_fwd_lo, m_bwd_lo)
            m_hi = max(m_fwd_hi, m_bwd_hi)

            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - n + 1
            jb = p - ms + n - 1
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            A = np.zeros(len(ms))
            A[valid_f] += (1.0 - h[jf[valid_f]])
            A[valid_b] -= h[jb[valid_b]]

            c_aff = c[ms].copy()
            c_N = c[n - 1]

            unaff_mask = np.ones(2 * n - 1, dtype=bool)
            unaff_mask[ms] = False
            unaff_max = c[unaff_mask].max() if unaff_mask.any() else -1e10

            idx_N = n - 1 - m_lo

            best_d = 0.0
            best_val = c5_unnorm

            n_coarse = 41
            d_grid = np.linspace(d_lo, d_hi, n_coarse)
            for d in d_grid:
                new_aff = c_aff + d * A
                if 0 <= idx_N < len(new_aff):
                    new_aff[idx_N] = c_N + d * (1.0 - 2.0 * h0) - d * d
                val = max(unaff_max, float(new_aff.max()))
                if val < best_val - 1e-15:
                    best_val = val
                    best_d = d

            if abs(best_d) < 1e-14:
                continue

            span = (d_hi - d_lo) / (n_coarse - 1)
            for _ in range(4):
                lo_r = max(d_lo, best_d - span)

                d_fine = np.linspace(lo_r, hi_r, 41)

                    if val < best_val - 1e-15:
                        best_val = val

                c[ms] = new_aff
                c5_unnorm = float(c.max())
                improved = True

        if not improved:

        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        c5_unnorm = float(c.max())
        c5 = c5_unnorm * dx

    return h, c5_unnorm * dx

def momentum_smooth(h, n, alpha, steps=200, deadline=None):
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    v = np.zeros(n)
    lr = 0.005
    mu = 0.9
    for step in range(steps):
        if deadline is not None and time.time() > deadline:
            break
        g = smooth_grad(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        v = mu * v - lr * g / (gn + 1e-10)
        h_new = project_feasible(h + v, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
    return best_h, best_c5
