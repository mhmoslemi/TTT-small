    m = c.max()
    exp_c = np.exp(alpha * (c - m))
    w = exp_c / exp_c.sum()
    # Gradient: grad[i] = sum_k w[k] * df_k/dh[i]
    # For each k, df_k/dh[i] is non-zero only for specific i

def project_feasible(h, n):

            grad[:je] += (1.0 - h[s:n])
            grad[s:n] -= h[:je]
        grad *= dx
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best - 1e-12:
            best = c5_new
            h = h_new
            lr = min(lr * 1.02, 0.08)
            stall = 0
        else:
            stall += 1
            lr *= 0.85
            if lr < 1e-6 or stall > 200:
                break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = 0.5 + 0.3 * np.sin(np.pi * np.linspace(0, 2, n))
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)

    # Phase 1: LP optimization at native resolution
    p1_end = t0 + 0.35 * (budget_s - 5)
    eps = 0.05
    stall = 0
    while time.time() < p1_end:
        h, mc = lp_step(h, n, eps)
        c5 = compute_c5(h, n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 40:
            eps *= 0.65
            if eps < 0.0008:
                break
            stall = 0

    # Phase 2: Greedy refinement (fast)
    p2_end = t0 + 0.48 * (budget_s - 5)
    while time.time() < p2_end:
        h, c5 = greedy_refine(best_h.copy(), n, max_iter=2000)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        else:
            break

    # Phase 3: Smooth SLSQP for diversity
    p3_end = t0 + 0.62 * (budget_s - 5)
    for alpha in [150, 800, 4000]:
        if time.time() > p3_end:
            break
        h_try = best_h.copy()
        h_try = smooth_optimize(h_try, n, alpha, maxiter=300)
        c5 = compute_c5(h_try, n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_try.copy()

    # Phase 4: Multi-start LP
    p4_end = t0 + 0.80 * (budget_s - 5)
    for trial in range(20):
        if time.time() > p4_end:
            break
        ns = 0.025 * (1.0 + 0.5 * rng.rand())
        h_try = project_feasible(best_h + ns * rng.randn(n), n)
        eps_r = 0.035
        for _ in range(80):
            if time.time() > p4_end:
                break
            h_try, _ = lp_step(h_try, n, eps_r)
            eps_r = max(eps_r * 0.99, 0.002)
        c5 = compute_c5(h_try, n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_try.copy()

    # Phase 5: Multi-resolution
    p5_end = deadline - 40
    for nh in [200, 600, 800]:
        if time.time() > p5_end:
            break
        if nh == n:
            continue
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        h_new = project_feasible(np.interp(xn, xo, best_h), nh)
        eps_r = 0.04
        for _ in range(120):
            if time.time() > p5_end:
                break
            h_new, _ = lp_step(h_new, nh, eps_r)
            eps_r = max(eps_r * 0.995, 0.002)
        h_new, c5g = greedy_refine(h_new, nh, max_iter=1000)
        c5 = compute_c5(h_new, nh)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
            n = nh

    # Phase 6: Final polish
    h = best_h.copy()
    eps_f = 0.004
    for _ in range(300):
        if time.time() > deadline:
            break
        h, _ = lp_step(h, n, eps_f)
        c5 = compute_c5(h, n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        eps_f = max(eps_f * 0.997, 0.0003)

    best_h, best_greedy_c5 = greedy_refine(best_h, n, max_iter=2000)
    best_c5 = min(compute_c5(best_h, n), best_greedy_c5)
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
