    sh = alpha * (c - m)

    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def project_box_sum(v, n):
    v = np.nan_to_num(np.asarray(v, dtype=float).reshape(-1))
    tgt = 0.5 * n
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > tgt:
            lo = mid
        else:
            hi = mid
    v = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(40):
        d = tgt - v.sum()
        if abs(d) < 1e-14:
            break
        m = (v > 1e-12) & (v < 1.0 - 1e-12)
        c = int(m.sum())
        if c:
            v[m] += d / c
        else:
            v += d / n
        v = np.clip(v, 0.0, 1.0)
    return v

def lse_obj(x, n, alpha):
    c = compute_c(x, n)
    m = float(c.max())
    if alpha <= 0:
        return m
    sh = alpha * (c - m)
    s = float(np.exp(sh).sum())
    if s <= 0:
        return m
    return m + np.log(s) / alpha

def lse_weights(c, alpha):
    m = float(c.max())
    sh = alpha * (c - m)
    e = np.exp(sh)
    s = float(e.sum())
    if s <= 0:
        q = np.zeros_like(c)
        q[int(np.argmax(c))] = 1.0
        return q
    q = e / s
    r = q.sum()
    if r > 0:
        q /= r
    return q

def lse_grad(x, n, alpha):
    c = compute_c(x, n)
    q = lse_weights(c, alpha)
    g = np.zeros(n)
    dx = 2.0 / n
    off = n - 1
    for idx in np.flatnonzero(q):
        k = idx - off
        w = q[idx]
        if k > 0:
            s = k
            m = n - s
            g[s:n] += w * (1.0 - x[0:m])
            g[:m] -= w * x[s:n]
        elif k < 0:
            s = -k
            m = n - s
            g[:m] += w * (1.0 - x[s:n])
            g[s:n] -= w * x[0:m]
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g

def coord_pass(h, n, rng=None, perturb_chance=0.0):
    dx = 2.0 / n
    off = n - 1
    c = compute_c(h, n)
    L = 2 * n - 1
    offsets = np.arange(-(n - 1), n)
    pos_mask_cache = {}
    for i in range(n):
        if rng is not None and rng.random() < perturb_chance:
            h[i] = np.clip(h[i] + 0.05 * (rng.random() - 0.5), 0.0, 1.0)
            c = compute_c(h, n)
        h_i = h[i]
        ip = i + offsets
        im = i - offsets
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        g = np.zeros(L)
        g[vp] += 1.0 - h[ip[vp]]
        g[vm] -= h[im[vm]]
        pos = g > 1e-15
        neg = g < -1e-15
        t_lo = 0.0
        t_hi = float(c.max()) + 1e-14
        best_delta = 0.0
        for _ in range(45):
            t_mid = 0.5 * (t_lo + t_hi)
            lo_b = -h_i
            hi_b = 1.0 - h_i
            ok = True
            if pos.any():
                ub = (t_mid - c[pos]) / (dx * g[pos])
                hi_b = min(hi_b, float(ub.min()))
            if neg.any():
                lb = (t_mid - c[neg]) / (dx * g[neg])
                lo_b = max(lo_b, float(lb.max()))
            if not pos.any() and not neg.any():
                pass
            if ok and lo_b <= hi_b:
                t_hi = t_mid
                best_delta = 0.5 * (lo_b + hi_b)
            else:
                t_lo = t_mid
        h[i] = np.clip(h_i + best_delta, 0.0, 1.0)
        c += dx * best_delta * g
    h = project_box_sum(h, n)
    c = compute_c(h, n)
    return h, float(c.max())

def lse_lbfgsb(h, n, alpha, mu, maxiter):
    h = project_box_sum(h, n)
    def pen_obj(x):
        d = float(x.sum() - 0.5 * n)
        return lse_obj(x, n, alpha) + 0.5 * mu * d * d
    def pen_grad(x):
        d = float(x.sum() - 0.5 * n)
        return lse_grad(x, n, alpha) + mu * d
    try:
        bounds = [(0.0, 1.0)] * n
        res = minimize(pen_obj, h, method="L-BFGS-B", jac=pen_grad,
                       bounds=bounds, options={"maxiter": maxiter, "maxfun": 4 * maxiter,
                                               "ftol": 1e-15, "gtol": 1e-9})
        if hasattr(res, "x") and res.x is not None:
            return res.x
    except Exception:
        pass
    return h

def slsqp_phase(h, n, alpha, maxiter):
    h = project_box_sum(h, n)
    def sfun(x):
        return float(x.sum() - 0.5 * n)
    def sjac(x):
        return np.ones(n)
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{"type": "eq", "fun": sfun, "jac": sjac}]
        res = minimize(lse_obj, h, args=(n, alpha), method="SLSQP",
                       jac=lambda x: lse_grad(x, n, alpha), bounds=bounds,
                       constraints=cons, options={"maxiter": maxiter, "ftol": 1e-14})
        if hasattr(res, "x") and res.x is not None:
            return res.x
    except Exception:
        pass
    return h

def pair_search(h, c_base, i, j, n):
    if i > j:
        i, j = j, i
    x0 = float(h[i])
    y0 = float(h[j])
    S = x0 + y0
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if hi - lo < 1e-13:
        return None
    xs = np.linspace(lo, hi, 64)
    xv0 = float(h[i])
    yv0 = float(h[j])
    omh = 1.0 - h
    omhrev = omh[::-1]
    dx = 2.0 / n
    off = n - 1
    L = 2 * n - 1
    def eval_xs(xx):
        delta = np.zeros(L)
        ax1 = xx - xv0
        ax2 = xv0 - xx
        yv = S - xx
        delta[i:i+n] += ax1 * omhrev
        si = off - i
        delta[si:si+n] += ax2 * h
        delta[j:j+n] += ax2 * omhrev
        sj = off - j
        delta[sj:sj+n] += ax1 * h
        idx0 = off
        base_self = ax1*(1.0-xv0) + xv0*ax2 + ax2*(1.0-yv0) + yv0*ax1
        exact_self = xx*(1.0-xx) + yv*(1.0-yv) - (xv0*(1.0-xv0) + yv0*(1.0-yv0))
        delta[idx0] += exact_self - base_self
        d = j - i
        idxp = off - d
        base_p = ax1*(1.0-yv0) + xv0*ax1
        exact_p = xx*(1.0-yv) - xv0*(1.0-yv0)
        delta[idxp] += exact_p - base_p
        idxm = off + d
        base_m = ax2*(1.0-xv0) + yv0*ax2
        exact_m = yv*(1.0-xx) - yv0*(1.0-xv0)
        delta[idxm] += exact_m - base_m
        return np.max(c_base + dx * delta)
    vals = np.array([eval_xs(x) for x in xs])
    for _ in range(3):
        k = int(np.argmin(vals))
        if k == 0:
            a, b = xs[0], xs[1]
        elif k == len(xs) - 1:

        else:
            a, b = xs[k-1], xs[k+1]
        if b - a < 1e-12:
            break
        xs2 = np.linspace(a, b, 48)
        vals = np.array([eval_xs(x) for x in xs2])
        xs = xs2
    k = int(np.argmin(vals))
    bv = float(vals[k])
    if bv < float(c_base.max()) - 1e-13:
        xb = float(np.clip(xs[k], lo, hi))
        yb = float(np.clip(S - xb, lo, hi))
        return xb, yb, bv
    return None

def pair_local_search(h, deadline, seed=1):

            d = int(rng.choice(n-1, p=w)) + 1

            noimp = 0

            noimp += 1
