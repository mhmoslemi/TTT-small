    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target - s

            break

        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        logsum = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

        return logsum + 1e6 * s * s

                   options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            sr = room.sum()
            if sr < 1e-14:
                break
            h = h + diff / sr * room
        else:
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def logsumexp_obj(x, n, alpha, target):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    s = x.sum() - target
    val += 1e5 * s * s
    return val

def logsumexp_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    cm = c.max()
    sh = alpha * (c - cm)
    p = np.exp(sh - sh.max())
    p /= p.sum()
    S_plus = np.zeros(n)
    S_minus = np.zeros(n)
    for li in range(2 * n - 1):
        l = li - (n - 1)
        w = p[li]
        if w < 1e-12:
            continue
        if l >= 0 and l < n:
            S_plus[:n - l] += w * x[l:]
            S_minus[l:] += w * x[:n - l]
        elif l < 0 and -l < n:
            ll = -l
            S_plus[ll:] += w * x[:n - ll]
            S_minus[:n - ll] += w * x[ll:]
    grad = dx * ((1.0 - S_plus) - S_minus)
    s = x.sum() - n / 2.0
    grad += 2e5 * s
    return grad

def lbfgs_run(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    target = n / 2.0
    x0 = h.copy()
    bounds = [(0.0, 1.0)] * n
    def fun(x):
        if time.time() > deadline:
            return logsumexp_obj(x, n, alpha, target) + 1e10
        return logsumexp_obj(x, n, alpha, target)
    def gfun(x):
        return logsumexp_grad(x, n, alpha)
    res = minimize(fun, x0, jac=gfun, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'maxfun': maxiter * 3})
    return project_feasible(res.x, n)

def adam_optimize(h, n, alpha, deadline, lr_init=0.05, beta1=0.9, beta2=0.999, eps=1e-8):
    dx = 2.0 / n
    target = n / 2.0
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    h = h.copy()
    while time.time() < deadline:
        t += 1
        grad = logsumexp_grad(h, n, alpha)
        lr = lr_init * np.sqrt(1.0 - beta2**t) / (1.0 - beta1**t)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h_new, n)
        if t % 500 == 0:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    c = compute_c5(h, n)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, best_c

def binary_init(h, n, rng):
    sorted_idx = np.argsort(-h)
    b = np.zeros(n)
    b[sorted_idx[:n // 2]] = 1.0
    return b

def coord_descent(h, n, deadline, max_sweeps=50):
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for sweep in range(max_sweeps):
        if time.time() > deadline - 5:
            break
        improved = False
        for i in range(n):
            old_val = h[i]
            h[i] = 0.0
            h = project_feasible(h, n)
            v0 = compute_c5(h, n)
            h[i] = 1.0
            h = project_feasible(h, n)
            v1 = compute_c5(h, n)
            if v0 <= v1:
                h[i] = 0.0
            else:
                h[i] = 1.0
            h = project_feasible(h, n)
            if abs(h[i] - old_val) > 1e-6:
                improved = True
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c
