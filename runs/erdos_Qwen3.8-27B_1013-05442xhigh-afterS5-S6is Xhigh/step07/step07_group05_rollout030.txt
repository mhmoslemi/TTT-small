import time
import numpy as np
import scipy.optimize as so

LS = {'n': 1, 'dx': 1.0, 'beta': 1.0}
SS = {'n': 1, 'dx': 1.0, 'beta': 1.0, 'cx': None, 'cg': None}


def erdos_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0)
    if abs(y.sum() - target) < 1e-13:
        return y
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        if y.sum() > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(5):
        d = target - y.sum()
        if abs(d) < 1e-14:
            break
        m = (y > 1e-12) & (y < 1.0 - 1e-12)
        c = int(m.sum())
        if c > 0:
            y[m] += d / c
        else:
            y += d / n
        y = np.clip(y, 0.0, 1.0)
    return y


def softmax_vals(c, beta):
    mc = float(np.max(c))
    z = beta * (c - mc)
    np.clip(z, -745.0, 0.0, out=z)
    ew = np.exp(z)
    ws = float(ew.sum())
    if ws < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ew / ws
    return mc + np.log(ws) / beta, w


def grad_smooth(h, n, beta):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    _, w = softmax_vals(c, beta)
    mid = n - 1
    if n == 1:
        return (1.0 - 2.0 * h) * w[0] * dx
    wp = np.zeros(n)
    wn = np.zeros(n)
    wp[1:] = w[mid + 1:]
    wn[1:] = w[:mid][::-1]
    s = wp + wn
    pp = np.cumsum(wp)
    pn = np.cumsum(wn)
    idx = np.arange(n)
    consts = pp + pn[n - 1 - idx]
    kern = np.zeros(2 * n - 1)
    kern[mid + 1:] = s[1:]
    kern[:mid][::-1] = s[1:]
    neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
    return (w[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx


def lbfgs_fun(x):
    st = LS
    n, beta = st['n'], st['beta']
    dx = st['dx']
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    f, _ = softmax_vals(c, beta)
    g = grad_smooth(x, n, beta)
    return f, g


def do_lbfgs(h, beta, maxiter, deadline):
    h = np.asarray(h, dtype=float)
    n = len(h)
    rem = deadline - time.time()
    if rem < 3.0:
        return h, erdos_c5(h, n)
    LS['n'], LS['dx'], LS['beta'] = n, 2.0 / n, float(beta)
    if n > 700:
        maxiter = max(10, maxiter // 2)
    mx = int(min(maxiter, max(8, int(rem / 2.5))))
    bounds = [(0.0, 1.0)] * n
    try:
        res = so.minimize(lbfgs_fun, h, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': mx, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 30})
        cand = project_box_sum(res.x, 0.5 * n)
    except Exception:
        cand = project_box_sum(h, 0.5 * n)
    return cand, erdos_c5(cand, n)


def sls_obj(x):
    st = SS
    cx = st.get('cx')
    if cx is not None and np.array_equal(x, cx):
        f, g = softmax_vals(np.correlate(x, 1.0 - x, mode='full') * st['dx'], st['beta'])
        g = grad_smooth(x, st['n'], st['beta'])
        st['cx'] = x.copy()
        st['cg'] = g
        return float(f)
    f = sls_full(x)
    return f


def sls_full(x):
    st = SS
    c = np.correlate(x, 1.0 - x, mode='full') * st['dx']
    f, _ = softmax_vals(c, st['beta'])
    st['cx'] = x.copy()
    return float(f)


def sls_jac(x):
    st = SS
    g = grad_smooth(x, st['n'], st['beta'])
    st['cg'] = g.copy()
    return g


def sls_con_fun(x):
    return float(x.sum() - SS['n'] * 0.5)


def sls_con_jac(x):
    return np.ones(SS['n'], dtype=float)


def do_slsqp(h, beta, maxiter, deadline):
    h = np.asarray(h, dtype=float)
    n = len(h)
    rem = deadline - time.time()
    if rem < 8.0:
        return h, erdos_c5(h, n)
    SS['n'], SS['dx'], SS['beta'] = n, 2.0 / n, float(beta)
    SS['cx'], SS['cg'] = None, None
    if n > 600:
        maxiter = max(5, maxiter // 2)
    mx = int(min(maxiter, max(5, int(rem / 10.0))))
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sls_con_fun, 'jac': sls_con_jac}]
    try:
        res = so.minimize(sls_obj, h, method='SLSQP', jac=sls_jac,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': mx, 'iprint': 0, 'ftol': 1e-14})
        cand = project_box_sum(res.x, 0.5 * n)
    except Exception:
        cand = project_box_sum(h, 0.5 * n)
    return cand, erdos_c5(cand, n)


def resample(h, n, nh):
    x = np.linspace(0.0, 2.0, int(n))
    y = np.linspace(0.0, 2.0, int(nh))
    return project_box_sum(np.interp(y, x, np.asarray(h, dtype=float)), 0.5 * nh)


def weighted_grad_step(h, n, beta, n_steps, lr, deadline):
    """Projected gradient with top-shift weighting."""
    h = np.asarray(h, dtype=float).copy()
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        # Weighted subgradient: emphasize top shifts
        top_k = max(3, n // 4)
        idx = np.argsort(c)[::-1][:top_k]
        weights = np.zeros_like(c)
        weights[idx] = 1.0 / top_k
        mid = n - 1
        dx = 2.0 / n
        wp = np.zeros(n)
        wn = np.zeros(n)
        wp[1:] = weights[mid + 1:]
        wn[1:] = weights[:mid][::-1]
        s = wp + wn
        pp = np.cumsum(wp)
        pn = np.cumsum(wn)
        ix = np.arange(n)
        consts = pp + pn[n - 1 - ix]
        kern = np.zeros(2 * n - 1)
        kern[mid + 1:] = s[1:]
        kern[:mid][::-1] = s[1:]
        neigh = np.convolve(h, kern, mode='full')[mid:mid + n]
        g = (weights[mid] * (1.0 - 2.0 * h) + consts - neigh) * dx
        h_new = np.clip(h - lr * g, 0.0, 1.0)
        h_new = project_box_sum(h_new, 0.5 * n)
        if erdos_c5(h_new, n) < erdos_c5(h, n) + 1e-15:
            h = h_new
    return h
