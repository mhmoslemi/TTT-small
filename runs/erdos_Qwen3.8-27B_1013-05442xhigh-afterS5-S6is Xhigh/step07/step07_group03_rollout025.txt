```python
def project_feasible(h, n):
    """Project h to feasible set: 0<=h<=1, sum(h)=n/2."""

    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def logsumexp_objective(h, n, alpha, corr_cache=None):
    """Log-sum-exp smoothed max of correlation."""
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    N = 1
    while N < 2*n:
        N *= 2
    H = np.fft.rfft(h, N)
    OH = np.fft.rfft(one_minus_h, N)
    corr = np.fft.irfft(H * np.conj(OH), N)[:2*n-1] * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    sm = shifted.max()
    logsum = sm + np.log(np.exp(shifted - sm).sum())
    return m + logsum / alpha, corr

        val, _ = logsumexp_objective(x, n, alpha)
        return val

        return x.sum() - n / 2.0

                   constraints=[{'type': 'eq', 'fun': cons_fun}],
                   options={'maxiter': maxiter, 'ftol': ftol})
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def corr_full(h, n, N):
    oh = 1.0 - h
    H = np.fft.rfft(h, N)
    OH = np.fft.rfft(oh, N)
    return np.fft.irfft(H * np.conj(OH), N)[:2*n-1]

def compute_c5(h, n, N):
    return float(corr_full(h, n, N).max() * (2.0/n))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def build_G(h, n):
    """Build the G matrix: G[p][j] = delta in corr when h[p] goes 0->1."""
    Ncorr = 2*n - 1
    G = np.zeros((n, Ncorr))
    lags = np.arange(-(n-1), n)
    for p in range(n):
        pk = p + lags
        mask1 = (pk >= 0) & (pk < n)
        G[p, mask1] += (1.0 - h[pk[mask1]])
        pl = p - lags
        mask2 = (pl >= 0) & (pl < n)
        G[p, mask2] -= h[pl[mask2]]
    return G

def update_G_swap(G, i, j, n):
    """Update G after swapping: h[i] 0->1, h[j] 1->0."""
    ncorr = 2*n - 1
    for q in range(n):
        idx1 = i - q + (n - 1)
        if 0 <= idx1 < ncorr:
            G[q, idx1] -= 1.0
        idx2 = q - i + (n - 1)
        if 0 <= idx2 < ncorr:
            G[q, idx2] -= 1.0
        idx3 = j - q + (n - 1)
        if 0 <= idx3 < ncorr:
            G[q, idx3] += 1.0
        idx4 = q - j + (n - 1)
        if 0 <= idx4 < ncorr:

def binary_ls(h_init, n, N, deadline, max_sweeps=500):
    dx = 2.0 / n
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:n//2]] = 1.0
    
    corr = corr_full(h, n, N)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    G = build_G(h, n)
    ncorr = 2*n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        improved = False
        
        for i in zeros:
            if time.time() > deadline:
                break
            if i not in zeros and h[i] == 1:
                continue  # i was already flipped this sweep
            delta = corr[None, :] + G[i][None, :] - G[ones, :, :]
            new_maxes = delta.max(axis=1)
            bi = new_maxes.argmin()
            if new_maxes[bi] < cur_max - 1e-10:
                j = ones[bi]
                corr += G[i] - G[j]
                cur_max = corr.max()
                h[i] = 1.0
                h[j] = 0.0
                update_G_swap(G, i, j, n)
                ones = np.where(h == 1)[0]
                improved = True
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
        if not improved:
            break
    
    return best_h, best_max * dx

def subgrad_polish(h, n, N, deadline, lr=0.3, beta=0.7, max_iter=50000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n, N)
    mom = np.zeros(n)

    dx = 2.0 / n
    def obj(x):
        c = corr_full(x, n, N) * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cfun(x):
        return x.sum() - n / 2.0
    try:

                       bounds=[(0.0,1.0)]*n,
                       constraints=[{'type':'eq','fun':cfun}],
                       options={'maxiter':maxiter, 'ftol':1e-16})
        return project_feasible(res.x, n)
    except:
        return h
```

```python
def build_G(h, n):
    ncorr = 2*n - 1
    lags = np.arange(-(n-1), n)
    # For position p, first term: (1-h[p+lags]) where 0<=p+lags<n
    # p+lags ranges from p-(n-1) to p+(n-1)
    # We need a (n, ncorr) matrix where entry [p, j] = h[p + (j-(n-1))] if valid
    
    # Create index matrix
    p_idx = np.arange(n)[:, None]
    lag_idx = (np.arange(ncorr) - (n-1))[None, :]
    pk = p_idx + lag_idx  # shape (n, ncorr)
    mask1 = (pk >= 0) & (pk < n)
    
    pl = p_idx - lag_idx  # shape (n, ncorr)
    mask2 = (pl >= 0) & (pl < n)
    
    # Gather h values
    # For valid indices, h[pk]
    h_expanded = h[np.clip(pk, 0, n-1)]  # shape (n, ncorr), with garbage for invalid
    G = (1.0 - h_expanded) * mask1 - h[np.clip(pl, 0, n-1)] * mask2
    return G
```

- Creating index arrays: O(n * ncorr) = O(n²) memory
- Masking: O(n²)
- Gathering h values: O(n²)
- Final computation: O(n²)

For n=400: 400 * 799 ≈ 320K elements. A few array operations on 320K elements takes ~1ms in numpy.

```python
import numpy as np
from scipy.optimize import minimize
import time

def corr_full(h, n, N):
    oh = 1.0 - h
    H = np.fft.rfft(h, N)
    OH = np.fft.rfft(oh, N)
    return np.fft.irfft(H * np.conj(OH), N)[:2*n-1]

def compute_c5(h, n, N):
    return float(corr_full(h, n, N).max() * (2.0/n))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def build_G(h, n):
    ncorr = 2*n - 1
    p_idx = np.arange(n)[:, None]
    lag_idx = (np.arange(ncorr) - (n-1))[None, :]
    pk = p_idx + lag_idx
    mask1 = (pk >= 0) & (pk < n)
    pl = p_idx - lag_idx
    mask2 = (pl >= 0) & (pl < n)
    h_pk = h[np.clip(pk, 0, n-1)]
    h_pl = h[np.clip(pl, 0, n-1)]
    return (1.0 - h_pk) * mask1 - h_pl * mask2

def update_G_after_swap(G, i, j, n):
    ncorr = 2*n - 1
    q = np.arange(n)
    idx_a = i - q + (n-1)
    va = (idx_a >= 0) & (idx_a < ncorr)
    G[va, idx_a[va]] -= 1.0
    idx_b = q - i + (n-1)
    vb = (idx_b >= 0) & (idx_b < ncorr)
    G[vb, idx_b[vb]] -= 1.0
    idx_c = j - q + (n-1)
    vc = (idx_c >= 0) & (idx_c < ncorr)
    G[vc, idx_c[vc]] += 1.0
    idx_d = q - j + (n-1)
    vd = (idx_d >= 0) & (idx_d < ncorr)
    G[vd, idx_d[vd]] += 1.0

def binary_ls(h_init, n, N, deadline, max_sweeps=500):
    dx = 2.0 / n
    h = np.zeros(n)
    idx = np.argsort(h_init)[::-1]
    h[idx[:n//2]] = 1.0
    corr = corr_full(h, n, N)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    G = build_G(h, n)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            if h[i] == 1:
                continue
            if len(ones) == 0:
                break
            delta = corr[None, :] + G[i][None, :] - G[ones, :, :]
            new_maxes = delta.max(axis=1)
            bi = new_maxes.argmin()
            if new_maxes[bi] < cur_max - 1e-10:
                j = ones[bi]
                corr += G[i] - G[j]
                cur_max = corr.max()
                h[i] = 1.0
                h[j] = 0.0
                update_G_after_swap(G, i, j, n)
                ones = np.where(h == 1)[0]
                improved = True
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
        if not improved:
            break
    return best_h, float(best_max * dx)

def subgrad_polish(h, n, N, deadline, lr=0.3, beta=0.7, max_iter=50000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n, N)
    mom = np.zeros(n)
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr_v = corr_full(h, n, N) * dx
        cur_c = float(corr_v.max())
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 3000:
            break
        top_k = 3
        top_idx = np.argsort(corr_v)[::-1][:top_k]
        grad = np.zeros(n)
        for j in top_idx:
            lag = int(j) - (n - 1)
            if lag >= 0:
                if lag == 0:
                    grad += 1.0 - 2.0 * h
                else:
                    grad[:n-lag] += (1.0 - h[lag:])
                    grad[lag:] -= h[:n-lag]
            else:
                kk = -lag
                grad[n-kk:] += (1.0 - h[:kk])
                grad[:n-kk] -= h[kk:]
        grad = grad / top_k * dx
        mom = beta * mom + (1.0 - beta) * grad
        step = lr / (1.0 + 0.05 * (it + 1))
        h = project_feasible(h - step * mom, n)
    return best_h, best_c

def slsqp_run(h, n, N, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        c = corr_full(x, n, N) * dx
        m = float(c.max())
        shifted = alpha * (c - m)
        sm = float(shifted.max())
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cfun(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cfun}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h
```
