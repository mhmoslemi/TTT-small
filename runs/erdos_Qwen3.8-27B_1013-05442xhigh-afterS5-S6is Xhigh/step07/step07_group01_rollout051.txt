                  alpha_start=100, alpha_end=1000000, beta1=0.9, beta2=0.999):

    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    total_steps = 500000
    for step in range(total_steps):
        if time.time() > deadline:
            break
        
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full') * dx
        c_max = corr.max()
        
        # Adaptive alpha
        if alpha_schedule == 'linear':
            alpha = alpha_start * (alpha_end / alpha_start) ** (step / total_steps)
        elif alpha_schedule == 'exp':
            alpha = alpha_start * np.exp(step / total_steps * np.log(alpha_end / alpha_start))

        weights = np.exp(shifted)

    if c < best_c:
        best_c = c
        best_h = h.copy()
    
    return best_h, best_c

        best_c = c
        best_h = h.copy()

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def grad_top_k(h, n, dx, top_indices, weights):
    grad = np.zeros(n)
    for idx, w in zip(top_indices, weights):
        lag = idx - (n - 1)
        if lag > 0:
            grad[:n-lag] += w * dx * (1.0 - h[lag:])
            grad[lag:] += w * dx * (-h[:n-lag])
        elif lag < 0:
            ell = -lag
            grad[ell:] += w * dx * (1.0 - h[:n-ell])
            grad[:n-ell] += w * dx * (-h[ell:])
        else:
            grad[:] += w * dx * (1.0 - 2.0 * h)
    return grad

    best_c = compute_c5(best_h, n)

    while time.time() < deadline:
        corr = np.correlate(h, 1 - h, mode='full') * dx
        c_max = corr.max()

        (4, 0.008, 200, 1.015),
        (10, 0.003, 80, 1.010),
    ]
    for (tk, lr, ai, ag) in configs:
        if time.time() > p2 - 30:
            break
        chunk_end = min(time.time() + 50, p2 - 10)
        h_try, c_try = adam_run(best_h.copy(), n, chunk_end,
                                 top_k=tk, lr=lr, alpha_init=ai, alpha_growth=ag)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()

    # Phase 3: Upsample to finer grid for better discretization (380-680s)
    p3 = t0 + 680
    for n_fine in [800, 1000]:
        if time.time() > p3 - 60:
            break
        h_fine = interpolate_h(best_h, n, n_fine)
        h_fine = project_feasible(h_fine, n_fine)
        chunk_end = min(time.time() + 100, p3 - 10)
        h_fine, c_fine = adam_run(h_fine, n_fine, chunk_end,
                                   top_k=10, lr=0.003, alpha_init=50, alpha_growth=1.007)
        # Interpolate back to n
        h_back = interpolate_h(h_fine, n_fine, n)
        h_back = project_feasible(h_back, n)
        c_back = compute_c5(h_back, n)
        if c_back < best_c:
            best_c = c_back
            best_h = h_back.copy()
        # Also try the fine grid solution directly
        if c_fine * n_fine / 2.0 < best_c * n / 2.0:
            pass  # fine grid solution is better in absolute terms but we need to report at its own resolution
            h_fine_final = project_feasible(h_fine, n_fine)
            c_ff = compute_c5(h_fine_final, n_fine)
            if c_ff < best_c:
                best_c = c_ff
                best_h = h_fine_final.copy()
                n = n_fine

    # Phase 4: SLSQP polish (680-920s)
    p4 = t0 + 920
    alphas = [3000, 30000, 300000, 3000000]
    for a in alphas:
        if time.time() > p4 - 15:
            break
        h_try = slsqp_polish(best_h, n, a, 1500, 1e-15)
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()

    # Phase 5: Noise perturbations + Adam (920-1040s)
    p5 = t0 + 1040
    for trial in range(8):
        if time.time() > p5 - 10:
            break
        noise = 0.004 * (1.0 + trial * 0.4)
        h_try = best_h + noise * rng.randn(n)
        h_try = project_feasible(h_try, n)
        chunk_end = min(time.time() + 12, p5 - 3)
        h_try, c_try = adam_run(h_try, n, chunk_end, top_k=8, lr=0.004,
                                 alpha_init=100, alpha_growth=1.01)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()

    # Phase 6: Final SLSQP polish
    if time.time() < deadline - 15:
        h_try = slsqp_polish(best_h, n, 5000000, 2500, 1e-16)
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
