def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def slsqp_obj_grad(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    p = np.exp(sh - sm)
    psum = p.sum()
    if psum < 1e-30:
        return m, np.zeros(n)
    p = p / psum
    val = m + (sm + np.log(psum)) / alpha
    grad = np.zeros(n)
    for ki in range(len(p)):
        w = p[ki]
        if w < 1e-10:
            continue
        lag = ki - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[:n - lag] += w * (1.0 - x[lag:]) * dx
                grad[lag:] -= w * x[:n - lag] * dx
        else:
            ll = -lag
            if ll < n:
                grad[ll:] += w * (1.0 - x[:n - ll]) * dx
                grad[:n - ll] -= w * x[ll:] * dx
    return val, grad

    y = h.copy()

            break

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def slsqp_obj_grad(x, n, alpha, topm=20):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    p = np.exp(sh - sm)
    psum = p.sum()
    if psum < 1e-30:
        return m, np.zeros(n)
    p = p / psum
    val = m + (sm + np.log(psum)) / alpha
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    if len(active) > topm:
        active = active[np.argsort(p[active])[::-1][:topm]]
    for ki in active:
        w = p[ki]
        lag = ki - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[:n - lag] += w * (1.0 - x[lag:]) * dx
                grad[lag:] -= w * x[:n - lag] * dx
        else:
            ll = -lag
            if ll < n:
                grad[ll:] += w * (1.0 - x[:n - ll]) * dx
                grad[:n - ll] -= w * x[ll:] * dx
    return val, grad

def slsqp_run(h, n, alpha, maxiter, ftol, deadline=None):
    def obj(x):
        if deadline is not None and time.time() > deadline:
            return 1e10
        v, _ = slsqp_obj_grad(x, n, alpha)
        return v
    def objg(x):
        if deadline is not None and time.time() > deadline:
            return 1e10, np.zeros(n)
        return slsqp_obj_grad(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(objg, h, jac=True, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def nesterov_subgrad(h, n, deadline, topk=3):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    y = h.copy()
    lr = dx * 3.0
    stall = 0
    for step in range(1000000):
        if time.time() > deadline:
            break
        c = np.correlate(y, 1 - y, mode='full') * dx
        idx = np.argsort(c)[::-1][:topk]
        grad = np.zeros(n)
        for ki in idx:
            lag = int(ki) - (n - 1)
            if lag >= 0:
                if lag < n:
                    grad[:n - lag] += (1.0 - y[lag:])
                    grad[lag:] -= y[:n - lag]
            else:
                ll = -lag
                if ll < n:
                    grad[ll:] += (1.0 - y[:n - ll])
                    grad[:n - ll] -= y[ll:]
        grad *= dx / topk
        t_acc = (1.0 + np.sqrt(1.0 + 4.0 * step * step)) / 2.0
        mom = (t_acc - 1.0) / (t_acc + step)
        h = project_feasible(y - lr * grad + mom * (y - h), n)
        y = h.copy()
        cv = compute_c5(h, n)
        if cv < best_c:
            best_c = cv
            best_h = h.copy()
            stall = 0
            lr = min(lr * 1.005, dx * 25.0)
        else:
            stall += 1
            if stall % 200 == 0:
                lr *= 0.93
                if lr < dx * 0.05:
                    lr = dx * 0.5
                    stall = 0
    return best_h, best_c

def upsample(h_coarse, n_fine):
    x_c = np.linspace(0, 2, len(h_coarse))
    x_f = np.linspace(0, 2, n_fine)
    return np.interp(x_f, x_c, h_coarse)

def coord_refine(h, n, deadline, max_sweeps=200):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        changed = False
        for i in range(n):
            old = h[i]
            best_t = old
            best_v = best_c
            h[i] = 0.0
            v0 = compute_c5(h, n)
            h[i] = 1.0
            v1 = compute_c5(h, n)
            if v0 < best_v:
                best_v, best_t = v0, 0.0
            if v1 < best_v:
                best_v, best_t = v1, 1.0
            if best_t != old:
                # Bisect between best_t and old
                lo, hi = min(best_t, old), max(best_t, old)
                for _ in range(8):
                    mid = (lo + hi) / 2.0
                    h[i] = mid
                    vm = compute_c5(h, n)
                    if vm < best_v:
                        best_v, best_t = vm, mid
                        hi = mid if best_t > old else lo
                        lo = mid if best_t > old else hi
                    else:
                        lo = mid if best_t < mid else hi
                        hi = mid if best_t < mid else lo
                h[i] = best_t
                changed = True
            else:
                h[i] = old
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if not changed:
            break
    return best_h, best_c
