```python
def subgradient_nesterov(h, n, max_iter=5000, deadline=None):
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    h = h.astype(float).copy()
    
    # Nesterov momentum
    h_nest = h.copy()
    beta = 0.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if deadline and time.time() > deadline:
            break
        
        # Compute correlation at h_nest
        c = np.correlate(h_nest, 1 - h_nest, mode='full') * dx
        
        # Find active lags (within epsilon of max)
        c_max = c.max()
        active = np.where(c > c_max - 0.01)[0]
        
        # Compute subgradient (average of active subgradients)
        grad = np.zeros(n)
        for k_idx in active:
            k = k_idx - (n - 1)  # lag
            # grad[j] = (1 - h_nest[j+k]) for j+k in [0,n) minus h_nest[j-k] for j-k in [0,n)
            for j in range(n):
                g = 0.0
                if j + k >= 0 and j + k < n:
                    g += (1.0 - h_nest[j + k])
                if j - k >= 0 and j - k < n:
                    g -= h_nest[j - k]
                grad[j] += g
        grad /= len(active)
        
        # Nesterov scheme
        beta_old = beta
        beta = (1 + np.sqrt(1 + 4 * beta_old**2)) / 2
        step = 2.0 / (beta + 1)
        
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        
        h_nest = h_new + (beta_old / beta) * (h_new - h)
        h_nest = project_feasible(h_nest, n)
        h = h_new
        
        c_val = compute_c5(h, n)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
    
    return best_h, best_c
```

```python
    """Binary simulated annealing with first-improvement."""
    h = to_binary(h, n).copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    
    zeros = list(np.where(h == 0)[0])
    ones = list(np.where(h == 1)[0])
    
    T_start = cur_max * 0.01
    T_end = cur_max * 0.0001
    n_its = max_iters
    rng = np.random.RandomState(42)
    
    for it in range(n_its):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (it / n_its)
        
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        d = swap_delta(h, i, j, n)
        new_max = (corr + d).max()
        
        if new_max < cur_max or rng.random() < np.exp(-(new_max - cur_max) / T):
            h[i], h[j] = 1.0, 0.0
            corr += d
            cur_max = new_max
            ones.remove(j); ones.append(i)
            zeros.remove(i); zeros.append(j)
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
    
    return best_h, best_max * dx
```
