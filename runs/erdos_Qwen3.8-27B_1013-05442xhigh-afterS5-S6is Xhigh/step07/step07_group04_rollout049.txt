        if nh == n0:
            time_alloc[nh] = 0.45
        else:
            time_alloc[nh] = 0.55 / max(1, L-1)
    for nh in res_list:
        if time.time() > deadline - 15: break
        alloc = time_alloc[nh] * (deadline - t0)
        dl_phase = t0 + alloc
        dl_phase = min(dl_phase, deadline - 10)
        if dl_phase <= time.time() + 8: continue
        if nh == n0:
            hinit = besth.copy()
        else:
            hinit = interp_h(besth, bestn, nh)
            hinit = project_box_sum(hinit + 0.003 * rng.randn(nh), nh)
        hopt, copt = optimize_at_n(hinit, nh, seed + 13*nh, dl_phase)
        if copt < bestc:
            bestc, besth, bestn = copt, hopt.copy(), nh
    # Final polish on best
    if time.time() < deadline - 10:
        hfin, cfin = focused_pair_search(besth, bestn, deadline - 3, seed + 777)
        if cfin < bestc: bestc, besth = cfin, hfin
        hfin, cfin = sweep_top_distances(besth, bestn, deadline - 2, top_k=5)
        if cfin < bestc: bestc, besth = cfin, hfin
    besth = project_box_sum(besth, bestn)
    bestc = float(compute_c(besth, bestn).max())
    return besth, bestc, bestn
