    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)

    y = beta * (c - m)

    B = np.convolve(P, h, mode="full")[n-1:2*n-1]

    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n: a = np.concatenate([a, np.zeros(n - a.size)])
        else: a = a[:n]

    target = 0.5 * n

        else: hi = mid

            if s < 1e-15: break
