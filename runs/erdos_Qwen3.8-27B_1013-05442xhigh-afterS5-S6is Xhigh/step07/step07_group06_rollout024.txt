For a binary swap (i: 1→0, j: 0→1), the change in the correlation at lag k is:
- At lag k = j - i: was h[i]=1, h[i+k]=h[j]=0, so contributes 1 to "overlap" (1*1). After swap: h[i]=0, h[j]=1, so contributes 0. Change: -1.

Let me be more careful. The full correlation at lag k (where k can be negative) is:
corr[k] = sum_i h[i] * (1 - h[i+k]) for valid i

When we swap h[i] from 1 to 0 and h[j] from 0 to 1:
- For each lag k, the terms involving position i change:
  - Term h[i]*(1-h[i+k]): goes from 1*(1-h[i+k]) to 0*(1-h[i+k]) = 0. Change: -(1-h[i+k]) = h[i+k]-1
  - Term h[i-k]*(1-h[i]): goes from h[i-k]*1 to h[i-k]*0 = 0. Change: -h[i-k]
- For each lag k, the terms involving position j change:
  - Term h[j]*(1-h[j+k]): goes from 0 to 1*(1-h[j+k]) = 1-h[j+k]. Change: 1-h[j+k]
  - Term h[j-k]*(1-h[j]): goes from h[j-k]*0 to h[j-k]*1 = h[j-k]. Change: h[j-k]

So the total change in corr[k] for each k:
Δcorr[k] = (h[i+k]-1) + (-h[i-k]) + (1-h[j+k]) + h[j-k]
         = h[i+k] - h[i-k] - h[j+k] + h[j-k]

where we use h[x]=0 for out-of-bounds x.

This is O(n) per swap (we need to check all 2n-1 lags), which is much faster than O(n^2) for full correlation.

from scipy.optimize import minimize
import time

_stop_time = 0.0

def stop_callback(*args):
    return time.time() > _stop_time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def top_n_binary(h, n):
    h = np.asarray(h, dtype=float).ravel()
    target = n // 2
    hb = np.zeros(n)
    if target > 0:
        idx = np.argsort(-h)[:target]
        hb[idx] = 1.0
    return hb

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(max(0, n - a.size))])
        else:
            a = a[:n]
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo = float(np.min(a)) - 1.0
    hi = float(np.max(a)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(a - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15:
                break
            h += diff * room / s
        else:
            s = h.sum()
            if s <= 1e-15:
                break
            h += diff * h / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)

def swap_delta(h, n, i, j):
    k = np.arange(-(n - 1), n)
    hik, himk, hjk, hjmk = i + k, i - k, j + k, j - k
    result = np.zeros(2 * n - 1)
    ma = (hik >= 0) & (hik < n)
    mb = (himk >= 0) & (himk < n)
    mc = (hjk >= 0) & (hjk < n)
    md = (hjmk >= 0) & (hjmk < n)
    result[ma] += h[hik[ma]]
    result[mb] += h[himk[mb]]
    result[mc] -= h[hjk[mc]]
    result[md] -= h[hjmk[md]]
    return result

def binary_sa(h_init, n, deadline, rng, t_start=0.8, t_end=0.01, cooling=0.9996, max_iters=80000):
    h = top_n_binary(h_init, n).copy()
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    best_h, best_M, curr_M = h.copy(), float(corr.max()), float(corr.max())
    T = t_start
    ones_list = list(np.nonzero(h == 1)[0])
    zeros_list = list(np.nonzero(h == 0)[0])
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = int(ones_list[rng.randint(len(ones_list))])
        j = int(zeros_list[rng.randint(len(zeros_list))])
        delta = swap_delta(h, n, i, j)
        new_M = float((corr + delta).max())
        d = new_M - curr_M
        if d <= 0 or (T > 0.005 and rng.random() < np.exp(min(0.0, -d / T))):
            h[i] = 0.0
            h[j] = 1.0
            corr = corr + delta
            curr_M = new_M
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
            if curr_M < best_M:
                best_M = curr_M
                best_h = h.copy()
        T = max(T * cooling, t_end)
    return best_h, best_M * (2.0 / float(n))

def binary_greedy(h_init, n, deadline, max_passes=50):

    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)

                    best_h = h.copy()
                    improved = True

def lse_obj_grad(h, n, beta):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h

    c = np.maximum(c, 0.0)
    m = float(c.max())

    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    R = p[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(p, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g

def lbfgs_obj(x, n, beta, pen):
    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    F, g = lse_obj_grad(h, n, beta)
    dev = float(h.sum() - 0.5 * n)
    if pen != 0.0:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g

                           options={"maxiter": 500000, "maxls": 50, "ftol": 1e-13, "gtol": 1e-8})
            x = res.x if (res is not None and hasattr(res, "x")) else work

            x = work
        work = project_feasible(x, n)
        cw = compute_c5(work, n)
        if cw < bc:
            bc, best = cw, work.copy()
    return best, bc
