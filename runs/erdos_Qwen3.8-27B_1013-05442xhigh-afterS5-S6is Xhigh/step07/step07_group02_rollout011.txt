def hessian_v(h, n, alpha, v):
    """Compute Hessian-vector product for the smoothed objective."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = c.max()
    p = np.exp(alpha * (c - max_c))
    p /= p.sum()
    
    # H * v = dx * sum_k p[k] * (-(v shifted by k) - (v shifted by -k))
    # For each k, the "shifted by k" means: (Tv)[i] = v[i+k] for valid i
    result = np.zeros(n)
    for m in range(2*n-1):
        if p[m] < 1e-16:
            continue
        k = m - (n-1)
        # v shifted by +k: result[i] += v[i+k] for valid i
        i1s = max(0, -k)
        i1e = min(n, n-k) if k > 0 else n
        if i1e > i1s:
            result[i1s:i1e] -= p[m] * v[i1s+k:i1e+k]
        # v shifted by -k: result[i] += v[i-k] for valid i
        i2s = max(0, k)
        i2e = min(n, n+k) if k < 0 else n
        if i2e > i2s:
            result[i2s:i2e] -= p[m] * v[i2s-k:i2e-k]
    result *= dx
    return result
