    p4_end = t0 + 920
    if time.time() < p4_end:
        h4 = best_h.copy()
        h4, c4 = anneal_blocks(h4, n, rng, p4_end, T_init=0.03, T_final=0.0005)
        if c4 < best_c:
            best_c, best_h = c4, h4.copy()
        # Quick SLSQP polish after annealing
        if time.time() < p4_end - 5:
            h4 = slsqp_wov(h4, n, 30, 0.0005, maxiter=400)
            c4 = compute_c5(h4, n)
            if c4 < best_c:
                best_c, best_h = c4, h4.copy()

    # Phase 5: noise restarts with quick re-optimization
    p5_end = deadline - 20
    for trial in range(10):
        if time.time() > p5_end - 5:
            break
        noise_scale = 0.005 + trial * 0.003
        h5 = best_h + noise_scale * rng.randn(n)
        h5 = project_feasible(h5, n)
        h5 = slsqp_wov(h5, n, max(20, n // 2), 0.005, maxiter=300)
        h5 = slsqp_wov(h5, n, 20, 0.0005, maxiter=300)
        c5 = compute_c5(h5, n)
        if c5 < best_c:
            best_c, best_h = c5, h5.copy()

    # Phase 6: final polish
    if time.time() < deadline - 5:
        hf = best_h.copy()
        hf = slsqp_wov(hf, n, 20, 0.0001, maxiter=500)
        hf = slsqp_wov(hf, n, 10, 0.00005, maxiter=500)
        hf, cf = block_cd(hf, n, rng, deadline - 3, block_sizes=(3, 5))
        if cf < best_c:
            best_c, best_h = cf, hf.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
