    s1_min = max(-nm1, -i)
    s1_max = min(nm1, nm1 - i)
    if s1_min <= s1_max:
        ss = np.arange(s1_min, s1_max + 1)
        delta[ss + nm1] -= (1.0 - h[i + ss])
    s2_min = max(-nm1, i - nm1)
    s2_max = min(nm1, i)
    if s2_min <= s2_max:
        ss = np.arange(s2_min, s2_max + 1)
        delta[ss + nm1] += h[i - ss]
    s3_min = max(-nm1, -j)
    s3_max = min(nm1, nm1 - j)
    if s3_min <= s3_max:
        ss = np.arange(s3_min, s3_max + 1)
        delta[ss + nm1] += (1.0 - h[j + ss])
    s4_min = max(-nm1, j - nm1)
    s4_max = min(nm1, j)
    if s4_min <= s4_max:
        ss = np.arange(s4_min, s4_max + 1)
        delta[ss + nm1] -= h[j - ss]
    return delta

def binary_search(h_init, n, deadline, mode='first', max_passes=200):
    h_b = to_binary(h_init, n).copy()
    corr = np.correlate(h_b, 1 - h_b, mode='full').astype(float)
    dx = 2.0 / n
    cur_c = corr.max() * dx
    best_h = h_b.copy()
    best_c = cur_c
    ones = [x for x in range(n) if h_b[x] == 1]
    zeros = [x for x in range(n) if h_b[x] == 0]
    rng = np.random.RandomState(123)
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if mode == 'best':
            best_nc = cur_c
            best_ij = None
            for i in zeros:
                for j in ones:
                    d = swap_delta_vec(i, j, h_b, n)
                    nc = (corr + d).max() * dx
                    if nc < best_nc - 1e-12:
                        best_nc = nc
                        best_ij = (i, j, d)
            if best_ij is None:
                break
            i, j, d = best_ij
        else:  # first improvement
            zi = zeros[:]
            oi = ones[:]
            rng.shuffle(zi)
            rng.shuffle(oi)
            found = False
            for i in zi:
                if found: break
                if time.time() > deadline: break
                for j in oi:
                    d = swap_delta_vec(i, j, h_b, n)
                    nc = (corr + d).max() * dx
                    if nc < cur_c - 1e-12:
                        # apply
                        h_b[i] = 0.0; h_b[j] = 1.0
                        corr += d
                        cur_c = nc
                        ones.remove(j); ones.append(i)
                        zeros.remove(i); zeros.append(j)
                        improved = True
                        found = True
                        break
            if not improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h_b.copy()
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    h_b = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_b[idx[:n // 2]] = 1.0
    return h_b

def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log((np.exp(shifted - sm) + 1e-300).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def swap_delta(i, j, h, n):
    d = np.zeros(2 * n - 1)
    m = n - 1
    a = max(-m, -i); b = min(m, m - i)
    if a <= b:
        s = np.arange(a, b + 1)
        d[s + m] -= 1.0 - h[i + s]
    a = max(-m, i - m); b = min(m, i)
    if a <= b:
        s = np.arange(a, b + 1)
        d[s + m] += h[i - s]
    a = max(-m, -j); b = min(m, m - j)
    if a <= b:
        s = np.arange(a, b + 1)
        d[s + m] += 1.0 - h[j + s]
    a = max(-m, j - m); b = min(m, j)
    if a <= b:
        s = np.arange(a, b + 1)
        d[s + m] -= h[j - s]
    return d

def bin_local(h_init, n, deadline, best_mode=True, max_passes=100):
    h_b = to_binary(h_init, n).copy()
    corr = np.correlate(h_b, 1.0 - h_b, mode='full').astype(float)
    dx = 2.0 / n
    cur = corr.max() * dx
    best_h = h_b.copy()
    best_c = cur
    ones = [x for x in range(n) if h_b[x] == 1]
    zeros = [x for x in range(n) if h_b[x] == 0]
    rng = np.random.RandomState(7)
    for p in range(max_passes):
        if time.time() > deadline:
            break
        if best_mode:
            bn = cur; bij = None
            for i in zeros:
                for j in ones:
                    d = swap_delta(i, j, h_b, n)
                    nc = (corr + d).max() * dx
                    if nc < bn - 1e-12:
                        bn = nc; bij = (i, j, d)
            if bij is None:
                break
            i, j, d = bij
        else:
            zi = zeros[:]; oi = ones[:]
            rng.shuffle(zi); rng.shuffle(oi)
            found = False
            for i in zi:
                if found or time.time() > deadline:
                    break
                for j in oi:
                    d = swap_delta(i, j, h_b, n)
                    nc = (corr + d).max() * dx
                    if nc < cur - 1e-12:
                        h_b[i] = 0.0; h_b[j] = 1.0
                        corr += d; cur = nc
                        ones.remove(j); ones.append(i)
                        zeros.remove(i); zeros.append(j)
                        found = True
                        break
            if not found:
                break
        if cur < best_c:
            best_c = cur; best_h = h_b.copy()
    return best_h, best_c
