min_h max_k g_k(h) s.t. constraints

The Lagrangian dual is:
max_{λ >= 0, ∑λ = 1} min_h [∑ λ_k g_k(h) + constraints]

For fixed λ, the inner problem is minimizing a weighted sum of quadratic functions, which is a QP. This could potentially be faster than the smooth max approach.

For fixed λ:
min_h ∑_k λ_k ∫ h(x)(1-h(x+k)) dx s.t. 0<=h<=1, ∫h=1

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_val_grad(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1.0 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    p = np.exp(sh - sm)
    ps = p.sum()
    p = p / ps
    val = m + (sm + np.log(ps)) / alpha
    grad = np.zeros(n)
    n2 = 2 * n - 1
    for li in range(n2):
        l = li - (n - 1)
        w = p[li]
        if w < 1e-10:
            continue
        if l >= 0:
            if l < n:
                grad[:n - l] += w * (1.0 - x[l:]) * dx
                grad[l:] -= w * x[:n - l] * dx
        else:
            ll = -l
            if ll < n:
                grad[ll:] += w * (1.0 - x[:n - ll]) * dx
                grad[:n - ll] -= w * x[ll:] * dx
    return val, grad

def slsqp_run(h, n, alpha, maxiter, ftol):

        return smooth_val_grad(x, n, alpha)[0]

        return smooth_val_grad(x, n, alpha)[1]
