```python
def compute_gradient(h, p, dx, n):
    """Compute gradient of smooth objective.
    p: softmax weights over shifts, length 2n-1
    """
    grad = np.zeros(n)
    # For each shift index m, k = m - (n-1)
    # Contribution to grad[j]: p[m] * ((1-h[j+k])*I(0<=j+k<n) - h[j-k]*I(0<=j-k<n))
    
    # Vectorize over m:
    # For a fixed m with k = m-(n-1):
    #   valid_j_for_plus = range(max(0,-k), min(n, n-k))  # where j+k is valid
    #   grad[valid_j_for_plus] += p[m] * (1 - h[valid_j_for_plus + k])
    #   valid_j_for_minus = range(max(0,k), min(n, n+k))  # where j-k is valid
    #   grad[valid_j_for_minus] -= p[m] * h[valid_j_for_minus - k]
    
    # This is O(n * 2n) = O(n^2) which for n=400 is 320K ops (fast)
    # For n=2000 it's 8M ops (still OK in a loop if vectorized per-m)
    
    # But a Python loop over 2n-1 iterations with numpy operations inside
    # is still fast for n up to a few thousand.
    
    for m in range(2 * n - 1):
        k = m - (n - 1)
        pm = p[m]
        if abs(pm) < 1e-15:
            continue
        # Plus term: (1-h[j+k]) for 0<=j+k<n => -k <= j < n-k
        j_start = max(0, -k)
        j_end = min(n, n - k)
        if j_end > j_start:
            grad[j_start:j_end] += pm * (1.0 - h[j_start + k:j_end + k])
        # Minus term: -h[j-k] for 0<=j-k<n => k <= j < n+k
        j_start2 = max(0, k)
        j_end2 = min(n, n + k)
        if j_end2 > j_start2:
            grad[j_start2:j_end2] -= pm * h[j_start2 - k:j_end2 - k]
    
    return dx * grad
```

This is O(n^2) but with small constant (each iteration is a vectorized numpy op of length O(n)). For n=400 and 799 iterations, that's about 799 vectorized ops of length ~200-400, which is very fast.

For n=2000, it's 3999 ops of length ~1000-2000, which should still be fast (maybe 1-2 ms total).

```python
import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize_scalar

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    f = corr * dx
    m = f.max()
    exp_f = np.exp(alpha * (f - m))
    Z = exp_f.sum()
    F = (m + np.log(Z)) / alpha  # Wait, this isn't right
    # F = m + log(Z)/alpha where Z = sum exp(alpha*(f-m))
    # = m + (1/alpha)*log(sum exp(alpha*(f-m)))
    F = m + np.log(Z) / alpha
    p = exp_f / Z  # softmax weights
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        k = mm - (n - 1)
        pm = p[mm]
        if pm < 1e-14: continue
        js = max(0, -k)
        je = min(n, n - k)
        if je > js:
            grad[js:je] += pm * (1.0 - h[js + k:je + k])
        js2 = max(0, k)
        je2 = min(n, n + k)
        if je2 > js2:
            grad[js2:je2] -= pm * h[js2 - k:je2 - k]
    grad *= dx
    return F, grad

def projected_grad_descent(h, n, alpha, time_limit, lr=0.01, momentum=0.9):
    v = np.zeros(n)
    t_start = time.time()
    best_h = h.copy()
    best_val = smooth_obj_grad(h, n, alpha)[0]
    for it in range(100000):
        if time.time() - t_start > time_limit:
            break
        F, g = smooth_obj_grad(h, n, alpha)
        if F < best_val:
            best_val = F
            best_h = h.copy()
        # Project gradient onto tangent space (remove mean component for equality constraint)
        g_proj = g - g.mean()
        # Momentum
        v = momentum * v - lr * g_proj
        # Step
        h_new = h + v
        h_new = project_feasible(h_new, n)
        # Check if we made progress
        F_new = compute_c5(h_new, n)  # Use the actual c5 for tracking
        if F_new > best_val * 1.5 and it > 100:
            # Reset
            v = np.zeros(n)
            h = best_h.copy()
            lr *= 0.5
        h = h_new
        # Adaptive learning rate
        if it % 100 == 0:
            lr *= 0.999
    return h, best_h

def coordinate_descent_step(h, n, time_limit):
    t_start = time.time()
    h = h.copy()
    alpha = 10000  # Use a high alpha for the smooth approximation in coord descent
    for sweep in range(10000):
        if time.time() - t_start > time_limit:
            break
        for j in range(n):
            if time.time() - t_start > time_limit:
                break
            # Optimize h[j] given all other values
            def obj_1d(val):
                h[j] = val
                corr = np.correlate(h, 1.0 - h, mode='full')
                f = corr * (2.0 / n)
                m = f.max()
                exp_f = np.exp(alpha * (f - m))
                return m + np.log(exp_f.sum()) / alpha
            res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 50, 'xatol': 1e-10})
            h[j] = res.x
        # Project after each sweep
        h = project_feasible(h, n)
    return h

def lp_step(h, n, eps):

    C = np.correlate(h, 1.0 - h, mode='full')
    mc = float(C.max())
    # Compute gradient matrix columns

        col = np.zeros(n)
        js = max(0, -k)

        if je > js:

        js2 = max(0, k)
        je2 = min(n, n + k)
        if je2 > js2:
            col[js2:je2] -= h[js2 - k:je2 - k]
        G_cols.append(col)
    G = np.array(G_cols).T  # shape (n, 2n-1)
    
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        dx * C + dx * G.T @ delta <= t,  # Wait, need to be careful
        # C[m] + G[:,m].T @ delta <= t (linearized)
        # Actually the linearization is: C[m] + G[:,m] @ delta <= t
        C + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in ['GLOP', 'CLARABEL', 'ECOS']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = delta.value
    best_h = h.copy()
    best_mc = mc
    for lam in [1.0, 0.5, 0.25, 0.1]:
        h_try = project_feasible(h + lam * d, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-14:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc
```

Let me optimize the coordinate descent. For a fixed j, the objective as a function of h[j] = x is:
F(x) = max_k f_k(x) * dx (or the smooth version)

For the smooth version with high alpha (essentially the max):
The k that achieves the max will be the same for nearby values of x (typically), so we can just compute the gradient of f_k w.r.t. x for the active k and do a 1D optimization.

Actually, for the actual max (not smooth), f_k as a function of x = h[j] is:
f_k(x) = const_k + x * (1-h[j+k]) - x * h[j-k] (for the terms involving h[j])
= const_k + x * (1 - h[j+k] - h[j-k]) (for valid indices)
= const_k + x * a_k

where a_k = (1-h[j+k])*I(valid) - h[j-k]*I(valid)

So f_k is linear in x for each k! The max of linear functions is convex in x, so we can find the minimum of the max by finding where the derivative changes sign.

The minimum of max_k (b_k + a_k * x) over x in [0,1] occurs where:
- The two largest values of (b_k + a_k * x) are equal, or
- At x = 0 or x = 1

This can be found in O(K log K) time where K = 2n-1 (the number of shifts).

But actually, for the coordinate descent, since f_k is linear in h[j], the max is a piecewise linear function of h[j], and its minimum over [0,1] can be found by:
1. Computing all a_k and b_k values
2. Finding the x in [0,1] that minimizes max_k (b_k + a_k * x)

This is a 1D convex optimization problem that can be solved efficiently.

The derivative of max_k (b_k + a_k * x) w.r.t. x is max_k a_k (at the point where the max is achieved, it's the a_k of the active constraint... actually it's the max of the a_k's of the active constraints).

The function is increasing where the active constraint has positive a_k and decreasing where it has negative a_k. The minimum is where the derivative changes from negative to positive, i.e., where the active constraint switches from one with negative a_k to one with positive a_k.

This can be found by:
- Computing all (b_k, a_k) pairs
- For a given x, the active set is argmax_k (b_k + a_k * x)
- The function g(x) = max_k (b_k + a_k * x) is piecewise linear
- Its breakpoints are at x = (b_k - b_l) / (a_l - a_k) for pairs where the active constraint changes
- We want the minimum of g on [0,1]

A simpler approach: binary search for where the subgradient contains 0.

Actually, the simplest approach: since g(x) is convex and piecewise linear on [0,1], we can use ternary search or just evaluate at a few points.

But even simpler: we can just use scipy.optimize.minimize_scalar with the actual (non-smooth) objective, and it should handle the piecewise linear function fine (it'll find the minimum at a kink point).

```python
            m1 = (jpk >= 0) & (jpk < n)
            m2 = (jmk >= 0) & (jmk < n)
            a = np.zeros(2 * n - 1)
            a[m1] = 1.0 - h[jpk[m1]]
            a[m2] -= h[jmk[m2]]
            b = corr - h[j] * a
            lo, hi = 0.0, 1.0
            for _ in range(45):
                m1v = lo + (hi - lo) / 3.0
                m2v = hi - (hi - lo) / 3.0
                if np.max(b + a * m1v) < np.max(b + a * m2v):
                    hi = m2v
                else:
                    lo = m1v
            x_new = (lo + hi) / 2.0
            if abs(x_new - h[j]) > 1e-10:
                corr += (x_new - h[j]) * a
                h[j] = x_new
                changed = True
        h = project_feasible(h, n)
        if not changed:
            break
    return h

def lp_step(h, n, eps):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    G = np.zeros((n, 2 * n - 1))
    for mm in range(2 * n - 1):
        k = mm - (n - 1)
        js, je = max(0, -k), min(n, n - k)
        if je > js:
            G[js:je, mm] = 1.0 - h[js + k:je + k]
        js2, je2 = max(0, k), min(n, n + k)
        if je2 > js2:
            G[js2:je2, mm] -= h[js2 - k:je2 - k]
    delta = cp.Variable(n)
    t = cp.Variable()
    cons = [C + G.T @ delta <= t, delta >= -eps, delta <= eps,
            delta >= -h, delta <= 1.0 - h, cp.sum(delta) == 0]
    prob = cp.Problem(cp.Minimize(t), cons)
    for sol in ['GLOP', 'CLARABEL', 'ECOS']:
        try:
            prob.solve(solver=sol)
            if delta.value is not None:
                break
        except Exception:
            continue
    if delta.value is None:
        return h, mc
    d = delta.value
    bh, bm = h.copy(), mc
    for lam in [1.0, 0.5, 0.25, 0.1]:
        ht = project_feasible(h + lam * d, n)
        mct = float(np.correlate(ht, 1.0 - ht, mode='full').max())
        if mct < bm - 1e-14:
            bm, bh = mct, ht
    return bh, bm

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 12
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    h = project_feasible(h, n)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()

    # Phase 1: Gradient descent with increasing alpha
    p1_end = t0 + 0.20 * (dl - t0)
    for alpha in [500, 2000, 10000, 50000]:
        if time.time() > p1_end - 3:
            break
        tl = min(p1_end - time.time() - 1, 50)
        if tl < 3:
            break
        h = grad_descent(h, n, alpha, tl, lr0=0.05, mu=0.85)
        h = project_feasible(h, n)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: Coordinate descent for local precision
    p2_end = t0 + 0.35 * (dl - t0)
    if time.time() < p2_end - 3:
        h = best_h.copy()
        h = coord_descend(h, n, p2_end - time.time() - 2)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 3: LP optimization with adaptive epsilon and restarts
    p3_end = dl - 4
    h = best_h.copy()
    eps = 0.03
    no_imp = 0
    for it in range(2000):
        if time.time() > p3_end - 3:
            break
        h_new, mc = lp_step(h, n, eps)
        c5_new = mc * 2.0 / n
        if c5_new < best_c5 - 1e-13:
            best_c5, best_h = c5_new, h_new.copy()
            no_imp = 0
            eps = max(eps * 0.98, 0.002)
        else:
            no_imp += 1
            if no_imp > 15:
                eps = max(eps * 0.7, 0.001)
                if eps <= 0.0015:
                    # Try perturbation restart
                    h = best_h + 0.01 * rng.randn(n)
                    h = project_feasible(h, n)
                    no_imp = 0
                    eps = 0.02
            h = h_new

    # Phase 4: Final coordinate descent polish
    if time.time() < dl - 2:
        h = best_h.copy()
        h = coord_descend(h, n, dl - time.time() - 1)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
```
