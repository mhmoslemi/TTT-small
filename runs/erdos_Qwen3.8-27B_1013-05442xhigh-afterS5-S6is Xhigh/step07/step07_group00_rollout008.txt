def jac_c(h, n):
    """Compute Jacobian of correlation c with respect to h. Shape: (2n-1, n)"""
    c_jac = np.zeros((2*n-1, n))
    for j in range(n):
        # c[m] = sum_i h[i] * (1-h[i+m-(n-1)])
        # dc[m]/dh[j] = (1-h[j+m-(n-1)]) [from h[j] term] - h[j-m+(n-1)] [from (1-h[...]) term where ...=j]
        # First term: for m such that j+m-(n-1) is in [0, n-1]
        #   => m in [n-1-j, n-1-j+n-1] = [n-1-j, 2n-2-j]
        m1_lo = max(0, n-1-j)
        m1_hi = min(2*n-2, 2*n-2-j)
        if m1_lo <= m1_hi:
            idx = np.arange(m1_lo, m1_hi+1)
            c_jac[idx, j] += 1.0 - h[idx + j - (n-1)]
        
        # Second term: for m such that j-m+(n-1) is in [0, n-1]
        #   => m in [j-(n-1), j]
        m2_lo = max(0, j-n+1)
        m2_hi = min(2*n-2, j)
        if m2_lo <= m2_hi:
            idx = np.arange(m2_lo, m2_hi+1)
            c_jac[idx, j] -= h[j - idx + (n-1)]
    return c_jac

def pnorm_obj(h, n, p, mu):
    c = np.correlate(h, 1-h, mode='full') * (2.0/n)
    # p-norm
    val = np.sum(c**p) ** (1.0/p)
    # penalty
    penalty = mu * (h.sum() - n/2.0)**2
    return val + penalty

def pnorm_grad(h, n, p, mu):
    c = np.correlate(h, 1-h, mode='full') * (2.0/n)
    # grad of (sum c^p)^(1/p) = (1/p) * (sum c^p)^(1/p - 1) * p * c^(p-1) * dc/dh
    # = (sum c^p)^(1/p - 1) * c^(p-1) * dc/dh
    base = np.sum(c**p) ** (1.0/p - 1.0)
    w = c**(p-1) * base  # weights for each c[m]
    # dc[m]/dh[j] as computed before
    jac = compute_jac(h, n)  # shape (2n-1, n)
    grad = jac.T @ w  # shape (n,)
    # add penalty gradient
    grad += 2*mu*(h.sum() - n/2.0)
    return grad

             'fun': lambda x, n=n: x.sum() - n / 2.0,
             'jac': lambda x, n=n: np.ones(n)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj_slsqp, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 800, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def coord_cd(h, n, deadline, grid_size=200, max_passes=200):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(c.max())
        improved = False
        order = np.random.default_rng(int(time.time()*1e6) % 2**31).permutation(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            a = np.zeros(2 * n - 1)
            m_lo_f = max(0, n - 1 - p)
            m_hi_f = min(2 * n - 2, 2 * n - 2 - p)
            if m_lo_f <= m_hi_f:
                idx = np.arange(m_lo_f, m_hi_f + 1)
                a[idx] += 1.0 - h[idx + p - (n - 1)]
            m_lo_b = max(0, p - n + 1)
            m_hi_b = min(2 * n - 2, p)
            if m_lo_b <= m_hi_b:
                idx2 = np.arange(m_lo_b, m_hi_b + 1)
                a[idx2] -= h[p - idx2 + (n - 1)]
            c_aff = c + a * h0
            c_base = c_aff - h0 * a
            diag_idx = n - 1
            c_at_diag = c_base[diag_idx] - h0 * h0
            unaff_lo = 0
            unaff_hi = 2 * n - 2
            affected = set()
            if m_lo_f <= m_hi_f:
                affected.update(range(m_lo_f, m_hi_f + 1))
            if m_lo_b <= m_hi_b:
                affected.update(range(m_lo_b, m_hi_b + 1))
            unaff_mask = np.ones(2 * n - 1, dtype=bool)
            for m in affected:
                unaff_mask[m] = False
            unaff_max = float(c[unaff_mask].max()) if unaff_mask.any() else 0.0
            m_start = min(affected) if affected else 0
            m_end = max(affected) if affected else 0
            d_grid = np.linspace(-h0, 1.0 - h0, grid_size)
            best_d = 0.0
            best_val = max(unaff_max, float(c.max()))
            for d in d_grid:
                if abs(d) < 1e-14:
                    continue
                nv = c_base + d * a
                nv[diag_idx] = c_at_diag + d * a[diag_idx] - d * h0
                val = max(unaff_max, float(nv.max()))
                if val < best_val:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-13:
                h[p] = float(np.clip(h0 + best_d, 0.0, 1.0))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def binarize(h, n):
    target = n // 2
    indices = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[indices[:target]] = 1.0
    return result

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n_init = 500
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Phase 1: L-BFGS-B with p-norm
    if time.time() < deadline - 250:
        best_h, best_c = lforb_opt(best_h, n_init, [10, 25, 50, 100], 50.0, deadline - 80)
    # Phase 2: SLSQP with annealed alpha
    if time.time() < deadline - 150:
        best_h, best_c = slsqp_opt(best_h, n_init, [1000, 10000, 50000, 200000], deadline - 60)
    # Phase 3: Binary cycle
    if time.time() < deadline - 100:
        x_orig = np.linspace(0, 2, n_init)
        h_bin = binarize(best_h, n_init)
        for ns in [0.08, 0.04, 0.02]:
            if time.time() > deadline - 60:
                break
            h_deb = project_feasible(h_bin + rng.normal(0, ns, n_init), n_init)
            h_o, c_o = slsqp_opt(h_deb, n_init, [5000, 20000], deadline - 50)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Phase 4: Multi-n exploration
    n_init_final = len(best_h)
    x_orig = np.linspace(0, 2, n_init_final)
    for n_try in [250, 400, 600, 800]:
        if time.time() > deadline - 120:
            break
        if n_try == n_init_final:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = lforb_opt(h_int, n_try, [25, 75], 50.0, deadline - 80)
        h_o, c_o = slsqp_opt(h_o, n_try, [5000, 50000], deadline - 60)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_orig = x_new
            n_init_final = n_try
    # Phase 5: Fine coordinate descent
    if time.time() < deadline - 10:
        best_h, best_c = coord_cd(best_h, n_init_final, deadline, grid_size=300, max_passes=300)
    # Phase 6: Final SLSQP polish
    if time.time() < deadline - 5:
        best_h, best_c = slsqp_opt(best_h, n_init_final, [500000, 2000000], deadline)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
