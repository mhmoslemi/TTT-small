                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def target_slsqp(h, n, k_target, tau, maxiter):
    """Specifically try to reduce correlation at lag k_target below tau."""
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        excess = np.maximum(c - tau, 0)
        # Weight the target lag more heavily
        w = np.ones(len(c))
        w[k_target] = 10.0
        return np.sum(w * excess ** 2)
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

                jm = j + m_range

            c_arr = np.correlate(best_h, 1 - best_h, mode='full') * (2.0 / n)

    # Phase 5: Resolution increase
    p5 = t0 + 0.86 * total
    x = np.linspace(0, 2, n)
    for nh in [450, 500, 550]:
        if time.time() > p5 - 25:
            break
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, x, best_h), nh)
        for alpha in [5000, 50000, 500000, 5000000]:
            if time.time() > p5 - 15:
                break
            hh = lse_slsqp(hh, nh, alpha, 3000, 1e-15)
        ch = compute_c5(hh, nh)
        if ch < best_c:
            best_c, best_h, n = ch, hh.copy(), nh
            x = xn

    # Phase 6: Final LSE polish at best resolution
    for alpha in [10000000, 50000000, 200000000, 500000000]:
        if time.time() > dl - 8:
            break
        h = lse_slsqp(best_h, n, alpha, 6000, 1e-16)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Final micro-push with reweighting
    if time.time() < dl - 5:
        c_arr = np.correlate(best_h, 1 - best_h, mode='full') * (2.0 / n)
        w = np.ones(len(c_arr))
        w[c_arr > 0.92 * c_arr.max()] = 15.0
        h = reweight_slsqp(best_h, n, w, 3000)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
