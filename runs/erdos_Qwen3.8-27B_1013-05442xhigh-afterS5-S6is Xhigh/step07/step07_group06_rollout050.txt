import time

_stop_time = 0.0

def stop_cb(*args):
    return time.time() > _stop_time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(a, n):
    """Project a to the feasible set: 0<=h<=1, sum(h)=n/2."""
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        a = a[:n] if a.size >= n else np.pad(a, (0, n - a.size))
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    # Bisection on threshold
    lo, hi = float(a.min()) - 1.0, float(a.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(a - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    # Iterate to fix sum
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15: break
            h += diff * room / s
        else:
            s = h.sum()
            if s < 1e-15: break
            h += diff * h / s
    h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def top_n_binary(h, n):
    h = np.asarray(h, dtype=float).ravel()
    target = n // 2
    idx = np.argsort(-h)[:target]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def lse_obj_grad(h, n, beta):
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if s <= 0:
        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    P = p
    R = P[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(P, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g

def lbfgs_obj(x, n, beta, pen):
    h = np.clip(x, 0.0, 1.0)
    F, g = lse_obj_grad(h, n, beta)
    dev = float(h.sum() - 0.5 * n)
    F += 0.5 * pen * dev * dev
    g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return F, g

def polish(h, n, betas, pen, deadline, stage_t):
    work = project_feasible(h, n)
    best = work.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        if time.time() > deadline - 5:
            break
        dl = min(deadline - 3, time.time() + stage_t)
        _stop_time = dl
        try:
            res = minimize(lbfgs_obj, work, method="L-BFGS-B", jac=True,
                          bounds=bounds, args=(n, float(beta), float(pen)),
                          callback=stop_cb,
                          options={"maxiter": 500000, "maxls": 30, "ftol": 1e-14, "gtol": 1e-9})
            x = res.x if res is not None and hasattr(res, 'x') else work
        except Exception:
            x = work
        work = project_feasible(x, n)
        cw = compute_c5(work, n)
        if cw < bc:
            bc = cw
            best = work.copy()
        # Binary projection
        hb = top_n_binary(work, n)
        cb = compute_c5(hb, n)
        if cb < bc:
            bc = cb
            best = hb.copy()
    return best, bc
