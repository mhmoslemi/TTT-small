def lse_obj_grad(h, n, beta, pen):
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    P = p
    R = P[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(P, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    dev = float(h.sum() - 0.5 * n)
    F += 0.5 * pen * dev * dev
    g = g + pen * dev
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g

def compute_base_binary(h, n):
    l = np.arange(-(n - 1), n, dtype=int)
    base = np.zeros((n, 2 * n - 1), dtype=float)
    for p in range(n):
        f = p + l
        mf = (f >= 0) & (f < n)
        base[p, mf] = 1.0 - h[f[mf]]
        b = p - l
        mb = (b >= 0) & (b < n)
        base[p, mb] -= h[b[mb]]
    return base

            s = h.sum()
            if s <= 1e-15: break
            h += diff * h / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmax(1.0 - h)); h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h)); h[idx] = max(0.0, h[idx] + diff)

    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full_like(c, 1.0 / (2 * n - 1)); F = m
    else:
        p = e / s; F = m + np.log(s) / beta
    R = p[::-1]
    A = np.convolve(R, v, mode="full")[n-1:2*n-1]
    B = np.convolve(p, h, mode="full")[n-1:2*n-1]
    g = dx * (A - B)
    dev = float(h.sum() - 0.5 * n)
    F += 0.5 * pen * dev * dev
    g = g + pen * dev
    if not np.all(np.isfinite(g)): g = np.zeros_like(h)
    return float(F), g
