    """Project y onto {0 <= h_i <= 1, sum(h_i) = target}"""
    # Bisection on Lagrange multiplier lambda
    # h_i = clip(y_i - lambda, 0, 1)
    lo, hi = y.min() - target, y.max()
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0, 1)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo+hi)/2, 0, 1)

                grad[a2s:a2e] -= p[idx] * x[a2s-k:a2e-k]
        grad += 2*lam*(x.sum()-n/2.0)
        return val, grad
    bounds = [(1e-7, 1-1e-7)] * n
    try:
        res = minimize(fg, h, jac=True, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        return np.clip(res.x, 0, 1)
    except Exception:
        return h

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_grad(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full').astype(float)
    a = alpha * c
    am = a.max()
    ea = np.exp(a - am)
    p = ea / ea.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if p[idx] < 1e-13:
            continue
        k = idx - (n - 1)
        s1, e1 = max(0, -k), min(n, n - k)
        s2, e2 = max(0, k), min(n, n + k)
        if e1 > s1:
            grad[s1:e1] += p[idx] * (1.0 - h[s1 + k:e1 + k])
        if e2 > s2:
            grad[s2:e2] -= p[idx] * h[s2 - k:e2 - k]
    return grad, float(c.max())

def lp_step(h, n, eps, solver='GLOP'):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    mc = float(C.max())
    G = np.zeros((n, 2 * n - 1))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        s1, e1 = max(0, -k), min(n, n - k)
        s2, e2 = max(0, k), min(n, n + k)
        if e1 > s1:

        if e2 > s2:
            G[s2:e2, m] -= h[s2 - k:e2 - k]
    delta = cp.Variable(n)
    t = cp.Variable()

        C + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in [solver, 'GLPK', 'ECOS', 'CLARABEL']:
        try:
            prob.solve(solver=sol)
            if delta.value is not None:
                solved = True
                break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, mc
    d = np.clip(delta.value, -eps, eps)

        h_try = np.clip(h + lam * d, 0.0, 1.0)
        h_try = project_feasible(h_try, n)

        if mc_try < best_mc - 1e-15:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc
