Let me reconsider the coordinate descent approach more carefully. For a single coordinate update:

Given h, and I want to optimally set h[i] = v + t:
- The change in overlap_k is: Δ_k = t * s_k where s_k = d(overlap_k)/d(h[i])
- New overlap_k = overlap_k + t * s_k
- I want to minimize max_k new_overlap_k over t ∈ [-v, 1-v]

This is a 1D convex optimization. The function f(t) = max_k (overlap_k + t*s_k) is convex piecewise linear. Its minimum can be found efficiently.

Key observation: f(t) is minimized at a point where the two "active" lines (those achieving the max) cross. If only one line is active at the minimum (i.e., the minimum is at a boundary t = -v or t = 1-v), then we just check the boundaries.

Algorithm:
1. Compute f(0) = max_k overlap_k (current max)
2. The subgradient of f at t=0 is: any convex combination of s_k for k in argmax(overlap_k)
3. If all s_k for k in argmax are ≥ 0, then decreasing h[i] (t < 0) helps → opt is at t = -v or where another line becomes active
4. If all s_k for k in argmax are ≤ 0, then increasing h[i] (t > 0) helps → opt is at t = 1-v or where another line becomes active
5. Otherwise, the optimum is interior where two lines cross

For a practical implementation, I can:
1. Compute s_k for all k
2. Find the two lines that are "most relevant" (highest overlap_k / s_k ratio)
3. Compute their intersection
4. Check if it's feasible and optimal
5. If not, check boundaries

Actually, the cleanest approach for 1D: the minimum of max of linear functions occurs at a breakpoint. The breakpoints are at t where two lines cross. But there are O(n^2) breakpoints.

A faster approach: The minimum value T* is the smallest T such that the interval [max_{s_k<0}(T-overlap_k)/s_k, min_{s_k>0}(T-overlap_k)/s_k] (intersected with [-v, 1-v]) is non-empty.

```python
def coord_sweep(h, n, overlaps, n_sweeps=5):
    """Perform coordinate descent sweeps with analytically optimal updates."""
    n_lags = 2 * n - 1
    lags = np.arange(n_lags) - (n - 1)
    
    for sweep in range(n_sweeps):
        for i in range(n):
            # Compute sensitivity s_k = d(overlap_k)/d(h[i])
            idx1 = i - lags
            idx2 = i + lags
            m1 = (idx1 >= 0) & (idx1 < n)
            m2 = (idx2 >= 0) & (idx2 < n)
            s = np.zeros(n_lags)
            s[m1] = 1.0 - h[np.clip(idx1[m1], 0, n-1)]
            s[m2] -= h[np.clip(idx2[m2], 0, n-1)]
            
            v = h[i]
            # Binary search for optimal t
            # Target: minimize max_k (overlaps[k] + t*s[k]) over t in [-v, 1-v]
            pos = s > 1e-14
            neg = s < -1e-14
            
            if not pos.any() and not neg.any():
                continue  # no effect
            
            # Lower bound on T: current max (can only improve by at most |s|*range)
            T_hi = overlaps.max()
            T_lo = T_hi - np.abs(s).max() * (1.0)  # rough lower bound
            T_lo = max(T_lo, 0.0)
            
            # Binary search
            for _ in range(50):
                T_mid = (T_lo + T_hi) / 2.0
                upper_t = 1.0 - v
                lower_t = -v
                if pos.any():
                    u = np.min((T_mid - overlaps[pos]) / s[pos])
                    if u < upper_t:
                        upper_t = u
                if neg.any():
                    l = np.max((T_mid - overlaps[neg]) / s[neg])
                    if l > lower_t:
                        lower_t = l
                if lower_t <= upper_t:
                    T_hi = T_mid
                else:
                    T_lo = T_mid
            
            # Find optimal t at T = T_hi
            upper_t = 1.0 - v
            lower_t = -v
            if pos.any():
                upper_t = min(upper_t, np.min((T_hi - overlaps[pos]) / s[pos]))
            if neg.any():
                lower_t = max(lower_t, np.max((T_hi - overlaps[neg]) / s[neg]))
            
            t_opt = (lower_t + upper_t) / 2.0
            t_opt = np.clip(t_opt, -v, 1.0 - v)
            
            if abs(t_opt) > 1e-10:
                h[i] = v + t_opt
                overlaps += t_opt * s
    
    return h, overlaps
```

  - If s_k = 0: need T ≥ overlap_k
- The feasible interval for t is [lower, upper] intersected with [-v, 1-v]
- We binary search for the smallest T where this interval is non-empty

```python
def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_grad(h, n, alpha):
    overlaps = np.correlate(h, 1 - h, mode='full').astype(float)
    shifted = alpha * (overlaps - overlaps.max())
    exp_v = np.exp(shifted)
    p = exp_v / exp_v.sum()
    n_lags = 2 * n - 1
    lags = np.arange(n_lags) - (n - 1)
    i_idx = np.arange(n)
    idx1 = i_idx[:, None] - lags[None, :]
    idx2 = i_idx[:, None] + lags[None, :]
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    idx1c = np.clip(idx1, 0, n - 1)
    idx2c = np.clip(idx2, 0, n - 1)
    term1 = (p[None, :] * (1.0 - h[idx1c]) * mask1).sum(axis=1)
    term2 = (p[None, :] * h[idx2c] * mask2).sum(axis=1)
    return term1 - term2, overlaps

def lp_step(h, n, eps, solver='GLOP'):
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    n_lags = 2 * n - 1
    lags = np.arange(n_lags) - (n - 1)
    i_idx = np.arange(n)
    idx1 = i_idx[:, None] - lags[None, :]
    idx2 = i_idx[:, None] + lags[None, :]
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    idx1c = np.clip(idx1, 0, n - 1)
    idx2c = np.clip(idx2, 0, n - 1)
    G = (1.0 - h[idx1c]) * mask1 - h[idx2c] * mask2  # shape (n, 2n-1)
    delta = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        C + G.T @ delta <= t,
        delta >= -eps,
        delta <= eps,
        delta >= -h,
        delta <= 1.0 - h,
        cp.sum(delta) == 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    solved = False
    for sol in [solver, 'CLARABEL', 'ECOS']:
        try:
            prob.solve(solver=sol)
            solved = True
            break
        except Exception:
            continue
    if not solved or delta.value is None:
        return h, C.max()
    d = delta.value
    best_h = h.copy()
    best_mc = C.max()
    for lam in [1.0, 0.5, 0.25, 0.1]:
        h_try = np.clip(h + lam * d, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        mc_try = float(np.correlate(h_try, 1.0 - h_try, mode='full').max())
        if mc_try < best_mc - 1e-14:
            best_mc = mc_try
            best_h = h_try
    return best_h, best_mc
```
