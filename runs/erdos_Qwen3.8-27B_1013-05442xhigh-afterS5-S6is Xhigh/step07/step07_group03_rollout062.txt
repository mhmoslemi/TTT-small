```python
import numpy as np

import time

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_deltas(h, n):
    """Fully vectorized computation of flip deltas."""
    m = 2*n - 1
    p = np.arange(n)[:, None]
    k = np.arange(-(n-1), n)[None, :]
    pk = p + k
    pmk = p - k
    valid_pk = (pk >= 0) & (pk < n)
    valid_pmk = (pmk >= 0) & (pmk < n)
    h_pk = np.where(valid_pk, h[np.clip(pk, 0, n-1)], 0.0)
    h_pmk = np.where(valid_pmk, h[np.clip(pmk, 0, n-1)], 0.0)
    sign = (1.0 - 2.0 * h)[:, None]
    return sign * ((1.0 - h_pk) - h_pmk)

def binary_steepest_descent(h, n, deadline, max_iters=5000):
    """Fast binary steepest descent using precomputed deltas."""
    m = 2*n - 1
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = float(corr.max())
    best_h = h.copy()
    best_max = cur_max
    total_iters = 0
    
    while time.time() < deadline:
        zeros_idx = np.where(h == 0)[0]
        ones_idx = np.where(h == 1)[0]
        nz, no = len(zeros_idx), len(ones_idx)
        if nz == 0 or no == 0:
            break
        
        deltas = compute_deltas(h, n)
        d_z = deltas[zeros_idx]  # (nz, m)
        d_o = deltas[ones_idx]  # (no, m)
        
        best_improvement = cur_max + 1e-15
        best_i, best_j = -1, -1
        
        # Batch over zeros
        batch = 40
        for bs in range(0, nz, batch):
            be = min(bs + batch, nz)
            # (batch, no, m)
            d_batch = d_z[bs:be, None, :] + d_o[None, :, :]
            new_maxs = (corr[None, None, :] + d_batch).max(axis=2)
            flat = new_maxs.flatten()
            li = flat.argmin()
            lv = float(flat[li])
            if lv < best_improvement:
                best_improvement = lv
                r, c = divmod(li, no)
                best_i = bs + r
                best_j = c
        
        if best_i < 0 or best_improvement >= cur_max - 1e-10:
            break
        
        pi = zeros_idx[best_i]
        pj = ones_idx[best_j]
        h[pi] = 1.0
        h[pj] = 0.0
        corr = np.correlate(h, 1 - h, mode='full').astype(float)
        cur_max = float(corr.max())
        total_iters += 1
        
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
    
    return best_h, best_max
```

```python
def slsqp_optimize(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_deltas(h, n):
    m = 2 * n - 1
    p = np.arange(n)[:, None]
    k = np.arange(-(n - 1), n)[None, :]
    pk = p + k
    pmk = p - k
    vp = (pk >= 0) & (pk < n)
    vm = (pmk >= 0) & (pmk < n)
    h_pk = np.where(vp, h[np.clip(pk, 0, n - 1)], 0.0)
    h_pmk = np.where(vm, h[np.clip(pmk, 0, n - 1)], 0.0)
    sign = (1.0 - 2.0 * h)[:, None]
    return sign * ((1.0 - h_pmk) - h_pk)

def find_best_swap(corr, zeros_idx, ones_idx, deltas):
    d_z = deltas[zeros_idx]
    d_o = deltas[ones_idx]
    nz, no = len(d_z), len(d_o)
    best_val = float(corr.max()) + 1e-14
    best_i, best_j = -1, -1
    batch = max(10, min(50, nz // 4))
    for bs in range(0, nz, batch):
        be = min(bs + batch, nz)
        db = d_z[bs:be, None, :] + d_o[None, :, :]
        nm = (corr[None, None, :] + db).max(axis=2)
        flat = nm.ravel()
        li = int(flat.argmin())
        lv = float(flat[li])
        if lv < best_val:

            r, c = divmod(li, no)
            best_i, best_j = bs + r, c
    return best_val, best_i, best_j

def binary_optimize(h, n, deadline, max_iters=200000):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = float(corr.max())
    best_h = h.copy()
    best_max = cur_max
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        zeros_idx = np.where(h == 0)[0]
        ones_idx = np.where(h == 1)[0]
        nz, no = len(zeros_idx), len(ones_idx)
        if nz == 0 or no == 0:
            break

        h[pi] = 1.0

        cur_max = float(corr.max())
        if cur_max < best_max:

            best_h = h.copy()

def slsqp_optimize(h, n, alpha, maxiter, ftol):

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha

        return x.sum() - n / 2.0
