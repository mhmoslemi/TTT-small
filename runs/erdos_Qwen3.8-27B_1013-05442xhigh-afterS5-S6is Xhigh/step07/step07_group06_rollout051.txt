def compute_columns(h, n):
    center = n - 1
    col = np.zeros((n, 2 * n - 1))
    i = np.arange(n)[:, None]  # (n, 1)
    m = np.arange(-(n-1), n)[None, :]  # (1, 2n-1)
    ip = i + m  # (n, 2n-1): i+m
    im = i - m  # (n, 2n-1): i-m
    
    # Source term: (1-h[i+m]) where 0<=i+m<n
    valid_src = (ip >= 0) & (ip < n)
    # We need h[ip] for valid entries
    ip_clipped = np.clip(ip, 0, n-1)
    col += np.where(valid_src, (1.0 - h[ip_clipped]), 0.0)
    
    # Target term: -h[i-m] where 0<=i-m<n
    valid_tgt = (im >= 0) & (im < n)
    im_clipped = np.clip(im, 0, n-1)
    col -= np.where(valid_tgt, h[im_clipped], 0.0)
    
    return col

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(n - a.size)])
        else:
            a = a[:n]
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo = float(np.min(a)) - 1.0
    hi = float(np.max(a)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(a - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-13:

        if diff > 0:

            if s <= 1e-15: break
            h += diff * room / s
        else:
            s = h.sum()
            if s <= 1e-15: break
            h += diff * h / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:

            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:

            h[idx] = max(0.0, h[idx] + diff)

    h = project_binary(h, n)
    c = compute_corr(h)
    M = int(np.round(c.max()))

        c_local = start_c.copy()
        M_local = start_M
        while time.time() < phase_deadline:
            col = compute_columns(h_local, n)
            ones = np.nonzero(h_local == 1)[0]
            zeros = np.nonzero(h_local == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            
            best_new_M = M_local
            best_i = -1
            best_j = -1

            chunk = 30

                ce = min(cs + chunk, len(ones))
                i_chunk = ones[cs:ce]
                # delta: (chunk, n_zeros, 2n-1)
                delta = col[zeros][None, :, :] - col[i_chunk][:, None, :]
                new_c = c_local[None, None, :] + delta
                new_max = new_c.max(axis=2)
                flat_idx = np.unravel_index(np.argmin(new_max), new_max.shape)
                li = i_chunk[flat_idx[0]]
                lj = zeros[flat_idx[1]]

                    best_j = lj

                break

            h_local[best_j] = 1.0
            c_local = c_local + col[best_j] - col[best_i]
            M_local = best_new_M
        
        return h_local, c_local, M_local
    
    # Main ILS loop
