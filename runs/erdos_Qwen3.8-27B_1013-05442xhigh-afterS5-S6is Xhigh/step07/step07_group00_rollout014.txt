import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=1000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

        # - p is in valid range for first arg: always (0 <= p < n)
        # - p+m-(n-1) is in [0, n-1] for second arg

    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_con(x, n):
    return x.sum() - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_run(h, n, alphas, maxiter_each=1000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-19})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def f1d_eval(d, c_aff, a, diag_pos):
    vals = c_aff + d * a
    if diag_pos >= 0:
        vals = vals.copy()
        vals[diag_pos] -= d * d
    return vals.max()

def solve_1d_vec(c_aff, a, diag_pos, lo, hi):
    n_grid = 40
    grid = np.linspace(lo, hi, n_grid)
    vals = c_aff[None, :] + grid[:, None] * a[None, :]
    if diag_pos >= 0:
        idx = np.arange(n_grid)
        vals[idx, diag_pos] -= grid ** 2
    row_max = vals.max(axis=1)
    best_idx = np.argmin(row_max)
    gl = grid[max(0, best_idx - 2)]
    gr = grid[min(n_grid - 1, best_idx + 2)]
    if gr - gl < 1e-14:
        return grid[best_idx]
    gr_ratio = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = gr - gr_ratio * (gr - gl)
    c2 = gl + gr_ratio * (gr - gl)
    f1 = f1d_eval(c1, c_aff, a, diag_pos)
    f2 = f1d_eval(c2, c_aff, a, diag_pos)
    for _ in range(50):
        if gr - gl < 1e-15:
            break
        if f1 < f2:
            gr = c2; c2 = c1; f2 = f1
            c1 = gr - gr_ratio * (gr - gl)
            f1 = f1d_eval(c1, c_aff, a, diag_pos)
        else:
            gl = c1; c1 = c2; f1 = f2
            c2 = gl + gr_ratio * (gr - gl)
            f2 = f1d_eval(c2, c_aff, a, diag_pos)
    return 0.5 * (gl + gr)

def coord_cd(h, n, max_passes=500, deadline=None):
    h = h.copy()
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = 0
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_lo = min(n - 1 - p, p)
            m_hi = max(2 * n - 2 - p, p + n - 1)
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c_aff = c[ms]
            dp = (ms == (n - 1)).nonzero()[0]
            diag_pos = int(dp[0]) if len(dp) > 0 else -1
            best_d = solve_1d_vec(c_aff, a, diag_pos, -h0, 1.0 - h0)
            if abs(best_d) > 1e-12:
                h[p] = h0 + best_d
                c[ms] += best_d * a
                if diag_pos >= 0:
                    c[ms[diag_pos]] -= best_d * best_d
                improved += 1
        if improved == 0:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def random_start(n, rng):
    h = rng.uniform(0, 1, n)
    return project_feasible(h, n)

    for n_try in [300, 400, 600, 800]:
        if time.time() > deadline - 100:
            break
        if n_try == n_init:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_opt, c_opt = slsqp_run(h_int, n_try, [500, 5000, 50000], 500)
        h_opt, c_opt = coord_cd(h_opt, n_try, 300, deadline)
        if c_opt < best_c:

            x_orig = x_new
    # Phase 4: Multi-restart
    n = len(best_h)

        n_restarts = max(2, int((deadline - time.time() - 40) / 70))

                break

                hp = project_feasible(best_h + rng.normal(0, s, n), n)

    # Phase 5: Final polish

        n = len(best_h)
        hf, cf = coord_cd(best_h, n, 300, deadline)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
