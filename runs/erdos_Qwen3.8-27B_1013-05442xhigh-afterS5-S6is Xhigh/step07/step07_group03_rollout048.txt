def proj_grad_descent(h, n, alpha, deadline, lr=0.01, max_iter=50000):
    """Projected gradient descent on the log-sum-exp objective."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        probs = np.exp(shifted - sm)
        probs /= probs.sum()
        
        # Gradient of the smoothed max w.r.t. h
        # c_k = sum_i h[i+k]*(1-h[i]) * dx (for appropriate i)
        # dc_k/dh[j] involves h at positions related to j and k
        # grad_obj[j] = sum_k probs[k] * dc_k/dh[j]
        
        # dc_k/dh[j] for k>0: 
        #   c_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i]) * dx
        #   dc_k/dh[j] = dx * (1-h[j-k]) [if j-k >= 0, j is the i+k term]
        #              + dx * (-h[j+k]) [if j+k < n, j is the i term]
        # For k<0 (k=-m):
        #   c_{-m} = sum_{i=m}^{n-1} h[i-m]*(1-h[i]) * dx
        #   dc_{-m}/dh[j] = dx * (1-h[j+m]) [if j+m < n, j is the i-m term... wait]
        #   Actually c_{-m} = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m]) * dx
        #   dc_{-m}/dh[j] = dx * (1-h[j+m]) [j is the h[i] term, i=j]
        #              + dx * (-h[j-m]) [j is the h[i+m] term, i=j-m]
        
        # Let me compute this more carefully.
        # c_k for k >= 0: sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        # dc_k/dh[j]:
        #   If j is in position i+k (i.e., i = j-k): contributes (1-h[j-k]) [if 0<=j-k<n-1-k, i.e., k<=j<=n-1]
        #   If j is in position i (i.e., i = j): contributes -h[j+k] [if 0<=j<n-1-k and 0<=j+k<n]
        
        # For simplicity, let me compute it as:
        # dc_k/dh[j] = (1-h[j-k])*[k<=j] + (-h[j+k])*[j+k<n] for k>=0
        #              (with appropriate bounds)
        
        # This is getting complex. Let me use a different approach:
        # Compute the gradient numerically using the fact that c_k is bilinear.
        
        # Actually, for the gradient, I can use:
        # grad[j] = sum_k probs[k] * dc_k/dh[j] * dx
        # 
        # For k > 0: dc_k/dh[j] = [j>=k]*(1-h[j-k]) - [j+k<n]*h[j+k]
        # For k < 0 (k = -m, m > 0): dc_k/dh[j] = [j+m<n]*(1-h[j+m]) - [j>=m]*h[j-m]
        # For k = 0: dc_0/dh[j] = (1-2*h[j])
        
        # Let me vectorize this:
        grad = np.zeros(n)
        # k = 0
        p0 = probs[n-1]  # index n-1 corresponds to k=0
        grad += p0 * (1.0 - 2.0 * h)
        
        # For k > 0:
        for k_idx in range(n, 2*n-1):
            k = k_idx - (n-1)
            p = probs[k_idx]
            if p < 1e-10:
                continue
            # Term 1: (1-h[j-k]) for j >= k
            if k < n:
                grad[k:] += p * (1.0 - h[:n-k])
            # Term 2: -h[j+k] for j+k < n, i.e., j < n-k
            if n-k > 0:
                grad[:n-k] -= p * h[k:]
        
        # For k < 0 (k = -m, m > 0):
        for k_idx in range(0, n-1):
            m = (n-1) - k_idx
            p = probs[k_idx]
            if p < 1e-10:
                continue
            # c_{-m} = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
            # dc/dh[j] = (1-h[j+m]) [j is h[i], i=j, valid if j < n-m]
            #          + (-h[j-m]) [j is h[i+m], i=j-m, valid if j-m >= 0, i.e., j >= m]
            if n-m > 0:
                grad[:n-m] += p * (1.0 - h[m:])
            if m < n:
                grad[m:] -= p * h[:n-m]
        
        # Scale by dx
        grad *= dx
        
        # Take a step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        # Check improvement
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            # Increase learning rate slightly
            lr = min(lr * 1.05, 0.5)
        else:
            lr = max(lr * 0.8, 0.0001)
        
        h = h_new
    
    return best_h, best_c

Let me vectorize the gradient computation:

```python
def compute_smooth_grad(h, n, probs, dx):
    """Compute gradient of sum_k probs[k] * c_k w.r.t. h."""
    grad = np.zeros(n)
    
    # k = 0
    p0 = probs[n-1]
    if p0 > 1e-12:
        grad += p0 * (1.0 - 2.0 * h)
    
    # For k > 0: k = 1, 2, ..., n-1
    # dc_k/dh[j] = (1-h[j-k]) for j=k..n-1, minus h[j+k] for j=0..n-1-k
    for k in range(1, n):
        p = probs[n-1+k]
        if p < 1e-10:
            continue
        grad[k:] += p * (1.0 - h[:n-k])
        if n-k > 0:
            grad[:n-k] -= p * h[k:]
    
    # For k < 0: k = -1, -2, ..., -(n-1), i.e., m = 1, 2, ..., n-1
    for m in range(1, n):
        p = probs[n-1-m]
        if p < 1e-10:
            continue
        if n-m > 0:
            grad[:n-m] += p * (1.0 - h[m:])
        grad[m:] -= p * h[:n-m]
    
    return grad * dx
```

This is still O(n²) due to the loops. But with the `if p < 1e-10` check, many iterations will be skipped if the probs are concentrated. With a good alpha, the probs will be concentrated on a few lags, making this much faster.

Actually, for the log-sum-exp with large alpha, the probs will be concentrated on the top few lags (maybe 5-20). So the gradient computation will be O(n * |S|) where |S| is the number of "active" lags. With |S| ≈ 10-20, that's O(4000-8000) per iteration, which is very fast!

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-15: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_ls(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        shifted -= shifted.max()
        return m + np.log(np.exp(shifted).sum()) / alpha
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    bounds = [(0.0, 1.0)] * n

        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:
        return h

def pgd_step(h, n, alpha, lr):
    """One projected gradient descent step."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    probs = np.exp(shifted)
    probs /= probs.sum()
    
    # Compute gradient: grad[j] = sum_k probs[k] * dc_k/dh[j] * dx
    grad = np.zeros(n)
    
    # Only process lags with significant probability
    active = np.where(probs > 1e-8)[0]
    
    for k_idx in active:
        k = k_idx - (n - 1)
        p = probs[k_idx]
        if k == 0:
            grad += p * (1.0 - 2.0 * h)
        elif k > 0:
            # c_k = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            # dc_k/dh[j] = (1-h[j-k]) for j >= k (j is the h[i+k] position)
            #            + (-h[j+k]) for j < n-k (j is the h[i] position)
            if n - k > 0:
                grad[k:] += p * (1.0 - h[:n-k])
                grad[:n-k] -= p * h[k:]
        else:
            m_lag = -k  # m_lag > 0
            # c_{-m} = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
            # dc/dh[j] = (1-h[j+m]) for j < n-m (j is the h[i] position)
            #          + (-h[j-m]) for j >= m (j is the h[i+m] position)
            if n - m_lag > 0:
                grad[:n-m_lag] += p * (1.0 - h[m_lag:])
            if m_lag < n:
                grad[m_lag:] -= p * h[:n-m_lag]
    
    grad *= dx
    
    h_new = h - lr * grad
    return project_feasible(h_new, n), c

def pgd_optimize(h, n, alpha, deadline, lr_init=0.05, max_iter=100000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:

        h_new, c = pgd_step(h, n, alpha, lr)
        cur_c = float(c.max())
        if cur_c < best_c:
            best_c = cur_c

            lr = min(lr * 1.1, 0.5)
            stall = 0
        else:
            lr = max(lr * 0.9, 0.001)
            stall += 1
        h = h_new
        if stall > 5000:
            break
    
    return best_h, best_c
