            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    c_max = corr.max()
    exp_c = np.exp(alpha * (corr - c_max))
    sum_exp = exp_c.sum()
    p = exp_c / sum_exp
    val = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):

        if w < 1e-12:

        if lag >= 0:
            grad[:n - lag] += (1.0 - h[lag:]) * w
            grad[lag:] -= h[:n - lag] * w
        else:
            ll = -lag
            grad[ll:] += (1.0 - h[:n - ll]) * w
            grad[:n - ll] -= h[ll:] * w
    return val, grad

def slsqp_with_grad(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        return smooth_max_and_grad(x, n, alpha)[0]
    def grad(x):
        return smooth_max_and_grad(x, n, alpha)[1]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if time.time() > deadline:
            return h
        return project_feasible(res.x, n)
    except Exception:

def lbfgs_auglag(h, n, alpha, deadline):

    dx = 2.0 / n
    lam = 0.0
    mu = 100.0

            c = x.sum() - n / 2.0

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    c_max = corr.max()
    exp_c = np.exp(alpha * (corr - c_max))
    sum_exp = exp_c.sum()
    p = exp_c / sum_exp
    val = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k] * dx
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += (1.0 - h[lag:]) * w
            grad[lag:] -= h[:n - lag] * w
        else:
            ll = -lag
            grad[ll:] += (1.0 - h[:n - ll]) * w
            grad[:n - ll] -= h[ll:] * w
    return val, grad

def slsqp_opt(h, n, alpha, maxiter, deadline):
    def obj(x):
        return smooth_max_and_grad(x, n, alpha)[0]
    def grad(x):
        return smooth_max_and_grad(x, n, alpha)[1]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if time.time() < deadline:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

    dx = 2.0 / n
    lam = 0.0
    mu = 50.0
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for outer in range(40):
        if time.time() > deadline - 5:
            break
        def obj(x):
            v, _ = smooth_max_and_grad(x, n, alpha)
            c = x.sum() - n/2.0
            return v + lam * c + 0.5 * mu * c * c
        def grad(x):
            _, g = smooth_max_and_grad(x, n, alpha)
            c = x.sum() - n/2.0
            return g + (lam + mu * c) * np.ones(n)
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 300, 'ftol': 1e-14})
            h = project_feasible(res.x, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            constraint_viol = abs(h.sum() - n/2.0)
            lam += mu * (h.sum() - n/2.0)
            if constraint_viol < 1e-5:
                mu *= 0.5
            else:
                mu = min(mu * 2.0, 1e6)
            if mu > 1e5 and constraint_viol < 1e-4:
                break
        except Exception:
            break
    return best_h

def subgrad_descent(h, n, alpha, deadline, mu=0.9):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = 1.0 / (alpha * dx + 1e-10)
    mom = np.zeros(n)
    stall = 0
    for step in range(500000):
        if time.time() > deadline:
            break
        _, grad = smooth_max_and_grad(h, n, alpha)
        mom = mu * mom - lr * grad
        h = project_feasible(h + mom, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-11:
            best_c = c
            best_h = h.copy()
            stall = 0
            lr = min(lr * 1.01, 10.0 / (alpha * dx + 1e-10))
        else:

            if stall > 200:

                if lr < 0.01 / (alpha * dx + 1e-10):
                    lr = 0.1 / (alpha * dx + 1e-10)
                    stall = 0
    return best_h, best_c

def coarse_refine(h_fine, n_fine, n_coarse, alpha, deadline):
    """Downsample, optimize, upsample."""
    h_c = np.interp(np.linspace(0, 2, n_coarse, endpoint=False),
                    np.linspace(0, 2, n_fine, endpoint=False), h_fine)
    h_c = project_feasible(h_c, n_coarse)

                     np.linspace(0, 2, n_coarse, endpoint=False), h_c)
    return project_feasible(h_up, n_fine)

import cvxpy as cp

def lp_linearize(h, n, max_iters, deadline):
    dx = 2.0 / n
    h_cvx = cp.Variable(n)
    t = cp.Variable()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Compute current c_k values and gradients
        corr = np.correlate(h, 1 - h, mode='full') * dx
        # Linearize each c_k around current h
        constraints = [h_cvx >= 0, h_cvx <= 1, h_cvx.sum() == n/2.0]
        for k in range(2 * n - 1):
            if corr[k] < corr.max() - 0.01:  # only active constraints
                # Compute gradient of c_k w.r.t. h
                lag = k - (n - 1)
                grad_k = np.zeros(n)
                if lag >= 0:
                    grad_k[:n-lag] += (1 - h[lag:]) * dx
                    grad_k[lag:] -= h[:n-lag] * dx
                else:
                    ll = -lag
                    grad_k[ll:] += (1 - h[:n-ll]) * dx
                    grad_k[:n-ll] -= h[ll:] * dx
                # Linearized constraint: grad_k · h + c_k - grad_k · h_current ≤ t
                # c_k_lin = c_k + grad_k · (h - h_current)
                # = (c_k - grad_k · h_current) + grad_k · h
                offset = corr[k] - grad_k.dot(h)
                constraints.append(grad_k.dot(h_cvx) + offset <= t)
        
        obj = cp.Minimize(t)
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.GLOP)  # or CBC
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                break
        h_new = h_cvx.value
        if h_new is None:
            break
        h = h_new
    return h

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    c_max = corr.max()
    exp_c = np.exp(alpha * (corr - c_max))
    sum_exp = exp_c.sum()
    p = exp_c / sum_exp
    val = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k] * dx
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += (1.0 - h[lag:]) * w
            grad[lag:] -= h[:n - lag] * w
        else:
            ll = -lag
            grad[ll:] += (1.0 - h[:n - ll]) * w
            grad[:n - ll] -= h[ll:] * w
    return val, grad

    def obj(x):
        return smooth_max_and_grad(x, n, alpha)[0]
    def grad(x):
        return smooth_max_and_grad(x, n, alpha)[1]

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})

            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def compute_lag_grad(h, n, k):
    dx = 2.0 / n
    lag = k - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        g[:n - lag] += (1.0 - h[lag:]) * dx
        g[lag:] -= h[:n - lag] * dx
    else:
        ll = -lag
        g[ll:] += (1.0 - h[:n - ll]) * dx
        g[:n - ll] -= h[ll:] * dx
    return g

def lp_refine(h, n, deadline, max_active=20, tolerance=0.003):
    h_cvx = cp.Variable(n)
    t = cp.Variable()
    base = [h_cvx >= 0, h_cvx <= 1, h_cvx.sum() == n / 2.0]
    prev_c = compute_c5(h, n)
    for it in range(10000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx_val
        c_max = corr.max()
        active = np.where(corr >= c_max - tolerance)[0]
        if len(active) > max_active:
            active = np.argsort(corr)[-max_active:]
        constraints = list(base)
        for k in active:
            gk = compute_lag_grad(h, n, k)
            offset = corr[k] - gk.dot(h)
            constraints.append(gk @ h_cvx - t <= -offset)
        prob = cp.Problem(cp.Minimize(t), constraints)
        solved = False
        for solver in [cp.GLOP, cp.CBC, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                solved = True
                break
            except Exception:
                continue
        if not solved:
            break
        h_new = np.array(h_cvx.value).flatten()
        if h_new is None or np.any(np.isnan(h_new)):
            break
        if np.max(np.abs(h_new - h)) < 1e-9:
            break
        h = h_new
    return project_feasible(h, n)

def subgrad_descent(h, n, alpha, deadline, mu=0.92):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    lr = 2.0 / (alpha * dx + 1e-10)
    mom = np.zeros(n)
    stall = 0
    for step in range(500000):
        if time.time() > deadline:
            break
        _, grad = smooth_max_and_grad(h, n, alpha)
        mom = mu * mom - lr * grad
        h = project_feasible(h + mom, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-11:
            best_c = c
            best_h = h.copy()
            stall = 0
            lr = min(lr * 1.015, 20.0 / (alpha * dx + 1e-10))
        else:
            stall += 1
            if stall > 300:
                lr *= 0.93
                if lr < 0.01 / (alpha * dx + 1e-10):
                    lr = 0.1 / (alpha * dx + 1e-10)
                    stall = 0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP with explicit gradient (much faster than numerical)
    p1_end = t0 + 80
    for a in [100, 500, 2000]:
        if time.time() > p1_end - 10:
            break
        h = slsqp_opt(h, n, a, 2000, p1_end - 5)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 2: LP refinement (successive linear programming)
    p2_end = t0 + 320
    h2 = lp_refine(best_h, n, p2_end, max_active=20, tolerance=0.003)
    c2 = compute_c5(h2, n)
    if c2 < best_c:
        best_c, best_h = c2, h2.copy()

    # Phase 3: SLSQP high alpha with gradient
    p3_end = t0 + 620
    for a in [10000, 100000, 1000000, 10000000, 100000000]:
        if time.time() > p3_end - 20:
            break
        h = slsqp_opt(best_h, n, a, 3000, p3_end - 10)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: LP refinement with tighter tolerance
    p4_end = t0 + 820
    if time.time() < p4_end - 30:
        h4 = lp_refine(best_h, n, p4_end - 10, max_active=15, tolerance=0.001)
        c4 = compute_c5(h4, n)
        if c4 < best_c:
            best_c, best_h = c4, h4.copy()

    # Phase 5: Subgradient descent for landscape exploration
    p5_end = t0 + 950
    if time.time() < p5_end - 20:
        h5, c5 = subgrad_descent(best_h, n, 50000, p5_end - 20, mu=0.93)
        if c5 < best_c:
            best_c, best_h = c5, h5.copy()

    # Phase 6: Noise restarts with LP + SLSQP polish
    p6_end = deadline - 40
    for trial in range(6):
        if time.time() > p6_end - 10:
            break
        noise = 0.003 * (1.0 + 0.5 * trial)
        h6 = best_h + noise * rng.randn(n)
        h6 = project_feasible(h6, n)
        h6 = slsqp_opt(h6, n, 50000, 800, p6_end - 5)
        h6 = lp_refine(h6, n, p6_end - 3, max_active=15, tolerance=0.002)
        c6 = compute_c5(h6, n)
        if c6 < best_c:
            best_c, best_h = c6, h6.copy()

    # Phase 7: Final polish - SLSQP + LP
    if time.time() < deadline - 15:
        h7 = slsqp_opt(best_h, n, 50000000, 3000, deadline - 10)
        c7 = compute_c5(h7, n)
        if c7 < best_c:
            best_c, best_h = c7, h7.copy()
    if time.time() < deadline - 3:
        h7 = lp_refine(best_h, n, deadline - 1, max_active=10, tolerance=0.0005)
        c7 = compute_c5(h7, n)
        if c7 < best_c:
            best_c, best_h = c7, h7.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
