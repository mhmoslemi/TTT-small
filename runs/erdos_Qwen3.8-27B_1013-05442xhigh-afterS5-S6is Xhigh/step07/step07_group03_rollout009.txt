import numpy as np
from scipy.optimize import minimize
import time

def compute_correlations(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_c5(h, n):
    return float(np.max(compute_correlations(h, n)))

def active_set_optimize(h, n, deadline, top_k=10, max_iters=5000):
    """Optimize using active set identification + SLSQP on reduced problem"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_correlations(h, n)
        cur_c = c.max()
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        
        # Identify active set
        threshold = cur_c - 0.001
        active = np.where(c >= threshold)[0]
        if len(active) < top_k:
            active = np.argsort(c)[::-1][:top_k]
        
        # Build objective: log-sum-exp over active set
        alpha = 50000.0
        c_active = c[active]
        
        def obj(x):
            cx = compute_correlations(x, n)
            ca = cx[active]
            m = ca.max()
            return m + np.log(np.exp((ca - m) * alpha).sum()) / alpha
        
        def grad(x):
            # Numerical gradient would be too slow; use analytical
            # grad of c_k at h[i]:
            # For k >= 1 (shift): c_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            # dc_k/dh[i] = (1-h[i+k]) for i <= n-1-k, plus -h[i-k] for i >= k
            # This is complex; use subgradient approach instead
            cx = compute_correlations(x, n)
            ca = cx[active]
            m = ca.max()
            weights = np.exp((ca - m) * alpha) / np.exp((ca - m) * alpha).sum()
            # Combined gradient
            g = np.zeros(n)
            for idx, w in zip(active, weights):
                k = idx - (n - 1)
                if k == 0:
                    g += w * (1.0 - 2.0 * x)
                elif k > 0:
                    g[k:] += w * (1.0 - x[:n-k])
                    g[:n-k] -= w * x[k:]
                else:
                    kk = -k
                    g[:n-kk] += w * (1.0 - x[kk:])
                    g[kk:] -= w * x[:n-kk]
            return dx * g
        
        # Use L-BFGS-B with the combined gradient
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 
                 'jac': lambda x: np.ones(n)}]
        
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < cur_c:
                h = h_new
        except:
            break
    
    return best_h, best_c
