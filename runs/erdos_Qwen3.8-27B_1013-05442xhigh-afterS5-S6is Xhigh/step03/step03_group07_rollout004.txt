def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.minimum(beta * shifted, 0))  # numerical stability
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s == 0:
            grad[:] += wk * (1.0 - 2.0 * h)
        elif s > 0:

            grad[ms] += wk * (1.0 - h[ms - s])

            grad[ms] += wk * (1.0 - h[ms + t])
            mp = np.arange(t, n)
            grad[mp] += wk * (-h[mp - t])

    """FISTA (fast iterative shrinkage-thresholding) with projected gradient."""
    n = len(h0)
    target = n / 2.0
    x = h0.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    
    for t in range(max_iters):
        # FISTA momentum
        if t == 0:
            y = x.copy()
            t_k = 1.0
        else:
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            mu = (t_k - 1.0) / t_new
            y = x + mu * (x - x_prev)
            t_k = t_new
        
        # Project y to feasible set
        y = project_feasible(y, target)
        
        # Compute gradient at y
        _, grad = smoothmax_val_grad(y, beta)
        
        # Adaptive step size (simplified: use fixed decay)
        lr_t = lr0 / (1.0 + 0.01 * t)
        
        # Update
        x_new = project_feasible(y - lr_t * grad, target)
        
        # Check improvement
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        
        x_prev = x
        x = x_new
    
    return best_x, best_v

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    
    def jac(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    
    def con_fun(x):
        return x.sum() - target
    
    con_jac = np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def interp_to_n(h, n_new):
    n_old = len(h)

    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

from scipy.interpolate import interp1d

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-10:
            free = (h > 1e-6) & (h < 1.0 - 1e-6)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.minimum(beta * shifted, 0.0))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s == 0:
            grad[:] += wk * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += wk * (1.0 - h[:n-s])
            grad[:n-s] += wk * (-h[s:])
        else:
            t = -s
            grad[:n-t] += wk * (1.0 - h[t:])
            grad[t:] += wk * (-h[:n-t])
    grad *= dx
    return val, grad

def targeted_perturb(h, n_top=3, amp=0.008):
    n = len(h)
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-n_top:]
    x = h.copy()
    for k in top_idx:
        s = k - (n - 1)
        freq = abs(s)
        if freq == 0:
            continue
        phase = np.random.uniform(0, 2 * np.pi)
        pert = amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        x = project_feasible(x + pert, n / 2.0)
    return x
