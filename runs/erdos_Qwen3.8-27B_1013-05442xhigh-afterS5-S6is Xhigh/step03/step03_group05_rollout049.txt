def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):

    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target):
    """Project onto {0<=x<=1, sum=target} via bisection on threshold."""
    n = len(h)
    lo, hi = -1.0, 2.0

        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid

            jp = j + k_vals
            jm = j - k_vals

            # = max_k [corr[k] + (t - cur[j]) * d[k]] + lam * (t - cur[j])

                    lo = m1

    best_h = cur.copy()

    # Phase 1: Brief SLSQP warmup
    h_init = project_h(h_parent, target)
    for beta in [150, 500]:
        if time_left() < 40:
            break
        obj = partial(obj_smooth_max, dx=dx, beta=beta)
        con = partial(con_sum, target=target)
        con_j = partial(con_sum_jac, target=target)
        cons = [{'type': 'eq', 'fun': con, 'jac': con_j}]
        try:
            res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 800, 'ftol': 1e-14})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_init = h_new.copy()
        except Exception:
            pass

    # Phase 2: Coordinate descent (main optimization)
    if time_left() > 80:
        h_cd = project_h(best_h, target)
        h_cd, c5_cd = coord_descent(h_cd, n, dx, target,
                                    max_sweeps=2000, time_fn=time_left, time_limit=60)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 3: Subgradient polish
    if time_left() > 20:
        h_pol = project_h(best_h, target)
        h_pol, c5_pol = subgradient_polish(h_pol, n, dx, target,
                                           max_iters=8000, time_fn=time_left, time_limit=10)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 4: Second coordinate descent round
    if time_left() > 40:
        h_cd2 = project_h(best_h, target)
        h_cd2, c5_cd2 = coord_descent(h_cd2, n, dx, target,
                                      max_sweeps=2000, time_fn=time_left, time_limit=15)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    # Phase 5: Try higher resolution
    if time_left() > 60:
        n_hi = min(n + 50, 400)
        target_hi = n_hi / 2.0
        dx_hi = 2.0 / n_hi
        bounds_hi = Bounds(np.zeros(n_hi), np.ones(n_hi))
        h_hi = project_h(resample(best_h, n_hi), target_hi)
        # Quick SLSQP at higher res
        for beta in [200, 600]:
            if time_left() < 20:
                break
            obj_hi = partial(obj_smooth_max, dx=dx_hi, beta=beta)
            con_hi = partial(con_sum, target=target_hi)
            con_hi_j = partial(con_sum_jac, target=target_hi)
            cons_hi = [{'type': 'eq', 'fun': con_hi, 'jac': con_hi_j}]
            try:
                res = minimize(obj_hi, h_hi, method='SLSQP', bounds=bounds_hi,
                               constraints=cons_hi, options={'maxiter': 500, 'ftol': 1e-14})
                h_new = project_h(res.x, target_hi)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_hi = h_new.copy()
            except Exception:
                pass
        # Coordinate descent at higher res
        if time_left() > 25:
            h_hi_cd = project_h(h_hi, target_hi)
            h_hi_cd, c5_hi = coord_descent(h_hi_cd, n_hi, dx_hi, target_hi,
                                           max_sweeps=1000, time_fn=time_left, time_limit=8)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi_cd.copy()

    # Phase 6: Final polish
    if time_left() > 5:
        n_f = len(best_h)
        target_f = n_f / 2.0
        dx_f = 2.0 / n_f
        h_f = project_h(best_h, target_f)
        h_f, c5_f = subgradient_polish(h_f, n_f, dx_f, target_f,
                                       max_iters=10000, time_fn=time_left, time_limit=2)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    return best_h, best_c5, len(best_h)
