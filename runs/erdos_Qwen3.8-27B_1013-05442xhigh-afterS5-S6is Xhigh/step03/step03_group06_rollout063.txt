import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    lam = (lo + hi) / 2.0
    p = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - p.sum()
    if abs(diff) > 1e-13:
        mask = (p > 1e-10) & (p < 1.0 - 1e-10)
        if mask.sum() > 0:
            p[mask] += diff / mask.sum()
            p = np.clip(p, 0.0, 1.0)
    return p

def smooth_max_val(corr_vals, beta):
    m = np.max(corr_vals)
    xs = np.clip(beta * (corr_vals - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(xs)) / beta

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    exps = np.exp(np.clip(beta * (vals - m), -500, 500))
    p = exps / exps.sum()
    
    grad = np.zeros(n)
    # Lag 0
    grad += p[n-1] * (1.0 - 2.0 * h)
    # Positive lags
    for k in range(1, n):
        pk = p[n-1+k]
        if pk > 1e-12:
            grad[k:] += pk * (1.0 - h[:n-k])
            grad[:n-k] -= pk * h[k:]
    # Negative lags
    for k in range(1, n):
        pk = p[n-1-k]
        if pk > 1e-12:
            grad[:n-k] += pk * (1.0 - h[k:])
            grad[k:] -= pk * h[:n-k]
    
    return grad * dx

def slsqp_step(h, target_sum, beta, maxiter=2000):
    n = len(h)
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max_val(corr * dx, beta)
    
    def gradient(x):
        return smooth_max_grad(x, dx, beta)
    
    def con(x):
        return np.sum(x) - target_sum
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': con, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h, jac=gradient, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_h(res.x, target_sum)
        return h_new
    except:
        return h

def subgrad_step(h, target_sum, n_lags=5, steps=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:n_lags]
    
    grad = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:n-lag])
            grad[:n-lag] -= h[lag:]
        else:
            m = -lag
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]
    
    gn = np.linalg.norm(grad)
    if gn < 1e-14:
        return h, compute_c5(h)
    grad /= gn
    
    best_h = h
    best_c5 = compute_c5(h)
    if steps is None:
        steps = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    for s in steps:
        h_trial = project_h(h - s * grad, target_sum)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5:
            best_h = h_trial
            best_c5 = c5_trial
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    # Initialize

        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.full(60, 0.5)
    
    best_c5 = compute_c5(best_h)
    n_best = len(best_h)
    
    # Phase 1: SLSQP with analytic gradient at multiple betas
    for n_pts in [n_best]:
        target = n_pts / 2.0
        h_cur = best_h.copy()
        c5_cur = best_c5
        
        for beta in [20, 50, 100, 200, 400, 800, 1600, 3200, 6400]:

            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-14:
                h_cur = h_new
                c5_cur = c5_new
            if c5_cur < best_c5:
                best_h = h_cur.copy()
                best_c5 = c5_cur
                n_best = n_pts
    
    # Phase 2: Multi-resolution exploration
    for n_pts in [80, 120, 160, 200, 240, 320, 400]:
        if remaining() < 60:
            break
        if n_pts == n_best:
            continue
        target = n_pts / 2.0
        x_old = np.linspace(0, 2, n_best)
        x_new = np.linspace(0, 2, n_pts)
        h_cur = project_h(np.interp(x_new, x_old, best_h, clip=True), target)
        c5_cur = compute_c5(h_cur)
        
        for beta in [50, 200, 800, 3200]:
            if remaining() < 25:
                break
            h_new = slsqp_step(h_cur, target, beta, maxiter=1000)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-14:
                h_cur = h_new
                c5_cur = c5_new
        if c5_cur < best_c5:
            best_h = h_cur.copy()
            best_c5 = c5_cur
            n_best = n_pts
    
    # Phase 3: Subgradient descent
    target = n_best / 2.0
    no_improve = 0
    for it in range(30000):
        if remaining() < 5:
            break
        h_new, c5_new = subgrad_step(best_h, target, n_lags=7,
                                      steps=[0.05, 0.025, 0.01, 0.005, 0.002, 0.001])
        if c5_new < best_c5 - 1e-15:
            best_h = h_new
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300 and it % 100 == 99:
                perturb = np.random.randn(n_best) * 0.008
                h_trial = project_h(best_h + perturb, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5:
                    best_h = h_trial
                    best_c5 = c5_trial
                    no_improve = 0
            if no_improve > 3000:

    target = n_best / 2.0
    for it in range(50000):
        if remaining() < 3:
            break
        improved = False
        # Try perturbing individual elements
        idx = it % n_best
        for d in [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]:
            h_trial = best_h.copy()
            h_trial[idx] = np.clip(h_trial[idx] + d, 0.0, 1.0)
            h_trial = project_h(h_trial, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-15:
                best_h = h_trial
                best_c5 = c5_trial
                improved = True
                break
        if not improved and it % 1000 == 999:
            # Random pair
            i, j = np.random.randint(0, n_best, 2)
            for d in [0.02, -0.02, 0.01, -0.01]:
                h_trial = best_h.copy()
                h_trial[i] += d
                h_trial[j] -= d
                if 0 <= h_trial[i] <= 1 and 0 <= h_trial[j] <= 1:
                    h_trial = project_h(h_trial, target)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < best_c5 - 1e-15:
                        best_h = h_trial
                        best_c5 = c5_trial
                        improved = True
                        break
    
    # Phase 5: Final SLSQP polish
    if remaining() > 15:
        target = n_best / 2.0
        h_new = slsqp_step(best_h, target, 8000, maxiter=5000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    return best_h, best_c5, len(best_h)

def lbfgs_with_penalty(h, target_sum, beta, rho_start=1.0, rho_mult=2.0, max_rho=1e4):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    rho = rho_start
    
    while rho <= max_rho and (time.time() - t0) < budget_s - 10:
        def obj(x):
            corr = np.correlate(x, 1.0-x, mode="full")
            sm = smooth_max_val(corr * dx, beta)
            penalty = 0.5 * rho * (np.sum(x) - target_sum)**2
            return sm + penalty
        
        def grad(x):
            g = smooth_max_grad(x, dx, beta)
            g += rho * (np.sum(x) - target_sum)
            return g
        
        bounds = Bounds(np.zeros(n), np.ones(n))
        res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 500, 'ftol': 1e-16})
        h_cur = np.clip(res.x, 0, 1)
        
        # Check constraint
        constraint_violation = abs(np.sum(h_cur) - target_sum)
        if constraint_violation < 1e-6:
            break
        rho *= rho_mult
    
    return project_h(h_cur, target_sum)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - p.sum()
    if abs(diff) > 1e-13:
        mask = (p > 1e-10) & (p < 1.0 - 1e-10)
        if mask.sum() > 0:
            p[mask] += diff / mask.sum()
            p = np.clip(p, 0.0, 1.0)
    return p

def smooth_max_val(corr_vals, beta):
    m = np.max(corr_vals)
    xs = np.clip(beta * (corr_vals - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(xs)) / beta

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    exps = np.exp(np.clip(beta * (vals - m), -500, 500))
    p = exps / exps.sum()
    grad = np.zeros(n)
    grad += p[n-1] * (1.0 - 2.0 * h)
    for k in range(1, n):
        pk = p[n-1+k]
        if pk > 1e-12:
            grad[k:] += pk * (1.0 - h[:n-k])
            grad[:n-k] -= pk * h[k:]
    for k in range(1, n):
        pk = p[n-1-k]
        if pk > 1e-12:
            grad[:n-k] += pk * (1.0 - h[k:])
            grad[k:] -= pk * h[:n-k]
    return grad * dx

def optimize_lbfgs(h, target_sum, beta, t0, budget_s, max_inner=300):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    
    for rho in [10.0, 100.0, 1000.0, 10000.0]:
        if time.time() - t0 > budget_s - 5:
            break
        def obj(x):
            corr = np.correlate(x, 1.0-x, mode="full")
            sm = smooth_max_val(corr * dx, beta)
            return sm + 0.5 * rho * (np.sum(x) - target_sum)**2
        def grad(x):
            g = smooth_max_grad(x, dx, beta)
            g += rho * (np.sum(x) - target_sum)
            return g
        try:
            res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                           bounds=Bounds(np.zeros(n), np.ones(n)),
                           options={'maxiter': max_inner, 'ftol': 1e-18})
            h_cur = np.clip(res.x, 0.0, 1.0)
        except:
            pass
    return project_h(h_cur, target_sum)

def subgrad_step(h, target_sum, n_lags=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:n_lags]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:n-lag])
            grad[:n-lag] -= h[lag:]
        else:
            m = -lag
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]
    gn = np.linalg.norm(grad)
    if gn < 1e-14:
        return h, compute_c5(h)
    grad /= gn
    best_h, best_c5 = h, compute_c5(h)
    for s in [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001]:
        ht = project_h(h - s * grad, target_sum)
        c5t = compute_c5(ht)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def rem():
        return budget_s - (time.time() - t0)
    
    # Initialize from parent or default
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
    else:
        best_h = np.full(60, 0.5)
    best_c5 = compute_c5(best_h)
    n_best = len(best_h)

    target = n_best / 2.0
    h_cur = best_h.copy()
    for beta in [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]:
        if rem() < 20:
            break
        h_new = optimize_lbfgs(h_cur, target, beta, t0, budget_s, max_inner=300)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h_cur) - 1e-14:
            h_cur = h_new
        if c5_new < best_c5:
            best_h, best_c5 = h_cur.copy(), c5_new
            n_best = len(h_cur)
    
    # Phase 2: Multi-resolution
    for n_pts in [80, 120, 160, 240, 320, 400, 500]:
        if rem() < 40:
            break
        target = n_pts / 2.0
        x_old = np.linspace(0, 2, n_best)
        x_new = np.linspace(0, 2, n_pts)
        h_cur = project_h(np.interp(x_new, x_old, best_h, clip=True), target)
        for beta in [50, 200, 1000, 5000]:
            if rem() < 15:
                break
            h_new = optimize_lbfgs(h_cur, target, beta, t0, budget_s, max_inner=200)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h_cur) - 1e-14:
                h_cur = h_new
        if compute_c5(h_cur) < best_c5:
            best_h, best_c5, n_best = h_cur.copy(), compute_c5(h_cur), n_pts
    
    # Phase 3: Random restarts at best resolution
    target = n_best / 2.0
    for restart in range(30):
        if rem() < 30:
            break
        # Generate diverse starting points
        if restart == 0:
            h_init = project_h(best_h + np.random.randn(n_best) * 0.1, target)
        elif restart % 3 == 1:
            # Block pattern
            n_blocks = np.random.randint(3, 8)
            h_init = np.tile(np.concatenate([[1.0]*np.random.randint(1,4), [0.0]*np.random.randint(1,4)]), n_best // 8 + 1)[:n_best]
            h_init = project_h(h_init, target)
        elif restart % 3 == 2:

        else:

        for beta in [100, 1000, 5000]:
            if rem() < 10:

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - p.sum()
    if abs(diff) > 1e-13:
        mask = (p > 1e-10) & (p < 1.0 - 1e-10)
        if mask.sum() > 0:
            p[mask] += diff / mask.sum()
            p = np.clip(p, 0.0, 1.0)
    return p

def smooth_max_val(corr_vals, beta):
    m = np.max(corr_vals)
    xs = np.clip(beta * (corr_vals - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(xs)) / beta

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    exps = np.exp(np.clip(beta * (vals - m), -500, 500))
    p = exps / exps.sum()
    grad = np.zeros(n)
    grad += p[n - 1] * (1.0 - 2.0 * h)
    for k in range(1, n):
        pk = p[n - 1 + k]
        if pk > 1e-12:
            grad[k:] += pk * (1.0 - h[:n - k])
            grad[:n - k] -= pk * h[k:]
    for k in range(1, n):
        pk = p[n - 1 - k]
        if pk > 1e-12:
            grad[:n - k] += pk * (1.0 - h[k:])
            grad[k:] -= pk * h[:n - k]
    return grad * dx

def optimized(x, h_init, target_sum, beta, dx, rho):
    corr = np.correlate(x, 1.0 - x, mode="full")
    sm = smooth_max_val(corr * dx, beta)
    return sm + 0.5 * rho * (np.sum(x) - target_sum) ** 2

def optimized_grad(x, dx, beta, target_sum, rho):
    g = smooth_max_grad(x, dx, beta)
    g += rho * (np.sum(x) - target_sum)
    return g

def lbfgs_round(h, target_sum, beta, t0, budget_s, max_iter=300):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    for rho in [5.0, 50.0, 500.0, 5000.0, 50000.0]:
        if time.time() - t0 > budget_s - 5:
            break
        bounds = Bounds(np.zeros(n), np.ones(n))
        def obj(x):
            return optimized(x, None, target_sum, beta, dx, rho)
        def g(x):
            return optimized_grad(x, dx, beta, target_sum, rho)
        try:
            res = minimize(obj, h_cur, jac=g, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-18})
            h_cur = np.clip(res.x, 0.0, 1.0)
        except Exception:
            pass
    return project_h(h_cur, target_sum)

def slsqp_round(h, target_sum, beta, t0, budget_s, max_iter=1000):
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max_val(corr * dx, beta)
    def g(x):
        return smooth_max_grad(x, dx, beta)
    def con(x):
        return np.sum(x) - target_sum
    def con_jac(x):
        return np.ones(n)
    constraints = [{'type': 'eq', 'fun': con, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=g, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': max_iter, 'ftol': 1e-18})
        return project_h(res.x, target_sum)
    except Exception:
        return h

def subgrad_step(h, target_sum, n_lags=7):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:n_lags]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
        else:
            m = -lag
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    gn = np.linalg.norm(grad)
    if gn < 1e-14:
        return h, compute_c5(h)
    grad /= gn
    best_h, best_c5 = h, compute_c5(h)
    for s in [0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]:
        ht = project_h(h - s * grad, target_sum)
        c5t = compute_c5(ht)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
    return best_h, best_c5
