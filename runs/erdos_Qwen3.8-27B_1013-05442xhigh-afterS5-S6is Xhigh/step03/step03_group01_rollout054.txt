        return m + (a + np.log(np.sum(np.exp(sh - a)))) / alpha

                      options={'maxiter': maxiter, 'ftol': ftol})

        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full') * dx))

def get_c(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    r = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    d = target - r.sum()
    if abs(d) > 1e-14:
        m = (r > 1e-10) & (r < 1 - 1e-10)
        if m.sum() > 0:
            r[m] += d / m.sum()
            r = np.clip(r, 0.0, 1.0)
    return r

def grad_c_m(h, n, m):
    dx = 2.0 / n
    off = m - (n - 1)
    g = np.zeros(n)
    if off >= 0:
        for i in range(off, n):
            g[i] += dx * (1.0 - h[i - off])
            g[i - off] -= dx * h[i]
    else:
        d = -off
        for i in range(n - d):
            g[i] += dx * (1.0 - h[i + d])
            g[i + d] -= dx * h[i]
    return g

    a = sh.max()
