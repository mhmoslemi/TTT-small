import time
from scipy.optimize import linprog

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13 * n:
            break
        if diff > 0:

            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:

def grad_B(h, k, n):
    """Gradient of B(k) = sum_n h[n]*h[n+k] w.r.t. h (0-indexed, k can be negative)"""
    g = np.zeros(n)
    ak = abs(k)
    if ak == 0:
        g[:] = 2.0 * h
    else:
        g[:n-ak] += h[ak:]
        g[ak:] += h[:n-ak]
    return g

    """Compute B(k) for all k from -(n-1) to (n-1)"""
    # Using np.correlate(h, h, mode='full')
    # Index m in [0, 2n-2] corresponds to k = m - (n-1)
    c = np.correlate(h, h, mode='full')
    return c  # length 2n-1

def lp_step(h0, n, active_lags, h0_B, h0_grads, maxiter=1):
    """Solve one LP step of the linearized max-min problem"""
    # Variables: [h_0, ..., h_{n-1}, t]
    # Maximize t subject to:
    #   grad_B(k_i)^T h >= t - B(k_i) + grad_B(k_i)^T h0  for each active lag
    #   0 <= h <= 1, sum(h) = n/2
    #
    # In linprog form (minimize c^T x):
    #   c = [0,...,0, -1]
    #   A_ub @ x <= b_ub:
    #     -grad_B(k_i)^T h + t <= B(k_i) - grad_B(k_i)^T h0
    #     -h_j <= 0, h_j <= 1
    #   A_eq @ x = b_eq:
    #     sum(h) = n/2
    
    n_lags = len(active_lags)
    # Build A_ub
    # Row for each lag: [-grad_k, 1]
    # Row for each h_j >= 0: [-e_j, 0]
    # Row for each h_j <= 1: [e_j, 0]
    
    A_ub = np.zeros((n_lags + 2*n, n+1))
    b_ub = np.zeros(n_lags + 2*n)
    
    for idx, ki in enumerate(range(n_lags)):
        k = active_lags[ki]
        Bk = h0_B[k]
        gk = h0_grads[k]
        rhs = Bk - np.dot(gk, h0)
        A_ub[ki, :n] = -gk
        A_ub[ki, n] = 1.0
        b_ub[ki] = rhs
    
    # -h_j <= 0
    A_ub[n_lags:n_lags+n, :n] = -np.eye(n)
    b_ub[n_lags:n_lags+n] = 0
    
    # h_j <= 1
    A_ub[n_lags+n:, :n] = np.eye(n)
    b_ub[n_lags+n:] = 1.0
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    c_obj = np.zeros(n+1)
    c_obj[n] = -1.0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(0,1)]*n + [(None, None)], method='highs')
    
    if res.success:
        new_h = res.x[:n]
        new_t = res.x[n]

    return h0, None

from scipy.optimize import linprog

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12 * n:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def grad_B(h, k, n):
    g = np.zeros(n)
    ak = abs(k)
    if ak == 0:
        g[:] = 2.0 * h
    else:
        g[:n-ak] += h[ak:]
        g[ak:] += h[:n-ak]
    return g

def lp_iteration(h0, n, use_all_lags=True, max_active=200):
    """One LP iteration of the linearized max-min problem. Returns new_h or None."""
    B_vals = np.correlate(h0, h0, mode='full')  # length 2n-1
    B_min = B_vals.min()
    
    # Select active lags
    if use_all_lags:
        active = np.arange(len(B_vals))
    else:
        threshold = B_min + 0.1 * (B_vals.max() - B_min)
        active = np.where(B_vals <= threshold)[0]
        if len(active) > max_active:
            # Take the worst max_active
            order = np.argsort(B_vals)
            active = order[:max_active]
    
    n_active = len(active)
    
    # Build LP
    # Variables: [h_0,...,h_{n-1}, t]
    # Max t s.t.:
    #   for each active lag i: grad_B(k_i)^T h - t >= -B(k_i) + grad_B(k_i)^T h0
    #   equivalently: -grad_B(k_i)^T h + t <= B(k_i) - grad_B(k_i)^T h0
    #   0 <= h <= 1, sum(h) = n/2
    
    n_vars = n + 1
    # A_ub: (n_active + 2n) x n_vars
    A_ub = np.zeros((n_active + 2 * n, n_vars))
    b_ub = np.zeros(n_active + 2 * n)
    
    for idx, lag_idx in enumerate(active):
        k = int(lag_idx) - (n - 1)
        Bk = B_vals[lag_idx]
        gk = grad_B(h0, k, n)
        rhs = Bk - np.dot(gk, h0)
        A_ub[idx, :n] = -gk
        A_ub[idx, n] = 1.0
        b_ub[idx] = rhs
    
    A_ub[n_active:n_active+n, :n] = -np.eye(n)
    b_ub[n_active:n_active+n] = 0.0
    
    A_ub[n_active+n:, :n] = np.eye(n)
    b_ub[n_active+n:] = 1.0
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    c_obj = np.zeros(n_vars)
    c_obj[n] = -1.0
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(0, 1)] * n + [(None, None)], method='highs')
    except Exception:
        return None
    
    if res.success:
        return project_feasible(res.x[:n], n)
    return None

def pair_local(h, n, deadline, deltas):
    """Pairwise local search"""
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            improved = False
            if time.time() > deadline:

            for i in range(n):

                    for s in (1.0, -1.0):

                            h[i] = hi
                            h[j] = hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i] -= s * d
                                h[j] += s * d
    return h, best_c

import numpy as np
import time
from scipy.optimize import linprog

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12 * n:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def grad_B(h, k, n):
    g = np.zeros(n)
    ak = abs(k)
    if ak == 0:
        g[:] = 2.0 * h
    else:
        g[:n - ak] += h[ak:]
        g[ak:] += h[:n - ak]
    return g

def lp_iteration(h0, n, max_active=250):
    B_vals = np.correlate(h0, h0, mode='full')
    B_min = B_vals.min()
    B_spread = B_vals.max() - B_min
    threshold = B_min + 0.15 * B_spread
    active = np.where(B_vals <= threshold)[0]
    if len(active) > max_active:
        order = np.argsort(B_vals)
        active = order[:max_active]
    n_act = len(active)
    n_vars = n + 1
    A_ub = np.zeros((n_act + 2 * n, n_vars))
    b_ub = np.zeros(n_act + 2 * n)
    for idx, li in enumerate(active):
        k = int(li) - (n - 1)
        Bk = B_vals[li]
        gk = grad_B(h0, k, n)
        rhs = Bk - np.dot(gk, h0)
        A_ub[idx, :n] = -gk
        A_ub[idx, n] = 1.0
        b_ub[idx] = rhs
    A_ub[n_act:n_act + n, :n] = -np.eye(n)
    b_ub[n_act:n_act + n] = 0.0
    A_ub[n_act + n:, :n] = np.eye(n)
    b_ub[n_act + n:] = 1.0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c_obj = np.zeros(n_vars)
    c_obj[n] = -1.0
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(0, 1)] * n + [(None, None)], method='highs')
    except Exception:
        return None
    if res.success:
        return project_feasible(res.x[:n], n)
    return None

def pair_local(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i + 1, n):
                    for s in (1.0, -1.0):
                        hi = h[i] + s * d
                        hj = h[j] - s * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i] = hi
                            h[j] = hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                improved = True
                            else:
                                h[i] -= s * d
                                h[j] += s * d
    return h, best_c

def subgradient_refine(h, n, deadline, max_iter=500):
    h = h.copy()
    c_current = compute_c5(h, n)
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        B_vals = np.correlate(h, h, mode='full')
        min_idx = B_vals.argmin()
        k = min_idx - (n - 1)
        g = grad_B(h, k, n)
        B_min = B_vals.min()
        B_max = B_vals.max()
        step = (B_max - B_min) / (np.dot(g, g) + 1e-8)
        step = min(step, 0.3)
        h_new = project_feasible(h + step * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < c_current - 1e-14:
            h = h_new
            c_current = c_new
    return h, c_current

def lp_phase(h, n, deadline):
    h = h.copy()
    c_current = compute_c5(h, n)
    stall_count = 0
    for _ in range(5000):
        if time.time() > deadline:
            break
        new_h = lp_iteration(h, n)
        if new_h is None:
            break
        c_new = compute_c5(new_h, n)
        if c_new < c_current - 1e-12:
            h = new_h
            c_current = c_new
            stall_count = 0
        elif c_new < c_current + 1e-10:
            # Try fractional step
            found = False
            for alpha in [0.7, 0.5, 0.3, 0.1]:
                h_f = (1.0 - alpha) * h + alpha * new_h
                h_f = project_feasible(h_f, n)
                c_f = compute_c5(h_f, n)
                if c_f < c_current - 1e-13:
                    h = h_f
                    c_current = c_f
                    found = True
                    break
            if not found:
                stall_count += 1
                if stall_count > 15:
                    break
        else:
            stall_count += 1
            if stall_count > 15:
                break
    return h, c_current
