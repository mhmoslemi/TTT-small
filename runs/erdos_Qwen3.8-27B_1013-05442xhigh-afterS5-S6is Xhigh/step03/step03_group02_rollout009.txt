```python
import time

def project_h(h, target_sum):

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new, endpoint=False),
                     np.linspace(0, 2, len(h), endpoint=False), h)

def sml_objective(h, beta, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * dx

def sml_grad(h, beta, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    x_shift = np.clip(beta * (c - m), -500, 500)
    weights = np.exp(x_shift) / np.sum(np.exp(x_shift))
    grad = np.zeros(len(h))
    n = len(h)
    for i, w in enumerate(weights):
        if w < 1e-15:

        lag = int(i) - (n - 1)
        if lag >= 0:
            nv = n - lag
            grad[:nv] += w * (1 - h[lag:])
            grad[lag:] -= w * h[:nv]
        else:
            la = -lag
            nv = n - la
            grad[la:] += w * (1 - h[:nv])
            grad[:nv] -= w * h[la:]
    return grad * dx

def slsqp_optimize(h0, n, target, beta, maxiter, t0, budget):

        return sml_objective(x, beta, 2.0 / n)

        return sml_grad(x, beta, 2.0 / n)

                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})

    except:
```

The subgradient of max_k f_k(h) at h is the weighted average of the subgradients of the active f_k's, where weights are proportional to how close f_k is to the max. This is exactly what the smooth max gradient approximates, but we can do it exactly:

```python
def exact_subgradient(h, top_k=3):
    """Compute exact subgradient of max_k f_k(h)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Use top-k lags for numerical stability
    max_val = corr.max()
    top_idx = np.argsort(corr)[::-1][:top_k]
    # Uniform weights on top-k
    weights = np.ones(top_k) / top_k
    grad = np.zeros(len(h))
    n = len(h)
    for idx, w in zip(top_idx, weights):
        lag = int(idx) - (n - 1)
        if lag >= 0:
            nv = n - lag
            grad[:nv] += w * (1 - h[lag:])
            grad[lag:] -= w * h[:nv]
        else:
            la = -lag
            nv = n - la
            grad[la:] += w * (1 - h[:nv])
            grad[:nv] -= w * h[la:]
    return grad, max_val
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def sml_obj(h, beta, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * dx

def sml_grad(h, beta, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    x_shift = np.clip(beta * (c - m), -500, 500)
    w = np.exp(x_shift) / np.sum(np.exp(x_shift))
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(w)):
        if w[i] < 1e-14:
            continue
        lag = int(i) - (n - 1)
        if lag >= 0:
            nv = n - lag
            grad[:nv] += w[i] * (1.0 - h[lag:])
            grad[lag:] -= w[i] * h[:nv]
        else:
            la = -lag
            nv = n - la
            grad[la:] += w[i] * (1.0 - h[:nv])
            grad[:nv] -= w[i] * h[la:]
    return grad * dx

def sum_eq(x, target):
    return x.sum() - target

def sum_eq_jac(x):
    return np.ones(x.shape)

def slsqp_optimize(h0, n, target, beta, maxiter, t0, budget):
    if budget - (time.time() - t0) < 5:
        return h0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, jac=sml_grad, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(r.x, target)
    except Exception:
        return h0

def resample_h(h, n_new):
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def differential_evolution_c5(n, target, pop_size, max_gen, seed_val, t0, budget):
    rng = np.random.RandomState(seed_val)
    pop = np.array([project_h(rng.rand(n), target) for _ in range(pop_size)])
    costs = np.array([compute_c5(p) for p in pop])
    best_h = pop[np.argmin(costs)].copy()
    best_c = costs.min()
    CR, F = 0.8, 0.7
    for gen in range(max_gen):
        if budget - (time.time() - t0) < 2:
            break
        for i in range(pop_size):
            if budget - (time.time() - t0) < 0.5:
                break
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, 3, replace=False)
            trial = pop[a] + F * (pop[b] - pop[c])
            mask = rng.rand(n) < CR
            if not mask.any():
                mask[rng.randint(n)] = True
            trial = np.where(mask, trial, pop[i])
            trial = project_h(trial, target)
            tc = compute_c5(trial)
            if tc <= costs[i]:
                pop[i] = trial
                costs[i] = tc
                if tc < best_c:
                    best_c = tc
                    best_h = trial.copy()
    return best_h, best_c

        nf = len(best_h)
        tf = nf / 2.0
        for beta in [3000, 10000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 5:
                break
            best_h = slsqp_optimize(best_h, nf, tf, beta, 5000, t0, budget_s)
            best_c5 = compute_c5(best_h)
        remaining = budget_s - (time.time() - t0)
        if remaining > 10:
            best_h, best_c5 = active_set_lp(best_h, tf, 8, 400, t0, budget_s)
        remaining = budget_s - (time.time() - t0)
        if remaining > 5:
            best_h, best_c5 = coord_descent(best_h, tf, 40, t0, budget_s)
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
