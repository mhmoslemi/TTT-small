def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)

            if free.sum() > 0:

from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -50, 0))  # clip for numerical stability
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[:n-s] += wk * (1.0 - h[s:])
            grad[s:] += wk * (-h[:n-s])
        elif s < 0:
            t = -s
            grad[t:] += wk * (1.0 - h[:n-t])
            grad[:n-t] += wk * (-h[t:])
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    grad *= dx
    return val, grad

def fista_optimize(h, max_iters, beta, lr_init):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    lr = lr_init
    
    for t in range(max_iters):
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2.0)
            y = x + mu * (x - x_prev)
            y = np.clip(y, 0.0, 1.0)
        
        val_y, grad_y = smoothmax_val_grad(y, beta)
        x_new = project_feasible(y - lr * grad_y, target)
        
        # Periodically check actual C5
        if t % 100 == 0 or t == max_iters - 1:
            c5 = compute_c5(x_new)
            if c5 < best_v:
                best_v = c5
                best_x = x_new.copy()
        
        x_prev = x
        x = x_new
    
    # Final check
    c5 = compute_c5(x)
    if c5 < best_v:
        best_v = c5
        best_x = x.copy()
    
    return best_x, best_v

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    
    def grad(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

    for t in range(max_iters):
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2.0)
            y = x + mu * (x - x_prev)
            y = np.clip(y, 0.0, 1.0)
        val_y, grad_y = smoothmax_val_grad(y, beta)
        x_new = project_feasible(y - lr * grad_y, target)
        if t % check_interval == 0 or t == max_iters - 1:
            c5 = compute_c5(x_new)
            if c5 < best_v:
                best_v = c5
                best_x = x_new.copy()
        x_prev = x
        x = x_new
    c5 = compute_c5(x)
    if c5 < best_v:
        best_v = c5
        best_x = x.copy()
    return best_x, best_v

def fista_topk(h, max_iters, lr_init, top_k, check_interval=200):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    lr = lr_init
    for t in range(max_iters):
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2.0)
            y = x + mu * (x - x_prev)
            y = np.clip(y, 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for ki in top_idx:
            s = ki - (n - 1)
            if s >= 0:
                grad[:n-s] += (1.0 - y[s:])
                grad[s:] += (-y[:n-s])
            elif s < 0:
                t2 = -s
                grad[t2:] += (1.0 - y[:n-t2])
                grad[:n-t2] += (-y[t2:])
            else:
                grad[:] += (1.0 - 2.0 * y)
        grad = grad / top_k * dx
        x_new = project_feasible(y - lr * grad, target)
        if t % check_interval == 0 or t == max_iters - 1:
            c5 = compute_c5(x_new)
            if c5 < best_v:
                best_v = c5
                best_x = x_new.copy()
        x_prev = x
        x = x_new
    c5 = compute_c5(x)
    if c5 < best_v:
        best_v = c5
        best_x = x.copy()
    return best_x, best_v

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    def gr(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    def con_f(x):
        return x.sum() - target
    def con_j(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    try:
        res = minimize(obj, h, jac=gr, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(400, 0.5)
    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: FISTA top-k aggressive (reduces max lags directly)
    for top_k in [4, 8, 12]:
        if tl() < 80:
            break
        iters = min(8000, max(2000, int(tl() * 1.5)))
        h_new, c_new = fista_topk(h_best, iters, lr_init=8.0, top_k=top_k)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new

    # Phase 2: FISTA smooth max with increasing beta (smooth convergence)
    for beta in [500, 2000, 5000, 10000]:
        if tl() < 60:
            break
        iters = min(6000, max(2000, int(tl() * 1.2)))
        h_new, c_new = fista_smoothmax(h_best, iters, beta, lr_init=10.0)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new

    # Phase 3: SLSQP polish at high beta
    for beta in [5000, 15000]:
        if tl() < 40:
            break
        h_new = slsqp_refine(h_best, maxiter=350, beta=beta)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 4: Multi-start with diverse perturbations
    for trial in range(8):
        if tl() < 60:
            break
        # Mix of smooth and rough perturbations
        if trial % 2 == 0:
            smooth = np.cumsum(np.random.randn(n)) * 0.005
            smooth -= smooth.mean()
            pert = smooth
        else:
            pert = np.random.randn(n) * 0.008
        h_pert = project_feasible(h_best + pert, n / 2.0)
        h_new, c_new = fista_topk(h_pert, 3000, lr_init=8.0, top_k=8)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new
        if tl() > 30:
            h_new2, c_new2 = fista_smoothmax(h_best, 2000, 8000, lr_init=10.0)
            if c_new2 < best_c5:
                h_best, best_c5 = h_new2, c_new2
            h_new3 = slsqp_refine(h_best, maxiter=200, beta=8000)
            c3 = compute_c5(h_new3)
            if c3 < best_c5:
                h_best, best_c5 = h_new3, c3

    # Phase 5: Higher resolution
    for n_try in [600, 800]:
        if tl() < 180:
            break
        h_interp = interp_to_n(h_best, n_try)
        for beta in [1000, 5000, 10000]:
            if tl() < 60:
                break
            iters = min(5000, max(1500, int(tl() * 1.0)))
            h_new, c_new = fista_smoothmax(h_interp, iters, beta, lr_init=6.0)
            if c_new < best_c5:
                h_best, best_c5 = h_new, c_new
                h_interp = h_best.copy()
        if tl() > 50:
            h_tk, c_tk = fista_topk(h_best, 4000, lr_init=5.0, top_k=10)
            if c_tk < best_c5:
                h_best, best_c5 = h_tk, c_tk
        n = len(h_best)

    # Phase 6: Final aggressive push
    if tl() > 40:
        h_new, c_new = fista_topk(h_best, min(5000, int(tl() * 1.5)), lr_init=6.0, top_k=5)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new
    if tl() > 30:
        h_final = slsqp_refine(h_best, maxiter=400, beta=20000)
        c = compute_c5(h_final)
        if c < best_c5:
            h_best, best_c5 = h_final, c

    # Final feasibility
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
