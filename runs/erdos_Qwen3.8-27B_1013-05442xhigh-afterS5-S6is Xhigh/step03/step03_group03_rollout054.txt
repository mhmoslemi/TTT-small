    w = np.exp(shifted)
    w /= w.sum()
    return w

    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            idx = np.arange(m)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
    else:

        m = n - la
        if m > 0:
            idx = np.arange(m)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_value(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    w = np.exp(shifted)
    w /= w.sum()
    return w

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            idx = np.arange(m)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
    else:
        la = -lag
        m = n - la
        if m > 0:
            idx = np.arange(m)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    return grad

def full_gradient(h, corr, n, beta):
    w = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        grad += w[k] * subgrad_for_lag(h, lag, n)
    return grad

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_value(corr, beta) * dx

def con_fun(h, target_sum, n):
    return h.sum() - target_sum

def con_jac(h, target_sum, n):
    return np.ones(n)

def make_slsqp_cons(target_sum, n):
    return {'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(210, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    best_h = project_feasible(h_init.copy(), target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: Nesterov gradient descent with full gradient and adaptive beta
    elapsed = time.time() - t0
    if budget_s - elapsed > 250:
        h = best_h.copy()
        v = np.zeros(n)
        beta = 50.0
        mu = 0.92
        lr = 0.03
        for it in range(60000):
            elapsed = time.time() - t0
            if budget_s - elapsed < 250:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            grad = full_gradient(h, corr, n, beta)
            v = mu * v - lr * grad
            h_new = project_feasible(h + v, target_sum)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c_new
            if c_new < best_c5 + 0.00005:
                h = h_new
            else:
                lr *= 0.7
                v *= 0.3
            if it % 200 == 199:
                beta = min(beta * 1.5, 50000.0)
                if beta > 200:
                    beta = 200.0

    # Phase 2: SLSQP continuation with increasing beta
    elapsed = time.time() - t0
    if budget_s - elapsed > 250:
        slsqp_cons = make_slsqp_cons(target_sum, n)
        for beta in [50, 200, 1000, 5000, 20000]:
            elapsed = time.time() - t0
            if budget_s - elapsed < 60:
                break
            h_start = best_h.copy()
            try:
                res = minimize(obj_smooth, h_start, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=[slsqp_cons],
                               options={'maxiter': 8000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 3: Structured restarts with re-optimization
    elapsed = time.time() - t0
    if budget_s - elapsed > 250:
        slsqp_cons = make_slsqp_cons(target_sum, n)
        x_grid = np.linspace(0, 2, n)
        for restart in range(30):
            elapsed = time.time() - t0
            if budget_s - elapsed < 80:
                break
            r = np.random.random()
            if r < 0.25:
                freq = np.random.randint(2, 12)
                phase = np.random.uniform(0, 2 * np.pi)
                amp = np.random.uniform(0.03, 0.12)
                pert = amp * np.sin(2 * np.pi * freq * x_grid / 2.0 + phase)
            elif r < 0.5:
                pert = np.random.randn(n) * np.random.uniform(0.03, 0.08)
                kernel = np.ones(7) / 7.0
                pert = np.convolve(pert, kernel, mode='same')
            elif r < 0.75:
                pert = np.zeros(n)
                blen = max(n // 6, 5)
                b1 = np.random.randint(0, n - blen)
                pert[b1:b1 + blen] = np.random.uniform(-0.1, 0.1)
                if b1 + blen < n:
                    b2 = np.random.randint(b1 + blen, n)
                    pert[b2:] = np.random.uniform(-0.08, 0.08)
            else:
                pert = np.zeros(n)
                idx = np.random.choice(n, size=n // 3, replace=False)
                pert[idx] = np.random.uniform(-0.06, 0.06, size=len(idx))

            h_try = project_feasible(best_h + pert, target_sum)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_h = h_try.copy()
                best_c5 = c_try

            if budget_s - elapsed > 40:
                for beta_s in [200, 2000]:
                    elapsed = time.time() - t0
                    if budget_s - elapsed < 25:
                        break
                    try:
                        res = minimize(obj_smooth, h_try, args=(dx, beta_s), method='SLSQP',
                                       bounds=bounds, constraints=[slsqp_cons],
                                       options={'maxiter': 5000, 'ftol': 1e-14})
                        h_new = project_feasible(res.x, target_sum)
                        c5 = compute_c5(h_new)
                        if c5 < best_c5:
                            best_h = h_new.copy()
                            best_c5 = c5
                            h_try = h_new
                    except Exception:
                        pass

    # Phase 4: Resolution exploration
    elapsed = time.time() - t0
    if budget_s - elapsed > 180:
        res_list = []
        for mult in [0.7, 0.8, 1.2, 1.5, 2.0]:
            res_list.append(int(n * mult))
        for offset in [-30, -20, -10, 10, 20, 30, 50, 80]:
            res_list.append(n + offset)
        res_list = sorted(set([r for r in res_list if 60 <= r <= 600]))

        for n_try in res_list:
            if n_try == n or budget_s - elapsed < 80:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_t = np.interp(x_new, x_old, best_h)
            h_t = project_feasible(h_t, ts_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            cons_t = make_slsqp_cons(ts_t, n_try)
            for beta in [100, 1000, 5000]:
                elapsed = time.time() - t0
                if budget_s - elapsed < 35:
                    break
                try:
                    res = minimize(obj_smooth, h_t, args=(dx_t, beta), method='SLSQP',
                                   bounds=bnds_t, constraints=[cons_t],
                                   options={'maxiter': 4000, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                        n = n_try
                        dx = dx_t
                        target_sum = ts_t
                        bounds = bnds_t
                        h_t = h_new
                except Exception:
                    pass

    # Phase 5: Final high-beta SLSQP polishing
    elapsed = time.time() - t0
    if budget_s - elapsed > 20:
        slsqp_cons = make_slsqp_cons(target_sum, n)
        for beta in [20000, 80000, 300000]:
            elapsed = time.time() - t0
            if budget_s - elapsed < 12:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=[slsqp_cons],
                               options={'maxiter': 12000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 6: Micro gradient polish
    elapsed = time.time() - t0
    if budget_s - elapsed > 10:
        h = best_h.copy()
        lr = 0.002
        for it in range(10000):
            elapsed = time.time() - t0
            if budget_s - elapsed < 5:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            grad = full_gradient(h, corr, n, 3000)
            h_new = project_feasible(h - lr * grad, target_sum)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c_new
                h = h_new
            else:
                lr *= 0.85
                if lr < 1e-9:
                    break

    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
