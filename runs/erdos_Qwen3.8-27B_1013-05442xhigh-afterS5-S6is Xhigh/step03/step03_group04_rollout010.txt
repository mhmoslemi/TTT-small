from scipy.optimize import minimize, Bounds
import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

        if free.sum() == 0:

    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500, 0)

def compute_grad(h, w, dx):
    n = len(h)
    g = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 1e-15:

def slsqp_polish(h, target, dx, t0, budget_s):

    best = h.copy()
    best_c5 = compute_c5(h)
    for beta in [500, 1000, 2000, 4000, 8000]:

            return smooth_max_val(corr, beta) * dx
