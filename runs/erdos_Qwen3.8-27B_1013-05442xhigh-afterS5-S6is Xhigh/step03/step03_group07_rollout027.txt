        if s >= 0:
            grad[s:] += wi * (1.0 - h[:n - s])
            grad[:n - s] += wi * (-h[s:])
        else:
            t = -s
            grad[:n - t] += wi * (1.0 - h[t:])
            grad[t:] += wi * (-h[:n - t])
    grad *= dx
    return val, grad

def construct_sym(g):
    n = len(g)
    h = np.empty(2 * n)
    h[:n] = g
    h[n:] = 1.0 - g
    return h

def obj_full(x):
    return smoothmax_val_grad(x, _beta_g)[0]

def grad_full(x):
    return smoothmax_val_grad(x, _beta_g)[1]

def con_sum(x):
    return x.sum() - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def obj_sym(x):
    h = construct_sym(x)
    return smoothmax_val_grad(h, _beta_g)[0]

def grad_sym(x):
    h = construct_sym(x)
    _, gh = smoothmax_val_grad(h, _beta_g)
    n = len(x)
    return gh[:n] - gh[n:]

def slsqp_full(h0, beta, maxiter):
    global _beta_g
    _beta_g = beta
    n = len(h0)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    try:
        res = minimize(obj_full, h0, jac=grad_full, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h0.copy(), target)

def lbfgs_sym(g0, beta, maxiter):
    global _beta_g
    _beta_g = beta
    n = len(g0)
    try:
        res = minimize(obj_sym, g0.copy(), jac=grad_sym, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return g0.copy()

def topk_grad(h, top_idx):
    n = len(h)
    grad = np.zeros(n)
    for i in top_idx:
        s = i - (n - 1)
        if s >= 0:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] += (-h[s:])
        else:
            t = -s
            grad[:n - t] += (1.0 - h[t:])
            grad[t:] += (-h[:n - t])
    return grad

def topk_nesterov(h0, max_iter, lr0, decay, top_k):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    x = h0.copy()
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iter):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = project_feasible(x + mu * (x - x_prev), target)
        corr = compute_corr(y)
        top_idx = np.argsort(corr)[-top_k:]
        grad = topk_grad(y, top_idx) * (dx / top_k)
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def nesterov_pg(h0, beta, max_iter, lr0, decay):
    n = len(h0)
    target = n / 2.0
    x = h0.copy()
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iter):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = project_feasible(x + mu * (x - x_prev), target)
        _, grad = smoothmax_val_grad(y, beta)
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    global _t0_g, _budget_g, _beta_g
    np.random.seed(seed)
    _t0_g = time.time()
    _budget_g = budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(400, 0.5)

    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: Symmetric L-BFGS-B (200 vars, no sum constraint needed)
    n_half = n // 2
    g_init = h_best[:n_half].copy()
    for beta in [20, 60, 150, 400, 1000]:
        if rem() < 25:
            break
        g_new = lbfgs_sym(g_init, beta, 500)
        h_sym = construct_sym(g_new)
        c = compute_c5(h_sym)
        if c < best_c5:
            h_best, best_c5 = h_sym, c
            g_init = g_new.copy()

    # Phase 2: Full SLSQP with increasing beta
    for beta in [80, 200, 500, 1200, 2500]:
        if rem() < 35:
            break
        h_new = slsqp_full(h_best, beta, 400)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 3: Top-k Nesterov (fast, focused on worst lags)
    if rem() > 60:
        for tk in [3, 5, 8]:
            if rem() < 30:
                break
            h_sg, c_sg = topk_nesterov(h_best, 10000, 0.06, 0.9999, top_k=tk)
            if c_sg < best_c5:
                h_best, best_c5 = h_sg, c_sg
        if rem() > 30:
            h_new = slsqp_full(h_best, 2000, 350)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c

    # Phase 4: Full smoothmax Nesterov
    if rem() > 50:
        h_sg, c_sg = nesterov_pg(h_best, 2000, 8000, 0.012, 0.99995)
        if c_sg < best_c5:
            h_best, best_c5 = h_sg, c_sg

    # Phase 5: Shake and re-optimize
    for trial in range(10):
        if rem() < 35:
            break
        scale = 0.003 + 0.003 * trial
        pert = np.random.randn(len(h_best)) * scale
        h_pert = project_feasible(h_best + pert, len(h_best) / 2.0)
        h_new = slsqp_full(h_pert, 1500, 400)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 6: Random binary restarts
    if rem() > 80:
        for _ in range(3):
            if rem() < 40:
                break
            perm = np.random.permutation(len(h_best))
            h_bin = np.zeros(len(h_best))
            h_bin[perm[:len(h_best) // 2]] = 1.0
            h_new = slsqp_full(h_bin, 200, 600)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c
            if rem() > 30:
                h_new = slsqp_full(h_best, 1500, 300)
                c = compute_c5(h_new)
                if c < best_c5:
                    h_best, best_c5 = h_new, c

    # Phase 7: Higher resolution
    for n_try in [500, 600]:
        if rem() < 130:
            break
        h_interp = interp_to_n(h_best, n_try)
        for beta in [150, 500, 1200]:
            if rem() < 45:
                break
            h_new = slsqp_full(h_interp, beta, 350)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c
                h_interp = h_best.copy()
        n = len(h_best)

    # Phase 8: Final very high beta polish
    if rem() > 30:
        h_final = slsqp_full(h_best, 5000, 500)
        c = compute_c5(h_final)
        if c < best_c5:
            h_best, best_c5 = h_final, c

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
