    """Compute the delta in corr when h[a]: 1->0 and h[b]: 0->1"""
    delta = np.zeros(2*n - 1)
    
    # a: 1 -> 0
    # As h[j+lag=a]: j ranges, lag = a-j, corr index = n-1+a-j
    j_range_a = range(max(0, a - n + 1), min(a, n - 1) + 1)

import numpy as np
from scipy.optimize import minimize, differential_evolution, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def binary_local_search(h, n, max_passes=50, t0=0, time_limit=200):
    """Fast 2-flip local search for binary h."""
    h = h.astype(float).copy()
    target = n // 2
    # Ensure exactly target ones
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > target:
        h[ones[target:]] = 0
    elif len(ones) < target:
        h[zeros[:target - len(ones)]] = 1
    
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    current_max = float(np.max(corr))
    lags = np.arange(2 * n - 1) - (n - 1)
    
    for p in range(max_passes):
        if time.time() - t0 > t0 + time_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        # Shuffle for random order
        rng = np.random.default_rng(42 + p)
        rng.shuffle(ones)
        rng.shuffle(zeros)
        
        improved = False
        for a in ones:
            for b in zeros:
                # Compute delta
                delta = np.zeros(2 * n - 1)
                m1 = (a + lags >= 0) & (a + lags < n)
                delta[m1] += h[a + lags[m1]] - 1.0
                m2 = (a - lags >= 0) & (a - lags < n)
                delta[m2] += h[a - lags[m2]]
                m3 = (b + lags >= 0) & (b + lags < n)
                delta[m3] += 1.0 - h[b + lags[m3]]
                m4 = (b - lags >= 0) & (b - lags < n)
                delta[m4] -= h[b - lags[m4]]
                
                new_max = float(np.max(corr + delta))
                if new_max < current_max - 1e-10:
                    h[a] = 0.0
                    h[b] = 1.0
                    corr += delta
                    current_max = new_max
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    
    return h, current_max * (2.0 / n)

def cutting_plane(h, n, max_iters=100, t0=0, time_limit=100):
    target = n / 2.0
    h = h.copy()
    dx = 2.0 / n
    
    for it in range(max_iters):
        if time.time() - t0 > t0 + time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        current_max = float(np.max(corr))
        
        # Find active lags (top 10)
        top_idx = np.argsort(corr)[::-1][:10]
        
        # Linearize at current h
        lags_arr = np.arange(2*n-1) - (n-1)
        
        # Gradient of corr[k] w.r.t. h[i]:
        # corr[k] = sum_j h[j]*(1-h[j+lag])
        # d(corr[k])/d(h[i]) = (1-h[i+lag]) [if valid] - h[i-lag] [if valid]
        
        # Solve: min t
        #        s.t. corr[k_0] + grad_k · (h - h_0) ≤ t for k in top_idx
        #             sum(h) = target
        #             0 ≤ h ≤ 1
        #
        # This is a LP with n+1 variables (h[0],...,h[n-1], t) and n+10+1 constraints
        
        from scipy.optimize import linprog
        
        # Variables: x[0:n] = h, x[n] = t
        c = np.zeros(n + 1)
        c[n] = 1.0  # minimize t
        
        # Inequality constraints: A_ub @ x ≤ b_ub
        # For each active lag k: grad_k · h ≤ t - corr[k_0] + grad_k · h_0
        # i.e., grad_k · h - t ≤ -corr[k_0] + grad_k · h_0
        A_ub = np.zeros((len(top_idx), n + 1))
        b_ub = np.zeros(len(top_idx))
        
        for ii, k in enumerate(top_idx):
            lag = lags_arr[k]
            grad = np.zeros(n)
            for i in range(n):
                # (1-h[i+lag]) if 0 ≤ i+lag < n
                if 0 <= i + lag < n:
                    grad[i] += (1.0 - h[i + lag])
                # -h[i-lag] if 0 ≤ i-lag < n
                if 0 <= i - lag < n:
                    grad[i] -= h[i - lag]
            
            A_ub[ii, :n] = grad
            A_ub[ii, n] = -1.0
            b_ub[ii] = -corr[k] + grad @ h
        
        # Equality: sum(h) = target
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        bounds = [(0, 1)] * n + [(current_max * 0.5, None)]
        
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
        
        if res.success:

            c5_new = compute_c5(h_new)

                h = h_new
            # Even if not improved, the linearization helps guide the next step
    
    return h

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def flip_delta(h, a, b, n, lags):
    d = np.zeros(2 * n - 1)
    m1 = (a + lags >= 0) & (a + lags < n) & (lags != 0)
    d[m1] += h[a + lags[m1]] - 1.0
    m2 = (a - lags >= 0) & (a - lags < n) & (lags != 0)
    d[m2] += h[a - lags[m2]]
    m3 = (b + lags >= 0) & (b + lags < n) & (lags != 0)
    d[m3] += 1.0 - h[b + lags[m3]]
    m4 = (b - lags >= 0) & (b - lags < n) & (lags != 0)
    d[m4] -= h[b - lags[m4]]
    return d

def binary_local_search(h_bin, n, t0, time_limit=150):
    h = h_bin.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    cur_max = float(np.max(corr))
    lags = np.arange(2 * n - 1) - (n - 1)
    for p in range(800):
        if time.time() - t0 > t0 + time_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng = np.random.default_rng(1000 + p)
        for _ in range(3000):
            a = ones[rng.integers(len(ones))]
            b = zeros[rng.integers(len(zeros))]
            d = flip_delta(h, a, b, n, lags)
            new_max = float(np.max(corr + d))
            if new_max < cur_max - 1e-12:
                h[a] = 0.0
                h[b] = 1.0
                corr += d
                cur_max = new_max
    return h, cur_max * (2.0 / n)

def coordinate_descent(h, n, t0, time_limit=200, max_passes=40):
    target = n / 2.0
    h = h.copy()
    lags = np.arange(2 * n - 1) - (n - 1)
    for p in range(max_passes):
        if time.time() - t0 > t0 + time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_before = float(np.max(corr))
        improved = False
        for i in range(n):
            slope = np.ones(2 * n - 1)
            m1 = (i + lags >= 0) & (i + lags < n)
            slope[m1] -= h[i + lags[m1]]
            m2 = (i - lags >= 0) & (i - lags < n)
            slope[m2] -= h[i - lags[m2]]
            A = corr - h[i] * slope
            grid = np.linspace(0, 1, 33)
            objs = np.max(A[None, :] + grid[:, None] * slope[None, :], axis=1)
            bi = np.argmin(objs)
            lo = max(0.0, grid[max(0, bi - 1)])
            hi = min(1.0, grid[min(32, bi + 1)])
            for _ in range(15):
                m1v = lo + (hi - lo) * 0.333
                m2v = hi - (hi - lo) * 0.333
                if np.max(A + m1v * slope) <= np.max(A + m2v * slope):
                    hi = m2v
                else:
                    lo = m1v
            bt = (lo + hi) / 2.0
            if abs(bt - h[i]) > 1e-10:
                h[i] = bt
                corr = A + bt * slope
                improved = True
        if improved:
            h = project_h(h, target)
        else:
            break
    return h

def slsqp_polish(h, n, t0, time_limit=120):
    target = n / 2.0
    dx = 2.0 / n
    best_h, best_c5 = h.copy(), compute_c5(h)
    h_cur = best_h.copy()
    bounds = Bounds(np.zeros(n), np.ones(n))
    ceq = lambda x: np.sum(x) - target
    cjac = lambda x: np.ones(n)
    constr = [{'type': 'eq', 'fun': ceq, 'jac': cjac}]
    for beta in [50, 150, 400, 1000, 2500, 5000]:
        if time.time() - t0 > t0 + time_limit:
            break
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode="full")
            return smooth_max(c, beta) * dx
        try:
            res = minimize(obj, h_cur, method='L-BFGS-B', bounds=bounds, options={'maxiter': 500, 'ftol': 1e-16})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h_cur) - 1e-15:
                h_cur = h_new
            if c5_new < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new
        except Exception:
            pass
    try:
        res = minimize(lambda x: np.max(np.correlate(x, 1.0-x, mode="full"))*dx,
                       best_h, method='SLSQP', bounds=bounds, constraints=constr,
                       options={'maxiter': 2000, 'ftol': 1e-14})
        h_new = project_h(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new.copy(), c5_new
    except Exception:
        pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    else:
        n = 280
        h0 = np.full(n, 0.5)

    h0 = project_h(h0, n / 2.0)
    best_h, best_c5 = h0.copy(), compute_c5(h0)

    def elapsed():
        return time.time() - t0

    # Phase 1: Binary local search from parent
    if budget_s - elapsed() > 60:
        hb = to_binary(best_h, n)
        hb, cb = binary_local_search(hb, n, t0, time_limit=120)
        if cb < best_c5:
            best_h, best_c5 = hb.copy(), cb

    # Phase 2: Coordinate descent from best binary
    if budget_s - elapsed() > 60:
        hc = coordinate_descent(best_h, n, t0, time_limit=150)
        cc = compute_c5(hc)
        if cc < best_c5:
            best_h, best_c5 = hc.copy(), cc

    # Phase 3: SLSQP polish
    if budget_s - elapsed() > 30:
        best_h, best_c5 = slsqp_polish(best_h, n, t0, time_limit=120)

    # Phase 4: Multi-start perturbations
    for s in range(8):
        if budget_s - elapsed() < 80:
            break
        rng = np.random.default_rng(seed * 100 + s)
        if s == 0:
            hp = best_h.copy()
        elif s == 1:
            hp = to_binary(best_h, n)
        elif s == 2:
            hp = np.zeros(n)
            idx = rng.choice(n, n // 2, replace=False)
            hp[idx] = 1.0
        elif s == 3:
            hp = np.zeros(n)
            idx = np.arange(0, n, 2)
            hp[idx] = 1.0
        elif s == 4:
            k = n // 4
            hp = np.tile([1.0, 1.0, 0.0, 0.0], n // 4).astype(float)
            hp = hp[:n]
        else:
            hp = best_h + rng.normal(0, 0.15, n)
            hp = np.clip(hp, 0, 1)
        hp = project_h(hp, n / 2.0)

        # Quick binary local search
        if budget_s - elapsed() < 40:
            continue
        hb = to_binary(hp, n)
        rem = min(budget_s - elapsed() - 30, 80)
        if rem > 10:
            hb, cb = binary_local_search(hb, n, t0 + elapsed(), time_limit=rem)
            if cb < best_c5:
                best_h, best_c5 = hb.copy(), cb
        # Coordinate descent
        if budget_s - elapsed() < 30:
            continue
        rem2 = min(budget_s - elapsed() - 20, 60)
        if rem2 > 10:
            hc = coordinate_descent(hb if hasattr(hb, '__len__') and (hb==1).sum()==n//2 else hp, n,
                                    t0 + elapsed(), time_limit=rem2)
            cc = compute_c5(hc)
            if cc < best_c5:
                best_h, best_c5 = hc.copy(), cc
        # Quick SLSQP
        if budget_s - elapsed() > 20:
            bh, bc = slsqp_polish(best_h, n, t0 + elapsed(), time_limit=min(budget_s - elapsed() - 5, 60))
            if bc < best_c5:
                best_h, best_c5 = bh.copy(), bc

    # Final: coordinate descent + SLSQP on the best
    if budget_s - elapsed() > 30:
        hc = coordinate_descent(best_h, n, t0 + elapsed(), time_limit=min(60, budget_s - elapsed() - 20))
        cc = compute_c5(hc)
        if cc < best_c5:
            best_h, best_c5 = hc.copy(), cc
    if budget_s - elapsed() > 20:
        best_h, best_c5 = slsqp_polish(best_h, n, t0 + elapsed(), time_limit=min(60, budget_s - elapsed() - 5))

    best_h = project_h(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, int(n)
