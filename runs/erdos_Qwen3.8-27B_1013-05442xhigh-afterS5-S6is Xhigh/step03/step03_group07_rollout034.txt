```python
def coordinate_minimize_1d(a_coeffs, b_coeffs, c_coeffs, n_grid=500):
    """Minimize max_s (a_s*h^2 + b_s*h + c_s) over h in [0,1]"""
    h_vals = np.linspace(0, 1, n_grid)
    # Compute all c_s(h_vals) as a matrix
    # c_s(h) = a_s*h^2 + b_s*h + c_s
    M = np.outer(h_vals**2, a_coeffs) + np.outer(h_vals, b_coeffs) + c_coeffs
    f = M.max(axis=1)
    best_idx = np.argmin(f)
    # Refine
    lo = h_vals[max(0, best_idx-1)]
    hi = h_vals[min(len(h_vals)-1, best_idx+1)]

        else:
            lo = h1
```

```python
def compute_grad(h, w, dx):
    """Compute gradient of smoothmax objective.
    h: array of length n
    w: weight vector of length 2n-1 (for lags s=-(n-1)..n-1)
    dx: step size
    """
    n = len(h)
    grad = np.zeros(n)
    for s_idx in range(len(w)):
        ws = w[s_idx]
        if ws < 1e-14:
            continue
        s = s_idx - (n-1)
        if s >= 0:
            # (1-h[j+s]) for j=0..n-1-s
            grad[:n-s] += ws * (1.0 - h[s:])
            # -h[j-s] for j=s..n-1
            grad[s:] -= ws * h[:n-s]
        else:
            t = -s
            # (1-h[j+s]) = (1-h[j-t]) for j=t..n-1
            grad[t:] += ws * (1.0 - h[:n-t])
            # -h[j-s] = -h[j+t] for j=0..n-1-t
            grad[:n-t] -= ws * h[t:]
    grad *= dx
    return grad
```

This is O(n²) but with good numpy vectorization. For n=400, each s does two vector operations of length ~n, so total is about 2*n*(2n-1) ≈ 640K simple operations. At Python overhead of ~10ns per numpy call, that's 798*2 ≈ 1600 numpy calls per gradient, each taking ~1μs for small arrays, so ~2ms per gradient. That's fast enough for 10K+ iterations.

FISTA (Beck-Teboulle 2009):
- x_{k+1} = proj(y_k - (1/L) * grad(f)(y_k))
- t_{k+1} = (1+sqrt(1+4t_k²))/2
- y_{k+1} = x_{k+1} + ((t_k-1)/t_{k+1})(x_{k+1} - x_k)

The Lipschitz constant L for the smoothmax gradient:
The smoothmax is a C^1 function. Its gradient is a convolution-like operation. The Lipschitz constant depends on the weights w and the problem size.

A safe upper bound: L ≤ dx * max_j (Σ_s |w_s| * 2) = 2 * dx * 2 = 4*dx (since Σ|w_s| = 1 and each term is bounded by 1).

Actually, let me think more carefully. The gradient is:
(∇f)[j] = dx * Σ_s w_s * [(1-h[j-s]) - h[j+s]]

The change in gradient when h changes by δ:
|(∇f)(h+δ) - (∇f)(h)|_∞ ≤ dx * Σ_s |w_s| * (|δ[j-s]| + |δ[j+s]|)
≤ dx * (max_j |δ[j]|) * (Σ_s |w_s| + Σ_s |w_s|)
= 2 * dx * ||δ||_∞

So L = 2*dx works as a Lipschitz constant (for the infinity norm).

But for FISTA to work well, we want a tighter bound. Let me just use backtracking.

Actually, for FISTA with backtracking, the standard approach is:
1. Compute grad at y
2. Try step size 1/L
3. If f(y - step*grad) > f(y) - 0.5*L*||step*grad||², increase L and retry

But for efficiency, let me just use a fixed L that's a bit larger than needed.

Let me estimate L more carefully. The Hessian of the smoothmax objective involves second derivatives of the correlation, times the weights. The second derivative of c_s w.r.t. h[j], h[k] is at most 1 (from the -h[j]*h[k] terms). So the Hessian of f is bounded by dx * Σ_s |w_s| * 1 = dx in operator norm (roughly). Actually it's more like 2*dx since each correlation has two bilinear terms.

Let me just use L = 4*dx * n / n = 4*dx for safety, and do backtracking if needed.

Actually, for the smoothmax, the Hessian involves not just the second derivatives of c_s but also the first derivatives (from differentiating the softmax weights). For large beta, the smoothmax approaches the max, and the Hessian of the max of quadratics is bounded. For small beta, it's smoother.

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h = h + diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    # Gradient
    grad = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        # Term 1: (1-h[j-s]) for j in [max(0,s), min(n-1, n-1+s)]
        j1_lo = max(0, s)
        j1_hi = min(n - 1, n - 1 + s)
        if j1_lo <= j1_hi:
            grad[j1_lo:j1_hi+1] += wk * (1.0 - h[j1_lo-s:j1_hi-s+1])
        # Term 2: -h[j+s] for j in [max(0,-s), min(n-1, n-1-s)]
        j2_lo = max(0, -s)
        j2_hi = min(n - 1, n - 1 - s)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi+1] -= wk * h[j2_lo+s:j2_hi+s+1]
    grad *= dx
    return val, grad

def fista_optimize(h_init, beta, n_iter, target, dx, max_time=None, t_start=None):
    n = len(h_init)
    x = h_init.copy()
    y = h_init.copy()
    t_val = 1.0
    L = 4.0 * dx  # initial Lipschitz estimate
    
    best_x = x.copy()
    best_val = compute_c5(x)
    
    for i in range(n_iter):
        if max_time is not None and (time.time() - t_start) > max_time:
            break
        
        val_y, grad_y = smoothmax_val_grad(y, beta, dx)
        
        # Backtracking
        step = 1.0 / L
        for bt in range(20):
            x_new = project_feasible(y - step * grad_y, target)
            val_new, _ = smoothmax_val_grad(x_new, beta, dx)
            if val_new <= val_y - 0.5 * L * step * np.dot(grad_y, grad_y):
                break
            L *= 2
            step = 1.0 / L
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = x_new + ((t_val - 1.0) / t_new) * (x_new - x)
        y = np.clip(y, 0.0, 1.0)  # keep y in bounds for numerical stability
        x = x_new
        t_val = t_new
        
        # Track best
        c = compute_c5(x)
        if c < best_val:
            best_val = c
            best_x = x.copy()
    
    return best_x, best_val
```

Alternative: use the analytical formula for c_s as a function of h[j].

For s=0: c_0(h_j) = (c_0 - h[j] + h[j]²) + h_j - h_j² = c_0 - h[j](1-h[j]) + h_j(1-h_j) ... hmm that's circular.

Let me define:
- Let h_cur = x[j] (current value)
- c_0 = Σ_i x[i](1-x[i]), so the contribution of x[j] to c_0 is x[j](1-x[j])
- c_0(h_j) = (c_0 - x[j](1-x[j])) + h_j(1-h_j) = c_0' + h_j - h_j²

For s≠0:
- c_s = Σ_n x[n+s](1-x[n])
- The contribution of x[j] to c_s:
  - From x[n+s] where n=j-s: x[j]*(1-x[j-s]) if 0≤j-s≤n-1
  - From x[n] where n=j: -x[j+s]*x[j] if 0≤j+s≤n-1
- So c_s(h_j) = (c_s - x[j]*(1-x[j-s])*I(0≤j-s≤n-1) + x[j+s]*x[j]*I(0≤j+s≤n-1)) + h_j*(1-x[j-s])*I(0≤j-s≤n-1) - h_j*x[j+s]*I(0≤j+s≤n-1)
  = c_s' + h_j * b_s

where c_s' = c_s - x[j]*[(1-x[j-s])*I(0≤j-s≤n-1) - x[j+s]*I(0≤j+s≤n-1)]
and b_s = (1-x[j-s])*I(0≤j-s≤n-1) - x[j+s]*I(0≤j+s≤n-1)

So for s=0: c_0(h_j) = c_0' - h_j² + h_j (a=−1, b=1)
For s≠0: c_s(h_j) = c_s' + b_s * h_j (a=0, b=b_s)

This is much better! The dependence is at most quadratic (and only for s=0).

Now, to find h_j that minimizes max_s c_s(h_j):
- c_0(h_j) = c_0' - h_j² + h_j (concave parabola)
- c_s(h_j) = c_s' + b_s * h_j (linear)
- We want to minimize the max of these over h_j ∈ [0,1]

This is the lower envelope problem. The minimum of the max of these functions can be found by:
1. The max function is continuous and piecewise (parabolic/linear)
2. The minimum occurs either at h_j=0, h_j=1, or at a point where two functions are equal and both are greater than all others

For a practical approach, I can use a fine grid (say 100 points) and evaluate:
- At each grid point h_j, compute max_s c_s(h_j)

  - max_{s≠0} (c_s' + b_s * h_j): this is the max of linear functions, which is a convex piecewise-linear function. I can precompute this!

```python
            h_cur = x[j]
            # Compute current correlations
            corr = np.correlate(x, 1.0 - x, mode="full")

            # s = k - (n-1) for k=0..2n-2
            c_prime = np.zeros(n_corr)
            b_coeff = np.zeros(n_corr)

                s = k - (n - 1)
                if s == 0:
                    # c_0(h_j) = c_0' + h_j - h_j^2
                    c_prime[k] = corr[k] - h_cur * (1.0 - h_cur)
                    b_coeff[k] = 1.0  # linear coefficient (quadratic handled separately)
                else:

                    # b_s = (1-x[j-s])*I(0≤j-s≤n-1) - x[j+s]*I(0≤j+s≤n-1)

                    if 0 <= j + s <= n - 1:
                        b -= x[j + s]
```

```python
term1[valid1] = 1.0 - x[j - s_range[valid1]]

term2 = np.zeros(n_corr)
valid2 = (s_range >= -j) & (s_range <= n-1-j)
term2[valid2] = x[j + s_range[valid2]]

b_coeff = term1 - term2
b_coeff[n-1] = 1.0  # special case s=0
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h = h + diff / len(h)
        else:
            h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        j1_lo = max(0, s)
        j1_hi = min(n - 1, n - 1 + s)
        if j1_lo <= j1_hi:
            grad[j1_lo:j1_hi+1] += wk * (1.0 - h[j1_lo-s:j1_hi-s+1])
        j2_lo = max(0, -s)
        j2_hi = min(n - 1, n - 1 - s)
        if j2_lo <= j2_hi:
            grad[j2_lo:j2_hi+1] -= wk * h[j2_lo+s:j2_hi+s+1]
    grad *= dx
    return val, grad

def fista_optimize(h_init, beta, n_iter, target, dx, max_time, t_start):
    n = len(h_init)
    x = h_init.copy()
    y = h_init.copy()
    t_val = 1.0
    L = 8.0 * dx
    best_x = x.copy()
    best_c5 = compute_c5(x)
    for i in range(n_iter):
        if time.time() - t_start > max_time:
            break
        val_y, grad_y = smoothmax_val_grad(y, beta, dx)
        step = 1.0 / L
        for _ in range(15):
            x_new = project_feasible(y - step * grad_y, target)
            val_new, _ = smoothmax_val_grad(x_new, beta, dx)
            if val_new <= val_y - 0.4 * L * step * np.dot(grad_y, grad_y):
                break
            L *= 1.5
            step = 1.0 / L
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = x_new + ((t_val - 1.0) / t_new) * (x_new - x)
        y = np.clip(y, 0.0, 1.0)
        x = x_new
        t_val = t_new
        if i % 200 == 0:
            c = compute_c5(x)
            if c < best_c5:
                best_c5 = c
                best_x = x.copy()
    c = compute_c5(x)
    if c < best_c5:
        best_c5, best_x = c, x.copy()
    return best_x, best_c5

def coord_desc(h_init, target, dx, n_grid=300):
    n = len(h_init)
    n_corr = 2 * n - 1
    h_grid = np.linspace(0.0, 1.0, n_grid)
    s_range = np.arange(n_corr) - (n - 1)
    x = h_init.copy()
    corr = np.correlate(x, 1.0 - x, mode="full")
    best_c5 = float(np.max(corr) * dx)
    path = []
    for j in range(n):
        path.append(j)
    t_cd_start = time.time()
    max_cd_time = 99999
    for p in range(500):
        if time.time() - t_cd_start > max_cd_time:
            break
        for j in path:
            h_cur = x[j]
            b_coeff = np.zeros(n_corr)
            valid1 = (s_range >= j - (n - 1)) & (s_range <= j)
            b_coeff[valid1] += (1.0 - x[j - s_range[valid1]])
            valid2 = (s_range >= -j) & (s_range <= n - 1 - j)
            b_coeff[valid2] -= x[j + s_range[valid2]]
            b_coeff[n - 1] = 1.0
            d_0 = corr[n - 1] - h_cur + h_cur * h_cur
            c_base = corr.copy() - h_cur * b_coeff
            c_base[n - 1] = d_0
            vals = c_base[None, :] + b_coeff[None, :] * h_grid[:, None]
            vals[:, n - 1] -= h_grid ** 2
            max_vals = vals.max(axis=1)
            best_gi = np.argmin(max_vals)
            new_hj = h_grid[best_gi]
            if abs(new_hj - h_cur) > 1e-12:
                x_old = x.copy()
                x[j] = new_hj
                x = project_feasible(x, target)
                corr = np.correlate(x, 1.0 - x, mode="full")
                c = float(np.max(corr) * dx)
                if c < best_c5:
                    best_c5 = c
                else:
                    x = x_old
                    corr = np.correlate(x, 1.0 - x, mode="full")
    return x, best_c5

def slsqp_refine(h_init, maxiter, beta):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0

    def obj_f(x):
        v, _ = smoothmax_val_grad(x, beta, dx)
        return v

    def obj_g(x):
        _, g = smoothmax_val_grad(x, beta, dx)
        return g

    def con_f(x):
        return x.sum() - target

    def con_j(x):
        return np.ones(len(x))

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    try:
        res = minimize(obj_f, h_init, jac=obj_g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_init.copy(), target)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(400, 0.5)

    n = len(h_best)
    target = n / 2.0
    dx = 2.0 / n
    best_c5 = compute_c5(h_best)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: FISTA with increasing beta
    for beta in [30, 80, 200, 500, 1000]:
        if tl() < 100:
            break
        alloc = min(80, tl() - 20)
        if alloc < 15:
            break
        h_new, c_new = fista_optimize(h_best, beta, 50000, target, dx, alloc, time.time())
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new

    # Phase 2: Coordinate descent
    if tl() > 80:
        h_new, c_new = coord_desc(h_best, target, dx, n_grid=300)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new

    # Phase 3: Multi-start with perturbation
    for trial in range(12):
        if tl() < 50:
            break
        scale = 0.008 * (1.0 + 0.4 * trial)
        h_pert = project_feasible(h_best + np.random.randn(n) * scale, target)
        h_new, c_new = coord_desc(h_pert, target, dx, n_grid=200)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new

    # Phase 4: FISTA at higher beta
    for beta in [500, 1500, 3000]:
        if tl() < 50:
            break
        alloc = min(50, tl() - 20)
        if alloc < 15:
            break
        h_new, c_new = fista_optimize(h_best, beta, 30000, target, dx, alloc, time.time())
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new

    # Phase 5: More coordinate descent
    if tl() > 40:
        h_new, c_new = coord_desc(h_best, target, dx, n_grid=400)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new

    # Phase 6: SLSQP polish
    for beta in [800, 2000]:
        if tl() < 30:
            break
        h_new = slsqp_refine(h_best, maxiter=300, beta=beta)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c
        if tl() > 20:
            h_new, c_new = coord_desc(h_best, target, dx, n_grid=300)
            if c_new < best_c5:
                h_best, best_c5 = h_new, c_new

    # Phase 7: Final multi-start at small scale
    for trial in range(6):
        if tl() < 30:
            break
        scale = 0.003 * (1.0 + 0.2 * trial)
        h_pert = project_feasible(h_best + np.random.randn(n) * scale, target)
        h_new, c_new = coord_desc(h_pert, target, dx, n_grid=300)
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new

    # Final FISTA + coord
    if tl() > 40:
        h_new, c_new = fista_optimize(h_best, 2000, 20000, target, dx, min(30, tl()-25), time.time())
        if c_new < best_c5:
            h_best, best_c5 = h_new, c_new
        if tl() > 20:
            h_new, c_new = coord_desc(h_best, target, dx, n_grid=400)
            if c_new < best_c5:
                h_best, best_c5 = h_new, c_new

    h_best = project_feasible(h_best, target)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, n
```
