import numpy as np
from scipy.optimize import minimize
import time

_beta_box = [100.0]
_target_box = [100.0]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(x_shift)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for k in active:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            grad[:n - m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n - m2]
    return val, grad

def _slsqp_obj(x):
    val, _ = smooth_max_and_grad(x, _beta_box[0])
    return val

def _slsqp_grad(x):
    _, g = smooth_max_and_grad(x, _beta_box[0])
    return g

def _slsqp_con(x):
    return x.sum() - _target_box[0]

def _slsqp_con_jac(x):
    return np.ones(len(x))

def do_slsqp(h0, betas, maxiter_each, time_fn):
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    _target_box[0] = target
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    constraints = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_con_jac}]
    for beta in betas:
        if time_fn() < 10:
            break
        _beta_box[0] = beta
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP',
                          bounds=bounds, jac=_slsqp_grad,
                          constraints=constraints,
                          options={'maxiter': maxiter_each, 'ftol': 1e-14})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                h_cur = h_new
                c5_cur = c5_new
        except Exception:
            pass
    return h_cur, c5_cur

def do_subgrad(h, target, n_iter, time_fn, top_k, steps_list, perturb_every, perturb_scale):
    n = len(h)
    no_imp = 0
    for it in range(n_iter):
        if time_fn() < 2:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        nk = min(top_k, max(3, n // 25))
        top_idx = np.argsort(corr)[::-1][:nk]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * h)
            elif 0 < lag < n:
                grad[lag:] += (1.0 - h[:-lag])
                grad[:n - lag] -= h[lag:]
            elif lag < 0 and abs(lag) < n:
                m2 = -lag
                grad[:n - m2] += (1.0 - h[m2:])
                grad[m2:] -= h[:n - m2]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        grad /= gnorm
        found = False
        for step in steps_list:
            ht = project_h(h - step * grad, target)
            c5t = compute_c5(ht)
            if c5t < compute_c5(h) - 1e-15:
                h = ht
                found = True
                no_imp = 0
                break
        if not found:
            no_imp += 1
            if no_imp % perturb_every == perturb_every - 1:
                sc = perturb_scale * min(5, no_imp // perturb_every + 1)
                ht = project_h(h + np.random.randn(n) * sc, target)
                if compute_c5(ht) < compute_c5(h):
                    h = ht
                    no_imp = 0
            if no_imp > n_iter // 10:
                break
    return h, compute_c5(h)
