    v = v.astype(float)
    lo, hi = v.min() - 1, v.max() + 1

def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx].astype(float)
    m = c_sel.max()
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w[i]
        if wk < 1e-14:
            continue

        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n - s])

        else:
            t = -s
            grad[:n - t] += wk * (1.0 - h[t:])
            grad[t:] += wk * (-h[:n - t])
    grad *= dx
    return val, grad

    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

             'fun': lambda x: np.sum(x) - target,

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})

    except Exception:

def structured_perturb(h, rng, type_id, magnitude):
    n = len(h)
    h_new = h.copy()

        # Gaussian noise
        h_new += rng.randn(n) * magnitude
    elif type_id == 1:

        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n) * 2.0 / n

    elif type_id == 2:

        end = start + rng.randint(5, min(30, n - start))

    elif type_id == 3:

        n1 = rng.randint(5, n // 3)
        n2 = n1

        h_new[idx] = h_new[idx][rng.permutation(subset_size)]
    return project_feasible(h_new, n / 2.0)

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
