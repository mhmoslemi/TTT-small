from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_and_grad_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700, 700)
    e = np.exp(shifted)
    s = e.sum()
    fm = m + np.log(s) / beta
    w = e / s
    return fm, w

def full_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    fm, w = smooth_max_and_grad_weights(corr, beta)
    active = np.where(w > 1e-10)[0]
    grad = np.zeros(n)
    center = n - 1
    for i in active:
        wi = w[i]
        lag = i - center
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h) * dx
        elif lag > 0:
            grad[lag:] += wi * (1.0 - h[:n-lag]) * dx
            grad[:n-lag] += wi * (-h[lag:]) * dx
        else:
            l = -lag
            grad[:n-l] += wi * (1.0 - h[l:]) * dx
            grad[l:] += wi * (-h[:n-l]) * dx
    return grad, fm

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, n_old), h)

def gd_optimize(h, target, beta_schedule, iters_per_beta, time_fn, min_time, best_state, use_sa=True):
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    if cur_c5 < best_state[0]:
        best_state[0] = cur_c5
        best_state[1] = cur.copy()
        best_state[2] = n
    
    for beta in beta_schedule:
        for it in range(iters_per_beta):
            if time_fn() < min_time:
                return cur, cur_c5
            grad, _ = full_grad(cur, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            step = 0.05
            improved = False
            for _ in range(30):
                trial = project_h(cur - step * grad, target)
                c = compute_c5(trial)
                if c < cur_c5 - 1e-15:
                    cur = trial
                    cur_c5 = c
                    improved = True
                    if c < best_state[0]:
                        best_state[0] = c
                        best_state[1] = cur.copy()
                        best_state[2] = n
                    break
                step *= 0.5
            if not improved and use_sa:
                T = 0.003 * np.exp(-it / 500.0)
                noise = np.random.normal(0, T, n)
                trial = project_h(cur + noise, target)
                c = compute_c5(trial)
                if c < cur_c5 + T * 2:
                    cur = trial
                    cur_c5 = c
    return cur, cur_c5

def slsqp_polish(h, target, n, time_fn, min_time, best_state):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cur = h.copy()
    
    for beta in [200, 500, 1000, 2000, 5000]:
        if time_fn() < min_time:
            break
        
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode="full") * dx
            return smooth_max_and_grad_weights(c, beta)[0]
        
        def obj_grad(x):
            g, _ = full_grad(x, beta)
            return g
        
        def con_fun(x):
            return np.sum(x) - target
        
        def con_jac(x):
            return np.ones(n)
        
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        try:
            res = minimize(obj, cur, jac=obj_grad, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 1000, 'ftol': 1e-15})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_state[0]:
                best_state[0] = c5_new
                best_state[1] = h_new.copy()
                best_state[2] = n
                cur = h_new
        except Exception:
            pass
    return cur

import numpy as np
from scipy.optimize import minimize, Bounds
from functools import partial
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700, 700)
    e = np.exp(shifted)
    s = e.sum()
    return m + np.log(s) / beta, e / s


def full_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    fm, w = smooth_max_val(corr, beta)
    active = np.where(w > 1e-11)[0]
    grad = np.zeros(n)
    center = n - 1
    for i in active:
        wi = w[i]
        lag = i - center
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h) * dx
        elif lag > 0:
            grad[lag:] += wi * (1.0 - h[:n - lag]) * dx
            grad[:n - lag] += wi * (-h[lag:]) * dx
        else:
            l = -lag
            grad[:n - l] += wi * (1.0 - h[l:]) * dx
            grad[l:] += wi * (-h[:n - l]) * dx
    return grad, fm


def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def resample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, n_old), h)


def obj_fn(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return smooth_max_val(corr, beta)[0]


def obj_fn_grad(h, dx, beta):
    g, _ = full_grad(h, beta)
    return g


def con_sum(x, target):
    return np.sum(x) - target


def con_sum_jac(x, target):
    return np.ones(len(x))


def gd_phase(h, target, beta_list, iters_per_beta, time_fn, min_time, best_state):
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    if cur_c5 < best_state[0]:
        best_state[0] = cur_c5
        best_state[1] = cur.copy()
        best_state[2] = n
    momentum = np.zeros(n)
    mu = 0.7

    for beta in beta_list:
        for it in range(iters_per_beta):
            if time_fn() < min_time:
                return cur, cur_c5
            grad, _ = full_grad(cur, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            momentum = mu * momentum + grad
            mnorm = np.linalg.norm(momentum)
            if mnorm > 0:
                momentum /= mnorm
            step = 0.06
            improved = False
            for _ in range(35):
                trial = project_h(cur - step * momentum, target)
                c = compute_c5(trial)
                if c < cur_c5 - 1e-15:
                    cur = trial
                    cur_c5 = c
                    improved = True
                    if c < best_state[0]:
                        best_state[0] = c
                        best_state[1] = cur.copy()
                        best_state[2] = n
                    break
                step *= 0.5
            if not improved:
                T = 0.004 * np.exp(-it / 400.0)
                noise = np.random.normal(0, T, n)
                trial = project_h(cur + noise, target)
                c = compute_c5(trial)
                if c < cur_c5 + T * 2:
                    cur = trial
                    cur_c5 = c
                    if c < best_state[0]:
                        best_state[0] = c
                        best_state[1] = cur.copy()
                        best_state[2] = n
    return cur, cur_c5


def slsqp_polish(h, target, n, time_fn, min_time, best_state):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cur = h.copy()
    for beta in [300, 600, 1200, 2500, 5000]:
        if time_fn() < min_time:
            break
        obj = partial(obj_fn, dx=dx, beta=beta)
        obj_g = partial(obj_fn_grad, dx=dx, beta=beta)
        con = partial(con_sum, target=target)
        con_j = partial(con_sum_jac, target=target)
        cons = [{'type': 'eq', 'fun': con, 'jac': con_j}]
        try:
            res = minimize(obj, cur, jac=obj_g, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 800, 'ftol': 1e-15})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_state[0]:
                best_state[0] = c5_new
                best_state[1] = h_new.copy()
                best_state[2] = n
                cur = h_new
        except Exception:
            pass
    return cur


def block_perturb(h, target, time_fn, min_time, best_state):
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    blocksizes = [3, 5, 8, 12, 20]
    for it in range(20000):
        if time_fn() < min_time:
            break
        bs = blocksizes[np.random.randint(len(blocksizes))]
        start = np.random.randint(0, max(1, n - bs))
        idx = np.arange(start, min(n, start + bs))
        noise = np.random.uniform(-0.05, 0.05, bs)
        trial = cur.copy()
        trial[idx] += noise
        trial = project_h(trial, target)
        c = compute_c5(trial)
        if c < cur_c5 - 1e-15:
            cur = trial
            cur_c5 = c
            if c < best_state[0]:
                best_state[0] = c
                best_state[1] = cur.copy()
                best_state[2] = n
    return cur


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    else:
        n_parent = 250
        h_parent = np.full(n_parent, 0.5)

    best_state = [compute_c5(h_parent), h_parent.copy(), n_parent]
    n = n_parent
    target = n / 2.0

    # Phase 1: Gradient descent with Nesterov momentum, increasing beta
    beta_schedule = [15, 30, 60, 120, 250, 500, 1000, 2000]
    h1, _ = gd_phase(h_parent, target, beta_schedule, 400, time_left, 60, best_state)

    # Phase 2: SLSQP polish
    if time_left() > 30:
        h2 = slsqp_polish(best_state[1], best_state[2] / 2.0, best_state[2],
                          time_left, 30, best_state)

    # Phase 3: Fine gradient descent at very high beta
    if time_left() > 20:
        n_f = len(best_state[1])
        target_f = n_f / 2.0
        h_f = best_state[1].copy()
        for beta in [3000, 6000, 12000]:
            for it in range(5000):
                if time_left() < 15:
                    break
                grad, _ = full_grad(h_f, beta)
                gnorm = np.linalg.norm(grad)
                if gnorm < 1e-15:
                    break
                grad /= gnorm
                step = 0.004
                for _ in range(25):
                    trial = project_h(h_f - step * grad, target_f)
                    c = compute_c5(trial)
                    if c < best_state[0] - 1e-16:
                        h_f = trial
                        best_state[0] = c
                        best_state[1] = h_f.copy()
                        best_state[2] = n_f
                        break
                    step *= 0.5

    # Phase 4: Block perturbations
    if time_left() > 30:
        block_perturb(best_state[1], best_state[2] / 2.0, time_left, 25, best_state)

    # Phase 5: Final SLSQP at highest beta
    if time_left() > 15:
        n_f2 = len(best_state[1])
        slsqp_polish(best_state[1], n_f2 / 2.0, n_f2, time_left, 5, best_state)

    # Phase 6: Try higher resolution
    if time_left() > 120:
        for n_try in [300, 400]:
            if time_left() < 100:
                break
            h_try = project_h(resample(best_state[1], n_try), n_try / 2.0)
            c_try = compute_c5(h_try)
            if c_try < best_state[0]:
                best_state[0] = c_try
                best_state[1] = h_try.copy()
                best_state[2] = n_try
            beta_sched = [50, 100, 200, 500, 1000, 2000]
            h_try, _ = gd_phase(h_try, n_try / 2.0, beta_sched, 300, time_left, 50, best_state)
            if time_left() > 20:
                slsqp_polish(best_state[1], best_state[2] / 2.0, best_state[2],
                             time_left, 15, best_state)

    # Final verification
    h_final = project_h(best_state[1], len(best_state[1]) / 2.0)
    c5_final = compute_c5(h_final)
    if c5_final < best_state[0]:
        best_state[0] = c5_final
        best_state[1] = h_final

    return best_state[1], best_state[0], len(best_state[1])
