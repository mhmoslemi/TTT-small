    n = len(h)
    dx = 2.0 / n
    x = h.copy().astype(float)

        top_idx = np.argpartition(corr, -top_k)[-top_k:]

        nz_mask = lags != 0
        idx_nz = top_idx[nz_mask]
        lags_nz = lags[nz_mask]
        k = len(idx_nz)
        
        if k == 0:
            break

        # Shape: (n, k)
        slopes = np.zeros((n, k))
        intercepts = np.zeros((n, k))
        base_0 = corr[center]  # we'll subtract old*(1-old) per coordinate
        
        for j in range(k):
            s = lags_nz[j]
            c_val = corr[idx_nz[j]]
            if s > 0:
                # slope[i] = (1-x[i+s])*(i+s<n) - x[i-s]*(i-s>=0)
                # contrib[i] = x[i]*(1-x[i+s])*(i+s<n) + x[i-s]*(1-x[i])*(i-s>=0)
                vp = np.arange(n) + s < n
                vm = np.arange(n) - s >= 0
                slopes[vp, j] += (1.0 - x[np.arange(n)[vp] + s])
                slopes[vm, j] -= x[np.arange(n)[vm] - s]
                contrib = np.zeros(n)
                contrib[vp] = x[vp] * (1.0 - x[np.arange(n)[vp] + s])
                contrib[vm] += x[np.arange(n)[vm] - s] * (1.0 - x[vm])
                intercepts[:, j] = c_val - contrib
            else:
                t = -s
                vp = np.arange(n) + t < n
                vm = np.arange(n) - t >= 0
                slopes[vp, j] += (1.0 - x[np.arange(n)[vp] + t])
                slopes[vm, j] -= x[np.arange(n)[vm] - t]
                contrib = np.zeros(n)
                contrib[vp] = x[vp] * (1.0 - x[np.arange(n)[vp] + t])
                contrib[vm] += x[np.arange(n)[vm] - t] * (1.0 - x[vm])
                intercepts[:, j] = c_val - contrib
        
        # For each coordinate i, solve 1D problem
        # F_i(u) = max(base_0_i + u - u^2, max_j (intercepts[i,j] + slopes[i,j]*u))
        # where base_0_i = corr[center] - x[i]*(1-x[i])
        
        base_0_all = corr[center] - x * (1.0 - x)  # shape (n,)

        u_col = u_grid[:, np.newaxis]  # (grid_n, 1)
        g0 = base_0_all[np.newaxis, :] + u_col - u_col**2  # (grid_n, n)
        glines = intercepts.T + slopes.T * u_col  # (grid_n, n, k) -> (n, k, grid_n)
        # Actually: intercepts.T is (k, n), slopes.T is (k, n)
        # intercepts.T + slopes.T * u_col: (k, n) + (k, n) * (grid_n, 1) -> broadcasting issue
        # Let me reshape: u_col is (grid_n, 1), slopes.T is (k, n)
        # I want result[i, j, m] = intercepts[i, j] + slopes[i, j] * u_grid[m]
        # = intercepts[i, j] + slopes[i, j] * u_grid[m]

        max_lines = np.max(glines, axis=1)  # (n, grid_n)

        new_x = u_grid[best_idx]  # (n,)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_smoothmax(h, maxiter, beta):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        return (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        exps = np.exp(beta * shifted)
        w = exps / exps.sum()
        g = np.zeros(n)
        for kk in range(len(corr)):
            wk = w[kk]
            if wk < 1e-12:
                continue
            s = kk - (n - 1)
            if s >= 0:
                ms = np.arange(s, n)
                g[ms] += wk * (1.0 - x[ms - s])
                mp = np.arange(0, n - s)
                g[mp] += wk * (-x[mp + s])
            elif s < 0:
                t = -s
                ms = np.arange(0, n - t)
                g[ms] += wk * (1.0 - x[ms + t])
                mp = np.arange(t, n)
                g[mp] += wk * (-x[mp - t])
            else:
                g[:] += wk * (1.0 - 2.0 * x)
        return g * dx
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except:
        return project_feasible(h.copy(), target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)
