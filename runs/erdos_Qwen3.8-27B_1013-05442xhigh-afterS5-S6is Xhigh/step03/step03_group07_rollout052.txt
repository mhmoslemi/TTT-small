import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-10:
            free = (h > 1e-9) & (h < 1.0 - 1e-9)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            ms = np.arange(s, n)
            grad[ms] += wk * (1.0 - h[ms - s])
            mp = np.arange(0, n - s)
            grad[mp] += wk * (-h[mp + s])
        elif s < 0:
            t = -s
            ms = np.arange(0, n - t)
            grad[ms] += wk * (1.0 - h[ms + t])
            mp = np.arange(t, n)
            grad[mp] += wk * (-h[mp - t])
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    grad *= dx
    return val, grad

def topk_grad(h, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for k in top_idx:
        s = k - (n - 1)
        if s >= 0:
            ms = np.arange(s, n)
            grad[ms] += (1.0 - h[ms - s])
            mp = np.arange(0, n - s)
            grad[mp] += (-h[mp + s])
        elif s < 0:
            t = -s
            ms = np.arange(0, n - t)
            grad[ms] += (1.0 - h[ms + t])
            mp = np.arange(t, n)
            grad[mp] += (-h[mp - t])
        else:
            grad[:] += (1.0 - 2.0 * h)
    grad *= dx / top_k
    return grad, float(np.max(corr) * dx)

def slsqp_optimize(h, beta, maxiter=500):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    def grad_fn(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad_fn, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def subgrad_backtrack(h, max_iters, top_k, lr0=0.01):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        lr = lr0 * (0.999 ** t)
        grad, cur_v = topk_grad(x, top_k)
        improved = False
        for _ in range(8):
            x_new = project_feasible(x - lr * grad, target)
            v_new = compute_c5(x_new)
            if v_new < cur_v - 1e-14:
                x = x_new
                improved = True
                break
            lr *= 0.4
        if not improved:
            x_new = project_feasible(x - lr * grad, target)
            x = x_new
            v_new = compute_c5(x)
        if v_new < best_v:
            best_v = v_new
            best_x = x.copy()
    return best_x, best_v

def equalize_top_lags(h, top_k, max_iters=500, lr0=0.005):
    """Reduce variance among top-k lags while keeping max low"""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        corr = compute_corr(x)
        top_idx = np.argsort(corr)[-top_k:]
        top_vals = corr[top_idx]
        mean_top = np.mean(top_vals)
        # Penalty gradient: push top lags toward their mean (equalize)
        grad = np.zeros(n)
        for k in top_idx:
            s = k - (n - 1)
            w_k = 1.0 / top_k
            if s >= 0:
                ms = np.arange(s, n)
                grad[ms] += w_k * (1.0 - x[ms - s])
                mp = np.arange(0, n - s)
                grad[mp] += w_k * (-x[mp + s])
            elif s < 0:
                tt = -s
                ms = np.arange(0, n - tt)
                grad[ms] += w_k * (1.0 - x[ms + tt])
                mp = np.arange(tt, n)
                grad[mp] += w_k * (-x[mp - tt])
            else:
                grad[:] += w_k * (1.0 - 2.0 * x)
        grad *= dx
        # Also add gradient to reduce the max
        max_k = np.argmax(corr)
        s = max_k - (n - 1)
        extra = np.zeros(n)
        if s >= 0:
            ms = np.arange(s, n)
            extra[ms] += (1.0 - x[ms - s])
            mp = np.arange(0, n - s)
            extra[mp] += (-x[mp + s])
        elif s < 0:
            tt = -s
            ms = np.arange(0, n - tt)
            extra[ms] += (1.0 - x[ms + tt])
            mp = np.arange(tt, n)
            extra[mp] += (-x[mp - tt])
        else:
            extra[:] += (1.0 - 2.0 * x)
        extra *= dx
        total_grad = 0.7 * grad + 0.3 * extra
        lr = lr0 * (0.998 ** t)
        x_new = project_feasible(x - lr * total_grad, target)
        v_new = compute_c5(x_new)
        if v_new < best_v:
            best_v = v_new
            best_x = x_new.copy()
        x = x_new
    return best_x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)
