def binary_local_search(h, max_iter=1000):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full").copy()
    current_max = corr.max()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for iteration in range(max_iter):
        improved = False
        for a in ones:
            for b in zeros:
                if a == b:
                    continue
                # Compute change in corr for this 2-swap
                # h[a]: 1->0, h[b]: 0->1
                # Effect on corr(d) for each d:
                # From a (1->0): for lag d, if a+d in [0,n): -(1-h[a+d]); if a-d in [0,n): +h[a-d]
                # From b (0->1): for lag d, if b+d in [0,n): +(1-h[b+d]); if b-d in [0,n): -h[b-d]
                
                # Instead of computing all d, let's just compute the new max
                # by updating corr incrementally
                old_corr = corr.copy()
                
                # Effect of h[a] 1->0
                for d in range(-n+1, n):
                    ap = a + d
                    am = a - d
                    delta = 0
                    if 0 <= ap < n:
                        delta -= (1 - h[ap])  # h[a] was 1, now 0, so h[a]*(1-h[ap]) goes from (1-h[ap]) to 0
                    if 0 <= am < n:
                        delta += h[am]  # h[am]*(1-h[a]) goes from 0 to h[am]
                    if delta != 0:
                        corr[int(n-1+d)] += delta
                
                # Effect of h[b] 0->1
                for d in range(-n+1, n):
                    bp = b + d
                    bm = b - d
                    delta = 0
                    if 0 <= bp < n:
                        delta += (1 - h[bp])  # h[b] was 0, now 1
                    if 0 <= bm < n:
                        delta -= h[bm]  # h[bm]*(1-h[b]) goes from h[bm] to 0
                    if delta != 0:
                        corr[int(n-1+d)] += delta
                
                new_max = corr.max()
                if new_max < current_max - 1e-12:
                    h[a], h[b] = 0, 1
                    current_max = new_max
                    improved = True
                    # Update ones and zeros sets
                    break
                else:
                    corr = old_corr
            if improved:
                break
        if not improved:
            break
    
    return h, current_max * dx

def compute_delta_flip_off(a, h, n):
    """Delta in corr when h[a] changes from 1 to 0"""
    delta = np.zeros(2*n - 1)
    # Positive lags d >= 0: index n-1+d
    # M[a, a+d] term: d in [0, n-1-a]
    d_pos = np.arange(0, n - a)
    delta[n-1 + d_pos] -= (1.0 - h[a + d_pos])
    # M[a-d, a] term: d in [0, a]
    d_pos2 = np.arange(0, a + 1)
    delta[n-1 + d_pos2] += h[a - d_pos2]
    # Negative lags d < 0: index n-1+d = n-1-|d|
    # For lag -k (k>0): corr(-k) = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    xc = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xc)) / len(x)) / beta

def objective(x, beta, dx):
    c = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max(c, beta) * dx

def slsqp_optimize(h, target, beta_list, maxiter=2000):
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    constr = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
               'jac': lambda x: np.ones(n)}]
    best = h.copy()
    best_c5 = compute_c5(best)
    
    for beta in beta_list:
        def obj(x):
            return objective(x, beta, dx)
        
        try:
            res = minimize(obj, best, method='SLSQP', bounds=bounds,
                          constraints=constr,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    
    return best, best_c5

def binary_round(h, target):
    n = len(h)
    best_h, best_c5 = h.copy(), compute_c5(h)
    for t in np.linspace(0.2, 0.8, 30):
        hb = (h >= t).astype(float)
        deficit = target - hb.sum()
        if abs(deficit) < 0.5:
            # Fine-tune
            if deficit > 0:
                order = np.argsort(-h)
                for i in order:
                    if hb[i] == 0:
                        hb[i] = 1.0
                        deficit -= 1
                        if deficit <= 0:
                            break
            else:
                order = np.argsort(h)
                for i in order:
                    if hb[i] == 1:
                        hb[i] = 0.0
                        deficit += 1
                        if deficit >= 0:
                            break
            if abs(hb.sum() - target) < 1e-6:
                c5 = compute_c5(hb)
                if c5 < best_c5:
                    best_h, best_c5 = hb.copy(), c5
    return best_h, best_c5

def binary_2swap_limited(h, top_k=5, max_rounds=500, time_limit=200):
    """2-swap local search focusing on top lags."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    current_max = float(corr.max())
    t0 = time.time()
    
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        # Find top lags
        top_idx = np.argsort(corr)[::-1][:top_k]
        
        # For each top lag, find positions that contribute
        # corr at index i corresponds to some lag
        # Try swaps that reduce top lags
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        # Build candidate set from top lags
        # For lag at index i, the contributing pairs are (n, n+L) where L = i-(n-1)
        candidates_a = set()  # positions with h=1 that contribute to top lags
        candidates_b = set()  # positions with h=0 that could help
        
        for idx in top_idx:
            L = idx - (n - 1)
            if L >= 0:
                for i in range(n - L):
                    if h[i] == 1.0 and h[i + L] == 0.0:
                        candidates_a.add(i)
                        candidates_b.add(i + L)
            else:
                k = -L
                for i in range(n - k):
                    if h[i + k] == 1.0 and h[i] == 0.0:
                        candidates_a.add(i + k)
                        candidates_b.add(i)
        
        cand_a = list(candidates_a & set(ones))
        cand_b = list(candidates_b & set(zeros))
        
        if not cand_a or not cand_b:
            # Fall back to all
            cand_a = ones.tolist()
            cand_b = zeros.tolist()
        
        best_gain = 0
        best_pair = None
        
        for a in cand_a:
            for b in cand_b:
                if a == b:
                    continue
                h[a], h[b] = 0.0, 1.0
                new_corr = np.correlate(h, 1.0 - h, mode="full")
                new_max = float(new_corr.max())
                gain = current_max - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_pair = (a, b, new_corr.copy())
                h[a], h[b] = 1.0, 0.0
        
        if best_pair is not None and best_gain > 1e-12:
            a, b, new_corr = best_pair
            h[a], h[b] = 0.0, 1.0
            corr = new_corr
            current_max -= best_gain
        else:
            break
    
    return h, current_max * dx

    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    xc = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xc)) / len(x)) / beta

def _obj(x, beta, dx):
    c = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max(c, beta) * dx

def _con_sum(x, target):
    return x.sum() - target

def _con_jac(x):
    return np.ones(x.shape)

def slsqp_refine(h, target, betas, maxiter_per_beta=2000):
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    best = h.copy()
    best_c5 = compute_c5(best)
    
    for beta in betas:
        def obj(x):
            return _obj(x, beta, dx)
        def con(x):
            return _con_sum(x, target)
        try:
            res = minimize(obj, best, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con, 'jac': _con_jac}],
                          options={'maxiter': maxiter_per_beta, 'ftol': 1e-14})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    
    return best, best_c5

def binary_round(h, target, n_thresholds=30):
    n = len(h)
    best_h, best_c5 = h.copy(), compute_c5(h)
    for t in np.linspace(0.15, 0.85, n_thresholds):
        hb = (h >= t).astype(float)
        deficit = target - hb.sum()
        if deficit > 0.5:
            zeros_idx = np.where(hb == 0)[0]
            order = zeros_idx[np.argsort(h[zeros_idx])[::-1]]
            hb[order[:int(np.ceil(deficit))]] = 1.0
        elif deficit < -0.5:
            ones_idx = np.where(hb == 1)[0]
            order = ones_idx[np.argsort(h[ones_idx])]
            hb[order[:int(np.ceil(-deficit))]] = 0.0
        if abs(hb.sum() - target) < 1e-6:
            c5 = compute_c5(hb)
            if c5 < best_c5:
                best_h, best_c5 = hb.copy(), c5
    return best_h, best_c5
