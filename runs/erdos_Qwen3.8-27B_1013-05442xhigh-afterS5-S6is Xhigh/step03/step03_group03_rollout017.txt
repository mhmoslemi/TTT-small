This is linear in h. So the problem becomes:
min ∑_k w_k * [f_k(h*) + dx * ∇f_k(h*) · (h - h*)]

s.t. 0 ≤ h ≤ 1, sum(h) = n/2

This is a LP! And we can solve it very fast.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_lag(h, lag, n):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:-m]
    return grad

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_val(corr, beta) * dx

def con_fun(x, target_sum):
    return x.sum() - target_sum

def con_jac(x, target_sum):
    return np.ones_like(x)

def slsqp_solve(h, n, target_sum, beta, maxiter, ftol):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
    res = minimize(obj_smooth, h, args=(dx, beta), method='SLSQP',
                    bounds=bounds, constraints=cons,
                    options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(res.x, target_sum)

def sg_step(h, n, topk, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_k = min(topk, len(corr))
    sorted_idx = np.argsort(corr)[-top_k:]
    w = np.exp(temp * (corr[sorted_idx] - corr[sorted_idx].max()))
    w /= w.sum()
    grad = np.zeros(n)
    for j, k in enumerate(sorted_idx):
        lag = k - (n - 1)
        grad += w[j] * subgrad_for_lag(h, lag, n)
    return grad * dx

def coord_polish(h, target_sum, n, max_rounds, t0, budget, reserve):
    best = h.copy()
    best_c5 = compute_c5(h)
    deltas = [0.005, -0.005, 0.01, -0.01, 0.02, -0.02, 0.003, -0.003, 0.002, -0.002]
    for rnd in range(max_rounds):
        if time.time() - t0 > budget - reserve:
            break
        improved = False
        for i in range(n):
            if time.time() - t0 > budget - reserve:
                break
            orig = best[i]
            for d in deltas:
                trial_val = orig + d
                if trial_val < -0.01 or trial_val > 1.01:
                    continue
                best[i] = trial_val
                best = project_feasible(best, target_sum)
                c5_try = compute_c5(best)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    improved = True
                else:
                    best[i] = orig
                    best = project_feasible(best, target_sum)
        if not improved:
            break
    return best, best_c5

                except Exception:
                    pass

    # Phase 6: Final subgradient micro-polish
    if time.time() - t0 < budget_s - 5:

        lr = 0.1

                    break

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
