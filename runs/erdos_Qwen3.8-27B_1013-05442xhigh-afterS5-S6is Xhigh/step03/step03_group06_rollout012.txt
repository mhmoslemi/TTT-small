import numpy as np

import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    e = np.exp(x_shift)
    return m + np.log(np.sum(e)) / beta


def smooth_max_weights(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    e = np.exp(x_shift)
    w = e / np.sum(e)
    return w

            h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        nv = n - lag
        if nv > 0:
            grad[:nv] += (1.0 - h[lag:])

    else:
        m = -lag
        nv = n - m
        if nv > 0:
            grad[m:] += (1.0 - h[:nv])

    return grad

        return g

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    e = np.exp(x_shift)
    return m + np.log(np.sum(e)) / beta


def smooth_max_weights(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    e = np.exp(x_shift)
    w = e / np.sum(e)
    return w


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def corr_grad_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        nv = n - lag
        if nv > 0:
            grad[:nv] += (1.0 - h[lag:])
            grad[lag:] -= h[:nv]
    else:
        m = -lag
        nv = n - m
        if nv > 0:
            grad[m:] += (1.0 - h[:nv])
            grad[:nv] -= h[m:]
    return grad


def obj_and_grad(h, beta, top_k=10):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    w = smooth_max_weights(scaled, beta)
    top_idx = np.argsort(scaled)[::-1][:top_k]
    w_top = w[top_idx].copy()
    s = w_top.sum()
    if s > 0:
        w_top = w_top / s
    obj = smooth_max(scaled, beta)
    grad = np.zeros(n)
    for i, idx in enumerate(top_idx):
        lag = int(idx) - (n - 1)
        grad += w_top[i] * dx * corr_grad_lag(h, lag)
    return obj, grad


def slsqp_optimize(h0, n, beta, maxiter=5000):
    dx = 2.0 / n
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(c * dx, beta)

    def jac_fn(x):
        _, g = obj_and_grad(x, beta, top_k=10)
        return g

    con = {'type': 'eq',
           'fun': lambda x: np.sum(x) - target,
           'jac': lambda x: np.ones(n)}
    try:
        res = minimize(obj_fn, h0, jac=jac_fn, method='SLSQP',
                       bounds=bounds, constraints=[con],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(res.x, target)
    except Exception:
        return h0


def coord_descent_sys(h, n, target, top_lags, steps, max_rounds=50):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for lag in top_lags:
            a = abs(lag)
            if a >= n:
                continue
            nv = n - a
            for i in range(nv):
                j = i + a if lag > 0 else i - lag
                if j < 0 or j >= n:
                    continue
                for d in steps:
                    h_t = best_h.copy()
                    h_t[i] += d
                    h_t[j] -= d
                    if 0 <= h_t[i] <= 1 and 0 <= h_t[j] <= 1:
                        c_t = compute_c5(h_t)
                        if c_t < best_c5 - 1e-17:
                            best_h = h_t
                            best_c5 = c_t
                            improved = True
                            break
        if not improved:
            break
    return best_h, best_c5
