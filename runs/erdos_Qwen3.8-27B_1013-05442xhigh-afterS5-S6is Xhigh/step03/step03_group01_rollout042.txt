```python
    f = (np.sum(c**p))**(1.0/p)
    w = c**(p-1) / (f**(p-1))  # length 2n-1
    # w[l_idx] corresponds to lag l = l_idx - (n-1)
    
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for k in range(2*n-1):
            l = k - (n-1)  # lag
            # term 1: (1 - h[j+l]) if j+l in [0, n-1]
            i1 = j + l
            if 0 <= i1 < n:
                s += w[k] * (1 - h[i1])
            # term 2: -h[j-l] if j-l in [0, n-1]
            i2 = j - l
            if 0 <= i2 < n:
                s -= w[k] * h[i2]
        grad[j] = dx * s
    return grad
```

```python
            if total < 1e-14:

def smooth_max_grad(h, n, p, mu):

    c = np.correlate(h, 1 - h, 'full') * dx
    m = c.max()
    shifted = p * (c - m)

    w = np.exp(shifted)

        s = 0.0

            l = k - (n - 1)
            i1 = j + l
            if 0 <= i1 < n:
                s += w[k] * (1.0 - h[i1])
            i2 = j - l
            if 0 <= i2 < n:
                s -= w[k] * h[i2]
        grad[j] = dx * s
    
    # Gradient of penalty term mu * (sum(h) - n/2)^2
    sum_dev = h.sum() - n / 2.0
    grad += 2.0 * mu * sum_dev * np.ones(n)
    
    return grad

    shifted -= shifted.max()

    val += mu * (h.sum() - n / 2.0) ** 2
    return val
```

```python
            total = h.sum()
            if total < 1e-14:

    c = np.correlate(h, 1 - h, 'full') * dx
    m = c.max()
    shifted = p * (c - m)

    w = np.exp(shifted)

    w /= w_sum
    obj = m + np.log(w_sum) / p
    obj += mu * (h.sum() - n / 2.0) ** 2

    grad = np.zeros(n)
    two_n_1 = 2 * n - 1
    for j in range(n):

        for k in range(two_n_1):

            if 0 <= i2 < n:
                s -= w[k] * h[i2]

    sum_dev = h.sum() - n / 2.0
```

This is the smooth max of linear functions, which is convex in t! So I can minimize it using a 1D method (bisection, ternary search, or scipy's minimize_scalar).

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, n, p, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, 'full') * dx
    m = c.max()
    sh = p * (c - m)
    sh -= sh.max()
    val = m + np.log(np.sum(np.exp(sh))) / p
    val += mu * (x.sum() - n / 2.0) ** 2
    return val

def smooth_max_grad(x, n, p, mu):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, 'full') * dx
    m = c.max()
    sh = p * (c - m)
    sh -= sh.max()
    w = np.exp(sh)
    w /= w.sum()
    
    # Vectorized gradient
    lags = np.arange(2 * n - 1) - (n - 1)
    j_idx = np.arange(n)[:, None]
    l_idx = lags[None, :]
    i1 = j_idx + l_idx
    i2 = j_idx - l_idx
    mask1 = (i1 >= 0) & (i1 < n)
    mask2 = (i2 >= 0) & (i2 < n)
    h_clip1 = np.clip(i1, 0, n - 1)
    h_clip2 = np.clip(i2, 0, n - 1)
    term1 = np.where(mask1, x[h_clip1], 0.0) @ w
    term2 = np.where(mask2, x[h_clip2], 0.0) @ w
    grad = dx * (1.0 - term1 - term2) + 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def phase_lbfgsb(h, n, deadline, p_list=None):
    if p_list is None:

def phase_active_lag(h, n, deadline, max_iter=3000):
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    rng = np.random.RandomState(789)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, 'full') * dx
        max_val = c.max()
        tol = max(0.0005, 0.003 * max_val)
        active = np.where(c >= max_val - tol)[0]
        
        d = np.zeros(n)
        lags = np.arange(2 * n - 1) - (n - 1)
        active_lags = lags[active]
        j_idx = np.arange(n)[:, None]
        l_idx = active_lags[None, :]
        i1 = j_idx + l_idx
        i2 = j_idx - l_idx
        mask1 = (i1 >= 0) & (i1 < n)
        mask2 = (i2 >= 0) & (i2 < n)
        h_c1 = np.clip(i1, 0, n - 1)
        h_c2 = np.clip(i2, 0, n - 1)
        term1 = np.where(mask1, 1.0 - h[h_c1], 0.0)
        term2 = np.where(mask2, h[h_c2], 0.0)
        d = (term1 - term2).sum(axis=1) * dx / len(active)
        d -= d.mean()
        
        step = 0.1
        improved = False
        for _ in range(25):
            h_new = h + step * d
            if h_new.min() >= -0.05 and h_new.max() <= 1.05:
                h_new = project_feasible(h_new, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-15:
                    h = h_new.copy()
                    best_c = c_new
                    improved = True
                    break
            step *= 0.5
        if not improved and rng.random() < 0.05:
            h_p = h + rng.normal(0, 0.003, n)
            h_p = project_feasible(h_p, n)
            if compute_c5(h_p, n) < best_c:
                h = h_p
                best_c = compute_c5(h, n)
    return h, best_c
