    return best_x, best_val

def lbfgs_penalty(h, max_iters, penalty, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj(x):
        v, g = smoothmax_val_grad(x, beta)
        pen = penalty * (x.sum() - target) ** 2
        return v + pen

    def grad(x):
        _, g = smoothmax_val_grad(x, beta)
        g = g + 2.0 * penalty * (x.sum() - target) * np.ones(n)
        return g

    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iters, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v

    def grad(x):
        _, g = smoothmax_val_grad(x, beta)
        return g

    def con_f(x):
        return x.sum() - target

    def con_j(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def nesterov_proj(h, max_iters, lr0, decay, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_val = compute_c5(x)
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2.0)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = compute_corr(y)
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            if s >= 0:
                ms = np.arange(s, n)
                grad[ms] += (1.0 - y[ms - s])
                mp = np.arange(0, n - s)
                grad[mp] += (-y[mp + s])
            elif s < 0:
                tt = -s
                ms = np.arange(0, n - tt)
                grad[ms] += (1.0 - y[ms + tt])
                mp = np.arange(tt, n)
                grad[mp] += (-y[mp - tt])
            else:
                grad[:] += (1.0 - 2.0 * y)
        grad = grad / top_k * dx
        x_new = project_feasible(y - lr_t * grad, target)
        val = compute_c5(x_new)
        if val < best_val:
            best_val = val
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_val

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)

    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: Quick SLSQP warm-start
    for beta in [100, 300]:
        if tl() < 40:
            break
        h_new = slsqp_refine(h_best, maxiter=200, beta=beta)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 2: Adam with adaptive smoothmax (main workhorse)
    if tl() > 120:
        h_a, c_a = adam_smoothmax(h_best, 15000, lr=0.05, b1=0.9, b2=0.999,
                                  eps=1e-8, beta0=200, beta_mult=1.5, beta_per=2000)
        if c_a < best_c5:
            h_best, best_c5 = h_a, c_a

    # Phase 3: L-BFGS-B with penalty
    if tl() > 80:
        h_lb = lbfgs_penalty(h_best, max_iters=300, penalty=50.0, beta=400)
        c = compute_c5(h_lb)
        if c < best_c5:
            h_best, best_c5 = h_lb, c
        if tl() > 60:
            h_lb2 = lbfgs_penalty(h_best, max_iters=300, penalty=200.0, beta=800)
            c = compute_c5(h_lb2)
            if c < best_c5:
                h_best, best_c5 = h_lb2, c

    # Phase 4: Nesterov top-k
    if tl() > 100:
        h_ng, c_ng = nesterov_proj(h_best, 8000, lr0=0.06, decay=0.9999, top_k=5)
        if c_ng < best_c5:
            h_best, best_c5 = h_ng, c_ng
        if tl() > 60:
            h_ng2, c_ng2 = nesterov_proj(h_best, 5000, lr0=0.03, decay=0.99995, top_k=10)
            if c_ng2 < best_c5:
                h_best, best_c5 = h_ng2, c_ng2

    # Phase 5: Multi-start perturbation
    for trial in range(12):
        if tl() < 50:
            break
        scale = 0.015 * (1.0 + 0.3 * trial)
        pert = np.random.randn(len(h_best)) * scale
        h_pert = project_feasible(h_best + pert, len(h_best) / 2.0)
        h_new, c_try = adam_smoothmax(h_pert, 3000, lr=0.04, b1=0.9, b2=0.999,
                                      eps=1e-8, beta0=300, beta_mult=1.4, beta_per=1500)
        if c_try < best_c5:
            h_best, best_c5 = h_new, c_try
            h_pert = h_best.copy()

    # Phase 6: Resolution upgrade
    for n_try in [600, 800]:
        if tl() < 200:
            break
        h_interp = interp_to_n(h_best, n_try)
        h_new, c_try = adam_smoothmax(h_interp, 8000, lr=0.03, b1=0.9, b2=0.999,
                                      eps=1e-8, beta0=200, beta_mult=1.5, beta_per=2000)
        if c_try < best_c5:
            h_best = h_new
            best_c5 = c_try
        if tl() > 80:
            h_lb = lbfgs_penalty(h_best, max_iters=200, penalty=100.0, beta=600)
            c = compute_c5(h_lb)
            if c < best_c5:
                h_best, best_c5 = h_lb, c
        if tl() > 60:
            h_ng, c_ng = nesterov_proj(h_best, 4000, lr0=0.04, decay=0.9999, top_k=5)
            if c_ng < best_c5:
                h_best, best_c5 = h_ng, c_ng

    # Phase 7: Final Adam polish at current resolution
    if tl() > 60:
        h_final, c_final = adam_smoothmax(h_best, 10000, lr=0.02, b1=0.9, b2=0.999,
                                          eps=1e-8, beta0=500, beta_mult=1.3, beta_per=2000)
        if c_final < best_c5:
            h_best, best_c5 = h_final, c_final

    # Phase 8: Final L-BFGS polish
    if tl() > 40:
        h_fp = lbfgs_penalty(h_best, max_iters=200, penalty=500.0, beta=1500)
        c = compute_c5(h_fp)
        if c < best_c5:
            h_best, best_c5 = h_fp, c

    # Final feasibility
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
