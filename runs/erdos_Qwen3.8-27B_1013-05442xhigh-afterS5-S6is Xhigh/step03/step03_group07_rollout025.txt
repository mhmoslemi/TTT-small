def frank_wolfe(h, max_iters, beta_smooth):

    x = h.copy().astype(float)
    best_x = x.copy()

        corr = compute_corr(x)
        m = np.max(corr)
        shifted = corr - m
        exps = np.exp(beta_smooth * shifted)
        w = exps / exps.sum()
        
        # Gradient
        grad = np.zeros(n)
        for k in range(len(corr)):
            if w[k] < 1e-10:
                continue
            s = k - (n - 1)
            if s >= 0:
                grad[s:] += w[k] * (1.0 - x[:n-s])
                grad[:n-s] += w[k] * (-x[s:])
            elif s < 0:
                t_ = -s
                grad[:n-t_] += w[k] * (1.0 - x[t_:])
                grad[t_:] += w[k] * (-x[:n-t_])
            else:
                grad[:] += w[k] * (1.0 - 2.0 * x)
        grad *= dx
        
        # Linear minimization oracle: minimize grad^T y s.t. sum(y) = target, 0 <= y <= 1
        # Sort grad, set y=1 where grad is smallest, y=0 where largest
        sorted_idx = np.argsort(grad)
        y = np.zeros(n)
        y[sorted_idx[:int(target)]] = 1.0
        # Handle the fractional part
        frac = target - int(target)
        if frac > 1e-10 and int(target) < n:
            y[sorted_idx[int(target)]] = frac
        
        # Actually target = n/2 which is an integer, so:
        # y = 1 for the bottom n/2 smallest grad values, 0 otherwise
        
        # Step size: use standard FW step size gamma = 2/(t+2)
        gamma = 2.0 / (t + 2)
        x_new = (1 - gamma) * x + gamma * y
        x_new = project_feasible(x_new, target)
        
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x = x_new
    
    return best_x, best_v

from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n-s])
            grad[:n-s] += wk * (-h[s:])
        elif s < 0:
            t_ = -s
            grad[:n-t_] += wk * (1.0 - h[t_:])
            grad[t_:] += wk * (-h[:n-t_])
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    grad *= dx
    return val, grad

def frank_wolfe(h, max_iters, beta):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    best_x = x.copy()

        grad = np.zeros(n)

            wk = w[k]
            if wk < 1e-10:
                continue
            s = k - (n - 1)
            if s >= 0:
                grad[s:] += wk * (1.0 - x[:n-s])

            elif s < 0:
                t_ = -s
                grad[:n-t_] += wk * (1.0 - x[t_:])
                grad[t_:] += wk * (-x[:n-t_])
            else:
                grad[:] += wk * (1.0 - 2.0 * x)

        y = np.zeros(n)
        y[sorted_idx[:int(target)]] = 1.0
        frac = target - int(target)
        if abs(frac) > 1e-10:
            y[sorted_idx[int(target)]] = frac
        # FW step

        x_new = np.clip(x_new, 0.0, 1.0)

        if abs(diff) > 1e-10:

        v = compute_c5(x_new)

            best_x = x_new.copy()
        x = x_new

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj_func(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    
    def obj_grad(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    
    def con_func(x):
        return x.sum() - target
    
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_func, 'jac': con_jac}]
    try:
        res = minimize(obj_func, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def binary_init(n, seed):
    """Create a random binary-like initialization."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    indices = rng.choice(n, size=n//2, replace=False)
    h[indices] = 1.0
    return h

def structured_init(n, seed):
    """Create a structured initialization based on low-discrepancy sequences."""
    rng = np.random.RandomState(seed)
    # Try a "block" pattern: alternate blocks of 1s and 0s
    # Block size chosen to create interesting correlations
    block_size = max(1, n // (2 * rng.randint(3, 8)))
    h = np.zeros(n)
    pos = 0
    is_one = rng.randint(0, 2)
    while pos < n:
        length = min(block_size + rng.randint(-block_size//2, block_size//2 + 1), n - pos)
        if length <= 0:
            length = 1
        if is_one:
            h[pos:pos+length] = 1.0
        is_one = 1 - is_one
        pos += length
    # clip to exactly n/2 ones
    ones = np.sum(h)
    if ones > n/2:
        idx = np.where(h == 1)[0]
        remove = rng.choice(idx, size=int(ones - n/2), replace=False)
        h[remove] = 0
    elif ones < n/2:
        idx = np.where(h == 0)[0]
        add = rng.choice(idx, size=int(n/2 - ones), replace=False)
        h[add] = 1
    return h

    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
from scipy.ndimage import uniform_filter1d
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n-s])
            grad[:n-s] += wk * (-h[s:])
        elif s < 0:
            t_ = -s
            grad[:n-t_] += wk * (1.0 - h[t_:])
            grad[t_:] += wk * (-h[:n-t_])
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    grad *= dx
    return val, grad

def adam_optimize(h, max_iters, beta, lr=0.05):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    m = np.zeros(n)
    v = np.zeros(n)
    best_x = x.copy()
    best_v = compute_c5(x)
    b1, b2, eps = 0.85, 0.999, 1e-8
    for t in range(1, max_iters + 1):
        val, grad = smoothmax_val_grad(x, beta)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x_new = project_feasible(x_new, target)
        c = compute_c5(x_new)
        if c < best_v:
            best_v = c
            best_x = x_new.copy()
        x = x_new
    return best_x, best_v

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj_func(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    def obj_grad(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    def con_func(x):
        return x.sum() - target
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_func, 'jac': con_jac}]
    try:
        res = minimize(obj_func, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def block_coordinate(h, n_blocks, block_beta, max_rounds):
    n = len(h)
    target = n / 2.0
    block_size = n // n_blocks
    x = h.copy().astype(float)
    best_c5 = compute_c5(x)
    
    for r in range(max_rounds):
        for b in range(n_blocks):
            start = b * block_size
            end = start + block_size if b < n_blocks - 1 else n
            x_block = x[start:end].copy()
            
            def block_obj(xb):
                x_full = x.copy()
                x_full[start:end] = xb
                return compute_c5(x_full)
            
            # Use SLSQP on just this block
            try:
                res = minimize(block_obj, x_block, method='SLSQP',
                              bounds=Bounds(np.zeros(block_size), np.ones(block_size)),
                              options={'maxiter': 50})
                x_block = np.clip(res.x, 0, 1)
            except:
                pass
            
            # Re-project to maintain global constraint
            x[start:end] = x_block
            x = project_feasible(x, target)
            c = compute_c5(x)
            if c < best_c5:
                best_c5 = c
    
    return x, best_c5

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
from scipy.ndimage import uniform_filter1d
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n - s])
            grad[:n - s] += wk * (-h[s:])
        elif s < 0:
            t_ = -s
            grad[:n - t_] += wk * (1.0 - h[t_:])
            grad[t_:] += wk * (-h[:n - t_])
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    grad *= dx
    return val, grad

def adam_optimize(h, max_iters, beta, lr, b1=0.8, b2=0.999):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    momentum = np.zeros(n)
    velocity = np.zeros(n)
    best_x = x.copy()
    best_v = compute_c5(x)
    eps = 1e-8
    for t in range(1, max_iters + 1):
        val, grad = smoothmax_val_grad(x, beta)
        momentum = b1 * momentum + (1 - b1) * grad
        velocity = b2 * velocity + (1 - b2) * grad ** 2
        m_hat = momentum / (1 - b1 ** t)
        v_hat = velocity / (1 - b2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x_new = x - step
        x_new = project_feasible(x_new, target)
        c = compute_c5(x_new)
        if c < best_v:
            best_v = c
            best_x = x_new.copy()
        x = x_new
    return best_x, best_v

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj_func(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v

    def obj_grad(x):
        _, g = smoothmax_val_grad(x, beta)
        return g

    def con_func(x):
        return x.sum() - target

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_func, 'jac': con_jac}]
    try:
        res = minimize(obj_func, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def nudge_search(h, n_iterations, seed, max_perturb=0.02):
    rng = np.random.RandomState(seed)
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(n_iterations):
        scale = max_perturb * (1.0 - 0.5 * t / n_iterations)
        pert = rng.randn(n) * scale
        x_new = project_feasible(x + pert, target)
        c = compute_c5(x_new)
        if c <= best_v:
            x = x_new
            if c < best_v:
                best_v = c
                best_x = x_new.copy()
    return best_x, best_v

def binary_smooth_init(n, seed, block_size):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    pos = 0
    is_one = rng.randint(0, 2)
    while pos < n:
        length = max(1, block_size + rng.randint(-block_size // 3, block_size // 3 + 1))
        if is_one:
            h[pos:pos + length] = 1.0
        is_one = 1 - is_one
        pos += length
    ones = np.sum(h)
    if ones > n // 2:
        idx = np.where(h == 1)[0]
        remove = rng.choice(idx, size=int(ones - n // 2), replace=False)
        h[remove] = 0
    elif ones < n // 2:
        idx = np.where(h == 0)[0]
        add = rng.choice(idx, size=int(n // 2 - ones), replace=False)
        h[add] = 1
    h_smooth = uniform_filter1d(h, size=max(3, n // 40))
    return project_feasible(h_smooth, n / 2.0)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

    # Phase 4: SLSQP with high beta
