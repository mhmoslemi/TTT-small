from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_simplex(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(h)
    x = h.copy()
    # Bisection on tau
    lo, hi = x.min() - 1.0, x.max() + 1.0
    for _ in range(60):
        tau = (lo + hi) / 2.0
        s = np.clip(x - tau, 0.0, 1.0).sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    return np.clip(x - tau, 0.0, 1.0)

def fft_corr(h):
    """Compute full linear correlation of h with (1-h) using FFT."""
    n = len(h)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    h_fft = np.fft.rfft(h, size)
    oh_fft = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(h_fft * np.conj(oh_fft), size)[:2*n-1]
    return corr

    """Compute smooth max of correlation and its gradient using FFT."""
    n = len(h)
    dx = 2.0 / n
    corr = fft_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = exps.sum()

    # G2[j] = (h * w_rev)[n-1+j]

    fft_w = np.fft.rfft(w, conv_size)
    fft_wf = np.fft.rfft(w_rev, conv_size)
    
    conv1 = np.fft.irfft(fft_1hm * fft_w, conv_size)
    conv2 = np.fft.irfft(fft_h * fft_wf, conv_size)
    
    G1 = conv1[n-1:2*n-1]
    G2 = conv2[n-1:2*n-1]
    
    grad = dx * (G1 - G2)
    return val, grad

def projected_gradient_step(h, grad, lr, target):
    """Take a projected gradient step."""
    h_new = h - lr * grad
    return project_onto_simplex(h_new, target)

def nesterov_optimize(h, max_iters, beta, lr0, decay=0.999, target_ratio=0.5):
    """Nesterov accelerated projected gradient for smooth max."""
    n = len(h)
    target = n * target_ratio
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_onto_simplex(y, target)
        
        v, grad = smoothmax_val_grad_fft(y, beta)
        x_new = project_onto_simplex(y - lr_t * grad, target)
        
        # Evaluate actual objective
        actual_v = compute_c5(x_new)
        if actual_v < best_v:
            best_v = actual_v
            best_x = x_new.copy()
        
        x_prev = x
        x = x_new
    
    return best_x, best_v

# Helper functions
def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode="full")) * 2.0/n)

def project_box_sum(h, target):
    n = len(h)
    x = h.copy()
    lo, hi = x.min() - 1.5, x.max() + 1.5
    for _ in range(50):
        tau = 0.5*(lo+hi)
        s = np.clip(x - tau, 0.0, 1.0).sum()
        if s > target: lo = tau
        else: hi = tau
    return np.clip(x - 0.5*(lo+hi), 0.0, 1.0)

def smoothmax_grad_fft(h, beta):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode="full")  # O(n^2) but fast in C
    m = np.max(corr)
    sh = corr - m
    exps = np.exp(beta * sh)
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw)/beta) * dx
    
    # FFT-based gradient
    w_rev = w[::-1]
    cs = 1
    while cs < 3*n-2: cs *= 2
    c1 = np.fft.irfft(np.fft.rfft(1.0-h, cs) * np.fft.rfft(w, cs), cs)
    c2 = np.fft.irfft(np.fft.rfft(h, cs) * np.fft.rfft(w_rev, cs), cs)
    grad = dx * (c1[n-1:2*n-1] - c2[n-1:2*n-1])
    return val, grad

def lbfgs_optimize(h0, beta, penalty, maxiter=2000):

        v, _ = smoothmax_grad_fft(x, beta)

        return v + 0.5 * penalty * s * s

        _, g = smoothmax_grad_fft(x, beta)
        s = x.sum() - target
        return g + penalty * s * np.ones(n)
