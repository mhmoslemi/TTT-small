        best_h, imp = subgrad_step(best_h, best_n / 2.0, 3, [0.02, 0.01, 0.005, 0.002, 0.001])
        if not imp and it > 2000:
            break
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
