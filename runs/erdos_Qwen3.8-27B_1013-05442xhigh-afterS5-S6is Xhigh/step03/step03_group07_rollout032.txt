import numpy as np
import cvxpy as cp
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    """Project onto {0<=h<=1, sum(h)=target_sum} using bisection."""
    h = np.clip(h, 0.0, 1.0)
    # Bisection for the Lagrange multiplier
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_trial = np.clip(h + mid, 0.0, 1.0)
        s = h_trial.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    # Final fix
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-10:
        free = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        if free.sum() > 0:
            h_proj[free] += diff / free.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.minimum(0, beta * shifted))  # numerical stability
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n-s])
            grad[:n-s] -= wk * h[s:]
        elif s < 0:
            t = -s
            grad[:n-t] += wk * (1.0 - h[t:])
            grad[t:] -= wk * h[:n-t]
        else:
            grad += wk * (1.0 - 2.0 * h)
    grad *= dx
    return val, grad

def adam_smoothmax(h_init, max_iters, beta_start, beta_end, lr_start, lr_end, target=None):
    """ADAM optimizer for smooth-max objective with projection."""
    n = len(h_init)
    if target is None:
        target = n / 2.0
    x = h_init.copy().astype(float)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    best_x = x.copy()
    best_val = compute_c5(x)
    
    for t in range(max_iters):
        progress = t / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        val, grad = smoothmax_val_grad(x, beta)
        
        # ADAM update
        m_v = 0.9 * m_v + 0.1 * grad
        v_v = 0.999 * v_v + 0.001 * grad**2
        m_hat = m_v / (1 - 0.9**(t+1))
        v_hat = v_v / (1 - 0.999**(t+1))
        x_new = project_feasible(x - lr * m_hat / (np.sqrt(v_hat) + 1e-8), target)
        
        x = x_new
        c = compute_c5(x)
        if c < best_val:
            best_val = c
            best_x = x.copy()
    
    return best_x, best_val

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    
    def obj_grad(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    
    def con_fun(x):
        return x.sum() - target
    
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    return project_feasible(h_new, n_new / 2.0)

    n = len(h)
    target = n / 2.0

from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-12:
        free = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        if free.sum() > 0:
            h_proj[free] += diff / free.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def smoothmax_val_grad(h, beta):
    """Corrected smooth-max value and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.minimum(0.0, beta * shifted))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[:n - s] += wk * (1.0 - h[s:])
            grad[s:] += wk * (-h[:n - s])
        elif s < 0:
            t = -s
            grad[t:] += wk * (1.0 - h[:n - t])
            grad[:n - t] += wk * (-h[t:])
        else:
            grad += wk * (1.0 - 2.0 * h)
    grad *= dx
    return val, grad

def adam_optimize(h_init, max_iters, beta_start, beta_end, lr_start, lr_end, target):
    n = len(h_init)
    x = h_init.copy().astype(float)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    best_x = x.copy()
    best_val = compute_c5(x)
    patience = 0
    for t in range(max_iters):
        progress = t / max(max_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        val, grad = smoothmax_val_grad(x, beta)
        m_v = 0.9 * m_v + 0.1 * grad
        v_v = 0.999 * v_v + 0.001 * grad * grad
        m_hat = m_v / (1.0 - 0.9 ** (t + 1))
        v_hat = v_v / (1.0 - 0.999 ** (t + 1))
        x_new = project_feasible(x - lr * m_hat / (np.sqrt(v_hat) + 1e-8), target)
        x = x_new
        c = compute_c5(x)
        if c < best_val:
            best_val = c
            best_x = x.copy()
            patience = 0
        else:
            patience += 1
        if patience > 500:
            x = best_x.copy()
            patience = 0
    return best_x, best_val

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    def obj_grad(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    def con_fun(x):
        return x.sum() - target
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    return project_feasible(h_new, n_new / 2.0)

def nesterov_smoothmax(h_init, max_iters, lr0, decay, beta, target):

    x = h_init.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_val = compute_c5(x)
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2.0)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        _, grad = smoothmax_val_grad(y, beta)
        x_new = project_feasible(y - lr_t * grad, target)
        c = compute_c5(x_new)
        if c < best_val:
            best_val = c
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_val
