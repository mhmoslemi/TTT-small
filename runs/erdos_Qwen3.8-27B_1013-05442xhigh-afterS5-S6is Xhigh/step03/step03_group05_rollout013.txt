import numpy as np
from scipy.optimize import minimize, Bounds
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def corr_values(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_for_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        m = n - lag
        grad[:m] += -h[lag:n] * dx
        grad[lag:] += (1.0 - h[:m]) * dx
    else:
        d = -lag
        m = n - d
        grad[:m] += (1.0 - h[d:n]) * dx
        grad[d:] += -h[:m] * dx
    return grad

def smooth_max(x, beta):
    m = np.max(x)
    shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(shift)) / len(x)) / beta

def objective_fn(h, dx, beta):
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return smooth_max(c, beta)

def sum_fn(h, target):
    return np.sum(h) - target

def sum_jac(h, target):
    return np.ones_like(h)

def gradient_descent(h, n, target, dx, max_iters, time_check, min_time, best_state, top_k=5):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    center = n - 1
    stale = 0
    for it in range(max_iters):
        if time_check() < min_time:
            break
        cv = corr_values(cur, dx)
        k = min(top_k, len(cv))
        top_idx = np.argsort(cv)[::-1][:k]
        top_vals = cv[top_idx]
        gap = top_vals[0] - top_vals[-1]
        temp = 1.0 / max(gap, 0.002)
        weights = np.exp((top_vals - top_vals[0]) * temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for t_idx, w in zip(top_idx, weights):
            lag = int(t_idx) - center
            grad += w * grad_for_lag(cur, lag, n, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        step = 0.04
        improved = False
        for _ in range(30):
            h_trial = project_h(cur - step * grad, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-16:
                cur = h_trial
                cur_c5 = c5_trial
                improved = True
                stale = 0
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
                break
            step *= 0.5
        if not improved:
            stale += 1
            if stale > 50:
                T = 0.008
            else:
                T = 0.003 * np.exp(-it / 1000.0) + 0.001
            noise = np.random.normal(0, T, n)
            h_trial = project_h(cur + noise, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 + T * 5:
                cur = h_trial
                cur_c5 = c5_trial
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
            stale = 0
    return cur, cur_c5

def pair_cd(h, n, target, dx, max_rounds, time_check, min_time, best_state):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    center = n - 1
    for rnd in range(max_rounds):
        if time_check() < min_time:
            break
        cv = corr_values(cur, dx)
        top_idx = np.argsort(cv)[::-1][:5]
        active_lags = [int(t) - center for t in top_idx]
        for _ in range(n // 3):
            if time_check() < min_time:
                break
            i, j = np.random.choice(n, 2, replace=False)
            vi, vj = cur[i], cur[j]
            s = vi + vj
            best_t = (vi, vj)
            best_c5 = cur_c5
            for t in np.linspace(0.0, s, 9):
                t1, t2 = t, s - t
                if t1 < -1e-10 or t1 > 1.0 + 1e-10 or t2 < -1e-10 or t2 > 1.0 + 1e-10:
                    continue
                cur[i], cur[j] = t1, t2
                c5_new = compute_c5(cur)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    best_t = (t1, t2)
            cur[i], cur[j] = best_t
            if best_c5 < cur_c5 - 1e-15:
                cur_c5 = best_c5
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
    return cur, cur_c5
