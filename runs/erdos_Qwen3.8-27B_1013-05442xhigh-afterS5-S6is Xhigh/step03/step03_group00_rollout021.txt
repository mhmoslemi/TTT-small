        if s > target:

        else:

    # Phase 3: Basin hopping at n=400 with structured perturbations
    n = len(h_cur)
    target = n / 2.0
    for hop in range(20):
        if remaining() < 60:
            break
        noise = 0.012 * (0.72 ** hop)
        if hop % 3 == 0:
            h_pert = h_cur + rng.randn(n) * noise
        elif hop % 3 == 1:
            bsize = max(4, n // 15)
            h_pert = h_cur.copy()
            b1 = rng.randint(0, n - 3 * bsize)
            b2 = b1 + bsize + rng.randint(0, 2 * bsize)
            h_pert[b1:b1+bsize], h_pert[b2:b2+bsize] = h_pert[b2:b2+bsize].copy(), h_pert[b1:b1+bsize].copy()
            h_pert = h_pert + rng.randn(n) * noise * 0.5
        else:
            freq = rng.uniform(0.5, 6.0)
            phase = rng.uniform(0, 2 * np.pi)
            h_pert = h_cur + noise * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        h_pert = project_box_sum(h_pert, target)
        h_local = fista(h_pert, 300, 200, top_k=30)
        h_local = fista(h_local, 600, 150, top_k=None)
        c_local = compute_c5(h_local)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()
            h_cur = h_local.copy()

    # Phase 4: Try n=600 if time permits
    if remaining() > 250:
        h_600 = interp_to(h_best, 600)
        h_cur = h_600
        for beta in [100, 300, 500, 800]:
            if remaining() < 80:
                break
            h_new = fista(h_cur, beta, 350, top_k=None)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                h_best = h_new.copy()
            h_cur = h_new

    # Phase 5: SLSQP polish on best solution
    if remaining() > 60:
        h_fin = slsqp_refine(h_best, 600, 400, top_k=None)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_c5 = c_fin
            h_best = h_fin

    # Phase 6: Final micro-perturbation search
    if remaining() > 40:
        n = len(h_best)
        target = n / 2.0
        for trial in range(12):
            if remaining() < 20:
                break
            eps = 0.004
            h_pert = h_best + rng.randn(n) * eps
            h_pert = project_box_sum(h_pert, target)
            h_local = fista(h_pert, 800, 150, top_k=None)
            c_local = compute_c5(h_local)
            if c_local < best_c5:
                best_c5 = c_local
                h_best = h_local.copy()

    h_best = project_box_sum(h_best, len(h_best) / 2.0)
    final_c5 = compute_c5(h_best)
    return h_best, final_c5, len(h_best)
