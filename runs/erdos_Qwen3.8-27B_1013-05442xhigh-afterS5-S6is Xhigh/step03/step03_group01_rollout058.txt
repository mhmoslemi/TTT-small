def compute_gradient(h, w, n):
    """Compute gradient of smooth objective.
    h: length n
    w: length 2n-1 (weights for each lag)
    Returns: gradient of length n
    """
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # (1-h[j-lag]) for j in [lag, n-1]
            grad[lag:] += wk * (1 - h[:-lag]) if lag > 0 else wk * (1 - h)
            # -h[j+lag] for j in [0, n-1-lag]
            if lag > 0:
                grad[:-lag] += wk * (-h[lag:])
            else:
                grad += wk * (-h)
        else:
            lag2 = -lag
            # (1-h[j+lag2]) for j in [0, n-1-lag2]
            if lag2 > 0:
                grad[:-lag2] += wk * (1 - h[lag2:])
            else:
                grad += wk * (1 - h)
            # -h[j-lag2] for j in [lag2, n-1]
            if lag2 > 0:
                grad[lag2:] += wk * (-h[:-lag2])
            else:
                grad += wk * (-h)
    
    return grad * dx

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.interpolate import interp1d
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = hp < 1.0
        else:
            mask = hp > 0.0
        cnt = mask.sum()
        if cnt > 0:
            hp[mask] += diff / cnt
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (np.log(np.exp(sh - mx).sum()) + mx) / alpha)

def slsqp_opt(h0, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def single_lag_grad(h, k, n):
    lag = k - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g += (1 - h) - h
        else:
            g[lag:] += (1 - h[:-lag])
            g[:-lag] += (-h[lag:])
    else:
        lag2 = -lag
        if lag2 == 0:
            g += (1 - h) - h
        else:
            g[:-lag2] += (1 - h[lag2:])
            g[lag2:] += (-h[:-lag2])
    return g

def lp_linearize(h0, n, max_iter, deadline, gap=0.005):
    h = h0.copy()
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = c.max()
        active = np.where(c > m - gap)[0]
        if len(active) == 0:
            active = np.array([c.argmax()])
        A_rows = []
        b_row = []
        for k in active:
            gk = single_lag_grad(h, k, n) * dx
            row = np.concatenate([gk, [-1.0]])
            A_rows.append(row)
            b_row.append(c[k] - gk @ h)
        A_ub = np.array(A_rows)
        b_ub = np.array(b_row)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        bounds_lp = [(0, 1)] * n + [(None, None)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds_lp, method='highs')
        except Exception:
            break
        if not res.success:
            break
        h_new = project_feasible(res.x[:n], n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            h = h_new
        else:
            if c_new < m + 1e-10:
                h = h_new
            else:

    return h, best_c

def local_search(h, n, deadline, steps, max_passes=5):
    h = h.copy()
    best = compute_c5(h, n)
    for step in steps:
        for _ in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i + 1, n):
                    for s in (1.0, -1.0):
                        hi = h[i] + s * step
                        hj = h[j] - s * step
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            oi, oj = h[i], h[j]
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best - 1e-15:
                                best = c
                                improved = True
                            else:
                                h[i], h[j] = oi, oj
            if not improved:
                break
    return h, best
