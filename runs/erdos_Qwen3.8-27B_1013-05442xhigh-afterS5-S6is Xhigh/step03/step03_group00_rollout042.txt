from scipy.interpolate import interp1d

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def compute_corr_array(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break

        else:

    return h

    dx = 2.0 / n

        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if hi2 > lo2:

    grad *= dx

                       bounds=bounds, constraints=cons,

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_arr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def proj(h, tgt):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = tgt - h.sum()
        if abs(d) < 1e-15:
            break
        fr = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not fr.any():
            h = np.clip(h + d / len(h), 0.0, 1.0)
        else:
            h[fr] += d / fr.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smx(h, beta, tk):
    n = len(h)
    dx = 2.0 / n
    c = corr_arr(h)
    if tk is not None and tk < len(c):
        idx = np.argsort(-c)[:tk]
    else:
        idx = np.arange(len(c))
    cs = c[idx]
    m = cs.max()
    e = np.exp(beta * (cs - m))
    sw = e.sum()
    w = e / sw
    v = (m + np.log(sw) / beta) * dx
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        k = idx[ii]
        s = k - n + 1
        a = max(0, s)
        b = min(n, n + s)
        if b > a:
            g[a:b] += w[ii] * (1.0 - h[a - s:b - s])
        a2 = max(0, -s)
        b2 = min(n, n - s)
        if b2 > a2:
            g[a2:b2] += w[ii] * (-h[a2 + s:b2 + s])
    g *= dx
    return v, g

def _obj(x, beta, tk):
    return smx(x, beta, tk)[0]

def _grad(x, beta, tk):
    return smx(x, beta, tk)[1]

def _cf(x, tgt):
    return x.sum() - tgt

def _cj(x, tgt):
    return np.ones(len(x))

def slsqp(h0, beta, mi, tgt, tk=None):
    n = len(h0)
    try:
        r = minimize(_obj, h0, args=(beta, tk), jac=_grad,
                     method='SLSQP',
                     bounds=Bounds(np.zeros(n), np.ones(n)),
                     constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cj,
                                   'args': (tgt,)}],
                     options={'maxiter': mi, 'ftol': 1e-15})
        return proj(r.x, tgt)
    except Exception:
        return proj(h0, tgt)

def sg_polish(h, tgt, ns, a0, dec):
    n = len(h)
    dx = 2.0 / n
    a = a0
    best = h.copy()
    bc = compute_c5(h)
    for _ in range(ns):
        c = corr_arr(h)
        km = int(np.argmax(c))
        s = km - n + 1
        g = np.zeros(n)
        a1 = max(0, s)
        b1 = min(n, n + s)
        if b1 > a1:
            g[a1:b1] += 1.0 - h[a1 - s:b1 - s]
        a2 = max(0, -s)
        b2 = min(n, n - s)
        if b2 > a2:
            g[a2:b2] += -h[a2 + s:b2 + s]
        g *= dx
        hn = proj(h - a * g, tgt)
        cn = compute_c5(hn)
        if cn < bc:
            best = hn.copy()
            bc = cn
        h = hn
        a *= dec
    return best, bc

def interp_n(h,nn):
    no = len(h)
    xo = (np.arange(no) + 0.5) * (2.0 / no)
    xn = (np.arange(nn) + 0.5) * (2.0 / nn)
    f = interp1d(xo, h, kind='cubic', fill_value='extrapolate')
    return proj(np.clip(f(xn), 0.0, 1.0), nn / 2.0)

def bhop(hb, bc, nh, sd, t0, bud, bts, mi, nb, nd, tk=30):
    rng = np.random.RandomState(sd)
    n = len(hb)
    t = n / 2.0
    for hp in range(nh):
        if bud - (time.time() - t0) < 45:
            break
        nz = nb * (nd ** hp)
        h = proj(hb + rng.randn(n) * nz, t)
        for b in bts:
            if bud - (time.time() - t0) < 25:
                break
            h = slsqp(h, b, mi, t, tk)
        if bud - (time.time() - t0) > 60:
            h, _ = sg_polish(h, t, 60, 0.04, 0.999)
        c = compute_c5(h)
        if c < bc:
            bc = c
            hb = h.copy()
    return hb, bc

def sa_opt(hb, bc, ns, t0, bud, ti, tf, sd, noise=0.015):
    rng = np.random.RandomState(sd)
    n = len(hb)
    t = n / 2.0
    h = hb.copy()
    c = bc
    for i in range(ns):
        if bud - (time.time() - t0) < 20:
            break
        temp = ti * (tf / ti) ** (i / max(ns - 1, 1))
        hn = proj(h + rng.randn(n) * noise, t)
        cn = compute_c5(hn)
        if cn < c or rng.rand() < np.exp(-(cn - c) / max(temp, 1e-10)):
            h = hn
            if cn < c:
                c = cn
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_cur = np.array(initial_h_values, dtype=float).copy()
    else:
        h_cur = np.full(200, 0.5)

    best_c5 = compute_c5(h_cur)
    h_best = h_cur.copy()
    n = len(h_cur)
    tgt = n / 2.0

    # Phase 1: SLSQP continuation
    for beta in [10, 50, 200, 600, 1500]:
        if rem() < 60:
            break
        h_cur = slsqp(h_cur, beta, 500, tgt, None)
        c = compute_c5(h_cur)
        if c < best_c5:
            best_c5 = c
            h_best = h_cur.copy()

    # Phase 2: Subgradient polish
    if rem() > 120:
        h2, c2 = sg_polish(h_best, tgt, 300, 0.1, 0.998)
        if c2 < best_c5:
            best_c5 = c2
            h_best = h2

    # Phase 3: Basin hopping
    if rem() > 200:
        h_best, best_c5 = bhop(h_best, best_c5, 20, seed + 1, t0, budget_s,
                                (200, 600), 350, 0.025, 0.7, 30)

    # Phase 4: Multi-resolution
    for nt in [150, 250, 300, 400]:
        if rem() < 180:
            break
        hi = interp_n(h_best, nt)
        for b in [50, 200, 800]:
            if rem() < 70:
                break
            hi = slsqp(hi, b, 400, nt / 2.0, 25)
        ct = compute_c5(hi)
        if ct < best_c5:
            best_c5 = ct
            h_best = hi

    # Phase 5: Fine basin hopping
    if rem() > 120:
        h_best, best_c5 = bhop(h_best, best_c5, 15, seed + 200, t0, budget_s,
                                (500, 1200), 400, 0.008, 0.78, 20)

    # Phase 6: Simulated annealing
    if rem() > 100:
        h_best, best_c5 = sa_opt(h_best, best_c5, 5000, t0, budget_s,
                                   0.005, 0.0003, seed + 500, 0.012)

    # Phase 7: Final polish
    n = len(h_best)
    tg = n / 2.0
    if rem() > 40:

        if cf2 < best_c5:
            best_c5 = cf2
            h_best = hf2

    return h_best, best_c5, len(h_best)

def sg_polish_multi(h, tgt, ns, a0, dec, K=5):
    n = len(h)
    dx = 2.0 / n
    a = a0

    bc = compute_c5(h)
    for _ in range(ns):
        c = corr_arr(h)
        top_k = np.argsort(-c)[:K]
        # Weight by proximity to max
        w = np.exp(100 * (c[top_k] - c.max()))
        w = w / w.sum()
        g = np.zeros(n)
        for ii, km in enumerate(top_k):
            s = km - n + 1
            a1 = max(0, s)
            b1 = min(n, n + s)

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def corr_arr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def proj(h, tgt):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = tgt - h.sum()
        if abs(d) < 1e-15:
            break
        fr = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not fr.any():
            h = np.clip(h + d / len(h), 0.0, 1.0)
        else:
            h[fr] += d / fr.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smx(h, beta, tk):
    n = len(h)
    dx = 2.0 / n
    c = corr_arr(h)
    if tk is not None and tk < len(c):
        idx = np.argsort(-c)[:tk]
    else:
        idx = np.arange(len(c))
    cs = c[idx]
    m = cs.max()
    e = np.exp(beta * (cs - m))
    sw = e.sum()
    w = e / sw
    v = (m + np.log(sw) / beta) * dx
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        s = idx[ii] - n + 1
        a = max(0, s); b = min(n, n + s)
        if b > a:
            g[a:b] += w[ii] * (1.0 - h[a - s:b - s])
        a2 = max(0, -s); b2 = min(n, n - s)
        if b2 > a2:
            g[a2:b2] += w[ii] * (-h[a2 + s:b2 + s])
    g *= dx
    return v, g

def _obj(x, beta, tk):
    return smx(x, beta, tk)[0]

def _grad(x, beta, tk):
    return smx(x, beta, tk)[1]

def _cfun(x, tgt):
    return x.sum() - tgt

def _cjac(x, tgt):
    return np.ones(len(x))

def slsqp(h0, beta, mi, tgt, tk=None):
    n = len(h0)
    try:
        r = minimize(_obj, h0, args=(beta, tk), jac=_grad,
                     method='SLSQP',
                     bounds=Bounds(np.zeros(n), np.ones(n)),
                     constraints=[{'type': 'eq', 'fun': _cfun, 'jac': _cjac,
                                   'args': (tgt,)}],
                     options={'maxiter': mi, 'ftol': 1e-15})
        return proj(r.x, tgt)
    except Exception:
        return proj(h0, tgt)

def sg_polish(h, tgt, ns, a0, dec, K=1):
    n = len(h)
    dx = 2.0 / n
    a = a0
    best = h.copy()
    bc = compute_c5(h)
    for _ in range(ns):
        c = corr_arr(h)
        if K > 1:
            topk = np.argsort(-c)[:K]
            wts = np.exp(50.0 * (c[topk] - c[topk][0]))
            wts = wts / wts.sum()
        else:
            topk = np.array([int(np.argmax(c))])
            wts = np.array([1.0])
        g = np.zeros(n)
        for ii, km in enumerate(topk):
            s = km - n + 1
            a1 = max(0, s); b1 = min(n, n + s)
            if b1 > a1:
                g[a1:b1] += wts[ii] * (1.0 - h[a1 - s:b1 - s])
            a2 = max(0, -s); b2 = min(n, n - s)
            if b2 > a2:
                g[a2:b2] += wts[ii] * (-h[a2 + s:b2 + s])
        g *= dx
        hn = proj(h - a * g, tgt)
        cn = compute_c5(hn)
        if cn < bc:
            best = hn.copy()
            bc = cn
        h = hn
        a *= dec
    return best, bc

def interp_n(h, nn):
    no = len(h)
    xo = (np.arange(no) + 0.5) * (2.0 / no)
    xn = (np.arange(nn) + 0.5) * (2.0 / nn)
    f = interp1d(xo, h, kind='cubic', fill_value='extrapolate')
    return proj(np.clip(f(xn), 0.0, 1.0), nn / 2.0)

def bhop(hb, bc, nh, sd, t0, bud, bts, mi, nb, nd, tk=30, K=1):
    rng = np.random.RandomState(sd)
    n = len(hb)
    t = n / 2.0
    for hp in range(nh):
        if bud - (time.time() - t0) < 45:
            break
        nz = nb * (nd ** hp)
        h = proj(hb + rng.randn(n) * nz, t)
        for b in bts:
            if bud - (time.time() - t0) < 25:
                break
            h = slsqp(h, b, mi, t, tk)
        if bud - (time.time() - t0) > 50:
            h, _ = sg_polish(h, t, 80, 0.04, 0.999, K)
        c = compute_c5(h)
        if c < bc:
            bc = c
            hb = h.copy()
    return hb, bc
