    grad = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag >= 0:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
        else:
            la = -lag
            grad[la:] += (1.0 - h[:n - la])
            grad[:n - la] -= h[la:]
    return grad / n_top

def adaptive_subgrad(h, target, n, t0, budget, max_iter=20000):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_top = 8
    step = 0.02
    patience = 0
    max_patience = 3000
    for it in range(max_iter):
        rem = budget - (time.time() - t0)
        if rem < 3:
            break
        corr = corr_full(h)
        g = subgrad_step(h, corr, n_top, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g_normed = g / gn
        trial = project_h(h - step * g_normed, target)
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-15:
            h = trial
            best_c5 = c_try
            best_h = trial.copy()
            patience = 0
            step = min(step * 1.02, 0.05)
        else:
            patience += 1
            step *= 0.95
            if patience % 10 == 0:
                trial2 = project_h(h - 0.5 * step * g_normed, target)
                c2 = compute_c5(trial2)
                if c2 < best_c5 - 1e-15:
                    h = trial2
                    best_c5 = c2
                    best_h = trial2.copy()
                    patience = 0
            if patience > max_patience:
                break
    return best_h, best_c5

def fine_coord_descent(h, target, n, t0, budget, phases=5):
    best_c5 = compute_c5(h)
    sizes = [0.05, 0.02, 0.008, 0.003, 0.001]
    for phase in range(min(phases, len(sizes))):
        rem = budget - (time.time() - t0)
        if rem < 2:
            break
        delta = sizes[phase]
        improved = True
        sweeps = 0
        while improved and sweeps < 50:
            improved = False
            sweeps += 1
            rem = budget - (time.time() - t0)
            if rem < 2:
                break
            for j in range(n):
                for sign in (1, -1):
                    trial = h.copy()
                    trial[j] = np.clip(trial[j] + sign * delta, 0, 1)
                    trial = project_h(trial, target)
                    c_try = compute_c5(trial)
                    if c_try < best_c5 - 1e-15:
                        h = trial
                        best_c5 = c_try
                        improved = True
                        break
    return h, best_c5

def pairwise_descent(h, target, n, t0, budget, max_sweeps=20):
    best_c5 = compute_c5(h)
    delta = 0.03
    for sw in range(max_sweeps):
        rem = budget - (time.time() - t0)
        if rem < 2:
            break
        improved = False
        for j in range(n - 1):
            rem = budget - (time.time() - t0)
            if rem < 1:
                break
            trial = h.copy()
            trial[j] = np.clip(trial[j] + delta, 0, 1)
            trial[j + 1] = np.clip(trial[j + 1] - delta, 0, 1)
            trial = project_h(trial, target)
            c_try = compute_c5(trial)
            if c_try < best_c5 - 1e-15:
                h = trial
                best_c5 = c_try
                improved = True
    return h, best_c5

def binary_perturb(h, target, n, t0, budget, n_trials=30):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    rem = budget - (time.time() - t0)
    per_trial = max(rem / n_trials, 0.5)
    for t in range(n_trials):
        if time.time() - t0 > budget - 5:
            break
        # Create binary version
        hb = (h > 0.5).astype(float)
        # Fix sum
        ones = np.where(hb == 1.0)[0]
        zeros = np.where(hb == 0.0)[0]
        need = int(target) - len(ones)
        if need > 0:
            idx = np.random.choice(zeros, min(need, len(zeros)), replace=False)
            hb[idx] = 1.0
        elif need < 0:
            idx = np.random.choice(ones, min(-need, len(ones)), replace=False)
            hb[idx] = 0.0
        c_bin = compute_c5(hb)
        if c_bin < best_c5:
            best_c5 = c_bin
            best_h = hb.copy()
        # Flip a few positions from binary, reoptimize
        if len(ones) > 5 and len(zeros) > 5:
            n_flip = np.random.randint(2, 6)
            flip_from_1 = np.random.choice(ones, min(n_flip, len(ones)), replace=False)
            flip_from_0 = np.random.choice(zeros, min(n_flip, len(zeros)), replace=False)
            n_f = min(len(flip_from_1), len(flip_from_0))
            h_pert = h.copy()
            for i in range(n_f):
                h_pert[flip_from_1[i]] = 0.0
                h_pert[flip_from_0[i]] = 1.0
            h_pert = project_h(h_pert, target)
            c_pert = compute_c5(h_pert)
            if c_pert < best_c5:
                best_c5 = c_pert
                best_h = h_pert.copy()
    return best_h, best_c5

def resample_h(h, n_new):
    if len(h) == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, len(h))
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 320
        h = np.full(n, 0.5)

    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)

    bounds = Bounds(np.zeros(n), np.ones(n))

    # Phase 1: SLSQP with top-k soft max, various parameters
    for k, beta in [(5, 80), (10, 60), (20, 40), (50, 30), (10, 100), (5, 200)]:
        rem = budget_s - (time.time() - t0)
        if rem < 20:
            break
        r = do_slsqp(best_h, n, target, k, beta, bounds, 5000, 1e-15)
        c = compute_c5(r)
        if c < best_c5:
            best_c5 = c
            best_h = r

    # Phase 2: Adaptive subgradient
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        best_h, best_c5 = adaptive_subgrad(best_h, target, n, t0, budget_s)

    # Phase 3: SLSQP with higher precision
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        for k, beta in [(3, 300), (5, 500), (8, 200)]:
            rem = budget_s - (time.time() - t0)
            if rem < 10:
                break
            r = do_slsqp(best_h, n, target, k, beta, bounds, 8000, 1e-16)
            c = compute_c5(r)
            if c < best_c5:
                best_c5 = c
                best_h = r

    # Phase 4: Fine coordinate descent
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        best_h, best_c5 = fine_coord_descent(best_h, target, n, t0, budget_s)

    # Phase 5: Pairwise descent
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        best_h, best_c5 = pairwise_descent(best_h, target, n, t0, budget_s)

    # Phase 6: Binary perturbation exploration
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h_bin, c_bin = binary_perturb(best_h, target, n, t0, min(rem - 5, 200))
        if c_bin < best_c5:
            best_c5 = c_bin
            best_h = h_bin

    # Phase 7: Final SLSQP polish
    rem = budget_s - (time.time() - t0)
    if rem > 10:
        r = do_slsqp(best_h, n, target, 3, 500, bounds, 10000, 1e-16)
        c = compute_c5(r)
        if c < best_c5:
            best_c5 = c
            best_h = r

    # Phase 8: Final ultra-fine coordinate descent
    rem = budget_s - (time.time() - t0)
    if rem > 5:
        best_h, best_c5 = fine_coord_descent(best_h, target, n, t0, budget_s, phases=3)

    return best_h, best_c5, n
