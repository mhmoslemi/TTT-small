    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_c5(h, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag >= 0:
            grad[:n-lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n-lag]
        else:
            la = -lag
            grad[la:] += (1.0 - h[:n-la])
            grad[:n-la] -= h[la:]
    return grad / top_k

  = -d + d*h[j] + h[i]*d - d^2
  = d*(h[i] + h[j] - 1 - d)

So if h[i] + h[j] > 1 + d, the change is positive (correlation increases - bad).
If h[i] + h[j] < 1 + d, the change is negative (correlation decreases - good).

But wait, this transfer also affects other correlations. The pair (i, j) participates in multiple lag correlations.

This is getting complex. Let me just stick with the simpler approach of trying both (+d,+d) and (-d,-d) updates and also the transfer (−d,+d) update, and accept if C5 improves.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_c5(h, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag >= 0:
            grad[:n-lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n-lag]
        else:
            la = -lag
            grad[la:] += (1.0 - h[:n-la])
            grad[:n-la] -= h[la:]
    return grad / top_k

def sml_obj(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * dx

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2.0, n_new), np.linspace(0, 2.0, len(h)), h)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol, t0, budget):
    if budget - (time.time() - t0) < 5:
        return h0
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0/n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def subgrad_desc(cur, target, n, n_iter, step, t0, budget, top_k=12):
    mom = np.zeros(n)
    best = compute_c5(cur)
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        g = subgrad_c5(cur, top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        mom = 0.8 * mom - step * g
        trial = project_h(cur + mom, target)
        c = compute_c5(trial)
        if c < best - 1e-15:
            cur, best, stall = trial, c, 0
        else:
            stall += 1
            mom *= 0.5
            trial = project_h(cur + mom, target)
            c = compute_c5(trial)
            if c < best - 1e-15:
                cur, best, stall = trial, c, 0
            else:
                mom = np.zeros(n)
                step *= 0.8
        if stall > 500:
            break
    return cur, best

def paired_cd(cur, target, n, t0, budget, max_rounds=150):
    best = compute_c5(cur)
    for rnd in range(max_rounds):
        if budget - (time.time() - t0) < 2:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:8]
        improved = False
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if abs(lag) < 1:
                continue
            a = min(0, lag)
            b = max(0, -lag)
            length = n - abs(lag)
            contrib = cur[b:b+length] * (1.0 - cur[a+lag:a+lag+length] if lag > 0 else cur[a:a+length])
            if lag > 0:
                contrib = cur[:length] * (1.0 - cur[lag:length+lag])
            else:
                contrib = cur[-lag:] * (1.0 - cur[:length])
            top_pairs = np.argsort(contrib)[::-1][:20]
            for i in top_pairs:
                if lag > 0:
                    j = i + lag
                else:
                    a2 = i
                    j = i - lag
                s = cur[a2 if lag < 0 else i] + cur[j]
                # Try transfer: move mass from larger to smaller
                if s > 1.01:
                    d = 0.02
                elif s < 0.99:
                    d = 0.02
                else:
                    continue
                # Try both directions
                for sign in (1, -1):
                    trial = cur.copy()
                    if lag > 0:
                        trial[i] += sign * d
                        trial[j] += sign * d
                    else:
                        trial[a2] += sign * d
                        trial[j] += sign * d
                    trial = project_h(trial, target)
                    c = compute_c5(trial)
                    if c < best - 1e-14:
                        cur, best, improved = trial, c, True
                        break
                if improved:
                    break
        if not improved:
            break
    return cur, best

def transfer_cd(cur, target, n, t0, budget, max_rounds=100):
    """Mass transfer between pairs in worst lags (preserves sum)."""
    best = compute_c5(cur)
    for rnd in range(max_rounds):
        if budget - (time.time() - t0) < 2:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:6]
        improved = False
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if abs(lag) < 1:
                continue
            if lag > 0:
                length = n - lag
                contrib = cur[:length] * (1.0 - cur[lag:length+lag])
                top_pairs = np.argsort(contrib)[::-1][:15]
                for i in top_pairs:
                    j = i + lag
                    if cur[i] > cur[j] + 0.05:
                        d = min(0.03, (cur[i] - cur[j]) * 0.5)
                        trial = cur.copy()
                        trial[i] -= d
                        trial[j] += d
                        trial = project_h(trial, target)
                        c = compute_c5(trial)
                        if c < best - 1e-14:
                            cur, best, improved = trial, c, True
                            break
                    elif cur[j] > cur[i] + 0.05:
                        d = min(0.03, (cur[j] - cur[i]) * 0.5)
                        trial = cur.copy()
                        trial[j] -= d
                        trial[i] += d
                        trial = project_h(trial, target)
                        c = compute_c5(trial)
                        if c < best - 1e-14:
                            cur, best, improved = trial, c, True
                            break
            else:
                la = -lag
                length = n - la
                contrib = cur[la:] * (1.0 - cur[:length])
                top_pairs = np.argsort(contrib)[::-1][:15]
                for i in top_pairs:
                    a2 = i
                    j = i + la
                    if cur[a2] > cur[j] + 0.05:
                        d = min(0.03, (cur[a2] - cur[j]) * 0.5)
                        trial = cur.copy()
                        trial[a2] -= d
                        trial[j] += d
                        trial = project_h(trial, target)
                        c = compute_c5(trial)
                        if c < best - 1e-14:
                            cur, best, improved = trial, c, True
                            break
        if not improved:
            break
    return cur, best

def simple_cd(cur, target, n, sweeps, t0, budget):
    best = compute_c5(cur)
    for sw in range(sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        deltas = (0.04, -0.04, 0.02, -0.02, 0.01, -0.01)
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best - 1e-14:
                    cur, best, improved = trial, c, True
                    break
        if not improved and sw > 4:
            break
    return cur, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(140, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(h_parent)

    n_list = sorted(set([len(h_parent)] + [r for r in [320, 400, 448] if r > len(h_parent)]))
    n_list = [r for r in n_list if r <= 512]

    for n_pts in n_list:
        remaining = budget_s - (time.time() - t0)
        if remaining < 50:
            break
        target = n_pts / 2.0
        bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))

        cur = project_h(resample(best_h, n_pts), target)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        for beta in [25, 60, 150, 400, 1000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 10:
                break
            cur = do_slsqp(cur, n_pts, target, beta, bounds, 3000, 1e-14, t0, budget_s)
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_c5:

        remaining = budget_s - (time.time() - t0)
        if remaining > 15:
            cur, cur_c5 = subgrad_desc(cur, target, n_pts, 8000, 0.02, t0, budget_s, top_k=12)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        remaining = budget_s - (time.time() - t0)
        if remaining > 10:
            cur, cur_c5 = paired_cd(cur, target, n_pts, t0, budget_s, max_rounds=120)
            if cur_c5 < best_c5:

        remaining = budget_s - (time.time() - t0)
        if remaining > 8:
            cur, cur_c5 = transfer_cd(cur, target, n_pts, t0, budget_s, max_rounds=100)
            if cur_c5 < best_c5:

        remaining = budget_s - (time.time() - t0)
        if remaining > 8:
            cur, cur_c5 = simple_cd(cur, target, n_pts, 30, t0, budget_s)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        remaining = budget_s - (time.time() - t0)
        if remaining > 45:
            for att in range(4):
                remaining = budget_s - (time.time() - t0)
                if remaining < 12:
                    break
                if att == 0:
                    hp = project_h(cur + np.random.randn(n_pts) * 0.02, target)
                elif att == 1:
                    hp = cur.copy()
                    bs = 16
                    for b in range(n_pts // bs):
                        hp[b*bs:(b+1)*bs] += np.random.randn(bs) * 0.035
                    hp = project_h(hp, target)
                elif att == 2:
                    ph = np.random.rand() * 2 * np.pi
                    hp = project_h(cur + 0.025 * np.sin(np.linspace(0, 6*np.pi, n_pts) + ph), target)
                else:
                    hp = project_h(0.7 * cur + 0.3 * np.clip(np.random.rand(n_pts), 0, 1), target)

                hp = do_slsqp(hp, n_pts, target, 200, bounds, 1000, 1e-13, t0, budget_s)
                hp, hp_c5 = subgrad_desc(hp, target, n_pts, 3000, 0.015, t0, budget_s, top_k=8)
                hp, hp_c5 = paired_cd(hp, target, n_pts, t0, budget_s, max_rounds=80)
                hp, hp_c5 = simple_cd(hp, target, n_pts, 15, t0, budget_s)

    remaining = budget_s - (time.time() - t0)
    if remaining > 10 and best_h is not None:

                break
            best_h = do_slsqp(best_h, nf, tf, beta, bf, 8000, 1e-15, t0, budget_s)
            best_c5 = compute_c5(best_h)
        remaining = budget_s - (time.time() - t0)
        if remaining > 5:
            best_h, best_c5 = subgrad_desc(best_h, tf, nf, 6000, 0.004, t0, budget_s, top_k=16)
        remaining = budget_s - (time.time() - t0)
        if remaining > 3:
            best_h, best_c5 = paired_cd(best_h, tf, nf, t0, budget_s, max_rounds=200)
