    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break

    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)

    lo, hi = 0.0, 1.0

            hi = mid
        elif np.max(a[active]) < -1e-14:
            lo = mid
        else:

            i0pL = i + Ls
            valid_p = (i0pL >= 0) & (i0pL < n)
            if np.any(valid_p):
                a[valid_p] -= h[i0pL[valid_p]]

    h = project_h(h, target)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def find_optimal_t(c, a):
    lo, hi = 0.0, 1.0
    for _ in range(25):
        mid = (lo + hi) * 0.5
        vals = c + a * mid
        mx = np.max(vals)
        active = vals >= mx - 1e-9
        a_min = np.min(a[active])
        a_max = np.max(a[active])
        if a_min > 1e-13:
            hi = mid
        elif a_max < -1e-13:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def coord_descent(h, n, max_sweeps, time_fn, min_time, best_state):
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full").copy()
    Ls = np.arange(-(n - 1), n)
    prev_c5 = compute_c5(h)
    no_improve = 0
    max_no_improve = 80

    for sweep in range(max_sweeps):
        if time_fn() < min_time:
            break
        for i in range(n):
            a = np.ones(2 * n - 1)
            ip = i + Ls
            vp = (ip >= 0) & (ip < n)
            if np.any(vp):
                a[vp] -= h[ip[vp]]
            im = i - Ls
            vm = (im >= 0) & (im < n)
            if np.any(vm):
                a[vm] -= h[im[vm]]
            hc = h[i]
            c = corr - a * hc
            t = find_optimal_t(c, a)
            d = t - hc
            if abs(d) > 1e-14:
                corr += a * d
                h[i] = t
        h = project_h(h, target)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = compute_c5(h)
        if c5 < best_state[0] - 1e-14:
            best_state[0] = c5
            best_state[1] = h.copy()
            best_state[2] = n
            no_improve = 0
        elif c5 < prev_c5 - 1e-11:
            no_improve = 0
        else:
            no_improve += 1
        prev_c5 = c5
        if no_improve >= max_no_improve:
            if time_fn() < min_time + 5:
                break
            n_pert = max(3, n // 25)
            idx = np.random.choice(n, n_pert, replace=False)
            h[idx] += np.random.normal(0, 0.06, n_pert)
            h = project_h(h, target)
            corr = np.correlate(h, 1.0 - h, mode="full")
            no_improve = 0

    return h, c5

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -400, 0)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def grad_for_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag]) * dx
        grad[:n - lag] += -h[lag:] * dx
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n - l] * dx
    return grad

def steepest_descend(h, n, target, dx, max_iters, time_fn, min_time, best_state):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    for it in range(max_iters):
        if time_fn() < min_time:
            break
        corr_raw = np.correlate(cur, 1.0 - cur, mode="full")
        corr_val = corr_raw * dx
        top_k = min(6, n // 40 + 3)
        top_idx = np.argsort(corr_val)[::-1][:top_k]
        vals = corr_val[top_idx]
        temp = 250.0 + 150.0 * it / max(1, max_iters)
        weights = np.exp((vals - vals[0]) * temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        center = n - 1
        for t_idx, w in zip(top_idx, weights):
            lag = int(t_idx) - center
            grad += w * grad_for_lag(cur, lag, n, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        step = 0.03
        improved = False
        for _ in range(20):
            h_trial = project_h(cur - step * grad, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-15:
                cur = h_trial
                cur_c5 = c5_trial
                improved = True
                if cur_c5 < best_state[0]:
                    best_state[0] = c5_trial
                    best_state[1] = cur.copy()
                    best_state[2] = n
                break
            step *= 0.5
        if not improved:
            T = 0.002 * np.exp(-it / 400.0)
            noise = np.random.normal(0, T, n)
            h_trial = project_h(cur + noise, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 + T * 2:
                cur = h_trial
                cur_c5 = c5_trial
    return cur, cur_c5

def slsqp_polish(h, n, time_fn, min_time, best_state):
    target = n / 2.0
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(c, 600) * dx

    def con_fun(x):
        return np.sum(x) - target

    def con_jac(x):
        return np.ones(x.shape)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': 3000, 'ftol': 1e-15})
        h_new = project_h(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_state[0]:
            best_state[0] = c5_new
            best_state[1] = h_new
            best_state[2] = n
    except Exception:
        pass
