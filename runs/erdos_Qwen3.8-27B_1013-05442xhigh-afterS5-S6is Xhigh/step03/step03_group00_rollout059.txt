            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += wk * (-h[j_lo + s:j_hi + s])
    grad *= dx
    return val, grad


def pg_optimize(h_init, betas, n_iter_per_beta, top_k, t0, budget_s, min_time=20):
    """Projected gradient with Nesterov momentum and line search."""
    h = h_init.copy().astype(float)
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)  # momentum velocity
    
    for beta_idx, beta in enumerate(betas):
        mu = 0.9  # momentum parameter
        step = 0.1  # initial step size
        
        for it in range(n_iter_per_beta):
            if budget_s - (time.time() - t0) < min_time:
                return best_h, best_c5
            
            f, g = smoothmax_val_grad(h, beta, top_k)
            
            # Nesterov: look-ahead
            h_look = project_feasible(h + mu * v, target)
            f_look, g_look = smoothmax_val_grad(h_look, beta, top_k)
            
            # Update velocity
            v = mu * v - step * g_look
            h_new = project_feasible(h + v, target)
            
            # Check actual objective
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
            
            # Adaptive step: if we got worse, reduce step
            c_cur = compute_c5(h)
            if c_new > c_cur + 1e-10:
                step *= 0.7
            elif c_new < c_cur - 1e-12:
                step = min(step * 1.1, 1.0)
            
            h = h_new
    
    return best_h, best_c5


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def slsqp_solve(h, beta, maxiter, top_k=None, t0=None, budget_s=None, min_time=15):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        return smoothmax_val_grad(x, beta, top_k)[0]
    
    def objg(x):
        return smoothmax_val_grad(x, beta, top_k)[1]
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = ({'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)})
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = project_feasible(res.x, target)
    except Exception:
        h_out = project_feasible(h, target)
    
    if t0 is not None and budget_s is not None:
        if budget_s - (time.time() - t0) < min_time:
            pass  # too late, just return
    return h_out

            h_best = h_local.copy()

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += wk * (-h[j_lo + s:j_hi + s])
    grad *= dx
    return val, grad


class ObjFunc:
    def __init__(self, beta, top_k):
        self.beta = beta
        self.top_k = top_k
    def __call__(self, x):
        v, _ = smoothmax_val_grad(x, self.beta, self.top_k)
        return v
    def jac(self, x):
        _, g = smoothmax_val_grad(x, self.beta, self.top_k)
        return g


def con_eq_fun(x):
    return np.sum(x) - len(x) / 2.0

def con_eq_jac(x):
    return np.ones(len(x))


def slsqp_solve(h, beta, maxiter, top_k=None):
    n = len(h)
    obj = ObjFunc(beta, top_k)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = ({'type': 'eq', 'fun': con_eq_fun, 'jac': con_eq_jac})
    try:
        res = minimize(obj, h, jac=obj.jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n / 2.0)
    except Exception:
        return project_feasible(h, n / 2.0)


def pg_optimize(h_init, betas, iters_per_beta, top_k, t0, budget_s, min_time=20):
    h = h_init.copy().astype(float)
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    for beta in betas:

        step = 0.4
        for it in range(iters_per_beta):
            if budget_s - (time.time() - t0) < min_time:
                return best_h, best_c5
            f, g = smoothmax_val_grad(h, beta, top_k)
            h_la = project_feasible(h + mu * v, target)
            _, g_la = smoothmax_val_grad(h_la, beta, top_k)
            v = mu * v - step * g_la
            h_new = project_feasible(h + v, target)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
            c_cur = compute_c5(h)
            if c_new > c_cur + 1e-12:
                step = max(step * 0.6, 0.02)
                v *= 0.5
            else:
                step = min(step * 1.04, 1.5)
            h = h_new
    return best_h, best_c5


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def generate_perturbation(h_best, noise, rng):
    n = len(h_best)
    r = rng.rand()
    if r < 0.35:
        return h_best + rng.randn(n) * noise
    elif r < 0.6:
        h_p = h_best.copy()
        blen = rng.randint(max(3, n // 10), max(5, n // 3))
        start = rng.randint(0, max(1, n - blen))
        h_p[start:start + blen] += rng.randn(blen) * noise * 2.0
        return h_p
    elif r < 0.8:
        h_p = h_best.copy()
        k = rng.randint(max(2, n // 15), max(3, n // 4))
        idx = rng.choice(n, k, replace=False)
        h_p[idx] += rng.randn(k) * noise * 3.0
        return h_p
    else:
        h_p = h_best.copy()
        h_p[:n // 2] += rng.randn(n // 2) * noise
        h_p[n // 2:] += rng.randn(n - n // 2) * noise
        return h_p


def basin_hop(h_best, best_c5, n_hops, inner_betas, noise_base, noise_decay,
              seed, t0, budget_s, n_pg_iters=150, pg_top_k=25):
    rng = np.random.RandomState(seed)
    n = len(h_best)
    target = n / 2.0
    for hop in range(n_hops):
        if budget_s - (time.time() - t0) < 30:
            break
        noise = noise_base * (noise_decay ** hop)
        h_pert = project_feasible(generate_perturbation(h_best, noise, rng), target)
        h_local, c_local = pg_optimize(h_pert, inner_betas, n_pg_iters,
                                       pg_top_k, t0, budget_s, min_time=15)
        if c_local < best_c5:
            best_c5 = c_local
            h_best = h_local.copy()
    return h_best, best_c5
