    max_c = np.max(corr)
    shifted = np.clip(t * (corr - max_c), -100, 100)
    return np.exp(shifted) / np.sum(np.exp(shifted))

def compute_grad(h, t):
    """Compute gradient of smooth_max(corr(h)) w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    w = smooth_max_weights(corr, t)
    
    # grad[i] = sum_k w[n-1+k] * ((1-h[i+k]) - h[i-k]) for valid indices
    i_idx = np.arange(n)
    k_idx = np.arange(-(n-1), n)
    ik = i_idx[:, None] + k_idx[None, :]
    mask1 = (ik >= 0) & (ik < n)
    ik2 = i_idx[:, None] - k_idx[None, :]
    mask2 = (ik2 >= 0) & (ik2 < n)
    
    one_minus_h = 1.0 - h
    part1 = np.where(mask1, one_minus_h[np.clip(ik, 0, n-1)], 0.0) @ w
    part2 = np.where(mask2, h[np.clip(ik2, 0, n-1)], 0.0) @ w
    
    return part1 - part2

def lbfgs_step(h, dx, target_sum, t, maxiter=500, penalty_weight=50.0):
    n = len(h)
    
    def obj(x):
        h_x = np.clip(x, 0, 1)
        corr = np.correlate(h_x, 1 - h_x, mode="full")
        sm = smooth_max_value(corr, t)
        penalty = penalty_weight * (np.sum(h_x) - target_sum) ** 2
        return sm * dx + penalty
    
    def grad(x):
        h_x = np.clip(x, 0, 1)
        g = compute_grad(h_x, t) * dx
        g += 2 * penalty_weight * (np.sum(h_x) - target_sum) * np.ones(n)
        return g
    
    try:
        res = minimize(obj, h.copy(), jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
        return fix_sum_iterative(np.clip(res.x, 0, 1), target_sum)
    except Exception:
        return h

def frequency_move(h, target_sum, freq, amp, phase):
    """Add a sinusoidal perturbation to h."""
    n = len(h)
    x = np.linspace(0, 2, n)
    pert = amp * np.sin(2 * np.pi * freq * x + phase)
    # Remove mean to preserve sum
    pert -= np.mean(pert)
    h_new = h + pert
    return fix_sum_iterative(np.clip(h_new, 0, 1), target_sum)

def top_k_shifts(corr, k=5):
    """Return indices of top k correlation values."""
    return np.argsort(corr)[-k:]

def active_set_optimize(h, dx, target_sum, t, top_k=5, maxiter=300):

        cur_h = best_h.copy()
        cur_c5 = best_c5
        freqs = [1, 2, 3, 4, 5, 6, 8, 10, 12, 15, 20]
        for it in range(5000):
            if elapsed() > budget_s - 170:
                break
            T = max(T_min, T * 0.9995)
            f = freqs[np.random.randint(len(freqs))]
            amp = 0.02 + 0.15 * T * 10
            ph = np.random.uniform(0, 2 * np.pi)
            h_trial = freq_perturb(cur_h, best_ts, f, amp, ph)
            c5_trial = compute_c5(h_trial, best_dx)
            if c5_trial < cur_c5 or np.random.uniform() < np.exp(-(c5_trial - cur_c5) / max(T, 1e-8)):
                cur_h = h_trial
                cur_c5 = c5_trial
                if c5_trial < best_c5:
                    best_h = cur_h.copy()
                    best_c5 = c5_trial
        # Post-SA refinement
        if elapsed() < budget_s - 80:
            for t in [300, 600]:
                if elapsed() > budget_s - 60:
                    break
                h_new = lbfgs_optimize(best_h, best_dx, best_ts, t=t, maxiter=2000)
                c5_new = compute_c5(h_new, best_dx)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                h_new = active_set_refine(best_h, best_dx, best_ts, top_k=3, t=400, maxiter=1000)
                c5_new = compute_c5(h_new, best_dx)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new

    # Phase 4: Try lower resolution n=80
    if elapsed() < budget_s - 80:
        n_low = 80
        dx_low = 2.0 / n_low
        ts_low = n_low / 2.0
        h_low = np.zeros(n_low)
        for i in range(n_low):
            s = int(i * len(best_h) / n_low)
            e = int((i + 1) * len(best_h) / n_low)
            h_low[i] = np.mean(best_h[s:e])
        h_low = fix_sum(h_low, ts_low)
        c5_low = compute_c5(h_low, dx_low)
        if c5_low < best_c5:
            for t in [10, 50, 200, 500]:
                if elapsed() > budget_s - 40:
                    break
                h_low_new = lbfgs_optimize(h_low, dx_low, ts_low, t=t, maxiter=2000)
                c5_low_new = compute_c5(h_low_new, dx_low)
                if c5_low_new < c5_low:
                    h_low = h_low_new
                    c5_low = c5_low_new
            h_low = active_set_refine(h_low, dx_low, ts_low, top_k=5, t=300, maxiter=1000)
            c5_low = compute_c5(h_low, dx_low)
            if c5_low < best_c5:
                best_h = h_low
                best_c5 = c5_low
                best_n = n_low
                best_dx = dx_low
                best_ts = ts_low

    # Phase 5: Try n=320 (higher resolution)
    if elapsed() < budget_s - 60:
        n_high = 2 * len(best_h)
        dx_high = 2.0 / n_high
        ts_high = n_high / 2.0
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_high, endpoint=False)
        h_high = np.interp(x_new, x_old, best_h, period=2)
        h_high = fix_sum(h_high, ts_high)
        c5_high = compute_c5(h_high, dx_high)
        if c5_high < best_c5 + 0.001:
            for t in [20, 100, 300]:
                if elapsed() > budget_s - 20:
                    break
                h_high_new = lbfgs_optimize(h_high, dx_high, ts_high, t=t, maxiter=1000)
                c5_high_new = compute_c5(h_high_new, dx_high)
                if c5_high_new < c5_high:
                    h_high = h_high_new
                    c5_high = c5_high_new
            if c5_high < best_c5:
                best_h = h_high
                best_c5 = c5_high
                best_n = n_high
                best_dx = dx_high
                best_ts = ts_high

    # Phase 6: Final polish
    if elapsed() < budget_s - 5:
        h_new = lbfgs_optimize(best_h, best_dx, best_ts, t=800, maxiter=3000)
        c5_new = compute_c5(h_new, best_dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        h_new = active_set_refine(best_h, best_dx, best_ts, top_k=2, t=1000, maxiter=2000)
        c5_new = compute_c5(h_new, best_dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, best_ts)
    final_dx = 2.0 / len(best_h)
    best_c5 = compute_c5(best_h, final_dx)
    return best_h, best_c5, len(best_h)
