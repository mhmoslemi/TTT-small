    h[idx] = 1.0
    return h


def sharpen(h, alpha):
    n = len(h)
    h_sharp = np.where(h > 0.5, 1.0, 0.0) * alpha + h * (1.0 - alpha)
    return project_h(h_sharp, n * 0.5)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = np.array(initial_h_values, dtype=np.float64)
    n = len(h)
    h = project_h(h, n * 0.5)
    c5 = compute_c5(h)
    h_best = h.copy()
    mu = 1000.0

    # Phase 1: Temperature annealing from parent solution
    temps = [0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.005, 0.003, 0.002,
             0.001, 0.0007, 0.0005, 0.0003, 0.0002, 0.0001]
    for temp in temps:
        if budget_s - (time.time() - t0) < 30:
            break
        iters = 500 if temp >= 0.01 else 800
        h_new, c5_new = lse_opt(h, temp, mu, max_iter=iters)
        if c5_new < c5 - 1e-10:
            h, c5 = h_new, c5_new
            h_best = h.copy()

    # Phase 2a: Perturbation restarts of current best
    for i in range(15):
        if budget_s - (time.time() - t0) < 25:
            break
        scale = 0.004 + 0.016 * (i / 14.0)
        h_p = h_best + rng.randn(len(h_best)) * scale
        h_p = project_h(h_p, len(h_best) * 0.5)
        for temp in [0.01, 0.005, 0.002, 0.001]:
            if budget_s - (time.time() - t0) < 15:
                break
            h_new, c5_new = lse_opt(h_p, temp, mu, max_iter=350)
            if c5_new < c5 - 1e-10:
                h, c5 = h_new, c5_new
                h_best = h.copy()

    # Phase 2b: Random block starts with smoothing
    for i in range(6):
        if budget_s - (time.time() - t0) < 35:
            break
        n_cur = len(h_best)
        h_r = random_block(n_cur, rng)
        kernel = np.ones(3) / 3.0
        h_r = np.convolve(h_r, kernel, mode='same')
        h_r = project_h(h_r, n_cur * 0.5)
        for temp in [0.03, 0.015, 0.008, 0.003]:
            if budget_s - (time.time() - t0) < 15:
                break
            h_new, c5_new = lse_opt(h_r, temp, mu, max_iter=400)
            if c5_new < c5 - 1e-10:
                h, c5 = h_new, c5_new
                h_best = h.copy()

    # Phase 2c: Sharpened starts
    for alpha in [0.3, 0.5, 0.7]:
        if budget_s - (time.time() - t0) < 30:
            break
        h_s = sharpen(h_best, alpha)
        for temp in [0.01, 0.005, 0.002]:
            if budget_s - (time.time() - t0) < 15:
                break
            h_new, c5_new = lse_opt(h_s, temp, mu, max_iter=350)
            if c5_new < c5 - 1e-10:
                h, c5 = h_new, c5_new
                h_best = h.copy()

    # Phase 3: Different resolutions
    for n_try in [120, 150, 200, 250, 320]:
        if budget_s - (time.time() - t0) < 80:
            break
        h_interp = interp_size(h_best, n_try)
        for temp in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            if budget_s - (time.time() - t0) < 25:
                break
            h_new, c5_new = lse_opt(h_interp, temp, mu, max_iter=500)
            if c5_new < c5 - 1e-10:
                h_best = h_new.copy()
                c5 = c5_new
                h_interp = h_new

    # Phase 4: Final fine-tuning at best resolution
    for temp in [0.0003, 0.0002, 0.0001, 0.00005, 0.00003]:
        if budget_s - (time.time() - t0) < 15:
            break
        h_new, c5_new = lse_opt(h_best, temp, mu, max_iter=600)
        if c5_new < c5 - 1e-11:
            h_best = h_new
            c5 = c5_new

    # Phase 5: Try with slightly different mu for final push
    if budget_s - (time.time() - t0) > 40:
        for mu2 in [500.0, 2000.0]:
            if budget_s - (time.time() - t0) < 20:
                break
            h_new, c5_new = lse_opt(h_best, 0.0001, mu2, max_iter=500)
            if c5_new < c5 - 1e-11:
                h_best = h_new
                c5 = c5_new

    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
