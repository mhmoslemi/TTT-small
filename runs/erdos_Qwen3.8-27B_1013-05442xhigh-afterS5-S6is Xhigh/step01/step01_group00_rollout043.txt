def adam_optimize(h0, temp_schedule, n_iter, lr=0.01, beta1=0.9, beta2=0.999):
    n = len(h0)
    target = n * 0.5
    h = h0.copy()
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for t in range(1, n_iter + 1):
        temp = temp_schedule[min(t - 1, len(temp_schedule) - 1)]
        corr = compute_corr(h)
        obj = logsumexp_val(corr, temp)
        grad = grad_logsumexp(h, temp)
        
        m_hat = beta1 * m_hat + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * grad**2
        m_hat_hat = m_hat / (1 - beta1**t)
        v_hat_hat = v_hat / (1 - beta2**t)
        
        h_new = h - lr * m_hat_hat / (np.sqrt(v_hat_hat) + 1e-8)
        h_new = project_feasible(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def slsqp_polish(h0, max_iter=300):

    v0 = np.append(h0.copy(), c5_cur + 0.01)

        {'type': 'ineq', 'fun': ineq_t},
        {'type': 'eq', 'fun': eq_t}

                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})

        return h_new, c5_new

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_corr(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_lse(h, temp):
    """Gradient of log-sum-exp of correlation values."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp((corr - m) / temp)
    w /= np.sum(w)
    A = np.correlate(w, 1.0 - h, mode='full')[:n][::-1]
    B = np.correlate(w, h, mode='full')[:n][::-1]
    return dx * (A - B)

def project_feasible(h, target):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target}."""
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h_proj < 1.0 - 1e-13
        else:
            mask = h_proj > 1e-13
        cnt = np.sum(mask)
        if cnt > 0:
            h_proj[mask] += diff / cnt
    return np.clip(h_proj, 0.0, 1.0)

def adam_lse(h0, temp, n_iter, lr=0.05, b1=0.9, b2=0.999):
    """Adam optimizer for log-sum-exp objective."""
    n = len(h0)
    target = n * 0.5
    h = h0.copy()
    mhat = np.zeros(n)
    vhat = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    eps = 1e-8
    
    for t in range(1, n_iter + 1):
        grad = grad_lse(h, temp)
        mhat = b1 * mhat + (1 - b1) * grad
        vhat = b2 * vhat + (1 - b2) * grad * grad
        mhat_c = mhat / (1 - b1 ** t)
        vhat_c = vhat / (1 - b2 ** t)
        h = project_feasible(h - lr * mhat_c / (np.sqrt(vhat_c) + eps), target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```

def slsqp_polish(h0, max_iter=300):
    n = len(h0)
    dx = 2.0 / n
    
    def obj(v):
        return v[-1]
    
    def ineq(v):
        h = v[:n]
        t = v[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - corr
    
    def eq(v):
        return np.sum(v[:n]) - n * 0.5
    
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.02)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq},
        {'type': 'eq', 'fun': eq}
    ]
    try:
        res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
        h_new = np.clip(res.x[:n], 0.0, 1.0)
        h_new = project_feasible(h_new, n * 0.5)

def compute_corr(h):

    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_lse(h, temp):
    n = len(h)
    dx = 2.0 / n

    m = np.max(corr)
    w = np.exp((corr - m) / temp)

    T2 = out_h[n-1:2*n-1][::-1]

def slsqp_obj(v):
