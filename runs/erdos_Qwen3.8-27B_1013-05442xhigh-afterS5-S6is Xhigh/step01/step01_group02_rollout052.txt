            if c < best_c5:
                best_h, best_c5 = h_try, c
        except Exception:
            pass

    # Phase 6: Random restart local search
    restart_deadline = time_left() - 30
    while time_left() > 35:
        scale = rng.choice([0.005, 0.01, 0.02, 0.03, 0.05])
        h_pert = best_h + rng.normal(0, scale, size=n)
        h_pert = project_feasible(h_pert, n)
        try:
            h_try = slsqp_step(h_pert, n, 5000, maxiter=2000, bounds=bounds, cons=cons)
            c = compute_c5(h_try, n)
            if c < best_c5 - 1e-12:
                best_h, best_c5 = h_try, c
                # Quick pair swap after improvement
                if time_left() > 50:
                    best_h, best_c5 = pair_swap_sweep(best_h, n, n_sweeps=10)
        except Exception:
            pass

    # Final polish
    if time_left() > 15:
        try:
            h_try = slsqp_step(best_h, n, 50000, maxiter=10000, bounds=bounds, cons=cons)
            c = compute_c5(h_try, n)
            if c < best_c5:
                best_h, best_c5 = h_try, c
        except Exception:
            pass

    # Final pair swap polish
    if time_left() > 20:
        best_h, best_c5 = pair_swap_sweep(best_h, n, n_sweeps=20)

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)

    return best_h, best_c5, n
