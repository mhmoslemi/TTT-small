def grad_step(h, n, alpha, eta, num_steps=100):
    """Gradient descent with projection."""
    best = h.copy()
    best_c5 = compute_c5(best, n)
    for step in range(num_steps):
        g = smax_grad(best, n, alpha)
        h_try = project_feasible(best - eta * g, n)
        c5 = compute_c5(h_try, n)
        if c5 < best_c5:
            best = h_try
            best_c5 = c5
        else:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                h_try = project_feasible(best - eta * factor * g, n)
                c5 = compute_c5(h_try, n)
                if c5 < best_c5:
                    best = h_try
                    best_c5 = c5
                    break
            else:
                break  # No improvement even with small steps
    return best, best_c5

def coordinate_descent(h, n, num_rounds=50, rng=None):
    """Coordinate descent: try adjusting 1 or 2 coordinates."""
    best = h.copy()
    best_c5 = compute_c5(best, n)
    if rng is None:
        rng = np.random.RandomState(0)
    
    for _ in range(num_rounds):
        improved = False
        # Single coordinate move
        i = rng.randint(n)
        for delta in [0.001, 0.005, 0.01, 0.02, -0.001, -0.005, -0.01, -0.02]:
            h_try = best.copy()
            h_try[i] = np.clip(h_try[i] + delta, 0, 1)
            h_try = project_feasible(h_try, n)
            c5 = compute_c5(h_try, n)
            if c5 < best_c5 - 1e-12:
                best = h_try
                best_c5 = c5
                improved = True
                break
        # Two coordinate move (preserve sum)
        if not improved:
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                for delta in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    for s in [1, -1]:
                        di = s * delta
                        hi = best[i] + di
                        hj = best[j] - di
                        if 0 <= hi <= 1 and 0 <= hj <= 1:
                            h_try = best.copy()
                            h_try[i] = hi
                            h_try[j] = hj
                            c5 = compute_c5(h_try, n)
                            if c5 < best_c5 - 1e-12:
                                best = h_try
                                best_c5 = c5
                                improved = True
                                break
                    if improved:
                        break
    return best, best_c5

import numpy as np
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-15:
                break
            h += diff * room / tr
        else:
            th = h.sum()
            if th < 1e-15:
                break
            h += diff * h / th
    return np.clip(h, 0.0, 1.0)

def _smax(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    s = sh.max()
    return m + (s + np.log(np.sum(np.exp(sh - s)))) / alpha

    return _smax(c5_vec(h, n), alpha)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-15:
                break
            h += diff * room / tr
        else:
            th = h.sum()
            if th < 1e-15:
                break
            h += diff * h / th
    return np.clip(h, 0.0, 1.0)

def _smax(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    s = sh.max()
    return m + (s + np.log(np.sum(np.exp(sh - s)))) / alpha

def smax_obj(h, n, alpha):
    return _smax(c5_vec(h, n), alpha)

def smax_grad(h, n, alpha):
    c = c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    s = sh.max()
    w = np.exp(sh - s)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for p in range(len(c)):
        k = p - (n - 1)
        wp = w[p]
        if wp < 1e-12:
            continue
        wk = wp * dx
        if k >= 0:
            idx1 = np.arange(k, n)
            grad[idx1] += wk * (1.0 - h[idx1 - k])
            idx2 = np.arange(0, n - k)
            grad[idx2] -= wk * h[idx2 + k]
        else:
            m2 = -k
            idx1 = np.arange(0, n - m2)
            grad[idx1] += wk * (1.0 - h[idx1 + m2])
            idx2 = np.arange(m2, n)
            grad[idx2] -= wk * h[idx2 - m2]
    return grad

def slsqp_opt(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smax_obj(x, n, alpha)
    def objg(x):
        return smax_grad(x, n, alpha)
    def eqf(x):
        return x.sum() - n / 2.0
    def eqj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': eqf, 'jac': eqj}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def grad_descent(h, n, alpha, eta, steps):
    best = h.copy()
    best_c5 = compute_c5(best, n)
    for _ in range(steps):
        g = smax_grad(best, n, alpha)
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_feasible(best - eta * f * g, n)
            c5 = compute_c5(h_try, n)
            if c5 < best_c5 - 1e-15:
                best = h_try
                best_c5 = c5
                improved = True
                break
        if not improved:
            break
    return best, best_c5

def coord_desc(h, n, rounds, rng):
    best = h.copy()
    best_c5 = compute_c5(best, n)
    deltas = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1]
    for _ in range(rounds):
        moved = False
        for _ in range(3):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in deltas:
                for s in [1, -1]:
                    di = s * d
                    hi = best[i] + di
                    hj = best[j] - di
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h_try = best.copy()
                        h_try[i] = hi
                        h_try[j] = hj
                        c5 = compute_c5(h_try, n)
                        if c5 < best_c5 - 1e-13:
                            best = h_try
                            best_c5 = c5
                            moved = True
                            break
                if moved:
                    break
            if moved:
                break
    return best, best_c5

        return budget_s - (time.time() - t0)

    while rem() > 80:
        h_try = slsqp_opt(best_h, n, 10000, 2000, 1e-15)

        if c5 < best_c5:
            best_h, best_c5 = h_try, c5

                break
            scale = rng.choice([0.01, 0.03, 0.08, 0.15, 0.3])
            h_p = project_feasible(best_h + rng.normal(0, scale, n), n)
            h_try = slsqp_opt(h_p, n, 5000, 1500, 1e-14)

    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
