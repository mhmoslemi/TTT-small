import numpy as np
from scipy.optimize import minimize
import time

def smooth_max_eval(corr, tau):
    c_max = np.max(corr)
    return float(c_max / tau + np.log(np.sum(np.exp((corr - c_max) / tau))) / tau)

def eval_c5_exact(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    return float(np.max(corr) * dx)

def fix_sum_and_clip(x, n):
    x = np.array(x, dtype=float)
    target = n / 2.0
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        x = x + diff / n
        x = np.clip(x, 0.0, 1.0)
    return x

def obj_c5_smooth(x, tau, n):
    x_clean = fix_sum_and_clip(x, n)
    corr = np.correlate(x_clean, 1.0 - x_clean, mode="full")
    return smooth_max_eval(corr, tau)

def obj_c5_plain(x, tau, n):
    # For methods that can handle the max without a smoothing path
    x_clean = fix_sum_and_clip(x, n)
    corr = np.correlate(x_clean, 1.0 - x_clean, mode="full")
    # Use a very small tau to ensure it's differentiable
    return smooth_max_eval(corr, 1e-2)
