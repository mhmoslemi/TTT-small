    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    obj = smooth_max(corr, beta) * dx
    
    # Softmax weights
    weights = np.exp(beta * (corr - np.max(corr)))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        # (1-h[j+lag]) for valid j
        if lag >= 0:
            # j ranges 0 to n-1-lag
            g1 = (1.0 - h[lag:n])  # h[j+lag] for j=0..n-1-lag
            grad[:n-lag] += w * g1
            # -h[j-lag] for valid j: j ranges lag to n-1
            g2 = -h[:n-lag]  # h[j-lag] for j=lag..n-1
            grad[lag:] += w * g2
        else:
            # lag < 0, let L = -lag
            L = -lag
            # (1-h[j+lag]) = (1-h[j-L]) for valid j: j ranges L to n-1
            g1 = (1.0 - h[:n-L])  # h[j-L] for j=L..n-1
            grad[L:] += w * g1
            # -h[j-lag] = -h[j+L] for valid j: j ranges 0 to n-1-L
            g2 = -h[L:n]  # h[j+L] for j=0..n-L-1
            grad[:n-L] += w * g2
    
    grad *= dx
    return obj, grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def smooth_max_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
