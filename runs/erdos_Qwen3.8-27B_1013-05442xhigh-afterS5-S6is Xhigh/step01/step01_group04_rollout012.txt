from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(60):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-13:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_fn(x, dx):

def con_fn(x, target):

def slsqp_polish(h_start, n, dx, target_sum, max_iter=300):

            constraints=[{'type': 'eq', 'fun': con_fn, 'args': (target_sum,)}],
            options={'maxiter': max_iter, 'ftol': 1e-15}
        )
        return project_feasible(res.x, n)

        return h_start
