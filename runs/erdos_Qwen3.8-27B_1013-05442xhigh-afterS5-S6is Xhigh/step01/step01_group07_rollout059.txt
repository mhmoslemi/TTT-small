from scipy.optimize import minimize
from functools import partial

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    h = h.copy()
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(x, t, dx):
    corr = np.correlate(x, 1 - x, mode="full")
    mc = np.max(corr)
    shifted = t * (corr - mc)
    return (mc + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx

def sum_constraint(x, target_sum):
    return np.sum(x) - target_sum

def slsqp_refine(h, dx, target_sum, t, maxiter):
    n = len(h)
    try:
        fun = partial(smooth_max_obj, t=t, dx=dx)
        con = partial(sum_constraint, target_sum=target_sum)
        res = minimize(
            fun, h.copy(), method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': con, 'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return fix_sum(res.x, target_sum)
    except Exception:
        return h
