import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj_grad(h, tau, mu=0.0):
    """Compute smooth max of correlation and its gradient, plus optional sum penalty."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    num_corr = len(corr)
    
    x = tau * corr
    x_max = np.max(x)
    exp_shifted = np.exp(x - x_max)
    sum_exp = np.sum(exp_shifted)
    p = exp_shifted / sum_exp
    obj = (x_max + np.log(sum_exp)) / tau
    
    grad = np.zeros(n)
    for j in range(num_corr):
        if p[j] < 1e-12:
            continue
        k = j - (n - 1)
        # Term 1: (1-h[i-k]) for 0 <= i-k < n
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            grad[idx] += p[j] * (1.0 - h[idx - k])
        # Term 2: -h[i+k] for 0 <= i+k < n
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            idx2 = np.arange(lo2, hi2)
            grad[idx2] -= p[j] * h[idx2 + k]
    
    if mu > 0:
        target = n / 2.0
        s = np.sum(h) - target
        obj += 0.5 * mu * s * s
        grad += mu * s
    
    return obj, grad

def project_sum(h, target):
    """Project h onto {0<=h<=1, sum(h)=target} using bisection on water level."""
    n = len(h)
    # Find alpha such that sum(clip(h - alpha, 0, 1)) = target
    lo, hi = h.min() - target, h.max() + target
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = projected.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fine adjustment
    diff = target - projected.sum()
    mask = (projected > 0) & (projected < 1)
    if mask.sum() > 0:
        projected[mask] += diff / mask.sum()
        projected = np.clip(projected, 0, 1)
    return projected

def lgb_optimize(h0, tau, mu, maxiter=500, ftol=1e-14):
    n = len(h0)
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def f(x):
        val, g = smooth_max_obj_grad(x, tau, mu)
        return val, g
    
    res = minimize(f, h0, jac=True, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
    return res.x

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj_grad(h, tau, mu):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    x_max = x.max()
    exp_s = np.exp(x - x_max)
    sum_exp = exp_s.sum()
    p = exp_s / sum_exp
    obj = (x_max + np.log(sum_exp)) / tau
    grad = np.zeros(n)
    for j in range(len(corr)):
        if p[j] < 1e-12:
            continue
        k = j - (n - 1)
        lo1, hi1 = max(0, k), min(n, n + k)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            grad[idx] += p[j] * (1.0 - h[idx - k])
        lo2, hi2 = max(0, -k), min(n, n - k)
        if lo2 < hi2:
            idx2 = np.arange(lo2, hi2)
            grad[idx2] -= p[j] * h[idx2 + k]
    if mu > 0:
        s = h.sum() - n / 2.0
        obj += 0.5 * mu * s * s
        grad += mu * s
    return obj, grad

def project_sum(h, target):
    lo = h.min() - target
    hi = h.max() + target
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0, 1)

    mask = (proj > 0) & (proj < 1)
    if mask.any():
        proj[mask] += diff / mask.sum()
        proj = np.clip(proj, 0, 1)
    return proj

                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 40})
    return res.x

    h = h.copy()
    for _ in range(steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        j_star = np.argmax(corr)
        k = j_star - (n - 1)
        grad = np.zeros(n)
        lo1, hi1 = max(0, k), min(n, n + k)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            grad[idx] += (1.0 - h[idx - k])
        lo2, hi2 = max(0, -k), min(n, n - k)
        if lo2 < hi2:
            idx2 = np.arange(lo2, hi2)
            grad[idx2] -= h[idx2 + k]

        lr = 0.1
        cur_val = corr[j_star]
        for _ in range(20):

            corr_t = np.correlate(h_trial, 1.0 - h_trial, mode="full")
            if corr_t[j_star] < cur_val:
                h = h_trial
                cur_val = corr_t[j_star]
                break
            lr *= 0.5
        else:
            break
    return h

    hi = h.max() + target
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0, 1)
    diff = target - proj.sum()
    mask = (proj > 0) & (proj < 1)
    if mask.any():
        proj[mask] += diff / mask.sum()
        proj = np.clip(proj, 0, 1)
    return proj

def lgb_opt(h0, tau, mu, maxiter=500):
    n = len(h0)
    bnds = Bounds(np.zeros(n), np.ones(n))
    def f(x):
        return smooth_max_obj_grad(x, tau, mu)
    res = minimize(f, h0, jac=True, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 40})
    return res.x

def active_set(h, steps=100, topk=1):
    n = len(h)
    h = h.copy()
    for _ in range(steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[-topk:][::-1]
        grad = np.zeros(n)
        for j in top_idx:
            k = j - (n - 1)
            lo1, hi1 = max(0, k), min(n, n + k)
            if lo1 < hi1:
                idx = np.arange(lo1, hi1)
                grad[idx] += (1.0 - h[idx - k]) / topk
            lo2, hi2 = max(0, -k), min(n, n - k)
            if lo2 < hi2:
                idx2 = np.arange(lo2, hi2)
                grad[idx2] -= h[idx2 + k] / topk
        lr = 0.1
        cur_val = np.max(corr)
        improved = False
        for _ in range(20):
            h_trial = project_sum(h - lr * grad, n / 2.0)
            corr_t = np.correlate(h_trial, 1.0 - h_trial, mode="full")
            if np.max(corr_t) < cur_val:
                h = h_trial
                cur_val = np.max(corr_t)
                improved = True
                break
            lr *= 0.5
        if not improved:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)

    starts = []
    if 'initial_h_values' in globals() and initial_h_values is not None:
        starts.append(np.array(initial_h_values, dtype=float).copy())
    if not starts:
        starts.append(np.full(120, 0.5))

    n0 = len(starts[0])
    best_h = starts[0].copy()
    best_c5 = evaluate_c5(best_h)

    x = np.linspace(0, 2, n0, endpoint=False)
    h_sin = 0.5 + 0.25 * np.sin(np.pi * x)
    starts.append(project_sum(np.clip(h_sin, 0, 1), n0 / 2.0))
    h_block = np.zeros(n0)
    h_block[:n0 // 2] = 1.0
    starts.append(h_block.copy())
    h_cos2 = 0.5 + 0.15 * np.cos(2 * np.pi * x) + 0.1 * np.sin(np.pi * x)
    starts.append(project_sum(np.clip(h_cos2, 0, 1), n0 / 2.0))

    tau_mu_sched = [(10, 200), (50, 100), (200, 50), (500, 20), (1000, 10), (3000, 5), (5000, 2)]
    for h_s in starts:
        if rem() < 30:
            break
        h_cur = h_s.copy()
        for tau, mu in tau_mu_sched:
            if rem() < 20:
                break
            mi = 300 if rem() > 400 else 100
            h_new = lgb_opt(h_cur, tau, mu, maxiter=mi)
            h_new = project_sum(h_new, len(h_new) / 2.0)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
            h_cur = h_new
        if rem() > 15:
            h_new = active_set(h_cur, steps=60, topk=1)
            h_new = project_sum(h_new, len(h_new) / 2.0)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
            h_new2 = active_set(h_new, steps=60, topk=3)
            h_new2 = project_sum(h_new2, len(h_new2) / 2.0)
            c5 = evaluate_c5(h_new2)
            if c5 < best_c5:
                best_h, best_c5 = h_new2, c5

    for tn in [60, 80, 100, 120, 150, 200]:
        if rem() < 30:
            break
        h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, len(best_h)), best_h)
        h_t = project_sum(h_t, tn / 2.0)
        for tau, mu in [(100, 50), (500, 20), (2000, 5), (5000, 1), (10000, 0.5)]:
            if rem() < 15:
                break
            h_new = lgb_opt(h_t, tau, mu, maxiter=200)
            h_new = project_sum(h_new, tn / 2.0)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
            h_t = h_new
        if rem() > 15:
            h_new = active_set(h_t, steps=50, topk=1)
            h_new = project_sum(h_new, tn / 2.0)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
            h_new2 = active_set(h_new, steps=50, topk=3)
            h_new2 = project_sum(h_new2, tn / 2.0)
            c5 = evaluate_c5(h_new2)
            if c5 < best_c5:
                best_h, best_c5 = h_new2, c5

    hn = len(best_h)
    for _ in range(20):
        if rem() < 15:
            break
        scale = 0.005 + 0.04 * np.random.random()
        hp = np.clip(best_h + np.random.normal(0, scale, hn), 0, 1)
        hp = project_sum(hp, hn / 2.0)
        for tau, mu in [(300, 10), (1000, 3)]:
            if rem() < 10:
                break
            h_new = lgb_opt(hp, tau, mu, maxiter=150)
            h_new = project_sum(h_new, hn / 2.0)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
            hp = h_new
        if rem() > 8:
            h_new = active_set(hp, steps=40, topk=2)
            h_new = project_sum(h_new, hn / 2.0)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5

    if rem() > 10:
        h_new = active_set(best_h, steps=150, topk=1)
        h_new = project_sum(h_new, len(h_new) / 2.0)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5
        h_new = active_set(best_h, steps=100, topk=2)
        h_new = project_sum(h_new, len(h_new) / 2.0)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
