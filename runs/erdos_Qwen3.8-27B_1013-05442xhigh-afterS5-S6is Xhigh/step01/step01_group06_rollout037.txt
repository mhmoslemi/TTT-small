def compute_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    total_w = np.sum(w)
    w /= total_w
    
    grad = np.zeros(n)
    indices = np.arange(n)
    
    # For positive lags k = 0, 1, ..., n-1 (index n-1+k in corr)
    for k in range(n):
        wk = w[n-1+k]
        if wk < 1e-12:
            continue
        # d(corr_pos[k])/dh[m]:
        #   (1-h[m-k]) for m >= k
        #   -h[m+k] for m <= n-1-k
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            # m >= k: (1-h[m-k])
            grad[k:] += wk * (1.0 - h[:n-k])
            # m <= n-1-k: -h[m+k]
            grad[:n-k] += wk * (-h[k:])
    
    # For negative lags k = 1, ..., n-1 (index n-1-k in corr)

        #   (1-h[m+k]) for m <= n-1-k
        #   -h[m-k] for m >= k
        grad[:n-k] += wk * (1.0 - h[k:])
        grad[k:] += wk * (-h[:n-k])

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    
    grad = np.zeros(n)
    # Positive lags k = 0, ..., n-1 at indices n-1, n, ..., 2n-2
    for k in range(n):
        wk = w[n-1+k]
        if wk < 1e-12:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            grad[k:] += wk * (1.0 - h[:n-k])
            grad[:n-k] += wk * (-h[k:])
    
    # Negative lags k = 1, ..., n-1 at indices n-2, n-3, ..., 0
    for k in range(1, n):
        wk = w[n-1-k]
        if wk < 1e-12:
            continue
        grad[:n-k] += wk * (1.0 - h[k:])
        grad[k:] += wk * (-h[:n-k])
    
    return grad * (2.0 / n)

def optimize_gradient(h, target, deadline, rng, alpha_init=100, alpha_max=500):
    n = len(h)
    best_h = project_simplex(h.copy(), target)
    best_c5 = compute_c5(best_h)
    lr = 0.01
    alpha = alpha_init
    no_improve = 0
    
    while time.time() < deadline:
        grad = compute_grad(best_h, alpha)
        # Line search
        for _ in range(10):
            h_new = project_simplex(best_h - lr * grad, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                no_improve = 0
                lr = min(lr * 1.1, 0.1)
                break
            lr *= 0.5
        else:
            no_improve += 1
            if no_improve > 20:
                # Restart with perturbation
                alpha = min(alpha * 1.5, alpha_max)
                h_pert = best_h + rng.randn(n) * 0.05
                best_h = project_simplex(h_pert, target)
                best_c5 = compute_c5(best_h)
                lr = 0.01
                no_improve = 0
                if alpha > alpha_max:
                    break
    
    return best_h, best_c5

def sa_binary(h_binary, n, deadline, rng):
    """Simulated annealing on binary sequences."""
    h = h_binary.copy().astype(float)
    curr_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = curr_c5
    
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    
    T = 5.0
    T_min = 0.01
    cooling = 0.9995
    
    while time.time() < deadline:
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        if a == b:
            continue
        
        # Perform swap
        h[a], h[b] = h[b], h[a]
        new_c5 = compute_c5(h)
        
        if new_c5 <= curr_c5 or rng.rand() < np.exp(-(new_c5 - curr_c5) / T):
            curr_c5 = new_c5
            ones[ones.index(a)] = b  # a was 1, now 0
            zeros[zeros.index(b)] = a  # b was 0, now 1
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Revert
            h[a], h[b] = h[b], h[a]
        
        T = max(T * cooling, T_min)
    
    return best_h, best_c5

import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_simplex(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-12:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            grad[k:] += wk * (1.0 - h[:n - k])
            grad[:n - k] += wk * (-h[k:])
    for k in range(1, n):
        wk = w[n - 1 - k]
        if wk < 1e-12:
            continue
        grad[:n - k] += wk * (1.0 - h[k:])
        grad[k:] += wk * (-h[:n - k])
    return grad * (2.0 / n)
