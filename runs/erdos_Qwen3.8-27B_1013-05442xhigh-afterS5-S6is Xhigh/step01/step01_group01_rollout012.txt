def minimax_subgradient(h, p, alpha, eta, target_sum):
    """One step of the multiplicative weights subgradient method."""
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    
    # Compute f_k(h) for all lags
    corr = np.correlate(h, 1.0 - h, mode="full")  # 2n-1 values
    f = corr * dx  # f_k = (1/n) * ∑ h[i+k](1-h[i]) ... wait, dx = 2/n
    
    # Actually f_k = corr[k] * dx where dx = 2/n
    # C5 = max(f)
    
    # Compute gradient: g[j] = sum_k p[k] * df_k/dh[j]
    g = np.zeros(n)
    for k_idx in range(n_lags):
        if p[k_idx] < 1e-10:
            continue
        lag = k_idx - (n - 1)
        for j in range(n):
            val = 0.0
            if 0 <= j - lag < n:
                val += (1.0 - h[j - lag])
            if 0 <= j + lag < n:
                val -= h[j + lag]
            g[j] += p[k_idx] * val
    g *= dx
    
    # Take step
    h_new = h - alpha * g
    
    # Project to [0,1] and sum constraint
    h_new = project_h(h_new, target_sum)
    
    # Update p
    f_scaled = eta * f
    f_scaled -= np.max(f_scaled)
    p_new = p * np.exp(f_scaled)
    p_new /= p_new.sum()
    
    return h_new, p_new

            h0 += np.random.randn(n_points) * 0.02
        else:
            h0 += np.random.randn(n_points) * (0.04 + 0.08 * np.exp(-iteration / 120.0))

        h0 = project_h(h0, target_sum)

        # Optimize
        try:
            beta = min(400.0, 50.0 + iteration * 2.5)
            pen = 300.0
            x_new = do_lbfgsb(h0, bounds, beta, pen, target_sum, dx, 200)
            h_new = project_h(x_new, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

        # Coordinate descent every 6 iterations
        if iteration % 6 == 0 and rem() > 50:
            best_h, best_c5 = coord_descent(best_h, target_sum, n_points, 8, 2)

    # Phase 3: SLSQP polisher (handles constraints natively, may find better geometry)
    if rem() > 20:
        try:
            def slesqp_obj(h):
                corr = np.correlate(h, 1.0 - h, mode="full")
                return smooth_max(corr, 200.0) * dx

            def con_fun(x):
                return np.sum(x) - target_sum

            def con_jac(x):
                return np.ones(n_points)

            res = minimize(slesqp_obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                          options={'maxiter': 1000, 'ftol': 1e-14})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 4: High-precision L-BFGS-B
    if rem() > 10:
        try:
            x_new = do_lbfgsb(best_h, bounds, 500, 500, target_sum, dx, 2000, 1e-15)
            h_new = project_h(x_new, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 5: Final fine coordinate descent
    if rem() > 5:
        for _ in range(5):
            improved = False
            for i in range(n_points):
                for v in np.linspace(0, 1, 11):
                    ht = best_h.copy()
                    ht[i] = v
                    ht = project_h(ht, target_sum)
                    ct = compute_c5(ht)
                    if ct < best_c5 - 1e-13:
                        best_h = ht.copy()
                        best_c5 = ct
                        improved = True
            if not improved:
                break

    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
