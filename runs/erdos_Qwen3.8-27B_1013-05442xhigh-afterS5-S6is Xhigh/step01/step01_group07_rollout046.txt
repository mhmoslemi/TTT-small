import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_simplex(v, S):
    n = len(v)
    lo, hi = v.min() - 0.5, v.max() + 0.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def get_smooth_grad(h, t, n):
    corrs = np.correlate(h, 1 - h, mode="full")
    mc = corrs.max()
    exps = np.exp(t * (corrs - mc))
    p = exps / exps.sum()
    grad = np.zeros(n)
    m = n - 1
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lag = k - m
        if lag == 0:
            grad += pk * (1.0 - 2.0 * h)
        elif lag > 0:
            j = np.arange(lag, n - lag)
            grad[j] += pk * ((1.0 - h[j - lag]) - h[j + lag])
        else:
            l = -lag
            j = np.arange(l, n - l)
            grad[j] += pk * ((1.0 - h[j + l]) - h[j - l])
    return grad

def get_topk_grad(h, k, n):
    corrs = np.correlate(h, 1 - h, mode="full")
    top_idx = np.argsort(corrs)[-k:]
    obj = corrs[top_idx].mean()
    grad = np.zeros(n)
    m = n - 1
    for ki in top_idx:
        lag = ki - m
        if lag == 0:
            grad += (1.0 / k) * (1.0 - 2.0 * h)
        elif lag > 0:
            j = np.arange(lag, n - lag)
            grad[j] += (1.0 / k) * ((1.0 - h[j - lag]) - h[j + lag])
        else:
            l = -lag
            j = np.arange(l, n - l)
            grad[j] += (1.0 / k) * ((1.0 - h[j + l]) - h[j - l])
    return obj, grad

def pgd_loop(h, dx, n, t, lr, max_iter, deadline):
    S = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = get_smooth_grad(h, t, n)
        improved = False
        for s in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            h_try = project_onto_simplex(h - s * grad, S)
            c5_try = compute_c5(h_try, dx)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                best_h = h_try.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            lr *= 0.8
            if stall > 50:
                break
    return best_h, best_c5, lr

def topk_pgd(h, dx, n, k, lr, max_iter, deadline):
    S = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        _, grad = get_topk_grad(h, k, n)
        improved = False
        for s in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            h_try = project_onto_simplex(h - s * grad, S)
            c5_try = compute_c5(h_try, dx)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                best_h = h_try.copy()
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            lr *= 0.7
            if stall > 40:
                break
    return best_h, best_c5

def pair_swap_search(h, dx, n, deadline, max_rounds=500):
    S = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(123)
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        n_try = rng.randint(50, 200)
        for _ in range(n_try):
            i, j = rng.choice(n, 2, replace=False)
            if h[i] <= 0 or h[j] >= 1:
                continue
            deltas = [0.01, 0.05, 0.1, 0.15, 0.2]
            for d in deltas:
                if h[i] + d > 1 or h[j] - d < 0:
                    continue
                h_try = h.copy()
                h_try[i] += d
                h_try[j] -= d
                c5_try = compute_c5(h_try, dx)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    improved = True
                    break
            if improved:
                break
    return best_h, best_c5

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n + 1)[:-1]
    x_new = np.linspace(0, 2, new_n + 1)[:-1]
    h_new = np.interp(x_new, x_old, h)
    return h_new

def downsample(h, new_n):
    return upsample(h, new_n)
