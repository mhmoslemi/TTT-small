```python
def subgradient_step(h, eta, n):
    # Compute correlations
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    s_star = np.argmax(corr)
    lag = s_star - (n - 1)
    
    # Compute subgradient
    grad = np.zeros(n)
    for j in range(n):
        # Term from h[j] * (1-h[j+lag]): derivative is (1-h[j+lag])
        if 0 <= j + lag < n:
            grad[j] += (1.0 - h[j + lag])
        # Term from h[j-lag] * (1-h[j]): derivative is -h[j-lag]
        if 0 <= j - lag < n:
            grad[j] -= h[j - lag]
    
    # Projected gradient step
    h_new = h - eta * grad
    # Project to feasible set
    h_new = np.clip(h_new, 0, 1)
    # Adjust sum
    target = n / 2
    diff = target - h_new.sum()
    if abs(diff) > 1e-10:
        # Add diff/n to non-bound elements
        interior = (h_new > 0.01) & (h_new < 0.99)
        if interior.sum() > 0:
            h_new[interior] += diff / interior.sum()
            h_new = np.clip(h_new, 0, 1)
    return h_new
```

    n = len(h)
    grad = np.zeros(n)
    # First term: (1-h[j+s]) for j+s in [0,n)
    idx1 = np.arange(n) + s
    valid1 = (idx1 >= 0) & (idx1 < n)
    grad[valid1] += (1 - h[idx1[valid1]])
    # Second term: -h[j-s] for j-s in [0,n)
    idx2 = np.arange(n) - s
    valid2 = (idx2 >= 0) & (idx2 < n)
    grad[valid2] -= h[idx2[valid2]]
    return grad

Actually, I just realized something important. The smooth_max with beta=100-200 approximates the max well, but the gradient is concentrated on the active lags. What if I use a "soft top-k" instead?

soft_top_k(x, k, beta) = smooth_max of the top k elements

This can be computed as: sort x, take top k, apply smooth_max to them. But this is non-smooth at the boundary. A better approach: use the Lagrangian of the top-k problem.

Actually, let me just use a very high beta (500-1000) for the smooth max. With n=70, the correlation vector has 2n-1 = 139 elements. The smooth max with beta=500 will be very close to the true max (within 1/beta of it), and the gradient will be concentrated on the active lags.

import numpy as np
from scipy.optimize import minimize, Bounds, differential_evolution
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_corrs(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def smooth_max(x, beta):
    m = np.max(x)
    shifted = beta * (x - m)
    shifted = np.clip(shifted, -50, 50)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        interior = (h > 0.02) & (h < 0.98)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
            h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target_sum) > 1e-10:
            h = np.clip(h + (target_sum - h.sum()) / len(h), 0.0, 1.0)
    return h


def subgradient_of_max_corr(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    s_star = np.argmax(corr)
    lag = s_star - (n - 1)
    grads = np.zeros(n)
    idx_fwd = np.arange(n) + lag
    valid_fwd = (idx_fwd >= 0) & (idx_fwd < n)
    grads[valid_fwd] += (1.0 - h[idx_fwd[valid_fwd]])
    idx_bwd = np.arange(n) - lag
    valid_bwd = (idx_bwd >= 0) & (idx_bwd < n)
    grads[valid_bwd] -= h[idx_bwd[valid_bwd]]
    return grads, s_star


def multi_lag_subgradient(h, top_k=3):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_indices = np.argsort(corr)[-top_k:]
    total_grad = np.zeros(n)
    for k_idx in top_indices:
        lag = k_idx - (n - 1)
        idx_fwd = np.arange(n) + lag
        valid_fwd = (idx_fwd >= 0) & (idx_fwd < n)
        total_grad[valid_fwd] += (1.0 - h[idx_fwd[valid_fwd]])
        idx_bwd = np.arange(n) - lag
        valid_bwd = (idx_bwd >= 0) & (idx_bwd < n)
        total_grad[valid_bwd] -= h[idx_bwd[valid_bwd]]
    return total_grad / top_k


def slqp_optimize(h, n, target, maxiter=3000, ftol=1e-14, beta=200.0):
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
        return smooth_max(corr, beta)

    def con_fun(x):
        return x.sum() - target

    def con_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def lbfgs_optimize(h, n, target, maxiter=10000, penalty=500.0, beta=150.0):
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)
        sm = smooth_max(corr, beta)
        pen = penalty * (x.sum() - target) ** 2
        return sm + pen

    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9,
                                'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h


def subgradient_refine(h, n, target, max_iters=20000, eta0=0.003, decay=0.9997):
    eta = eta0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if it % 500 == 0:
            elapsed = time.time() - subgradient_refine._start
            if elapsed > subgradient_refine._budget_left:
                break
        top_k = 1 if it > 5000 else 2
        grad = multi_lag_subgradient(h, top_k)
        h = project_feasible(h - eta * grad, target)
        eta = max(eta * decay, 0.00005)
        if it % 200 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5


def structured_perturb(h, n, mode, rng):
    h0 = h.copy()
    if mode == 0:
        # Wavy perturbation
        freq = rng.randint(2, 8)
        phase = rng.uniform(0, 2 * np.pi)
        h0 += 0.1 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    elif mode == 1:
        # Block swap
        blen = max(3, n // 5)
        pos1 = rng.randint(0, n - blen)
        pos2 = rng.randint(0, n - blen)
        if abs(pos1 - pos2) > blen:
            h0[pos1:pos1 + blen], h0[pos2:pos2 + blen] = h0[pos2:pos2 + blen].copy(), h0[pos1:pos1 + blen].copy()
    elif mode == 2:
        # Antisymmetric perturbation
        anti = np.zeros(n)
        for j in range(n // 2):
            anti[j] = rng.uniform(-0.05, 0.05)
            anti[n - 1 - j] = -anti[j]
        h0 += anti
    elif mode == 3:
        # Gaussian bump
        center = rng.randint(n // 4, 3 * n // 4)
        width = max(2, n // 10)
        x = np.arange(n)
        bump = np.exp(-((x - center) ** 2) / (2 * width ** 2))
        h0 += 0.1 * (bump - bump.mean())
    else:
        # Random sparse large change
        n_change = rng.randint(3, max(4, n // 8))
        idx = rng.choice(n, n_change, replace=False)
        h0[idx] += rng.uniform(-0.2, 0.2, n_change)
    return h0


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 70
        h_init = np.full(n_init, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: L-BFGS-B on initial solution
    n = n_init
    target = n / 2.0
    h_work = project_feasible(h_init, target)

    if time_left() > 100:
        h_work = lbfgs_optimize(h_work, n, target, maxiter=8000, penalty=300.0, beta=120.0)
        c5 = compute_c5(h_work)
        if c5 < best_c5:
            best_h, best_c5 = h_work.copy(), c5

    # Phase 2: SLSQP polish
    if time_left() > 60:
        h_work = slqp_optimize(h_work, n, target, maxiter=3000, ftol=1e-14, beta=200.0)
        c5 = compute_c5(h_work)
        if c5 < best_c5:
            best_h, best_c5 = h_work.copy(), c5

    # Phase 3: Subgradient refinement
    if time_left() > 60:
        subgradient_refine._start = time.time()
        subgradient_refine._budget_left = time_left() - 30
        h_work, c5_sg = subgradient_refine(h_work, n, target, max_iters=30000, eta0=0.004, decay=0.9998)
        if c5_sg < best_c5:
            best_h, best_c5 = h_work.copy(), c5_sg

    # Phase 4: SLSQP after subgradient
    if time_left() > 40:
        h_work = slqp_optimize(h_work, n, target, maxiter=2000, ftol=1e-15, beta=300.0)
        c5 = compute_c5(h_work)
        if c5 < best_c5:
            best_h, best_c5 = h_work.copy(), c5

    # Phase 5: Diverse restarts with structured perturbations
    attempt = 0
    while time_left() > 40:
        attempt += 1
        mode = attempt % 5
        h0 = structured_perturb(best_h, n, mode, rng)
        target = n / 2.0
        h0 = project_feasible(h0, target)

        # Quick L-BFGS-B
        h0 = lbfgs_optimize(h0, n, target, maxiter=3000, penalty=300.0, beta=100.0)
        c5 = compute_c5(h0)

        if c5 < best_c5:
            best_h, best_c5 = h0.copy(), c5
            # Immediate SLSQP polish on improvement
            if time_left() > 30:
                h0 = slqp_optimize(h0, n, target, maxiter=1500, ftol=1e-14, beta=250.0)
                c5 = compute_c5(h0)
                if c5 < best_c5:
                    best_h, best_c5 = h0.copy(), c5
        elif attempt % 3 == 0:
            # Even if no improvement, try SLSQP on perturbed point (might find different basin)
            if time_left() > 30:
                h0 = slqp_optimize(h0, n, target, maxiter=1000, ftol=1e-13, beta=150.0)
                c5 = compute_c5(h0)
                if c5 < best_c5:
                    best_h, best_c5 = h0.copy(), c5

    # Phase 6: Try different resolutions
    for n_try in [n_init // 2, n_init * 3 // 2, n_init + 14]:
        if time_left() < 60:
            break
        if n_try < 10 or n_try > 200:
            continue
        target_try = n_try / 2.0
        # Interpolate best solution to new resolution
        x_old = np.linspace(0.0, 2.0, len(best_h))
        x_new = np.linspace(0.0, 2.0, n_try)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_feasible(h_try, target_try)

        h_try = lbfgs_optimize(h_try, n_try, target_try, maxiter=5000, penalty=400.0, beta=130.0)
        if time_left() > 40:
            h_try = slqp_optimize(h_try, n_try, target_try, maxiter=2000, ftol=1e-14, beta=250.0)

        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_h, best_c5 = h_try.copy(), c5_try

    # Phase 7: Final high-precision polish
    if time_left() > 20:
        n_final = len(best_h)
        target_final = n_final / 2.0
        best_h = slqp_optimize(best_h, n_final, target_final, maxiter=5000, ftol=1e-16, beta=500.0)
        final_c5 = compute_c5(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5

    # Ensure feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
