    dx = 2.0 / n

    return float(np.max(corr) * dx), corr

    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-12 and s > 0:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)

    n = len(h)
    grad = np.zeros(n)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-10:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def logsumexp_obj_grad(h, beta):
    """Objective and gradient of smooth max (log-sum-exp) approximation."""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Objective: (1/beta) * log(sum exp(beta * corr * dx))
    scaled = beta * corr * dx
    m = np.max(scaled)
    obj = (m + np.log(np.sum(np.exp(scaled - m)))) / beta
    # Gradient
    w = np.exp(scaled - m)
    w /= np.sum(w)
    grad = np.zeros(N)
    for k in range(1, 2 * N - 1):
        r = k - (N - 1)
        wi = w[k] * dx
        if wi < 1e-15:
            continue
        lo1 = max(0, -r)
        hi1 = min(N, N - r)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + r:hi1 + r])
        lo2 = max(0, r)
        hi2 = min(N, N + r)
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - r:hi2 - r]
    return obj, grad

    best_c5, _ = compute_c5(best_h)
    
    def time_left():
        return budget_s - (time.time() - start_time)

    betas = [20, 50, 100, 200, 500]

                return logsumexp_obj_grad(x, beta)[0]
            def obj_grad(x):
                return logsumexp_obj_grad(x, beta)[1]
            
            cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                     'jac': lambda x: np.ones(N)}]
            
            res = minimize(obj, best_h, jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
