        if s > target:
            lo = t
        else:
            hi = t
    t = (lo + hi) / 2.0
    return np.clip(h - t, 0.0, 1.0)


def compute_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    log_w = beta * (corr - mc)
    log_w -= np.max(log_w)
    w = np.exp(log_w)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
        w_sum = 1.0
    w /= w_sum
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        s = idx - (n - 1)
        if s >= 0:
            grad[s:] += wi * one_minus_h[:n - s]
            grad[:n - s] -= wi * h[s:]
        else:
            sp = -s
            grad[:n - sp] += wi * one_minus_h[sp:]
            grad[sp:] -= wi * h[:n - sp]
    return dx * grad


def lbfgs_optimize(h, target, n, beta, lam, maxiter=500, ftol=1e-16):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max_val(corr, beta) * (2.0 / n) + lam * (x.sum() - target) ** 2

    def grad(x):
        g = compute_gradient(x, beta)
        g += 2.0 * lam * (x.sum() - target) * np.ones(n)
        return g

    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        return project_exact(r.x, target)
    except Exception:
        return h


def proximal_grad(h, target, n_iters, beta_start, beta_end, step_base):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iters):
        frac = it / max(1, n_iters - 1)
        beta = beta_start * (beta_end / max(1, beta_start)) ** frac
        step = step_base * (1.0 - 0.7 * frac)
        grad = compute_gradient(h, beta)
        h_new = project_exact(h - step * grad, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c5


def gen_start(n, mode, seed):
    rng = np.random.RandomState(seed)
    target = n / 2.0
    if mode == 0:
        h = np.full(n, 0.5)
    elif mode == 1:
        x = np.arange(n) * 2.0 / n
        a = rng.uniform(0.15, 0.4)
        w = rng.uniform(1.0, 6.0) * np.pi
        p = rng.uniform(0, 2 * np.pi)
        h = 0.5 + a * np.sin(w * x + p)
    elif mode == 2:
        h = rng.uniform(0.0, 1.0, n)
    elif mode == 3:
        h = np.zeros(n)
        nb = rng.randint(3, max(4, n // 8))
        for b in range(nb):
            s, e = b * n // nb, (b + 1) * n // nb
            h[s:e] = rng.uniform(0.2, 0.8)
    elif mode == 4:
        x = np.arange(n) * 2.0 / n
        h = 0.5 + 0.25 * np.sin(3 * np.pi * x) + 0.15 * np.sin(7 * np.pi * x + 0.5)
    elif mode == 5:
        h = 0.5 + 0.3 * rng.randn(n)
    else:
        h = 0.5 + 0.2 * np.sin(5.0 * np.pi * np.arange(n) / n) + 0.1 * np.cos(11.0 * np.pi * np.arange(n) / n)
    return project_exact(h, target)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        n0 = len(initial_h_values)
        h0 = project_exact(np.array(initial_h_values, dtype=float).copy(), n0 / 2.0)
    else:
        n0 = 70
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    n_list = sorted(set([n0, 70, 80, 90, 100, 120, 140, 160]))

    for n in n_list:
        if tl() < 45:
            break
        target = n / 2.0

        # Build starting points
        starts = []
        if n == n0:
            starts.append(h0.copy())
        else:
            x_old = np.linspace(0, 2, n0, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            starts.append(project_exact(h_interp, target))
        for mode in range(7):
            starts.append(gen_start(n, mode, seed + n * 7 + mode))

        # Optimize each start
        for si, h_start in enumerate(starts):
            if tl() < 20:
                break
            h_cur = h_start.copy()

            # Stage 1: L-BFGS with moderate beta
            for beta in [50, 150, 400]:
                if tl() < 10:
                    break
                h_cur = lbfgs_optimize(h_cur, target, n, beta, 30.0, maxiter=400)
                c5 = compute_c5(h_cur)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_cur.copy()

            # Stage 2: Proximal gradient
            if tl() > 10:
                h_cur, c5 = proximal_grad(h_cur, target, 300, 100, 2000, 0.008)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_cur.copy()

            # Stage 3: L-BFGS with high beta
            for beta in [800, 2000]:
                if tl() < 8:
                    break
                h_cur = lbfgs_optimize(h_cur, target, n, beta, 100.0, maxiter=500)
                c5 = compute_c5(h_cur)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_cur.copy()

    # Final intensive polish on best solution
    n_best = len(best_h)
    target_best = n_best / 2.0

    if tl() > 25:
        for beta in [3000, 8000, 20000]:
            if tl() < 12:
                break
            best_h = lbfgs_optimize(best_h, target_best, n_best, beta, 500.0,
                                    maxiter=3000, ftol=1e-18)
            c5 = compute_c5(best_h)
            if c5 < best_c5:
                best_c5 = c5

    if tl() > 15:
        best_h, c5 = proximal_grad(best_h, target_best, 2000, 2000, 20000, 0.002)
        if c5 < best_c5:
            best_c5 = c5

    if tl() > 10:
        best_h = lbfgs_optimize(best_h, target_best, n_best, 50000, 1000.0,
                                maxiter=5000, ftol=1e-18)
        c5 = compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5

    best_h = project_exact(best_h, target_best)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
