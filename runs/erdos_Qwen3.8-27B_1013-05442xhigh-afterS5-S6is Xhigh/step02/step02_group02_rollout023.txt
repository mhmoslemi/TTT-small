import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def c5_lag_gradient(h, n, lag):
    dx = 2.0 / n
    j = lag - (n - 1)
    grad = np.zeros(n)
    if j >= 0:
        i1 = np.arange(n - j)
        grad[i1] += (1.0 - h[i1 + j]) * dx
        i2 = np.arange(j, n)
        grad[i2] -= h[i2 - j] * dx
    else:
        jp = -j
        i1 = np.arange(jp, n)
        grad[i1] += (1.0 - h[i1 - jp]) * dx
        i2 = np.arange(n - jp)
        grad[i2] -= h[i2 + jp] * dx
    return grad

def soft_max(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 700:
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    return soft_max(c, alpha)

def _smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 700:
        w = np.exp(s - mx)
    else:
        w = np.exp(s)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(c)):
        if w[k] > 1e-7:
            grad += w[k] * c5_lag_gradient(x, n, k)
    return grad

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def _slsqp_cons(n):
    return [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]

def slsqp_optimize(h, n, alpha, maxiter=500, ftol=1e-14):
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = _slsqp_cons(n)
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha), method='SLSQP',
                       jac=_smooth_grad, bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgradient_step(h, n, step):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    lag = int(np.argmax(c))
    grad = c5_lag_gradient(h, n, lag)
    return project_feasible(h - step * grad, n)

def de_optimize(h_init, n, pop_size, n_gens, t0, deadline, rng):
    pop = np.zeros((pop_size, n))
    pop[0] = h_init
    for i in range(1, pop_size):
        r = i % 4
        if r == 1:
            hh = np.zeros(n)
            hh[rng.choice(n, n // 2, replace=False)] = 1.0
        elif r == 2:
            x = np.linspace(0, 2, n)
            hh = 0.5 + 0.45 * np.sin(rng.uniform(1, 4) * np.pi * x + rng.uniform(0, 6.28))
        elif r == 3:
            hh = h_init + rng.normal(0, 0.15, n)
        else:
            hh = h_init + rng.normal(0, 0.05, n)
        pop[i] = project_feasible(hh, n)
    fit = np.array([compute_c5(pop[i], n) for i in range(pop_size)])
    best_h = pop[np.argmin(fit)].copy()
    best_f = fit.min()
    F, CR = 0.7, 0.8
    for gen in range(n_gens):
        if time.time() > deadline - 30:
            break
        order = np.argsort(fit)
        for i in range(pop_size):
            sel = rng.choice(order[:max(4, int(pop_size * 0.7))], 3, replace=False)
            mut = pop[sel[0]] + F * (pop[sel[1]] - pop[sel[2]])
            mask = rng.random(n) < CR
            trial = np.where(mask, mut, pop[i])
            trial = project_feasible(trial, n)
            f = compute_c5(trial, n)
            if f < fit[i]:

                fit[i] = f
        idx = np.argmin(fit)

        if (gen + 1) % 12 == 0:

            for wi in order[-nr:]:
                pop[wi] = project_feasible(pop[wi] + rng.normal(0, 0.25, n), n)
                fit[wi] = compute_c5(pop[wi], n)
    return best_h, best_f

def upsample(h, n_old, n_new):
    xo = np.linspace(0, 2, n_old)
    xn = np.linspace(0, 2, n_new)
    return project_feasible(np.interp(xn, xo, h), n_new)

def coord_descent(h, n, deadline):
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(30):
        if time.time() > deadline - 5:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline - 5:
                break
            for j in range(i + 1, n):
                for d in [0.01, 0.004]:
                    for s in [1, -1]:
                        ni, nj = h[i] + s * d, h[j] - s * d
                        if 0 <= ni <= 1 and 0 <= nj <= 1:
                            h[i], h[j] = ni, nj
                            c = compute_c5(h, n)
                            if c < best - 1e-15:
                                best = c
                                improved = True
                            else:
                                h[i], h[j] = ni - s * d, nj + s * d
        if not improved:
            break
    return h, best
