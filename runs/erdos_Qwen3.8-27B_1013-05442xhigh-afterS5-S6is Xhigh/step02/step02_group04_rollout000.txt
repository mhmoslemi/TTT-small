    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    w = e / e.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for idx in range(len(corr)):
        if w[idx] < 1e-12:
            continue
        lag = idx - (n - 1)
        wv = w[idx] * dx
        for j in range(n):
            jp = j + lag
            jn = j - lag
            if 0 <= jp < n:
                g[j] += wv * (1.0 - x[jp])
            if 0 <= jn < n:
                g[j] -= wv * x[jn]
    return g

def fast_smooth_max_grad_h(x, beta, n):
    """Vectorized gradient computation."""
    corr = corr_full(x)
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    w = e / e.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    # For each lag, compute contribution to gradient
    # g[j] = dx * sum_lag w[lag_idx] * [(1-x[j+lag]) - x[j-lag]]
    # Use FFT-like approach or direct computation
    for lag in range(-(n-1), n):
        idx = lag + (n - 1)
        wv = w[idx]
        if wv < 1e-12:
            continue
        # Indices j where j+lag is valid: 0 <= j+lag < n => max(0,-lag) <= j < min(n, n-lag)
        jstart = max(0, -lag)
        jend = min(n, n - lag)
        if jstart >= jend:
            continue
        # (1 - x[j+lag]) contribution
        g[jstart:jend] += wv * dx * (1.0 - x[jstart + lag:jend + lag])
        # -x[j-lag] contribution: j-lag valid means 0 <= j-lag < n => lag <= j < lag+n
        jstart2 = max(0, lag)
        jend2 = min(n, n + lag)
        if jstart2 >= jend2:
            continue
        g[jstart2:jend2] -= wv * dx * x[jstart2 - lag:jend2 - lag]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_points = len(h_init)
    else:
        n_points = 70
        h_init = np.full(n_points, 0.5)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = project_h(h_init, target_sum)
    best_c5 = compute_c5(best_h)
    n = n_points
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]

    # Phase 1: SLSQP with moderate beta for quick local improvement
    for beta in [30, 80, 150]:
        if time_left() < 100:
            break
        def obj1(x):
            return smooth_max_val(corr_full(x), beta) * dx
        def obj1_grad(x):
            return fast_smooth_max_grad_h(x, beta, n)
        try:
            res = minimize(obj1, best_h, jac=obj1_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 2000, 'ftol': 1e-15})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 2: Diversified local search with SLSQP
    # Try many different perturbations and refine each
    n_diversify = 60
    for i in range(n_diversify):
        if time_left() < 80:
            break
        h0 = best_h.copy()
        strat = i % 8
        if strat == 0:
            scale = 0.005 + 0.015 * np.exp(-i / 20.0)
            h0 += rng.randn(n) * scale
        elif strat == 1:
            # Binary push
            thresh = 0.5 - 0.1 * rng.randn()
            h0 = np.where(h0 > thresh, h0 + 0.08, h0 - 0.08)
        elif strat == 2:
            # Target worst lag reduction
            corr = corr_full(h0)
            worst_idx = np.argmax(corr)
            lag = worst_idx - (n - 1)
            g = np.zeros(n)
            for j in range(n):
                jp = j + lag
                jn = j - lag
                if 0 <= jp < n:
                    g[j] += (1.0 - h0[jp])
                if 0 <= jn < n:
                    g[j] -= h0[jn]
            g = g - g.mean()
            h0 -= 0.02 * g
        elif strat == 3:
            # Block perturbation
            nblocks = rng.randint(2, 5)
            bsizes = np.sort(rng.choice(np.arange(1, n), size=min(nblocks, n-1), replace=False))
            for bs in bsizes:
                h0[max(0, bs-1):bs+2] += rng.uniform(-0.06, 0.06, min(3, n - max(0, bs-1)))
        elif strat == 4:
            # Mirror mix
            mix = rng.uniform(0.3, 0.7)
            h0 = mix * h0 + (1 - mix) * h0[::-1] + rng.randn(n) * 0.008
        elif strat == 5:
            # Sinusoidal perturbation
            freq = rng.randint(2, 8)
            h0 += 0.02 * np.sin(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 6.28))
        elif strat == 6:
            # Multi-pair swap
            for _ in range(rng.randint(3, 8)):
                ii, jj = rng.choice(n, 2, replace=False)
                d = rng.uniform(-0.04, 0.04)
                h0[ii] += d
                h0[jj] -= d
        else:
            # Ripples
            for _ in range(rng.randint(1, 3)):
                center = rng.randint(0, n)
                radius = rng.randint(2, 6)
                lo, hi = max(0, center-radius), min(n, center+radius)
                h0[lo:hi] += rng.uniform(-0.03, 0.03, hi-lo)

        h0 = project_h(h0, target_sum)
        beta = rng.choice([100, 200, 400, 800])
        def obj2(x):
            return smooth_max_val(corr_full(x), beta) * dx
        def obj2_grad(x):
            return fast_smooth_max_grad_h(x, beta, n)
        try:
            res = minimize(obj2, h0, jac=obj2_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 1500, 'ftol': 1e-14})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 3: High-beta SLSQP polishing (increasingly aggressive)
    for beta in [500, 1000, 2000, 4000, 8000]:
        if time_left() < 25:
            break
        def obj3(x):
            return smooth_max_val(corr_full(x), beta) * dx
        def obj3_grad(x):
            return fast_smooth_max_grad_h(x, beta, n)
        try:
            res = minimize(obj3, best_h, jac=obj3_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-16})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 4: Final intensive local search around best
    for i in range(30):
        if time_left() < 20:
            break
        h0 = best_h.copy()
        scale = 0.002 + 0.005 * np.exp(-i / 10.0)
        h0 += rng.randn(n) * scale
        # Also try targeted reduction of worst lag
        corr = corr_full(h0)
        top3 = np.argsort(corr)[-3:]
        for idx in top3:
            lag = idx - (n - 1)
            g = np.zeros(n)
            for j in range(n):
                jp = j + lag
                jn = j - lag
                if 0 <= jp < n:
                    g[j] += (1.0 - h0[jp])
                if 0 <= jn < n:
                    g[j] -= h0[jn]
            g = g - g.mean()
            h0 -= 0.003 * g
        h0 = project_h(h0, target_sum)

        beta = 3000
        def obj4(x):
            return smooth_max_val(corr_full(x), beta) * dx
        def obj4_grad(x):
            return fast_smooth_max_grad_h(x, beta, n)
        try:
            res = minimize(obj4, h0, jac=obj4_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Final ultra-high beta polish
    if time_left() > 15:
        for beta in [10000, 20000]:
            if time_left() < 10:
                break
            def obj5(x):
                return smooth_max_val(corr_full(x), beta) * dx
            def obj5_grad(x):
                return fast_smooth_max_grad_h(x, beta, n)
            try:
                res = minimize(obj5, best_h, jac=obj5_grad, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 8000, 'ftol': 1e-17})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
