def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, target_sum):
    n = y.size
    lo = y.min() - n
    hi = y.max() + n
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def grad_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    w = np.exp(np.clip(beta * (corr - m), -500, 500))
    w /= w.sum()
    omh = 1.0 - h
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wk * omh[lag:]
            grad[lag:] -= wk * h[:n - lag]
        else:
            L = -lag
            grad[L:] += wk * omh[:n - L]
            grad[:n - L] -= wk * h[L:]
    return dx * grad

                best_h, best_c5 = h_sl3, c5_sl3

    return best_h, best_c5, best_n
