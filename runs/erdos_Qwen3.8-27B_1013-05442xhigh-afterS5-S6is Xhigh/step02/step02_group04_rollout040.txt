                    best_c5 = c5_trial
                    best_h = h_trial
                    stagnation = 0
                    break
                step *= 0.7
            else:
                stagnation += 1
                step = 0.003
                if stagnation > 20:
                    # Perturb and retry
                    best_h = project_h(best_h + rng.randn(n) * 0.01, target)
                    best_c5 = compute_c5(best_h)
                    stagnation = 0

    # Phase 5: Final SLSQP polish with multiple betas
    for beta in [5000, 10000, 50000]:
        if budget_s - elapsed() < 20:
            break
        obj = SmoothObj(beta, dx)
        try:
            res = minimize(obj, best_h, jac=obj.grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 10000, 'ftol': 1e-17})
            h_new = project_h(res.x, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new
        except Exception:
            pass

    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
