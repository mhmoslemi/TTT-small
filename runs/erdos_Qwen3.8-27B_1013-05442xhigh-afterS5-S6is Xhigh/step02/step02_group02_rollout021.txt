    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]

        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def cd_pass(h, n, t0, t_end, min_gap=1e-5, max_sweeps=100):
    best_c5 = compute_c5(h, n)
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        free = np.where((h > min_gap) & (h < 1.0 - min_gap))[0]
        if len(free) < 2:
            break
        nf = len(free)
        for ia in range(nf):
            if time.time() > t_end:
                return h, best_c5
            i = free[ia]
            for ib in range(ia + 1, nf):
                j = free[ib]
                d_min = max(-h[i], -(1.0 - h[j]))
                d_max = min(1.0 - h[i], h[j])
                if d_max - d_min < 1e-8:
                    continue
                best_d = 0.0
                best_c = best_c5
                samples = np.linspace(d_min, d_max, 5)
                for d in samples:
                    if abs(d) < 1e-12:
                        continue
                    h[i] += d
                    h[j] -= d
                    c = compute_c5(h, n)
                    h[i] -= d
                    h[j] += d
                    if c < best_c - 1e-14:
                        best_c = c
                        best_d = d
                if best_d == 0.0:
                    continue
                span = (d_max - d_min) / 2.5
                a = max(d_min, best_d - span)
                b = min(d_max, best_d + span)
                for _ in range(15):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    h[i] += m1
                    h[j] -= m1
                    f1 = compute_c5(h, n)
                    h[i] -= m1
                    h[j] += m1
                    h[i] += m2
                    h[j] -= m2
                    f2 = compute_c5(h, n)
                    h[i] -= m2
                    h[j] += m2
                    if f1 < f2:
                        b = m2
                    else:
                        a = m1
                d_opt = (a + b) / 2.0
                h[i] += d_opt
                h[j] -= d_opt
                c_new = compute_c5(h, n)
                h[i] -= d_opt
                h[j] += d_opt
                if c_new < best_c5 - 1e-14:
                    h[i] += d_opt
                    h[j] -= d_opt
                    best_c5 = c_new
                    improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def targeted_sa(h, n, t0, t_end, rng, n_iter, e_start, e_min, d_start, d_min):
    E = e_start
    D = d_start
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    current_c5 = best_c5
    e_ratio = (e_min / e_start) ** (1.0 / max(n_iter, 1))
    d_ratio = (d_min / d_start) ** (1.0 / max(n_iter, 1))
    for it in range(n_iter):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.0005, D) * rng.choice([1.0, -1.0])
        hi = h[i] + d
        hj = h[j] - d
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        h[i] = hi
        h[j] = hj
        new_c5 = compute_c5(h, n)
        delta = new_c5 - current_c5
        if delta < 0 or (delta < 10.0 * E and rng.random() < np.exp(-delta / max(E, 1e-12))):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        else:
            h[i] = hi - d
            h[j] = hj + d
        E *= e_ratio
        D = max(D * d_ratio, 0.0005)
    return best_h, best_c5

def optimize_n(h, n, t0, t_end, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)

    alphas = [500, 2000, 10000, 50000, 200000]
    for a in alphas:
        if time.time() > t_end - 20:
            break
        ht = slsqp_optimize(best_h, n, a, 3000, 1e-16)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = ht, c

    t_end2 = min(t_end, t0 + (t_end - t0) * 0.4)
    best_h, best_c5 = cd_pass(best_h, n, t0, t_end2)

    for a in [50000, 200000]:
        if time.time() > t_end - 15:
            break
        ht = slsqp_optimize(best_h, n, a, 5000, 1e-18)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = ht, c

    t_end3 = min(t_end, t0 + (t_end - t0) * 0.6)
    best_h, best_c5 = cd_pass(best_h, n, t0, t_end3)

    if time.time() < t_end - 80:
        best_h, best_c5 = targeted_sa(best_h, n, t0, min(t_end, t0 + (t_end - t0) * 0.8),
                                       rng, 200000, 0.008, 1e-7, 0.08, 0.002)
        t_end4 = min(t_end, t0 + (t_end - t0) * 0.92)
        best_h, best_c5 = cd_pass(best_h, n, t0, t_end4)
        for a in [100000, 500000]:
            if time.time() > t_end - 10:
                break
            ht = slsqp_optimize(best_h, n, a, 3000, 1e-18)
            ht, c = cd_pass(ht, n, t0, t_end)
            if c < best_c5 - 1e-14:
                best_h, best_c5 = ht, c

    while time.time() < t_end - 40:
        scale = rng.choice([0.02, 0.04, 0.08])
        h_pert = best_h + rng.normal(0, scale, size=n)
        h_pert = project_feasible(h_pert, n)
        ht = slsqp_optimize(h_pert, n, 20000, 1500, 1e-14)
        ht, c = cd_pass(ht, n, t0, t_end)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = ht, c

    best_h, best_c5 = cd_pass(best_h, n, t0, t_end)
    if time.time() < t_end - 5:
        ht = slsqp_optimize(best_h, n, 500000, 3000, 1e-18)
        c = compute_c5(ht, n)
        if c < best_c5:
            best_h, best_c5 = ht, c
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    t_end = t0 + budget_s - 10

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 80

    if h_init is not None:
        h_init = project_feasible(h_init, n_init)

    n_list = []
    if h_init is not None:
        n_list.append(n_init)
    for nn in [64, 80, 96, 100, 120]:
        if nn not in n_list and nn < 200:
            n_list.append(nn)

    best_overall_h = None
    best_overall_c5 = float('inf')
    best_overall_n = 0
    total_n = len(n_list)
    time_per = (budget_s - 30) / total_n

    for idx_n in range(total_n):
        elapsed = time.time() - t0
        remaining = budget_s - elapsed - 10
        if remaining < 60:
            break
        n = n_list[idx_n]
        alloc = min(remaining * 0.6, time_per * 1.5)
        t_n_end = t0 + elapsed + alloc

        if n == n_init and h_init is not None:
            h_start = h_init.copy()
        else:
            h_start = np.full(n, 0.5)
            x = np.linspace(0, 2, n)
            h_start = 0.5 + 0.3 * np.sin(np.pi * x)
            h_start = project_feasible(h_start, n)

        h_opt, c5_opt, n_opt = optimize_n(h_start, n, t0, t_n_end, rng)
        if c5_opt < best_overall_c5:
            best_overall_h = h_opt
            best_overall_c5 = c5_opt
            best_overall_n = n_opt

    if best_overall_h is None:
        best_overall_h = np.full(80, 0.5)
        best_overall_n = 80
        best_overall_c5 = compute_c5(best_overall_h, best_overall_n)

    return best_overall_h, best_overall_c5, best_overall_n
