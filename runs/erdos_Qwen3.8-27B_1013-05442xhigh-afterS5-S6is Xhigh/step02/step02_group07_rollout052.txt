import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)

def corr_gradient(h, weights):
    """Compute gradient of weighted sum of correlations."""
    n = len(h)
    lag_offset = n - 1
    grad = np.zeros(n)
    for m in range(2*n - 1):
        w = weights[m]
        if w < 1e-15:
            continue
        lag = m - lag_offset
        if lag >= 0:
            # h[i] as h[n+lag]: i from lag to n-1, contributes (1-h[i-lag])
            for i in range(lag, n):
                grad[i] += w * (1.0 - h[i-lag])
            # h[i] as (1-h[n]): i from 0 to n-1-lag, contributes -h[i+lag]
            for i in range(0, n-lag):
                grad[i] += w * (-h[i+lag])
        else:
            p = -lag
            # h[i] as h[n-p]: i from 0 to n-1-p, contributes (1-h[i+p])
            for i in range(0, n-p):
                grad[i] += w * (1.0 - h[i+p])
            # h[i] as (1-h[n]): i from p to n-1, contributes -h[i-p]
            for i in range(p, n):
                grad[i] += w * (-h[i-p])
    return grad

                            h_new[j] -= 0.04 + 0.04 * rng.random()
                    elif h_cur[i] < 0.6 and h_cur[j] < 0.6:
                        if rng.random() < 0.5:
                            h_new[i] += 0.03 + 0.03 * rng.random()
                        else:
                            h_new[j] += 0.03 + 0.03 * rng.random()
    elif r < 0.84:
        blen = rng.randint(2, min(8, n // 4 + 1))
        i = rng.randint(0, n - blen + 1)
        j = rng.randint(0, n - blen + 1)
        if abs(i - j) >= blen:
            tmp = h_new[i:i + blen].copy()
            h_new[i:i + blen] = h_new[j:j + blen]
            h_new[j:j + blen] = tmp
    else:
        n_sw = rng.randint(1, 4)
        for _ in range(n_sw):
            i, j = rng.choice(n, 2, replace=False)
            h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0

    best_h = project_h(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: Gradient descent with line search
    for beta in [80, 150, 250, 400, 600]:
        if time_left() < 250:
            break
        for _ in range(300):
            if time_left() < 250:
                break
            h_new, c5_new, improved = grad_step(
                best_h, beta, target_sum, 0.002,
                [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
            )
            if not improved:
                break
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Simulated annealing
    if time_left() > 200:
        h_cur = best_h.copy()
        c5_cur = best_c5
        T = 0.008
        T_min = 0.00005
        cooling = 0.9993
        no_improve = 0

        while time_left() > 100:
            h_new = anneal_move(h_cur, n, T, rng)
            h_new = project_h(h_new, target_sum)
            c5_new = compute_c5(h_new)
            delta = c5_new - c5_cur
            if delta < 0 or rng.random() < np.exp(-min(150.0, delta / T)):
                h_cur = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_h = h_cur.copy()
                    best_c5 = c5_cur
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

            T = max(T * cooling, T_min)
            if no_improve > 3000:
                T = max(T, 0.005)
                no_improve = 0

    # Phase 3: SLSQP polish with multi-start
    if time_left() > 40:
        bounds = Bounds(np.zeros(n), np.ones(n))
        constraints = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_jac}]

        starts = [best_h]
        for _ in range(3):
            h0 = best_h.copy()
            n_p = rng.randint(5, 15)
            idx = rng.choice(n, n_p, replace=False)
            h0[idx] += rng.randn(n_p) * 0.04
            starts.append(project_h(h0, target_sum))

        for h_start in starts:
            if time_left() < 25:
                break
            for beta_val in [400, 700, 1200]:
                if time_left() < 20:
                    break
                try:
                    res = minimize(slsqp_obj, h_start, args=(beta_val, dx),
                                   method='SLSQP', bounds=bounds,
                                   constraints=constraints,
                                   options={'maxiter': 3000, 'ftol': 1e-15})
                    h_new = project_h(res.x, target_sum)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h, best_c5 = h_new.copy(), c5
                        h_start = h_new.copy()
                except Exception:
                    pass

    # Phase 4: Final fine gradient polish
    if time_left() > 15:
        for _ in range(100):
            if time_left() < 10:
                break
            h_new, c5_new, improved = grad_step(
                best_h, 500, target_sum, 0.001,
                [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]
            )
            if not improved:
                break
            best_h = h_new
            best_c5 = c5_new

    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
