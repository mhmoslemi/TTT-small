            # Contribution: -h[j+s]
            grad[0:n-s] += -w * dx * h[s:n]
    
    return grad

Let me think about the "critical shift descent" more carefully:

At the optimum, say shifts k1, k2, ..., km are all achieving the maximum C5. The subdifferential of the max function at this point is the convex hull of {∇c_k1, ∇c_k2, ..., ∇c_km}. A descent direction is any vector d such that <d, ∇c_ki> < 0 for all i.

One natural choice: d = -sum_i ∇c_ki (the sum of gradients, negated). This is guaranteed to be a descent direction if the gradients are not "balanced" (i.e., if the minimum of the convex hull doesn't contain the origin in a specific way).

Actually, more precisely: d = -(sum_i α_i ∇c_ki) where α_i > 0 and sum α_i = 1. Any such d is a descent direction for the max function at a point where all k_i are active.

The choice α_i = 1/m (uniform) is the "natural" one. This is essentially what the soft-max gradient does when alpha → ∞.

So my "critical shift descent" is:
1. Find the top-m shifts (those with c_k within epsilon of the max)
2. Compute d = -(1/m) * sum_{i=1}^m ∇c_{k_i}
3. Take a step: h_new = project(h + eta * d)
4. Choose eta by line search (or backtracking)

This is simpler and potentially more effective than the full soft-max approach when the number of active shifts is small.

        E *= e_ratio

    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] / ws
        if wk < 1e-14:
            continue
        sh = k - (n - 1)
        if sh >= 0:
            grad[:n - sh] += wk * dx * (1.0 - h[sh:])
            grad[sh:] -= wk * dx * h[:n - sh]
        else:
            a = -sh
            grad[a:] += wk * dx * (1.0 - h[:-a])
            grad[:-a] -= wk * dx * h[a:]
    return grad

def con_eq_fun(x, n, alpha):
    return x.sum() - n / 2.0

def con_eq_jac(x, n, alpha):
    return np.ones(len(x))

def slsqp_opt(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq_fun, 'jac': con_eq_jac}]
    try:
        res = minimize(soft_max_obj, h, args=(n, alpha), jac=soft_max_grad,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def critical_descent(h, n, t0, end_time):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    dx = 2.0 / n
    for it in range(200000):
        if time.time() > end_time:
            break
        c = compute_c5_vec(h, n)
        m = c.max()
        if float(m) < best_c5:
            best_c5 = float(m)
            best_h = h.copy()
        thresh = m - 0.02 * m
        crit = np.where(c >= thresh)[0]
        if len(crit) > 15:
            crit = np.argsort(c)[-15:]
        grad_sum = np.zeros(n)
        for k in crit:
            sh = k - (n - 1)
            if sh >= 0:
                grad_sum[:n - sh] += dx * (1.0 - h[sh:])
                grad_sum[sh:] -= dx * h[:n - sh]
            else:
                a = -sh
                grad_sum[a:] += dx * (1.0 - h[:-a])
                grad_sum[:-a] -= dx * h[a:]
        d = -grad_sum / len(crit)
        dn = np.linalg.norm(d)
        if dn < 1e-15:
            break
        d = d / dn
        improved = False
        for eta in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
            h_new = project_feasible(h + eta * d, n)
            f_new = compute_c5(h_new, n)
            if f_new < best_c5:
                h = h_new
                best_h = h.copy()
                best_c5 = f_new
                improved = True
                break
        if not improved:
            h_new = project_feasible(h - 0.001 * d, n)
            f_new = compute_c5(h_new, n)
            if f_new < best_c5:
                h = h_new
                best_h = h.copy()
                best_c5 = f_new
            else:
                break
    return best_h, best_c5

def structured_sa(h, n, t0, end_time, rng, n_iter):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    curr_c5 = best_c5
    E = 0.03
    e_ratio = (0.0001 / 0.03) ** (1.0 / max(n_iter, 1))
    for it in range(n_iter):
        if time.time() > end_time:
            break
        c = compute_c5_vec(h, n)
        m = c.max()
        crit = np.where(c >= m - 0.08 * m)[0]
        k = crit[rng.randint(len(crit))]
        sh = k - (n - 1)
        if sh >= 0:
            if n - sh < 2:
                i = rng.randint(n)
                j = (i + sh) % n
            else:
                i = rng.randint(n - sh)
                j = i + sh
        elif sh < 0:
            a = -sh
            if a >= 1:
                j = rng.randint(n - a)
                i = j + a
            else:
                i = rng.randint(n)
                j = rng.randint(n)
        else:
            i = rng.randint(n)
            j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-0.08, 0.08)
        hi, hj = h[i] + delta, h[j] - delta
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        h[i], h[j] = hi, hj
        new_c5 = compute_c5(h, n)
        if new_c5 < curr_c5 or (curr_c5 - new_c5 > -0.05 * E and
           rng.random() < np.exp(-(new_c5 - curr_c5) / max(E, 1e-12))):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = h.copy(), new_c5
        else:
            h[i], h[j] = hi - delta, hj + delta
        E *= e_ratio
    return best_h, best_c5

def binary_local_search(h, n, t0, end_time, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    for it in range(100000):
        if time.time() > end_time:
            break
        i = rng.randint(n)
        if h[i] < 0.5:
            h[i] = min(1.0, h[i] + 0.3)
        else:
            h[i] = max(0.0, h[i] - 0.3)
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c5:
            best_h, best_c5 = h.copy(), c
        else:
            h = best_h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 80
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)

    elapsed = time.time() - t0
    phase1_end = t0 + min(200, budget_s * 0.18)
    phase2_end = t0 + min(550, budget_s * 0.50)
    phase3_end = t0 + min(850, budget_s * 0.78)

    for alpha in [200, 800, 3000, 12000, 50000, 200000, 800000]:
        if time.time() > phase1_end:
            break
        h_new = slsqp_opt(h, n, alpha, 8000, 1e-18)
        c = compute_c5(h_new, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = h_new, c
            h = h_new

    h = best_h.copy()
    h, c = critical_descent(h, n, t0, phase2_end)
    if c < best_c5:
        best_h, best_c5 = h, c

    for alpha in [200000, 800000, 3000000]:
        if time.time() > phase2_end:
            break
        h_new = slsqp_opt(best_h, n, alpha, 10000, 1e-20)
        c = compute_c5(h_new, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = h_new, c

    h, c = structured_sa(best_h, n, t0, phase3_end, rng, 500000)
    if c < best_c5:
        best_h, best_c5 = h, c

    for alpha in [500000, 2000000, 8000000]:
        if time.time() > budget_s - 80:
            break
        h_new = slsqp_opt(best_h, n, alpha, 10000, 1e-20)
        c = compute_c5(h_new, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = h_new, c

    h, c = critical_descent(best_h, n, t0, budget_s - 30)
    if c < best_c5:
        best_h, best_c5 = h, c

    if time.time() < budget_s - 40:
        h, c = structured_sa(best_h, n, t0, budget_s - 40, rng, 300000)
        if c < best_c5:
            best_h, best_c5 = h, c

    if time.time() < budget_s - 20:
        h_new = slsqp_opt(best_h, n, 10000000, 10000, 1e-20)
        c = compute_c5(h_new, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = h_new, c

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
