```python
def compute_grad(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0-h, mode="full")
    corr_max = np.max(corr)
    w = np.exp(beta * (corr - corr_max))
    w /= w.sum()
    
    # k_m = m - (n-1), m = 0..2n-2
    k_m = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    j = np.arange(n)  # shape (n,)
    
    # First term: (1-h[j+k_m]) where 0<=j+k_m<n
    idx1 = j[:, None] + k_m[None, :]  # shape (n, 2n-1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    term1 = np.where(valid1, 1.0 - h[idx1.clip(0, n-1)], 0.0)
    grad_first = term1 @ w  # shape (n,)
    
    # Second term: h[j-k_m] where 0<=j-k_m<n
    idx2 = j[:, None] - k_m[None, :]
    valid2 = (idx2 >= 0) & (idx2 < n)
    term2 = np.where(valid2, h[idx2.clip(0, n-1)], 0.0)
    grad_second = term2 @ w
    
    grad = (grad_first - grad_second) * dx / beta
    return grad, corr
```

```python
def project(h_new, S):
    n = len(h_new)
    lo, hi = h_new.min() - 1, h_new.max() + 1
    for _ in range(60):

        h_proj = np.clip(h_new - mid, 0, 1)
        s = h_proj.sum()
        if s > S:

        else:
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_polytope(h_new, S):
    """Project onto {h: 0<=h<=1, sum(h)=S}"""
    n = len(h_new)
    lo = h_new.min() - 1.0
    hi = h_new.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h_new - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h_new - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad_smooth_max(h, beta, dx):
    """Gradient of smooth_max(corr, beta) * dx"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    corr_max = np.max(corr)
    w = np.exp(beta * (corr - corr_max))
    w /= w.sum()
    
    k_m = np.arange(2 * n - 1) - (n - 1)
    j = np.arange(n)
    
    idx1 = j[:, None] + k_m[None, :]
    valid1 = (idx1 >= 0) & (idx1 < n)
    term1 = np.where(valid1, 1.0 - h[np.clip(idx1, 0, n-1)], 0.0)
    grad_first = term1 @ w
    
    idx2 = j[:, None] - k_m[None, :]
    valid2 = (idx2 >= 0) & (idx2 < n)
    term2 = np.where(valid2, h[np.clip(idx2, 0, n-1)], 0.0)
    grad_second = term2 @ w
    
    grad = (grad_first - grad_second) * dx / beta
    return grad, corr

def pgd_optimize(h, S, budget_remain, beta_init=50.0, beta_final=300.0, 
                 lr_init=0.05, momentum=0.9):
    """Projected gradient descent with momentum and adaptive beta"""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    t0 = time.time()
    n_iter = 0
    stale = 0
    
    while time.time() - t0 < budget_remain:
        n_iter += 1
        # Anneal beta
        progress = min(1.0, n_iter / max(1, int(budget_remain * 10)))
        beta = beta_init + (beta_final - beta_init) * progress
        
        grad, corr = compute_grad_smooth_max(h, beta, dx)
        # Nesterov-like step
        h_new = h + momentum * (h - h_prev) - lr * grad
        h_new = project_polytope(h_new, S)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stale = 0
            lr = min(lr * 1.05, 0.5)
        else:
            stale += 1
            lr *= 0.9
        
        h_prev = h
        h = h_new
        
        if stale > 100:
            # Restart with perturbation
            h = best_h + np.random.randn(n) * 0.02
            h = project_polytope(h, S)
            h_prev = h.copy()
            lr = lr_init * 0.5
            stale = 0
    
    return best_h, best_c5, n_iter

def interpolate_h(h_old, n_new):
    """Interpolate h from old resolution to new resolution"""
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    # Periodic interpolation
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return h_new
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_polytope(h_new, S):
    n = len(h_new)
    lo = h_new.min() - 1.0
    hi = h_new.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h_new - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h_new - lam, 0.0, 1.0)

def compute_grad(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    corr_max = corr.max()
    w = np.exp(beta * (corr - corr_max))
    w /= w.sum()
    g = 1.0 - 2.0 * h
    idx = np.arange(n)[:, None] + np.arange(n)[None, :]
    W_mat = w[idx]
    grad = (W_mat * g[None, :]).sum(axis=1) * dx / beta
    return grad, corr

def pgd_optimize(h_init, S, time_budget, beta_start=50.0, beta_end=400.0,
                 lr_init=0.05, momentum=0.85, restart_every=200):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    t0 = time.time()
    it = 0
    stale = 0
    
    while time.time() - t0 < time_budget:
        it += 1
        progress = min(1.0, it / max(1, int(time_budget * 8)))
        beta = beta_start + (beta_end - beta_start) * progress ** 0.7
        
        grad, corr = compute_grad(h, beta, dx)
        # Momentum step
        h_new = h + momentum * (h - h_prev) - lr * grad
        h_new = project_polytope(h_new, S)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stale = 0
            lr = min(lr * 1.02, 0.3)
        else:
            stale += 1
            lr *= 0.95
        
        h_prev = h
        h = h_new
        
        if stale > restart_every:
            # Restart
            h = best_h + np.random.randn(n) * 0.01 * (stale / restart_every)
            h = project_polytope(h, S)
            h_prev = h.copy()
            lr = lr_init * 0.5
            stale = 0
    
    return best_h, best_c5

    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old, period=2.0)

    best_n = len(h_init)

    remaining = budget_s - time.time() - 120

        if c5_opt < best_c5:

            current_best = h_opt.copy()

        S_f = n_f / 2.0
        dx_f = 2.0 / n_f
        bounds_f = Bounds(np.zeros(n_f), np.ones(n_f))
        
        def con_f(x):
            return np.sum(x) - S_f
        def jac_f(x):
            return np.ones(n_f)

                m = corr.max()
                return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx_f
            try:

                h_new = project_polytope(res.x, S_f)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

            if cf < best_c5:
                best_h, best_c5 = hf, cf
    
    return best_h, best_c5, best_n
```

```python
    w_top = np.exp(beta * (corr_top - corr_max))
    w_top /= w_top.sum()
    
    g = 1.0 - 2.0 * h
    # For each j: sum_{m in top_idx} w_top[m] * (1-2h[m-j]) * I(0≤m-j<n)
    grad = np.zeros(n)
    for i, m in enumerate(top_idx):
        l = m  # m-j = l means j = m-l
        # (1-2h[m-j]) * I(0≤m-j<n) for j=0..n-1
        # l = m-j ranges from m-(n-1) to m
        # I(0≤l<n) means m-n+1 ≤ j ≤ m
        j_min = max(0, m - n + 1)
        j_max = min(n-1, m)
        if j_min <= j_max:
            l_range = m - np.arange(j_min, j_max+1)
            grad[j_min:j_max+1] += w_top[i] * (1.0 - 2.0 * h[l_range])
    grad *= dx
    return grad, corr
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_polytope(h_new, S):
    n = len(h_new)
    lo = h_new.min() - 1.0
    hi = h_new.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h_new - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h_new - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    corr_max = corr.max()
    w = np.exp(beta * (corr - corr_max))
    w /= w.sum()
    g = 1.0 - 2.0 * h
    idx = np.arange(n)[:, None] + np.arange(n)[None, :]
    W_mat = w[idx]
    grad = (W_mat * g[None, :]).sum(axis=1) * dx
    return grad, corr

def pgd_optimize(h_init, S, time_budget, beta_start=50.0, beta_end=400.0,
                 lr_init=0.04, momentum=0.7, restart_thresh=150):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    t0 = time.time()
    it = 0
    stale = 0
    while time.time() - t0 < time_budget:
        it += 1
        progress = min(1.0, it / max(1, int(time_budget * 6)))
        beta = beta_start + (beta_end - beta_start) * progress ** 0.6
        grad, corr = compute_grad(h, beta, dx)
        h_new = h + momentum * (h - h_prev) - lr * grad
        h_new = project_polytope(h_new, S)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stale = 0
            lr = min(lr * 1.015, 0.25)
        else:
            stale += 1
            lr *= 0.97
        h_prev = h
        h = h_new
        if stale > restart_thresh:
            scale = 0.01 + 0.02 * min(1.0, stale / (restart_thresh * 3))
            h = best_h + np.random.randn(n) * scale
            h = project_polytope(h, S)
            h_prev = h.copy()
            lr = lr_init * 0.4
            stale = 0
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old, period=2.0)

def slsqp_polish(h_init, S, time_budget, beta_list=(200, 400, 600)):
    n = len(h_init)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    t0 = time.time()
    for beta in beta_list:
        if time.time() - t0 > time_budget:
            break
        def obj_slsqp(h):
            corr = np.correlate(h, 1.0 - h, mode="full")
            m = corr.max()
            return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx
        def con_fun(x):
            return np.sum(x) - S
        def con_jac(x):
            return np.ones(n)
        try:
            res = minimize(obj_slsqp, best_h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                           options={'maxiter': 400, 'ftol': 1e-14})
            h_new = project_polytope(res.x, S)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    best_n = len(h_init)
    n0 = len(h_init)
    S0 = n0 / 2.0

    # Phase 1: Quick refinement at base resolution
    t_alloc1 = min(50.0, budget_s * 0.06)
    h1, c5_1 = pgd_optimize(h_init, S0, t_alloc1, beta_start=30, beta_end=150, lr_init=0.03)
    if c5_1 < best_c5:
        best_h, best_c5, best_n = h1, c5_1, n0

    # Phase 2: Multi-resolution expansion
    resolutions = sorted(set([n0 * 2, n0 * 4, n0 * 8]))
    resolutions = [r for r in resolutions if r <= 560]

    time_remaining = budget_s - time.time() - 200
    n_res = len(resolutions)
    # Allocate more time to lower resolutions
    weights = [1.0 / (1.0 + 0.3 * i) for i in range(n_res)]
    w_sum = sum(weights)
    time_each = [time_remaining * w / w_sum for w in weights]

    current_best = best_h.copy()

    for i, n_new in enumerate(resolutions):
        if time.time() - start_time > budget_s - 150:
            break
        S_new = n_new / 2.0
        t_alloc = min(time_each[i], budget_s - time.time() - 120)
        if t_alloc < 20:
            t_alloc = 20

        # Interpolate from current best
        h_interp = project_polytope(interpolate_h(current_best, n_new), S_new)

        # Main optimization
        h_opt, c5_opt = pgd_optimize(h_interp, S_new, t_alloc * 0.65,
                                       beta_start=40, beta_end=350, lr_init=0.025)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n_new
            current_best = h_opt.copy()

        # Also try from original init with different params
        if time.time() - start_time < budget_s - 130:
            h_from_init = project_polytope(interpolate_h(h_init, n_new), S_new)
            t_alloc2 = min(t_alloc * 0.35, budget_s - time.time() - 120)
            if t_alloc2 > 15:
                h_opt2, c5_opt2 = pgd_optimize(h_from_init, S_new, t_alloc2,
                                                 beta_start=30, beta_end=250, lr_init=0.02)
                if c5_opt2 < best_c5:
                    best_h, best_c5, best_n = h_opt2.copy(), c5_opt2, n_new
                    current_best = h_opt2.copy()

    # Phase 3: SLSQP polish on best solution
    if time.time() - start_time < budget_s - 40:
        n_f = best_n
        S_f = n_f / 2.0
        t_slq = min(120.0, budget_s - time.time() - 30)
        h_s, c5_s = slsqp_polish(best_h, S_f, t_slq, beta_list=(200, 400, 600))
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s

    # Phase 4: Final PGD polish with high beta
    if time.time() - start_time < budget_s - 10:
        n_f = best_n
        S_f = n_f / 2.0
        tl = budget_s - time.time() - 5
        if tl > 15:
            hf, cf = pgd_optimize(best_h, S_f, tl, beta_start=300, beta_end=800, lr_init=0.008)
            if cf < best_c5:
                best_h, best_c5 = hf, cf

    return best_h, best_c5, best_n
```
