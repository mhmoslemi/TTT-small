    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

import numpy as np
from scipy.optimize import minimize, Bounds, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def obj_smooth(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta)

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_jac(x):
    return np.ones(len(x))

def sca_optimize(h0, n, dx, max_iter=300):
    target_sum = n / 2.0
    h = project_h(h0, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds_lp = [(0, 1)] * n + [(0, None)]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    lags = []
    for d in range(-(n - 1), n):
        i_s = max(0, -d)
        i_e = min(n, n - d)
        if i_s < i_e:
            lags.append((d, i_s, i_e))
    for it in range(max_iter):
        A_rows = []
        for d, i_s, i_e in lags:
            row = np.zeros(n + 1)
            for i in range(i_s, i_e):
                j = i + d
                row[i] += (1.0 - h[j])
                row[j] += (-h[i])
            row[n] = -1.0 / dx
            A_rows.append(row)
        A = np.array(A_rows)
        b = np.zeros(len(A_rows))
        try:
            res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds_lp, method='highs')
        except Exception:
            break
        if not res.success:
            break
        h_lp = np.clip(res.x[:n], 0.0, 1.0)
        h_lp = project_h(h_lp, target_sum)
        c5_lp = compute_c5(h_lp)
        if c5_lp < best_c5:
            best_h = h_lp.copy()
            best_c5 = c5_lp
        h = h_lp
    return best_h, best_c5

def resample_h(h, n_new):
    x_old = np.linspace(0, 2, len(h), endpoint=False) + 1.0 / (2 * len(h))
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / (2 * n_new)
    return np.interp(x_new, x_old, h)

def slsqp_polish(h0, n, max_iter=2000):
    target_sum = n / 2.0
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_jac}]
    h = project_h(h0, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for beta in [100, 250, 500, 1000, 2000]:
        try:
            res = minimize(obj_smooth, best_h, args=(beta,),
                         method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': max_iter, 'ftol': 1e-15})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass
    return best_h, best_c5

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
