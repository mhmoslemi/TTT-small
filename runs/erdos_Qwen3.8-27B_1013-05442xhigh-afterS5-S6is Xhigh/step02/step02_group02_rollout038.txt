    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_correlation(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def smooth_max_and_grad(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    if shifted.max() > 700:
        s = shifted - shifted.max()
        e = np.exp(s)
        logsum = s.max() + np.log(e.sum())
        val = m + logsum / alpha
        weights = e / e.sum()
    else:
        e = np.exp(shifted)
        logsum = np.log(e.sum())
        val = m + logsum / alpha
        weights = e / e.sum()
    return val, weights

    c = get_correlation(h, n)
    _, weights = smooth_max_and_grad(c, alpha)
    grad = grad_correlation(h, weights, n, 2.0/n)
    # Add penalty gradient
    s = h.sum() - n/2
    grad += 2 * penalty * s * np.ones(n)
    return grad

    val, _ = smooth_max_and_grad(c, alpha)
    s = h.sum() - n/2
    return val + penalty * s * s

Now let me think about the coordinate descent. The idea is: for each pair (i, j), find the optimal values of h[i] and h[j] given all other values fixed. Since the objective is max_k c_k, and each c_k depends linearly on h[i] and h[j], the problem for a fixed pair is:

minimize max_k (a_k * h[i] + b_k * h[j] + c_k)
subject to h[i] + h[j] = const (to maintain sum constraint)
0 <= h[i], h[j] <= 1

where a_k = dc_k/dh[i], b_k = dc_k/dh[j], c_k = c_k - a_k*h[i] - b_k*h[j] (the part not involving i or j).

With h[i] + h[j] = s (constant), we can write h[j] = s - h[i], so:
minimize max_k (a_k * h[i] + b_k * (s - h[i]) + c_k)
= maximize... minimize over h[i] in [max(0, s-1), min(1, s)] of max_k ((a_k - b_k) * h[i] + b_k * s + c_k)

2. Finding the lowest point on the envelope

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def smooth_max_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    if s.max() > 700:
        ss = s - s.max()
        return m + (ss.max() + np.log(np.exp(ss).sum())) / alpha
    return m + np.log(np.exp(s).sum()) / alpha

def smooth_max_w(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    if s.max() > 700:
        s = s - s.max()
    e = np.exp(s)
    return e / e.sum()

def grad_corr(w, h, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        lag = p - (n - 1)
        wp = w[p] * dx
        if abs(wp) < 1e-18:
            continue
        if lag == 0:
            grad += wp * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wp * (1.0 - h[:n - lag])
            grad[:n - lag] -= wp * h[lag:]
        else:
            L = -lag
            grad[L:] += wp * (1.0 - h[:n - L])
            grad[:n - L] -= wp * h[L:]
    return grad
