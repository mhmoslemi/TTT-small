            if time_left() > 20:
                h_new2 = slsqp_refine(best_h, 2000, n, target, maxiter=3000, ftol=1e-16)
                c52 = compute_c5(h_new2)
                if c52 < best_c5:
                    best_c5 = c52
                    best_h = h_new2.copy()

    # Phase 5: Final high-precision polish
    if time_left() > 10:
        for beta in [3000, 8000, 15000]:
            if time_left() < 8:
                break
            h_new = slsqp_refine(best_h, beta, n, target, maxiter=8000, ftol=1e-16)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
