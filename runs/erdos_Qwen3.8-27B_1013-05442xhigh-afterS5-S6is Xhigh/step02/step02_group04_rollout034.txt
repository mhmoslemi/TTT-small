    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def tleft():
        return budget_s - (time.time() - t0)

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float)

    configs = [(70, 3, 320), (60, 2, 220), (80, 2, 220)]

    for n, n_starts, time_alloc in configs:
        if tleft() < time_alloc + 40:
            continue
        ts = n / 2.0
        _TGT[0] = ts
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]

        for si in range(n_starts):
            if tleft() < 50:
                break
            per_start = time_alloc / n_starts
            if si == 0 and parent is not None and n == len(parent):
                h0 = parent.copy()
            elif si == 0 and parent is not None:
                h0 = np.interp(np.linspace(0, len(parent) - 1, n),
                               np.arange(len(parent)), parent)
            else:
                h0 = np.zeros(n)
                h0[rng.choice(n, n // 2, replace=False)] = 1.0
            h0 = project_sum_box(h0, ts)

            # Subgradient phase
            sub_t0 = time.time()
            max_it = int(per_start * 0.55 / (n * 2.0 / 1000.0))
            max_it = max(500, min(max_it, 5000))
            bh, bc5 = subgrad_optimize(h0, n, ts, t0, budget_s, rng)

            # Binary refine
            bh, bc5 = binary_refine(bh, n, ts, t0, budget_s)

            # SLSQP polish
            if tleft() > 25:
                for beta in [200, 1500]:
                    if tleft() < 12:
                        break
                    hp = slsqp_polish(bh, n, ts, bounds, cons, beta, 5000)
                    cp = compute_c5(hp)
                    if cp < bc5:
                        bc5 = cp
                        bh = hp

            if bc5 < best_c5:
                best_c5 = bc5
                best_h = bh.copy()
                best_n = n

    # Final high-beta polish
    if best_h is not None and tleft() > 15:
        n = best_n
        ts = n / 2.0
        _TGT[0] = ts
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
        for beta in [5000, 20000]:
            if tleft() < 8:
                break
            hp = slsqp_polish(best_h, n, ts, bounds, cons, beta, 10000)
            cp = compute_c5(hp)
            if cp < best_c5:
                best_h = hp
                best_c5 = cp

    best_h = project_sum_box(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
