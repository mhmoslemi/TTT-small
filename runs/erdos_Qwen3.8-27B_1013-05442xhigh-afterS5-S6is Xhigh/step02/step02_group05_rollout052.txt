```python
    n = len(h)
    # Compute all overlaps
    corr = np.correlate(h, 1.0 - h, mode='full')
    # shift = corr.index - (n-1), ranges from -(n-1) to n-1
    # corr[k] corresponds to shift s = k - (n-1)
    
    max_corr = np.max(corr)
    weights = np.exp(beta * (corr - max_corr))
    weights /= np.sum(weights)
    
    # Gradient: for each i, grad[i] = dx * sum_s weights[s] * ((1-h[i+s]) - h[i-s])
    # where s ranges from -(n-1) to n-1 (i.e., k ranges from 0 to 2n-2)
    # h[j] = 0 for j outside [0, n-1]

        if weights[k] < 1e-15:
            continue
        w = weights[k] * dx
        for i in range(n):

            if i+s >= 0 and i+s < n:
                term += (1.0 - h[i+s])
            if i-s >= 0 and i-s < n:
                term -= h[i-s]
            grad[i] += w * term
    
    return grad
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = h.copy()
    # Binary search for lambda
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = h_proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0, 1)
    # Fine-tune
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-10:
        # Add small correction to non-boundary elements
        free = (h_proj > 0.01) & (h_proj < 0.99)
        if free.sum() > 0:
            h_proj[free] += diff / free.sum()
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def compute_gradient(h, beta, dx):
    """Compute gradient of smooth_max(corr, beta) * dx"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    exps = np.exp(beta * (corr - max_corr))
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-16:
            continue
        s = k - (n - 1)
        w = weights[k] * dx
        for i in range(n):
            term = 0.0
            is1 = i + s
            is2 = i - s
            if 0 <= is1 < n:
                term += 1.0 - h[is1]
            if 0 <= is2 < n:
                term -= h[is2]
            grad[i] += w * term
    return grad

        grad = compute_gradient(h, float(beta), dx)
```

```python
def grad_sm_vec(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    w = np.exp(beta * (corr - mc))
    w /= w.sum()
    
    # For each shift s = k-(n-1), compute:
    # contrib[i] = (1-h[i+s]) - h[i-s]
    # Sum: grad[i] = dx * sum_k w[k] * contrib[i]
    
    # Vectorized approach:
    # (1-h[i+s]) for all i: this is (1-h) shifted by -s, with zeros where out of bounds
    # h[i-s] for all i: this is h shifted by s, with zeros where out of bounds
    
    # For a shift s:
    #   term1[i] = 1-h[i+s] if 0<=i+s<n else 0  → this is (1-h) with the first max(0,-s) elements zeroed and last max(0,s) elements zeroed, shifted
    #   Actually: term1[i] = (1-h)[i+s] if valid else 0
    #   term2[i] = h[i-s] if valid else 0
    
    # We can compute this as:
    #   extended_h = pad(h, n-1)  # length 3n-1, h centered
    #   term1[i] = (1 - extended_h[i + (n-1) + s]) if in range
    #   term2[i] = extended_h[i + (n-1) - s] if in range
    
    # Simpler: for each s, the contribution to grad is:
    #   w[s_idx] * dx * ((1-h_shifted_neg_s) - h_shifted_pos_s)
    
    # Let's use the convolution approach:
    # grad[i] = dx * (sum_k w_k - sum_k w_k * h[i+s_k] - sum_k w_k * h[i-s_k])
    #         = dx * (sum_k w_k - (w * h_rev_conv)[i] - (w * h_conv)[i])
    # Hmm, not quite.
    
    # Let me just do it with a vectorized loop over k:
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-17:
            continue
        s = k - (n - 1)
        # term1[i] = 1-h[i+s] for valid i (i+s in [0,n))
        # term2[i] = h[i-s] for valid i (i-s in [0,n))
        
        # term1: i ranges from max(0,-s) to min(n-1, n-1-s)
        i_lo = max(0, -s)
        i_hi = min(n, n - s)  # exclusive
        if i_hi > i_lo:
            idx = np.arange(i_lo, i_hi)
            grad[idx] += wk * (1.0 - h[idx + s])
        
        # term2: i ranges from max(0,s) to min(n-1, n-1+s)
        i_lo2 = max(0, s)
        i_hi2 = min(n, n + s)  # exclusive
        if i_hi2 > i_lo2:
            idx2 = np.arange(i_lo2, i_hi2)
            grad[idx2] += -wk * h[idx2 - s]
    
    return grad
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -1.0, 2.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    hp = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - hp.sum()
    if abs(diff) > 1e-13:
        free = (hp > 1e-6) & (hp < 1.0 - 1e-6)
        cnt = free.sum()
        if cnt > 0:
            hp[free] += diff / cnt
            hp = np.clip(hp, 0.0, 1.0)
