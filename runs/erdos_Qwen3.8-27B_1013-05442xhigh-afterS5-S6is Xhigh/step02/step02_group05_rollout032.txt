```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def smooth_max_grad(h, beta):
    """Gradient of smooth_max(correlate(h, 1-h), beta) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    shifted = beta * (corr - np.max(corr))
    w = np.exp(shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    for i in range(n):
        for k in range(2 * n - 1):
            m_lag = k - (n - 1)
            if 0 <= i - m_lag < n:
                grad[i] += w[k] * (1.0 - h[i - m_lag])
            if 0 <= i + m_lag < n:
                grad[i] -= w[k] * h[i + m_lag]
    grad *= dx
    return grad

def project_to_constraint(y, target):
    """Project y onto {h in [0,1]^n : sum(h) = target}"""
    y = np.clip(y, 0.0, 1.0)
    # Bisection for the shift
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    return x

def projected_gradient_descent(h0, target, beta, n_iters, dx):

    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    step = 0.1

                h = h_new
                best_h = h_new.copy()
                best_val = val
                step = min(step * 1.2, 1.0)
                break
            step *= 0.5
        else:
            # No improvement, reset step
            step = max(step * 2.0, 0.01)
    
    return best_h, best_val
```

```python
def smooth_max_val(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    shifted = beta * (corr - np.max(corr))
    w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for i in range(n):
        for k in range(2 * n - 1):
            m_lag = k - (n - 1)
            if 0 <= i - m_lag < n:
                grad[i] += w[k] * (1.0 - h[i - m_lag])
            if 0 <= i + m_lag < n:
                grad[i] -= w[k] * h[i + m_lag]
    return grad * dx

def project_to_constraint(y, target):
    y = np.clip(y, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_beta = 100.0
_g_target = 0.0
_g_n = 70

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    shifted = beta * (corr - np.max(corr))
    w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for i in range(n):
        for k in range(2 * n - 1):
            m_lag = k - (n - 1)
            if 0 <= i - m_lag < n:
                grad[i] += w[k] * (1.0 - h[i - m_lag])
            if 0 <= i + m_lag < n:
                grad[i] -= w[k] * h[i + m_lag]
    return grad * dx

def project_to_constraint(y, target):
    y = np.clip(y, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def _obj(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max_val(corr, _g_beta) * (2.0 / _g_n)

def _obj_grad(h):
    return smooth_max_grad(h, _g_beta)

def _con_fun(h):
    return np.sum(h) - _g_target

def _con_jac(h):
    return np.ones(_g_n)

def pd_refine(h0, target, max_iters, beta):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    step = 0.05
    for it in range(max_iters):
        grad = smooth_max_grad(h, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        grad_norm = grad / gn
        accepted = False
        for _ in range(20):
            h_new = project_to_constraint(h - step * grad_norm, target)
            val = compute_c5(h_new)
            if val < best_val - 1e-14:
                h = h_new
                best_h = h_new.copy()
                best_val = val
                step = min(step * 1.25, 0.5)
                accepted = True
                break
            step *= 0.5
        if not accepted:
            step = max(step * 1.8, 0.01)
    return best_h, best_val

def slsqp_step(h0, target, beta, n, maxiter, ftol):
    global _g_beta, _g_target, _g_n
    _g_beta = float(beta)
    _g_target = float(target)
    _g_n = int(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h0, jac=_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_constraint(res.x, target)
    except Exception:
        return h0

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    else:
        parent = np.full(70, 0.5)

    n = len(parent)
    target = n / 2.0
    h = project_to_constraint(parent, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_n = n

    # Phase 1: PD with increasing beta (quick convergence)
    for beta in [10, 20, 40, 80, 160, 320]:
        if time.time() - t_start > budget_s * 0.15:
            break
        h, c5 = pd_refine(h, target, 300, beta)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    # Phase 2: SLSQP with analytical gradient, sweeping beta
    for beta in [30, 60, 120, 250, 500, 1000, 2000]:
        if time.time() - t_start > budget_s * 0.55:
            break
        h_new = slsqp_step(h, target, beta, n, 3000, 1e-14)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h, best_c5 = h_new.copy(), c5
            h = h_new.copy()
        else:
            h = h_new  # use as warm start for next beta

    # Phase 3: Multi-start perturbations
    for r in range(20):
        if time.time() - t_start > budget_s * 0.85:
            break
        scale = 0.005 + 0.005 * r
        h_pert = best_h + np.random.randn(n) * scale
        h_pert = project_to_constraint(h_pert, target)
        # Quick PD then SLSQP
        h_pert, c5_p = pd_refine(h_pert, target, 80, 100)
        if c5_p < best_c5 + 0.001:
            h_pert = slsqp_step(h_pert, target, 500, n, 1000, 1e-13)
            c5_p = compute_c5(h_pert)
        if c5_p < best_c5:
            best_h, best_c5 = h_pert.copy(), c5_p
            h = h_pert.copy()

    # Phase 4: Final ultra-precise polish
    remaining = budget_s - (time.time() - t_start)
    if remaining > 30:
        h_new = slsqp_step(best_h, target, 5000, n, 10000, 1e-16)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h, best_c5 = h_new.copy(), c5

    # Phase 5: One last attempt with even higher beta
    remaining = budget_s - (time.time() - t_start)
    if remaining > 20:
        h_new = slsqp_step(best_h, target, 10000, n, 5000, 1e-16)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h, best_c5 = h_new.copy(), c5

    return best_h, best_c5, best_n
```
