            grad[:L] -= wi * h[m2:]

    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    dx = 2.0 / n
    c5_cur = compute_c5(h)
    best_obj = c5_cur

        grad_proj = grad - grad.mean()
        gnorm = np.linalg.norm(grad_proj)

def grad_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    w = softmax_weights(corr, beta)
    grad = np.zeros(n)
    wmax = w.max()
    active = w > wmax * 0.01
    for idx in np.where(active)[0]:
        m = int(idx) - (n - 1)
        wi = w[idx]
        if m >= 0:
            L = n - m
            if L > 0:
                grad[:L] += wi * (1.0 - h[m:])
                grad[m:] -= wi * h[:L]
        else:
            m2 = -m
            L = n - m2
            if L > 0:
                grad[m2:] += wi * (1.0 - h[:L])
                grad[:L] -= wi * h[m2:]
    return dx * grad

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = free.sum()
        if nf == 0:
            free = np.ones(len(h), dtype=bool)
            nf = len(h)
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h_init, beta, target, maxiter=500):
    n = len(h_init)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = corr.max()
        xs = np.clip(beta * (corr - m), -500, 500)
        return (m + np.log(np.exp(xs).sum() / len(xs)) / beta) * dx
    
    def obj_grad(x):
        return grad_smooth_max(x, beta)
    
    def con(x):
        return np.sum(x) - target
    
    def con_jac(x):
        return np.ones(x.shape)
    
    constraints = [{'type': 'eq', 'fun': con, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h_init, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_h(res.x, target)
        c5 = compute_c5(h_new)
        return h_new, c5
    except Exception:
        return h_init, compute_c5(h_init)

def coordinate_descent(h, target, max_iter=5000, seed=42):

    c5_cur = compute_c5(h)
    h = h.copy()

        # Random coordinate

            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_cur - 1e-15:
                h = h_trial
                c5_cur = c5_trial
                break
    return h, c5_cur
