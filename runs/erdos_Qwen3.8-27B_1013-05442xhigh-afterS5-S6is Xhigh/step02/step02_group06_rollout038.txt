def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 0.0) & (h < 1.0)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)

def smooth_max(x, beta):
    m = np.max(x)
    if m > 1e10:  # overflow protection
        return m
    shifted = beta * (x - m)

    return m + np.log(np.sum(np.exp(shifted))) / beta

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 0.0) & (h < 1.0)
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        m = (h > 0) & (h < 1)
        if m.sum() > 0:
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def pairwise_search(h, n_iters, rng, temperature=0.0):

    best_score = compute_c5(h)
    
    for it in range(n_iters):

        j2 = rng.randint(n)
        if j == j2:

            continue
        
        n_cand = 15 if (hi - lo) > 0.1 else 25
        candidates = np.linspace(lo, hi, n_cand)

        best_val = old_j
        best_local = old_score

            h[j] = cv
            h[j2] = s - cv
            score = compute_c5(h)
            if score < best_local - 1e-15:
                best_local = score
                best_val = cv

        if temperature > 0 and best_local >= old_score:
            # Try a random candidate

                best_val = cv

        h[j2] = s - best_val

    h_new = h - lr * g
    h_new = h_new + (h_new.sum() - target) / n
    return project_box_sum(h_new, target)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 70
        h_init = np.full(n_init, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_init

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: Explore nearby n values
    n_list = list(range(max(55, n_init - 6), n_init + 7))
    if n_init not in n_list:
        n_list.append(n_init)
    n_list = sorted(set(n_list))
    warmup_end = min(budget_s * 0.18, 150)

    for n in n_list:
        if time.time() - t0 > warmup_end or remaining() < 100:
            break
        if n == n_init:
            h = h_init.copy()
        else:
            x_old = np.linspace(0, 2, n_init, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_new, x_old, h_init)
            h = project_box_sum(h, n / 2.0)
        h, c5 = pairwise_search(h, 4000, rng)
        if c5 < best_c5:
            best_h, best_c5, best_n = h.copy(), c5, n

    n = best_n
    h = best_h.copy()
    target = n / 2.0

    # Phase 2: Intensive pairwise with annealing
    stagnation = 0
    phase = 0
    while remaining() > 80:
        phase += 1
        temp = 0.003 * np.exp(-phase * 0.4)
        n_iters = min(5000, int(remaining() * 8))
        h, c5 = pairwise_search(h, n_iters, rng, temperature=temp)
        if c5 < best_c5 - 1e-13:
            best_h, best_c5 = h.copy(), c5
            stagnation = 0
        else:
            stagnation += n_iters
            if stagnation > 30000:
                h = best_h.copy()
                scale = 0.005 + 0.015 * np.exp(-stagnation / 40000.0)
                h += scale * rng.randn(n)
                h = project_box_sum(h, target)
                stagnation = 0

    # Phase 3: Directed gradient steps
    if remaining() > 50:
        h = best_h.copy()
        for lr in [0.008, 0.004, 0.002, 0.001, 0.0005]:
            if remaining() < 8:
                break
            for _ in range(300):
                h = directed_step(h, n, lr, target)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h, best_c5 = h.copy(), c5
        # Follow with pairwise
        h, c5 = pairwise_search(best_h, 3000, rng)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    # Phase 4: L-BFGS-B polish
    if remaining() > 25:
        bnds = [(0.0, 1.0)] * n
        for beta, lam in [(400, 500.0), (1000, 2000.0), (2000, 5000.0)]:
            if remaining() < 10:
                break
            try:
                r = minimize(lbfgs_obj, best_h, args=(n, target, beta, lam),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-12})
                h_new = project_box_sum(r.x, target)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new, c5
            except Exception:
                pass

    # Phase 5: Final dense pairwise polish
    if remaining() > 8:
        n_final = int(min(15000, remaining() * 40))
        h, c5 = pairwise_search(best_h, n_final, rng)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
