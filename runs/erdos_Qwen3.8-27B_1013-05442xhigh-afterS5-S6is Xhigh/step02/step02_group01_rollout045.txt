    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad


def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0

    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v

    def objg(x):
        _, g = smoothmax_val_grad(x, beta)
        return g

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)


def nesterov_subgrad(h, max_iters, lr0, decay, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx.astype(int) - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])
        grad = grad / top_k * dx
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def make_block_pattern(n, period):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < period // 2:
            h[i] = 1.0
    return project_feasible(h, n / 2.0)


def perturb_gauss(h, sigma, rng):
    h_new = h + rng.normal(0, sigma, len(h))
    return project_feasible(h_new, len(h) / 2.0)


def perturb_block(h, rng, n_blocks):
    n = len(h)
    h_new = h.copy()
    bsize = max(1, n // n_blocks)
    idx = rng.choice(n // bsize - 1, 2, replace=False)
    a, b = int(idx[0]), int(idx[1])
    tmp = h_new[a*bsize:(a+1)*bsize].copy()
    h_new[a*bsize:(a+1)*bsize] = h_new[b*bsize:(b+1)*bsize]
    h_new[b*bsize:(b+1)*bsize] = tmp
    return project_feasible(h_new, n / 2.0)


def perturb_invert(h, rng, n_blocks):
    n = len(h)
    h_new = h.copy()
    bsize = max(1, n // n_blocks)
    idx = rng.randint(0, n // bsize - 1)
    a = int(idx)
    h_new[a*bsize:(a+1)*bsize] = 1.0 - h_new[a*bsize:(a+1)*bsize]
    return project_feasible(h_new, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = make_block_pattern(200, 8)

    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: Initial polish
    if tl() > 40:
        h_t = slsqp_refine(h_best, maxiter=500, beta=200)
        c_t = compute_c5(h_t)
        if c_t < best_c5:
            h_best, best_c5 = h_t, c_t

    # Phase 2: Nesterov exploration
    if tl() > 80:
        h_n, c_n = nesterov_subgrad(h_best, 5000, 0.05, 0.9999, top_k=10)
        if c_n < best_c5:
            h_best, best_c5 = h_n, c_n
        h_t = slsqp_refine(h_best, maxiter=400, beta=300)
        c_t = compute_c5(h_t)
        if c_t < best_c5:
            h_best, best_c5 = h_t, c_t

    # Phase 3: Multi-start with structured patterns
    if tl() > 100:
        patterns = []
        for p in [4, 6, 8, 10, 12, 16, 20, 25, 33, 40, 50]:
            patterns.append(make_block_pattern(n, p))
        for _ in range(8):
            hp = rng.random(n)
            patterns.append(project_feasible(hp, n / 2.0))
        for pat in patterns:
            if tl() < 60:
                break
            h_opt = slsqp_refine(pat, maxiter=200, beta=100)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                h_best, best_c5 = h_opt, c_opt
                h_opt2 = slsqp_refine(h_best, maxiter=300, beta=250)
                c_opt2 = compute_c5(h_opt2)
                if c_opt2 < best_c5:
                    h_best, best_c5 = h_opt2, c_opt2

    # Phase 4: Iterated Local Search (ILS) with annealing
    n_ilsw = 30
    for i in range(n_ilsw):
        if tl() < 50:
            break
        temp = max(0.02, 0.12 * (1.0 - i / n_ilsw))
        mode = i % 4
        if mode == 0:
            h_pert = perturb_gauss(h_best, temp, rng)
        elif mode == 1:
            h_pert = perturb_block(h_best, rng, 8)
        elif mode == 2:
            h_pert = perturb_invert(h_best, rng, 6)
        else:
            h_pert = perturb_gauss(h_best, temp * 2, rng)

        h_opt = slsqp_refine(h_pert, maxiter=250, beta=150)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            h_best, best_c5 = h_opt, c_opt
            h_opt2 = slsqp_refine(h_best, maxiter=400, beta=350)
            c_opt2 = compute_c5(h_opt2)
            if c_opt2 < best_c5:
                h_best, best_c5 = h_opt2, c_opt2

    # Phase 5: Higher resolution
    for n_try in [300, 400, 500]:
        if tl() < 150:
            break
        h_interp = interp_to_n(h_best, n_try)
        h_opt = slsqp_refine(h_interp, maxiter=300, beta=150)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            h_best, best_c5 = h_opt, c_opt
            if tl() > 100:
                h_n2, c_n2 = nesterov_subgrad(h_best, 2000, 0.015, 0.99995, top_k=5)
                if c_n2 < best_c5:
                    h_best, best_c5 = h_n2, c_n2
                h_opt2 = slsqp_refine(h_best, maxiter=400, beta=300)
                c_opt2 = compute_c5(h_opt2)
                if c_opt2 < best_c5:
                    h_best, best_c5 = h_opt2, c_opt2
        else:
            # Try ILS at this resolution too
            for i in range(5):
                if tl() < 40:
                    break
                h_pert = perturb_gauss(h_interp, 0.05, rng)
                h_opt = slsqp_refine(h_pert, maxiter=200, beta=120)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c5:
                    h_best, best_c5 = h_opt, c_opt
                    h_opt2 = slsqp_refine(h_best, maxiter=300, beta=300)
                    c_opt2 = compute_c5(h_opt2)
                    if c_opt2 < best_c5:
                        h_best, best_c5 = h_opt2, c_opt2

    # Phase 6: Final aggressive polish
    if tl() > 20:
        h_f = slsqp_refine(h_best, maxiter=800, beta=600)
        c_f = compute_c5(h_f)
        if c_f < best_c5:
            h_best, best_c5 = h_f, c_f

    # One more SLSQP with even higher beta
    if tl() > 15:
        h_f = slsqp_refine(h_best, maxiter=500, beta=1000)
        c_f = compute_c5(h_f)
        if c_f < best_c5:
            h_best, best_c5 = h_f, c_f

    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
