    """Compute (1,0) and (0,1) counts for all positive lags of binary h."""

The difference is non-zero only when either p ∈ S or p + m - (n-1) ∈ S.

Case 1: p ∈ S and p + m - (n-1) ∉ S
- For p = i: h_new[i]=0, h[i]=1. (1-h_new)[i+m-(n-1)] = (1-h)[i+m-(n-1)] (since the index is not in S).
  Contribution to Δ: 0*(1-h)[...] - 1*(1-h)[...] = -(1-h)[i+m-(n-1)]
- For p = j: h_new[j]=1, h[j]=0. 
  Contribution: 1*(1-h)[j+m-(n-1)] - 0 = +(1-h)[j+m-(n-1)]

Case 2: p ∉ S and p + m - (n-1) ∈ S

- For q = i: (1-h_new)[i]=1, (1-h)[i]=0. h[p] = h_new[p] (p ∉ S).
  Contribution: h[p]*1 - h[p]*0 = +h[p] = h[q-m+(n-1)]
- For q = j: (1-h_new)[j]=0, (1-h)[j]=1.
  Contribution: h[p]*0 - h[p]*1 = -h[p] = -h[q-m+(n-1)]

Case 3: p ∈ S and p + m - (n-1) ∈ S
- This happens when both p and p+m-(n-1) are in {i, j}.
- Sub-case 3a: p=i, p+m-(n-1)=j, i.e., m = (n-1) + j - i.
  h_new[i]=0, (1-h_new)[j]=0. h[i]=1, (1-h)[j]=1.
  Contribution: 0*0 - 1*1 = -1
- Sub-case 3b: p=j, p+m-(n-1)=i, i.e., m = (n-1) + i - j.
  h_new[j]=1, (1-h_new)[i]=1. h[j]=0, (1-h)[i]=0.
  Contribution: 1*1 - 0*0 = +1
- Sub-case 3c: p=i, i+m-(n-1)=i, i.e., m=0.
  h_new[i]=0, (1-h_new)[i]=1. h[i]=1, (1-h)[i]=0.
  Contribution: 0*1 - 1*0 = 0
- Sub-case 3d: p=j, j+m-(n-1)=j, i.e., m=0.
  h_new[j]=1, (1-h_new)[j]=0. h[j]=0, (1-h)[j]=1.
  Contribution: 1*0 - 0*1 = 0

So for m=0: Δcorr[0] = 0 (makes sense, since binary sequence has 0 overlap at lag 0 regardless of arrangement).

For m ≠ 0:
Δcorr[m] = -(1-h)[i+m-(n-1)] * valid(i+m-(n-1)) 
          + (1-h)[j+m-(n-1)] * valid(j+m-(n-1))
          + h[i-m+(n-1)] * valid(i-m+(n-1)) * [i-m+(n-1) ∉ S]

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    count = int(h_bin.sum())
    if count > target:
        ones = np.where(h_bin == 1)[0]
        flip = rng.choice(ones, count - target, replace=False)
        h_bin[flip] = 0
    elif count < target:
        zeros = np.where(h_bin == 0)[0]
        flip = rng.choice(zeros, target - count, replace=False)
        h_bin[flip] = 1
    return h_bin

def binary_steepest_descent(h, time_limit, rng):
    n = len(h)
    h = h.copy()
    start = time.time()
    best_c5 = compute_c5(h)
    
    while time.time() - start < time_limit:
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        
        best_delta = 1e-10
        best_move = None
        
        for i in ones_idx:
            for j in zeros_idx:
                h_trial = h.copy()
                h_trial[i] = 0
                h_trial[j] = 1
                new_max = np.max(np.correlate(h_trial, 1.0 - h_trial, mode="full"))
                delta = new_max - max_val
                if delta < best_delta:
                    best_delta = delta
                    best_move = (i, j)
                if best_delta < -1:  # Can't get better than -1 for integer counts
                    break
            if best_delta < -1:
                break
        
        if best_move is not None:
            h[best_move[0]] = 0
            h[best_move[1]] = 1
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
        else:
            break
    
    return h, best_c5

def binary_annealing(h, time_limit, rng):
    n = len(h)
    h = h.copy()
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c5_cur = best_c5
    
    T = 2.0
    T_min = 0.1
    cooling = 0.999
    
    while time.time() - start < time_limit:
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = int(rng.choice(ones_idx))
        j = int(rng.choice(zeros_idx))
        
        h_trial = h.copy()
        h_trial[i] = 0
        h_trial[j] = 1
        c5_trial = compute_c5(h_trial)
        
        delta = c5_trial - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
            h = h_trial
            c5_cur = c5_trial
            if c5_cur < best_c5:
                best_h = h.copy()
                best_c5 = c5_cur
        
        T = max(T * cooling, T_min)
    
    return best_h, best_c5

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def cont_obj(x, beta, n):
    h_arr = np.clip(x, 0, 1)
    dx = 2.0 / n
    corr = np.correlate(h_arr, 1.0 - h_arr, mode="full")
    return smooth_max(corr, beta) * dx
