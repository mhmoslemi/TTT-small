    h = np.clip(h, 0.0, 1.0)

            if total < 1e-15:

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def grad_lag(h, n, lag):
    g = np.zeros(n)
    a = np.arange(n)
    if lag == 0:
        return 1.0 - 2.0 * h
    elif lag > 0:
        idx1 = np.arange(lag, n)
        g[idx1] += (1.0 - h[idx1 - lag])
        idx2 = np.arange(0, n - lag)
        g[idx2] += (-h[idx2 + lag])
    else:
        kp = -lag
        idx1 = np.arange(0, n - kp)
        g[idx1] += (1.0 - h[idx1 + kp])
        idx2 = np.arange(kp, n)
        g[idx2] += (-h[idx2 - kp])
    return g

def subgradient_descent(h, n, lr0, decay, momentum, n_steps, t0, budget_s, active_M):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    vel = np.zeros(n)
    mu = momentum
    for step in range(n_steps):
        if budget_s - (time.time() - t0) < 10:
            break
        c = np.correlate(h, 1 - h, mode='full')
        top_idx = np.argsort(c)[-active_M:]
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            g += grad_lag(h, n, lag)
        g /= active_M
        lr = lr0 / (1.0 + decay * step)

        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
    return best_h, best_c5

        else:
            h[i], h[j] = hi - d, hj + d
        E *= er
        D *= dr
    return best_h, best_c5

def coordinate_descent(h, n, n_pass, t0, budget_s):
    best_c5 = compute_c5(h, n)
    for _ in range(n_pass):
        if budget_s - (time.time() - t0) < 10:
            break
        improved = False
        for i in range(n):
            if budget_s - (time.time() - t0) < 10:
                break
            for j in range(i + 1, n):
                hi_opt = h[i]
                for frac in [0.3, 0.5, 0.7, 0.1, 0.9]:
                    new_i = h[i] + (h[j] - h[i]) * frac
                    new_j = h[j] - (h[j] - h[i]) * frac
                    if 0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0:
                        h[i], h[j] = new_i, new_j
                        c = compute_c5(h, n)

        if not improved:

    return h, best_c5
