        return g

             'jac': lambda x: np.ones(n)}]

                       bounds=bounds, constraints=cons,

    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]

        x = x_new

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -60.0, 60.0)
    exps = np.exp(shifted)
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    j = np.arange(n)
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    return val, grad

def slsqp_refine(h, maxiter, beta, ftol=1e-15):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        return smoothmax_val_grad(x, beta)[0]
    def objg(x):
        return smoothmax_val_grad(x, beta)[1]
    def con_fun(x):
        return x.sum() - target
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def lbfgs_refine(h, maxiter, beta, penalty):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    ones = np.ones(n)
    def obj(x):
        v, g = smoothmax_val_grad(x, beta)
        c = x.sum() - target
        return v + 0.5 * penalty * c * c
    def objg(x):
        v, g = smoothmax_val_grad(x, beta)
        c = x.sum() - target
        return g + penalty * c * ones
    try:

                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12,
                                'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def nesterov_subgrad(h, max_iters, lr0, decay, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])
        grad = grad / top_k * dx
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])
        grad = grad / top_k * dx
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def local_perturb(h, n_trials, scale):
    n = len(h)
    target = n / 2.0
    best = h.copy()
    best_v = compute_c5(h)
    for _ in range(n_trials):
        k = max(1, n // 25)
        idx = np.random.choice(n, k, replace=False)
        h_try = best.copy()
        h_try[idx] += scale * np.random.randn(k)
        h_try = project_feasible(h_try, target)
        v = compute_c5(h_try)
        if v < best_v:
            best = h_try.copy()
            best_v = v
    return best, best_v

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(70, 0.5)

    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: Subgradient warmup
    if tl() > 60:
        h1, c1 = nesterov_subgrad(h_best, 3000, 0.05, 0.9996, top_k=5)
        if c1 < best_c5:
            h_best, best_c5 = h1, c1

    # Phase 2: SLSQP annealing schedule
    for b in [60, 120, 250, 500, 800, 1200, 2000]:
        if tl() < 40:
            break
        h2 = slsqp_refine(h_best, maxiter=600, beta=b)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            h_best, best_c5 = h2, c2

    # Phase 3: Multi-start with diverse perturbations
    for trial in range(8):
        if tl() < 100:
            break
        if trial == 0:
            h_pert = h_best + 0.04 * np.random.randn(n)
        elif trial == 1:
            h_pert = h_best.copy()
            st = np.random.randint(0, n // 3)
            ln = np.random.randint(n // 8, n // 3)
            en = min(st + ln, n)
            h_pert[st:en] += 0.12 * np.random.randn(en - st)
        elif trial == 2:
            freq = np.random.randint(2, 7)
            x_arr = np.arange(n) / n * 2.0 * np.pi
            h_pert = h_best + 0.05 * np.sin(freq * x_arr + np.random.rand() * np.pi)
        elif trial == 3:
            h_pert = h_best.copy()
            h_pert[h_pert < 0.35] -= 0.15
            h_pert[h_pert > 0.65] += 0.15
        elif trial == 4:
            h_pert = h_best + 0.06 * np.random.randn(n)
        elif trial == 5:
            h_pert = h_best.copy()
            perm = np.random.permutation(n)
            h_pert[perm[:n//4]] += 0.08
            h_pert[perm[3*n//4:]] -= 0.08
        elif trial == 6:
            h_pert = h_best + 0.03 * np.random.randn(n)
        else:
            h_pert = h_best.copy()
            h_pert[np.random.choice(n, n//5, replace=False)] = np.random.rand(n//5)
        h_pert = project_feasible(h_pert, n / 2.0)

        if tl() > 70:
            h_nr, c_nr = nesterov_subgrad(h_pert, 1500, 0.03, 0.9997, top_k=5)
            if c_nr < best_c5:
                h_best, best_c5 = h_nr, c_nr
        if tl() > 35:
            h_pol = slsqp_refine(h_best, maxiter=500, beta=600)
            c_pol = compute_c5(h_pol)
            if c_pol < best_c5:
                h_best, best_c5 = h_pol, c_pol

    # Phase 4: L-BFGS-B with penalty (faster for high effective beta)
    if tl() > 80:
        for b, pen in [(800, 80.0), (1500, 150.0), (3000, 400.0)]:
            if tl() < 35:
                break
            h_lb = lbfgs_refine(h_best, maxiter=5000, beta=b, penalty=pen)
            c_lb = compute_c5(h_lb)
            if c_lb < best_c5:
                h_best, best_c5 = h_lb, c_lb

    # Phase 5: Local perturbation search + final SLSQP
    if tl() > 30:
        h_loc, c_loc = local_perturb(h_best, 80, 0.015)
        if c_loc < best_c5:
            h_best, best_c5 = h_loc, c_loc

    if tl() > 25:
        h_f = slsqp_refine(h_best, maxiter=1000, beta=3000)
        c_f = compute_c5(h_f)
        if c_f < best_c5:
            h_best, best_c5 = h_f, c_f

    # Phase 6: Resolution changes
    for n_try in [150, 250, 300]:
        if tl() < 70:
            break
        h_interp = interp_to_n(h_best, n_try)
        if tl() > 50:
            h_nr, c_nr = nesterov_subgrad(h_interp, 1500, 0.03, 0.9997, top_k=5)
            if c_nr < best_c5:
                h_best, best_c5 = h_nr, c_nr
                h_interp = h_best.copy()
        if tl() > 35 and len(h_best) == n_try:
            h_pol = slsqp_refine(h_interp, maxiter=400, beta=800)
            c_pol = compute_c5(h_pol)
            if c_pol < best_c5:
                h_best, best_c5 = h_pol, c_pol

    # Phase 7: Final aggressive polish if time remains
    if tl() > 30:
        h_f2 = slsqp_refine(h_best, maxiter=1500, beta=5000)
        c_f2 = compute_c5(h_f2)
        if c_f2 < best_c5:
            h_best, best_c5 = h_f2, c_f2

    # Verify feasibility
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)

    return h_best, best_c5, len(h_best)
