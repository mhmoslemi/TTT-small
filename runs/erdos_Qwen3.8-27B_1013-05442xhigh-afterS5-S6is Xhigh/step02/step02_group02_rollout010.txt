        d = rng.uniform(0.001, D) * (1 if rng.random() < 0.5 else -1)
        hi, hj = h[i]+d, h[j]-d
        if not (0 <= hi <= 1 and 0 <= hj <= 1): continue
        h[i], h[j] = hi, hj
        new_c5 = compute_c5(h, n)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta/max(E,1e-10)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = h.copy(), new_c5
        else:
            h[i], h[j] = hi-d, hj+d
        E *= 0.9998
        D *= 0.9999
    return best_h, best_c5
