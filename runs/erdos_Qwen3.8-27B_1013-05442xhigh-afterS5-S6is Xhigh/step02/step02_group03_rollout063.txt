```python
    # Try pairs of flips: flip h[i] 0->1 and h[j] 1->0
    best = h_binary.copy()
    best_corr = compute_corr_array(best)
    best_max = np.max(best_corr)
    
    ones = np.where(h_binary == 1)[0]
    zeros = np.where(h_binary == 0)[0]
    
    for it in range(num_iterations):
        # Pick random i from zeros, j from ones
        i = zeros[np.random.randint(len(zeros))]
        j = ones[np.random.randint(len(ones))]
        
        # Try the flip
        h_new = h_binary.copy()
        h_new[i] = 1
        h_new[j] = 0
        
        corr_new = compute_corr_array(h_new)
        max_new = np.max(corr_new)
        
        if max_new < best_max:
            h_binary = h_new
            best = h_new.copy()
            best_max = max_new
            best_corr = corr_new
            # Update ones and zeros sets
            ones = np.where(h_binary == 1)[0]
            zeros = np.where(h_binary == 0)[0]
    
    return best, best_max
```

For the local search, I'll use a simulated annealing approach:
- At each step, pick a random pair (a from zeros, b from ones)
- Compute the new max correlation
- Accept if it improves, or with probability exp(-Δ/T) if it worsens
- Decrease T over time

Let me also think about a "smart" local search:
- Identify the worst lag(s)
- Try to reduce the correlation at those lags specifically
- This means finding pairs (a, b) such that flipping them reduces corr[worst_lag]

For a specific lag k, corr[k] = sum_i h[i]*(1-h[i+k]). To reduce this:
- Find i where h[i]=1, h[i+k]=0 (these contribute 1 to corr[k])
- Flip h[i] from 1 to 0 (and some h[b] from 0 to 1 to maintain constraint)
- This reduces corr[k] by 1 but might increase other lags

```python
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    if m == 0:
        return 0.0
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, target_sum, n):
    return h.sum() - target_sum

def con_jac(h, target_sum, n):
    return np.ones(n)

def binary_local_search(h0, n, n_iters, rng, T0=5.0, T1=0.1):
    """Simulated annealing on binary sequences with pair flips."""
    h = h0.copy()
    best_h = h.copy()
    corr = np.correlate(h, 1.0-h, mode='full')
    best_max = float(np.max(corr))
    cur_max = best_max
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_iters):
        T = T0 * (T1/T0) ** (it / max(n_iters, 1))
        
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        h[i] = 1
        h[j] = 0
        
        corr_new = np.correlate(h, 1.0-h, mode='full')
        max_new = float(np.max(corr_new))
        
        delta = max_new - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_max = max_new
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
            # Update ones/zeros
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[i] = 0
            h[j] = 1
    
    return best_h, best_max
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, target_sum, n):
    return h.sum() - target_sum

def con_jac(h, target_sum, n):
    return np.ones(n)

def make_binary_init(n, rng, seed_type=0):
    """Create a binary initialization with exactly n/2 ones."""
    h = np.zeros(n)
    if seed_type == 0:
        idx = rng.choice(n, n//2, replace=False)
        h[idx] = 1
    elif seed_type == 1:
        # Alternating with perturbation
        h[::2] = 1
        h[1::2] = 0
        if n % 2 == 1:
            h[-1] = 0
        # Random swap to break symmetry
        for _ in range(n // 10):
            i = rng.randint(n)
            j = rng.randint(n)
            h[i], h[j] = h[j], h[i]
        # Fix count
        cnt = h.sum()
        if cnt > n // 2:
            ones = np.where(h == 1)[0]
            for i in ones[cnt - n//2:]:
                h[i] = 0
        elif cnt < n // 2:
            zeros = np.where(h == 0)[0]
            for i in zeros[:n//2 - int(cnt)]:
                h[i] = 1
    elif seed_type == 2:
        # Block pattern: 11..100..0 with specific block sizes
        block = n // 4
        h[:block] = 1
        h[2*block:3*block] = 1
        # Fix count
        cnt = int(h.sum())
        if cnt > n // 2:
            ones = np.where(h == 1)[0]
            rng.shuffle(ones)
            for i in ones[:cnt - n//2]:
                h[i] = 0
        elif cnt < n // 2:
            zeros = np.where(h == 0)[0]
            rng.shuffle(zeros)
            for i in zeros[:n//2 - cnt]:
                h[i] = 1
    return h

def binary_local_search(h0, n, n_iters, rng, T0=3.0, T1=0.01):
    """Simulated annealing on binary sequences."""
    h = h0.copy()
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_max = float(np.max(corr))
    cur_max = best_max
    
    for it in range(n_iters):
        T = T0 * (T1 / T0) ** (it / max(n_iters - 1, 1))
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        h[i] = 1.0
        h[j] = 0.0
        
        corr_new = np.correlate(h, 1.0 - h, mode='full').astype(float)
        max_new = float(np.max(corr_new))
        
        delta = max_new - cur_max
        if delta < 0 or (T > 0.01 and rng.random() < np.exp(-delta / T)):
            cur_max = max_new
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            h[i] = 0.0
            h[j] = 1.0
    
    return best_h, best_max

def slsqp_refine(h0, n, time_left_fn, max_time=30):
    """Refine a solution using SLSQP with smooth max continuation."""
    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_feasible(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
    
    for beta in [10, 100, 500, 2000, 10000]:
        if time_left_fn() < 10:
            break
        try:
            res = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-13})
            h_new = project_feasible(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
        except Exception:
            pass
    
    return best_h, best_c5

def subgradient_refine(h0, n, n_iters, time_left_fn):
    """Subgradient method with momentum."""
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h0.copy()
    v = np.zeros(n)
    momentum = 0.9
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for it in range(n_iters):
        if time_left_fn() < 10:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_worst = int(np.argmax(corr))
        lag = k_worst - (n - 1)
        
        grad = np.zeros(n)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
        
        lr = 0.002 / (1.0 + it * 0.001)
        v = momentum * v - lr * grad
        h = h + v
        h = project_feasible(h, target_sum)
        
        if it % 5000 == 4999:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_h = h.copy()
        best_c5 = c5
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)
    t0 = time.time()

        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    def tl():
        return budget_s - (time.time() - t0)
    
    # Phase 1: Binary local search at the initial resolution
    n_init = len(h_init)
    h_bin = (h_init > 0.5).astype(float)
    # Fix count
    cnt = int(h_bin.sum())
    if cnt > n_init // 2:
        ones = np.where(h_bin == 1)[0]
        rng.shuffle(ones)
        h_bin[ones[cnt - n_init//2:]] = 0
    elif cnt < n_init // 2:
        zeros = np.where(h_bin == 0)[0]
        rng.shuffle(zeros)
        h_bin[zeros[:n_init//2 - cnt]] = 1
    
    n_sa = max(20000, min(200000, int(tl() // 20)))
    h_bin, max_corr = binary_local_search(h_bin, n_init, n_sa, rng)
    c5_bin = max_corr * 2.0 / n_init
    if c5_bin < best_c5:
        best_h = h_bin.copy()
        best_c5 = c5_bin
    
    # Phase 2: SLSQP refinement
    if tl() > 30:
        best_h, best_c5 = slsqp_refine(best_h, n_init, tl)
    
    # Phase 3: Subgradient refinement
    if tl() > 30:
        best_h, best_c5 = subgradient_refine(best_h, n_init, 50000, tl)
    
    # Phase 4: Try other resolutions
```

```python
def smooth_max(x, beta):

    shifted = np.clip(beta * (x - m), -700.0, 700.0)
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def con_fun(h, ts, n):
    return h.sum() - ts

def con_jac(h, ts, n):
    return np.ones(n)

def to_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    cnt = int(h_bin.sum())
    if cnt > target:
        ones = np.where(h_bin == 1)[0]
        rng.shuffle(ones)
        h_bin[ones[cnt - target:]] = 0
    elif cnt < target:
        zeros = np.where(h_bin == 0)[0]
        rng.shuffle(zeros)
        h_bin[zeros[:target - cnt]] = 1
    return h_bin

def binary_sa(h0, n, n_iters, rng, T0=8.0, T1=0.05):
    h = h0.copy()
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    best_max = float(np.max(corr))
    cur_max = best_max
    for it in range(n_iters):
        T = T0 * (T1 / T0) ** (it / max(n_iters - 1, 1))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h[i] = 1.0
        h[j] = 0.0
        c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
        mx = float(np.max(c))
        delta = mx - cur_max
        if delta < 0 or (T > 0.02 and rng.random() < np.exp(-delta / T)):
            cur_max = mx
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            h[i] = 0.0
            h[j] = 1.0
    return best_h, best_max

def slsqp_refine(h0, n, time_left_fn):
    dx = 2.0 / n
    ts = n / 2.0
    h = project_feasible(h0.astype(float).copy(), ts)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts, n)}]
    for beta in [5, 50, 200, 1000, 5000, 20000]:
        if time_left_fn() < 8:
            break
        for trial in range(2):
            if time_left_fn() < 5:
                break
            x0 = best_h.copy()
            if trial > 0:
                x0 += np.random.randn(n) * 0.005
                x0 = project_feasible(x0, ts)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 4000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, ts)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new
                    best_c5 = c5
            except Exception:
                pass
    return best_h, best_c5

def coord_descent(h0, n, n_passes, t_budget):
    t0 = time.time()
    dx = 2.0 / n
    ts = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    x_grid = np.linspace(0.0, 1.0, 150)
    for p in range(n_passes):
        if p * len(x_grid) > t_budget * 50:
            break
        if time.time() - t0 > t_budget:
            break
        corr_cur = np.correlate(h, 1.0 - h, mode='full')
        for i in range(n):
            a = np.zeros(2 * n - 1)
            for k in range(1, n):
                ip = n - 1 + k
                im = n - 1 - k
                if i + k < n:
                    a[ip] += (1.0 - h[i + k])
                    a[im] -= h[i + k]
                if i - k >= 0:
                    a[ip] -= h[i - k]
                    a[im] += (1.0 - h[i - k])
            b = corr_cur - a * h[i]
            b[n - 1] = corr_cur[n - 1] + h[i] ** 2 - h[i]
            best_val = np.inf
            best_x = h[i]
            for x in x_grid:
                c = a * x + b
                c[n - 1] = -x * x + x + b[n - 1]
                v = np.max(c)
                if v < best_val:
                    best_val = v
                    best_x = x
            h[i] = best_x
            corr_cur = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, ts)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    best_h = project_feasible(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def subgrad_refine(h0, n, n_iters, time_left_fn):
    dx = 2.0 / n
    ts = n / 2.0
    h = h0.copy()
    v = np.zeros(n)
    mu = 0.85
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(n_iters):
        if time_left_fn() < 10:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_worst = int(np.argmax(corr))
        lag = k_worst - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            idx = np.arange(n - lag)
            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
        lr = 0.003 / (1.0 + 0.0005 * it)
        v = mu * v - lr * grad
        h = h + v
        h = project_feasible(h, ts)
        if it % 3000 == 2999:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_h = h.copy()
        best_c5 = c5
    return best_h, best_c5

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2.0, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    n_main = len(h_init)

    # Phase 1: Binary SA at main resolution
    h_bin = to_binary(h_init, n_main, rng)
    n_sa = int(min(50_000_000, max(1_000_000, tl() * 1_000_000 / 30)))
    h_bin, max_corr = binary_sa(h_bin, n_main, n_sa, rng)
    c5_bin = max_corr * 2.0 / n_main
    if c5_bin < best_c5:
        best_h = h_bin.copy()
        best_c5 = c5_bin

    # Phase 2: SLSQP from best
    if tl() > 40:
        best_h, best_c5 = slsqp_refine(best_h, n_main, tl)

    # Phase 3: Coordinate descent
    if tl() > 30:
        cd_budget = min(60.0, tl() * 0.25)
        best_h, best_c5 = coord_descent(best_h, n_main, 30, cd_budget)

    # Phase 4: Subgradient refinement
    if tl() > 30:
        best_h, best_c5 = subgrad_refine(best_h, n_main, int(min(200000, tl() * 3000)), tl)

    # Phase 5: SLSQP + CD again
    if tl() > 30:
        best_h, best_c5 = slsqp_refine(best_h, n_main, tl)
        if tl() > 20:
            best_h, best_c5 = coord_descent(best_h, n_main, 20, min(40.0, tl() * 0.5))

    # Phase 6: Try other resolutions
    try_res = [n_main + 10, n_main - 10, n_main + 20, n_main + 40,
              n_main - 20, 50, 80, 100, 120, 60, 90, 130]
    for n_try in try_res:
        if n_try < 30 or n_try > 200 or tl() < 40:
            continue
        h_t = resample_h(best_h, n_try)
        ts_t = n_try / 2.0
        h_t = project_feasible(h_t, ts_t)
        h_bin_t = to_binary(h_t, n_try, rng)
        n_sa_t = int(min(5_000_000, max(500_000, tl() * 50_000)))
        h_bin_t, mc_t = binary_sa(h_bin_t, n_try, n_sa_t, rng)
        c5_t = mc_t * 2.0 / n_try
        if c5_t < best_c5 or c5_t < best_c5 + 0.005:
            h_ref, c5_ref = slsqp_refine(h_bin_t, n_try, tl)
            if c5_ref < best_c5:
                best_h = h_ref
                best_c5 = c5_ref
            if tl() > 20:
                h_cd, c5_cd = coord_descent(best_h, len(best_h), 15, min(30.0, tl() * 0.4))
                if c5_cd < best_c5:
                    best_h = h_cd
                    best_c5 = c5_cd

    # Phase 7: Iterated perturbation + SLSQP
    for pert in [0.01, 0.005, 0.002, 0.001]:
        if tl() < 20:
            break
        n_cur = len(best_h)
        h_p = best_h + rng.randn(n_cur) * pert
        h_p = project_feasible(h_p, n_cur / 2.0)
        h_r, c5_r = slsqp_refine(h_p, n_cur, tl)
        if c5_r < best_c5:
            best_h = h_r
            best_c5 = c5_r

    if tl() > 15:
        best_h, best_c5 = coord_descent(best_h, len(best_h), 30, min(40.0, tl() * 0.5))

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
```
