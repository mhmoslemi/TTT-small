def smooth_max_grad(corr, beta):
    """Returns (smooth_max_value, gradient_weights)"""
    M = len(corr)
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 500)

def project_to_constraint(h, target_sum):

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_constraint(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            if diff > 0:
                order = np.argsort(h)
            else:
                order = np.argsort(-h)
            for i in order:
                if diff > 0:
                    space = 1.0 - h[i]
                    use = min(space, diff)
                    h[i] += use
                    diff -= use
                else:
                    space = h[i]
                    use = min(space, -diff)
                    h[i] -= use
                    diff += use
                if abs(diff) < 1e-14:
                    break
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    M = len(corr)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exp_s = np.exp(shifted)
    weights = exp_s / exp_s.sum()
    value = m + np.log(exp_s.sum()) / beta

    grad = np.zeros(n)
    for m_idx in range(M):
        w = weights[m_idx]
        if w < 1e-12:
            continue
        lag = m_idx - (n - 1)
        if lag >= 0:
            idx1 = np.arange(lag, n)
            grad[idx1] += w * (1.0 - h[idx1 - lag])
            idx2 = np.arange(0, n - lag)
            grad[idx2] -= w * h[idx2 + lag]

            grad[idx1] += w * (1.0 - h[idx1 + L])
            idx2 = np.arange(L, n)
            grad[idx2] -= w * h[idx2 - L]
    grad *= dx
    return value * dx, grad

    """Momentum gradient descent with projection."""
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    velocity = np.zeros_like(cur)
    n = len(cur)
    
    for it in range(n_iter):
        val, grad = compute_smooth_max_and_grad(cur, beta)
        grad = grad - grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            break
        grad = grad / gn
        step = base_step / (1.0 + 0.001 * it)
        velocity = momentum_rate * velocity - step * grad
        h_trial = cur + velocity
        h_trial = project_to_constraint(h_trial, target_sum)
        c5_trial = compute_c5(h_trial)
        if c5_trial < cur_c5 - 1e-15:

                cur = h_trial
                cur_c5 = c5_trial
            else:
                velocity = np.zeros_like(cur)
    return cur, cur_c5
