import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = c.max()
    shifted = c - mx
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    obj = mx + np.log(total) / alpha
    weights = exp_s / (total * alpha)
    n_full = 2 * n - 1
    grad = np.zeros(n)
    for k in range(n_full):
        w = weights[k]
        if w < 1e-15:
            continue
        r = k - (n - 1)
        j_lo1 = max(0, -r)
        j_hi1 = min(n - 1, n - 1 - r)
        if j_lo1 <= j_hi1:
            idx = np.arange(j_lo1, j_hi1 + 1)
            grad[idx] += w * (1.0 - h[idx + r])
        j_lo2 = max(0, r)
        j_hi2 = min(n - 1, n - 1 + r)
        if j_lo2 <= j_hi2:
            idx2 = np.arange(j_lo2, j_hi2 + 1)
            grad[idx2] -= w * h[idx2 - r]
    return obj, grad

def proj_grad_desc(h, n, t0, budget_s, alpha_start, alpha_end, n_steps, lr_init):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    current_c5 = best_c5
    lr = lr_init
    for step in range(n_steps):
        if time.time() - t0 > budget_s - 40:
            break
        frac = step / max(1, n_steps - 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        cur_lr = lr_init * (1.0 - 0.7 * frac)
        obj, grad = lse_obj_grad(h, n, alpha)
        for trial in range(5):
            h_new = h - cur_lr * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < current_c5:
                current_c5 = c_new
                h = h_new
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h.copy()
                break
            cur_lr *= 0.4
    return best_h, best_c5

def coord_descent(h, n, t0, budget_s, max_rounds=20):
    best_c5 = compute_c5(h, n)
    half = n // 2
    for rnd in range(max_rounds):
        if time.time() - t0 > budget_s - 15:
            break
        for i in range(n):
            if time.time() - t0 > budget_s - 15:
                break
            j = (i + half) % n
            if i == j:
                j = (i + 1) % n
            hi_old = h[i]
            hj_old = h[j]
            for di in np.concatenate([
                np.linspace(-0.04, 0.04, 17),
                np.linspace(-0.01, 0.01, 9),
                np.linspace(-0.003, 0.003, 7)
            ]):
                hi_new = hi_old + di
                hj_new = hj_old - di
                if 0.0 <= hi_new <= 1.0 and 0.0 <= hj_new <= 1.0:
                    h[i] = hi_new
                    h[j] = hj_new
                    c = compute_c5(h, n)
                    if c < best_c5:
                        best_c5 = c
                        hi_old, hj_old = hi_new, hj_new
                    else:
                        h[i] = hi_old
                        h[j] = hj_old
    return h, best_c5

def fourier_optimize(h, n, t0, budget_s, M):
    x = np.linspace(0, 2, n, endpoint=False)
    dx = 2.0 / n
    coefs = np.zeros(2 * M + 1)
    coefs[0] = 0.5
    for m in range(1, M + 1):
        coefs[2*m - 1] = np.sum(h * np.cos(m * np.pi * x)) * dx
        coefs[2*m] = np.sum(h * np.sin(m * np.pi * x)) * dx

    def fourier_func(coefs):
        h_val = np.full(n, coefs[0])
        for m in range(1, M + 1):
            h_val += coefs[2*m - 1] * np.cos(m * np.pi * x)
            h_val += coefs[2*m] * np.sin(m * np.pi * x)
        return h_val

    def fourier_obj_grad(cc):
        h_val = project_feasible(fourier_func(cc), n)
        obj, grad_h = lse_obj_grad(h_val, n, 3000)
        gc = np.zeros_like(cc)
        for m in range(1, M + 1):
            gc[2*m - 1] = np.sum(grad_h * np.cos(m * np.pi * x)) * dx
            gc[2*m] = np.sum(grad_h * np.sin(m * np.pi * x)) * dx
        return obj, gc

    try:
        res = minimize(fourier_obj_grad, coefs.copy(), jac=True, method='L-BFGS-B',
                       options={'maxiter': 2000, 'gtol': 1e-10})
        h_result = project_feasible(fourier_func(res.x), n)
        return h_result
    except Exception:
        return h

def slsqp_polish(h, n, t0, budget_s, alpha):
    try:
        res = minimize(
            lambda x: lse_obj_grad(x, n, alpha)[0],
            h.copy(),
            jac=lambda x: lse_obj_grad(x, n, alpha)[1],
            method='SLSQP',
            bounds=[(0, 1)] * n,
            constraints=[{'type': 'eq',
                          'fun': lambda x: x.sum() - n / 2.0,
                          'jac': lambda x: np.ones(n)}],
            options={'maxiter': 1000, 'ftol': 1e-16}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h
