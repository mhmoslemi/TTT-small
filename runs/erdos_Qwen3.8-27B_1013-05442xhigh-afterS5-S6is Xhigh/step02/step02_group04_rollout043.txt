from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def softmax_weights(corr, beta):
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    return exps / np.sum(exps)

def compute_gradient(h, w):
    """Compute gradient of smooth max objective using structured formula."""
    n = len(h)
    # w has length 2n-1, indexed from 0 to 2n-2
    # w[i] corresponds to lag L = i - (n-1)
    # g[p] = sum_{L=-p}^{n-1-p} w_L - sum_j (w_{j-p} + w_{p-j}) * h[j]
    
    # Convert w to lag-indexed: w_lag[L] = w[L + (n-1)] for L in [-(n-1), n-1]
    # w_lag has length 2n-1
    # First term for coordinate p: sum of w_lag[L] for L in [-p, n-1-p]
    # = cumsum approach
    
    g = np.zeros(n)
    
    # Precompute cumulative sums of w for the first term
    # w is indexed 0..2n-2, corresponding to lags -(n-1)..n-1
    # For coordinate p, first term = sum_{L=-p}^{n-1-p} w[L + n - 1]
    # = sum of w from index (n-1-p) to index (2n-2-p)
    cumw = np.concatenate(([0], np.cumsum(w)))
    for p in range(n):
        lo = n - 1 - p
        hi = 2 * n - 2 - p
        first_term = cumw[hi + 1] - cumw[lo]
        
        # Second term: sum_j (w_{j-p} + w_{p-j}) * h[j]
        # w_{j-p} means w indexed at (j-p + n-1) = j - p + n - 1
        # w_{p-j} means w indexed at (p-j + n-1) = p - j + n - 1
        second_term = 0.0
        for j in range(n):
            idx1 = j - p + n - 1
            idx2 = p - j + n - 1
            second_term += (w[idx1] + w[idx2]) * h[j]
        
        g[p] = first_term - second_term
    
    return g

def project_h(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()

            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            # All at bounds, can't satisfy constraint
            break

    return h

    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

    corr = np.correlate(h, 1.0 - h, mode="full")
    # softmax weights
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w /= w.sum()
    # Compute gradient using structured formula
    # g[p] = sum_{L=-p}^{n-1-p} w[L+n-1] - sum_j (w[j-p+n-1] + w[p-j+n-1]) * h[j]
    g = np.zeros(n)

            idx2 = p - j + n - 1
            second += (w[idx1] + w[idx2]) * h[j]

    return g

from scipy.optimize import minimize, minimize_scalar, Bounds

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    h = np.array(h, dtype=float).copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nfree = free.sum()
        if nfree == 0:
            break
        h[free] += diff / nfree
    return h

    s = np.sum(np.exp(beta * (corr - m)))

def slsqp_phase(h, target_sum, duration, beta):

        return smooth_max_val(corr, beta) * dx
    
    def con_fun(x):
        return x.sum() - target_sum
    
    def con_jac(x):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    start = time.time()
    best_h = h.copy()
    best_c = compute_c5(h)
    
    while time.time() - start < duration:
        try:
            res = minimize(obj_fun, best_h, method='SLSQP', bounds=bounds,
                          constraints=constraints, options={'maxiter': 3000, 'ftol': 1e-16})
            h_new = project_h(res.x, target_sum)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_h, best_c = h_new, c_new
            else:
                break
        except Exception:
            break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        np.clip(h, 0.0, 1.0, out=h)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        mf = mask.sum()
        if mf == 0:
            break
        h[mask] += diff / mf
    return h

def smax(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta*(corr-m))))/beta

def grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0-h, mode="full")
    m = np.max(corr)
    w = np.exp(beta*(corr-m))
    w /= w.sum()
    g = np.zeros(n)
    cw = np.concatenate(([0.], np.cumsum(w)))
    for p in range(n):
        g[p] = cw[2*n-1-p] - cw[n-1-p]
        for j in range(n):
            g[p] -= (w[j-p+n-1] + w[p-j+n-1]) * h[j]
    return g

def nesterov(h, ts, dur, b0, b1, rng):
    n = len(h)
    hc, hp = h.copy(), h.copy()
    t, beta = 1.0, b0
    bc, bh = compute_c5(hc), hc.copy()
    t0 = time.time()
    while time.time()-t0 < dur:
        g = grad(hc, beta)
        mom = (t-1)/(t+2)
        y = hc + mom*(hc-hp)
        gn = np.max(np.abs(g))
        if gn < 1e-14: break
        y -= (1.0/(beta*gn))*g
        hp = hc.copy()
        hc = project_h(y, ts)
        t += 1
        beta = b0*(b1/b0)**((time.time()-t0)/dur)
        if int(t) % 1000 == 0:
            c = compute_c5(hc)
            if c < bc: bc, bh = c, hc.copy()
        if int(t) % 5000 == 0:
            hc = project_h(hc + rng.randn(n)*0.005, ts)
    c = compute_c5(hc)
    if c < bc: bc, bh = c, hc.copy()
    return bh, bc

def coord_desc(h, ts, dur, beta, rng):
    n = len(h)
    hc = h.copy()
    bc = compute_c5(hc)
    t0 = time.time()
    while time.time()-t0 < dur:
        imp = False
        for p in range(n):
            if time.time()-t0 > dur: break
            # Try modifying h[p]
            best_t = hc[p]
            best_f = smax(np.correlate(hc,1-hc,mode="full"), beta)
            for dt in np.array([0.1,-0.1,0.05,-0.05,0.2,-0.2,0.02,-0.02]):
                t_new = hc[p] + dt
                if t_new < 0 or t_new > 1: continue
                ht = hc.copy()
                ht[p] = t_new
                ht = project_h(ht, ts)
                f = smax(np.correlate(ht,1-ht,mode="full"), beta)
                if f < best_f:
                    best_f = f
                    best_t = t_new
            if best_t != hc[p]:
                hc[p] = best_t
                hc = project_h(hc, ts)
                imp = True
        c = compute_c5(hc)
        if c < bc: bc = c
        if not imp: break
    return hc, bc

def slsqp(h, ts, dur, beta):
    n = len(h)
    dx = 2.0/n
    def o(x):
        return smax(np.correlate(x,1-x,mode="full"), beta)*dx
    def cf(x):
        return x.sum()-ts
    def cj(x):
        return np.ones(n)
    cons=[{'type':'eq','fun':cf,'jac':cj}]
    bnds=Bounds(np.zeros(n),np.ones(n))
    t0=time.time()
    bh,bc = h.copy(), compute_c5(h)
    while time.time()-t0 < dur:
        try:
            r=minimize(o,bh,method='SLSQP',bounds=bnds,constraints=cons,
                      options={'maxiter':5000,'ftol':1e-16})
            hn=project_h(r.x,ts)
            cn=compute_c5(hn)
            if cn<bc: bh,bc=hn,cn
            else: break
        except: break
    return bh,bc
