import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Distribute evenly among free variables
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        # Second correction
        diff2 = target - h.sum()
        if abs(diff2) > 1e-10:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
            else:
                h += diff2 / len(h)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, beta, target, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Smooth max
    m = np.max(corr)
    shifted = beta * (corr - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exps = np.exp(shifted)
    log_sum = m + np.log(np.sum(exps)) / beta
    w = exps / np.sum(exps)
    
    # Penalty for sum constraint
    sum_diff = h.sum() - target
    obj = log_sum * dx + penalty * sum_diff ** 2
    
    # Gradient
    grad = np.zeros(n)
    j = np.arange(n)
    
    # Only consider lags with significant weight
    active = np.where(w > 1e-10)[0]
    for k in active:
        wk = w[k]
        s = k - (n - 1)
        d = np.ones(n)
        if s > 0:
            d[:n-s] -= h[s:]
            d[s:] -= h[:n-s]
        elif s < 0:
            sp = -s
            d[sp:] -= h[:n-sp]
            d[:n-sp] -= h[sp:]
        else:
            d -= 2.0 * h
        grad += wk * d
    
    grad *= dx
    grad += 2.0 * penalty * sum_diff
    
    return obj, grad

def lbfgs_optimize(h, beta, target, maxiter=500, penalty=1e3):
    n = len(h)
    h0 = project_feasible(h.copy(), target)
    
    res = minimize(
        smooth_max_obj_grad, h0,
        args=(beta, target, penalty),
        method='L-BFGS-B',
        jac=True,
        bounds=[(0, 1)] * n,
        options={'maxiter': maxiter, 'gtol': 1e-8, 'ftol': 1e-14}
    )
    h_opt = project_feasible(res.x, target)
    return h_opt

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)

from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - h.sum()
        if abs(diff2) > 1e-10:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            n_free = free.sum()
            if n_free > 0:
                h[free] += diff2 / n_free
            else:
                h += diff2 / len(h)
            h = np.clip(h, 0.0, 1.0)

    s_exp = np.sum(exps)
    val = (m + np.log(s_exp) / beta) * dx
    w = exps / s_exp
    
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        wk = w[k]
        s = k - (n - 1)
        # d(corr[k])/d(h[j]) = (1-h[j+s])*I(0<=j+s<n) - h[j-s]*I(0<=j-s<n)
        if s > 0:
            d = np.zeros(n)
            d[:n-s] = 1.0 - h[s:]       # (1-h[j+s]) for j=0..n-s-1
            d[s:] -= h[:n-s]             # -h[j-s] for j=s..n-1
        elif s < 0:
            sp = -s
            d = np.zeros(n)
            d[sp:] = 1.0 - h[:n-sp]     # (1-h[j-s]) for j=sp..n-1  [j+s = j-sp]
            d[:n-sp] -= h[sp:]           # -h[j-s] for j=0..n-sp-1   [j-s = j+sp]
        else:
            d = 1.0 - 2.0 * h
        grad += wk * d
    
    grad *= dx
    return val, grad

def obj_pen(h, beta, target, pen):
    """Objective with penalty for sum constraint."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    val, grad = smax_grad(corr, h, beta)
    sd = h.sum() - target
    return val + pen * sd * sd, grad + 2.0 * pen * sd

def obj_slsqp(h, beta, target):
    """Objective for SLSQP (no penalty, use constraint)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smax_grad(corr, h, beta)

def lbfgs_opt(h, beta, target, maxiter=300, pen=1e4):
    n = len(h)
    h0 = project_feasible(h.copy(), target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h0, args=(beta, target, pen),
                       method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-8, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h0

def slsqp_opt(h, beta, target, maxiter=200):
    n = len(h)
    h0 = project_feasible(h.copy(), target)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_slsqp, h0, args=(beta, target),
                       method='SLSQP', jac=True, bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h0

    return project_feasible(h_new, n_new / 2.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        nf = free.sum()
        if nf > 0:
            h[free] += diff / nf
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - h.sum()
        if abs(diff2) > 1e-10:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            nf = free.sum()
            if nf > 0:
                h[free] += diff2 / nf
            else:
                h += diff2 / len(h)
            h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad_single(s, h, n):
    """Gradient of corr at shift s w.r.t. h: (1-h[j+s])*I(0<=j+s<n) - h[j-s]*I(0<=j-s<n)"""
    d = np.zeros(n)
    if s > 0:
        d[:n-s] = 1.0 - h[s:]
        d[s:] -= h[:n-s]
    elif s < 0:
        sp = -s
        d[sp:] = 1.0 - h[:n-sp]
        d[:n-sp] -= h[sp:]
    else:
        d = 1.0 - 2.0 * h
    return d

def smax_val_grad(h, beta):
    """Smooth max of corr * dx, with CORRECT gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    exps = np.exp(shifted)

    val = (m + np.log(s_exp) / beta) * dx

        grad += w[k] * corr_grad_single(k - (n - 1), h, n)
    grad *= dx
    return val, grad

    w = exps / s_exp
    sd = x.sum() - target
    val += pen * sd * sd

    active = np.where(w > 1e-12)[0]
    for k in active:
        grad += w[k] * corr_grad_single(k - (n - 1), x, n)
    grad *= (2.0 / n)
    grad += 2.0 * pen * sd
    return val, grad
