```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def optimize_1d_coord(A, B, dx):
    """Minimize dx * max_k (A_k * x + B_k) over x in [0,1]"""
    # Grid evaluation
    xs = np.linspace(0.0, 1.0, 40)
    M = np.outer(xs, A) + B[None, :]
    f_vals = M.max(axis=1) * dx
    best_idx = np.argmin(f_vals)
    best_x = xs[best_idx]
    best_f = f_vals[best_idx]
    
    # Refine around best with golden section
    lo = max(0.0, xs[max(0, best_idx-1)])
    hi = min(1.0, xs[min(len(xs)-1, best_idx+1)])
    if hi - lo < 1e-8:
        return best_x
    
    gr = (np.sqrt(5) - 1) / 2
    for _ in range(50):
        x1 = hi - gr * (hi - lo)
        x2 = lo + gr * (hi - lo)
        f1 = np.max(A * x1 + B) * dx
        f2 = np.max(A * x2 + B) * dx
        if f1 < f2:
            hi = x2
        else:

        if hi - lo < 1e-10:
            break
    return (lo + hi) / 2

def coord_descent(h, target_sum, max_sweeps=100, order=None, rng=None):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    delta_arr = np.arange(n_lags) - n + 1

        improved = False

            mask_plus = (idx_plus >= 0) & (idx_plus < n)
            A = np.zeros(n_lags)
```

```python
def coord_descent_fast(h, target_sum, max_sweeps=200, rng=None):

    dx = 2.0 / n
    n_lags = 2 * n - 1
    delta_arr = np.arange(n_lags) - n + 1
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = corr.max() * dx
    
    for sweep in range(max_sweeps):
        indices = list(range(n))
        if rng is not None:
            rng.shuffle(indices)
        
        for j in indices:
            idx_plus = j + delta_arr
            mask_plus = (idx_plus >= 0) & (idx_plus < n)
            A = np.zeros(n_lags)
            A[mask_plus] = 1.0 - h[idx_plus[mask_plus]]
            
            idx_minus = j - delta_arr
            mask_minus = (idx_minus >= 0) & (idx_minus < n)
            A[mask_minus] -= h[idx_minus[mask_minus]]
            
            B = corr - A * h[j]
            
            new_val = optimize_1d_coord(A, B, dx)
            delta = new_val - h[j]
            if abs(delta) > 1e-12:
                h[j] = new_val
                corr += delta * A  # Incremental update
    
        # Project
        h = project_feasible(h, target_sum)
        # Recheck c5 (projection might have changed things)
        c5 = compute_c5(h)
        corr = np.correlate(h, 1.0 - h, mode="full")  # Resync
        if c5 < best_c5:
            best_c5 = c5
    
    return h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(80):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)


def optimize_1d_coord(A, B, dx):
    xs = np.linspace(0.0, 1.0, 30)
    M = np.outer(xs, A) + B[None, :]
    f_vals = M.max(axis=1) * dx
    bi = int(np.argmin(f_vals))
    lo = float(xs[max(0, bi - 1)])
    hi = float(xs[min(len(xs) - 1, bi + 1)])
    if hi - lo < 1e-10:
        return xs[bi]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(60):
        x1 = hi - gr * (hi - lo)
        x2 = lo + gr * (hi - lo)
        f1 = float(np.max(A * x1 + B)) * dx
        f2 = float(np.max(A * x2 + B)) * dx
        if f1 < f2:
            hi = x2
        else:
            lo = x1
        if hi - lo < 1e-12:
            break
    return (lo + hi) / 2.0


def coord_descent(h, target_sum, max_sweeps, rng, start_time, budget_s):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    delta_arr = np.arange(n_lags) - n + 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = float(np.max(corr) * dx)
    for sweep in range(max_sweeps):
        if budget_s - (time.time() - start_time) < 60:
            break
        indices = list(range(n))
        rng.shuffle(indices)
        for j in indices:
            ip = j + delta_arr
            mp = (ip >= 0) & (ip < n)
            A = np.zeros(n_lags)
            A[mp] = 1.0 - h[ip[mp]]
            im = j - delta_arr
            mm = (im >= 0) & (im < n)
            A[mm] -= h[im[mm]]
            B = corr - A * h[j]
            nv = optimize_1d_coord(A, B, dx)
            d = nv - h[j]
            if abs(d) > 1e-13:
                h[j] = nv
                corr += d * A
        h = project_feasible(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        else:
            corr = np.correlate(h, 1.0 - h, mode="full")
    return h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(70, 0.5)

    best_h = None
    best_c5 = np.inf
    best_n = 70

    n_list = [50, 60, 70, 80, 100, 140, 200, 300, 500]
    for n in n_list:
        if budget_s - (time.time() - start_time) < 150:
            break
        ts = n / 2.0
        cands = []
        hp = np.interp(np.linspace(0, len(h_parent) - 1, n),
                       np.arange(len(h_parent)), h_parent)
        cands.append(project_feasible(hp, ts))
        cands.append(np.full(n, 0.5))
        t = np.arange(n)
        for fr in [1, 2, 3]:
            cands.append(project_feasible(0.5 + 0.25 * np.sin(2 * np.pi * fr * t / n), ts))
        hb = np.zeros(n)
        hb[::2] = 1.0
        cands.append(project_feasible(hb, ts))
        for hc in cands:
            c5v = compute_c5(hc)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = hc.copy()
                best_n = n

    n = best_n
    ts = n / 2.0
    h = best_h.copy()

    # Coordinate descent main
    h, c5v = coord_descent(h, ts, 200, rng, start_time, budget_s)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h.copy()

    # More CD with perturbations

        if budget_s - (time.time() - start_time) < 120:
            break
        hp2 = best_h + rng.randn(n) * 0.03
        hp2 = project_feasible(hp2, ts)
        hp2, cv = coord_descent(hp2, ts, 80, rng, start_time, budget_s)
        if cv < best_c5:
            best_c5 = cv
            best_h = hp2.copy()

    for n2 in [best_n + 20, best_n - 20, best_n + 50, best_n * 2]:

            best_n = n2
            n = n2
            ts = ts2

    # SLSQP refinement
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts,
                   'jac': lambda x: np.ones(x.shape[0])}]

            corr = np.correlate(hv, 1.0 - hv, mode="full")
            m = np.max(corr)
            return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * (2.0 / len(hv))
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 5000, 'ftol': 1e-15})
            hn = project_feasible(res.x, ts)
            cv = compute_c5(hn)
            if cv < best_c5:
                best_c5 = cv
                best_h = hn.copy()
        except Exception:
            pass

    # Final CD polish
    if budget_s - (time.time() - start_time) > 40:
        h3 = best_h.copy()
        h3, cv3 = coord_descent(h3, ts, 200, rng, start_time, budget_s)

            corr = np.correlate(hv, 1.0 - hv, mode="full")
            m = np.max(corr)
            return (m + np.log(np.sum(np.exp(10000.0 * (corr - m)))) / 10000.0) * (2.0 / len(hv))
        try:
            res = minimize(obj_f, best_h, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 5000, 'ftol': 1e-16})
            hn = project_feasible(res.x, ts)
            cv = compute_c5(hn)
            if cv < best_c5:
                best_c5 = cv
                best_h = hn.copy()
        except Exception:
            pass

    best_h = project_feasible(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
