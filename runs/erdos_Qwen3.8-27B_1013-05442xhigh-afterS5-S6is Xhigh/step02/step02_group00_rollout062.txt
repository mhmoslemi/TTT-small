```python
def grad_for_lag(h, lag, n, dx):
    """Gradient of f_lag with respect to h."""
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        # f_lag = sum_{j=0}^{n-1-lag} h[j+lag]*(1-h[j])*dx
        # d/dh[m] = (1-h[m-lag]) for m in [lag, n-1]
        #          - h[m+lag] for m in [0, n-1-lag]
        grad[lag:] += (1.0 - h[:n-lag]) * dx
        grad[:n-lag] += -h[lag:] * dx
    else:
        l = -lag
        # f_{-l} = sum_{j=0}^{n-1-l} h[j]*(1-h[j+l])*dx
        # d/dh[m] = (1-h[m+l]) for m in [0, n-1-l]
        #          - h[m-l] for m in [l, n-1]
        grad[:n-l] += (1.0 - h[l:]) * dx
        grad[l:] += -h[:n-l] * dx
    return grad
```

```python
    """One step of steepest descent using weighted active lags."""

    center = n - 1

    gnorm = np.linalg.norm(grad)

    return h, corr, grad / gnorm
```

```python
                grad += w * grad_for_lag(best_h, lag, n_f, dx_f)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            step = 0.003
            for _ in range(20):
                h_trial = project_h(best_h - step * grad, target_f)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-16:
                    best_h = h_trial
                    best_c5 = c5_trial
                    break
                step *= 0.5

    return best_h, best_c5, len(best_h)
```
