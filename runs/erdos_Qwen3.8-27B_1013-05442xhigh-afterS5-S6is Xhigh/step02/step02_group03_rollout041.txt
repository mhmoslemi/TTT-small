import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        step = diff / cnt
        # Don't overshoot bounds
        if diff > 0:
            headroom = 1.0 - h[mask]
            max_step = headroom.min() if cnt > 0 else 0
            step = min(step, max_step) if step > 0 else step
        else:
            headroom = h[mask]
            max_step = headroom.min() if cnt > 0 else 0
            step = max(step, -max_step) if step < 0 else step
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def pair_descent_sweep(h, dx, beta, n_evals=30):
    """One full sweep of pair coordinate descent"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    obj_current = smooth_max(corr, beta) * dx
    
    for i in range(n):
        for j in range(i+1, n):
            h_i = h[i]
            h_j = h[j]
            # Feasible range for delta
            delta_min = max(-h_i, -(1.0 - h_j))  # h[i]+delta >= 0, h[j]-delta <= 1
            delta_max = min(1.0 - h_i, h_j)       # h[i]+delta <= 1, h[j]-delta >= 0
            
            if delta_max - delta_min < 1e-12:
                continue
            
            def f(delta):
                h[i] += delta
                h[j] -= delta
                corr_new = np.correlate(h, 1.0 - h, mode='full')
                val = smooth_max(corr_new, beta) * dx
                h[i] -= delta
                h[j] += delta
                return val
            
            res = minimize_scalar(f, bounds=(delta_min, delta_max), method='bounded',
                                  options={'maxiter': n_evals})
            if res.fun < obj_current:
                h[i] += res.x
                h[j] -= res.x
                obj_current = res.fun
    
    return h, obj_current

    # Phase 2: SLSQP polish
    deadline2 = t0 + budget_s * 0.32
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (target_sum, n)}]
    for beta in [500, 2000, 10000]:
        if time.time() > deadline2:
            break
        try:
            res = minimize(obj_for_slsqp, best_h, args=(dx, beta),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        except Exception:
            pass

    # Phase 3: Try different resolutions
    deadline3 = t0 + budget_s * 0.55
    for n_try in [50, 60, 80, 90, 40, 100, 55, 65, 45, 75, 35, 110]:
        if time.time() > deadline3:
            break
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        h_t = resample_h(best_h, n_try)
        h_t = project_feasible(h_t, ts_t)
        for beta in [100, 500, 2000]:
            if time.time() > deadline3 - 5:
                break
            h_t = pair_sweep_seq(h_t, dx_t, beta, n_evals=15)
            c5 = compute_c5(h_t)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_t.copy()
                n = n_try
                dx = dx_t
                target_sum = ts_t
                bounds = Bounds(np.zeros(n), np.ones(n))
                cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (target_sum, n)}]

    # Phase 4: Multi-start exploration
    deadline4 = t0 + budget_s * 0.75
    starts = generate_starts(n, target_sum, seed + 777)
    for h_s in starts:
        if time.time() > deadline4:
            break
        for beta in [50, 300, 1000]:
            if time.time() > deadline4 - 3:
                break
            h_s = pair_sweep_seq(h_s, dx, beta, n_evals=15)
            c5 = compute_c5(h_s)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_s.copy()

    # Phase 5: Extended pair descent with random ordering
    deadline5 = t0 + budget_s * 0.92
    h = best_h.copy()
    for beta in [500, 1000, 2000, 5000, 10000]:
        while time.time() < deadline5 - 15:
            pair_sweep(h, dx, beta, n_evals=25, rng=rng)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 6: Final SLSQP + pair polish
    if time.time() < t0 + budget_s - 10:
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (target_sum, n)}]
        for beta in [5000, 20000, 50000]:
            if time.time() > t0 + budget_s - 5:
                break
            try:
                res = minimize(obj_for_slsqp, best_h, args=(dx, beta),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 10000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            except Exception:
                pass
        # One more pair sweep at very high beta
        if time.time() < t0 + budget_s - 3:
            h = best_h.copy()
            pair_sweep_seq(h, dx, 20000, n_evals=30)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    return best_h, best_c5, len(best_h)
