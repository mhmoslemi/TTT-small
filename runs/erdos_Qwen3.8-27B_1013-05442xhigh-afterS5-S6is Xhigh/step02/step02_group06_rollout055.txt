            lag_idx = L + (n - 1)
            w = p[lag_idx]
            if w < 1e-8:
                continue
            g += w * lag_gradient(h, L, n)
        gn = np.max(np.abs(g)) + 1e-12
        alpha = 0.03 / gn
        h = h - alpha * g
        h = project_box_sum(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c_max = np.max(corr)
        if c_max * (2.0 / n) < best_c:
            best_c = c_max * (2.0 / n)
            best_h = h.copy()
        eta = 0.15
        p = p * np.exp(eta * corr / (c_max + 1e-10))
        p = p / p.sum()
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    else:
        n = 70
        h_init = np.full(n, 0.5)

    ts = n / 2.0
    best_h = project_box_sum(h_init, ts)
    best_c5 = compute_c5(best_h)
    bnds = [(0.0, 1.0)] * n

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: Fourier global search
    if tl() > 200:
        for m_modes in [3, 4, 5, 6, 8]:
            if tl() < 80:
                break
            bounds_fe = [(-0.45, 0.45)] * (2 * m_modes)
            try:
                res = differential_evolution(
                    fourier_obj, bounds_fe, args=(n,),
                    seed=seed + m_modes * 7, maxiter=200, tol=1e-10,
                    popsize=14, mutation=(0.3, 1.0), recombination=0.8,
                    updating='deferred', init='latinhypercube'
                )
                h_new = fourier_to_h(res.x, n)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    # Phase 2: Local L-BFGS-B with top-k smooth max
    for beta, lam, ktop in [(80, 40, 5), (200, 150, 5), (500, 400, 4), (1000, 800, 3)]:
        if tl() < 30:
            break
        try:
            r = minimize(objective_full, best_h, args=(n, ts, beta, lam, ktop),
                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': 600, 'ftol': 1e-16})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # Phase 3: Hedge-style minimax
    if tl() > 120:
        h_hedged, c_hedged = hedge_solve(best_h, n, ts, t0, budget_s, n_iter=400)
        if c_hedged < best_c5:
            best_h, best_c5 = h_hedged, c_hedged

    # Phase 4: Lag-targeted coordinate descent
    if tl() > 60:
        for rd in range(400):
            if tl() < 15:
                break
            h = coordinate_descent_worst(best_h, n, ts, max_rounds=15, top_lags=3)
            c = compute_c5(h)
            if c < best_c5:
                best_h, best_c5 = h, c
            if rd % 8 == 4:
                h2 = best_h + np.random.randn(n) * 0.01
                h2 = project_box_sum(h2, ts)
                h2 = coordinate_descent_worst(h2, n, ts, max_rounds=10, top_lags=4)
                c2 = compute_c5(h2)
                if c2 < best_c5:
                    best_h, best_c5 = h2, c2

    # Phase 5: Diverse restarts
    it = 0
    while tl() > 100:
        it += 1
        h = best_h.copy()
        strat = it % 6
        if strat == 0:
            scale = max(0.005, 0.08 * np.exp(-it / 50.0))
            h += np.random.randn(n) * scale
        elif strat == 1:
            corr = np.correlate(h, 1.0 - h, mode="full")
            top_idx = np.argpartition(corr, -3)[-3:]
            g_total = np.zeros(n)
            for idx in top_idx:
                k_shift = int(idx) - (n - 1)
                g_total += lag_gradient(h, k_shift, n)
            gn = np.max(np.abs(g_total)) + 1e-10
            h = h - (0.02 / gn) * g_total
        elif strat == 2:
            blen = np.random.randint(5, max(6, n // 4))
            bs = np.random.randint(0, n - blen)
            h[bs:bs + blen] = 0.5 + 0.35 * np.random.randn(blen)
        elif strat == 3:
            x = 2.0 * np.arange(n) / n
            for j in range(1, 5):
                h += 0.015 * np.random.randn() * np.cos(j * np.pi * x)
                h += 0.015 * np.random.randn() * np.sin(j * np.pi * x)
        elif strat == 4:
            corr = np.correlate(h, 1.0 - h, mode="full")
            k_star = int(np.argmax(corr)) - (n - 1)
            if k_star > 0:
                mask = np.zeros(n, dtype=bool)
                mask[:n - k_star] = True
                mask[k_star:] = True
                h[mask] += 0.01 * np.random.randn(mask.sum())
            elif k_star < 0:
                kk = -k_star
                mask = np.zeros(n, dtype=bool)
                mask[kk:] = True
                mask[:n - kk] = True
                h[mask] += 0.01 * np.random.randn(mask.sum())
        else:
            for _ in range(np.random.randint(2, 7)):
                i, j = np.random.choice(n, 2, replace=False)
                h[i], h[j] = h[j], h[i]
            h += np.random.randn(n) * 0.004

        h = project_box_sum(h, ts)
        for beta, ktop in [(300, 4), (800, 3)]:
            if tl() < 20:
                break
            try:
                r = minimize(objective_full, h, args=(n, ts, beta, 400.0, ktop),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 250, 'ftol': 1e-15})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    # Phase 6: Final high-precision polish
    if tl() > 20:
        for beta in [3000, 8000, 15000]:
            if tl() < 8:
                break
            try:
                r = minimize(objective_all, best_h, args=(n, ts, beta, 3000.0),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 5000, 'ftol': 1e-18})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    # Phase 7: Final coordinate descent
    if tl() > 8:
        h = coordinate_descent_worst(best_h, n, ts, max_rounds=200, top_lags=5)
        c = compute_c5(h)
        if c < best_c5:
            best_h, best_c5 = h, c

    best_h = project_box_sum(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
