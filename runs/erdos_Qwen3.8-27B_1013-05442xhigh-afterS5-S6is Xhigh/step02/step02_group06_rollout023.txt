import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.sum() == 0:
            if d > 0:
                idx = np.argmin(h)
                h[idx] = min(1.0, h[idx] + d)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + d)
            h = np.clip(h, 0.0, 1.0)
            continue
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def get_correlation(h, n):
    return np.correlate(h, 1.0 - h, mode="full")
