    return float(np.max(corr) * dx)

                    h[j] = np.clip(h[j], 0.0, 1.0)
                    # Fix sum
                    diff = target_sum - h.sum()
                    if abs(diff) > 1e-13:
                        free = (h > 1e-10) & (h < 1.0 - 1e-10)
                        if free.sum() > 0:
                            h[free] += diff / free.sum()
                            h = np.clip(h, 0.0, 1.0)
                    sweep_improvements += 1
                    # Update coeff for changed coordinates
                    for m in range(M):
                        k = active_lags[m]
                        c_i = 0.0
                        if i - k >= 0 and i - k < n:
                            c_i += (1.0 - h[i - k])
                        if i + k >= 0 and i + k < n:
                            c_i -= h[i + k]
                        coeff[i, m] = c_i
                        c_j = 0.0
                        if j - k >= 0 and j - k < n:
                            c_j += (1.0 - h[j - k])
                        if j + k >= 0 and j + k < n:
                            c_j -= h[j + k]
                        coeff[j, m] = c_j

        if sweep_improvements == 0:
            no_improve_count += 1
            if no_improve_count >= 2:
                break
        else:
            no_improve_count = 0

        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now

    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 140
        h = 0.5 + 0.3 * np.sin(np.pi * np.linspace(0, 2, n_init))
        h = project_h(h, n_init / 2.0)

    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = project_h(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_n = n

    # Phase 1: Paired coordinate descent with increasing M
    M_values = [8, 12, 16, 20]
    for M in M_values:
        if time_left() < 60:
            break
        h, c5 = pair_descent(h, n, target_sum, M, max_sweeps=50, time_fn=time_left)
        if c5 < best_c5:
            best_h, best_c5, best_n = h.copy(), c5, n

    # Phase 2: SLSQP polish with smooth max
    if time_left() > 30:
        bounds = Bounds(np.zeros(n), np.ones(n))
        h_cur = best_h.copy()
        for beta in [50, 100, 200, 500]:
            if time_left() < 20:
                break
            def _obj(h_in):
                c = np.correlate(h_in, 1.0 - h_in, mode="full")
                return smooth_max(c, beta) * dx
            def _con(x_in):
                return np.sum(x_in) - target_sum
            def _con_jac(x_in):
                return np.ones(x_in.shape)
            constraints = [{'type': 'eq', 'fun': _con, 'jac': _con_jac}]
            try:
                res = minimize(_obj, h_cur, method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 2000, 'ftol': 1e-15})
                h_new = project_h(res.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h, best_c5 = h_new.copy(), c5_new
                    h_cur = h_new.copy()
            except Exception:
                pass

    # Phase 3: Final paired descent on the polished solution
    if time_left() > 30:
        h = best_h.copy()
        for M in [15, 25]:
            if time_left() < 15:
                break
            h, c5 = pair_descent(h, n, target_sum, M, max_sweeps=30, time_fn=time_left)
            if c5 < best_c5:
                best_h, best_c5, best_n = h.copy(), c5, n

    # Phase 4: Try slightly different resolutions by resampling
    if time_left() > 45:
        for n_try in [130, 150, 160]:
            if time_left() < 30:
                break
            x_b = np.linspace(0, 2, best_n)
            x_n = np.linspace(0, 2, n_try)
            h_try = project_h(np.interp(x_n, x_b, best_h), n_try / 2.0)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                # Optimize at this resolution
                h_try, c5_try2 = pair_descent(h_try, n_try, n_try / 2.0, 15, 20, time_left)
                if c5_try2 < best_c5:
                    best_h, best_c5, best_n = h_try, c5_try2, n_try

    return best_h, best_c5, best_n
