```python
import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smoothmax_value(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return float((m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx)

def smoothmax_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return grad

def penalty_objective(x, beta, target):
    v = smoothmax_value(x, beta)
    penalty = 100.0 * (x.sum() - target) ** 2
    return v + penalty

def penalty_grad(x, beta, target):
    g = smoothmax_grad(x, beta)
    g += 200.0 * (x.sum() - target) * np.ones_like(x)
    return g

def slsqp_refine(h, maxiter, beta, target):

    res = minimize(penalty_objective, h, jac=penalty_grad, method='SLSQP',
                   bounds=bounds, args=(beta, target),
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return np.clip(res.x, 0.0, 1.0)
```

```python
def nesterov_subgrad(h, max_iters, lr0, decay, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)
        # Fix sum
        diff = target - y.sum()
        y += diff / n
        y = np.clip(y, 0.0, 1.0)
        
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])
        grad = grad / top_k * dx
        x_new = np.clip(y - lr * grad, 0.0, 1.0)
        diff2 = target - x_new.sum()
        x_new += diff2 / n
        x_new = np.clip(x_new, 0.0, 1.0)
        
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    
    return best_x, best_v
```

```python
def perturb_binary(h, n_flips, rng):

    h_p[idx] = 1.0 - h_p[idx]
    # Fix sum
    diff = target - h_p.sum()
    if abs(diff) > 1e-8:
        # Adjust
        h_p += diff / n
        h_p = np.clip(h_p, 0.0, 1.0)
    return h_p
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smoothmax_value_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    s = np.sum(exps)
    w = exps / s
    val = (m + np.log(s) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s_shift = k - (n - 1)
        js = j - s_shift
        jp = j + s_shift
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad

def _obj(x, beta, target):
    v, _ = smoothmax_value_grad(x, beta)
    pen = 1000.0 * (np.sum(x) - target) ** 2
    return v + pen

def _objg(x, beta, target):
    _, g = smoothmax_value_grad(x, beta)
    g = g + 2000.0 * (np.sum(x) - target) * np.ones(len(x))
    return g

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(_obj, h, jac=_objg, method='SLSQP',
                       bounds=bounds, args=(beta, target),
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h

def nesterov_subgrad(h, max_iters, lr0, decay, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        # Project
        y = np.clip(y, 0.0, 1.0)
        diff = target - np.sum(y)
        if abs(diff) > 1e-10:
            y += diff / n
            y = np.clip(y, 0.0, 1.0)
        
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s_shift in lags:
            js = j - s_shift
            jp = j + s_shift
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])
        grad = grad / top_k * dx
        x_new = np.clip(y - lr * grad, 0.0, 1.0)
        diff2 = target - np.sum(x_new)
        if abs(diff2) > 1e-10:
            x_new += diff2 / n
            x_new = np.clip(x_new, 0.0, 1.0)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v
```

```python
def perturb_gaussian(h, temp, rng):
    n = len(h)
    target = n / 2.0
    h_p = h + temp * rng.standard_normal(n)
    h_p = np.clip(h_p, 0.0, 1.0)
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-10:
        h_p += diff / n
        h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def perturb_flip(h, n_flips, rng):
    n = len(h)
    target = n / 2.0
    h_p = h.copy()
    idx = rng.choice(n, size=min(n_flips, n), replace=False)
    h_p[idx] = 1.0 - h_p[idx]
    h_p = np.clip(h_p, 0.0, 1.0)
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-10:
        h_p += diff / n
        h_p = np.clip(h_p, 0.0, 1.0)
    return h_p
```

```python
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)
    
    n = len(h_best)
    best_c5 = compute_c5(h_best)
    
    # Phase 1: Initial local search
    if budget_s - (time.time() - t0) > 120:
        h_best, best_c5 = nesterov_subgrad(h_best, 3000, 0.05, 0.9998, top_k=10)
    
    if budget_s - (time.time() - t0) > 100:
        h_best = slsqp_refine(h_best, 300, beta=200)
        c = compute_c5(h_best)
        if c < best_c5:
            best_c5 = c
    
    # Phase 2: ILS
    temp = 0.08
    cooling = 0.998
    patience = 0
    max_patience = 200
    
    while budget_s - (time.time() - t0) > 80:
        # Choose perturbation
        r = rng.random()
        if r < 0.5:
            h_p = perturb_gaussian(h_best, temp, rng)
        elif r < 0.8:
            h_p = perturb_flip(h_best, max(2, int(n * temp / 0.5)), rng)
        else:
            # Segment swap
            h_p = h_best.copy()
            seg_len = rng.randint(n // 20, n // 5)
            pos1 = rng.randint(0, n - seg_len)
            pos2 = rng.randint(0, n - seg_len)
            if abs(pos1 - pos2) >= seg_len:
                h_p[pos1:pos1+seg_len] = h_best[pos2:pos2+seg_len]
                h_p[pos2:pos2+seg_len] = h_best[pos1:pos1+seg_len]
            else:
                h_p = perturb_gaussian(h_best, temp, rng)
        
        # Local search
        h_new, c_new = nesterov_subgrad(h_p, 800, 0.04, 0.9995, top_k=8)
        
        # SLSQP polish
        if budget_s - (time.time() - t0) > 85:
            h_new = slsqp_refine(h_new, 150, beta=200)
            c_new = compute_c5(h_new)
        
        # Accept/reject
        if c_new < best_c5:
            h_best = h_new
            best_c5 = c_new
            temp = 0.08
            patience = 0
        else:
            patience += 1
            # Simulated annealing: accept worse with some probability
            delta = c_new - best_c5
            if delta < 0.005 and rng.random() < np.exp(-delta / (temp * 0.005)):
                h_best = h_new
            temp *= cooling
        
        if patience > max_patience:
            temp = 0.15  # Reheat
            patience = 0
    
    # Phase 3: Final polish
    if budget_s - (time.time() - t0) > 20:
        h_best = slsqp_refine(h_best, 500, beta=500)
        c = compute_c5(h_best)
        if c < best_c5:
            best_c5 = c
    
    return h_best, best_c5, n
```

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for idx in range(len(corr)):
        wi = w[idx]
        if wi < 1e-12:
            continue
        s = idx - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wi * (-h[js[ms]])
        if mp.any():
            grad[mp] += wi * (1.0 - h[jp[mp]])
    grad *= dx
    return val, grad

def _slsqp_obj(x, beta, target):
    v, _ = smoothmax_val_grad(x, beta)
    pen = 1000.0 * (np.sum(x) - target) ** 2
    return v + pen

def _slsqp_grad(x, beta, target):
    _, g = smoothmax_val_grad(x, beta)
    g = g + 2000.0 * (np.sum(x) - target) * np.ones(len(x))
    return g

    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad, method='SLSQP',
                       bounds=bounds, args=(beta, target),
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h

    target = n / 2.0

            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)
        diff = target - np.sum(y)
        if abs(diff) > 1e-10:
            y += diff / n
            y = np.clip(y, 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)

            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (-y[js[ms]])
            if mp.any():
                grad[mp] += (1.0 - y[jp[mp]])
        grad = grad / top_k * dx
        x_new = np.clip(y - lr * grad, 0.0, 1.0)
        diff2 = target -
