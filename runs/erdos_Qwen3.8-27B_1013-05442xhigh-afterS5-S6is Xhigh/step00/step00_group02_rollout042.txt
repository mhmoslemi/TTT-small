def simulated_annealing(n, iterations, seed, init_h=None):
    """Run simulated annealing on binary sequences of length n."""
    rng = np.random.RandomState(seed)
    
    if init_h is not None and len(init_h) == n:
        h = (init_h > 0.5).astype(float)
        # Adjust sum if needed
        current_sum = h.sum()
        target_sum = n / 2
        if current_sum > target_sum:
            # Flip some 1s to 0s
            ones = np.where(h == 1)[0]
            extra = int(current_sum - target_sum)
            for i in rng.choice(ones, extra, replace=False):
                h[i] = 0
        elif current_sum < target_sum:
            zeros = np.where(h == 0)[0]
            deficit = int(target_sum - current_sum)
            for i in rng.choice(zeros, deficit, replace=False):
                h[i] = 1
    else:
        # Random binary sequence with n/2 ones
        h = np.zeros(n)
        h[rng.choice(n, n//2, replace=False)] = 1.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    current_c5 = best_c5
    
    T = 0.1
    cooling = 0.9999
    
    for it in range(iterations):
        if it % 1000 == 0:
            T *= cooling
        
        # Pick a random 1 and a random 0 to swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        
        # Swap
        h[i] = 0
        h[j] = 1
        
        new_c5 = compute_c5(h, n)
        
        if new_c5 < current_c5 or rng.random() < np.exp((current_c5 - new_c5) / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Revert
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c5

def continuous_refine(h, n, max_iter=500):
    from scipy.optimize import minimize
    h0 = h.astype(float).copy()
    
    def obj(x):
        x = np.clip(x, 0, 1)
        return compute_c5(x, n)
    
    def obj_grad(x):
        x = np.clip(x, 0, 1)
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        k_max = np.argmax(corr)
        l = k_max - n + 1
        grad = np.zeros(n)
        if l == 0:
            grad[:] = (1.0 - 2.0 * x) * dx
        else:
            for m in range(n):
                g = 0.0
                if 0 <= m - l < n:
                    g += (1.0 - x[m - l])
                if 0 <= m + l < n:
                    g -= x[m + l]
                grad[m] = g * dx
        return grad
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    bounds = [(0.0, 1.0)] * n
    result = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': 1e-10})
    if result.success and result.fun < obj(h0):
        return np.clip(result.x, 0, 1), float(result.fun)
    return h0, obj(h0)

        rng.shuffle(zeros)
        for idx in zeros[:int(target - s)]:
            h[idx] = 1.0
    return h

def fix_binary(h, n):
    h = (h > 0.5).astype(float)
    s = h.sum()
    t = n / 2.0
    if s > t:
        ones = np.where(h == 1)[0]
        for idx in ones[:int(s - t + 0.5)]:
            h[idx] = 0.0
    elif s < t:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:int(t - s + 0.5)]:
            h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    init_h = None
    try:
        init_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    init_binary = None
    init_n = None
    if init_h is not None:
        in_ = len(init_h)
        ne = in_ if in_ % 2 == 0 else in_ + 1
        h0 = np.zeros(ne)
        h0[:in_] = init_h
        h0 = fix_binary(h0, ne)
        init_binary = h0
        init_n = ne
        c5_0 = compute_c5(h0, ne)
        best_h, best_c5, best_n = h0.copy(), c5_0, ne

    n_list = [46, 48, 50, 52, 54, 56, 60, 64, 70, 80, 96, 112]
    if init_n is not None and init_n not in n_list and 40 <= init_n <= 128:
        n_list.insert(0, init_n)

    weights = {}
    for n in n_list:
        if n <= 52:
            weights[n] = 5.0
        elif n <= 64:
            weights[n] = 3.5
        elif n <= 96:
            weights[n] = 2.0
        else:
            weights[n] = 1.2
    total_w = sum(weights.values())
    avail = max(1.0, budget_s * 0.82 - (time.time() - t0))

    for n in n_list:
        if time.time() - t0 > budget_s * 0.88:
            break
        t_n = avail * weights[n] / total_w
        n_restarts = max(1, min(5, int(t_n / 14)))

        for r in range(n_restarts):
            if time.time() - t0 > budget_s * 0.88:
                break
            t_per = t_n / n_restarts
            s = (seed * 7919 + n * 31 + r * 104729) % (2**31)

            if r == 0 and n == init_n and init_binary is not None:
                ini = init_binary
            elif r == 1 and n >= 50:
                ini = make_thue_morse(n, seed=s)
            else:
                ini = None

            t_sa = t_per * 0.65
            h_sa, c5_sa = sa_optimize(n, seed=s, time_budget=t_sa, init=ini)

            t_ls = t_per * 0.35
            h_ls, c5_ls = local_search_full(n, h_sa, time_budget=t_ls)

            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                best_n = n

    # Final polish on best solution
    if best_h is not None and time.time() - t0 < budget_s * 0.95:
        t_final = min(60, budget_s * 0.95 - (time.time() - t0))
        if t_final > 2:
            h_pol, c5_pol = local_search_full(best_n, best_h, time_budget=t_final)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    if best_h is None:
        n = 50
        best_h = np.zeros(n)
        best_h[::2] = 1.0
        best_n = n
        best_c5 = compute_c5(best_h, n)

    return best_h, best_c5, best_n
