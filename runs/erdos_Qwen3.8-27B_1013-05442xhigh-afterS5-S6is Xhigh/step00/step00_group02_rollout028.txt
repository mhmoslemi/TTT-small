import numpy as np
import time
from scipy.optimize import minimize, linprog

def project(h, n):
    """Project h onto {0 <= x <= 1, sum(x) = n/2}"""
    target = n / 2.0
    # Use bisection on Lagrange multiplier
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo+hi)/2, 0, 1)
    return proj

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    return np.max(c) * dx, c * dx

                break

            if val < best_c5:

                best_h = h.copy()

            k_max = np.argmax(c_vals)

def grad_shift(h, s, n):
    """Gradient of c[s] (the correlation at shift s) w.r.t. h"""
    g = np.zeros(n)
    if s >= 0:
        m = s
        if m < n:

def project_h(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        c = np.clip(h - mid, 0.0, 1.0)
        if c.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def eval_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def grad_shift(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        m = s
        if m < n:
            g[m:] += (1.0 - h[:n - m])
            g[:n - m] -= h[m:]
    else:
        m = -s
        if m < n:
            g[:n - m] += (1.0 - h[m:])
            g[m:] -= h[:n - m]
    return g


def grad_at_max(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    k = int(np.argmax(c))
    s = k - (n - 1)
    return grad_shift(h, s, n) * dx


def smoothed_grad(h, tau):
