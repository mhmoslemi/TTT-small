import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta)

def smooth_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    ev = np.exp(beta * (corr - m))
    w = ev / np.sum(ev)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        s = k - (n - 1)
        wk = w[k] * dx
        if s == 0:
            grad[:] += wk * (1.0 - 2.0 * h)
        elif s > 0:
            if n - s > 0:
                grad[:n - s] += wk * (1.0 - h[s:])
            if s < n:
                grad[s:] -= wk * h[:n - s]
        else:
            sp = -s
            if sp < n:
                grad[sp:] += wk * (1.0 - h[:n - sp])
            if n - sp > 0:
                grad[:n - sp] -= wk * h[sp:]
    return grad

def local_search(h, n, max_iter, t0, budget, seed_rng):
    c5 = compute_c5(h, n)
    for it in range(max_iter):

            break
        scale = 0.15 * max(0.3, 1.0 - it / max_iter)
        h_p = h + seed_rng.randn(n) * scale
        h_p = project_feasible(h_p, n)
        c5_p = compute_c5(h_p, n)
        if c5_p < c5:
            h = h_p
            c5 = c5_p
    return h, c5

def slsqp_optimize(h0, n, beta, maxiter):
    def obj(x):
        return smooth_obj(x, n, beta)
    def jac(x):
        return smooth_grad(x, n, beta)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=jac, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-10})
    return project_feasible(res.x, n)
