from scipy import optimize
import time

def compute_correlations(h):
    """Compute all correlations correlate(h, 1-h, 'full')"""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    return np.max(compute_correlations(h)) * dx

def smooth_max(corr, T=10.0):
    """Smooth approximation to max using log-sum-exp"""
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m))) / T

def smooth_max_grad(corr, T=10.0):
    """Gradient of smooth max (soft assignment weights)"""
    m = np.max(corr)
    exp_corr = np.exp(corr - m)
    weights = exp_corr / np.sum(exp_corr)
    return weights

def grad_c(h):
    """Gradient of all correlations with respect to h"""
    N = len(h)
    # corr[shift] = sum_n h[n] * (1-h[n+shift])
    # d corr[shift] / d h[i] = (1-h[i+shift]) - h[i-shift] (with bounds)
    # Shape: (2N-1, N)
    # corr index k corresponds to shift = k - (N-1)
    # For shift s, d corr / d h[i] = (1-h[i+s]) * [0<=i+s<N] - h[i-s] * [0<=i-s<N]
    
    # Let's compute it more efficiently

    # d corr[k] / d h[i] = (1-h[i+shift]) - h[i-shift]
    # where shift = k - (N-1)

    # d corr[k] / d h[i] = (1-h[i+k-(N-1)]) * [valid] + sum_n h[n] * (-1) * [n+k-(N-1) == i]
    # = (1-h[i+k-(N-1)]) * [0<=i+k-(N-1)<N] - h[i-k+(N-1)] * [0<=i-k+(N-1)<N]

import numpy as np
import time

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    return float(np.max(compute_correlations(h)) * dx)

def round_to_binary(h, N_target_sum=None):
    """Round h to binary, fixing sum to N/2."""
    N = len(h)
    target = N_target_sum if N_target_sum is not None else N / 2.0
    h_bin = (h >= 0.5).astype(float)
    n_ones = h_bin.sum()
    if n_ones < target:
        zi = np.where(h_bin == 0)[0]
        # Prioritize positions where h is closest to 0.5 from above
        order = zi[np.argsort(-h[zi])]
        for idx in order[:int(target - n_ones)]:
            h_bin[idx] = 1.0
    elif n_ones > target:
        oi = np.where(h_bin == 1)[0]
        order = oi[np.argsort(h[oi])]
        for idx in order[:int(n_ones - target)]:
            h_bin[idx] = 0.0
    return h_bin

def hill_climb_binary(h, max_passes=1000, time_limit=None, t0=None):
    """Hill climbing: try all pair flips, accept improvements."""
    N = len(h)
    c5 = compute_c5(h)
    for p in range(max_passes):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        for i in zeros:
            for j in ones:
                h[i] = 1.0
                h[j] = 0.0
                new_c5 = compute_c5(h)
                if new_c5 < c5 - 1e-15:
                    c5 = new_c5
                    improved = True
                    # Restart search from current state
                    break
                else:
                    h[i] = 0.0
                    h[j] = 1.0
            if improved:
                break
        if not improved:
            break
    return h, c5

def simulated_annealing_binary(h, t0_factor=0.1, time_budget=None, t0_abs=None, seed=42):
    """Simulated annealing on binary sequences."""
    np.random.seed(seed)
    N = len(h)
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    
    T = t0_factor * c5  # initial temperature
    T_min = T * 1e-4
    cooling = 0.9995
    
    while T > T_min:
        if time_budget is not None and t0_abs is not None:
            if time.time() - t0_abs > time_budget:
                break
        
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        i = zeros[np.random.randint(len(zeros))]
        j = ones[np.random.randint(len(ones))]
        
        h[i], h[j] = 1.0, 0.0
        new_c5 = compute_c5(h)
        
        if new_c5 < c5 or np.random.rand() < np.exp(-(new_c5 - c5) / T):
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i], h[j] = 0.0, 1.0
        
        T *= cooling
    
    return best_h, best_c5

def continuous_refinement(h, N_target_sum, time_budget, t0_abs, seed=42):
    """Continuous optimization with projected gradient and smooth max."""
    np.random.seed(seed)
    N = len(h)
    target = N_target_sum
    
    # Start from a smoothed version of h
    h_c = h.copy().astype(float)
    # Add some randomness
    h_c = np.clip(h_c + 0.05 * np.random.randn(N), 0, 1)
    # Project to constraint
    diff = target - h_c.sum()
    h_c += diff / N
    h_c = np.clip(h_c, 0, 1)
    
    c5 = compute_c5(h_c)
    best_c5 = c5
    best_h = h_c.copy()
    
    # Smoothed objective: F(h) = dx * logsumexp(T * corr) / T
    # Gradient: dx * sum_s softmax_s(corr) * dcorr[s]/dh
    
    T_smooth = 3.0
    lr = 0.5
    dx = 2.0 / N
    
    for step in range(100000):
        if time.time() - t0_abs > time_budget:
            break
        
        corr = compute_correlations(h_c)
        m = np.max(corr)
        exp_c = np.exp(T_smooth * (corr - m))
        w = exp_c / np.sum(exp_c)
        
        # Compute gradient
        grad = np.zeros(N)
        for i in range(N):
            # d corr[s] / d h[i] = (1-h[i+shift]) - h[i-shift]
            shift = np.arange(2*N-1) - (N-1)
            # First term: (1-h[i+shift]) where 0<=i+shift<N
            valid1 = (i + shift >= 0) & (i + shift < N)
            term1 = np.zeros(2*N-1)
            term1[valid1] = 1.0 - h_c[i + shift[valid1]]
            # Second term: -h[i-shift] where 0<=i-shift<N
            valid2 = (i - shift >= 0) & (i - shift < N)
            term2 = np.zeros(2*N-1)
            term2[valid2] = h_c[i - shift[valid2]]
            
            grad[i] = np.dot(w, term1 - term2)
        
        # Gradient step
        h_c = h_c - lr * grad * dx
        
        # Project onto [0,1]^N ∩ {sum = target}
        h_c = np.clip(h_c, 0, 1)
        diff = target - h_c.sum()
        if abs(diff) > 1e-10:
            # Distribute proportionally to available room
            if diff > 0:
                room = 1.0 - h_c
            else:
                room = h_c.copy()
            total_room = room.sum()
            if total_room > 1e-10:
                h_c += diff * room / total_room
                h_c = np.clip(h_c, 0, 1)
        
        new_c5 = compute_c5(h_c)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_c.copy()
        
        # Adaptive learning rate
        if step % 1000 == 999:
            lr *= 0.95
    
    return best_h, best_c5

```python
def continuous_optimize(h_init, S, T_schedule, lr_schedule, max_steps, deadline):
    """Projected gradient descent with smooth max."""
    N = len(h_init)
    dx = 2.0 / N
    h = h_init.copy()
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for step in range(max_steps):
        if time.time() > deadline:
            break
        
        T = T_schedule[step] if step < len(T_schedule) else T_schedule[-1]
        lr = lr_schedule[step] if step < len(lr_schedule) else lr_schedule[-1]
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # Smooth max weights
        max_c = np.max(corr)
        exp_c = np.exp(T * (corr - max_c))
        w = exp_c / exp_c.sum()
        
        # Gradient
        grad = np.zeros(N)
        shifts = np.arange(2*N-1) - (N-1)
        for i in range(N):
            # (1-h[i+shift]) for valid i+shift
            valid1 = (i + shifts >= 0) & (i + shifts < N)
            # -h[i-shift] for valid i-shift
            valid2 = (i - shifts >= 0) & (i - shifts < N)
            
            term1 = np.zeros(2*N-1)
            term1[valid1] = 1.0 - h[i + shifts[valid1]]
            term2 = np.zeros(2*N-1)
            term2[valid2] = h[i - shifts[valid2]]
            
            grad[i] = np.dot(w, term1 - term2)
        
        # Gradient step
        h_new = h - lr * dx * grad
        
        # Project
        h_new = project(h_new, S)
        
        # Accept or reject (simple: always accept, or use line search)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5
```
