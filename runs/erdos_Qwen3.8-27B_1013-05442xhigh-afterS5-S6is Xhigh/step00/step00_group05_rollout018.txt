def project_to_feasible(x, target_sum):

    lo, hi = float(x.min()) - target_sum, float(x.max())
    for _ in range(80):

        s = np.clip(x - tau, 0.0, 1.0).sum()

        else:

    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float((corr * dx).max())

    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[:m] += 1.0 - h[lag:]
            grad[lag:] -= h[:m]
    else:
        L = -lag
        m = n - L
        if m > 0:
            grad[L:] += 1.0 - h[:m]
            grad[:m] -= h[L:]
    return grad

def smooth_obj(h, dx, beta):
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full") * dx
    m = corr.max()
    shifted = np.clip(beta * (corr - m), -500, 500)
    return m + np.log(np.exp(shifted).sum()) / beta

def smooth_grad(h, dx, beta):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full") * dx
    m = corr.max()
    shifted = np.clip(beta * (corr - m), -500, 500)
    w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        grad += w[k] * active_gradient(h, lag)
    return grad * dx

    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    grad = active_gradient(h, lag)
    h_new = h - alpha * dx * grad
    return project_to_feasible(h_new, n * 0.5)

    n = len(h)
    t0 = time.time()
    best = h.copy()
    best_val = c5_eval(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    for beta in [50.0, 100.0, 200.0, 500.0]:
        if time.time() - t0 > max_time:
            break
        def f(x):
            return smooth_obj(x, dx, beta)
        def g(x):
            return smooth_grad(x, dx, beta)
        for attempt in range(3):
            if time.time() - t0 > max_time:
                break
            try:
