    # Validate
    assert best_h is not None
    assert 0 <= np.min(best_h) and np.max(best_h) <= 1 + 1e-12
    s = best_h.sum()
    assert abs(s - best_n / 2.0) < 1.0, f"sum={s}, target={best_n/2}"

    return best_h, best_c5, best_n
