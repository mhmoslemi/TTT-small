        elif r == 2:

            x0 = (h0 > np.median(h0)).astype(float)

                      options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-8})
        
        h_opt = normalize_h(res.x, n)
        c5_opt = evaluate_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 2: Subgradient refinement
    if time.time() - t_start < budget_s - 30:
        h = best_h.copy()
        for it in range(5000):
            if time.time() - t_start > budget_s - 10:
                break
            
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            l = max_idx - (n - 1)
            
            # Gradient of corr at lag l w.r.t. h
            grad = np.zeros(n)
            # corr at lag l = sum_m h[m+l]*(1-h[m]) for valid m
            # d/dh[j] = (1-h[j-l]) if 0<=j-l<n, minus h[j+l] if 0<=j+l<n
            jm = j - l  # where h[j-l] appears as the "1-h[m]" term (m=j, so j+l = j+l... wait)
            
            # Let me re-derive: c_l = sum_m h[m+l]*(1-h[m])
            # d c_l / d h[j] = sum_m [d h[m+l]/d h[j] * (1-h[m]) + h[m+l] * d(1-h[m])/d h[j]]
            # = (1-h[j-l]) * I(0<=j-l<n) - h[j+l] * I(0<=j+l<n)
            # (first term: m+l=j => m=j-l, valid if 0<=j-l<n)
            # (second term: m=j, valid if 0<=j+l<n)
            
            for j in range(n):
                g = 0.0
                if 0 <= j - l < n:
                    g += (1.0 - h[j - l])
                if 0 <= j + l < n:
                    g -= h[j + l]
                grad[j] = g
            grad *= dx
            
            # Projected subgradient step
            lr = 0.5 / (1.0 + 0.01 * it)
            h_new = h - lr * grad
            h_new = normalize_h(h_new, n)
            
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
    
    return (best_h, best_c5, n)

Let me try yet another approach: use the fact that for binary h (h ∈ {0,1}), the problem becomes combinatorial. For binary h with sum = n/2 (i.e., n/2 ones and n/2 zeros), the overlap at lag l is the number of positions m where h[m+l]=1 and h[m]=0 (times dx). This is the Hamming-like overlap.

For the binary case, at lag 0: overlap = 0 (since h(1-h) = 0 always for binary h).
At lag l: overlap = number of m where h[m+l]=1 and h[m]=0.

We want to minimize the maximum over l of this count, times dx.

This is related to the problem of finding a binary sequence with low auto-correlation. In fact, it's related to Barker sequences or similar constructs.

For a binary sequence of length n with n/2 ones, the "asymmetric" auto-correlation at lag l is:
R(l) = sum_m (h[m+l] - 0.5)(h[m] - 0.5)

But our quantity is different: it's the count of (1,0) pairs at lag l, which is:
C(l) = sum_m h[m+l](1-h[m])

Note that C(l) + C(-l) = sum_m h[m+l](1-h[m]) + h[m](1-h[m+l]) = sum_m |h[m+l]-h[m]|... no that's not right either.

Actually: C(l) = #(m: h[m+l]=1, h[m]=0) and C(-l) = #(m: h[m-l]=1, h[m]=0) = #(m: h[m]=1, h[m+l]=0).

So C(l) + C(-l) = #(m: h[m+l]≠h[m]) = Hamming distance between h and shift_l(h).

And C(l) - C(-l) = #(m: h[m+l]=1, h[m]=0) - #(m: h[m]=1, h[m+l]=0) = sum_m (h[m+l] - h[m]) = (number of ones in the overlap region of h shifted by l) - (number of ones in the overlap region of h).

import numpy as np

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def normalize_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_val(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

        pen = 100.0 * (np.sum(h) - target_sum) ** 2
        return soft_max_val(h, alpha, n) + pen
    return obj
