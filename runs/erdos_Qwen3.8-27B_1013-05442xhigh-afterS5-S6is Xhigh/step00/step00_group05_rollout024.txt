    return float(np.max(corr) * dx)

def soft_max_c5(h, dx, eps):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Soft max approximation
    m = np.max(corr)
    return m + eps * np.log(np.mean(np.exp((corr - m) / eps)))

def fix_sum(h, target):
    """Fix the sum of h to equal target while keeping 0<=h<=1"""
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Try to adjust entries not at boundary
    mask = (h > 1e-6) & (h < 1.0 - 1e-6)
    n_mask = mask.sum()
    if n_mask > 0:
        adj = diff / n_mask
        if -h[mask].min() + 1e-9 <= adj <= 1.0 - h[mask].max() - 1e-9:
            h = h.copy()
            h[mask] += adj
        else:
            # Need more complex adjustment
            h = h.copy()
            if diff > 0:
                # Need to increase sum, prioritize entries closer to 1
                order = np.argsort(-h)
                for i in order:
                    space = 1.0 - h[i]
                    add = min(space, diff)
                    h[i] += add
                    diff -= add
                    if abs(diff) < 1e-12:

                for i in order:
                    space = h[i]
                    rem = min(space, -diff)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_c5(h, dx, eps):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + eps * np.log(np.mean(np.exp((corr - m) / eps)))

def fix_sum(h, target):
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(-h)
        for i in order:
            space = 1.0 - h[i]
            add = min(space, diff)
            h[i] += add
            diff -= add
            if abs(diff) < 1e-14:
                break
    else:
        order = np.argsort(h)
        for i in order:
            space = h[i]
            rem = min(space, -diff)
            h[i] -= rem
            diff += rem
            if abs(diff) < 1e-14:
                break
    return h

def make_initial(n, seed_val, style=0):
    rng = np.random.RandomState(seed_val)
    if style == 0:
        h = rng.random(n) * 0.6 + 0.2
    elif style == 1:
        h = np.zeros(n)
        bs = max(1, n // 6)
        for b in range(0, n, bs * 2):
            h[b:b+bs] = 1.0
    elif style == 2:
        h = np.ones(n) * 0.5
        h[:n//4] = 0.7
        h[-n//4:] = 0.3
    else:
        h = np.zeros(n)
        h[n//4:3*n//4] = 0.5
    return fix_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h_parent = None

        best_h = h_parent.copy()
        best_c5 = compute_c5(best_h, 2.0 / len(best_h))

    def time_left():
        return budget_s - (time.time() - start_time)

    n_list = []
    if h_parent is not None:
        n_list.append(len(h_parent))
    for n in [60, 70, 80, 90, 100, 110, 120, 140, 160, 200, 250, 300]:
        if n not in n_list:
            n_list.append(n)

    all_inits = []
    if h_parent is not None:
        all_inits.append(h_parent.copy())

    for n in n_list:
        if time_left() < 90:
            break
        dx = 2.0 / n
        bounds = Bounds(0.0, 1.0)
        target_sum = n / 2.0

        # Build initial guesses for this n

        if best_h is not None and len(best_h) != n:

        best_c5 = 0.5

    return best_h, best_c5, len(best_h)
