            break

def interpolate_to_n(src, n_new):
    """Interpolate a binary sequence to a new length and binarize."""
    n_src = len(src)
    x_old = np.linspace(0, 2, n_src, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    interp = np.interp(x_new, x_old, np.append(src, src[0]))  # periodic
    # Binarize while maintaining the constraint
    # Sort by interpolated value, pick the top n_new/2
    threshold = np.sort(interp)[n_new//2]

def structured_init(n, rng):
    """Create a near-equipspaced binary sequence with perturbation."""
    h = np.zeros(n)
    k = n // 2
    # Start with equidistant positions
    base = np.arange(k) * (n / k)
    # Add small perturbations
    pert = rng.randint(-2, 3, size=k)
    positions = np.mod(np.round(base + pert).astype(int), n)
    # Handle collisions
    positions = np.unique(positions)
    if len(positions) < k:
        # Fill in gaps
        used = set(positions)
        for i in range(n):
            if i not in used:
                positions = np.append(positions, i)
            if len(positions) == k:
                break
    positions = positions[:k]
    h[positions] = 1.0
    return h

def evaluate_score(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def interp_binarize(src, n_new):
    n = len(src)
    x_old = np.arange(n, dtype=float)
    x_new = np.linspace(0, n, n_new, endpoint=False)
    src_ext = np.concatenate([src, [src[0], src[1]] if n > 1 else [src[0]]*2])
    x_old_ext = np.concatenate([x_old, [n, n+1]])
    interp = np.interp(x_new, x_old_ext, src_ext)
    k = n_new // 2
    thresh = np.sort(interp)[-k]
    result = (interp >= thresh).astype(float)
    cnt = int(result.sum())
    if cnt < k:
        zi = np.where(result == 0)[0]
        zi = zi[np.argsort(interp[zi])[::-1]][:k - cnt]
        result[zi] = 1.0
    elif cnt > k:
        oi = np.where(result == 1)[0]
        oi = oi[np.argsort(interp[oi])[:cnt - k]]
        result[oi] = 0.0
    return result

def structured_random(n, rng):
    h = np.zeros(n)
    k = n // 2
    step = n / k
    base = np.arange(k) * step
    pert = rng.randint(-2, 3, size=k)
    pos = np.mod((base + pert).astype(int), n)
    pos = np.unique(pos)

        pos = np.concatenate([pos, extra[:k - len(pos)]])
    pos = pos[:k]
    h[pos] = 1.0
    return h
