import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def compute_slopes(h, i, j):
    """Compute slopes s[k] for the pair (i, j) swap"""
    n = len(h)
    m = np.arange(2*n-1) - (n-1)
    
    s = np.zeros(2*n-1)
    
    # For h[i] += delta:
    # contribution = (1-h[i-m]) if i-m valid, minus h[i+m] if i+m valid
    im = i - m
    ip = i + m
    valid_im = (im >= 0) & (im < n)
    valid_ip = (ip >= 0) & (ip < n)
    s[valid_im] += 1 - h[im[valid_im]]
    s[valid_ip] -= h[ip[valid_ip]]
    
    # For h[j] -= delta (so contribution is negated):
    jm = j - m
    jp = j + m
    valid_jm = (jm >= 0) & (jm < n)
    valid_jp = (jp >= 0) & (jp < n)
    s[valid_jm] -= (1 - h[jm[valid_jm]])
    s[valid_jp] += h[jp[valid_jp]]
    
    return s

def find_optimal_delta(corr, slopes, h, i, j):
    """Find delta that minimizes max(corr + delta*slopes) subject to constraints"""
    n = len(h)
    # Constraints: 0 <= h[i]+delta <= 1, 0 <= h[j]-delta <= 1
    lo = max(0, -h[i])
    hi = min(1 - h[i], h[j])
    
    if lo >= hi:
        return 0
    
    def objective(delta):
        return np.max(corr + delta * slopes)
    
    # Try to find the minimum of this piecewise linear convex function
    # Use minimize_scalar
    result = minimize_scalar(objective, bounds=(lo, hi), method='bounded')
    return result.x

def pair_swap_optimize(h, max_passes=50):
    """Coordinate descent with pair swaps"""
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    h = h.copy()
    
    for _pass in range(max_passes):
        improved = False
        corr = np.correlate(h, 1-h, mode="full")
        
        # Shuffle order for diversity
        indices = np.arange(n)
        np.random.shuffle(indices)
        
        for ii in indices:
            for jj in indices:
                if ii == jj:
                    continue
                slopes = compute_slopes(h, ii, jj)
                delta = find_optimal_delta(corr, slopes, h, ii, jj)
                if abs(delta) > 1e-10:
                    h[ii] += delta
                    h[jj] -= delta
                    # Clip to [0,1]
                    h[ii] = np.clip(h[ii], 0, 1)
                    h[jj] = np.clip(h[jj], 0, 1)
                    # Recompute correlation
                    corr = np.correlate(h, 1-h, mode="full")
                    improved = True
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        
        if not improved:

    return h, best_c5

        if time.time() > dl:
            break
        hr = (np.random.rand(n0) < 0.5).astype(float)
        h1, c1 = optimize_h(hr, 12, dl)
        if c1 < best_c5:
            best_c5, best_h = c1, h1
    # Phase 3: upscale to 2n
    if time.time() < start + budget_s * 0.35:
        h2 = upscale(best_h)
        dl = start + budget_s * 0.68
        h2, c2 = optimize_h(h2, 40, dl)
        if c2 < best_c5:
            best_c5, best_h = c2, h2
    # Phase 4: upscale to 4n
    if time.time() < start + budget_s * 0.7:
        h3 = upscale(best_h)
        dl = start + budget_s * 0.9
        h3, c3 = optimize_h(h3, 8, dl)
        if c3 < best_c5:
            best_c5, best_h = c3, h3
    # Phase 5: final polish
    remaining = (start + budget_s - 12) - time.time()
    if remaining > 20:
        hf, cf = optimize_h(best_h, 15, start + budget_s - 12)
        if cf < best_c5:
            best_c5, best_h = cf, hf
    # Ensure valid return
    n_final = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h += (n_final / 2.0 - best_h.sum()) / n_final
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_final
