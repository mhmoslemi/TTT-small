import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode="full")
    return np.max(c) * dx

def local_search_optimize(h, n, max_iters, seed):
    rng = np.random.RandomState(seed)
    current = h.copy()
    current_score = compute_c5(current)
    best = current.copy()
    best_score = current_score
    
    ones = np.where(current == 1)[0]
    zeros = np.where(current == 0)[0]
    
    T = 0.1  # initial temperature
    T_min = 1e-6
    
    for it in range(max_iters):
        # Pick a random 1 and 0 to swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        
        # Make the swap
        current[i] = 0
        current[j] = 1
        
        new_score = compute_c5(current)
        
        delta = new_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # Accept
            current_score = new_score
            ones = np.where(current == 1)[0]
            zeros = np.where(current == 0)[0]
            if new_score < best_score:
                best_score = new_score
                best = current.copy()
        else:
            # Reject
            current[i] = 1
            current[j] = 0
        
        # Cool
        T = max(T_min, T * 0.9999)
    
    return best, best_score

    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        flip = rng.choice(idx, n_ones - target, replace=False)
        h[flip] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        flip = rng.choice(idx, target - n_ones, replace=False)
        h[flip] = 1
    return h

def thue_morse_n(n):
    return np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)

def sa_optimize(h0, n, rng, deadline, T_init=0.04):
    h = h0.copy()
    score = compute_c5(h)
    best_h = h.copy()
    best_score = score
    T = T_init
    stall_count = 0
    n_improves = 0

    while time.time() < deadline:
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        for _ in range(300):
            if time.time() >= deadline:
                break
            a = ones_idx[rng.randint(len(ones_idx))]
            b = zeros_idx[rng.randint(len(zeros_idx))]
            h[a], h[b] = 0.0, 1.0
            new_score = compute_c5(h)
            delta = new_score - score
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                score = new_score
                ones_idx = np.where(h == 1)[0]
                zeros_idx = np.where(h == 0)[0]
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
                    stall_count = 0
                else:
                    stall_count += 1
            else:
                h[a], h[b] = 1.0, 0.0
            T = max(1e-9, T * 0.99997)
        if stall_count > 3000:
            T = T_init * 0.5
            stall_count = 0

    # Final hill-climb polish
    for _ in range(500):
        if time.time() >= deadline:
            break
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        a = ones_idx[rng.randint(len(ones_idx))]
        b = zeros_idx[rng.randint(len(zeros_idx))]
        h[a], h[b] = 0.0, 1.0
        new_score = compute_c5(h)
        if new_score < score:
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[a], h[b] = 1.0, 0.0

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    deadline = t0 + budget_s * 0.92

    try:
        parent = np.array(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError):
        parent = None

    best_h = None
    best_score = float('inf')

    n_parent = len(parent) if parent is not None and len(parent) % 2 == 0 else 0
    parent_bin = None
    if parent is not None and n_parent >= 20 and n_parent <= 200:
        parent_bin = make_binary(parent, n_parent, rng)

    n_list = []
    if n_parent > 0:
        n_list.append(n_parent)
    for n in [44, 46, 48, 50, 52, 54, 56, 60, 64, 68, 72, 80, 96, 112, 128, 160, 200]:
        if n not in n_list:
            n_list.append(n)

    # Time allocation: more for smaller n (faster per iter), extra for parent
    weights = []
    for n in n_list:
        w = 1.0 / np.sqrt(n)
        if n == n_parent:
            w *= 2.5
        weights.append(w)
    total_w = sum(weights)
    avail = deadline - time.time()

    for i, n in enumerate(n_list):
        if time.time() >= deadline:
            break
        time_alloc = avail * weights[i] / total_w
        n_deadline = min(deadline, time.time() + max(5, time_alloc))

        # Build starting points
        h_starts = []
        if parent_bin is not None and n == n_parent:
            h_starts.append(parent_bin.copy())
        # Thue-Morse if it has exactly n/2 ones
        tm = thue_morse_n(n)
        if int(tm.sum()) == n // 2:
            h_starts.append(tm.copy())
        # Random starts
        for _ in range(3):
            hr = np.zeros(n)
            hr[rng.choice(n, n // 2, replace=False)] = 1.0
            h_starts.append(hr)

        # Run SA from each start with time split
        sub_deadlines = []
        t_start = time.time()
        for j in range(len(h_starts)):
            frac = (j + 1) / len(h_starts)
            sub_deadlines.append(min(n_deadline, t_start + (n_deadline - t_start) * frac))

        for j, h_s in enumerate(h_starts):
            if time.time() >= sub_deadlines[j]:
                continue
            h_opt, s = sa_optimize(h_s, n, rng, sub_deadlines[j])
            if s < best_score:
                best_score = s
                best_h = h_opt.copy()

    if best_h is None:
        n = n_parent if n_parent > 0 else 42
        best_h = np.zeros(n)
        best_h[rng.choice(n, n // 2, replace=False)] = 1.0
        best_score = compute_c5(best_h)

    return best_h.tolist(), best_score, len(best_h)
