from scipy.special import logsumexp

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj_grad(h, alpha, dx, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = alpha * dx * corr
    m = np.max(vals)
    exps = np.exp(vals - m)
    s = np.sum(exps)
    p = exps / s
    obj = (m + np.log(s)) / alpha
    
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        wi = p[mi]
        if wi < 1e-12:
            continue
        k = mi - (n - 1)
        t1 = np.zeros(n)
        t2 = np.zeros(n)
        if k >= 0:
            t1[:n-k] = 1.0 - h[k:]
            t2[k:] = h[:n-k]
        else:
            t1[-k:] = 1.0 - h[:n+k]
            t2[:n+k] = h[-k:]
        grad += wi * (t1 - t2)
    grad *= dx
    return obj, grad

def project(feasible_h, target_sum):

        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h
