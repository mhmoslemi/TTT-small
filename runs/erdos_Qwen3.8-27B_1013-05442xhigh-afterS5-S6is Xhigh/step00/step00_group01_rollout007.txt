import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def upsample_h(h, target_n):
    x_old = np.linspace(0.0, 2.0, len(h), endpoint=False)
    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    s = h_new.sum()
    if s > 1e-12:
        h_new *= (target_n / 2.0) / s
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = h_new.sum() - target_n / 2.0
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h_new)[::-1]
        else:
            order = np.argsort(h_new)
        for idx in order:
            if diff > 0:
                adj = min(diff, h_new[idx] - 1e-12)
                h_new[idx] -= adj
                diff -= adj
            else:
                adj = min(-diff, 1.0 - h_new[idx] - 1e-12)
                h_new[idx] += adj
                diff += adj
            if abs(diff) < 1e-12:
                break
    return h_new

def best_improve(h, max_time, eps_list=None):
    if eps_list is None:

    end = time.time() + max_time

        cur = compute_c5(h)
        best_delta = 0.0

        for eps in eps_list:
            for i in range(n):
                vi = h[i] + eps
                if vi < -1e-12 or vi > 1.0 + 1e-12:
                    continue
                for j in range(n):
                    if j == i:
                        continue

            break

        h[best_j] -= best_eps

    stall = 0

        T = T_start * (T_end / T_start) ** min(frac, 1.0)
