def project_onto_simplex(h, target_sum, lower=0, upper=1):
    """Project h onto the simplex {x : lower <= x_i <= upper, sum(x) = target_sum}."""
    # Use iterative projection
    # First clip to bounds
    h = np.clip(h, lower, upper)
    # Then adjust to meet sum constraint

        free = ~at_lower & ~at_upper

        h[free] -= excess / np.sum(free)

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(x, target_sum):

    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
