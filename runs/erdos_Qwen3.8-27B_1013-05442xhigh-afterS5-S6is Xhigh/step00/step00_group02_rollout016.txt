```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1-h, mode='full')
    return np.max(conv) * dx

def gradient_smooth_max(h, n, T=0.01):
    """Compute gradient of smooth max of correlation * dx"""
    dx = 2.0 / n
    conv = np.correlate(h, 1-h, mode='full')
    # Softmax weights
    conv_shifted = conv - np.max(conv)
    weights = np.exp(conv_shifted / T)
    weights /= np.sum(weights)
    
    # Now compute gradient
    # For each lag k (corresponding to index m = k + n - 1 in conv):
    # d(overlap[k])/d(h[j]) = (1-h[j+k])*I(0<=j+k<n) - h[j-k]*I(0<=j-k<n)
    # 
    # So grad[j] = dx * sum_k weights[k+n-1] * [(1-h[j+k])*I(0<=j+k<n) - h[j-k]*I(0<=j-k<n)]
    
    grad = np.zeros(n)
    for j in range(n):
        for k in range(-(n-1), n):
            m = k + n - 1
            term = 0.0
            if 0 <= j+k < n:
                term += (1 - h[j+k])
            if 0 <= j-k < n:
                term -= h[j-k]
            grad[j] += weights[m] * term
    grad *= dx
    return grad

def project_to_feasible(v, S):
    """Project v onto {h : 0 <= h <= 1, sum(h) = S} using bisection on lambda"""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(v - mid, 0, 1)
        s = np.sum(projected)
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial values
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
        n = len(h)
    else:
        n = 45
        h = np.full(n, 0.5)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Gradient descent with projection
    T = 0.05  # temperature for smooth max
    lr = 0.1
    S = n / 2.0
    
    for iteration in range(5000):
        if time.time() - start_time > budget_s - 5:
            break
        
        grad = gradient_smooth_max(h, n, T)
        h_new = h - lr * grad
        h = project_to_feasible(h_new, S)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            lr = max(0.01, lr * 0.95)
            T = max(0.005, T * 0.98)
    
    return best_h, best_c5, n
```

```python
    dx = 2.0 / n
    conv = np.correlate(h, 1-h, mode='full')
    conv_max = np.max(conv)
    conv_shifted = (conv - conv_max) / T
    weights = np.exp(conv_shifted)
    weights /= np.sum(weights)

            m = k + n - 1
            term = 0.0
            jk = j + k
            jmk = j - k
            if 0 <= jk < n:
                term += (1.0 - h[jk])
            if 0 <= jmk < n:
                term -= h[jmk]
```

```python
def gradient_smooth_max(h, n, T):
    dx = 2.0 / n
    conv = np.correlate(h, 1-h, mode='full')  # length 2n-1
    conv_max = np.max(conv)
    weights = np.exp((conv - conv_max) / T)
    weights /= np.sum(weights)
    
    # For each j, compute:
    # grad[j] = dx * sum_k weights[k] * [(1-h[j+k])*I(0<=j+k<n) - h[j-k]*I(0<=j-k<n)]
    
    # Vectorize over j:
    # Create a matrix M where M[j, k] = (1-h[j+k])*I(0<=j+k<n) - h[j-k]*I(0<=j-k<n)
    # Then grad = dx * M @ weights
    
    # For the first term: (1-h[j+k]) for valid j+k
    j_idx = np.arange(n)
    k_idx = np.arange(-(n-1), n)
    
    # j+k matrix
    jk = j_idx[:, None] + k_idx[None, :]  # shape (n, 2n-1)
    valid1 = (jk >= 0) & (jk < n)
    term1 = np.where(valid1, 1.0 - h[jk.clip(0, n-1)], 0.0)
    
    # j-k matrix
    jmk = j_idx[:, None] - k_idx[None, :]
    valid2 = (jmk >= 0) & (jmk < n)
    term2 = np.where(valid2, h[jmk.clip(0, n-1)], 0.0)
    
    M = term1 - term2  # shape (n, 2n-1)
    grad = dx * (M @ weights)
    return grad
```

```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
```
```python
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx

def gradient_smooth_max(h, n, T):
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full')
    conv_max = np.max(conv)
    weights = np.exp((conv - conv_max) / T)
    weights /= np.sum(weights)
    j_idx = np.arange(n)
    k_idx = np.arange(-(n - 1), n)
    jk = j_idx[:, None] + k_idx[None, :]
    valid1 = (jk >= 0) & (jk < n)
    jk_c = np.clip(jk, 0, n - 1)
    term1 = np.where(valid1, 1.0 - h[jk_c], 0.0)
    jmk = j_idx[:, None] - k_idx[None, :]
    valid2 = (jmk >= 0) & (jmk < n)
    jmk_c = np.clip(jmk, 0, n - 1)
    term2 = np.where(valid2, h[jmk_c], 0.0)
    return dx * ((term1 - term2) @ weights)

def project_feasible(v, S):
```
