from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def smooth_grad(h, T):
    """Gradient of (1/T)logsumexp(T*corr) w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    w = np.exp(T * corr - np.max(T * corr))
    w /= w.sum()
    
    grad = np.zeros(n)
    for m in range(len(corr)):
        k = m - n + 1
        wm = w[m]
        if wm < 1e-15:

            grad[k:] += wm * (1.0 - h[:n-k])
            grad[:n-k] -= wm * h[k:]
        else:
            a = -k
            grad[:n-a] += wm * (1.0 - h[a:])
            grad[a:] -= wm * h[:n-a]

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    Tc = T * corr
    w = np.exp(Tc - Tc.max())
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(len(corr)):
        wm = w[m]
        if wm < 1e-14:
            continue
        k = m - n + 1
        if k >= 0:
            grad[k:] += wm * (1.0 - h[:n - k])
            grad[:n - k] -= wm * h[k:]
        else:
            a = -k
            grad[:n - a] += wm * (1.0 - h[a:])
            grad[a:] -= wm * h[:n - a]
    return grad * dx

def project_feasible(h, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def projected_grad_descent(h, n, max_iter, T_start, alpha_start, time_limit):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = T_start
    alpha = alpha_start
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        g = smooth_grad(h, T)
        h_new = project_feasible(h - alpha * g, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        if it % 80 == 79:
            T = min(T * 1.8, 400.0)
            alpha *= 0.85
    return best_h, best_c5

def slsqp_polish(h, n, time_limit):

    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        return (1.0 / T) * logsumexp(T * corr)
    def obj_grad(x):
        return smooth_grad(x, T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 3000, 'ftol': 1e-14})
        if compute_c5(res.x) < compute_c5(h):
            return res.x, compute_c5(res.x)
    except Exception:
        pass
    return h, compute_c5(h)
