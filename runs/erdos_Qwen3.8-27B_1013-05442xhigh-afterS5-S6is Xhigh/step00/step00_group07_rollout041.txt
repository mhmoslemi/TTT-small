    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            idx = idx[np.argsort(h[idx])[::-1]]
        else:
            idx = np.where(h > 0.0)[0]
            idx = idx[np.argsort(h[idx])]
        per = diff / len(idx) if len(idx) > 0 else 0
        for i in idx:
            if diff > 0:
                add = min(per, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-per, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def coord_descent(h, max_sweeps=5000, time_limit=None, start=None, n_grid=41):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    grid = np.linspace(0, 1, n_grid)
    omh = 1.0 - h
    for sweep in range(max_sweeps):
        if time_limit is not None and start is not None:
            if time.time() - start > time_limit:
                break
        improved = False
        for j in range(n):
            orig = h[j]
            best_v = orig
            best_val = np.max(np.correlate(h, omh, mode="full"))
            for v in grid:
                if abs(v - orig) < 1e-12:
                    continue
                h[j] = v
                omh[j] = 1.0 - v
                val = np.max(np.correlate(h, omh, mode="full"))
                if val < best_val:
                    best_val = val
                    best_v = v
            if best_v != orig:
                h[j] = best_v
                omh[j] = 1.0 - best_v
                improved = True
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        if not improved:
            break
    return h, best_c5

def simulated_annealing(h, time_limit, start, rng, n_iter=100000):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c5_cur = best_c5
    temp = 0.001
    for i in range(n_iter):
        if time.time() - start > time_limit:
            break
        j = rng.randint(n)
        old = h[j]
        step = rng.normal(0, 0.08)
        new = np.clip(old + step, 0.0, 1.0)
        if new == old:
            continue
        h[j] = new
        new_c5 = float(np.max(np.correlate(h, 1 - h, mode="full")) * dx)
        delta_c = new_c5 - c5_cur
        if delta_c < 0 or rng.random() < np.exp(-delta_c / max(temp, 1e-9)):
            c5_cur = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[j] = old
        temp = max(temp * 0.9998, 1e-8)
    return best_h, best_c5

def try_binary_projection(h):
    n = len(h)
    sorted_idx = np.argsort(h)[::-1]
    target = n / 2.0
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:int(target)]] = 1.0
    if target != int(target):
        frac = target - int(target)
        if int(target) < n:
            h_bin[sorted_idx[int(target)]] = frac
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(80, 0.5)

    n_init = len(h_init)
    h_init = repair_constraint(h_init)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent on initial resolution
    t_limit1 = budget_s * 0.45
    h1, c5_1 = coord_descent(h_init, max_sweeps=10000, time_limit=t_limit1, start=start, n_grid=41)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Try binary projection
    h_bin = try_binary_projection(best_h)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_h, best_c5 = h_bin, c5_bin
        h1b, c5_1b = coord_descent(best_h, max_sweeps=2000,
                                   time_limit=min(budget_s*0.1, budget_s - (time.time()-start) - 60),
                                   start=start, n_grid=21)
        if c5_1b < best_c5:
            best_h, best_c5 = h1b, c5_1b

    # Phase 2: Try higher resolutions
    elapsed = time.time() - start
    if elapsed < budget_s * 0.65:
        for n_try in [100, 130, 160]:
            if time.time() - start > budget_s - 60:
                break
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_try = np.interp(x_new, x_old, best_h)
            h_try = repair_constraint(h_try)
            rem = min(budget_s * 0.15, budget_s - (time.time() - start) - 30)
            if rem < 5:
                break
            h_opt, c5_opt = coord_descent(h_try, max_sweeps=3000, time_limit=rem, start=start, n_grid=31)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # Phase 3: Refinement with coordinate descent + SA on best solution
    elapsed = time.time() - start
    if elapsed < budget_s * 0.8:
        rem = min(budget_s * 0.1, budget_s - (time.time() - start) - 20)
        if rem > 5:
            h_ref, c5_ref = coord_descent(best_h, max_sweeps=5000, time_limit=rem, start=start, n_grid=61)
            if c5_ref < best_c5:
                best_h, best_c5 = h_ref, c5_ref

    # Phase 4: Simulated annealing
    elapsed = time.time() - start
    if elapsed < budget_s * 0.88:
        t_sa = budget_s - (time.time() - start) - 10
        if t_sa > 10:
            h_sa, c5_sa = simulated_annealing(best_h, t_sa, start, rng, n_iter=200000)
            if c5_sa < best_c5:
                best_h, best_c5 = h_sa, c5_sa

    # Final: one more coord descent pass if time remains
    elapsed = time.time() - start
    if elapsed < budget_s - 5:
        h_final, c5_final = coord_descent(best_h, max_sweeps=2000,
                                          time_limit=budget_s - (time.time() - start) - 2,
                                          start=start, n_grid=51)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final

    best_h = repair_constraint(best_h)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
