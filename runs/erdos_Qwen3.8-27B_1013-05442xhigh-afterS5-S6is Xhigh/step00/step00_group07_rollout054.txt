A better approach for the continuous optimization might be:
- Use the "t" parameter approach: minimize t subject to f_k(h) ≤ t for all k
- This converts the problem to a (non-convex) constrained optimization with a smooth objective (minimize t)

import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def smooth_max_obj(h, n, target, tau, penalty):
    dx = 2.0 / n
    c = correlate(h, 1.0 - h, mode='full') * dx
    c_max = float(np.max(c))
    smooth = c_max + np.log(np.sum(np.exp(tau * (c - c_max)))) / tau
    sp = penalty * (np.sum(h) - target) ** 2
    return smooth + sp

def make_binary(h0, target):
    h = (h0 >= 0.5).astype(float)
    n_ones = int(np.sum(h))
    if n_ones > target:
        idx = np.where(h == 1)[0]
        order = idx[np.argsort(h0[idx])]
        h[order[:n_ones - target]] = 0.0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        order = idx[np.argsort(-h0[idx])[target - n_ones:]]
        h[order] = 1.0
    return h

def sa_binary(h, target, seed, max_time):

    toc = time.time()
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = 0.008
    T_min = 1e-7
    cooling = 0.99995
    stuck = 0
    while time.time() - toc < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - best_c5
        if delta < 0 or rng.random() < np.exp(-max(delta, 0.0) / max(T, T_min)):
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T * cooling, T_min)
        if stuck > 5000:
            # Restart with perturbation
            h = best_h.copy()
            for _ in range(3):
                o = np.where(h == 1)[0]
                z = np.where(h == 0)[0]
                h[o[rng.randint(len(o))]] = 0.0
                h[z[rng.randint(len(z))]] = 1.0
            stuck = 0
    return best_h, best_c5

def hill_climb_binary(h, max_time):
    toc = time.time()
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    improved = True
    while improved and time.time() - toc < max_time:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_move = None
        best_new = best_c5
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c5 = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                if c5 < best_new:
                    best_new = c5
                    best_move = (i, j)
        if best_move is not None:
            i, j = best_move
            h[i] = 0.0
            h[j] = 1.0
            best_c5 = best_new
            improved = True
    return h, best_c5

def optimize_continuous(h0, n, target, max_time, tau=200.0, penalty=200.0):

    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_max_obj(x, n, target, tau, penalty)

    result = minimize(obj, h0, method='L-BFGS-B', bounds=bounds,

            t = x[n]
            c = correlate(h, 1.0 - x[:n], mode='full') * dx
            return t - c

        def cons_eq(x):

        x0 = np.append(hx, c5)
        b2 = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        cons = [{'type': 'ineq', 'fun': cons_ineq},
                {'type': 'eq', 'fun': cons_eq}]
        r2 = minimize(obj2, x0, method='SLSQP', bounds=b2,

            hx2 += (target - np.sum(hx2)) / n
            hx2 = np.clip(hx2, 0.0, 1.0)
            c52 = compute_c5(hx2)
            if c52 < c5:
                hx = hx2
                c5 = c52
    return hx, c5
