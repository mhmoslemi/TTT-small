from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def subgradient_c5(h, n_points):
    """Compute subgradient of C5 at h."""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    tol = 1e-8
    active = np.where(np.abs(corr - max_val) <= tol)[0]
    
    grad = np.zeros(n_points)
    n_active = len(active)
    
    for m in active:
        d = m - (n_points - 1)
        # For each j, compute dc/dh[j]
        # Term 1: (1 - h[j+d]) for valid j+d
        # Term 2: -h[j-d] for valid j-d
        for j in range(n_points):

            if 0 <= j + d < n_points:
                val += (1.0 - h[j + d])
            if 0 <= j - d < n_points:
                val -= h[j - d]
            grad[j] += val / n_active

            # j - d valid when d ≤ j ≤ n-1
            # Term 1: (1-h[j+d]) for j = 0..n-1-d
            # Term 2: -h[j-d] for j = d..n-1
            grad[:n_points-d] += (1.0 - h[d:])  # (1-h[j+d]) for j=0..n-1-d
            grad[d:] -= h[:n_points-d]           # -h[j-d] for j=d..n-1

            grad[d_abs:] += (1.0 - h[:n_points-d_abs])  # (1-h[j+d]) for j=d_abs..n-1
            grad[:n_points-d_abs] -= h[d_abs:]           # -h[j-d] for j=0..n-1-d_abs
    
    return grad / n_active * dx

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def subgradient_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(np.abs(corr - max_val) <= 1e-8)[0]
    grad = np.zeros(n_points)
    for m in active:
        d = m - (n_points - 1)
        if d >= 0:
            grad[:n_points - d] += (1.0 - h[d:])
            grad[d:] -= h[:n_points - d]
        else:
            da = -d
            grad[da:] += (1.0 - h[:n_points - da])
            grad[:n_points - da] -= h[da:]
    return grad / len(active) * dx


def project_feasible(h, n_points):
    target = n_points / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def smooth_obj(x, t):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    shifted = t * (corr - np.max(corr))
    return float(np.log(np.sum(np.exp(shifted))) / t * dx)


def smooth_grad(x, t):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    shifted = t * (corr - np.max(corr))
    w = np.exp(shifted)
    w /= np.sum(w)
    grad = np.zeros(n)
    for m in range(len(corr)):
        wm = w[m]
        if wm < 1e-12:
            continue
        d = m - (n - 1)
        if d >= 0:
            grad[:n - d] += wm * (1.0 - x[d:])
            grad[d:] -= wm * x[:n - d]
        else:
            da = -d
            grad[da:] += wm * (1.0 - x[:n - da])
            grad[:n - da] -= wm * x[da:]
    return grad * dx


def eq_fun(x):
    return np.sum(x) - len(x) / 2.0


def eq_jac(x):
    return np.ones(len(x))
