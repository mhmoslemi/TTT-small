        if d == 0:
            contrib = 1.0 - 2.0 * h

            contrib = np.zeros(n)
            contrib[:n-d] = 1.0 - h[d:]
            contrib[d:] = -h[:n-d]
        else:
            dp = -d
            contrib = np.zeros(n)
            contrib[dp:] = 1.0 - h[:n-dp]
            contrib[:n-dp] = -h[dp:]
        grad += w[k] * contrib
    grad *= dx
    grad += 2.0 * pen * (h.sum() - target)
    
    return total, grad
