To minimize this, we want h[i]² to be as large as possible given sum h[i] = n/2.
By Jensen's inequality, sum h[i]² is minimized when all h[i] are equal (= 1/2), giving sum h[i]² = n/4.
Then conv[0] = (n/2 - n/4) * 2/n = 1/4 = 0.25.

For binary: conv[0] = 0.

So binary is better at lag 0! But at other lags, continuous values might allow more "spread" in the correlation.

import numpy as np
from scipy.optimize import minimize
from itertools import combinations
import time

def compute_c5(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def make_binary(h, n, rng):
    target = n // 2
    h_bin = np.zeros(n)
    order = np.argsort(-h)
    h_bin[order[:target]] = 1.0
    return h_bin

def binary_local_search(h_bin, n, rng, start_time, max_time=300):
    c5 = compute_c5(h_bin, n)
    h = h_bin.copy()
    best_c5 = c5
    best_h = h.copy()
    plateau = 0
    
    while time.time() - start_time < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_swap_c5 = c5
        best_i = best_j = -1
        n_tries = min(400, len(ones) * len(zeros))
        
        for _ in range(n_tries):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0; h[j] = 1
            c5_t = compute_c5(h, n)
            h[i] = 1; h[j] = 0
            if c5_t < best_swap_c5:
                best_swap_c5 = c5_t
                best_i, best_j = i, j
        
        if best_i >= 0:
            h[best_i] = 0; h[best_j] = 1
            c5 = best_swap_c5
            plateau = 0
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            plateau += 1
            if plateau > 25:
                if rng.random() < 0.4:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i] = 0; h[j] = 1
                    c5 = compute_c5(h, n)
                    if c5 > best_c5 * 1.1:
                        h = best_h.copy()
                        c5 = best_c5
                plateau = 0
    
    return best_h, best_c5

def exhaustive_search(n, target=None):
    if target is None:
        target = n // 2
    best_c5 = float('inf')
    best_h = None
    for combo in combinations(range(n), target):
        h = np.zeros(n)
        h[list(combo)] = 1.0
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
    else:
        n_init = 45
        h_init = np.ones(n_init) * 0.5
    
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, n_init)
    best_n = n_init
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    # Phase 1: Binary local search at initial resolution
    h_bin = make_binary(h_init, n_init, rng)
    t1 = min(250, remaining() * 0.3)
    h_b, c5_b = binary_local_search(h_bin, n_init, rng, start_time, max_time=t1)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()
        best_n = n_init
    
    # Phase 2: Exhaustive search at small n
    for n_try in [20, 22, 24]:
        if remaining() < 80:
            break
        h_opt, c5_opt = exhaustive_search(n_try)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_try
    
    # Phase 3: Local search at various n values
    n_candidates = [28, 32, 36, 40, n_init, 50, 60, 80, 100]
    n_candidates = sorted(set(x for x in n_candidates if 10 <= x <= 200))
    
    for n_try in n_candidates:
        if remaining() < 80:
            break
        # Interpolate from best solution
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bin_try = make_binary(h_interp, n_try, rng)
        
        t_alloc = min(120, remaining() * 0.25)
        h_opt, c5_opt = binary_local_search(h_bin_try, n_try, rng, start_time, max_time=t_alloc)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_try
    
    # Phase 4: Multiple restarts at best_n
    n_restarts = 8
    for _ in range(n_restarts):
        if remaining() < 40:
            break
        h_rand = np.zeros(best_n)
        positions = rng.choice(best_n, best_n // 2, replace=False)
        h_rand[positions] = 1.0
        t_alloc = min(80, remaining() * 0.2)
        h_opt, c5_opt = binary_local_search(h_rand, best_n, rng, start_time, max_time=t_alloc)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    return best_h, best_c5, len(best_h)
