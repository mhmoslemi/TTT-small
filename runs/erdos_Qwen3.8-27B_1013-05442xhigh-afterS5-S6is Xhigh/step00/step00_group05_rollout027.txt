        t = t_high * (t_low / t_high) ** (i / max(n_iter, 1))
        # Choose move type: single swap or double swap
        if rng.random() < 0.7:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            pa = ones[rng.randint(len(ones))]
            pb = zeros[rng.randint(len(zeros))]
            h[pa], h[pb] = 0, 1
            new_c5 = compute_c5(h)
            d = new_c5 - cur
            if d < 0 or rng.random() < np.exp(-d / max(t, 1e-15)):
                cur = new_c5
                if cur < best:
                    best = cur
                    best_h = h.copy()
            else:
                h[pa], h[pb] = 1, 0
        else:
            # Double swap: two ones with two zeros
            if len(np.where(h == 1)[0]) < 2 or len(np.where(h == 0)[0]) < 2:
                continue
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            pa = ones[rng.randint(len(ones))]
            zeros_no_pb = zeros[zeros != pa]
            pb = zeros_no_pb[rng.randint(len(zeros_no_pb))]
            h2 = h.copy()
            # Find two ones
            oa = pa
            ones_other = ones[ones != oa]
            ob = ones_other[rng.randint(len(ones_other))]
            # Find two zeros
            za = pb
            zeros_other = zeros[zeros != za]
            zb = zeros_other[rng.randint(len(zeros_other))]
            # Apply: oa,ob become 0; za,zb become 1
            h_new = h.copy()
            h_new[oa], h_new[ob] = 0, 0
            h_new[za], h_new[zb] = 1, 1
            new_c5 = compute_c5(h_new)
            d = new_c5 - cur
            if d < 0 or rng.random() < np.exp(-d / max(t, 1e-15)):
                cur = new_c5
                h = h_new
                if cur < best:
                    best = cur
                    best_h = h.copy()
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError):
        h_init = None

    best_h = None
    best_c5 = float('inf')

    if h_init is not None:
        n = len(h_init)
        if n % 2 != 0:
            n += 1
    else:
        n = 70

    if h_init is not None and len(h_init) == n:
        h_start = h_init.copy()
    elif h_init is not None:
        xo = np.linspace(0, 2, len(h_init), endpoint=False)
        xn = np.linspace(0, 2, n, endpoint=False)
        h_start = np.interp(xn, xo, h_init)
    else:
        h_start = np.ones(n) * 0.5

    # Phase 1: Binarize + greedy
    h_bin = binarize(h_start, n)
    h_bin, c5 = greedy_swaps(h_bin, n_rounds=300, rng=np.random.RandomState(seed))
    best_c5 = c5
    best_h = h_bin.copy()

    # Phase 2: SA with multiple restarts (main optimization)
    time_left = deadline - time.time()
    n_restarts = max(4, int(time_left / 50))

    for r in range(n_restarts):
        if time.time() > deadline - 30:
            break

        if r == 0:
            h_cur = best_h.copy()
        elif r % 3 == 1:
            # Mild perturbation
            h_cur = best_h.copy()
            n_pert = max(2, n // 10)
            rng_p = np.random.RandomState(seed + r * 7 + 3)
            for _ in range(n_pert):
                ones = np.where(h_cur == 1)[0]
                zeros = np.where(h_cur == 0)[0]
                h_cur[ones[rng_p.randint(len(ones))]] = 0
                h_cur[zeros[rng_p.randint(len(zeros))]] = 1
        else:
            # Stronger perturbation
            h_cur = best_h.copy()
            n_pert = max(4, n // 5)
            rng_p = np.random.RandomState(seed + r * 13 + 1)
            for _ in range(n_pert):
                ones = np.where(h_cur == 1)[0]
                zeros = np.where(h_cur == 0)[0]
                h_cur[ones[rng_p.randint(len(ones))]] = 0
                h_cur[zeros[rng_p.randint(len(zeros))]] = 1

        h_cur, _ = greedy_swaps(h_cur, n_rounds=50, rng=np.random.RandomState(seed + r + 1))

        # Alternate between single-swap SA and pair SA
        if r % 2 == 0:
            t_h = 0.03 * (1.0 + 0.15 * min(r, 6))
            h_cur, c5_sa = sa_search(h_cur, t_h, 0.0002, n_iter=10**9,
                                     deadline=deadline - 30,
                                     rng=np.random.RandomState(seed + r * 100 + 10))
        else:
            t_h = 0.035 * (1.0 + 0.15 * min(r, 6))
            h_cur, c5_sa = pair_sa_search(h_cur, t_h, 0.0002, n_iter=10**9,
                                          deadline=deadline - 30,
                                          rng=np.random.RandomState(seed + r * 100 + 20))
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_cur.copy()

    # Phase 3: Try different n values
    if time.time() < deadline - 20:
        for n_alt in [60, 80, 100, 120, 90, 110]:
            if n_alt == n or time.time() > deadline - 10 or n_alt % 2 != 0:
                continue
            xo = np.linspace(0, 2, len(best_h), endpoint=False)
            xn = np.linspace(0, 2, n_alt, endpoint=False)
            h_alt = np.interp(xn, xo, best_h)
            h_alt = binarize(h_alt, n_alt)
            h_alt, c5_alt = greedy_swaps(h_alt, n_rounds=150,
                                         rng=np.random.RandomState(seed + n_alt))
            h_alt, c5_alt = sa_search(h_alt, 0.03, 0.0002, n_iter=10**9,
                                      deadline=deadline - 5,
                                      rng=np.random.RandomState(seed + n_alt + 50))
            if c5_alt < best_c5:
                best_c5 = c5_alt
                best_h = h_alt.copy()

    # Validate
    best_h = np.clip(best_h, 0, 1)
    n_final = len(best_h)
    if np.all((best_h == 0) | (best_h == 1)):
        n_ones = int(np.sum(best_h))
        if n_ones != n_final // 2:
            if n_ones > n_final // 2:
                idx = np.where(best_h == 1)[0]
                best_h[idx[-(n_ones - n_final // 2):]] = 0
            else:
                idx = np.where(best_h == 0)[0]
                best_h[idx[:n_final // 2 - n_ones]] = 1

    c5_final = compute_c5(best_h)
    return (list(best_h), c5_final, n_final)
