    # Phase 1: L-BFGS-B with increasing smoothness and penalty
    for attempt in range(15):
        if time.time() - t0 > budget_s * 0.45:
            break
        beta = 20.0 + attempt * 15.0
        pen = 50.0 + attempt * 80.0
        res = minimize(pen_obj, h.copy(), args=(dx, beta, pen, target_sum),
                       jac=pen_grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 400, 'ftol': 1e-15, 'gtol': 1e-13})
        h_new = project_feasible(np.clip(res.x, 0, 1), target_sum)
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Nelder-Mead on exact (non-smooth) max
    if time.time() - t0 < budget_s * 0.75:
        res = minimize(lambda x: compute_c5(x, dx), best_h.copy(),
                       method='Nelder-Mead',
                       options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-12,
                                'initial_simplex': best_h + 0.01, 'adaptive': True})
        h_new = project_feasible(np.clip(res.x, 0, 1), target_sum)
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 3: Fine-tune with high-beta L-BFGS-B
    for attempt in range(15):
        if time.time() - t0 > budget_s * 0.95:
            break
        beta = 300.0 + attempt * 100.0
        pen = 500.0 + attempt * 200.0
        res = minimize(pen_obj, best_h.copy(), args=(dx, beta, pen, target_sum),
                       jac=pen_grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 600, 'ftol': 1e-16, 'gtol': 1e-15})
        h_new = project_feasible(np.clip(res.x, 0, 1), target_sum)
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 4: Pair coordinate descent for final polish
    if time.time() - t0 < budget_s * 0.98:
        cur = best_h.copy()
        cur_c5 = best_c5
        for _ in range(50000):
            if time.time() - t0 > budget_s * 0.98:
                break
            cur = pair_cd_step(cur, dx, n, target_sum)
            c5 = compute_c5(cur, dx)
            if c5 < cur_c5:
                cur_c5 = c5
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = cur.copy()

    final_c5 = compute_c5(best_h, dx)
    return best_h, final_c5, n
