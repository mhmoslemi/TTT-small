= h[j] * (1 - h[j+d] - h[j-d]) + h[j-d]

The coefficient of h[j] is (1 - h[j+d] - h[j-d]), which can be positive or negative.

So for each lag k = n-1+d that involves j, the correlation is linear in h[j]. The max of these linear functions in h[j] is a convex function of h[j]. We want to minimize this max, so we're minimizing a convex function of h[j] — which is a 1D convex optimization problem that can be solved in closed form!

This gives us an excellent coordinate descent algorithm:
1. For each j from 0 to n-1:
   a. Determine all lags k that involve h[j]
   b. For each such lag, compute the coefficient and constant of corr[k] as a linear function of h[j]
   c. Find h[j] ∈ [0,1] that minimizes max_k (a_k * h[j] + b_k)
   d. Update h[j]
2. Repeat until convergence

The 1D minimization in step 1c: minimize max_k (a_k * t + b_k) subject to 0 ≤ t ≤ 1.

This is a linear programming problem in 1D, which can be solved by:
- The optimal t is where two of the linear functions intersect (or at a boundary)
- We can find it by checking all pairs of lines and the boundaries

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h

def coord_descend(h, n, max_sweeps=200, tol=1e-13):
    h = h.copy()
    for sweep in range(max_sweeps):
        max_change = 0.0
        for j in range(n):
            old = h[j]
            def obj_1d(t):
                h[j] = t
                corr = np.correlate(h, 1 - h, mode="full")
                return np.max(corr)
            res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 150, 'xatol': tol})
            h[j] = res.x
            max_change = max(max_change, abs(h[j] - old))
        h = fix_sum(h, n)
        if max_change < tol * 10:
            break
    return h

def try_binary_greedy(h, n, rng, max_iter=500):
    """For even n, try to find a good binary solution near h."""
    if n % 2 != 0:
        return None
    target = n // 2
    h_bin = (h > 0.5).astype(float)
    curr = int(np.sum(h_bin))
    if curr > target:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h_bin[ones])
        for idx in order[:curr - target]:
            h_bin[idx] = 0
    elif curr < target:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h_bin[zeros])
        for idx in order[:target - curr]:
            h_bin[idx] = 1
    c5 = compute_c5(h_bin, n)
    for _ in range(max_iter):
        improved = False
        idx = rng.integers(0, n)
        if h_bin[idx] == 1:
            zeros = np.where(h_bin == 0)[0]
            if len(zeros) == 0:
                continue
            cand = zeros[rng.integers(0, len(zeros))]
        else:
            ones = np.where(h_bin == 1)[0]
            if len(ones) == 0:
                continue
            cand = ones[rng.integers(0, len(ones))]
        h_bin[idx], h_bin[cand] = 1.0 - h_bin[idx], 1.0 - h_bin[cand]
        c5_new = compute_c5(h_bin, n)
        if c5_new < c5:
            c5 = c5_new
            improved = True
        else:
            h_bin[idx], h_bin[cand] = 1.0 - h_bin[idx], 1.0 - h_bin[cand]
        if not improved and rng.random() < 0.1:
            break
    return h_bin, c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except (NameError, UnboundLocalError):
        n0 = 45
        h0 = np.ones(n0) * 0.5

    best_h = fix_sum(h0.copy(), n0)
    best_c5 = compute_c5(best_h, n0)

    def time_left():
        return budget_s - (time.time() - start_time)

    # Phase 1: Coordinate descent at original resolution
    if time_left() > 30:
        h_ref = coord_descend(best_h, n0, max_sweeps=150)
        h_ref = fix_sum(h_ref, n0)
        c5 = compute_c5(h_ref, n0)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_ref

    # Phase 2: Multi-resolution refinement
    resolutions = []
    for mult in [1, 2, 3, 4, 5, 6, 8, 10]:
        resolutions.append(n0 * mult)
    resolutions.extend([60, 90, 120, 150, 200, 250, 300, 400, 500])
    resolutions = sorted(set([r for r in resolutions if 2 * n0 <= r <= 600]))

    curr_n = n0
    curr_h = best_h.copy()
    for n_new in resolutions:
        if time_left() < 60:
            break
        if n_new <= curr_n:
            continue
        x_old = np.linspace(0, 1, curr_n, endpoint=False)
        x_new = np.linspace(0, 1, n_new, endpoint=False)
        h_interp = np.interp(x_new, x_old, curr_h)
        h_interp = fix_sum(h_interp, n_new)
        max_sw = min(80, max(10, int(time_left() / 15)))
        h_opt = coord_descend(h_interp, n_new, max_sweeps=max_sw)
        h_opt = fix_sum(h_opt, n_new)
        c5 = compute_c5(h_opt, n_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            curr_n = n_new
            curr_h = h_opt.copy()

    # Phase 3: Random restarts
    n_attempts = 0
    while time_left() > 40 and n_attempts < 40:
        n_attempts += 1
        n = int(rng.integers(40, 350))
        style = n_attempts % 4
        if style == 0:
            h = (rng.random(n) < 0.5).astype(float)
        elif style == 1:
            x = np.linspace(0, 2, n, endpoint=False)
            k1 = rng.integers(1, 4)
            k2 = rng.integers(4, 8)
            p1 = rng.random() * 2 * np.pi
            p2 = rng.random() * 2 * np.pi
            h = np.clip(0.5 + 0.2 * np.sin(2*np.pi*k1*x + p1) + 0.15*np.sin(2*np.pi*k2*x + p2), 0, 1)
        elif style == 2:
            # Interpolate best to n
            x_old = np.linspace(0, 1, len(best_h), endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            h = np.interp(x_new, x_old, best_h) + 0.15 * rng.standard_normal(n)
            h = np.clip(h, 0, 1)
        else:
            # Block pattern
            blk = int(rng.integers(2, 10))
            pattern = (rng.random(blk) < 0.5).astype(float)
            h = np.tile(pattern, n // blk + 1)[:n]
        
        h = fix_sum(h, n)
        max_sw = min(40, max(5, int(time_left() / 30)))
        h_opt = coord_descend(h, n, max_sweeps=max_sw)
        h_opt = fix_sum(h_opt, n)
        c5 = compute_c5(h_opt, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    # Phase 4: Binary refinement on best solution
    if time_left() > 20:
        nb = len(best_h)
        if nb % 2 == 0:
            res = try_binary_greedy(best_h, nb, rng, max_iter=1000)
            if res is not None:
                h_bin, c5_bin = res
                if c5_bin < best_c5:
                    best_c5 = c5_bin
                    best_h = h_bin
        # Also try even resolutions that are multiples for binary
        for mult in [2, 3, 4, 5]:
            if time_left() < 10:
                break
            nb2 = len(best_h) * mult
            if nb2 > 600 or nb2 % 2 != 0:
                continue
            x_old = np.linspace(0, 1, len(best_h), endpoint=False)
            x_new = np.linspace(0, 1, nb2, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = fix_sum(h_interp, nb2)
            res = try_binary_greedy(h_interp, nb2, rng, max_iter=500)
            if res is not None:
                h_bin, c5_bin = res
                if c5_bin < best_c5:
                    best_c5 = c5_bin
                    best_h = h_bin

    # Final polish: one more coordinate descent
    if time_left() > 15:
        n_f = len(best_h)
        h_final = coord_descend(best_h, n_f, max_sweeps=50)
        h_final = fix_sum(h_final, n_f)
        c5_f = compute_c5(h_final, n_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_final

    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h, len(best_h))
    return best_h, best_c5, len(best_h)
