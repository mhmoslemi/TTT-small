import numpy as np

def fft_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def normalize(a, S):
    """Clamp to [0, 1000] and scale to sum = S."""
    a = np.clip(np.asarray(a, float), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-12:
        return np.full(len(a), S / len(a))
    return a * (S / s)


def lse_grad(a, c, beta, n):
    """
    Compute subgradient of LSE(beta*c) w.r.t. a using FFT cross-correlation.
    Returns gradient vector of length n.
    """
    bc = beta * c
    bc -= bc.max()
    w = np.exp(bc)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w = w / ws
    
    # grad[m] = 2 * sum_j w[m+j] * a[j]  (cross-correlation)
    fsize = 1 << (3 * n - 2).bit_length()
    W = fft(w, fsize)
    A = fft(a, fsize)
    corr = np.real(ifft(W * np.conj(A)))[:n]
    return 2.0 * corr


def topk_grad(a, c, n, top_k):
    """Compute subgradient using only top-k lags (uniform weights among them)."""
    idx = np.argsort(c)[::-1][:top_k]
    w = np.zeros(2 * n - 1)
    w[idx] = 1.0 / top_k

    A = fft(a, fsize)

def peak_shave(a, c, n, S, num_pairs=100):
    """Directly reduce the peak by balancing pairs of elements."""
    a = a.copy()
    # Find the peak lag
    k_peak = np.argmax(c)
    # Elements contributing to c[k_peak]: a[i] * a[k_peak - i] for valid i
    i_range = range(max(0, k_peak - n + 1), min(n, k_peak + 1))
    
    # For each pair, move mass from larger to smaller
    for _ in range(num_pairs):
        i = rng.choice(i_range)  # need rng
        j = k_peak - i
        if 0 <= i < n and 0 <= j < n and i != j:
            if a[i] > a[j]:
                delta = min(0.01 * (a[i] - a[j]), a[i] * 0.01)
                a[i] -= delta
                a[j] += delta
            else:
                delta = min(0.01 * (a[j] - a[i]), a[j] * 0.01)
                a[j] -= delta
                a[i] += delta
    
    a = normalize(a, S)

    """
    a = normalize(a_init, S)
    c = fft_conv_self(a)
    best_a = a.copy()
    best_val = c.max()

    start = time.monotonic()
    t = 0
    step_count = int(1.0 / max(0.001, (max_time - 0.1) / max(1.0, max_time)))

    while time.monotonic() - start < max_time - 0.05:
        t += 1
        # Anneal beta
        frac = min(1.0, t / max(1, step_count * 0.8))
        beta = beta_start + (beta_end - beta_start) * frac

        # Compute gradient
        if use_lse:
            grad = compute_lse_gradient(a, c, beta, n)
        else:
            grad = compute_topk_gradient(a, c, n, top_k)

        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad = grad / gn

        # Adaptive learning rate
        eta = eta_init / (1.0 + 0.003 * t)

        # Multiplicative update
        a_new = a * np.exp(-eta * grad)
        a_new = np.clip(a_new, 0.0, 1000.0)
        s = a_new.sum()
        if s < 1e-12:
            a = np.full(n, S / n, dtype=np.float64)
        else:
            a = a_new * (S / s)

        # Periodic peak shaving
        if shave_every > 0 and t % shave_every == 0:
            c = fft_conv_self(a)
            k_peak = int(np.argmax(c))
            # Balance pairs contributing to the peak
            i_lo = max(0, k_peak - n + 1)
            i_hi = min(n - 1, k_peak)
            for _ in range(50):
                i = i_lo + rng.integers(0, i_hi - i_lo + 1)
                j = k_peak - i
                if 0 <= i < n and 0 <= j < n and i != j:
                    diff = a[i] - a[j]
                    if abs(diff) > 1e-10:
                        delta = 0.005 * abs(diff)
                        if diff > 0:
                            a[i] -= delta
                            a[j] += delta
                        else:
                            a[j] -= delta
                            a[i] += delta
            a = normalize(a, S)

        # Evaluate every few steps
        if t % 3 == 0 or t < 5:
            c = fft_conv_self(a)
            cur = c.max()
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    # Final check
    c = fft_conv_self(a)
    if c.max() < best_val:
        best_val = c.max()
        best_a = a.copy()

    return best_a, best_val


def run_adam_logspace(a_init, n, S, max_time, rng):
    """Adam optimizer in log-space for smooth multiplicative updates."""
    a0 = normalize(a_init, S)
    x = np.log(np.maximum(a0, 1e-10))

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    c = fft_conv_self(best_a)
    best_val = c.max()
    start = time.monotonic()

    while time.monotonic() - start < max_time - 0.05:
        t += 1
        a = np.exp(x)
        a = a / a.sum() * S

        beta_lse = min(80.0, 1.0 + 0.003 * t)
        c = fft_conv_self(a)

        grad = compute_lse_gradient(a, c, beta_lse, n)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad = grad / gn

        # Natural gradient in log-space
        grad_x = a * grad

        m = b1 * m + (1.0 - b1) * grad_x
        v = b2 * v + (1.0 - b2) * (grad_x ** 2)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)

        lr = 0.3 / (1.0 + 0.002 * t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x = np.clip(x, -25.0, 7.0)

        if t % 8 == 0:
            a = np.exp(x)
            a = a / a.sum() * S
            c = fft_conv_self(a)
            cur = c.max()
            if cur < best_val:
                best_val = cur
                best_a = a.copy()

    return best_a, best_val


def make_initializations(rng):
    """Create diverse initial sequences."""
    inits = []
    n = 8192

    # From the given sequence
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        given = np.clip(given, 0.0, 1000.0)
        if given.size >= 16 and given.sum() > 0:
            inits.append(("given", given.copy()))
    except Exception:
        pass

    # Random
    inits.append(("rand", rng.random(n) + 0.05))

    # Cosine taper (peaks in middle)
    tt = np.linspace(0.0, 1.0, n)
    inits.append(("cos2", 0.3 + 0.7 * np.cos(np.pi * tt) ** 2))

    # Sine modulated
    inits.append(("sine3", np.abs(1.0 + 0.4 * np.sin(2 * np.pi * 3.0 * tt))))

    # Two Gaussians
    c1, c2 = n // 4, 3 * n // 4
    dd = np.arange(n, dtype=np.float64)
    g1 = np.exp(-(dd - c1) ** 2 / (2.0 * (n // 8) ** 2))
    g2 = np.exp(-(dd - c2) ** 2 / (2.0 * (n // 8) ** 2))
    inits.append(("bimodal", 0.1 + g1 + g2))

    # Wide Gaussian
    gc = np.exp(-(dd - n // 2) ** 2 / (2.0 * (n // 4) ** 2))
    inits.append(("wide_gauss", 0.05 + 0.95 * gc))

    # Four-peaks pattern
    peaks = np.zeros(n)
    for p in range(4):
        center = (2 * p + 1) * n // 8
        peaks += np.exp(-(dd - center) ** 2 / (2.0 * (n // 16) ** 2))
    inits.append(("4peak", 0.1 + peaks))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses LSE-smoothed mirror descent with annealed temperature,
    multiple initializations, and final polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, budget_s - 10.0)

    inits = make_initializations(rng)
    n_inits = len(inits)

    # Time allocation: 70% exploration, 30% polishing
    total_exploration = 0.70 * (budget_s - 15.0)
    per_init_time = total_exploration / max(1, n_inits)

    global_best = np.inf
    global_a = None

    print(f"Starting search: {n_inits} inits, {per_init_time:.1f}s each, total budget {budget_s}s")

    for idx, (name, arr) in enumerate(inits):
        remaining = deadline - 20.0 - time.monotonic()
        if remaining < 5.0:
            break

        n = len(arr)
        if n < 16:
            continue
        S = np.sqrt(2.0 * float(n))

        t0 = time.monotonic()
        seg = min(per_init_time, remaining)

        # Phase 1: Mirror descent with LSE (coarse-to-fine)
        a1, val1 = run_mirror_lse(arr, n, S, seg * 0.55, rng,
                                   eta_init=0.12, beta_start=1.0, beta_end=30.0,
                                   use_lse=True, shave_every=300)
        t_el = time.monotonic() - t0
        print(f"[{idx}] {name:12s} mirror_lse: {val1:.6f} ({t_el:.1f}s)")
        if val1 < global_best:
            global_best = val1
            global_a = a1.copy()

        # Phase 2: Continue with sharper settings
        remaining2 = deadline - 25.0 - time.monotonic()
        if remaining2 > 3.0:
            seg2 = min(seg * 0.45, remaining2)
            a2, val2 = run_mirror_lse(a1, n, S, seg2, rng,
                                       eta_init=0.06, beta_start=10.0, beta_end=80.0,
                                       use_lse=True, shave_every=200)
            print(f"[{idx}] {name:12s} refine:     {val2:.6f}")
            if val2 < global_best:
                global_best = val2
                global_a = a2.copy()

        # Phase 3: Log-space Adam as alternative
        remaining3 = deadline - 30.0 - time.monotonic()
        if remaining3 > 5.0:
            seg3 = min(seg * 0.3, remaining3)
            a3, val3 = run_adam_logspace(a2 if val2 < val1 else a1, n, S, seg3, rng)
            print(f"[{idx}] {name:12s} adam_log:   {val3:.6f}")
            if val3 < global_best:
                global_best = val3
                global_a = a3.copy()

        # Verify with authoritative function
        test_a = np.clip(a1, 0, 1000)
        ver = evaluate_sequence(test_a.tolist())
        if np.isfinite(ver):
            print(f"[{idx}] {name:12s} verify:     {ver:.6f}")
            if ver < global_best:
                global_best = ver
                global_a = test_a

    # === POLISH PHASE ===
    if global_a is None:
        global_a = np.ones(8192, dtype=np.float64)

    n = len(global_a)
    S = np.sqrt(2.0 * float(n))
    polish_time = max(1.0, deadline - time.monotonic() - 5.0)
    print(f"\nPolishing for {polish_time:.1f}s, current best: {global_best:.6f}")

    # Fine-tune with very sharp LSE
    a_p1, val_p1 = run_mirror_lse(global_a, n, S, polish_time * 0.4, rng,
                                   eta_init=0.03, beta_start=20.0, beta_end=150.0,
                                   use_lse=True, shave_every=100)
    print(f"polish1: {val_p1:.6f}")
    if val_p1 < global_best:
        global_best = val_p1
        global_a = a_p1

    # Top-k refinement on the sharpest peaks
    remaining_p = max(1.0, deadline - time.monotonic() - 3.0)
    if remaining_p > 2.0:
        a_p2, val_p2 = run_mirror_lse(global_a, n, S, remaining_p, rng,
                                       eta_init=0.02, beta_start=50.0, beta_end=300.0,
                                       use_lse=False, top_k=15, shave_every=50)
        print(f"polish2: {val_p2:.6f}")
        if val_p2 < global_best:
            global_best = val_p2
            global_a = a_p2

    # Final verification
    final_a = np.clip(global_a, 0.0, 1000.0)
    final_ver = evaluate_sequence(final_a.tolist())
    print(f"\nFINAL ANSWER: {final_ver:.6f}")

    if np.isfinite(final_ver):
        global_a = final_a

    if global_a is None or global_a.sum() < 0.01:
        global_a = np.ones(256, dtype=np.float64)

    return [float(x) for x in global_a]
