"""
LP-based best-response search for low max autocorrelation step functions.
Uses iterative LP solving to minimize the peak of the autocorrelation.
Tries multiple sequence lengths and diverse starting points.
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def conv_full(p):
    """Full self-convolution of 1D vector."""
    n = len(p)
    L = next_fast_len(2 * n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1]

def score_normalized(p):
    """Score assuming sum(p) = 1, p >= 0."""
    n = len(p)
    c = conv_full(p)
    return 2.0 * n * np.max(c)

def normalize(p):
    p = np.maximum(p, 0.0)
    s = np.sum(p)
    if s < 1e-15:
        return np.ones_like(p) / len(p)
    return p / s

def make_uniform(n):
    return np.ones(n) / n

def make_cosine(n, alpha=1.0):
    x = np.linspace(0, 1, n)
    p = 1.0 + alpha * np.cos(2 * np.pi * x)
    return normalize(p)

def make_geometric(n, rate):
    p = np.exp(-rate * np.linspace(0, 1, n))
    return normalize(p)

def make_power(n, a):
    x = np.linspace(1e-3, 1.0, n)
    p = x ** a
    return normalize(p)

def make_sinusoidal(n, freq, phase=0):
    x = np.linspace(0, 1, n)
    p = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x + phase)
    return normalize(np.maximum(p, 0.01))

def make_two_peak(n, mu1, mu2, w1, w2, sigma=0.05):
    x = np.linspace(0, 1, n)
    p = w1 * np.exp(-0.5 * ((x - mu1) / sigma) ** 2) + w2 * np.exp(-0.5 * ((x - mu2) / sigma) ** 2)
    return normalize(p)

def random_smooth(n, rng):
    x = np.linspace(0, 1, n)
    coeffs = {}
    p = np.ones(n)
    for freq in [1, 2, 3, 5, 8]:
        coeff = rng.normal(0, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        p += coeff * np.sin(2 * np.pi * freq * x + phase)
    return normalize(np.maximum(p, 0.01))

def lp_best_response(p, M=None, tolerance=1e-10):
    """
    Solve: max sum(g) s.t. (p*g)[k] <= M for all k, g >= 0.
    Returns normalized g and the optimal sum.
    """
    n = len(p)
    if M is None:
        c = conv_full(p)
        M = np.max(c)
    
    if M <= 0:
        return p.copy(), 1.0
    
    # Build constraint matrix A where A[m, k] = p[m-k] if 0 <= m-k < n
    # Constraint m: sum_k A[m,k] * g[k] <= M
    rows = []
    cols = []
    data = []
    for m in range(2 * n - 1):
        # g[k] appears in constraint m when 0 <= m-k < n, i.e., max(0, m-n+1) <= k <= min(n-1, m)
        k_start = max(0, m - n + 1)
        k_end = min(n - 1, m)
        for k in range(k_start, k_end + 1):
            rows.append(m)
            cols.append(k)
            data.append(p[m - k])
    
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs',
                  options={'disp': False, 'dual_feasibility_tolerance': tolerance})
    
    if res is None or not res.success:
        return p.copy(), np.max(p if n > 0 else [1.0])
    
    g = res.x
    g = np.maximum(g, 0)
    s = np.sum(g)
    if s < 1e-15:
        return p.copy(), 1.0
    
    return g / s, s
