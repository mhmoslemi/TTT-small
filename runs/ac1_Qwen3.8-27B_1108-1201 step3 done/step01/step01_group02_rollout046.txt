"""
Hybrid multiplicative-update and LP-based optimization for minimizing
peak auto-convolution of non-negative sequences. Combines fast mirror-descent
type multiplicative updates with periodic LP refinement and local search polishing.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, csc_matrix

import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def score_sequence(a):
    n = len(a)
    if n <= 3000:
        conv = np.convolve(a, a, mode='full')
    else:
        conv = fftconvolve(a, a, mode='full')
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def compute_conv(a):
    n = len(a)
    if n <= 3000:
        return np.convolve(a, a, mode='full')
    else:
        return fftconvolve(a, a, mode='full')


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_flat_taper(n, taper_frac=0.3):
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    taper = taper_frac
    a = np.ones(n)
    mask_left = x < -taper
    mask_right = x > taper
    a[mask_left] = 0.5 * (1.0 + np.cos(np.pi * x[mask_left] / taper))
    a[mask_right] = 0.5 * (1.0 + np.cos(np.pi * x[mask_right] / taper))
    return a


def initial_triangle(n):
    t = np.arange(n, dtype=np.float64)

    return np.maximum(0.0, 1.0 - np.abs(t - center) / (center + 0.5))

def compute_subgradient(a, conv, K=20):
    """Compute approximate subgradient of max(conv) w.r.t. a."""
    n = len(a)
    top_idx = np.argsort(conv)[-K:]
    grad = np.zeros(n)

        hi = min(n - 1, k)
        if hi >= lo:
            indices = np.arange(lo, hi + 1)
            grad[indices] += a[k - indices]
    return grad

def build_conv_matrix(csr_a):
    """Build the convolution matrix A(a) where A(a) @ g = conv(a, g)."""
    n = csr_a.shape[1]
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(csr_a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, n_constraints=None, timeout=30.0):
    """Solve the LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if n_constraints is None:
        n_constraints = 2 * n - 1
    A = build_conv_matrix(a)

    # Optionally reduce constraints to top ones
    if n_constraints < 2 * n - 1:
        # Keep only the most binding constraints (all are binding at M)
        # But for efficiency, sub-sample
        pass  # For now use all constraints

    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout,
                               'dual_feasibility_tolerance': 1e-7,
                               'primal_feasibility_tolerance': 1e-7})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
        elif res.status in (2, 3):  # time limit or iter limit
            if res.x is not None:
                return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=300.0, verbose=True, max_iters=200):
    """LP-based iterative refinement."""
    start = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    LP start: score={best_score:.6f}, n={n}")
    stagnant = 0

    for it in range(max_iters):
        elapsed = time.monotonic() - start
        if elapsed > time_limit_s:
            break
        remaining = time_limit_s - elapsed
        if remaining < 5.0:
            break

        conv = compute_conv(a)
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 10 == 0 or it < 5):
                print(f"    LP iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1

        if stagnant > 30:
            if verbose:
                print(f"    LP stagnant at iter {it}, best={best_score:.6f}")
            break

        timeout = max(2.0, min(45.0, remaining * 0.25))
        g = solve_lp_step(a, M, timeout=timeout)
        if g is None:
            break
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-8:
            if verbose:
                print(f"    LP no improvement (sum_g={sum_g:.10f}) at iter {it}")
            stagnant += 5
            if stagnant > 50:
                break
            # Slightly perturb to escape
            M = M * 1.001
            continue
        g = g / sum_g

        # Line search for step size
        best_alpha = 0.1
        best_new_score = score
        for alpha in [0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
            a_try = (1.0 - alpha) * a + alpha * g
            a_try = normalize_nonneg(a_try)
            s_try = 2.0 * n * compute_conv(a_try).max()
            if s_try < best_new_score:
                best_new_score = s_try
                best_alpha = alpha

        a = (1.0 - best_alpha) * a + best_alpha * g
        a = normalize_nonneg(a)

    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    LP done: best={best_score:.6f} after {it+1} iters, {time.monotonic()-start:.1f}s")
    return best_a, best_score


def local_polish(a, time_limit_s=60.0, rng=None, verbose=True):
    """Local search polishing with random perturbations."""
    if rng is None:
        rng = np.random.default_rng(123)
    start = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    PS start: score={best_score:.6f}, n={n}")
    accept_count = 0

    while time.monotonic() - start < time_limit_s:
        # Random smooth perturbation
        width = max(3, int(rng.integers(n // 200, n // 6)))
        pos = int(rng.integers(0, max(1, n - width)))
        candidate = a.copy()

        # Apply a smooth bump or dip
        t = np.arange(width, dtype=np.float64)
        center_w = width / 2.0
        sigma_w = max(1.0, width / 4.0)
        smooth = np.exp(-0.5 * ((t - center_w) / sigma_w) ** 2)
        magnitude = rng.normal(0, 0.05)
        candidate[pos:pos + width] *= (1.0 + magnitude * smooth)

        # Occasionally try symmetry
        if rng.random() < 0.3:
            candidate = 0.5 * (candidate + candidate[::-1])

        candidate = np.maximum(0, candidate)
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)

        if score < best_score - 1e-12:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 30 == 0:
                print(f"    PS accept {accept_count}: score={best_score:.6f}")

    if verbose:
        print(f"    PS done: best={best_score:.6f}, {time.monotonic()-start:.1f}s")
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses hybrid multiplicative updates + LP refinement with multi-start.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(10.0, budget_s - 25.0)

    best_score = float('inf')
    best_seq = None

    # Try parent sequence
    parent = None
    try:
        parent_list = height_sequence_1
        parent = np.array(parent_list, dtype=np.float64)
        parent = normalize_nonneg(parent)
        parent_score = score_sequence(parent)
        print(f"Parent score: {parent_score:.6f}, n={len(parent)}")
        if parent_score < best_score:
            best_score = parent_score
            best_seq = parent.copy()
    except Exception as e:
        print(f"No parent available: {e}")

    # Define search configurations
    configs = []

    # Parent-based (most important if available)
    if parent is not None:
        configs.append((len(parent), 'parent', 0.20))

    # Various lengths and initializations
    configs.extend([
        (500, 'raised_cosine', 0.08),
        (1000, 'raised_cosine', 0.10),
        (1500, 'raised_cosine', 0.08),
        (2000, 'raised_cosine', 0.10),
        (3000, 'raised_cosine', 0.08),
        (1000, 'flat_taper', 0.10),
        (2000, 'flat_taper', 0.08),
        (3000, 'flat_taper', 0.08),
        (1000, 'gaussian', 0.06),
        (2000, 'gaussian', 0.06),
        (500, 'triangle', 0.05),
        (1000, 'triangle', 0.05),
    ])

    for n, init_type, time_frac in configs:
        remaining = deadline - time.monotonic()
        if remaining < 25.0:
            break
        time_limit = min(time_frac * budget_s, remaining * 0.85)
        if time_limit < 15.0:
            continue

        print(f"\n=== n={n}, init={init_type}, time_limit={time_limit:.0f}s ===")

        # Initialize
        if init_type == 'parent' and parent is not None:
            a = parent.copy()
        elif init_type == 'raised_cosine':
            a = initial_raised_cosine(n)
        elif init_type == 'gaussian':
            a = initial_gaussian(n, sigma_frac=rng.uniform(0.2, 0.4))
        elif init_type == 'flat_taper':
            a = initial_flat_taper(n, taper_frac=rng.uniform(0.2, 0.5))
        elif init_type == 'triangle':
            a = initial_triangle(n)
        elif init_type == 'hann':
            a = initial_hann(n)
        else:
            a = initial_raised_cosine(n)

        a = normalize_nonneg(a)
        init_score = score_sequence(a)
        print(f"    Init score: {init_score:.6f}")

        # Phase 1: Fast multiplicative updates (40% of time)
        mu_time = time_limit * 0.40
        a, score = multiplicative_refine(a, time_limit_s=mu_time, verbose=True, rng=rng)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  *** AFTER MU: {score:.6f}, n={n}")

        # Phase 2: LP refinement (40% of time)
        remaining = deadline - time.monotonic()
        lp_time = min(time_limit * 0.40, remaining * 0.6)
        if lp_time > 10.0:
            a, score = lp_refine(a, time_limit_s=lp_time, verbose=True)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"  *** AFTER LP: {score:.6f}, n={n}")

        # Phase 3: Final multiplicative polish (20% of time)
        remaining = deadline - time.monotonic()
        mu2_time = min(time_limit * 0.20, remaining * 0.4)
        if mu2_time > 5.0:
            a, score = multiplicative_refine(a, time_limit_s=mu2_time, verbose=False, rng=rng)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"  *** AFTER MU2: {score:.6f}, n={n}")

    # Final polishing on best sequence
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 30.0:
            print(f"\n=== Final polishing (remaining={remaining:.0f}s) ===")
            # Local search
            ps_time = min(remaining * 0.3, 120.0)
            polished, score = local_polish(best_seq, time_limit_s=ps_time, rng=rng, verbose=True)
            if score < best_score:
                best_score = score
                best_seq = polished.copy()
                print(f"  *** After PS: {score:.6f}")

            # LP polish
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20.0:
                lp_time = min(remaining2 * 0.5, 180.0)
                polished2, score2 = lp_refine(best_seq, time_limit_s=lp_time, verbose=True)
                if score2 < best_score:
                    best_score = score2
                    best_seq = polished2.copy()
                    print(f"  *** After final LP: {score2:.6f}")

            # One more multiplicative pass
            remaining3 = deadline - time.monotonic()
            if remaining3 > 10.0:
                mu3_time = min(remaining3 * 0.5, 60.0)
                polished3, score3 = multiplicative_refine(best_seq, time_limit_s=mu3_time,
                                                          verbose=False, rng=rng)
                if score3 < best_score:
                    best_score = score3
                    best_seq = polished3.copy()
                    print(f"  *** After final MU: {score3:.6f}")

    if best_seq is None:
        best_seq = initial_raised_cosine(500)
        best_score = score_sequence(best_seq)
        print(f"  Fallback: score={best_score:.6f}")

    # Verify
    print(f"\nFinal result: score={best_score:.6f}, n={len(best_seq)}")
    verify_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified score via evaluate_sequence: {verify_score:.6f}")
    print(f"Total time: {time.monotonic()-total_start:.1f}s")

    return [float(x) for x in best_seq]
