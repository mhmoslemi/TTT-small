def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n+1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0/n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / (k + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s <= 1e-300:
        return np.full(n, 1.0/n)
    return out / s

    n = int(len(p))
    if n == 0:
        return np.empty(0)
    L = _fl(n)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2*n-1].astype(np.float64, copy=False)


def score_seq(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if s < 1e-300:
        return np.inf
    c = conv_full(p)
    mb = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(mb):
        return np.inf
    return float(2.0 * n * mb / (s * s))


def smooth_grad(p, T):
    """Gradient and surrogate value of smoothed max of conv(p,p)."""
    n = int(p.size)
    if n == 0:
        return np.empty(0), 0.0
    L = _fl(n)
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2*n-1]
    cmax = float(np.max(c))
    if cmax <= 1e-300 or not np.isfinite(cmax):
        return np.zeros(n), 0.0
    T = max(float(T), 1e-300)
    diff = (c - cmax) / T
    mx = float(np.max(diff))
    if not np.isfinite(mx):
        mx = 0.0
    diff = diff - mx
    e = np.exp(diff)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = e / s
    surr = cmax + mx * T + T * np.log(s)
    # grad[j] = 2 * sum_k w[k] * p[k-j] for valid k-j in [0,n-1]
    # = 2 * (p * w_flip)[n-1+j]
    w_flip = w[::-1].copy()
    Lc = _fl(n)
    Fp2 = fft(p, Lc)
    Fw2 = fft(w_flip, Lc)
    cpw = np.real(ifft(Fp2 * Fw2))[:2*n-1]
    gh = cpw[n-1:]
    if gh.size < n:
        tmp = np.zeros(n)
        tmp[:gh.size] = gh
        gh = tmp
    grad = 2.0 * gh
    grad = np.maximum(grad, 0.0)
    return grad, float(surr)


def proj_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (css - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    k = int(np.count_nonzero(cond) - 1)
    theta = (css[k] - 1.0) / (k + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s < 1e-300:
        return np.full(n, 1.0 / n)
    return out / s


def norm_p(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-300:
        return np.full(p.size, 1.0 / max(1, p.size))
    return p / s


def lp_bresp(p, budget_s=30.0):
    """LP: max sum(g) s.t. p*g <= M, g >= 0. Return (best_q, best_score)."""
    n = int(p.size)
    if n > LP_NMAX or n < 2:
        return p, score_seq(p)
    if budget_s < 1.0:
        return p, score_seq(p)
    c = conv_full(p)
    M = float(np.max(c))
    if M < 1e-300:
        return p, score_seq(p)
    pn = p + 1e-14
    pn = pn / pn.sum()
    t0 = time.monotonic()
    try:
        rows_list = []
        cols_list = []
        data_list = []
        for k in range(2 * n - 1):
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            for j in range(j_min, j_max + 1):
                idx = k - j
                rows_list.append(k)
                cols_list.append(j)
                data_list.append(pn[idx])
        A = csr_matrix((data_list, (rows_list, cols_list)), shape=(2*n-1, n))
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M),
                      bounds=(0.0, None), method='highs',
                      options={'disp': False, 'time_limit': max(1.0, budget_s * 0.7)})
        if res is not None and res.status == 0:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-12:
                g = g / gs
                cur_best = p.copy()
                cur_sc = score_seq(p)
                for t in (1.0, 0.9, 0.75, 0.6, 0.45, 0.3, 0.15, 0.05):
                    q = proj_simplex((1 - t) * p + t * g)
                    sc = score_seq(q)
                    if np.isfinite(sc) and sc < cur_sc:
                        cur_sc = sc
                        cur_best = q
                return cur_best, cur_sc
    except Exception:
        pass
    return p, score_seq(p)


def make_shapes(n, rng):
    """Generate diverse initial shapes of length n."""
    out = []
    x = np.linspace(0.0, 1.0, n)
    xm = 2.0 * x - 1.0

    # Uniform
    out.append(np.full(n, 1.0 / n))

    # Geometric
    for s in (0.05, 0.1, 0.2, 0.4, 0.7, 1.0, 1.5, 2.5, 4.0, 8.0):
        p = np.exp(-s * x)
        out.append(norm_p(p))
        out.append(norm_p(p[::-1]))

    # Power
    for a in (-0.5, -0.2, 0.0, 0.3, 0.7, 1.5, 3.0, 6.0):
        p = np.power(x + 1e-8, a)
        out.append(norm_p(np.maximum(p, 0)))
        p2 = np.power(1.0 - x + 1e-8, a)
        out.append(norm_p(np.maximum(p2, 0)))

    # Quadratic (cap/cup)
    for a in (0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        out.append(norm_p(np.maximum(1.0 - a * xm * xm, 0.01)))
        out.append(norm_p(1.0 + a * xm * xm))

    # Sine-based
    for f in (1, 2, 3, 5, 8):
        for ph in (0.0, 0.5, 1.0):
            out.append(norm_p(np.maximum(1.0 + 0.6 * np.sin(np.pi * f * x + ph), 0.01)))
            out.append(norm_p(np.maximum(1.0 + 0.3 * np.cos(2 * np.pi * f * x + ph), 0.01)))

    # Gaussian-like
    for mu in (0.2, 0.3, 0.5, 0.7, 0.8):
        for sig in (0.05, 0.1, 0.15, 0.25):
            out.append(norm_p(np.exp(-((x - mu) ** 2) / (2 * sig * sig))))

    # Two-peaked
    for mu1, mu2, w1, w2 in ((0.25, 0.75, 1, 1), (0.2, 0.8, 2, 1), (0.3, 0.7, 1, 2),
                              (0.15, 0.5, 1, 1), (0.5, 0.85, 1, 1)):
        p = w1 * np.exp(-((x - mu1) ** 2) / 0.01) + w2 * np.exp(-((x - mu2) ** 2) / 0.01)
        out.append(norm_p(p))

    # Random smooth
    for _ in range(20):
        f1 = int(rng.integers(1, max(2, n // 100)))
        f2 = int(rng.integers(1, max(2, n // 50)))
        f3 = int(rng.integers(1, max(2, n // 20)))
        p1 = float(rng.uniform(0, 2 * np.pi))
        p2 = float(rng.uniform(0, 2 * np.pi))
        p3 = float(rng.uniform(0, 2 * np.pi))
        a1 = float(rng.uniform(0.1, 0.8))
        a2 = float(rng.uniform(0.05, 0.5))
        a3 = float(rng.uniform(0.02, 0.3))
        p = 1.0 + a1 * np.sin(2 * np.pi * f1 * x + p1) + a2 * np.sin(2 * np.pi * f2 * x + p2) + a3 * np.sin(2 * np.pi * f3 * x + p3)
        out.append(norm_p(np.maximum(p, 0.01)))

    return out


def optimize_core(p0, time_limit, rng, allow_lp=True, verbose=False):
    """Main local optimizer: adaptive-T smoothed gradient + LP + restarts."""
    p = proj_simplex(np.asarray(p0, dtype=np.float64).copy())
    n = int(p.size)
    best_p = p.copy()
    best_score = score_seq(p)
    cur_score = best_score
    end = time.monotonic() + max(1.0, float(time_limit))

    T_fac = 0.5
    prev_p = p.copy()
    since_imp = 0
    it = 0
    momentum = 0.0

    if verbose:
        print(f"    start: score={best_score:.8f} n={n}", flush=True)

    # Initial LP
    if allow_lp and n <= LP_NMAX:
        t0 = time.monotonic()
        p_lp, sc_lp = lp_bresp(p, budget_s=min(35.0, (end - t0) * 0.15))
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_p = p_lp.copy()
            p = p_lp.copy()
            prev_p = p_lp.copy()
            cur_score = sc_lp
            if verbose:
                print(f"    after LP: score={best_score:.8f}", flush=True)

    while time.monotonic() < end:
        it += 1
        remaining = end - time.monotonic()
        if remaining < 1.0:
            break

        # Adaptive temperature: high early, low later
        frac = remaining / max(1.0, time_limit)
        T_fac = 0.02 + 0.6 * (frac ** 2)
        if since_imp > 200:
            T_fac = max(0.01, T_fac * 0.5)

        c_cur = conv_full(p)
        cmax = float(np.max(c_cur))
        T = T_fac * cmax

        g, surr = smooth_grad(p, T)

        accepted = False
        if np.all(np.isfinite(g)) and np.max(g) > 1e-300:
            gscale = float(np.max(g))
            base = (1.0 / n) / max(gscale, 1e-300)
            lrs = [0.4 * base, 0.15 * base, 0.05 * base, 0.015 * base, 0.005 * base, 0.001 * base]
            lrs = [min(max(lr, 1e-12), 5.0) for lr in lrs if np.isfinite(lr) and lr > 0]
            if not lrs:
                lrs = [0.1, 0.03, 0.01]

            if it > 100:
                momentum = 0.3
            else:
                momentum = 0.0

            for lr in lrs:
                v = p - lr * g
                if momentum > 0:
                    v = v + momentum * (p - prev_p)
                pt = proj_simplex(v)
                sc = score_seq(pt)
                if np.isfinite(sc):
                    if sc < surr - 1e-12 * max(1, abs(surr)):
                        accept = True
                    elif sc < best_score - 1e-12:
                        accept = True
                    elif sc < cur_score - 1e-12 and sc < surr + 0.5 * max(1, surr):
                        accept = True
                    else:
                        accept = False
                    if accept:
                        if sc < best_score:
                            best_score = sc
                            best_p = pt.copy()
                            since_imp = 0
                        else:
                            since_imp += 1
                        cur_score = sc
                        prev_p = p.copy()
                        p = pt
                        accepted = True
                        break

        if not accepted:
            since_imp += 1

            # Random perturbation
            if since_imp > 30:
                sig = 0.15 / n
                pt = proj_simplex(p + rng.normal(0, sig, n))
                sc = score_seq(pt)
                if np.isfinite(sc) and sc < cur_score:
                    if sc < best_score:
                        best_score = sc
                        best_p = pt.copy()
                        since_imp = 0
                    cur_score = sc
                    prev_p = p.copy()
                    p = pt
                    accepted = True

            # Pair move: shift mass from high-gradient to low-gradient
            if not accepted and np.all(np.isfinite(g)) and n > 1:
                hi = int(np.argmax(g))
                lo = int(np.argmin(g))
                if hi != lo and g[hi] > g[lo] + 1e-15:
                    for fr in (0.3, 0.1, 0.03, 0.01):
                        d = fr * p[hi]
                        if d <= 0:
                            continue
                        pt = p.copy()
                        pt[hi] -= d
                        pt[lo] += d
                        pt = proj_simplex(pt)
                        sc = score_seq(pt)
                        if np.isfinite(sc) and sc < cur_score:
                            if sc < best_score:
                                best_score = sc
                                best_p = pt.copy()
                                since_imp = 0
                            else:
                                since_imp += 1
                            cur_score = sc
                            prev_p = p.copy()
                            p = pt
                            accepted = True
                            break

            # Restart from best
            if not accepted and since_imp > 250:
                if since_imp > 600:
                    p = proj_simplex(0.5 * best_p + 0.5 * p + rng.normal(0, 0.3 / n, n))
                else:
                    p = proj_simplex(0.8 * best_p + 0.2 * p + rng.normal(0, 0.08 / n, n))
                cur_score = score_seq(p)
                prev_p = p.copy()
                since_imp = 0

        # Periodic LP
        if allow_lp and n <= LP_NMAX and it % 200 == 0 and remaining > 45:
            t0 = time.monotonic()
            p_lp, sc_lp = lp_bresp(p, budget_s=min(35.0, (end - t0) * 0.2))
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                best_p = p_lp.copy()
                p = p_lp.copy()
                prev_p = p_lp.copy()
                cur_score = sc_lp
                since_imp = 0
                if verbose:
                    print(f"    iter={it} LP improved to {best_score:.8f}", flush=True)

        if verbose and it % 500 == 0:
            print(f"    iter={it} best={best_score:.8f} cur={cur_score:.8f} T={T_fac:.4f} since_imp={since_imp}", flush=True)

    return best_p, float(best_score) if np.isfinite(best_score) else np.inf


def shape_param_search(n, rng, budget_s):
    """Optimize a low-dim shape parameterization."""
    if budget_s < 10:
        return None, np.inf
    x = np.linspace(0.0, 1.0, n)
    z = 2.0 * x - 1.0
    DEG = 8

    def build(theta):
        theta = np.asarray(theta, dtype=np.float64)
        need = DEG + 2
        if theta.size < need:
            theta = np.concatenate([theta, np.zeros(need - theta.size)])
        else:
            theta = theta[:need].copy()
        logp = np.zeros(n)
        # Chebyshev
        T0 = np.ones(n)
        T1 = z.copy()
        logp += theta[0] * T1
        Tnm = T0
        Tn = T1
        for d in range(2, DEG + 1):
            Tnp = 2.0 * z * Tn - Tnm
            logp += theta[d - 1] * Tnp
            Tnm = Tn
            Tn = Tnp
        # Endpoint taper
        eps = 1e-4
        logp += theta[DEG] * np.log(x + eps)
        logp += theta[DEG + 1] * np.log(1.0 - x + eps)
        if not np.all(np.isfinite(logp)):
            return None
        logp -= np.max(logp)
        p = np.exp(logp)
        return norm_p(p)

    def obj(theta):
        p = build(theta)
        if p is None:
            return 1e100
        sc = score_seq(p)
        return sc if np.isfinite(sc) else 1e100

    best_p = None
    best_sc = np.inf

    def try_update(theta):
        nonlocal best_p, best_sc
        sc = obj(theta)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_p = build(theta)

    dim = DEG + 2
    end = time.monotonic() + budget_s

    # Starting points
    starts = [np.zeros(dim), rng.uniform(-1, 1, dim), rng.uniform(-1, 1, dim)]
    t = np.zeros(dim)
    t[DEG] = 2.0
    t[DEG + 1] = 2.0
    starts.append(t)
    t = np.zeros(dim)
    t[0] = 1.5
    starts.append(t)
    t = np.zeros(dim)
    for i in range(DEG):
        t[i] = 0.7 * (1 if i % 2 == 0 else -1)
    starts.append(t)

    for s in starts:
        try_update(s)

    if _HAVE_MIN and time.monotonic() < end - 30:
        bounds = [(-3, 3)] * DEG + [(-3, 5), (-3, 5)]
        try:
            res = differential_evolution(obj, bounds, maxiter=30, popsize=12,
                                          tol=0.3, seed=int(rng.integers(0, 10**6)),
                                          polish=False, init='latinhypercube')
            if res is not None and np.isfinite(getattr(res, 'fun', np.inf)):
                try_update(res.x)
        except Exception:
            pass

    for s in starts:
        if time.monotonic() > end - 5:
            break
        if _HAVE_MIN:
            try:
                res = minimize(obj, np.asarray(s, dtype=np.float64), method='Nelder-Mead',
                               options={'maxiter': 200, 'xatol': 1e-5, 'fatol': 1e-7})
                if res is not None and np.isfinite(getattr(res, 'fun', np.inf)):
                    try_update(res.x)
            except Exception:
                pass

    return best_p, float(best_sc) if np.isfinite(best_sc) else np.inf


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for non-negative step function minimizing evaluate_sequence."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)

    # Load parent
    parent = None
    try:
        pa = np.asarray(height_sequence_1, dtype=np.float64)
        pa = np.where(np.isfinite(pa), pa, 0.0)
        pa = np.maximum(pa, 0.0)
        if pa.ndim == 1 and pa.size >= 32 and pa.sum() > 0:
            parent = norm_p(pa)
            print(f"Loaded parent: n={parent.size}, score={score_seq(parent):.8f}", flush=True)
    except Exception:
        pass

    best_p = None
    best_score = np.inf

    def record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = norm_p(np.asarray(p, dtype=np.float64))
        if not np.isfinite(sc):
            sc = score_seq(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print(f"  NEW BEST: {best_score:.8f} (n={p.size})", flush=True)

    if parent is not None:
        record(parent, score_seq(parent))

    # Determine lengths to try
    n_parent = int(parent.size) if parent is not None else 700
    lengths = [n_parent]
    for candidate_n in [512, 700, 800, 1024, 1200, 1400]:
        if candidate_n not in lengths and candidate_n >= 64:
            lengths.append(candidate_n)
    lengths = sorted(set(lengths))

    # Build jobs: (initial_score, p, allow_lp, label)
    jobs = []

    for n in lengths:
        if time.monotonic() > deadline - 120:
            break
        shapes = make_shapes(n, rng)
        scored = []
        for p in shapes:
            p = norm_p(p)
            sc = score_seq(p)
            if np.isfinite(sc):
                scored.append((sc, p))
        scored.sort(key=lambda x: x[0])
        for sc, p in scored[:4]:
            jobs.append((sc, p, n <= LP_NMAX, f"n={n}"))
        # Record top few
        for sc, p in scored[:3]:
            record(p, sc)

    # Also try shape parameter search at a few lengths
    for n in lengths[:3]:
        if time.monotonic() > deadline - 300:
            break
        sb = min(120.0, (deadline - time.monotonic()) * 0.15)
        sp, ss = shape_param_search(n, rng, sb)
        if sp is not None and np.isfinite(ss):
            record(sp, ss)
            jobs.append((ss, sp, n <= LP_NMAX, f"shape_n={n}"))

    # Add parent as a job
    if parent is not None:
        jobs.append((score_seq(parent), parent, n_parent <= LP_NMAX, "PARENT"))

    jobs.sort(key=lambda x: x[0])

    # Optimize top jobs
    final_reserve = max(60.0, min(200.0, (deadline - time.monotonic()) * 0.15))
    n_jobs = min(len(jobs), 5)
    if n_jobs > 0:
        avail = max(30.0, (deadline - final_reserve) - time.monotonic())
        time_per = max(30.0, min(360.0, avail / n_jobs))

        for j in range(n_jobs):
            if time.monotonic() > deadline - final_reserve:
                break
            sc, p, allow_lp, label = jobs[j]
            remaining = max(10.0, deadline - final_reserve - time.monotonic())
            limit = min(time_per, remaining)
            if limit < 15:
                break
            print(f"  Optimizing job {label} (init={sc:.6f}, time={limit:.0f}s)", flush=True)
            t0 = time.monotonic()
            p_best, sc_best = optimize_core(p, limit, rng, allow_lp=allow_lp, verbose=True)
            record(p_best, sc_best)
            elapsed = time.monotonic() - t0
            print(f"  Job {label} done: {sc_best:.8f} ({elapsed:.0f}s)", flush=True)

    # Final polish on global best
    if best_p is not None and time.monotonic() < deadline - 10:
        limit = max(5.0, deadline - time.monotonic() - 5.0)
        print(f"  Final polish (time={limit:.0f}s)", flush=True)
        p_best, sc_best = optimize_core(best_p, limit, rng,
                                        allow_lp=(best_p.size <= LP_NMAX), verbose=True)
        record(p_best, sc_best)

    if best_p is None:
        best_p = np.full(n_parent, 1.0 / n_parent)
        best_score = score_seq(best_p)

    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    best_p = np.minimum(best_p, 1000.0)
    return [float(x) for x in best_p]
