```python
def project_simplex(v, s=1):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    r = np.arange(1, n+1)
    cond = u * r > (cssv - s)
    rho = np.max(r[cond])
    theta = (cssv[rho-1] - s) / rho
    return np.maximum(v - theta, 0)
```

This is the standard algorithm from Duchi et al. (2012).

```python
def smooth_gradient_descent(f, budget_s, T_start=0.1, T_end=100.0, lr=0.01):
    f = f / f.sum()
    n = len(f)
    best = f.copy()
    best_score = fast_score(f)
    
    # Anneal temperature
    n_iter = 10000  # or until time runs out
    
    for it in range(n_iter):
        if time.monotonic() > budget_s:
            break
        
        T = T_start * (T_end / T_start) ** (it / n_iter)
        
        # Compute convolution
        conv = np.convolve(f, f, mode='full')  # length 2n-1
        
        # Compute softmax weights
        log_w = T * conv - T * conv.max()  # for numerical stability
        w = np.exp(log_w)
        w /= w.sum()
        
        # Compute gradient
        grad = 2 * np.correlate(w, f, mode='valid')

        f_new = project_simplex(f_new)
        
        f = f_new
```

```python
"""
Smooth gradient descent on the simplex for minimizing the autocorrelation peak
of a non-negative step function.

Uses the log-sum-exp (soft-max) as a smooth surrogate for the max in the
objective, combined with projected gradient descent on the probability simplex.
The temperature is annealed from low to high to progressively sharpen the
focus on the peak.
"""
import numpy as np
import time

def fast_score(f):
    """Compute score = 2*n*max(conv)/sum^2."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    r = np.arange(1, n + 1)
    cond = u * r > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.max(r[cond])
    theta = (cssv[rho - 1] - s) / rho
    return np.maximum(v - theta, 0.0)

def smooth_grad_descent(f, budget_s, T_start=1.0, T_end=500.0, 
                        lr_init=0.05, lr_decay=0.999):
    """
    Smooth gradient descent on the simplex.
    f: initial point (will be normalized to sum=1)
    budget_s: time budget in seconds
    T_start, T_end: temperature range for softmax
    """
    f = np.asarray(f, dtype=np.float64).copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    
    n = len(f)
    best = f.copy()
    best_sc = fast_score(f)
    start_time = time.monotonic()
    lr = lr_init
    it = 0
    
    while time.monotonic() - start_time < budget_s:
        it += 1
        # Anneal temperature
        frac = min(1.0, it / 5000.0)
        T = T_start * (T_end / T_start) ** frac
        
        # Compute convolution
        conv = np.convolve(f, f)  # length 2n-1
        
        # Softmax weights
        conv_shifted = conv - conv.max()  # numerical stability
        log_w = T * conv_shifted
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-300:
            # Fallback: concentrate on argmax
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w /= w_sum
        
        # Gradient: grad[j] = 2 * sum_k w[k] * f[k-j]
        # This is 2 * (w cross-correlate f) at valid lags
        grad = 2.0 * np.correlate(w, f, mode='valid')
        
        # Scale gradient to have unit norm for stable learning rate
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Gradient step with adaptive learning rate
        f_new = f - lr * grad
        f_new = project_simplex(f_new)
        
        # Simple accept/reject: accept if score doesn't worsen too much
        if it % 10 == 0:
            sc_new = fast_score(f_new)
            if np.isfinite(sc_new):
                if sc_new < best_sc:
                    best_sc = sc_new
                    best = f_new.copy()
                # Adaptive learning rate
                if sc_new < fast_score(f) + 0.01:
                    f = f_new
                    lr = lr * 1.01  # reward
                else:
                    lr = lr * 0.9  # penalty
                    f = f_new  # still accept to explore
            else:
                f = f_new
        else:
            f = f_new
        
        lr = max(lr, 0.001)
        lr = min(lr, 0.5)
    
    return best, best_sc
```

dL/dtheta[j] = sum_i (dL/df[i]) * (df[i]/dtheta[j])
= sum_i (grad_L[i]) * (f[i] * (delta(i,j) - f[j]))
= grad_L[j] * (1 - f[j]) - f[j] * sum_i grad_L[i] * f[i]

where grad_L is the gradient of L with respect to f.

This is the chain rule for the softmax parameterization. It's clean and well-defined.

```python
def smooth_objective_and_grad(theta, T):
```

```python
"""
Minimize autocorrelation peak of non-negative step function using:
1. Smoothed objective (log-sum-exp) with BFGS optimization in softmax parameterization
2. LP-based direction finding with golden-section line search
3. Multiple starting points and restarts

The score = 2*n*max(f*f)/(sum(f))^2 is scale-invariant; we work on the simplex.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

def fast_score(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def project_simplex(v, s=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    r = np.arange(1, n + 1)
    cond = u * r > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.max(r[cond])
    theta = (cssv[rho - 1] - s) / rho
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(n, f):
    """Sparse A (2n-1 x n) where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    A = sp.coo_matrix((data, (row_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()
    return A

def lp_direction(f, M, time_limit=5.0):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None

def golden_section_line_search(f, g, score_fn, a=0.0, b=0.5, tol=1e-4):
    """Find t in [a,b] minimizing score_fn((1-t)*f + t*g)."""
    gr = (np.sqrt(5) + 1) / 2
    c = b - (b - a) / gr
    d = a + (b - a) / gr
    fc = score_fn((1 - c) * f + c * g)
    fd = score_fn((1 - d) * f + d * g)
    while abs(b - a) > tol:
        if fc > fd:
            a = c
            c = d
            d = a + (b - a) / gr
            fc = fd
            fd = score_fn((1 - d) * f + d * g)
        else:
            b = d
            d = c
            c = b - (b - a) / gr
            fd = fc
            fc = score_fn((1 - c) * f + c * g)
    t = (a + b) / 2
    return t

def bfgs_smoothed(f_init, T, time_budget, max_iter=2000):
    """
    BFGS optimization of the smoothed objective in softmax parameterization.
    """
    n = len(f_init)
    f0 = f_init.copy()
    f0 = np.maximum(f0, 1e-10)
    f0 = f0 / f0.sum()
    theta0 = np.log(f0)
    
    def obj_grad(theta):
        f = softmax(theta)
        conv = np.convolve(f, f)
        max_c = conv.max()
        exp_part = np.exp(T * (conv - max_c))
        w = exp_part / exp_part.sum()
        obj = (np.log(exp_part.sum()) + T * max_c) / T
        # Gradient w.r.t. f
        grad_f = 2.0 * np.correlate(w, f, mode='valid')
        # Gradient w.r.t. theta
        grad_theta = f * (grad_f - np.dot(f, grad_f))
        return obj, grad_theta
    
    start_time = time.monotonic()
    result = minimize(obj_grad, theta0, method='BFGS', jac=True,
                      options={'maxiter': max_iter, 'gtol': 1e-8})
    
    f_opt = softmax(result.x)
    return f_opt
```

```python
Minimize autocorrelation peak of non-negative step function using a combination of:
1. BFGS optimization on a smoothed (log-sum-exp) objective in softmax parameterization
2. LP-based direction finding with golden-section line search
3. Stochastic local search with block perturbations

def fast_score(f):
    """Compute score = 2*n*max(f*f)/(sum(f))^2."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def softmax_fn(x):
    """Numerically stable softmax."""
    x = x - np.max(x)
    e = np.exp(x)
    return e / e.sum()


def project_simplex(v, s=1.0):
    """Project v onto the probability simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    r = np.arange(1, n + 1)
    cond = u * r > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.max(r[cond])
    theta = (cssv[rho - 1] - s) / rho
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(n, f):
    """Sparse (2n-1 x n) matrix A where A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    return sp.coo_matrix((data, (row_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()


def lp_direction(f, M, time_limit=5.0):
    """LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0. Returns normalized g or None."""
    n = len(f)

    c = -np.ones(n)
    b = np.full(2 * n - 1, M)

                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def golden_line_search(f, g, n, a=0.0, b=0.6, tol=1e-5):
    """Minimize 2n*max(((1-t)f+tg)*((1-t)f+tg)) over t in [a,b]."""
    gr = (np.sqrt(5.0) + 1.0) / 2.0
    c = b - (b - a) / gr
    d = a + (b - a) / gr

    def score_t(t):
        ft = (1.0 - t) * f + t * g
        ft = np.maximum(ft, 0.0)
        s = ft.sum()
        if s < 1e-10:
            return np.inf
        ft = ft / s
        conv_t = np.convolve(ft, ft)
        return 2.0 * n * conv_t.max()

    fc = score_t(c)
    fd = score_t(d)
    for _ in range(60):
        if abs(b - a) < tol:
            break
        if fc > fd:
            a = c
            c = d
            d = a + (b - a) / gr
            fc = fd
            fd = score_t(d)
        else:
            b = d
            d = c
            c = b - (b - a) / gr
            fd = fc
            fc = score_t(c)
    return (a + b) / 2.0, min(fc, fd)


def bfgs_smoothed(f_init, T, max_iter=1000, tol=1e-8):
    """
    BFGS on the smoothed objective L(f) = (1/T)log(sum exp(T*(f*f)[k]))
    parameterized via f = softmax(theta).
    """
    n = len(f_init)
    f0 = np.maximum(f_init, 1e-10)
    f0 = f0 / f0.sum()
    theta0 = np.log(f0)

    def obj_and_grad(theta):
        f = softmax_fn(theta)
        conv = np.convolve(f, f)
        max_c = conv.max()
        exp_part = np.exp(T * (conv - max_c))
        w = exp_part / exp_part.sum()
        val = (np.log(exp_part.sum()) + T * max_c) / T
        # grad w.r.t. f
        grad_f = 2.0 * np.correlate(w, f, mode='valid')
        # grad w.r.t. theta via chain rule
        grad_theta = f * (grad_f - np.dot(f, grad_f))
        return val, grad_theta

    result = minimize(obj_and_grad, theta0, method='BFGS', jac=True,
                      options={'maxiter': max_iter, 'gtol': tol})
    return softmax_fn(result.x)
```

"""
Minimize autocorrelation peak of non-negative step function using:
1. BFGS on log-sum-exp smoothed objective in softmax parameterization
2. LP-based direction finding with golden-section line search
3. Stochastic local search with block perturbations

Score = 2*n*max(f*f)/(sum(f))^2, scale-invariant, optimized on the simplex.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
import scipy.sparse as sp


def fast_score(f):
    """Compute score = 2*n*max(f*f)/(sum(f))^2."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def softmax_fn(x):
    """Numerically stable softmax."""
    x = x - np.max(x)
    e = np.exp(x)
    return e / e.sum()


def project_simplex(v, s=1.0):
    """Project v onto the simplex {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    r = np.arange(1, n + 1)
    cond = u * r > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.max(r[cond])
    theta = (cssv[rho - 1] - s) / rho
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(n, f):
    """Sparse (2n-1 x n) matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    i_idx = np.tile(np.arange(n), n)
    j_idx = np.repeat(np.arange(n), n)
    row_idx = j_idx + i_idx
    data = f[i_idx]
    return sp.coo_matrix((data, (row_idx, j_idx)), shape=(2 * n - 1, n)).tocsr()


def lp_direction(f, M, time_limit=5.0):
    """LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def compute_line_score(t, f, g, n):
    """Score at parameter t along direction g from base f (both normalized)."""
    ft = (1.0 - t) * f + t * g
    ft = np.maximum(ft, 0.0)
    s = ft.sum()
    if s < 1e-10:
        return np.inf
    ft = ft / s
    conv_t = np.convolve(ft, ft)
    return 2.0 * n * conv_t.max()


def golden_line_search(f, g, n, a=0.0, b=0.6, tol=1e-5):
    """Minimize score along direction g from base f over t in [a,b]."""
    gr = (np.sqrt(5.0) + 1.0) / 2.0
    c = b - (b - a) / gr
    d = a + (b - a) / gr
    fc = compute_line_score(c, f, g, n)
    fd = compute_line_score(d, f, g, n)

def smoothed_obj_and_grad(theta, T):
    """Smoothed log-sum-exp objective and gradient w.r.t. softmax parameter theta."""
    n = len(theta)
    f = softmax_fn(theta)
    conv = np.convolve(f, f)
    max_c = conv.max()
    exp_part = np.exp(T * (conv - max_c))
    w_sum = exp_part.sum()
    w = exp_part / w_sum
    val = (np.log(w_sum) + T * max_c) / T
    grad_f = 2.0 * np.correlate(w, f, mode='valid')
    grad_theta = f * (grad_f - np.dot(f, grad_f))
    return val, grad_theta

def make_start(n, seed, method):

    if method == 'sin2':
        x = np.linspace(0.0, np.pi, n)

        return a / a.sum()

    return np.ones(n) / n
