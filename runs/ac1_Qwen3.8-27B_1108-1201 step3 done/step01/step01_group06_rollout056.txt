```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences using Nesterov-accelerated
subgradient descent with adaptive top-k, momentum, and plateau escaping.

Key improvements over previous approach:
- Nesterov momentum for faster convergence
- Adaptive top_k that grows over iterations for better gradient quality
- Polyak-style adaptive step size
- Multi-length search (8192, 16384)
- Structured initializations (Gaussian mixtures, symmetric patterns)
- Aggressive plateau detection with multiple escape strategies
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def fft_convolve_self(a):
    """Full self-convolution via FFT (length 2n-1)."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def compute_grad(a, c, top_indices):
    """Weighted subgradient of max(c) using soft-max weighted top-k active lags."""
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    top_vals = c[top_indices]
    # Softmax weighting with temperature based on spread
    spread = top_vals.max() - top_vals.min()
    if spread > 1e-15:
        temp = spread / 3.0
        weights = np.exp((top_vals - top_vals.max()) / temp)
        weights /= weights.sum()
    else:
        weights = np.ones(len(top_vals)) / len(top_vals)
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def momentum_subgrad(a_init, n, S, max_time, rng, top_k_init=16, top_k_max=64,
                     lr_scale=1.0, beta=0.9, noise_scale=0.03):
    """Momentum subgradient descent with simplex projection and plateau escape."""
    a = project_simplex(np.asarray(a_init, dtype=np.float64).copy(), S)
    v = np.zeros(n, dtype=np.float64)

    c = fft_convolve_self(a)
    best_score = c.max()
    best_a = a.copy()
    stagnation = 0
    best_stagnation = 0

    start = time.monotonic()
    step = 0

    while time.monotonic() - start < max_time:
        step += 1
        elapsed = time.monotonic() - start

        # Adaptive top_k
        top_k = min(top_k_max, top_k_init + step // 500)

        c = fft_convolve_self(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1
        best_stagnation = max(best_stagnation, stagnation)

        # Find top-k active lags
        if top_k >= len(c):
            top_indices = np.arange(len(c))
        else:
            top_indices = np.argpartition(c, -top_k)[-top_k:]

        grad = compute_grad(a, c, top_indices)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Adaptive step size
        eta = lr_scale * S / (2.0 * n * np.sqrt(step + 1.0))

        # Momentum update
        v = beta * v - eta * grad
        a = project_simplex(a + v, S)

        # Plateau escape strategies
        if stagnation > 800:
            r = rng.random()
            if r < 0.02:
                # Add structured noise (smooth)
                noise = rng.normal(0, noise_scale * S / n, n)
                # Smooth the noise
                kernel = np.ones(5) / 5.0
                noise = np.convolve(noise, kernel, mode='same')
                a = project_simplex(a + noise, S)
                v = np.zeros(n, dtype=np.float64)
                stagnation = 0
            elif r < 0.03 and step > 2000:
                # Restart with partial reset
                uniform = np.full(n, S / n, dtype=np.float64)
                alpha = 0.3
                a = project_simplex(alpha * a + (1 - alpha) * uniform, S)
                v = np.zeros(n, dtype=np.float64)
                stagnation = 0

        # Periodic logging
        if step % 2000 == 0:
            print(f"    step={step} t={elapsed:.1f}s best={best_score:.6f} cur={cur:.6f} stag={stagnation}")

    return best_a, best_score


def flatten_step(a, c, n, S, amount=0.01):
    """Direct flattening: transfer mass from high-conv contributors to low-conv positions."""
    k_max = int(np.argmax(c))
    k_min_pos = int(np.argmin(c))
    # Find worst contributor to k_max
    lo = max(0, k_max - n + 1)
    hi = min(n - 1, k_max)
    if lo > hi:
        return a
    contribs = a[lo:hi + 1] * a[k_max - hi:k_max - lo + 1][::-1]
    worst_local = int(np.argmax(contribs))
    worst_pos = lo + worst_local

    # Find a position that helps k_min_pos
    lo2 = max(0, k_min_pos - n + 1)
    hi2 = min(n - 1, k_min_pos)
    if lo2 > hi2 or worst_pos < a.nbytes:
        pass
    best_target = -1
    best_gain = 0
    for m in range(n):
        if m == worst_pos:
            continue
        # Effect on k_min_pos
        if 0 <= k_min_pos - m < n:
            gain = a[k_min_pos - m]
            if gain > best_gain:
                best_gain = gain
                best_target = m
    if best_target >= 0 and a[worst_pos] > amount * S / n:
        transfer = min(amount * S / n, a[worst_pos] * 0.1)
        a[worst_pos] -= transfer
        a[best_target] += transfer
    return a


def generate_init_gaussian(n, rng):
    """Generate a Gaussian mixture initialization."""
    n_components = rng.integers(3, 7)
    a = np.zeros(n, dtype=np.float64)
    for _ in range(n_components):
        mu = rng.uniform(0.1, 0.9) * n
        sigma = rng.uniform(0.05, 0.25) * n
        amp = rng.uniform(0.5, 1.5)
        x = np.arange(n, dtype=np.float64)
        a += amp * np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    return a


def generate_init_symmetric(n, rng):
    """Generate a symmetric initialization (a[i] = a[n-1-i])."""
    half = n // 2
    a_half = rng.uniform(0.5, 1.5, half)
    # Smooth
    kernel = np.ones(7) / 7.0
    a_half = np.convolve(a_half, kernel, mode='same')
    a = np.zeros(n, dtype=np.float64)
    a[:half] = a_half
    a[half:] = a_half[::-1]
    return np.maximum(a, 0.0) + 0.1


def generate_init_periodic(n, rng):
    """Generate a periodic-ish initialization."""
    period = rng.integers(max(4, n // 64), max(8, n // 32))
    base = np.linspace(0, 2 * np.pi, period)
    pattern = 1.0 + 0.3 * np.sin(base + rng.uniform(0, 2 * np.pi)) + \
              0.15 * np.sin(2 * base + rng.uniform(0, 2 * np.pi))
    a = np.zeros(n, dtype=np.float64)
    for i in range(n):
        a[i] = pattern[i % period]
    a = np.maximum(a, 0.0)
    return a
```
