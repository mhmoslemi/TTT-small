def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Compute 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b) / (s * s)

def _score_unit(a):
    """Score for unit-sum sequence: 2*n*max(b)."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)

    Subgradient descent with optional backtracking line search.
    Minimizes max_k conv(a,a)[k] subject to sum(a)=1, a>=0.
    """
    n = len(a)

    best_score = _score_unit(best)

        eta_init = 0.02 / np.sqrt(n)
    eta = eta_init
    no_improve = 0
    max_no_improve = 200
    
    if rng is None:

            lo = max(0, k - n + 1)
            hi = min(n - 1, k)

                grad[idx] += 2.0 * a[k - idx]

            best_obj = cur_score

    return best, best_score

def _lp_direction(a, M):
    """
    Solve LP to find a non-negative direction g that keeps conv(a,g) <= M.
    Maximizes sum(g) to find the largest such direction.
    Returns the normalized direction (sum=1) or None if infeasible.
    """
    n = len(a)
    # Variables: g[0], ..., g[n-1]
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: conv(a, g)[k] <= M for k = 0, ..., 2n-2
    # conv(a, g)[k] = sum_i a[i] * g[k-i] (valid i)
    # For a given k, the constraint is: sum_{i} a[i] * g[k-i] <= M
    # where i ranges over max(0, k-n+1) to min(n-1, k)
    # In terms of g[j] (where j = k-i, so j ranges over k-min(n-1,k) to k-max(0,k-n+1)):
    #   sum_j a[k-j] * g[j] <= M
    
    # Build constraint matrix A_ub (each row is one constraint)
    # Number of constraints: 2n-1
    n_cons = 2 * n - 1
    A_ub = np.zeros((n_cons, n))
    b_ub = np.full(n_cons, M)
    
    for k in range(n_cons):
        # g[j] appears with coefficient a[k-j] where j = k-i, i in valid range
        # i in [max(0,k-n+1), min(n-1,k)]
        # j = k - i, so j in [k - min(n-1,k), k - max(0,k-n+1)]
        # j in [max(0, k-(n-1)), min(n-1, k)]
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        for j in range(j_lo, j_hi + 1):
            A_ub[k, j] = a[k - j]
    
    # Bounds: g[j] >= 0
    bounds = [(0, None)] * n
    
    # Solve
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        g = result.x
        s = np.sum(g)
        if s > 1e-10:
            return g / s
    return None

Alternative to LP: **Mirror descent / Multiplicative weights update**

For the problem of minimizing `max_k b[k]` subject to `a >= 0, sum(a) = 1`, we can use the multiplicative weights (Hedge) algorithm:

- Initialize `a[i] = 1/n`
- At each step:
  1. Compute `b = conv(a, a)`
  2. Find the lag `k*` with maximum `b[k*]`
  3. The "loss" for each coordinate `j` is `L[j] = 2*a[k*-j]` (the gradient of `b[k*]` at `j`)
  4. Update: `a[j] = a[j] * exp(-eta * (L[j] - mean(L)))`
  5. Normalize: `a = a / sum(a)`

This is equivalent to mirror descent with the negative entropy regularizer, which is well-suited for simplex constraints.

Actually, this is essentially the exponential weights / multiplicative weights method for online convex optimization. It's known to have good convergence guarantees.

The continuation idea is interesting: instead of directly minimizing `max_k b[k]`, minimize a smooth approximation like `log-sum-exp` of the b[k] values:

`f(a) = (1/tau) * log(sum_k exp(tau * b[k]))`

For large tau, this approximates `max_k b[k]`. For small tau, it's smoother and easier to optimize. We can start with small tau and increase it.

The gradient of `f(a)` is:
`grad[j] = (1/tau) * sum_k exp(tau * b[k]) * (db[k]/da[j]) / sum_k exp(tau * b[k])`
`= sum_k p[k] * (db[k]/da[j])`

where `p[k] = exp(tau * b[k]) / sum exp(tau * b[k])` is a softmax distribution over the lags.

This is a weighted average of the gradients, with weights concentrated on the peak lags. As tau increases, more weight is on the actual maximum.

This is actually a great approach because:
- It's smooth (differentiable).
- The gradient is well-defined even when multiple lags are tied at the max.
- We can use standard optimization methods (Adam, L-BFGS, etc.).

Optimized subgradient and smooth-max (log-sum-exp) descent to minimize the
normalized autocorrelation peak of a non-negative sequence. Uses continuation
on the smoothness parameter, FFT-based gradient computation, Adam optimizer,
and multiple diverse starting points.

import numpy as np
import time

def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
