```python
"""
Minimize 2*n*max(f*f)/(sum f)^2 for non-negative f using iterative LP and
coordinate descent. Scale-invariant: fix sum(f)=1, minimize 2*n*max(f*f).

Key LP: min t s.t. (f_cur * g)[k] <= t ∀k, sum(g)=1, g>=0.
This finds the best "direction" to move that minimizes the weighted peak.
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time

def fast_score(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def build_A(n, f):
    """Convolution matrix A (2n-1 x n): A[k,j] = f[k-j] if 0<=k-j<n."""
    row_idx = np.tile(np.arange(n), n) + np.repeat(np.arange(n), n)
    col_idx = np.repeat(np.arange(n), n)
    data = np.tile(f, n)
    return sp.coo_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n)).tocsr()

def solve_lp(f, time_limit=5.0):
    """Solve min t s.t. A@g <= t, sum(g)=1, g>=0. Returns (g, t) or None."""
    n = len(f)
    A = build_A(n, f)
    
    # Variables: x = [g_0, ..., g_{n-1}, t], length n+1
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequality: A@g - t <= 0
    # A_ub has shape (2n-1, n+1)
    A_aug = np.zeros((2*n-1, n+1))
    A_aug[:, :n] = A.toarray()  # This is dense, might be slow for large n
    A_aug[:, n] = -1.0
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_aug, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': time_limit})
    if res.status == 0:
        g = np.maximum(res.x[:n], 0.0)
        t = res.x[n]
        return g, t
    return None
```

This coordinate descent is O(n^2) per sweep (each coordinate update requires computing convolutions which are O(n) each, and there are n coordinates). For n=1024, each sweep is about 1024 * 20 * O(n) = 1024 * 20 * 1024 ≈ 20M operations. With the fine grid, maybe 30M. That's fast enough.

But wait, I'm computing np.convolve(f_try, f_try) which is O(n^2) for each evaluation. So each coordinate update is O(n * n^2) = O(n^3) in the worst case (20 evaluations * O(n^2) each). For n=1024, that's 20 * 1024^2 ≈ 20M per coordinate, and 1024 coordinates per sweep = 20 billion. That's too slow!

I need to be smarter. Instead of recomputing the full convolution each time, I can update it incrementally.

Let me also think about whether the LP formulation I have is correct and effective.

The LP: min t s.t. (f_cur * g)[k] <= t for all k, sum(g) = 1, g >= 0.

The solution g* is the probability vector that minimizes the peak of f_cur * g*. Intuitively, g* "avoids" the regions where f_cur is large. If f_cur is concentrated in the middle, g* will be concentrated at the edges, and vice versa.

When we set f_new = g*, we're moving to a new distribution that "complements" the old one. The hope is that f_new * f_new has a lower peak than f_cur * f_cur.

This is similar to the "alternating minimization" approach for matrix factorization, where you alternate between fixing one factor and optimizing the other.

One concern: the iteration might oscillate. If f is concentrated on the left, g will be concentrated on the right, and vice versa. To prevent this, I should blend: f_new = (1-α)*f + α*g.

Let me also think about a completely different approach: **directed local search**.

Idea: Start with a good initial sequence. At each step, pick a random subset of coordinates (a "block") and solve a small optimization problem to find the best values for those coordinates (keeping the others fixed). This is like a "Gaussian elimination" approach.

"""
Minimize 2*n*max(f*f)/(sum(f))^2 for non-negative sequences using:
1. Iterative LP: min peak(f_cur * g) s.t. sum(g)=1, g>=0
2. Line search along the LP direction
3. Multi-start with different lengths and initializations
4. Final coordinate descent polish
"""
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog, minimize_scalar
import time

def fast_score(f):
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))

def build_A(n, f):
    row_idx = np.tile(np.arange(n), n) + np.repeat(np.arange(n), n)
    col_idx = np.repeat(np.arange(n), n)
    data = np.tile(f, n)
    return sp.coo_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n)).tocsr()

def solve_lp_minpeak(f, time_limit=5.0):
    """Solve min t s.t. (f*g)[k] <= t for all k, sum(g)=1, g>=0."""
    n = len(f)
    A = build_A(n, f)
    
    A_eq_row = np.ones(n)
    # A_ub: [A, -1]
    neg_col = sp.csr_matrix(np.array([[-1.0]] * (2*n-1)))
    A_ub = sp.hstack([A, neg_col], format='csr')
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    b_ub = np.zeros(2*n-1)
    A_eq = A_eq_row.reshape(1, -1)
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    if res.status == 0:
        g = np.maximum(res.x[:n], 0.0)
        gs = g.sum()
        if gs < 1e-12:
            return None
        g = g / gs
        return g
    return None

def lp_line_search(f, g, n_evals=50):
    """Line search: minimize score((1-a)*f + a*g) for a in [0,1]."""
    f = np.asarray(f, dtype=np.float64)
    g = np.asarray(g, dtype=np.float64)
    
    def obj(a):
        f_a = (1 - a) * f + a * g
        f_a = np.maximum(f_a, 0.0)
        s = f_a.sum()
        if s < 1e-10:
            return np.inf
        f_a = f_a / s
        return fast_score(f_a)
    
    res = minimize_scalar(obj, bounds=(0, 1), method='bounded',
                          options={'xatol': 1e-6, 'maxiter': n_evals})
    a_best = res.x
    f_best = (1 - a_best) * f + a_best * g
    f_best = np.maximum(f_best, 0.0)
    s = f_best.sum()
    if s < 1e-10:
        return f
    return f_best / s

def make_initial(n, method, rng):
    if method == 'uniform':
        f = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        f = 0.5 * (1 + np.cos(x))
    elif method == 'gaussian':
        x = np.arange(n)
        f = np.exp(-0.5 * ((x - n/2) / (n/4))**2)
    elif method == 'bimodal':
        x = np.arange(n)
        f = np.exp(-0.5 * ((x - n*0.25) / (n/8))**2) + np.exp(-0.5 * ((x - n*0.75) / (n/8))**2)
    elif method == 'random':
        f = rng.random(n)
    elif method == 'sin2':
        x = np.linspace(0, np.pi, n)
        f = np.sin(x)**2
    else:
        f = np.ones(n)
    f = np.maximum(f, 0.01)
    f = f / f.sum()
    return f
