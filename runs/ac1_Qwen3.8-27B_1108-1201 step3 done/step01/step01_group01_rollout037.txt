import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import fftconvolve


def _compute_score_normalized(a):
    """Compute 2*n*max(b) for normalized a (sum=1)."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _make_anti_smooth(n, eps):
    """Smooth anti-symmetric: 1 + eps*cos(pi*i/(n-1))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_two_level(n, eps):
    """Two-level anti-symmetric."""
    a = np.ones(n)
    a[:n // 2] = 1.0 + eps
    a[n // 2:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    return a / S

    Objective and gradient for softmax parameterization.
    a = softmax(z), F = (sum_k b_k^p)^(1/p)
    """
    n = len(z)

    z_shift = z - np.max(z)

    a = exp_z / np.sum(exp_z)
    
    # Compute b = convolve(a, a)
    b = fftconvolve(a, a, mode='full')
    
    # Compute F and gradient

        # grad_a = 2 * convolve(ones, a[::-1])[n-1:2n-1] = 2 * (length of valid overlap)

        w = np.ones_like(b)
    else:
        b_p = np.sum(b ** p)
        F = b_p ** (1.0 / p)
        w = b ** (p - 1)
        norm_factor = b_p ** (1.0 - 1.0 / p)
    
    # grad_a_j = 2 * sum_k w_k * a_{k-j} / norm_factor (for p > 1)
    # = 2 * convolve(w, a[::-1])[n-1:2n-1] / norm_factor
    w_a_conv = fftconvolve(w, a[::-1], mode='full')

    if p > 1:

    # grad_z = a * (grad_a - sum(a * grad_a))
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)
    
    return F, grad_z


def _lbfgs_optimize(a_start, p, deadline, max_iter=500, tol=1e-10):

    Optimize using L-BFGS with softmax parameterization.
    """
    n = len(a_start)
    # Initialize z from a_start: z_i = log(a_i) (approximately)
    z0 = np.log(np.maximum(a_start, 1e-30))

        return F, g
    
    result = minimize(

    # Check time

    # Also check intermediate results... (L-BFGS doesn't easily expose these)
    # Just return the final result
    a_opt = _softmax(result.x)
    score = _compute_score_normalized(a_opt)
    return a_opt, score


def _softmax(z):

"""
P-norm annealing with L-BFGS (softmax parameterization) for minimizing
the normalized autocorrelation peak. Uses smooth p-norm surrogates that
sharpen toward the max, combined with subgradient refinement on the true
objective at the end.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import fftconvolve


def _compute_score_normalized(a):
    """Compute 2*n*max(convolve(a,a)) for a with sum=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax(z):
    """Numerically stable softmax."""
    z_shift = z - np.max(z)
    exp_z = np.exp(z_shift)
    return exp_z / np.sum(exp_z)


def _pnorm_func_grad(z, p):
    """
    Compute p-norm of convolve(a,a) and its gradient w.r.t. z,
    where a = softmax(z).
    """
    n = len(z)
    a = _softmax(z)
    b = fftconvolve(a, a, mode='full')

    if p <= 1.0:
        # L1-like: sum of b (smooth for positive b)
        F = np.sum(b)
        w = np.ones_like(b)
        norm_factor = 1.0
    else:
        b_p = np.sum(b ** p)
        F = b_p ** (1.0 / p)
        w = b ** (p - 1)
        norm_factor = b_p ** (1.0 - 1.0 / p)

    # grad_a = 2 * convolve(w, a_rev)[n-1 : 2n-1] / norm_factor
    a_rev = a[::-1]
    w_a_conv = fftconvolve(w, a_rev, mode='full')
    grad_a = 2.0 * w_a_conv[n - 1: 2 * n - 1]
    grad_a /= norm_factor

    # grad_z = a * (grad_a - dot(a, grad_a))
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)

    return F, grad_z


def _pbfgs_optimize(a_start, p, deadline, maxiter=300):
    """
    Run L-BFGS-B to minimize p-norm using softmax param.
    Returns (best_a, best_true_score).
    """
    n = len(a_start)
    z0 = np.log(np.maximum(a_start, 1e-30))
    z0 -= np.mean(z0)

    best_a = a_start.copy()
    best_score = _compute_score_normalized(a_start)
    call_count = [0]
    start_time = time.monotonic()

    def wrapped_func(z):
        call_count[0] += 1
        F, g = _pnorm_func_grad(z, p)
        return F, g

    try:
        result = minimize(
            wrapped_func, z0, jac=True,
            method='L-BFGS-B',
            options={
                'maxiter': maxiter,
                'ftol': 1e-14,
                'gtol': 1e-6,
                'maxfun': maxiter * 5
            }
        )
        a_opt = _softmax(result.x)
        score = _compute_score_normalized(a_opt)
        if score < best_score:
            best_a = a_opt
            best_score = score
    except Exception:
        pass

    return best_a, best_score


def _subgradient_refine(a, deadline, initial_eta=None, M_weight=5.0):
    """
    Subgradient descent on the true max objective with momentum.
    Minimizes 2n*max(b) subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a = a / S
    else:
        a = np.ones(n) / n

    best = a.copy()
    best_score = _compute_score_normalized(best)

    if initial_eta is None:
        initial_eta = 0.05 / np.sqrt(n)
    eta = initial_eta
    momentum = np.zeros(n)
    beta_m = 0.9

    no_improve = 0
    patience = 200

    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-12:
                break

        # Weighted subgradient using all peaks with exponential weighting
        w = np.exp(-M_weight * (b_max - b) / max(b_max, 1e-15))
        w /= np.sum(w)

        # Compute weighted gradient
        grad_a = 2.0 * fftconvolve(w, a[::-1], mode='full')[n - 1: 2 * n - 1]

        # Center for sum constraint
        grad_a -= np.mean(grad_a)

        # Momentum update
        momentum = beta_m * momentum + grad_a
        a = a - eta * momentum

        # Project
        a = _project_simplex(a, 1.0)

    return best, best_score


def _make_anti_smooth(n, eps):
    """Smooth anti-symmetric profile."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_two_level(n, eps):
    """Two-level anti-symmetric sequence."""
    a = np.ones(n)
    half = n // 2
    a[:half] = 1.0 + eps
    a[half:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    return a / S


def _make_four_level(n, eps):
    """Four-level anti-symmetric sequence."""
    a = np.ones(n)
    q = n // 4
    a[:q] = 1.0 + eps
    a[q:2*q] = 1.0 + eps / 2.0
    a[2*q:3*q] = 1.0 - eps / 2.0
    a[3*q:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    return a / S

    """
    Run the full p-annealing + refinement pipeline from a starting point.
    Returns (best_a, best_score).
    """
    n = len(a_start)

    S = np.sum(a)
    if S > 1e-10:
        a = a / S

    t_start = time.monotonic()

    # P-annealing schedule
    p_schedule = [2, 5, 20, 50, 100, 200]
    total_p_time = (deadline - t_start) * 0.65

    for p in p_schedule:
        if time.monotonic() >= deadline:
            break
        # Allocate time proportional to remaining
        remaining_ps = p_schedule[p_schedule.index(p):]
        time_for_p = total_p_time / len(remaining_ps)
        p_deadline = min(time.monotonic() + time_for_p, deadline)

        # More iterations for smaller p
        if p <= 5:

        elif p <= 50:
            maxiter = 150
        else:
            maxiter = 100

        a_new, score_new = _pbfgs_optimize(a, p, p_deadline, maxiter=maxiter)

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new.copy()
            print(f"  {label} p={p}: improved to {score_new:.6f}")
        else:
            print(f"  {label} p={p}: no improvement ({score_new:.6f} vs {best_score:.6f})")

            a = a_new.copy()
