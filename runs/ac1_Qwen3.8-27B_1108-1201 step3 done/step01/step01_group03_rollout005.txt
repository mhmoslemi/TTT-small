    g = solve_lp_direction(f, b_vec, lp_time)
    if g is None:
        return f.copy(), fast_score(f), False

    gs = g.sum()
    if gs < 1e-10:
        return f.copy(), fast_score(f), False
    g = g / gs

    # Line search
    base_sc = fast_score(f)
    best_f = f.copy()
    best_sc = base_sc
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        cand = (1.0 - t) * f + t * g
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-10:
            continue
        cand = cand / s
        sc = fast_score(cand)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_f = cand.copy()

    improved = best_sc < base_sc - 1e-10
    return best_f, best_sc, improved


def block_perturb(f, rng, w_min=1, w_max=None, noise_std=0.25):
    """Multiplicative random block perturbation, renormalised to sum 1."""
    n = len(f)
    if w_max is None:
        w_max = max(2, n // 10)
    w = int(rng.integers(w_min, max(w_min + 1, min(w_max, n))))
    w = max(1, min(w, n))
    start = int(rng.integers(0, max(1, n - w + 1)))
    stop = min(n, start + w)
    cand = f.copy()
    factor = float(np.exp(rng.normal(0.0, noise_std)))
    cand[start:stop] *= factor
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f.copy()
    return cand / s


def make_initial(n, method, rng):
    """Generate a non-negative sum-normalised starting vector."""
    if method == 'uniform_perturb':
        a = np.ones(n) + rng.normal(0, 0.3, n)
    elif method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = 0.5 * (1.0 + np.cos(x))
    elif method == 'sine_sq':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
    elif method == 'random':
        a = rng.random(n) + 0.01
    elif method == 'triangular':
        x = np.arange(n) / max(1, n - 1)
        a = 1.0 - np.abs(2.0 * x - 1.0)
    elif method == 'two_gaussians':
        x = np.arange(n, dtype=float)
        s = max(1, n // 8)
        a = np.exp(-0.5 * ((x - 0.25 * n) / s) ** 2) + \
            np.exp(-0.5 * ((x - 0.75 * n) / s) ** 2)
    elif method == 'staircase':
        a = np.linspace(0.1, 1.0, n)
    elif method == 'chirp_like':
        x = np.arange(n, dtype=float) / n
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x * 3.0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def subgradient_polish(f, budget_s, alpha_init=0.02, alpha_decay=0.999):
    """Projected subgradient descent with momentum for polishing."""
    n = len(f)
    best = f.copy()
    best_sc = fast_score(f)
    f_curr = f.copy()
    momentum = np.zeros(n)
    alpha = alpha_init
    beta = 0.9  # momentum coefficient

    while time.monotonic() < budget_s:
        conv = np.convolve(f_curr, f_curr)
        M = conv.max()
        if M < 1e-15:
            break
        # Find peak positions (within 1% of max)
        threshold = M * 0.995
        peak_mask = conv >= threshold
        peak_count = peak_mask.sum()
        if peak_count == 0:
            break

        # Build subgradient: average of A[k,:] over peak rows
        A = build_conv_matrix(n, f_curr)
        grad = np.asarray(A[peak_mask, :].tocsr().sum(axis=0)).flatten()
        grad /= float(peak_count)

        # Momentum update
        momentum = beta * momentum - alpha * grad
        f_new = f_curr + momentum
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            alpha *= 0.5
            continue
        f_new = f_new / s
        f_curr = f_new
        alpha *= alpha_decay

        sc = fast_score(f_curr)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = f_curr.copy()

    return best, best_sc


def coordinate_descent_polish(f, budget_s, max_iter=5000):
    """Try adjusting individual elements to reduce peak."""
    n = len(f)
    best = f.copy()
    best_sc = fast_score(f)
    f_curr = f.copy()
    rng = np.random.default_rng(12345)

    for it in range(max_iter):
        if time.monotonic() > budget_s:
            break
        # Pick a random element to adjust
        i = int(rng.integers(0, n))
        old_val = f_curr[i]
        if old_val < 1e-10:
            continue
        # Try multiplying by various factors
        best_local = old_val
        best_local_sc = best_sc
        # Get current conv
        conv = np.convolve(f_curr, f_curr)
        cur_M = conv.max()
        for factor in [0.5, 0.7, 0.85, 0.9, 1.1, 1.15, 1.3, 1.5]:
            new_val = old_val * factor
            if new_val < 0:
                continue
            f_curr[i] = new_val
            # Quick score check
            conv2 = np.convolve(f_curr, f_curr)
            sc = 2.0 * n * conv2.max() / (f_curr.sum() ** 2)
            if sc < best_local_sc:
                best_local_sc = sc
                best_local = new_val
        f_curr[i] = best_local
        f_curr = np.maximum(f_curr, 0.0)
        s = f_curr.sum()
        if s > 1e-10:
            f_curr = f_curr / s
        sc = fast_score(f_curr)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best = f_curr.copy()

    return best, best_sc


def run_lp_phase(f, budget_s, t_levels=None, lp_time=5.0,
                 perturb_every=15, perturb_prob=0.3):
    """Run LP iterations until budget exhausted. Returns (best_f, best_score)."""
    if t_levels is None:
        t_levels = [1.0, 0.99, 0.98, 0.97, 0.95]

    best = f.copy()
    best_sc = fast_score(f)
    f_curr = f.copy()
    stall = 0
    it = 0
    rng = np.random.default_rng(99999)

    while time.monotonic() < budget_s:
        it += 1
        # Cycle through t_levels
        tl = t_levels[it % len(t_levels)]
        # Occasionally use a more aggressive level
        if it % 50 == 0:
            tl = max(0.9, tl - 0.02)

        f_new, sc_new, improved = lp_step(f_curr, tl, lp_time)

        if improved:
            f_curr = f_new
            stall = 0
            if sc_new < best_sc:
                best_sc = sc_new
                best = f_curr.copy()
        else:
            stall += 1
            # Try perturbation to escape
            if stall > 5 and (rng.random() < perturb_prob):
                f_curr = block_perturb(f_curr, rng, w_min=1,
                                       w_max=max(2, len(f_curr) // 8),
                                       noise_std=0.15)
                stall = 0
            elif stall > 20:
                f_curr = block_perturb(f_curr, rng, w_min=2,
                                       w_max=max(4, len(f_curr) // 5),
                                       noise_std=0.3)
                stall = 0

        # Periodic larger perturbation
        if it % 100 == 0:
            f_pert = block_perturb(f_curr, rng, w_min=1,
                                   w_max=max(3, len(f_curr) // 12),
                                   noise_std=0.1)
            sc_pert = fast_score(f_pert)
            if sc_pert < fast_score(f_curr):
                f_curr = f_pert

    return best, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Search for the best non-negative sequence minimizing the evaluation score."""
    start_time = time.monotonic()
    deadline = start_time + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(seed)

    # Get baseline from provided sequence
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                         0.0, 1000.0)
    except Exception:
        parent = np.ones(512, dtype=np.float64)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(512, dtype=np.float64)

    best = np.maximum(parent, 0.0).copy()
    bs = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(bs):
        bs = fast_score(best)
    print("Baseline score = %.6f (n=%d)" % (bs, best.size), flush=True)

    def try_update(candidate, score):
        nonlocal best, bs
        if np.isfinite(score) and score < bs - 1e-10:
            bs = score
            best = np.maximum(candidate, 0.0).copy()
            print("  NEW BEST: %.6f (n=%d) at t=%.0fs" %
                  (bs, best.size, time.monotonic() - start_time), flush=True)
            return True
        return False

    # Phase 1: Coarse multi-start search
    print("=== Phase 1: Coarse multi-start search ===", flush=True)
    phase1_budget = min(300.0, (deadline - start_time) * 0.2)
    phase1_deadline = start_time + phase1_budget

    lengths = [384, 512, 640, 768, 1024, 1280, 1536, 2048]
    methods = ['uniform_perturb', 'cosine', 'sine_sq', 'random',
               'triangular', 'two_gaussians', 'staircase', 'chirp_like']

    # Include parent length if it's reasonable
    if best.size <= 2048 and best.size not in lengths:
        lengths.append(int(best.size))
    lengths = sorted(set(lengths))

    # Allocate time per (length, method) combo
    n_combos = min(len(lengths) * 2, 12)  # Test 2 methods per length
    per_combo_time = max(10.0, phase1_budget / max(1, n_combos))

    combos = []
    for n in lengths:
        for m in methods[:2]:  # Only first 2 methods per length for speed
            combos.append((n, m))
    # Also try a few extra random methods
    for n in [512, 1024]:
        combos.append((n, 'random'))

    for (n, m) in combos:
        if time.monotonic() > phase1_deadline:
            break
        start_f = make_initial(n, m, rng)
        local_deadline = min(phase1_deadline, time.monotonic() + per_combo_time)
        print("  Coarse: n=%d method=%s budget=%.0fs" % (n, m, per_combo_time),
              flush=True)
        c_best, c_sc = run_lp_phase(start_f, local_deadline,
                                    t_levels=[1.0, 0.98, 0.96],
                                    lp_time=3.0, perturb_every=10)
        if np.isfinite(c_sc):
            c_conf = float(evaluate_sequence(c_best.tolist()))
            if np.isfinite(c_conf):
                c_sc = min(c_sc, c_conf)
            try_update(c_best, c_sc)

    # Also try from the parent sequence
    if best.size >= 100 and best.size <= 2048:
        parent_norm = best / best.sum()
        local_deadline = min(phase1_deadline, time.monotonic() + 60.0)
        print("  Coarse from parent: n=%d budget=%.0fs" %
              (best.size, local_deadline - time.monotonic()), flush=True)
        c_best, c_sc = run_lp_phase(parent_norm, local_deadline,
                                    t_levels=[1.0, 0.99, 0.98],
                                    lp_time=4.0)
        if np.isfinite(c_sc):
            c_conf = float(evaluate_sequence(c_best.tolist()))
            if np.isfinite(c_conf):
                c_sc = min(c_sc, c_conf)
            try_update(c_best, c_sc)

    # Phase 2: Deep LP search on best
    print("=== Phase 2: Deep LP search ===", flush=True)
    print("  Current best: %.6f (n=%d)" % (bs, best.size), flush=True)
    phase2_budget = min(1000.0, (deadline - start_time) * 0.55)
    phase2_deadline = start_time + phase2_budget

    f_deep = best.copy()
    if f_deep.sum() > 0:
        f_deep = f_deep / f_deep.sum()

    # Run multiple LP sub-phases with different parameters
    sub_phases = [
        {'t_levels': [1.0], 'lp_time': 5.0, 'frac': 0.3},
        {'t_levels': [0.99, 0.98], 'lp_time': 4.0, 'frac': 0.3},
        {'t_levels': [0.97, 0.95, 0.93], 'lp_time': 4.0, 'frac': 0.2},
        {'t_levels': [0.92, 0.90, 0.88], 'lp_time': 3.0, 'frac': 0.2},
    ]

    for sp_cfg in sub_phases:
        if time.monotonic() > phase2_deadline:
            break
        sub_budget = phase2_budget * sp_cfg['frac']
        sub_deadline = time.monotonic() + sub_budget
        sub_deadline = min(sub_deadline, phase2_deadline)
        if sub_deadline <= time.monotonic():
            break
        print("  Sub-phase: t_levels=%s lp_time=%.1f" %
              (sp_cfg['t_levels'], sp_cfg['lp_time']), flush=True)
        c_best, c_sc = run_lp_phase(f_deep, sub_deadline,
                                    t_levels=sp_cfg['t_levels'],
                                    lp_time=sp_cfg['lp_time'])
        if np.isfinite(c_sc):
            c_conf = float(evaluate_sequence(c_best.tolist()))
            if np.isfinite(c_conf):
                c_sc = min(c_sc, c_conf)
            if try_update(c_best, c_sc):
                f_deep = best.copy()
                if f_deep.sum() > 0:
                    f_deep = f_deep / f_deep.sum()

    # Phase 3: Polishing
    print("=== Phase 3: Polishing ===", flush=True)
    polish_deadline = min(deadline, start_time + budget_s - 15.0)
    remaining = max(10.0, polish_deadline - time.monotonic())
    print("  Time remaining for polish: %.0fs" % remaining, flush=True)

    # 3a: Subgradient polish
    sg_budget = remaining * 0.4
    if sg_budget > 5.0:
        print("  Subgradient polish: %.0fs" % sg_budget, flush=True)
        f_sg = best.copy()
        if f_sg.sum() > 0:
            f_sg = f_sg / f_sg.sum()
        c_best, c_sc = subgradient_polish(f_sg, time.monotonic() + sg_budget)
        if np.isfinite(c_sc):
            c_conf = float(evaluate_sequence(c_best.tolist()))
            if np.isfinite(c_conf):
                c_sc = min(c_sc, c_conf)
            try_update(c_best, c_sc)

    # 3b: Coordinate descent
    cd_budget = remaining * 0.3
    if cd_budget > 5.0 and time.monotonic() < polish_deadline:
        print("  Coordinate descent polish: %.0fs" % cd_budget, flush=True)
        f_cd = best.copy()
        if f_cd.sum() > 0:
            f_cd = f_cd / f_cd.sum()
        c_best, c_sc = coordinate_descent_polish(f_cd, time.monotonic() + cd_budget)
        if np.isfinite(c_sc):
            c_conf = float(evaluate_sequence(c_best.tolist()))
            if np.isfinite(c_conf):
                c_sc = min(c_sc, c_conf)
            try_update(c_best, c_sc)

    # 3c: Fine block perturbations
    if time.monotonic() < polish_deadline:
        print("  Fine perturbation polish...", flush=True)
        f_poly = best.copy()
        if f_poly.sum() > 0:
            f_poly = f_poly / f_poly.sum()
        rng2 = np.random.default_rng(seed + 7777)
        best_poly_sc = fast_score(f_poly)
        while time.monotonic() < polish_deadline:
            for noise in [0.05, 0.1, 0.15]:
                if time.monotonic() >= polish_deadline:
                    break
                cand = block_perturb(f_poly, rng2, w_min=1,
                                     w_max=max(2, len(f_poly) // 20),
                                     noise_std=noise)
                sc = fast_score(cand)
                if np.isfinite(sc) and sc < best_poly_sc - 1e-10:
                    f_poly = cand
                    best_poly_sc = sc
                    sc_conf = float(evaluate_sequence(cand.tolist()))
                    if np.isfinite(sc_conf):
                        try_update(cand, sc_conf)
            # Also try a slightly larger perturbation every few iterations
            if time.monotonic() < polish_deadline:
                cand = block_perturb(f_poly, rng2, w_min=2,
                                     w_max=max(5, len(f_poly) // 10),
                                     noise_std=0.08)
                sc = fast_score(cand)
                if np.isfinite(sc) and sc < best_poly_sc - 1e-10:
                    f_poly = cand
                    best_poly_sc = sc
                    sc_conf = float(evaluate_sequence(cand.tolist()))
                    if np.isfinite(sc_conf):
                        try_update(cand, sc_conf)

    # Final validation
    best = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    if best.size == 0 or not np.isfinite(best).all() or best.sum() < 0.01:
        best = np.ones(512, dtype=np.float64)

    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score):
        final_score = fast_score(best)
    print("FINAL: score=%.6f (n=%d) time=%.0fs" %
          (final_score, best.size, time.monotonic() - start_time), flush=True)
    return [float(x) for x in best]
