from scipy.optimize import linprog
import time

def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))

def solve_lp(a, M, timeout=30.0):
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except:
        pass
    return None

def subgradient_polish(a, n_steps=20, step_size=0.05):
    """Reduce peak autocorrelation using subgradient descent."""
    n = len(a)
    for _ in range(n_steps):
        conv = np.convolve(a, a)
        max_val = conv.max()
        # Find peak indices (within tolerance)
        peak_idx = np.where(conv >= max_val * (1 - 1e-4))[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in peak_idx[:20]:  # Limit to 20 peaks for speed
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            indices = np.arange(j_min, j_max + 1)
            vals = a[k - indices]
            factor = np.full(len(indices), 2.0)
            if k % 2 == 0 and k // 2 >= j_min and k // 2 <= j_max:
                factor[k // 2 - j_min] = 1.0
            grad[indices] += factor * vals
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        a = a - step_size * grad / gn
        a = np.maximum(0, a)
        a = normalize_nonneg(a)
    
    return a

"""
Minimize peak auto-convolution of non-negative sequences using
LP-based refinement with backtracking line search, subgradient polishing,
and local search across multiple lengths and initializations.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
from scipy.special import i0
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def build_conv_matrix(a):
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp(a, M, timeout=30.0):
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def compute_subgradient(a, k):
    """Compute dc[k]/da for a given lag k."""
    n = len(a)
    grad = np.zeros(n)
    j_min = max(0, k - n + 1)
    j_max = min(n - 1, k)
    if j_min > j_max:
        return grad
    # a[k-j] for j in [j_min, j_max]
    a_start = k - j_max
    a_end = k - j_min
    vals = a[a_start:a_end + 1][::-1].copy()
    grad[j_min:j_max + 1] += 2.0 * vals
    # Fix diagonal: if k = 2j for some j in range
    if k % 2 == 0:
        j_diag = k // 2
        if j_min <= j_diag <= j_max:
            grad[j_diag] -= a[j_diag]
    return grad


def subgradient_polish(a, n_steps=30, step_size=0.05, verbose=False):
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = score_sequence(a)
    for step in range(n_steps):
        conv = np.convolve(a, a)
        max_val = conv.max()
        # Find peaks within 1% of max
        threshold = max_val * 0.99
        peak_indices = np.where(conv >= threshold)[0]
        # Limit to top 10 peaks
        if len(peak_indices) > 10:
            peak_indices = peak_indices[np.argsort(conv[peak_indices])[-10:]]
        
        grad = np.zeros(n)
        for k in peak_indices:
            grad += compute_subgradient(a, int(k))
        
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        a_new = a - step_size * grad / gn
        a_new = np.maximum(0.0, a_new)
        a_new = normalize_nonneg(a_new)
        s_new = score_sequence(a_new)
        if s_new < best_score:
            best_score = s_new
            best_a = a_new.copy()
        a = a_new
    
    if verbose:
        print(f"  SG done: best={best_score:.6f}")
    return best_a, best_score

    n = len(a)
