import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.fft import fft, ifft, rfft, irfft

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]

def score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def make_odd_symmetric_init(n, k, c, rng):
    """Create a sequence with odd-symmetric perturbation of uniform."""
    j = np.arange(n, dtype=np.float64)
    a = 1.0 + c * np.sin(2.0 * np.pi * k * (j + 0.5) / n)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

def lp_step(a, M=None):
    """
    Solve the LP: maximize sum(g) subject to g >= 0 and (a*g)[k] <= M for all k.
    Returns the optimal g.
    """
    n = len(a)
    if M is None:
        c = fft_convolve(a, a)
        M = c.max()
    
    # Build constraint matrix A where A[k,j] = a[k-j] if valid
    # (a*g)[k] = sum_j A[k,j] * g[j] <= M
    # For k = 0, ..., 2n-2:
    #   valid j: max(0, k-n+1) <= j <= min(n-1, k)
    #   A[k,j] = a[k-j]
    
    # We use scipy linprog: minimize c^T x subject to A_ub x <= b_ub, x >= 0
    # Objective: minimize -sum(g), so c = -1
    c_obj = -np.ones(n)
    
    # Build A_ub (sparse would be better but for n<=1024 dense is OK)
    # A_ub has shape (2n-1, n)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(n-1, k)
        for j in range(j_min, j_max+1):
            A_ub[k, j] = a[k-j]
    
    b_ub = np.full(2*n-1, M)
    
    # Bounds: g[j] >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        g = res.x
        return g
    else:
        return None

def lp_iterate(a, n_steps=50, blend_factor=0.3, time_limit=60):
    """
    Iterate the LP-based approach.
    """
    t0 = time.monotonic()
    current = a.astype(np.float64).copy()
    current_score = score(current)
    best = current.copy()
    best_score = current_score
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Normalize
        s = current.sum()
        if s < 1e-10:
            break
        current_norm = current / s
        
        # Compute M
        conv = fft_convolve(current_norm, current_norm)
        M = conv.max()
        
        # Solve LP
        g = lp_step(current_norm, M)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        
        # Rescale g to have same sum as current
        g = g * s / g_sum
        
        # Blend
        new_a = (1 - blend_factor) * current + blend_factor * g
        
        new_score = score(new_a)
        if new_score < best_score:
            best_score = new_score
            best = new_a.copy()
        
        # Accept if not much worse (allow some uphill for exploration)
        if new_score < current_score * 1.01:
            current = new_a
            current_score = new_score
        else:
            # Try smaller blend
            new_a2 = (1 - blend_factor/3) * current + (blend_factor/3) * g
            new_score2 = score(new_a2)
            if new_score2 < current_score:
                current = new_a2
                current_score = new_score2
            else:
                # Reduce M and retry
                pass
    
    return best, best_score

"""
Multi-engine optimization combining LP-based iteration, gradient descent with
softmax parameterization, and targeted local search. Exploits odd-symmetric
initializations and sparse LP solves to find low-autocorrelation sequences.
"""

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute the score: 2n * max(conv(a,a)) / (sum(a))^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)


def score_with_conv(a):
    """Return score and the convolution."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf, None
    c = fft_convolve(a, a)
    sc = 2.0 * n * c.max() / (s * s)
    return sc, c


def softmax_stable(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    xc = np.fft.irfft(W * np.conj(A), size)[:n]

    # Full gradient: d(2n*m/s^2)/da[j] = 2n/s^2 * (2*xc[j] - 2*m*a[j]/s)
    grad = (4.0 * n / (s * s)) * (xc - m_val * a / s)
    return grad, m_val, s


def lp_step_sparse(a, M):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (a*g)[k] <= M for all k=0..2n-2.
    Returns optimal g or None.
    """
    n = len(a)
    c_obj = -np.ones(n)

    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 30})
    if res.success:
        return res.x
    return None


def lp_iteration(a_init, time_limit, rng, max_iter=200):
    """
    Run LP-based iteration to minimize the score.
    """
    t0 = time.monotonic()
    current = a_init.astype(np.float64).copy()
    current_score = score_fast(current)
    best = current.copy()
    best_score = current_score
    n = len(current)

    no_improve = 0
    blend = 0.5

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        if current_score > 10.0:
            break

        s = current.sum()
        if s < 1e-10:
            break
        a_norm = current / s
        conv = fft_convolve(a_norm, a_norm)
        M = conv.max()

        g = lp_step_sparse(a_norm, M)
        if g is None:
            # Try tighter M
            g = lp_step_sparse(a_norm, M * 0.98)
            if g is None:
                break

        g_sum = g.sum()
        if g_sum < 1e-10:
            break

        g = g * s / g_sum

        # Try multiple blend factors
        best_new = None
        best_new_score = np.inf
        for bf in [blend, blend * 0.5, blend * 0.2, 0.1, 0.05]:
            new_a = (1 - bf) * current + bf * g
            ns = score_fast(new_a)
            if ns < best_new_score:
                best_new_score = ns
                best_new = new_a

        if best_new_score < best_score:
            best_score = best_new_score
            best = best_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            # Perturb to escape
            current = current * np.exp(rng.normal(0, 0.05, n))
            no_improve = 0
            blend = max(0.1, blend * 0.8)
        else:
            current = best_new
            current_score = best_new_score

    return best, best_score


def gradient_descent_theta(a_init, time_limit, rng):
    """
    Gradient descent on theta (log-softmax parameterization) with Adam.
    """
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    a0 = a_init.astype(np.float64).copy()
    a0 = np.clip(a0, 1e-6, None)
    theta = np.log(a0 / a0.sum() + 1e-10)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    step = 0
    t0 = time.monotonic()
    best_theta = theta.copy()
    best_score = score_fast(a_init)
    tau_start = 20.0 / n
    tau_end = 0.1 / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.7)

        a = softmax_stable(theta) * S
        grad_a, m_val, s = compute_gradient(a, tau)

        # Project gradient to theta space (accounting for softmax normalization)
        inner = np.dot(a, grad_a)
        g_theta = a * (grad_a - inner / S)

        step += 1
        m = beta1 * m + (1 - beta1) * g_theta
        v = beta2 * v + (1 - beta2) * (g_theta * g_theta)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        lr = 1.5 / np.sqrt(step)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if step % 100 == 0:
            a_check = softmax_stable(theta) * S
            sc = score_fast(a_check)
            if sc < best_score:
                best_score = sc
                best_theta = theta.copy()

    a_best = softmax_stable(best_theta) * S
    final_score = score_fast(a_best)
    return a_best, final_score


def local_polish(a_init, time_limit, rng):
    """
    Targeted local search with peak-reduction perturbations.
    """
    current = a_init.astype(np.float64).copy()
    current_score = score_fast(current)
    best = current.copy()
    best_score = current_score
    n = len(current)
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.9998

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.30:
            # Reduce top contributors to peak
            conv = fft_convolve(current, current)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = current[i] * current[j]
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = min(6, len(other))
                inc_idx = rng.choice(other, n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, n_inc))
        elif r < 0.50:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.02, width))
        elif r < 0.65:
            # Odd-symmetric nudge
            sym_part = 0.5 * (current - current[::-1])
            candidate = current + 0.1 * sym_part * np.exp(rng.normal(0, 0.01, n))
        elif r < 0.80:
            # Frequency-domain perturbation
            A = np.fft.rfft(current)
            A *= np.exp(rng.normal(0, 0.02, len(A)) + 1j * rng.normal(0, 0.01, len(A)))
            candidate = np.fft.irfft(A, n)
        else:
            # Small global multiplicative
            candidate = current * np.exp(rng.normal(0, 0.008, n))

        candidate = np.clip(candidate, 1e-6, 1000.0)
        new_score = score_fast(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            current = candidate
            current_score = new_score

        temp *= cooling

    return best, best_score


def make_odd_symmetric_init(n, rng):
    """Initialize with odd-symmetric perturbation of uniform (exploits structure)."""
    j = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    # Add several odd-symmetric harmonics
    for k in [1, 2, 3, 5]:
        amp = 0.3 / k
        a += amp * np.sin(2.0 * np.pi * k * (j + 0.5) / n)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_chirp_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    alpha = 0.3 + 0.3 * rng.random()
    a = 1.0 + alpha * np.cos(np.pi * t * t / (2.0 * n))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k, amp in [(1, 0.2), (3, 0.12), (5, 0.08), (7, 0.05), (11, 0.03)]:
        a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_linear_init(n, rng):
    """Try a linear ramp (exploits asymmetry)."""
    t = np.linspace(0.5, 1.5, n)
    a = t + 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining LP iteration, gradient descent, and local polish.
    Returns the best sequence found within the time budget.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 15.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
            s = score_fast(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_seq = start_seq.copy()
        except Exception:
            pass

    # Define search configurations: (n, list_of_init_types)
    configs = [
        (256, ['odd_sym', 'chirp', 'multitone', 'linear', 'provided']),
        (384, ['odd_sym', 'chirp', 'multitone']),
        (512, ['odd_sym', 'chirp']),
        (192, ['odd_sym', 'chirp', 'multitone']),
        (128, ['odd_sym', 'chirp']),
    ]

    def get_init(n, init_type, rng):
        if init_type == 'provided' and start_seq is not None:
            if len(start_seq) == n:
                return start_seq.copy()
            else:
                x_old = np.linspace(0, 1, len(start_seq))
                x_new = np.linspace(0, 1, n)
                return np.interp(x_new, x_old, start_seq)
        elif init_type == 'odd_sym':
            return make_odd_symmetric_init(n, rng)
        elif init_type == 'chirp':
            return make_chirp_init(n, rng)
        elif init_type == 'multitone':
            return make_multitone_init(n, rng)
        elif init_type == 'linear':
            return make_linear_init(n, rng)
        else:
            return make_odd_symmetric_init(n, rng)

    # Phase 1: LP-based optimization
    print("\n=== Phase 1: LP-based optimization ===")
    for n, init_types in configs:
        if time.monotonic() > deadline - 120:
            break
        remaining = deadline - time.monotonic()
        time_per_init = min(60.0, remaining / (len(init_types) + 1))

        for init_type in init_types:
            if time.monotonic() > deadline - 60:
                break
            a0 = get_init(n, init_type, rng)
            a0 = np.clip(a0, 0.01, 1000.0)
            init_score = score_fast(a0)
            if not np.isfinite(init_score):
                continue

            # Run LP iteration
            t0 = time.monotonic()
            a_out, lp_score = lp_iteration(a0, time_per_init, rng)
            elapsed = time.monotonic() - t0
            print(f"  LP: n={n}, init='{init_type}', score={lp_score:.6f} (was {init_score:.6f}), t={elapsed:.1f}s")

            if lp_score < best_score:
                best_score = lp_score
                best_seq = a_out.copy()
                print(f"    *** New best: {best_score:.6f} ***")

        if time.monotonic() > deadline - 60:
            break

    # Phase 2: Gradient-based refinement on best candidates
    print(f"\n=== Phase 2: Gradient refinement ===")
    remaining = deadline - time.monotonic()
    if remaining > 60 and best_seq is not None:
        # Refine the best sequence with gradient descent
        n_best = len(best_seq)
        grad_time = min(120.0, remaining * 0.4)
        print(f"  Refining n={n_best}, time={grad_time:.0f}s")
        a_grad, grad_score = gradient_descent_theta(best_seq, grad_time, rng)
        if grad_score < best_score:
            best_score = grad_score
            best_seq = a_grad.copy()
            print(f"    *** New best after grad: {best_score:.6f} ***")
        else:
            print(f"  Grad result: {grad_score:.6f} (best remains {best_score:.6f})")

        # Also try gradient from other inits at same n
        remaining = deadline - time.monotonic()
        if remaining > 30:
            for init_type in ['odd_sym', 'chirp']:
                if time.monotonic() > deadline - 30:
                    break
                a0 = get_init(n_best, init_type, rng)
                t_per = min(40.0, remaining * 0.3)
                a_g, sc = gradient_descent_theta(a0, t_per, rng)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_g.copy()
                    print(f"    *** New best (grad {init_type}): {best_score:.6f} ***")

    # Phase 3: Local search polish
    print(f"\n=== Phase 3: Local search polish ===")
    remaining = deadline - time.monotonic()
    if remaining > 20 and best_seq is not None:
        polish_time = min(100.0, remaining * 0.5)
        print(f"  Polishing, time={polish_time:.0f}s")
        a_pol, pol_score = local_polish(best_seq, polish_time, rng)
        if pol_score < best_score:
            best_score = pol_score
            best_seq = a_pol.copy()
            print(f"    *** New best after polish: {best_score:.6f} ***")
        else:
            print(f"  Polish result: {pol_score:.6f} (best remains {best_score:.6f})")

    # Phase 4: Final LP iteration on polished result
    remaining = deadline - time.monotonic()
    if remaining > 30 and best_seq is not None:
        print(f"\n=== Phase 4: Final LP polish ===")
        lp_time = min(60.0, remaining * 0.6)
        a_final, fin_score = lp_iteration(best_seq, lp_time, rng, max_iter=100)
        if fin_score < best_score:
            best_score = fin_score
            best_seq = a_final.copy()
            print(f"    *** New best after final LP: {best_score:.6f} ***")

    # Phase 5: One more local polish
    remaining = deadline - time.monotonic()
    if remaining > 20 and best_seq is not None:
        a_pol2, pol2_score = local_polish(best_seq, min(40.0, remaining * 0.5), rng)
        if pol2_score < best_score:
            best_score = pol2_score
            best_seq = a_pol2.copy()
            print(f"    *** Final best: {best_score:.6f} ***")

    # Verify with evaluate_sequence
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
