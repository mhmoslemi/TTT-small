```python
import numpy as np
import time

def _fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    L = 1
    while L < n + m - 1:
        L *= 2
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    result = np.fft.irfft(fa * fb, L)
    return result[:n + m - 1]

def _fft_self_convolve(a):
    """Compute a * a (self-convolution) using FFT."""
    n = len(a)
    L = 1
    while L < 2 * n - 1:
        L *= 2
    fa = np.fft.rfft(a, L)
    result = np.fft.irfft(fa * fa, L)
    return result[:2 * n - 1]

def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = _fft_self_convolve(a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def _soft_peak_gradient(p, tau, L=None):
    """
    Compute the gradient of the soft-peak objective using FFT.
    p: probability vector (sum=1, non-negative)
    tau: temperature parameter
    Returns: gradient vector of same length as p
    """
    n = len(p)
    # If L not provided, compute it
    if L is None:
        L = 1
        while L < 3 * n - 2:
            L *= 2
    
    # Step 1: Compute c = conv(p, p) (length 2n-1)
    fp = np.fft.rfft(p, L)
    c_full = np.fft.irfft(fp * fp, L)
    c = c_full[:2 * n - 1]
    
    # Step 2: Compute softmax weights
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w /= w.sum()
    
    # Step 3: Compute gradient = 2 * cross_corr(w, p)
    # cross_corr(w, p)[j] = sum_k w[k] * p[k-j]
    # = conv(w, p_flip)[n-1+j]
    p_flip = p[::-1]  # reverse p
    fw = np.fft.rfft(w, L)
    fpf = np.fft.rfft(p_flip, L)
    z = np.fft.irfft(fw * fpf, L)
    grad = 2.0 * z[n - 1 : 2 * n - 1]  # extract the relevant part
    
    return grad, c

def _optimize_fft(a0, n, deadline, tau_start=10.0, tau_end=200.0, 
                  lr_start=0.05, lr_end=0.001, n_iters=5000):
    """
    FFT-accelerated gradient descent to minimize 2n*max(conv(p,p)) 
    where p is a probability distribution.
    """
    p = np.clip(np.asarray(a0, dtype=np.float64), 0.0, None).copy()
    p = p / p.sum()  # normalize to sum=1
    
    # Precompute FFT size
    L = 1
    while L < 3 * n - 2:
        L *= 2
    
    best_p = p.copy()
    best_score = _compute_obj(best_p)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Anneal parameters
        frac = it / max(n_iters, 1)
        tau = tau_start + (tau_end - tau_start) * frac
        lr = lr_start * (1.0 + np.cos(np.pi * frac)) * 0.5
        lr = max(lr, lr_end)
        
        # Compute gradient
        grad, c = _soft_peak_gradient(p, tau, L)
        
        # Project out mean (maintain sum=1)
        grad -= grad.mean()
        
        # Gradient step
        p = p - lr * grad
        
        # Project: non-negative, normalize
        p = np.maximum(p, 0.0)
        S = p.sum()
        if S < 1e-12:
            p = np.ones(n) / n
        else:
            p = p / S
        
        # Track best
        if (it + 1) % 50 == 0:
            score = _compute_obj(p)
            if score < best_score:
                best_score = score
                best_p = p.copy()
    
    return best_p, best_score

def _lp_refine(f, M, n):
    """Solve LP: max sum(g) s.t. convolve(f,g) <= M, g >= 0."""
    from scipy.optimize import linprog
    if n > 1024:
        return None
    
    c_obj = -np.ones(n)
    
    # Build constraint matrix
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[k, j] = f[k - j]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None
```

```python
"""
FFT-accelerated optimization of non-negative sequences to minimize
2*n*max(convolve(a,a))/sum(a)^2.

Uses soft-peak gradient descent with FFT-accelerated convolution and
cross-correlation, combined with LP-based refinement (alternating
minimization) and multiple diverse initializations including chirp-based
constructs with naturally flat autocorrelation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(x):
    """Next power of 2 >= x."""
    x = max(1, x)
    return 1 << (x - 1).bit_length()


def _fft_self_conv(a, L=None):
    """Compute linear self-convolution a*a via FFT. Returns array of length 2n-1."""
    n = len(a)
    if L is None:
        L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    result = np.fft.irfft(fa * fa, L)
    return result[:2 * n - 1]


def _fft_conv2(a, b, L=None):
    """Compute linear convolution a*b via FFT. Returns array of length len(a)+len(b)-1."""
    na, nb = len(a), len(b)
    if L is None:
        L = _next_pow2(na + nb - 1)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    result = np.fft.irfft(fa * fb, L)
    return result[:na + nb - 1]


def _compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = _fft_self_conv(a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _soft_peak_grad(p, tau, L):
    """
    Compute gradient of soft-peak objective for probability vector p.
    
    Soft-peak: SP(c, tau) = (1/tau) * log(sum_k exp(tau * c[k]))
    Gradient: grad[j] = 2 * sum_k softmax(tau*c)[k] * p[k-j]
    
    Uses FFT for O(n log n) computation.
    Returns (grad, c) where c = conv(p, p).
    """
    n = len(p)
    # Self-convolution
    fp = np.fft.rfft(p, L)
    c_full = np.fft.irfft(fp * fp, L)
    c = c_full[:2 * n - 1]
    
    # Softmax weights (numerically stable)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[c.argmax()] = 1.0
    else:
        w /= w_sum
    
    # Gradient via cross-correlation: grad[j] = 2 * sum_k w[k] * p[k-j]
    # = 2 * conv(w, p_flip)[n-1+j]
    p_flip = p[::-1]
    fw = np.fft.rfft(w, L)
    fpf = np.fft.rfft(p_flip, L)
    z = np.fft.irfft(fw * fpf, L)
    grad = 2.0 * z[n - 1:2 * n - 1]
    
    return grad, c


def _hard_peak_grad(p, L=None):
    """
    Compute subgradient for hard max peak.
    Returns (grad, c, peak_lag).
    """
    n = len(p)
    c = _fft_self_conv(p)
    peak = int(c.argmax())
    
    # grad[j] = 2 * p[peak - j] for valid indices
    grad = np.zeros(n)
    lo = max(0, peak - n + 1)
    hi = min(n - 1, peak)
    for j in range(lo, hi + 1):
        grad[j] = 2.0 * p[peak - j]
    
    return grad, c, peak


def _normalize_prob(a):
    """Normalize to probability vector (sum=1, non-negative)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _optimize_fft_grad(p0, n, deadline, tau_start=5.0, tau_end=300.0,
                       lr_start=0.03, lr_end=0.0005, n_iters=10000,
                       use_soft=True):
    """
    Main FFT-accelerated gradient descent on probability vectors.
    Minimizes 2n * max(conv(p, p)) via soft-peak surrogate.
    """
    p = _normalize_prob(np.asarray(p0, dtype=np.float64).copy())
    L = _next_pow2(3 * n)  # For cross-correlation step
    
    best_p = p.copy()
    best_score = _compute_obj(p)
    c = _fft_self_conv(p)
    peak = float(c.max())
    
    # Track for adaptive tau
    soft_peak_val = peak
    it = 0
    while it < n_iters:
        if time.monotonic() > deadline:
            break
        
        frac = it / max(n_iters, 1)
        
        # Anneal: tau increases, lr decreases (cosine)
        tau = tau_start + (tau_end - tau_start) * frac ** 0.7
        lr = lr_start * 0.5 * (1.0 + np.cos(np.pi * frac))
        lr = max(lr, lr_end)
        
        if use_soft:
            grad, c = _soft_peak_grad(p, tau, L)
        else:
            grad, c, _ = _hard_peak_grad(p)
        
        # Project gradient (maintain sum=1)
        grad -= grad.mean()
        
        # Adaptive step: scale by current peak to handle magnitude
        peak_now = float(c.max())
        if peak_now > 1e-12:
            lr_eff = lr * peak_now  # Scale learning rate with peak magnitude
        
        # Gradient step
        p = p - lr_eff * grad
        
        # Project: non-negative, normalize
        p = _normalize_prob(p)
        
        it += 1
        
        # Periodic evaluation
        if it % 200 == 0:
            score = _compute_obj(p)
            if score < best_score:
                best_score = score
                best_p = p.copy()
                if it % 2000 == 0:
                    print(f"    iter={it}, score={best_score:.6f}, peak={float(_fft_self_conv(p).max()):.6f}")
    
    # Final check
    score = _compute_obj(p)
    if score < best_score:
        best_score = score
        best_p = p.copy()
    
    return best_p, best_score


def _optimizer_with_restarts(p0, n, deadline, n_restarts=3):
    """Run FFT gradient with periodic random restarts to escape local minima."""
    p = np.asarray(p0, dtype=np.float64).copy()
    best_p = p.copy()
    best_score = _compute_obj(p)
    rng_local = np.random.default_rng(12345 + n)
    
    per_restart = max(1, (int(deadline - time.monotonic()) - 5) // n_restarts)
    
    for r in range(n_restarts):
        if time.monotonic() > deadline - 5:
            break
        
        p_res, score = _optimize_fft_grad(
            p, n, time.monotonic() + per_restart,
            tau_start=5.0 + r * 20.0, tau_end=200.0 + r * 100.0,
            lr_start=0.03 / (1 + 0.3 * r), lr_end=0.0003,
            n_iters=5000
        )
        if score < best_score:
            best_score = score
            best_p = p_res.copy()
        
        # Perturb for next restart
        noise = rng_local.random(n) * 0.1
        p = 0.9 * best_p + 0.1 * noise
        p = _normalize_prob(p)
    
    return best_p, best_score


def _lp_solve(f, M, n):
    """
    Solve LP: maximize sum(g) subject to conv(f, g) <= M (elementwise), g >= 0.
    Returns g or None on failure.
    """
    if n > 2048:
        return None
    
    c_obj = -np.ones(n)
    
    # Build A matrix: A[k, j] = f[k-j] if 0<=k-j<n, else 0
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[k, j] = f[k - j]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _lp_refine(f, n, deadline, max_iters=100):
    """
    LP-based refinement (SOTA approach).
    Alternates: solve LP for g given f, blend, update M.
    """
    f = _normalize_prob(f.astype(np.float64).copy())
    best_p = f.copy()
    best_score = _compute_obj(f)
    
    c = _fft_self_conv(f)
    M = float(c.max())
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = _lp_solve(f, M, n)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum  # normalize to probability
        
        # Try multiple blend ratios
        improved = False
        for t in [0.6, 0.4, 0.25, 0.15, 0.08, 0.04]:
            new_p = (1.0 - t) * f + t * g
            new_p = _normalize_prob(new_p)
            s = _compute_obj(new_p)
            if s < best_score:
                f = new_p
                best_p = f.copy()
                best_score = s
                improved = True
                break
        
        if not improved:
            # Try small gradient nudge to escape
            grad, c_new = _soft_peak_grad(f, 50.0, _next_pow2(3 * n))
            f = f - 0.005 * (grad - grad.mean())
            f = _normalize_prob(f)
            s = _compute_obj(f)
            if s < best_score:
                best_p = f.copy()
                best_score = s
        
        # Update M (slightly tighten to force progress)
        c = _fft_self_conv(f)
        M = float(c.max()) * 0.999  # Slight tightening
    
    return best_p, best_score


def _make_chirp_init(n, freq=1.0):
    """Create chirp-based initialization with good autocorrelation properties."""
    i = np.arange(n, dtype=np.float64)
    # Quadratic chirp: |cos(pi * i^2 / n)| has flat-ish autocorrelation
    a = 0.5 + 0.5 * np.cos(np.pi * i * i * freq / n)
    return np.maximum(a, 0.001)


def _make_symmetric_init(n, freq=1.0):
    """Create symmetric initialization (a[j] = a[n-1-j])."""
    half = n // 2
    i = np.arange(half, dtype=np.float64)
    a_half = 0.5 + 0.5 * np.cos(np.pi * i / half)
    if n % 2 == 0:
        a = np.concatenate([a_half, a_half[::-1]])
    else:
        a = np.concatenate([a_half, a_half[::-1][1:]])
    return np.maximum(a, 0.001)


def _make_random_init(n, seed):
    """Create a random initialization with some structure."""
    rng = np.random.default_rng(seed)
    i = np.arange(n, dtype=np.float64)
    # Mix of random and structured
    a = rng.random(n) * 0.3 + 0.35 + 0.3 * np.cos(2 * np.pi * rng.random(5) * i / n)
    return np.maximum(a, 0.01)


def _try_global_sequence(deadline, best_score_tracker):
    """Try optimizing the provided global sequence."""
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size < 8 or not np.all(np.isfinite(gs)):
            return None, float('inf')
    except NameError:
        return None, float('inf')
    
    best_p = None
    best_s = float('inf')
    
    for n in [min(len(gs), 512), min(len(gs), 256)]:
        if time.monotonic() > deadline:
            break
        p0 = gs[:n].copy()
        p0 = _normalize_prob(p0)
        s0 = _compute_obj(p0)
        print(f"  Global seq n={n}: initial score={s0:.6f}")
        
        p, s = _optimizer_with_restarts(p0, n, deadline, n_restarts=2)
        if s < best_s:
            best_s = s
            best_p = p
    
    return best_p, best_s


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function: finds sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    
    Strategy:
    1. Generate diverse initializations (chirps, global seq, random)
    2. FFT-accelerated soft-peak gradient descent with annealing
    3. LP-based refinement (alternating minimization)
    4. Final polishing
    
    Returns best sequence found within time budget.
    """
    t_start = time.monotonic()
    hard_deadline = t_start + max(0.0, float(budget_s) - 8.0)
    
    best_seq = None
    best_score = float('inf')
    
    def update_best(seq, label):
        nonlocal best_seq, best_score
        if seq is None:
            return
        s = _compute_obj(seq)
        if s < best_score:
            best_score = s
            best_seq = seq.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{elapsed:.0f}s] New best ({label}): {best_score:.6f} (n={len(seq)})")
    
    print(f"=" * 60)
    print(f"Starting optimization. Budget: {budget_s}s")
    print(f"=" * 60)
    
    # --- Stage 1: Try global sequence (if available) ---
    print("\n[Stage 1] Global sequence optimization...")
    stage1_deadline = min(t_start + 300, hard_deadline - 400)
    p_glob, s_glob = _try_global_sequence(stage1_deadline, best_score)
    if p_glob is not None:
        update_best(p_glob, "global")
    
    # --- Stage 2: FFT gradient descent with diverse inits ---
    print(f"\n[Stage 2] FFT gradient descent (remaining: {hard_deadline - time.monotonic():.0f}s)")
    
    # Generate diverse initializations
    inits = []
    for n in [256, 512, 1024]:
        inits.append(('chirp', n, _make_chirp_init(n)))
        inits.append(('chirp2', n, _make_chirp_init(n, freq=0.7)))
        inits.append(('sym', n, _make_symmetric_init(n)))
        inits.append(('rand', n, _make_random_init(n, seed + n)))
    
    # If we have a global, add trimmed versions
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 256 and np.all(np.isfinite(gs)):
            for n in [256, 512]:
                if n <= len(gs):
                    inits.append(('glob_trim', n, _normalize_prob(gs[:n])))
    except NameError:
        pass
    
    # Evaluate and sort by initial score
    scored_inits = []
    for label, n, a in inits:
        a = _normalize_prob(a)
        s = _compute_obj(a)
        scored_inits.append((s, label, n, a))
        print(f"  Init '{label}' n={n}: {s:.6f}")
    
    scored_inits.sort(key=lambda x: x[0])
    
    # Optimize top candidates
    n_optimize = 6  # number of inits to fully optimize
    time_per_init = max(30, (hard_deadline - 300 - time.monotonic()) / max(1, n_optimize))
    
    for idx, (s_init, label, n, a) in enumerate(scored_inits[:n_optimize]):
        if time.monotonic() > hard_deadline - 250:
            break
        
        t0 = time.monotonic()
        alloc = min(time_per_init, hard_deadline - 250 - time.monotonic())
        if alloc < 15:
            break
        
        print(f"  Optimizing '{label}' n={n} (init={s_init:.6f}, alloc={alloc:.0f}s)...")
        
        # Phase A: Soft-peak gradient
        p, s = _optimize_fft_grad(a, n, t0 + alloc * 0.6,
                                   tau_start=5.0, tau_end=200.0,
                                   lr_start=0.04, lr_end=0.001,
                                   n_iters=8000)
        print(f"    Phase A done: {s:.6f} ({time.monotonic()-t0:.1f}s)")
        
        # Phase B: Continue with higher tau
        p2, s2 = _optimize_fft_grad(p, n, t0 + alloc,
                                     tau_start=100.0, tau_end=500.0,
                                     lr_start=0.015, lr_end=0.0003,
                                     n_iters=5000)
        if s2 < s:
            p, s = p2, s2
            print(f"    Phase B done: {s:.6f}")
        
        elapsed = time.monotonic() - t0
        print(f"  '{label}' n={n} final: {s:.6f} ({elapsed:.1f}s)")
        update_best(p, label)
    
    # --- Stage 3: LP refinement on best candidate ---
    print(f"\n[Stage 3] LP refinement (remaining: {hard_deadline - time.monotonic():.0f}s)")
    
    if best_seq is not None and time.monotonic() < hard_deadline - 30:
        n = len(best_seq)
        if n <= 1024:
            lp_deadline = hard_deadline - 20
            p_lp, s_lp = _lp_refine(best_seq, n, lp_deadline, max_iters=50)
            if s_lp < best_score:
                print(f"  LP improved: {best_score:.6f} -> {s_lp:.6f}")
                best_score = s_lp
                best_seq = p_lp.copy()
            else:
                print(f"  LP no improvement (lp_score={s_lp:.6f})")
        else:
            print(f"  Skipping LP (n={n} too large)")
    
    # --- Stage 4: Final polish ---
    print(f"\n[Stage 4] Final polish (remaining: {hard_deadline - time.monotonic():.0f}s)")
    
    if best_seq is not None and time.monotonic() < hard_deadline - 5:
        n = len(best_seq)
        p_final, s_final = _optimize_fft_grad(
            best_seq, n, hard_deadline - 2,
            tau_start=200.0, tau_end=800.0,
            lr_start=0.008, lr_end=0.0001,
            n_iters=20000
        )
        if s_final < best_score:
            print(f"  Polish improved: {best_score:.6f} -> {s_final:.6f}")
            best_score = s_final
            best_seq = p_final.copy()
        else:
            print(f"  Polish no improvement")
    
    # --- Final validation ---
    if best_seq is not None:
        final_score = _compute_obj(best_seq)
        print(f"\n{'=' * 60}")
        print(f"FINAL: n={len(best_seq)}, score={final_score:.6f}")
        print(f"{'=' * 60}")
        
        # Validate with evaluate_sequence
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence validation: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        
        return [float(x) for x in best_seq]
    
    # Fallback
    print("FALLBACK: returning uniform sequence")
    return [1.0] * 512
```
