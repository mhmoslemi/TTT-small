def solve_lp_direction(a, M, top_k=200):
    n = len(a)
    c = fft_convolve_self(a)
    active_lags = np.argsort(c)[::-1][:top_k]

    # vector [A[k,0], A[k,1], ..., A[k,n-1]] where A[k,j] = a[k-j] if valid
    n_cons = len(active_lags)
    rows = np.zeros(n_cons * n, dtype=np.int32)  # worst case
    cols = np.zeros(n_cons * n, dtype=np.int32)
    vals = np.zeros(n_cons * n, dtype=np.float64)
    count = 0
    
    for idx, k in enumerate(active_lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows[count] = idx
            cols[count] = j
            vals[count] = a[k - j]
            count += 1
    
    A = csr_matrix((vals[:count], (rows[:count], cols[:count])), shape=(n_cons, n))
    
    # LP: minimize -1^T g subject to A @ g <= M, g >= 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(n_cons, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy:
  * LP-based direction finding: at each iteration, solve an LP to find the
    best direction g that keeps cross-convolution (a*g) bounded by current max.
  * Combine with adaptive subgradient method for fine polishing.
  * Multiple initializations (structured + random) at multiple lengths.
  * Time-managed exploration and polishing phases.

import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def cross_convolve(a, g):
    """Full cross-convolution of a and g via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    G = fft(g, fsize)
    c = np.real(ifft(A * G))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n, S=None):
    """Clamp to [0,1000] and rescale to sum = S (default sqrt(2n))."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if S is None:
        S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)


def score_normalized(a):
    """Compute the normalized score (objective value) for a sequence."""
    n = len(a)
    c = fft_convolve_self(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s ** 2)


def solve_lp_direction(a, M, top_k):
    """
    Solve LP: max 1^T g s.t. (a*g)[k] <= M for k in active_top_k, g >= 0.
    Returns g or None if LP fails.
    """
    n = len(a)
    c = fft_convolve_self(a)
    active = np.argsort(c)[::-1][:top_k]
    
    n_cons = len(active)
    # Build sparse constraint matrix
    # For each active lag k, constraint is sum_j a[k-j]*g[j] <= M
    # where a[k-j] is defined only for 0 <= k-j < n
    row_data = []
    col_data = []
    val_data = []
    
    for r, k in enumerate(active):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        row_data.extend([r] * len(js))
        col_data.extend(js.tolist())
        val_data.extend(a[k - js].tolist())
    
    if not val_data:
        return np.ones(n) / n
    
    A_mat = csr_matrix((val_data, (row_data, col_data)), shape=(n_cons, n))
    
    c_obj = -np.ones(n)
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 10})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def run_lp_iteration(a, n, S, max_time, top_k, step_size, rng, 
                     lp_every=1, min_improve=1e-8):
    """
    LP-based iteration method.
    At each LP step, solve for direction g and move towards it.
    Between LP steps, use subgradient descent.
    """
    a = safe_normalize(a, n, S)
    best_a = a.copy()
    best_score = score_normalized(a)
    start = time.monotonic()
    no_improve_count = 0
    
    iteration = 0
    while time.monotonic() - start < max_time:
        iteration += 1
        c = fft_convolve_self(a)
        M = c.max()
        current_score = 2.0 * n * M / (a.sum() ** 2)
        
        if current_score < best_score - min_improve:
            best_score = current_score
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        if no_improve_count > 100:
            # Add perturbation
            a = project_simplex(a + rng.normal(0, 0.01 * S / np.sqrt(n), n), S)
            no_improve_count = 0
            continue
        
        # Every lp_every iterations, solve the LP
        if iteration % lp_every == 0:
            g = solve_lp_direction(a, M, top_k)
            if g is not None:
                g_sum = g.sum()
                if g_sum > 1e-10:
                    g = g * (S / g_sum)
                    # Blend
                    t = step_size
                    a_new = (1 - t) * a + t * g
                    a = project_simplex(a_new, S)
                    continue
            # LP failed or g is zero, fall through to subgradient
        
        # Subgradient step
        c = fft_convolve_self(a)
        top_indices = np.argsort(c)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_indices:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm
        
        lr = 0.1 * S / (n ** 0.5 * (1 + 0.01 * iteration))
        a = project_simplex(a - lr * grad, S)
        
        # Periodic perturbation
        if iteration % 500 == 0:
            a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
    
    return best_a, best_score


def run_adam_fine(a, n, S, max_time, rng, lr_init, top_k):
    """Fine Adam polishing with better subgradient."""
    a = safe_normalize(a, n, S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = score_normalized(a)
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_score = 2.0 * n * c.max() / (a.sum() ** 2)
        
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        
        # Use weighted subgradient (exponential weighting by value)
        top_indices = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_indices]
        # Exponential weighting
        max_v = top_vals[0]
        weights = np.exp((top_vals - max_v) / (max_v * 0.1 + 1e-15))
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for w_val, k in zip(weights, top_indices):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi+1] += w_val * 2.0 * a[k-hi:k-lo+1][::-1]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        lr = lr_init * (0.99 ** (t / 100.0))
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)
        
        if t % 2000 == 0:
            a = project_simplex(a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)
    
    return best_a, best_score

    vals_arr = np.array(all_vals, dtype=np.float64)
    
    return csr_matrix((vals_arr, (rows_arr, cols_arr)), shape=(n_cons, n))

"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.

Strategy:
  * LP-based direction finding (from SOTA): at each iteration, solve an LP to
    find the best direction g maximizing sum(g) subject to (a*g)[k] <= M
    (current peak) and g >= 0. Move towards g to reduce the peak.
  * Adaptive subgradient method with exponential weighting for fine polishing.
  * Multiple diverse initializations (structured + random) at multiple lengths.
  * Time-managed exploration and polishing phases with safety margins.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve_self(a):
    """Full self-convolution of a (length 2*n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def project_simplex(v, S):
    """Euclidean projection of v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def safe_normalize(a, n, S=None):
    """Clamp to [0,1000] and rescale to sum = S (default sqrt(2n))."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if S is None:
        S = np.sqrt(2.0 * float(n))
    s = a.sum()
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a * (S / s)


def compute_score(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_convolve_self(a)
    return 2.0 * n * c.max() / (s ** 2)


def build_conv_matrix_sparse(a, lags):
    """Build sparse convolution matrix rows for given lags."""
    n = len(a)
    lags_arr = np.asarray(lags, dtype=np.int64)
    n_cons = len(lags_arr)
    if n_cons == 0:
        return csr_matrix((0, n))

    rows_list = []
    cols_list = []
    vals_list = []

    for r, k in enumerate(lags_arr):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        js = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(js), r, dtype=np.int32))
        cols_list.append(js.astype(np.int32))
        vals_list.append(a[k - js])

    if not vals_list:
        return csr_matrix((n_cons, n))

    rows_arr = np.concatenate(rows_list)
    cols_arr = np.concatenate(cols_list)
    vals_arr = np.concatenate(vals_list)

    return csr_matrix((vals_arr, (rows_arr, cols_arr)), shape=(n_cons, n))


def solve_lp_direction(a, M, top_k, time_limit=8.0):
    """
    Solve LP: max 1^T g s.t. (a*g)[k] <= M for k in top_k active lags, g >= 0.
    Returns normalized g or None if LP fails.
    """
    n = len(a)
    c = fft_convolve_self(a)
    active = np.argsort(c)[::-1][:top_k]

    A_mat = build_conv_matrix_sparse(a, active)
    n_cons = len(active)

    c_obj = -np.ones(n)
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.clip(res.x, 0, 1000)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def compute_subgradient_weighted(a, c, top_k, temperature=0.1):
    """Compute exponentially-weighted subgradient of max(c)."""
    n = len(a)
    top_indices = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_indices]
    max_v = top_vals[0]
    if max_v <= 1e-15:
        return np.zeros(n)
    weights = np.exp((top_vals - max_v) / (max_v * temperature + 1e-15))
    weights /= weights.sum()

    grad = np.zeros(n)
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w_val * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def run_lp_method(a_init, n, S, max_time, top_k, step_size, rng,
                  lp_every=3, lp_time_limit=8.0):
    """
    LP-based iteration: alternate between LP direction and subgradient steps.
    Returns (best_a, best_score).
    """
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_score = compute_score(a)
    start = time.monotonic()
    iteration = 0
    no_improve = 0

    while time.monotonic() - start < max_time:
        iteration += 1
        c = fft_convolve_self(a)
        M = c.max()
        cur_score = 2.0 * n * M / (a.sum() ** 2)

        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            a = project_simplex(
                a + rng.normal(0, 0.02 * S / np.sqrt(n), n), S)
            no_improve = 50
            continue

        time_left = max_time - (time.monotonic() - start)
        if time_left < 2.0:
            break

        if iteration % lp_every == 0:
            g = solve_lp_direction(a, M, top_k,
                                   time_limit=min(lp_time_limit, time_left - 1.0))
            if g is not None:
                g = g * S
                t_blend = min(step_size, 0.5)
                a = project_simplex((1.0 - t_blend) * a + t_blend * g, S)
                continue

        # Subgradient step
        grad = compute_subgradient_weighted(a, c, top_k, temperature=0.05)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm
        lr = 0.15 * S / (np.sqrt(n) * (1.0 + 0.05 * iteration))
        a = project_simplex(a - lr * grad, S)

    return best_a, best_score


def run_adam_polish(a_init, n, S, max_time, rng, lr_init, top_k):
    """Adam optimizer with weighted subgradient for fine polishing."""
    a = safe_normalize(a_init, n, S)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = compute_score(a)
    start = time.monotonic()

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        grad = compute_subgradient_weighted(a, c, top_k, temperature=0.03)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_init * (0.995 ** (t / 50.0))
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps), S)

        if t % 1000 == 0:
            a = project_simplex(
                a + rng.normal(0, 0.003 * S / np.sqrt(n), n), S)

    return best_a, best_score


def run_mirror_descent(a_init, n, S, max_time, rng, eta_init, top_k):
    """Mirror descent / multiplicative weights respecting non-negativity."""
    a = safe_normalize(a_init, n, S)
    best_a = a.copy()
    best_score = compute_score(a)
    start = time.monotonic()
    t = 0

    while time.monotonic() - start < max_time:
        t += 1
        c = fft_convolve_self(a)
        cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        grad = compute_subgradient_weighted(a, c, top_k, temperature=0.05)
        gmax = np.abs(grad).max()
        if gmax < 1e-14:
            break
        grad_normed = grad / gmax

        eta = eta_init * max(0.01, 1.0 / (1.0 + 0.003 * t))
        a_new = a * np.exp(-eta * grad_normed)
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s <= 1e-15:
            a = np.full(n, S / n)
        else:
            a = a_new * (S / s)

    return best_a, best_score

        t_2b = polish_budget * 0.25
        if time.monotonic() < deadline - 5.0 - t_2b:
            a_2b, s_2b = run_mirror_descent(
                best_a, n, S, max(1.0, t_2b), rng,
                eta_init=0.2, top_k=20)
            try_update(a_2b, s_2b, "Polish Mirror")

        # Step 2c: Adam fine polish
        t_2c = max(5.0, polish_budget - t_2a - t_2b - 3.0)
        if time.monotonic() < deadline - 5.0 - t_2c:
            a_2c, s_2c = run_adam_polish(
                best_a, n, S, max(1.0, t_2c), rng,
                lr_init=0.1 * S / n, top_k=50)
            try_update(a_2c, s_2c, "Polish Adam")

        # Step 2d: One more LP round if time remains
        if time.monotonic() < deadline - 5.0:
            t_2d = deadline - 5.0 - time.monotonic() - 1.0
            if t_2d > 5.0:
                a_2d, s_2d = run_lp_method(
                    best_a, n, S, t_2d,
                    top_k=300, step_size=0.1, rng=rng,
                    lp_every=1, lp_time_limit=12.0)
                try_update(a_2d, s_2d, "Polish LP2")

    # === Phase 3: Final verification ===
    if best_a is None:
        print("WARNING: No valid sequence found, using fallback")
        best_a = np.ones(4096, dtype=np.float64)

    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:
        best_a = np.ones(4096, dtype=np.float64)

    # Verify with the authoritative evaluate_sequence
    final_score = evaluate_sequence(best_a.tolist())
    print("Final: estimated=%.6f, verified=%.6f, n=%d"
          % (best_score, final_score, len(best_a)))

    if not np.isfinite(final_score):
        print("WARNING: evaluation returned inf, using uniform fallback")
        best_a = np.ones(4096, dtype=np.float64)
        final_score = evaluate_sequence(best_a.tolist())

    return [float(x) for x in best_a]
