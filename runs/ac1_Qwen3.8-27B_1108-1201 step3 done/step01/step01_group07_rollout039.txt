    n = len(a)
    a = a.copy()

        conv = fast_convolve(a)

        M = conv.max()

                contributions[i] = a[i] * a[j]

        # Elements with low contribution are "safe" to increase
        dangerous_idx = np.where(contributions > contributions.mean())[0]
        safe_idx = np.where(contributions <= 0.5 * contributions.mean())[0]
        
        if len(dangerous_idx) == 0 or len(safe_idx) == 0:
            break

        best_o = best_obj

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using iterative LP-based optimization
with golden-section line search and targeted local refinement.
The LP step finds a non-negative direction g maximizing sum(g) subject to
conv(f,g) <= current_peak, providing a principled descent direction.
Multiple sequence lengths and diverse initializations are explored.

import numpy as np

import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    """Objective: 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def solve_lp(f, M):
    """Find g >= 0 maximizing sum(g) subject to conv(f,g)[k] <= M for all k."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = f[:end - j]
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()

def make_initializations(n, rng):
    """Generate diverse initializations for sequence of length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    for sf in [0.2, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1))) if n > 1 else np.ones(n)
    hann = np.maximum(hann, 0.0)
    inits['hann'] = hann / hann.sum()

    triangle = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = triangle / triangle.sum()

    r = rng.dirichlet(np.ones(n) * 0.3)
    inits['random'] = r

    edges[-m:] = 1.0
    inits['edges'] = edges / edges.sum()

    parent = get_parent_clipped(n)

            # LP step
            lp_time = min(remaining * 0.5, 120.0)
            a_opt, obj_opt = lp_optimize(best_sequence, time_limit=lp_time)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish LP: score={score:.6f}")

            # Local search step
            remaining = polish_deadline - time.monotonic()
            if remaining < 10:
                break
            ls_time = min(remaining * 0.5, 60.0)
            a_ls, obj_ls = local_search(best_sequence, time_limit=ls_time)
            score = float(evaluate_sequence(a_ls.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_ls.copy()
                print(f"  Polish LS: score={score:.6f}")

    # Final evaluation
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(best_sequence)}")
    return [float(x) for x in best_sequence]
