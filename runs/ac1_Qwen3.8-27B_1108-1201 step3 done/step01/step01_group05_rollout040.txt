def _fft_len(n, factor=2):
    return int(next_fast_len(max(1, factor * int(n) - 1)))


def conv_sq(p):

    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = _fft_len(n, 2)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def internal_score(p):

def project_simplex(v):

    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):

    u = np.sort(v)[::-1]

    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)

    s = float(out.sum())

def solve_lp(p, M):

    # Row k (0 to 2n-2), column j (0 to n-1):

def gradient_surrogate(p, T_factor=0.1):
    """Compute gradient of smoothed max surrogate."""
    n = int(p.size)
    c = conv_sq(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax <= 0:
        return np.zeros(n), 0.0
    
    T = max(T_factor * cmax, 1e-300)
    diff = (c - cmax) / T
    e = np.exp(diff)
    raw = float(e.sum())
    if not np.isfinite(raw) or raw <= 0:
        e = np.zeros_like(c)
        e[int(np.argmax(c))] = 1.0
        raw = 1.0
    w = e / raw
    
    # Gradient: grad_j = 2 * sum_k w_k * p_{k-j} (appropriate indexing)
    L = _fft_len(n, 3)
    Fp = fft(p, L)
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))
    
    if n == 1:
        G = h[:1]

        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp
    g = 2.0 * np.maximum(G, 0.0)
    return g, cmax

"""
LP-driven optimization for minimum peak self-convolution of non-negative sequences.

Strategy:
1. Global shape search via differential evolution on Fourier+taper parameter space
2. Iterative best-response LP with adaptive constraint tightening (primary local method)
3. Smoothed-max gradient refinement for final polish
4. Multi-start with diverse initial shapes, keeping global best
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog, minimize, differential_evolution
from scipy.sparse import csr_matrix


def _fft_len(n, factor=2):
    return int(next_fast_len(max(1, factor * int(n) - 1)))


def conv_sq(p):
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = _fft_len(n, 2)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def internal_score(p):
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = conv_sq(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (s * s))


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or p.size == 0 or s <= 1e-12:
        n = max(1, int(p.size))
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s


def fourier_shape(params, n):
    """Build a positive sequence from Fourier modes + endpoint taper."""
    x = np.arange(n, dtype=np.float64) / n
    K = 12
    log_p = np.zeros(n, dtype=np.float64)
    log_p += params[0]
    for k in range(1, K + 1):
        log_p += params[2 * k] * np.cos(2 * np.pi * k * x) + params[2 * k + 1] * np.sin(2 * np.pi * k * x)
    taper_L = params[2 * (K + 1)]
    taper_R = params[2 * (K + 1) + 1]
    eps = 1e-4
    log_p += taper_L * np.log(x + eps) + taper_R * np.log(1.0 - x + eps)
    if not np.all(np.isfinite(log_p)):
        return np.full(n, 1.0 / n, dtype=np.float64)
    log_p -= np.max(log_p)
    p = np.exp(log_p)
    return safe_normalize(p)


def fourier_objective(params, n):
    try:
        p = fourier_shape(params, n)
        return internal_score(p)
    except Exception:
        return 1e10


def solve_best_response_lp(p, M):
    """Solve: max sum(g) s.t. (p*g)_k <= M for all k, g >= 0."""
    n = int(p.size)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        jmin = max(0, k - (n - 1))
        jmax = min(n - 1, k)
        if jmin > jmax:
            continue
        js = np.arange(jmin, jmax + 1)
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        data.extend(p[k - js].tolist())
    if not data:
        return None
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0.0, None)] * n,
                  method="highs", options={"disp": False, "presolve": True})
    if res is None or res.status not in (0, 2):
        return None
    g = res.x
    if g is None:
        return None
    g = np.asarray(g, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-12:
        return None
    return g / s


def lp_step(p, tighten_factors=(1.0, 0.97, 0.94, 0.90, 0.85)):
    """Try LP with various constraint tightenings, return best g."""
    c = conv_sq(p)
    M = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(M) or M <= 0:
        return None, np.inf, np.inf
    sc_p = float(2.0 * int(p.size) * M)
    best_g = None
    best_sc = sc_p
    for tf in tighten_factors:
        M_t = M * tf
        if M_t <= 0:
            continue
        try:
            g = solve_best_response_lp(p, M_t)
        except Exception:
            continue
        if g is None:
            continue
        sc_g = internal_score(g)
        if np.isfinite(sc_g) and sc_g < best_sc:
            best_sc = sc_g
            best_g = g
    return best_g, best_sc, sc_p


def gradient_of_smoothed_max(p, T_factor=0.15):
    """Gradient of T * logsumexp(conv(p,p)/T) w.r.t. p."""
    n = int(p.size)
    c = conv_sq(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax <= 0:
        return np.zeros(n, dtype=np.float64), 0.0
    T = max(T_factor * cmax, 1e-300)
    diff = (c - cmax) / T
    e = np.exp(diff)
    raw = float(e.sum())
    if not np.isfinite(raw) or raw <= 0:
        e = np.zeros_like(c)
        e[np.argmax(c)] = 1.0
        raw = 1.0
    w = e / raw
    L = _fft_len(n, 3)
    Fp = fft(p, L)
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))
    if n == 1:
        G = h[:1]
    else:
        G = h[2 * n - 2:n - 2:-1]
    G = np.asarray(G, dtype=np.float64)
    if G.size != n:
        tmp = np.zeros(n)
        m = min(n, G.size)
        tmp[:m] = G[:m]
        G = tmp
    return 2.0 * np.maximum(G, 0.0), cmax


def make_diverse_shapes(n, rng):
    """Generate a diverse set of initial shapes."""
    shapes = []
    x = np.linspace(0.0, 1.0, n)
    shapes.append(np.full(n, 1.0 / n))
    for alpha in (-2.0, -1.0, -0.5, 0.5, 1.0, 2.0, 3.0, 5.0):
        p = np.power(np.maximum(x, 1e-6), alpha)
        shapes.append(safe_normalize(p))
        shapes.append(safe_normalize(np.power(np.maximum(1.0 - x, 1e-6), alpha)))
    for a in (0.3, 0.7, 1.5, 3.0, 6.0, 12.0):
        shapes.append(safe_normalize(1.0 + a * (2.0 * x - 1.0) ** 2))
        p = 1.0 - a * (2.0 * x - 1.0) ** 2
        shapes.append(safe_normalize(np.maximum(p, 0.01)))
    for mu, sig in [(0.25, 0.05), (0.3, 0.08), (0.5, 0.1), (0.7, 0.05), (0.75, 0.08)]:
        p = np.exp(-((x - mu) ** 2) / (2 * sig ** 2)) + np.exp(-((x + mu - 1) ** 2) / (2 * sig ** 2))
        shapes.append(safe_normalize(p))
    for f in (2, 3, 4, 5, 7):
        for amp in (0.3, 0.5, 0.7):
            p = 1.0 + amp * np.sin(2 * np.pi * f * x)
            shapes.append(safe_normalize(np.maximum(p, 0.005)))
    for _ in range(15):
        f1 = int(rng.integers(1, n // 40 + 1))
        f2 = int(rng.integers(1, n // 15 + 1))
        p = 1.0
        p += 0.4 * rng.uniform(0.3, 1.0) * np.sin(2 * np.pi * f1 * x + rng.uniform(0, 2 * np.pi))
        p += 0.2 * rng.uniform(0.3, 1.0) * np.sin(2 * np.pi * f2 * x + rng.uniform(0, 2 * np.pi))
        shapes.append(safe_normalize(np.maximum(p, 0.005)))
    return shapes


def lp_iterate(p0, time_limit, rng, max_iters=500):
    """Iterative LP optimization with line search."""
    p = safe_normalize(np.asarray(p0, dtype=np.float64))
    n = int(p.size)
    best_p = p.copy()
    best_score = internal_score(p)
    current_score = best_score
    end = time.monotonic() + max(0.0, float(time_limit))
    no_improve = 0
    tighten_seq = [(1.0, 0.97, 0.94, 0.90), (1.0, 0.95, 0.90, 0.85), (1.0, 0.95, 0.88, 0.82)]

    for it in range(max_iters):
        if time.monotonic() > end - 3:
            break
        phase = it // 100
        factors = tighten_seq[min(phase, len(tighten_seq) - 1)]
        if no_improve > 60:
            factors = (1.0, 0.98, 0.96, 0.94)

        g, sc_g, sc_p = lp_step(p, factors)
        if g is None:
            no_improve += 5
            if no_improve > 200:
                break
            continue

        improved = False
        for t in (1.0, 0.9, 0.8, 0.7, 0.6, 0.4, 0.2, 0.1):
            q = project_simplex((1.0 - t) * p + t * g)
            sc_q = internal_score(q)
            if np.isfinite(sc_q) and sc_q < current_score:
                if sc_q < best_score:
                    best_score = sc_q
                    best_p = q.copy()
                current_score = sc_q
                p = q.copy()
                improved = True
                break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 150:
                p = project_simplex(0.6 * best_p + 0.4 * p + rng.normal(0, 2.0 / n, n))
                current_score = internal_score(p)
                no_improve = 0
            if no_improve > 400:
                break

        if it % 50 == 0:
            t_elapsed = time.monotonic() - (end - time_limit)
            print("    LP iter %d: best=%.10f cur=%.10f no_improve=%d (%.0fs)" %
                  (it, best_score, current_score, no_improve, t_elapsed), flush=True)

    return best_p, best_score


def gradient_refine(p0, time_limit, rng):
    """Gradient-based local refinement with periodic LP calls."""
    p = project_simplex(np.asarray(p0, dtype=np.float64))
    n = int(p.size)
    best_p = p.copy()
    best_score = internal_score(p)
    current_score = best_score
    end = time.monotonic() + max(0.0, float(time_limit))
    T_factor = 0.3
    no_improve = 0
    it = 0

    while time.monotonic() < end - 5:
        it += 1
        if it % 40 == 0:
            remaining = end - time.monotonic()
            if remaining > 8:
                g, sc_g, _ = lp_step(p, (1.0, 0.95, 0.9))
                if g is not None:
                    for t in (1.0, 0.7, 0.5, 0.3):
                        q = project_simplex((1 - t) * p + t * g)
                        sc_q = internal_score(q)
                        if np.isfinite(sc_q) and sc_q < current_score:
                            if sc_q < best_score:
                                best_score = sc_q
                                best_p = q.copy()
                            p = q.copy()
                            current_score = sc_q
                            no_improve = 0
                            break

        g_grad, cmax = gradient_of_smoothed_max(p, T_factor)
        if not np.all(np.isfinite(g_grad)):
            g_grad = np.zeros(n)

        gabs = float(np.max(g_grad)) if g_grad.size else 0.0
        if gabs > 1e-300:
            base = (1.0 / n) / gabs
            lrs = (0.4 * base, 0.15 * base, 0.05 * base, 0.01 * base, 0.002 * base)
        else:
            lrs = (0.05, 0.02, 0.005)

        accepted = False
        for lr in lrs:
            lr = min(max(lr, 1e-10), 3.0)
            v = p - lr * g_grad
            pt = project_simplex(v)
            sc = internal_score(pt)
            if np.isfinite(sc) and sc < current_score - 1e-14:
                if sc < best_score:
                    best_score = sc
                    best_p = pt.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                current_score = sc
                p = pt.copy()
                accepted = True
                break

        if not accepted:
            T_factor = max(0.02, T_factor * 0.9)
            no_improve += 1
            if no_improve > 300:
                for sigma in (1.0 / n, 0.3 / n, 0.05 / n):
                    pt = project_simplex(p + rng.normal(0, sigma, n))
                    sc = internal_score(pt)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_p = pt.copy()
                        p = pt.copy()
                        current_score = sc
                        no_improve = 0
                        break
            if no_improve > 800:
                p = project_simplex(0.65 * best_p + 0.35 * p + rng.normal(0, 0.5 / n, n))
                current_score = internal_score(p)
                no_improve = 0
                T_factor = 0.3

        if it % 200 == 0:
            print("    grad iter %d: best=%.10f cur=%.10f T=%.3f" %
                  (it, best_score, current_score, T_factor), flush=True)

        if best_score < 1.495 and no_improve > 3000:
            break

    return best_p, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search function: LP-iterative + gradient + shape search."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 15.0)

    n = 700
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        parent_arr = np.where(np.isfinite(parent_arr), parent_arr, 0.0)
        parent_arr = np.maximum(parent_arr, 0.0)
        if parent_arr.ndim == 1 and parent_arr.size >= 32 and parent_arr.sum() > 0:
            parent = safe_normalize(parent_arr)
            n = int(parent_arr.size)
    except Exception:
        pass

    if n < 100:
        n = 700

    print("n=%d, parent available: %s" % (n, parent is not None), flush=True)

    best_p = None
    best_score = np.inf

    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        if p.size == 0 or not np.all(np.isfinite(p)):
            return
        p = safe_normalize(p)
        if not np.isfinite(sc):
            sc = internal_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print("  ** NEW BEST: %.10f (t=%.0fs)" % (best_score, time.monotonic() - start_time), flush=True)

    # Phase 1: Generate initial shapes and score them
    print("Phase 1: Initial shapes...", flush=True)
    t1_start = time.monotonic()
    shapes = make_diverse_shapes(n, rng)
    if parent is not None:
        shapes.insert(0, parent)
    scored = []
    for p in shapes:
        p = safe_normalize(p)
        sc = internal_score(p)
        if np.isfinite(sc):
            scored.append((sc, p))
    scored.sort(key=lambda x: x[0])
    for sc, p in scored[:3]:
        try_record(p, sc)
    print("  Top 5 initial scores: %s" % ["%.6f" % s for s, _ in scored[:5]], flush=True)
    print("  Phase 1 done in %.1fs" % (time.monotonic() - t1_start), flush=True)

    # Phase 2: Fourier shape search (global)
    print("Phase 2: Fourier shape search...", flush=True)
    t2_start = time.monotonic()
    if deadline - time.monotonic() > 300:
        dim = 2 * 13 + 2  # 13 Fourier modes (DC + 12) + 2 tapers = 28 params
        bounds = [(0.0, 0.0)]  # DC fixed
        for _ in range(12):
            bounds.append((-2.0, 2.0))
            bounds.append((-2.0, 2.0))
        bounds.append((-4.0, 0.0))
        bounds.append((-4.0, 0.0))

        fourier_n_objective = fourier_objective
        # Wrap with n
        class _ObjWrapper:
            def __init__(self, n_val):
                self.n_val = n_val
            def __call__(self, params):
                return fourier_objective(np.asarray(params, dtype=np.float64), self.n_val)

        obj_wrapper = _ObjWrapper(n)

        de_budget = min(240.0, (deadline - time.monotonic()) * 0.3)
        try:
            res = differential_evolution(
                obj_wrapper, bounds,
                maxiter=50, popsize=15,
                tol=0.3, seed=int(rng.integers(0, 2**31 - 1)),
                polish=False, init="latinhypercube",
                update_interval=1,
            )
            if res is not None and np.isfinite(getattr(res, "fun", np.inf)):
                p_de = fourier_shape(res.x, n)
                sc_de = internal_score(p_de)
                try_record(p_de, sc_de)
                print("  DE result: %.10f" % sc_de, flush=True)
                scored.append((sc_de, p_de))
        except Exception as e:
            print("  DE failed: %s" % str(e), flush=True)

    print("  Phase 2 done in %.1fs" % (time.monotonic() - t2_start), flush=True)

    # Phase 3: LP iterative optimization on top candidates
    print("Phase 3: LP iteration...", flush=True)
    scored.sort(key=lambda x: x[0])
    top_k = min(4, len(scored))
    lp_budget_total = deadline - time.monotonic() - 180.0
    if lp_budget_total < 120:
        lp_budget_total = max(60, lp_budget_total)
    lp_per = max(30.0, lp_budget_total / float(max(1, top_k)))

    for i in range(top_k):
        if time.monotonic() > deadline - 180:
            break
        sc, p = scored[i]
        remaining = min(lp_per, deadline - time.monotonic() - 180)
        if remaining < 20:
            break
        print("  LP on candidate %d (score=%.8f), budget=%.0fs" % (i, sc, remaining), flush=True)
        p_best, sc_best = lp_iterate(p, remaining, rng)
        try_record(p_best, sc_best)
        scored[i] = (sc_best, p_best)

    # Phase 4: Gradient refinement on global best
    if best_p is not None and time.monotonic() < deadline - 30:
        print("Phase 4: Gradient refinement...", flush=True)
        remaining = deadline - time.monotonic() - 10
        p_ref, sc_ref = gradient_refine(best_p, remaining, rng)
        try_record(p_ref, sc_ref)

    # Phase 5: Final LP push if time remains
    if best_p is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic() - 5
        if remaining > 30:
            print("Phase 5: Final LP push, %.0fs" % remaining, flush=True)
            p_fin, sc_fin = lp_iterate(best_p, remaining, rng, max_iters=200)
            try_record(p_fin, sc_fin)

    # Final cleanup
    if best_p is None:
        best_p = np.full(n, 1.0 / n, dtype=np.float64)

    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    s = float(best_p.sum())
    if s <= 1e-12:
        best_p = np.full(n, 1.0 / n, dtype=np.float64)
        s = 1.0

    final_sc = internal_score(best_p)
    print("FINAL: score=%.10f, n=%d, sum=%.4f, time=%.1fs" %
          (final_sc, n, s, time.monotonic() - start_time), flush=True)

    return [float(x) for x in best_p]
