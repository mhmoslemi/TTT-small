def _gauss_mixture_score(params, n):
    """Score for a Gaussian mixture representation. params = [w1,m1,s1, w2,m2,s2, ...]"""
    K = len(params) // 3
    a = np.zeros(n)
    for k in range(K):
        w_k = abs(params[3 * k])
        m_k = params[3 * k + 1]
        s_k = max(abs(params[3 * k + 2]), 1.0)
        idx = np.arange(n)
        a += w_k * np.exp(-((idx - m_k) ** 2) / (2.0 * s_k ** 2))
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-10:
        return 1e10, np.zeros(len(params))
    return _score(a), None

                a = np.zeros(n)
                for k in range(K):
                    w_k = abs(p[3 * k])
                    m_k = p[3 * k + 1]
                    s_k = max(abs(p[3 * k + 2]), 1.0)
                    idx = np.arange(n)
                    a += w_k * np.exp(-((idx - m_k) ** 2) / (2.0 * s_k ** 2))
                a = np.maximum(a, 0.0)
                if a.sum() < 1e-10:
                    return 1e10
                return _score(a)

            try:
                res = minimize(obj_gm, params, method='Nelder-Mead',
                               options={'maxiter': 500, 'xatol': 1.0,
                                        'fatol': 1e-7, 'adaptive': True})
                p_opt = res.x
                a = np.zeros(n)
                for k in range(K):
                    w_k = abs(p_opt[3 * k])
                    m_k = p_opt[3 * k + 1]
                    s_k = max(abs(p_opt[3 * k + 2]), 1.0)
                    idx = np.arange(n)
                    a += w_k * np.exp(-((idx - m_k) ** 2) / (2.0 * s_k ** 2))
                a = np.maximum(a, 0.0)
                if a.sum() > 1e-10:
                    sc = _score(a)
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
            except Exception:
                pass

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 15.0

    global_best = np.inf
    global_a = None

    def _try(a, label):
        nonlocal global_best, global_a
        if a is None:
            return
        a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        sc = _score(a)
        if sc < global_best:
            global_best = sc
            global_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST: {label} score={sc:.6f} n={len(a)} t={elapsed:.0f}s")

    # Determine lengths to try
    lengths = [4096, 8192, 16384]
    given_given = False
    try:
        given_raw = np.asarray(height_sequence_1, dtype=np.float64).flatten()
        if given_raw.size >= 256 and np.clip(given_raw, 0, 1000).sum() > 0:
            given_len = given_raw.size
            lengths = [given_len] + [l for l in lengths if l != given_len]
            given_given = True
    except Exception:
        pass

    print(f"Lengths to try: {lengths}")
    print(f"Budget: {budget_s}s, deadline in {budget_s-15:.0f}s")

    # === Phase 1: L-BFGS annealing ===
    explore_budget = (deadline - time.monotonic()) * 0.65

    for n in lengths:
        if time.monotonic() > t_start + explore_budget:
            break
        S = np.sqrt(2.0 * n)
        per_len = min(300.0, (t_start + explore_budget - time.monotonic()) / max(1, len(lengths) - lengths.index(n)))
        if per_len < 20.0:
            per_len = 20.0

        inits = []

        # Given sequence
        if given_given:
            try:
                given_raw = np.asarray(height_sequence_1, dtype=np.float64).flatten()
                if given_raw.size == n:
                    g = np.clip(given_raw, 0, 1000)
                    if g.sum() > 0:
                        inits.append(("given", g * (S / g.sum())))
            except Exception:
                pass

        # Raised cosine
        tt = np.linspace(0, 1, n)
        rc = 0.5 * (1.0 + np.cos(np.pi * (2.0 * tt - 1.0)))
        rc = np.maximum(rc, 1e-6)
        inits.append(("cosine", rc))

        # Smooth random (averaged random walk)
        rw = np.cumsum(rng.normal(0, 1, n))
        rw -= rw.min()
        rw += 0.1
        inits.append(("randwalk", rw))

        # B-spline-like: convolution of uniform windows
        m = max(2, n // 64)
        u = np.zeros(n)
        u[:m] = 1.0
        for _ in range(3):
            u = np.convolve(u, u, 'full')[:n]
            u = u / u.sum() * n
        inits.append(("bspline", u))

        # Flat with tapered edges
        flat = np.ones(n)
        ew = n // 8
        taper = np.linspace(0, 1, ew)
        flat[:ew] = taper
        flat[-ew:] = taper[::-1]
        inits.append(("flat_taper", flat))

        # Pure random
        inits.append(("random", np.maximum(rng.random(n), 1e-6)))

        n_inits = len(inits)
        per_init = per_len / n_inits

        for iname, a_init in inits:
            if time.monotonic() > t_start + explore_budget:
                break
            a_norm = np.clip(a_init, 0, 1000)
            if a_norm.sum() < 1e-10:
                a_norm = np.full(n, S / n)
            else:
                a_norm = a_norm * (S / a_norm.sum())
            w0 = _seq_to_w(a_norm, S)

            # Run annealing
            a_res, sc, w_res = _run_anneal(w0, S, n, min(per_init, 80.0), rng,
                                           beta_lo=5.0, beta_hi=3000.0, n_stages=7)
            _try(a_res, f"n={n}/{iname}")

            # Also run with different annealing (start sharper)
            if time.monotonic() < t_start + explore_budget:
                a_res2, sc2, _ = _run_anneal(w0, S, n, min(per_init * 0.6, 50.0), rng,
                                             beta_lo=50.0, beta_hi=8000.0, n_stages=5)
                _try(a_res2, f"n={n}/{iname}/sharp")

    # === Phase 2: Gaussian mixture exploration ===
    if time.monotonic() < deadline - 200.0:
        for n in [4096, 8192]:
            if time.monotonic() > deadline - 200.0:
                break
            print(f"\nGaussian mixture search n={n}...")
            gm_a, gm_sc = _gauss_mix_optimize(n, 60.0, rng)
            if gm_a is not None:
                _try(gm_a, f"gaussmix_n={n}")
                # Refine with L-BFGS
                if gm_a is not None and time.monotonic() < deadline - 150.0:
                    S = np.sqrt(2.0 * n)
                    a_n = gm_a * (S / gm_a.sum())
                    w0 = _seq_to_w(a_n, S)
                    a_ref, sc_ref, _ = _run_anneal(w0, S, n, 60.0, rng,
                                                   beta_lo=10.0, beta_hi=5000.0, n_stages=7)
                    _try(a_ref, f"gaussmix_refine_n={n}")

    # === Phase 3: Cross-pollination ===
    # Take the best sequence and try perturbations
    if global_a is not None and time.monotonic() < deadline - 100.0:
        n_best = len(global_a)
        S = np.sqrt(2.0 * n_best)
        a_best = global_a * (S / global_a.sum())
        w_best = _seq_to_w(a_best, S)

        # Perturb and re-optimize
        for perturb_scale in [0.05, 0.15]:
            if time.monotonic() > deadline - 100.0:
                break
            w_pert = w_best + rng.normal(0, perturb_scale, n_best)
            a_res, sc, _ = _run_anneal(w_pert, S, n_best, 40.0, rng,
                                       beta_lo=20.0, beta_hi=5000.0, n_stages=6)
            _try(a_res, f"perturb_{perturb_scale}")

    # === Phase 4: Subgradient polish ===
    if global_a is not None and time.monotonic() < deadline - 20.0:
        polish_time = min(150.0, deadline - time.monotonic() - 10.0)
        if polish_time > 10.0:
            print(f"\nSubgradient polish ({polish_time:.0f}s)...")
            a_p, sc_p = _subgrad_polish(global_a, polish_time, top_k=6)
            _try(a_p, "subgrad_polish")

    # === Phase 5: Final L-BFGS with very high beta ===
    if global_a is not None and time.monotonic() < deadline - 10.0:
        final_time = deadline - time.monotonic() - 5.0
        if final_time > 10.0:
            n_f = len(global_a)
            S_f = np.sqrt(2.0 * n_f)
            a_f = global_a * (S_f / global_a.sum())
            w_f = _seq_to_w(a_f, S_f)
            print(f"\nFinal high-beta L-BFGS ({final_time:.0f}s)...")
            a_fin, sc_fin, _ = _run_anneal(w_f, S_f, n_f, final_time, rng,
                                           beta_lo=2000.0, beta_hi=50000.0, n_stages=5)
            _try(a_fin, "final_highbeta")

    # Verify with evaluate_sequence
    if global_a is not None:
        fa = np.clip(global_a, 0, 1000).tolist()
        try:
            ver = evaluate_sequence(fa)
            if np.isfinite(ver):
                print(f"\nVerification score: {ver:.6f}")
                if ver < global_best:
                    global_best = ver
                else:
                    # Use verified score
                    print(f"(internal score was {global_best:.6f})")
            else:
                print("Verification returned inf, using internal score")
        except Exception as e:
            print(f"Verification error: {e}")

    # Fallback
    if global_a is None:
        global_a = np.ones(256)

    global_a = np.clip(global_a, 0.0, 1000.0)
    if global_a.sum() < 0.01:
        global_a = np.ones(256)

    final_sc = _score(global_a)
    print(f"\n=== FINAL: score={final_sc:.6f}, length={len(global_a)}, time={time.monotonic()-t_start:.1f}s ===")
    return [float(x) for x in global_a]
