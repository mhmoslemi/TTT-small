            # Targeted: reduce top peak contributors, boost non-peak
            valid_start = max(0, peak_idx - n + 1)
            valid_end = min(n, peak_idx + 1)
            contribs = np.zeros(n)
            for i in range(valid_start, valid_end):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]

            top_k = max(3, min(15, n // 12))
            top_idx = np.argsort(contribs)[-top_k:]

            # Reduce contributors (more aggressive for larger ones)
            reductions = rng.uniform(0.03, 0.12, size=len(top_idx))
            a_new[top_idx] *= (1.0 - reductions)

            # Distribute freed mass to positions far from peak
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            # Prefer indices that don't pair with the peak
            peak_partners = set()
            for i in range(valid_start, valid_end):
                j = peak_idx - i
                if 0 <= j < n:
                    peak_partners.add(i)
                    peak_partners.add(j)
            safe_idx = np.array([i for i in other_idx if i not in peak_partners])
            if len(safe_idx) < 3:
                safe_idx = other_idx
            n_boost = max(2, min(8, len(safe_idx) // 4))
            boost_idx = rng.choice(safe_idx, n_boost, replace=False)
            a_new[boost_idx] *= (1.0 + rng.uniform(0.03, 0.12, size=n_boost))

        elif r < 0.50:
            # Multi-peak reduction: target second-highest peak too
            sorted_peaks = np.argsort(c)[::-1]
            second_peak = int(sorted_peaks[1]) if c[sorted_peaks[1]] > 0.9 * peak_val else peak_idx

            valid_s = max(0, second_peak - n + 1)
            valid_e = min(n, second_peak + 1)
            for i in range(valid_s, valid_e):
                j = second_peak - i
                if 0 <= j < n and a[i] * a[j] > 0.02 * peak_val:
                    a_new[i] *= (1.0 - rng.uniform(0.02, 0.06))

        elif r < 0.65:
            # Block shift: move a block's mass to a different location
            width = max(2, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, n - width))
            block = a_new[start:start + width].copy()
            a_new[start:start + width] = 0
            # Insert at new location
            new_start = int(rng.integers(0, n - width))
            a_new[new_start:new_start + width] += block

        elif r < 0.75:
            # Symmetrize with noise (exploit that symmetric solutions are common)
            sym = 0.5 * (a + a[::-1])
            a_new = sym * np.exp(rng.normal(0, 0.008, n))

        elif r < 0.85:
            # Gaussian perturbation on a random subset
            subset_size = max(5, n // 6)
            subset = rng.choice(n, subset_size, replace=False)
            a_new[subset] *= np.exp(rng.normal(0, 0.02, subset_size))

        else:
            # Global small perturbation
            a_new = a * np.exp(rng.normal(0, 0.004, n))

        a_new = np.clip(a_new, 0, None)
        s = np.sum(a_new)
        if s < 1e-10:
            a_new = a
        else:
            a_new = a_new / s

        new_score = local_score(a_new)

        if new_score <= current_score:
            a = a_new
            current_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                if iters % 5000 == 0:
                    print(f"      LS: iter={iters}, new_best={best_score:.8f}, t={time.monotonic()-t0:.1f}s")
        else:
            # Accept with SA probability
            delta = new_score - current_score
            if delta < 0.01 and rng.random() < np.exp(-delta / max(temperature, 1e-10)):
                a = a_new
                current_score = new_score
            no_improve += 1

        # Anneal temperature
        temperature *= cooling
        if no_improve > 3000:
            # Restart with perturbation
            a = a * np.exp(rng.normal(0, 0.03, n))
            a = np.clip(a, 1e-8, None)
            a = a / np.sum(a)
            current_score = local_score(a)
            no_improve = 0
            temperature = 0.0005

        iters += 1
        if iters % 20000 == 0:
            print(f"    LS: iter={iters}, score={current_score:.8f}, best={best_score:.8f}, temp={temperature:.2e}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_score


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    """Multi-tone sinusoidal initialization."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in [1, 2, 3, 5, 8]:
        amp = 0.2 / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_bell_init(n, rng, center=0.5, width=0.3):
    """Gaussian bell initialization (concentrated in middle)."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.exp(-((t - center) ** 2) / (2 * width ** 2))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search:
    Phase 1: Quick global exploration with different inits and lengths.
    Phase 2: Gradient optimization from best candidates.
    Phase 3: Intensive peak-aware local search.
    Phase 4: Final polish.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided sequence
    start_seq = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
            # Normalize
            start_seq = start_seq / np.sum(start_seq)
            s = local_score(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.8f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()
    except Exception as e:
        print(f"Could not load provided sequence: {e}")

    # Phase 1: Quick global exploration
    print("\n=== Phase 1: Global Exploration ===")
    exploration_configs = [
        (256, 'chirp', 0.5),
        (256, 'multi_tone', 0.0),
        (256, 'random', 0.3),
        (256, 'bell', 0.0),
        (384, 'chirp', 0.4),
        (512, 'chirp', 0.5),
        (512, 'multi_tone', 0.0),
    ]

    candidates = []
    for n, init_type, param in exploration_configs:
        if time.monotonic() > deadline - 600:
            break

        if init_type == 'chirp':
            a0 = make_chirp_init(n, param, rng)
        elif init_type == 'multi_tone':
            a0 = make_multi_tone_init(n, rng)
        elif init_type == 'bell':
            a0 = make_bell_init(n, param * 0 + 0.5, rng)
        else:
            a0 = make_random_init(n, rng, param)

        a0 = a0 / np.sum(a0)
        score = local_score(a0)
        print(f"  n={n}, init='{init_type}': score={score:.8f}")
        candidates.append((n, a0, score, init_type))

        if score < best_score:
            best_score = score
            best_seq = a0.copy()

    # If we have a provided sequence, include it
    if start_seq is not None:
        candidates.insert(0, (len(start_seq), start_seq.copy(), local_score(start_seq), 'provided'))

    # Phase 2: Gradient optimization on best candidates
    print(f"\n=== Phase 2: Gradient Optimization ===")
    # Sort by initial score and pick top 4
    candidates.sort(key=lambda x: x[2])
    top_candidates = candidates[:4]

    grad_time_each = min(150.0, max(30.0, (deadline - time.monotonic()) * 0.3 / len(top_candidates)))
    print(f"  Time per gradient run: {grad_time_each:.1f}s")

    for n, a0, s0, init_type in top_candidates:
        if time.monotonic() > deadline - 400:
            break
        t0 = time.monotonic()
        a_opt, grad_score = optimize_gradient(a0, grad_time_each, rng)
        actual_score = local_score(a_opt)
        elapsed = time.monotonic() - t0
        print(f"  n={n}, init='{init_type}': grad_score={grad_score:.8f}, actual={actual_score:.8f}, time={elapsed:.1f}s")

        if actual_score < best_score:
            best_score = actual_score
            best_seq = a_opt.copy()
            print(f"    *** New best: {best_score:.8f} ***")

    # Phase 3: Intensive local search
    print(f"\n=== Phase 3: Intensive Local Search ===")
    if best_seq is not None:
        ls_time = min(900.0, max(100.0, deadline - time.monotonic() - 60))
        print(f"  Starting local search, score={best_score:.8f}, time_available={ls_time:.1f}s")
        polished, ls_score = local_search_peak(best_seq, ls_time, rng)
        if ls_score < best_score:
            best_score = ls_score
            best_seq = polished.copy()
            print(f"  After local search: {best_score:.8f}")
        else:
            print(f"  Local search didn't improve (score={ls_score:.8f} vs best={best_score:.8f})")

    # Phase 4: Final short gradient polish + local search
    print(f"\n=== Phase 4: Final Polish ===")
    if best_seq is not None:
        remaining = max(10.0, deadline - time.monotonic() - 20)
        # Short gradient refinement
        if remaining > 60:
            grad_time = remaining * 0.4
            a_opt, g_score = optimize_gradient(best_seq, grad_time, rng)
            if local_score(a_opt) < best_score:
                best_score = local_score(a_opt)
                best_seq = a_opt.copy()
                print(f"  After gradient polish: {best_score:.8f}")

        # Final local search
        remaining2 = max(10.0, deadline - time.monotonic() - 20)
        if remaining2 > 20:
            a_pol, p_score = local_search_peak(best_seq, remaining2, rng)
            if p_score < best_score:
                best_score = p_score
                best_seq = a_pol.copy()
                print(f"  After final local search: {best_score:.8f}")

    # Return the best sequence found
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.8f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
