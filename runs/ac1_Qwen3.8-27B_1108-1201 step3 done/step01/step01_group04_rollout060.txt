"""
Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max smoothed gradient descent with momentum, LP-based refinement
with constraint generation, and multi-start strategy.
"""

import numpy as np
import time
from scipy.optimize import linprog

def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=None):
    """Normalize a to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-15:
        return a
    return a * (target / S)

def softmax_obj_and_grad(a, tau, top_m=10):
    """Compute soft-max objective and gradient for peak convolution."""
    n = len(a)
    conv = np.convolve(a, a, mode='full')
    # Soft-max approximation to max
    shifted = conv - conv.max()
    exp_vals = np.exp(tau * shifted)
    weights = exp_vals / exp_vals.sum()
    
    # Objective: (1/tau) * log(sum(exp(tau * conv))) = conv.max() + (1/tau)*log(sum(exp(shifted)))
    obj = conv.max() + np.log(exp_vals.sum()) / tau
    
    # Gradient: weighted sum of per-peak gradients
    grad = np.zeros(n)
    # Get top_m indices by weight
    idx = np.argsort(weights)[::-1][:top_m]
    for k in idx:
        w = weights[k]
        if w < 1e-10:
            continue
        # d conv[k]/d a[i] = 2*a[k-i] for valid k-i
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * w * a[j]
    
    return obj, grad, conv

def soft_obj_and_grad_fast(a, tau, top_m=5):
    """Faster version: only track top_m peaks for gradient."""
    n = len(a)
    conv = np.convolve(a, a, mode='full')
    
    shifted = conv - conv.max()
    exp_vals = np.exp(tau * shifted)
    z = exp_vals.sum()
    weights = exp_vals / z
    
    obj = conv.max() + np.log(z) / tau
    
    grad = np.zeros(n)
    idx = np.argsort(weights)[::-1][:top_m]
    for k in idx:
        w = weights[int(k)]
        if w < 1e-12:
            continue
        k = int(k)
        # d conv[k]/d a[i] = 2*a[k-i] for 0 <= k-i < n
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * w * a[k - i]
    
    return obj, grad

def project_onto_simplex_direction(a, grad, target_sum):
    """Project gradient step onto {a >= 0, sum(a) = target_sum}."""
    # Remove mean component of gradient (tangent to sum constraint)
    grad_centered = grad - grad.mean()
    return grad_centered

def grad_descent_smooth(a0, deadline, tau_start=5.0, tau_end=50.0, 
                        lr_start=0.5, momentum=0.9, max_iters=5000):
    """Gradient descent with soft-max objective and momentum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    target_sum = np.sqrt(2.0 * n)
    a = normalize(a, target_sum)
    
    best = a.copy()
    best_score = compute_obj(a)
    vel = np.zeros(n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Anneal tau
        frac = it / max_iters
        tau = tau_start + (tau_end - tau_start) * frac
        lr = lr_start * 0.5 * (1 + np.cos(np.pi * frac))
        lr = max(lr, 0.01)
        
        obj, grad = soft_obj_and_grad_fast(a, tau, top_m=5)
        
        # Center gradient (remove component along all-ones)
        grad = grad - grad.mean()
        
        # Nesterov momentum
        vel = momentum * vel - lr * grad
        a_new = a + vel
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, target_sum)
        
        # Check improvement
        score = compute_obj(a_new)
        if score < best_score:
            best_score = score
            best = a_new.copy()
        a = a_new
    
    return best, best_score

def lp_refine(f, M, n_constraints=100):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Uses constraint generation for efficiency."""
    n = len(f)
    if n > 2048:
        return None
    
    # Start with a subset of constraints
    fg = np.convolve(f, np.ones(n) / n)  # initial guess for active constraints
    active = np.argsort(fg)[::-1][:min(n_constraints, 2*n-1)]
    
    for iteration in range(20):
        # Build constraint matrix for active constraints
        m = len(active)
        A = np.zeros((m, n))
        for idx, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A[idx, j] = f[k - j]
        
        c_obj = -np.ones(n)
        b = np.full(m, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
            if res.status != 0:
                return None
            g = res.x
        except Exception:
            return None
        
        # Check for violated constraints
        conv_fg = np.convolve(f, g)
        violations = np.where(conv_fg > M + 1e-8)[0]
        if len(violations) == 0:
            return g
        
        # Add violated constraints (up to a limit)
        for v in violations[:50]:
            if v not in active:
                active = np.append(active, v)
        
        if len(active) > 2*n - 1:
            active = np.arange(2*n - 1)
    
    return g

def lp_iteration(a, deadline, max_iters=50, min_n=32, max_n=2048):
    """Iterative LP-based refinement."""
    n = len(a)
    if n < min_n or n > max_n:
        return a, compute_obj(a)
    
    target_sum = np.sqrt(2.0 * n)
    best = a.copy()
    best_score = compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(a, a)
        M = float(conv.max())
        
        g = lp_refine(a, M)
        if g is None:
            break
        
        # Line search for blend factor
        improved = False
        for t in [0.9, 0.7, 0.5, 0.3, 0.1, 0.05]:
            a_new = (1 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, target_sum)
            s = compute_obj(a_new)
            if s < best_score:
                a = a_new
                best = a.copy()
                best_score = s
                improved = True
                break
        
        if not improved:
            # Small perturbation to escape
            a = a + 0.01 * (np.random.random(n) - 0.5)
            a = np.maximum(a, 0.0)

"""
Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Primary method: iterative LP-based refinement (block coordinate descent on 
the peak convolution constraint). Secondary: soft-max gradient descent with 
momentum. Multiple diverse starting points of various lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def _normalize(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-15:
        return a.copy()
    return a * (target / S)


def _lp_solve_constraints(f, M, active_indices):
    """Solve LP with a specific set of active constraints.
    Returns solution g or None."""
    n = len(f)
    m = len(active_indices)
    
    A = np.zeros((m, n))
    for idx, k in enumerate(active_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[idx, j] = f[k - j]
    
    c_obj = -np.ones(n)
    b = np.full(m, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _lp_solve(f, M, max_rounds=30, max_constraints=200):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Uses constraint generation."""
    n = len(f)
    total = 2 * n - 1
    
    # Initial active set: top convolution values of conv(f, ones/n)
    test_g = np.ones(n) / n
    test_conv = np.convolve(f, test_g)
    active = np.argsort(test_conv)[::-1][:min(50, total)]
    
    g = None
    for round_num in range(max_rounds):
        g = _lp_solve_constraints(f, M, active)
        if g is None:
            return None
        
        # Check all constraints
        conv_fg = np.convolve(f, g)
        violated = np.where(conv_fg > M * (1 + 1e-9) + 1e-12)[0]
        
        if len(violated) == 0:
            return g
        
        # Add violated constraints
        for v in violated[:50]:
            if v not in active:
                active = np.append(active, v)
        
        if len(active) >= max_constraints:
            # Use the most violated ones
            conv_fg = np.convolve(f, g)
            active = np.argsort(conv_fg)[::-1][:max_constraints]
    
    return g


def _lp_refine_sequence(a, deadline, max_iters=200):
    """Iterative LP-based refinement of sequence a."""
    n = len(a)
    if n < 16 or n > 4096:
        return a, _compute_obj(a)
    
    target_sum = np.sqrt(2.0 * n)
    a = _normalize(np.maximum(a, 0.0), target_sum)
    best = a.copy()
    best_score = _compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(a, a)
        M = float(conv.max())
        
        g = _lp_solve(a, M)
        if g is None:
            break
        
        # Line search
        best_t = 0.0
        best_s = best_score
        for t in np.linspace(0.05, 0.95, 19):
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0.0)
            a_try = _normalize(a_try, target_sum)
            s = _compute_obj(a_try)
            if s < best_s:
                best_s = s
                best_t = t
        
        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0.0)
            a = _normalize(a, target_sum)
            if best_s < best_score:
                best_score = best_s
                best = a.copy()
        else:
            # No improvement from LP; try a small perturbation
            break
    
    return best, best_score


def _grad_step(a, lr, tau=20.0, top_m=5):
    """One soft-max gradient step (projected)."""
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    
    conv = np.convolve(a, a)
    shifted = conv - conv.max()
    exp_vals = np.exp(tau * shifted)
    w = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    idx = np.argsort(w)[::-1][:top_m]
    for k in idx:
        wk = w[int(k)]
        if wk < 1e-12:
            continue
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            grad[i] += 2.0 * wk * a[k - i]
    
    grad -= grad.mean()
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = _normalize(a_new, target_sum)
    return a_new
