```python
import numpy as np
from scipy.optimize import linprog
import time

def score(f):
    """Fast score computation."""
    f = np.asarray(f, dtype=np.float64)
    f = np.clip(f, 0, 1000)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = np.convolve(f, f)
    return 2.0 * len(f) * c.max() / (s * s)

def lp_step(f, M, n):
    """Solve the LP to find best g given current f and peak M."""
    # Build constraint matrix A (2n-1 x n)
    # A[k, i] = f[i] if 0 <= k-i < n, else 0
    # So A[k, i] = f[i] * (k-i >= 0) * (k-i < n)
    #            = f[i] * (i <= k) * (i > k-n)
    #            = f[i] * (max(0, k-n+1) <= i <= min(k, n-1))
    
    # For efficiency, build A as a banded matrix
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_lo = max(0, k-n+1)
        i_hi = min(k, n-1)
        if i_lo <= i_hi:
            rows.extend([k]*(i_hi-i_lo+1))
            cols.extend(range(i_lo, i_hi+1))
            data.extend(f[i_lo:i_hi+1])
    
    A = sp.coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None

def optimize(f, budget_s):
    """Main optimization loop for a single sequence."""
    n = len(f)
    f = f / f.sum()  # normalize
    
    best = f.copy()
    best_sc = score(f)
    
    t = time.monotonic()
    deadline = t + budget_s
    stall = 0
    
    while time.monotonic() < deadline:
        c = np.convolve(f, f)
        M = c.max()
        
        g = lp_step(f, M, n)
        if g is None:
            # Fallback: subgradient step
            g = subgradient_step(f, M, c)
        
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g / gs
        
        # Line search
        best_t = 0.5
        best_val = np.inf
        for t_candidate in [0.3, 0.5, 0.7, 0.9]:
            f_trial = (1 - t_candidate) * f + t_candidate * g
            f_trial = np.maximum(f_trial, 0)
            s = f_trial.sum()
            if s > 1e-10:
                f_trial = f_trial / s
                val = score(f_trial)
                if val < best_val:
                    best_val = val
                    best_t = t_candidate
        
        f_new = (1 - best_t) * f + best_t * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s > 1e-10:
            f_new = f_new / s
        
        sc = score(f_new)
        if sc < best_sc - 1e-8:
            best = f_new.copy()
            best_sc = sc
            stall = 0
        else:
            stall += 1
            if stall > 20:
                # Perturb to escape
                f_new = perturb(f_new)
                sc = score(f_new)
        
        f = f_new
    
    return best, best_sc
```
