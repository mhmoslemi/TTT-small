    C = fft_convolve(a, a)
    S = np.sum(a)
    current_score = 2.0 * n * np.max(C) / (S * S)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        # Random order for this pass
        order = np.random.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            a_i = a[i]
            if a_i < 1e-10:
                continue

            # Determine affected C indices
            k_min = i
            k_max = min(i + n - 1, 2 * n - 2)
            is_2i = (2 * i <= 2 * n - 2)

            # Precompute max of unaffected C values
            # Affected: k in [k_min, k_max]
            if k_min == 0:
                old_max_unaffected = np.max(C[k_max + 1:]) if k_max + 1 < 2 * n - 1 else 0.0
            elif k_max + 1 >= 2 * n - 1:
                old_max_unaffected = np.max(C[:k_min]) if k_min > 0 else 0.0
            else:
                old_max_unaffected = max(
                    np.max(C[:k_min]) if k_min > 0 else 0.0,
                    np.max(C[k_max + 1:]) if k_max + 1 < 2 * n - 1 else 0.0
                )

            # 1D search for optimal x
            best_x = a_i
            best_f = current_score

            # Coarse search: try range of x values
            # The optimal x is likely near a_i (small change per step)
            search_lo = max(0.0, a_i * 0.01)
            search_hi = a_i * 2.0 + 1.0 / n

            # Generate candidate x values
            xs = np.concatenate([
                [0.0],
                np.linspace(search_lo, search_hi, 80),
                [a_i * 0.5, a_i * 1.5, a_i * 0.1, a_i]
            ])
            xs = np.clip(xs, 0.0, 1000.0)
            xs = np.unique(xs)

            for x in xs:
                if abs(x - a_i) < 1e-12:
                    f = current_score
                else:
                    S_new = S - a_i + x
                    if S_new < 0.01:
                        continue
                    # Compute new max of affected C values
                    max_affected = 0.0
                    for k in range(k_min, k_max + 1):
                        ki = k - i
                        if k == 2 * i and is_2i:
                            new_val = C[k] + (x * x - a_i * a_i)
                        else:
                            new_val = C[k] + 2.0 * (x - a_i) * a[ki]
                        if new_val > max_affected:
                            max_affected = new_val
                    new_max = max(max_affected, old_max_unaffected)
                    f = 2.0 * n * new_max / (S_new * S_new)

                if f < best_f:
                    best_f = f
                    best_x = x

            # Fine refinement around best_x
            if best_x != a_i:
                span = max(abs(best_x - a_i), a_i * 0.1, 0.01 / n)
                xs_fine = np.linspace(max(0, best_x - span), best_x + span, 40)
                xs_fine = np.clip(xs_fine, 0.0, 1000.0)

                for x in xs_fine:
                    S_new = S - a_i + x
                    if S_new < 0.01:
                        continue
                    max_affected = 0.0
                    for k in range(k_min, k_max + 1):
                        ki = k - i
                        if k == 2 * i and is_2i:
                            new_val = C[k] + (x * x - a_i * a_i)
                        else:
                            new_val = C[k] + 2.0 * (x - a_i) * a[ki]
                        if new_val > max_affected:
                            max_affected = new_val
                    new_max = max(max_affected, old_max_unaffected)
                    f = 2.0 * n * new_max / (S_new * S_new)

                    if f < best_f:
                        best_f = f
                        best_x = x

            # Apply update
            if best_x != a_i and best_f < current_score:
                delta = best_x - a_i
                # Update C
                for k in range(k_min, k_max + 1):
                    ki = k - i
                    if k == 2 * i and is_2i:
                        C[k] += (best_x * best_x - a_i * a_i)
                    else:
                        C[k] += 2.0 * delta * a[ki]
                a[i] = best_x
                S += delta
                current_score = best_f

                if current_score < best_score:
                    best_score = current_score
                    best_a = a.copy()

        it += 1
        if n_iters is not None and it >= n_iters:
            break

    return best_a, best_score


def swap_local_search(a_init, time_limit, rng):
    """
    Local search using swap operations: transfer mass between pairs of coordinates.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    current_score = compute_score(a)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        # Pick two random indices
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Try transferring mass from i to j (or vice versa)
        # a[i] -> a[i] - delta, a[j] -> a[j] + delta (preserves sum)
        candidates = []
        a_i, a_j = a[i], a[j]
        if a_i > 0.01:
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                delta = frac * a_i
                candidates.append((i, j, delta))
        if a_j > 0.01:
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                delta = frac * a_j
                candidates.append((j, i, delta))

        for src, dst, delta in candidates:
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dst] += delta
            if a_new[src] < 0:
                continue
            s = compute_score(a_new)
            if s < current_score:
                a = a_new
                current_score = s
                if s < best_score:
                    best_score = s
                    best_a = a.copy()
                break  # Accept first improvement

    return best_a, best_score


def make_chirp_init(n, alpha, rng):
    """Chirp-like initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n + 1))
    a = np.clip(a, 0.01, None)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    """Random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    """Multi-tone initialization."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0
    for k in range(1, 8):
        amp = 0.15 / np.sqrt(k)
        a += amp * np.cos(2.0 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_anti_palindrome_init(n, rng):
    """Initialize to minimize reversal correlation."""
    a = rng.standard_normal(n)
    # Make a[i] and a[n-1-i] decorrelated
    b = rng.standard_normal(n)
    a = a + 0.5 * np.roll(b, n // 2)
    a = np.abs(a) + 0.1
    return a


def make_gradient_init(n, rng):
    """Random walk initialization (smooth)."""
    a = np.ones(n) * 0.5
    for i in range(1, n):
        a[i] = a[i - 1] + rng.normal(0, 0.3)
        a[i] = max(a[i], 0.05)
    return np.clip(a, 0.05, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Multi-stage: gradient optimization with diverse inits,
    then coordinate descent polish, then swap local search.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Try provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = compute_score(start_seq)
        print(f"Provided: n={len(start_seq)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = start_seq.copy()

    # Phase 1: Gradient optimization with diverse initializations
    print("\n=== Phase 1: Gradient optimization ===")
    configurations = [
        # (length, init_type, label)
        (256, 'chirp05', 'chirp a=0.5'),
        (256, 'chirp03', 'chirp a=0.3'),
        (256, 'multitone', 'multi-tone'),
        (256, 'apalin', 'anti-palindrome'),
        (256, 'random', 'random'),
        (384, 'chirp05', 'chirp a=0.5'),
        (384, 'multitone', 'multi-tone'),
        (384, 'random', 'random'),
        (512, 'chirp05', 'chirp a=0.5'),
        (512, 'multitone', 'multi-tone'),
        (512, 'gradient', 'gradient'),
        (768, 'chirp05', 'chirp a=0.5'),
        (768, 'multitone', 'multi-tone'),
        (1024, 'chirp05', 'chirp a=0.5'),
        (1024, 'multitone', 'multi-tone'),
    ]

    # If we have a provided sequence, add it as a configuration
    if start_seq is not None:
        configurations.insert(0, (len(start_seq), 'provided', 'provided seq'))

    # Time per config
    n_configs = len(configurations)
    phase1_time = time_budget * 0.5
    time_per_config = phase1_time / n_configs

    candidates = []  # (score, sequence, n)

    for n, init_type, label in configurations:
        if time.monotonic() > deadline - 60:
            break

        time_left = min(time_per_config * 1.5, deadline - time.monotonic() - 60)
        if time_left < 5:
            break

        # Generate initialization
        if init_type == 'provided' and start_seq is not None:
            if len(start_seq) == n:
                a0 = start_seq.copy()
            else:
                x_old = np.linspace(0, 1, len(start_seq))
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, start_seq)
        elif init_type == 'chirp05':
            a0 = make_chirp_init(n, 0.5, rng)
        elif init_type == 'chirp03':
            a0 = make_chirp_init(n, 0.3, rng)
        elif init_type == 'multitone':
            a0 = make_multi_tone_init(n, rng)
        elif init_type == 'apalin':
            a0 = make_anti_palindrome_init(n, rng)
        elif init_type == 'gradient':
            a0 = make_gradient_init(n, rng)
        else:
            a0 = make_random_init(n, rng)

        a0 = np.clip(a0, 0.01, 1000.0)
        theta_init = np.log(a0 / np.sum(a0) + 1e-10)

        t0 = time.monotonic()
        theta_best, grad_score = gradient_optimize(n, theta_init, time_left)
        elapsed = time.monotonic() - t0

        a_best = softmax_normalize(theta_best) * np.sqrt(2.0 * n)
        actual_score = compute_score(a_best)
        print(f"  n={n}, init='{label}', score={actual_score:.6f}, time={elapsed:.1f}s")

        if actual_score < best_score:
            best_score = actual_score
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

        candidates.append((actual_score, a_best.copy(), n))
        print(f"    Candidates: {len(candidates)}, best={best_score:.6f}")

    # Phase 2: Coordinate descent polish on top candidates
    print(f"\n=== Phase 2: Coordinate descent polish ===")
    candidates.sort(key=lambda x: x[0])
    phase2_time = time_budget * 0.35
    top_k = min(5, len(candidates))
    time_per_polish = phase2_time / top_k

    for idx in range(top_k):
        if time.monotonic() > deadline - 30:
            break
        score_c, seq_c, n_c = candidates[idx]
        t_left = min(time_per_polish * 1.5, deadline - time.monotonic() - 30)
        if t_left < 5:
            break

        print(f"  Polishing candidate {idx}: n={n_c}, score={score_c:.6f}, time_avail={t_left:.1f}s")
        t0 = time.monotonic()
        polished, polish_score = coord_descent_polish(seq_c, t_left)
        elapsed = time.monotonic() - t0
        print(f"    After polish: score={polish_score:.6f} (time={elapsed:.1f}s)")

        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"    *** New best: {best_score:.6f} ***")

        # Swap local search on best from coord descent
        t_left2 = min(30.0, deadline - time.monotonic() - 30)
        if t_left2 > 5:
            swapped, swap_score = swap_local_search(polished, t_left2, rng)
            if swap_score < best_score:
                best_score = swap_score
                best_seq = swapped.copy()
                print(f"    After swap: score={swap_score:.6f} *** New best!")

    # Phase 3: Final refinement on the best candidate
    print(f"\n=== Phase 3: Final refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        t_left = deadline - time.monotonic() - 10
        n_best = len(best_seq)

        # More coordinate descent
        t_cd = min(t_left * 0.6, 200.0)
        print(f"  More coord descent: time={t_cd:.1f}s")
        refined, r_score = coord_descent_polish(best_seq, t_cd)
        if r_score < best_score:
            best_score = r_score
            best_seq = refined.copy()
            print(f"    After more CD: {best_score:.6f}")

        # More swap search
        t_swap = min(t_left * 0.4, 100.0)
        if t_swap > 5:
            print(f"  More swap search: time={t_swap:.1f}s")
            swapped2, s2_score = swap_local_search(best_seq, t_swap, rng)
            if s2_score < best_score:
                best_score = s2_score
                best_seq = swapped2.copy()
                print(f"    After more swaps: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
