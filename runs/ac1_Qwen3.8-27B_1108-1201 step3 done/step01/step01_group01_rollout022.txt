import numpy as np
import time

def project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - target)
    rho = np.max(rho[cond]) if np.any(cond) else 0
    theta = (cssv[rho-1] - target) / rho
    return np.maximum(v - theta, 0.0)

def smooth_obj_grad(a, tau, n=None):
    """Compute smoothed objective value and gradient."""
    if n is None:
        n = len(a)
    b = np.convolve(a, a)
    b_max = np.max(b)
    exp_b = np.exp((b - b_max) / tau)
    s = np.sum(exp_b)
    p = exp_b / s
    smooth_max = b_max + tau * np.log(s)
    obj = 2.0 * n * smooth_max
    
    # Gradient
    s_stride = p.strides[0]
    P = np.lib.stride_tricks.as_strided(p, shape=(n, n), strides=(s_stride, s_stride))
    grad = 4.0 * n * (np.dot(a, P.T))
    
    return obj, grad

    FISTA to minimize the smoothed objective on the simplex.

        f_y, grad_y = smooth_obj_grad(y, tau_schedule[0], n)

        f_x, _ = smooth_obj_grad(x_new, tau_schedule[0], n)

def project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - target)
    if not np.any(cond):
        return np.full(n, target / n)
    r = rho[np.argmax(cond)]  # largest rho satisfying condition
    # Actually, we want the largest rho where cond is True
    # Since cond is monotonically decreasing (u is sorted descending),
    # the last True is what we want.
    # np.argmax gives the first True. We want the last True.
    # Since cond is [True, True, ..., True, False, False, ...],
    # the last True index is the count of Trues minus 1.
    r = np.sum(cond) - 1
    theta = (cssv[r] - target) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute full autocorrelation (convolution with itself)."""

def smooth_obj_and_grad(a, tau, n):
    """
    Compute the smoothed objective 2*n*tau*log(sum(exp(b_k/tau))) and its gradient.
    """

    grad = 4.0 * n * np.dot(a, P.T)
    
    return obj, grad
