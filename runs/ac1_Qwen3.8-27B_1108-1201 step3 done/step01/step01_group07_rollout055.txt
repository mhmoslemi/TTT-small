Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided alternating optimization

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def build_conv_matrix(f):
    """Build matrix A (shape 2n-1 x n) where (A @ g) = cross_conv(f, g)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = f
    return A


def solve_lp(f, M):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M. Returns g or None."""
    n = len(f)
    c = -np.ones(n)
    A = build_conv_matrix(f)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': 30})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def mirror_step(f, t_param, eta, n):
    """One mirror descent step on the simplex."""
    conv = fast_convolve(f)
    M = conv.max()
    if M < 1e-15:
        return f
    
    scaled = t_param * conv / M
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Subgradient: sg[i] = sum_k w[k] * f[k-i]
    sg = np.zeros(n)
    thresh = w.max() * 0.001
    active = np.where(w > thresh)[0]
    for k in active:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        sg[j_start:j_end + 1] += w[k] * f[k - j_end:k - j_start + 1][::-1]
    
    sg_centered = sg - sg.mean()
    log_factor = -eta * sg_centered
    log_factor -= log_factor.max()
    f_new = f * np.exp(log_factor)
    f_new = np.maximum(f_new, 0.0)
    f_new_sum = f_new.sum()
    if f_new_sum < 1e-12:
        return f
    f_new /= f_new_sum
    return f_new


def lp_optimize(f0, time_limit, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    t0 = time.monotonic()
    t = 1.0
    patience = 0

        if g is None:

            g = solve_lp(f, M * 1.001)
            if g is None:

        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum
        
        # Check if g is essentially the same as f (convergence)
        if np.sum(np.abs(g - f)) < 1e-10:
            break

        improved = False

            f_new = np.maximum(f_new, 0.0)
            f_new_sum = f_new.sum()
            if f_new_sum < 1e-12:
                continue
            f_new /= f_new_sum
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-12:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.05)
                improved = True
                break
        
        if not improved:
            patience += 1
            t *= 0.5
            if patience > 30:
                break
    
    return best_f, best_obj

def generate_inits(n, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    for sf in [0.1, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g
    
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()
    
    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    cos_b = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cosine'] = cos_b / cos_b.sum()
    
    r = rng.dirichlet(np.ones(n) * 0.3)
    inits['random'] = r
    
    # "Triangle" - peaks at center
    tri = 1.0 - np.abs(x - n * 0.5) / (n * 0.5)
    tri = np.maximum(tri, 0.1)
    inits['triangle'] = tri / tri.sum()
    
    # "U-shaped" - peaks at edges
    u = np.abs(x - n * 0.5) / (n * 0.5) + 0.1
    inits['u_shaped'] = u / u.sum()
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided alternating optimization.
The LP step finds the maximizer of sum(g) subject to g>=0 and (f*g)<=M (cross-conv bound).
Convex combination with backtracking ensures monotone objective decrease.
Combined with mirror descent polishing and diverse initializations across multiple lengths.

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def make_lp_structure(n):
    """Precompute row/col indices for the cross-convolution constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, rows, cols, n, timeout=20):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_optimize(f0, time_limit, discount=0.95):
    """LP-based alternating optimization with backtracking line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    rows, cols = make_lp_structure(n)
    
    t0 = time.monotonic()
    t = 1.0
    patience = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        g = solve_lp(f, M, rows, cols, n)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum
        
        # Check convergence
        diff = np.max(np.abs(g - f))
        if diff < 1e-12:
            break
        
        # Backtracking line search
        improved = False
        for t_try in [t, 0.7*t, 0.5*t, 0.3*t, 0.1*t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-13:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.08)
                improved = True
                break
        
        if not improved:
            patience += 1
            t *= discount
            if patience > 40:
                break
    
    return best_f, best_obj

        if w_sum < 1e-12:

    tri = np.maximum(1.0 - np.abs(x - 0.5*n)/(0.5*n), 0.05)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-guided alternating optimization
with backtracking line search, mirror descent polishing, and diverse initializations.
The LP step solves: max sum(g) s.t. g>=0, (f*g)[k]<=M for all k (cross-conv bound).
This provides a globally-informed direction at each step, avoiding local minima
that gradient-only methods can get stuck in.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def make_lp_structure(n):
    """Precompute row/col indices for the cross-convolution constraint matrix (2n-1 x n).
    A[k,i] = f[k-i] when 0<=k-i<=n-1, else 0.
    Column i has f[0..n-1] at rows i..i+n-1."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, rows, cols, n, timeout=15):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k=0..2n-2."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_optimize(f0, time_limit):
    """LP-based alternating optimization with backtracking line search.
    f0: initial vector (non-negative). Returns (best_f, best_normalized_obj)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    rows, cols = make_lp_structure(n)

    t0 = time.monotonic()
    t = 1.0
    patience = 0
    iter_count = 0
    last_print = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        g = solve_lp(f, M, rows, cols, n)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        # Convergence check
        if np.max(np.abs(g - f)) < 1e-13:
            break

        # Backtracking line search
        improved = False
        for t_try in [t, 0.7 * t, 0.5 * t, 0.3 * t, 0.1 * t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-13:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.1)
                improved = True
                break

        if not improved:
            patience += 1
            t *= 0.8
            if patience > 50:
                break

        if iter_count - last_print > 50:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}, obj={best_obj:.8f}, t={t:.3f}, elapsed={elapsed:.1f}s")
            last_print = iter_count

    return best_f, best_obj


def mirror_polish(f0, time_limit, t_param=50.0, eta0=0.3):
    """Polish with multiplicative mirror descent on the probability simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    eta = eta0
    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        scaled = t_param * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        thresh = w.max() * 0.002
        active = np.where(w > thresh)[0]
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            sg[j_start:j_end + 1] += w[k] * f[k - j_end:k - j_start + 1][::-1]

        sg_c = sg - sg.mean()
        lf = -eta * sg_c
        lf -= lf.max()
        f = f * np.exp(lf)
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-12:
            f = np.ones(n) / n
        else:
            f /= s

        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            patience = 0
        else:
            patience += 1
            if patience % 25 == 0:
                eta *= 0.7
                if eta < 0.008:
                    t_param = min(t_param * 1.5, 500.0)
                    eta = 0.2
            if patience > 250:
                break

    return best_f, best_obj


def gen_inits(n, rng):
    """Generate diverse initializations of length n, each summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uni'] = np.ones(n) / n

    for sf in [0.1, 0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g

    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()

    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()

    rr = rng.dirichlet(np.ones(n) * 0.3)
    inits['rnd'] = rr

    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.05)
    inits['tri'] = tri / tri.sum()

    return inits
