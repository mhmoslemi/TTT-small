   3. We want to minimize t subject to p on the simplex and all other convolution values <= t.
   4. This is a non-convex problem (the feasible set depends on t).

   A bisection approach:
   - For a given t, check if there exists p on the simplex with max(conv(p,p)) <= t.
   - This is still non-convex in general.

   However, we can use a heuristic:
   - Start with a good p.
   - Find the peak lag k*.
   - Try to reduce conv(p,p)_{k*} by moving mass in p.
   - Iterate.

   This is essentially what the gradient method does, but we can make it more targeted.

   **Another idea: Water-filling on the convolution**

   Think of the convolution values as "buckets" and the sequence entries as "water sources." We want to fill the buckets as evenly as possible. The gradient tells us which "direction" in sequence-space decreases the fullest bucket.

def _fft_len(n):
    return int(next_fast_len(max(1, 2 * int(n) - 1)))


def _fft_len_grad(n):
    return int(next_fast_len(max(1, 3 * int(n) - 2)))


def conv_sq(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = _fft_len(n)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def internal_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = conv_sq(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (s * s))


def smooth_max_grad(p, T_factor):
    """Return (gradient, c, current_max, surrogate_value)."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0), np.empty(0), 0.0, 0.0
    L = _fft_len_grad(n)
    Fp = fft(p, L)
    c = np.real(ifft(Fp * Fp))[:2 * n - 1]
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.zeros(n, dtype=np.float64), c, 0.0, 0.0
    T = max(float(T_factor) * cmax, 1e-15)
    diff = (c - cmax) / T
    # Clip for numerical stability
    diff = np.clip(diff, -50.0, 50.0)
    e = np.exp(diff)
    raw = float(e.sum())
    if not np.isfinite(raw) or raw <= 0.0:
        e = np.zeros_like(c)
        e[int(np.argmax(c))] = 1.0
        raw = 1.0
    w = e / raw
    # Gradient: g_j = 2 * sum_k w_k * p_{k-j} (0-valid)
    # This is 2 * (reverse_conv(w, p))_j
    # w has length 2n-1, p has length n
    # (w * p)_j = sum_m w_m * p_{m+j} ... let's use FFT
    # Actually: sum_k w_k * p_{k-j} for 0<=k-j<n, 0<=k<2n-1
    # = sum_m w_{m+j} * p_m for 0<=m<n, 0<=m+j<2n-1
    # This is a cross-correlation-like operation
    # Let's compute via FFT: take w padded to length L, multiply by conjugate of p padded, ifft
    Fw = fft(w[::-1], L)
    h = np.real(ifft(Fp * Fw))
    if n == 1:
        G = h[:1]
    else:
        # We need h indexed such that G_j = sum_k w_k * p_{k-j}
        # h[t] = sum_i Fp^{-1}[i] * Fw^{-1}[i] ... 
        # Actually h = ifft(fft(p) * fft(w_rev)) gives circular cross-correlation
        # For linear: we need h[j] = sum_k p[k] * w_rev[j-k] = sum_k p[k] * w[j-k] (circular)
        # Hmm, let me just do it directly for correctness when n is small, FFT when large
        if n <= 512:
            G = np.zeros(n, dtype=np.float64)
            for j in range(n):
                # G_j = sum_k w_k * p_{k-j} where 0<=k-j<n and 0<=k<2n-1
                # Let m = k-j, so k = m+j, and 0<=m<n, 0<=m+j<2n-1
                idx = np.arange(n)
                k_idx = idx + j
                valid = k_idx < (2 * n - 1)
                G[j] = np.dot(w[k_idx[valid]], p[idx[valid]])
        else:
            # FFT approach: pad w to length L, compute ifft(fft(p)*fft(w_rev))
            # w_rev[j] = w[L-1-j] for circular
            G = h[2 * n - 2:n - 2:-1] if n > 1 else h[:1]
            if G.size != n:
                tmp = np.zeros(n, dtype=np.float64)
                m = min(n, G.size)
                tmp[:m] = G[:m]
                G = tmp
    G = np.maximum(np.asarray(G, dtype=np.float64), 0.0)
    g = 2.0 * G
    surr = float(cmax + T * np.log(raw))
    return g, c.astype(np.float64, copy=False), cmax, surr


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or p.size == 0 or s <= 1e-12:
        n = max(1, int(p.size))
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s


def make_uniform(n):
    n = int(max(1, n))
    return np.full(n, 1.0 / n, dtype=np.float64)


def make_geometric(n, s, reverse=False):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = np.exp(-float(s) * x)
    if reverse:
        p = p[::-1]
    return safe_normalize(p)


def make_power(n, a, reverse=False):
    n = int(max(2, n))
    x = np.linspace(1e-6, 1.0, n)
    p = np.power(x, float(a))
    if reverse:
        p = p[::-1]
    return safe_normalize(np.maximum(p, 0.0))


def make_cup(n, a):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 + float(a) * (2.0 * x - 1.0) ** 2
    return safe_normalize(np.maximum(p, 0.01))


def make_cap(n, a):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 - float(a) * (2.0 * x - 1.0) ** 2
    return safe_normalize(np.maximum(p, 0.01))


def make_gauss(n, sigma, mu=0.5):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = np.exp(-((x - float(mu)) ** 2) / (2.0 * max(1e-6, float(sigma)) ** 2))
    return safe_normalize(np.maximum(p, 1e-10))


def make_two_gauss(n, mu1, s1, w1, mu2, s2, w2):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = float(w1) * np.exp(-((x - mu1) ** 2) / (2.0 * max(1e-6, float(s1)) ** 2))
    p += float(w2) * np.exp(-((x - mu2) ** 2) / (2.0 * max(1e-6, float(s2)) ** 2))
    return safe_normalize(np.maximum(p, 1e-10))


def make_wiggle(n, freq, amp, phase=0.0):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = 1.0 + float(amp) * np.sin(2.0 * np.pi * float(freq) * x + float(phase))
    p = np.maximum(p, 0.02)
    return safe_normalize(p)


def make_beta_dist(n, a, b):
    n = int(max(2, n))
    x = np.linspace(1e-5, 1.0 - 1e-5, n)
    p = np.power(x, float(a)) * np.power(1.0 - x, float(b))
    return safe_normalize(np.maximum(p, 1e-10))


def make_skew(n, skew):
    """Asymmetric shape: skewed to right or left."""
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = np.exp(-float(skew) * x * (1.0 - x) ** 2 * (3.0 / (x + 0.01) ** 2))
    return safe_normalize(np.maximum(p, 1e-10))


def make_random_smooth(n, rng, strength=1.0):
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    p = 1.0
    for _ in range(3):
        f = int(rng.integers(1, max(2, n // 50)))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        p += 0.4 * float(strength) / (_ + 1) * np.sin(2.0 * np.pi * f * x + ph)
    p = np.maximum(p, 0.02)
    return safe_normalize(p)

def lp_best_response(p, best_score, rng, budget_s):

        return p, best_score
    f = f / fs
    end = time.monotonic() + max(0.5, float(budget_s))

        return p, best_score
    try:
        mid = np.arange(n, dtype=np.int32)
        rows = (np.arange(n, dtype=np.int32)[:, None] + mid[None, :]).ravel().astype(np.int32)
        cols = np.repeat(np.arange(n, dtype=np.int32), n)
        data = np.tile(f, n).astype(np.float64, copy=False)
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=(0.0, None), method="highs",
                      options={"disp": False, "time_limit": max(1.0, end - time.monotonic())})
    except Exception:
        return p, best_score
    if res is None or getattr(res, 'x', None) is None:
        return p, best_score

    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-12:
        return p, best_score
    g = g / s
    for t in (1.0, 0.8, 0.6, 0.4, 0.2, 0.1):
        q = project_simplex((1.0 - t) * p + t * g)
        sc = internal_score(q)
        if np.isfinite(sc) and sc < best_score:

def optimize_local(p0, rng, time_limit, label=""):
    """Main local optimizer with adaptive temperature and peak-targeting."""
    p = project_simplex(np.asarray(p0, dtype=np.float64))
    n = int(p.size)
    if n == 0:
        return p, np.inf
    best_p = p.copy()
    best_score = internal_score(best_p)
    current_score = best_score
    prev_p = p.copy()
    end = time.monotonic() + max(0.0, float(time_limit))
    T_factor = 0.15
    iter_no = 0
    since_improve = 0
    # Adaptive temperature: start moderate, decrease, occasionally increase for exploration
    T_mult = 1.0

    # Initial LP if short enough
    if n <= LP_NMAX_VAL and time_limit > 40:
        lp_budget = min(30.0, float(time_limit) * 0.12)
        p_lp, sc_lp = lp_best_response(p, best_score, rng, lp_budget)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            current_score = sc_lp
            best_p = p_lp.copy()
            p = p_lp.copy()
            prev_p = p_lp.copy()

    while time.monotonic() < end:
        iter_no += 1
        # Adaptive temperature schedule
        progress = min(1.0, float(iter_no) / max(1, int(time_limit * 3)))
        T_factor = 0.15 * (1.0 - 0.8 * progress) * T_mult
        if since_improve > 200:
            T_mult = min(3.0, T_mult * 1.1)
        else:
            T_mult = max(0.3, T_mult * 0.97)
        T_factor = max(T_factor, 0.005)

        g, c_cur, cmax, surr = smooth_max_grad(p, T_factor)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n, dtype=np.float64)

        accepted = False
        pmax = float(np.max(p)) if n > 0 else 0.0
        p_scale = max(1.0 / float(n), 0.1 * pmax, 1e-12)
        gabs = float(np.max(np.abs(g))) if g.size else 0.0

        if gabs > 1e-300:
            base = p_scale / gabs
            lrs = (0.6 * base, 0.15 * base, 0.03 * base, 0.006 * base, 0.001 * base, 0.0002 * base)
        else:
            lrs = (0.1, 0.03, 0.01, 0.003)

        momentum = 0.0
        if iter_no > 100:
            momentum = 0.3

        for lr in lrs:
            if not np.isfinite(lr) or lr <= 0.0:
                continue
            lr = min(max(lr, 1e-10), 50.0)
            v = p - lr * g
            if momentum > 0.0 and np.all(np.isfinite(prev_p)):
                v = v + momentum * (p - prev_p)
            pt = project_simplex(v)
            sc = internal_score(pt)
            c_t = conv_sq(pt)
            cmax_t = float(np.max(c_t)) if c_t.size else 0.0
            _, _, _, surr_t = smooth_max_grad(pt, T_factor)

            accept = False
            if np.isfinite(sc) and np.isfinite(surr_t):
                if surr_t < surr - 1e-10 * max(1.0, abs(surr)):
                    accept = True
                elif sc < best_score - 1e-12:
                    if surr_t < surr + 1.0 * max(1.0, abs(surr)):
                        accept = True
                elif sc < current_score - 1e-12:
                    if surr_t < surr + 0.5 * max(1.0, abs(surr)):
                        accept = True
            if accept:
                if sc < best_score:
                    best_score = sc
                    best_p = pt.copy()
                    since_improve = 0
                else:
                    since_improve += 1
                current_score = sc
                prev_p = p
                p = pt
                accepted = True
                break

        if not accepted:
            # Peak-targeting move: find which lag has max, try to reduce it
            if n > 2 and np.all(np.isfinite(c_cur)) and c_cur.size > 0:
                peak_lag = int(np.argmax(c_cur))
                # The peak is at lag peak_lag: c[peak_lag] = sum_i p_i * p[peak_lag - i]
                # To reduce it, we want to decrease products at this lag
                # Strategy: find indices i where p_i * p[peak_lag - i] is large
                # and transfer mass from those to indices that don't contribute to this peak
                if 0 < peak_lag < 2 * n - 2:
                    # Find the top contributing pairs
                    contrib = np.zeros(n, dtype=np.float64)
                    for i in range(n):
                        j = peak_lag - i
                        if 0 <= j < n:
                            contrib[i] = p[i] * p[j]
                    # Sort by contribution
                    order = np.argsort(contrib)[::-1]
                    # Try reducing the top contributor
                    if order[0] > 0 and contrib[order[0]] > 1e-15:
                        hi = int(order[0])
                        # Find a target that doesn't contribute to this peak
                        # A good target is one where both it and (peak_lag - it) have low p
                        targets = np.zeros(n, dtype=np.float64)
                        for i in range(n):
                            j = peak_lag - i
                            if 0 <= j < n:
                                targets[i] = p[i] * p[j]
                            else:
                                targets[i] = 0.0  # no contribution to this peak
                        # Pick target with zero contribution and some capacity
                        zero_contrib = np.where(targets < 1e-10)[0]
                        if len(zero_contrib) > 0:
                            # Pick the one with least current p (to avoid creating new peaks)
                            lo = int(zero_contrib[np.argmin(p[zero_contrib])])
                        else:
                            lo = int(np.argmin(p))
                        if hi != lo:
                            for frac in (0.15, 0.05, 0.02, 0.005):
                                delta = frac * p[hi]
                                if delta <= 0:
                                    continue
                                pt = p.copy()
                                pt[hi] -= delta
                                pt[lo] += delta
                                pt = project_simplex(pt)
                                sc = internal_score(pt)
                                if np.isfinite(sc) and sc < best_score:
                                    best_score = sc
                                    best_p = pt.copy()
                                    since_improve = 0
                                    current_score = sc
                                    prev_p = p
                                    p = pt
                                    accepted = True
                                    break
                                elif np.isfinite(sc) and sc < current_score and rng.random() < 0.1:
                                    current_score = sc
                                    prev_p = p
                                    p = pt
                                    accepted = True
                                    break

        if not accepted:
            # Random perturbation
            sigmas = (0.5 / n, 0.1 / n, 0.02 / n, 0.004 / n)
            for sigma in sigmas:
                pt = project_simplex(p + rng.normal(0.0, float(sigma), n))
                sc = internal_score(pt)
                if np.isfinite(sc) and (sc < best_score or (sc < current_score and rng.random() < 0.05)):
                    if sc < best_score:
                        best_score = sc
                        best_p = pt.copy()
                        since_improve = 0
                    current_score = sc
                    prev_p = p
                    p = pt
                    accepted = True
                    break

        if not accepted:
            # Pair swap
            if n > 2:
                hi = int(np.argmax(p))
                lo = int(np.argmin(p))
                if hi != lo and p[hi] > p[lo] + 1e-15:
                    for frac in (0.1, 0.03, 0.01):
                        delta = frac * p[hi]
                        pt = p.copy()
                        pt[hi] -= delta
                        pt[lo] += delta
                        pt = project_simplex(pt)
                        sc = internal_score(pt)
                        if np.isfinite(sc) and sc < best_score:
                            best_score = sc
                            best_p = pt.copy()
                            since_improve = 0
                            current_score = sc
                            prev_p = p
                            p = pt
                            accepted = True
                            break

        if not accepted:
            since_improve += 1
            # Periodic restart
            if since_improve > 400:
                alpha = 0.3 + 0.4 * rng.random()
                p = project_simplex(alpha * best_p + (1 - alpha) * p + rng.normal(0.0, 0.3 / n, n))
                current_score = internal_score(p)
                prev_p = p
                T_mult = 2.0
                since_improve = 0
            elif since_improve > 150:
                p = project_simplex(0.8 * best_p + 0.2 * p + rng.normal(0.0, 0.05 / n, n))
                current_score = internal_score(p)
                prev_p = p

        # Periodic LP for short sequences
        if n <= LP_NMAX_VAL and iter_no % 2000 == 0 and (end - time.monotonic()) > 15.0:
            p_lp, sc_lp = lp_best_response(p, best_score, rng, min(30.0, end - time.monotonic()))
            if np.isfinite(sc_lp) and sc_lp < best_score:
                best_score = sc_lp
                current_score = sc_lp
                best_p = p_lp.copy()
                p = p_lp.copy()
                prev_p = p_lp.copy()
                since_improve = 0

        if iter_no % 3000 == 0:
            print(label + " iter=%d best=%.10f cur=%.10f T=%.4f" % (iter_no, best_score, current_score, T_factor), flush=True)

        if best_score < 1.4990 and since_improve > 8000:
            break

    best_p = project_simplex(np.asarray(best_p, dtype=np.float64))
    return best_p, float(best_score) if np.isfinite(best_score) else np.inf


LP_NMAX_VAL = 600


def shape_parameterize(theta, n):
    """Build shape from Chebyshev + endpoint logs."""
    global _SHAPE_N, _SHAPE_X
    n = int(max(16, n))
    x = np.linspace(0.0, 1.0, n)
    D = 8
    theta = np.asarray(theta, dtype=np.float64)
    need = D + 2
    if theta.size < need:
        theta = np.concatenate((theta, np.zeros(need - theta.size)))
    else:
        theta = theta[:need].copy()
    logp = np.zeros(n, dtype=np.float64)
    z = 2.0 * x - 1.0
    T_nm = np.ones(n)
    T_n = z.copy()
    if D >= 1:
        logp += theta[0] * T_n
    for d in range(2, D + 1):
        T_np = 2.0 * z * T_n - T_nm
        logp += theta[d - 1] * T_np
        T_nm = T_n
        T_n = T_np
    eps = 1e-4
    logp += theta[D] * np.log(x + eps)
    logp += theta[D + 1] * np.log(1.0 - x + eps)
    if not np.all(np.isfinite(logp)):
        return make_uniform(n)
    logp -= np.max(logp)
    p = np.exp(logp)
    return safe_normalize(p)


def shape_objective(theta, n):
    try:
        p = shape_parameterize(theta, n)
        return internal_score(p)
    except Exception:
        return 1e100


def search_shapes(n, rng, budget_s):
    """Global shape search using differential evolution + Nelder-Mead."""
    if budget_s <= 10.0:
        return None, np.inf
    D = 8
    dim = D + 2
    best_p = None
    best_score = np.inf
    end = time.monotonic() + max(0.0, float(budget_s))

    def update(theta_c):
        nonlocal best_p, best_score
        try:
            sc = shape_objective(theta_c, n)
            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_p = shape_parameterize(theta_c, n)
        except Exception:
            pass

    # Initial guesses
    starts = [np.zeros(dim)]
    starts.append(rng.uniform(-1.5, 1.5, dim))
    starts.append(rng.uniform(-1.5, 1.5, dim))
    t = np.zeros(dim)
    t[D] = 2.5
    t[D + 1] = 2.5
    starts.append(t)
    t = np.zeros(dim)
    t[0] = 2.0
    t[1] = -1.0
    starts.append(t)
    t = np.zeros(dim)
    t[D] = -1.5
    t[D + 1] = -1.5
    starts.append(t)

    for x0 in starts:
        if time.monotonic() > end - 2.0:
            break
        update(x0)

    if _HAVE_MIN:
        bounds = [(-3.0, 3.0)] * D + [(-4.0, 5.0), (-4.0, 5.0)]
        # Differential evolution
        de_time = max(0.0, end - time.monotonic())
        if de_time > 30.0:
            try:
                maxiter = min(25, int(de_time / 15))
                res = differential_evolution(
                    lambda t: shape_objective(t, n),
                    bounds, maxiter=max(maxiter, 5), popsize=15,
                    tol=0.3, seed=int(rng.integers(0, 2**31 - 1)),
                    polish=False, init="latinhypercube",
                )
                if res is not None and np.isfinite(getattr(res, "fun", np.inf)):
                    update(res.x)
                    starts.append(np.asarray(res.x, dtype=np.float64))
            except Exception:
                pass

        # Nelder-Mead polish from best starts
        for x0 in starts[:8]:
            if time.monotonic() > end - 3.0:
                break
            try:
                res = minimize(
                    lambda t: shape_objective(t, n),
                    np.asarray(x0, dtype=np.float64),
                    method="Nelder-Mead",
                    options={"maxiter": 300, "xatol": 1e-5, "fatol": 1e-7},
                )
                if res is not None and np.isfinite(getattr(res, "fun", np.inf)):
                    update(res.x)
            except Exception:
                pass

    return best_p, float(best_score) if np.isfinite(best_score) else np.inf


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search for non-negative step function minimizing evaluate_sequence score."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 15.0)

    # Load parent if available
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            pa = np.asarray(height_sequence_1, dtype=np.float64)
            pa = np.where(np.isfinite(pa), pa, 0.0)
            pa = np.maximum(pa, 0.0)
            if pa.ndim == 1 and pa.size >= 32 and pa.sum() > 0:
                parent = pa
    except Exception:
        pass

    # Determine lengths to try
    if parent is not None:
        base_len = int(parent.size)
    else:
        base_len = 4096
    lengths = set()
    lengths.add(base_len)
    lengths.add(max(512, base_len // 2))
    lengths.add(min(8192, base_len * 2))
    lengths.add(2048)
    lengths.add(1024)
    lengths = sorted([l for l in lengths if 64 <= l <= 16384])

    best_p = None
    best_score = np.inf

    def try_record(p, sc):
        nonlocal best_p, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64)
        if p.size == 0 or not np.all(np.isfinite(p)):
            return
        p = safe_normalize(p)
        if not np.isfinite(sc):
            sc = internal_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()
            print("NEW BEST: score=%.10f n=%d" % (best_score, p.size), flush=True)

    # Phase 1: Quick shape search at moderate length
    if deadline - time.monotonic() > 120.0:
        shape_n = min(1024, max(256, lengths[-1] // 4))
        shape_budget = min(120.0, (deadline - time.monotonic()) * 0.15)
        shape_p, shape_sc = search_shapes(shape_n, rng, shape_budget)
        try_record(shape_p, shape_sc)
        print("Shape search done: score=%s" % shape_sc, flush=True)

    # Phase 2: Score initial candidates at each length
    all_jobs = []
    for n_len in lengths:
        if deadline - time.monotonic() <= 60.0:
            break
        inits = generate_initials(n_len, rng, parent if (parent is not None and parent.size == n_len) else None)
        if shape_p is not None:
            # Upsample/reshape shape to this length
            sp = np.interp(np.linspace(0, 1, n_len), np.linspace(0, 1, len(shape_p)), shape_p)
            inits.append(safe_normalize(sp))
        scored = []
        for inp in inits:
            inp = safe_normalize(inp)
            sc = internal_score(inp)
            if np.isfinite(sc):
                scored.append((sc, inp))
                try_record(inp, sc)
        scored.sort(key=lambda x: x[0])
        for sc, inp in scored[:4]:
            allow_lp = (n_len <= LP_NMAX_VAL)
            all_jobs.append((sc, inp, allow_lp, n_len))

    all_jobs.sort(key=lambda x: x[0])

    # Phase 3: Local optimization on best jobs
    n_jobs = max(1, min(len(all_jobs), 8))
    final_reserve = max(30.0, (deadline - start_time) * 0.1)
    available = max(0.0, (deadline - final_reserve) - time.monotonic())
    time_per = max(20.0, min(300.0, available / float(n_jobs)))

    for j in range(n_jobs):
        if time.monotonic() > deadline - final_reserve:
            break
        sc, p, allow_lp, n_len = all_jobs[j]
        remaining = max(0.0, deadline - final_reserve - time.monotonic())
        if remaining < 20.0:
            break
        limit = max(15.0, min(time_per, remaining))
        label = "n=%d job%d" % (n_len, j)
        p_best, sc_best = optimize_local(p, rng, limit, label + ": ", allow_lp)
        try_record(p_best, sc_best)
        all_jobs[j] = (sc_best, p_best, allow_lp, n_len)
        print("Job n=%d done: score=%.10f" % (n_len, sc_best), flush=True)

    # Phase 4: Final polish
    if best_p is not None and time.monotonic() < deadline - 10.0:
        limit = max(5.0, deadline - time.monotonic() - 5.0)
        p_final, sc_final = optimize_local(best_p, rng, limit, "FINAL: ", best_p.size <= LP_NMAX_VAL)
        try_record(p_final, sc_final)

    # Fallback
    if best_p is None:
        best_p = make_uniform(4096)
        best_score = internal_score(best_p)

    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    s = float(best_p.sum())
    if s <= 1e-12:
        best_p = make_uniform(4096)
    best_p = np.minimum(best_p, 1000.0)
    print("FINAL: score=%.10f n=%d" % (best_score, len(best_p)), flush=True)
    return [float(x) for x in best_p]
