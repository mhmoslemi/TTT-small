import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def simplex_project(v, target_sum=1.0):
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0)

def compute_grad_lse(a, conv, t):
    """Compute gradient of LSE(t*conv(a)) w.r.t. a"""
    n = len(a)
    scaled = t * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0

        p /= p_sum

    L = 1 << (3 * n).bit_length()
    P = np.fft.rfft(p, L)
    A = np.fft.rfft(a, L)
    corr = np.fft.irfft(np.conj(P) * A, L)

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1

def simplex_project(v, target_sum=1.0):
    """Project v onto {x: sum(x)=target_sum, x>=0}."""
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0)


def compute_smooth_gradient(a, conv, t, L):
    """
    Compute gradient of LSE(t*conv(a)) w.r.t. a.
    Returns gradient vector of length n.
    Uses FFT cross-correlation for efficiency.
    """

    scaled = t * conv
    scaled -= scaled.max()
    p = np.exp(scaled)

    if p_sum < 1e-300:

        p /= p_sum

            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                return pp / pp.sum()
        return None

    # Phase 1: FISTA scan
    print("Phase 1: FISTA scan...")
    lengths = [128, 256, 512]
    init_types = ['uniform', 'cosine', 'gauss', 'bimodal', 'random', 'hamming', 'sinc_like']

    configs = []
    for n in lengths:
        for it in init_types:
            a0 = get_init(n, it, rng)
            configs.append((n, it, a0))
        # Parent
        ap = try_parent(n, rng)
        if ap is not None:
            configs.append((n, 'parent', ap))

    n_configs = len(configs)
    phase1_budget = min(600.0, (deadline - time.monotonic()) * 0.4)
    time_per_config = max(5.0, phase1_budget / n_configs)

    scan_results = []
    for i, (n, name, a0) in enumerate(configs):
        if time.monotonic() > deadline - 120:
            break
        t_alloc = min(time_per_config, deadline - time.monotonic() - 120)
        if t_alloc < 3:
            break

        a_opt, obj_opt = fista_optimize(a0, time_limit=t_alloc, t_start=5.0, t_growth=1.5)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  FISTA: n={n}, {name}, score={score:.6f}")

    print(f"Phase 1 done. Best: {best_score:.6f}")

    # Phase 2: Deep optimization with FISTA + multiplicative polish
    print("Phase 2: Deep optimization...")
    scan_results.sort(key=lambda x: x[0])

    # Select top configs with diversity
    selected = []
    seen_n = set()
    for score, n, name, a_opt in scan_results:
        if len(selected) < 8:
            selected.append((n, name, a_opt))
            seen_n.add(n)
        elif n not in seen_n and len(selected) < 10:
            selected.append((n, name, a_opt))
            seen_n.add(n)

    # Always include current best
    if best_sequence is not None:
        selected.append((len(best_sequence), 'best', best_sequence.copy()))

    remaining = deadline - time.monotonic() - 60
    time_per_deep = max(15.0, remaining / max(1, len(selected)))

    for i, (n, name, a0) in enumerate(selected):
        if time.monotonic() > deadline - 40:
            break
        t_alloc = min(time_per_deep, deadline - time.monotonic() - 40)
        if t_alloc < 5:
            break

        # FISTA phase
        t_fista = t_alloc * 0.6
        a_opt, obj_opt = fista_optimize(a0, time_limit=t_fista, t_start=10.0, t_growth=1.4,
                                        max_per_phase=2000)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep FISTA: n={n}, {name}, score={score:.6f}")

        # Multiplicative polish
        t_mpd = t_alloc * 0.4
        if t_mpd > 3:
            for t_p in [30.0, 80.0, 150.0]:
                if time.monotonic() > deadline - 40:
                    break
                t_single = min(10.0, t_mpd / 3.0)
                a_polish, obj_polish = multiplicative_polish(a_opt, time_limit=t_single,
                                                             t_param=t_p, eta0=0.25)
                score = float(evaluate_sequence(a_polish.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_polish.copy()
                    a_opt = a_polish
                    obj_opt = obj_polish
                    print(f"  Polish: n={n}, t={t_p:.0f}, score={score:.6f}")

    # Phase 3: Final intensive polish
    print(f"Phase 3: Final polish. Best so far: {best_score:.6f}")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            n = len(best_sequence)
            # Try very high temperature FISTA
            t_f = remaining * 0.5
            a_opt, obj_opt = fista_optimize(best_sequence, time_limit=t_f,
                                            t_start=50.0, t_growth=1.3, max_per_phase=1000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final FISTA: score={score:.6f}")

            # Multiplicative with various params
            t_m = remaining * 0.5
            for t_p in [100.0, 200.0, 400.0]:
                if time.monotonic() > deadline - 10:
                    break
                t_single = min(15.0, t_m / 3.0)
                a_polish, obj_polish = multiplicative_polish(best_sequence, time_limit=t_single,
                                                             t_param=t_p, eta0=0.15)
                score = float(evaluate_sequence(a_polish.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_polish.copy()
                    print(f"  Final Polish t={t_p}: score={score:.6f}")
                t_m = deadline - 10 - time.monotonic()

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"FINAL SCORE: {best_score:.6f}")
    return best_sequence


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses FISTA with log-sum-exp smoothing combined with
    multiplicative mirror descent polishing to minimize 2n*max(conv)/(sum)^2.
    Multi-phase: scan, deep optimize, polish.
    """
    result = run_full_search(seed=seed, budget_s=budget_s)
    return [float(x) for x in result]
