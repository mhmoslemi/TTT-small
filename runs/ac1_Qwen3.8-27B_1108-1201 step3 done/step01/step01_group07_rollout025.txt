"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a 2-block (bimodal) initialization
that achieves ~1.488 for n=128, combined with subgradient descent on the simplex,
multiplicative mirror descent, and local coordinate search. Explores multiple

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def simplex_project(v):
    """Project v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_2block(n):
    """Create a 2-block (bimodal) sequence of length n with optimal proportions."""
    m = (n + 1) // 3
    while m > 1 and n - 2 * m < m - 1:
        m -= 1
    if m < 1:
        m = 1
    a = np.zeros(n)
    a[:m] = 1.0 / (2.0 * m)
    a[n - m:] = 1.0 / (2.0 * m)
    return a


def make_smooth_2block(n, alpha=0.3):
    """Create a 2-block sequence with smooth (tapered) edges."""
    a = make_2block(n)
    m = (n + 1) // 3
    while m > 1 and n - 2 * m < m - 1:
        m -= 1
    # Apply a smooth taper at the edges of each block
    taper_len = max(1, int(alpha * m))
    for start, end in [(0, m), (n - m, n)]:
        for i in range(taper_len):
            frac = (i + 1) / (taper_len + 1)
            a[start + i] *= 0.5 * (1.0 - np.cos(np.pi * frac))
            a[end - 1 - i] *= 0.5 * (1.0 - np.cos(np.pi * frac))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-12:
        a /= s
    return a


def subgradient_descent(a0, time_limit, t_smooth=20.0, eta0=0.1, momentum=0.9):
    """Subgradient descent with simplex projection and momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    t0 = time.monotonic()
    best_obj = float('inf')
    best_a = a.copy()
    eta = eta0
    v = np.zeros(n)  # momentum
    patience = 0
    max_patience = 50

    conv = fast_convolve(a)

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        # Soft-max weights over convolution peaks
        scaled = t_smooth * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        # Compute subgradient: sg[j] = sum_k w[k] * 2 * a[k-j] (valid range)
        sg = np.zeros(n)
        thresh = w.max() * 0.005
        active_idx = np.where(w > thresh)[0]

        for k_int in active_idx:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                # sg[j] += w[k] * 2 * a[k-j] for j in [j_start, j_end]
                # a[k-j] for j=j_start..j_end = a[k-j_start], a[k-j_start-1], ...
                sg[j_start:j_end + 1] += w[k] * 2.0 * a[k - j_end:k - j_start + 1][::-1]

        # Momentum update
        v = momentum * v + (1.0 - momentum) * sg
        v_centered = v - v.mean()

        # Step and project
        a_new = simplex_project(a - eta * v_centered)

        # Evaluate
        conv = fast_convolve(a_new)
        current_obj = 2.0 * n * conv.max()

        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a_new.copy()
            patience = 0
        else:
            a = a_new  # still update a for exploration
            patience += 1

        if patience >= max_patience:
            patience = 0
            eta *= 0.5
            t_smooth *= 1.5
            max_patience = max(20, max_patience - 5)
            if eta < 0.001:
                break

    return best_a, best_obj


def mirror_descent(a0, time_limit, t_smooth=40.0, eta0=0.5):
    """Multiplicative mirror descent (exponential weights) on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    t0 = time.monotonic()
    best_obj = float('inf')
    best_a = a.copy()
    eta = eta0
    patience = 0
    max_patience = 60
    current_t = t_smooth

    conv = fast_convolve(a)

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        scaled = current_t * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        thresh = w.max() * 0.003
        active_idx = np.where(w > thresh)[0]

        for k_int in active_idx:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                sg[j_start:j_end + 1] += w[k] * a[k - j_end:k - j_start + 1][::-1]

        sg_centered = sg - sg.mean()
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.ones(n) / n
        else:
            a /= a_sum

        conv = fast_convolve(a)
        current_obj = 2.0 * n * conv.max()

        if current_obj < best_obj - 1e-9:
            best_obj = current_obj
            best_a = a.copy()
            patience = 0
        else:
            patience += 1

        if patience >= max_patience:
            patience = 0
            eta *= 0.7
            current_t = np.clip(current_t * 1.3, 5.0, 300.0)
            if eta < 0.01:
                break

    return best_a, best_obj


def local_search(a0, time_limit, n_iter_per_elem=3, delta_scale=0.01):
    """Coordinate-wise local search on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)

    if s < 1e-12:

    # Iterate through elements in random order

    idx = 0

    while time.monotonic() - t0 < time_limit:

                a_trial[j] += delta
                # Redistribute: reduce other elements proportionally

                non_j[j] = 0.0
                excess = a_trial[j] - a[j]
                a_trial = a.copy()

                others_sum = a.sum() - a[j]
                if others_sum > 1e-12:
                    a_trial = a.copy()
                    a_trial[j] = a[j] + excess
                    # Scale down others to maintain sum=1
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a_trial[mask] = a_trial[mask] * (1.0 - a[j]) / others_sum if others_sum > 0 else 0
                    # Simpler: just subtract proportionally

                    a_trial[j] = a[j] + excess
                    rest = a.copy()
                    rest[j] = 0.0
                    rest_sum = rest.sum()
                    if rest_sum > 1e-12:

                    a_trial = np.maximum(a_trial, 0.0)

                        a_trial /= ts

                    obj = compute_obj(a_trial)
                    if obj < best_obj - 1e-10:
                        best_obj = obj
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        break

            # Try decreasing (if a[j] is large enough)
            if a[j] > delta * 2:
                a_trial = a.copy()
                a_trial[j] = a[j] - delta
                # Redistribute the freed mass to other elements proportionally
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                rest_sum = a.sum() - a[j]
                if rest_sum > 1e-12:
                    a_trial[mask] = a[mask] + delta * a[mask] / rest_sum

                    obj = compute_obj(a_trial)

def run_optimization(a0, time_limit):

    best_a = a.copy()
    best_obj = compute_obj(a)

    t0 = time.monotonic()
