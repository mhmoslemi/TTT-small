   ```python
   import numpy as np
   import time
   from scipy.fft import fft, ifft
   from scipy.optimize import minimize

   def fft_convolve_self(a):
       n = len(a)
       size = 2 * n - 1
       fsize = 1 << (size - 1).bit_length()
       A = fft(a, fsize)
       c = np.real(ifft(A * A))
       return c[:size]

   def softmax_obj_and_grad(a, beta):
       """Compute 2*n*softmax(c, beta)/sum(a)^2 and its gradient."""
       n = len(a)
       s = a.sum()
       c = fft_convolve_self(a)
       c_max = c.max()
       scaled = beta * (c - c_max)
       scaled = np.clip(scaled, -500, 0)  # clip upper to 0 for stability (since c-c_max <= 0)
       exp_s = np.exp(scaled)
       Z = exp_s.sum()
       soft_max_val = c_max + np.log(Z) / beta
       val = 2.0 * n * soft_max_val / (s * s)
       
       # Gradient of soft_max_val w.r.t. a
       w = exp_s / Z  # weights on each lag
       grad_sm = np.zeros(n)
       # Only process lags with significant weight
       active = np.where(w > 1e-10)[0]
       for k in active:
           wk = w[k]
           # d(c[k])/d(a[i]) = a[k-i] + a[i-k] (with bounds)
           # But we need to be careful: a[k-i] is non-zero only if 0 <= k-i < n
           # and a[i-k] is non-zero only if 0 <= i-k < n
           lo1 = max(0, k - n + 1)
           hi1 = min(n - 1, k)
           if lo1 <= hi1:
               idx = np.arange(lo1, hi1 + 1)
               grad_sm[lo1:hi1+1] += wk * a[k - idx]
           lo2 = max(0, k)
           hi2 = min(n - 1, k + n - 1)
           if lo2 <= hi2:
               idx2 = np.arange(lo2, hi2 + 1)
               grad_sm[lo2:hi2+1] += wk * a[idx2 - k]
       
       # grad of val = 2n * soft_max / s^2 w.r.t. a
       # d(val)/d(a[i]) = 2n * (grad_sm[i] / s^2 - 2 * soft_max * a[i] / s^3)
       grad = (2.0 * n * grad_sm / (s * s)) - (4.0 * n * soft_max_val * a / (s ** 3))
       return val, grad

   def run_lbfgs(a_init, beta, max_time, maxiter=500):
       """Run L-BFGS-B with soft-max objective."""
       a_init = np.clip(a_init, 0, 1000)
       if a_init.sum() < 0.01:
           a_init = np.ones(len(a_init))
       
       def fun(x):
           val, _ = softmax_obj_and_grad(x, beta)
           return val
       
       def grad(x):
           _, g = softmax_obj_and_grad(x, beta)
           return g
       
       bounds = [(0, 1000)] * len(a_init)
       t0 = time.monotonic()
       res = minimize(fun, a_init, jac=grad, method='L-BFGS-B', 
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 
                      'gtol': 1e-14, 'maxfun': 2000, 'maxls': 50})
       elapsed = time.monotonic() - t0
       best_a = res.x
       best_obj = res.fun
       return best_a, best_obj, elapsed
   ```
