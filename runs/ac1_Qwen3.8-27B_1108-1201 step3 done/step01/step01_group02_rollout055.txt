    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:

    """Project a onto {x : sum(x) = 1, x >= 0}"""
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)

    # a - lambda >= 0 and sum(a - lambda) = 1
    # => lambda = sum(a) - 1... but we need a - lambda >= 0
    # Better: sort and find threshold
    n = len(a)

    cumsum = np.cumsum(sorted_a)
    rho = np.where(sorted_a - (cumsum - 1) / np.arange(1, n+1) > 0)[0]

    rho = rho[-1]
    thresh = (cumsum[rho] - 1) / (rho + 1)
    proj = np.maximum(a - thresh, 0)
    return proj / proj.sum()  # ensure exact sum = 1

    return grad

- We want `|A(ω)|^2` (which is the DFT of the auto-convolution) to have a flat inverse DFT
- Equivalently, we want the auto-convolution to be flat
- In the frequency domain, `C(ω) = |A(ω)|^2` must be a non-negative trigonometric polynomial that factors as a square
- We can try to design `A(ω)` (the DFT of `a`) such that `|A(ω)|^2` has a flat inverse DFT
- This is related to the design of "flat-top" windows

Actually, the flat-top window (like the Harris or Nuttall window) has a very flat main lobe in the frequency domain, which means its auto-correlation has low sidelobes. But we want the opposite: we want the auto-convolution to be flat in the time domain.

Hmm, let me think about this differently. The DFT of the auto-convolution is `|A(ω)|^2`. If `|A(ω)|^2` is constant (say `= 1/n`), then the auto-convolution is a delta function (all at the center). That's the opposite of what we want.

If `|A(ω)|^2` has a specific shape, the auto-convolution will have the corresponding inverse DFT shape. We want the inverse DFT of `|A(ω)|^2` to be as flat as possible.

The inverse DFT of a constant is a delta. The inverse DFT of a "broad" function is "narrow". So we want `|A(ω)|^2` to be as "broad" (spread out in frequency) as possible, which means `A(ω)` should have energy spread across all frequencies.

For a non-negative sequence `a`, `A(ω)` is a polynomial with non-negative coefficients. The "broadest" such polynomial (in terms of frequency spread) would be one that's close to having equal magnitude at all frequencies. This is related to the concept of "equiripple" filters.

So the ideal solution would be a non-negative sequence whose DFT has (approximately) constant magnitude. This is the "flat spectrum" or "white noise" property. But for a finite non-negative sequence, perfect flatness is impossible (by the uncertainty principle and the non-negativity constraint).

The closest we can get is a sequence whose DFT magnitude is as flat as possible. This is related to the design of "minimum peak" auto-correlation sequences, which is a well-studied problem in radar and communications.

For our purposes, I think the most effective approach is a combination of:
1. Start from a good initial sequence (like the parent or a well-designed window)
2. Use gradient-based optimization (energy minimization + subgradient on max)
3. Polish with LP and local search

    """Project onto {x >= 0, sum(x) = 1} using water-filling."""
    n = len(a)
    if n == 0:
        return a

    css = np.cumsum(u)
    rho = np.arange(1, n + 1)

        return np.ones(n) / n

def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)

                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:

        pass
    return None

        else:
            stagnant += 1
            eta *= 0.9
            if stagnant > 100:
                break
        a = a_new

    fs = score_sequence(a)
    if fs < best_score:
        best_score = fs
        best_a = a.copy()
    if verbose:
        print(f"    Subgrad max: score={best_score:.6f}, time={time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def smoothed_max_minimize(a, time_limit=120.0, beta=50.0, verbose=False):
    """Minimize log-sum-exp of c_k as smooth approximation to max."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    eta = 0.5 / n
    stagnant = 0

    while time.monotonic() - start_time < time_limit:
        c = np.convolve(a, a)
        c_shifted = c - c.max()
        weights = np.exp(beta * c_shifted)
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        w = weights / wsum
        # Gradient of log-sum-exp: sum_k w_k * grad_k
        g = 2.0 * np.correlate(w, a, mode='valid')
        gscale = np.linalg.norm(g) / n + 1e-30
        a_new = a - eta * g / gscale * n
        a_new = project_onto_simplex(a_new)
        score = score_sequence(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            stagnant = 0
            eta = min(eta * 1.005, 3.0 / n)
        else:
            stagnant += 1
            eta *= 0.93
            if stagnant > 120:
                break
        a = a_new

    fs = score_sequence(a)
    if fs < best_score:
        best_score = fs
        best_a = a.copy()
    if verbose:
        print(f"    Smoothed max: score={best_score:.6f}, time={time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def solve_lp_step(a, M, timeout=30.0):
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=300.0, verbose=False):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    it = 0

    while time.monotonic() - start_time < time_limit_s:
        conv = np.convolve(a, a)
        max_b = conv.max()
        score = 2.0 * n * max_b / (a.sum() ** 2)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 40:
            break
        remaining = time_limit_s - (time.monotonic() - start_time)
        if remaining < 5:
            break
        found = False
        for eps in [0.0, 0.002, 0.008, 0.02, 0.05]:
            M_tight = max_b * (1.0 - eps)
            timeout = max(2.0, min(30.0, remaining * 0.25))
            g = solve_lp_step(a, M_tight, timeout=timeout)
            if g is None:
                continue
            sum_g = g.sum()
            if sum_g <= 1.0 + 1e-10:
                continue
            g = g / sum_g
            for alpha in [0.4, 0.6, 0.8, 1.0]:
                a_new = (1 - alpha) * a + alpha * g
                a_new = normalize_nonneg(a_new)
                new_score = score_sequence(a_new)
                if new_score < score:
                    a = a_new
                    found = True
                    break
            if found:
                break
        it += 1
        if verbose and it % 30 == 0:
            print(f"    LP iter {it}: score={score_sequence(a):.6f}")

    fs = score_sequence(a)
    if fs < best_score:
        best_score = fs
        best_a = a.copy()
    if verbose:
        print(f"    LP refine: score={best_score:.6f}, time={time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search(a, time_limit=120.0, rng=None, verbose=False):
    if rng is None:
        rng = np.random.default_rng(0)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    accept_count = 0

    while time.monotonic() - start_time < time_limit:
        width = max(2, int(rng.integers(n // 100, max(3, n // 6))))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        candidate[start:stop] *= np.exp(rng.normal(0.0, 0.08, size=stop - start))
        if rng.random() < 0.2:
            candidate[0:width] *= np.exp(rng.normal(0.0, 0.05, size=width))
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 30 == 0:
                print(f"    LS accept {accept_count}: score={best_score:.6f}")

    if verbose:
        print(f"    Local search: score={best_score:.6f}, time={time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def cyclic_shift_search(a, n_shifts=5, verbose=False):
    """Try cyclic shifts and find the best scoring one."""
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    for s in range(1, n_shifts + 1):
        shifted = np.roll(a, s)
        shifted = normalize_nonneg(shifted)
        sc = score_sequence(shifted)
        if sc < best_score:
            best_score = sc
            best_a = shifted.copy()
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: energy minimization, subgradient/smoothed max, LP refinement, local search.
    Uses diverse initializations including parent sequence.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(10.0, budget_s - 20.0)

    best_score = float('inf')
    best_seq = None

    def update_best(a, label=""):
        nonlocal best_score, best_seq
        sc = score_sequence(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - total_start
            print(f"  *** NEW BEST ({label}): score={sc:.6f}, n={len(a)}, elapsed={elapsed:.1f}s")
            return True
        return False

    # Collect initial sequences
    inits = []

    # Try parent sequence
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = normalize_nonneg(parent)
        if len(parent) > 10:
            inits.append((parent, "parent"))
            print(f"  Parent loaded: n={len(parent)}, score={score_sequence(parent):.6f}")
    except NameError:
        pass
    except Exception as e:
        print(f"  Could not load parent: {e}")

    # Other initializations
    init_configs = [
        (500, 'raised_cosine'),
        (400, 'gaussian'),
        (600, 'gaussian'),
        (500, 'blackman'),
        (400, 'asymmetric'),
        (500, 'skew'),
        (400, 'random'),
        (600, 'random'),
    ]

    for n, itype in init_configs:
        if itype == 'raised_cosine':
            a = initial_raised_cosine(n)
        elif itype == 'gaussian':
            a = initial_gaussian(n, sigma_frac=rng.uniform(0.2, 0.35))
        elif itype == 'blackman':
            a = initial_blackman(n)
        elif itype == 'asymmetric':
            a = initial_asymmetric(n)
        elif itype == 'skew':
            a = initial_skew(n, alpha=rng.uniform(1.5, 3.5))
        elif itype == 'random':
            a = random_init(n, rng, n_bumps=int(rng.integers(2, 6)))
        a = normalize_nonneg(a)
        inits.append((a, f"{itype}_{n}"))

    print(f"\nTotal initial sequences: {len(inits)}")
    for a, name in inits:
        print(f"  {name}: n={len(a)}, score={score_sequence(a):.6f}")

    # Pipeline function
    def run_pipeline(a, name, time_budget):
        nonlocal best_score, best_seq
        if time.monotonic() > deadline:
            return
        print(f"\n  --- Pipeline: {name}, budget={time_budget:.0f}s ---")
        t_start = time.monotonic()
        remaining = time_budget

        # Phase 1: Energy minimization (multiplicative update)
        t_rem = deadline - time.monotonic()
        if t_rem < 20:
            return
        t1 = min(remaining * 0.3, t_rem * 0.4, 200.0)
        a1, s1 = energy_minimize(a, time_limit=t1, verbose=True)
        update_best(a1, f"energy_{name}")
        remaining -= (time.monotonic() - t_start)
        t_start = time.monotonic()

        # Phase 2: Projection GD on energy
        t_rem = deadline - time.monotonic()
        if t_rem < 20:
            return
        t2 = min(remaining * 0.2, t_rem * 0.3, 120.0)
        a2, s2 = projection_gd_energy(a1, time_limit=t2, verbose=True)
        update_best(a2, f"projGD_{name}")
        a_best_local = a2 if s2 < s1 else a1
        remaining -= (time.monotonic() - t_start)
        t_start = time.monotonic()

        # Phase 3: Smoothed max minimization
        t_rem = deadline - time.monotonic()
        if t_rem < 20:
            return
        t3 = min(remaining * 0.25, t_rem * 0.35, 180.0)
        a3, s3 = smoothed_max_minimize(a_best_local, time_limit=t3, beta=rng.uniform(30, 80), verbose=True)
        update_best(a3, f"smooth_{name}")
        a_best_local = a3 if s3 < min(s1, s2) else a_best_local
        remaining -= (time.monotonic() - t_start)
        t_start = time.monotonic()

        # Phase 4: Subgradient on max
        t_rem = deadline - time.monotonic()
        if t_rem < 20:
            return
        t4 = min(remaining * 0.2, t_rem * 0.3, 120.0)
        a4, s4 = subgradient_max_minimize(a_best_local, time_limit=t4, verbose=True)
        update_best(a4, f"subgrad_{name}")
        a_best_local = a4 if s4 < s3 else a_best_local
        remaining -= (time.monotonic() - t_start)
        t_start = time.monotonic()

        # Phase 5: LP Refinement
        t_rem = deadline - time.monotonic()
        if t_rem < 15:
            return
        t5 = min(remaining * 0.5, t_rem * 0.5, 300.0)
        a5, s5 = lp_refine(a_best_local, time_limit_s=t5, verbose=True)
        update_best(a5, f"lp_{name}")
        remaining -= (time.monotonic() - t_start)
        t_start = time.monotonic()

        # Phase 6: Local search
        t_rem = deadline - time.monotonic()
        if t_rem < 10:
            return
        t6 = min(remaining * 0.3, t_rem * 0.4, 100.0)
        a6, s6 = local_search(a5, time_limit=t6, rng=rng, verbose=False)
        update_best(a6, f"ls_{name}")
        remaining -= (time.monotonic() - t_start)

        # Phase 7: Final LP polish
        t_rem = deadline - time.monotonic()
        if t_rem > 15:
            t7 = min(remaining * 0.5, t_rem * 0.5, 150.0)
            a7, s7 = lp_refine(a6, time_limit_s=t7, verbose=False)
            update_best(a7, f"lp_final_{name}")

        # Cyclic shift search
        a8, s8 = cyclic_shift_search(a_best_local if True else a6, n_shifts=min(10, len(a_best_local) // 10))
        update_best(a8, f"shift_{name}")

    # Sort inits by initial score (best first)
    inits.sort(key=lambda x: score_sequence(x[0]))

    # Run pipelines with time allocation
    n_inits = len(inits)
    # Give more time to better initial points
    for i, (a_init, name) in enumerate(inits):
        t_rem = deadline - time.monotonic()
        if t_rem < 30:
            break
        # Allocate more time to top inits
        frac = max(0.15, 1.0 - 0.15 * i / max(n_inits - 1, 1))
        t_budget = min(t_rem * 0.7, frac * 400.0)
        if t_budget < 30:
            t_budget = t_rem * 0.5
        run_pipeline(a_init.copy(), name, t_budget)

    # Final polishing phase on the best found
    if best_seq is not None:
        t_rem = deadline - time.monotonic()
        if t_rem > 30:
            print(f"\n  === Final Polish (remaining={t_rem:.0f}s) ===")
            # LP polish
            t_lp = min(t_rem * 0.4, 200.0)
            a_p, s_p = lp_refine(best_seq, time_limit_s=t_lp, verbose=True)
            update_best(a_p, "final_lp")
            best_seq = a_p if s_p < best_score else best_seq

            t_rem = deadline - time.monotonic()
            if t_rem > 20:
                # Local search polish
                t_ls = min(t_rem * 0.3, 100.0)
                a_ls, s_ls = local_search(best_seq, time_limit=t_ls, rng=rng, verbose=False)
                update_best(a_ls, "final_ls")
                best_seq = a_ls if s_ls < best_score else best_seq

            t_rem = deadline - time.monotonic()
            if t_rem > 15:
                # One more LP
                t_lp2 = min(t_rem * 0.6, 100.0)
                a_p2, s_p2 = lp_refine(best_seq, time_limit_s=t_lp2, verbose=False)
                update_best(a_p2, "final_lp2")
                best_seq = a_p2 if s_p2 < best_score else best_seq

    if best_seq is None:
        best_seq = normalize_nonneg(initial_raised_cosine(500))
        best_score = score_sequence(best_seq)

    print(f"\nFinal: score={best_score:.6f}, n={len(best_seq)}, total_time={time.monotonic()-total_start:.1f}s")
    verify = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified via evaluate_sequence: {verify:.6f}")

    return [float(x) for x in best_seq]
