```python
def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_obj_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def smooth_grad(a, conv, t):
    """Gradient of logsumexp(t*conv)/t w.r.t. a. conv = auto-convolution of a."""
    n = len(a)
    max_cv = conv.max()
    if max_cv < 1e-30:
        return np.zeros(n)
    
    scaled = t * (conv - max_cv)
    exp_s = np.exp(scaled)
    w = exp_s / exp_s.sum()
    
    sg = np.zeros(n)
    thresh = w.max() * 1e-5
    active = np.where(w > thresh)[0]
    
    for k in active:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += w[k] * a[k - je:k - js + 1][::-1]
    
    return sg

def mult_optimize(a0, time_limit, t_init=40.0, eta_init=0.5):
    """Multiplicative (exponential) mirror descent on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
```

Actually, let me think about a completely different approach: **alternating minimization**.

The objective `max_k (a*a)[k]` can be written as:
`min_{a >= 0, sum a = 1} max_k sum_j a_j * a_{k-j}`

By minimax theorem (Sion's theorem), this equals:
`max_{p in simplex} min_{a >= 0, sum a = 1} sum_k p_k * (a*a)[k]`
= `max_{p in simplex} min_{a >= 0, sum a = 1} a^T (sum_k p_k P_k) a`

where P_k is the shift matrix for lag k.

The inner problem is: minimize `a^T Q a` subject to `a >= 0, sum a = 1`, where `Q = sum_k p_k P_k`.

This is a quadratic program! And for a fixed p, it can be solved (approximately) using standard QP methods.

The outer problem is: maximize over p in the simplex the minimum value of the inner QP.

This is a bilevel optimization, but we can use a iterative approach:
1. Start with p = uniform (which corresponds to minimizing the average of all (a*a)[k])
2. Solve the inner QP to get a*
3. Update p to put more weight on the lags where (a***)[k] is largest
4. Repeat

This is essentially a cutting-plane / column-generation approach.

For the inner QP: minimize `a^T Q a` s.t. `a >= 0, sum a = 1`.

```python
def qprod(Q_p, a):
    """Compute Q*a where Q = sum_k p_k S_k, using FFT.
    Q_p is the vector p of length 2n-1.
    """
    n = len(a)
    # Q*a = conv(a, p[::-1])[2*n-2 : n-1 : -1]
    p_rev = Q_p[::-1]
    full_conv = fast_convolve_combined(a, p_rev)
    return full_conv[2*n-2 : n-1 : -1]

def fast_convolve_combined(a, b):
    """Linear convolution of a and b using FFT."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]
```

```python
            best_a = a.copy()
            if verbose:

        # Update p: focus on lags near the maximum
        max_cv = conv.max()
        # Use temperature to sharpen p
        t_p = 10.0 + outer_iter * 0.5  # gradually sharpen
        p_new = np.exp(t_p * (conv - max_cv))
        p_new /= p_new.sum()
```

"""
Minimizes 2*n*max(auto-conv(a,a))/(sum(a))^2 using a combination of:
1. Alternating minimax (exploiting the saddle-point structure)
2. Smooth gradient descent with logsumexp approximation
3. Multiplicative mirror descent
With multiple sequence lengths and diverse initializations.
"""

import numpy as np
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_convolve2(a, b):
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def q_times_a(p, a):
    """Compute Q*p * a where Q = sum_k p_k S_k, using FFT.
    p has length 2n-1, a has length n.
    """
    n = len(a)
    full = fast_convolve2(a, p[::-1])
    return full[2*n-2 : n-1 : -1]


def smooth_grad(a, conv, t):
    n = len(a)
    max_cv = conv.max()
    if max_cv < 1e-30:
        return np.zeros(n)
    scaled = t * (conv - max_cv)
    exp_s = np.exp(scaled)
    w = exp_s / exp_s.sum()
    sg = np.zeros(n)
    thresh = w.max() * 1e-5
    active = np.where(w > thresh)[0]
    for k in active:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je+1] += w[k] * a[k-je:k-js+1][::-1]
    return sg


def alternating_method(n, time_limit, p_temp_init=5.0, inner_iters=30):
    """
    Alternating minimax:
    - p: weight distribution over lags (2n-1 total)
    - a: sequence to optimize
    Inner: min a^T Q(a) a over simplex (projected gradient)
    Outer: update p to focus on active lags
    """
    a = np.ones(n) / n
    p = np.ones(2*n-1) / (2*n-1)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    eta = 1.0  # will be adapted
    p_temp = p_temp_init
    t0 = time.monotonic()
    outer = 0
    
    while time.monotonic() - t0 < time_limit:
        outer += 1
        
        # Estimate Lipschitz constant for step size
        # Use ||Q||_F / n as a rough bound
        Qa_init = q_times_a(p, a)
        lip_est = 2.0 * np.linalg.norm(Qa_init) / np.linalg.norm(a) + 1e-10
        eta = min(eta, 0.5 / lip_est)
        
        # Inner: projected gradient on the QP
        for it in range(inner_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            Qa = q_times_a(p, a)
            grad = 2.0 * Qa
            
            a_new = a - eta * grad
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = np.ones(n) / n
            else:
                a_new = a_new / s
            
            # Check if step was good (decrease in objective)
            obj_old = np.dot(a, q_times_a(p, a))
            obj_new = np.dot(a_new, q_times_a(p, a_new))
            
            if obj_new > obj_old * 1.01 + 1e-12:
                eta *= 0.5
            else:
                if obj_new < obj_old * 0.99:
                    # Good step, maybe increase eta
                    pass
                a = a_new
        
        # Evaluate actual objective
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        
        # Update p: sharpen focus on active lags
        max_cv = conv.max()
        p_new = np.exp(p_temp * (conv - max_cv))
        p_new /= p_new.sum()
        p = 0.8 * p + 0.2 * p_new
        
        # Gradually sharpen p
        p_temp *= 1.05
    
    return best_a, best_obj


def grad_optimize(a0, time_limit, t_init=40.0, lr_init=0.5):

    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
