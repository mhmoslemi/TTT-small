def gradient_descent_smooth_max(a, n_iters=1000, initial_temp=1.0, final_temp=0.01, lr_initial=0.1, lr_final=0.001, time_limit=60.0):
    start = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    for it in range(n_iters):
        if time.monotonic() - start > time_limit:
            break
        
        # Anneal temperature and learning rate
        frac = it / max(1, n_iters - 1)
        temp = initial_temp * (final_temp / initial_temp) ** frac
        lr = lr_initial * (lr_final / lr_initial) ** frac
        
        # Compute gradient
        grad = gradient_smooth_max(a, temperature=temp)
        
        # Project gradient onto tangent space of simplex (sum(a) = const)
        # The tangent space is {v : sum(v) = 0}
        grad = grad - np.mean(grad)  # project to sum-zero
        
        # Gradient step
        a_new = a - lr * grad
        a_new = np.maximum(0, a_new)
        a_new = normalize(a_new)
        
        # Evaluate
        score = score_sequence(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_new = a - lr * 0.1 * grad
            a_new = np.maximum(0, a_new)
            a_new = normalize(a_new)
            score = score_sequence(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new
    
    return best_a, best_score
