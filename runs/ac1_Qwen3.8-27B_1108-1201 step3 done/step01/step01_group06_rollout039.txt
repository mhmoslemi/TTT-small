Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Uses smoothed (log-sum-exp) gradient descent with FFT-based gradient
computation, adaptive temperature, and multiple initializations.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.signal import correlate

def normalize_seq(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s < 1e-15:
        a[:] = 1.0
        s = float(n)
    a *= S / s
    return a

def self_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

def proj_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    w = np.sort(v)[::-1]
    cs = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cs - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.where(valid)[0].max()) + 1
    theta = (cs[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

    n = len(a)
    shift = c - c.max()

    w = e / e.sum()

    return g

"""
Minimize 2*n*max_conv / sum^2 over non-negative sequences.

Strategy: Smoothed (log-sum-exp) gradient descent with FFT-based gradient
computation via cross-correlation. Uses adaptive temperature (T), Adam
momentum, multiple initializations, and a polishing phase with high T.
The key improvement over the previous top-k subgradient approach is that
the smoothed gradient uses ALL lags (weighted by exp(T*c[k])), giving a
much richer signal, and is computed in O(n log n) via FFT-based correlation.
"""
import numpy as np
import time
from scipy.fft import fft, ifft
from scipy.signal import correlate


def normalize_seq(a, n):
    """Clamp to [0,1000] and rescale to sum = sqrt(2*n)."""
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s < 1e-15:
        a[:] = 1.0
        s = float(n)
    a *= S / s
    return a


def self_conv(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def proj_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cs = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tc = (cs - S) / idx
    valid = tc <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.where(valid)[0].max()) + 1
    theta = (cs[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def smooth_grad(a, c, T):
    """Gradient of (1/T)log(sum exp(T*c_k)) w.r.t. a.
    Returns raw (unnormalized) gradient."""
    n = len(a)
    shift = c - c.max()
    e = np.exp(T * shift)
    w = e / e.sum()
    # grad[m] = 2 * sum_j w[m+j] * a[j] = 2 * correlate(w, a, 'valid')[m]
    g = 2.0 * correlate(w, a, mode='valid')[:n]
    return g


def run_adam_smoothed(a_init, n, S, budget, rng, T_start=3.0, T_end=300.0,
                      lr_init=0.1, name="", verbose=True):
    """Adam optimizer with smoothed (log-sum-exp) gradient and simplex projection."""
    a = normalize_seq(a_init, n)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    lr = lr_init

    c = self_conv(a)
    best_score = c.max()
    best_a = a.copy()

    start = time.monotonic()
    target_iters = max(200, int(budget / 0.03))
    no_improve = 0
    improve_patience = 2000

    while time.monotonic() - start < budget:
        t += 1
        # Adaptive T: exponential ramp
        frac = min(1.0, t / target_iters)
        T = T_start * (T_end / T_start) ** frac

        c = self_conv(a)
        cur = c.max()
        if cur < best_score - 1e-12:
            best_score = cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Smoothed gradient
        g = smooth_grad(a, c, T)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Learning rate decay
        lr_t = lr_init * (1.0 / (1.0 + 0.001 * t))

        a = proj_simplex(a - lr_t * m_hat / (np.sqrt(v_hat) + eps), S)

        # Exploration noise
        if t % 1000 == 0:
            noise_amp = 0.02 * S / n
            a = proj_simplex(a + rng.normal(0, noise_amp, n), S)

        # Restart if stuck
        if no_improve > improve_patience and budget - (time.monotonic() - start) > 5:
            a = proj_simplex(a + rng.normal(0, 0.1 * S / n, n), S)
            best_a_local = a.copy()
            m[:] = 0
            v[:] = 0
            no_improve = 0
            if verbose:
                print(f"    {name} RESTART at t={t}, best={best_score:.8f}")

        if verbose and t % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"    {name} t={t} T={T:.1f} maxc={cur:.8f} best={best_score:.8f} lr={lr_t:.4f} elapsed={elapsed:.1f}s")

    return best_a, best_score


def run_mirror_smoothed(a_init, n, S, budget, rng, T_start=5.0, T_end=200.0,
                        eta_init=0.5, name=""):
    """Mirror descent (multiplicative weights) with smoothed gradient."""
    a = normalize_seq(a_init, n)
    t = 0
    start = time.monotonic()
    c = self_conv(a)
    best_score = c.max()
    best_a = a.copy()
    target_iters = max(100, int(budget / 0.03))

    while time.monotonic() - start < budget:
        t += 1
        frac = min(1.0, t / target_iters)
        T = T_start * (T_end / T_start) ** frac
        eta = eta_init / (1.0 + 0.005 * t)

        c = self_conv(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        g = smooth_grad(a, c, T)
        gmax = np.abs(g).max()
        if gmax < 1e-15:
            break
        g = g / gmax

        a_new = a * np.exp(-eta * g)
        s = a_new.sum()
        if s < 1e-15:

        else:
            a = a_new * (S / s)
