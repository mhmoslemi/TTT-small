"""
FISTA-based optimization with smooth max approximation and coordinate descent polishing
for minimizing normalized autocorrelation peak of non-negative sequences.
"""

import numpy as np
import time


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _fft_convolve(a):
    """Compute linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2 for efficient FFT
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    fa2 = fa * fa
    b = np.fft.irfft(fa2, fft_size)[:size]
    return b


def _compute_b(a):
    """Compute autocorrelation (linear convolution)."""
    return np.convolve(a, a)


def _grad_smooth_max(a, b, t, top_k_vals=None):
    """
    Compute gradient of (1/t)*log(sum_k exp(t*b_k)) with respect to a.
    Uses top-k peak positions for efficiency.
    """
    n = len(a)
    if top_k_vals is not None and len(top_k_vals) < len(b):
        b_work = np.full(len(b), -np.inf)
        b_work[top_k_vals] = b[top_k_vals]
        # Add a small baseline to avoid -inf issues
        b_work = np.maximum(b_work, b[top_k_vals].min() - 1.0)
    else:
        b_work = b.copy()
    
    # Softmax with temperature t
    b_shifted = t * (b_work - np.max(b_work))
    p = np.exp(b_shifted)
    p_sum = np.sum(p)
    if p_sum < 1e-300:
        return np.zeros(n)
    p /= p_sum
    
    # Compute gradient: sum_k p_k * 2 * a[k-i] for valid i
    grad = np.zeros(n)
    # For efficiency, only process k with significant p_k
    threshold = 1e-10
    active_k = np.where(p > threshold)[0]
    
    for k in active_k:
        pk = p[k]
        if pk < 1e-12:
            continue
        # grad[i] += pk * 2 * a[k-i] for valid i (0 <= k-i <= n-1)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += pk * 2.0 * a[k - idx]
    
    return grad


def _fista_smooth(a, deadline, t_init=5.0, t_max=50.0, M=100, L_scale=1.0, 
                    momentum_decay=1.0, verbose=True):
    """
    FISTA to minimize smooth approximation of max(b_k) subject to a>=0, sum(a)=1.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n
    
    best_a = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a))
    
    y = a.copy()  # FISTA momentum variable
    t_k = 1.0  # FISTA momentum parameter
    t_temp = t_init  # Temperature for log-sum-exp
    
    iter_count = 0
    t_last_check = time.monotonic()
    
    # Adaptive temperature: increase over time
    total_iters_estimate = 5000
    iters_to_check = 50
    
    while time.monotonic() < deadline:
        iter_count += 1
        
        # Compute convolution
        b = np.convolve(y, y)
        
        # Find top M peak positions
        if M < len(b):
            top_idx = np.argpartition(b, -M)[-M:]
        else:
            top_idx = np.arange(len(b))
        
        # Compute smooth max gradient
        grad = _grad_smooth_max(y, b, t_temp, top_idx)
        
        # Center gradient for sum constraint
        grad -= np.mean(grad)
        
        # Lipschitz estimate
        L = L_scale * 2.0 * t_temp / max(np.sum(y), 1e-10)
        # Actually, let's use a simpler bound
        L = 2.0 * t_temp
        step_size = 1.0 / L
        
        # FISTA update
        x_new = y - step_size * grad
        x_new = _project_simplex(x_new, 1.0)
        
        # Update momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - a)
        y = _project_simplex(y, 1.0)  # Keep y in feasible set
        t_k = t_new
        
        a = x_new
        
        # Periodic evaluation
        if iter_count % iters_to_check == 0:
            b_actual = np.convolve(a, a)
            cur_score = 2.0 * n * np.max(b_actual)
            if cur_score < best_score - 1e-10:
                best_score = cur_score
                best_a = a.copy()
            
            # Increase temperature
            if t_temp < t_max:
                t_temp = min(t_max, t_temp * 1.05)
            
            if verbose and iter_count % (iters_to_check * 10) == 0:
                print(f"  FISTA iter {iter_count}: score={cur_score:.6f}, best={best_score:.6f}, t_temp={t_temp:.1f}")
    
    # Final check
    b_actual = np.convolve(a, a)
    cur_score = 2.0 * n * np.max(b_actual)
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()
    
    return best_a, best_score


def _coordinate_polish(a, n_iterations=5, verbose=True):

    Coordinate descent polishing. For each coordinate, find the optimal value
    that minimizes the max autocorrelation peak.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    
    best_a = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a))
    
    for iteration in range(n_iterations):
        improved = False
        for i in range(n):
            # Save current value
            old_val = a[i]
            
            # Set a_i = 0 and normalize
            a[i] = 0.0
            S_rest = np.sum(a)
            if S_rest < 1e-10:
                a[i] = old_val
                continue
            a = a * (1.0 / S_rest)  # Now sum = 1 with a_i = 0
            
            # Compute b with a_i = 0
            b0 = np.convolve(a, a)
            M0 = np.max(b0)
            
            # Find optimal a_i value t
            # b_k(t) = b0_k + t * (a_{k-i} + a_{i-k}) for k where valid
            # Actually, when we set a_i = t and others are a_j * (1/S_rest) * S_rest = ...
            # Hmm, this is getting complicated with the renormalization.
            
            # Let me use a simpler approach: binary search on t
            # We want to find t that minimizes max_k b_k(t)
            # where b_k(t) = b0_k + t*a_{k-i} + t*a_{i-k} - t^2*(correction for normalization)
            
            # Actually, let's just try a few values of t and pick the best

            t_candidates = np.concatenate([

            ])
            
            best_t = 0
            best_max_b = M0

                a[i] = t * S_rest  # Scale back
                # Renormalize
                S_new = np.sum(a)
                if S_new < 1e-10:
                    continue
                a_norm = a / S_new
                b_t = np.convolve(a_norm, a_norm)
                max_b = np.max(b_t)
                if max_b < best_max_b:

            a[i] = best_t * S_rest
            a = a / np.sum(a)
            
            # Check improvement
            b_check = np.convolve(a, a)
            cur_score = 2.0 * n * np.max(b_check)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                improved = True
        
        if not improved and iteration > 0:
            break
    
    return best_a, best_score

```python
"""
FISTA (Nesterov-accelerated gradient) with adaptive log-sum-exp smoothing
and anti-symmetric restarts to minimize the normalized autocorrelation peak
of a non-negative sequence. Combines smooth approximation with momentum
acceleration for O(1/T^2) convergence, plus diversification via structured
anti-symmetric initializations.
"""

import numpy as np
import time


def _project_simplex(v, target=1.0):
    """Project vector v onto {x >= 0, sum(x) = target} in O(n log n)."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_candidates = np.nonzero(cond)[0]
    if len(rho_candidates) == 0:
        return np.full(n, target / n)
    rho = rho_candidates[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_score_direct(a):
    """Compute 2*n*max(convolve(a,a)) / (sum(a))^2."""
    S = np.sum(a)
    if S < 1e-10:
        return np.inf
    a_norm = a / S
    b = np.convolve(a_norm, a_norm)
    return 2.0 * len(a) * np.max(b)


def _grad_at_peaks(a, b, peak_indices, temps):
    """
    Compute gradient of (1/t)*log(sum_k exp(t*b_k)) at given peak indices.
    Returns the gradient vector of length n.
    """
    n = len(a)
    b_sub = b[peak_indices]
    t = temps
    shifted = t * (b_sub - np.max(b_sub))
    max_shifted = np.max(shifted)
    exp_vals = np.exp(shifted - max_shifted)
    p = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for m in range(len(peak_indices)):
        pm = p[m]
        if pm < 1e-12:
            continue
        k = peak_indices[m]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += pm * 2.0 * a[k - idx]
    return grad


def _fista_optimize(a, deadline, t_init=5.0, t_final=200.0, M=150, 
                    L_base=2.0, momentum_reset=200, verbose=True, tag=""):
    """
    FISTA to minimize smooth max of autocorrelation peaks.
    Minimizes (1/t)*log(sum_k exp(t*b_k)) subject to a>=0, sum(a)=1.
    Temperature t increases geometrically from t_init to t_final.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    else:
        a = np.ones(n) / n
    
    best_a = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a))
    
    y = a.copy()
    x_prev = a.copy()
    t_mom = 1.0
    t_temp = t_init
    step_size = 1.0 / (L_base * t_temp)
    
    iter_count = 0
    eval_interval = max(10, n // 100)
    no_improve = 0
    max_no_improve = eval_interval * 50
    
    # Time budget for temperature increase
    t_start = time.monotonic()
    
    while time.monotonic() < deadline:
        iter_count += 1
        
        # Compute convolution of momentum point
        b = np.convolve(y, y)
        b_max = np.max(b)
        
        # Find top M peaks
        if M < len(b):
            peak_idx = np.argpartition(b, -M)[-M:]
        else:
            peak_idx = np.arange(len(b))
        
        # Compute gradient
        grad = _grad_at_peaks(y, b, peak_idx, t_temp)
        
        # Center for sum constraint
        grad -= np.mean(grad)
        
        # Adaptive step size (Barzilai-Borwein style)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Take step
        x_new = y - step_size * grad / grad_norm * np.linalg.norm(grad) * 0.5
        # Actually, standard FISTA: x_new = y - (1/L) * grad
        x_new = y - step_size * grad
        
        # Project onto simplex
        x_new = _project_simplex(x_new, 1.0)
        
        # Check for improvement (periodic)
        if iter_count % eval_interval == 0:
            cur_score = 2.0 * n * b_max * (np.sum(y)**2) / (np.sum(y)**2)
            # More accurate: recompute
            b_actual = np.convolve(x_new, x_new)
            cur_score = 2.0 * n * np.max(b_actual)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = x_new.copy()
                no_improve = 0
            else:
                no_improve += eval_interval
            
            # Increase temperature geometrically
            if t_temp < t_final:
                progress = min(1.0, iter_count / (t_final * 20.0))
                t_temp = t_init * (t_final / t_init) ** progress
                step_size = 1.0 / (L_base * t_temp)
            
            # Reset momentum periodically
            if iter_count % momentum_reset == 0:
                t_mom = 1.0
            
            # Break if stuck
            if no_improve > max_no_improve:
                break
            
            if verbose and iter_count % (eval_interval * 20) == 0:
                elapsed = time.monotonic() - t_start
                print(f"  [{tag}] iter={iter_count}, score={cur_score:.6f}, "
                      f"best={best_score:.6f}, t={t_temp:.1f}, "
                      f"elapsed={elapsed:.1f}s")
        
        # FISTA momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        beta = min(beta, 0.85)  # Damp momentum for stability
        y = x_new + beta * (x_new - x_prev)
        y = _project_simplex(y, 1.0)
        t_mom = t_new
        x_prev = x_new.copy()
        a = x_new
    
    # Final evaluation
    b_final = np.convolve(a, a)
    final_score = 2.0 * n * np.max(b_final)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    return best_a, best_score


def _subgradient_polish(a, deadline, M=50, eta_init=None, cool_rate=0.8, 
                        min_eta=1e-10, tag=""):
    """
    Subgradient descent with top-M peaks for final polishing.
    More robust than FISA when very close to optimum.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    S = np.sum(a)
    if S > 1e-10:
        a /= S
    
    best_a = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a))
    
    if eta_init is None:
        eta_init = 0.05 / np.sqrt(n)
    eta = eta_init
    no_improve = 0
    reheat_count = 0
    
    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        b_max = np.max(b)
        cur_score = 2.0 * n * b_max
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 500:
            eta *= cool_rate
            no_improve = 0
            reheat_count += 1
            if eta < min_eta or reheat_count > 20:
                break
        
        # Top M peaks with exponential weighting
        if M < len(b):
            k_idx = np.argpartition(b, -M)[-M:]
        else:
            k_idx = np.arange(len(b))
        
        b_sel = b[k_idx]
        wts = np.exp(-10.0 * (b_max - b_sel) / max(b_max, 1e-15))
        wts /= np.sum(wts)
        
        grad = np.zeros(n)
        for m in range(len(k_idx)):
            if wts[m] < 1e-10:
                continue
            k = k_idx[m]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                idx = np.arange(lo, hi + 1)
                grad[idx] += wts[m] * 2.0 * a[k - idx]
        
        grad -= np.mean(grad)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        
        a = a - eta * grad / gn
        a = _project_simplex(a, 1.0)
    
    b_final = np.convolve(a, a)
    final_score = 2.0 * n * np.max(b_final)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    return best_a, best_score


def _make_anti_symmetric_two_level(n, eps):
    """Two-level anti-symmetric: first half (1+eps), second half (1-eps), normalized."""
    a = np.ones(n)
    half = n // 2
    a[:half] = 1.0 + eps
    a[half:] = 1.0 - eps
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_smooth_anti(n, eps):
    """Smooth anti-symmetric profile."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + eps * np.cos(np.pi * i / max(1, n - 1))
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_quartic_anti(n, eps):
    """Quartic anti-symmetric: 1 + eps*(2*i/(n-1) - 1)^3, creating S-shape."""
    i = np.arange(n, dtype=np.float64)
    x = 2.0 * i / max(1, n - 1) - 1.0
    a = 1.0 + eps * x**3
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_gaussian_anti(n, center_frac, width_frac, height):
    """Gaussian bump anti-symmetric: add positive Gaussian on one side, negative on other."""
    i = np.arange(n, dtype=np.float64)
    x = i / max(1, n - 1)
    center = center_frac
    width = width_frac
    a = np.exp(-((x - center) / width) ** 2) - np.exp(-((x - (1 - center)) / width) ** 2)
    a = 1.0 + height * a
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _perturb(a, rng, scale=0.05):
    """Add smooth random perturbation."""
    n = len(a)
    freqs = rng.integers(1, max(2, n // 8), size=min(5, max(2, n // 4)))
    noise = np.zeros(n)
    i = np.arange(n, dtype=np.float64)
    for f in freqs:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(-1, 1)
        noise += amp * np.cos(2.0 * np.pi * f * i / n + phase)
    noise_max = np.max(np.abs(noise))
    if noise_max > 1e-10:
        noise /= noise_max
    a_new = a + scale * noise
    a_new = _project_simplex(a_new, 1.0)
    return a_new
