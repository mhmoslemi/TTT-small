    n = len(a)
    size = 2 * n - 1

```python
def compute_smooth_grad(a, weights, N):
    """Compute the gradient of the smooth objective.
    
    a: length n
    weights: length 2n-1 (the softmax weights on convolution coefficients)
    N: FFT size (power of 2, >= 3n-2)
    """
    n = len(a)
    W_flip = weights[::-1]  # Reverse weights, length 2n-1
    # Compute linear convolution of W_flip (2n-1) and a (n)
    # Result has length (2n-1) + n - 1 = 3n-2
    A = np.zeros(N)
    W = np.zeros(N)
    A[:n] = a
    W[:2*n-1] = W_flip
    conv = np.real(np.fft.ifft(np.fft.fft(W, N) * np.fft.fft(A, N)))
    # We want conv[2n-2-m] for m=0,...,n-1, which is conv[n-1 : 2n-1] reversed
    G = conv[n-1:2*n-1][::-1]
    return 2.0 * G
```

import numpy as np

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n+1)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho-1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.fft(a, fsize)
    c = np.real(np.fft.ifft(A * A))
    return c[:size]

def compute_gradient(a, weights, N_fft):
    """Compute the weighted gradient of the convolution.
    
    grad[m] = 2 * sum_k weights[k] * a[k-m] (valid terms)
    
    Using FFT: grad = 2 * ifft(fft(weights_flipped, N) * fft(a, N))[n-1:2n-1][::-1]
    """
    n = len(a)
    W = np.zeros(N_fft)
    A = np.zeros(N_fft)
    W[:2*n-1] = weights[::-1]
    A[:n] = a
    conv = np.real(np.fft.ifft(np.fft.fft(W) * np.fft.fft(A)))
    # We want the cross-correlation result
    grad = 2.0 * conv[n-1:2*n-1][::-1]
    return grad

"""
Minimize 2*n*max(self_convolution)/sum^2 over non-negative sequences.

Uses FISTA (Nesterov-accelerated projected gradient) with a smooth
log-sum-exp approximation to the max, combined with mirror descent.
Starts from diverse initializations including chirp sequences.
"""
import numpy as np
import time

def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    if S <= 0:
        return np.zeros(n)
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)

def fft_convolve(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.fft(a, fsize)
    c = np.real(np.fft.ifft(A * A))
    return c[:size]

def compute_gradient(a, weights, N_fft):
    """Compute weighted gradient: grad[m] = 2*sum_k weights[k]*a[k-m]."""
    n = len(a)
    W = np.zeros(N_fft, dtype=np.float64)
    A = np.zeros(N_fft, dtype=np.float64)
    W[:2*n-1] = weights[::-1]
    A[:n] = a
    conv = np.real(np.fft.ifft(np.fft.fft(W) * np.fft.fft(A)))
    return 2.0 * conv[n-1:2*n-1][::-1]

def run_fista(a_init, n, S, time_budget, rng, T_init=1.0, T_min=0.005, lr_init=0.5):
    """FISTA with smooth log-sum-exp objective and simplex projection."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S/n)
    else:
        a = a * (S / s)
    a = project_simplex(a, S)
    
    N_fft = 1 << (3*n - 2).bit_length()
    a_prev = a.copy()
    t = 1
    best_a = a.copy()
    c = fft_convolve(a)
    best_val = c.max()
    lr = lr_init
    
    start = time.monotonic()
    while time.monotonic() - start < time_budget:
        # Nesterov lookahead
        kappa = (1.0 + np.sqrt(1.0 + 4.0 * float(t) * float(t))) / 2.0
        y = (1.0 + 1.0/kappa) * a - (1.0/kappa) * a_prev
        y = project_simplex(y, S)
        
        # Smooth gradient at y
        c_y = fft_convolve(y)
        T_local = max(T_min, T_init * np.exp(-0.002 * float(t)))
        c_shifted = (c_y - c_y.max()) / T_local
        exp_c = np.exp(c_shifted)
        sum_exp = exp_c.sum()
        if sum_exp < 1e-300:
            break
        weights = exp_c / sum_exp
        grad = compute_gradient(y, weights, N_fft)
        
        # Gradient step
        a_new = project_simplex(a - lr * grad, S)
        
        # Evaluate actual objective
        c_new = fft_convolve(a_new)
        val = c_new.max()
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
        
        a_prev = a
        a = a_new
        t += 1
        
        # Decay learning rate
        if t % 200 == 0:
            lr *= 0.98
        
        # Periodic noise for exploration
        if t % 5000 == 0 and rng.random() < 0.3:
            a = project_simplex(a + rng.normal(0, 0.005 * S / np.sqrt(n), n), S)
            a_prev = a.copy()
        
        if t % 10000 == 0:
            print("    fista t=%d val=%.6f T=%.4f lr=%.4f" % (t, best_val, T_local, lr))
    
    return best_a, best_val

def run_mirror(a_init, n, S, time_budget, rng, eta_init=0.3):
    """Mirror descent (multiplicative weights) for simplex-constrained optimization."""
    a = np.clip(np.asarray(a_init, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        a = np.full(n, S/n)
    else:
        a = a * (S / s)
    
    N_fft = 1 << (3*n - 2).bit_length()
    c = fft_convolve(a)
    best_a = a.copy()
    best_val = c.max()
    eta = eta_init
    t = 0
    
    start = time.monotonic()
    while time.monotonic() - start < time_budget:
        t += 1
        c = fft_convolve(a)
        cur = c.max()
        if cur < best_val:
            best_val = cur
            best_a = a.copy()
        
        T_local = max(0.01, 0.3 * cur / np.sqrt(n))
        c_shifted = (c - c.max()) / T_local
        exp_c = np.exp(c_shifted)
        weights = exp_c / exp_c.sum()
        grad = compute_gradient(a, weights, N_fft)
        
        gmax = np.abs(grad).max()
        if gmax < 1e-15:
            break
        grad = grad / gmax
        
        a_new = a * np.exp(-eta * T_local * grad)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = np.full(n, S/n)
        else:
            a = a_new * (S / s_new)
        
        eta *= 0.9995
        if t % 10000 == 0:
            print("    mirror t=%d val=%.6f eta=%.4f" % (t, best_val, eta))
    
    return best_a, best_val
