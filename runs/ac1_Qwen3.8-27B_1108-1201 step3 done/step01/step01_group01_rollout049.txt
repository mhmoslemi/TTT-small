       n = len(a)
       # Build constraint matrix: A[k, i] = a[k-i] if 0 <= k-i < n, else 0
       # A is (2n-1) x n

       A = lil_matrix((2*n-1, n))
       for k in range(2*n-1):

           hi = min(n - 1, k)
           for i in range(lo, hi + 1):

    rows = np.empty(K_actual * n, dtype=np.int32)
    cols = np.empty(K_actual * n, dtype=np.int32)
    vals = np.empty(K_actual * n, dtype=np.float64)

    row_ptr = 0
    for idx in range(K_actual):
        k = active_lags[idx]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        width = hi - lo + 1
        rows[row_ptr:row_ptr + width] = idx
        cols[row_ptr:row_ptr + width] = np.arange(lo, hi + 1)
        vals[row_ptr:row_ptr + width] = a[lo:hi + 1][::-1]
        row_ptr += width

    A = coo_matrix((vals[:row_ptr], (rows[:row_ptr], cols[:row_ptr])),
                   shape=(K_actual, n)).tocsr()

    c = -np.ones(n)
    b_ub = np.full(K_actual, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if not res.success:
            return None
        g = res.x
        S = np.sum(g)
        if S < 1e-10:
            return None
        return g / S
    except Exception:
        return None


def _subgradient_step(a, top_k_indices, eta):
    """Subgradient step using top K peak locations."""
    n = len(a)
    grad = np.zeros(n)
    for k in top_k_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[k - idx]
    grad /= len(top_k_indices)
    grad -= np.mean(grad)
    a_new = a - eta * grad
    return _project_simplex(a_new, 1.0)


def _make_linear_anti(n, alpha):
    """Linear anti-symmetric profile."""
    x = 2.0 * np.arange(n, dtype=np.float64) / (n - 1) - 1.0
    a = 1.0 + alpha * x
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_smooth_anti(n, alpha):
    """Smooth anti-symmetric: 1 + alpha*cos(pi*x) where x in [-1,1]."""
    x = 2.0 * np.arange(n, dtype=np.float64) / (n - 1) - 1.0
    a = 1.0 + alpha * np.cos(np.pi * x)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _make_two_level(n, frac, ratio):
    """Two-level: first frac*n get value 1, rest get 1/ratio."""
    a = np.ones(n)
    a[int(frac * n):] = 1.0 / ratio
    S = np.sum(a)
    return a / S


def _make_quadratic_anti(n, alpha):
    """Quadratic anti-symmetric."""
    x = 2.0 * np.arange(n, dtype=np.float64) / (n - 1) - 1.0
    a = 1.0 + alpha * x * np.abs(x)
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-10:
        return np.ones(n) / n
    return a / S


def _optimize_lp(a_init, deadline, K_init=50, K_max=500, verbose=True):
    """
    Main LP-based optimization loop with adaptive step size and K-increase.
    a_init should be normalized to sum=1.
    Returns (best_a, best_score).
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a /= S

    best = a.copy()
    best_score = _score(best)
    t = 0.4
    K = K_init
    stagnant = 0
    iter_count = 0

    if verbose:
        print(f"    Init: score={best_score:.8f}, K={K}, t={t:.3f}")

    while time.monotonic() < deadline:
        b = _conv(a)
        M = np.max(b)

        # Solve active LP
        g = _solve_active_lp(a, M, K)

        if g is not None:
            # Try the update with current t, reduce if score increases
            improved = False
            for _ in range(5):
                a_new = (1 - t) * a + t * g
                a_new = np.maximum(a_new, 0.0)
                a_new = _project_simplex(a_new, 1.0)
                new_score = _score(a_new)
                if new_score < best_score - 1e-12:
                    a = a_new
                    best = a.copy()
                    best_score = new_score
                    stagnant = 0
                    improved = True
                    break
                t = max(0.02, t * 0.6)

            if improved:
                # Increase t slightly on success
                t = min(0.9, t * 1.1)
                # Gradually increase K
                if iter_count % 20 == 0 and K < K_max:
                    K = min(K_max, K + 15)
            else:
                stagnant += 1
                if stagnant == 15:
                    # Reheat: restore t, increase K
                    t = 0.4
                    K = min(K_max, K + 20)
                elif stagnant > 50:
                    # Perturb and restart
                    a = best.copy()
                    noise = np.random.default_rng(iter_count).normal(0, 0.01, n)
                    a = _project_simplex(a + noise, 1.0)
                    stagnant = 20
                    t = 0.3
        else:
            # Fallback: subgradient step
            top_n = min(30, len(b))
            top_idx = np.argpartition(b, -top_n)[-top_n:]
            eta = 0.1 / np.sqrt(n)
            a = _subgradient_step(a, top_idx, eta)
            new_score = _score(a)
            if new_score < best_score - 1e-12:
                best = a.copy()
                best_score = new_score
                stagnant = 0
            else:
                stagnant += 1
                a = best.copy()
                if stagnant > 40:
                    break

        iter_count += 1

        if verbose and iter_count % 200 == 0:
            remaining = deadline - time.monotonic()
            print(f"    iter={iter_count}, score={best_score:.8f}, K={K}, "
                  f"t={t:.3f}, stag={stagnant}, rem={remaining:.1f}s")

    return best, best_score


def _optimize_subgradient(a_init, deadline, M_smooth=5.0, verbose=True):
    """
    Smoothed subgradient optimization using softmax approximation of max.
    Good for polishing.
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    S = np.sum(a)
    if S < 1e-10:
        a = np.ones(n) / n
    else:
        a /= S

    best = a.copy()
    best_score = _score(best)
    eta = 0.5 / np.sqrt(n)
    T = M_smooth  # softmax temperature
    stagnant = 0

    if verbose:
        print(f"    Smooth init: score={best_score:.8f}, T={T}")

    while time.monotonic() < deadline:
        b = _conv(a)
        b_max = np.max(b)

        # Softmax weights
        shifted = (b - b_max) / max(T, 1e-10)
        wts = np.exp(shifted)
        wts /= np.sum(wts)

        # Compute weighted gradient
        grad = np.zeros(n)
        # Only use lags with significant weight
        active = wts > 1e-6
        for k in np.where(active)[0]:
            w = wts[k]
            if w < 1e-8:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * 2.0 * a[k - idx]

        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        a_new = a - eta * grad / gnorm
        a_new = _project_simplex(a_new, 1.0)
        new_score = _score(a_new)

        if new_score < best_score - 1e-12:
            a = a_new
            best = a.copy()
            best_score = new_score
            stagnant = 0
            eta = min(1.0, eta * 1.02)
        else:
            stagnant += 1
            eta *= 0.9
            if stagnant > 200:
                T *= 0.7  # sharpen the softmax
                eta = 0.3 / np.sqrt(n)
                stagnant = 0
                a = best.copy()

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-start LP-based + smoothed subgradient optimization to minimize
    the normalized autocorrelation peak of a non-negative sequence of length n.
    """
    rng_seed = seed
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)

    # Get starting sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except NameError:
        parent = np.ones(6121)
    parent = np.clip(parent, 0.0, None)
    if parent.size == 0:
        parent = np.ones(6121)
    n = parent.size

    # Global best
    global_best = None
    global_best_score = np.inf

    # Prepare diverse starting points
    starts = []

    # Start 0: Previous best (normalized)
    a0 = parent.copy()
    S0 = np.sum(a0)
    if S0 > 1e-10:
        a0 = a0 / S0
    else:
        a0 = np.ones(n) / n
    starts.append(("parent", a0))

    # Start 1: Linear anti-symmetric, alpha=1.0
    starts.append(("linear_a1.0", _make_linear_anti(n, 1.0)))

    # Start 2: Linear anti-symmetric, alpha=1.5
    starts.append(("linear_a1.5", _make_linear_anti(n, 1.5)))

    # Start 3: Smooth anti-symmetric
    starts.append(("smooth_a0.95", _make_smooth_anti(n, 0.95)))

    # Start 4: Two-level (70% high, 30% low)
    starts.append(("twolevel", _make_two_level(n, 0.6, 2.0)))

    # Start 5: Quadratic anti-symmetric
    starts.append(("quadratic", _make_quadratic_anti(n, 0.8)))

    n_starts = len(starts)
    total_time = max(1.0, deadline - time.monotonic())

    # Time allocation: first start gets 40%, rest split
    time_first = 0.40 * total_time
    time_rest_each = 0.60 * total_time / (n_starts - 1)

    print(f"n={n}, total_time={total_time:.1f}s, {n_starts} starts")

    # Phase 1: Multi-start LP optimization
    for i, (name, a_start) in enumerate(starts):
        if time.monotonic() >= deadline:
            break

        if i == 0:
            t_alloc = time_first
        else:
            t_alloc = time_rest_each

        start_deadline = min(time.monotonic() + t_alloc, deadline)

        print(f"\n[{time.monotonic():.1f}s] Start {i+1}/{n_starts} ({name}): "
              f"{t_alloc:.1f}s")

        best_a, best_s = _optimize_lp(a_start, start_deadline,
                                      K_init=50, K_max=400, verbose=True)

        if best_s < global_best_score:
            global_best = best_a.copy()
            global_best_score = best_s
            print(f"  *** NEW GLOBAL BEST: {global_best_score:.8f} ***")
        else:
            print(f"  Best: {best_s:.8f} (global: {global_best_score:.8f})")

    # Phase 2: Polish with smoothed subgradient
    remaining = deadline - time.monotonic()
    print(f"\nPhase 2: Polishing, remaining={remaining:.1f}s, "
          f"score={global_best_score:.8f}")

    if remaining > 30 and global_best is not None:
        # Polish 1: smoothed subgradient
        polish1_deadline = min(time.monotonic() + remaining * 0.5, deadline)
        best_a, best_s = _optimize_subgradient(global_best, polish1_deadline,
                                               M_smooth=3.0, verbose=True)
        if best_s < global_best_score:
            global_best = best_a.copy()
            global_best_score = best_s
            print(f"  After smooth polish: {global_best_score:.8f}")

        # Polish 2: LP with large K
        remaining2 = deadline - time.monotonic()
        if remaining2 > 20:
            best_a, best_s = _optimize_lp(global_best, deadline,
                                          K_init=300, K_max=12000, verbose=True)
            if best_s < global_best_score:
                global_best = best_a.copy()
                global_best_score = best_s
                print(f"  After LP polish: {global_best_score:.8f}")

    # Phase 3: Diversification - perturb and re-optimize
    remaining = deadline - time.monotonic()
    if remaining > 60 and global_best is not None:
        print(f"\nPhase 3: Diversification, remaining={remaining:.1f}s")
        n_div = max(1, int(remaining / 80))
        per_div = remaining / n_div
        div_rng = np.random.default_rng(rng_seed + 1000)

        for d in range(n_div):
            if time.monotonic() >= deadline:
                break
            # Random perturbations of different types
            pert_type = d % 3
            if pert_type == 0:
                # Smooth perturbation
                freqs = div_rng.integers(1, max(2, n // 6), size=3)
                noise = np.zeros(n)
                for f in freqs:
                    phase = div_rng.uniform(0, 2 * np.pi)
                    amp = div_rng.uniform(-1, 1)
                    noise += amp * np.cos(2 * np.pi * f * np.arange(n) / n + phase)
                noise /= (np.max(np.abs(noise)) + 1e-15)
                a_pert = _project_simplex(global_best + 0.08 * noise, 1.0)
            elif pert_type == 1:
                # Random restart with bias towards anti-symmetric
                x = 2.0 * np.arange(n) / (n - 1) - 1.0
                alpha = div_rng.uniform(0.8, 1.5)
                a_pert = _make_linear_anti(n, alpha)
            else:
                # Gaussian bump perturbation
                a_pert = global_best.copy()
                center = int(div_rng.integers(n // 4, 3 * n // 4))
                width = int(div_rng.integers(n // 8, n // 4))
                height = div_rng.uniform(-0.3, 0.3)
                a_pert = a_pert + height * np.exp(
                    -0.5 * ((np.arange(n) - center) / width) ** 2)
                a_pert = _project_simplex(a_pert, 1.0)

            div_deadline = min(time.monotonic() + per_div, deadline)
            best_a_d, best_s_d = _optimize_lp(a_pert, div_deadline,
                                              K_init=80, K_max=300,
                                              verbose=False)
            if best_s_d < global_best_score:
                global_best = best_a_d.copy()
                global_best_score = best_s_d
                print(f"  Div {d+1}: NEW BEST {global_best_score:.8f}")

    # Final result
    if global_best is None:
        global_best = a0

    result = [float(x) for x in global_best]
    final_score = float(evaluate_sequence(result))
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {final_score:.8f}")
    print(f"{'='*50}")
    return result
