"""
Step-function search for low max autocorrelation.
Uses adaptive subgradient methods with multiple strategies:
- Smoothed-max (softmax) gradients with geometric cooling
- Hard subgradient steps at stagnation
- Simulated annealing escape
- Frequency-domain flat-spectrum initialization
- Multi-start with diverse shapes and the parent sequence

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len


def _fft_len_score(n):
    return int(next_fast_len(max(1, 2 * int(n) - 1)))


def _fft_len_grad(n):
    return int(next_fast_len(max(1, 3 * int(n) - 1)))


def conv_sq(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = _fft_len_score(n)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2 * n - 1].astype(np.float64, copy=False)


def internal_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.inf
    s = float(np.sum(p))
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = conv_sq(p)
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax):
        return np.inf
    return float(2.0 * n * cmax / (s * s))


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = (u * idx) > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n, dtype=np.float64)
    k = int(np.count_nonzero(cond) - 1)
    theta = (cssum[k] - 1.0) / float(k + 1)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return out / s


def safe_normalize(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if not np.isfinite(s) or s <= 1e-12:
        n = max(1, int(p.size))
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s


def softmax_gradient(p, T_frac):
    """
    Compute gradient of smoothed max of conv(p,p) using softmax weights.
    Returns (gradient, surrogate_value, cmax, weights).
    """
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.empty(0, dtype=np.float64), 0.0, 0.0, np.empty(0, dtype=np.float64)
    c = conv_sq(p)
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 0:
        return np.zeros(n, dtype=np.float64), 0.0, 0.0, c
    T = max(float(T_frac) * cmax, 1e-300)
    diff = (c - cmax) / T
    e = np.exp(np.clip(diff, -50.0, 50.0))
    raw = float(e.sum())
    if not np.isfinite(raw) or raw <= 0.0:
        e = np.zeros_like(c)
        e[int(np.argmax(c))] = 1.0
        raw = 1.0
    w = e / raw
    surr = cmax + T * np.log(raw)
    # Gradient: g[m] = 2 * sum_k w[k] * p[k-m] for valid indices
    # = 2 * cross_correlation(w, p)[m]
    L = _fft_len_grad(n)
    W = fft(w, L)
    P = fft(p, L)
    cross = np.real(ifft(W * np.conj(P)))[:n]
    g = 2.0 * np.maximum(cross, 0.0)
    return g, surr, cmax, w


def hard_subgradient(p):
    """Subgradient of max(conv(p,p)) using only the argmax lag."""
    p = np.asarray(p, dtype=np.float64)
    n = int(p.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    c = conv_sq(p)
    k_star = int(np.argmax(c))
    # dc[k_star]/dp[m] = 2*p[k_star-m] if 0<=k_star-m<n
    g = np.zeros(n, dtype=np.float64)
    lo = max(0, k_star - n + 1)
    hi = min(k_star, n - 1)
    if lo <= hi:
        m_range = np.arange(lo, hi + 1)
        g[m_range] = 2.0 * p[k_star - m_range]
    return g


def flat_spectrum_init(n, rng, num_iter=50):
    """
    Frequency-domain approach: try to make |FFT(p)| flat.
    Alternates between:
    1. Given p, compute c=conv(p,p), find peaks, adjust spectrum to reduce peaks
    2. Project back to p >= 0
    """
    p = np.full(n, 1.0 / n, dtype=np.float64)
    L = _fft_len_score(n)
    for it in range(num_iter):
        Fp = fft(p, L)
        mag = np.abs(Fp)
        phase = np.angle(Fp)
        # We want mag to be as flat as possible
        # Target: mag_front/mag_back ratio = 1
        # Adjust: compress dynamic range
        if it < 10:
            alpha = 0.3
        else:
            alpha = 0.7
        target_mag = np.power(mag, alpha) * np.full_like(mag, np.mean(mag) / np.mean(np.power(mag, alpha)))
        Fp_new = target_mag * np.exp(1j * phase)
        p_new = np.real(ifft(Fp_new, n))
        p_new = np.maximum(p_new, 0.0)
        s = p_new.sum()
        if s > 1e-12:
            p_new = p_new / s
        if np.all(np.isfinite(p_new)):
            p = p_new
    return p


def make_smooth_shape(n, rng, style):
    """Generate a smooth nonnegative shape."""
    n = int(max(2, n))
    x = np.linspace(0.0, 1.0, n)
    if style == 'uniform':
        p = np.ones(n)
    elif style == 'gaussian':
        sigma = float(rng.uniform(0.15, 0.4))
        p = np.exp(-((x - 0.5) ** 2) / (2.0 * sigma * sigma))
    elif style == 'super_gauss':
        sigma = float(rng.uniform(0.2, 0.4))
        alpha = float(rng.uniform(1.5, 4.0))
        p = np.exp(-np.abs(x - 0.5) ** alpha / (sigma ** alpha))
    elif style == 'cosine':
        p = 0.5 + 0.5 * np.cos(np.pi * (x - 0.5))
    elif style == 'beta_dist':
        a = float(rng.uniform(0.5, 3.0))
        b = a
        p = np.power(np.maximum(x, 1e-8), a - 1) * np.power(np.maximum(1 - x, 1e-8), b - 1)
    elif style == 'two_peak':
        m1 = float(rng.uniform(0.2, 0.35))
        m2 = 1.0 - m1
        s = float(rng.uniform(0.03, 0.1))
        p = np.exp(-((x - m1) ** 2) / (2 * s * s)) + np.exp(-((x - m2) ** 2) / (2 * s * s))
    elif style == 'random_fourier':
        p = np.ones(n)
        for k in range(1, int(rng.integers(2, 8))):
            amp = float(rng.uniform(0.05, 0.3)) / k
            phi = float(rng.uniform(0, 2 * np.pi))
            p += amp * np.cos(2 * np.pi * k * x + phi)
        p = np.maximum(p, 0.01)
    else:
        p = np.ones(n)
    return safe_normalize(p)

def multi_temp_optimize(p0, rng, time_limit, label=""):
    """
    Try multiple temperature schedules and keep the best result.
    """
    p = project_simplex(np.asarray(p0, dtype=np.float64))
    n = int(p.size)
    best_overall = p.copy()
    best_score = internal_score(best_overall)

    end = time.monotonic() + max(0.0, float(time_limit))
    schedules = [
        (0.5, 0.999, "hot"),
        (0.3, 0.9995, "med"),
        (0.1, 0.9999, "cold"),
    ]

    for T0, decay, slabel in schedules:
        if time.monotonic() > end - 20:
            break
        remaining = end - time.monotonic()
        np_ = safe_normalize(p0)
        p_try, sc_try = _optimize_with_schedule(np_, rng, min(remaining - 5, time_limit / 3.0), T0, decay, label + slabel)
        if np.isfinite(sc_try) and sc_try < best_score:
            best_score = sc_try
            best_overall = p_try.copy()
            print("  %s schedule %s: %.10f" % (label, slabel, sc_try), flush=True)

    return best_overall, best_score

            p_best, sc_best = optimize_local(p_init, rng, t_alloc, "init%d" % i)
            try_record(p_best, sc_best)

    # Final polish on the best
    if best_p is not None and time.monotonic() < deadline - 30:
        limit = max(10.0, deadline - 10.0 - time.monotonic())
        print("Final polish with %.0fs..." % limit, flush=True)
        # Try multiple temperature schedules
        p_final, sc_final = optimize_local(best_p, rng, limit, "FINAL")
        try_record(p_final, sc_final)

        # One more try with a fresh perturbation
        if time.monotonic() < deadline - 15:
            n = best_p.size
            p_pert = project_simplex(best_p + rng.normal(0, 0.5 / n, n))
            p_final2, sc_final2 = optimize_local(p_pert, rng, max(5, deadline - 10 - time.monotonic()), "FINAL2")
            try_record(p_final2, sc_final2)

    # Fallback
    if best_p is None:
        n = parent_n if parent_n >= 32 else 512
        best_p = np.full(n, 1.0 / n)
        best_score = internal_score(best_p)

    # Final validation
    best_p = np.asarray(best_p, dtype=np.float64)
    best_p = np.where(np.isfinite(best_p), best_p, 0.0)
    best_p = np.maximum(best_p, 0.0)
    s = float(best_p.sum())
    if s <= 1e-12:
        best_p = np.full(max(1, int(best_p.size)), 1.0)
    best_p = best_p / best_p.sum()
    best_p = np.minimum(best_p, 1000.0)

    final_sc = internal_score(best_p)
    print("DONE. Final score: %.10f, n=%d, time=%.1fs" % (final_sc, best_p.size, time.monotonic() - t_start), flush=True)

    return [float(x) for x in best_p]
