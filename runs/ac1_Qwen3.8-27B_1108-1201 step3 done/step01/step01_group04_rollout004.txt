"""
Optimizes a non-negative sequence to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated softmax gradient descent, coordinate descent with 1D
optimization, simulated annealing, and LP refinement. Explores multiple
sequence lengths and diverse initializations within the time budget.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 1e-12:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize_to_sum(a, target=None):
    """Normalize sequence to target sum (default sqrt(2n))."""
    n = len(a)
    if target is None:
        target = np.sqrt(2.0 * n)
    S = a.sum()
    if S < 1e-12:
        return np.full(n, target / n)
    return a * (target / S)


def softmax_gradient(a, tau):
    """Compute gradient of softmax-smoothed peak autoconvolution w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    # Numerically stable softmax
    c_shift = c - c.max()
    w = np.exp(tau * c_shift)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum

    # Gradient: dL/da[j] = 2 * sum_m w[m]*a[m-j] - w[2j]*a[j]
    # Compute cross-correlation: sum_m w[m]*a[m-j] for j=0..n-1
    a_rev = a[::-1]
    cross = np.convolve(w, a_rev)
    cross_part = cross[n - 1: 2 * n - 1]

    grad = 2.0 * cross_part
    # Correction for diagonal: subtract w[2j]*a[j]
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    if np.any(valid):
        grad[valid] -= w[idx[valid]] * a[valid]

    return grad


def grad_step(a, lr, tau):
    """One projected gradient descent step."""
    n = len(a)
    grad = softmax_gradient(a, tau)
    grad -= grad.mean()
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    return normalize_to_sum(a_new)


def coordinate_step(a, j, n_evals=12):
    """Optimize a single coordinate a[j] via 1D search."""
    n = len(a)
    old_val = a[j]
    if old_val < 1e-15:
        return a

    # Remove a[j] contribution to get rest
    a_zero = a.copy()
    a_zero[j] = 0.0
    c_rest = np.convolve(a_zero, a_zero)
    S_rest = a_zero.sum()

    # Coefficients: c[m] = c_rest[m] + t*L[m] + t^2*Q[m]
    # L[m] = 2*a[m-j] if valid and m != 2j; a[m-j] if m==2j (but Q handles that)
    # Actually: L[m] = 2*a[m-j]*I(0<=m-j<n) - a[m-j]*I(0<=m-j<n, m==2j)
    # Q[m] = I(m==2j)
    L = np.zeros(2 * n - 1)
    Q = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        k = m - j
        if 0 <= k < n:
            if m == 2 * j:
                L[m] = 0.0
                Q[m] = 1.0
            else:
                L[m] = 2.0 * a[k]

    # Search over t values
    candidates = [0.0, old_val * 0.1, old_val * 0.3, old_val * 0.5,
                  old_val * 0.7, old_val * 0.9, old_val,
                  old_val * 1.1, old_val * 1.3, old_val * 1.7,
                  old_val * 2.5, old_val * 5.0]
    # Also try values that might equalize top peaks
    c_old = c_rest + old_val * L + old_val ** 2 * Q
    top2 = np.argsort(c_old)[-2:]
    if len(top2) == 2:
        m1, m2 = top2[0], top2[1]
        # Solve c_rest[m1] + t*L[m1] + t^2*Q[m1] = c_rest[m2] + t*L[m2] + t^2*Q[m2]
        # (Q[m1]-Q[m2])t^2 + (L[m1]-L[m2])t + (c_rest[m1]-c_rest[m2]) = 0
        A_q = Q[m1] - Q[m2]
        B_q = L[m1] - L[m2]
        C_q = c_rest[m1] - c_rest[m2]
        if abs(A_q) > 1e-15:
            disc = B_q ** 2 - 4 * A_q * C_q
            if disc >= 0:
                t1 = (-B_q + np.sqrt(disc)) / (2 * A_q)
                t2 = (-B_q - np.sqrt(disc)) / (2 * A_q)
                if t1 > 0:
                    candidates.append(t1)
                if t2 > 0:
                    candidates.append(t2)
        elif abs(B_q) > 1e-15:
            t = -C_q / B_q
            if t > 0:
                candidates.append(t)

    best_t = old_val
    best_score = float('inf')
    for t in candidates:
        t = max(0.0, t)
        c_new = c_rest + t * L + t ** 2 * Q
        max_c = float(c_new.max())
        S_new = S_rest + t
        if S_new < 1e-12:
            continue
        score = 2.0 * n * max_c / (S_new * S_new)
        if score < best_score:
            best_score = score
            best_t = t

    a_new = a.copy()
    a_new[j] = best_t
    return a_new


def coord_descent(a, max_rounds=5, deadline=None):
    """Cyclic coordinate descent."""
    n = len(a)
    best = a.copy()
    best_score = compute_objective(a)
    for r in range(max_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            a_new = coordinate_step(a, j)
            if not np.array_equal(a_new, a):
                a = a_new
                improved = True
        if not improved:
            break
    s = compute_objective(a)
    if s < best_score:
        best = a.copy()
        best_score = s
    return best, best_score


def simulated_annealing(a, n_iter=2000, deadline=None):
    """Simulated annealing with small perturbations."""
    n = len(a)
    best = a.copy()
    best_score = compute_objective(a)
    T = best_score * 0.05
    T_min = best_score * 0.001

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        T = max(T_min, T * 0.999)
        a_try = a.copy()
        # Perturb 2-5 random elements
        n_perturb = np.random.randint(2, 6)
        indices = np.random.choice(n, n_perturb, replace=False)
        for idx in indices:
            frac = np.random.uniform(0.5, 1.5)
            a_try[idx] = a[idx] * frac
        a_try = np.maximum(a_try, 0.0)
        a_try = normalize_to_sum(a_try)
        score_try = compute_objective(a_try)
        delta = score_try - compute_objective(a)
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            a = a_try
            if score_try < best_score:
                best_score = score_try
                best = a.copy()
    return best, best_score


def lp_refine(a, n_iters=30, deadline=None):
    """LP-based refinement: solve max sum(g) s.t. conv(a,g) <= M."""
    n = len(a)
    if n > 512:
        return a, compute_objective(a)

    best = a.copy()
    best_score = compute_objective(a)
    S_t = np.sqrt(2.0 * n)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = np.convolve(a, a)
        M = float(c.max())

        # Build LP
        c_obj = -np.ones(n)
        A = np.zeros((2 * n - 1, n))
        for m in range(2 * n - 1):
            lo = max(0, m - n + 1)
            hi = min(n - 1, m)
            for j in range(lo, hi + 1):
                A[m, j] = a[m - j]
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                          options={'maxiter': 5000})
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break

        # Blend
        improved = False
        for t in [0.7, 0.5, 0.3, 0.15, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_to_sum(a_new, S_t)
            s_new = compute_objective(a_new)
            if s_new < best_score:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break

        if not improved:
            # Small gradient nudge to escape
            a = grad_step(a, 0.02, 50.0)
            s = compute_objective(a)
            if s < best_score:
                best = a.copy()
                best_score = s
            elif time.monotonic() - (deadline if deadline else 0) > 5:
                break

    return best, best_score

        tau = tau0 + (tau1 - tau0) * frac
        lr = lr_max * 0.5 * (1.0 + np.cos(np.pi * frac))

        if (it + 1) % 200 == 0:
