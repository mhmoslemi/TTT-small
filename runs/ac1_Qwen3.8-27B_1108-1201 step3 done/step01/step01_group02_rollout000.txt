"""
Hybrid gradient + LP optimization for minimizing peak auto-convolution of non-negative sequences.
Uses exact gradient of smoothed max, Frank-Wolfe iterations, LP refinement, and local search.
Enforces symmetry to reduce effective dimensionality.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def make_symmetric(a):
    n = len(a)
    return 0.5 * (a + a[::-1])


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_seq(a):
    n = len(a)
    conv = np.convolve(a, a)
    sum_a = a.sum()
    if sum_a < 1e-15:
        return float('inf')
    return 2.0 * n * conv.max() / (sum_a ** 2)


def compute_gradient_smoothed_max(a, beta=50.0):
    """Compute exact gradient of smoothed max of auto-convolution."""
    n = len(a)
    conv = np.convolve(a, a)
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_s = np.exp(shifted)
    weights = exp_s / exp_s.sum()

    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-12:
            continue
        i_start = max(0, m - n + 1)
        i_end = min(n - 1, m)
        j_start = m - i_end
        j_end = m - i_start
        grad[i_start:i_end + 1] += 2.0 * w * a[j_start:j_end + 1][::-1]
    return grad, max_c


def fw_step(grad, a):
    """Frank-Wolfe step: find s = e_argmin(grad), return updated a with line search."""
    n = len(a)
    j = int(np.argmin(grad))
    s = np.zeros(n)
    s[j] = 1.0
    s = make_symmetric(s)
    s = normalize_nonneg(s)

    # Line search
    best_t = 0.0
    best_score = score_seq(a)
    for t in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
        candidate = (1.0 - t) * a + t * s
        candidate = normalize_nonneg(candidate)
        sc = score_seq(candidate)
        if sc < best_score:
            best_score = sc
            best_t = t
    if best_t > 0:
        a_new = (1.0 - best_t) * a + best_t * s
        a_new = normalize_nonneg(a_new)
        return a_new, best_score, best_t
    return a, best_score, 0.0


def projected_grad_step(a, grad, lr):
    """One step of projected gradient descent onto simplex."""
    a_new = a - lr * grad
    a_new = np.maximum(0.0, a_new)
    a_new = normalize_nonneg(a_new)
    return a_new


def gradient_optimize(a, time_limit=60.0, beta_start=20.0, beta_end=200.0, verbose=True):
    """Optimize using gradient of smoothed max with adaptive beta and FW steps."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    a = make_symmetric(a)
    a = normalize_nonneg(a)
    n = len(a)
    best_a = a.copy()
    best_score = score_seq(a)
    n_iter = 0
    stagnant = 0

    while time.monotonic() - start_time < time_limit:
        progress = min(1.0, (time.monotonic() - start_time) / time_limit)
        beta = beta_start * (beta_end / beta_start) ** progress

        grad, max_c = compute_gradient_smoothed_max(a, beta)
        current_score = 2.0 * n * max_c / (a.sum() ** 2)

        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 200:
            break

        # Try FW step
        a_fw, sc_fw, t_fw = fw_step(grad, a)
        if t_fw > 0 and sc_fw < current_score:
            a = a_fw
        else:
            # Projected gradient step with adaptive lr
            lr = 1.0 / (np.max(np.abs(grad)) + 1e-8)
            lr = min(lr, 0.1)
            a_pg = projected_grad_step(a, grad, lr)
            sc_pg = score_seq(a_pg)
            if sc_pg < current_score:
                a = a_pg
            else:
                # Try smaller step

                a_pg2 = projected_grad_step(a, grad, lr2)
                sc_pg2 = score_seq(a_pg2)
                if sc_pg2 < current_score:
                    a = a_pg2
                else:
                    stagnant += 50

        n_iter += 1
        if verbose and (n_iter % 200 == 0):
            elapsed = time.monotonic() - start_time
            print(f"      grad iter {n_iter}: score={best_score:.8f}, elapsed={elapsed:.1f}s")

    # Final check
    final_score = score_seq(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    Grad done: best={best_score:.8f}, {n_iter} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def build_conv_sparsity(a):
    """Build sparse matrix for cross-correlation conv(a, x)."""
    n = len(a)
    n_rows = 2 * n - 1
    # For row k (0-indexed in full conv), the valid j are max(0,k-n+1) <= j <= min(n-1,k)
    # conv(a,x)[k] = sum_j a[j] * x[k-j]
    # So row k has non-zeros at columns j where a[j] * x[k-j] contributes
    # i.e., x index = k-j, so column = k-j for valid j
    # Let's build it as: A[k, col] where col = k - j, A[k, col] = a[j]
    # Valid: 0 <= j < n and 0 <= k-j < n => j in [max(0,k-n+1), min(n-1,k)]
    
    row_indices = []
    col_indices = []
    data = []
    for k in range(n_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            col = k - j
            if 0 <= col < n:
                row_indices.append(k)
                col_indices.append(col)
                data.append(a[j])
    
    A = csr_matrix((data, (row_indices, col_indices)), shape=(n_rows, n))
    return A


def lp_refine(a, time_limit=120.0, verbose=True):
    """LP-based iterative refinement."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    a = make_symmetric(a)
    a = normalize_nonneg(a)
    n = len(a)
    best_a = a.copy()
    best_score = score_seq(a)
    n_rows = 2 * n - 1
   
    # Pre-build the matrix structure (doesn't depend on a values for sparsity pattern)
    # Actually it does depend on a values. Let's rebuild each time.
    
    stagnant = 0
    it = 0
    max_iters = 200

    while time.monotonic() - start_time < time_limit and it < max_iters:
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit - 5:
            break

        conv = np.convolve(a, a)
        M = conv.max()
        current_score = 2.0 * n * M / (a.sum() ** 2)

        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        if stagnant > 50:
            break

        # Build LP: max sum(b) s.t. A*b <= M, b >= 0
        A = build_conv_sparsity(a)
        c = -np.ones(n)
        b_ub = np.full(n_rows, M)
        bounds = [(0.0, None)] * n

        remaining = time_limit - elapsed
        timeout = max(5.0, min(60.0, remaining * 0.4))

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': timeout})
            if not (res.success and res.x is not None):
                break
            g = np.maximum(0.0, res.x)
        except Exception:
            break

        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        g = g / sum_g
        g = make_symmetric(g)
        g = normalize_nonneg(g)

        # Line search for step size
        best_alpha = 0.0
        best_sc = current_score
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
            a_new = (1.0 - alpha) * a + alpha * g
            a_new = normalize_nonneg(a_new)
            sc = score_seq(a_new)
            if sc < best_sc:
                best_sc = sc
                best_alpha = alpha

        if best_alpha > 0:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = normalize_nonneg(a)
        else:
            # Try very small step
            a = (1.0 - 0.01) * a + 0.01 * g
            a = normalize_nonneg(a)

        it += 1
        if verbose and (it % 10 == 0):
            print(f"      LP iter {it}: score={best_score:.8f}, elapsed={time.monotonic()-start_time:.1f}s")

    final_score = score_seq(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"    LP done: best={best_score:.8f}, {it} iters, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def local_search(a, time_limit=60.0, rng=None, verbose=True):
    """Local search with random window perturbations."""
    if rng is None:
        rng = np.random.default_rng(123)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_seq(a)
    accept_count = 0

    while time.monotonic() - start_time < time_limit:
        width = max(2, int(rng.integers(n // 200, n // 6)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        candidate = a.copy()
        scale = np.exp(rng.normal(0.0, 0.15, size=stop - start))
        candidate[start:stop] *= scale
        if rng.random() < 0.4:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = normalize_nonneg(candidate)
        sc = score_seq(candidate)
        if sc < best_score:
            best_score = sc
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            if verbose and accept_count % 30 == 0:
                print(f"      LS accept {accept_count}: score={best_score:.8f}")

    if verbose:
        print(f"    LS done: best={best_score:.8f}, {accept_count} accepts, {time.monotonic()-start_time:.1f}s")
    return best_a, best_score


def init_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def init_gaussian(n, sigma_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def init_flat_with_edges(n, edge_width=0.15):
    t = np.arange(n, dtype=np.float64)
    x = t / (n - 1) if n > 1 else np.array([0.0])
    # Smoother edge falloff
    left = np.minimum(1.0, x / edge_width)
    right = np.minimum(1.0, (1.0 - x) / edge_width)
    # Use cosine taper for smoothness
    left = 0.5 * (1.0 - np.cos(np.pi * np.minimum(1.0, x / edge_width)))
    right = 0.5 * (1.0 - np.cos(np.pi * np.minimum(1.0, (1.0 - x) / edge_width)))
    return left * right


def init_random_smooth(n, rng, n_bumps=4):
    a = np.zeros(n)
    t = np.arange(n, dtype=np.float64)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(2, n // 15), max(3, n // 4)))
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    a = make_symmetric(a)
    s = a.sum()
    if s < 1e-15:
        return init_raised_cosine(n)
    return a / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses gradient optimization, LP refinement, and local search.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    deadline = total_start + max(10.0, budget_s - 20.0)

    best_score = float('inf')
    best_seq = None

    # Try parent sequence if available
    try:
        parent = np.array(height_sequence_1, dtype=np.float64)
        parent = np.maximum(0, parent)
        parent = make_symmetric(parent)
        parent = normalize_nonneg(parent)
        ps = score_seq(parent)
        print(f"Parent sequence: n={len(parent)}, score={ps:.8f}")
        if ps < best_score:
            best_score = ps
            best_seq = parent.copy()
    except NameError:
        print("No parent sequence available.")
    except Exception as e:
        print(f"Error loading parent: {e}")

    # Define configurations to try
    # (n, init_type, grad_time_frac, lp_time_frac)
    configs = [
        (400, 'cosine', 0.08, 0.05),
        (500, 'cosine', 0.08, 0.05),
        (600, 'cosine', 0.06, 0.04),
        (700, 'cosine', 0.05, 0.04),
        (500, 'gaussian', 0.06, 0.04),
        (600, 'gaussian', 0.05, 0.03),
        (800, 'gaussian', 0.04, 0.03),
        (500, 'flat_edge', 0.06, 0.04),
        (700, 'flat_edge', 0.04, 0.03),
        (500, 'random', 0.04, 0.03),
        (600, 'random', 0.03, 0.02),
    ]

    for n, init_type, gf, lf in configs:
        remaining = deadline - time.monotonic()
        if remaining < 30.0:
            break

        grad_time = min(gf * budget_s, remaining * 0.4)
        lp_time = min(lf * budget_s, remaining * 0.3)
        if grad_time < 10.0:
            continue

        print(f"\n=== n={n}, init={init_type}, grad_time={grad_time:.0f}s ===")

        if init_type == 'cosine':
            a = init_raised_cosine(n)
        elif init_type == 'gaussian':
            a = init_gaussian(n, sigma_frac=0.25 + 0.1 * rng.random())
        elif init_type == 'flat_edge':
            a = init_flat_with_edges(n, edge_width=0.1 + 0.1 * rng.random())
        elif init_type == 'random':
            a = init_random_smooth(n, rng, n_bumps=int(rng.integers(3, 8)))
        else:
            a = init_raised_cosine(n)

        a = make_symmetric(a)
        a = normalize_nonneg(a)
        init_score = score_seq(a)
        print(f"    Init score: {init_score:.8f}")

        # Phase 1: Gradient optimization
        a_grad, sc_grad = gradient_optimize(a, time_limit=grad_time,
                                            beta_start=10.0, beta_end=300.0, verbose=True)
        if sc_grad < best_score:
            best_score = sc_grad
            best_seq = a_grad.copy()
            print(f"  *** NEW BEST (gradient): score={sc_score if False else sc_grad:.8f}, n={n}")

        # Phase 2: LP refinement
        remaining = deadline - time.monotonic()
        if remaining > 15.0 and len(a_grad) <= 1000:
            lp_budget = min(lp_time, remaining * 0.5)
            if lp_budget > 10.0:
                a_lp, sc_lp = lp_refine(a_grad, time_limit=lp_budget, verbose=True)
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = a_lp.copy()
                    print(f"  *** NEW BEST (LP): score={sc_lp:.8f}, n={n}")

    # Final polishing on best sequence
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20.0:
            print(f"\n=== Final polishing (remaining={remaining:.0f}s) ===")

            # LS polish
            ls_time = min(remaining * 0.3, 120.0)
            if ls_time > 10.0:
                polished, sc_pol = local_search(best_seq, time_limit=ls_time, rng=rng, verbose=True)
                if sc_pol < best_score:
                    best_score = sc_pol
                    best_seq = polished.copy()

            # Gradient polish with new beta schedule
            remaining2 = deadline - time.monotonic()
            if remaining2 > 15.0:
                grad_time2 = min(remaining2 * 0.4, 120.0)
                polished2, sc_pol2 = gradient_optimize(best_seq, time_limit=grad_time2,
                                                       beta_start=50.0, beta_end=500.0, verbose=True)
                if sc_pol2 < best_score:
                    best_score = sc_pol2
                    best_seq = polished2.copy()

            # LP polish
            remaining3 = deadline - time.monotonic()
            if remaining3 > 15.0 and len(best_seq) <= 1000:
                lp_time2 = min(remaining3 * 0.6, 120.0)
                polished3, sc_pol3 = lp_refine(best_seq, time_limit=lp_time2, verbose=True)
                if sc_pol3 < best_score:
                    best_score = sc_pol3
                    best_seq = polished3.copy()

            # Final LS
            remaining4 = deadline - time.monotonic()
            if remaining4 > 10.0:
                polished4, sc_pol4 = local_search(best_seq, time_limit=remaining4 * 0.8, rng=rng, verbose=True)
                if sc_pol4 < best_score:
                    best_score = sc_pol4
                    best_seq = polished4.copy()

    if best_seq is None:
        best_seq = init_raised_cosine(500)
        best_seq = make_symmetric(best_seq)
        best_seq = normalize_nonneg(best_seq)
        best_score = score_seq(best_seq)
        print(f"  Fallback: score={best_score:.8f}")

    # Final verification
    print(f"\nFinal result: score={best_score:.8f}, n={len(best_seq)}")
    verify_score = evaluate_sequence([float(x) for x in best_seq])
    print(f"Verified score via evaluate_sequence: {verify_score:.8f}")
    print(f"Total time: {time.monotonic() - total_start:.1f}s")

    return [float(x) for x in best_seq]
