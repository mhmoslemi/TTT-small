```python
from scipy.fft import fft, ifft

def fft_convolve_full(a, b):
    """Full linear convolution of a and b via FFT."""
    n = len(a) + len(b) - 1
    fsize = 1 << (n - 1).bit_length()
    A = fft(a, fsize)
    B = fft(b, fsize)
    c = np.real(ifft(A * B))
    return c[:n]

def self_convolve(a):
    """Full self-convolution of a (length 2n-1) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]

    c is the full self-convolution (length 2n-1).
    Returns gradient of shape (n,).

    n = len(a)
    # Softmax weights
    c_shifted = c - c.max()  # for numerical stability
    w = np.exp(beta * c_shifted)
    w /= w.sum()
```

```python
"""
Minimize 2*n*max_conv/sum^2 over non-negative sequences.
Uses soft subgradient (FFT-based) with multiplicative weights and Adam,
adaptive learning rates, and periodic LP-based refinement.
Starts from the provided height_sequence_1.
"""
import numpy as np
import time
from scipy.fft import fft, ifft


def project_simplex(v, S):
    """Euclidean projection onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta_candidates = (cumsum - S) / idx
    valid = theta_candidates <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - S) / rho
    return np.maximum(v - theta, 0.0)


def self_convolve(a):
    """Full self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    c = np.real(ifft(A * A))
    return c[:size]


def soft_subgradient(a, c, beta):
    """Soft subgradient using exp(beta*c) weights, computed via FFT."""
    n = len(a)
    c_shifted = c - c.max()
    w = np.exp(beta * c_shifted)
    total = w.sum()
    if total < 1e-30:
        return np.zeros(n)
    w /= total
    
    w_flip = w[::-1]
    size = 3 * n - 2
    fsize = 1 << (size - 1).bit_length()
    A = fft(a, fsize)
    Wf = fft(w_flip, fsize)
    conv_full = np.real(ifft(A * Wf))[:size]
    
    grad = 2.0 * conv_full[n-1:2*n-1][::-1]
    return grad


def topk_subgradient(a, c, k):
    """Top-k subgradient."""
    n = len(a)
    indices = np.argsort(c)[::-1][:k]
    vals = c[indices]
    total = vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    w = np.zeros(2*n - 1)
    w[indices] = vals / total
    return soft_subgradient(a, w, 1.0)  # beta=1 means no additional exp weighting


def safe_normalize(a, n):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    s = a.sum()
    S = np.sqrt(2.0 * float(n))
    if s <= 1e-15:
        a = np.ones(n)
        s = n
    return a * (S / s)


def run_adam_soft(a_init, n, S, max_time, rng, beta_start=30.0, beta_end=100.0,
                  lr_init=None, topk_fallback=10):
    """Adam with adaptive-soft subgradient."""
    if lr_init is None:
        lr_init = 0.3 * S / n
    a = safe_normalize(a_init, n)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    c = self_convolve(a)
    best_score = c.max()
    best_a = a.copy()
    start = time.monotonic()
    
    while time.monotonic() - start < max_time:
        t += 1
        c = self_convolve(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            if t % 500 == 0:
                print(f"  adam_soft t={t} score={cur:.6f}")
        
        # Adaptive beta: increase over time
        frac = min(1.0, t / max(1, int(max_time * 10)))  # approximate
        beta = beta_start + (beta_end - beta_start) * frac
        
        grad = soft_subgradient(a, c, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad ** 2)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        
        lr = lr_init / (1.0 + 0.005 * t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a, S)
        
        # Periodic noise to escape local minima
        if t % 2000 == 0 and t > 0:
            noise_scale = 0.05 * S / n
            a = project_simplex(a + noise_scale * rng.standard_normal(n), S)
    
    return best_a, best_score


def run_multiplicative(a_init, n, S, max_time, rng, eta_init=0.5, beta=50.0):
    """Multiplicative weights (mirror descent) with soft subgradient."""
    a = safe_normalize(a_init, n)
    c = self_convolve(a)
    best_score = c.max()
    best_a = a.copy()
    start = time.monotonic()
    t = 0
    
    while time.monotonic() - start < max_time:
        t += 1
        c = self_convolve(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            if t % 500 == 0:
                print(f"  mult_weights t={t} score={cur:.6f}")
        
        grad = soft_subgradient(a, c, beta)
        gmax = np.abs(grad).max()
        if gmax < 1e-12:
            break
        grad = grad / gmax
        
        # Multiplicative update: a_new = a * exp(-eta * grad), then normalize
        eta = eta_init / (1.0 + 0.002 * t)
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - eta * grad
        log_a_new = log_a_new - log_a_new.max()  # for numerical stability
        a_new = np.exp(log_a_new)
        s = a_new.sum()
        if s < 1e-30:
            a = np.full(n, S / n)
        else:
            a = a_new * (S / s)
        
        if t % 5000 == 0:
            a = np.maximum(a + 0.02 * (S/n) * rng.standard_normal(n), 0)
            a = a * (S / a.sum())
    
    return best_a, best_score


def polish_topk(a_init, n, S, max_time, k_start=5, k_end=50):
    """Polish with top-k subgradient, increasing k."""
    a = safe_normalize(a_init, n)
    c = self_convolve(a)
    best_score = c.max()
    best_a = a.copy()
    start = time.monotonic()
    t = 0
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    while time.monotonic() - start < max_time:
        t += 1
        c = self_convolve(a)
        cur = c.max()
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            if t % 200 == 0:
                print(f"  polish t={t} score={cur:.6f}")
        
        k = min(k_end, k_start + int((k_end - k_start) * min(1.0, t / 1000.0)))
        indices = np.argsort(c)[::-1][:k]
        vals = c[indices]
        total = vals.sum()
        if total < 1e-30:
            break
        w = np.zeros(2*n - 1)
        w[indices] = vals / total
        
        # Compute gradient via FFT
        w_flip = w[::-1]
        size = 3 * n - 2
        fsize = 1 << (size - 1).bit_length()
        A = fft(a, fsize)
        Wf = fft(w_flip, fsize)
        conv_full = np.real(ifft(A * Wf))[:size]
        grad = 2.0 * conv_full[n-1:2*n-1][::-1]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad ** 2)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        
        lr = 0.1 * S / n / (1.0 + 0.01 * t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a, S)
    
    return best_a, best_score
```
