"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using softmax parameterization
with L-BFGS-B optimization on a smoothed (soft-max) objective.
Tries multiple lengths and initializations with temperature scheduling.
"""

import numpy as np
from scipy.optimize import minimize
import time

def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()

def auto_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    return np.fft.irfft(A * A, fsize)[:size]

def smooth_obj_grad(b, t):
    """Smoothed objective and gradient with respect to b (log-softmax param)."""
    a = softmax(b)
    n = len(a)
    R = auto_convolve(a)
    
    tr = t * R
    tr_max = tr.max()
    exp_tr = np.exp(tr - tr_max)
    exp_sum = exp_tr.sum()
    w = exp_tr / exp_sum
    
    # Objective
    obj = 2.0 * n * (tr_max + np.log(exp_sum)) / t
    
    # S2 = sum(w * R)
    S2 = np.dot(w, R)
    
    # S1[p] = sum_j a[j] * w[j+p] (cross-correlation)
    L = n + (2*n - 1) - 1  # = 3n - 2
    fsize = 1 << (L - 1).bit_length()
    Af = np.fft.rfft(a, fsize)
    Wf = np.fft.rfft(w, fsize)
    S1_full = np.fft.irfft(np.conj(Af) * Wf, fsize)
    S1 = S1_full[:n]
    
    # Gradient
    grad = 4.0 * n * a * (S1 - S2)
    
    return obj, grad

def optimize_softmax(b0, t_schedule, max_iter=200, time_limit=60):
    """Optimize using L-BFGS-B with temperature schedule."""
    b = b0.copy()
    best_b = b.copy()
    best_a = softmax(b)
    best_obj = 2.0 * len(best_a) * auto_convolve(best_a).max()
    
    t0 = time.monotonic()
    n_temps = len(t_schedule)
    time_per_temp = time_limit / n_temps
    
    for ti, t in enumerate(t_schedule):
        tl = min(time_per_temp, time_limit - (time.monotonic() - t0))
        if tl < 1:
            break
        
        def fun(b):
            obj, grad = smooth_obj_grad(b, t)
            return obj, grad
        
        res = minimize(fun, b, method='L-BFGS-B', 
                      options={'maxiter': max_iter, 'maxfun': 500,
                               'time_limit': tl if hasattr(minimize, 'time_limit') else None})
        # L-BFGS-B doesn't have time_limit in scipy, so we rely on maxiter
        # Let's use maxiter as the control
        
        b = res.x
        a = softmax(b)
        R = auto_convolve(a)
        obj_true = 2.0 * len(a) * R.max()
        
        if obj_true < best_obj:
            best_obj = obj_true
            best_b = b.copy()
            best_a = a.copy()
    
    return best_a, best_obj

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using softmax parameterization
with L-BFGS-B and Adam optimizers on a smoothed objective.
Explores multiple lengths (128-4096) and diverse initializations.
Uses temperature scheduling for soft-max smoothing.

import numpy as np
from scipy.optimize import minimize
import time


def _softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def _auto_conv(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    return np.fft.irfft(A * A, fsize)[:size]


def _smooth_obj_grad(b, t):
    a = _softmax(b)
    n = len(a)
    R = _auto_conv(a)
    
    tr = t * R
    tr_max = tr.max()
    exp_tr = np.exp(tr - tr_max)
    exp_sum = exp_tr.sum()
    w = exp_tr / exp_sum
    
    obj = 2.0 * n * (tr_max + np.log(exp_sum)) / t
    S2 = np.dot(w, R)
    
    L = n + 2 * n - 2
    fsize = 1 << (L - 1).bit_length()
    Af = np.fft.rfft(a, fsize)
    Wf = np.fft.rfft(w, fsize)
    S1_full = np.fft.irfft(np.conj(Af) * Wf, fsize)
    S1 = S1_full[:n]
    
    grad = 4.0 * n * a * (S1 - S2)
    return obj, grad


def _true_obj(a):
    n = len(a)
    R = _auto_conv(a)
    return 2.0 * n * R.max()


def _lbfgs_optimize(b0, t_schedule, max_iter_per_phase=300, time_limit=45):
    b = b0.copy()
    best_a = _softmax(b)
    best_obj = _true_obj(best_a)
    t_start = time.monotonic()

        elapsed = time.monotonic() - t_start
        if elapsed > time_limit - 2:
            break
        
        def fun(b):
            return _smooth_obj_grad(b, t)
        
        try:
            res = minimize(fun, b, method='L-BFGS-B',
                          options={'maxiter': max_iter_per_phase, 'ftol': 1e-10, 'gtol': 1e-8})
            b = res.x
        except Exception:
            pass
        
        a = _softmax(b)
        obj = _true_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj

    m = np.zeros(n)

    obj_true = _true_obj(a)
    if obj_true < best_obj:
        best_obj = obj_true
        best_a = a.copy()

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using softmax parameterization
with L-BFGS-B optimization on a soft-max smoothed objective.
Derives efficient gradient: df/db[p] = 4n*a[p]*(S1[p]-S2) where
S1=cross_correlate(a,w), S2=dot(w,R). Tries multiple lengths and
diverse initializations with temperature annealing schedule.

import numpy as np
from scipy.optimize import minimize
import time


def _softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def _auto_conv(a):
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    return np.fft.irfft(A * A, fsize)[:size]


def _smooth_obj_grad(b, t):
    """Compute soft-max smoothed objective and gradient w.r.t. b (log-softmax param)."""
    a = _softmax(b)
    n = len(a)
    R = _auto_conv(a)

    tr = t * R
    tr_max = tr.max()
    exp_tr = np.exp(tr - tr_max)
    exp_sum = exp_tr.sum()
    w = exp_tr / exp_sum

    obj = 2.0 * n * (tr_max + np.log(exp_sum)) / t
    S2 = np.dot(w, R)

    # S1[p] = sum_j a[j]*w[j+p] via FFT cross-correlation
    L = n + 2 * n - 2
    fsize = 1 << (L - 1).bit_length()
    Af = np.fft.rfft(a, fsize)
    Wf = np.fft.rfft(w, fsize)
    S1_full = np.fft.irfft(np.conj(Af) * Wf, fsize)
    S1 = S1_full[:n]

    grad = 4.0 * n * a * (S1 - S2)
    return obj, grad

    n = len(a)
    R = _auto_conv(a)
    return 2.0 * n * R.max()

                                    'ftol': 1e-12, 'gtol': 1e-9,

        except Exception:

        obj = _true_obj(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

def _resample_to_length(a, n):
    """Resample sequence a to length n via linear interpolation."""
    if len(a) == n:
        return a.copy()
    if len(a) == 1:
        return np.full(n, a[0])
    x_old = np.linspace(0, 1, len(a))
