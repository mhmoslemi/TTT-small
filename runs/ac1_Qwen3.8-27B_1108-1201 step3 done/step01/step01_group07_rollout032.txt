def resample_sequence(parent, new_len):
    """Resample a sequence to a new length using interpolation."""
    old_len = len(parent)
    if old_len == new_len:
        return parent.copy()
    x_old = np.linspace(0, 1, old_len)
    x_new = np.linspace(0, 1, new_len)
    return np.interp(x_new, x_old, parent)


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for a given length."""
    inits = {}

        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # Cosine window (tapered edges)
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_w = np.maximum(cos_w, 1e-6)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(2, n // 6)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi = np.maximum(bi, 1e-6)
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Alternating pattern
    alt = np.ones(n)
    alt[1::2] = 0.3
    alt /= alt.sum()
    inits['alternating'] = alt

    # Random
    r = rng.dirichlet(np.ones(n) * 0.3)
    inits['random'] = r

    # From parent
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0).copy()

        init_names = ['cosine', 'alternating', 'gauss_0.3', 'bimodal', 'random']
        tried = 0
        for name in init_names:
            if time.monotonic() > deadline - 60:
                break
            if name not in inits:
                continue
            if tried >= 3:
                break
            a0 = inits[name]
            t_alloc = min(40.0, (deadline - time.monotonic() - 60) * 0.15)
            if t_alloc < 5:
                break
            a_opt, M_opt = lp_iterate(a0, time_limit=t_alloc, max_iters=1000, verbose=False)
            update_best(a_opt, f"LP_{name}_n{n}")
            tried += 1

    # Phase 3: Mirror descent refinement on best found so far
    print(f"\n=== Phase 3: Mirror descent refinement. Best: {best_score:.6f} ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 30
        # Try MD refinement with different parameters
        for eta0 in [0.5, 1.0, 2.0]:
            if time.monotonic() > deadline - 20:
                break
            t_alloc = min(60.0, remaining / 3.0)
            a_opt, M_opt = mirror_descent_exact(best_sequence, time_limit=t_alloc, eta0=eta0)
            update_best(a_opt, f"MD_refine_eta{eta0}")

    # Phase 4: Combined LP + MD on best, with perturbation
    print(f"\n=== Phase 4: Combined refinement. Best: {best_score:.6f} ===")
    if best_sequence is not None:
        n = len(best_sequence)
        remaining = deadline - time.monotonic() - 20
        if remaining > 30:
            # LP from best
            t_alloc = remaining * 0.5
            a_opt, M_opt = lp_iterate(best_sequence, time_limit=t_alloc, max_iters=2000)
            update_best(a_opt, f"LP_refine_n{n}")

            # MD from best
            if time.monotonic() < deadline - 15:
                t_alloc = min(40.0, (deadline - time.monotonic() - 15) * 0.8)
                a_opt, M_opt = mirror_descent_exact(best_sequence, time_limit=t_alloc, eta0=0.3)
                update_best(a_opt, f"MD_polish_n{n}")

    # Phase 5: Try a few more lengths with LP if time remains
    print(f"\n=== Phase 5: Additional exploration. Best: {best_score:.6f} ===")
    for n in [96, 160, 224, 288, 448, 576]:
        if time.monotonic() > deadline - 30:
            break
        a0 = np.ones(n) / n
        # Add some structure
        x = np.arange(n)
        a0 = 0.5 * (1 + np.cos(2 * np.pi * x / n))
        a0 = np.maximum(a0, 0.01)
        a0 /= a0.sum()
        t_alloc = min(30.0, deadline - time.monotonic() - 20)
        if t_alloc < 5:
            break
        a_opt, M_opt = lp_iterate(a0, time_limit=t_alloc, max_iters=500)
        update_best(a_opt, f"LP_explore_n{n}")

    # Final fallback
    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
