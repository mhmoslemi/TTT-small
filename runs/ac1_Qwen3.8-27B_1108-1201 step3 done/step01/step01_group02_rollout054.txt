"""
Smoothed mirror descent for minimizing peak auto-convolution of non-negative sequences.
Uses FFT-based gradient computation, asymmetric initializations, and adaptive softmax
peak weighting to flatten the autocorrelation. Combines coarse-to-fine multi-start
search with local polishing.

import numpy as np
import time


def score_sequence(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _next_pow2(n):
    return 1 << (n - 1).bit_length()


def mirror_descent(a, time_limit, rng, beta_start=3.0, beta_end=60.0, verbose=False):
    """
    Smoothed mirror descent (exponential family) to minimize max(conv(a,a)).
    Uses FFT for efficient cross-correlation gradient computation.
    """
    start_t = time.monotonic()
    a = normalize(a.copy().astype(np.float64))
    n = len(a)
    N_fft = _next_pow2(3 * n)

    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"      MD start: score={best_score:.6f}")

    beta = beta_start
    stagnant = 0
    it = 0
    eta_decay = 1.0

    while time.monotonic() - start_t < time_limit:
        # Compute convolution and score
        r = np.convolve(a, a)
        cur_score = 2.0 * n * r.max() / (a.sum() ** 2)

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        # Escape local minimum via perturbation
        if stagnant > 400:
            noise_scale = 0.15 + 0.1 * rng.random()
            a = a * np.exp(rng.normal(0, noise_scale, n))
            a = normalize(a)
            beta = max(beta_start, beta * 0.5)
            stagnant = 0
            it += 1
            continue

        # Peak weights via softmax
        r_shift = r - r.max()
        w = np.exp(beta * r_shift)
        w_sum = w.sum()
        if w_sum < 1e-30 or not np.isfinite(w_sum):
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w /= w_sum

        # Cross-correlation gradient: g[j] = sum_k w[k]*a[k-j]
        W = np.fft.rfft(w, N_fft)
        A = np.fft.rfft(a, N_fft)
        cross = np.fft.irfft(W * np.conj(A), N_fft)[:n]
        cross = np.maximum(0.0, cross)

        c_max = cross.max()
        if c_max < 1e-15:
            break

        # Adaptive step size
        eta = (1.5 * eta_decay) / c_max
        a = a * np.exp(-eta * cross)
        a = normalize(a)

        # Adaptively increase beta (focus more on peaks)
        progress = min(1.0, it / 5000.0)
        beta = beta_start + (beta_end - beta_start) * progress
        # Slight eta decay for stability
        eta_decay = max(0.3, eta_decay * 0.9995)
        it += 1

        if verbose and it % 2000 == 0:
            print(f"      MD iter {it}: score={score_sequence(a):.6f}, beta={beta:.1f}")

    # Final check
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    if verbose:
        print(f"      MD done: best={best_score:.6f} after {it} iters, {time.monotonic()-start_t:.1f}s")
    return best_a, best_score


def hard_peak_reduction(a, time_limit, rng, verbose=False):
    """
    Additional optimization: directly target the single peak using a

    """
    start_t = time.monotonic()

            best_score = cur_score
            best_a = a.copy()

        k_star = int(r.argmax())
        # Gradient from single peak: g[j] = a[k_star - j] (where valid)
        # k_star ranges from 0 to 2n-2
        # a[k_star - j] is valid when 0 <= k_star - j < n, i.e., k_star - n + 1 <= j <= k_star
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        if j_lo > j_hi:
            break
        indices = np.arange(j_lo, j_hi + 1)
        contrib = a[k_star - indices]  # a[k_star - j] for j in [j_lo, j_hi]

        # Multiplicative reduction
        eta = 0.5 / max(contrib.max(), 1e-15)
        a[indices] *= np.exp(-eta * contrib)
        a = normalize(a)

        # Occasionally add small noise to escape

            a = a * np.exp(rng.normal(0, 0.02, n))
            a = normalize(a)

        it += 1

    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    if verbose:
        print(f"      HPR done: best={best_score:.6f} after {it} iters, {time.monotonic()-start_t:.1f}s")
    return best_a, best_score


def local_search_polish(a, time_limit, rng, verbose=False):
    """Polish with random multiplicative perturbations on random windows."""
    start_t = time.monotonic()

    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    accept_count = 0

        pos = int(rng.integers(0, max(1, n - width)))
        candidate = a.copy()

    if verbose:
        print(f"        LS done: best={best_score:.6f}, {time.monotonic()-start_t:.1f}s")
    return best_a, best_score


def _init_exponential(n, rate, direction='left', rng=None):
    """Exponential decay initialization."""
    t = np.arange(n, dtype=np.float64)
    if direction == 'left':
        a = np.exp(-rate * t / n)

    # Add small noise for diversity

        a *= np.exp(rng.normal(0, 0.1, n))
    return a


def _init_two_bumps(n, c1, c2, s1, s2, h2, rng=None):

    a += h2 * np.exp(-0.5 * ((t - c2 * n) / (s2 * n)) ** 2)
    if rng is not None:
        a *= np.exp(rng.normal(0, 0.05, n))
    return np.maximum(0, a)
