"""
Optimizes non-negative step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-based directional search (primary), softmax gradient descent with momentum (secondary),
and coordinate-level refinement. Explores multiple sequence lengths and starting points.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * float(c.max()) / (S * S))


def _normalize(a, target=1.0):
    """Normalize sequence to have a specific sum."""
    S = float(a.sum())
    if S < 1e-15:
        return np.ones_like(a) / len(a)
    return a * (target / S)


def _softmax_gradient(a, beta):
    """Compute softmax-weighted gradient of peak autocorrelation efficiently."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(beta * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
    else:
        w = w / w_sum

    # grad[i] = 2 * sum_{m=0}^{n-1} w[i+m] * a[m]
    # = 2 * np.convolve(w, a[::-1])[i + n - 1]
    a_flipped = a[::-1]
    w_conv_af = np.convolve(w, a_flipped)
    grad = 2.0 * w_conv_af[n - 1:n - 1 + n]
    return grad


def _lp_find_direction(a, step_scale=1.0):
    """
    Solve LP to find direction d minimizing linearized peak of conv(a + t*d, a + t*d).
    Returns (d, predicted_peak) or (None, None).
    
    LP: minimize M_new
    s.t. c[k] + 2*t*conv(a,d)[k] <= M_new  for all k
         d[i] >= -a[i]/t  for all i
         sum(d) = 0
    """
    n = len(a)
    if n > 2048:
        return None, None
    c = np.convolve(a, a)
    t = step_scale

    # Variables: d[0..n-1], M_new (index n)
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[n] = 1.0

    # Build sparse constraint matrix
    # Type 1: 2*t*conv(a,d)[k] - M_new <= -c[k]  for k=0..2n-2
    # Type 2: -d[i] <= a[i]/t  for i=0..n-1
    n_ineq = (2 * n - 1) + n

    rows = []
    cols = []
    vals = []
    b_ub = np.zeros(n_ineq)

    # Type 1: conv(a,d)[k] = sum_{m} a[k-m] * d[m] where 0<=k-m<n
    for k in range(2 * n - 1):
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        for m in range(m_lo, m_hi + 1):
            rows.append(k)
            cols.append(m)
            vals.append(2.0 * t * a[k - m])
        # -M_new term
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
        b_ub[k] = -c[k]

    # Type 2: -d[i] <= a[i]/t
    for i in range(n):
        rows.append(2 * n - 1 + i)
        cols.append(i)
        vals.append(-1.0)
        b_ub[2 * n - 1 + i] = a[i] / t

    A_ub = csr_matrix((vals, (rows, cols)), shape=(n_ineq, n_vars))

    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] * n_vars

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, None


def _lp_optimize(a, deadline, max_iters=500, verbose=False):
    """Iteratively apply LP direction with backtracking line search."""
    n = len(a)
    a = _normalize(a.copy(), 1.0)
    best = a.copy()
    best_score = _compute_obj(a)
    no_improve_count = 0
    base_step = 0.3

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        d, pred_peak = _lp_find_direction(a, step_scale=base_step)
        if d is None:
            break

        # Backtracking line search
        improved = False
        for attempt in range(10):
            step = base_step / (2 ** attempt)
            a_new = a + step * d
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new, 1.0)
            s_new = _compute_obj(a_new)
            if s_new < best_score * (1.0 - 1e-9):
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                no_improve_count = 0
                break
            elif s_new < best_score:
                # Accept even tiny improvements
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                no_improve_count = 0
                break

        if not improved:
            no_improve_count += 1
            if no_improve_count >= 10:
                # Perturb and restart
                rng = np.random.default_rng(it)
                a = a + 0.01 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a = _normalize(a, 1.0)
                no_improve_count = 0
            # Reduce step scale
            base_step *= 0.7
            if base_step < 1e-6:
                break

    return best, best_score


def _softmax_optimize(a, deadline, max_iters=3000, beta_start=10.0, beta_end=200.0,
                      lr_start=0.5, lr_end=0.01, momentum=0.9):
    """Softmax gradient descent with Nesterov momentum."""
    n = len(a)
    a = _normalize(np.maximum(a, 0.0).copy(), 1.0)
    best = a.copy()
    best_score = _compute_obj(a)
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        frac = it / max(max_iters - 1, 1)
        beta = beta_start + (beta_end - beta_start) * frac
        lr = lr_start + (lr_end - lr_start) * frac

        # Compute gradient
        grad = _softmax_gradient(a, beta)
        # Project onto tangent space (sum = const)
        grad = grad - grad.mean()

        # Nesterov momentum
        v = momentum * v + grad
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        a_new = _normalize(a_new, 1.0)

        s_new = _compute_obj(a_new)
        if s_new < best_score:
            best = a_new.copy()
            best_score = s_new
            a = a_new
        else:
            # Reduce step
            v = 0.5 * v
            a_new = a - 0.1 * lr * v
            a_new = np.maximum(a_new, 0.0)
            a_new = _normalize(a_new, 1.0)
            a = a_new

    return best, best_score


def _coordinate_refine(a, deadline, max_sweeps=20):
    """
    Coordinate descent: for each i, find optimal a[i] given others fixed.
    Since the peak is minimized at a[i]=0 (for non-neg constraint), we instead
    do: try removing contributions from the peak by zeroing coordinates
    that contribute most to the peak, then redistribute.
    """
    n = len(a)
    a = _normalize(a.copy(), 1.0)
    best = a.copy()
    best_score = _compute_obj(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline - 5.0:
            break

        c = np.convolve(a, a)
        peak_idx = int(c.argmax())
        peak_val = c[peak_idx]

        # Find coordinates contributing most to the peak
        # c[peak_idx] = sum_j a[j] * a[peak_idx - j]
        j_lo = max(0, peak_idx - n + 1)
        j_hi = min(n - 1, peak_idx)
        contributions = np.zeros(n)
        for j in range(j_lo, j_hi + 1):
            contributions[j] = a[j] * a[peak_idx - j]

        # Try zeroing the top contributors (one at a time) and redistributing
        top_contribs = np.argsort(contributions)[::-1][:min(10, n)]

        for idx in top_contribs:
            if contributions[idx] < 1e-10:
                continue
            a_trial = a.copy()
            mass_to_remove = a_trial[idx]
            a_trial[idx] = 0.0
            # Redistribute mass to coordinates NOT contributing to peak
            non_peak = [(j for j in range(n) if not (j_lo <= j <= j_hi))]
            if not non_peak:
                non_peak = list(range(n))
            non_peak = np.array(list(non_peak))
            if len(non_peak) == 0:
                continue
            # Redistribute proportionally to existing values
            existing = a_trial[non_peak]
            total_existing = existing.sum()
            if total_existing < 1e-15:
                a_trial[non_peak] += mass_to_remove / len(non_peak)
            else:
                a_trial[non_peak] += mass_to_remove * existing / total_existing

            a_trial = np.maximum(a_trial, 0.0)
            a_trial = _normalize(a_trial, 1.0)
            s_trial = _compute_obj(a_trial)
            if s_trial < best_score:
                best = a_trial.copy()
                best_score = s_trial
                a = a_trial

    return best, best_score


def _try_start_point(a0, deadline, verbose=True):
    """Try optimizing a starting point with multiple methods."""
    n = len(a0)
    a0 = np.maximum(a0, 0.0)
    if a0.sum() < 1e-10:
        a0 = np.ones(n)

    best_seq = None
    best_score = float('inf')

    # Method 1: Softmax gradient descent
    if time.monotonic() < deadline - 30.0:
        t_method = min(deadline - 10.0, time.monotonic() + 30.0)
        seq, score = _softmax_optimize(a0, t_method, max_iters=500)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        if verbose:
            print(f"    Softmax: score={score:.6f}, n={n}")

    # Method 2: LP-based optimization (from best so far)
    start_for_lp = best_seq if best_seq is not None else _normalize(a0, 1.0)
    if time.monotonic() < deadline - 20.0:
        t_method = min(deadline - 10.0, time.monotonic() + 60.0)
        seq, score = _lp_optimize(start_for_lp, t_method, max_iters=200)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        if verbose:
            print(f"    LP: score={score:.6f}, n={n}")

    # Method 3: Coordinate refinement
    start_for_coord = best_seq if best_seq is not None else start_for_lp
    if time.monotonic() < deadline - 10.0:
        t_method = min(deadline - 5.0, time.monotonic() + 20.0)
        seq, score = _coordinate_refine(start_for_coord, t_method, max_sweeps=10)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        if verbose:
            print(f"    Coord: score={score:.6f}, n={n}")

    # Method 4: Final LP polish
    if best_seq is not None and time.monotonic() < deadline - 5.0:
        t_method = min(deadline - 2.0, time.monotonic() + 30.0)
        seq, score = _lp_optimize(best_seq, t_method, max_iters=100)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()

    return best_seq, best_score

            _update_best(seq, score, f"phase1:{label}")

        elapsed = time.monotonic() - t0
        if (idx + 1) % 5 == 0:
            print(f"  Phase 1: {idx+1}/{n_starts} done, best={best_score:.6f}, elapsed={elapsed:.1f}s")

    # --- Phase 2: Deep optimization on top candidates ---
    # Sort phase 1 results by score
    phase1_results.sort(key=lambda x: x[0])

    # Take top candidates and give them more time
    n_deep = min(8, len(phase1_results))
    deep_time_each = min(120.0, (deadline - time.monotonic() - 30.0) / max(n_deep, 1))
    deep_time_each = max(deep_time_each, 15.0)

    print(f"\nPhase 2: Deep optimization on top {n_deep} candidates")
    for idx in range(n_deep):
        if time.monotonic() > deadline - 30.0:
            break
        score0, a0, label = phase1_results[idx]
        t0 = time.monotonic()
        t_alloc = min(t0 + deep_time_each, deadline - 10.0)
        if t_alloc <= t0 + 3.0:
            continue

        # Also try perturbed versions
        for pert in range(3):
            if time.monotonic() > t_alloc - 5.0:
                break
            if pert == 0:
                a_try = a0.copy()
            else:
                a_try = a0 + 0.05 * rng.standard_normal(len(a0))
                a_try = np.maximum(a_try, 0.0)
                a_try = _normalize(a_try, 1.0)

            t_sub = min(time.monotonic() + deep_time_each / 3.0, t_alloc - 5.0)
            seq, score = _try_start_point(a_try, t_sub, verbose=False)
            if seq is not None:
                _update_best(seq, score, f"phase2:{label}:p{pert}")

        elapsed = time.monotonic() - t0
        print(f"  Deep {idx+1}/{n_deep}: {label}, elapsed={elapsed:.1f}s, best={best_score:.6f}")

    # --- Phase 3: Intensive refinement of the global best ---
    if best_seq is not None and time.monotonic() < deadline - 20.0:
        print(f"\nPhase 3: Intensive refinement (current best={best_score:.6f})")
        n = len(best_seq)

        # Multiple LP passes with decreasing step size
        for pass_i in range(5):
            if time.monotonic() > deadline - 10.0:
                break
            t_remaining = deadline - 10.0 - time.monotonic()
            if t_remaining < 10.0:
                break
            t_alloc = time.monotonic() + min(60.0, t_remaining * 0.6)
            seq, score = _lp_optimize(best_seq, t_alloc, max_iters=300)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  LP pass {pass_i}: score={best_score:.6f}")
            else:
                break

        # Softmax fine-tuning
        if time.monotonic() < deadline - 10.0:
            t_alloc = min(time.monotonic() + 40.0, deadline - 5.0)
            seq, score = _softmax_optimize(best_seq, t_alloc, max_iters=2000,
                                           beta_start=50.0, beta_end=300.0,
                                           lr_start=0.1, lr_end=0.001, momentum=0.95)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Softmax refine: score={best_score:.6f}")

        # Coordinate refinement
        if time.monotonic() < deadline - 5.0:
            t_alloc = min(time.monotonic() + 20.0, deadline - 2.0)
            seq, score = _coordinate_refine(best_seq, t_alloc, max_sweeps=30)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  Coord refine: score={best_score:.6f}")

    # --- Phase 4: Try to improve with block LP ---
    if best_seq is not None and time.monotonic() < deadline - 15.0:
        print(f"\nPhase 4: Block LP refinement (current best={best_score:.6f})")
        n = len(best_seq)
        a = _normalize(best_seq.copy(), 1.0)

        # Split into blocks and optimize each block
        n_blocks = max(1, n // 64)
        block_size = n // n_blocks
        for bi in range(n_blocks):
            if time.monotonic() > deadline - 10.0:
                break
            lo = bi * block_size
            hi = min(n, (bi + 1) * block_size) if bi < n_blocks - 1 else n

            # Extract block and optimize it with others fixed
            # For a block, the LP is smaller
            a_block = a[lo:hi].copy()
            if a_block.sum() < 1e-15:
                continue

            # Build reduced LP: variables are a_block only
            # conv(a,a)[k] = R[k] + 2*sum_{j in block} a[j]*a[k-j] - (remove terms already in R)
            # This is complex, so just do gradient steps on the block
            for it in range(200):
                if time.monotonic() > deadline - 10.0:
                    break
                # Gradient of peak wrt block elements
                c = np.convolve(a, a)
                peak_k = int(c.argmax())
                # Gradient for block elements
                grad_block = np.zeros(hi - lo)
                for di in range(hi - lo):
                    i_abs = lo + di
                    g = 0.0
                    for k in [peak_k]:  # Just the peak for speed
                        m = k - i_abs
                        if 0 <= m < n:
                            if m != i_abs:
                                g += 2.0 * a[m]
                            else:
                                g += 2.0 * a[i_abs]
                    grad_block[di] = g

                # Normalize and take step
                grad_block = grad_block - grad_block.mean()
                a_block = a_block - 0.01 * grad_block
                a_block = np.maximum(a_block, 0.0)
                a[lo:hi] = a_block

            a = _normalize(a, 1.0)
            s = _compute_obj(a)
            if s < best_score:
                best_score = s
                best_seq = a.copy()
                print(f"  Block {bi}: score={best_score:.6f}")

    # --- Final validation ---
    if best_seq is not None:
        final_score = _compute_obj(best_seq)
        print(f"\nFinal: n={len(best_seq)}, score={final_score:.8f}")

        # Ensure all values are valid
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = np.minimum(best_seq, 1000.0)

        result = [float(x) for x in best_seq]
        print(f"Returning sequence of length {len(result)}")
        return result

    # Fallback
    print("Falling back to uniform sequence.")
    n_fallback = 256
    return [1.0] * n_fallback
