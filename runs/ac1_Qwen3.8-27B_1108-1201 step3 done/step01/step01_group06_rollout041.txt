def random_smooth(n, rng):
    x=np.arange(n, dtype=np.float64)
    v=np.ones(n, dtype=np.float64)
    for f in range(1,5):
        amp=0.25/(f+0.5)
        phase=rng.uniform(0,2*np.pi)
        v += amp*np.cos(2*np.pi*f*x/n + phase)
    nbumps=int(rng.integers(1,4))
    for _ in range(nbumps):
        center=rng.uniform(0,n)
        width=rng.uniform(0.05*n,0.2*n)
        amp=rng.uniform(0.05,0.2)
        v += amp*np.exp(-0.5*((x-center)/width)**2)
    v=np.maximum(v,0.05)

def upsample_sequence(seq, m):
    seq = np.asarray(seq, dtype=np.float64).reshape(-1)
    n=seq.size
    if m <= n: return seq.copy()
    if n == 1: return np.full(m, seq[0])
    x_old = np.linspace(0.0,1.0,n)
    x_new = np.linspace(0.0,1.0,m)
    vals = np.interp(x_new, x_old, seq)
    vals = np.maximum(vals,0.0)
    if vals.sum() <= 0: vals=np.ones(m)
    return vals

```python
"""
Search for non-negative sequences minimizing
    2*n*max(np.convolve(a,a)) / sum(a)^2.

Approach:
  * Normalize sequences to sum sqrt(2n), so the score is max convolution.
  * Use FFT-based self-convolution for all internal evaluations.
  * Polish with a smoothed log-sum-exp objective using dense FFT gradients
    and exponential (mirror) updates, then finish with exact max-convolution
    top-k mirror descent and a projected-Adam stage.
  * Explore several lengths/initializations, but heavily polish the best
    known candidate when available.
"""

import numpy as np
import time
from scipy.fft import fft, ifft


def _next_pow2(x):
    x = int(max(1, x))
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def safe_normalize(a, S=None):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        a = np.ones(64, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = int(a.size)
    if S is None:
        S = np.sqrt(2.0 * float(n))
    s = float(a.sum())
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    a = a * (float(S) / s)
    a = np.clip(a, 0.0, 1000.0)
    s2 = float(a.sum())
    if s2 > 1e-15:
        a = a * (float(S) / s2)
    return a


def positive_for_run(a, S, floor_frac=1e-8):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = int(a.size)
    if n == 0:
        return a
    f = max(1e-300, float(floor_frac) * float(S) / float(n))
    b = np.maximum(a, f)
    s = float(b.sum())
    if s <= 1e-15 or not np.isfinite(s):
        b = np.full(n, float(S) / n, dtype=np.float64)
    else:
        b = b * (float(S) / s)
    return b


def c_from_A(A, n):
    c = np.real(ifft(A * A))[: int(2 * n - 1)]
    c = np.maximum(c, 0.0)
    return c


def calc_c_padded(a, fsize, n=None):
    if n is None:
        n = int(a.size)
    A = fft(a, fsize)
    return c_from_A(A, n)


def calc_c(a):
    n = int(a.size)
    fsize = _next_pow2(2 * n - 1)
    return calc_c_padded(a, fsize, n)


def smooth_value(c, T):
    c = np.asarray(c, dtype=np.float64)
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    if T <= 1e-6:
        return m
    x = (c - m) / T
    x = np.clip(x, -745.0, 0.0)
    s = float(np.exp(x).sum())
    if not np.isfinite(s) or s <= 0.0:
        return m
    return float(m + T * np.log(s))


def soft_weights(c, T):
    c = np.asarray(c, dtype=np.float64)
    m = float(np.max(c))
    if T <= 1e-6:
        idx = np.flatnonzero(c == m)
        if idx.size == 0:
            idx = np.array([int(np.argmax(c))], dtype=int)
        w = np.zeros_like(c)
        w[idx] = 1.0
        return w / w.sum()
    x = (c - m) / T
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def dense_gradient(a, A, w, fsize, n):
    if n == 1:
        return np.array([2.0 * float(a[0]) * float(w[0])], dtype=np.float64)
    wr = np.zeros(fsize, dtype=np.float64)
    wr[: int(2 * n - 1)] = w[::-1]
    W = fft(wr, fsize)
    h = np.real(ifft(A * W))
    g = 2.0 * h[int(2 * n - 2): int(n - 2): -1]
    g = np.maximum(g, 0.0)
    return g


def limit_top_k(c, top_k):
    m = int(c.size)
    top_k = int(top_k)
    if top_k <= 0:
        top_k = 1
    if top_k > 64:
        top_k = 64
    if top_k > m:
        top_k = m
    return top_k


def select_top_idx(c, top_k):
    c = np.asarray(c, dtype=np.float64)
    top_k = limit_top_k(c, top_k)
    m = int(c.size)
    if top_k >= m:
        return np.arange(m, dtype=int)
    idx = np.argpartition(c, -top_k)[-top_k:].astype(int, copy=True)
    max_idx = int(np.argmax(c))
    if max_idx not in idx:
        pos = int(np.argmin(c[idx]))
        idx[pos] = max_idx
    return idx


def topk_weights(c, idx, mode="prop"):
    c = np.asarray(c, dtype=np.float64)
    idx = np.asarray(idx, dtype=int)
    if idx.size == 0:
        return np.zeros(0, dtype=np.float64)
    if mode == "equal":
        return np.full(idx.size, 1.0 / float(idx.size), dtype=np.float64)
    w = c[idx].astype(np.float64, copy=True)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 0.0:
        w.fill(1.0 / float(idx.size))
    else:
        w = w / s
    return w


def topk_subgradient(a, idx, w, n):
    n = int(n)
    a = np.asarray(a, dtype=np.float64)
    idx = np.asarray(idx, dtype=int)
    w = np.asarray(w, dtype=np.float64)
    g = np.zeros(n, dtype=np.float64)
    for j in range(idx.size):
        k = int(idx[j])
        wj = float(w[j])
        if wj <= 0.0:
            continue
        lo = 0 if k - n + 1 < 0 else int(k - n + 1)
        hi = n - 1 if k > n - 1 else int(k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * wj * a[k - hi:k - lo + 1][::-1]
    return g


def project_simplex(v, S):
    v = np.asarray(v, dtype=np.float64).reshape(-1)
    n = int(v.size)
    if n == 0:
        return v
    w = np.sort(v)[::-1]
    cumsum = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    theta = (cumsum - float(S)) / idx
    valid = theta <= w
    if not np.any(valid):
        rho = n
    else:
        rho = int(np.max(np.where(valid)[0]) + 1)
    theta = (cumsum[rho - 1] - float(S)) / float(rho)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s <= 1e-15:
        out = np.full(n, float(S) / n, dtype=np.float64)
    else:
        out = out * (float(S) / s)
    return out


def upsample_sequence(seq, m):
    seq = np.asarray(seq, dtype=np.float64).reshape(-1)
    n = int(seq.size)
    m = int(max(1, m))
    if m <= n:
        return seq.copy()
    if n == 0:
        return np.ones(m, dtype=np.float64)
    if n == 1:
        return np.full(m, float(seq[0]), dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, m)
    vals = np.interp(x_new, x_old, seq)
    vals = np.maximum(vals, 0.0)
    if float(vals.sum()) <= 1e-15:
        vals = np.ones(m, dtype=np.float64)
    return vals


def downsample_sequence(seq, m):
    seq = np.asarray(seq, dtype=np.float64).reshape(-1)
    n = int(seq.size)
    m = int(max(1, m))
    if m >= n:
        return seq.copy()
    if m <= 0:
        return np.zeros(0, dtype=np.float64)
    out = []
    blocks = np.array_split(seq, m)
    for b in blocks:
        if b.size:
            out.append(float(np.mean(b)))
        else:
            out.append(0.0)
    return np.asarray(out, dtype=np.float64)


def random_smooth(n, rng):
    n = int(max(32, n))
    x = np.arange(n, dtype=np.float64)
    v = np.ones(n, dtype=np.float64)
    for f in range(1, 5):
        amp = 0.22 / (float(f) + 0.5)
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        v += amp * np.cos(2.0 * np.pi * f * x / float(n) + phase)
    nb = int(rng.integers(1, 4))
    for _ in range(nb):
        center = float(rng.uniform(0.0, float(n)))
        width = float(rng.uniform(0.05 * n, 0.20 * n))
        amp = float(rng.uniform(0.04, 0.18))
        v += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
    v = np.maximum(v, 0.05)
    return v


def sort_results(results):
    tmp = []
    for i, r in enumerate(results):
        # r = [score, a, name]
        tmp.append((r[0], i, r[2], r[1]))
    tmp.sort()
    out = []
    for item in tmp:
        out.append([item[0], item[3], item[2]])
    return out


def run_soft_mirror(a_init, S, budget, rng, T, eta=0.15, eta_max=0.45,
                    max_tries=5, stall_s=4.0):
    a_init = np.asarray(a_init, dtype=np.float64).reshape(-1)
    n = int(a_init.size)
    if n == 0:
        return a_init, 0.0, a_init, 0.0
    a = positive_for_run(safe_normalize(a_init, S), S)
    if budget <= 0.0:
        c = calc_c(a)
        s = float(c.max())
        return a.copy(), s, a.copy(), s

    fsize = _next_pow2(3 * n - 2)
    A = fft(a, fsize)
    c = c_from_A(A, n)
    cur_score = float(c.max())
    best_score = cur_score
    best_a = a.copy()
    val = smooth_value(c, T)
    eta_cur = float(eta)
    start = time.monotonic()
    last_progress = start
    it = 0

    while time.monotonic() - start < float(budget):
        now = time.monotonic()
        if now - last_progress > float(stall_s):
            if float(rng.random()) < 0.6:
                base = best_a
            else:
                base = a
            scale = max(1e-10, 0.08 * float(S) / n)
            a = positive_for_run(safe_normalize(base + rng.normal(0.0, scale, n), S), S)
            A = fft(a, fsize)
            c = c_from_A(A, n)
            cur_score = float(c.max())
            val = smooth_value(c, T)
            last_progress = now

        w = soft_weights(c, T)
        g = dense_gradient(a, A, w, fsize, n)
        gmax = float(g.max())
        if not np.isfinite(gmax) or gmax <= 1e-15:
            break
        g = g / gmax
        old_val = float(val)
        old_max = float(cur_score)
        accepted = False
        cand = None

        base_max = max(old_max, best_score)
        if base_max > 1.8:
            max_tol = 0.30
        elif base_max > 1.6:
            max_tol = 0.15
        elif base_max > 1.5:
            max_tol = 0.05
        else:
            max_tol = 0.02

        for tr in range(max_tries):
            a_new = a * np.exp(-eta_cur * g)
            a_new = np.maximum(a_new, 1e-300)
            s = float(a_new.sum())
            if not np.isfinite(s) or s <= 1e-15:
                eta_cur *= 0.5
                continue
            a_new = a_new * (float(S) / s)
            A_new = fft(a_new, fsize)
            c_new = c_from_A(A_new, n)
            new_max = float(c_new.max())
            new_val = smooth_value(c_new, T)
            if not np.isfinite(new_val):
                eta_cur *= 0.5
                continue

            if cand is None or new_val < cand[3] or new_max < cand[4]:
                cand = (a_new, A_new, c_new, new_val, new_max)

            if (new_val <= old_val + 1e-11) and (new_max <= base_max + max_tol):
                a = a_new
                A = A_new
                c = c_new
                cur_score = new_max
                val = new_val
                if new_max < best_score:
                    best_score = new_max
                    best_a = a.copy()
                if new_val < old_val - 1e-11:
                    last_progress = time.monotonic()
                if tr == 0:
                    eta_cur = min(float(eta_max), eta_cur * 1.05)
                accepted = True
                break
            else:
                eta_cur *= 0.5
                if eta_cur < 1e-6:
                    break

        if not accepted:
            if cand is not None:
                a_c, A_c, c_c, v_c, mx_c = cand
                if mx_c < best_score or v_c < old_val - 1e-8:
                    a = a_c
                    A = A_c
                    c = c_c
                    cur_score = mx_c
                    val = v_c
                    if mx_c < best_score:
                        best_score = mx_c
                        best_a = a.copy()
                    last_progress = time.monotonic()
                    accepted = True

            if not accepted:
                base = best_a if float(rng.random()) < 0.7 else a
                scale = max(1e-10, 0.12 * float(S) / n)
                a = positive_for_run(safe_normalize(base + rng.normal(0.0, scale, n), S), S)
                A = fft(a, fsize)
                c = c_from_A(A, n)
                cur_score = float(c.max())
                val = smooth_value(c, T)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                last_progress = time.monotonic()
                eta_cur = max(1e-5, eta_cur * 0.5)

        it += 1
        if it % 500 == 0:
            print("soft_mirror n=%d T=%.3f cur=%.6f best=%.6f eta=%.3f"
                  % (n, T, cur_score, best_score, eta_cur))

    return best_a, best_score, a, cur_score


def run_topk_mirror(a_init, S, budget, rng, top_k=16, eta=0.15,
                    eta_max=0.60, mode="prop", max_tries=5, stall_s=5.0):
    a_init = np.asarray(a_init, dtype=np.float64).reshape(-1)
    n = int(a_init.size)
    if n == 0:
        return a_init, 0.0, a_init, 0.0
    a = positive_for_run(safe_normalize(a_init, S), S)
    if budget <= 0.0:
        c = calc_c(a)
        s = float(c.max())
        return a.copy(), s, a.copy(), s

    fsize = _next_pow2(2 * n - 1)
    c = calc_c_padded(a, fsize, n)
    cur_score = float(c.max())
    best_score = cur_score
    best_a = a.copy()
    eta_cur = float(eta)
    start = time.monotonic()
    last_progress = start
    it = 0

    while time.monotonic() - start < float(budget):
        now = time.monotonic()
        if now - last_progress > float(stall_s):
            base = best_a if float(rng.random()) < 0.65 else a
            scale = max(1e-10, 0.08 * float(S) / n)
            a = positive_for_run(safe_normalize(base + rng.normal(0.0, scale, n), S), S)
            c = calc_c_padded(a, fsize, n)
            cur_score = float(c.max())
            last_progress = now

        idx = select_top_idx(c, top_k)
        w = topk_weights(c, idx, mode)
        g = topk_subgradient(a, idx, w, n)
        gmax = float(g.max())
        if not np.isfinite(gmax) or gmax <= 1e-15:
            break
        g = g / gmax
        old = float(cur_score)

        up_tol = 0.0
        if it < 250:
            up_tol = 0.003 * (1.0 - it / 250.0)

        accepted = False
        cand = None
        for tr in range(max_tries):
            a_new = a * np.exp(-eta_cur * g)
            a_new = np.maximum(a_new, 1e-300)
            s = float(a_new.sum())
            if not np.isfinite(s) or s <= 1e-15:
                eta_cur *= 0.5
                continue
            a_new = a_new * (float(S) / s)
            c_new = calc_c_padded(a_new, fsize, n)
            new = float(c_new.max())
            if not np.isfinite(new):
                eta_cur *= 0.5
                continue

            if cand is None or new < cand[2]:
                cand = (a_new, c_new, new)

            if (new <= old + up_tol) or (new < best_score - 1e-14):
                a = a_new
                c = c_new
                cur_score = new
                if new < best_score:
                    best_score = new
                    best_a = a.copy()
                if new < old - 1e-11 or new < best_score:
                    last_progress = time.monotonic()
                if tr == 0:
                    eta_cur = min(float(eta_max), eta_cur * 1.05)
                accepted = True
                break
            else:
                eta_cur *= 0.5
                if eta_cur < 1e-6:
                    break

        if not accepted:
            if cand is not None:
                a_c, c_c, new_c = cand
                if new_c < best_score or new_c < old:
                    a = a_c
                    c = c_c
                    cur_score = new_c
                    if new_c < best_score:
                        best_score = new_c
                        best_a = a.copy()
                    if new_c < old - 1e-11:
                        last_progress = time.monotonic()
                    accepted = True
            if not accepted:
                base = best_a if float(rng.random()) < 0.7 else a
                scale = max(1e-10, 0.12 * float(S) / n)
                a = positive_for_run(safe_normalize(base + rng.normal(0.0, scale, n), S), S)
                c = calc_c_padded(a, fsize, n)
                cur_score = float(c.max())
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                last_progress = time.monotonic()
                eta_cur = max(1e-5, eta_cur * 0.5)

        it += 1
        if it % 1000 == 0:
            print("topk_mirror n=%d k=%d cur=%.6f best=%.6f eta=%.3f"
                  % (n, top_k, cur_score, best_score, eta_cur))

    return best_a, best_score, a, cur_score


def run_adam_topk(a_init, S, budget, rng, top_k=8, lr=None):
    a_init = np.asarray(a_init, dtype=np.float64).reshape(-1)
    n = int(a_init.size)
    if n == 0:
        return a_init, 0.0, a_init, 0.0
    a = positive_for_run(safe_normalize(a_init, S), S)
    if budget <= 0.0:
        c = calc_c(a)
        s = float(c.max())
        return a.copy(), s, a.copy(), s

    fsize = _next_pow2(2 * n - 1)
    c = calc_c_padded(a, fsize, n)
    cur_score = float(c.max())
    best_score = cur_score
    best_a = a.copy()

    if lr is None:
        lr = 0.5 * float(S) / n
    lr = float(lr)
    lr_init = lr
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    b1 = 0.85
    b2 = 0.999
    eps = 1e-8
    t = 0
    start = time.monotonic()

    while time.monotonic() - start < float(budget):
        now = time.monotonic()
        cur = float(c.max())
        if cur < best_score:
            best_score = cur
            best_a = a.copy()

        idx = select_top_idx(c, top_k)
        w = topk_weights(c, idx, "prop")
        g = topk_subgradient(a, idx, w, n)
        gnorm = float(np.linalg.norm(g))
        if not np.isfinite(gnorm) or gnorm <= 1e-13:
            break
        g = g / gnorm
        t += 1

        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * (g * g)
        mhat = m / (1.0 - b1 ** t)
        vhat = v / (1.0 - b2 ** t)
        a2 = a - lr * (mhat / (np.sqrt(vhat) + eps))
        a = project_simplex(a2, S)
        if float(np.min(a)) <= 0.0:
            a = positive_for_run(a, S)

        lr = lr_init * (0.5 ** (float(t) / 400.0))

        if t % 5000 == 0:
            a = project_simplex(a + rng.normal(0.0, 0.05 * float(S) / n, n), S)
            a = positive_for_run(a, S)

        c = calc_c_padded(a, fsize, n)
        cur_score = float(c.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if t % 1000 == 0:
            print("adam_topk n=%d k=%d cur=%.6f best=%.6f lr=%.5f"
                  % (n, top_k, cur_score, best_score, lr))

    return best_a, best_score, a, cur_score


def polish_candidate(a, budget, rng, level="full"):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = int(a.size)
    if n == 0:
        a = np.ones(1024, dtype=np.float64)
        n = int(a.size)
    S = np.sqrt(2.0 * float(n))
    a = safe_normalize(a, S)
    c = calc_c(a)
    best_score = float(c.max())
    best_a = a.copy()

    floor = max(1e-12, 1e-9 * float(S) / n)
    work = np.maximum(a, floor)
    ws = float(work.sum())
    if ws <= 1e-15:
        work = np.full(n, float(S) / n, dtype=np.float64)
    else:
        work = work * (float(S) / ws)

    B = max(0.0, float(budget))
    start = time.monotonic()

    def _finite(x):
        x = np.asarray(x)
        return bool(x.size > 0 and np.all(np.isfinite(x)))

    if level == "quick":
        used = time.monotonic() - start
        t1 = min(0.55 * B, B - used)
        if t1 > 1.0:
            ba, bs, ca, cs = run_soft_mirror(work, S, t1, rng, T=0.22, eta=0.16)
            if np.isfinite(bs) and bs < best_score:
                best_score = bs
                best_a = ba.copy()
            if _finite(ca):
                work = ca
        else:
            t1 = 0.0
        used = time.monotonic() - start
        t2 = min(0.45 * B, B - used)
        if t2 > 1.0:
            tk = 16 if n <= 9000 else 10
            ba, bs, ca, cs = run_topk_mirror(work, S, t2, rng, top_k=tk, eta=0.16)
            if np.isfinite(bs) and bs < best_score:
                best_score = bs
                best_a = ba.copy()
            if _finite(ba):
                work = ba
            elif _finite(ca):
                work = ca

        return best_a, best_score

    # full polishing schedule
    if n <= 9000:
        stages = [
            ("soft", 0.25, 0.08),
            ("soft", 0.13, 0.12),
            ("soft", 0.065, 0.15),
            ("soft", 0.035, 0.15),
            ("topk", 32, 0.15),
            ("topk", 12, 0.15),
            ("adam", 8, 0.10),
            ("topk", 4, 0.10),
        ]
    else:
        stages = [
            ("soft", 0.12, 0.12),
            ("soft", 0.06, 0.12),
            ("topk", 16, 0.25),
            ("topk", 8, 0.25),
            ("topk", 4, 0.26),
        ]

    for kind, param, frac in stages:
        elapsed = time.monotonic() - start
        t = frac * B
        t = min(t, B - elapsed)
        if t < 0.5:
            continue

        if kind == "soft":
            ba, bs, ca, cs = run_soft_mirror(work, S, t, rng, T=param, eta=0.15)
        elif kind == "topk":
            tk = int(param)
            eta = 0.15 if tk >= 16 else 0.12
            ba, bs, ca, cs = run_topk_mirror(work, S, t, rng, top_k=tk, eta=eta)
        elif kind == "adam":
            ba, bs, ca, cs = run_adam_topk(work, S, t, rng, top_k=int(param), lr=None)
        else:
            continue

        if np.isfinite(bs) and bs < best_score:
            best_score = bs
            best_a = ba.copy()

        if kind == "soft":
            if _finite(ca):
                work = ca
            elif _finite(ba):
                work = ba
        else:
            if _finite(ba) and (bs < cs or not np.isfinite(cs)):
                work = positive_for_run(best_a if bs <= best_score else ba, S)
            elif _finite(ca):
                work = positive_for_run(ca, S)
            elif _finite(ba):
                work = positive_for_run(ba, S)

    elapsed = time.monotonic() - start
    final_t = B - elapsed
    if final_t > 3.0:
        base = positive_for_run(best_a, S)
        tk = 6 if n <= 9000 else 4
        ba, bs, ca, cs = run_topk_mirror(base, S, min(final_t, 2.0), rng, top_k=tk, eta=0.11)
        if np.isfinite(bs) and bs < best_score:
            best_score = bs
            best_a = ba.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    total_budget = max(120.0, float(budget_s))
    reserve = 35.0
    deadline = time.monotonic() + max(1.0, total_budget - reserve)

    raw_inits = []
    given = None
    try:
        given = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
        given = np.nan_to_num(given, nan=0.0, posinf=1000.0, neginf=0.0)
        given = np.clip(given, 0.0, 1000.0)
        if given.size >= 32 and float(given.sum()) > 1e-12:
            raw_inits.append(("given", given.copy()))
    except Exception as exc:
        print("no previous sequence:", exc)

    if given is not None:
        n0 = int(given.size)
    else:
        n0 = 8192

    if n0 > 0 and n0 <= 20000:
        max_n = max(16384, min(n0, 16384))
    else:
        max_n = 16384

    lengths = set()
    if n0 > 0:
        lengths.add(n0)
    for factor in (1.25, 1.5, 2.0):
        m = int(round(n0 * factor))
        m = max(1024, min(m, max_n))
        lengths.add(m)

    if given is None:
        for m in (4096, 8192, 12288):
            lengths.add(m)

    lengths = sorted([m for m in lengths if 512 <= m <= max_n])
    if not lengths:
        lengths = [8192]

    for m in lengths:
        if given is not None:
            if m == n0:
                arr = given
            elif m > n0:
                arr = upsample_sequence(given, m)
            else:
                arr = downsample_sequence(given, m)
        else:
            arr = random_smooth(m, rng)
        raw_inits.append(("len%d" % m, arr))

    if given is not None and n0 >= 64:
        k = 5 if n0 > 20 else 3
        filt = np.convolve(given, np.ones(k) / float(k), mode="same")
        raw_inits.append(("filt%d" % n0, filt))
        x = np.arange(n0, dtype=np.float64)
        for f, amp in ((1, 0.08), (2, 0.08)):
            mod = 1.0 + amp * np.cos(2.0 * np.pi * f * x / float(n0))
            raw_inits.append(("mod%d" % f, given * mod))

    rand_lens = []
    rand_lens.append(n0)
    m2 = int(round(1.5 * n0))
    m2 = max(1024, min(m2, max_n))
    if m2 not in rand_lens:
        rand_lens.append(m2)
    for m in rand_lens:
        raw_inits.append(("rand%d" % m, random_smooth(m, rng)))

    if not raw_inits:
        raw_inits.append(("ones", np.ones(8192, dtype=np.float64)))

    scored = []
    for name, arr in raw_inits:
        try:
            a = safe_normalize(arr, None)
            if a.size < 32:
                continue
            c = calc_c(a)
            s = float(c.max())
            if np.isfinite(s):
                scored.append((s, name, a))
        except Exception as exc:
            pass

    if not scored:
        a = safe_normalize(np.ones(8192, dtype=np.float64), None)
        return [float(x) for x in a]

    best_by_len = {}
    for s, name, a in scored:
        L = int(a.size)
        if L not in best_by_len or s < best_by_len[L][0]:
            best_by_len[L] = (s, name, a)

    init_items = []
    for item in best_by_len.values():
        init_items.append(item)
    tmp = []
    for i, item in enumerate(init_items):
        tmp.append((item[0], i, item[1], item[2]))
    tmp.sort()
    top_init = []
    for item in tmp[:4]:
        top_init.append((item[1], item[3]))

    results = []
    if top_init:
        avail = max(0.0, deadline - time.monotonic() - reserve - 5.0)
        quick_total = min(150.0, max(40.0, avail * 0.07))
        per_quick = quick_total / float(len(top_init))
        for name, a in top_init:
            if time.monotonic() > deadline - reserve - 5.0:
                break
            t = min(per_quick, max(0.0, deadline - time.monotonic() - reserve - 5.0))
            if t < 2.0:
                continue
            print("quick polish %s tint=%.1f" % (name, t))
            ba, bs = polish_candidate(a, t, rng, "quick")
            if np.isfinite(bs):
                results.append([bs, ba, name + "-quick"])

    if not results:
        for name, a in top_init:
            try:
                s = float(calc_c(a).max())
                if np.isfinite(s):
                    results.append([s, a, name])
            except Exception:
                pass

    if not results:
        a = safe_normalize(np.ones(8192, dtype=np.float64), None)
        return [float(x) for x in a]

    results = sort_results(results)
    print("after quick best=%.6f (%s)" % (results[0][0], results[0][2]))

    R0 = max(0.0, deadline - time.monotonic() - reserve)
    if R0 > 30.0:
        long_cands = results[:3]
        base_score = long_cands[0][0]
        if len(long_cands) == 1:
            allocs = [0.90]
        elif len(long_cands) == 2:
            allocs = [0.60, 0.30]
            if long_cands[1][0] > base_score + 0.04:
                allocs[1] *= 0.7
        else:
            allocs = [0.50, 0.25, 0.15]
            if long_cands[1][0] > base_score + 0.04:
                allocs[1] *= 0.65
            if long_cands[2][0] > base_score + 0.07:
                allocs[2] *= 0.6

        s_alloc = sum(allocs[: len(long_cands)])
        if s_alloc <= 0.0:
            s_alloc = 1.0

        for i, cand in enumerate(long_cands):
            if time.monotonic() > deadline - reserve - 10.0:
                break
            t = R0 * (allocs[i] / s_alloc)
            if t < 5.0:
                continue
            print("full polish %s budget=%.1f" % (cand[2], t))
            ba, bs = polish_candidate(cand[1], t, rng, "full")
            if np.isfinite(bs):
                results.append([bs, ba, cand[2] + "-full"])

    results = sort_results(results)
    print("after full best=%.6f (%s)" % (results[0][0], results[0][2]))

    Rfinal = max(0.0, deadline - time.monotonic() - reserve)
    if results and Rfinal > 12.0:
        best_score, best_a, best_name = results[0]
        t = min(Rfinal * 0.25, 220.0)
        if t > 5.0:
            print("final polish %s budget=%.1f" % (best_name, t))
            ba, bs = polish_candidate(best_a, t, rng, "full" if t > 120.0 else "quick")
            if np.isfinite(bs):
                results.append([bs, ba, best_name + "-final2"])
            results = sort_results(results)

    if not results:
        out = safe_normalize(np.ones(8192, dtype=np.float64), None)
        return [float(x) for x in out]

    best_return = results[0][1]
    best_fft_score = results[0][0]
    best_eval_value = best_fft_score

    top_eval = []
    seen_lens = set()
    for r in results:
        L = int(r[1].size)
        if L in seen_lens:
            continue
        seen_lens.add(L)
        top_eval.append((r[1], r[0], r[2]))
        if len(top_eval) >= 3:
            break

    if "evaluate_sequence" in globals():
        for a, s, name in top_eval:
            if time.monotonic() > deadline - 4.0:
                break
            L = int(a.size)
            if L > 16384:
                continue
            if L > 12288 and time.monotonic() > deadline - 18.0:
                continue
            try:
                val = float(evaluate_sequence([float(x) for x in a]))
            except Exception as exc:
                val = np.inf
                print("eval failed for %s: %s" % (name, exc))
            if np.isfinite(val):
                print("evaluate_sequence %s %.6f" % (name, val))
                if val < best_eval_value:
                    best_eval_value = val
                    best_return = a

    out = np.asarray(best_return, dtype=np.float64).reshape(-1)
    out = np.nan_to_num(out, nan=0.0, posinf=1000.0, neginf=0.0)
    out = np.clip(out, 0.0, 1000.0)
    if out.size == 0:
        out = np.ones(8192, dtype=np.float64)
    s = float(out.sum())
    if s <= 0.01 or not np.isfinite(s):
        out = safe_normalize(np.ones(8192, dtype=np.float64), None)
    else:
        out = out * (np.sqrt(2.0 * float(out.size)) / s)
        out = np.clip(out, 0.0, 1000.0)
    if out.size > 20000:
        out = downsample_sequence(out, 16384)
        out = safe_normalize(out, None)

    return [float(x) for x in out]
```
