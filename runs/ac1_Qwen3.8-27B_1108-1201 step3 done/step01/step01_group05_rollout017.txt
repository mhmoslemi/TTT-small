```python
"""
Step-function search minimizing 2n*max(conv(p,p))/(sum(p))^2.
Uses iterated LP best-response with gradient refinement and multi-length search.
"""

import time
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _next_fast_len(n):
    try:
        return next_fast_len(n)
    except:
        m = 1
        while m < n:
            m <<= 1
        return m

def conv_full(p):
    n = len(p)
    L = _next_fast_len(2*n - 1)
    F = fft(p, L)
    c = np.real(ifft(F * F))
    return c[:2*n-1]

def score(p):
    n = len(p)
    s = np.sum(p)
    if s < 1e-12:
        return np.inf
    c = conv_full(p)
    return 2.0 * n * np.max(c) / (s*s)

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssum = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx > (cssum - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0/n)
    k = np.count_nonzero(cond) - 1
    theta = (cssum[k] - 1.0) / (k + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s < 1e-12:
        return np.full(n, 1.0/n)
    return out / s

def normalize(p):
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-12:
        return np.full(len(p), 1.0/len(p))
    return p / s

def lp_solve(p, M, n):
    """Solve: max sum(g) s.t. (p*g)[k] <= M for all k, g >= 0."""
    # Build sparse matrix A where A[k, j] = sum_i p[i] * delta[i + (k-j) == valid]
    # Actually A[k, j] = (p*g)[k] when g has 1 at position j and 0 elsewhere
    # (p*g)[k] = sum_i p[i] * g[k-i], so if g = e_j, (p*g)[k] = p[k-j] (if valid)
    # So A[k, j] = p[k-j] if 0 <= k-j <= n-1, else 0
    
    # A is (2n-1) x n
    # A[k, j] = p[k-j] if 0 <= k-j <= n-1
    
    # This is a Toeplitz-like matrix (actually it's the convolution matrix)
    # Row k, column j: A[k,j] = p[k-j] if 0 <= k-j < n, else 0
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(p[idx])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs', options={'disp': False, 'presolve': True})
    
    if res is None or not res.success:
        return None
    g = res.x
    g = np.maximum(g, 0)
    s = g.sum()
    if s < 1e-12:
        return None
    return g / s

def optimize_lp_iterative(p, rng, time_limit):
    """Main optimization loop using iterated LP."""
    n = len(p)
    p = project_simplex(p.copy())
    best_p = p.copy()
    best_sc = score(p)
    end = time.monotonic() + time_limit
    
    no_improve = 0
    lp_count = 0
    
    while time.monotonic() < end:
        # Compute convolution and max
        c = conv_full(p)
        M = np.max(c)
        
        # Solve LP
        remaining = end - time.monotonic()
        if remaining < 2.0:
            break
        
        g = lp_solve(p, M, n)
        lp_count += 1
        
        if g is None:
            break
        
        # Try mixing with various t values
        improved = False
        for t in [1.0, 0.9, 0.7, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02]:
            q = project_simplex((1-t)*p + t*g)
            sc = score(q)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_p = q.copy()
                p = q
                improved = True
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            # Try a perturbation to escape
            if no_improve > 3:
                p = project_simplex(best_p + rng.normal(0, 0.1/n, n))
                no_improve = 0
            elif no_improve > 10:
                # More aggressive restart
                p = project_simplex(0.7*best_p + 0.3*g + rng.normal(0, 0.2/n, n))
                no_improve = 0
        
        if lp_count % 10 == 0:
            print(f"  LP iter {lp_count}: best={best_sc:.8f}", flush=True)
    
    return best_p, best_sc
```

But wait, I realize there's an issue with the LP formulation. Let me re-derive it.

The problem is: min_p max_k (p*p)_k subject to sum(p) = 1, p >= 0.

At the current point p, we want to find a direction d such that moving to p + t*d reduces the max.

The LP approach from the previous code is:
- Given f = p (normalized), M = max(f*f)
- Solve: max sum(g) subject to (f*g)_k <= M for all k, g >= 0
- Then mix: p_new = (1-t)*p + t*g

Why does this work? Consider the new sequence q = (1-t)*p + t*g (after renormalization to sum 1).

The autocorrelation of q:
(q*q)_k = (1-t)^2 (p*p)_k + 2t(1-t)(p*g)_k + t^2 (g*g)_k

We know (p*p)_k <= M for all k, and (p*g)_k <= M for all k (by LP constraint).
So (q*q)_k <= (1-t)^2 * M + 2t(1-t) * M + t^2 * max(g*g)
= M * [(1-t)^2 + 2t(1-t)] + t^2 * max(g*g)
= M * (1-t^2) + t^2 * max(g*g)

For this to be less than M, we need: M(1-t^2) + t^2 * max(g*g) < M
=> t^2 * max(g*g) < t^2 * M
=> max(g*g) < M

So if max(g*g) < M, then for any t in (0,1), the new max autocorrelation is strictly less than M. This is the key property!

But we need to verify that max(g*g) < M. Since g is normalized (sum = 1), and g >= 0:
max(g*g) >= (sum g)^2 / (2n-1) = 1/(2n-1) (by the "balanced" bound)
max(g*g) <= max(g)^2 * n (crude bound)

In practice, the LP solution g tends to be more "spread out" than p (because the LP objective is to maximize sum, which favors spreading), so max(g*g) is often less than max(p*p) = M.

This gives us a guarantee of monotone decrease in max autocorrelation, which is great!

However, the mixing ratio t matters for the rate of decrease. A larger t gives more decrease per step but risks the "t^2 * max(g*g)" term being too large.

- We can express this as: for each k, sum_j p[k-j] * g[j] <= M (where the sum is over valid j).

```python
def precompute_indices(n):
    all_rows = []
    all_cols = []
    all_val_idx = []  # index into p for each entry
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2*n - 2) + 1
        ks = np.arange(k_start, k_end)
        all_rows.append(ks)
        all_cols.append(np.full(len(ks), j, dtype=np.int32))
        all_val_idx.append(ks - j)
    return np.concatenate(all_rows).astype(np.int32), np.concatenate(all_cols).astype(np.int32), np.concatenate(all_val_idx).astype(np.int32)

def build_matrix_from_p(p, indices):
    rows, cols, val_idx = indices
    vals = p[val_idx]
    n = len(p)
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
```
