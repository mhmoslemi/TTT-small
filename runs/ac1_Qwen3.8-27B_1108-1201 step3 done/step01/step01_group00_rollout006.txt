    n = len(a)
    S = np.sqrt(2.0 * n)

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

    """Autocorrelation (linear self-convolution)."""
    return fft_convolve(a, a)

    if s < 1e-15:
        return a
    return a * target / s

        idx_hi = k - j_min
        idx_lo = k - j_max
        A[k, j_min:j_max+1] = a[idx_lo:idx_hi+1][::-1]

        A = build_conv_matrix(a)

        try:

            print(f"  LP error at iter {it}: {e}")
            break
        
        if not result.success:

        sum_b = np.sum(b)

        g = normalize_sum(b, S)
