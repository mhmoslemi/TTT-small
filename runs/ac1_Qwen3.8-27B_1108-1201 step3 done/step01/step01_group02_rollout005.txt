import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time

def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

def build_active_constraints(a, M, top_k=20):
    """Build LP constraints only for the top-k peaks."""
    n = len(a)
    conv = np.convolve(a, a)
    top_indices = np.argsort(conv)[-top_k:]
    # For each active constraint k, the constraint is sum_j a_j * g_{k-j} <= M
    # This is a linear constraint in g
    # (a * g)_k = sum_j a_j * g[k-j]
    # We need to express this as a row in the constraint matrix
    
    n_rows = len(top_indices)
    # Build sparse matrix
    row_idx = []
    col_idx = []
    data = []
    for i, k in enumerate(top_indices):
        # (a * g)_k = sum_j a_j * g[k-j]
        # g index = k - j, must be in [0, n-1]
        # So j must be in [max(0, k-n+1), min(n-1, k)]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            g_idx = k - j
            row_idx.append(i)
            col_idx.append(g_idx)
            data.append(a[j])
    
    A = csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))
    b = np.full(n_rows, M)
    return A, b

def solve_lp_step(a, M, top_k=20, timeout=10.0):
    n = len(a)
    A, b = build_active_constraints(a, M, top_k)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None

def lp_refine(a, time_limit_s=300.0, verbose=True):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    it = 0
    max_iters = 1000
    
    while time.monotonic() - start_time < time_limit_s and it < max_iters:
        elapsed = time.monotonic() - start_time
        conv = np.convolve(a, a)
        M = conv.max()
        score = 2.0 * n * M / (a.sum() ** 2)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 50:
            break
        
        # Use adaptive top_k: more constraints when close to optimum
        top_k = min(30, max(10, int(30 - stagnant * 0.3)))
        
        remaining = time_limit_s - elapsed
        if remaining < 2.0:
            break
        timeout = max(1.0, min(20.0, remaining * 0.2))
        
        g = solve_lp_step(a, M, top_k=top_k, timeout=timeout)
        if g is None:
            break
        
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        
        g = g / sum_g
        alpha = min(0.5, max(0.03, 1.0 / (1.0 + 0.2 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-12:
            a = a_new
        else:
            alpha2 = alpha * 0.1
            a_new = (1.0 - alpha2) * a + alpha2 * g
            a_new = normalize_nonneg(a_new)
            a = a_new
        it += 1
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    return best_a, best_score

Multi-strategy optimization for minimum peak autocorrelation of non-negative sequences.
Combines LP-based refinement, spectral flattening, peak shaving, and local search.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def normalize_nonneg(a):
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25, center_frac=0.5):
    t = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_blackman(n):
    if n == 1:
        return np.ones(1, dtype=np.float64)
    t = np.arange(n, dtype=np.float64)
    x = t / (n - 1)
    return np.maximum(0.0, 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x))


def initial_ramp(n, bias=0.3):
    """Linear ramp from (1-bias) to (1+bias)."""
    t = np.linspace(1.0 - bias, 1.0 + bias, n)
    return t


def initial_two_bumps(n, rng):
    """Two gaussian bumps at random positions."""
    a = np.zeros(n)
    for _ in range(2):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 8), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    return a


def build_active_constraints(a, M, top_k=20):
    n = len(a)
    conv = np.convolve(a, a)
    top_indices = np.argsort(conv)[-top_k:]
    
    n_rows = len(top_indices)
    row_idx = []
    col_idx = []
    data = []
    for i, k in enumerate(top_indices):
        j_start = max(0, int(k) - n + 1)
        j_end = min(n - 1, int(k))
        for j in range(j_start, j_end + 1):
            g_idx = int(k) - j
            row_idx.append(i)
            col_idx.append(g_idx)
            data.append(a[j])
    
    A = csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))
    b = np.full(n_rows, M)
    return A, b


def solve_lp_step(a, M, top_k=20, timeout=10.0):
    n = len(a)
    A, b = build_active_constraints(a, M, top_k)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None


def lp_refine(a, time_limit_s=300.0, verbose=False):
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    stagnant = 0
    it = 0
    
    while time.monotonic() - start_time < time_limit_s:
        elapsed = time.monotonic() - start_time
        conv = np.convolve(a, a)
        M = conv.max()
        score = 2.0 * n * M / (a.sum() ** 2)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 60:
            break
        
        top_k = min(40, max(10, 30 - stagnant // 5))
        remaining = time_limit_s - elapsed
        if remaining < 2.0:
            break
        timeout = max(1.0, min(15.0, remaining * 0.25))
        
        g = solve_lp_step(a, M, top_k=top_k, timeout=timeout)
        if g is None:
            break
        
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
            break
        
        g = g / sum_g
        alpha = min(0.4, max(0.03, 0.8 / (1.0 + 0.3 * it)))
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = normalize_nonneg(a_new)
        
        new_score = score_sequence(a_new)
        if new_score <= score + 1e-12:
            a = a_new
        else:
            alpha2 = alpha * 0.1
            a_new = (1.0 - alpha2) * a + alpha2 * g
            a_new = normalize_nonneg(a_new)
            a = a_new
        it += 1
    
    final_score = score_sequence(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    return best_a, best_score


def spectral_flatten(a, time_limit_s=60.0, verbose=False):
    """Iteratively flatten the power spectrum to reduce autocorrelation peak."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    # Use zero-padded FFT for efficiency
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size *= 2
    
    while time.monotonic() - start_time < time_limit_s:
        A_hat = np.fft.rfft(a, n=fft_size)
        P = np.abs(A_hat) ** 2
        P_mean = P.mean()
        
        # Soft flattening: compress dynamic range
        alpha = 0.3  # compression strength
        P_flat = P ** alpha
        P_flat = P_flat / P_flat[0] * P[0]  # preserve DC
        
        # Reconstruct
        phase = np.angle(A_hat)
        A_hat_new = np.sqrt(P_flat / P_flat[0] * P[0]) * np.exp(1j * phase)
        # Actually, let me be more careful
        mag_new = np.sqrt(P_flat)
        mag_new = mag_new * (np.sqrt(P[0]) / mag_new[0])  # preserve DC magnitude
        A_hat_new = mag_new * np.exp(1j * phase)
        
        a_new = np.fft.irfft(A_hat_new, n=fft_size)[:n]
        a_new = np.maximum(0.0, a_new)
        a_new = normalize_nonneg(a_new)
        
        score = score_sequence(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new.copy()
    
    return best_a, best_score


def peak_shave(a, time_limit_s=60.0, verbose=False):
    """Directly reduce values contributing most to the peak autocorrelation."""
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    
    while time.monotonic() - start_time < time_limit_s:
        conv = np.convolve(a, a)
        k_star = int(np.argmax(conv))
        
        # Find the top contributing indices
        # (a*a)_{k_star} = sum_j a_j * a_{k_star - j}
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        contributions = np.array([a[j] * a[k_star - j] for j in range(j_start, j_end + 1)])
        top_contrib = np.argsort(contributions)[-5:]
        top_indices = np.array([j_start + i for i in top_contrib])
        
        # Reduce top contributors and redistribute to random positions
        reduce_amt = 0.01 * a[top_indices].mean()
        total_reduced = 0.0
        mask = np.zeros(n, dtype=bool)
        for idx in top_indices:
            amt = min(a[idx], reduce_amt)
            a[idx] -= amt
            total_reduced += amt
            mask[idx] = True
        
        # Redistribute to random non-contributing positions
        available = np.where(~mask)[0]
        if len(available) > 0:
            targets = available[rng_peak_shave.integers(0, len(available), size=min(10, len(available)))]
            per_target = total_reduced / len(targets)
            for t_idx in targets:
                a[t_idx] += per_target
        
        a = normalize_nonneg(a)
        score = score_sequence(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

def local_search(a, time_limit_s=60.0, seed=0, verbose=False):
    """Random local search with adaptive step size."""
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    a = normalize_nonneg(a.copy().astype(np.float64))
    n = len(a)
    best_a = a.copy()
    best_score = score_sequence(a)
    step_size = 0.05
    accept_count = 0
    
    while time.monotonic() - start_time < time_limit_s:
        # Random perturbation
        candidate = a.copy()
        # Option 1: multiplicative perturbation on a random window
        if rng.random() < 0.5:
            width = max(2, int(rng.integers(n // 50, n // 5)))
            start = int(rng.integers(0, max(1, n - width)))
            candidate[start:start+width] *= np.exp(rng.normal(0, step_size, size=width))
        else:
            # Option 2: add random smooth noise
            t = np.arange(n)
            sigma = rng.uniform(0.01, step_size)
            candidate += sigma * np.exp(-0.5 * ((t - rng.integers(0, n)) / max(1, n//10))**2) * rng.normal(1, 1, n)
        
        # Optionally symmetrize
        if rng.random() < 0.2:
            candidate = 0.5 * (candidate + candidate[::-1])
        
        candidate = normalize_nonneg(candidate)
        score = score_sequence(candidate)
        
        if score < best_score:
            best_score = score
            best_a = candidate.copy()
            a = candidate.copy()
            accept_count += 1
            step_size = min(0.2, step_size * 1.1)
        else:
            step_size = max(0.01, step_size * 0.95)
        
        if accept_count > 50 and (accept_count % 50 == 0):
            step_size = max(0.01, step_size * 0.7)
    
    return best_a, best_score
