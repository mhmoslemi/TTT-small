Hybrid LP + Simulated Annealing + Coordinate Descent optimizer for
minimum peak autocorrelation of non-negative sequences.

Improvements over previous approach:
1. Multi-M LP: solve LP with M slightly above current max to find
   more aggressive improvement directions.
2. Adaptive step size based on observed improvement.
3. Perturb-and-reoptimize phase to escape local minima.
4. Coordinate descent (pairwise) polish for fine improvements.
5. Multiple sequence lengths and structured initializations.

import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
import time


def fast_score(f):
    """Return 2*n*max(conv)/sum^2 or inf."""
    f = np.asarray(f, dtype=np.float64)
    if f.size == 0:
        return np.inf
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(f, f)
    return float(2.0 * len(f) * conv.max() / (s * s))


def build_conv_matrix_fast(n, f):
    """Sparse A (2n-1 x n) with A[k,j] = f[k-j] for valid indices."""
    cols = np.repeat(np.arange(n), n)
    rows = np.tile(np.arange(n), n) + np.repeat(np.arange(n), n)
    data = np.tile(f, n)
    A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    return A


def solve_lp(f, M, time_limit=5.0):
    """Solve max sum(g) s.t. conv(f,g) <= M, g >= 0. Return g or None."""
    n = len(f)
    if n > 3000:
        return None
    A = build_conv_matrix_fast(n, f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_step(f, M, t, time_limit=5.0):
    """One LP-driven update step. Returns (new_f, improved)."""
    g = solve_lp(f, M, time_limit)
    if g is None:
        return f.copy(), False
    gs = g.sum()
    if gs < 1e-10:
        return f.copy(), False
    g = g / gs
    if np.linalg.norm(g - f) < 1e-10:
        return f.copy(), False
    f_new = (1.0 - t) * f + t * g
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f.copy(), False
    f_new = f_new / s
    return f_new, True


def make_initial(n, rng, method='uniform'):
    """Create initial non-negative sequence of length n, normalized to sum=1."""
    if method == 'uniform':
        a = np.ones(n)
    elif method == 'cosine':
        x = np.linspace(0.0, np.pi, n)
        a = 0.5 * (1.0 + np.cos(x))
    elif method == 'sine_sq':
        x = np.linspace(0.0, np.pi, n)
        a = np.sin(x) ** 2
    elif method == 'random':
        a = rng.random(n)
    elif method == 'bimodal':
        x = np.arange(n, dtype=np.float64)
        s = max(1, n // 16)
        a = (np.exp(-0.5 * ((x - n // 4) / s) ** 2) +
             np.exp(-0.5 * ((x - 3 * n // 4) / s) ** 2))
    elif method == 'modulated':
        x = np.arange(n, dtype=np.float64)
        freq = rng.integers(3, max(4, n // 20))
        a = 0.5 * (1.0 + np.cos(2 * np.pi * freq * x / n))
    elif method == 'ramp':
        a = np.linspace(0.1, 1.0, n)
    elif method == 'arch':
        x = np.arange(n, dtype=np.float64) / (n - 1)
        a = 4.0 * x * (1.0 - x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a


def block_perturb(f, rng, block_size=None):
    """Multiplicative block perturbation, renormalized to sum 1."""
    n = len(f)
    cand = f.copy()
    if block_size is None:
        block_size = int(rng.integers(1, max(2, n // 10 + 1)))
    block_size = min(block_size, n)
    start = int(rng.integers(0, max(1, n - block_size + 1)))
    stop = min(n, start + block_size)
    scale = float(np.exp(rng.normal(0.0, 0.3)))
    cand[start:stop] *= scale
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s < 1e-10:
        return f.copy()
    return cand / s


def single_perturb(f, rng, sigma=0.1):
    """Perturb a single random element."""
    n = len(f)
    cand = f.copy()
    idx = int(rng.integers(0, n))
    cand[idx] = max(0.0, cand[idx] * float(np.exp(rng.normal(0.0, sigma))))
    s = cand.sum()
    if s < 1e-10:
        return f.copy()
    return cand / s


def coordinate_descent_step(f, rng, n_pairs=50):
    """Try pairwise coordinate adjustments to improve score."""
    n = len(f)
    cur_sc = fast_score(f)
    improved = False
    for _ in range(n_pairs):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        best_val_i = f[i]
        best_val_j = f[j]
        best_sc = cur_sc
        # Try several adjustments
        for di in [0.5, 0.7, 0.8, 1.2, 1.5]:
            for dj in [0.5, 0.7, 0.8, 1.2, 1.5]:
                new_i = max(0.0, f[i] * di)
                new_j = max(0.0, f[j] * dj)
                if abs(new_i) < 1e-12 and abs(new_j) < 1e-12:
                    continue
                cand = f.copy()
                cand[i] = new_i
                cand[j] = new_j
                s = cand.sum()
                if s < 1e-10:
                    continue
                cand = cand / s
                sc = fast_score(cand)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_val_i = new_i
                    best_val_j = new_j
                    improved = True
    if improved:
        f = f.copy()
        f[i] = best_val_i
        f[j] = best_val_j
        s = f.sum()
        f = f / s
    return f, improved

                        best_f_new = f_new
                    stall = 0
                    t = t_init
        if best_f_new is not None:
            f = best_f_new
            if best_sc_new < best_sc - 1e-12:
                best_sc = best_sc_new
                best = f.copy()
                stall = 0
                # Increase step size on improvement
                t = min(0.8, t * 1.1)
            else:
                stall += 1
                t = max(t_min, t * 0.95)
        cur_sc = best_sc_new
    return best, best_sc


def run_annealing(f0, deadline, n_iter=5000):
    """Simulated annealing with structured moves."""
    f = f0.copy()
    cur_sc = fast_score(f)
    best = f.copy()
    best_sc = cur_sc
    rng = np.random.default_rng(12345)
    T_init = 0.01
    T = T_init
    cool_rate = max(1.0 - (T_init / max(1, n_iter)), 0.99)
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        T *= cool_rate
        # Choose move type
        move_type = rng.random()
        if move_type < 0.5:
            cand = block_perturb(f, rng, block_size=int(rng.integers(1, max(2, len(f) // 20 + 1))))
        elif move_type < 0.8:
            cand = single_perturb(f, rng, sigma=0.15)
        else:
            # Swap two elements
            cand = f.copy()
            i = int(rng.integers(0, len(f)))
            j = int(rng.integers(0, len(f)))
            if i != j:
                cand[i], cand[j] = cand[j], cand[i]
                s = cand.sum()
                if s > 1e-10:
                    cand = cand / s
        sc = fast_score(cand)
        if sc < cur_sc or (T > 1e-6 and rng.random() < np.exp(-(sc - cur_sc) / T)):
            f = cand
            cur_sc = sc
            if sc < best_sc - 1e-12:
                best_sc = sc
                best = f.copy()
    return best, best_sc


def run_perturb_reoptimize(f0, deadline, n_restarts=10):
    """Perturb and re-optimize: make big changes, then briefly re-optimize."""
    best = f0.copy()
    best_sc = fast_score(f0)
    rng = np.random.default_rng(999)
    n = len(f0)
    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        # Make a significant perturbation
        f_pert = f0.copy()
        for _ in range(5):
            f_pert = block_perturb(f_pert, rng, block_size=int(rng.integers(3, max(4, n // 5))))
        # Brief LP re-optimization
        sub_deadline = min(deadline, time.monotonic() + 30.0)
        f_opt, sc_opt = run_lp_refinement(f_pert, sub_deadline, lp_time=2.0,
                                           t_init=0.4, t_min=0.1)
        if sc_opt < best_sc - 1e-12:
            best = f_opt.copy()
            best_sc = sc_opt
            f0 = f_opt.copy()  # Update base for next restart
    return best, best_sc


def coordinate_descent_polish(f0, deadline, n_rounds=100):
    """Polish with coordinate descent."""
    f = f0.copy()
    cur_sc = fast_score(f)
    best = f.copy()
    best_sc = cur_sc
    rng = np.random.default_rng(777)
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        f, improved = coordinate_descent_step(f, rng, n_pairs=200)
        if improved:
            new_sc = fast_score(f)
            if new_sc < best_sc - 1e-12:
                best_sc = new_sc
                best = f.copy()
    return best, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """Search for the best non-negative sequence minimizing peak autocorrelation score.
    
    Uses a multi-phase approach:
    1. Multi-start with different lengths and initializations
    2. LP refinement with multi-M strategy
    3. Perturb-and-reoptimize to escape local minima
    4. Simulated annealing
    5. Coordinate descent polish
    """
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(seed)

    # Get parent sequence if available
    try:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        parent = np.ones(512, dtype=np.float64)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(512, dtype=np.float64)
    if parent.sum() < 0.01:
        parent = np.ones(512, dtype=np.float64)

    parent = parent / parent.sum()
    parent_sc = fast_score(parent)
    print("Parent: n=%d, score=%.6f" % (parent.size, parent_sc), flush=True)

    best = parent.copy()
    best_sc = parent_sc

    def try_update(f, sc):
        nonlocal best, best_sc
        if np.isfinite(sc) and sc < best_sc - 1e-12:
            best_sc = sc
            best = f.copy()
            print("    New best: %.6f (n=%d) at t=%.0f" % (sc, f.size,
                   time.monotonic() - (deadline - float(budget_s))), flush=True)

    # Phase 1: Multi-start with LP refinement
    # Try various lengths and methods
    lengths = [256, 384, 512, 768, 1024, 1536, 2048]
    methods = ['uniform', 'cosine', 'sine_sq', 'bimodal', 'modulated', 'arch', 'ramp']

    # If parent is good, also try its length
    if parent.size not in lengths and 128 <= parent.size <= 3000:
        lengths.append(int(parent.size))
    lengths.sort()

    # Allocate time: Phase 1 gets 40%, Phase 2 gets 25%, Phase 3 gets 20%, Phase 4 gets 15%
    p1_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.40
    per_combo = max(5.0, (p1_deadline - time.monotonic()) / max(1, len(lengths) * 3))

    print("\n=== Phase 1: Multi-start LP refinement ===", flush=True)
    # Start with parent first
    sub_dl = min(p1_deadline, time.monotonic() + max(30.0, per_combo * 3))
    f_opt, sc_opt = run_lp_refinement(parent, sub_dl, lp_time=3.0)
    try_update(f_opt, sc_opt)

    # Try other (n, method) combos
    for n in lengths:
        if time.monotonic() > p1_deadline:
            break
        if n == parent.size:
            continue
        for m in methods[:3]:  # Limit to 3 methods per length in phase 1
            if time.monotonic() > p1_deadline:
                break
            start = make_initial(n, rng, m)
            sub_dl = min(p1_deadline, time.monotonic() + per_combo)
            f_opt, sc_opt = run_lp_refinement(start, sub_dl, lp_time=2.0,
                                               t_init=0.3, t_min=0.1)
            if sc_opt < best_sc:
                print("  Phase1: n=%d method=%s score=%.6f" % (n, m, sc_opt),
                      flush=True)
                try_update(f_opt, sc_opt)

    # Phase 2: Deep LP refinement on the current best
    print("\n=== Phase 2: Deep LP refinement ===", flush=True)
    p2_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.30
    print("  Deep refinement on n=%d, score=%.6f" % (best.size, best_sc), flush=True)
    f_opt, sc_opt = run_lp_refinement(best, p2_deadline, lp_time=4.0,
                                       t_init=0.4, t_min=0.05)
    try_update(f_opt, sc_opt)

    # Also try deep refinement on parent (different trajectory)
    if time.monotonic() < p2_deadline - 30:
        f_opt, sc_opt = run_lp_refinement(parent, min(p2_deadline, time.monotonic() + 45),
                                           lp_time=3.0, t_init=0.5, t_min=0.1)
        try_update(f_opt, sc_opt)

    # Phase 3: Perturb and reoptimize
    print("\n=== Phase 3: Perturb and reoptimize ===", flush=True)
    p3_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.30
    f_opt, sc_opt = run_perturb_reoptimize(best, p3_deadline, n_restarts=15)
    try_update(f_opt, sc_opt)

    # Phase 4: Simulated annealing
    print("\n=== Phase 4: Simulated annealing ===", flush=True)
    p4_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.50
    f_opt, sc_opt = run_annealing(best, min(p4_deadline, deadline - 20),
                                   n_iter=20000)
    try_update(f_opt, sc_opt)

    # Phase 5: Coordinate descent polish
    print("\n=== Phase 5: Coordinate descent polish ===", flush=True)
    print("  Polish: score before = %.6f" % best_sc, flush=True)
    f_opt, sc_opt = coordinate_descent_polish(best, min(deadline, time.monotonic() + 60),
                                               n_rounds=200)
    try_update(f_opt, sc_opt)

    # Final: one more round of everything on the best
    if time.monotonic() < deadline - 30:
        print("\n=== Final polish ===", flush=True)
        # Quick LP
        f_opt, sc_opt = run_lp_refinement(best, min(deadline - 10, time.monotonic() + 20),
                                           lp_time=2.0, t_init=0.3, t_min=0.05)
        try_update(f_opt, sc_opt)
        # Quick coordinate descent
        f_opt, sc_opt = coordinate_descent_polish(best, min(deadline - 5,
                                                             time.monotonic() + 20),
                                                   n_rounds=100)
        try_update(f_opt, sc_opt)

    # Verify with authoritative evaluator
    best = np.clip(np.asarray(best, dtype=np.float64), 0.0, 1000.0)
    if best.size == 0 or not np.isfinite(best).all() or best.sum() < 0.01:
        best = np.ones(512, dtype=np.float64)

    final_sc = float(evaluate_sequence([float(x) for x in best]))
    print("\nFINAL: score=%.6f (n=%d)" % (final_sc, best.size), flush=True)
    return [float(x) for x in best]
