"""
LP-based iterative optimization combined with projected gradient descent
and local search for minimizing peak autocorrelation of non-negative sequences.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def proj_gradient_descent(f, n, lr_init, num_steps, time_limit):
    """
    Projected gradient descent on the constraint set {g >= 0, sum(g) = 1}.
    Objective: max(convolve(f,f)).
    """
    t0 = time.monotonic()
    f = f.astype(np.float64).copy()
    sum_f = np.sum(f)
    f = f / sum_f  # normalize to sum = 1
    
    best = f.copy()
    best_peak = np.max(fft_convolve(f, f))
    
    for step in range(num_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute gradient of max(convolve(f,f)) using soft-max
        c = fft_convolve(f, f)
        # Use a smooth max approximation
        tau = best_peak * 0.01
        c_max = np.max(c)
        w = np.exp((c - c_max) / tau)
        w = w / np.sum(w)
        
        # Gradient: g[j] = 2 * sum_k w[k] * A[k,j] = 2 * convolve(f_rev, w)[j + n - 1]
        # Actually: dc[k]/df[j] = 2*f[k-j] if valid
        # So grad[j] = 2 * sum_k w[k] * f[k-j] (for valid k-j)
        # This is 2 * (w * f_rev) convolved appropriately
        # Or: grad = 2 * convolve(w, f)[n-1 : 2n-1] ... let me think
        # sum_k w[k] * f[k-j] for valid k-j means: for each j, sum over k where 0<=k-j<n
        # = sum over k of w[k] * f[k-j] * 1(0<=k-j<n)
        # Let m = k-j, then k = m+j: sum_m w[m+j] * f[m] * 1(0<=m<n)
        # = convolve(w, f)[j + n - 1] ... hmm this is getting complicated
        # Let me just use the direct formula
        # grad[j] = 2 * sum_k w[k] * f[k-j] where 0 <= k-j < n
        
        # Using FFT: define h[k] = w[k] for k in [0, 2n-2], 0 otherwise
        # Then grad[j] = 2 * sum_k h[k] * f[k-j] (where k-j in [0,n-1])
        # This is the cross-correlation of h and f at lag j
        # cross_corr(j) = sum_k h[k] * f[k-j]
        # = sum_k h[k+j] * f[k] (substituting m=k, k=m+j)
        # Wait, let me just compute it directly for clarity (n=256, should be fast enough)
        
        # Actually, let me use the formula from the previous program:
        # g[j] = 2 * corr[n-1:2n-1] where corr = convolve(f[::-1], w)
        f_rev = f[::-1]
        corr = fft_convolve(f_rev, w)
        grad = 2.0 * corr[n - 1:2 * n - 1]
        
        # Gradient step
        lr = lr_init / (1.0 + step * 0.01)
        f_new = f - lr * grad
        
        # Project onto {g >= 0, sum(g) = 1}
        # This is: find lambda such that sum(max(0, f_new - lambda)) = 1
        # Binary search for lambda
        f_sorted = np.sort(f_new)
        cumsum = np.cumsum(f_sorted)
        n_inner = n
        # Find lambda: sum of (f_new_i - lambda) for f_new_i > lambda = 1
        # If we sort f_new in decreasing order: f_(1) >= f_(2) >= ...
        # Then lambda = (sum of top k values - 1) / k for the right k
        f_desc = np.sort(f_new)[::-1]
        cumsum_desc = np.cumsum(f_desc)
        # Find k such that (cumsum_desc[k-1] - 1) / k <= f_desc[k-1] and (cumsum_desc[k-1] - 1) / k >= f_desc[k]
        # Actually: lambda = (sum of values above lambda) / count - 1/count
        # Let's use the standard water-filling algorithm
        # For i from n down to 1:
        #   lambda_i = (cumsum_desc[i-1] - 1) / i
        #   Check if lambda_i < f_desc[i-1] (i.e., the i-th largest is above lambda)
        #   and lambda_i >= f_desc[i] (if i < n)
        
        # Vectorized approach:
        # lambda_i = (cumsum_desc[i-1] - 1) / i for i = 1, ..., n
        lambdas = (cumsum_desc - 1.0) / np.arange(1, n + 1)
        # We want the largest i such that lambda_i < f_desc[i-1] (0-indexed)
        # i.e., (cumsum_desc[i-1] - 1) / i < f_desc[i-1]
        # i.e., cumsum_desc[i-1] - 1 < i * f_desc[i-1]
        # i.e., cumsum_desc[i-1] - i * f_desc[i-1] < 1
        # This is equivalent to: sum of (f_desc[j] - f_desc[i-1]) for j < i-1 < 1
        # Hmm, let me just use a simpler approach
        
        # Simpler: binary search for lambda
        lo = np.min(f_new) - 1
        hi = np.max(f_new) + 1
        for _ in range(50):
            mid = (lo + hi) / 2
            s = np.sum(np.maximum(f_new - mid, 0))
            if s > 1:
                lo = mid
            else:
                hi = mid
        lambda_val = (lo + hi) / 2
        f_new = np.maximum(f_new - lambda_val, 0)
        f_new = f_new / np.sum(f_new)  # Ensure exact sum = 1 (numerical safety)
        
        peak = np.max(fft_convolve(f_new, f_new))
        if peak < best_peak:
            best_peak = peak
            best = f_new.copy()
        f = f_new
    
    return best, best_peak


def lp_refine(f, n, max_iters, time_limit, alpha=1.0):
    """
    LP-based iterative refinement.
    f: current sequence (will be normalized to sum=1 internally)
    Returns improved sequence and its peak.
    """
    t0 = time.monotonic()
    f = f.astype(np.float64).copy()
    f = f / np.sum(f)  # normalize
    
    best = f.copy()
    best_peak = np.max(fft_convolve(f, f))
    
    # Build the convolution matrix (depends on f, so we rebuild each iteration)
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = np.max(fft_convolve(f, f))
        target = alpha * M
        
        # Build A matrix
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            A[k, j_min:j_max + 1] = f[k - j_max:k - j_min + 1][::-1]
        
        # Solve LP: max sum(g) s.t. A@g <= target, g >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, target)
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, 
                         bounds=[(0, None)] * n, 
                         method='highs',
                         options={'presolve': True, 'time_limit': 10})
        except:
            break
        
        if not res.success or res.x is None:
            break
        
        g = res.x
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        
        # Rescale g to have same sum as f
        g = g * np.sum(f) / sum_g
        
        # Try different blend factors
        improved = False
        for t in [0.5, 0.3, 0.7, 0.1, 0.9, 0.2]:
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0)
            f_new = f_new / np.sum(f_new)
            peak = np.max(fft_convolve(f_new, f_new))
            if peak < best_peak:
                best_peak = peak
                best = f_new.copy()
            if peak < M:
                f = f_new
                improved = True
                break
        
        if not improved:
            # Try smaller steps
            for t in [0.05, 0.02, 0.01]:
                f_new = (1 - t) * f + t * g
                f_new = np.maximum(f_new, 0)
                f_new = f_new / np.sum(f_new)
                peak = np.max(fft_convolve(f_new, f_new))
                if peak < M:
                    f = f_new
                    if peak < best_peak:
                        best_peak = peak
                        best = f_new.copy()
                    break
            else:
                # No improvement, maybe tighten alpha
                alpha = max(0.5, alpha * 0.9)
                if alpha < 0.5:
                    break
    
    return best, best_peak


def make_init(n, rng, init_type='random'):
    """Create initial sequence."""
    if init_type == 'uniform':
        a = np.ones(n)
    elif init_type == 'chirp':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.5 * np.cos(np.pi * t * t / n)
    elif init_type == 'multi_tone':
        t = np.arange(n, dtype=np.float64)
        a = np.ones(n)
        for k in [1, 3, 5, 8, 13]:
            a += 0.2 / np.sqrt(k) * np.cos(2 * np.pi * k * t / n + rng.uniform(0, 2 * np.pi))
    elif init_type == 'triangle':
        t = np.arange(n, dtype=np.float64)
        a = 1.0 + 0.5 * (1 - 2 * np.abs(t / n - 0.5))
    elif init_type == 'bump':
        t = np.arange(n, dtype=np.float64)
        for center, width, amp in [(0.25*n, 0.1*n, 0.5), (0.75*n, 0.1*n, 0.5), (0.5*n, 0.05*n, -0.3)]:
            a += amp * np.exp(-0.5 * ((t - center) / width) ** 2)
    else:  # random
        a = 1.0 + 0.3 * rng.standard_normal(n)
    
    a = np.clip(a, 0.01, None)
    return a

LP-based iterative optimization combined with projected gradient descent
and local search for minimizing peak autocorrelation of non-negative sequences.
Key innovation: uses linear programming to find complementary directions that
reduce the autocorrelation peak while maintaining the L1 norm constraint.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize as scipy_minimize


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score for a sequence (same as evaluate_sequence)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def compute_peak(a):
    """Compute max of convolution (for normalized sequences with sum=1)."""
    c = fft_convolve(a, a)
    return np.max(c)


def build_conv_matrix(f, n):
    """
    Build matrix A (shape 2n-1 x n) such that A @ g = convolve(f, g).
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    """
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        # A[k, j] = f[k-j] for j in [j_min, j_max]
        # Indices into f: k-j_max, k-j_max+1, ..., k-j_min
        A[k, j_min:j_max + 1] = f[k - j_max:k - j_min + 1][::-1]
    return A


def proj_to_simplex(v):
    """Project vector v onto the probability simplex {g >= 0, sum(g) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    # Water-filling algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    r = np.arange(1, n + 1)
    cond = u - (cssv - 1) / r > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0]) + 1
    theta = (cssv[rho - 1] - 1) / rho
    return np.maximum(v - theta, 0)


def proj_gradient_descent(f, n, time_limit, lr_init=0.05, verbose=True):
    """
    Projected gradient descent minimizing max(convolve(f,f))
    subject to f in simplex (sum=1, f>=0).
    """
    t0 = time.monotonic()
    f = f.astype(np.float64).copy()
    f = proj_to_simplex(f)
    
    best = f.copy()
    best_peak = compute_peak(f)
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute soft-max gradient
        c = fft_convolve(f, f)
        c_max = np.max(c)
        tau = max(c_max * 0.005, 1e-10)
        exp_c = np.exp((c - c_max) / tau)
        w = exp_c / np.sum(exp_c)
        
        # Gradient of soft-max(conv(f,f)) w.r.t. f
        # dc[k]/df[j] = 2*f[k-j] if 0<=k-j<n
        # grad[j] = 2 * sum_k w[k] * f[k-j] (for valid k-j)
        # = 2 * (cross-correlation of w and f) at appropriate lag
        # Use: grad = 2 * convolve(f[::-1], w)[n-1 : 2n-1]
        f_rev = f[::-1]
        corr = fft_convolve(f_rev, w)
        grad = 2.0 * corr[n - 1:2 * n - 1]
        
        # Adaptive learning rate
        lr = lr_init / (1.0 + 0.005 * step)
        
        # Gradient step + projection
        f_new = f - lr * grad
        f_new = proj_to_simplex(f_new)
        
        peak = compute_peak(f_new)
        if peak < best_peak:
            best_peak = peak
            best = f_new.copy()
        
        f = f_new
        step += 1
        
        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PGD step={step}, peak={best_peak:.8f}, elapsed={elapsed:.1f}s")
    
    return best, best_peak


def lp_solve(f, n, target, time_per_lp=5.0):
    """
    Solve LP: maximize sum(g) subject to A@g <= target, g >= 0.
    Returns g (optimal solution) or None if infeasible.
    """
    A = build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, target)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                     bounds=[(0, None)] * n,
                     method='highs',
                     options={'presolve': True, 'time_limit': time_per_lp})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def lp_refine(f, n, time_limit, alpha_init=1.0, verbose=True):
    """
    LP-based iterative refinement.
    Repeatedly: solve LP to find complementary g, blend with f.
    """
    t0 = time.monotonic()
    f = f.astype(np.float64).copy()
    f = proj_to_simplex(f)
    
    best = f.copy()
    best_peak = compute_peak(f)
    alpha = alpha_init
    stall_count = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = compute_peak(f)
        target = alpha * M
        
        # Solve LP
        g = lp_solve(f, n, target, time_per_lp=min(5.0, max(1.0, (time_limit - (time.monotonic() - t0)) / 10)))
        
        if g is None or np.sum(g) < 1e-10:
            stall_count += 1
            if stall_count > 20:
                break
            continue
        
        # Rescale g to have sum = 1
        g = g / np.sum(g)
        
        # Try blending
        improved = False
        for t_blend in [0.5, 0.3, 0.7, 0.1, 0.8, 0.2, 0.4]:
            f_new = (1 - t_blend) * f + t_blend * g
            f_new = proj_to_simplex(f_new)
            peak = compute_peak(f_new)
            
            if peak < best_peak:
                best_peak = peak
                best = f_new.copy()
            
            if peak < M * 0.999:
                f = f_new
                improved = True
                stall_count = 0
                break

                f_new = (1 - t_blend) * f + t_blend * g

                peak = compute_peak(f_new)
                if peak < M * 0.999:
                    f = f_new
                    if peak < best_peak:
                        best_peak = peak
                        best = f_new.copy()
                    stall_count = 0

                    break

            print(f"    LP iter={it}, M={M:.8f}, sum_g/f={sum_g:.4f}, best_peak={best_peak:.8f}, alpha={alpha:.2f}, elapsed={elapsed:.1f}s")
    
    return best, best_peak


def make_init(n, rng, init_type='random'):

    elif init_type == 'bump':

    else:
