"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based peak reduction.
The key idea: at each iteration, solve a linear program to find the "largest"
non-negative sequence b compatible with the current autocorrelation peak,
then convex combine with the current sequence. This directly targets the
bottleneck (the max peak) and provides a principled descent direction.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_obj(a):
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def build_lp_matrix(f, n):
    """Build the constraint matrix A for the LP.
    A[m, j] = f[m-j] if 0 <= m-j < n, else 0.
    Size: (2n-1) x n
    """
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        # b[j] contributes to conv[m] for m = j, j+1, ..., j+n-1
        # conv[m] includes term f[m-j] * b[j] when 0 <= m-j < n
        # So for fixed j, m ranges from j to j+n-1, and A[m, j] = f[m-j]
        m_start = j
        m_end = min(j + n - 1, 2 * n - 2)
        for m in range(m_start, m_end + 1):
            A[m, j] = f[m - j]
    return A

def solve_lp(f, M, n):
    """Solve: max sum(b) s.t. A @ b <= M * 1, b >= 0."""
    A = build_lp_matrix(f, n)
    c = -np.ones(n)  # minimize -sum(b)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    else:
        return np.zeros(n)

def mirror_descent_step(a, conv, eta, temperature):
    """One step of multiplicative mirror descent."""
    n = len(a)
    max_conv = conv.max()
    if max_conv < 1e-15:
        return a
    
    # Soft-max weights
    scaled = temperature * conv / max_conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    w = w / w_sum
    
    # Subgradient
    sg = np.zeros(n)
    thresh = w.max() * 0.001
    active = np.where(w > thresh)[0]
    for k in active:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        sg[j_start:j_end+1] += w[k] * a[k - j_end:k - j_start + 1][::-1]
    
    sg_centered = sg - sg.mean()
    log_factor = -eta * sg_centered
    log_factor -= log_factor.max()
    a = a * np.exp(log_factor)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based peak reduction combined
with mirror descent. At each iteration, solves a linear program to find the
"largest" non-negative sequence compatible with the current autocorrelation peak,
then takes a convex step toward it. This directly targets the bottleneck peak.
Includes Nesterov momentum, adaptive step sizes, and multi-length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def build_lp_matrix(f, n):
    """Build constraint matrix A (2n-1 x n) where A[m,j] = f[m-j] if valid."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        m_start = j
        m_end = min(j + n - 1, 2 * n - 2)
        A[m_start:m_end + 1, j] = f[m_end - j: m_start - j - 1: -1][::-1]
    # Fix: do it properly
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(j, min(j + n, 2 * n - 1)):
            A[m, j] = f[m - j]
    return A


def solve_lp(f, M, n):
    """Solve: max 1'b s.t. A b <= M*1, b >= 0. Returns optimal b."""
    A = build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
        if res.success:
            return res.x
        return np.zeros(n)
    except Exception:
        return np.zeros(n)


def lp_with_momentum(a, time_limit, t0=0.12, beta_mom=0.5, eps_M=0.0):

def mirror_descent_optimize(a0, time_limit, eta0=0.4, temperature=50.0):
    """Multiplicative mirror descent with soft-max peak smoothing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    eta = eta0
    temp = temperature
    t_start = time.monotonic()
    stuck = 0

    while time.monotonic() - t_start < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        scaled = temp * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = w / w_sum

        sg = np.zeros(n)
        thresh = w.max() * 0.002
        active = np.where(w > thresh)[0]
        for k_int in active:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                sg[j_start:j_end + 1] += w[k] * a[k - j_end:k - j_start + 1][::-1]

        sg_centered = sg - sg.mean()
        log_factor = -eta * sg_centered
        log_factor -= log_factor.max()
        a = a * np.exp(log_factor)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n

            a = a / s

        current_obj = 2.0 * n * conv.max()

            best_obj = current_obj
            best_a = a.copy()

            stuck += 1

        if stuck >= 60:
            eta *= 0.65
            temp = min(temp * 1.4, 300.0)
            stuck = 0
            if eta < 0.003:

def coordinate_descent_polish(a, time_limit):
    """Local polish: try reducing the max peak by adjusting individual coordinates."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-12:
        return a
    a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t_start = time.monotonic()

    while time.monotonic() - t_start < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        current_obj = 2.0 * n * M

        if current_obj < best_obj - 1e-10:
            best_obj = current_obj
            best_a = a.copy()

        # Find peak location
        peak_idx = int(np.argmax(conv))

        # For peak at index peak_idx, conv[peak_idx] = sum_j a[j]*a[peak_idx-j]
        # Find the j that contributes most
        j_start = max(0, peak_idx - n + 1)
        j_end = min(n - 1, peak_idx)

        if j_start > j_end:
            break

        contributions = np.array([a[j] * a[peak_idx - j] for j in range(j_start, j_end + 1)])
        if contributions.max() < 1e-15:
            break

        # Try reducing the largest contributor
        best_j_offset = int(np.argmax(contributions))
        best_j = j_start + best_j_offset

        # Try reducing best_j and its partner
        partner = peak_idx - best_j

        for target in [best_j, partner]:
            if target < 0 or target >= n:
                continue
            # Try several reduction amounts
            for frac in [0.3, 0.5, 0.7]:
                a_trial = a.copy()
                a_trial[target] *= (1.0 - frac)
                # Redistribute to partner or to a random other element
                removed = a[target] * frac
                if target != partner and 0 <= partner < n:
                    a_trial[partner] += removed * 0.5
                # Add rest to a low element
                low_idx = int(np.argmin(a_trial))
                a_trial[low_idx] += removed * 0.5
                a_trial = np.maximum(a_trial, 0)
                s_trial = a_trial.sum()
                if s_trial < 1e-12:
                    continue
                a_trial = a_trial / s_trial

                conv_trial = fast_convolve(a_trial)
                obj_trial = 2.0 * n * conv_trial.max()

                if obj_trial < current_obj - 1e-12:
                    a = a_trial
                    break

    return best_a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses LP-based peak reduction with momentum, combined
    with mirror descent and coordinate descent polishing.
    Explores multiple sequence lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + budget_s - 15.0

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={p.size}, sum={p.sum():.4f}, score={evaluate_sequence(p.tolist()):.6f}")
    except Exception:
        pass

    # Phase 1: LP from parent (if available) + quick scans
    print("=" * 60)
    print("Phase 1: LP optimization (quick) + exploration")
    print("=" * 60)

    results = []  # (score, n, name, array)

    if parent is not None:
        n = len(parent)
        a0 = np.maximum(parent, 0)
        s = a0.sum()
        a0 = a0 / s
        t_alloc = min(400.0, (total_deadline - time.monotonic()) * 0.25)
        a_opt, _ = lp_with_momentum(a0, time_limit=t_alloc, t0=0.1, beta_mom=0.5)
        score = float(evaluate_sequence(a_opt.tolist()))
        print(f"  LP parent(n={n}): score={score:.6f}")
        if np.isfinite(score):
            results.append((score, n, 'parent_lp', a_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()

    # Quick scan: multiple n and inits
    lengths_to_try = [64, 96, 128, 160, 192, 256, 320, 512]
    if parent is not None and len(parent) not in lengths_to_try:
        lengths_to_try.append(len(parent))

    scan_time_total = min(500.0, (total_deadline - time.monotonic()) * 0.30)
    num_scans = sum(4 for _ in lengths_to_try)
    time_per_scan = max(8.0, scan_time_total / max(1, num_scans))

    for n in lengths_to_try:
        if time.monotonic() > total_deadline - 120:
            break
        x = np.arange(n, dtype=np.float64)
        inits = {}

        inits['uniform'] = np.ones(n) / n

        g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
        inits['gauss'] = g / g.sum()

        bi = np.zeros(n)
        bi[:max(1, n // 5)] = 1.0
        bi[-max(1, n // 5):] = 1.0
        inits['bimodal'] = bi / bi.sum()

        r = rng.dirichlet(np.ones(n) * 0.4)
        inits['random'] = r

        if parent is not None:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                inits['parent'] = pp / pp.sum()

        for name, a0 in inits.items():
            if time.monotonic() > total_deadline - 120:
                break
            t_alloc = min(time_per_scan, total_deadline - time.monotonic() - 120)
            if t_alloc < 3:
                break

            # Use LP
            a_opt, _ = lp_with_momentum(a0, time_limit=t_alloc, t0=0.1, beta_mom=0.4)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Scan n={n} {name}: score={score:.6f}")

    # Phase 2: Deep optimization on top results
    print("=" * 60)
    print(f"Phase 2: Deep optimization. Best so far: {best_score:.6f}")
    print("=" * 60)

    results.sort(key=lambda x: x[0])

    top_configs = []
    seen = set()
    for score, n, name, a_opt in results:
        key = (n, name)
        if key not in seen and len(top_configs) < 10:
            top_configs.append((n, name, a_opt.copy()))
            seen.add(key)

    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    remaining = total_deadline - time.monotonic() - 100
    time_per_deep = max(20.0, remaining / max(1, len(top_configs)))

    for n, name, a0 in top_configs:
        if time.monotonic() > total_deadline - 80:
            break
        t_alloc = min(time_per_deep, total_deadline - time.monotonic() - 80)
        if t_alloc < 5:
            break

        # Multiple LP runs with different parameters
        for t_init, beta in [(0.15, 0.5), (0.08, 0.3), (0.25, 0.6)]:
            if time.monotonic() > total_deadline - 80:
                break
            t_single = t_alloc / 3.0
            a_opt, _ = lp_with_momentum(a0, time_limit=t_single, t0=t_init, beta_mom=beta)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep LP n={n} {name} t={t_init}: score={score:.6f}")

        # Mirror descent
        if time.monotonic() < total_deadline - 80:
            t_md = (total_deadline - time.monotonic() - 80) / 2.0
            a_opt, _ = mirror_descent_optimize(a_opt if 'a_opt' in dir() else a0,
                                               time_limit=min(60.0, t_md),
                                               eta0=0.3, temperature=80.0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Deep MD n={n} {name}: score={score:.6f}")

    # Phase 3: Polish the best
    print("=" * 60)
    print(f"Phase 3: Polish. Score: {best_score:.6f}")
    print("=" * 60)

    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = total_deadline - time.monotonic() - 20

        # LP polish with small steps
        t_lp = min(200.0, remaining * 0.4)
        if t_lp > 10:
            a_opt, _ = lp_with_momentum(best_sequence, time_limit=t_lp, t0=0.03, beta_mom=0.3)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish LP: score={score:.6f}")

        # Mirror descent polish
        remaining = total_deadline - time.monotonic() - 20
        t_md = min(150.0, remaining * 0.4)
        if t_md > 10:
            a_opt, _ = mirror_descent_optimize(best_sequence, time_limit=t_md,
                                               eta0=0.15, temperature=120.0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish MD: score={score:.6f}")

        # Coordinate descent
        remaining = total_deadline - time.monotonic() - 10
        if remaining > 10:
            a_opt = coordinate_descent_polish(best_sequence, time_limit=remaining)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish CD: score={score:.6f}")

        # Final LP with very small step
        remaining = total_deadline - time.monotonic() - 5
        if remaining > 5:
            a_opt, _ = lp_with_momentum(best_sequence, time_limit=remaining, t0=0.01, beta_mom=0.2)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: score={score:.6f}")

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
