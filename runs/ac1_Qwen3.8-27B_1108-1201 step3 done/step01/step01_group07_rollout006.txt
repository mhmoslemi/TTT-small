"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Improved softmax mirror descent with FFT-based subgradient
2. LP-based direction finding (polar step) for escape from local optima
3. Multiple diverse initializations and sequence lengths
4. Adaptive parameter tuning with periodic restarts

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)
    return result[:size]


def compute_score_from_conv(conv, n):
    """Compute 2*n*max(conv) assuming sum(a)=1."""
    return 2.0 * n * conv.max()


def mirror_descent_step(a, C, eta, t_param, top_k=50):
    """
    One step of softmax mirror descent.
    Returns updated a (normalized to sum=1).
    """
    n = len(a)
    max_C = C.max()
    if max_C < 1e-15:
        return a
    
    # Softmax weights over convolution peaks
    if top_k is not None and top_k < len(C):
        # Sparse: only top-k
        idx = np.argpartition(C, -top_k)[-top_k:]
        w = np.zeros(len(C))
        C_top = C[idx]
        w[idx] = np.exp(t_param * (C_top - C_top.max()) / max_C)
        w /= w.sum()
    else:
        scaled = t_param * C / max_C
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
    
    # Subgradient via cross-correlation
    # sg[j] = sum_k w[k] * a[k-j] for valid k-j
    sg = np.correlate(w, a, mode='valid')[:n]
    
    # Center
    sg -= sg.mean()
    
    # Multiplicative update
    log_factor = -eta * sg
    log_factor -= log_factor.max()
    a_new = a * np.exp(log_factor)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a
    a_new /= s
    return a_new


def lp_direction(f, M, n):
    """
    Solve LP: max sum(b) s.t. b>=0, conv(f,b)[k] <= M for all k.
    Returns b or None if infeasible.
    """
    c = -np.ones(n)  # minimize -sum(b)
    
    # Build constraint matrix: for each k, sum_i f[i]*b[k-i] <= M
    # Only need constraints where the sum could potentially exceed M
    # (i.e., where f has significant mass)
    A_ub = np.zeros((2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            A_ub[k, j] = f[i]
    
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        b = np.maximum(res.x, 0)
        if b.sum() > 1e-10:
            return b
    return None


def run_mirror_descent(a0, time_limit, t_param=30.0, eta0=0.5, top_k=None, 
                       use_lp=False, lp_every=50, seed=0):

    Run mirror descent with optional LP corrections.
    Returns (best_a, best_objective_normalized).
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    C = fast_convolve(a)
    best_obj = compute_score_from_conv(C, n)
    best_a = a.copy()
    
    eta = eta0
    t_cur = t_param
    patience = 0
    max_patience = 100
    iter_count = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Mirror descent step
        a_new = mirror_descent_step(a, C, eta, t_cur, top_k)
        
        # Evaluate
        C = fast_convolve(a_new)
        cur_obj = compute_score_from_conv(C, n)
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a_new.copy()
            patience = 0
            a = a_new
        else:
            patience += 1
            a = a_new  # Still update, might help later
        
        iter_count += 1
        
        # Periodic LP correction
        if use_lp and iter_count % lp_every == 0 and patience > 20:
            M = C.max()
            b = lp_direction(a, M, n)
            if b is not None:
                b_norm = b / b.sum()
                # Try different step sizes
                for t_step in [0.5, 0.3, 0.1, 0.05]:
                    a_try = (1 - t_step) * a + t_step * b_norm
                    a_try = np.maximum(a_try, 0)
                    C_try = fast_convolve(a_try)
                    obj_try = compute_score_from_conv(C_try, n)
                    if obj_try < best_obj - 1e-10:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try
                        C = C_try
                        patience = 0
                        break
        
        # Adapt parameters
        if patience >= max_patience:
            patience = 0
            eta *= 0.7
            t_cur *= 1.5
            if eta < 0.01:
                # Restart with perturbation
                a = best_a * (1 + 0.1 * rng.standard_normal(n))

                s = a.sum()
                if s > 1e-10:
                    a /= s
                else:
                    a = best_a.copy()
                C = fast_convolve(a)

                t_cur = t_param

def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for sequence length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Edge-concentrated (exploits the endpoint structure)
    for frac in [0.1, 0.2, 0.3]:
        w = np.zeros(n)
        edge = max(1, int(n * frac))
        w[:edge] = 1.0
        w[-edge:] = 1.0
        w /= w.sum()
        inits[f'edge_{frac}'] = w
    
    # Gaussian (center)
    for sf in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g
    
    # Two Gaussians (at 1/4 and 3/4)
    for sf in [0.05, 0.1, 0.15]:
        g = np.exp(-0.5 * ((x - n * 0.25) / (sf * n)) ** 2) + \
            np.exp(-0.5 * ((x - n * 0.75) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'bi_gauss_{sf}'] = g
    
    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w
    
    # Random (Dirichlet)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Random sparse
    sparse = np.zeros(n)
    k = max(2, n // 4)
    idx = rng.choice(n, size=k, replace=False)
    sparse[idx] = rng.dirichlet(np.ones(k))
    inits['sparse'] = sparse
    
    # Parent (if available)
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0)
            if pp.sum() > 1e-10:
                inits['parent'] = pp / pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n), 
                          np.linspace(0, 1, parent.size), 
                          np.maximum(parent, 0))
            if pp.sum() > 1e-10:
                inits['parent_interp'] = pp / pp.sum()
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Improved softmax mirror descent with FFT-based subgradient computation
2. LP-based direction finding (polar step) for escape from local optima
3. Multiple diverse initializations and sequence lengths
4. Adaptive parameter tuning with periodic perturbations
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)
    return result[:size]


def compute_score_from_conv(conv, n):
    """Compute 2*n*max(conv) assuming sum(a)=1."""
    return 2.0 * n * conv.max()


def mirror_descent_step(a, C, eta, t_param, top_k=None):
    """
    One step of softmax mirror descent.
    Returns updated a (normalized to sum=1).
    """
    n = len(a)
    max_C = C.max()
    if max_C < 1e-15:
        return a

    # Softmax weights over convolution peaks
    if top_k is not None and top_k < len(C):
        idx = np.argpartition(C, -top_k)[-top_k:]
        w = np.zeros(len(C))
        C_top = C[idx]
        w[idx] = np.exp(t_param * (C_top - C_top.max()) / max_C)
        w_sum = w.sum()
        if w_sum > 1e-30:
            w /= w_sum
        else:
            w = np.zeros(len(C))
            w[np.argmax(C)] = 1.0
    else:
        scaled = t_param * C / max_C
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 1e-30:
            w /= w_sum
        else:
            w = np.zeros(len(C))
            w[np.argmax(C)] = 1.0

    # Subgradient via cross-correlation: sg[j] = sum_{m=0}^{n-1} w[j+m]*a[m]
    sg = np.correlate(w, a, mode='valid')[:n]

    # Center
    sg -= sg.mean()

    # Multiplicative update
    log_factor = -eta * sg
    log_factor -= log_factor.max()
    a_new = a * np.exp(log_factor)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a
    a_new /= s
    return a_new


def lp_direction(f, M, n):
    """
    Solve LP: max sum(b) s.t. b>=0, conv(f,b)[k] <= M for all k.
    Returns b or None if infeasible.
    """
    c = -np.ones(n)

    A_ub = np.zeros((2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)

    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            A_ub[k, j] = f[i]

    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0)
            if b.sum() > 1e-10:
                return b
    except Exception:
        pass
    return None

def greedy_step(a, C, n):
    """
    Greedy peak reduction: move mass from peak-contributing indices to non-contributing ones.
    Returns a new sequence.
    """
    k_star = int(np.argmax(C))
    # Contribution to C[k_star] from index j: a[k_star - j] (if valid)
    contrib = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            contrib[j] = a[idx]

    # Find best donor (high contribution) and receiver (low contribution)
    max_contrib_idx = np.argmax(contrib)
    min_contrib_idx = np.argmin(contrib)

    if max_contrib_idx == min_contrib_idx:
        return a

    # Move a small amount of mass
    delta = 0.05 * a[max_contrib_idx]
    if delta < 1e-10:
        return a

    a_new = a.copy()
    a_new[max_contrib_idx] -= delta
    a_new[min_contrib_idx] += delta
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations for sequence length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Edge-concentrated
    for frac in [0.1, 0.2, 0.3]:
        w = np.zeros(n)
        edge = max(1, int(n * frac))
        w[:edge] = 1.0
        w[-edge:] = 1.0
        w /= w.sum()
        inits['edge_{:.1f}'.format(frac)] = w

    # Gaussian center
    for sf in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - n * 0.5) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_{:.1f}'.format(sf)] = g

    # Two Gaussians
    for sf in [0.05, 0.1, 0.15]:
        g = np.exp(-0.5 * ((x - n * 0.25) / (sf * n)) ** 2) + \
            np.exp(-0.5 * ((x - n * 0.75) / (sf * n)) ** 2)
        g /= g.sum()
        inits['bi_{:.2f}'.format(sf)] = g

    # Cosine
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w

    # Random Dirichlet
    inits['rand'] = rng.dirichlet(np.ones(n) * 0.3)

    # Sparse random
    sparse = np.zeros(n)
    k = max(2, n // 4)
    idx = rng.choice(n, size=k, replace=False)
    sparse[idx] = rng.dirichlet(np.ones(k))
    inits['sparse'] = sparse

    # Parent
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0)
            if pp.sum() > 1e-10:
                inits['parent'] = pp / pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, parent.size),
                           np.maximum(parent, 0))
            if pp.sum() > 1e-10:
                inits['parent_i'] = pp / pp.sum()

    return inits

                        print("  LP: t={:.2f}, score={:.6f}".format(t_step, score))
                        break

        # Greedy refinement
        if remaining > 10:
            a_cur = best_sequence.copy()
            C_cur = fast_convolve(a_cur)
            for _ in range(min(200, int(remaining * 50))):
                if time.monotonic() > deadline - 5:
                    break
                a_try = greedy_step(a_cur, C_cur, n)
                C_try = fast_convolve(a_try)
                obj_try = compute_score_from_conv(C_try, n)
                obj_cur = compute_score_from_conv(C_cur, n)
                if obj_try < obj_cur - 1e-12:
                    a_cur = a_try
                    C_cur = C_try
                else:
                    break
            score = float(evaluate_sequence(a_cur.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_cur
                print("  Greedy: score={:.6f}".format(score))

    if best_sequence is None:
        best_sequence = np.ones(128)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    total_time = time.monotonic() - start_time
    print("FINAL: score={:.6f}, n={}, time={:.1f}s".format(best_score, len(best_sequence), total_time))
    return [float(x) for x in best_sequence]
