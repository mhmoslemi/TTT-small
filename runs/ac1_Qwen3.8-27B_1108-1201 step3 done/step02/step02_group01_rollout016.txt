def fft_convolve(a, b):

def local_score(a):

def grad_smooth_max_conv(p, tau):
    """
    Compute gradient of tau*log(sum exp(conv(p,p)/tau)) w.r.t. p.

        w[np.argmax(conv)] = 1.0

        w /= w_sum

"""
Smoothed subgradient descent with multiplicative updates and adaptive tau
annealing for minimizing peak autocorrelation of non-negative step functions.
Uses Nesterov momentum, multiple diverse initializations, local search polish,
and multi-length exploration to find low-score sequences.

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (s * s))


def compute_gradient_smooth(p, tau):
    """
    Compute gradient of tau*log(sum exp(conv(p,p)/tau)) w.r.t. p.
    Returns (gradient, conv, peak_value).
    """
    n = len(p)
    conv = fft_convolve(p, p)
    conv_max = np.max(conv)
    conv_shifted = conv - conv_max
    w = np.exp(conv_shifted / tau)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    grad = 2.0 * np.correlate(w, p, mode='valid')
    return grad, conv, conv_max


def smooth_subgrad_optimize(p0, time_limit, verbose=False):

    Smoothed subgradient descent with multiplicative updates and momentum
    for minimizing max conv(p,p) subject to p >= 0, sum(p) = 1.
    """
    n = len(p0)
    p = p0.copy().astype(np.float64)
    p = np.maximum(p, 1e-15)
    p /= p.sum()

    best_p = p.copy()
    best_peak = np.max(fft_convolve(p, p))

    v = np.zeros(n)
    momentum = 0.8

    conv0 = fft_convolve(p, p)
    peak0 = np.max(conv0)
    tau0 = peak0 * 2.0
    tau_final = peak0 / (n * 100.0)

        progress = min(1.0, elapsed / max(1.0, time_limit))

        tau = tau_final + 0.5 * (tau0 - tau_final) * (1.0 + np.cos(np.pi * progress))

        grad, conv, peak = compute_gradient_smooth(p, tau)

            print(f"        iter {i}, t={elapsed:.1f}s, peak={best_peak:.10f}, "
                  f"tau={tau:.8f}, mom={best_momentum:.3f}")

    return best_p, best_peak


def smooth_subgrad_mult(p0, time_limit, verbose=False):
    """
    Smoothed subgradient with multiplicative updates (preserves non-negativity).
    p_new = p * exp(-lr * grad), then normalize.
    """
    n = len(p0)
    p = p0.copy().astype(np.float64)
    p = np.maximum(p, 1e-15)
    p /= p.sum()

    best_p = p.copy()
    best_peak = np.max(fft_convolve(p, p))

    conv0 = fft_convolve(p, p)
    peak0 = np.max(conv0)
    tau0 = peak0 * 3.0
    tau_final = peak0 / (n * 80.0)

    t_start = time.monotonic()
    lr = 0.3
    i = 0
    stall = 0

    while time.monotonic() - t_start < time_limit:
        i += 1
        elapsed = time.monotonic() - t_start
        progress = min(1.0, elapsed / max(1.0, time_limit))

        tau = tau_final + 0.5 * (tau0 - tau_final) * (1.0 + np.cos(np.pi * progress))

        grad, conv, peak = compute_gradient_smooth(p, tau)
        grad_max = np.max(np.abs(grad))
        if grad_max < 1e-18:
            break

        # Multiplicative update
        step = lr * grad / (grad_max + 1e-15)
        p_new = p * np.exp(-step)
        p_new = np.maximum(p_new, 1e-15)
        p_new /= p_new.sum()

        p = p_new
        current_peak = np.max(fft_convolve(p, p))

        if current_peak < best_peak - 1e-14:
            best_peak = current_peak
            best_p = p.copy()
            stall = 0
            lr = min(lr * 1.02, 1.0)
        else:
            stall += 1
            if stall > 30:
                lr *= 0.85
                stall = 0

        if verbose and i % 200 == 0:
            print(f"        mult iter {i}, t={elapsed:.1f}s, peak={best_peak:.10f}, lr={lr:.4f}")

    return best_p, best_peak
