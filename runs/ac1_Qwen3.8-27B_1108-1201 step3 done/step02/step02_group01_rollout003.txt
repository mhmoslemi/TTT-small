        peak_idx = int(np.argmax(conv))
        peak_history.append(peak_idx)
        
        # Subgradient: grad[j] = 2*a[peak_idx - j] if valid
        grad = np.zeros(n)
        valid = (peak_idx >= 0) & (peak_idx < 2 * n - 1)
        if valid:
            # For each j, check if peak_idx - j is in [0, n-1]
            j_range = np.arange(n)
            i_vals = peak_idx - j_range
            mask = (i_vals >= 0) & (i_vals < n)
            grad[mask] = 2.0 * a[i_vals[mask]]
        
        # Adaptive step size
        if len(peak_history) > 50:
            recent = peak_history[-50:]
            unique_recent = len(set(recent))
            if unique_recent <= 3:
                lr *= 0.97  # Stuck in a cycle
            elif unique_recent > 15:
                lr = min(lr * 1.03, 5.0 / n)
        
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        
        # Ensure minimum positivity to avoid degeneracy
        a_new = np.maximum(a_new, 1e-8)
        a_new = normalize(a_new, S)
        
        new_peak = np.max(fft_convolve(a_new))
        if new_peak < best_peak:
            best_peak = new_peak
            best_a = a_new.copy()
            a = a_new
            stall = 0
        elif new_peak < np.max(conv):
            a = a_new
            stall += 1
        else:
            stall += 2
            lr *= 0.93
        
        if stall > 200:
            lr *= 0.5
            stall = 0
            if lr < 1e-5:
                break
    
    return best_a


def anneal_search(a, time_limit, verbose=False):
    """
    Simulated annealing with structured moves targeted at reducing peak correlation.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S).copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a))
    rng = np.random.default_rng(777)
    
    T = 0.05 * best_peak
    T_min = T * 1e-5
    cooling = 0.9995
    accept_count = 0
    total_count = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1:
            break
        if T < T_min:
            break
        
        conv = fft_convolve(a)
        peak_idx = int(np.argmax(conv))
        current_peak = conv[peak_idx]
        
        # Choose move type
        move_type = rng.integers(0, 4)
        candidate = a.copy()
        
        if move_type == 0:
            # Block perturbation
            width = max(1, int(n * rng.uniform(0.01, 0.1)))
            start = int(rng.integers(0, max(1, n - width)))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.03, size=width))
        elif move_type == 1:
            # Targeted: reduce peak contributors, boost others
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(3, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            for idx in top_idx:
                candidate[idx] *= np.exp(rng.normal(-0.04, 0.015))
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                n_inc = max(3, n // 20)
                inc_idx = rng.choice(others, size=min(n_inc, len(others)), replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=len(inc_idx)))
        elif move_type == 2:
            # Gaussian-smoothed perturbation
            noise = rng.normal(0, 0.02, size=n)
            noise = gaussian_filter1d(noise, sigma=max(1, n // 100))
            candidate *= np.exp(noise)
        else:
            # Oscillatory perturbation
            freq = rng.uniform(1, 10)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.01, 0.05)
            t = np.arange(n)
            candidate *= np.exp(amp * np.sin(2 * np.pi * freq * t / n + phase))
        
        candidate = np.maximum(candidate, 1e-8)
        candidate = normalize(candidate, S)
        
        new_peak = np.max(fft_convolve(candidate))
        delta = new_peak - current_peak
        total_count += 1
        
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            a = candidate
            accept_count += 1
            if new_peak < best_peak:
                best_peak = new_peak
                best_a = a.copy()
        
        T *= cooling
    
    if verbose:
        print(f"    Anneal: accepted {accept_count}/{total_count}, best_peak={best_peak:.6f}")
    
    return best_a


def ift_preprocess(a, n_iterations=50):
    """
    Iterative Fourier Transform preprocessing to flatten the autocorrelation spectrum.
    Alternates between time-domain non-negativity projection and frequency-domain
    spectral flattening.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S).copy()
    
    for _ in range(n_iterations):
        # Time domain: ensure non-negative and normalize
        a = np.maximum(a, 0.0)
        a = normalize(a, S)
        
        # Frequency domain: flatten |A|^2
        A = np.fft.fft(a)
        mag = np.abs(A)
        # Target: make |A|^2 as flat as possible (but keep DC)
        # Use a soft target: blend current magnitudes with flat target
        flat_target = np.median(mag[1:])  # exclude DC
        # Partial flattening
        blend = 0.3
        target_mag = np.ones(n)
        target_mag[0] = mag[0]  # keep DC
        target_mag[1:] = flat_target
        new_mag = (1 - blend) * mag + blend * target_mag
        new_mag[0] = mag[0]
        
        # Preserve phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        a = np.real(np.fft.ifft(A_new))
    
    a = np.maximum(a, 0.0)
    a = normalize(a, S)
    return a


def make_initialization(n, rng, init_type):
    """Create various asymmetric initial sequences."""
    t = np.arange(n, dtype=np.float64)
    
    if init_type == 'asym_chirp':
        alpha = rng.uniform(1.0, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + 0.5 * np.sin(alpha * np.pi * t / n + phase)
    elif init_type == 'ramp_osc':
        a = 0.5 + 0.4 * t / max(1, n - 1) + 0.2 * np.sin(2 * np.pi * rng.integers(1, 8) * t / n)
    elif init_type == 'exp_decay':
        sigma = n * rng.uniform(0.2, 0.6)
        f = rng.integers(2, max(3, n // 5))
        a = np.exp(-t / sigma) + 0.3 * np.cos(2 * np.pi * f * t / n)
    elif init_type == 'two_bump':
        c1 = int(n * rng.uniform(0.2, 0.4))
        c2 = int(n * rng.uniform(0.6, 0.8))
        s1 = n * rng.uniform(0.05, 0.15)
        s2 = n * rng.uniform(0.05, 0.15)
        a = np.exp(-0.5 * ((t - c1) / s1) ** 2) + 0.7 * np.exp(-0.5 * ((t - c2) / s2) ** 2)
    elif init_type == 'random_struct':
        a = 1.0 + 0.3 * rng.standard_normal(n)
        a = gaussian_filter1d(a, sigma=max(1, n // 50))
    elif init_type == 'multi_frac':
        a = np.ones(n)
        for _ in range(4):
            f = rng.integers(1, n // 3)
            amp = rng.uniform(0.1, 0.3)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'log_spread':
        a = np.exp(-0.5 * (np.log1p(t) / np.log1p(n)) ** 2 * 4) + 0.1 * rng.standard_normal(n)
    else:
        a = 1.0 + 0.3 * np.sin(np.pi * t / n + rng.uniform(0, np.pi))
    
    return np.clip(a, 0.01, None)


def interpolate_seq(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: Broad exploration with diverse inits and lengths (brief LP)
    Phase 2: IFT preprocessing + LP refinement on top candidates
    Phase 3: Deep LP refinement on best
    Phase 4: Gradient polish
    Phase 5: Annealing for local optima escape
    Phase 6: Final LP + gradient polish
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    def update_best(a):
        nonlocal best_score, best_seq
        if a is None:
            return False
        a_c = np.clip(a, 0.0, 1000.0)
        s = local_score(a_c)
        if s < best_score:
            old_score = best_score
            best_score = s
            best_seq = a_c.copy()
            print(f"  *** New best: {best_score:.6f} (n={len(a_c)}, prev={old_score:.6f}) ***")
            return True
        return False
    
    # Get provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        print(f"Provided: n={len(provided)}, score={local_score(provided):.6f}")
        update_best(provided)
    
    # Phase 1: Exploration
    print("\n=== Phase 1: Exploration ===")
    results = []
    
    init_types = ['asym_chirp', 'ramp_osc', 'exp_decay', 'two_bump', 'random_struct', 'multi_frac', 'log_spread']
    lengths = [512, 640, 800, 1000, 1200, 1500]
    
    # Include provided at its length
    if provided is not None:
        results.append(('provided', provided.copy()))
    
    # Generate candidates
    candidates_to_try = []
    for n in lengths:
        for it in init_types[:4]:
            candidates_to_try.append((f'init_{it}_{n}', n, it))
    
    # Also try interpolated provided at other lengths
    if provided is not None:
        for n in [800, 1000, 1200, 1500]:
            if n != len(provided):
                candidates_to_try.append((f'interp_{n}', n, 'interp'))
    
    # Time budget for phase 1
    phase1_time = min(400, (deadline - t_global) * 0.25)
    t_phase1_start = time.monotonic()
    per_candidate_time = phase1_time / max(1, len(candidates_to_try) + len(results))
    
    # First, do brief LP on provided
    if provided is not None:
        t_per = min(30, per_candidate_time * 2)
        print(f"  LP on provided (n={len(provided)}, {t_per:.0f}s)...")
        res = lp_refine(provided, t_per, verbose=False)
        results.append(('provided_lp', res))
        update_best(res)
    
    # Try other candidates with very brief LP
    for name, n, it in candidates_to_try:
        elapsed = time.monotonic() - t_phase1_start
        if elapsed > phase1_time:
            break
        
        remaining_per = per_candidate_time
        if time.monotonic() - t_global > deadline - 500:
            break
        
        if it == 'interp' and provided is not None:
            a0 = interpolate_seq(provided, n)
        else:
            a0 = make_initialization(n, rng, it)
        
        a0 = np.clip(a0, 0.01, None)
        
        # Very brief LP (just a few iterations to see if promising)
        t_brief = min(8.0, remaining_per)
        if t_brief < 3:
            continue
        res = lp_refine(a0, t_brief, verbose=False)
        s = local_score(res)
        results.append((name, res))
        update_best(res)
    
    print(f"  Phase 1 done: best={best_score:.6f}, candidates={len(results)}")
    
    # Phase 2: IFT + LP on top candidates
    print("\n=== Phase 2: IFT + Deep LP ===")
    phase2_time = min(400, (deadline - t_global) * 0.2)
    t_phase2_start = time.monotonic()
    
    # Sort results by score, take top 3
    scored = []
    for name, arr in results:
        s = local_score(arr)
        scored.append((s, name, arr))
    scored.sort(key=lambda x: x[0])
    
    for s, name, arr in scored[:3]:
        if time.monotonic() - t_phase2_start > phase2_time:
            break
        if time.monotonic() - t_global > deadline - 400:
            break
        
        n = len(arr)
        print(f"  IFT+LP on {name} (n={n}, score={s:.6f})...")
        
        # IFT preprocessing
        a_ift = ift_preprocess(arr, n_iterations=30)
        s_ift = local_score(a_ift)
        print(f"    After IFT: {s_ift:.6f}")
        update_best(a_ift)
        
        # LP refinement
        t_lp = min(60, phase2_time - (time.monotonic() - t_phase2_start))
        if t_lp > 10:
            res = lp_refine(a_ift, t_lp, verbose=False)
            update_best(res)
            res2 = lp_refine(arr, t_lp, verbose=False)
            update_best(res2)
    
    print(f"  Phase 2 done: best={best_score:.6f}")
    
    # Phase 3: Deep LP on best
    print("\n=== Phase 3: Deep LP ===")
    if best_seq is not None:
        remaining = min(deadline - t_global, 300) - 300
        t_lp = max(30, min(300, remaining))
        print(f"  Deep LP (n={len(best_seq)}, {t_lp:.0f}s)...")
        res = lp_refine(best_seq, t_lp, verbose=True)
        update_best(res)
    
    print(f"  Phase 3 done: best={best_score:.6f}")
    
    # Phase 4: Gradient polish
    print("\n=== Phase 4: Gradient ===")
    if best_seq is not None:
        remaining = max(30, min(150, deadline - t_global - 200))
        t_grad = max(20, remaining)
        print(f"  Gradient (n={len(best_seq)}, {t_grad:.0f}s)...")
        res = gradient_polish(best_seq, t_grad, verbose=True)
        update_best(res)
    
    print(f"  Phase 4 done: best={best_score:.6f}")
    
    # Phase 5: Annealing
    print("\n=== Phase 5: Annealing ===")
    if best_seq is not None:
        remaining = max(20, min(150, deadline - t_global - 100))
        t_anneal = max(20, remaining)
        print(f"  Anneal (n={len(best_seq)}, {t_anneal:.0f}s)...")
        res = anneal_search(best_seq, t_anneal, verbose=True)
        update_best(res)
    
    print(f"  Phase 5 done: best={best_score:.6f}")
    
    # Phase 6: Final polish (LP + gradient alternating)
    print("\n=== Phase 6: Final Polish ===")
    if best_seq is not None:
        remaining = deadline - t_global - 10
        if remaining > 30:
            # Alternate LP and gradient
            a_cur = best_seq.copy()
            for round_i in range(4):
                rem = deadline - t_global - 10
                if rem < 15:
                    break
                t_lp = min(rem * 0.5, 40)
                t_grad = min(rem * 0.4, 30)
                if t_lp < 5:
                    break
                print(f"  Round {round_i}: LP ({t_lp:.0f}s) + Grad ({t_grad:.0f}s)")
                res = lp_refine(a_cur, t_lp, verbose=False)
                update_best(res)
                a_cur = best_seq.copy()
                
                if time.monotonic() < deadline - 20:
                    res = gradient_polish(a_cur, t_grad, verbose=False)
                    update_best(res)
                    a_cur = best_seq.copy()
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate={final_score:.6f}, n={len(final_list)}, local={best_score:.6f} ===")
        return final_list
    
    # Fallback
    if provided is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided]
    
    return [1.0] * 256
