def fast_convolve(a):

    n = len(a)

    A = np.fft.rfft(a, fft_size)

    # First term: sum_{j=0}^{n-1} w[i+j] * a[j]
    idx = np.arange(n)
    M = w[idx[None, :] + idx[:, None]]  # M[j, i] = w[i+j]
    grad_first = M.T @ a
    
    # Second term: sum_{k=0}^{i} w[k] * a[i-k] = conv(w[:n], a)[i]
    grad_second = np.convolve(w[:n], a, mode='full')[:n]
    
    return grad_first + grad_second


def gradient_optimize(a0, t0=10.0, t1=200.0, eta0=0.01, time_limit=60.0):
    """
    Soft-max gradient descent to minimize peak of auto-convolution.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t_start = time.monotonic()
    max_iters = 50000
    no_improve = 0
    max_no_improve = 500
    
    for it in range(max_iters):
        if time.monotonic() - t_start > time_limit:
            break
        if no_improve >= max_no_improve:
            break
        
        conv = fast_convolve(a)
        curr_max = conv.max()
        
        if curr_max < best_obj - 1e-12:
            best_obj = curr_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Annealing: increase t over time
        progress = min(1.0, it / max_iters)
        t = t0 + (t1 - t0) * progress**0.5
        
        w = softmax(conv, t)
        grad = compute_grad_smoothed(a, w)
        
        # Adaptive learning rate
        eta = eta0 / (1.0 + 0.01 * it)
        
        # Gradient step
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        ns = a_new.sum()
        if ns < 1e-12:
            break
        a_new /= ns
        
        # Check if objective improved
        new_conv_max = fast_convolve(a_new).max()
        if new_conv_max >= curr_max + 1e-8:
            # Step too large, reduce

            a_new = a - eta * grad
            a_new = np.maximum(a_new, 0.0)
            ns = a_new.sum()
            if ns < 1e-12:
                break
            a_new /= ns
            new_conv_max = fast_convolve(a_new).max()
        
        a = a_new
    
    return best_a, best_obj


def solve_lp(f, M):
    """LP: find g >= 0 maximizing sum(g) s.t. conv(f,g) <= M."""
    n = len(f)
    # Build constraint matrix A where A[k,j] = f[j] if k-j is valid, else 0
    # conv(f,g)[k] = sum_j f[j] * g[k-j] for valid j, k-j
    # For fixed k: sum_j f[j] * g[k-j] where 0<=j<n and 0<=k-j<n
    # = sum_j f[j] * g[k-j] where j in [max(0,k-n+1), min(n-1,k)]
    
    # A has shape (2n-1, n), A[k,j] = f[j] if 0<=k-j<n, else 0
    # i.e., A[k,j] = f[j] if j <= k < j+n, else 0
    
    # For efficiency, build A as a banded matrix
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f[:n]
    
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def lp_optimize(f0, time_limit=60.0):
    """LP-based iterative optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-12:
        f = np.ones(len(f))
    else:
        f /= s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        
        g = solve_lp(f, M)
        
        if np.allclose(g, f, atol=1e-8):
            no_improve += 5
        else:
            # Line search for best t in [0, 1]
            best_t = 0.0
            best_f_val = best_obj
            for t in [0.1, 0.25, 0.5, 0.75, 1.0]:
                trial = (1-t)*f + t*g
                trial = np.maximum(trial, 0.0)
                ts = trial.sum()
                if ts < 1e-12:
                    continue
                trial /= ts
                obj = fast_convolve(trial).max()
                if obj < best_f_val:
                    best_f_val = obj
                    best_t = t
            
            if best_t > 0 and best_f_val < best_obj - 1e-12:
                f = (1-best_t)*f + best_t*g
                f = np.maximum(f, 0.0)
                f /= f.sum()
                best_obj = best_f_val
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve >= 50:
            break
    
    return best_f, best_obj


def local_search_peek(a, time_limit=30.0):
    """Local search: move mass away from peak-contributing positions."""
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a /= a.sum()
    n = len(a)
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    stuck = 0
    
    while time.monotonic() - t0 < time_limit and stuck < 200:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        
        # Compute contribution of each pair to the peak
        # conv[k_star] = sum_i a[i]*a[k_star-i]
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        
        # Move mass from high-contributing to low-contributing positions
        sorted_idx = np.argsort(contrib)
        high_idx = sorted_idx[-10:]  # top 10 contributors
        low_idx = sorted_idx[:10]    # bottom 10
        
        found = False
        for hi in high_idx:
            if found:
                break
            for li in low_idx:
                if hi == li:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    amt = a[hi] * frac

                        continue

                    trial = np.maximum(trial, 0.0)

                        continue
                    trial /= ts
                    obj = fast_convolve(trial).max()

                        best_obj = obj
                        a = trial
                        found = True

            stuck = 0
        else:
            stuck += 1

def get_parent(n):
    """Get parent sequence clipped to length n."""
    try:
        p = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim != 1 or p.size == 0:
            return None

        else:

def make_inits(n, rng):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    inits['uniform'] = np.ones(n) / n
    
    # Raised cosine (Hann window)
    if n > 1:
        hann = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
        hann = np.maximum(hann, 0.0)
        inits['hann'] = hann / hann.sum()
    
    # Gaussian center
    for sf in [0.2, 0.3, 0.4, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5*n) / (sf*n))**2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g
    
    # U-shaped (edges high)
    u = (x / (n-1))**2 + ((x/(n-1)) - 1)**2 + 0.1 if n > 1 else np.ones(n)
    inits['u_shaped'] = u / u.sum()
    
    # Triangle
    tri = np.minimum(x, n-1-x) + 1.0
    inits['triangle'] = tri / tri.sum()
    
    # Random (Dirichlet)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Parent
    parent = get_parent(n)
    if parent is not None:
        inits['parent'] = parent
    
    # Perturbed parent
    if parent is not None:
        for scale in [0.01, 0.05, 0.1]:
            pert = parent + scale * rng.standard_normal(n)
            pert = np.maximum(pert, 0.0)
            ps = pert.sum()
            if ps > 1e-10:
                inits[f'parent_pert_{scale}'] = pert / ps
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using soft-max gradient descent,
LP-based direction finding, and local search with multi-start.
The gradient of the soft-max smoothed peak is computed in closed form:
  grad[i] = 2 * sum_j w[i+j] * a[j]
where w are softmax weights over the convolution values.
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_weights(conv, t):
    """Compute stable softmax weights from temperature t."""
    shifted = t * conv
    shifted -= shifted.max()
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return e / s


def compute_grad(a, w):
    """
    Gradient of soft-max smoothed peak of auto-convolution.
    grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    """
    n = len(a)
    idx = np.arange(n)
    M = w[idx[None, :] + idx[:, None]]  # shape (n, n), M[j, i] = w[i+j]
    return 2.0 * (M.T @ a)


def gradient_optimize(a0, t0, t1, eta0, time_limit):
    """Soft-max gradient descent for minimizing peak of auto-convolution."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(len(a), dtype=np.float64)
    else:
        a /= s
    
    n = len(a)
    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    
    t_start = time.monotonic()
    max_iters = 100000
    no_improve = 0
    max_no_improve = 2000
    
    for it in range(max_iters):
        if time.monotonic() - t_start > time_limit:
            break
        if no_improve >= max_no_improve and t0 >= t1:
            break
        
        conv = fast_convolve(a)
        curr_peak = conv.max()
        
        if curr_peak < best_peak - 1e-12:
            best_peak = curr_peak
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Anneal temperature
        frac = min(1.0, it / max(max_iters * 0.8, 1))
        t = t0 + (t1 - t0) * frac
        
        w = softmax_weights(conv, t)
        grad = compute_grad(a, w)
        
        # Adaptive step size with backtracking
        eta = eta0
        for _ in range(10):
            trial = a - eta * grad
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-12:
                eta *= 0.5
                continue
            trial /= ts
            new_peak = fast_convolve(trial).max()
            if new_peak <= curr_peak + 1e-10:
                break
            eta *= 0.5
        else:
            break
        
        a = trial
    
    return best_a, best_peak
