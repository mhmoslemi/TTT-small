```python
def lp_direction(a, M):
    """Find direction g >= 0, sum(g) maximized, s.t. (a*g)[k] <= M for all k."""
    n = len(a)
    # Variables: g_0, ..., g_{n-1}
    # Objective: minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (a*g)[k] <= M for k = 0, ..., 2n-2
    # (a*g)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * g[k-i]
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            # a[i] * g[k-i] contributes to (a*g)[k]
            j = k - i  # index in g
            A_ub[k, j] = a[i]
    
    b_ub = np.full(2*n-1, M)
    
    # g >= 0 handled by bounds
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x
```

```python
def lp_step(a, M):
    """Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    n = len(a)
    # Build constraint matrix
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # which means (a*g)[k] = sum_j A[k,j] * g[j]
    
    # Use scipy linprog
    # Variables: g (n of them)
    # Objective: minimize -sum(g)
    # Constraints: A @ g <= M (2n-1 constraints)
    # Bounds: g >= 0
    
    c = -np.ones(n)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # (a*g)[k] = sum_{j: 0<=k-j<n} a[k-j] * g[j]
        # j ranges from max(0, k-n+1) to min(n-1, k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    return res.x
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based descent.
The key innovation is using a linear program at each step to find the
optimal direction that minimizes peak growth, rather than using simple
subgradients. This provides a principled, globally-informed step that
converges faster and to better solutions than naive subgradient methods.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(a, M):
    """
    Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns the optimal g vector.
    """
    n = len(a)
    # Build A matrix: A[k, j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end+1] = a[k - np.arange(j_start, j_end + 1)]
    
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 5.0})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None


def lp_descent(a0, time_limit, verbose=False):
    """
    LP-based descent to minimize max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    
    t = 0.3  # blend parameter
    t_min = 0.05
    t_max = 0.8
    consecutive_fail = 0
    
    start = time.monotonic()
    it = 0
    
    while time.monotonic() - start < time_limit:
        b = fast_convolve(a)
        M = b.max()
        
        g = solve_lp_direction(a, M)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum  # normalize to sum=1
        
        # Try the blend
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        ns = a_new.sum()
        if ns < 1e-15:
            t *= 0.5
            continue
        a_new = a_new / ns
        
        new_M = fast_convolve(a_new).max()
        
        if new_M < best_M - 1e-15:
            best_a = a_new.copy()
            best_M = new_M
            a = a_new
            consecutive_fail = 0
            # Increase step size cautiously
            t = min(t_max, t * 1.05)
        elif new_M < M:
            # Improvement over current but not best
            a = a_new
            t = min(t_max, t * 1.02)
            consecutive_fail = 0
        else:
            # No improvement
            consecutive_fail += 1
            t = max(t_min, t * 0.7)
            if consecutive_fail > 50:
                # Perturb and restart
                rng = np.random.default_rng(it)
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                t = 0.3
                consecutive_fail = 0
        
        it += 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"  LP iter {it}, M={best_M:.8f}, t={t:.3f}, time={elapsed:.1f}s")
    
    return best_a, best_M


def subgradient_polish(a0, time_limit):
    """
    Fine-grained subgradient descent for polishing.
    """
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    
    v = np.zeros(n)  # momentum
    lr = 0.01
    momentum = 0.9
    top_k = max(10, n // 20)
    
    start = time.monotonic()
    
    while time.monotonic() - start < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        # Weighted subgradient over top peaks
        peaks = np.argpartition(conv, -top_k)[-top_k:]

        scaled = peak_vals / (0.01 * M + 1e-15)
        scaled -= scaled.max()

                continue

            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += wi * a[k - idx]

        v = momentum * v - lr * sg
        a_new = project_simplex(a + v)
        a = a_new

        if cur_M < best_M - 1e-15:
            best_a = a.copy()
            best_M = cur_M
        
        lr = max(0.001, lr * 0.999)
    
    return best_a, best_M
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based descent with
adaptive step sizing, periodic perturbation, and subgradient polishing.
The key innovation is solving a linear program at each step to find the
globally optimal direction under the peak constraint, which provides
more informed steps than naive subgradient methods.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a):
    """Build the constraint matrix A for the LP. A[k,j] = a[k-j] if valid."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A[k, j_start:j_end + 1] = a[k - np.arange(j_start, j_end + 1)]
    return A


def solve_lp_direction(a, M):
    """
    Solve: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns optimal g or None on failure.
    """
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def lp_descent(a0, time_limit, verbose=False, perturb_interval=500,
               perturb_scale=0.008):

    LP-based descent to minimize max(conv(a,a)) with sum(a)=1.
    Returns (best_a, best_max_conv).
    """

    else:
        a = a / s
    n = len(a)

    best_M = fast_convolve(a).max()

    t = 0.4
    t_min = 0.02
    t_max = 0.9
    consecutive_fail = 0
    total_iter = 0

    start = time.monotonic()

    while time.monotonic() - start < time_limit:
        b = fast_convolve(a)
        M = b.max()

        g = solve_lp_direction(a, M)
        if g is None:

        g = g / g_sum

def subgradient_step(a, conv, top_k, temperature):
    """Compute weighted subgradient over top-k peaks of convolution."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

    if temperature > 1e-15:
        scaled = peak_vals / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        wi = w[i] * 2.0
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += wi * a[k - idx]

    return sg


def subgradient_polish(a0, time_limit, lr_init=0.008, momentum_init=0.9):
    """
    Subgradient descent with momentum for final polishing.
    Returns (best_a, best_max_conv).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_convolve(a).max()

    v = np.zeros(n)
    lr = lr_init
    momentum = momentum_init
    top_k = max(10, n // 25)
    temp = 0.005 * best_M
    no_improve = 0

    start = time.monotonic()
    it = 0

    while time.monotonic() - start < time_limit:
        for _ in range(20):
            if time.monotonic() - start > time_limit:
                break
            conv = fast_convolve(a)
            sg = subgradient_step(a, conv, top_k, temp)
            v = momentum * v - lr * sg
            a_new = project_simplex(a + v)
            a = a_new
            cur_M = fast_convolve(a).max()
            it += 1

            if cur_M < best_M - 1e-15:
                best_M = cur_M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        if no_improve >= 500:
            rng = np.random.default_rng(it)
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            v *= 0.3
            lr = max(0.001, lr * 0.7)
            no_improve = 0

        lr *= 0.9995
        momentum = min(0.97, momentum + 0.0002)
        temp *= 1.001

    return best_a, best_M


def mirror_descent(a0, time_limit, t_param=60.0, eta_init=0.3):
    """
    Multiplicative weight update (mirror descent) as an alternative optimizer.
    Returns (best_a, best_max_conv).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_a = a.copy()
    best_M = fast_convolve(a).max()

    eta = eta_init
    t_cur = t_param
    no_improve = 0

    start = time.monotonic()

    while time.monotonic() - start < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        if M < 1e-15:
            break

        scaled = t_cur * conv / M
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        thresh = w.max() * 0.01
        active = np.where(w > thresh)[0]
        for k_int in active:
            k = int(k_int)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[k_int] * 2.0 * a[k - idx]

        sg_centered = sg - sg.mean()
        log_f = -eta * sg_centered
        log_f -= log_f.max()
        a = a * np.exp(log_f)
        a = np.maximum(a, 0.0)
        s2 = a.sum()
        a = a / s2 if s2 > 1e-15 else np.ones(n) / n

        cur_M = fast_convolve(a).max()
        if cur_M < best_M - 1e-14:
            best_M = cur_M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 80:
            no_improve = 0
            eta *= 0.75
            t_cur *= 1.4
            if eta < 0.02:
                eta = 0.25
                t_cur = t_param

    return best_a, best_M


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n, all summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    ramp_inc = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp_inc'] = ramp_inc / ramp_inc.sum()
    inits['ramp_dec'] = ramp_inc[::-1].copy()

    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    for sf in [0.15, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits[f'gauss_{sf}'] = g / g.sum()

    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    bump = np.exp(-x / (0.3 * n))
    inits['bump_left'] = bump / bump.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.4)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.1)

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase LP-based search with subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: len={len(p)}, sum={p.sum():.2f}, "
                  f"score={parent_score:.6f}")
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
    except Exception:
        pass

    # Phase 1: Quick LP scan over lengths and initializations
    print("=== Phase 1: Quick LP Scan ===")
    lengths = [128, 256, 512, 1024]

    for n in lengths:
        if time.monotonic() > deadline - 120:
            break

        inits = generate_initializations(n, parent, rng)
        # Prioritize: parent, uniform, gauss, random
        priority = ['parent', 'parent_up', 'parent_ds', 'uniform',
                    'gauss_0.3', 'random', 'gauss_0.15', 'bimodal']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in init_names]
        init_names = init_names[:6]

        time_per = max(15.0, 250.0 / len(lengths) / 4)

        for name in init_names:
            if time.monotonic() > deadline - 100:
                break
            t_alloc = min(time_per, deadline - time.monotonic() - 80)
            if t_alloc < 8:
                break

            a0 = inits[name]
            a_opt, M_opt = lp_descent(a0, time_limit=t_alloc)
            score = 2.0 * n * M_opt
            actual = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best_sequence = a_opt.copy()
                print(f"  n={n}, {name}: score={actual:.6f} "
                      f"(M={M_opt:.8f})")
            elif score < best_score:
                # Use LP estimate
                pass

    print(f"After Phase 1: best_score={best_score:.6f}")

    # Phase 2: Deep LP optimization
    print("=== Phase 2: Deep LP Optimization ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 60

        # Multiple deep runs with different strategies
        runs = 6
        t_per_run = remaining / runs

        for run_i in range(runs):
            if time.monotonic() > deadline - 40:
                break
            t_alloc = min(t_per_run * (1.2 if run_i < 3 else 0.8),
                          deadline - time.monotonic() - 30)
            if t_alloc < 10:
                break

            if run_i == 0:
                a_start = best_sequence.copy()
            elif run_i == 1:
                noise = rng.standard_normal(n_best) * 0.003
                a_start = project_simplex(best_sequence + noise)
            elif run_i == 2:
                a_start = project_simplex(best_sequence +
                                          rng.standard_normal(n_best) * 0.01)
            elif run_i == 3:
                # Mirror descent from best
                a_start = best_sequence.copy()
                a_opt, M_opt = mirror_descent(a_start, time_limit=t_alloc * 0.5)
                score = 2.0 * n_best * M_opt
                actual = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(actual) and actual < best_score:
                    best_score = actual
                    best_sequence = a_opt.copy()
                    print(f"  Mirror run: score={actual:.6f}")
                # Then LP from mirror result
                a_opt, M_opt = lp_descent(a_opt, time_limit=t_alloc * 0.5)
                actual = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(actual) and actual < best_score:
                    best_score = actual
                    best_sequence = a_opt.copy()
                    print(f"  LP after mirror: score={actual:.6f}")
                continue
            elif run_i == 4:
                # Subgradient then LP
                a_opt_sg, _ = subgradient_polish(best_sequence,
                                                 time_limit=t_alloc * 0.4,
                                                 lr_init=0.005,
                                                 momentum_init=0.92)
                a_opt, M_opt = lp_descent(a_opt_sg, time_limit=t_alloc * 0.6)
            else:
                # Fresh random start at best length
                a_start = rng.dirichlet(np.ones(n_best) * 0.3)

            a_opt, M_opt = lp_descent(a_start, time_limit=t_alloc)
            actual = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best_sequence = a_opt.copy()
                print(f"  Deep run {run_i}: score={actual:.6f} "
                      f"(n={n_best}, M={M_opt:.10f})")

    # Phase 3: Try longer sequences
    print("=== Phase 3: Longer Sequences ===")
    if time.monotonic() < deadline - 80:
        for n_long in [512, 1024, 2048]:
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - time.monotonic() - 40
            if remaining < 40:
                break

            if best_sequence is not None:
                a0 = np.interp(np.linspace(0, 1, n_long),
                               np.linspace(0, 1, len(best_sequence)),
                               best_sequence)
                a0 = np.maximum(a0, 0)
                a0 = a0 / a0.sum()
            else:
                a0 = np.ones(n_long) / n_long

            t_alloc = min(remaining / 2, 400)
            a_opt, M_opt = lp_descent(a0, time_limit=t_alloc)
            actual = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best_sequence = a_opt.copy()
                print(f"  Long n={n_long}: score={actual:.6f}")

    # Phase 4: Final polish
    print("=== Phase 4: Final Polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 10:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 5

        # LP polish
        if remaining > 30:
            t_lp = min(remaining * 0.4, 300)
            a_opt, M_opt = lp_descent(best_sequence, time_limit=t_lp,
                                       perturb_interval=1000,
                                       perturb_scale=0.004)
            actual = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best_sequence = a_opt.copy()
                print(f"  LP polish: score={actual:.6f}")
            remaining = deadline - time.monotonic() - 5

        # Subgradient polish
        if remaining > 20:
            t_sg = min(remaining * 0.3, 200)
            a_opt, M_opt = subgradient_polish(best_sequence, time_limit=t_sg,
                                               lr_init=0.004,
                                               momentum_init=0.93)
            actual = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best_sequence = a_opt.copy()
                print(f"  SG polish: score={actual:.6f}")
            remaining = deadline - time.monotonic() - 5

        # One more LP from polished result
        if remaining > 20:
            t_lp2 = min(remaining * 0.5, 200)
            a_opt, M_opt = lp_descent(best_sequence, time_limit=t_lp2,
                                       perturb_interval=800,
                                       perturb_scale=0.003)
            actual = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(actual) and actual < best_score:
                best_score = actual
                best_sequence = a_opt.copy()
                print(f"  Final LP: score={actual:.6f}")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Ensure all values are valid
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}, "
          f"sum={best_sequence.sum():.4f}")
    return [float(x) for x in best_sequence]
```
