"""
Multi-phase optimization for minimizing peak auto-convolution of non-negative sequences.
Uses smooth log-sum-exp optimization, LP-based refinement, and coordinate descent polish.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def smooth_optimize(a_init, time_limit, alpha_start=5.0, alpha_end=200.0, lr=0.05, verbose=True):
    """Adam optimizer with log-sum-exp smoothing of the peak."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    a = a / max(np.sum(a), 1e-10)
    
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step = 0
    n_fft = 1
    while n_fft < 2 * n - 1:
        n_fft <<= 1
    
    while time.monotonic() - t0 < time_limit - 2.0:
        c = fft_convolve(a, a)
        
        # Compute alpha (anneal)
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        
        # Softmax weights
        c_max = np.max(c)
        exp_c = np.exp(alpha * (c - c_max))
        sum_exp = np.sum(exp_c)
        p = exp_c / sum_exp
        
        # Gradient: grad[j] = 2n * (2 * corr[j] - correction[j])
        # where corr[j] = sum_k p[k] * a[k-j] (cross-correlation)
        P = rfft(p, n_fft)
        A = rfft(a, n_fft)
        corr = irfft(P * np.conjugate(A), n_fft)[:n]
        
        # Correction: for each j, subtract p[2*j] * a[j] if 2*j < 2*n-1
        idx2 = 2 * np.arange(n)
        corr_mask = idx2 < 2 * n - 1
        correction = np.zeros(n)
        correction[corr_mask] = p[idx2[corr_mask]] * a[np.arange(n)[corr_mask]]
        
        grad = 2.0 * n * (2.0 * corr - correction)
        
        # Adam update
        step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, None)
        
        S = np.sum(a)
        if S < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / S
        
        step += 1  # Count the normalization step too for timing
        
        if step % 2000 == 0 or step < 5:
            s = score_fast(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    [smooth {elapsed:5.1f}s] step={step:5d} alpha={alpha:6.1f} score={s:.6f} best={best_score:.6f}")
    
    s = score_fast(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()
    return best_a, best_score


def build_conv_matrix(f):
    """Build the (2n-1) x n matrix A such that A @ b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A


def lp_refine(f, time_limit, M_override=None, verbose=True):
    """LP-based: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    f = f.astype(np.float64).copy()
    best_f = f.copy()
    S = np.sum(f)
    best_score = score_fast(f)
    t0 = time.monotonic()
    
    # Build A matrix (fixed for this phase)
    A = build_conv_matrix(f)
    
    while time.monotonic() - t0 < time_limit - 3.0:
        c = fft_convolve(f, f)
        M = float(np.max(c)) if M_override is None else M_override
        
        # Solve LP: max sum(b) s.t. A@b <= M, b >= 0
        c_obj = -np.ones(n)
        try:
            res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M),
                         bounds=[(0, None)]*n, method='highs',
                         options={'time_limit': 4.0, 'presolve': True})
            if not res.success:
                break
            b = res.x
        except Exception:
            break
        
        Sb = np.sum(b)
        if Sb < 1e-12:
            break
        
        # Scale b to match sum(f)
        b_scaled = b * (S / Sb)
        
        # Line search
        best_t = 0.05
        best_s = np.inf
        for t_try in [0.05, 0.1, 0.2, 0.3]:
            f_cand = (1 - t_try) * f + t_try * b_scaled
            f_cand = np.clip(f_cand, 0.0, None)
            sc = score_fast(f_cand)
            if sc < best_s:
                best_s = sc
                best_t = t_try
        
        f = (1 - best_t) * f + best_t * b_scaled
        f = np.clip(f, 0.0, None)
        
        if best_s < best_score - 1e-9:
            best_score = best_s
            best_f = f.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    [lp {elapsed:5.1f}s] M={M:.4f} score={best_s:.6f} best={best_score:.6f}")
        
        # Check for stagnation
        if best_s > best_score * (1 - 1e-8):
            # Try to escape
            rng = np.random.default_rng(int(time.monotonic()*1000) % 100000)
            f += 0.01 * rng.standard_normal(n)
            f = np.clip(f, 0.01, None)
            f = f / np.sum(f) * S
    
    return best_f, best_score


def coordinate_descent(a_init, time_limit, verbose=True):
    """Coordinate descent: optimize one element at a time."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    S = float(np.sum(a))
    best_a = a.copy()
    best_score = score_fast(a)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit - 2.0:
        improved = False
        # Compute current convolution
        c = fft_convolve(a, a)
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit - 2.0:
                break
            
            # c_k = sum_i a_i * a_{k-i}
            # When we change a_j: c_k_new = c_k + delta * (a_{k-j} + a_{j-k} - a_j * delta_{k,2j})
            # where delta = a_j_new - a_j_old
            # Actually: c_k = ... + a_j * a_{k-j} + a_{k-j} * a_j + ...
            # If k-j != j: two terms, so dc_k/d a_j = 2 * a_{k-j}
            # If k-j == j (i.e., k=2j): one term a_j^2, so dc_k/d a_j = 2*a_j (derivative of a_j^2)
            # Wait: c_{2j} includes the term a_j * a_j = a_j^2, and dc_{2j}/da_j = 2*a_j
            # c_k for k != 2j: includes a_j*a_{k-j} + a_{k-j}*a_j = 2*a_j*a_{k-j} (two terms), so dc_k/da_j = 2*a_{k-j}
            # But only if k-j is in [0, n-1]!
            
            # So: dc_k/da_j = 2*a_{k-j} if 0 <= k-j < n and k != 2j
            #       dc_k/da_j = 2*a_j if k = 2j (i.e., a_{k-j} = a_j, and it's a_j^2 not 2*a_j^2)
            # Actually: c_{2j} = ... + a_j*a_{2j-j} + ... = ... + a_j*a_j + ... 
            # The term a_j appears in c_{2j} as a_j * a_{2j-j} = a_j * a_j = a_j^2
            # dc_{2j}/da_j = 2*a_j (derivative of a_j^2)
            # But also, does a_j appear elsewhere in c_{2j}? c_{2j} = sum_i a_i * a_{2j-i}
            # The term i=j gives a_j * a_{2j-j} = a_j^2. No other term has a_j.
            # So dc_{2j}/da_j = 2*a_j. ✓
            # For k != 2j with 0<=k-j<n: c_k contains a_j*a_{k-j} (term i=j) and a_{k-j}*a_j (term i=k-j).
            # dc_k/da_j = a_{k-j} + a_{k-j} = 2*a_{k-j}. ✓
            
            # So: w_k = 2*a_{k-j} if 0<=k-j<n, else 0
            #      AND w_{2j} should be 2*a_j (which is already 2*a_{2j-j} = 2*a_j) ✓
            # Wait, that's consistent! w_k = 2*a_{k-j} for all k where 0<=k-j<n, and 0 otherwise.
            # But at k=2j: w_{2j} = 2*a_{2j-j} = 2*a_j. And dc_{2j}/da_j should be 2*a_j. ✓
            
            # So the gradient contribution is simply: w_k = 2 * a[k-j] if 0 <= k-j < n else 0
            
            # New a_j: let x = a_j (the variable we're optimizing)
            # c_k_new = c_k + (x - a_j_old) * w_k
            # S_new = S - a_j_old + x
            # score = 2n * max_k(c_k_new) / S_new^2
            
            # For the 1D minimization, the max is a piecewise-linear function of x.
            # The optimal x for a fixed argmax k* is:
            # minimize (c_k* + (x - a_j_old) * w_k*) / (S - a_j_old + x)^2
            # Let A = c_k* - a_j_old * w_k*, B = w_k*, C = S - a_j_old
            # minimize (A + B*x) / (C + x)^2
            # d/dx [(A+Bx)/(C+x)^2] = [B(C+x)^2 - (A+Bx)*2(C+x)] / (C+x)^4
            # = [(C+x)(B(C+x) - 2(A+Bx))] / (C+x)^4
            # = [BC + Bx - 2A - 2Bx] / (C+x)^3
            # = [BC - 2A - Bx] / (C+x)^3
            # Set = 0: x* = (BC - 2A) / B (if B > 0)
            
            # But we need to check which k* is the argmax at x*, which may change.
            # For simplicity, use a few iterations of "find argmax, optimize, repeat."
            
            a_j_old = a[j]
            
            # Find which k gives the max
            # c_k_new = c_k + (x - a_j_old) * w_k
            # w_k = 2*a[k-j] if 0 <= k-j < n else 0
            
            k_valid = np.arange(max(0, j - (n-1) ... ))
            
            # Actually, w_k = 2*a[k-j] for k in [j, j+n-1] (where 0 <= k-j < n)
            lo = j
            hi = min(2*n-2, j+n-1)
            w = np.zeros(2*n-1)
            w[lo:hi+1] = 2.0 * a[0:hi+1-lo]
            
            # For a range of x, find the k that maximizes c_k + (x - a_j_old) * w_k
            # This is an upper envelope of lines, each with slope w_k and intercept c_k - a_j_old*w_k
            
            # For a practical approach: evaluate at a few candidate x values
            # x = 0, x = a_j_old, x = optimal for current argmax, etc.
            
            # Current argmax
            k_star = int(np.argmax(c))
            
            # Optimal x for this k_star:
            A_val = c[k_star] - a_j_old * w[k_star]
            B_val = w[k_star]
            C_val = S - a_j_old
            
            if abs(B_val) > 1e-15:
                x_opt = (B_val * C_val - 2.0 * A_val) / B_val
            else:
                x_opt = a_j_old  # If B=0, the term c_k* doesn't change, just keep current
            
            x_opt = max(0.0, x_opt)
            # Upper bound: don't let a_j get too large
            x_opt = min(x_opt, 10.0 * S / n)
            
            # Also try x=0 and x=a_j_old and a few other candidates
            candidates = [0.0, a_j_old, x_opt, 0.5 * x_opt, 2.0 * x_opt]
            candidates = [max(0.0, min(c, 10.0 * S / n)) for c in candidates]
            
            best_x = a_j_old
            best_s = score_fast(a)
            
            for xc in candidates:
                a[j] = xc
                S_new = S - a_j_old + xc
                if S_new < 0.01:
                    continue
                c_new = c + (xc - a_j_old) * w
                s_new = 2.0 * n * np.max(c_new) / (S_new * S_new)
                if s_new < best_s:
                    best_s = s_new
                    best_x = xc
            
            a[j] = best_x
            S = S - a_j_old + best_x
            c = c + (best_x - a_j_old) * w
            
            if best_x != a_j_old:
                improved = True
        
        s = score_fast(a)
        if s < best_score:
            best_score = s
            best_a = a.copy()
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    [coord {elapsed:5.1f}s] score={s:.6f} best={best_score:.6f} improved={improved}")
        
        if not improved:
            break
    
    return best_a, best_score


def make_init(n, kind, rng):
    t = np.arange(n, dtype=np.float64)
    if kind == 'uniform':
        return np.ones(n)
    elif kind == 'hann':
        return 0.5 * (1 + np.cos(2 * np.pi * t / max(1, n-1)))
    elif kind == 'cos_mod':
        freq = 1 + rng.integers(0, max(2, n // 16))
        return np.clip(1.0 + 0.3 * np.cos(2 * np.pi * freq * t / n), 0.01, None)
