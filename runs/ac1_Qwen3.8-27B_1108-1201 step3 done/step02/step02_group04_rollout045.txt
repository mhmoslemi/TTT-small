    return result[:size]


def compute_grad_fft(p, w):
    """Compute gradient of smoothed max of p*p w.r.t. p using FFT.
    w: softmax weights over convolution values (length 2n-1).
    Returns gradient vector of length n."""
    n = len(p)
    w_flip = w[::-1]
    total_len = n + (2 * n - 1) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    W = np.fft.rfft(w_flip, fft_size)
    full = np.fft.irfft(P * W, fft_size)
    grad = full[n - 1:2 * n - 1][::-1]
    return grad


def make_lp_structure(n):
    """Precompute sparse indices for the cross-convolution LP constraint matrix."""
    cols = np.repeat(np.arange(n), n)
    rows = (np.tile(np.arange(n), n) + np.repeat(np.arange(n), n))
    return rows.astype(np.int32), cols.astype(np.int32)


def solve_lp(f, M, rows, cols, n, timeout=8):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                               options={'disp': False, 'time_limit': timeout})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_take_step(p, M, rows, cols, n, timeout=6):
    """Attempt one LP-guided step. Returns (p_new, M_new) or (None, M)."""
    g = solve_lp(p, M, rows, cols, n, timeout=timeout)
    if g is None:
        return None, M
    g_sum = g.sum()
    if g_sum < 1e-12:
        return None, M
    g /= g_sum

    for t in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1]:
        p_new = (1.0 - t) * p + t * g
        p_new = np.maximum(p_new, 0.0)
        s = p_new.sum()
        if s < 1e-12:
            continue
        p_new /= s
        conv_new = fast_convolve(p_new)
        M_new = conv_new.max()
        if M_new < M - 1e-15:
            return p_new, M_new
    return None, M


def gradient_step(p, conv, T, eta):
    """Single gradient step. Returns (p_new, conv_new, M_new)."""
    n = len(p)
    c_scaled = conv / T
    c_scaled -= c_scaled.max()
    w = np.exp(c_scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum

    grad = compute_grad_fft(p, w)
    grad -= grad.mean()

    p_new = p * np.exp(-eta * grad)
    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s < 1e-300:
        p_new = p.copy()
    else:
        p_new /= s

    conv_new = fast_convolve(p_new)
    M_new = conv_new.max()
    return p_new, conv_new, M_new


def gradient_optimize(p, time_limit, T_init, eta_init, max_iters=None):
    """Adaptive gradient optimization. Returns (best_p, best_M)."""
    n = len(p)
    p = p.copy()
    conv = fast_convolve(p)
    M = conv.max()
    best_M = M
    best_p = p.copy()

    T = T_init
    eta = eta_init
    patience = 0
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        if max_iters is not None and iters >= max_iters:
            break
        iters += 1

        improved = False
        for eta_try in [eta, eta * 0.6, eta * 0.3, eta * 0.1]:
            p_new, conv_new, M_new = gradient_step(p, conv, T, eta_try)
            if M_new < best_M - 1e-14:
                best_M = M_new
                best_p = p_new.copy()
                p = p_new
                conv = conv_new
                M = M_new
                patience = 0
                T = max(T * 0.997, T_init * 0.1)
                eta = min(eta * 1.005, 5.0)
                improved = True
                break

        if not improved:
            patience += 1
            eta *= 0.92
            T *= 0.99
            if patience > 80:
                break

    return best_p, best_M


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry: hybrid LP + gradient optimization for n=512."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 20.0)

    best_score = float('inf')
    best_seq = None
    n = 512

    # Get parent sequence
    parent = None
    try:
        p_arr = np.asarray(height_sequence_1, dtype=np.float64)
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        if p_arr.ndim == 1 and p_arr.size > 0 and np.isfinite(p_arr).all() and p_arr.sum() > 1e-10:
            parent = p_arr / p_arr.sum()
    except Exception:
        pass

    # Build initial candidate pool
    candidates = []
    if parent is not None:
        if parent.size == n:
            candidates.append(('parent', parent.copy()))
        else:
            p_interp = np.interp(np.linspace(0, 1, n),
                                 np.linspace(0, 1, parent.size), parent)
            p_interp = np.maximum(p_interp, 0.0)
            if p_interp.sum() > 1e-10:
                candidates.append(('parent_interp', p_interp / p_interp.sum()))

    # Add diverse initializations
    x = np.arange(n, dtype=np.float64)
    candidates.append(('uniform', np.ones(n) / n))

    for sigma in [0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sigma * n)) ** 2)
        g /= g.sum()
        candidates.append((f'gauss_{sigma}', g))

    # Bimodal (mass at edges)
    bim = np.zeros(n)
    w = max(1, n // 5)
    bim[:w] = 1.0
    bim[-w:] = 1.0
    candidates.append(('bimodal', bim / bim.sum()))

    # Cosine
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    candidates.append(('cosine', cb / cb.sum()))

    # Random Dirichlet
    for conc in [0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * conc)
        candidates.append((f'dir_{conc}', r))

    # Perturbed parent
    if parent is not None:
        if parent.size == n:
            for noise_level in [0.02, 0.05, 0.1]:
                pp = parent * (1 + noise_level * rng.standard_normal(n))
                pp = np.maximum(pp, 0)
                if pp.sum() > 1e-10:
                    candidates.append((f'pert_{noise_level}', pp / pp.sum()))
        else:
            p_interp = np.interp(np.linspace(0, 1, n),
                                 np.linspace(0, 1, parent.size), parent)
            p_interp = np.maximum(p_interp, 0)
            pp = p_interp * (1 + 0.05 * rng.standard_normal(n))
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                candidates.append(('pert_interp', pp / pp.sum()))

    # LP structure
    rows, cols = make_lp_structure(n)

    # Phase 0: Quick scan of candidates (30s each, top ones)
    print(f"Phase 0: Quick scan of {len(candidates)} candidates")
    scan_results = []
    for name, p0 in candidates:
        if time.monotonic() > deadline - 400:
            break
        tr = min(25.0, deadline - time.monotonic() - 400)
        if tr < 5:
            break
        p_scan, M_scan = gradient_optimize(p0, time_limit=tr, T_init=0.001, eta_init=0.5)
        obj = 2.0 * n * M_scan
        score = float(evaluate_sequence(p_scan.tolist()))
        scan_results.append((score, obj, name, p_scan))
        print(f"  {name:15s}: score={score:.6f}, M={M_scan:.8f}")

    scan_results.sort(key=lambda x: x[0])

    # Pick the best starting point
    if scan_results:
        best_start = scan_results[0][3].copy()
        best_obj = 2.0 * n * fast_convolve(best_start).max()
        best_score = scan_results[0][0]
        best_seq = best_start.copy()
    else:
        best_start = np.ones(n) / n
        best_obj = 2.0 * n / n
        best_seq = best_start.copy()

    print(f"Phase 0 done. Best: {best_score:.6f}")

    # Phase 1: Main LP + gradient hybrid loop
    print("Phase 1: LP + gradient hybrid")
    p = best_start.copy()
    conv = fast_convolve(p)
    M = conv.max()
    best_p = best_start.copy()
    best_M = M

    lp_count = 0
    total_iter = 0
    stuck_counter = 0
    last_print = 0

    while time.monotonic() < deadline - 250:
        total_iter += 1

        # Every ~10-20 iterations, try an LP step
        if total_iter % 12 == 1:
            p_new, M_new = lp_take_step(p, M, rows, cols, n, timeout=6)
            if p_new is not None:
                p = p_new
                M = M_new
                conv = fast_convolve(p)
                lp_count += 1
                stuck_counter = 0
                if M < best_M:
                    best_M = M
                    best_p = p.copy()
                    score = float(evaluate_sequence(p.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_seq = p.copy()
                        print(f"  LP[{lp_count}]: score={score:.6f}, M={M:.8f}, t={time.monotonic()-deadline+budget_s:.0f}s")
            else:
                stuck_counter += 1
        else:
            # Gradient step
            T = max(M * 0.03, 1e-7)
            eta_try = 0.4
            improved = False
            for e in [eta_try, eta_try * 0.5, eta_try * 0.25]:
                p_g, conv_g, M_g = gradient_step(p, conv, T, e)
                if M_g < best_M - 1e-14:
                    p = p_g
                    conv = conv_g
                    M = M_g
                    best_M = M_g
                    best_p = p.copy()
                    stuck_counter = 0
                    score = float(evaluate_sequence(p.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_seq = p.copy()
                        if total_iter - last_print > 200:
                            print(f"  Grad: score={score:.6f}, M={M:.8f}, iter={total_iter}")
                            last_print = total_iter
                    improved = True
                    break

            if not improved:
                stuck_counter += 1

        # Perturbation when stuck
        if stuck_counter > 60:
            print(f"  Stuck (counter={stuck_counter}), perturbing... best_M={best_M:.8f}")
            perturbation = rng.standard_normal(n) * 0.03
            p = best_p * (1.0 + perturbation)
            p = np.maximum(p, 0.0)
            s = p.sum()
            if s < 1e-10:
                p = np.ones(n) / n
            else:
                p /= s
            conv = fast_convolve(p)
            M = conv.max()
            stuck_counter = 0

    # Phase 2: Intensive gradient polishing
    print(f"Phase 2: Intensive polishing (best_score={best_score:.6f})")
    p = best_seq.copy()
    p /= p.sum()
    conv = fast_convolve(p)
    M = conv.max()

    remaining = deadline - time.monotonic() - 80
    if remaining > 30:
        # Multiple temperature levels
        for frac in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08]:
            tr = min(40.0, remaining / 2.0, deadline - time.monotonic() - 80)
            if tr < 5:
                break
            T_init = M * frac
            p2, M2 = gradient_optimize(p, time_limit=tr, T_init=T_init, eta_init=0.6)
            if M2 < M:
                M = M2
                p = p2
                score = float(evaluate_sequence(p.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = p.copy()
                    print(f"  Polish T_frac={frac}: score={score:.6f}")
            remaining = deadline - time.monotonic() - 80

    # Phase 3: Final LP kicks
    print("Phase 3: Final LP kicks")
    if time.monotonic() < deadline - 15:
        p = best_seq.copy()
        p /= p.sum()
        conv = fast_convolve(p)
        M = conv.max()

        k = 0
        while time.monotonic() < deadline - 10:
            p_new, M_new = lp_take_step(p, M, rows, cols, n, timeout=5)
            if p_new is not None:
                p = p_new
                M = M_new
                k += 1
                score = float(evaluate_sequence(p.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = p.copy()
                    print(f"  Final LP[{k}]: score={score:.6f}")
            else:
                # Try a gradient nudge then LP again
                T = M * 0.02
                p_g, conv_g, M_g = gradient_step(p, conv, T, 0.3)
                if M_g < M:
                    p = p_g
                    M = M_g
                    conv = conv_g
                else:
                    break

    if best_seq is None:
        best_seq = np.ones(128)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}")
    return [float(x) for x in best_seq]
