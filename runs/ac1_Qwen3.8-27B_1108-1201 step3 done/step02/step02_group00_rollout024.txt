    """Linear auto-convolution via FFT."""
    n = len(a)

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)

    rho = np.nonzero(cond)[0][-1]

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: (1) Nesterov-accelerated gradient descent with softmax-smoothed max,
(2) LP-based best-response alternating method, (3) multi-start with diverse
initializations and length exploration. Uses FFT for fast convolution and
gradient computation.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_correlation_ftt(p, a, n):
    """Compute q[m] = sum_k p[k] * a[k-m] for m=0..n-1 (a[i]=0 outside [0,n))."""
    N = 1 << (2 * n - 2).bit_length()
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(np.conj(P) * A, N)
    return corr[:n]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def compute_smoothed_obj_and_grad(a, temp):
    """Compute smoothed objective and gradient using softmax smoothing of max."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / temp
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    smoothed_obj = temp * (np.log(p_sum) + (conv.max() / temp))
    # Actually: smoothed_obj = temp * log(sum exp(c_k/temp)) = temp * (log(p_sum) + max_c/temp)
    # = temp * log(p_sum) + max_c
    # Let me just compute it directly:
    smoothed_obj = conv.max() + temp * np.log(p_sum)
    
    grad = 2.0 * cross_correlation_ftt(p, a, n)
    return smoothed_obj, grad, conv


def nesterov_optimize(a0, time_limit, lr_init=0.01, momentum=0.9,
                      temp_init=0.001, temp_decay=1.003,
                      perturb_every=500, perturb_scale=0.003,
                      use_bb=True, verbose=False):
    """
    Nesterov-accelerated projected gradient descent with softmax-smoothed max.
    Optimizes over simplex {a >= 0, sum(a) = 1}.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = compute_obj(a)
    best_a = a.copy()
    
    v = np.zeros(n)  # momentum velocity
    y = a.copy()  # Nesterov lookahead
    t_start = time.monotonic()
    lr = lr_init
    temp = temp_init
    step_count = 0
    no_improve = 0
    prev_grad = None
    prev_a = None
    bb_alpha = lr_init
    
    while time.monotonic() - t_start < time_limit:
        # Nesterov lookahead
        y = a + momentum * v
        
        # Compute gradient at y
        conv_y = fast_convolve(y)
        scaled = conv_y / temp
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(conv_y))
            p[np.argmax(conv_y)] = 1.0
        else:
            p /= p_sum
        grad = 2.0 * cross_correlation_ftt(p, y, n)
        
        # Barzilai-Borwein step size update
        if use_bb and prev_grad is not None and prev_a is not None:
            s_vec = a - prev_a
            y_vec = grad - prev_grad
            sy = np.dot(s_vec, y_vec)
            yy = np.dot(y_vec, y_vec)
            ss = np.dot(s_vec, s_vec)
            if yy > 1e-15 and sy > 1e-15:
                bb_alpha = ss / sy
                bb_alpha = min(max(bb_alpha, 0.001), 0.5)
        
        # Update
        alpha = lr * (bb_alpha / lr_init if use_bb else 1.0)
        v = momentum * v - alpha * grad
        a_new = project_simplex(a + v)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv_new.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        prev_a = a.copy()
        prev_grad = grad.copy()
        a = a_new
        conv = conv_new
        step_count += 1
        
        # Periodic perturbation
        if no_improve >= perturb_every:
            rng = np.random.default_rng(step_count)
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            v *= 0.3
            no_improve = 0
            perturb_every = int(perturb_every * 1.3)
            perturb_scale *= 0.8
        
        # Schedule
        temp *= temp_decay
        lr = max(0.0005, lr * 0.9999)
        
        if verbose and step_count % 10000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    Nesterov: step={step_count}, obj={best_obj:.6f}, "
                  f"temp={temp:.6f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def lp_best_response(a_fix, max_time=0.5):
    """
    Solve LP: minimize t s.t. conv(a_fix, b)[k] <= t for all k,
    b >= 0, sum(b) = 1.
    Returns optimal b and t.
    """
    n = len(a_fix)
    num_k = 2 * n - 1
    
    # Variables: b_0, ..., b_{n-1}, t (total n+1)
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: conv(a_fix, b)[k] - t <= 0
    A_ub = np.zeros((num_k, n + 1))
    b_ub = np.zeros(num_k)
    
    for k in range(num_k):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = a_fix[idx]
        A_ub[k, -1] = -1.0
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n], result.x[-1]
    return None, None


def lp_alternating_optimize(a0, time_limit, step_size=0.3, max_lp_iter=5000):
    """
    LP-based alternating method:
    1. Given a, solve LP to find b minimizing max_k conv(a, b)[k]
    2. Update a <- (1-t)*a + t*b
    3. Track best self-convolution peak
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t_start = time.monotonic()
    iter_count = 0
    
    while time.monotonic() - t_start < time_limit and iter_count < max_lp_iter:
        b, t_lp = lp_best_response(a)
        if b is None:
            break
        
        # Update
        a_new = (1.0 - step_size) * a + step_size * b
        a_new = project_simplex(a_new)
        
        conv_new = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv_new.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
        
        a = a_new
        conv = conv_new
        iter_count += 1
        
        # Adapt step size: reduce if oscillating
        if iter_count % 100 == 0 and cur_obj > best_obj + 1e-8:
            step_size *= 0.8
    
    return best_a, best_obj


def mirror_descent_optimize(a0, time_limit, t_param=50.0, eta_init=0.2,
                            top_k_frac=0.1):
    """
    Multiplicative weights / mirror descent on the simplex.
    Uses softmax over top-k convolution peaks for the subgradient.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t_start = time.monotonic()
    eta = eta_init
    t_cur = t_param
    no_improve = 0
    
    while time.monotonic() - t_start < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        
        # Softmax weights over top peaks
        top_k = max(5, int(len(conv) * top_k_frac))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        
        scaled = t_cur * peak_vals / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(top_k)
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(top_k):
            k = int(peaks[i])
            wi = w[i] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += wi * a[k - idx]
        
        # Center the subgradient (for mirror descent on simplex)
        sg_centered = sg - sg.mean()
        
        # Multiplicative update
        log_update = -eta * sg_centered
        log_update -= log_update.max()
        a = a * np.exp(log_update)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 100:
            no_improve = 0
            eta *= 0.7
            t_cur *= 1.5
            if eta < 0.01:
                # Reset with perturbation
                rng = np.random.default_rng()
                a = project_simplex(a + rng.standard_normal(n) * 0.01)
                eta = eta_init * 0.5
                t_cur = t_param
    
    return best_a, best_obj


def generate_initializations(n, parent=None, rng=None):
    """Generate diverse initializations of length n, all on the simplex."""
    if rng is None:
        rng = np.random.default_rng()
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    rc = np.maximum(rc, 1e-10)
    rc /= rc.sum()
    inits['raised_cosine'] = rc
    
    # Gaussian (centered)
    for sf in [0.1, 0.2, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.1f}'] = g
    
    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(2, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    inits['ramp_rev'] = ramp[::-1].copy()
    
    # Truncated sinc (rectangular spectrum)
    sinc = np.sinc(x / (n * 0.3) - 0.5)
    sinc = np.abs(sinc)
    sinc = np.maximum(sinc, 1e-10)
    sinc /= sinc.sum()
    inits['sinc'] = sinc
    
    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Different random
    inits['random2'] = rng.dirichlet(np.ones(n) * 0.1)
    
    # Parent sequence
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps
    
    return inits


def try_optimization(a0, n, time_budget, methods=['nesterov', 'lp', 'mirror']):
    """Try multiple optimization methods on a0, return best result."""
    best_a = a0.copy()
    best_obj = compute_obj(a0)
    
    elapsed = time.monotonic() - (-1)  # placeholder
    t0 = time.monotonic()
    
    remaining = time_budget
    time_per_method = remaining / len(methods)
    
    for method in methods:
        if time.monotonic() - t0 > time_budget - 1:
            break
        t_alloc = min(time_per_method, time_budget - (time.monotonic() - t0) - 0.5)
        if t_alloc < 1:
            break
        
        if method == 'nesterov':
            a_opt, obj_opt = nesterov_optimize(
                best_a, time_limit=t_alloc, lr_init=0.01, momentum=0.9,
                temp_init=0.001, temp_decay=1.003,
                perturb_every=500, perturb_scale=0.003)
        elif method == 'lp':
            a_opt, obj_opt = lp_alternating_optimize(
                best_a, time_limit=t_alloc, step_size=0.3)
        elif method == 'mirror':
            a_opt, obj_opt = mirror_descent_optimize(
                best_a, time_limit=t_alloc, t_param=50.0, eta_init=0.2)
        else:
            continue
