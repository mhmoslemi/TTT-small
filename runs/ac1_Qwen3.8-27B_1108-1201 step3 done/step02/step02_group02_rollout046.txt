import numpy as np
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def compute_subgradient(a, top_k=1, peak_frac=0.0):
    """Compute subgradient of max_k (a*a)_k w.r.t. a."""
    n = len(a)
    conv = fft_convolve(a, a)
    max_val = np.max(conv)
    
    if peak_frac > 0:
        threshold = max_val * (1.0 - peak_frac)
        peak_indices = np.where(conv >= threshold)[0]
    else:
        peak_indices = np.argsort(conv)[-top_k:]
    
    if len(peak_indices) == 0:
        return np.zeros(n), conv
    
    grad = np.zeros(n)
    for k in peak_indices:
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        indices = np.arange(i_start, i_end + 1)
        j = k - indices
        grad[indices] += a[j] + a[indices]
    
    grad /= len(peak_indices)
    return grad, conv

def subgradient_optimize(a0, time_limit, rng, top_k=5, verbose=True):
    """Subgradient optimization with backtracking line search."""
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    S = np.sqrt(2.0 * len(a))
    sum_a = np.sum(a)
    if sum_a < 0.01:
        a = np.ones(len(a)) * S / len(a)
    else:
        a = a * (S / sum_a)
    
    current_score = local_score(a)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    eta = 0.05
    stuck = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        grad, conv = compute_subgradient(a, top_k=top_k)
        
        # Backtracking line search
        improved = False
        for trial in range(10):
            eta_try = eta * (0.5 ** trial)
            a_new = a - eta_try * grad
            a_new = np.clip(a_new, 0, None)
            s = np.sum(a_new)
            if s < 0.01:
                continue
            a_new = a_new * (S / s)
            new_score = local_score(a_new)
            if new_score < current_score:
                a = a_new
                current_score = new_score
                eta = eta_try * 1.1
                eta = min(eta, 0.2)
                improved = True
                stuck = 0
                break
            else:
                # Still accept if it's close (for exploration)
                if trial == 0 and new_score < current_score * 1.001:
                    a = a_new
                    current_score = new_score
                    stuck += 1
                    improved = True
                    break
        
        if not improved:
            stuck += 1
            eta *= 0.8
            if stuck > 50:
                # Perturb
                a = a * np.exp(rng.normal(0, 0.05, len(a)))
                a = np.clip(a, 0, None)
                s = np.sum(a)
                if s > 0.01:
                    a = a * (S / s)
                    current_score = local_score(a)
                stuck = 0
                eta = 0.05
        
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
        
        iter_count += 1
        if verbose and iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter={iter_count}, score={current_score:.6f}, best={best_score:.6f}, eta={eta:.4f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score, current_score

def coordinate_descent(a0, time_limit, rng, n_sweeps=None, verbose=True):
    """Coordinate descent refinement."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 0.01:
        a = np.ones(n) * S / n
    else:
        a = a * (S / s)
    
    current_score = local_score(a)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    sweep = 0
    
    while time.monotonic() - t0 < time_limit:
        # Randomize coordinate order
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Golden section search on a[i]
            old_val = a[i]
            lo = 0.0
            hi = max(2.0 * old_val, S / n * 3)
            
            # Golden section search
            gr = (np.sqrt(5) - 1) / 2
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            
            a[i] = c
            fc = local_score(a)
            a[i] = d
            fd = local_score(a)
            
            for _ in range(20):  # 20 iterations of golden section
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    a[i] = c
                    fc = local_score(a)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    a[i] = d
                    fd = local_score(a)
            
            # Set to the best
            a[i] = c if fc < fd else d
            best_val = min(fc, fd)
            
            if best_val < current_score:
                current_score = best_val
                if current_score < best_score:
                    best_score = current_score
                    best_a = a.copy()
            else:
                # Revert
                a[i] = old_val
                current_score = local_score(a)
        
        sweep += 1
        # Renormalize after each sweep
        s = np.sum(a)
        if s > 0.01:
            a = a * (S / s)
            current_score = local_score(a)
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    sweep={sweep}, score={current_score:.6f}, best={best_score:.6f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score

def simulate_annealing(a0, time_limit, rng, verbose=True):
    """Simulated annealing for final polish."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = np.clip(a, 0, None)
    s = np.sum(a)
    a = a * (S / s)
    
    current_score = local_score(a)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    T = 0.002
    cooling = 0.9995
    min_T = 1e-6
    
    while time.monotonic() - t0 < time_limit and T > min_T:
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.4:
            # Peak-targeted modification
            conv = fft_convolve(candidate, candidate)
            peak_idx = int(np.argmax(conv))
            # Find contributing pairs
            i_start = max(0, peak_idx - n + 1)
            i_end = min(peak_idx, n - 1)
            indices = np.arange(i_start, i_end + 1)
            contribs = candidate[indices] * candidate[peak_idx - indices]
            top_k = max(3, n // 8)
            top_idx = indices[np.argsort(contribs)[-top_k:]]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=len(top_idx)))
            # Increase others
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 3:
                inc_n = min(10, len(other) // 3)
                inc_idx = rng.choice(other, size=inc_n, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=inc_n))
        elif r < 0.6:
            # Block modification
            width = max(1, int(rng.integers(1, max(2, n // 4))))
            start = int(rng.integers(0, n - width))
            candidate[start:start+width] *= np.exp(rng.normal(0, 0.02, size=width))
        elif r < 0.8:
            # Pair swap
            i, j = rng.choice(n, 2, replace=False)
            delta = rng.uniform(-0.1, 0.1) * max(a[i], a[j])
            candidate[i] = max(0, candidate[i] - delta)
            candidate[j] = max(0, candidate[j] + delta)
        elif r < 0.9:
            # Smooth local modification
            i = int(rng.integers(0, n))
            w = max(1, int(n * 0.1))
            start = max(0, i - w // 2)
            end = min(n, i + w // 2 + 1)
            candidate[start:end] *= np.exp(rng.normal(0, 0.01, size=end-start))
        else:
            # Global small
            candidate *= np.exp(rng.normal(0, 0.005, n))
        
        candidate = np.clip(candidate, 0, 1000)
        cs = np.sum(candidate)
        if cs < 0.01:
            continue
        candidate = candidate * (S / cs)
        
        new_score = local_score(candidate)
        
        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        else:
            accept_prob = np.exp(-(new_score - current_score) / max(T, min_T))
            if rng.random() < accept_prob:
                a = candidate
                current_score = new_score
        
        T *= cooling
    
    return best_a, best_score

"""
Smoothed-max subgradient optimization for minimizing peak autocorrelation
of non-negative step functions. Uses O(n^2) per-iteration gradient computed
via sliding window, with backtracking line search, Nesterov momentum,
coordinate descent refinement, and simulated annealing polish.

import numpy as np
import time


def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def smooth_gradient(a, beta=20.0):
    """Compute gradient of smoothed max of autocorrelation O(n^2)."""
    n = len(a)
    conv = fft_convolve(a, a)
    M = np.max(conv)
    if M < 1e-15:
        return np.zeros(n), conv, M
    
    shifted = (conv - M) / M
    weights = np.exp(beta * shifted)
    w_sum = np.sum(weights)
    
    # h_first[i] = sum_j a[j] * w[j+i]
    W = np.lib.stride_tricks.sliding_window_view(weights, n)
    h_first = a @ W
    
    # h_second[i] = a[i] * sum_{k=i}^{i+n-1} w[k]
    window_sum = np.convolve(weights, np.ones(n), mode='valid')
    h_second = a * window_sum
    
    grad = (h_first + h_second) / w_sum * (M / beta)
    return grad, conv, M


def hard_subgradient(a, top_k=5):
    """Compute subgradient using top-k peaks."""
    n = len(a)
    conv = fft_convolve(a, a)
    M = np.max(conv)
    peak_indices = np.argsort(conv)[-top_k:]
    
    grad = np.zeros(n)
    for k in peak_indices:
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        indices = np.arange(i_start, i_end + 1)
        j = k - indices
        grad[indices] += a[j] + a[indices]
    
    grad /= len(peak_indices)
    return grad, conv, M


def subgradient_optimize(a0, time_limit, rng, beta=20.0, use_smooth=True, momentum=0.9, verbose=True):

    a = a0.astype(np.float64).copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 0.01:

    else:
        a = a * (S / s)
    
    current_score = local_score(a)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    
    eta = 0.1  # Initial step size
    vel = np.zeros(n)  # Momentum velocity
    stuck = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        if use_smooth:
            grad, conv, M = smooth_gradient(a, beta=beta)
        else:
            grad, conv, M = hard_subgradient(a, top_k=5)
        
        # Nesterov-like momentum
        vel = momentum * vel - eta * grad
        a_try = a + vel
        
        # Project and normalize
        a_try = np.clip(a_try, 0, None)
        s_try = np.sum(a_try)
        if s_try < 0.01:
            a_try = a.copy()
            s_try = np.sum(a_try)
        a_try = a_try * (S / s_try)
        
        # Line search (backtracking)
        improved = False
        for backtrack in range(8):
            new_score = local_score(a_try)
            if new_score < current_score * (1.0 - 1e-8):
                a = a_try
                current_score = new_score
                eta = min(eta * 1.01, 0.5)
                improved = True
                stuck = 0
                break
            else:
                eta *= 0.7
                vel = momentum * vel - eta * grad
                a_try = np.clip(a + vel, 0, None)
                s_try = np.sum(a_try)
                if s_try < 0.01:
                    a_try = a.copy()
                    s_try = S
                a_try = a_try * (S / s_try)
        
        if not improved:
            stuck += 1
            if stuck > 30:
                # Perturb to escape local min
                scale = 1.0 + 0.03 * stuck / 30
                a = a * np.exp(rng.normal(0, 0.05 * scale, n))
                a = np.clip(a, 0, None)
                s = np.sum(a)
                if s > 0.01:
                    a = a * (S / s)
                    current_score = local_score(a)
                vel = np.zeros(n)
                stuck = 0
                eta = 0.05
            elif stuck > 10:
                # Try switching between smooth and hard
                use_smooth = not use_smooth
                vel = np.zeros(n)
                eta = 0.05
        
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
        
        iter_count += 1
        if verbose and iter_count % 10000 == 0:

def coordinate_descent(a0, time_limit, rng, verbose=True):
    """Coordinate descent with golden section search."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s > 0.01:
        a = a * (S / s)
    
    current_score = local_score(a)
    best_a = a.copy()
    best_score = current_score
    t0 = time.monotonic()
    sweep = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        order = rng.permutation(n)
        sweep_improved = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[i]
            lo = 0.0
            hi = max(3.0 * old_val, S / n * 2)
            
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            
            a[i] = c
            fc = local_score(a)
            a[i] = d
            fd = local_score(a)
            
            for _ in range(15):
                if fc <= fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    a[i] = c
                    fc = local_score(a)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    a[i] = d
                    fd = local_score(a)
            
            best_c = c if fc <= fd else d
            best_f = min(fc, fd)
            a[i] = best_c
            
            if best_f < current_score:
                current_score = best_f
                sweep_improved = True
                if current_score < best_score:
                    best_score = current_score
                    best_a = a.copy()
            else:
                a[i] = old_val
        
        # Renormalize
        s = np.sum(a)
        if s > 0.01:
            a = a * (S / s)
            current_score = local_score(a)
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        
        sweep += 1
        if sweep_improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break  # Converged
        
        if verbose and sweep % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"      sweep={sweep}, score={current_score:.6f}, best={best_score:.6f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score
