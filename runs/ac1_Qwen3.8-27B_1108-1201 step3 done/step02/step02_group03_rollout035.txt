import numpy as np
import time
from scipy.optimize import linprog

def compute_obj(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (S * S))

def normalize_sum(a, target=1.0):
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)

def softmax_gradient(a, tau):
    """Compute gradient of soft-max autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    # grad[i] = 2 * sum_j w[i+j] * a[j] for j=0..n-1
    # = 2 * (valid cross-correlation of w and a)
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n-1:n-1+n]
    return grad

def hard_gradient(a):
    """Compute gradient of hard-max autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad

class AdamOptimizer:
    def __init__(self, dim, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        self.m = np.zeros(dim)
        self.v = np.zeros(dim)
        self.t = 0
        self.lr = lr
        self.beta1 = beta1
        self.beta2 = beta2
        self.eps = eps
    
    def step(self, grad):
        self.t += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * grad ** 2
        m_hat = self.m / (1 - self.beta1 ** self.t)
        v_hat = self.v / (1 - self.beta2 ** self.t)
        return self.lr * m_hat / (np.sqrt(v_hat) + self.eps)

def project(a, target_sum=1.0):
    """Project onto non-negative simplex with given sum."""
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones_like(a) / len(a)
        return a
    return a * (target_sum / S)

def opt_adam(a0, deadline, num_leap_seconds=5):
    """Adam optimization with multi-peak tracking."""
    n = len(a0)
    a = project(a0.copy(), 1.0)
    S_target = 1.0
    score = compute_obj(a)
    best_a = a.copy()
    best_score = score
    
    adam = AdamOptimizer(n, lr=0.05)
    tau = 10.0
    no_improve = 0
    
    while time.monotonic() < deadline:
        # Compute gradient
        grad = softmax_gradient(a, tau)
        
        # Project out the mean direction (to maintain sum constraint)
        grad -= grad.mean()
        
        # Adam step
        step = adam.step(grad)
        a_new = a + step
        a_new = project(a_new, S_target)
        
        # Check improvement
        new_score = compute_obj(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Anneal tau (increase to make gradient sharper)
        tau = min(tau * 1.001, 100.0)
        
        # Reset Adam if stuck
        if no_improve > 200:
            adam = AdamOptimizer(n, lr=0.05 * (0.5 ** (no_improve // 200)))
            no_improve = 0
            # Add small noise
            a = a + np.random.normal(0, 0.01, n)
            a = project(a, S_target)
            if compute_obj(a) > best_score * 1.1:
                a = best_a.copy()
        
        if no_improve > 1000:
            break
    
    return best_a, best_score

        self.b2 = b2
        self.eps = eps

def compute_obj(a):
    """Compute 2*n*max(convolve(a,a))/(sum(a))^2."""
    n = len(a)

def normalize_sum(a, target=1.0):
    """Scale a so that sum(a) = target."""
    S = a.sum()
    if S < 1e-15:
        return np.full_like(a, target / len(a))
    return a * (target / S)

    """Project onto {x >= 0, sum(x) = target_sum}."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:

        grad += conv_res[n - 1:n - 1 + n]
    grad *= 2.0 / len(taus)
    return grad


def hard_grad(a):
    """Gradient of hard max autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


class Adam:
    """Adam optimizer."""

    def __init__(self, dim, lr=0.005, b1=0.9, b2=0.999, eps=1e-8):
        self.m = np.zeros(dim)
        self.v = np.zeros(dim)
        self.t = 0
        self.lr = lr
        self.b1 = b1
        self.b2 = b2
        self.eps = eps

    def step(self, grad):
        self.t += 1
        self.m = self.b1 * self.m + (1.0 - self.b1) * grad
        self.v = self.b2 * self.v + (1.0 - self.b2) * grad * grad
        mh = self.m / (1.0 - self.b1 ** self.t)
        vh = self.v / (1.0 - self.b2 ** self.t)
        return self.lr * mh / (np.sqrt(vh) + self.eps)


def lp_solve(a, M):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    if n > 512:
        return None

    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        m_lo = j
        m_hi = min(j + n, 2 * n - 1)
        for m in range(m_lo, m_hi):
            A[m, j] = a[m - j]

    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def adam_optimize(a0, deadline, tau_start=5.0, tau_end=60.0, lr=0.005,
                  use_multi_tau=False, reheat_interval=200):
    """Adam optimization with annealed softmax gradient."""
    n = len(a0)
    a = project_nn_sum(a0.copy(), 1.0)
    score = compute_obj(a)
    best_a = a.copy()
    best_score = score

    adam = Adam(n, lr=lr)
    no_improve = 0
    t0 = time.monotonic()
    total_time = max(1.0, deadline - t0)

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        frac = min(1.0, (time.monotonic() - t0) / total_time)
        tau = tau_start + (tau_end - tau_start) * frac

        if use_multi_tau:
            taus = [tau * 0.5, tau, tau * 2.0]
            grad = multi_tau_grad(a, taus)
        else:
            grad = softmax_grad(a, tau)

        grad -= grad.mean()

        step = adam.step(grad)
        a_new = a + step
        a_new = project_nn_sum(a_new, 1.0)

        new_score = compute_obj(a_new)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Reheat: add noise and reset optimizer when stuck
        if no_improve > reheat_interval:
            noise_scale = 0.02 * (1.0 + no_improve / reheat_interval)
            rng_local = np.random.default_rng(iteration)
            a = best_a + rng_local.normal(0, noise_scale, n)
            a = project_nn_sum(a, 1.0)
            adam = Adam(n, lr=lr * 0.7)
            no_improve = 0
            if compute_obj(a) > best_score * 1.08:
                a = best_a.copy()

        if no_improve > reheat_interval * 5:
            break

    return best_a, best_score


def lp_refine(a, deadline, max_iters=40):
    """LP-based refinement: find g with low cross-correlation with a."""
    n = len(a)
    if n > 512:
        return a, compute_obj(a)

    a = project_nn_sum(a.copy(), 1.0)
    score = compute_obj(a)
    best_a = a.copy()
    best_score = score

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = np.convolve(a, a)
        M = float(c.max())

        g = lp_solve(a, M)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-6:
            break

        g_norm = g * (1.0 / g_sum)

        improved = False
        for t_blend in [0.5, 0.35, 0.2, 0.1, 0.05]:
            a_new = (1.0 - t_blend) * a + t_blend * g_norm
            a_new = project_nn_sum(a_new, 1.0)
            s_new = compute_obj(a_new)
            if s_new < best_score - 1e-12:
                a = a_new
                best_a = a.copy()
                best_score = s_new
                improved = True
                break

        if not improved:
            rng_local = np.random.default_rng(1000 + it)
            a = a + rng_local.normal(0, 0.02, n)
            a = project_nn_sum(a, 1.0)
            s = compute_obj(a)
            if s < best_score:
                best_a = a.copy()
                best_score = s

    return best_a, best_score


def coord_descent_polish(a, deadline, max_rounds=4):
    """Coordinate descent: try reducing peak-contributing coordinates."""
    n = len(a)
    a = project_nn_sum(a.copy(), 1.0)
    score = compute_obj(a)
    best_a = a.copy()
    best_score = score

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = np.convolve(a, a)
        # Get top peaks
        top_k = np.argsort(c)[-3:]
        peak_lags = [int(k) for k in top_k if k < n]

        if not peak_lags:
            peak_lags = [int(np.argmax(c))]

        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break

            orig = a[i]
            if orig < 1e-10:
                continue

            for k_star in peak_lags:
                j = k_star - i
                if 0 <= j < n and a[j] > 1e-8:
                    break

            best_val = orig
            best_s = best_score
            for factor in [0.0, 0.3, 0.5, 0.7]:
                a[i] = orig * factor
                a_norm = project_nn_sum(a, 1.0)
                s = compute_obj(a_norm)
                if s < best_s - 1e-12:
                    best_s = s
                    best_val = orig * factor
                    a = a_norm.copy()

            a[i] = best_val
            a = project_nn_sum(a, 1.0)
            if best_val != orig:
                best_score = best_s
                best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_score


def generate_starts(n, rng):
    """Generate diverse initial sequences of length n."""
    starts = []

    # From previous best
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size >= n and np.all(np.isfinite(prev)):
            starts.append(prev[:n].copy())
            starts.append(prev[:n][::-1].copy())
    except (NameError, UnboundLocalError, Exception):
        pass

    # Random uniform
    starts.append(rng.random(n) * 0.5 + 0.25)

    # Random exponential
    starts.append(rng.exponential(1.0, n))

    # Structured: chirp
    i = np.arange(n, dtype=np.float64)
    starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i * 3.0 / n))
    starts.append(0.5 + 0.5 * np.sin(2 * np.pi * i * 5.0 / n))
    starts.append(np.abs(np.sin(2 * np.pi * i * 2.0 / n)) + 0.05)

    # Two bumps
    starts.append(
        np.exp(-0.5 * ((i - n * 0.25) / (n * 0.12)) ** 2) +
        np.exp(-0.5 * ((i - n * 0.75) / (n * 0.12)) ** 2)
    )

    # Sparse pattern
    a_sparse = np.zeros(n)
    a_sparse[::4] = rng.random(n // 4 + 1) * 0.6 + 0.1
    starts.append(a_sparse)

    return starts


def optimize_single(a0, deadline, n):
    """Full optimization pipeline from a single starting point."""
    t = time.monotonic()
    remaining = deadline - t
    if remaining < 15:
        a = project_nn_sum(a0.copy(), 1.0)
        return a, compute_obj(a)

    # Phase A: Adam with annealed tau (60% of time)
    adam_deadline = t + remaining * 0.6
    a, score = adam_optimize(a0, adam_deadline, tau_start=5.0, tau_end=50.0,
                             lr=0.005, use_multi_tau=True, reheat_interval=150)
    best_a, best_score = a, score

    # Phase B: LP refinement (25% of time, only for n <= 512)
    if n <= 512:
        lp_deadline = t + remaining * 0.85
        a_lp, score_lp = lp_refine(best_a, lp_deadline, max_iters=25)
        if score_lp < best_score:
            best_a, best_score = a_lp, score_lp

    # Phase C: Adam again with sharper gradient (10% of time)
    adam2_deadline = t + remaining * 0.95
    if time.monotonic() < adam2_deadline:
        a2, score2 = adam_optimize(best_a, adam2_deadline, tau_start=30.0,
                                   tau_end=100.0, lr=0.002,
                                   use_multi_tau=False, reheat_interval=100)
        if score2 < best_score:
            best_a, best_score = a2, score2

    # Phase D: Coordinate descent polish (5% of time)
    if n <= 512 and time.monotonic() < deadline - 3:
        cd_deadline = deadline - 2
        a_cd, score_cd = coord_descent_polish(best_a, cd_deadline, max_rounds=3)
        if score_cd < best_score:
            best_a, best_score = a_cd, score_cd

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Multi-start Adam optimization with LP refinement,
    coordinate descent, across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Try lengths: start with 384 (was best before), then explore others
    lengths_to_try = [384, 512, 640]

    print(f"Starting optimization, budget={budget_s}s, deadline in {deadline - t_start:.0f}s")

    # --- First pass: quick optimization at each length ---
    for n in lengths_to_try:
        if time.monotonic() > deadline - 90:
            break

        starts = generate_starts(n, rng)
        # Use top 3 starts per length
        num_starts = min(3, len(starts))

        remaining = deadline - time.monotonic() - 60
        time_per_start = max(20.0, remaining / max(1, num_starts) * 0.7)

        for idx in range(num_starts):
            if time.monotonic() > deadline - 60:
                break

            t_alloc = min(time_per_start, deadline - time.monotonic() - 30)
            if t_alloc < 10:
                break

            a0 = starts[idx]
            print(f"  n={n}, start={idx}, alloc={t_alloc:.0f}s")

            a, score = optimize_single(a0, time.monotonic() + t_alloc, n)

            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"    NEW BEST: score={best_score:.6f}, n={n}")
            else:
                print(f"    score={score:.6f}, n={n}")

    # --- Second pass: focus on best length with more diverse starts ---
    if best_seq is not None and time.monotonic() < deadline - 60:
        best_n = len(best_seq)
        print(f"\nSecond pass at n={best_n} (current best: {best_score:.6f})")

        # Generate more diverse starts at the best length
        extra_starts = []
        for trial in range(4):
            a0 = rng.random(best_n) * 0.4 + 0.1
            extra_starts.append(a0)

        # Perturbed versions of best
        for trial in range(3):
            noise = rng.normal(0, 0.05, best_n)
            a0 = np.maximum(best_seq + noise, 0)
            extra_starts.append(a0)

        # Reversed + noise
        a0 = best_seq[::-1].copy() + rng.normal(0, 0.02, best_n)
        extra_starts.append(np.maximum(a0, 0))

        remaining = deadline - time.monotonic() - 30
        time_per_extra = max(15.0, remaining / max(1, len(extra_starts)) * 0.8)

        for idx, a0 in enumerate(extra_starts):
            if time.monotonic() > deadline - 30:
                break
            t_alloc = min(time_per_extra, deadline - time.monotonic() - 15)
            if t_alloc < 10:
                break

            a, score = optimize_single(a0, time.monotonic() + t_alloc, best_n)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"  [extra {idx}] NEW BEST: score={best_score:.6f}")
            else:
                print(f"  [extra {idx}] score={score:.6f}")

    # --- Third pass: intensive LP + Adam on the global best ---
    if best_seq is not None and time.monotonic() < deadline - 20:
        best_n = len(best_seq)
        remaining = deadline - time.monotonic() - 10
        print(f"\nFinal intensive pass (n={best_n}, score={best_score:.6f}, time={remaining:.0f}s)")

        if best_n <= 512:
            # More LP iterations
            lp_deadline = time.monotonic() + remaining * 0.4
            a, score = lp_refine(best_seq, lp_deadline, max_iters=50)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"  LP: score={best_score:.6f}")

        # Final Adam with very sharp gradient
        if time.monotonic() < deadline - 5:
            adam_deadline = deadline - 5
            a, score = adam_optimize(best_seq, adam_deadline, tau_start=50.0,
                                     tau_end=200.0, lr=0.001,
                                     use_multi_tau=False, reheat_interval=80)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"  Final Adam: score={best_score:.6f}")

    # --- Final coordinate descent on global best ---
    if best_seq is not None and len(best_seq) <= 512 and time.monotonic() < deadline - 3:
        a, score = coord_descent_polish(best_seq, deadline - 1, max_rounds=5)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  Final CD: score={best_score:.6f}")

    # --- Validation and return ---
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\nFinal result: n={len(best_seq)}, score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence: {eval_score:.6f}")
        except Exception as e:
            print(f"Evaluation error: {e}")
        # Ensure all values are valid
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        return [float(x) for x in best_seq]

    # Fallback
    print("Fallback: uniform sequence n=256")
    return [1.0] * 256
