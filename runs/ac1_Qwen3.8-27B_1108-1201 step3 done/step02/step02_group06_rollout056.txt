"""
Gradient-based optimization for minimizing normalized peak convolution
of non-negative step functions. Uses analytical gradients of the score
function with Adam optimization, diverse asymmetric initializations,
annealed perturbations, and multi-length search.

Key insight: For non-negative sequences, symmetric sequences give score >= 2.
To beat 2, the sequence must be asymmetric so that the convolution peak
is reduced below (sum)^2/n. The gradient exploits this structure directly.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    M = np.max(conv)
    return 2.0 * n * M / (S * S)


def make_trend_init(n, rng, trend_strength=1.0):
    """Create asymmetric initialization with linear trend."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 + trend_strength * (t - 0.3)
    a += 0.15 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_bump_init(n, rng):
    """Create initialization with off-center bump."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    center = 0.2 + 0.6 * rng.random()
    width = 0.1 + 0.3 * rng.random()
    a = 1.0 + 1.5 * np.exp(-((t - center) ** 2) / (2 * width ** 2))
    a += 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_quadratic_init(n, rng):
    """Create initialization with quadratic asymmetry."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 + 0.8 * (t - 0.25) ** 2 - 0.3 * t
    a += 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_sinusoidal_init(n, rng):
    """Create initialization with asymmetric sinusoid."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    k = int(rng.integers(1, max(2, n // 10)))
    phase = rng.uniform(0, np.pi)
    a = 1.0 + 0.5 * np.cos(2 * np.pi * k * t + phase) + 0.3 * (t - 0.5)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.5):
    """Create random initialization."""
    a = 1.0 + spread * rng.standard_normal(n)
    a += 0.3 * np.linspace(-1, 1, n)
    return np.clip(a, 0.01, None)


def gradient_step(f, lr, k_peak, M, S, mom_m, mom_v, beta1, beta2, t_step, eps=1e-8):
    """
    One Adam gradient step for the score function.
    
    Score = 2n*M/S^2 where M=max(conv), S=sum(f)
    Gradient:
      For i in active set (0<=k_peak-i<n): (4n/S^2)*(f[k_peak-i] - M/S)
      For i not in active set: -4n*M/S^3
    """
    n = len(f)
    factor = 4.0 * n / (S * S)
    
    # Active set: i where 0 <= k_peak - i < n
    i_start = max(0, k_peak - n + 1)
    i_end = min(n - 1, k_peak)
    
    # Build gradient
    # Base gradient for all elements: -4n*M/S^3 = -factor * M/S
    base_grad = -factor * M / S
    
    # Active set gradient: factor * (f[k_peak-i] - M/S)
    active_indices = np.arange(i_start, i_end + 1)
    contrib = f[k_peak - active_indices]
    active_grad = factor * (contrib - M / S)
    
    # Combine into full gradient
    grad = np.full(n, base_grad)
    grad[active_indices] = active_grad
    
    # Adam update
    mom_m = beta1 * mom_m + (1 - beta1) * grad
    mom_v = beta2 * mom_v + (1 - beta2) * (grad * grad)
    m_hat = mom_m / (1 - beta1 ** t_step)
    v_hat = mom_v / (1 - beta2 ** t_step)
    
    f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
    f_new = np.clip(f_new, 0.0, 1000.0)
    
    return f_new, mom_m, mom_v


def topk_gradient_step(f, lrs, peaks, conv, S, mom_m, mom_v, beta1, beta2, t_step, eps=1e-8):
    """
    Adam gradient step using top-k peaks for smoother gradient.
    """
    n = len(f)
    k_count = len(peaks)
    factor = 4.0 * n / (S * S)
    
    # Base gradient: -4n*M/S^3 where M = max(conv)
    M = conv[peaks[0]]
    base_grad = -factor * M / S
    grad = np.full(n, base_grad)
    
    # For each peak, add its contribution
    for k_p in peaks:
        i_start = max(0, k_p - n + 1)
        i_end = min(n - 1, k_p)
        active_indices = np.arange(i_start, i_end + 1)
        contrib = f[k_p - active_indices]
        grad[active_indices] += factor * contrib / k_count
    
    # Adam update
    mom_m = beta1 * mom_m + (1 - beta1) * grad
    mom_v = beta2 * mom_v + (1 - beta2) * (grad * grad)
    m_hat = mom_m / (1 - beta1 ** t_step)
    v_hat = mom_v / (1 - beta2 ** t_step)
    
    f_new = f - lrs * m_hat / (np.sqrt(v_hat) + eps)
    f_new = np.clip(f_new, 0.0, 1000.0)
    
    return f_new, mom_m, mom_v


def optimize_sequence(f_init, time_budget, rng, verbose=False, use_topk=3, use_anneal=True):
    """
    Main gradient-based optimization loop.
    Uses analytical gradient with Adam optimizer and annealing.
    """
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = compute_score(f)
    
    S = np.sum(f)
    base_lr = 0.3
    beta1, beta2 = 0.9, 0.999
    mom_m = np.zeros(n)
    mom_v = np.zeros(n)
    t_step = 0
    
    t0 = time.monotonic()
    stagnation = 0
    no_improve_count = 0
    consecutive_worse = 0
    
    while time.monotonic() - t0 < time_budget:
        conv = fft_convolve(f, f)
        S = np.sum(f)
        if S < 0.01:
            f = f_init.copy()
            S = np.sum(f)
            continue
        
        M = np.max(conv)
        current_score = 2.0 * n * M / (S * S)
        
        # Adaptive learning rate
        lr = base_lr / (1.0 + 0.005 * t_step)
        
        # Determine peaks for gradient
        if use_topk > 1 and n <= 2048:
            # Use top-k peaks
            threshold = M * 0.95
            peak_indices = np.where(conv >= threshold)[0]
            if len(peak_indices) == 0:
                peak_indices = np.array([np.argmax(conv)])
            elif len(peak_indices) > use_topk:
                top_vals = conv[peak_indices]
                top_order = np.argsort(top_vals)[::-1][:use_topk]
                peak_indices = peak_indices[top_order]
        else:
            peak_indices = np.array([np.argmax(conv)])
        
        # Take gradient step
        if use_topk > 1 and len(peak_indices) > 1:
            f, mom_m, mom_v = topk_gradient_step(f, lr, peak_indices, conv, S,
                                                  mom_m, mom_v, beta1, beta2, t_step)
        else:
            f, mom_m, mom_v = gradient_step(f, lr, int(peak_indices[0]), M, S,
                                             mom_m, mom_v, beta1, beta2, t_step)
        
        t_step += 1
        
        # Check new score
        new_S = np.sum(f)
        if new_S < 0.01:
            f = best_f.copy()
            continue
        new_score = 2.0 * n * np.max(fft_convolve(f, f)) / (new_S * new_S)
        
        if new_score < best_score:
            best_score = new_score
            best_f = f.copy()
            no_improve_count = 0
            stagnation = 0
        else:
            no_improve_count += 1
            consecutive_worse += 1
        
        # Annealing: periodically perturb to escape local minima
        if use_anneal and consecutive_worse > 50:
            perturbation_scale = min(0.3, 0.01 * np.sqrt(consecutive_worse / 10.0))
            if rng.random() < 0.3:
                # Local perturbation around the peak
                k_peak = int(np.argmax(fft_convolve(f, f)))
                i_start = max(0, k_peak - n + 1)
                i_end = min(n - 1, k_peak)
                noise_idx = np.arange(i_start, i_end + 1)
                f[noise_idx] += perturbation_scale * rng.standard_normal(len(noise_idx))
            elif rng.random() < 0.5:
                # Global small perturbation
                f += perturbation_scale * 0.5 * rng.standard_normal(n)
            else:
                # Block perturbation
                block_size = max(1, n // 8)
                start = int(rng.integers(0, n - block_size))
                f[start:start + block_size] += perturbation_scale * rng.standard_normal(block_size)
            
            f = np.clip(f, 0.0, 1000.0)
            consecutive_worse = 0
            # Reinitialize momentum to avoid bad direction
            mom_m *= 0.5
            mom_v *= 0.5
        
        if no_improve_count > 500:
            # Stronger perturbation
            f = best_f.copy()
            f += 0.1 * rng.standard_normal(n)
            f = np.clip(f, 0.01, None)
            mom_m = np.zeros(n)
            mom_v = np.zeros(n)
            no_improve_count = 0
            consecutive_worse = 0
        
        if verbose and t_step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    step={t_step}, score={current_score:.6f}, best={best_score:.6f}, "
                  f"elapsed={elapsed:.1f}s, lr={lr:.4f}")
    
    return best_f, best_score


def sa_polish(f_init, time_budget, rng):
    """Simulated annealing polish for final refinement."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = compute_score(f)
    current_score = best_score
    S = np.sum(f)
    
    t0 = time.monotonic()
    temperature = 0.005
    cooling_rate = 0.9999
    iter_count = 0
    
    while time.monotonic() - t0 < time_budget:
        candidate = f.copy()
        r = rng.random()
        
        if r < 0.4:
            # Perturb elements contributing to the peak
            conv = fft_convolve(f, f)
            k_peak = int(np.argmax(conv))
            i_start = max(0, k_peak - n + 1)
            i_end = min(n - 1, k_peak)
            active = np.arange(i_start, i_end + 1)
            top_k = max(5, len(active) // 5)
            # Find the most contributing elements
            contribs = f[active] * f[k_peak - active]
            top_idx_local = np.argsort(contribs)[-top_k:]
            target = active[top_idx_local]
            candidate[target] *= np.exp(rng.normal(-0.02, 0.01, size=len(target)))
            # Increase some non-contributing elements
            other = np.setdiff1d(np.arange(n), target)
            if len(other) > 0:
                n_inc = min(10, len(other))
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.01, size=n_inc))
        elif r < 0.7:
            # Gaussian perturbation
            scale = 0.01 + 0.01 * temperature / 0.005
            candidate *= np.exp(rng.normal(0, scale, size=n))
        elif r < 0.85:
            # Block operation
            block_size = max(1, int(rng.integers(1, max(2, n // 10))))
            start = int(rng.integers(0, n - block_size))
            candidate[start:start + block_size] *= np.exp(rng.normal(0, 0.03, size=block_size))
        else:
            # Swap two random elements (partial)
            i, j = rng.choice(n, size=2, replace=False)
            swap_amount = 0.1 * f[i]
            candidate[i] -= swap_amount
            candidate[j] += swap_amount
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        cand_S = np.sum(candidate)
        if cand_S < 0.01:
            continue
        candidate = candidate * (S / cand_S)
        
        new_score = compute_score(candidate)
        iter_count += 1
        
        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            f = candidate
            current_score = new_score
        
        temperature *= cooling_rate
        
        if iter_count % 5000 == 0 and temperature < 0.001:
            # Reheat occasionally
            temperature = 0.002
    
    return best_f, best_score


def coordinate_polish(f_init, time_budget):
    """Simple coordinate descent polish - modify pairs of elements."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = compute_score(f)
    S = np.sum(f)
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_budget:
        # Pick a random pair of indices
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        
        # Try a small adjustment: decrease i, increase j (or vice versa)
        # to maintain sum
        delta = 0.05 * min(f[i], f[j])
        
        # Try direction 1: decrease i, increase j
        cand = f.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, 1000.0)
        cand_S = np.sum(cand)
        if cand_S > 0.01:
            cand = cand * (S / cand_S)
            s = compute_score(cand)
            if s < best_score:
                best_score = s
                best_f = cand.copy()
                f = cand
        
        # Try direction 2: increase i, decrease j
        cand2 = f.copy()
        cand2[i] += delta
        cand2[j] -= delta
        cand2 = np.clip(cand2, 0.0, 1000.0)
        cand2_S = np.sum(cand2)
        if cand2_S > 0.01:
            cand2 = cand2 * (S / cand2_S)
            s2 = compute_score(cand2)
            if s2 < best_score:
                best_score = s2
                best_f = cand2.copy()
                f = cand2
    
    return best_f, best_score

        sa_time = min(150.0, remaining * 0.5)
        if sa_time > 20:
            print(f"  SA polish, n={len(best_seq)}, time={sa_time:.0f}s")
            a_sa, score_sa = sa_polish(best_seq, sa_time, rng)
            print(f"  SA result: {score_sa:.6f}")
            if score_sa < best_score:
                best_score = score_sa
                best_seq = a_sa.copy()
                print(f"  *** New best: {best_score:.6f} ***")
    
    # === Phase 5: Final gradient polish ===
    print("\n" + "=" * 60)
    print("Phase 5: Final gradient polish")
    print("=" * 60)
    
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        final_time = min(100.0, remaining * 0.6)
        if final_time > 15:
            print(f"  Final gradient, n={len(best_seq)}, time={final_time:.0f}s")
            a_final, score_final = optimize_sequence(best_seq, final_time, rng,
                                                      verbose=False, use_topk=5, use_anneal=False)
            print(f"  Final gradient result: {score_final:.6f}")
            if score_final < best_score:
                best_score = score_final
                best_seq = a_final.copy()
                print(f"  *** New best: {best_score:.6f} ***")
    
    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 60}")
        return final_list
    
    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
