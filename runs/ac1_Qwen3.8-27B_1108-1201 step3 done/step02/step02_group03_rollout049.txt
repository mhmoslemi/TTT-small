"""

Uses a combination of:

- Simulated annealing with gradient-guided sum-preserving moves

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def simplex_project(a, S):
    """Project vector a onto {x >= 0, sum(x) = S} via water-filling."""
    n = len(a)
    if n == 0:
        return a
    u = np.sort(a)[::-1]
    cv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cv - S)
    if not np.any(cond):
        return np.full(n, S / n)
    rho = np.max(np.where(cond)[0])
    theta = (cv[rho] - S) / (rho + 1.0)
    return np.maximum(a - theta, 0.0)


def gradient_soft_fast(a, tau):
    """Compute soft-max gradient using FFT-based convolution."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    max_exp = np.max(c_shifted)
    if max_exp < -700:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = np.exp(tau * (c_shifted - max_exp))
        w /= w.sum()
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev)
    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    return grad


def gradient_hard_fast(a):
    """Compute hard-max gradient."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)

    for i in range(valid_start, valid_end):
        j = k_star - i

    return grad

def simulated_annealing(a, deadline, epochs=200, t_start=0.5, t_end=0.01):
    """
    Simulated annealing with sum-preserving moves.
    Move: pick two random indices, transfer mass.
    Returns (best_sequence, best_score).
    """
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = simplex_project(np.maximum(a, 0.0), S_t)
    best = a.copy()
    best_score = compute_obj(a)
    rng = np.random.default_rng(12345)

    for epoch in range(epochs):
        if time.monotonic() > deadline:
            break
        T = t_start * (t_end / t_start) ** (epoch / epochs)
        moves_per_epoch = max(50, n // 2)

        for _ in range(moves_per_epoch):
            if time.monotonic() > deadline:
                break
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            # Transfer mass from i to j
            delta = rng.uniform(0, max(a[i], 0.1)) * 0.5
            if a[i] < delta:
                delta = a[i]
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = simplex_project(np.maximum(a_new, 0.0), S_t)

            s_new = compute_obj(a_new)
            delta_obj = s_new - best_score
            if delta_obj < 0 or rng.random() < np.exp(-delta_obj / (T + 1e-10)):
                a = a_new
                if s_new < best_score:
                    best_score = s_new
                    best = a.copy()

    return best, best_score


def lp_refine(a, deadline, max_iters=50):
    """
    LP-based refinement: solve max sum(g) s.t. convolve(a, g) <= M, g >= 0.
    Blend the solution with current a.
    """
    n = len(a)
    if n > 600:
        return a, compute_obj(a)

    S_t = np.sqrt(2.0 * n)
    a = simplex_project(np.maximum(a, 0.0), S_t)
    best_score = compute_obj(a)
    best = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = np.convolve(a, a)
        M = float(c.max())

        # Build constraint matrix
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            m_start = j
            m_end = min(2 * n - 1, j + n)
            A[m_start:m_end, j] = a[:m_end - m_start]

        c_obj = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs', options={'presolve': True})
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break

        g_sum = g.sum()
        if g_sum < 1e-6:
            break
        g_norm = g * (S_t / g_sum)

        improved = False
        for t_blend in [0.8, 0.6, 0.4, 0.2, 0.1]:
            a_new = simplex_project((1.0 - t_blend) * a + t_blend * g_norm, S_t)
            s_new = compute_obj(a_new)
            if s_new < best_score - 1e-10:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break

        if not improved:
            break

    return best, best_score

        lr *= 0.999
        if it % 200 == 0:
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best = a.copy()
    print(f"      Phase 3 (fine Nesterov): {best_score:.6f}")

    # Phase 4: LP (15%)
    if n <= 512:
        p4_deadline = time.monotonic() + remaining * 0.12
        a4, s4 = lp_refine(best, p4_deadline)
        if s4 < best_score:
            best = a4.copy()
            best_score = s4
        print(f"      Phase 4 (LP): {best_score:.6f}")

    # Phase 5: Coordinate descent (5%)
    p5_deadline = time.monotonic() + remaining * 0.08
    a5, s5 = coordinate_descent(best, p5_deadline)
    if s5 < best_score:
        best = a5.copy()
        best_score = s5
    print(f"      Phase 5 (coord): {best_score:.6f}")

    return best, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Multi-phase optimization from diverse starting points.
    Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Try to use height_sequence_1 if available
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)):
            best_seq = gs.copy()
            best_score = compute_obj(gs)
            print(f"Starting from height_sequence_1: n={len(gs)}, score={best_score:.6f}")
    except NameError:
        pass

    # Define lengths to try
    all_lengths = [128, 192, 256, 320, 384, 448, 512, 640]

    # Phase A: Quick exploration (40% of budget)
    print(f"\n=== Phase A: Quick exploration ===")
    phase_a_deadline = t_start + (deadline - t_start) * 0.35

    candidates = []  # (score, sequence)

    # First, quickly try improving height_sequence_1 if we have it
    if best_seq is not None:
        n0 = len(best_seq)
        quick_deadline = t_start + (deadline - t_start) * 0.12
        a_quick, s_quick = nesterov_optimize(best_seq.copy(), quick_deadline,
                                             tau_start=10.0, tau_end=50.0,
                                             lr_start=0.3, lr_end=0.02,
                                             max_iters=10000)
        if s_quick < best_score:
            best_score = s_quick
            best_seq = a_quick.copy()
            print(f"  Quick improvement on h_seq_1: {best_score:.6f}")

    # Try quick gradient descent at various lengths
    for n in all_lengths:
        if time.monotonic() > phase_a_deadline - 30:
            break
        inits = generate_inits(n, rng)
        # Use 2 random inits + 1 structured
        for a0 in inits[:4]:
            if time.monotonic() > phase_a_deadline - 10:
                break
            t_alloc = min(40.0, (phase_a_deadline - time.monotonic()) * 0.7)
            if t_alloc < 10:
                break
            a_res, s_res = nesterov_optimize(a0, time.monotonic() + t_alloc,
                                             tau_start=5.0, tau_end=40.0,
                                             lr_start=0.5, lr_end=0.05,
                                             max_iters=8000)
            candidates.append((s_res, a_res))
            if s_res < best_score:
                best_score = s_res
                best_seq = a_res.copy()
                print(f"  n={n}: new best {best_score:.6f}")

    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    print(f"  Best from Phase A: {candidates[0][0]:.6f} (n={len(candidates[0][1])})")

    # Phase B: Deep optimization on top candidates (45% of budget)
    print(f"\n=== Phase B: Deep optimization ===")
    phase_b_deadline = t_start + (deadline - t_start) * 0.80

    # Take top candidates + the global best
    top_candidates = []
    seen_lengths = set()
    for s, a in candidates[:5]:
        if len(a) not in seen_lengths:
            top_candidates.append((s, a.copy()))
            seen_lengths.add(len(a))

    if best_seq is not None and len(best_seq) not in seen_lengths:
        top_candidates.insert(0, (best_score, best_seq.copy()))

    # Also add the best length with a fresh random init
    if top_candidates:
        best_n = len(top_candidates[0][1])
        top_candidates.append((float('inf'), rng.random(best_n) * 0.5 + 0.25))

    for idx, (s_init, a_init) in enumerate(top_candidates[:4]):
        if time.monotonic() > phase_b_deadline - 60:
            break
        t_alloc = min(400.0, (phase_b_deadline - time.monotonic()) * 0.8)
        if t_alloc < 30:
            break
        print(f"  Candidate {idx}: n={len(a_init)}, initial_score={s_init:.6f}, alloc={t_alloc:.0f}s")
        a_res, s_res = full_optimize(a_init, time.monotonic() + t_alloc)
        if s_res < best_score:
            best_score = s_res
            best_seq = a_res.copy()
            print(f"    -> New best: {best_score:.6f}")
        else:
            print(f"    -> {s_res:.6f} (not better)")

    # Phase C: Focus on the single best sequence (remaining time)
    print(f"\n=== Phase C: Final focus ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 60:
            n_best = len(best_seq)
            print(f"  Focusing on n={n_best}, current score={best_score:.6f}")

            # Multiple rounds of refinement
            for round_num in range(3):
                if time.monotonic() > deadline - 15:
                    break
                t_alloc = min(remaining / (3 - round_num), 300.0)
                if t_alloc < 20:
                    break

                # Nesterov with different params
                tau_s = 20.0 + round_num * 20.0
                lr_s = 0.15 / (round_num + 1)
                a_r, s_r = nesterov_optimize(best_seq.copy(),
                                             time.monotonic() + t_alloc * 0.5,
                                             tau_start=tau_s, tau_end=tau_s + 40,
                                             lr_start=lr_s, lr_end=lr_s * 0.1,
                                             max_iters=15000)
                if s_r < best_score:
                    best_score = s_r
                    best_seq = a_r.copy()
                    print(f"  Round {round_num} Nesterov: {best_score:.6f}")

                # SA
                if time.monotonic() < deadline - 20:
                    a_sa, s_sa = simulated_annealing(best_seq,
                                                     time.monotonic() + t_alloc * 0.3,
                                                     epochs=100,
                                                     t_start=0.3 / (round_num + 1),
                                                     t_end=0.005)
                    if s_sa < best_score:
                        best_score = s_sa
                        best_seq = a_sa.copy()
                        print(f"  Round {round_num} SA: {best_score:.6f}")

                # LP if n is small enough
                if n_best <= 512 and time.monotonic() < deadline - 20:
                    a_lp, s_lp = lp_refine(best_seq, time.monotonic() + t_alloc * 0.2,
                                           max_iters=30)
                    if s_lp < best_score:
                        best_score = s_lp
                        best_seq = a_lp.copy()
                        print(f"  Round {round_num} LP: {best_score:.6f}")

            # Final coordinate descent
            if time.monotonic() < deadline - 10 and n_best <= 640:
                a_cd, s_cd = coordinate_descent(best_seq, deadline - 5, max_rounds=3)
                if s_cd < best_score:
                    best_score = s_cd
                    best_seq = a_cd.copy()
                    print(f"  Final coord descent: {best_score:.6f}")

    # Final validation
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\n=== Final: n={len(best_seq)}, score={final_score:.6f} ===")
        try:
            eval_score = evaluate_sequence([float(x) for x in best_seq])
            print(f"evaluate_sequence: {eval_score:.6f}")
        except Exception as e:
            print(f"Eval error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Fallback: uniform sequence")
    return [1.0] * 128
