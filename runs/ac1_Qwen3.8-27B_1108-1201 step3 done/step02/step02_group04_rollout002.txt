import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def eval_normalized(f):
    """Evaluate 2*n*max(conv(f,f)) assuming sum(f)=1."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def solve_lp_cross(f, M, n, time_limit=5.0):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k=0..2n-2."""
    rows_list = []
    cols_list = []
    vals_list = []
    for i in range(n):
        j_start = 0
        j_end = min(n - 1, 2 * n - 2 - i)
        for j in range(j_start, j_end + 1):
            k = i + j
            rows_list.append(k)
            cols_list.append(i)
            vals_list.append(f[j])

    if not rows_list:
        return np.ones(n) / n

    A = sparse.csc_matrix(
        (np.array(vals_list, dtype=np.float64),
         (np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32))),
        shape=(2 * n - 1, n)
    ).tocsr()

    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                               options={'disp': False, 'time_limit': time_limit,
                                        'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def gradient_polish(f0, n, time_limit, seed=123):
    """Gradient-based polish using subgradient at active set with momentum."""
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    eta = 0.1
    momentum = 0.5
    v = np.zeros(n)
    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        max_val = conv.max()

        # Find active set (peaks within 5% of max)
        thresh = max_val * 0.95
        active = np.where(conv >= thresh)[0]

        # Compute weighted subgradient
        weights = conv[active]
        w_sum = weights.sum()
        if w_sum < 1e-15:
            break
        weights = weights / w_sum

        sg = np.zeros(n)
        for w, k in zip(weights, active):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            # d(conv[k])/df[i] for i in [j_start, j_end]: f[k-i]
            sg[j_start:j_end + 1] += w * f[k - j_end:k - j_start + 1][::-1]

        # Momentum update
        v = momentum * v + (1.0 - momentum) * sg
        f_new = f - eta * v
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-12:
            break
        f_new = f_new / s

        obj_new = eval_normalized(f_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            patience = 0
            # Slowly increase eta if making steady progress
            if patience == 0 and eta < 0.5:
                eta = min(eta * 1.02, 0.5)
        else:
            patience += 1
            if patience > 50:
                eta *= 0.7
            if patience > 300:
                break

    return best_f, best_obj


def block_perturb_restart(f0, n, rng, n_blocks=3, block_frac=0.1):
    """Create perturbed version by zeroing and re-randomizing blocks."""
    f = f0.copy()
    block_size = max(5, int(n * block_frac))
    for _ in range(n_blocks):
        start = rng.integers(0, max(1, n - block_size))
        f[start:start + block_size] = rng.uniform(0.1, 1.0, size=block_size)
    f = np.maximum(f, 0)
    s = f.sum()
    if s > 1e-12:
        f = f / s
    return f


def interpolate_sequence(src, target_n):
    """Interpolate a sequence to a different length."""
    if len(src) == target_n:
        return src.copy()
    x_old = np.linspace(0, 1, len(src))
    x_new = np.linspace(0, 1, target_n)
    interp = np.interp(x_new, x_old, src)
    interp = np.maximum(interp, 0)
    s = interp.sum()
    if s > 1e-12:
        interp = interp / s
    else:
        interp = np.ones(target_n) / target_n
    return interp


def gen_diverse_inits(n, rng):

                f_pol, obj_pol = gradient_polish(f_opt, n, time_limit=tp, seed=seed + 100 + idx)
                score_pol = float(evaluate_sequence(f_pol.tolist()))
                if np.isfinite(score_pol) and score_pol < score:
                    f_opt = f_pol
                    score = score_pol
                    obj_opt = obj_pol

        if np.isfinite(score):
            all_results.append((score, n, name, f_opt))
            if score < best_score:
                best_score = score
                best_seq = f_opt.copy()
                print(f"    NEW BEST: n={n}, score={score:.6f}")
            else:
                print(f"    score={score:.6f}")

    print(f"Phase 1 done. Best: {best_score:.6f}, elapsed={time.monotonic()-t_start:.0f}s")

    # Phase 2: Deep optimization on top results with shake-and-restart
    all_results.sort(key=lambda x: x[0])
    top = all_results[:5]

    remaining = deadline - time.monotonic() - 30
    if remaining > 60 and top:
        print(f"Phase 2: Deep optimization, {remaining:.0f}s remaining")
        t_each = remaining / len(top)

        for idx, (sc, n, name, f) in enumerate(top):
            if time.monotonic() > deadline - 20:
                break
            ta = min(t_each, deadline - time.monotonic() - 15)
            if ta < 10:
                break

            print(f"  Deep[{idx}]: {name}, n={n}, score={sc:.6f}, time={ta:.0f}s")

            # LP with longer time
            f2, obj2 = lp_optimize(f, n, time_limit=ta * 0.6, seed=seed + 200 + idx)
            sc2 = float(evaluate_sequence(f2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = f2.copy()
                print(f"    LP improved: {sc2:.6f}")

            # Shake and re-optimize
            if time.monotonic() < deadline - 15:
                ts = min(ta * 0.2, deadline - time.monotonic() - 10)
                if ts > 10:
                    f_shake = block_perturb_restart(f2, n, rng, n_blocks=2, block_frac=0.08)
                    f3, obj3 = lp_optimize(f_shake, n, time_limit=ts, seed=seed + 300 + idx)
                    sc3 = float(evaluate_sequence(f3.tolist()))
                    if np.isfinite(sc3) and sc3 < best_score:
                        best_score = sc3
                        best_seq = f3.copy()
                        print(f"    Shake improved: {sc3:.6f}")

            # Gradient polish
            if time.monotonic() < deadline - 10:
                tp = min(ta * 0.15, deadline - time.monotonic() - 5)
                if tp > 3:
                    f4, _ = gradient_polish(f2, n, time_limit=tp, seed=seed + 400 + idx)
                    sc4 = float(evaluate_sequence(f4.tolist()))
                    if np.isfinite(sc4) and sc4 < best_score:
                        best_score = sc4
                        best_seq = f4.copy()
                        print(f"    Polish improved: {sc4:.6f}")

    print(f"Phase 2 done. Best: {best_score:.6f}, elapsed={time.monotonic()-t_start:.0f}s")

    # Phase 3: Final intensive polish on the best solution
    if best_seq is not None and time.monotonic() < deadline - 10:
        print("Phase 3: Final polish")
        n_best = len(best_seq)
        tr = deadline - time.monotonic() - 5
        tr = max(5.0, tr)

        # Multiple gradient polish passes with different parameters
        for gi in range(3):
            if tr < 3:
                break
            tp = tr / 4.0
            etas = [0.05, 0.15, 0.3]
            f5, _ = gradient_polish(best_seq, n_best, time_limit=tp, seed=seed + 500 + gi)
            sc5 = float(evaluate_sequence(f5.tolist()))
            if np.isfinite(sc5) and sc5 < best_score:
                best_score = sc5
                best_seq = f5.copy()
                print(f"    Final polish {gi}: {sc5:.6f}")
            tr = deadline - time.monotonic() - 5

        # One more LP run on the best
        if tr > 20:
            f6, _ = lp_optimize(best_seq, n_best, time_limit=tr * 0.5, seed=seed + 600)
            sc6 = float(evaluate_sequence(f6.tolist()))
            if np.isfinite(sc6) and sc6 < best_score:
                best_score = sc6
                best_seq = f6.copy()
                print(f"    Final LP: {sc6:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}, time={time.monotonic()-t_start:.0f}s")
    return [float(x) for x in best_seq]
