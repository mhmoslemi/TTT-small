"""
Gradient-based optimization with log-sum-exp smoothing for minimizing
peak autocorrelation of non-negative step functions. Uses adaptive step sizes,
momentum, peak-directed moves, and simulated annealing polish.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def compute_gradient_smoothed(a, beta, N):
    """Compute gradient of smoothed max of conv(a,a).
    Returns (smoothed_value, gradient_vector)."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    exp_c = np.exp(beta * (c - c_max))
    w = exp_c / np.sum(exp_c)
    val = c_max + np.log(np.sum(exp_c)) / beta  # (1/beta)*log(sum exp(beta*c))
    
    # Gradient: g[i] = 2 * sum_k w[k] * a[k-i] (valid indices)
    # = 2 * cross_corr(w, a)[i]
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(W * np.conj(A), N)[:n]
    g = 2.0 * cross
    
    return val, g


def gradient_descent_smoothed(a0, time_limit, rng, beta_schedule='linear', verbose=True):
    """Gradient descent with log-sum-exp smoothing."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 0.01:
        a = np.ones(n) / n
    else:
        a = a / s
    
    N = 1
    while N < 3 * n + 2:
        N <<= 1
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    
    lr = 0.1
    momentum = 0.9
    v = np.zeros(n)
    beta = 50.0
    beta_max = 500.0
    total_iter = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        # Anneal beta
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        if beta_schedule == 'linear':
            beta = 50.0 + (beta_max - 50.0) * progress
        elif beta_schedule == 'log':
            beta = 50.0 * (10.0 ** (progress))
        
        val, g = compute_gradient_smoothed(a, beta, N)
        total_iter += 1
        
        # Momentum update
        v = momentum * v - lr * g
        a_new = a + v
        
        # Project: non-negativity and sum=1
        a_new = np.clip(a_new, 0, None)
        s_new = np.sum(a_new)
        if s_new < 0.01:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Check improvement
        score_new = compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        
        a = a_new
        
        # Adaptive learning rate
        if stall > 50:
            lr *= 0.9
            stall = 0
        elif total_iter % 200 == 0:
            lr = min(lr * 1.05, 0.5)
        
        if verbose and total_iter % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GD iter={total_iter}, score={score_new:.6f}, best={best_score:.6f}, beta={beta:.1f}, lr={lr:.4f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score


def peak_directed_move(a, rng, n_trials=10):
    """Try to reduce the peak convolution bin directly."""
    n = len(a)
    c = fft_convolve(a, a)
    peak_idx = int(np.argmax(c))
    peak_val = c[peak_idx]
    
    # Find contributions to the peak
    contribs = np.zeros(n)
    for i in range(n):
        j = peak_idx - i
        if 0 <= j < n:
            contribs[i] = 2.0 * a[i] * a[j]  # factor of 2 for symmetry handling
        # Actually, the contribution of a[i] to c[peak_idx] is:
        # a[i] appears in a[i]*a[peak_idx-i] and a[peak_idx-i]*a[i] if different
        # dc[peak_idx]/da[i] = 2*a[peak_idx-i] (as we derived)
    
    # Gradient at the peak
    grad_peak = np.zeros(n)
    for i in range(n):
        j = peak_idx - i
        if 0 <= j < n:
            grad_peak[i] = 2.0 * a[j]
    
    # Try to reduce the peak by moving mass from high-contribution elements
    # to low-contribution elements
    score0 = compute_score(a)
    
    # Identify top and bottom contributors
    sorted_idx = np.argsort(contribs)
    top_k = max(5, n // 20)
    bottom_k = max(5, n // 20)
    top = sorted_idx[-top_k:]
    bottom = sorted_idx[:bottom_k]
    
    best_a = a.copy()
    best_s = score0
    
    for trial in range(n_trials):
        candidate = a.copy()
        # Reduce top contributors
        reduce_factors = np.exp(rng.normal(-0.05, 0.02, size=len(top)))
        candidate[top] *= reduce_factors
        # Increase bottom contributors to maintain sum
        mass_moved = np.sum(a[top] - candidate[top])
        if mass_moved > 0:
            current_bottom = candidate[bottom].copy()
            add_factors = np.ones(len(bottom)) + mass_moved / max(np.sum(current_bottom), 1e-10)
            candidate[bottom] = current_bottom * add_factors
        
        candidate = np.clip(candidate, 0, 1000.0)
        s_c = np.sum(candidate)
        if s_c < 0.01:
            continue
        candidate = candidate / s_c  # normalize
        
        sc = compute_score(candidate)
        if sc < best_s:
            best_s = sc
            best_a = candidate.copy()
    
    return best_a, best_s


def spectral_init(n, rng):
    """Initialize using spectral method: try to make autocorrelation flat."""
    # Start with random phase, flat amplitude
    N = n
    A_mag = np.ones(N)
    A_phase = 2 * np.pi * rng.random(N)
    
    # Iterate: enforce non-negativity in time domain
    a = np.zeros(n)
    for it in range(200):
        A = A_mag * np.exp(1j * A_phase)
        a_full = np.fft.irfft(A, N)
        a = np.clip(a_full[:n], 0, None)
        if np.sum(a) < 0.01:
            break
        # Update spectrum from current time-domain signal
        A_new = np.fft.rfft(a, N)
        A_mag = np.abs(A_new)
        A_phase = np.angle(A_new if len(A_new) == N//2+1 else np.fft.fft(a, N))
    
    # Alternative: just use the time-domain signal after clipping
    if np.sum(a) < 0.01:
        a = np.ones(n) / n
    else:

def simulated_annealing_polish(a0, time_limit, rng):
    """Simulated annealing for final polish."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = a / np.sum(a)  # normalize
    current_score = compute_score(a)
    best = a.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9998
    
    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.35:
            # Targeted peak reduction
            c = fft_convolve(a, a)
            peak_idx = int(np.argmax(c))
            # Identify elements contributing most to peak
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(3, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 5:
                n_inc = min(10, len(other) // 3)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.01, size=n_inc))
        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.015, size=width))
        elif r < 0.70:
            # Pair swap
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.1, 0.1) * a[i]
                candidate[i] = max(0, a[i] - delta)
                candidate[j] = max(0, a[j] + delta)
        elif r < 0.85:
            # Multi-point perturbation
            for _ in range(3):
                i = int(rng.integers(0, n))
                candidate[i] *= np.exp(rng.normal(0.0, 0.02))
        else:
            # Global small perturbation
            candidate = a * np.exp(rng.normal(0.0, 0.004, size=n))
        
        candidate = np.clip(candidate, 0.0, 1000.0)
        s_c = np.sum(candidate)
        if s_c < 0.01:
            continue
        candidate = candidate / s_c
        
        new_score = compute_score(candidate)
        
        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            a = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best, best_score


def resample_sequence(seq, new_n):
    """Resample a sequence to a new length."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)

import numpy as np

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_gradient_smoothed(a, beta, N):
    """
    Compute gradient of (1/beta)*log(sum exp(beta*conv(a,a))) w.r.t. a.
    Returns (smoothed_value, gradient_vector of length n).

    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    exp_c = np.exp(beta * (c - c_max))

    val = c_max + np.log(np.sum(exp_c)) / beta

        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        beta = 20.0 + (beta_final - 20.0) * (progress ** 0.7)

        val, g = compute_gradient_smoothed(a, beta, N)
        total_iter += 1

        v = momentum * v - lr * g
        a_new = a + v
        a_new = project_nonneg_sum1(a_new)

        # Compute score every few iterations
        if total_iter % 3 == 0 or total_iter < 10:
            score_new = compute_score(a_new)
            score_cache = score_new
        else:
            score_new = score_cache

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new

        # Adaptive learning rate
        if stall > 100:
            lr *= 0.85
            v *= 0.5
            stall = 0
        elif stall > 300:
            lr *= 0.7
            v *= 0.3
            stall = 0
            # Perturb to escape
            a = a * np.exp(rng.normal(0, 0.005, n))
            a = project_nonneg_sum1(a)
            best_score = min(best_score, compute_score(a))
            if compute_score(a) < best_score:
                best_a = a.copy()
                best_score = compute_score(a)
        elif total_iter % 500 == 0:
            lr = min(lr * 1.03, 0.3)

        if verbose and total_iter % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GD iter={total_iter}, score={score_cache:.6f}, best={best_score:.6f}, beta={beta:.1f}, lr={lr:.5f}, t={elapsed:.1f}s")

    return best_a, best_score


def peak_directed_move(a, rng, n_trials=15):
    """Try to reduce the peak convolution bin by redistributing mass."""
    n = len(a)
    c = fft_convolve(a, a)
    peak_idx = int(np.argmax(c))

    # Contributions to the peak bin
    contribs = np.zeros(n)
    for i in range(n):
        j = peak_idx - i
        if 0 <= j < n:
            contribs[i] = a[i] * a[j]

    score0 = compute_score(a)
    sorted_idx = np.argsort(contribs)
    top_k = max(4, n // 20)
    bottom_k = max(4, n // 20)
    top = sorted_idx[-top_k:]
    bottom = np.setdiff1d(sorted_idx[:bottom_k * 2], top)
    if len(bottom) < bottom_k:
        bottom = np.setdiff1d(np.arange(n), top)
        bottom = bottom[:bottom_k]

    best_a = a.copy()
    best_s = score0

    for _ in range(n_trials):
        candidate = a.copy()
        reduce_factors = np.exp(rng.normal(-0.04, 0.015, size=len(top)))
        candidate[top] *= reduce_factors
        mass_moved = np.sum(a[top] - candidate[top])
        if mass_moved > 1e-10:
            current_bottom = candidate[bottom].copy()
            if np.sum(current_bottom) > 1e-10:
                scale = 1.0 + mass_moved / np.sum(current_bottom)
                candidate[bottom] = current_bottom * scale

        candidate = np.clip(candidate, 0, 1000.0)
        s_c = np.sum(candidate)
        if s_c < 0.01:
            continue
        candidate = candidate / s_c
        sc = compute_score(candidate)
        if sc < best_s:
            best_s = sc
            best_a = candidate.copy()

    return best_a, best_s


def spectral_init(n, rng, n_iter=300):
    """Initialize using spectral method targeting flat autocorrelation."""
    N = n
    # Start with random phases
    A = np.fft.rfft(np.ones(N))
    A[:N//2+1] = np.abs(A[:N//2+1]) * np.exp(1j * 2 * np.pi * rng.random(N//2+1))

    a = np.ones(n) / n
    for it in range(n_iter):
        a_full = np.fft.irfft(A, N)
        a = np.clip(a_full[:n], 0, None)
        s = np.sum(a)
        if s < 0.01:
            a = np.ones(n) / n
        else:
            a = a / s
        A = np.fft.rfft(a, N)
        # Blend: keep current phase, target magnitude
        target_mag = np.ones(N // 2 + 1)
        current_mag = np.abs(A)
        blend = 0.5 + 0.5 * (it / n_iter)
        A = (blend * target_mag + (1 - blend) * current_mag) * np.exp(1j * np.angle(A))

    return a


def simulated_annealing_polish(a0, time_limit, rng):
    """Simulated annealing for final polish."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = a / np.sum(a)
    current_score = compute_score(a)
    best = a.copy()
    best_score = current_score

    t0 = time.monotonic()
    temperature = 0.0015
    cooling = 0.9999
    it = 0

    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()

        if r < 0.35:
            # Targeted peak reduction
            c = fft_convolve(a, a)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(3, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.025, 0.01, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 5:
                n_inc = min(8, len(other) // 4)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.02, 0.008, size=n_inc))
        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.012, size=width))
        elif r < 0.70:
            # Pair redistribution
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.08, 0.08) * a[i]
                candidate[i] = max(0, a[i] - delta)
                candidate[j] = max(0, a[j] + delta)
        elif r < 0.85:
            # Multi-point
            for _ in range(4):
                i = int(rng.integers(0, n))
                candidate[i] *= np.exp(rng.normal(0.0, 0.015))
        else:
            # Global small
            candidate = a * np.exp(rng.normal(0.0, 0.003, size=n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        s_c = np.sum(candidate)
        if s_c < 0.01:
            continue
        candidate = candidate / s_c

        new_score = compute_score(candidate)
        it += 1

        if new_score < current_score:
            a = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = a.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            a = candidate
            current_score = new_score

        temperature *= cooling
        if it % 5000 == 0:
            print(f"      SA iter={it}, current={current_score:.6f}, best={best_score:.6f}, temp={temperature:.6f}")

    return best, best_score


def resample_sequence(seq, new_n):
    """Resample a sequence to a new length."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)


def make_chirp_init(n, alpha, rng):
    """Create chirp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Gradient-based optimization with log-sum-exp smoothing, peak-directed moves,
    spectral initialization, and simulated annealing polish.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 30.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(start_seq) > 0:
            s = compute_score(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()

    # Phase 1: Gradient descent from provided sequence
    print("\n=== Phase 1: Gradient descent (primary) ===")
    if best_seq is not None and time.monotonic() < deadline - 500:
        t1 = min(350.0, deadline - time.monotonic() - 500)
        a_best, s_best = gradient_descent_smoothed(best_seq, t1, rng, verbose=True)
        print(f"  Phase 1 result: score={s_best:.6f}")
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 2: Peak-directed refinement
    print(f"\n=== Phase 2: Peak-directed refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_cur = best_seq.copy()
        a_cur = a_cur / np.sum(a_cur)
        s_cur = compute_score(a_cur)
        no_improve = 0
        for step in range(100):
            if time.monotonic() > deadline - 300:
                break
            a_new, s_new = peak_directed_move(a_cur, rng, n_trials=10)
            if s_new < s_cur - 1e-8:
                a_cur = a_new
                s_cur = s_new
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 20:
                    a_cur = a_cur * np.exp(rng.normal(0, 0.02, len(a_cur)))
                    a_cur = np.clip(a_cur, 0, None)
                    a_cur = a_cur / np.sum(a_cur)
                    s_cur = compute_score(a_cur)
                    no_improve = 0
        print(f"  Phase 2 result: score={s_cur:.6f}")
        if s_cur < best_score:
            best_score = s_cur
            best_seq = a_cur.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 3: Exploration with different initializations
    print(f"\n=== Phase 3: Exploration ===")
    exploration_inits = []

    # Spectral init
    exploration_inits.append(('spectral_512', spectral_init(512, rng)))
    # Chirp inits
    exploration_inits.append(('chirp_512', make_chirp_init(512, 0.5, rng)))
    # Resampled best
    if best_seq is not None:
        exploration_inits.append(('resampled_384', resample_sequence(best_seq, 384)))
        exploration_inits.append(('resampled_768', resample_sequence(best_seq, 768)))
    # Random
    rand_a = 1.0 + 0.3 * rng.standard_normal(512)
    exploration_inits.append(('random_512', np.clip(rand_a, 0.05, None)))

    n_exp = len(exploration_inits)
    for ei in range(n_exp):
        if time.monotonic() > deadline - 200:
            break
        name, a0 = exploration_inits[ei]
        time_left = deadline - time.monotonic() - 200
        remaining = n_exp - ei
        t_per = max(25.0, min(80.0, time_left / max(1, remaining)))

        print(f"  [{name}] starting...")
        a_best, s_best = gradient_descent_smoothed(a0, t_per, rng, verbose=False)

        # Resample back to 512 if needed
        if len(a_best) != 512:
            a_best = resample_sequence(a_best, 512)
            s_best = compute_score(a_best)

        print(f"  [{name}] score={s_best:.6f}")
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # Phase 4: Second gradient polish on best
    print(f"\n=== Phase 4: Second GD polish ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        final_time = min(150.0, max(20.0, deadline - time.monotonic() - 150))
        a_best, s_best = gradient_descent_smoothed(best_seq, final_time, rng, verbose=False)
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"  After 2nd GD: {best_score:.6f}")
        else:
            print(f"  2nd GD didn't improve: {best_score:.6f}")

    # Phase 5: Peak-directed + GD interleaved
    print(f"\n=== Phase 5: Interleaved refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_cur = best_seq.copy()
        a_cur = a_cur / np.sum(a_cur)
        s_cur = compute_score(a_cur)
        t5 = min(100.0, deadline - time.monotonic() - 100)
        t5_start = time.monotonic()
        while time.monotonic() - t5_start < t5:
            a_new, s_new = peak_directed_move(a_cur, rng, n_trials=8)
            if s_new < s_cur:
                a_cur = a_new
                s_cur = s_new
            # Short GD burst
            t_burst = min(5.0, deadline - time.monotonic() - 100)
            if t_burst > 2.0:
                a_gd, s_gd = gradient_descent_smoothed(a_cur, t_burst, rng, verbose=False)
                if s_gd < s_cur:
                    a_cur = a_gd
                    s_cur = s_gd
        if s_cur < best_score:
            best_score = s_cur
            best_seq = a_cur.copy()
            print(f"  After interleaved: {best_score:.6f}")
        else:
            print(f"  Interleaved didn't improve: {best_score:.6f}")

    # Phase 6: Final SA polish
    print(f"\n=== Phase 6: Final SA polish ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        sa_time = min(120.0, max(20.0, deadline - time.monotonic() - 20))
        polished, polish_score = simulated_annealing_polish(best_seq, sa_time, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  After SA: {best_score:.6f}")
        else:
            print(f"  SA didn't improve (best={best_score:.6f})")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 512
