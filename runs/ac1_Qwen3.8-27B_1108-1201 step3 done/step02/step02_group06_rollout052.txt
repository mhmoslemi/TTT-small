def lp_solve(f, M, time_limit=30):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0."""
    n = len(f)
    # A[k, j] = f[k-j] if 0 <= k-j < n
    A = np.zeros((2 * n - 1, n))

    return None

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        M = np.max(conv)
        
        g = lp_solve(f, M, time_limit=20)
        if g is None:
            # Fallback: use subgradient direction
            sg, _, _, _ = compute_subgradient(f)
            g = np.maximum(0, sum_f/n - sg)
            if np.sum(g) < 1e-10:
                break
        
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        
        g_rescaled = g * (sum_f / sum_g)
        
        # Line search for step size
        best_t = 0.1
        best_score_ls = np.inf
        for t_try in [t, t*0.5, t*0.25, t*0.75, min(0.9, t*1.5)]:
            f_new = (1.0 - t_try) * f + t_try * g_rescaled
            f_new = np.clip(f_new, 0.0, None)
            s = np.sum(f_new)
            if s < 1e-10:
                continue
            f_new = f_new * (sum_f / s)
            sc = local_score(f_new)
            if sc < best_score_ls:
                best_score_ls = sc
                best_t = t_try
        
        f_new = (1.0 - best_t) * f + best_t * g_rescaled
        f_new = np.clip(f_new, 0.0, None)
        s = np.sum(f_new)
        if s < 1e-10:
            t *= 0.5
            continue
        f_new = f_new * (sum_f / s)
        
        score_new = local_score(f_new)
        
        if score_new < best_score - 1e-12:
            f = f_new
            best_score = score_new
            best_f = f.copy()
            stagnant = 0
            t = min(0.8, t * 1.05)
        else:
            f = f_new
            stagnant += 1
            if stagnant > 15:
                t *= 0.7
                stagnant = 0
            if stagnant > 50:
                f = f + 0.05 * rng_local.standard_normal(n)
                f = np.clip(f, 0.0, None)
                f = f * (sum_f / np.sum(f))
                t = 0.3
                stagnant = 0
            if stagnant > 150:
                break
    
    return best_f, best_score

def subgradient_optimize(a_init, time_limit, verbose=False):
    """Subgradient method with adaptive step and restarts."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    sum_a = np.sum(a)
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    # Polyak-style step size
    step = 0.1
    iter_count = 0
    patience = 0
    rng = np.random.default_rng(123)
    
    while time.monotonic() - t0 < time_limit:
        sg, M, peak_lags, conv = compute_subgradient(a)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        
        # Step direction: -sg (we want to minimize max conv)
        # But we need to maintain non-negativity, so use projection
        a_new = a - step * sg / sg_norm
        a_new = np.clip(a_new, 0.0, None)
        new_sum = np.sum(a_new)
        if new_sum < 1e-10:
            step *= 0.5
            continue
        a_new = a_new * (sum_a / new_sum)
        
        score = local_score(a_new)
        iter_count += 1
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            patience = 0
            step = min(0.5, step * 1.01)
        else:
            a = a_new
            patience += 1
            if patience > 50:
                step *= 0.9
                patience = 0
            if patience > 200:
                # Restart with perturbation
                a = a + 0.02 * rng.standard_normal(n)
                a = np.clip(a, 0.0, None)
                a = a * (sum_a / np.sum(a))
                step = 0.1
                patience = 0
        
        if verbose and iter_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD iter={iter_count}, score={score:.6f}, best={best_score:.6f}, t={elapsed:.1f}s")
    
    return best_a, best_score

def multiplicative_update(a_init, time_limit, verbose=False):
    """Multiplicative update method focused on peak reduction."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    sum_a = np.sum(a)
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    alpha = 0.3
    iter_count = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        M = np.max(conv)
        eps = 1e-6 * M
        peak_lags = np.where(conv >= M - eps)[0]
        
        # Compute "pressure" at each position
        pressure = np.zeros(n)
        for k in peak_lags:
            for i in range(n):
                if 0 <= k - i < n:
                    pressure[i] += a[k - i]
        
        # Multiplicative update: reduce where pressure is high
        a_new = a / (1.0 + alpha * pressure / (np.max(pressure) + 1e-15))
        a_new = np.clip(a_new, 1e-10, None)
        a_new = a_new * (sum_a / np.sum(a_new))
        
        score = local_score(a_new)
        iter_count += 1
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            # Slowly increase alpha
            alpha = min(0.8, alpha * 1.005)
        else:
            a = a_new
            alpha = max(0.05, alpha * 0.95)
            if iter_count % 500 == 0 and time.monotonic() - t0 > 60:
                # Perturbation
                a = a * np.exp(0.01 * rng.standard_normal(n))
                a = a * (sum_a / np.sum(a))
        
        if verbose and iter_count % 200 == 0:
            print(f"    MU iter={iter_count}, score={score:.6f}, best={best_score:.6f}, "
                  f"alpha={alpha:.3f}, t={time.monotonic()-t0:.1f}s")
    
    return best_a, best_score

def sa_polish(f_init, time_limit, rng):
    """Simulated annealing for final polishing."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    current_score = best_score
    t0 = time.monotonic()
    temperature = 0.005
    cooling = 0.9998
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        candidate = f.copy()
        r = rng.random()
        
        if r < 0.3:
            # Peak-targeted: reduce elements contributing to peak
            conv = fft_convolve(f, f)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = f[i] * f[j]
            top_k = max(3, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=top_k))
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = min(8, len(other_idx))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_inc))
        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 5))))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.75:
            # Multi-block
            for _ in range(3):
                width = max(1, int(rng.integers(1, max(2, n // 10))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.9:
            # Asymmetric shift
            shift = int(rng.integers(1, max(2, n // 25)))
            candidate = np.roll(f, shift)
            candidate *= np.exp(rng.normal(0.0, 0.005, size=n))
        else:
            # Global small perturbation
            candidate *= np.exp(rng.normal(0.0, 0.008, size=n))
        
        candidate = np.clip(candidate, 0.0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (sum_f / cand_sum)
        
        new_score = local_score(candidate)
        iter_count += 1
        
        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
            f = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best_f, best_score

def gradient_polish(f_init, time_limit, verbose=False):
    """Adam gradient descent on log parameterization with annealed softmax."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f_safe = np.clip(f_init, 1e-10, None)
    theta = np.log(f_safe / np.sum(f_safe))
    
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.5
    step_count = 0
    best_theta = theta.copy()
    best_score = local_score(f_init)
    t0 = time.monotonic()
    tau_start = 10.0 / n
    tau_end = 0.2 / n
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac
        tau = max(tau, 0.05 / n)
        
        t_shifted = theta - np.max(theta)
        e = np.exp(t_shifted)
        a = e / np.sum(e) * S
        
        conv = fft_convolve(a, a)
        conv_max = np.max(conv)
        shifted_conv = (conv - conv_max) / tau
        # Clip to avoid overflow
        shifted_conv = np.clip(shifted_conv, -50, 50)
        exp_conv = np.exp(shifted_conv)
        sum_exp = np.sum(exp_conv)
        w = exp_conv / sum_exp
        
        rev_a = a[::-1]
        corr = fft_convolve(rev_a, w)
        g_a = 2.0 * corr[n-1:2*n-1]
        
        inner = np.dot(a, g_a) / S
        g_theta = a * (g_a - inner)
        
        step_count += 1
        momentum1 = beta1 * momentum1 + (1.0 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1.0 - beta2) * (g_theta ** 2)
        m1_hat = momentum1 / (1.0 - beta1 ** step_count)
        m2_hat = momentum2 / (1.0 - beta2 ** step_count)
        lr = base_lr / np.sqrt(1.0 + 0.05 * step_count)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        t_shifted = theta - np.max(theta)
        e_new = np.exp(t_shifted)
        a_check = e_new / np.sum(e_new) * S
        score = local_score(a_check)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()
        
        if verbose and step_count % 300 == 0:
            print(f"    Grad step={step_count}, score={score:.6f}, best={best_score:.6f}")
    
    t_shifted = best_theta - np.max(best_theta)
    a_best = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
    return a_best, local_score(a_best)

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining parametric search, subgradient methods,
    LP-based direction finding, gradient polishing, and simulated annealing.
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s_prov = local_score(provided_seq)
        print(f"Provided sequence: n={len(provided_seq)}, score={s_prov:.6f}")
        if s_prov < best_score:
            best_score = s_prov
            best_seq = provided_seq.copy()
    
    n_base = len(provided_seq) if provided_seq is not None else 256
    
    # === Phase 1: Parametric + diverse initialization (120s) ===
    print("\n" + "=" * 60)
    print("Phase 1: Parametric search and diverse initialization")
    print("=" * 60)
    
    p1_time = min(120.0, (deadline - time.monotonic()) * 0.08)
    print(f"  Time budget: {p1_time:.0f}s, base n={n_base}")
    
    t_start = time.monotonic()
    
    # Try multiple n values
    n_candidates = []
    if provided_seq is not None:
        n_candidates.append(n_base)
    n_candidates.extend([256, 384, 512])
    n_candidates = list(set(n_candidates))
    
    for nc in n_candidates:
        if time.monotonic() > deadline - 60:
            break
        remaining_p1 = (t_start + p1_time) - time.monotonic()
        if remaining_p1 < 10:
            break
        a_param, s_param = parametric_search(nc, remaining_p1 * 0.7, rng)
        if a_param is not None:
            # Quick local polish
            a_quick, s_quick = subgradient_optimize(a_param, min(20, remaining_p1 * 0.2), verbose=False)
            if s_quick < best_score:
                best_score = s_quick
                best_seq = a_quick.copy()

            else:
                print(f"  n={nc}: parametric+SGD score={s_quick:.6f}")

            remaining_p1 = (t_start + p1_time) - time.monotonic()

                if s_interp < best_score:

                    best_seq = a_interp.copy()
                    print(f"  n={nc}: interp+SGD score={s_interp:.6f} (NEW BEST)")

    if provided_seq is not None:
        remaining_p1 = (t_start + p1_time) - time.monotonic()
        if remaining_p1 > 20:
            a_sgd, s_sgd = subgradient_optimize(provided_seq, min(40, remaining_p1), verbose=False)
            print(f"  Provided+SGD: score={s_sgd:.6f}")
            if s_sgd < best_score:
                best_score = s_sgd
                best_seq = a_sgd.copy()
