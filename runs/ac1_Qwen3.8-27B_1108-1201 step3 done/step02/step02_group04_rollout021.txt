Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) LP-guided direction finding, (2) accelerated subgradient method
with the mirror-gradient insight, (3) diversification via perturbation and

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def make_lp_structure(n):
    """Precompute sparse matrix structure for cross-convolution."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, rows, cols, n, timeout=10):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_optimize(f0, time_limit, verbose=False):
    """LP-based alternating optimization with line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    rows, cols = make_lp_structure(n)

    t0 = time.monotonic()
    t = 1.0
    patience = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        g = solve_lp(f, M, rows, cols, n, timeout=min(10, max(2, time_limit - (time.monotonic() - t0))))
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        if np.max(np.abs(g - f)) < 1e-14:
            break

        improved = False
        for t_try in [t, 0.7*t, 0.5*t, 0.3*t, 0.2*t, 0.1*t, 0.05*t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-14:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.05)
                improved = True
                break

            patience += 1
            t *= 0.85
            if patience > 40:

    return best_f, best_obj


def subgradient_optimize(f0, time_limit, verbose=False):
    """Accelerated subgradient method exploiting the mirror-gradient structure.
    
    Key insight: d(max_conv)/da[j] at k* is 2*a[k*-j] (if valid).
    So the subgradient is a "reversed" version of a around k*/2.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f /= s

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    # Nesterov momentum
    y = f.copy()
    beta = 0.9
    t_param = 1.0
    eta = 1.0 / np.sqrt(n)  # initial step size
    
    t0 = time.monotonic()
    patience = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_param += 1.0
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        k_star = int(np.argmax(conv_y))
        
        # Subgradient: sg[j] = 2*y[k_star - j] if 0 <= k_star - j < n
        sg = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        # sg[j] = 2*y[k_star - j] for j in [k_star-(n-1), k_star] intersect [0, n-1]
        # Let i = k_star - j, so j = k_star - i
        # i ranges from k_star-(n-1) to k_star, intersected with [0, n-1]
        # j ranges from k_star to k_star-(n-1)... 
        # More simply: for j in range(n): if 0 <= k_star-j < n: sg[j] = 2*y[k_star-j]
        idx = np.arange(n)
        valid = (k_star - idx >= 0) & (k_star - idx < n)
        sg[valid] = 2.0 * y[k_star - idx[valid]]
        
        # Normalize subgradient (it's on the order of 2*max(y) ~ 2/n)
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        
        # Adaptive step size: decrease if stuck
        step = eta / t_param  # Polyak-like decay
        
        # Update: y = y - step * sg, project onto simplex
        y_new = y - step * sg
        # Project onto probability simplex
        y_new = project_simplex(y_new)
        
        # Nesterov averaging
        f = (1.0 - beta_new / t_param) * y_new + (beta_new / t_param) * f
        # Actually, standard Nesterov is: x_{k+1} = (t_k * x_k + y_k) / (t_k + 1)
        # and y_{k+1} = x_{k+1} - (t_k/(t_k+1)) * (x_{k+1} - x_k)
        # Let me use a simpler momentum version
        
        y = y_new
        beta = beta_new
        
        # Evaluate
        conv_f = fast_convolve(f)
        obj = 2.0 * n * conv_f.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            patience = 0
        else:
            patience += 1
            if patience % 100 == 0:
                eta *= 0.8
            if patience > 1000:
                break

    return best_f, best_obj


def project_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def mirror_polish(f0, time_limit, t_param=50.0, eta0=0.3):
    """Polish with multiplicative weights / mirror descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    eta = eta0
    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        # Weight towards reducing the peak
        scaled = t_param * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum

        # Compute gradient of max_conv using weighted combination
        sg = np.zeros(n)
        thresh = w.max() * 0.001
        active = np.where(w > thresh)[0]
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            # d(conv[k])/da[j] = 2*a[k-j] for j in [k-n+1, k] intersect [0,n-1]
            # So contribution to sg[j] from weight w[k] is w[k] * 2 * f[k-j]
            for j in range(j_start, j_end + 1):
                sg[j] += 2.0 * w[k] * f[k - j]
        
        # Mirror descent update
        lf = -eta * sg
        lf -= lf.max()
        f = f * np.exp(lf)
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-12:
            f = np.ones(n) / n
        else:
            f /= s

        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            patience = 0
        else:
            patience += 1
            if patience % 50 == 0:
                eta *= 0.7
                if eta < 0.01:
                    t_param = min(t_param * 2.0, 1000.0)
                    eta = 0.2
            if patience > 300:
                break

    return best_f, best_obj


def gen_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uni'] = np.ones(n) / n

    for sf in [0.1, 0.15, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf*100)}'] = g

    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    # Bimodal
    w = max(1, n // 5)
    bi = np.zeros(n)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()

    # Cosine
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()

    # Random
    rr = rng.dirichlet(np.ones(n) * 0.5)
    inits['rnd'] = rr

    # Triangle
    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.01)
    inits['tri'] = tri / tri.sum()

    # Parent
    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0)
            if pp.sum() > 1e-10:
                inits['par'] = pp / pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0.0)
            if pp.sum() > 1e-10:
                inits['par'] = pp / pp.sum()

    return inits


def diverse_perturb(f, n_perturbs=5, scale=0.1, rng=None):
    """Create perturbed versions of f for diversification."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(f)
    perturbations = [f.copy()]
    for _ in range(n_perturbs):
        noise = rng.normal(0, scale, size=n)
        f_new = f + noise
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
            perturbations.append(f_new)
    return perturbations
