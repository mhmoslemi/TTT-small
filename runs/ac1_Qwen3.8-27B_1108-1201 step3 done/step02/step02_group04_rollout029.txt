"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated smooth gradient
descent with beta-annealing, coordinate descent polishing, and LP-based refinement.
Key innovation: uses log-sum-exp smooth approximation of max with Nesterov momentum
for fast convergence, combined with coordinate descent for fine-tuning.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def project_simplex(v):
    """Project onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho_val = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_convObjective(a, beta):
    """Compute smooth approximation of 2n*max(conv(a,a)) and gradient w.r.t. a."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    # Log-sum-exp for numerical stability
    shifted = beta * (c - c_max)
    max_shifted = shifted.max()
    log_sum = c_max + max_shifted / beta + np.log(np.sum(np.exp(shifted - max_shifted))) / beta
    # Softmax weights
    p = np.exp(shifted - max_shifted)
    p /= p.sum()
    # Gradient: grad[j] = 2n * sum_k p[k] * a[k-j] (for valid k, j)
    # = 2n * (a ⋆ p)[j]  (cross-correlation)
    fft_size = 1 << (2 * n - 1 + n - 1).bit_length()  # need length at least (2n-1) + n - 1
    # Actually for cross-correlation of a (length n) and p (length 2n-1):
    # result length is n + (2n-1) - 1 = 3n - 2
    # We need indices 0 to n-1
    fft_size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(np.conj(A) * P, fft_size)
    grad = 2.0 * n * cross[:n]
    obj = 2.0 * n * log_sum
    return obj, grad

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated smooth gradient
descent with log-sum-exp beta-annealing, coordinate descent polishing, and LP refinement.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def smooth_obj_and_grad(a, beta):
    """Compute smooth approximation of 2n*max(conv(a,a)) and gradient."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    shifted = beta * (c - c_max)
    m = shifted.max()
    log_sum = c_max + (m + np.log(np.sum(np.exp(shifted - m)))) / beta
    p = np.exp(shifted - m)
    p /= p.sum()
    # Gradient: 2n * (a cross-correlated with p), first n elements
    fft_size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(np.conj(A) * P, fft_size)
    grad = 2.0 * n * cross[:n]
    return 2.0 * n * log_sum, grad

def make_lp_structure(n):
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp_for_direction(f, M, rows, cols, n, timeout=10):
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-13:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.1)
                improved = True
                break
        
        if not improved:
            patience += 1
            t *= 0.8
            if patience > 30:
                break
    
    return best_f, best_obj


def coord_descent_refine(a, time_limit):
    """Coordinate descent refinement."""
    n = len(a)
    a = a.astype(np.float64).copy()
    t0 = time.monotonic()
    c = fast_convolve(a)
    s = a.sum()
    obj = 2.0 * n * c.max() / (s * s)
    best_obj = obj
    best_a = a.copy()
    rng = np.random.default_rng(123)
    
    while time.monotonic() - t0 < time_limit:
        sweep_improved = False
        # Random order for diversity
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            old = a[i]
            if old < 1e-15:
                candidates = [0.0, s * 0.005, s * 0.01, s * 0.02]
            else:
                candidates = [0.0, old * 0.25, old * 0.5, old * 0.75, old, old * 1.25, old * 1.5, old * 2.0]
            
            best_t = old
            best_c = c.max()
            
            for t in candidates:
                if t < 0:
                    continue
                a[i] = t
                new_s = a.sum()
                if new_s < 1e-10:
                    a[i] = old
                    continue
                new_c = fast_convolve(a)
                new_obj = 2.0 * n * new_c.max() / (new_s * new_s)
                a[i] = old  # restore
                if new_obj < best_c:
                    best_c = new_obj
                    best_t = t
            
            if best_t != old:
                a[i] = best_t
                s = a.sum()
                c = fast_convolve(a)
                obj = 2.0 * n * c.max() / (s * s)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                sweep_improved = True
        
        if not sweep_improved:
            break
    
    return best_a, best_obj


def gen_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    for sf in [0.1, 0.15, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits[f'gauss_{sf}'] = g / g.sum()
    
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    
    # Cosine
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cosine'] = cb / cb.sum()
    
    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Random
    rr = rng.dirichlet(np.ones(n) * 0.5)
    inits['random'] = rr
    
    # Triangle
    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.01)
    inits['triangle'] = tri / tri.sum()
    
    # Power律
    pw = x ** 2 + 1
    inits['quad'] = pw / pw.sum()
    
    # Sinc-like
    sinc = np.sinc(x / (n / 4.0))
    sinc = np.maximum(sinc, 0)
    if sinc.sum() > 1e-10:
        inits['sinc'] = sinc / sinc.sum()
    
    # Parent
    if parent is not None:
        if parent.size >= n:
            p = np.maximum(parent[:n], 0.0)
            if p.sum() > 1e-10:
                inits['parent'] = p / p.sum()
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), np.maximum(parent, 0))
            p = np.maximum(p, 0)
            if p.sum() > 1e-10:
                inits['parent'] = p / p.sum()
    
    return inits


def random_perturb_search(a, time_limit, rng):
    """Try random perturbations of a to find improvements."""
    n = len(a)
    best_a = a.copy()
    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max() / (a.sum() ** 2)
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Random perturbation
        scale = rng.uniform(0.01, 0.1)
        noise = rng.randn(n) * scale
        a_new = np.maximum(best_a + noise, 0)
        if a_new.sum() < 1e-10:
            continue
        c_new = fast_convolve(a_new)
        obj_new = 2.0 * n * c_new.max() / (a_new.sum() ** 2)
        if obj_new < best_obj - 1e-12:
            best_a = a_new.copy()
            best_obj = obj_new
    
    return best_a, best_obj
