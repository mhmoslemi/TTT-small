    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated mirror descent
with Bregman projections, LP-based direction finding, and local search.
Combines fast iterative methods with principled optimization to escape
local minima.

import numpy as np
import scipy.optimize
import time
import math


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    """Objective: 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def mirror_descent_step(a, grad, eta):
    """Mirror descent step with negative entropy."""
    log_a = np.log(np.maximum(a, 1e-300))
    new_log_a = log_a - eta * grad
    new_a = np.exp(new_log_a - new_log_a.max())
    s = new_a.sum()
    return new_a / s


def compute_subgrad(a, k_star):
    """Compute subgradient of max_k conv(a,a)[k] at peak k_star."""
    n = len(a)
    grad = np.zeros(n)
    for m in range(n):
        j = k_star - m
        if 0 <= j < n:
            grad[m] = 2.0 * a[j]
    return grad

    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    a_prev = a.copy()

    best_obj = obj_norm(a)

    eta = 1.0 / np.sqrt(n)  # initial step size

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def simplex_project(v):

def lp_direction(f, M):
    """
    Find direction g >= 0, sum(g)=1, maximizing sum(g) subject to
    conv(f,g) <= M. Returns normalized g.

    n = len(f)
    # Build constraint matrix
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n))
    for j in range(n):
        for k in range(n):
            idx = j + k
            if idx < num_constraints:
                A[idx, j] = f[k]
    
    c = -np.ones(n)
    b = np.full(num_constraints, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()

def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    """Objective assuming sum(a)=1: 2n*max(conv(a,a))."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def simplex_project(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgrad(a, k_star):
    """Compute subgradient of max_k conv(a,a)[k] at peak position k_star.
    grad[m] = 2 * a[k_star - m] with zero padding."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    if lo <= hi:
        grad[lo:hi + 1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]
    return grad

def quadratic_local_search(a0, time_limit, max_pairs=50):
    """
    Local search using analytical quadratic optimization for 2-point exchanges.
    For moving mass t from i to j: conv[k] changes quadratically in t.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    best_obj = obj_norm(a)
    t0 = time.monotonic()
    stuck = 0
    max_stuck = 800

    while time.monotonic() - t0 < time_limit and stuck < max_stuck:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))

        # Compute contribution of each position to the peak
        contrib = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            j = k_star - i
            contrib[i] = a[i] * a[j]

        # Find danger (high contribution) and safe (low contribution) positions
        sorted_idx = np.argsort(contrib)[::-1]
        top_k = min(max_pairs, n // 2)
        danger = sorted_idx[:top_k]
        safe = sorted_idx[n - top_k:n]

        found_improvement = False
        for di in danger:
            if found_improvement:
                break
            ai = a[di]
            if ai < 1e-14:
                continue
            for si in safe:
                if found_improvement:
                    break
                if di == si:
                    continue
                if a[si] >= ai:
                    continue

                # Quadratic coefficients for conv[k_star]
                k_di = k_star - di
                k_si = k_star - si

                a_k_di = a[k_di] if 0 <= k_di < n else 0.0
                a_k_si = a[k_si] if 0 <= k_si < n else 0.0

                B = 2.0 * (a_k_si - a_k_di)
                A_quad = (1.0 if k_star == 2 * di else 0.0) \
                       - 2.0 * (1.0 if k_star == di + si else 0.0) \
                       + (1.0 if k_star == 2 * si else 0.0)

                t_max = ai * 0.95

                if A_quad > 1e-15:
                    t_opt = -B / (2.0 * A_quad)
                elif B < 0:
                    t_opt = t_max
                else:
                    continue

                t_opt = max(1e-10, min(t_opt, t_max))

                # Try optimal and nearby values
                candidates_t = [t_opt, t_opt * 0.7, t_opt * 1.3, t_opt * 0.4]
                for t_try in candidates_t:
                    t_try = max(1e-10, min(t_try, t_max))
                    a_trial = a.copy()
                    a_trial[di] -= t_try
                    a_trial[si] += t_try
                    if a_trial[di] < -1e-12:
                        continue
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial /= a_trial.sum()
                    o = obj_norm(a_trial)
                    if o < best_obj - 1e-13:
                        best_obj = o
                        a = a_trial
                        found_improvement = True
                        break

        if found_improvement:
            stuck = 0
        else:
            stuck += 1

    return a, best_obj

def lp_direction(f, M):
    """Find g >= 0, sum(g)=1, maximizing sum(g) subject to conv(f,g) <= M."""
    n = len(f)
    num_constraints = 2 * n - 1

    # Build constraint matrix efficiently
    A = np.zeros((num_constraints, n))
    for j in range(n):
        end = min(num_constraints, j + n)
        A[j:end, j] = f[:end - j]

    c = -np.ones(n)
    b = np.full(num_constraints, M)
    bounds = [(0.0, None)] * n

    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                      method='highs', options={'time_limit': 60})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def lp_line_search(f, g):
    """Golden section search for best mixing parameter t."""
    def eval_t(t):
        a = (1.0 - t) * f + t * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            return np.inf
        a /= s
        return obj_norm(a)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = eval_t(c)
    fd = eval_t(d)
    for _ in range(35):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = eval_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = eval_t(d)
    t = (lo + hi) / 2.0
    return t, eval_t(t)

    b = np.zeros(n)
    m = n // 2
    b[:m] = np.exp(-0.5 * ((x[:m] - 0.25 * n) / (0.15 * n)) ** 2)
    b[m:] = np.exp(-0.5 * ((x[m:] - 0.75 * n) / (0.15 * n)) ** 2)
    b = np.maximum(b, 0)
    if b.sum() > 0:
        inits['bimodal'] = b / b.sum()

    # Random (concentrated)
    r = rng.dirichlet(np.ones(n) * 0.15)
    inits['random'] = r

    # Random (spread)
    r2 = rng.dirichlet(np.ones(n) * 1.0)
    inits['random_spread'] = r2

    # Parent
    p = get_parent(n)
    if p is not None:
        inits['parent'] = p
        # Perturbed parents
        for scale in [0.005, 0.02, 0.05]:
            pert = p + scale * rng.standard_normal(n)
            pert = np.maximum(pert, 0.0)
            if pert.sum() > 1e-10:
                inits['parent_{:.3f}'.format(scale)] = pert / pert.sum()

    return inits


def hybrid_optimize(a0, time_limit, use_lp=False):
    """
    Hybrid optimization: alternate between PGD and mirror descent,
    with optional LP restarts when stuck.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    best_obj = obj_norm(a)
    t0 = time.monotonic()

    # Alternate between methods
    method_idx = 0
    methods = [pgd_optimize, mirror_descent_optimize]

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5.0:
            break

        chunk_time = min(remaining, max(5.0, remaining * 0.5))
        method = methods[method_idx % len(methods)]
        method_idx += 1

        a_opt, obj_opt = method(a, time_limit=chunk_time)

        if obj_opt < best_obj - 1e-13:
            best_obj = obj_opt
            best_a = a_opt.copy()

        # Try LP restart if stuck
        if use_lp and (time.monotonic() - t0) > time_limit * 0.3:
            remaining = time_limit - (time.monotonic() - t0)
            if remaining > 30.0:
                conv = fast_convolve(best_a)
                M = conv.max()
                g = lp_direction(best_a, M)
                if not np.allclose(g, best_a, atol=1e-8):
                    t, obj_t = lp_line_search(best_a, g)
                    a_lp = (1 - t) * best_a + t * g
                    a_lp = np.maximum(a_lp, 0.0)
                    ls = a_lp.sum()
                    if ls > 1e-12:
                        a_lp /= ls
                        if obj_t < best_obj - 1e-13:
                            best_obj = obj_t
                            best_a = a_lp.copy()
                            # Continue from LP result
                            a = a_lp.copy()
                    # Use remaining time for PGD from LP result
                    rem = time_limit - (time.monotonic() - t0)
                    if rem > 10.0:
                        a_opt2, obj_opt2 = pgd_optimize(a_lp, time_limit=rem)
                        if obj_opt2 < best_obj - 1e-13:
                            best_obj = obj_opt2
                            best_a = a_opt2.copy()

        a = best_a.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Combines Nesterov PGD, mirror descent, quadratic
    local search, and LP-based direction finding across multiple sequence
    lengths and initializations. Three-phase strategy: scan, deep optimize, polish.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(10.0, float(budget_s))
    deadline = time.monotonic() + total_budget - 30.0

    best_sequence = None
    best_score = float('inf')

    lengths = [256, 384, 512, 640]

    # Phase 1: Quick scan (40% of budget)
    phase1_frac = 0.40
    print("=== Phase 1: Quick scan ({} lengths, multiple inits) ===".format(len(lengths)))

    all_inits = {}
    for n in lengths:
        all_inits[n] = make_inits(n, rng)

    configs = []
    for n in lengths:
        for name, a0 in all_inits[n].items():
            if name == 'parent':
                pri = 0
            elif 'parent' in name:
                pri = 1
            else:
                pri = 2
            configs.append((pri, n, name, a0))
    configs.sort(key=lambda x: (x[0], x[1]))

    n_configs = len(configs)
    phase1_time = (total_budget - 30.0) * phase1_frac
    time_per_scan = max(5.0, phase1_time / max(1, n_configs))

    phase1_results = []
    for i, (pri, n, name, a0) in enumerate(configs):
        remaining = deadline - time.monotonic()
        if remaining < 90.0:
            break
        t_alloc = min(time_per_scan, 45.0, remaining - 90.0)
        if t_alloc < 3.0:
            break

        # Use PGD for speed
        a_opt, obj_opt = pgd_optimize(a0, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score):
            phase1_results.append((score, n, name, a_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print("  [{}] n={}, {}, score={:.6f}".format(
                    time.monotonic() - (deadline - total_budget + 30.0), n, name, score))

    print("  Phase 1 best: {:.6f}".format(best_score))

    # Phase 2: Deep optimization (35% of budget)
    phase2_frac = 0.35
    print("=== Phase 2: Deep optimization (hybrid + LP) ===")

    phase1_results.sort(key=lambda x: x[0])
    top_candidates = []
    seen = set()
    for score, n, name, a_opt in phase1_results:
        key = (n, name)
        if key not in seen:
            top_candidates.append((n, name, a_opt))
            seen.add(key)
        if len(top_candidates) >= 8:
            break

    if best_sequence is not None:
        top_candidates.append((len(best_sequence), 'best', best_sequence.copy()))

    n_deep = len(top_candidates)
    phase2_time = (total_budget - 30.0) * phase2_frac
    time_per_deep = max(20.0, phase2_time / max(1, n_deep))

    for n, name, a0 in top_candidates:
        remaining = deadline - time.monotonic()
        if remaining < 60.0:
            break
        t_alloc = min(time_per_deep, remaining - 30.0)
        if t_alloc < 10.0:
            break

        # Hybrid optimization with LP
        a_opt, obj_opt = hybrid_optimize(a0, time_limit=t_alloc, use_lp=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print("  Deep: n={}, {}, score={:.6f}".format(n, name, score))

    # Phase 3: Intensive polish (25% of budget)
    print("=== Phase 3: Intensive polish ===")
    if best_sequence is not None:
        polish_end = deadline - 10.0
        polish_iter = 0
        while time.monotonic() < polish_end:
            remaining = polish_end - time.monotonic()
            if remaining < 10.0:
                break

            # Quadratic local search
            ls_time = min(remaining * 0.35, 50.0)
            a_ls, obj_ls = quadratic_local_search(best_sequence, time_limit=ls_time)
            score = float(evaluate_sequence(a_ls.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_ls.copy()
                print("  Polish QLS [{}]: {:.6f}".format(polish_iter, best_score))

            # Multi-step local search
            remaining = polish_end - time.monotonic()
            if remaining < 10.0:
                break
            ms_time = min(remaining * 0.25, 40.0)
            a_ms, obj_ms = multi_step_local_search(best_sequence, time_limit=ms_time)
            score = float(evaluate_sequence(a_ms.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_ms.copy()
                print("  Polish MSL [{}]: {:.6f}".format(polish_iter, best_score))

            # PGD refinement
            remaining = polish_end - time.monotonic()
            if remaining < 10.0:
                break
            pgd_time = min(remaining * 0.25, 40.0)
            a_pgd, obj_pgd = pgd_optimize(best_sequence, time_limit=pgd_time)
            score = float(evaluate_sequence(a_pgd.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_pgd.copy()
                print("  Polish PGD [{}]: {:.6f}".format(polish_iter, best_score))

            polish_iter += 1

    # Final: try mirror descent from best
    if best_sequence is not None:
        remaining = deadline - time.monotonic()
        if remaining > 15.0:
            a_md, obj_md = mirror_descent_optimize(best_sequence, time_limit=remaining - 5.0)
            score = float(evaluate_sequence(a_md.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_md.copy()
                print("  Final MD: {:.6f}".format(best_score))

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final evaluation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = min(best_score, final_score)

    print("=== FINAL: score={:.6f}, length={} ===".format(best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
