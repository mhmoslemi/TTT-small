import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score(a):
    """Compute the score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def peak_peel(a, n_iters=1000, rng=None):
    """Iteratively reduce peaks in the convolution."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a)
    a = a.astype(np.float64).copy()
    S = np.sum(a)
    best_a = a.copy()
    best_s = score(a)
    
    for it in range(n_iters):
        c = fft_convolve(a, a)
        k_star = int(np.argmax(c))
        
        # Find contributing indices
        I = []
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                I.append(i)
        I = np.array(I)
        I_complement = np.setdiff1d(np.arange(n), I)
        
        if len(I_complement) == 0:
            break
        
        # Compute importance weights
        weights = np.zeros(len(I))
        for idx, i in enumerate(I):
            j = k_star - i
            if i == j:
                weights[idx] = a[i]
            else:
                weights[idx] = 2.0 * a[j]
        
        # Reduce top contributors, increase non-contributors
        p = max(1, len(I) // 4)
        top_idx = np.argsort(weights)[-p:]
        R = I[top_idx]
        
        q = min(p, len(I_complement))
        # Prefer to increase elements that don't contribute to the peak
        Z = rng.choice(I_complement, size=q, replace=False)
        
        # Build direction
        d = np.zeros(n)
        for i in R:
            d[i] = -weights[np.where(I == i)[0][0]]
        for j in Z:
            d[j] = 1.0
        
        # Normalize to preserve sum
        d_sum = np.sum(d)
        if abs(d_sum) < 1e-10:
            continue
        # Adjust to make sum zero
        # Reduce the "to increase" elements slightly
        d[Z] -= d_sum / q
        
        # Line search
        best_t = 0.0
        best_t_score = best_s
        for t in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
            a_trial = a + t * d
            a_trial = np.clip(a_trial, 0.0, None)
            a_trial = a_trial * (S / np.sum(a_trial))
            s_trial = score(a_trial)
            if s_trial < best_t_score:
                best_t_score = s_trial
                best_t = t
        
        if best_t == 0.0:
            # Try smaller steps
            for t in [0.001, 0.005]:
                a_trial = a + t * d
                a_trial = np.clip(a_trial, 0.0, None)
                a_trial = a_trial * (S / np.sum(a_trial))
                s_trial = score(a_trial)
                if s_trial < best_t_score:
                    best_t_score = s_trial
                    best_t = t
        
        if best_t > 0:
            a = a + best_t * d
            a = np.clip(a, 0.0, None)
            a = a * (S / np.sum(a))
            
            if best_t_score < best_s:
                best_s = best_t_score
                best_a = a.copy()
    
    return best_a, best_s
