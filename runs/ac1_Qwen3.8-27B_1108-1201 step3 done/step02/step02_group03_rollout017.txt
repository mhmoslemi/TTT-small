```python
def qp_peak_shave(a, S_target, M_target, top_k=5):
    """
    QP: minimize 0.5 * ||a' - a||^2
    subject to: for top_k peak lags, sum_i a'[i] * a[k-i] <= M_target
                sum(a') = S_target
                a' >= 0
    """
    n = len(a)
    c = np.convolve(a, a)
    # Get top k peak lags
    top_indices = np.argsort(c)[-top_k:][::-1]
    
    def objective(x):
        return 0.5 * np.sum((x - a)**2)
    
    def objective_grad(x):
        return x - a
    
    constraints = []
    # Peak constraints (linearized)
    for k in top_indices:
        # sum_i x[i] * a[k-i] <= M_target
        coeff = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                coeff[i] = a[j]
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, c=coeff, m=M_target: m - np.dot(x, c),
            'jac': lambda x, c=coeff: -c
        })
    # Sum constraint
    constraints.append({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - S_target,
        'jac': lambda x: np.ones(n)
    })
    
    bounds = [(0, None)] * n
    
    result = minimize(objective, a, jac=objective_grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 1000, 'ftol': 1e-10})
    
    if result.success:
        return result.x
    return a
```

```python
"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses progressive LP-based peak shaving, gradient polishing, FFT-accelerated
operations, and multi-start with structured initializations. Key innovation:
iterative LP with tightening peak constraint and smart blending strategy.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to target sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def get_peak(a):
    """Get peak of autocorrelation."""
    return float(np.convolve(a, a).max())


def solve_lp(a, M):
    """
    Solve LP: max sum(g) s.t. conv(a,g) <= M (elementwise), g >= 0.
    Returns optimal g or None.
    """
    n = len(a)
    c_obj = -np.ones(n)

    # Build constraint matrix efficiently
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(j, j + n):
            A[m, j] = a[m - j]

    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def solve_lp_sparse(a, M, top_k=None):
    """
    LP with only top_k peak constraints (faster for large n).
    """
    n = len(a)
    c = np.convolve(a, a)

    if top_k is None or top_k >= 2 * n - 1:
        top_indices = list(range(2 * n - 1))
    else:
        top_indices = list(np.argsort(c)[-top_k:])

    c_obj = -np.ones(n)
    m = len(top_indices)
    A = np.zeros((m, n))
    for idx, k in enumerate(top_indices):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[idx, j] = a[i]

    b = np.full(m, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 20})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def gradient_hard(a):
    """Hard-max gradient of peak autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def gradient_soft(a, tau):
    """Soft-max gradient using weighted peaks."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    return grad


def gradient_descent_phase(a, S_t, deadline, max_iter=30000):
    """Phase 1: Gradient descent with momentum."""
    n = len(a)
    a = normalize(a.copy(), S_t)
    momentum = np.zeros(n)
    decay = 0.9
    lr = 0.3
    best = a.copy()
    best_score = compute_obj(a)
    no_improve = 0

    it = 0
    while time.monotonic() < deadline and it < max_iter:
        it += 1
        lr = 0.3 * (0.9995 ** it)
        lr = max(lr, 0.003)

        grad = gradient_hard(a)
        grad -= grad.mean()
        momentum = decay * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a = normalize(a_new, S_t)

        if it % 100 == 0:
            s = compute_obj(a)
            if s < best_score - 1e-9:
                best_score = s
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 10:
                    break

    return best, best_score, it

    lr = 0.05
    no_improve = 0

    it = 0
    while time.monotonic() < deadline and it < max_iter:
        it += 1
        tau = 10.0 + 5.0 * (it / 1000.0)
        lr = 0.05 * (0.9999 ** it)
        lr = max(lr, 0.002)

        grad = gradient_soft(a, tau)
        grad -= grad.mean()
        momentum = decay * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a = normalize(a_new, S_t)

        if it % 200 == 0:
            s = compute_obj(a)
            if s < best_score - 1e-10:
                best_score = s
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 8:
                    break

    return best, best_score, it


def peak_shave_step(a, S_t, top_k=3):
    """
    Direct peak shaving: for the top-k peak lags, reduce contributions
    by redistributing mass from high-contribution to low-contribution entries.
    """
    n = len(a)
    c = np.convolve(a, a)
    peak_indices = np.argsort(c)[-top_k:]

    # Compute contribution weights
    weight = np.zeros(n)
    for k in peak_indices:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                weight[i] += a[j]

    # Redistribute: reduce entries with high weight, increase those with low weight
    # Use a "water-filling" inspired redistribution
    mean_a = a.mean()
    weight_mean = weight.mean()
    if weight_mean < 1e-12:
        return a.copy()

    # Direction: move mass from high-weight to low-weight positions
    direction = weight - weight_mean
    direction -= direction.mean()  # Preserve sum

    # Try several step sizes
    best_a = a.copy()
    best_score = compute_obj(a)
    for alpha in [0.02, 0.05, 0.1, 0.2, 0.03, 0.08]:
        a_new = a - alpha * direction / (np.linalg.norm(direction) + 1e-12) * a.mean()
        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new, S_t)
        s = compute_obj(a_new)
        if s < best_score:
            best_score = s
            best_a = a_new

    return best_a, best_score


def optimize_single(a0, time_budget):
    """
    Full optimization pipeline for a single starting point.
    Returns (best_sequence, best_score).
    """
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    S_t = np.sqrt(2.0 * n)
    a = normalize(a, S_t)

    best = a.copy()
    best_score = compute_obj(a)

    t_start = time.monotonic()
    deadline = t_start + time_budget

    print(f"      n={n}, initial_score={best_score:.6f}")

    # Phase 1: Hard gradient descent (30% of time)
    if time.monotonic() < deadline - 0.2 * time_budget:
        p1_deadline = t_start + time_budget * 0.25
        a, score, it1 = gradient_descent_phase(a, S_t, p1_deadline)
        if score < best_score:
            best_score = score
            best = a.copy()
        print(f"      P1 (hard grad): {score:.6f} ({it1} iters)")

    # Phase 2: Soft gradient (20% of time)
    remaining = deadline - time.monotonic()
    if remaining > 15:
        p2_deadline = time.monotonic() + remaining * 0.3
        a, score, it2 = soft_gradient_phase(a, S_t, p2_deadline)
        if score < best_score:
            best_score = score
            best = a.copy()
        print(f"      P2 (soft grad): {score:.6f} ({it2} iters)")

    # Phase 3: LP-based peak shaving (40% of time)
    remaining = deadline - time.monotonic()
    if remaining > 20:
        a = best.copy()
        p3_deadline = time.monotonic() + remaining * 0.55
        a, score, it3 = lp_phase(a, S_t, p3_deadline)
        if score < best_score:
            best_score = score
            best = a.copy()
        print(f"      P3 (LP): {score:.6f} ({it3} iters)")

    # Phase 4: Gradient polish (remaining time)
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a = best.copy()
        a, score, it4 = gradient_polish(a, S_t, time.monotonic() + remaining * 0.7)
        if score < best_score:
            best_score = score
            best = a.copy()
        print(f"      P4 (polish): {score:.6f} ({it4} iters)")

    return best, best_score


def generate_initializations(n, seed):
    """Generate diverse initializations for given length."""
    rng = np.random.default_rng(seed)
    inits = []
    i = np.arange(n, dtype=np.float64)

    # Structured initializations
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * i / n))
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * i * 2 / n))
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * i * 3 / n))
    inits.append(0.3 + 0.7 * np.abs(np.sin(np.pi * i / n)))
    inits.append(np.exp(-0.5 * ((i - n / 2) / (n * 0.25)) ** 2))
    inits.append(np.exp(-0.5 * ((i - n * 0.3) / (n * 0.12)) ** 2) +
                 np.exp(-0.5 * ((i - n * 0.7) / (n * 0.12)) ** 2))

    # Chirp-like
    inits.append(0.3 + 0.7 * np.abs(np.cos(np.pi * i ** 2 / n)))
    inits.append(0.3 + 0.7 * np.abs(np.cos(2 * np.pi * i ** 2 / (2 * n))))

    # Multi-frequency
    inits.append(0.2 + 0.2 * np.abs(np.sin(2 * np.pi * i * 3 / n)) +
                 0.2 * np.abs(np.sin(2 * np.pi * i * 7 / n)) +
                 0.2 * np.abs(np.sin(2 * np.pi * i * 11 / n)))

    # Random
    inits.append(rng.random(n) * 0.5 + 0.25)
    inits.append(rng.random(n) * 0.3 + 0.35)

    # Smoothed random
    x = rng.random(n)
    kernel = np.ones(5) / 5
    x_smooth = np.convolve(x, kernel, mode='same')
    inits.append(x_smooth / x_smooth.max() * 0.5 + 0.25)

    # Binary-like (rounded random)
    b = (rng.random(n) > 0.5).astype(float) * 0.4 + 0.1
    inits.append(b)

    return inits


def add_previous_best(inits, n, prev_best, seed):
    """Add previous best sequence (adapted to length n) to initializations."""
    if prev_best is None:
        return inits

    if len(prev_best) == n:
        inits.insert(0, prev_best.copy())
        # Perturbed versions
        rng = np.random.default_rng(seed)
        for scale in [0.02, 0.05, 0.1]:
            perturbed = prev_best + rng.normal(0, scale * prev_best.mean(), n)
            inits.append(np.maximum(perturbed, 0))
    else:
        # Pad or truncate
        if n < len(prev_best):
            inits.append(prev_best[:n].copy())
        else:
            padded = np.zeros(n)
            offset = (n - len(prev_best)) // 2
            padded[offset:offset + len(prev_best)] = prev_best
            inits.append(padded)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses multi-phase optimization (gradient, LP with
    progressive tightening, gradient polish) from multiple diverse starting
    points across several sequence lengths. Returns the best sequence found.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s)) - 20.0

    best_seq = None
    best_score = float('inf')

    # Get previous best
    prev_best = None
    try:
        prev_best = np.array(height_sequence_1, dtype=np.float64)
        if prev_best is not None and not np.all(np.isfinite(prev_best)):
            prev_best = None
    except (NameError, TypeError):
        prev_best = None

    if prev_best is not None:
        print(f"Previous best available: n={len(prev_best)}, "
              f"score={compute_obj(prev_best):.6f}")

    # Lengths to try, ordered by priority
    lengths = [384, 448, 512, 576, 640]
    if prev_best is not None:
        pn = len(prev_best)
        if pn not in lengths:
            lengths.insert(0, pn)
        else:
            lengths.remove(pn)
            lengths.insert(0, pn)

    print(f"Starting optimization. Budget={budget_s}s. Lengths to try: {lengths}")

    # Time allocation strategy:
    # - Give the first (previous best) length more time
    # - Give subsequent lengths less time
    # - Reserve 30s for final polish

    time_weights = []
    total_weight = 0
    for idx, n in enumerate(lengths):
        if idx == 0:
            w = 4.0
        elif idx == 1:
            w = 2.5
        else:
            w = 1.5
        time_weights.append(w)
        total_weight += w

    available_time = deadline - t_start - 30.0

    for n_idx, n in enumerate(lengths):
        if time.monotonic() > deadline - 40:
            break

        time_for_n = available_time * time_weights[n_idx] / total_weight
        time_for_n = max(60.0, min(400.0, time_for_n))

        remaining = deadline - time.monotonic() - 30
        time_for_n = min(time_for_n, remaining)
        if time_for_n < 40:
            continue

        print(f"\n=== Length n={n}, time budget={time_for_n:.0f}s ===")

        # Generate initializations
        inits = generate_initializations(n, seed + n_idx * 1000)
        inits = add_previous_best(inits, n, prev_best, seed + n)

        # Limit to top initializations (by initial score)
        scored_inits = []
        for a0 in inits:
            a0 = np.clip(np.asarray(a0, dtype=np.float64), 0, 1000)
            if a0.sum() < 1e-6:
                continue
            a0_norm = normalize(a0)
            s = compute_obj(a0_norm)
            scored_inits.append((s, a0_norm))

        scored_inits.sort(key=lambda x: x[0])
        # Take top 5
        scored_inits = scored_inits[:5]

        print(f"  {len(scored_inits)} initializations, best initial score={scored_inits[0][0]:.6f}")

        # Allocate time among initializations
        n_inits = len(scored_inits)
        time_per_init = time_for_n / n_inits
        # Give more time to better initializations
        for init_idx, (init_score, a0) in enumerate(scored_inits):
            if time.monotonic() > deadline - 40:
                break

            # Better initializations get more time
            if init_idx == 0:
                t_alloc = time_per_init * 1.5
            else:
                t_alloc = time_per_init * 0.85

            t_alloc = min(t_alloc, deadline - time.monotonic() - 30)
            if t_alloc < 25:
                t_alloc = 25
            if time.monotonic() + t_alloc > deadline - 30:
                break

            print(f"  [init {init_idx}] score={init_score:.6f}, time={t_alloc:.0f}s")
            try:
                seq, score = optimize_single(a0, t_alloc)
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(f"    *** NEW BEST: score={score:.6f}, n={n} ***")
            except Exception as e:
                print(f"    Error: {e}")

    # Final polish on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"\n=== Final polish (score={best_score:.6f}, n={len(best_seq)}) ===")
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            try:
                seq, score = optimize_single(best_seq, remaining)
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(f"  After polish: score={score:.6f}")
            except Exception as e:
                print(f"  Polish error: {e}")

    # Validation
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\nFinal result: n={len(best_seq)}, score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 384
```
