"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses improved soft-max gradient descent with adaptive step sizes, peak equalization,
simulated annealing local search, and multi-start strategy with diverse n values.

import numpy as np
import time


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return float('inf')
    c = np.convolve(a, a)
    max_c = float(c.max())
    return float(2.0 * n * max_c / (S * S))


def convolve_fast(a):
    """Compute convolution using FFT for speed."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)[:size]
    return result


def gradient_soft(a, tau):
    """
    Compute gradient of soft-max autocorrelation.
    grad[i] = 2 * sum_k w[k] * a[k-i] for valid k-i.
    Uses FFT for efficiency.
    """
    n = len(a)
    c = convolve_fast(a)
    c_max = c.max()
    c_shifted = c - c_max
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is the cross-correlation of w (length 2n-1) with a (length n)
    # Compute via: convolve(w, a_flipped) and extract appropriate slice
    a_flipped = a[::-1]
    # np.convolve(w, a_flipped, 'full') has length (2n-1) + n - 1 = 3n-2
    # We need: sum_j w[i+j]*a[j] = (w * a_flipped)[i + n - 1]
    conv_wa = np.convolve(w, a_flipped, 'full')
    grad = 2.0 * conv_wa[n-1:n-1+n]
    return grad


def gradient_hard(a):
    """Gradient using hard max (single peak)."""
    n = len(a)
    c = convolve_fast(a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    # grad[i] = 2*a[k_star - i] if 0 <= k_star - i < n
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def gradient_multi_peak(a, top_k=3):
    """Gradient using top-k peaks with equal weighting."""
    n = len(a)
    c = convolve_fast(a)
    top_indices = np.argsort(c)[-top_k:]
    grad = np.zeros(n)
    for k in top_indices:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    grad /= top_k
    return grad


def project_normalize(a, target_sum=1.0):
    """Project onto non-negative simplex with given sum."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) * target_sum / len(a)
        return a
    return a * (target_sum / s)


def backtrack_line_search(a, grad, f_obj, lr_init=0.1, max_iter=30):
    """
    Backtracking line search with projection.
    Minimizes f(a + t * (-grad)) subject to a >= 0, sum(a) = 1.
    """
    n = a.shape[0]
    # Remove mean component to maintain sum constraint
    grad_centered = grad - grad.mean()
    
    lr = lr_init
    f_curr = f_obj
    for _ in range(max_iter):
        a_new = a - lr * grad_centered
        a_new = project_normalize(a_new, 1.0)
        f_new = compute_obj(a_new)
        if f_new < f_curr - 1e-12:
            return a_new, f_new, lr
        lr *= 0.5
        if lr < 1e-10:
            break
    return a, f_curr, lr

def local_search(a_init, deadline, n_perturb=5000):
    """
    Simulated annealing style local search.
    Randomly perturb coordinates and accept if objective improves.
    """
    n = len(a_init)
    a = project_normalize(a_init.copy(), 1.0)
    score = compute_obj(a)
    best = a.copy()
    best_score = score
    
    # Compute which lags are near the peak
    c = convolve_fast(a)
    c_max = c.max()
    # Identify "active" lags (within 5% of max)
    active_lags = np.where(c >= 0.95 * c_max)[0]
    
    step_size = 0.02
    rng_local = np.random.default_rng(12345)
    
    it = 0
    while time.monotonic() < deadline - 3 and it < n_perturb:
        it += 1
        # Pick 1-5 random coordinates to perturb
        n_pert = rng_local.integers(1, 6)
        indices = rng_local.choice(n, size=n_pert, replace=False)
        
        a_trial = a.copy()
        for idx in indices:
            delta = rng_local.normal(0, step_size)
            a_trial[idx] += delta
        a_trial = project_normalize(a_trial, 1.0)
        
        s_trial = compute_obj(a_trial)
        if s_trial < score - 1e-12:
            a = a_trial
            score = s_trial
            if score < best_score - 1e-12:
                best_score = score
                best = a.copy()
            # Decrease step size on improvement
            step_size *= 0.98
        else:
            # Occasionally accept bad moves (annealing)
            if rng_local.random() < 0.05 * step_size:
                a = a_trial
                score = s_trial
        
        # Update active lags periodically
        if it % 500 == 0:
            c = convolve_fast(a)
            c_max = c.max()
            active_lags = np.where(c >= 0.95 * c_max)[0]
            step_size = max(0.005, step_size * 0.9)
    
    return best, best_score


def peak_equalize(a_init, deadline):

    score = compute_obj(a)
    best = a.copy()
    best_score = score

    for it in range(max_iters):
        if time.monotonic() > deadline - 5:
            break
        
        c = convolve_fast(a)
        c_max = c.max()
        # Get top 4 peaks
        top_idx = np.argsort(c)[-4:]
        top_vals = c[top_idx]
        
        if len(top_vals) < 2:
            break
        
        # If the spread is small, we're close to optimal
        if top_vals.max() - top_vals.min() < 1e-4 * c_max:
            break
        
        # Move to reduce the highest peak and increase the lowest
        k_high = top_idx[0]  # position of max
        k_low = top_idx[-1]  # position of min among top 4

        grad_high = np.zeros(n)
        for i in range(n):
            j = k_high - i
            if 0 <= j < n:

        grad_low = np.zeros(n)
        for i in range(n):
            j = k_low - i
            if 0 <= j < n:

            if s_trial < score - 1e-12:
                a = a_trial
                score = s_trial
                if score < best_score:
                    best_score = score
                    best = a.copy()
                break
    
    return best, best_score


def lp_refine(a_init, deadline):
    """
    LP-based refinement: solve max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Then blend g with f.
    """
    from scipy.optimize import linprog
    
    n = len(a_init)
    if n > 512:
        return a_init, compute_obj(a_init)
    
    a = project_normalize(a_init.copy(), 1.0)
    score = compute_obj(a)
    best = a.copy()
    best_score = score
    
    max_iters = 50
    for it in range(max_iters):
        if time.monotonic() > deadline - 5:
            break
        
        c = convolve_fast(a)
        M = float(c.max())
        
        # Build LP: max sum(g) s.t. A@g <= M, g >= 0
        # A[m, j] = a[m-j] if 0 <= m-j < n, else 0
        # For efficiency, build sparsely
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            lo = max(0, j - n + 1)
            hi = min(2 * n - 2, j + n - 1)
            for m in range(lo, hi + 1):
                A[m, j] = a[m - j]
        
        c_obj = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 30})
            if res.status != 0:
                break
            
            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            g_norm = g / g_sum  # normalize to sum=1
            
        except Exception:
            break
        
        # Try blending
        improved = False
        for t_blend in [0.3, 0.2, 0.15, 0.1, 0.05]:
            a_trial = (1.0 - t_blend) * a + t_blend * g_norm
            a_trial = project_normalize(a_trial, 1.0)
            s_trial = compute_obj(a_trial)
            if s_trial < score - 1e-10:
                a = a_trial
                score = s_trial
                best = a.copy()
                best_score = s_trial
                improved = True
                break
        
        if not improved:
            # Small random nudge to escape
            nudge = np.random.default_rng(it).normal(0, 0.01, n)
            a_trial = a + nudge
            a_trial = project_normalize(a_trial, 1.0)
            s_trial = compute_obj(a_trial)
            if s_trial < best_score:
                best = a_trial.copy()
                best_score = s_trial
                a = a_trial
                score = s_trial
            else:
                break
    
    return best, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences of length n."""
    inits = []
    i = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n))
    
    # Cosine modulation (various frequencies)
    for freq in [1, 2, 3, 5, 7, 11]:
        inits.append(0.5 + 0.5 * np.cos(2 * np.pi * freq * i / n))
    
    # Two-sided (high at edges)
    inits.append(1.0 + 0.5 * np.cos(np.pi * i / (n - 1)))
    
    # Gaussian centered
    center = (n - 1) / 2.0
    inits.append(np.exp(-0.5 * ((i - center) / (n * 0.3)) ** 2))
    
    # Two Gaussians
    inits.append(np.exp(-0.5 * ((i - n*0.2) / (n*0.1))**2) + 
                 np.exp(-0.5 * ((i - n*0.8) / (n*0.1))**2) + 0.1)
    
    # Sine-based
    inits.append(np.abs(np.sin(2 * np.pi * 3.0 * i / n)) + 0.2)
    
    # Random
    inits.append(rng.random(n) + 0.1)
    
    return [project_normalize(x, 1.0) for x in inits]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses multi-phase optimization with diverse strategies.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load previous best
    prev_seq = None
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)) and gs.min() >= 0:
            prev_seq = gs
            print(f"Loaded previous sequence: n={len(gs)}, score={compute_obj(gs):.6f}")
    except NameError:
        print("No previous sequence available.")
    
    # Determine n values to try
    n_values = [384, 380, 388, 390, 392, 384]  # 384 appears twice for extra effort
    
    # Build list of (n, init) pairs
    tasks = []
    
    if prev_seq is not None:
        # Use previous best as primary starting point
        for target_n in [384]:
            if len(prev_seq) >= target_n:
                tasks.append((target_n, prev_seq[:target_n].copy()))
            elif len(prev_seq) < target_n:
                # Pad or interpolate
                a_pad = np.zeros(target_n)
                a_pad[:len(prev_seq)] = prev_seq
                tasks.append((target_n, a_pad))
        
        # Also try previous best at its native length
        tasks.append((len(prev_seq), prev_seq.copy()))
    
    # Add diverse inits for each n
    for n in [384, 380, 388]:
        inits = diverse_init(n, rng)
        for init in inits[:3]:  # First 3 diverse inits
            tasks.append((n, init.copy()))
    
    print(f"Total tasks: {len(tasks)}")
    
    # Allocate time to tasks
    # Give more time to promising tasks (those starting from previous best)
    total_time = deadline - t_start - 60  # Reserve 60s for final polish
    
    # Sort: previous-best starts first
    tasks.sort(key=lambda x: (0 if prev_seq is not None and np.array_equal(
        project_normalize(x[1], 1.0), project_normalize(prev_seq[:x[0]], 1.0)[:min(x[0],len(prev_seq))]) else 1,
        x[0]))
    
    seen = set()
    unique_tasks = []
    for task in tasks:
        n, init = task
        key = (n, tuple(init[:8].round(2)))
        if key not in seen:
            seen.add(key)
            unique_tasks.append(task)
    tasks = unique_tasks
    
    print(f"Unique tasks: {len(tasks)}")
    
    # Run optimization on each task
    for idx, (n, init) in enumerate(tasks):
        if time.monotonic() > deadline - 60:
            break
        
        remaining = deadline - time.monotonic() - 60
        if remaining < 30:
            break
        
        # Allocate time based on task priority
        if idx < 3:
            t_alloc = min(300.0, remaining * 0.3)
        elif idx < 6:
            t_alloc = min(200.0, remaining * 0.2)
        else:
            t_alloc = min(100.0, remaining * 0.15)
        
        t_alloc = max(20.0, t_alloc)
        t_deadline = time.monotonic() + t_alloc
        
        print(f"\n--- Task {idx}: n={n}, time_budget={t_alloc:.0f}s ---")
        t_task_start = time.monotonic()
        
        # Phase 1: Soft gradient
        a1, s1 = optimize_gradient(init, t_deadline * 0.4, method='soft')
        print(f"  Soft grad: {s1:.6f}")
        
        # Phase 2: Hard gradient
        t_phase2 = min(t_deadline - time.monotonic(), t_alloc * 0.3)
        if t_phase2 > 10:
            a2, s2 = optimize_gradient(a1, time.monotonic() + t_phase2, method='hard')
            print(f"  Hard grad: {s2:.6f}")
            if s2 < s1:
                a1, s1 = a2, s2
        
        # Phase 3: Multi-peak gradient
        t_phase3 = min(t_deadline - time.monotonic(), t_alloc * 0.2)
        if t_phase3 > 10:
            a3, s3 = optimize_gradient(a1, time.monotonic() + t_phase3, method='multi')
            print(f"  Multi-peak: {s3:.6f}")
            if s3 < s1:
                a1, s1 = a3, s3
        
        # Phase 4: Peak equalization
        t_phase4 = min(t_deadline - time.monotonic(), t_alloc * 0.15)
        if t_phase4 > 10:
            a4, s4 = peak_equalize(a1, time.monotonic() + t_phase4)
            print(f"  Peak eq: {s4:.6f}")
            if s4 < s1:
                a1, s1 = a4, s4
        
        # Phase 5: Local search
        t_phase5 = min(t_deadline - time.monotonic(), t_alloc * 0.15)
        if t_phase5 > 10:
            a5, s5 = local_search(a1, time.monotonic() + t_phase5)
            print(f"  Local search: {s5:.6f}")
            if s5 < s1:
                a1, s1 = a5, s5
        
        # Phase 6: LP refinement (only for smaller n)
        if n <= 512:
            t_phase6 = min(t_deadline - time.monotonic(), t_alloc * 0.15)
            if t_phase6 > 10:
                a6, s6 = lp_refine(a1, time.monotonic() + t_phase6)
                print(f"  LP refine: {s6:.6f}")
                if s6 < s1:
                    a1, s1 = a6, s6
        
        # Update global best
        if s1 < best_score:
            best_score = s1
            best_seq = a1.copy()
            print(f"  *** New global best: {best_score:.6f} (n={n}) ***")
        
        elapsed = time.monotonic() - t_task_start
        print(f"  Task {idx} done in {elapsed:.1f}s, best_score={best_score:.6f}")
    
    # --- Final polish phase ---
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n--- Final polish (current best: {best_score:.6f}) ---")
        n_best = len(best_seq)
        t_polish = deadline - time.monotonic() - 20
        
        # Multiple short refinement passes
        a_curr = best_seq.copy()
        s_curr = best_score
        
        for polish_round in range(5):
            if time.monotonic() > deadline - 25:
                break
            t_round = min(t_polish / (5 - polish_round), 60.0)
            
            # Local search
            a_ls, s_ls = local_search(a_curr, time.monotonic() + t_round * 0.5, 
                                       n_perturb=int(2000 * t_round / 30))
            if s_ls < s_curr - 1e-10:
                a_curr, s_curr = a_ls, s_ls
            
            # Peak equalization
            a_pe, s_pe = peak_equalize(a_curr, time.monotonic() + t_round * 0.3)
            if s_pe < s_curr - 1e-10:
                a_curr, s_curr = a_pe, s_pe
            
            # Short gradient
            if t_round > 15:
                a_g, s_g = optimize_gradient(a_curr, time.monotonic() + t_round * 0.2, method='soft')
                if s_g < s_curr - 1e-10:
                    a_curr, s_curr = a_g, s_g
            
            if s_curr < best_score:
                best_score = s_curr
                best_seq = a_curr.copy()
                print(f"  Polish round {polish_round}: {best_score:.6f}")
    
    # --- Final validation ---
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\nFinal result: n={len(best_seq)}, score={final_score:.6f}")
        
        # Ensure all values are valid (non-negative, finite)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq = np.clip(best_seq, 0, 1000.0)
        
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        
        return [float(x) for x in best_seq]
    
    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 384
