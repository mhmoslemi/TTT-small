def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution a * b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    result = np.fft.irfft(A * B, fft_size)
    return result[:size]

def mirror_descent_fast(f0, time_limit, tau0=0.05, tau_rate=0.995, eta0=0.5, momentum0=0.9):
    """Fast mirror descent with momentum and adaptive temperature.
    Works on probability simplex (sum=1, non-negative)."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    momentum_buf = np.zeros(n)
    tau = tau0
    eta = eta0
    momentum = momentum0
    patience = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        c_max = conv.max()
        
        if c_max < 1e-15:
            break
        
        # Softmax weights for peak approximation
        scaled = conv / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum
        
        # Gradient: g[j] = sum_k w[k] * a[k-j] + a[j] * sum(w)
        # = cross_correlate(w, a)[j] + a[j] (since sum(w) = 1)
        
        # Cross-correlation: (w ⊛ a)[j] = sum_k w[k] * a[k-j]
        # This is conv(w, a_reversed) or we can use FFT
        # Actually: (w ⊛ a)[j] = sum_k w[k] * a[k-j]
        # Let's compute this via FFT: it's the cross-correlation of w and a
        
        # Using the fact that corr(w, a)[j] = sum_k w[k] a[k-j]
        # This equals conv(w, flip(a))[j + n - 1]... let me just compute directly
        # For efficiency, use FFT:
        
        # Actually, the simplest: corr[j] = sum_k w[k] * a[(k-j) mod ...] 
        # But we have linear (not circular) correlation. Let me use the direct formula:
        # corr[j] = sum_{k: 0 <= k-j < n} w[k] * a[k-j]
        # = sum_{m: 0 <= m < n, k = j+m} w[j+m] * a[m]
        # For j >= 0: sum over m from 0 to min(n-1, len(w)-1-j) of w[j+m] * a[m]
        
        # This is actually (a * w)[j] (cross-convolution of a and w, but w has length 2n-1)
        # Let me just compute it via FFT
        
        size = n + len(w) - 1
        fft_size = 1 << (size - 1).bit_length()
        W = np.fft.rfft(w, fft_size)
        A = np.fft.rfft(a, fft_size)
        corr_full = np.fft.irfft(A * W, fft_size)
        # corr_full[k] = sum_i a[i] * w[k-i], which is what we want
        # corr_full[j] = sum_i a[i] * w[j-i] = sum_i a[i] * w[j-i]
        # But we want sum_k w[k] * a[k-j] = sum_m w[j+m] * a[m] (substituting m = k-j)
        # = sum_m a[m] * w[j+m]
        # This is (a * w_flip)[j] where w_flip[m] = w[n-1-m]... 
        # Actually, it's simpler: sum_m a[m] * w[j+m] = (a convolved with w)[j] (with appropriate indexing)
        
        # Let me just use the cross-convolution I defined earlier
        # fast_cross_convolve(a, w) gives c[k] = sum_i a[i] * w[k-i]
        # We want d[j] = sum_k w[k] * a[k-j] = sum_m a[m] * w[m+j]
        # = (a * w)[j] shifted... 
        # Actually d[j] = sum_m a[m] * w[m+j]
        # Let u = m+j, then d[j] = sum_u a[u-j] * w[u] = (a * w)[j] (cross-conv)
        # Wait no. (a * w)[k] = sum_i a[i] w[k-i]. So (a*w)[j] = sum_i a[i] w[j-i].
        # But we want sum_m a[m] w[m+j] = sum_i a[i] w[i+j]
        # Let j' = -j: sum_i a[i] w[i-j'] = (a * w)[j'] ... no
        # sum_i a[i] w[i+j] ... this is (a * w_flipped)[-j] where w_flipped[k] = w[-k]
        # This is getting confusing. Let me just compute it directly for small n or use a different approach.
        
        # Alternative: d[j] = sum_k w[k] a[k-j]
        # For fixed j, this is the dot product of w with a shifted by j.
        # We can compute all j simultaneously using FFT.
        # d = IDFT(FFT(a) * conj(FFT(w)))... no, that's circular correlation.
        
        # Let me just use the cross-convolve function:
        # c = cross_convolve(w, a) gives c[k] = sum_i w[i] a[k-i]
        # So c[j] = sum_i w[i] a[j-i] = sum_i w[i] a[j-i]
        # We want d[j] = sum_k w[k] a[k-j] = sum_i w[j+i] a[i] (letting i = k-j)
        # = sum_i a[i] w[j+i]
        # This is NOT the same as c[j]. c[j] = sum_i w[i] a[j-i].
        
        # Actually, c[j] = sum_i w[i] a[j-i]. Let m = j-i: c[j] = sum_m w[j-m] a[m] = sum_m a[m] w[j-m].
        # And d[j] = sum_m a[m] w[j+m].
        # So d[j] = c[j] with w reversed? Not exactly.
        
        # The simplest approach: since w has length 2n-1 and a has length n,
        # d[j] = sum_{m=0}^{n-1} a[m] * w[j+m] (where w is zero-padded outside [0, 2n-2])
        # This is a valid computation that I can do with a loop or with FFT.
        
        # For n=512, the loop approach: for each j (0 to n-1), compute sum of n terms.
        # Total: n^2 = 262144 operations. That's fast enough.
        # But let me use a smarter approach.
        
        # d[j] = sum_{m=0}^{n-1} a[m] * w[j+m]
        # = cross_correlation of a and w, evaluated at lag j
        # In numpy, np.correlate(w, a, mode='valid') gives this... 
        # Actually np.correlate(w, a, mode='full')[n-1:n-1+n] = what we want?
        # np.correlate(c, a, mode='full')[k] = sum_i c[i] * a[i-k+n-1] (approximately)
        # This is getting too complicated. Let me just use the FFT approach directly.
        
        # d[j] = sum_{m=0}^{n-1} a[m] * w[j+m]
        # Let's define v[k] = w[k] for k = 0..2n-2, 0 otherwise.
        # d[j] = sum_m a[m] v[j+m]
        # This is the cross-correlation of a and v, evaluated at lag j.
        # cross_corr(j) = sum_m a[m] v[m+j]
        # = sum_m a[m] v[m+j]
        # Using FFT: cross_corr = IFFT(FFT(a) * conj(FFT(v)))
        # But this gives circular cross-correlation. For linear, we need to zero-pad.
        
        # Let me just do it the simple way for now (n=512 means n^2 = 262K ops, which is ~0.1ms)
        
        # Actually, for n=512, let me just use a vectorized approach:
        # d[j] = sum_m a[m] * w[j+m]
        # For j=0: d[0] = dot(a, w[0:n])
        # For j=1: d[1] = dot(a, w[1:n+1])
        # etc.
        # This can be done with a sliding dot product, but numpy doesn't have a built-in.
        # Let me use the Toeplitz matrix approach or just the FFT.
        
        # FFT approach for linear cross-correlation:
        # We want d[j] = sum_{m=0}^{n-1} a[m] * w[j+m] for j = 0, 1, ..., n-1
        # This is equivalent to: d = a ⊛ w (cross-correlation)
        # In FFT: compute FFT of a (zero-padded) and FFT of w (zero-padded), multiply by conjugate, IFFT
        
        # Size needed: max(n, len(w)) + min(n, len(w)) - 1 = max(n, 2n-1) + min(n, 2n-1) - 1
        # = (2n-1) + n - 1 = 3n - 2. So fft_size should be >= 3n-2.
        
        fft_size2 = 1 << (3 * n - 2 - 1).bit_length()
        Af = np.fft.rfft(a, fft_size2)
        Wf = np.fft.rfft(w, fft_size2)
        # Cross-correlation: IFFT(Af * conj(Wf))
        corr = np.fft.irfft(Af * np.conj(Wf), fft_size2)
        # corr[k] = sum_m a[m] * w[(m-k) mod fft_size2] (circular)
        # For linear: corr[0:n-1] or corr[fft_size2-n+1:fft_size2]... 
        # Actually, the linear cross-correlation (a ⊛ w)[k] = sum_m a[m] w[m-k]
        # We want sum_m a[m] w[m+j] = (a ⊛ w)[-j] = (a ⊛ w)[N-1-j] for circular of size N
        # Hmm, this is getting complicated. Let me just use a different formulation.
        
        # SIMPLEST APPROACH: Just compute the gradient directly.
        # g[j] = sum_k w[k] * a[k-j] + a[j]
        # The first term: for each j, sum over k where 0 <= k-j < n
        # Let me just compute the cross-convolution of w and a:
        # cross_conv(k) = sum_i w[i] * a[k-i]
        # Then g[j] = cross_conv of a and w... 
        
        # OK let me just think about this more carefully.
        # C[k] = sum_i a[i] a[k-i]
        # dC[k]/da[j] = a[k-j] + a[j] (with a[m]=0 if m out of [0,n-1])
        # 
        # Gradient[j] = sum_k w[k] (a[k-j] + a[j])
        #            = sum_k w[k] a[k-j] + a[j] sum_k w[k]
        #            = T[j] + a[j]  (since sum_k w[k] = 1)
        #
        # where T[j] = sum_k w[k] a[k-j]
        #
        # T[j] = sum_{k: 0<=k-j<=n-1} w[k] a[k-j]
        #      = sum_{m: 0<=m<=n-1} w[m+j] a[m]  (letting m = k-j)
        #      = dot(a, w[j:j+n])  (treating w as zero-padded)
        
        # So T[j] = dot(a, w[j:j+n]) for j = 0, ..., n-1
        # This is a "sliding dot product" which I can compute efficiently.
        
        # Using FFT: T = IFFT(FFT(a) * conj(FFT(w))) evaluated at the right indices
        # For linear cross-correlation: the result has length n + (2n-1) - 1 = 3n-2
        # T[j] for j=0..n-1 corresponds to indices n-1-j in the full correlation... 
        
        # You know what, for n=512, let me just do it with a loop over j.
        # Each T[j] is a dot product of length n, so total is n^2 = 262144 multiply-adds.
        # In numpy, this is essentially instantaneous.
        
        # Vectorized approach using stride or just a loop:
        T = np.zeros(n)
        for j in range(n):
            # T[j] = sum_{m=0}^{n-1} a[m] * w[j+m]
            # w has length 2n-1, so w[j:j+n] is valid for j=0..n-1
            T[j] = np.dot(a, w[j:j+n])
        
        grad = T + a  # Full gradient
        
        # Project onto simplex tangent space
        grad_proj = grad - grad.mean()
        
        # Momentum update
        momentum_buf = momentum * momentum_buf + (1 - momentum) * grad_proj
        
        # Mirror descent step
        lf = -eta * momentum_buf
        lf -= lf.max()  # numerical stability
        a = a * np.exp(lf)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a /= s
        
        # Evaluate
        conv = fast_convolve(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            patience = 0
            # Slightly increase eta if improving
            eta = min(eta * 1.02, 2.0)
        else:
            patience += 1
            if patience % 20 == 0:
                eta *= 0.7
                if eta < 0.01:
                    eta = 0.1
                    momentum = min(momentum * 1.01, 0.99)
            if patience > 200:
                break
        
        # Cool down temperature
        tau = max(tau * tau_rate, 0.001)
        
    return best_a, best_obj


def lp_step(f, rows, cols, n, timeout=10):
    """One LP step: find g minimizing max(f*g) subject to g>=0, sum(g)=1.
    Returns (g, new_M) or (None, None)."""
    vals = np.tile(f, n)
    A_ub_rows = np.concatenate([rows, np.arange(2 * n - 1)])
    A_ub_cols = np.concatenate([cols, np.full(2 * n - 1, n, dtype=np.int32)])
    A_ub_data = np.concatenate([vals, -np.ones(2 * n - 1)])
    A_ub = sparse.csc_matrix((A_ub_data, (A_ub_rows, A_ub_cols)), shape=(2 * n - 1, n + 1))
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    b_ub = np.zeros(2 * n - 1)
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                           bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        g = np.maximum(res.x[:n], 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            return None, None
        g /= g_sum
        return g, res.x[n]
    return None, None


def lp_optimize(f0, rows, cols, n, time_limit, M_init=None):
    """LP-based alternating optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f /= f.sum()
    
    conv = fast_convolve(f)
    M = conv.max() if M_init is None else M_init
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    t = 0.8
    patience = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        g, M_lp = lp_step(f, rows, cols, n, timeout=8)
        if g is None:
            break
        
        # Check convergence
        if np.max(np.abs(g - f)) < 1e-12:
            break
        
        # Line search
        improved = False
        for t_try in [t, 0.7 * t, 0.5 * t, 0.3 * t, 0.1 * t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            if obj_new < best_obj - 1e-13:
                f = f_new
                M = obj_new
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.05)
                improved = True
                break
        
        if not improved:
            patience += 1
            t *= 0.85
            if patience > 40:
                break
    
    return best_f, best_obj


def gen_inits(n, rng):
    """Generate diverse initializations of length n, each summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uni'] = np.ones(n) / n
    
    # Gaussian bumps at different widths
    for sf in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100):02d}'] = g
    
    # Ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    
    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()
    
    # Cosine
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()
    
    # Random Dirichlet (various concentrations)
    for conc in [0.1, 0.3, 1.0, 3.0]:
        rr = rng.dirichlet(np.ones(n) * conc)
        inits[f'rnd{int(conc * 10)}'] = rr
    
    # Triangle
    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.01)
    inits['tri'] = tri / tri.sum()
    
    # Split (half high, half low)
    sp = np.ones(n)
    sp[:n//3] = 3.0
    sp[2*n//3:] = 3.0
    inits['split3'] = sp / sp.sum()
    
    # Periodic (alternating)
    per = np.ones(n)
    per[::2] = 2.0
    inits['alt'] = per / per.sum()
    
    # Multi-scale cosine
    mc = 1.0
    for freq in [2, 3, 5]:
        mc = mc * (0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n))
    inits['mscos'] = mc / mc.sum()
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point: Multi-phase optimization combining mirror descent,
    LP polishing, and random restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Try parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass
    
    # Try different lengths
    lengths = [256, 384, 512, 768]
    
    # Precompute LP structures
    lp_structures = {}
    for n in lengths:
        lp_structures[n] = make_lp_structure(n)
    
    all_results = []
    
    # === Phase 1: Mirror descent scan ===
    print("Phase 1: Mirror descent scan")
    for n in lengths:
        if time.monotonic() > deadline - 120:
            break
        
        inits = gen_inits(n, rng)
        
        if parent is not None:
            if parent.size >= n:
                pp = np.maximum(parent[:n], 0.0)
                if pp.sum() > 1e-10:
                    inits['par'] = pp / pp.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
                pp = np.maximum(pp, 0.0)
                if pp.sum() > 1e-10:
                    inits['par'] = pp / pp.sum()
        
        n_cfgs = len(inits)
        remaining = deadline - time.monotonic() - 120
        time_per = max(5.0, remaining / max(1, n_cfgs * len(lengths)))
        
        for name, a0 in inits.items():
            if time.monotonic() > deadline - 60:
                break
            ta = min(time_per, deadline - time.monotonic() - 60)
            if ta < 3:
                break
            
            f_opt, obj_opt = mirror_descent_fast(a0, time_limit=ta, tau0=0.03, tau_rate=0.998, eta0=0.3)
            score = float(evaluate_sequence(f_opt.tolist()))
            
            if np.isfinite(score):
                all_results.append((score, n, name, f_opt))
                if score < best_score:
                    best_score = score
                    best_seq = f_opt.copy()
                    print(f"  MD: n={n:4d} {name:6s} score={score:.6f}")
    
    print(f"Phase 1 done. Best score: {best_score:.6f}")
    
    # === Phase 2: LP polishing on top results ===
    all_results.sort(key=lambda x: x[0])
    top_k = min(10, len(all_results))
    top = all_results[:top_k]
    
    remaining = deadline - time.monotonic() - 60
    if remaining > 30 and top:
        print(f"Phase 2: LP polishing, {remaining:.0f}s remaining")
        t_each = remaining / max(1, len(top))
        for idx, (sc, n, name, f) in enumerate(top):
            if time.monotonic() > deadline - 30:
                break
            ta = min(t_each, deadline - time.monotonic() - 30)
            if ta < 5:
                break
            
            rows, cols = lp_structures[n]
            f2, obj2 = lp_optimize(f, rows, cols, n, time_limit=ta * 0.7)
            sc2 = float(evaluate_sequence(f2.tolist()))
            if np.isfinite(sc2):
                all_results.append((sc2, n, f'lp_{name}', f2))
                if sc2 < best_score:
                    best_score = sc2
                    best_seq = f2.copy()
                    print(f"  LP: n={n} {name} score={sc2:.6f}")
            
            # Mirror descent polish after LP
            if time.monotonic() < deadline - 15:
                tr = min(ta * 0.3, deadline - time.monotonic() - 15)
                f3, _ = mirror_descent_fast(f2, time_limit=tr, tau0=0.01, tau_rate=0.999, eta0=0.2)
                sc3 = float(evaluate_sequence(f3.tolist()))
                if np.isfinite(sc3):
                    if sc3 < best_score:
                        best_score = sc3
                        best_seq = f3.copy()
                        print(f"  MD-polish: n={n} {name} score={sc3:.6f}")
    
    # === Phase 3: Random restarts from best ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        print("Phase 3: Random restarts")
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 10
        n_restarts = min(20, int(remaining / 10))
        
        for r in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            
            # Add random perturbation
            noise_scale = 0.1 * (1.0 - r / max(1, n_restarts)) + 0.02
            a_pert = best_seq + noise_scale * rng.normal(0, 1, size=n_best)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert /= a_pert.sum()
            
            # Quick mirror descent
            tr = max(3.0, remaining / (n_restarts - r))
            f_new, _ = mirror_descent_fast(a_pert, time_limit=tr, tau0=0.02, tau_rate=0.997, eta0=0.3)
            sc = float(evaluate_sequence(f_new.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = f_new.copy()
                print(f"  Restart {r}: score={sc:.6f}")
    
    # === Phase 4: Deep LP on the very best ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        print("Phase 4: Deep LP on best")
        n_best = len(best_seq)
        rows, cols = lp_structures[n_best]
        tr = deadline - time.monotonic() - 10
        f_deep, obj_deep = lp_optimize(best_seq, rows, cols, n_best, time_limit=tr * 0.6)
        sc_deep = float(evaluate_sequence(f_deep.tolist()))
        if np.isfinite(sc_deep) and sc_deep < best_score:
            best_score = sc_deep
            best_seq = f_deep.copy()
            print(f"  Deep LP: score={sc_deep:.6f}")
        
        # Final mirror descent polish
        if time.monotonic() < deadline - 5:
            tr2 = deadline - time.monotonic() - 5
