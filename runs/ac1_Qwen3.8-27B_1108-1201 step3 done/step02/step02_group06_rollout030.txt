"""
Multi-phase optimization for minimizing normalized peak autocorrelation.
Uses LP-based proximal iterations with line search, adaptive-temperature 
gradient descent, multi-length exploration, and simulated annealing restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def build_constraint_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def lp_solve(f, M):
    n = len(f)
    A = build_constraint_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': 20})
    if result.success:
        return result.x
    return None

def lp_step_with_linesearch(f, time_per_lp=2.0):
    """LP step with backtracking line search."""
    n = len(f)
    sum_f = np.sum(f)
    conv = fft_convolve(f, f)
    M = np.max(conv)
    
    g = lp_solve(f, M)
    if g is None:
        return f, local_score(f)
    
    sum_g = np.sum(g)
    if sum_g < 1e-10:
        return f, local_score(f)
    
    g_rescaled = g * (sum_f / sum_g)
    base_score = local_score(f)
    
    # Try multiple step sizes
    best_f = f.copy()
    best_s = base_score
    for t in [0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1]:
        f_new = (1.0 - t) * f + t * g_rescaled
        f_new = np.clip(f_new, 0.0, None)
        s_new = np.sum(f_new)
        if s_new < 1e-10:
            continue
        f_new = f_new * (sum_f / s_new)
        score = local_score(f_new)
        if score < best_s:
            best_s = score
            best_f = f_new.copy()
    
    return best_f, best_s

def gradient_optimize(f_init, time_limit, tau_init=None, tau_min=None, verbose=False):
    """Gradient descent with adaptive (SA-style) temperature."""
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f_safe = np.clip(f_init, 1e-10, None)
    theta = np.log(f_safe / np.sum(f_safe))
    
    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step_count = 0
    best_theta = theta.copy()
    best_score = local_score(f_init)
    t0 = time.monotonic()
    
    if tau_init is None:
        tau_init = 10.0 / n
    if tau_min is None:
        tau_min = 0.1 / n
    
    tau = tau_init
    best_tau = tau
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        
        # Adaptive temperature: decrease gradually, with occasional increases
        target_tau = tau_min + (tau_init - tau_min) * (1 - elapsed_frac) ** 0.7
        tau = 0.9 * tau + 0.1 * target_tau
        
        # If stuck, increase temperature to escape
        if no_improve > 300:
            tau = min(tau_init, tau * 1.5)
            no_improve = 0
        
        t_shifted = theta - np.max(theta)
        e = np.exp(t_shifted)
        a = e / np.sum(e) * S
        
        conv = fft_convolve(a, a)
        conv_max = np.max(conv)
        shifted_conv = (conv - conv_max) / tau
        exp_conv = np.exp(shifted_conv)
        sum_exp = np.sum(exp_conv)
        w = exp_conv / sum_exp
        
        rev_a = a[::-1]
        corr = fft_convolve(rev_a, w)
        g_a = 2.0 * corr[n - 1:2 * n - 1]
        
        inner = np.dot(a, g_a) / S
        g_theta = a * (g_a - inner)
        
        step_count += 1
        momentum1 = beta1 * momentum1 + (1.0 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1.0 - beta2) * (g_theta ** 2)
        m1_hat = momentum1 / (1.0 - beta1 ** step_count)
        m2_hat = momentum2 / (1.0 - beta2 ** step_count)
        
        # Cosine-annealed learning rate
        lr = 0.5 * (1 + np.cos(np.pi * elapsed_frac)) + 0.05
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        t_shifted = theta - np.max(theta)
        e_new = np.exp(t_shifted)
        a_check = e_new / np.sum(e_new) * S
        score = local_score(a_check)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and step_count % 500 == 0:
            print(f"    Grad step={step_count}, score={score:.6f}, best={best_score:.6f}, tau={tau:.5f}")
    
    t_shifted = best_theta - np.max(best_theta)
    a_best = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
    return a_best, local_score(a_best)

def sa_optimize(f_init, time_limit, rng, verbose=False):
    """Simulated annealing with diverse moves."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    current_score = best_score
    t0a = time.monotonic()
    temperature = 0.005
    cooling = 0.9999
    min_temp = 0.0005
    iter_count = 0
    
    while time.monotonic() - t0a < time_limit:
        # Intra-epoch cooling
        for _ in range(100):
            if time.monotonic() - t0a > time_limit:
                break
            if temperature < min_temp:
                break
            
            candidate = f.copy()
            r = rng.random()
            
            if r < 0.3:
                # Peak-directed move
                conv = fft_convolve(f, f)
                peak_idx = int(np.argmax(conv))
                contribs = np.zeros(n)
                for i in range(n):
                    j = peak_idx - i
                    if 0 <= j < n:
                        contribs[i] = f[i] * f[j]
                top_k = max(3, n // 8)
                top_idx = np.argsort(contribs)[-top_k:]
                candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=top_k))
                other_idx = np.setdiff1d(np.arange(n), top_idx)
                if len(other_idx) > 0:
                    n_inc = min(8, len(other_idx))
                    inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                    candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))
            elif r < 0.5:
                # Block perturbation
                width = max(1, int(rng.integers(1, max(2, n // 5))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
            elif r < 0.7:
                # Two-block perturbation
                for _ in range(2):
                    width = max(1, int(rng.integers(1, max(2, n // 10))))
                    start = int(rng.integers(0, n - width))
                    candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
            elif r < 0.85:
                # Global noise
                candidate *= np.exp(rng.normal(0.0, 0.008, size=n))
            else:
                # Swap-like move
                i1 = int(rng.integers(0, n))
                i2 = int(rng.integers(0, n))
                if i1 != i2:
                    tmp = candidate[i1]
                    candidate[i1] = candidate[i2]
                    candidate[i2] = tmp
            
            candidate = np.clip(candidate, 0.0, None)
            cand_sum = np.sum(candidate)
            if cand_sum < 1e-10:
                continue
            candidate = candidate * (sum_f / cand_sum)
            
            new_score = local_score(candidate)
            iter_count += 1
            
            if new_score < current_score:
                f = candidate
                current_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_f = candidate.copy()
            elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-12)):
                f = candidate
                current_score = new_score
            
            temperature *= cooling
        
        # End of epoch: reset temperature
        if temperature < min_temp:
            temperature = 0.002
            # Add a larger perturbation
            f = np.clip(f + 0.05 * rng.standard_normal(n), 0.0, None)
            f = f * (sum_f / np.sum(f))
            current_score = local_score(f)
    
    return best_f, best_score

def make_init(n, init_type, rng, provided=None):
    t = np.arange(n, dtype=np.float64)
    if init_type == 'provided' and provided is not None:
        if len(provided) == n:
            a = provided.copy()
        else:
            x_old = np.linspace(0, 1, len(provided))
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, provided)
    elif init_type == 'uniform':
        a = np.ones(n)
    elif init_type == 'chirp':
        alpha = 0.3 + 0.4 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / n)
        a = np.clip(a, 0.01, None)
        a += 0.03 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
    elif init_type == 'multitone':
        a = np.ones(n)
        for _ in range(5):
            k = int(rng.integers(1, max(2, n // 4)))
            amp = rng.uniform(0.05, 0.25)
            phase = rng.uniform(0, 2.0 * np.pi)
            a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
        a += 0.02 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
    elif init_type == 'raised_cosine':
        a = 1.0 + 0.3 * np.cos(np.pi * n * t / (2.0 * n))
        a = np.clip(a, 0.01, None)
        a += 0.02 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
    else:  # random
        a = 1.0 + 0.3 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
    return a

def hybrid_optimize(f_init, time_limit, rng, verbose=False):
    """Combined LP + gradient + SA in a single loop."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    elapsed = 0.0
    
    # Phase A: LP for first 40% of time
    lp_end = t0 + 0.4 * time_limit
    while time.monotonic() < lp_end:
        f, s = lp_step_with_linesearch(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()
        if verbose:
            print(f"  LP: score={s:.6f}, best={best_score:.6f}")
    
    # Phase B: Gradient for next 35%
    grad_end = t0 + 0.75 * time_limit
    f = best_f.copy()
    f, s = gradient_optimize(f, grad_end - time.monotonic(), verbose=verbose)
    if s < best_score:

    # Phase C: SA for remaining time
    f = best_f.copy()
    f, s = sa_optimize(f, time_limit - (time.monotonic() - t0), rng, verbose=verbose)
    if s < best_score:

        best_f = f.copy()
    
    return best_f, best_score

Optimized peak autocorrelation minimization using adaptive-temperature gradient
descent with subgradient mixing, multi-length exploration, and SA polishing.
Key innovations: (1) mixing soft-max gradient with exact subgradient, (2)

import numpy as np

from scipy.optimize import linprog

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A

def lp_solve(f, M):
    n = len(f)
    A = build_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 15})
    if res.success:
        return res.x
    return None

def compute_gradient(a, tau):

    n = len(a)
    conv = fft_convolve(a, a)
    conv_max = np.max(conv)
    shifted = (conv - conv_max) / tau
    exp_conv = np.exp(shifted)
    w = exp_conv / np.sum(exp_conv)
    
    rev_a = a[::-1]
    corr = fft_convolve(rev_a, w)
    grad = 2.0 * corr[n - 1:2 * n - 1]
    return grad, conv

def compute_subgradient(a):
    """Compute the exact subgradient at the argmax lag."""
    n = len(a)
    conv = fft_convolve(a, a)
    peak_idx = int(np.argmax(conv))
    grad = np.zeros(n)
    for i in range(n):
        j = peak_idx - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad, conv, peak_idx
