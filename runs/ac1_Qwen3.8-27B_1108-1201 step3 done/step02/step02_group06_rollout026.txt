"""
Multi-phase optimization for minimizing normalized peak autocorrelation.
Uses Adam gradient descent with annealed soft-peak, peak-targeted transfers,
and simulated annealing for polishing. Explores multiple sequence lengths
and initializations.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (sum_a ** 2))


def compute_soft_peak_gradient(f, w):
    """Compute gradient of soft peak. Returns (grad, c_soft, c)."""
    n = len(f)
    S = np.sum(f)
    c = fft_convolve(f, f)
    c_soft = np.dot(w, c)
    
    # g[m] = 2 * sum_j f[j] * w[j+m]
    g = np.zeros(n)
    for m in range(n):
        valid = min(n, 2 * n - 1 - m)
        g[m] = 2.0 * np.dot(f[:valid], w[m:m + valid])
    
    # score = 2n * c_soft / S^2
    # d(score)/df[m] = 2n * (g[m]*S^2 - c_soft*2*S) / S^4 = 2n*(g[m]*S - 2*c_soft)/S^3
    score_grad = 2.0 * n * (g * S - 2.0 * c_soft) / (S ** 3)
    
    return score_grad, c_soft, c


def adam_optimize(f_init, time_limit, tau_start, tau_end, base_lr, 
                  restart_interval=500, restart_strength=0.1, verbose=False):
    """Adam optimization with annealed soft-peak in log-parameterization."""
    n = len(f_init)
    S = np.sum(f_init)
    
    # Log-parameterization
    f_safe = np.clip(f_init, 1e-10, None)
    theta = np.log(f_safe / S)
    
    mom1 = np.zeros(n)
    mom2 = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step_count = 0
    
    best_f = f_init.copy()
    best_score = local_score(f_init)
    t0 = time.monotonic()
    stagnant = 0
    rng_local = np.random.default_rng(9999)
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Exponential decay of temperature
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac
        
        # Compute f from theta
        t_shifted = theta - np.max(theta)
        e = np.exp(t_shifted)
        f = e / np.sum(e) * S
        
        # Compute soft peak weights
        c = fft_convolve(f, f)
        c_max = np.max(c)
        shifted_c = (c - c_max) / tau
        exp_c = np.exp(shifted_c)
        w = exp_c / np.sum(exp_c)
        
        # Gradient
        score_grad, c_soft, _ = compute_soft_peak_gradient(f, w)
        
        # Convert to theta-space
        inner = np.dot(score_grad, f) / S
        g_theta = f * (score_grad - inner)
        
        # Adam step
        step_count += 1
        mom1 = beta1 * mom1 + (1.0 - beta1) * g_theta
        mom2 = beta2 * mom2 + (1.0 - beta2) * (g_theta ** 2)
        m1_hat = mom1 / (1.0 - beta1 ** step_count)
        m2_hat = mom2 / (1.0 - beta2 ** step_count)
        lr = base_lr / (1.0 + 0.005 * step_count)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)
        
        # Evaluate
        t_shifted = theta - np.max(theta)
        f_new = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
        score = local_score(f_new)
        
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Periodic restart to escape local minima
        if stagnant > restart_interval:
            strength = restart_strength * min(1.0, stagnant / (2 * restart_interval))
            theta += strength * rng_local.standard_normal(n)
            stagnant = 0
        
        if verbose and step_count % 100 == 0:
            print(f"    Adam step={step_count}, score={score:.6f}, "
                  f"best={best_score:.6f}, tau={tau:.5f}, lr={lr:.4f}")
    
    return best_f, best_score


def peak_targeted_search(f_init, time_limit, verbose=False):
    """Direct peak-targeted multiplicative update search."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    S = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    rng_local = np.random.default_rng(777)
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(f, f)
        c_max = np.max(c)
        
        # Find all near-peak positions
        peak_threshold = c_max * 0.95
        peak_positions = np.where(c >= peak_threshold)[0]
        
        for p in peak_positions:
            if p < 0 or p >= 2 * n - 1:
                continue
            
            # Compute contributions to this peak
            contribs = np.zeros(n)
            for i in range(n):
                j = p - i
                if 0 <= j < n:
                    contribs[i] = f[i] * f[j]
            
            total_contrib = np.sum(contribs)
            if total_contrib < 1e-10:
                continue
            
            # Find top and bottom contributors
            n_top = max(3, n // 8)
            n_bot = max(3, n // 8)
            
            order = np.argsort(contribs)
            top_idx = order[-n_top:]
            bot_idx = order[:n_bot]
            
            # Multiplicative update: reduce top, increase bottom
            # Scale factor based on contribution ratio
            top_contrib = contribs[top_idx]
            bot_contrib = contribs[bot_idx]
            
            eta = 0.05  # learning rate for multiplicative update
            
            # Reduce top contributors
            for i in top_idx:
                reduction = 1.0 - eta * (contribs[i] / total_contrib) * (c_max / S)
                reduction = max(0.5, min(1.0, reduction))
                f[i] *= reduction
            
            # Increase bottom contributors
            for i in bot_idx:
                if contribs[i] < total_contrib / n:
                    increase = 1.0 + eta * (1.0 - contribs[i] * n / total_contrib) * 0.1
                    f[i] *= increase
        
        # Rescale
        f = np.clip(f, 0.0, None)
        f_sum = np.sum(f)
        if f_sum < 1e-10:
            break
        f = f * (S / f_sum)
        
        iteration += 1
        score = local_score(f)
        
        if score < best_score:
            best_score = score
            best_f = f.copy()
        elif iteration % 50 == 0:
            # Add small perturbation to escape
            f *= np.exp(rng_local.normal(0, 0.01, size=n))
            f = np.clip(f, 0, None)
            f = f * (S / np.sum(f))
    
    return best_f, best_score

            candidate[bot] *= np.exp(rng.normal(0.03, 0.015, size=len(bot)))

def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(8):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.03, 0.15)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_golden_ratio_init(n, rng):
    """Initialize using golden ratio related frequencies."""
    t = np.arange(n, dtype=np.float64)
    golden = (1 + np.sqrt(5)) / 2
    a = np.ones(n)
    for _ in range(5):
        freq = rng.uniform(0.1, 0.5) * (1 + 1/golden)
        amp = rng.uniform(0.05, 0.2)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)
