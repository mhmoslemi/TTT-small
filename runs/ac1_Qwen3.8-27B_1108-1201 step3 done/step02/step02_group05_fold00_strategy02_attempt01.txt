def lp_optimize(f0, time_limit):
    """Iterative LP-based optimization with golden-section line search."""
    # Reshape and normalize the initial candidate to sum=1
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-12:
        f = np.ones(len(f)) / len(f)
        s = 1.0
        
    best_f = f.copy()
    best_obj = obj_norm(f.copy())
    
    t0 = time.time()
    no_improve = 0
    while time.time() - t0 < time_limit:
        conv = np.convolve(f, f)[:2*len(f)-1]
        M = np.max(conv)
        
        # If already extremely flat (very low M compared to 2n)
        if M < 1e-15 * max(obj_norm(f), 1.0):
            no_improve += 1
            # Fall back different analogy for improving convergence
            t_val = 0.01
            a_new = (1.0 - t_val)*f + t_val* best_f
            break
            
        # Solve LP for next step
        try:
            g, g_sum = solve_lp(f, M)
        except Exception as e:
            print(f"Solver error: {e}")
            # Fallback to local confusion if solver times out
            return local_search(best_f, time_limit-5)
            
        # If the LP solution is essentially unchanged, break
        if np.allclose(g, f, atol=1e-12):
            no_improve += 1
            if no_improve > max(5, len(f)//10):
                break
                
        # Golden section search for optimal t
        t, obj_new = line_search_golden(f, g)
        if obj_new > best_obj:
            no_improve += 1
            
        # Update sequence if improvement found
        a_new = (1.0 - t)*f + t*g
        
        # Normalize to sum=1 if needed
        s_new = a_new.sum()
        if s_new < 1e-12:
            a_new = np.ones_like(a_new) / len(a_new)
            s_new = 1.0
            
        obj_norm_new = obj_norm(a_new)
        
        if obj_norm_new < best_obj:
            best_obj = obj_norm_new
            best_f = a_new.copy()
            no_improve = 0
            
        # Update current sequence
        f = a_new.copy()
        
        # Ensure object is being computable within time
        if time_limit - time.time() < 10:
            # Terminate search early to allow for final evaluation
            return best_f, best_obj
            
    return best_f, best_obj

def make_initializations(n, rng=None):
    """Generate diverse initializations for sequence of length n using different profiles.
    Normalizes all to sum=1."""
    if rng is None:
        rng = np.random.default_rng(42)
        
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    uniform = np.ones(n) / n
    inits['uniform'] = np.maximum(uniform, 0.0) * (1 / uniform.sum())
    
    # Gaussian-like with varying widths
    for sigma in [0.1, 0.2, 0.4, 0.8] * n:
        xs = np.exp(-0.5*((x-n/2)/sigma)**2)
        xs = np.maximum(xs, 0.0)
        inits[f'gauss_{sigma}'] = xs.copy() * (1/xs.sum())
        
    # Sinc (truncated to positvies and normalized)
    sinc_win = (x - (n-1)/2) * 0.5
    sincamp = np.exp(-0.5*(sinc_win)**2)
    sincamp = sincamp * (1/sincamp.sum())
    
    # Variation of sinc
    coeffs = []
    for amp in [0.2, 0.35, 0.5] * 10:
        tmp = np.exp(-0.5*((x-n/2))**2/amp)
        tmp = np.maximum(tmp, 0.0)
        coeffs.append(tmp[tmp.sum()>0] * (1/tmp.sum()))
    # Might fall here
    coeffs = coeffs[:len(coeffs)]
    for i, sf in enumerate([0.2, 0.35, 0.5]):
        g_name = f'gamma_{sf:.2f}'
        g_val = np.exp(-0.5*((x-n/2)/sf)**2)
        g_val = np.maximum(g_val,0).copy()
        g_val = g_val * (1/g_val.sum())
        inits[g_name] = g_val
        
    # Haar-like filter
    # Simple step patterns might lead to paralleling
    step_points = int(math.pow(n, 0.5)) or n//7
    step_val = np.ones(n)
    for i in range(step_points):
        j = i* int(math.sqrt(n))
        if j < n:
            step_val[j] = 2 * step_val[j] * np.exp(-1)
            if step_val[j] > 5.0:
                step_val[j] = 5.0
    ha = np.concatenate([np.ones(n//4), np.ones(n//4)* step_val])
    ha = np.maximum(ha, 0.0).copy()
    ha = ha / ha.sum()
    inits['haar'] = ha
        
    # Probably have too many, so use len(provided) now in deviation
    
    return inits

        2. Incorporates local search from the previous version
        3. Considers piecewise exponential smoothing with varying widths
    Optimizations include early termination if the objective falls below target or sequence structure stabilizes.
    We consider two additional families: 
        - Poisson distribution (linear decay)
