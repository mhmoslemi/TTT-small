"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_objective(f):
    """Compute the objective (normalized) for a sequence f with sum=1."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()

def make_lp_structure(n):
    """Precompute row/col indices for the cross-convolution constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)

def solve_lp(f, M, rows, cols, n, timeout=10):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k=0..2n-2."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def lp_step(f, M, rows, cols, n, timeout=10):
    """One LP-guided step: find g maximizing sum(g) s.t. (f*g)<=M, g>=0."""
    g = solve_lp(f, M, rows, cols, n, timeout=timeout)
    if g is None:
        return f
    g_sum = g.sum()
    if g_sum < 1e-12:
        return f
    g /= g_sum
    return g

def lp_optimize(f0, time_limit, verbose=False):
    """LP-based alternating optimization with backtracking line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    rows, cols = make_lp_structure(n)

    t0 = time.monotonic()
    t = 1.0
    patience = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        g = solve_lp(f, M, rows, cols, n, timeout=5)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        if np.max(np.abs(g - f)) < 1e-14:
            break

        improved = False
        for t_try in [t, 0.7*t, 0.5*t, 0.3*t, 0.1*t, 0.05*t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-14:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.05)
                improved = True
                break

        if not improved:
            patience += 1
            t *= 0.9
            if patience > 30:
                break
        if verbose and iter_count % 50 == 0:
            print(f"    LP iter {iter_count}, obj={best_obj:.8f}, t={t:.3f}")

    return best_f, best_obj

def simulated_annealing(f0, time_limit, T0=0.01, T_end=0.0001, cool_rate=0.999):
    """Simulated annealing with peak-targeted perturbations."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    n = len(f)
    f /= f.sum()
    
    conv = fast_convolve(f)
    current_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    best_obj = current_obj
    
    T = T0
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Peak-targeted perturbation
        conv = fast_convolve(f)
        M = conv.max()
        k_peak = int(np.argmax(conv))
        
        # Find elements contributing to the peak
        j_start = max(0, k_peak - n + 1)
        j_end = min(n - 1, k_peak)
        idx = np.arange(j_start, j_end + 1)
        
        # Perturbation: reduce high-contributing elements, increase low ones
        noise = np.random.randn(n) * T
        # Extra reduction at peak-contributing elements
        contrib = f[idx] * f[k_peak - idx]
        peak_weight = contrib / (T + 1e-15) * 0.1
        
        f_new = f + noise - np.zeros(n)
        # Apply peak reduction
        f_new[idx] -= peak_weight * T
        
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-12:
            f_new = f.copy()
        else:
            f_new /= s
        
        conv_new = fast_convolve(f_new)
        new_obj = 2.0 * n * conv_new.max()
        
        delta = new_obj - current_obj
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-10)):
            f = f_new
            current_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        T *= cool_rate
        if T < T_end:
            T = T_end * 2
            no_improve = min(no_improve, 100)
        
        if no_improve > 500:
            break
    
    return best_f, best_obj

def mirror_polish(f0, time_limit, t_param=50.0, eta0=0.3):
    """Polish with multiplicative mirror descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    eta = eta0
    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        scaled = t_param * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        thresh = w.max() * 0.002
        active = np.where(w > thresh)[0]
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            sg[j_start:j_end + 1] += w[k] * f[k - j_end:k - j_start + 1][::-1]

        sg_c = sg - sg.mean()
        lf = -eta * sg_c
        lf -= lf.max()
        f = f * np.exp(lf)
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-12:
            f = np.ones(n) / n
        else:
            f /= s

        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            patience = 0
            eta = min(eta * 1.02, 0.8)
        else:
            patience += 1
            if patience % 20 == 0:
                eta *= 0.8
                if eta < 0.01:
                    t_param = min(t_param * 1.5, 500.0)
                    eta = 0.3
            if patience > 200:
                break

    return best_f, best_obj

def gen_inits(n, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uni'] = np.ones(n) / n
    
    for sf in [0.1, 0.15, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf*100)}'] = g
    
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    
    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()
    
    for freq in [1, 2, 3, 5]:
        cb = 0.5 * (1 + np.cos(2 * np.pi * freq * x / n))
        cb = np.maximum(cb, 0.01)
        inits[f'cos{freq}'] = cb / cb.sum()
    
    rr = rng.dirichlet(np.ones(n) * 0.5)
    inits['rnd'] = rr
    
    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.05)
    inits['tri'] = tri / tri.sum()
    
    # Ring/U-shape
    ring = np.abs(x - 0.5 * (n - 1)) / (0.5 * (n - 1))
    ring = np.maximum(ring, 0.01)
    inits['ring'] = ring / ring.sum()
    
    # Power law
    pl = np.power(r, 2.0)
    inits['pow2'] = pl / pl.sum()
    
    return inits

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_norm_obj(f):
    """Compute 2n*max(conv) for f with sum=1."""
    n = len(f)
    conv = fast_convolve(f)
    return 2.0 * n * conv.max()


def make_lp_structure(n):
    """Precompute row/col indices for cross-convolution matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp_max_sum(f, M, rows, cols, n, timeout=8):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

    A_eq = np.zeros((1, n + 1))

    no_improve = 0
    reheat_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        k_peak = int(np.argmax(conv))
        j_start = max(0, k_peak - n + 1)
        j_end = min(n - 1, k_peak)
        idx = np.arange(j_start, j_end + 1)
        contrib = f[idx] * f[k_peak - idx]
        total_c = contrib.sum()

        # Structured perturbation
        noise_scale = T * np.sqrt(n)
        noise = np.random.randn(n) * noise_scale

        # Extra push away from peak contributors
        if total_c > 1e-15:
            frac = contrib / total_c
            noise[idx] -= frac * T * 2.0 * n

        # Randomly boost a few non-peak elements
        if np.random.rand() < 0.3:
            non_peak = np.setdiff1d(np.arange(n), idx)
            if len(non_peak) > 0:
                pick = np.random.choice(non_peak, size=min(3, len(non_peak)), replace=False)
                noise[pick] += T * np.sqrt(n) * 0.5

        f_new = f + noise
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-12:
            f_new = f.copy()
        else:
            f_new /= s

        conv_new = fast_convolve(f_new)
        new_obj = 2.0 * n * conv_new.max()
        delta = new_obj - current_obj

        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            f = f_new
            current_obj = new_obj
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        T *= 0.9995
        if T < T_min:
            # Reheat
            T = max(T0 * 0.5, T_min * 10)
            reheat_count += 1
            no_improve = 0
            if reheat_count > 8:
                break

        if no_improve > 1000:
            T = T0 * 0.3
            no_improve = 0
            reheat_count += 1
            if reheat_count > 10:
                break

    return best_f, best_obj


def mirror_polish(f0, time_limit, t_param=50.0, eta0=0.3):
    """Polish with multiplicative mirror descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    eta = eta0
    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break
        scaled = t_param * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        thresh = w.max() * 0.003
        active = np.where(w > thresh)[0]
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            sg[j_start:j_end + 1] += w[k] * f[k - j_end:k - j_start + 1][::-1]

        sg_c = sg - sg.mean()
        lf = -eta * sg_c
        lf -= lf.max()
        f = f * np.exp(lf)
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-12:
            f = np.ones(n) / n
        else:
            f /= s
        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            patience = 0
            eta = min(eta * 1.01, 0.7)
        else:
            patience += 1
            if patience % 15 == 0:
                eta *= 0.85
                if eta < 0.01:
                    t_param = min(t_param * 1.6, 800.0)
                    eta = 0.25
            if patience > 150:
                break

    return best_f, best_obj


def peak_reduction(f0, time_limit):
    """Directly target the peak bin for reduction with random restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    n = len(f)
    f /= f.sum()
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        iteration += 1
        conv = fast_convolve(f)
        M = conv.max()
        current_obj = 2.0 * n * M
        k_peak = int(np.argmax(conv))
        j_start = max(0, k_peak - n + 1)
        j_end = min(n - 1, k_peak)
        idx = np.arange(j_start, j_end + 1)
        contrib = f[idx] * f[k_peak - idx]
        total_c = contrib.sum()
        if total_c < 1e-15:
            break
        frac = contrib / total_c
        rate = 0.03 + 0.02 * np.random.rand()
        f[idx] *= (1.0 - rate * frac)
        # Boost non-peak elements
        non_peak = np.setdiff1d(np.arange(n), idx)
        if len(non_peak) > 0:
            low_thresh = np.percentile(f[non_peak], 30)
            boost_mask = np.zeros(n, dtype=bool)
            boost_mask[non_peak] = f[non_peak] <= low_thresh
            f[boost_mask] *= (1.0 + rate * 0.3)
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-12:
            break
        f /= s
        conv_new = fast_convolve(f)
        new_obj = 2.0 * n * conv_new.max()
        if new_obj < best_obj:
            best_obj = new_obj
            best_f = f.copy()
        if iteration % 80 == 0:
            f += np.random.randn(n) * 0.005
            f = np.maximum(f, 0.0)
            f /= f.sum()
        if best_obj < 2.0 * n / (2.0 * n - 1.0) * 1.005:
            break

    return best_f, best_obj


def gen_inits(n, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uni'] = np.ones(n) / n
    for sf in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    bi = np.zeros(n)
    w = max(1, n // 4)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()
    for freq in [1, 2, 3, 5, 8]:
        cb = 0.5 * (1 + np.cos(2 * np.pi * freq * x / n))
        cb = np.maximum(cb, 0.01)
        inits[f'cos{freq}'] = cb / cb.sum()
    rr = rng.dirichlet(np.ones(n) * 0.4)
    inits['rnd'] = rr
    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.05)
    inits['tri'] = tri / tri.sum()
    ring = np.abs(x - 0.5 * (n - 1)) / (0.5 * (n - 1) + 1)
    ring = np.maximum(ring, 0.01)
    inits['ring'] = ring / ring.sum()
    pl = np.power(r, 1.5)
    inits['pow15'] = pl / pl.sum()
    # Binary-like: random subset
    mask = rng.random(n) < 0.3
    if mask.sum() > 0:
        inits['bin'] = mask.astype(float) / mask.sum()
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry point: multi-phase hybrid optimization."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    def try_update(f, label):
        nonlocal best_seq, best_score
        score = float(evaluate_sequence(f.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = f.copy()
            print(f"  NEW BEST: {label} score={score:.8f} n={len(f)}")
        return score

    # Phase 1: Quick scan with LP (300s)
    print("Phase 1: LP scan across lengths and inits")
    lengths = [256, 512, 1024]
    results = []

    for n in lengths:
        if time.monotonic() > deadline - 1000:
            break
        inits = gen_inits(n, rng)
        if parent is not None:
            if parent.size >= n:
                pp = np.maximum(parent[:n], 0.0)
                if pp.sum() > 1e-10:
                    inits['par'] = pp / pp.sum()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
                pp = np.maximum(pp, 0.0)
                if pp.sum() > 1e-10:
                    inits['par'] = pp / pp.sum()

        n_cfgs = len(inits)
        remaining = deadline - time.monotonic() - 1000
        time_per = max(12.0, remaining / max(1, n_cfgs * len(lengths)))

        for name, a0 in inits.items():
            if time.monotonic() > deadline - 980:
                break
            ta = min(time_per, deadline - time.monotonic() - 980)
            if ta < 10:
                break
            use_mp = (name == 'par' or name == 'uni')
            f_opt, obj_opt = lp_optimize(a0, time_limit=ta, use_min_peak=use_mp)
            score = try_update(f_opt, f"LP n={n} {name}")
            if np.isfinite(score):
                results.append((score, n, name, f_opt))

    # Also run adaptive LP on parent if available
    if parent is not None and len(parent) >= 256:
        for n in [256, 512]:
            if time.monotonic() > deadline - 950:
                break
            if parent.size >= n:
                pp = np.maximum(parent[:n], 0.0) / np.maximum(parent[:n], 0.0).sum()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
                pp = np.maximum(pp, 0.0) / pp.sum()
            ta = min(60.0, deadline - time.monotonic() - 950)
            f_ad, _ = adaptive_lp_optimize(pp, time_limit=ta)
            score = try_update(f_ad, f"AdaptLP n={n} par")
            if np.isfinite(score):
                results.append((score, n, 'adapt_par', f_ad))

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}")

    # Phase 2: Deep optimization on top candidates (600s)
    top = results[:12] if len(results) >= 12 else results
    remaining = deadline - time.monotonic() - 350
    if remaining > 100 and top:
        n_deep = min(len(top), 8)
        print(f"Phase 2: Deep opt on top {n_deep}, {remaining:.0f}s left")
        t_each = remaining / n_deep

        for idx in range(n_deep):
            if time.monotonic() > deadline - 330:
                break
            sc, n, name, f = top[idx]
            ta = min(t_each * 0.95, deadline - time.monotonic() - 330)
            if ta < 10:
                break

            # Adaptive LP
            f2, _ = adaptive_lp_optimize(f, time_limit=ta * 0.3)
            try_update(f2, f"DeepLP[{idx}] n={n} {name}")

            # LP with min-peak formulation
            if time.monotonic() < deadline - 330:
                f3, _ = lp_optimize(f2, time_limit=ta * 0.2, use_min_peak=True)
                try_update(f3, f"MinPeak[{idx}] n={n} {name}")

            # SA
            if time.monotonic() < deadline - 330:
                tr = min(ta * 0.2, deadline - time.monotonic() - 330)
                f4, _ = simulated_annealing(f3, time_limit=tr, T0=0.006, T_min=0.00005,
                                            seed=100 + idx)
                try_update(f4, f"SA[{idx}] n={n} {name}")

            # Peak reduction
            if time.monotonic() < deadline - 330:
                tr2 = min(ta * 0.15, deadline - time.monotonic() - 330)
                f5, _ = peak_reduction(f4, time_limit=tr2)
                try_update(f5, f"Peak[{idx}] n={n} {name}")

            # Mirror polish
            if time.monotonic() < deadline - 330:
                tr3 = min(ta * 0.1, deadline - time.monotonic() - 330)
                f6, _ = mirror_polish(f5, time_limit=tr3, t_param=60.0, eta0=0.25)
                try_update(f6, f"Mirror[{idx}] n={n} {name}")

    # Phase 3: Extended SA and peak reduction on best (200s)
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 50
        if remaining > 30:
            print(f"Phase 3: Extended optimization, {remaining:.0f}s left")
            n = len(best_seq)

            # Long SA with reheat
            f_sa, _ = simulated_annealing(best_seq, time_limit=remaining * 0.35,
                                          T0=0.004, T_min=0.00002, seed=999)
            try_update(f_sa, f"LongSA n={n}")

            # Peak reduction
            if time.monotonic() < deadline - 50:
                tr = min(remaining * 0.2, deadline - time.monotonic() - 50)
                f_pk, _ = peak_reduction(f_sa, time_limit=tr)
                try_update(f_pk, f"LongPeak n={n}")

            # Multiple mirror polish passes with different params
            for tp, eta in [(30.0, 0.3), (80.0, 0.2), (150.0, 0.15), (400.0, 0.08), (1000.0, 0.04)]:
                if time.monotonic() > deadline - 50:
                    break
                tr = min(remaining * 0.08, deadline - time.monotonic() - 50)
                if tr < 3:
                    break
                f_mp, _ = mirror_polish(f_pk, time_limit=tr, t_param=tp, eta0=eta)
                try_update(f_mp, f"MP tp={tp:.0f} n={n}")

    # Phase 4: Try a different length for the best approach found
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        alt_lengths = [l for l in [384, 768, 1536, 2048] if l != n_best]
        remaining = deadline - time.monotonic() - 30

        if remaining > 60 and alt_lengths:
            print(f"Phase 4: Try alternative lengths, {remaining:.0f}s left")
            target_n = alt_lengths[0]
            # Interpolate best to new length
            f_interp = np.interp(np.linspace(0, 1, target_n),
                                 np.linspace(0, 1, n_best), best_seq)
            f_interp = np.maximum(f_interp, 0.0)
            f_interp /= f_interp.sum()

            ta = min(remaining * 0.4, deadline - time.monotonic() - 30)
            f_alt, _ = adaptive_lp_optimize(f_interp, time_limit=ta)
            try_update(f_alt, f"AltLen n={target_n}")

            if time.monotonic() < deadline - 30:
                tr = min(remaining * 0.2, deadline - time.monotonic() - 30)
                f_alt2, _ = mirror_polish(f_alt, time_limit=tr, t_param=100.0, eta0=0.2)
                try_update(f_alt2, f"AltLenPolish n={target_n}")

    # Phase 5: Final combined polish
    if best_seq is not None and time.monotonic() < deadline - 5:
        tr = deadline - time.monotonic() - 5
        # Short SA
        f_f1, _ = simulated_annealing(best_seq, time_limit=tr * 0.4,
                                      T0=0.002, T_min=0.00001, seed=777)
        try_update(f_f1, "FinalSA")
        if time.monotonic() < deadline - 5:
            tr2 = deadline - time.monotonic() - 5
            f_f2, _ = mirror_polish(f_f1, time_limit=tr2, t_param=80.0, eta0=0.2)
            try_update(f_f2, "FinalMirror")

    if best_seq is None:
        best_seq = np.ones(512)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.8f}")
    return [float(x) for x in best_seq]
