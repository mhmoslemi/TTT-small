"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth mirror descent with
log-sum-exp regularization, adaptive temperature annealing, and basin hopping.
Explores multiple sequence lengths and diverse initializations.
"""

import numpy as np
import time
import math


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(x, T):
    """Temperature-scaled softmax."""
    x = x / T
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def compute_gradient(a, p):
    """Compute gradient of soft_max(conv(a,a)) with respect to a.
    p is the softmax distribution over convolution entries (length 2n-1).
    Returns gradient of shape (n,).
    """
    n = len(a)
    # grad[i] = 2 * sum_j a[j] * p[i+j]
    # This is np.correlate(p, a, 'full')[0:n]
    corr = np.correlate(p, a, mode='full')
    return 2.0 * corr[:n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    condition = u - (cssv - 1) / rho > 0
    if not condition.any():
        return np.ones(n) / n
    rho_val = np.max(np.where(condition)[0]) + 1
    theta = (cssv[rho_val - 1] - 1) / rho_val
    return np.maximum(v - theta, 0)


def mirror_descent_step(a, grad, lr):
    """Mirror descent update with negative entropy."""
    # a_new[i] = a[i] * exp(-lr * grad[i]) / Z
    log_ratio = -lr * grad
    log_ratio -= log_ratio.max()  # numerical stability
    weights = np.exp(log_ratio)
    a_new = a * weights
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    a_new /= s
    return a_new


def smooth_optimize(a0, time_limit, T_start=0.05, T_min=0.003, lr_start=2.0, lr_decay=0.9995, seed=0):
    """Smooth mirror descent optimization."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 1e-15).copy()
    a = a / a.sum()
    n = len(a)
    
    T = T_start
    lr = lr_start
    t0 = time.monotonic()
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    obj_ema = best_obj  # exponential moving average for adaptive lr
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        p = softmax(b, T)
        grad = compute_gradient(a, p)
        
        # Center the gradient (project onto tangent space of sum=1)
        grad_centered = grad - grad.mean()
        
        # Mirror descent step
        a_new = mirror_descent_step(a, grad_centered, lr)
        a = a_new
        
        # Clamp very small values (prevent underflow issues)
        a[np.where(a < 1e-15)[0]] = 1e-15
        a = a / a.sum()
        
        # Update best
        b = fast_convolve(a)
        obj = 2.0 * n * b.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Anneal
        T = max(T * 0.999, T_min)
        lr = max(lr * lr_decay, 0.01)
        step += 1
    
    return best_a, best_obj


def basin_hopping(a0, time_limit, n_trials=5):
    """Basin hopping: random perturbations + re-optimization."""
    rng = np.random.default_rng()
    n = len(a0)
    best_a = a0.copy()
    best_obj = 2.0 * n * fast_convolve(a0).max()
    t0 = time.monotonic()
    
    time_per_trial = time_limit / n_trials
    for trial in range(n_trials):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Perturb
        noise = rng.laplace(0, 0.1, size=n)
        a_perturbed = best_a + noise
        a_perturbed = np.maximum(a_perturbed, 0)
        s = a_perturbed.sum()
        if s < 1e-10:

        a_perturbed /= s
        
        # Optimize
        a_opt, obj = smooth_optimize(a_perturbed, time_limit=min(time_per_trial, remaining - 2), seed=trial)
        if obj < best_obj:

            best_a = a_opt.copy()
    
    return best_a, best_obj

def get_parent(n):
    """Get parent sequence resized to length n."""
    try:

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim != 1 or p.size == 0:
            return None
        if p.size >= n:
            pc = p[:n].copy()
        else:
            pc = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s < 1e-10:

        return pc / s

def make_inits(n, rng):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    
    inits['uniform'] = np.ones(n) / n
    
    # Gaussian at different positions and widths
    for center_frac in [0.25, 0.5, 0.75]:
        for width_frac in [0.1, 0.2, 0.3]:
            g = np.exp(-0.5 * ((x - center_frac * n) / (width_frac * n)) ** 2)
            g = np.maximum(g, 1e-10)
            inits[f'gau_{center_frac}_{width_frac}'] = g / g.sum()
    
    # Bimodal
    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    bim = g1 + g2
    inits['bimodal'] = bim / bim.sum()
    
    # Cosine-based (concentrated at edges)
    cos = np.cos(np.pi * x / (n - 1)) ** 2 if n > 1 else np.ones(n)
    cos = np.maximum(cos, 1e-10)
    inits['cos_edge'] = cos / cos.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Random with structure
    k = max(2, n // 8)
    centers = rng.choice(n, size=k, replace=False)
    sig = n / (4 * k)
    g = np.zeros(n)
    for c in centers:
        g += np.exp(-0.5 * ((x - c) / sig) ** 2)
    g = np.maximum(g, 1e-10)
    inits['random_peaks'] = g / g.sum()
    
    # Parent
    parent = get_parent(n)
    if parent is not None:
        inits['parent'] = parent
    
    return inits

The Euclidean projection onto the simplex {x >= 0, sum(x) = 1} is:
1. Find theta such that sum(max(v - theta, 0)) = 1

To find theta, sort v in decreasing order: v_1 >= v_2 >= ... >= v_n.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed proximal gradient
descent with log-sum-exp regularization, temperature continuation, adaptive
learning rate, basin hopping, and local search. Explores multiple sequence
lengths and diverse initializations including the parent sequence.
"""

import numpy as np
import time
import math


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def temp_softmax(x, T):
    """Temperature-scaled softmax, numerically stable."""
    x = x / T
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s


def compute_grad(a, p):
    """Gradient of soft_max(conv(a,a),T) w.r.t. a.
    p is softmax distribution over conv entries (len 2n-1).
    Returns grad of shape (n,).
    """
    n = len(a)
    corr = np.correlate(p, a, mode='full')
    return 2.0 * corr[n - 1: 2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    rho_vals = np.where(cond)[0]
    if len(rho_vals) == 0:
        return np.ones(n) / n
    rho_val = rho_vals[-1] + 1
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def obj_val(a):
    """Objective value (assumes sum(a)=1)."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def smooth_prox_grad(a0, time_limit, T_start=0.1, T_min=0.002, T_decay=0.9985,
                     lr_init=1.0, seed=0):
    """Smoothed proximal gradient with temperature annealing and adaptive LR."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones_like(a) / len(a)
    else:
        a /= s

    n = len(a)
    T = T_start
    lr = lr_init
    t0 = time.monotonic()

    best_a = a.copy()
    best_obj = obj_val(a)
    cur_obj = best_obj
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        p = temp_softmax(b, T)
        grad = compute_grad(a, p)
        grad = grad - grad.mean()

        # Proximal gradient step
        a_new = a - lr * grad
        a_new = project_simplex(a_new)

        obj_new = obj_val(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive learning rate
        if obj_new > cur_obj * 1.005 + 1e-10:
            lr = max(lr * 0.5, 0.01)
        elif obj_new < cur_obj * 0.995:
            lr = min(lr * 1.05, 10.0)
        cur_obj = obj_new

        a = a_new
        T = max(T * T_decay, T_min)

        if stall > 200 and T <= T_min * 1.01:
            break

    return best_a, best_obj


def basin_hop(a0, time_limit, n_trials=4, noise_scale=0.15, seed=0):
    """Basin hopping with Laplace noise perturbations."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    best_a = a0.copy()
    best_obj = obj_val(a0)
    t0 = time.monotonic()
    time_per = time_limit / max(1, n_trials)

    for trial in range(n_trials):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3.0:
            break
        # Perturb
        noise = rng.laplace(0, noise_scale, size=n)
        a_p = best_a + noise
        a_p = np.maximum(a_p, 0.0)
        sp = a_p.sum()
        if sp < 1e-12:
            continue
        a_p /= sp
        a_opt, obj = smooth_prox_grad(a_p, time_limit=min(time_per, rem - 2.0),
                                       T_start=0.05, T_min=0.003, T_decay=0.998,
                                       lr_init=1.5, seed=trial + 100)
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()

    return best_a, best_obj


def local_search(a0, time_limit, seed=0):
    """Local search via pairwise mass transfers targeting peak reduction."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    best_obj = obj_val(a)
    t0 = time.monotonic()
    stuck = 0
    max_stuck = 200

    while time.monotonic() - t0 < time_limit and stuck < max_stuck:
        b = fast_convolve(a)
        k_star = int(np.argmax(b))

        # Contribution to peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        max_c = contrib.max()
        if max_c < 1e-18:
            break

        found = False
        n_tries = 80
        for _ in range(n_tries):
            # Weighted source selection (high contribution)
            w = contrib + 1e-18
            w /= w.sum()
            src = rng.choice(n, p=w)
            dst = rng.integers(n)
            if src == dst:
                continue
            for frac in [0.03, 0.1, 0.25, 0.5, 0.8]:
                amt = a[src] * frac
                if amt < 1e-16:
                    continue
                a_t = a.copy()
                a_t[src] -= amt
                a_t[dst] += amt
                a_t = np.maximum(a_t, 0.0)
                st = a_t.sum()
                if st < 1e-12:
                    continue
                a_t /= st
                o = obj_val(a_t)
                if o < best_obj - 1e-12:
                    best_obj = o
                    a = a_t.copy()
                    found = True
                    break
        if found:
            stuck = 0
        else:
            stuck += 1

    return a, best_obj

def make_inits(n, rng):
    """Generate diverse initializations for length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    for cf in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for wf in [0.08, 0.15, 0.25]:
            g = np.exp(-0.5 * ((x - cf * n) / (wf * n)) ** 2)
            g = np.maximum(g, 1e-12)
            inits[f'g_{cf}_{wf}'] = g / g.sum()

    # Bimodal
    g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.08 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.08 * n)) ** 2)
    inits['bimodal'] = (g1 + g2) / (g1 + g2).sum()

    # Trimodal
    g1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.07 * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - 0.5 * n) / (0.07 * n)) ** 2)
    g3 = np.exp(-0.5 * ((x - 0.8 * n) / (0.07 * n)) ** 2)
    inits['trimodal'] = (g1 + g2 + g3) / (g1 + g2 + g3).sum()

    # Cosine edge
    c = np.cos(np.pi * x / max(1, n - 1)) ** 2
    c = np.maximum(c, 1e-12)
    inits['cos_edge'] = c / c.sum()

    # Random Dirichlet
    inits['rand'] = rng.dirichlet(np.ones(n) * 0.3)

    # Random peaks
    k = max(3, n // 6)
    centers = rng.choice(n, size=k, replace=False)
    sig = n / (3.0 * k)
    g = np.zeros(n)
    for c_i in centers:
        g += np.exp(-0.5 * ((x - c_i) / sig) ** 2)
    g = np.maximum(g, 1e-12)
    inits['rand_peaks'] = g / g.sum()

    # Parent
    parent = get_parent(n)
    if parent is not None:
        inits['parent'] = parent

    return inits
