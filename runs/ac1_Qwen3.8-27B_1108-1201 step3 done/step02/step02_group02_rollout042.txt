    size = 1

        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:len(a)+len(b)-1]

    """Project a onto {x >= 0, sum(x) = S}."""
    a = np.clip(a, 0, None)

def lp_direction_reduced(f, M, n_constraints=100):
    """Solve reduced LP: maximize sum(b) s.t. b>=0, (f*b)[k] <= M for top k constraints."""
    n = len(f)
    c_conv = fast_conv(f, f)
    # Get top n_constraints lags
    top_lags = np.argsort(c_conv)[-n_constraints:]

    A = np.zeros((len(top_lags), n))
    for idx, k in enumerate(top_lags):

import numpy as np
from scipy.optimize import linprog

def fft_conv(a, b=None):
    """FFT-based linear convolution a * b (or a * a if b is None)."""
    if b is None:
        b = a
    size = 1
    while size < len(a) + len(b) - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:len(a)+len(b)-1]

def compute_score(a):
    """Compute evaluation score (fast version)."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_conv(a)
    return 2.0 * n * np.max(c) / (S * S)

def normalize(a, S):
    """Normalize to have sum S, clip to non-negative."""
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s > 0:
        a = a * (S / s)
    return a

def subgradient(a, n_active=5):
    """
    Compute subgradient of max_k conv(a,a)[k] at point a.
    Uses average over top n_active lags.
    Returns (gradient, conv, max_val).
    """
    n = len(a)
    c = fft_conv(a)
    max_c = np.max(c)
    # Get top lags
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:n_active]
    
    grad = np.zeros(n)
    for k in active:
        # Subgradient of conv(a,a)[k]: g[m] = 2*a[k-m] if 0 <= k-m < n
        m_min = max(0, k - n + 1)
        m_max = min(n - 1, k)
        if m_min <= m_max:
            # a[k-m] for m = m_min,...,m_max
            # = a[k-m_min], a[k-m_min-1], ..., a[k-m_max]
            src = slice(max(k - m_max, 0), k - m_min + 1)
            grad[m_min:m_max+1] += 2.0 * a[src][::-1]
    
    grad /= len(active)
    return grad, c, max_c

def lp_direction_reduced(f, M, n_cons=100):
    """Solve reduced LP: max sum(b) s.t. b>=0, (f*b)[k]<=M for top n_cons lags."""
    n = len(f)
    c = fft_conv(f)
    # Most important constraints are those where f*b could exceed M
    # Since b >= 0 and f >= 0, the most binding constraints are where f has mass
    # aligned with b. Use top lags of conv(f,f) as proxy.
    top_lags = np.argsort(c)[-n_cons:]
    
    A = np.zeros((n_cons, n))
    for idx, k in enumerate(top_lags):
        m_min = max(0, k - n + 1)
        m_max = min(n - 1, k)
        for j in range(m_min, m_max + 1):
            A[idx, j] = f[k - j]
    
    c_obj = -np.ones(n)
    b_ub = np.full(n_cons, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return f / np.sum(f)

def lp_direction_full(f, M):
    """Solve full LP with all 2n-1 constraints."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return f / np.sum(f)

def nesterov_subgradient(a, n_iter, n_active=5, lr_init=0.01):
    """Nesterov-accelerated subgradient descent with projection."""
    n = len(a)
    S = np.sum(a)
    x = a.copy()
    y = a.copy()
    best = a.copy()
    best_score = compute_score(a)
    
    for i in range(n_iter):
        lr = lr_init / (1.0 + i * 0.005)
        grad, c, max_c = subgradient(y, n_active)
        # Update
        x_new = y - lr * grad
        x_new = normalize(x_new, S)
        # Nesterov momentum
        t_i = 2.0 / (i + 2.0)
        y = x_new + (t_i - 1.0) * (x_new - x)
        y = np.clip(y, 0, None)
        x = x_new
        s = compute_score(x)
        if s < best_score:
            best_score = s
            best = x.copy()
    
    return best, best_score

The LP maximizes sum(g), so g tends to be "spread out" which helps keep (g*g) small. But there's no guarantee.

This is why the line search is important: we try different t values and check if the score actually decreases.

    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n+m-1]

def score(a):
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_conv(a)
    return 2.0 * n * np.max(c) / (S * S)

def norm_to(a, S):
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a
    return a * (S / s)

def get_subgrad(a, n_top=3):
    """Subgradient of peak autocorrelation, averaged over top n_top lags."""
    n = len(a)
    c = fft_conv(a)
    peak = np.max(c)

    return g, c, peak

"""
Optimized peak autocorrelation minimization using reduced-constraint LP,
Nesterov subgradient, and coordinate descent. Key improvements:
- Faster LP via constraint reduction (top-K active lags)
- Aggressive step sizes with geometric line search
- Nesterov momentum for subgradient acceleration
- Targeted coordinate descent that exploits 1D structure

import numpy as np
from scipy.optimize import linprog
import time


def fft_conv(a, b=None):
    """FFT-based linear convolution."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def compute_score(a):
    """Fast evaluation score."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_conv(a)
    return 2.0 * n * np.max(c) / (S * S)


def norm_to(a, S):
    """Clip to non-negative and rescale to sum S."""
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-15:
        return np.full_like(a, S / len(a))
    return a * (S / s)


def get_subgrad(a, n_top=3):
    """Subgradient of peak autocorrelation averaged over top lags."""
    n = len(a)
    c = fft_conv(a)
    peak = c.max()
    thr = peak * (1.0 - 1e-8)
    active = np.where(c >= thr)[0]
    if len(active) > n_top:
        active = active[np.argsort(c[active])[::-1][:n_top]]
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= len(active)
    return g, c, peak

def nesterov_descent(f0, time_budget):
    """Nesterov-accelerated subgradient with projection."""
    n = len(f0)
    S = np.sum(f0)
    x = f0.copy()
    y = f0.copy()
    best = f0.copy()
    best_s = compute_score(f0)
    t0 = time.monotonic()
    it = 0
    lr_base = 0.8

    while time.monotonic() - t0 < time_budget:
        lr = lr_base / np.sqrt(it + 1.0)
        grad, c, peak = get_subgrad(y, 4)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        # Scale step: we want to move by roughly lr * peak / n in L2
        step_vec = lr * peak * grad / (gn * n)
        x_new = y - step_vec
        x_new = norm_to(x_new, S)
        mu = 2.0 / (np.sqrt(it + 1.0) + 1.0)
        y = x_new + (mu - 1.0) * (x_new - x)
        y = np.clip(y, 0.0, None)
        x = x_new
        s = compute_score(x)
        if s < best_s:
            best_s = s
            best = x.copy()
        it += 1
        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    nesterov it={it}, best={best_s:.6f}, elapsed={elapsed:.1f}s")

    return best, best_s


def coord_descent(a0, time_budget):
    """
    Coordinate descent exploiting 1D structure.
    For each coordinate i, the peak is a convex PWL function of a[i]
    (with one quadratic term at k=2i). We use ternary search.
    """
    n = len(a0)
    S = np.sum(a0)
    a = a0.copy()
    best = a0.copy()
    best_s = compute_score(a0)
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_budget:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_budget:
                break
            old = a[i]
            if old < 1e-10:
                # Try small positive values
                for trial in [S / n * 0.1, S / n * 0.5]:
                    a[i] = trial
                    a = np.clip(a, 0, None)
                    s = np.sum(a)
                    if s < 0.01:
                        a[i] = 0
                        continue
                    a = a * (S / s)
                    sc = compute_score(a)
                    if sc < best_s - 1e-9:
                        best_s = sc
                        best = a.copy()
                        improved = True
                        break
                else:
                    a[i] = 0
                continue

            # Ternary search around old value
            # Range: [old*0.3, old*3]
            lo = old * 0.3
            hi = old * 3.0
            # Quick evalu
            for _ in range(8):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                a[i] = m1
                a = np.clip(a, 0, None)
                s1 = np.sum(a)
                if s1 < 0.01:
                    s1 = np.inf
                else:
                    a_temp = a * (S / s1)
                    s1 = compute_score(a_temp)
                a[i] = m2
                a = np.clip(a, 0, None)
                s2 = np.sum(a)
                if s2 < 0.01:
                    s2 = np.inf
                else:
                    a_temp2 = a * (S / s2)
                    s2 = compute_score(a_temp2)
                if s1 < s2:
                    hi = m2
                else:
                    lo = m1

            a[i] = (lo + hi) / 2.0
            a = np.clip(a, 0, None)
            s = np.sum(a)
            if s > 0.01:
                a = a * (S / s)
                sc = compute_score(a)
                if sc < best_s - 1e-9:
                    best_s = sc
                    best = a.copy()
                    improved = True
            else:
                a[i] = old
                a = a * (S / np.sum(a))

        if not improved:
            break

    return best, best_s


def simulated_annealing(a0, time_budget, rng):
    """Simulated annealing with targeted peak reduction moves."""
    n = len(a0)
    S = np.sum(a0)
    cur = a0.copy()
    cur_s = compute_score(a0)
    best = a0.copy()
    best_s = cur_s
    t0 = time.monotonic()
    temp = 0.002
    cooling = 0.99985

    while time.monotonic() - t0 < time_budget:
        cand = cur.copy()
        r = rng.random()

        if r < 0.35:
            # Targeted: reduce top contributors to peak, increase elsewhere
            c = fft_conv(cur)
            pk = int(np.argmax(c))
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)
            contribs = np.zeros(n)
            for j in range(lo, hi + 1):
                p = pk - j
                if 0 <= p < n:
                    contribs[j] += cur[j] * cur[p]
            top_k = max(3, n // 15)
            top_idx = np.argsort(contribs)[-top_k:]
            cand[top_idx] *= np.exp(rng.normal(-0.03, 0.012, size=len(top_idx)))
            rest = np.setdiff1d(np.arange(n), top_idx)
            if len(rest) > 3:
                n_inc = min(10, len(rest) // 3)
                inc_idx = rng.choice(rest, size=n_inc, replace=False)
                cand[inc_idx] *= np.exp(rng.normal(0.03, 0.012, size=n_inc))
        elif r < 0.55:
            # Block move
            w = max(1, int(rng.integers(1, max(2, n // 8))))
            st = int(rng.integers(0, n - w))
            cand[st:st + w] *= np.exp(rng.normal(0.0, 0.025, size=w))
        elif r < 0.70:
            # Pair swap
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                d = rng.uniform(-0.2, 0.2) * cur[i]
                cand[i] = max(0, cur[i] - d)
                cand[j] = max(0, cur[j] + d)
        elif r < 0.85:
            # Multi-point
            for _ in range(4):
                k = int(rng.integers(0, n))
                cand[k] *= np.exp(rng.normal(0.0, 0.02))
        else:
            # Global small
            cand *= np.exp(rng.normal(0.0, 0.005, n))

        cand = np.clip(cand, 0.0, 1000.0)
        s_c = np.sum(cand)
        if s_c < 0.01:
            continue
        cand = cand * (S / s_c)
        new_s = compute_score(cand)

        if new_s < cur_s:
            cur = cand
            cur_s = new_s
            if cur_s < best_s:
                best_s = cur_s
                best = cur.copy()
        elif rng.random() < np.exp(-(new_s - cur_s) / max(temp, 1e-15)):
            cur = cand
            cur_s = new_s

        temp *= cooling

    return best, best_s


def make_init(n, rng, style='chirp'):
    """Create various initial sequences."""
    S = np.sqrt(2.0 * n)
    t = np.arange(n, dtype=np.float64)
    if style == 'chirp':
        a = 1.0 + 0.6 * np.cos(np.pi * t * t / (n * 0.8))
        a += 0.04 * rng.standard_normal(n)
    elif style == 'multi':
        a = np.ones(n)
        for k in range(1, 8):
            a += (0.3 / k) * np.cos(2 * np.pi * rng.uniform(0.2, 3.0) * k * t / n + rng.uniform(0, 2 * np.pi))
        a += 0.03 * rng.standard_normal(n)
    elif style == 'block':
        a = np.ones(n)
        for _ in range(n // 32):
            w = rng.integers(4, n // 8)
            st = rng.integers(0, n - w)
            a[st:st + w] *= rng.uniform(0.4, 1.6)
        a += 0.02 * rng.standard_normal(n)
    else:
        a = 1.0 + 0.35 * rng.standard_normal(n)
    a = np.clip(a, 0.01, None)
    return a * (S / np.sum(a))


def resample(seq, new_n):
    """Resample sequence to new length."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Fast reduced-LP from parent
    Phase 2: Full LP refinement
    Phase 3: Nesterov subgradient
    Phase 4: Coordinate descent
    Phase 5: Simulated annealing polish
    Phase 6: Final LP boost
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 40.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load parent
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(parent) > 0:
            n_par = len(parent)
            S_par = np.sqrt(2.0 * n_par)
            parent = norm_to(parent, S_par)
            best_score = compute_score(parent)
            best_seq = parent.copy()
            print(f"Parent loaded: n={n_par}, score={best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(512)
        best_score = compute_score(best_seq)
        print(f"Default init: score={best_score:.6f}")

    n = len(best_seq)

    # Phase 1: Reduced LP from parent (fast, many iterations)
    print(f"\n=== Phase 1: Reduced LP (n={n}) ===")
    t1 = min(600.0, max(10.0, deadline - time.monotonic() - 600))
    print(f"  Time budget: {t1:.0f}s")
    f, s = lp_iterate(best_seq, t1, rng, reduced=True, n_c=60)
    if s < best_score:
        best_score = s
        best_seq = f
        print(f"  Score: {best_score:.6f} (improved)")
    else:
        print(f"  No improvement (score={best_score:.6f})")

    # Phase 2: Full LP for accuracy
    print(f"\n=== Phase 2: Full LP ===")
    t2 = min(500.0, max(10.0, deadline - time.monotonic() - 400))
    if t2 > 30:
        print(f"  Time budget: {t2:.0f}s")
        f, s = lp_iterate(best_seq, t2, rng, reduced=False, n_c=80)
        if s < best_score:
            best_score = s
            best_seq = f
            print(f"  Score: {best_score:.6f} (improved)")
        else:
            print(f"  No improvement (score={best_score:.6f})")

    # Phase 3: Nesterov subgradient
    print(f"\n=== Phase 3: Nesterov subgradient ===")
    t3 = min(250.0, max(10.0, deadline - time.monotonic() - 250))
    if t3 > 20:
        print(f"  Time budget: {t3:.0f}s")
        f, s = nesterov_descent(best_seq, t3)
        if s < best_score:
            best_score = s
            best_seq = f
            print(f"  Score: {best_score:.6f} (improved)")
        else:
            print(f"  No improvement (score={best_score:.6f})")

    # Phase 4: Coordinate descent
    print(f"\n=== Phase 4: Coordinate descent ===")
    t4 = min(200.0, max(10.0, deadline - time.monotonic() - 100))
    if t4 > 20:
        print(f"  Time budget: {t4:.0f}s")
        f, s = coord_descent(best_seq, t4)
        if s < best_score:
            best_score = s
            best_seq = f
            print(f"  Score: {best_score:.6f} (improved)")
        else:
            print(f"  No improvement (score={best_score:.6f})")

    # Phase 5: Simulated annealing
    print(f"\n=== Phase 5: Simulated annealing ===")
    t5 = min(200.0, max(10.0, deadline - time.monotonic() - 50))
    if t5 > 20:
        print(f"  Time budget: {t5:.0f}s")
        f, s = simulated_annealing(best_seq, t5, rng)
        if s < best_score:
            best_score = s
            best_seq = f
            print(f"  Score: {best_score:.6f} (improved)")
        else:
            print(f"  No improvement (score={best_score:.6f})")

    # Phase 6: Final short LP if time remains
    remaining = deadline - time.monotonic() - 10
    if remaining > 60:
        print(f"\n=== Phase 6: Final LP boost ({remaining:.0f}s) ===")
        f, s = lp_iterate(best_seq, remaining * 0.6, rng, reduced=True, n_c=80)
        if s < best_score:
            best_score = s
            best_seq = f
            print(f"  Score: {best_score:.6f} (improved)")
        # Quick SA polish
        rem2 = deadline - time.monotonic() - 5
        if rem2 > 15:
            f, s = simulated_annealing(best_seq, rem2, rng)
            if s < best_score:
                best_score = s
                best_seq = f
                print(f"  After SA: {best_score:.6f} (improved)")

    print(f"\n{'='*50}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}")
    print(f"{'='*50}")

    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return final_list
