def peak_shave(f0, time_limit, num_peaks=3):

    f = np.maximum(f0.astype(np.float64), 0.0)

    f /= f.sum()
    
    c = fast_convolve(f)
    best_obj = c.max()

    while time.monotonic() - t0 < time_limit:
        c_max = c.max()
        # Find top peaks
        peak_indices = np.argsort(c)[-num_peaks:][::-1]
        
        # Compute contribution of each f[i] to the peaks
        sg = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    sg[i] += f[j]
        
        # Reduce f at high-contribution positions, increase at low-contribution
        sg_mean = sg.mean()
        f_new = f * np.exp(-alpha * (sg - sg_mean))
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()

            best_f = f_new.copy()
            f = f_new
        else:
            alpha *= 0.9
            if alpha < 0.001:

            f = f_new  # accept anyway for exploration

def solve_lp(f, M, rows, cols, n, timeout=10):

    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)

                           options={'disp': False, 'time_limit': timeout})

    return None

def gen_inits(n, rng):
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uni'] = np.ones(n) / n
    
    for sf in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5*n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf*100)}'] = g
    
    r = np.arange(1, n+1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    
    # Bimodal (concentrated at edges)
    bi = np.zeros(n)
    w = max(1, n // 6)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()
    
    # Cosine
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()
    
    # Random
    rr = rng.dirichlet(np.ones(n) * 0.5)
    inits['rnd'] = rr
    
    # Triangular
    tri = np.maximum(1.0 - np.abs(x - 0.5*n) / (0.5*n), 0.05)
    inits['tri'] = tri / tri.sum()
    
    # Inverse cosine (Concentrated in middle)
    icb = 0.5 * (1 - np.cos(2 * np.pi * x / n))
    icb += 0.01
    inits['icb'] = icb / icb.sum()
    
    # Sparse random
    sp = np.zeros(n)
    num_nonzero = max(1, n // 4)
    idx = rng.choice(n, size=num_nonzero, replace=False)
    sp[idx] = rng.uniform(0.5, 1.0, size=num_nonzero)
    inits['sparse'] = sp / sp.sum()
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated mirror descent
with active-set focusing, random restarts, LP-guided refinements, and multi-phase
time allocation. Key improvements over prior approach: (1) Nesterov momentum for
faster convergence, (2) active-set weighting that emphasizes the top peaks,
(3) periodic random perturbations to escape local minima, (4) symmetric sequence
exploitation, and (5) careful multi-phase time budgeting.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT with zero-padding."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_subgrad_active(f, c, frac=0.85):
    """Compute subgradient of max(c) using active-set weighting.
    Active set = {k : c[k] >= frac * max(c)}.
    Returns gradient vector of length n."""
    n = len(f)
    c_max = c.max()
    if c_max < 1e-15:
        return np.zeros(n)
    thresh = frac * c_max
    active = c >= thresh
    w = active.astype(np.float64)
    w_sum = w.sum()
    if w_sum < 1e-10:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        w_sum = 1.0
    else:
        w /= w_sum
    # sg[i] = sum_k w[k] * f[k-i] (with f[x]=0 outside [0,n-1])
    sg = np.zeros(n)
    for i in range(n):
        k_start = i
        k_end = min(i + n, len(w))
        j_end = k_end - k_start
        if j_end > 0:
            sg[i] = np.dot(f[:j_end], w[k_start:k_end])
    return sg


def compute_subgrad_softmax(f, c, temp=5.0):
    """Compute subgradient using softmax weighting of convolution values."""
    n = len(f)
    c_max = c.max()
    if c_max < 1e-15:
        return np.zeros(n)
    scaled = temp * c / c_max
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-10:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    sg = np.zeros(n)
    for i in range(n):
        k_start = i
        k_end = min(i + n, len(w))
        j_end = k_end - k_start
        if j_end > 0:
            sg[i] = np.dot(f[:j_end], w[k_start:k_end])
    return sg


def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) < 1e-15:
        return np.ones(n) / n
    # Bisection for the simplex projection
    lo = v.min()
    hi = v.max()
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.maximum(v - mid, 0.0))
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    result = np.maximum(v - tau, 0.0)
    s = result.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return result / s


def md_nesterov(f0, time_limit, temp0=15.0, temp_end=3.0, eta0=0.4,
                use_active=True, frac=0.85, nesterov=True,
                perturb_every=200, perturb_scale=0.05):
    """Nesterov-accelerated mirror descent on probability simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f = f / f_sum

    y = f.copy()  # Nesterov momentum variable
    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()
    best_score = 2.0 * n * best_obj

    eta = eta0
    temp = temp0
    beta = 0.9  # Momentum parameter
    patience = 0
    max_patience = 300
    t0 = time.monotonic()
    iter_count = 0
    total_iter = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        total_iter += 1
        c_max = c.max()
        if c_max < 1e-15:
            break

        # Compute subgradient
        if use_active:
            sg = compute_subgrad_active(f, c, frac=frac)
        else:
            sg = compute_subgrad_softmax(f, c, temp=temp)

        # Center the subgradient
        sg_c = sg - sg.mean()

        # Nesterov update
        if nesterov:
            y = f + beta * (f - y) if iter_count > 1 else f.copy()
            # Recompute gradient at y for better Nesterov
            c_y = fast_convolve(y)
            if use_active:
                sg_y = compute_subgrad_active(y, c_y, frac=frac)
            else:
                sg_y = compute_subgrad_softmax(y, c_y, temp=temp)
            sg_y_c = sg_y - sg_y.mean()
            # Update: z = y - eta * sg_y_c, then project
            z = y - eta * sg_y_c
            f_new = simplex_project(z)
        else:
            z = f - eta * sg_c
            f_new = simplex_project(z)

        # Evaluate
        c = fast_convolve(f_new)
        obj = c.max()
        score = 2.0 * n * obj

        if score < best_score - 1e-14:
            best_score = score
            best_obj = obj
            best_f = f_new.copy()
            patience = 0
        else:
            patience += 1

        f = f_new

        # Adaptive step size and temperature
        if patience > 50:
            eta *= 0.85
            temp = max(temp_end, temp * 0.95)
            if nesterov:
                beta = max(0.5, beta * 0.95)
            patience = 0
        else:
            temp = max(temp_end, temp * 0.999)
            eta = min(1.0, eta * 1.001)

        # Periodic perturbation to escape local minima
        if perturb_every > 0 and total_iter % perturb_every == 0:
            rng_p = np.random.default_rng(total_iter)
            f_pert = f + perturb_scale * rng_p.standard_normal(n)
            f_pert = simplex_project(f_pert)
            c_pert = fast_convolve(f_pert)
            score_pert = 2.0 * n * c_pert.max()
            if score_pert < best_score:
                f = f_pert
                y = f_pert.copy()
                c = c_pert
                best_score = score_pert
                best_obj = c_pert.max()
                best_f = f_pert.copy()
                patience = 0

        if patience > max_patience:
            break

        # Progress reporting
        if total_iter % 200 == 1:
            elapsed = time.monotonic() - t0
            print(f"    MD-N: iter={total_iter}, score={best_score:.8f}, "
                  f"eta={eta:.4f}, temp={temp:.2f}, beta={beta:.3f}, "
                  f"pat={patience}, t={elapsed:.1f}s")

    return best_f, best_score


def lp_optimize(f0, time_limit, max_iters=150):
    """LP-guided alternating optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum

    c = fast_convolve(f)
    M = c.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    # Build LP structure
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)

    t0 = time.monotonic()
    t_mix = 1.0
    patience = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for all k
        vals = np.tile(f, n)
        A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = optimize.linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds,
                               method='highs', options={'disp': False, 'time_limit': 10})
        if not res.success:
            break

        g = np.maximum(res.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        if np.max(np.abs(g - f)) < 1e-14:
            break

        improved = False
        for t_try in [t_mix, 0.7 * t_mix, 0.5 * t_mix, 0.3 * t_mix, 0.1 * t_mix]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            c_new = fast_convolve(f_new)
            obj_new = 2.0 * n * c_new.max()
            if obj_new < best_obj - 1e-13:
                f = f_new
                M = c_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t_mix = min(1.0, t_mix * 1.03)
                improved = True
                break

        if not improved:
            patience += 1
            t_mix *= 0.85
            if patience > 25:
                break

    return best_f, best_obj


def gen_diverse_inits(n, rng):
    """Generate diverse initializations for length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uni'] = np.ones(n) / n

    # Gaussians with various widths
    for sf in [0.08, 0.12, 0.18, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100):02d}'] = g

    # Bimodal (concentrated at edges)
    for w_frac in [0.1, 0.15, 0.2, 0.3]:
        bi = np.zeros(n)
        w = max(1, int(w_frac * n))
        bi[:w] = 1.0
        bi[-w:] = 1.0
        inits[f'bi{int(w_frac * 100):02d}'] = bi / bi.sum()

    # Cosine (concentrated in middle)
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.05
    inits['cos'] = cb / cb.sum()

    # Inverse cosine
    icb = 0.5 * (1 - np.cos(2 * np.pi * x / n)) + 0.05
    inits['icb'] = icb / icb.sum()

    # Ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    # Random Dirichlet
    inits['rnd'] = rng.dirichlet(np.ones(n) * 0.5)

    # Sparse random
    sp = np.zeros(n)
    nnz = max(2, n // 3)
    idx = rng.choice(n, size=nnz, replace=False)
    sp[idx] = rng.uniform(0.3, 1.0, size=nnz)
    inits['sp'] = sp / sp.sum()

    # Symmetric random
    half = n // 2
    sym = rng.dirichlet(np.ones(half) * 0.5)
    full = np.zeros(n)
    full[:half] = sym
    full[half:] = sym[::-1]
    if n % 2 == 1:
        full[half] += sym[half - 1]
    full = np.maximum(full, 0)
    inits['sym'] = full / full.sum()

    return inits
