def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    sum_a = np.sum(a)

    cols = []

"""
Multi-phase optimization for minimizing normalized peak autocorrelation of non-negative step functions.
Combines active-set LP iterations, smoothed max (log-sum-exp) with Adam, and subgradient polishing.
Uses multiple initializations and adaptive time allocation to beat the 1.5027 target.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_A_sparse_fast(f, active_lags):
    """Build sparse constraint matrix A[k,j] = f[k-j] if valid."""
    n = len(f)
    indptr = [0]
    indices = []
    data = []
    for k in active_lags:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            indices.append(j)
            data.append(f[k - j])
        indptr.append(len(indices))
    return csr_matrix((data, indices, indptr), shape=(len(active_lags), n))


def solve_lp_fast(f, M, active_lags, time_limit=5.0):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    A = build_A_sparse_fast(f, active_lags)
    c_obj = -np.ones(n)
    b_ub = np.full(len(active_lags), M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': time_limit})
    if result.success:
        return result.x
    return None


def lp_optimize_active(f_init, time_limit, verbose=False):
    """LP-based iterative optimization with active set and adaptive step."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    t = 0.4
    stagnant = 0
    iteration = 0
    rng_local = np.random.default_rng(99)

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        M = np.max(conv)

        # Active set: lags with conv >= 0.92*M (or top 20)
        threshold = 0.92 * M
        active_mask = conv >= threshold
        if np.sum(active_mask) < 10:
            top_k = max(10, len(conv) // 10)
            active_mask = np.zeros(len(conv), dtype=bool)
            active_mask[np.argsort(conv)[-top_k:]] = True
        active_lags = np.where(active_mask)[0].tolist()

        g = solve_lp_fast(f, M, active_lags, time_limit=3.0)
        if g is None:
            t *= 0.9
            if t < 0.05:
                break
            continue

        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break

        best_s = np.inf

            s = np.sum(f_try)
            if s < 1e-10:

            f_try = f_try * (sum_f / s)
            sc = local_score(f_try)
            if sc < best_s:

        t = np.clip(best_t, 0.05, 0.9)

            best_score = score_new

    return best_f, best_score


def smooth_gradient(a, w, n):
    """Compute gradient of smoothed max: grad[i] = 2*sum_k w[k]*a[k-i]."""
    a_flipped = a[::-1]
    N = 1
    while N < (2 * n - 1) + n - 1:
        N <<= 1
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a_flipped, N)
    conv = np.fft.irfft(W * A, N)
    G = conv[n - 1:n - 1 + n]
    return 2.0 * G


def smooth_max_optimize(f_init, time_limit, T_start=30.0, T_end=300.0, verbose=False):
    """Optimize using smoothed max (log-sum-exp) with Adam."""
    n = len(f_init)
    S = np.sum(f_init)
    a = f_init.astype(np.float64).copy()
    a = a / S  # normalize to sum=1
    S_norm = 1.0

    best_a = a.copy()
    best_score = local_score(a * np.sqrt(2 * n))
    t0 = time.monotonic()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    step = 0
    base_lr = 2.0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_start * (T_end / T_start) ** elapsed_frac

        conv = fft_convolve(a, a)

        # Softmax weights
        conv_scaled = T * conv
        conv_scaled = conv_scaled - np.max(conv_scaled)
        w = np.exp(conv_scaled)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            break
        w = w / w_sum

        # Gradient
        grad = smooth_gradient(a, w, n)
        grad = grad - np.mean(grad)  # project to maintain sum=1

        step += 1
        lr = base_lr / np.sqrt(1.0 + 0.02 * step)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0, None)
        sum_a = np.sum(a)
        if sum_a < 1e-10:
            a = best_a.copy()
            break
        a = a / sum_a

        if step % 10 == 0:
            score = local_score(a * np.sqrt(2 * n))
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose and step % 200 == 0:
            score = local_score(a * np.sqrt(2 * n))
            print(f"    Smooth step={step}, T={T:.1f}, score={score:.7f}, best={best_score:.7f}")

    result = best_a * np.sqrt(2 * n)
    return result, local_score(result)


def subgradient_polish(f_init, time_limit, verbose=False):
    """Subgradient method with momentum for final polishing."""
    n = len(f_init)
    S = np.sum(f_init)
    a = f_init.copy() / S  # normalize to sum=1
    best_a = a.copy()
    best_score = local_score(a * np.sqrt(2 * n))
    t0 = time.monotonic()

    y = a.copy()
    momentum = 0.92
    base_lr = 0.3
    step = 0

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        lr = base_lr / (1.0 + 0.03 * step)
        mu = 0.5 * (1 + momentum) / (1 + elapsed_frac * 0.5)

        conv = fft_convolve(y, y)
        k_star = int(np.argmax(conv))

        # Subgradient at peak lag
        s = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                s[i] = 2.0 * y[j]
        s = s - np.mean(s)

        y = y - lr * s
        y = np.clip(y, 0, None)
        sum_y = np.sum(y)
        if sum_y < 1e-10:
            y = best_a.copy()
            break
        y = y / sum_y

        a = (1 - mu) * a + mu * y
        a = np.clip(a, 0, None)
        sum_a = np.sum(a)
        if sum_a < 1e-10:
            a = best_a.copy()
            break
        a = a / sum_a

        step += 1
        if step % 20 == 0:
            score = local_score(a * np.sqrt(2 * n))
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose and step % 500 == 0:
            score = local_score(a * np.sqrt(2 * n))
            print(f"    Subgrad step={step}, score={score:.7f}, best={best_score:.7f}")

    result = best_a * np.sqrt(2 * n)
    return result, local_score(result)


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / (n * 0.5))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(8):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.03, 0.15)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.25):

    return np.clip(a, 0.01, None)


def make_gaussian_lobes_init(n, rng):
    """Sequence with Gaussian lobes at random positions."""
    a = np.zeros(n)
    n_lobes = int(rng.integers(2, 5))
    for _ in range(n_lobes):
        center = rng.uniform(0, n)
        width = rng.uniform(n * 0.05, n * 0.2)
        amp = rng.uniform(0.5, 1.5)
        t = np.arange(n)
        a += amp * np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)

            a_fin, score_fin = smooth_max_optimize(best_seq, final_smooth, T_start=50, T_end=500, verbose=False)
            print(f"  Final smooth: score={score_fin:.7f}")
            if score_fin < best_score:
                best_score = score_fin
                best_seq = a_fin.copy()
                print(f"  *** New best: {best_score:.7f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL RESULT: score={final_score:.7f}, n={len(final_list)}")
        print(f"{'=' * 60}")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
