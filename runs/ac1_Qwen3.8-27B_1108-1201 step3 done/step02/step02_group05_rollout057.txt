```python
    """Optimize in softmax parameterization."""
    n = len(b0)
    b = b0.copy().astype(np.float64)
    best_b = b.copy()
    best_score = np.inf
    
    def get_a(b):
        b_max = b.max()
        a = np.exp(b - b_max)
        a /= a.sum()
        return a
    
    t0 = time.monotonic()
    alpha = 0.1  # step size
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a = get_a(b)
        conv = fast_convolve(a)
        max_conv = conv.max()

        beta = beta_start + (beta_end - beta_start) * it / 100000
        shifted = conv - max_conv
        exp_vals = np.exp(beta * shifted)
        weights = exp_vals / exp_vals.sum()
        
        # Gradient of (1/beta)*log(sum exp(beta*conv_k)) with respect to a[j]
        # = sum_k weights[k] * d(conv_k)/d(a[j])
        # d(conv_k)/d(a[j]) = a[k-j] (if valid) + a[j-k] (if valid)

        corr2 = np.fft.irfft(W * np.conj(A), size)  # sum_k w[k]*a[j-k]
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smoothed gradient descent (softmax approximation of max)
2. Softmax parametrization for non-negativity constraint
3. Adaptive beta annealing
4. Periodic LP-based refinement
5. Local search polishing

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]


def compute_smoothed_grad(a, beta):
    """Compute gradient of (1/beta)*log(sum_k exp(beta*conv_k)) w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    weights = np.exp(beta * (conv - max_c))
    weights /= weights.sum()
    
    size = 1 << (2 * n - 1).bit_length()
    W = np.fft.rfft(weights, size)
    A = np.fft.rfft(a, size)
    
    # term1[j] = sum_k weights[k] * a[k-j] (for valid) = cross-correlation
    term1 = np.fft.irfft(W * np.conj(A), size)[:n]
    # term2[j] = sum_k weights[k] * a[j-k] (for valid) = convolution
    term2 = np.fft.irfft(W * A, size)[:n]
    
    return (term1 + term2) / beta


def softmax_obj(a, beta):
    """Smoothed objective: (1/beta)*log(sum exp(beta*conv_k)) + max_conv."""
    conv = fast_convolve(a)
    max_c = conv.max()
    return (1.0/beta) * np.log(np.sum(np.exp(beta * (conv - max_c)))) + max_c


def gradient_descent_smoothed(a0, time_limit, beta_start=5.0, beta_end=500.0):
    """Gradient descent in softmax parametrization."""
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    # Initialize b in softmax parametrization
    b = np.log(a + 1e-30)
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    alpha = 0.05
    n_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        n_iters += 1
        progress = min(1.0, n_iters / 5000.0)
        beta = beta_start + (beta_end - beta_start) * progress
        
        # Compute gradient in a-space
        grad_a = compute_smoothed_grad(a, beta)
        
        # Convert to b-space gradient
        # a[j] = exp(b[j]) / sum(exp(b)), so d(a[j])/d(b[i]) = a[j]*(delta_ji - a[i])
        # grad_b[i] = sum_j grad_a[j] * d(a[j])/d(b[i]) = sum_j grad_a[j] * a[j] * (delta_ji - a[i])
        #           = a[i]*grad_a[i] - a[i]*sum_j(grad_a[j]*a[j])
        dot = np.dot(grad_a, a)
        grad_b = a * (grad_a - dot)
        
        # Step in b-space (no constraints needed)
        b_new = b - alpha * grad_b
        
        # Get new a
        b_max = b_new.max()
        a_new = np.exp(b_new - b_max)
        a_new /= a_new.sum()
        
        # Evaluate
        obj_new = 2.0 * n * fast_convolve(a_new).max()
        
        if obj_new < best_obj - 1e-14:
            a = a_new
            b = b_new
            best_obj = obj_new
            best_a = a.copy()
            # Increase step size slightly if we made good progress
            alpha = min(alpha * 1.05, 0.5)
        else:
            alpha *= 0.7
            if alpha < 1e-6:
                alpha = 0.01
                # Try a different direction: random perturbation
                b = b + 0.01 * np.random.randn(n)
    
    return best_a, best_obj
```

`∝ a[j] · exp(-η·grad_a[j])` (the dot term is a constant across j, so it cancels in the normalization)

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smoothed gradient (log-sum-exp approximation of max)
2. Mirror descent / softmax parametrization for non-negativity
3. Adaptive beta annealing and step size
4. Periodic LP-based direction refinement
5. Frequency-domain power spectrum flattening
6. Local search polishing with pairwise exchanges
7. Multiple sequence lengths and diverse initializations
8. Restart mechanism to escape local minima
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def compute_smoothed_grad(a, beta):
    """Compute gradient of (1/beta)*log(sum_k exp(beta*conv_k)) w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    weights = np.exp(beta * (conv - max_c))
    weights /= weights.sum()

    size = 1 << (2 * n - 1).bit_length()
    W = np.fft.rfft(weights, size)
    A = np.fft.rfft(a, size)

    # term1[j] = sum_k weights[k]*a[k-j] (cross-correlation)
    term1 = np.fft.irfft(W * np.conj(A), size)[:n]
    # term2[j] = sum_k weights[k]*a[j-k] (convolution)
    term2 = np.fft.irfft(W * A, size)[:n]

    return (term1 + term2) / beta


def obj_value(a):
    """Objective: 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def mirror_descent(a0, time_limit, beta_start=5.0, beta_end=1000.0,
                   eta_init=0.1, use_ada=False):
    """
    Mirror descent (equivalent to gradient descent in softmax parametrization).
    Uses log-sum-exp smoothed objective with adaptive beta.
    """
    n = len(a0)
    a = normalize_nonneg(a0.copy())
    best_a = a.copy()
    best_obj = obj_value(a)

    t0 = time.monotonic()
    eta = eta_init
    n_iters = 0
    stall_count = 0
    max_stall = 200

    # Adaptive learning rate components
    exp_avg_sq = np.zeros(n)
    step_count = 0

    while time.monotonic() - t0 < time_limit and stall_count < max_stall:
        n_iters += 1
        progress = min(1.0, n_iters / max(1, int(time_limit * 100)))
        beta = beta_start + (beta_end - beta_start) * (progress ** 1.5)

        grad_a = compute_smoothed_grad(a, beta)

        if use_ada:
            step_count += 1
            exp_avg_sq = 0.9 * exp_avg_sq + 0.1 * (grad_a ** 2)
            a_new = a * np.exp(-eta * grad_a / (np.sqrt(exp_avg_sq) + 1e-8))
        else:
            a_new = a * np.exp(-eta * grad_a)

        a_new = normalize_nonneg(a_new)
        obj_new = obj_value(a_new)

        if obj_new < best_obj - 1e-14:
            a = a_new
            best_obj = obj_new
            best_a = a.copy()
            stall_count = 0
            eta = min(eta * 1.01, 2.0)
        else:
            stall_count += 1
            eta *= 0.85
            if eta < 1e-8:
                # Restart with perturbation
                a = normalize_nonneg(a + 0.05 * np.random.randn(n))
                eta = 0.05

    return best_a, best_obj


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g."""
    n = len(f)
    n_constraints = 2 * n - 1
    A = np.zeros((n_constraints, n))
    for j in range(n):
        start_k = 0
        end_k = min(n_constraints, j + n)
        A[start_k:end_k, j] = f[:end_k - start_k]
    c = -np.ones(n)
    b = np.full(n_constraints, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def lp_step(f, time_limit=5.0):
    """Perform one LP-based step with line search."""
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    if M < 1e-15:
        return f.copy(), obj_value(f)

    t0 = time.monotonic()
    g = solve_lp(f, M)
    if time.monotonic() - t0 > time_limit:
        return f.copy(), obj_value(f)

    # Line search: minimize obj((1-t)*f + t*g) for t in [0, 1]
    best_t = 0.0
    best_obj = obj_value(f)
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        a_trial = normalize_nonneg((1 - t) * f + t * g)
        o = obj_value(a_trial)
        if o < best_obj:
            best_obj = o
            best_t = t

    if best_t > 0:
        return normalize_nonneg((1 - best_t) * f + best_t * g), best_obj
    return f.copy(), obj_value(f)


def freq_flatten(a, eta=0.5):
    """Flatten power spectrum in frequency domain."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    P = np.abs(A) ** 2
    mean_P = P.mean()
    if mean_P < 1e-15:
        return a.copy()
    A_new = A * np.exp(-eta * (P - mean_P) / (2.0 * mean_P))
    a_new = np.fft.irfft(A_new, size)[:n]
    return normalize_nonneg(a_new)


def local_search(a, time_limit):
    """Pairwise exchange local search focused on peak reduction."""
    n = len(a)
    a = normalize_nonneg(a.copy())
    best_obj = obj_value(a)
    t0 = time.monotonic()
    stuck = 0
    max_stuck = 100

    while time.monotonic() - t0 < time_limit and stuck < max_stuck:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))

        # Compute contribution of each pair to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        max_c = contrib.max()
        if max_c < 1e-15:
            break

        # Find high-contrib and low-contrib indices
        sorted_idx = np.argsort(contrib)
        high = sorted_idx[-20:]
        low = sorted_idx[:20]

        found = False
        for di in high:
            if found:
                break
            for si in low:
                if found:
                    break
                if di == si:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    amount = a[di] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[di] -= amount
                    a_trial[si] += amount
                    a_trial = normalize_nonneg(a_trial)
                    o = obj_value(a_trial)
                    if o < best_obj - 1e-12:
                        best_obj = o
                        a = a_trial.copy()
                        found = True
                        break

        if found:
            stuck = 0
        else:
            stuck += 1

    return a, best_obj


def alternating_optimize(a0, time_limit):
    """Alternate between mirror descent, LP step, and frequency flattening."""
    n = len(a0)
    a = normalize_nonneg(a0.copy())
    best_a = a.copy()
    best_obj = obj_value(a)
    t0 = time.monotonic()

    n_iters = 0
    while time.monotonic() - t0 < time_limit:
        n_iters += 1
        phase = n_iters % 10

        if phase < 6:
            # Mirror descent (60% of time)
            md_time = min(1.0, (time_limit - (time.monotonic() - t0)) * 0.1)
            a_new, o_new = mirror_descent(a, time_limit=md_time,
                                          beta_start=10.0, beta_end=500.0,
                                          eta_init=0.1)
            if o_new < best_obj:
                a = a_new
                best_obj = o_new
                best_a = a.copy()
        elif phase < 8:
            # LP step (20% of time)
            lp_time = min(3.0, (time_limit - (time.monotonic() - t0)) * 0.1)
            a_new, o_new = lp_step(a, time_limit=lp_time)
            if o_new < best_obj:
                a = a_new
                best_obj = o_new
                best_a = a.copy()
        else:
            # Frequency flattening (20% of time)
            for eta_f in [0.1, 0.3, 0.5, 1.0]:
                a_new = freq_flatten(a, eta=eta_f)
                o_new = obj_value(a_new)
                if o_new < best_obj:
                    a = a_new
                    best_obj = o_new
                    best_a = a.copy()
                    break

        # Check time
        if time.monotonic() - t0 > time_limit:
            break

    return best_a, best_obj


def get_parent(n):
    """Get parent sequence clipped/resized to length n."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim != 1 or p.size == 0 or not np.isfinite(p).all():
            return None
        if p.size >= n:
            pc = p[:n].copy()
        else:
            pc = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s < 1e-10:
            return None
        return pc / s
    except Exception:
        return None


def make_initializations(n, rng):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Gaussian
    for sf in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits[f'gauss_{sf}'] = g / g.sum()

    # Hann window
    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    else:
        hann = np.ones(n)
    hann = np.maximum(hann, 0.0)
    inits['hann'] = hann / hann.sum()

    # Edges
    m = max(1, n // 4)
    edges = np.zeros(n)
    edges[:m] = 1.0
    edges[-m:] = 1.0
    inits['edges'] = edges / edges.sum()

    # Random Dirichlet
    inits['dirichlet'] = rng.dirichlet(np.ones(n) * 0.5)

    # Random Dirichlet (concentrated)
    inits['dirichlet_conc'] = rng.dirichlet(np.ones(n) * 0.1)

    # Parent
    parent = get_parent(n)
    if parent is not None:
        inits['parent'] = parent
        # Perturbed parent
        for eps in [0.05, 0.1, 0.2]:
            p_pert = parent * (1.0 + eps * rng.randn(n))
            inits[f'parent_pert_{eps}'] = normalize_nonneg(p_pert)

    # Frequency-flattened uniform (iterate a few times)
    a_ff = np.ones(n) / n
    for _ in range(20):
        a_ff = freq_flatten(a_ff, eta=0.5)
    inits['freq_flat'] = a_ff

    return inits

                a_ff = freq_flatten(a_current, eta=0.3)
                a_new, o_new = mirror_descent(a_ff, time_limit=min(3.0, remaining * 0.2),
                                              beta_start=50.0, beta_end=500.0,
                                              eta_init=0.1)
                if o_new < obj_value(a_current):
                    a_current = a_new
                try_candidate(a_new, f"polish_ff_{polish_iter}")

    # Phase 4: Final intensive local search (5%)
    print("Phase 4: Final local search polish")
    if best_sequence is not None:
        remaining = deadline - time.monotonic()
        if remaining > 5:
            a_final, o_final = local_search(best_sequence, time_limit=min(remaining - 5, 60.0))
            try_candidate(a_final, "final_ls")

            # Additional mirror descent with very high beta (focus on peak)
            remaining = deadline - time.monotonic()
            if remaining > 5:
                a_md, o_md = mirror_descent(a_final, time_limit=min(remaining - 5, 30.0),
                                            beta_start=200.0, beta_end=5000.0,
                                            eta_init=0.03)
                try_candidate(a_md, "final_md")

    # Final evaluation
    if best_sequence is None:
        best_sequence = np.ones(512)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
