Optimized step function peak autocorrelation minimization using
subgradient descent with momentum, LP-based direction finding,

def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

        peak_threshold = max_val * (1.0 - 1e-5)
        peak_indices = np.where(conv >= peak_threshold)[0]
        if len(peak_indices) < 2:
            # Use top 2 positions
            peak_indices = np.argsort(conv)[-k_peaks:]
        
        # Compute subgradient
        grad = compute_subgradient_fft(a, peak_indices)
        
        # Project to sum=0 (tangent to sum(a)=1 constraint)
        grad_proj = grad - np.mean(grad)
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad_proj
        a_new = a - lr * momentum
        a_new = np.clip(a_new, 0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new / s_new
        
        new_conv = fft_convolve(a_new, a_new)
        new_peak = np.max(new_conv)
        new_score = 2.0 * n * new_peak
        
        if new_peak < current_peak:
            a = a_new
            current_peak = new_peak
            lr = min(lr * 1.01, 2.0 / n)
            peak_count = 0
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            lr = max(lr * 0.95, 0.01 / n)
            peak_count += 1
            if peak_count > 200:
                # Reset momentum and try smaller step
                momentum = np.zeros(n)
                lr = 0.1 / n
                peak_count = 0
                if verbose:
                    print(f"  SG: reset at iter {int(peak_count)}, peak={current_peak:.8f}")
        
        if verbose and peak_count % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SG: peak={current_peak:.8f}, score={best_score:.6f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score


def build_constraint_matrix(f):
    """Build the (2n-1) x n constraint matrix."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M):
    """Solve LP to find best descent direction."""
    n = len(f)
    A = build_constraint_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return np.ones(n) / n


def lp_iterate(f0, time_limit, rng, verbose=False):
    """LP-based iterative optimization."""
    n = len(f0)
    a = f0.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
    a = a / s
    
    current_peak = np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_score = 2.0 * n * best_peak
    t0 = time.monotonic()
    t = 0.1
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit:
        M = np.max(fft_convolve(a, a))
        b_0 = solve_lp_direction(a, M)
        
        sum_b = np.sum(b_0)
        if sum_b < 1e-10:
            break
        g = b_0 / sum_b  # normalize to sum=1
        
        improved = False
        for trial_t in [t, t * 0.5, t * 0.25]:
            a_new = (1.0 - trial_t) * a + trial_t * g
            a_new = np.clip(a_new, 0, None)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            new_peak = np.max(fft_convolve(a_new, a_new))
            if new_peak < current_peak:
                a = a_new
                current_peak = new_peak
                t = min(trial_t * 1.1, 0.3)
                improved = True
                stall_count = 0
                new_score = 2.0 * n * current_peak
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                break
        
        if not improved:
            stall_count += 1
            t *= 0.8
            if stall_count > 50:
                # Perturb
                a = a * np.exp(rng.normal(0, 0.05, n))
                a = np.clip(a, 0, None)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                    current_peak = np.max(fft_convolve(a, a))
                stall_count = 0
                t = 0.05
        
        if verbose and stall_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP: peak={current_peak:.8f}, best={best_score:.6f}, stale={stall_count}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, rng, verbose=False):
    """Coordinate descent: try increasing each coordinate."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a0, local_score(a0)
    a = a / s
    
    current_peak = np.max(fft_convolve(a, a))
    best_a = a.copy()
    best_score = 2.0 * n * current_peak
    
    t0 = time.monotonic()
    step_size = 0.1
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        # Try each coordinate in random order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            # Try increasing coordinate i
            for delta in [step_size * a[i] * 0.1, step_size * a[i] * 0.05, step_size * a[i] * 0.01]:
                a_new = a.copy()
                a_new[i] += delta
                a_new = np.clip(a_new, 0, None)
                s_new = np.sum(a_new)
                if s_new < 1e-10:
                    continue
                a_new = a_new / s_new
                new_peak = np.max(fft_convolve(a_new, a_new))
                if new_peak < current_peak:
                    a = a_new
                    current_peak = new_peak
                    improved = True
                    new_score = 2.0 * n * current_peak
                    if new_score < best_score:
                        best_score = new_score
                        best_a = a.copy()
                    break
        
        if not improved:
            # Try decreasing some coordinates
            for i in order:
                if time.monotonic() - t0 > time_limit:
                    break
                for delta in [step_size * a[i] * 0.1, step_size * a[i] * 0.05]:
                    a_new = a.copy()
                    a_new[i] = max(0, a_new[i] - delta)
                    s_new = np.sum(a_new)
                    if s_new < 1e-10:
                        continue
                    a_new = a_new / s_new
                    new_peak = np.max(fft_convolve(a_new, a_new))
                    if new_peak < current_peak:
                        a = a_new
                        current_peak = new_peak
                        improved = True
                        new_score = 2.0 * n * current_peak
                        if new_score < best_score:
                            best_score = new_score
                            best_a = a.copy()
                        break
            if not improved:
                step_size *= 0.5
                if step_size < 0.001:
                    break
    
    return best_a, best_score


def simulated_annealing(a0, time_limit, rng, verbose=False):
    """Simulated annealing for final polish."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a0, local_score(a0)
    a = a / s
    
    current_peak = np.max(fft_convolve(a, a))
    current_score = 2.0 * n * current_peak
    best_a = a.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9998
    
    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()
        
        if r < 0.35:
            # Peak-targeted: reduce at high-contribution positions for the peak
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            # Find positions contributing most to the peak
            contribs = np.zeros(n)
            lo = max(0, peak_idx - n + 1)
            hi = min(peak_idx, n - 1)
            for i in range(lo, hi + 1):
                contribs[i] = a[i] * a[peak_idx - i]
            top_k = max(3, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.01, size=len(top_idx)))
            # Increase at random other positions
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 5:
                n_inc = min(10, len(other) // 3)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.01, size=n_inc))
        elif r < 0.55:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.70:
            # Multi-block
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 16))))
                start = int(rng.integers(0, max(1, n - width)))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
        elif r < 0.85:
            # Pair exchange
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                amount = rng.uniform(0.02, 0.1) * a[i]
                candidate[i] = max(0, a[i] - amount)
                candidate[j] = a[j] + amount
        else:
            # Global small perturbation
            candidate = a * np.exp(rng.normal(0.0, 0.008, size=n))
        
        candidate = np.clip(candidate, 0.0, None)
        s_c = np.sum(candidate)
        if s_c < 1e-10:
            continue
        candidate = candidate / s_c
        
        new_conv = fft_convolve(candidate, candidate)
        new_peak = np.max(new_conv)
        new_score = 2.0 * n * new_peak
        
        if new_score < current_score:
            a = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            a = candidate
            current_score = new_score
        
        temperature *= cooling
    
    return best_a, best_score


def resample_sequence(seq, new_n):
    """Resample sequence to new length."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)

            # Ternary search for optimal x

            lo_x, hi_x = 0.0, x_max
            for _ in range(30):

                m2 = hi_x - (hi_x - lo_x) / 3.0
                # f(x) = max_k(c[k] + 2*x*b[k] + x^2*delta_{k,2i})
                # = max(max_k≠2i(c[k] + 2*x*b[k]), c[2i] + 2*x*b[2i] + x^2)

                a[i] = x_opt

    return best_a, best_score

"""
Optimized step function peak autocorrelation minimization using:
1. Fast subgradient descent with FFT-based gradient and momentum
2. Optimal coordinate descent with 1D ternary search per coordinate
3. LP-based direction finding (reduced time)
4. Simulated annealing polish
5. Final coordinate descent refinement

Key innovation over previous approach: The optimal coordinate descent
solves the exact 1D subproblem at each coordinate using the analytical
structure of the convolution, enabling much more efficient local
optimization than the previous LP-heavy approach.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the evaluation score for a sequence (fast local version)."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def compute_subgradient_fft(a, peak_indices):
    """Compute subgradient of max_k Q_k(a) using FFT."""
    n = len(a)
    nconv = 2 * n - 1
    P = np.zeros(nconv)
    for k in peak_indices:
        P[k] = 1.0 / len(peak_indices)
    a_rev = a[::-1]
    P_rev = P[::-1]
    size = 1
    while size < 2 * n + 1:
        size <<= 1
    fa = np.fft.rfft(a_rev, size)
    fP = np.fft.rfft(P_rev, size)
    conv = np.fft.irfft(fa * fP, size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad


def subgradient_descent(a0, time_limit, rng, verbose=False):
    """Subgradient descent with momentum for minimizing peak autocorrelation."""
    n = len(a0)
    nconv = 2 * n - 1
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s

    current_conv = fft_convolve(a, a)
    current_peak = np.max(current_conv)
    best_a = a.copy()
    best_peak = current_peak
    best_score = 2.0 * n * best_peak

    t0 = time.monotonic()
    lr = 0.3 / n
    momentum = np.zeros(n)
    beta = 0.85
    stall = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        max_val = np.max(current_conv)
        # Find top peaks (within 1e-6 relative)
        thresh = max_val * (1.0 - 5e-6)
        peak_mask = current_conv >= thresh
        peak_indices = np.where(peak_mask)[0]
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(current_conv)])
        elif len(peak_indices) > 10:
            # Use top 10 by value
            top_vals = np.sort(current_conv)[-10:]
            thresh2 = top_vals[0]
            peak_indices = np.where(current_conv >= thresh2)[0]

        # Compute subgradient
        grad = compute_subgradient_fft(a, peak_indices)
        # Project to tangent (sum=0)
        grad_proj = grad - np.mean(grad)

        # Momentum
        momentum = beta * momentum + (1.0 - beta) * grad_proj
        a_new = a - lr * momentum
        a_new = np.clip(a_new, 0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            lr *= 0.5
            continue
        a_new = a_new / s_new

        new_conv = fft_convolve(a_new, a_new)
        new_peak = np.max(new_conv)

        if new_peak < current_peak:
            a = a_new
            current_conv = new_conv
            current_peak = new_peak
            lr = min(lr * 1.005, 1.5 / n)
            stall = 0
            new_score = 2.0 * n * current_peak
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            lr = max(lr * 0.97, 0.02 / n)
            stall += 1
            if stall > 150:
                momentum = np.zeros(n)
                lr = 0.1 / n
                stall = 0
                # Small perturbation
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = np.clip(a, 0, None)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                    current_conv = fft_convolve(a, a)
                    current_peak = np.max(current_conv)

        iters += 1
        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG: iter={iters}, score={best_score:.6f}, peak={current_peak:.8f}, lr={lr:.5f}, t={elapsed:.1f}s")

    return best_a, best_score


def coordinate_descent_optimal(a0, time_limit, rng, verbose=False):
    """
    Optimal coordinate descent: at each coordinate, solve the exact 1D
    minimization of max_k (c_k + 2x*b_k + x^2*delta) via ternary search.
    Much faster than LP-based approaches for local optimization.
    """
    n = len(a0)
    nconv = 2 * n - 1
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a0, local_score(a0)
    a = a / s

    current_conv = fft_convolve(a, a)
    current_peak = np.max(current_conv)
    best_a = a.copy()
    best_score = 2.0 * n * current_peak

    t0 = time.monotonic()
    sweeps = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        sweep_start_peak = current_peak
        any_improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[i]

            # b[k] = a[k-i] for valid k
            b = np.zeros(nconv)
            b[i:i + n] = a

            # c = current_conv minus contribution of a[i]
            c = current_conv.copy()
            c[i:i + n] -= 2.0 * x_old * a
            c[2 * i] += x_old * x_old  # correct: subtracted 2*x_old^2, need to subtract only x_old^2

            # Ternary search for x minimizing max_k(c[k] + 2*x*b[k] + x^2*delta_{k,2i})
            x_max = max(5.0 * x_old, 4.0 / n)
            lo_x = 0.0
            hi_x = x_max

            for _ in range(30):
                m1 = lo_x + (hi_x - lo_x) / 3.0
                m2 = hi_x - (hi_x - lo_x) / 3.0
                # Evaluate f(m1)
                v1 = c + 2.0 * m1 * b
                v1[2 * i] += m1 * m1
                f1 = np.max(v1)
                # Evaluate f(m2)
                v2 = c + 2.0 * m2 * b
                v2[2 * i] += m2 * m2
                f2 = np.max(v2)
                if f1 < f2:
                    hi_x = m2
                else:
                    lo_x = m1

            x_opt = (lo_x + hi_x) / 2.0

            if abs(x_opt - x_old) > 1e-12:
                delta = x_opt - x_old
                a[i] = x_opt
                # Incremental convolution update
                current_conv[i:i + n] += 2.0 * delta * b
                current_conv[2 * i] += delta * delta

                new_peak = np.max(current_conv)
                if new_peak < sweep_start_peak:
                    any_improved = True

        current_peak = np.max(current_conv)
        current_score = 2.0 * n * current_peak
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        sweeps += 1
        if not any_improved and no_improve_count > 2:
            break
        if verbose and sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD: sweep={sweeps}, score={best_score:.6f}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def build_constraint_matrix(f):
    """Build the (2n-1) x n constraint matrix for LP."""
    n = len(f)
    nconv = 2 * n - 1
    A = np.zeros((nconv, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M):
    """Solve LP to find best descent direction under peak constraint."""
    n = len(f)
    nconv = 2 * n - 1
    A = build_constraint_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(nconv, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return np.ones(n) / n


def lp_iterate(f0, time_limit, rng, verbose=False):
    """LP-based iterative optimization with line search."""
    n = len(f0)
    a = f0.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s

    current_conv = fft_convolve(a, a)
    current_peak = np.max(current_conv)
    best_a = a.copy()
    best_score = 2.0 * n * current_peak
    t0 = time.monotonic()
    t_mix = 0.12
    stall = 0

    while time.monotonic() - t0 < time_limit:
        M = current_peak
        b_0 = solve_lp_direction(a, M)
        sum_b = np.sum(b_0)
        if sum_b < 1e-10:
            break
        g = b_0 / sum_b

        improved = False
        for trial_t in [t_mix, t_mix * 0.5, t_mix * 0.25]:
            a_new = (1.0 - trial_t) * a + trial_t * g
            a_new = np.clip(a_new, 0, None)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            new_conv = fft_convolve(a_new, a_new)
            new_peak = np.max(new_conv)
            if new_peak < current_peak:
                a = a_new
                current_conv = new_conv
                current_peak = new_peak
                t_mix = min(trial_t * 1.08, 0.3)
                improved = True
                stall = 0
                new_score = 2.0 * n * current_peak
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                break

        if not improved:
            stall += 1
            t_mix *= 0.85
            if stall > 60:
                a = a * np.exp(rng.normal(0, 0.04, n))
                a = np.clip(a, 0, None)
                s = np.sum(a)
                if s > 1e-10:
                    a = a / s
                    current_conv = fft_convolve(a, a)
                    current_peak = np.max(current_conv)
                stall = 0
                t_mix = 0.06

        if verbose and stall % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP: score={best_score:.6f}, stale={stall}, t={elapsed:.1f}s")

    return best_a, best_score


def simulated_annealing(a0, time_limit, rng, verbose=False):
    """Simulated annealing with peak-targeted moves."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        return a0, local_score(a0)
    a = a / s

    current_conv = fft_convolve(a, a)
    current_peak = np.max(current_conv)
    current_score = 2.0 * n * current_peak
    best_a = a.copy()
    best_score = current_score

    t0 = time.monotonic()
    temperature = 0.0015
    cooling = 0.99985
    iters = 0

    while time.monotonic() - t0 < time_limit:
        candidate = a.copy()
        r = rng.random()

        if r < 0.40:
            # Peak-reducing: decrease top contributors to peak
            conv = current_conv  # use cached
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            k_lo = max(0, peak_idx - n + 1)
            k_hi = min(peak_idx, n - 1)
            for i in range(k_lo, k_hi + 1):
                contribs[i] = a[i] * a[peak_idx - i]
            top_k = max(5, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.03, 0.012, size=len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 5:
                n_inc = min(15, len(other) // 3)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.012, size=n_inc))
        elif r < 0.60:
            # Block perturbation
            width = max(1, int(rng.integers(1, max(2, n // 10))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.025, size=width))
        elif r < 0.75:
            # Multi-block
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 20))))
                start = int(rng.integers(0, max(1, n - width)))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.018, size=width))
        elif r < 0.90:
            # Pair redistribution
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                amount = rng.uniform(0.01, 0.08) * a[i]
                candidate[i] = max(0.0, a[i] - amount)
                candidate[j] = a[j] + amount
        else:
            # Global
            candidate = a * np.exp(rng.normal(0.0, 0.006, size=n))

        candidate = np.clip(candidate, 0.0, None)
        s_c = np.sum(candidate)
        if s_c < 1e-10:
            continue
        candidate = candidate / s_c

        new_conv = fft_convolve(candidate, candidate)
        new_peak = np.max(new_conv)
        new_score = 2.0 * n * new_peak

        if new_score < current_score:
            a = candidate
            current_conv = new_conv
            current_peak = new_peak
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            a = candidate
            current_conv = new_conv
            current_peak = new_peak
            current_score = new_score

        temperature *= cooling
        iters += 1
        if verbose and iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA: iter={iters}, score={best_score:.6f}, T={temperature:.6f}, t={elapsed:.1f}s")

    return best_a, best_score


def resample_sequence(seq, new_n):
    """Resample sequence to new length via linear interpolation."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)

        if time_left < 30:
            break
        a_res = resample_sequence(best_seq, target_n)
        # Quick coordinate descent
        cd_time = min(40.0, time_left * 0.4)
        a_best, s_best = coordinate_descent_optimal(a_res, cd_time, rng, verbose=False)
        print(f"  n={target_n}: score={s_best:.6f}")
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # Phase 5: Simulated annealing
    print("\n=== Phase 5: Simulated annealing ===")
    t5 = min(200.0, deadline - time.monotonic() - 50.0)
    if t5 > 20:
        a_best, s_best = simulated_annealing(best_seq, t5, rng, verbose=True)
        print(f"  Phase 5: score={s_best:.6f}")
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # Phase 6: Final coordinate descent polish
    print("\n=== Phase 6: Final CD polish ===")
    t6 = min(150.0, deadline - time.monotonic() - 20.0)
    if t6 > 10:
        a_best, s_best = coordinate_descent_optimal(best_seq, t6, rng, verbose=True)
        print(f"  Phase 6: score={s_best:.6f}")
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"  *** New best: {best_score:.6f} ***")

    # Phase 7: Quick final subgradient if time remains
    if time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            print(f"\n=== Phase 7: Final SG ({remaining:.0f}s) ===")
            a_best, s_best = subgradient_descent(best_seq, remaining, rng, verbose=False)
            if s_best < best_score:
                best_score = s_best
                best_seq = a_best.copy()
                print(f"  Phase 7: score={s_best:.6f} *** New best ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*50}")
        return final_list

    # Fallback
    return [1.0] * 256
