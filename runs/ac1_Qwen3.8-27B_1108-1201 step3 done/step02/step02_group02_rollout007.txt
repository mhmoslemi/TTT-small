```python
"""
Smooth gradient-based optimization for minimizing peak autoconvolution of
non-negative step functions. Uses log-sum-exp smoothing for differentiability,
mirror descent (multiplicative updates) for non-negativity, coarse-to-fine
beta continuation, hard subgradient refinement, and coordinate descent polish.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Fast score computation."""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def smooth_grad_and_score(a, beta):
    """
    Compute the smoothed (log-sum-exp) objective and gradient.
    Returns: (smoothed_score, gradient_vector)
    The gradient is of the smoothed objective w.r.t. each a[i].
    """
    n = len(a)
    c = fft_convolve(a, a)
    S = np.sum(a)

    # Log-sum-exp for numerical stability
    c_max = np.max(c)
    shifted = beta * (c - c_max)
    # Clip for overflow protection
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    log_sum = c_max + (np.log(np.sum(exp_shifted)) - np.log(beta))
    # LSE = (1/beta) * log(sum(exp(beta*c)))
    lse = c_max + np.log(np.sum(exp_shifted)) / beta

    # Softmax weights p[k] = d(LSE)/d(c[k])
    p = exp_shifted / np.sum(exp_shifted)

    # Gradient computation:
    # d(LSE)/d(a[i]) = g1[i] + a[i] * window_sum[i]
    # where g1[i] = sum_j a[j] * p[j+i]  (cross-correlation)
    #       window_sum[i] = sum_{k=i}^{i+n-1} p[k]

    # g1 via correlate: np.correlate(p, a, 'full')[:n]
    # np.correlate(p, a, 'full')[i] = sum_j p[j+i] * a[j]
    g1 = np.correlate(p, a, mode='full')[:n]

    # window_sum via prefix sums
    p_prefix = np.zeros(2 * n)
    np.cumsum(p, out=p_prefix[1:])
    # window_sum[i] = p_prefix[i+n] - p_prefix[i] for i=0..n-1
    window_sum = p_prefix[n:n + n] - p_prefix[:n]

    grad = g1 + a * window_sum

    # The actual score (using true max, not smoothed):
    true_score = 2.0 * n * c_max / (S ** 2)

    return true_score, lse, grad, p, c


def hardsub_grad(a):

    Compute the hard subgradient at the actual peak position.
    Returns gradient vector.
    """
    n = len(a)
    c = fft_convolve(a, a)
    peak_idx = int(np.argmax(c))

    grad = np.zeros(n)
    for i in range(n):

        if 0 <= j < n:
            if j == i:
                grad[i] = 2.0 * a[i]
            else:
                grad[i] = a[j] + a[i]
        # else grad[i] = 0

    return grad, c, peak_idx


def mirror_descent_step(a, grad, lr, S_target):
    """
    Mirror descent (exponential) step.
    a[i] *= exp(-lr * a[i] * grad[i])
    Then renormalize to sum = S_target.
    """
    n = len(a)
    log_update = -lr * a * grad
    # Clip for numerical stability
    log_update = np.clip(log_update, -10, 10)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 1e-15)
    S_new = np.sum(a_new)
    if S_new > 0.01:
        a_new = a_new * (S_target / S_new)
    return a_new

    best_score = current_score
    best_a = a.copy()

    t0 = time.monotonic()
    total_time = time_limit
    lr = 0.1
    stall_count = 0

    # Phase the beta: start low, end high
    n_phases = 5
    phase_time = total_time / n_phases

    for phase in range(n_phases):
        beta = beta_start * (beta_end / beta_start) ** (phase / max(n_phases - 1, 1))
        phase_t0 = time.monotonic()

        while time.monotonic() - phase_t0 < phase_time and time.monotonic() - t0 < total_time - 5:
            _, lse, grad, p, c = smooth_grad_and_score(a, beta)

            # Mirror descent step
            a_new = mirror_descent_step(a, grad, lr, S_target)
            new_score = compute_score(a_new)

            if new_score < current_score:
                a = a_new
                current_score = new_score
                stall_count = 0
                if current_score < best_score:
                    best_score = current_score
                    best_a = a.copy()
                # Slightly increase lr if improving
                lr = min(lr * 1.02, 1.0)
            else:
                stall_count += 1
                lr *= 0.9
                if stall_count > 50:
                    # Add small perturbation
                    a = a * np.exp(rng.normal(0, 0.01, n))
                    a = np.maximum(a, 1e-10)
                    a = a * (S_target / np.sum(a))
                    current_score = compute_score(a)
                    stall_count = 0
                    lr = min(lr * 2, 0.5)
                elif stall_count > 10:
                    lr = max(lr * 0.8, 1e-4)

            if verbose and (int((time.monotonic() - phase_t0) * 2) % 10 == 0):
                elapsed = time.monotonic() - t0
                print(f"      phase={phase}, beta={beta:.1f}, "
                      f"score={current_score:.6f}, best={best_score:.6f}, "
                      f"lr={lr:.4f}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def optimize_hardsub(a_init, time_limit, rng, verbose=True):
    """
    Hard subgradient optimization: use the actual peak direction.
    """
    n = len(a_init)
    S_target = np.sum(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 1e-10)
    a = a * (S_target / np.sum(a))

    current_score = compute_score(a)
    best_score = current_score
    best_a = a.copy()

    t0 = time.monotonic()
    lr = 0.05
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        grad, c, peak_idx = hardsub_grad(a)

        # Also consider reducing the peak specifically
        # The peak is at peak_idx, contributions are a[j]*a[peak_idx-j]
        # Reduce the largest contributors
        a_new = mirror_descent_step(a, grad, lr, S_target)
        new_score = compute_score(a_new)

        if new_score < current_score:
            a = a_new
            current_score = new_score
            stall_count = 0
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
            lr = min(lr * 1.01, 0.5)
        else:
            stall_count += 1
            lr *= 0.95
            if stall_count > 30:
                # Perturb
                a = a * np.exp(rng.normal(0, 0.02, n))
                a = np.maximum(a, 1e-10)
                a = a * (S_target / np.sum(a))
                current_score = compute_score(a)
                stall_count = 0
                lr = 0.1

            # Every 100 stalls, try targeted peak reduction
            if stall_count > 0 and stall_count % 50 == 0:
                # Reduce top contributors to the peak
                contribs = np.zeros(n)
                for i in range(n):
                    j = peak_idx - i
                    if 0 <= j < n:
                        contribs[i] = a[i] * a[j]
                top_k = max(3, n // 10)
                top_idx = np.argsort(contribs)[-top_k:]
                a[top_idx] *= 0.95
                a = np.maximum(a, 1e-10)
                a = a * (S_target / np.sum(a))
                new_score = compute_score(a)
                if new_score < current_score:
                    current_score = new_score
                    if current_score < best_score:
                        best_score = current_score
                        best_a = a.copy()
                    stall_count = 0

        if verbose and int((time.monotonic() - t0) * 3) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      hardsub: score={current_score:.6f}, best={best_score:.6f}, "
                  f"lr={lr:.4f}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def coordinate_descent(a_init, time_limit, rng, verbose=True):
    """
    Coordinate descent: optimize one coordinate at a time.
    """
    n = len(a_init)
    S_target = np.sum(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 1e-10)
    a = a * (S_target / np.sum(a))

    current_score = compute_score(a)
    best_score = current_score
    best_a = a.copy()

    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 2:
                break

            old_val = a[i]
            if old_val < 1e-10:
                continue

            # Compute base convolution (without a[i]'s contribution)
            # c_k = c_k_base + a[i] * d_k
            # where d_k = a[k-i] (if valid) + a[k-i] (if valid and k-i != i)
            #          = 2*a[k-i] if k=2i
            d = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                j = k - i
                if 0 <= j < n:
                    if j == i:
                        d[k] = 2.0 * a[j]
                    else:
                        d[k] = a[j] + a[i]

            S_base = S_target - old_val  # sum of all other elements

            # Try several candidate values for a[i]
            candidates = [0.0, old_val]
            # Analytical optimum for current peak
            c_base = fft_convolve(a, a) - old_val * d  # convolution without a[i]
            peak_k = int(np.argmax(c_base + old_val * d))
            if d[peak_k] > 1e-15:
                # Optimize 2n*(c_base[peak_k] + ai*d[peak_k]) / (S_base+ai)^2
                # Derivative zero: c_base[peak_k]*d[peak_k] + ai*d[peak_k]^2
                #   - 2*(c_base[peak_k] + ai*d[peak_k]) = 0
                # c_base[peak_k]*d[peak_k] + ai*d[peak_k]^2
                #   - 2*c_base[peak_k] - 2*ai*d[peak_k] = 0
                # ai*(d[peak_k]^2 - 2*d[peak_k]) + c_base[peak_k]*(d[peak_k] - 2) = 0
                # ai = c_base[peak_k]*(2 - d[peak_k]) / (d[peak_k]^2 - 2*d[peak_k])
                denom = d[peak_k] ** 2 - 2.0 * d[peak_k]
                if abs(denom) > 1e-15:
                    ai_opt = c_base[peak_k] * (2.0 - d[peak_k]) / denom
                    if ai_opt > 0:
                        candidates.append(ai_opt)
            # Add some geometric probes
            for factor in [0.3, 0.5, 0.7, 1.3, 1.5, 2.0]:
                candidates.append(old_val * factor)

            best_val = old_val
            best_local_score = current_score

            for cv in candidates:
                cv = max(0.0, cv)
                if abs(cv - old_val) < 1e-12:
                    continue
                a[i] = cv
                S_new = S_base + cv
                if S_new < 0.01:
                    continue
                c_new = c_base + cv * d
                local_s = 2.0 * n * np.max(c_new) / (S_new ** 2)
                if local_s < best_local_score:
                    best_local_score = local_s
                    best_val = cv

            a[i] = best_val
            if best_val != old_val:
                improved = True
                current_score = best_local_score
                if current_score < best_score:
                    best_score = current_score
                    best_a = a.copy()

        sweep += 1
        if verbose and sweep % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      coord: sweep={sweep}, score={current_score:.6f}, "
                  f"best={best_score:.6f}, elapsed={elapsed:.1f}s")
        if not improved and sweep > 5:
            # Try perturbation
            a = a * np.exp(rng.normal(0, 0.01, n))
            a = np.maximum(a, 1e-10)
            a = a * (S_target / np.sum(a))
            current_score = compute_score(a)
            sweep = 0

    return best_a, best_score


def targeted_peak_reduction(a_init, time_limit, rng, verbose=True):
    """
    Directly target the peak: find which pairs contribute most to the peak
    and reduce them while maintaining the sum.
    """
    n = len(a_init)
    S_target = np.sum(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 1e-10)
    a = a * (S_target / np.sum(a))

    current_score = compute_score(a)
    best_score = current_score
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        peak_val = np.max(c)
        peak_mask = c >= peak_val * 0.999
        peak_indices = np.where(peak_mask)[0]

        for pk in peak_indices:
            if time.monotonic() - t0 > time_limit - 5:
                break
            # Find top contributors to this peak
            contribs = np.zeros(n)
            for i in range(n):
                j = pk - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]

            top_k = max(5, n // 8)
            top_idx = np.argsort(contribs)[-top_k:]

            # Reduce top contributors, increase others
            n_reduce = max(2, top_k // 2)
            reduce_idx = top_idx[:n_reduce]
            increase_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(increase_idx) > 0:
                n_inc = min(n_reduce, len(increase_idx))
                inc_idx = rng.choice(increase_idx, size=n_inc, replace=False)
            else:
                inc_idx = top_idx[top_k // 2:]

            a[reduce_idx] *= 0.97
            a[inc_idx] *= 1.03
            a = np.maximum(a, 1e-10)
            a = a * (S_target / np.sum(a))

            new_score = compute_score(a)
            if new_score < current_score:
                current_score = new_score
                if current_score < best_score:
                    best_score = current_score
                    best_a = a.copy()
            else:
                # Revert
                a[reduce_idx] /= 0.97
                a[inc_idx] /= 1.03
                a = np.maximum(a, 1e-10)
                a = a * (S_target / np.sum(a))
                current_score = compute_score(a)

        if verbose and int((time.monotonic() - t0) * 2) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      targeted: score={current_score:.6f}, best={best_score:.6f}, "
                  f"elapsed={elapsed:.1f}s")

    return best_a, best_score


def resample_sequence(seq, new_n):
    """Resample a sequence to a new length using interpolation."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)


def make_init(n, init_type, rng):
    """Create various initialization patterns."""
    if init_type == 'uniform':
        return np.ones(n)
    elif init_type == 'geometric':
        r = rng.uniform(0.7, 1.3)
        a = r ** np.arange(n)
        return a / np.max(a)
    elif init_type == 'bell':
        t = np.arange(n) / n
        a = np.exp(-5.0 * (t - 0.5) ** 2)
        return a
    elif init_type == 'two_lobe':
        t = np.arange(n) / n
        a = np.exp(-20.0 * (t - 0.2) ** 2) + np.exp(-20.0 * (t - 0.8) ** 2)
        return a
    elif init_type == 'polynomial':
        t = np.arange(n) / n
        power = rng.uniform(1.0, 4.0)
        direction = rng.choice([1, -1])
        a = (1.0 + direction * t) ** power
        return np.maximum(a, 0.01)
    elif init_type == 'random_smooth':
        # Random but smooth (low-pass filtered)
        raw = rng.standard_normal(n)
        # Simple moving average
        kernel = np.ones(8) / 8
        smooth = np.convolve(raw, kernel, mode='same')
        return np.maximum(1.0 + 0.3 * smooth, 0.01)
    else:
        return np.ones(n)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: smooth gradient descent, hard subgradient,
    coordinate descent, and targeted peak reduction.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 40.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Try to get provided sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(start_seq) > 0 and np.sum(start_seq) > 0.01:
            s = compute_score(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()

    # Define optimization pipeline for a given sequence
    def run_pipeline(a0, time_alloc, label=""):
        nonlocal best_score, best_seq
        n = len(a0)
        S_target = np.sum(a0)
        a0 = np.maximum(a0, 1e-10)
        a0 = a0 * (S_target / np.sum(a0))

        remaining = deadline - time.monotonic()
        if remaining < 30:
            return

        # Split time among methods
        t_smooth = min(time_alloc * 0.30, remaining * 0.35)
        t_hard = min(time_alloc * 0.20, remaining * 0.25)
        t_coord = min(time_alloc * 0.25, remaining * 0.25)
        t_targeted = min(time_alloc * 0.20, remaining * 0.20)
        t_smooth = max(10.0, t_smooth * (remaining / max(time_alloc, 1)))
        t_hard = max(5.0, t_hard * (remaining / max(time_alloc, 1)))
        t_coord = max(5.0, t_coord * (remaining / max(time_alloc, 1)))
        t_targeted = max(5.0, t_targeted * (remaining / max(time_alloc, 1)))

        # Phase 1: Smooth gradient
        if t_smooth > 5 and time.monotonic() < deadline - 20:
            print(f"  [{label}] Smooth optimization ({t_smooth:.0f}s)...")
            a_best, s_best = optimize_smooth(a0, t_smooth, rng,
                                             beta_start=2.0, beta_end=200.0,
                                             verbose=False)
            if s_best < best_score:
                best_score = s_best
                best_seq = a_best.copy()
                print(f"    New best: {best_score:.6f}")
            a0 = a_best

        # Phase 2: Hard subgradient
        if t_hard > 5 and time.monotonic() < deadline - 15:
            print(f"  [{label}] Hard subgradient ({t_hard:.0f}s)...")
            a_best, s_best = optimize_hardsub(a0, t_hard, rng, verbose=False)
            if s_best < best_score:
                best_score = s_best
                best_seq = a_best.copy()
                print(f"    New best: {best_score:.6f}")
            a0 = a_best

        # Phase 3: Coordinate descent
        if t_coord > 5 and time.monotonic() < deadline - 10:
            print(f"  [{label}] Coordinate descent ({t_coord:.0f}s)...")
            a_best, s_best = coordinate_descent(a0, t_coord, rng, verbose=False)
            if s_best < best_score:
                best_score = s_best
                best_seq = a_best.copy()
                print(f"    New best: {best_score:.6f}")
            a0 = a_best

        # Phase 4: Targeted peak reduction
        if t_targeted > 5 and time.monotonic() < deadline - 10:
            print(f"  [{label}] Targeted peak reduction ({t_targeted:.0f}s)...")
            a_best, s_best = targeted_peak_reduction(a0, t_targeted, rng, verbose=False)
            if s_best < best_score:
                best_score = s_best
                best_seq = a_best.copy()
                print(f"    New best: {best_score:.6f}")
            a0 = a_best

    # === Main search strategy ===

    # Strategy 1: Optimize from provided sequence
    if start_seq is not None and time.monotonic() < deadline - 200:
        print("\n=== Strategy 1: Optimize provided sequence ===")
        n_prov = len(start_seq)
        t_alloc = min(400.0, deadline - time.monotonic() - 300)
        run_pipeline(start_seq.copy(), t_alloc, f"provided n={n_prov}")

    # Strategy 2: Try different sequence lengths and initializations
    print("\n=== Strategy 2: Explore different configurations ===")
    configs = []
    for n in [256, 384, 512, 768, 1024]:
        for init_type in ['two_lobe', 'bell', 'geometric', 'polynomial',
                          'random_smooth', 'uniform']:
            configs.append((n, init_type))

    # Shuffle and give each a time budget
    rng.shuffle(configs)

    # Give more time to configs near the best n
    n_best = len(best_seq) if best_seq is not None else 512
    configs.sort(key=lambda x: abs(x[0] - n_best))

    n_configs = len(configs)
    time_left = deadline - time.monotonic() - 100
    time_per_config = max(20.0, time_left / max(n_configs, 1))

    for ci, (n, init_type) in enumerate(configs):
        if time.monotonic() > deadline - 80:
            break

        remaining_configs = n_configs - ci
        t_alloc = max(15.0, min(80.0, (deadline - time.monotonic() - 60) /
                                 max(remaining_configs, 1)))

        # Sometimes resample best
        if best_seq is not None and rng.random() < 0.3:
            a0 = resample_sequence(best_seq, n)
            label = f"resampled n={n}"
        else:
            a0 = make_init(n, init_type, rng)
            label = f"n={n} {init_type}"

        print(f"\n  Config {ci+1}/{n_configs}: {label}")
        run_pipeline(a0, t_alloc, label)

    # Strategy 3: Intensive polish on best
    if best_seq is not None and time.monotonic() < deadline - 50:
        print("\n=== Strategy 3: Intensive polish on best ===")
        t_polish = min(300.0, deadline - time.monotonic() - 30)
        run_pipeline(best_seq.copy(), t_polish, "polish")

    # Strategy 4: Multiple short smooth passes with different betas
    if best_seq is not None and time.monotonic() < deadline - 30:
        print("\n=== Strategy 4: Multiple beta passes ===")
        remaining = deadline - time.monotonic() - 20
        t_per = remaining / 4
        for beta in [5.0, 20.0, 50.0, 100.0]:
            if time.monotonic() > deadline - 20:
                break
            print(f"  Beta={beta} pass ({t_per:.0f}s)...")
            a_best, s_best = optimize_smooth(best_seq, t_per, rng,
                                             beta_start=beta, beta_end=beta * 3,
                                             verbose=False)
            if s_best < best_score:
                best_score = s_best
                best_seq = a_best.copy()
                print(f"    New best: {best_score:.6f}")

    # Strategy 5: Final coordinate descent
    if best_seq is not None and time.monotonic() < deadline - 15:
        print("\n=== Strategy 5: Final coordinate descent ===")
        t_coord = min(200.0, deadline - time.monotonic() - 10)
        a_best, s_best = coordinate_descent(best_seq, t_coord, rng, verbose=True)
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"  Final: {best_score:.6f}")

    # Strategy 6: One more targeted reduction
    if best_seq is not None and time.monotonic() < deadline - 10:
        t_targeted = min(150.0, deadline - time.monotonic() - 5)
        if t_targeted > 10:
            print(f"\n=== Strategy 6: Final targeted reduction ({t_targeted:.0f}s) ===")
            a_best, s_best = targeted_peak_reduction(best_seq, t_targeted, rng,
                                                     verbose=False)
            if s_best < best_score:
                best_score = s_best
                best_seq = a_best.copy()
                print(f"  Final: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 512
```
