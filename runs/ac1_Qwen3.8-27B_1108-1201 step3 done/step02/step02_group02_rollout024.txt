Let me also think about one more approach: **Alternating Direction Method of Multipliers (ADMM)** or **Proximal methods**.

The problem: minimize max_k c(k) subject to a >= 0, sum(a) = S.

We can introduce a variable M >= c(k) for all k, and minimize M:
minimize M
subject to c(k) <= M for all k = 0, ..., 2n-2
         a >= 0
         sum(a) = S

This is a problem with quadratic constraints (c(k) = a^T B_k a <= M). It's non-convex (as we discussed, the B_k matrices are not PSD in general).

But we can solve it iteratively:
1. Fix a, update M = max_k c(k).
2. Fix M, update a to minimize some regularizer subject to c(k) <= M for "active" k and a >= 0, sum(a) = S.

Step 2 is a quadratically constrained optimization (QCO). We can solve it approximately with scipy.optimize.minimize.

This is essentially what the LP approach does (but with linear constraints instead of quadratic). The LP is easier to solve but less accurate.

For a practical approach, I'll stick with subgradient + coordinate descent + LP.

"""
Optimized sequence search using subgradient descent, coordinate descent,
and LP-based refinement to minimize peak autocorrelation of non-negative sequences.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    if n == 0 or m == 0:
        return np.array([])
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    """Compute score = 2*n*max(convolve(a,a)) / sum(a)^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def compute_subgrad(a, c):
    """Compute subgradient of max(c) w.r.t. a."""
    n = len(a)
    max_c = np.max(c)
    # Find peak positions (within tolerance)
    tol = max_c * 1e-10 + 1e-15
    peak_mask = c >= max_c - tol
    peak_indices = np.where(peak_mask)[0]
    
    g = np.zeros(n)
    for k in peak_indices:
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_max >= i_min:
            j_vals = k - np.arange(i_min, i_max + 1)
            g[i_min:i_max+1] += 2.0 * a[j_vals]
    g /= len(peak_indices)
    return g, c, max_c, peak_indices

def subgrad_descent(a0, time_limit, rng, verbose=False):
    """Subgradient descent with backtracking line search."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0.0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n)
        s = n
    a = a / s  # Normalize to sum = 1
    
    c = fft_convolve(a, a)
    current_peak = np.max(c)
    best_a = a.copy()
    best_peak = current_peak
    current_score = 2.0 * n * current_peak  # Since sum = 1, score = 2n*peak
    
    t0 = time.time()
    lr = 0.01
    stall = 0
    it = 0
    
    while time.time() - t0 < time_limit:
        g, c, max_c, peaks = compute_subgrad(a, c)
        it += 1
        
        # Backtracking line search
        improved = False
        for trial_lr in [lr, lr*0.5, lr*0.25, lr*0.1]:
            a_new = a - trial_lr * g
            a_new = np.clip(a_new, 0.0, None)
            s_new = np.sum(a_new)
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            c_new = fft_convolve(a_new, a_new)
            peak_new = np.max(c_new)
            if peak_new < current_peak:

                c = c_new
                current_peak = peak_new

def build_A(f):
    """Build constraint matrix for LP."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def solve_lp(f, M):
    """Solve LP to find descent direction."""
    n = len(f)
    A = build_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None
