```python
def compute_soft_max_grad(a, T):
    n = len(a)
    conv = fft_convolve(a)
    max_c = np.max(conv)
    exp_c = np.exp((conv - max_c) / T)
    Z = np.sum(exp_c)
    weights = exp_c / Z
    
    # We need FFT of weights (length 2n-1) and a (length n)
    # Pad both to same size
    size = 1
    while size < 2 * (2*n - 1):
        size <<= 1
    
    W = np.fft.rfft(weights, size)
    A = np.fft.rfft(a, size)
    
    # Cross-correlation: (weights cross-corr a)[j] = IDFT(FFT(weights) * conj(FFT(a)))[j]
    term1 = np.fft.irfft(W * np.conj(A), size)[:n]
    
    grad = 2.0 * term1
    soft_max_val = max_c + T * np.log(Z)
    return grad, soft_max_val, conv
```

```python
import numpy as np
import time

def fft_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a)
    return 2.0 * n * np.max(conv) / (sum_a ** 2)

    rho_candidates = np.where(vals >= S)[0]
    if len(rho_candidates) == 0:

        return np.full(n, S / n)
    rho = rho_candidates[-1]  # 0-indexed
    theta = (cumsum[rho] - S) / (rho + 1)
    return np.maximum(a - theta, 0)
```

```python
"""
Multi-phase optimization for minimizing normalized peak autocorrelation of non-negative step functions.
Combines soft-max smoothed gradient descent with Nesterov momentum, LP-based directional refinement,
and targeted local search. Uses FFT-based acceleration for all gradient and convolution computations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b=None):
    """Compute linear convolution of two 1-D arrays using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def project_simplex(a, S):
    """Project vector a onto the simplex {x >= 0, sum(x) = S}."""
    n = len(a)
    if n == 1:
        return np.array([S])
    a_sorted = np.sort(a)[::-1]
    cumsum = np.cumsum(a_sorted)
    idx = np.arange(1, n + 1)
    vals = cumsum - idx * a_sorted
    # Find largest rho (0-indexed) where vals[rho] > 0 and vals[rho] >= S is NOT required
    # We need: cumsum[rho] - (rho+1) * a_sorted[rho] >= S
    mask = vals >= S
    if not np.any(mask):
        return np.full(n, S / n)
    rho = np.max(np.where(mask)[0])
    theta = (cumsum[rho] - S) / (rho + 1)
    return np.maximum(a - theta, 0.0)


def compute_soft_max_grad(a, T):
    """Compute soft-max value and gradient using FFT-based cross-correlation."""
    n = len(a)
    conv = fft_convolve(a)
    max_c = np.max(conv)
    exp_c = np.exp((conv - max_c) / T)
    Z = np.sum(exp_c)
    weights = exp_c / Z
    soft_max_val = max_c + T * np.log(Z)
    
    # Gradient: grad[j] = 2 * sum_{l=0}^{n-1} weights[j+l] * a[l]
    # This is 2 * (weights cross-corr a)[j]
    # = 2 * IDFT(FFT(weights) * conj(FFT(a)))[j]
    size = 1
    while size < 2 * (2 * n - 1):
        size <<= 1
    W = np.fft.rfft(weights, size)
    A = np.fft.rfft(a, size)
    term = np.fft.irfft(W * np.conj(A), size)[:n]
    grad = 2.0 * term
    return grad, soft_max_val, conv


def subgradient_step(a, S):
    """One subgradient step toward reducing the peak autocorrelation."""
    n = len(a)
    conv = fft_convolve(a)
    peak = int(np.argmax(conv))
    # Subgradient of max(conv) at peak lag:
    # grad[j] = 2*a[peak-j] if 0<=peak-j<n else 0
    grad = np.zeros(n)
    for j in range(n):
        k = peak - j
        if 0 <= k < n:
            grad[j] = 2.0 * a[k]
    return grad, peak


def soft_max_descent(a0, time_limit, S, T_init=None, T_min=None, verbose=False):
    """Soft-max smoothed gradient descent with Nesterov momentum."""
    n = len(a0)
    if T_init is None:
        conv0 = fft_convolve(a0)
        T_init = max(0.01, 0.1 * np.max(conv0))
    if T_min is None:
        T_min = max(0.001, T_init / 1000.0)
    
    a = a0.copy()
    v = np.zeros(n)
    T = T_init
    lr = 1.0 / (n * np.max(np.abs(a0)) + 1e-10)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    patience = 0
    history = []
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        grad, sm_val, conv = compute_soft_max_grad(a, T)
        
        # Adaptive learning rate based on gradient norm
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            # Normalize gradient for consistent step size
            grad = grad / (grad_norm + 1e-10)
        
        # Nesterov momentum
        mu = min(0.9, 0.9 + 0.05 * (step / 1000))
        v = mu * v + (1 - mu) * lr * grad
        a_new = a - v
        a_new = project_simplex(a_new, S)
        
        a = a_new
        
        # Periodic score evaluation
        if step % 50 == 0:
            score = compute_score(a)
            if score < best_score:

                best_a = a.copy()
                patience = 0
            else:
                patience += 50
            
            # Adaptive T
            if patience > 1000:
                T = max(T_min, T * 0.7)
                lr *= 0.6
                patience = 0
            elif patience > 500:
                T = max(T_min, T * 0.85)
                lr *= 0.8
            elif step % 500 == 0 and step > 0:
                T = max(T_min, T * 0.95)
        
        # Periodic subgradient correction (helps when soft-max is too smooth)
        if step % 200 == 100:
            sg, peak = subgradient_step(a, S)
            a = a - 0.1 * lr * sg
            a = project_simplex(a, S)
        
        step += 1
        
        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SoftMax step {step}: best_score={best_score:.6f}, T={T:.4f}, lr={lr:.2e}, t={elapsed:.0f}s")
    
    return best_a


def build_lp_matrix(a):
    """Build convolution constraint matrix for LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def lp_refine(a0, time_limit, S, max_iters=200, verbose=False):
    """LP-based iterative refinement."""
    n = len(a0)
    a = a0.copy()
    a = project_simplex(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    
    A = build_lp_matrix(a)
    c = -np.ones(n)
    
    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        a = project_simplex(a, S)
        A = build_lp_matrix(a)
        conv_a = fft_convolve(a)
        M = np.max(conv_a)
        
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs')
        if res.status != 0:
            break
        
        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break
        
        # Normalize g to have sum S
        g_norm = g * S / g_sum
        
        # Try different blend parameters
        best_t = 0.0
        best_score_t = np.max(fft_convolve(a, a))
        for t in [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.05]:
            b = (1 - t) * a + t * g_norm
            b = np.maximum(b, 0.0)
            b = project_simplex(b, S)
            b_conv_max = np.max(fft_convolve(b, b))
            if b_conv_max <= best_score_t + 1e-10:
                best_t = t
                best_score_t = b_conv_max
                break
        
        if best_t == 0.0:
            # Try more aggressive: just go to g_norm
            b = project_simplex(g_norm, S)
            b_conv_max = np.max(fft_convolve(b, b))
            if b_conv_max < best_score_t:
                a = b
            break
        
        a = (1 - best_t) * a + best_t * g_norm
        a = project_simplex(a, S)
        
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iteration}: score={best_score:.6f}, t={best_t:.2f}, elapsed={elapsed:.0f}s")
        
        # Convergence check
        if best_t < 0.02 and iteration > 10:
            break
    
    return best_a


def targeted_local_search(a0, time_limit, S, rng, verbose=False):
    """Targeted local search focused on reducing the peak lag."""
    n = len(a0)
    a = a0.copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a)
        peak = int(np.argmax(conv))
        current_peak_val = conv[peak]
        
        # Compute contributions to the peak
        contribs = np.zeros(n)
        for j in range(n):
            k = peak - j
            if 0 <= k < n:
                contribs[j] = a[j] * a[k]
        
        # Strategy 1: Reduce top contributors, increase others
        total_contrib = np.sum(contribs)
        if total_contrib > 1e-10:
            # Top 10% contributors
            n_top = max(5, n // 10)
            top_idx = np.argsort(contribs)[-n_top:]
            
            candidate = a.copy()
            # Reduce top contributors by a small amount
            reduce_frac = rng.uniform(0.05, 0.2)
            for idx in top_idx:
                candidate[idx] *= (1.0 - reduce_frac * rng.uniform(0.5, 1.5))
            
            # Redistribute to non-top elements
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = max(3, n // 15)
                inc_idx = rng.choice(other_idx, size=min(n_inc, len(other_idx)), replace=False)
                inc_frac = rng.uniform(0.1, 0.3)
                candidate[inc_idx] *= (1.0 + inc_frac)
            
            candidate = np.maximum(candidate, 0.0)
            candidate = project_simplex(candidate, S)
            
            new_score = compute_score(candidate)
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
                a = candidate
        
        # Strategy 2: Random perturbation with acceptance
        if rng.random() < 0.3:

def make_initializer(n, rng, init_type):
    """Create various types of initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'constant':
        a = np.ones(n)
    elif init_type == 'chirp':
        alpha = 0.3 + 0.3 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / (n * 0.5))
    elif init_type == 'sine':
        f = 1 + int(rng.integers(0, 4))
        amp = 0.2 + 0.3 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'multi_tone':
        a = np.ones(n)
        for _ in range(3):
            f = int(rng.integers(1, max(2, n // 4)))
            amp = rng.uniform(0.05, 0.2)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'random':
        a = 1.0 + 0.3 * rng.standard_normal(n)
    elif init_type == 'bimodal':
        center = n // 2
        sigma = n * (0.15 + 0.1 * rng.random())
        a = 1.0 + 0.5 * np.exp(-0.5 * ((t - center) / sigma) ** 2)
    elif init_type == 'triangular':
        a = 1.0 + 0.5 * np.abs(2.0 * t / n - 1.0)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)
```

```python
"""
Multi-phase optimization for minimizing normalized peak autocorrelation of non-negative step functions.
Combines soft-max smoothed gradient descent with Nesterov momentum, LP-based directional refinement,
targeted local search, and momentum-based subgradient methods.
Uses FFT-based acceleration for all gradient and convolution computations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b=None):
    """Compute linear convolution of two 1-D arrays using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def project_simplex(a, S):
    """Project vector a onto the simplex {x >= 0, sum(x) = S}."""
    n = len(a)
    if n == 1:
        return np.array([S])
    a_sorted = np.sort(a)[::-1]
    cumsum = np.cumsum(a_sorted)
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = cumsum - idx * a_sorted
    mask = vals >= S
    if not np.any(mask):
        return np.full(n, S / n)
    rho = np.max(np.where(mask)[0])
    theta = (cumsum[rho] - S) / (rho + 1)
    return np.maximum(a - theta, 0.0)


def compute_soft_max_grad(a, T):
    """Compute soft-max value and gradient using FFT-based cross-correlation.
    
    The gradient of soft_max(conv(a,a); T) with respect to a[j] is:
    grad[j] = sum_k weights[k] * 2*a[k-j] * I[0<=k-j<n]
    which equals 2 * (weights cross-corr a)[j]
    """
    n = len(a)
    conv = fft_convolve(a)
    max_c = np.max(conv)
    exp_c = np.exp((conv - max_c) / T)
    Z = np.sum(exp_c)
    weights = exp_c / Z
    soft_max_val = max_c + T * np.log(Z)

    size = 1
    while size < 2 * (2 * n - 1):
        size <<= 1
    W = np.fft.rfft(weights, size)
    A = np.fft.rfft(a, size)
    # Cross-correlation: (weights cross-corr a)[j] = IDFT(FFT(weights) * conj(FFT(a)))[j]
    term = np.fft.irfft(W * np.conj(A), size)[:n]
    grad = 2.0 * term
    return grad, soft_max_val, conv


def subgradient_at_peak(a):
    """Compute the subgradient of max(conv(a,a)) at the current peak lag.
    
    The subgradient at peak lag k* is:
    grad[j] = 2*a[k*-j] if 0<=k*-j<n else 0
    """
    n = len(a)
    conv = fft_convolve(a)
    peak = int(np.argmax(conv))
    grad = np.zeros(n)
    for j in range(n):
        k = peak - j
        if 0 <= k < n:
            grad[j] = 2.0 * a[k]
    return grad, peak, conv


def soft_max_descent(a0, time_limit, S, T_init=None, T_min=None, lr_init=None, verbose=False):
    """Soft-max smoothed gradient descent with Nesterov momentum and adaptive scheduling."""
    n = len(a0)
    if T_init is None:
        conv0 = fft_convolve(a0)
        T_init = max(0.01, 0.1 * np.max(conv0))
    if T_min is None:
        T_min = max(0.001, T_init / 2000.0)
    if lr_init is None:
        lr_init = 1.0 / (n * 2.0)

    a = a0.copy()
    v = np.zeros(n)
    T = T_init
    lr = lr_init
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    patience = 0
    step = 0

    while time.monotonic() - t0 < time_limit:
        grad, sm_val, conv = compute_soft_max_grad(a, T)

        # Normalize gradient for consistent step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-14:
            grad = grad / grad_norm

        # Nesterov momentum
        mu = min(0.92, 0.85 + 0.002 * step)
        v = mu * v + lr * grad
        a_new = a - v
        a_new = project_simplex(a_new, S)
        a = a_new

        # Periodic score evaluation
        if step % 30 == 0:
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                patience = 0
            else:
                patience += 30

            # Adaptive T and lr
            if patience > 2000:
                T = max(T_min, T * 0.6)
                lr *= 0.5
                patience = 0
            elif patience > 1000:
                T = max(T_min, T * 0.8)
                lr *= 0.7
            elif patience > 500:
                T = max(T_min, T * 0.9)
            elif step % 300 == 0 and step > 0:
                T = max(T_min, T * 0.97)

        # Occasional subgradient correction for sharper peak targeting
        if step % 500 == 250:
            sg, peak, _ = subgradient_at_peak(a)
            sg_norm = np.linalg.norm(sg)
            if sg_norm > 1e-14:
                a = a - 0.5 * lr * sg / sg_norm
                a = project_simplex(a, S)

        step += 1

        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SoftMax step {step}: best_score={best_score:.6f}, T={T:.4f}, lr={lr:.2e}, t={elapsed:.0f}s")

    return best_a


def build_lp_matrix_fast(a):
    """Build convolution constraint matrix using FFT-based approach."""
    n = len(a)
    # A[k, j] = a[k-j] if 0 <= k-j < n else 0
    # For each k, the row is a shifted version of a
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def lp_refine(a0, time_limit, S, max_iters=500, verbose=False):
    """LP-based iterative refinement with aggressive blending."""
    n = len(a0)
    a = project_simplex(a0, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    c = -np.ones(n)

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        a = project_simplex(a, S)
        conv_a = fft_convolve(a)
        M = np.max(conv_a)

        # Build LP matrix
        A = build_lp_matrix_fast(a)

        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs')
        if res.status != 0:
            break

        g = res.x
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break

        g_norm = g * S / g_sum
        current_conv_max = np.max(fft_convolve(a, a))

        # Try blend parameters from aggressive to conservative
        best_t = 0.0
        best_conv_max = current_conv_max
        for t in [0.7, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.05]:
            b = (1.0 - t) * a + t * g_norm
            b = project_simplex(b, S)
            b_conv_max = np.max(fft_convolve(b, b))
            if b_conv_max <= current_conv_max + 1e-9:
                best_t = t
                best_conv_max = b_conv_max
                break

        if best_t == 0.0:
            # Try direct move to g_norm
            b = project_simplex(g_norm, S)
            b_conv_max = np.max(fft_convolve(b, b))
            if b_conv_max < current_conv_max:
                a = b
                score = compute_score(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
            break

        a = (1.0 - best_t) * a + best_t * g_norm
        a = project_simplex(a, S)

        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and iteration % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iteration}: score={best_score:.6f}, t={best_t:.2f}, elapsed={elapsed:.0f}s")

        if best_t < 0.03 and iteration > 20:
            break

    return best_a


def momentum_subgradient(a0, time_limit, S, verbose=False):
    """Heavy-ball momentum subgradient method for the non-smooth peak objective."""
    n = len(a0)
    a = a0.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    lr = 1.0 / n
    mu = 0.8
    patience = 0
    step = 0

    while time.monotonic() - t0 < time_limit:
        grad, peak, conv = subgradient_at_peak(a)
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-14:
            grad = grad / grad_norm

        v = mu * v + lr * grad
        a_new = a - v
        a_new = project_simplex(a_new, S)
        a = a_new

        if step % 50 == 0:
            score = compute_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                patience = 0
            else:
                patience += 50
                if patience > 1000:
                    lr *= 0.7
                    mu = min(mu + 0.02, 0.95)
                    patience = 0

        step += 1

        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step {step}: best_score={best_score:.6f}, lr={lr:.2e}, mu={mu:.2f}, t={elapsed:.0f}s")

    return best_a


def targeted_local_search(a0, time_limit, S, rng, verbose=False):
    """Targeted local search focused on reducing the peak lag contributions."""
    n = len(a0)
    a = a0.copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a)
        peak = int(np.argmax(conv))

        # Compute contributions to the peak lag
        contribs = np.zeros(n)
        for j in range(n):
            k = peak - j
            if 0 <= k < n:
                contribs[j] = a[j] * a[k]

        total_contrib = np.sum(contribs)

        # Strategy 1: Reduce top contributors, boost others
        if total_contrib > 1e-10:
            n_top = max(5, n // 12)
            top_idx = np.argsort(contribs)[-n_top:]

            candidate = a.copy()
            for idx in top_idx:
                candidate[idx] *= (1.0 - rng.uniform(0.05, 0.25))

            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = max(3, n // 15)
                inc_idx = rng.choice(other_idx, size=min(n_inc, len(other_idx)), replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.08, 0.05, size=n_inc))

            candidate = np.maximum(candidate, 0.0)
            candidate = project_simplex(candidate, S)
            new_score = compute_score(candidate)
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
                a = candidate

        # Strategy 2: Local window perturbation around peak
        if rng.random() < 0.4:
            candidate = a.copy()
            window = max(3, n // 25)
            center = peak // 2
            start = max(0, center - window)
            end = min(n, center + window)
            candidate[start:end] *= np.exp(rng.normal(0, 0.06, size=end - start))
            candidate = np.maximum(candidate, 0.0)
            candidate = project_simplex(candidate, S)
            new_score = compute_score(candidate)
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
                a = candidate
            elif new_score < best_score * 1.01 and rng.random() < 0.15:
                a = candidate

        # Strategy 3: Pair swap
        if rng.random() < 0.3:
            candidate = a.copy()
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                amount = min(candidate[i], candidate[j]) * rng.uniform(0.1, 0.4)
                candidate[i] -= amount
                candidate[j] += amount
                candidate = np.maximum(candidate, 0.0)
                candidate = project_simplex(candidate, S)
                new_score = compute_score(candidate)
                if new_score < best_score:
                    best_score = new_score
                    best_a = candidate.copy()
                    a = candidate

        # Strategy 4: Gaussian perturbation
        if rng.random() < 0.2:
            candidate = a * np.exp(rng.normal(0, 0.03, size=n))
            candidate = np.maximum(candidate, 0.0)
            candidate = project_simplex(candidate, S)
            new_score = compute_score(candidate)
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
                a = candidate

        iteration += 1
        if verbose and iteration % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LocalSearch iter {iteration}: best_score={best_score:.6f}, t={elapsed:.0f}s")

    return best_a


def make_initializer(n, rng, init_type):
    """Create various types of initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'constant':
        a = np.ones(n)
    elif init_type == 'chirp':
        alpha = 0.3 + 0.3 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / (n * 0.5))
    elif init_type == 'sine':
        f = 1 + int(rng.integers(0, 4))
        amp = 0.2 + 0.3 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'multi_tone':
        a = np.ones(n)
        for _ in range(3):
            f = int(rng.integers(1, max(2, n // 4)))
            amp = rng.uniform(0.05, 0.2)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'random':
        a = 1.0 + 0.3 * rng.standard_normal(n)
    elif init_type == 'bimodal':
        center = n // 2
        sigma = n * (0.15 + 0.1 * rng.random())
        a = 1.0 + 0.5 * np.exp(-0.5 * ((t - center) / sigma) ** 2)
    elif init_type == 'U_shape':
        # U-shaped: high at edges, low in middle
        a = 1.0 + 0.8 * (2.0 * t / n - 1.0) ** 2
    elif init_type == 'inverted_U':
        # Inverted U: low at edges, high in middle
        a = 1.5 - 0.8 * (2.0 * t / (n - 1) - 1.0) ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization for minimizing normalized peak autocorrelation.
    
    Strategy:
    1. Soft-max smoothed gradient descent (main engine, fast iterative)
    2. Momentum subgradient method (handles non-smoothness directly)
    3. LP-based directional refinement (principled, slower)
    4. Targeted local search (final polish)
    
    Explores multiple sequence lengths and diverse initializations to avoid local minima.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Get provided sequence if available
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n_prov = len(provided_seq)
        prov_score = compute_score(provided_seq)
        print(f"Provided sequence: n={n_prov}, score={prov_score:.6f}")
        best_score = min(best_score, prov_score)
        best_seq = provided_seq.copy()

    # Phase 1: Soft-max gradient descent from provided sequence
    print("\n=== Phase 1: Soft-max gradient descent (provided seq) ===")
    if best_seq is not None:
        n = len(best_seq)
        S = np.sqrt(2.0 * n)
        a = project_simplex(best_seq, S)
        t_limit = min(240.0, (deadline - time.monotonic()) * 0.25)
        print(f"  n={n}, time_limit={t_limit:.0f}s")
        result = soft_max_descent(a, t_limit, S, verbose=True)
        r_score = compute_score(result)
        if r_score < best_score:
            best_score = r_score
            best_seq = result.copy()
            print(f"  *** Improved to {best_score:.6f} ***")
        print(f"  Phase 1 best: {best_score:.6f}")

    # Phase 2: Multi-start exploration with diverse inits
    print("\n=== Phase 2: Multi-start exploration ===")
    lengths_to_try = [448, 512, 576, 640, 704, 768]
    if provided_seq is not None:
        lengths_to_try.insert(0, len(provided_seq))

    init_types = ['chirp', 'U_shape', 'sine', 'multi_tone', 'bimodal', 'random']
    candidates = []

    for n in lengths_to_try:
        if time.monotonic() > deadline - 400:
            break
        for init_type in init_types[:3]:
            if time.monotonic() > deadline - 350:
                break
            a0 = make_initializer(n, rng, init_type)
            S = np.sqrt(2.0 * n)
            a0 = project_simplex(a0, S)
            t_per = min(50.0, (deadline - time.monotonic()) * 0.08)
            if t_per < 5:
                break
            result = soft_max_descent(a0, t_per, S, verbose=False)
            r_score = compute_score(result)
            if r_score < best_score:
                best_score = r_score
                best_seq = result.copy()
                print(f"  *** n={n}, init={init_type}: new best {best_score:.6f} ***")
            if r_score < 1.55:
                candidates.append((r_score, result.copy(), n, init_type))

    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 2 best: {best_score:.6f}, {len(candidates)} good candidates")

    # Phase 3: Momentum subgradient on best
    print("\n=== Phase 3: Momentum subgradient ===")
    if best_seq is not None:
        n = len(best_seq)
        S = np.sqrt(2.0 * n)
        a = project_simplex(best_seq, S)
        t_limit = min(120.0, (deadline - time.monotonic()) * 0.2)
        print(f"  n={n}, time_limit={t_limit:.0f}s")
        result = momentum_subgradient(a, t_limit, S, verbose=True)
        r_score = compute_score(result)
        if r_score < best_score:
            best_score = r_score
            best_seq = result.copy()
            print(f"  *** Improved to {best_score:.6f} ***")
        print(f"  Phase 3 best: {best_score:.6f}")

    # Phase 4: LP refinement on best
    print("\n=== Phase 4: LP refinement ===")
    if best_seq is not None:
        n = len(best_seq)
        S = np.sqrt(2.0 * n)
        a = project_simplex(best_seq, S)
        t_limit = min(180.0, (deadline - time.monotonic()) * 0.3)
        print(f"  n={n}, time_limit={t_limit:.0f}s")
        result = lp_refine(a, t_limit, S, max_iters=300, verbose=True)
        r_score = compute_score(result)
        if r_score < best_score:
            best_score = r_score
            best_seq = result.copy()
            print(f"  *** Improved to {best_score:.6f} ***")
        print(f"  Phase 4 best: {best_score:.6f}")

    # Phase 5: Second round of soft-max with lower temperature
    print("\n=== Phase 5: Soft-max round 2 (lower T) ===")
    if best_seq is not None:
        n = len(best_seq)
        S = np.sqrt(2.0 * n)
        a = project_simplex(best_seq, S)
        conv = fft_convolve(a)
        T_init = 0.03 * np.max(conv)
        t_limit = min(120.0, (deadline - time.monotonic()) * 0.3)
        print(f"  n={n}, T_init={T_init:.4f}, time_limit={t_limit:.0f}s")
        result = soft_max_descent(a, t_limit, S, T_init=T_init, T_min=T_init / 100.0, verbose=True)
        r_score = compute_score(result)
        if r_score < best_score:
            best_score = r_score
            best_seq = result.copy()
            print(f"  *** Improved to {best_score:.6f} ***")
        print(f"  Phase 5 best: {best_score:.6f}")

    # Phase 6: Targeted local search
    print("\n=== Phase 6: Targeted local search ===")
    if best_seq is not None:
        n = len(best_seq)
        S = np.sqrt(2.0 * n)
        a = project_simplex(best_seq, S)
        t_limit = min(120.0, deadline - time.monotonic() - 30)
        if t_limit > 10:
            print(f"  n={n}, time_limit={t_limit:.0f}s")
            result = targeted_local_search(a, t_limit, S, rng, verbose=True)
            r_score = compute_score(result)
            if r_score < best_score:
                best_score = r_score
                best_seq = result.copy()
                print(f"  *** Improved to {best_score:.6f} ***")
            print(f"  Phase 6 best: {best_score:.6f}")

    # Phase 7: Final LP polish
    remaining = deadline - time.monotonic() - 5
    if best_seq is not None and remaining > 20:
        print(f"\n=== Phase 7: Final LP polish (limit={remaining:.0f}s) ===")
        n = len(best_seq)
        S = np.sqrt(2.0 * n)
        a = project_simplex(best_seq, S)
        result = lp_refine(a, remaining, S, max_iters=500, verbose=True)
        r_score = compute_score(result)
        if r_score < best_score:
            best_score = r_score
            best_seq = result.copy()
            print(f"  *** Final improvement to {best_score:.6f} ***")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: evaluate_score={final_score:.6f}, n={len(final_list)}, local_best={best_score:.6f} ===")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]

    return [1.0] * 256
```
