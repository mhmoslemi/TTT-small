def peak_shave(a, n_shave=5, strength=0.05, n_iter=100):

        hi = min(n-1, peak_k)
        for j in range(lo, hi+1):
            contribs[j] = a[j] * a[peak_k - j]
        
        # Reduce top contributors, increase others
        top_n = max(3, n // 20)
        order = np.argsort(contribs)[::-1]

        top_mask[order[:top_n]] = True

        other_mask = ~top_mask

            a[other_mask] += deficit / other_mask.sum()
        
        a = np.clip(a, 0.0, None)
        S = float(np.sum(a))

            hi = min(n-1, ki)
            for j in range(lo, hi+1):

        # Multiplicative update: reduce heavily-loaded elements
        # a_new[j] = a[j] * (1/mean(load))^alpha * (mean(load)/load[j])^alpha

def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)

def build_conv_matrix(f):
    """Build matrix A where A[k,:] @ b = conv(f, b)[k]."""

        f = (1.0 - best_t) * f + best_t * b
        f = np.clip(f, 0.0, None)
        Sc = float(np.sum(f))
        if Sc < 1e-10:
            break
        f = f / Sc
        it += 1
        
        if best_s < best_score - 1e-12:
            best_score = best_s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
        
        if verbose and it % 20 == 0:
            print(f"    [LP {time.monotonic()-t0:6.1f}s] it={it:4d} score={best_s:.6f} best={best_score:.6f} stall={stall}")
        
        if stall >= 60:
            rng_esc = np.random.default_rng(42 + it * 7919)
            f += 0.03 * rng_esc.standard_normal(n)
            f = np.clip(f, 0.001, None)
            f = f / float(np.sum(f))
            stall = 0
    
    return best_f, best_score


def multiplicative_update(f_init, time_limit, alpha=0.4, verbose=False):
    """
    Multiplicative (Sinkhorn-like) update to flatten the convolution.
    Reduces elements that contribute heavily to peaks.
    """
    n = len(f_init)
    a = f_init.astype(np.float64).copy()
    S = float(np.sum(a))
    a = a / S
    best_a = a.copy()
    best_score = score_fast(best_a)
    t0 = time.monotonic()
    it = 0
    n_peaks = min(30, 2 * n - 1)
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 0.5:
            break
        
        c = fft_convolve(a, a)
        top_idx = np.argsort(c)[-n_peaks:]
        
        # Compute load on each element from top peaks
        load = np.zeros(n)
        for ki in top_idx:
            ki = int(ki)
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            for j in range(lo, hi + 1):
                load[j] += a[ki - j]
        
        load /= float(n_peaks)
        mean_load = float(np.mean(load))
        if mean_load < 1e-15:
            break
        
        # Multiplicative correction
        ratio = np.where(load > 1e-15, mean_load / load, 1.0)
        a = a * np.power(ratio, alpha)
        a = np.clip(a, 1e-8, None)
        S = float(np.sum(a))
        a = a / S
        
        it += 1
        if it % 100 == 0:
            s = score_fast(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
            if verbose:
                print(f"    [mult {time.monotonic()-t0:6.1f}s] it={it} score={s:.6f} best={best_score:.6f}")
    
    s = score_fast(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()
    return best_a, best_score


def subgradient_descent(f_init, time_limit, verbose=False):
    """Subgradient descent with momentum for minimizing peak convolution."""
    n = len(f_init)
    a = f_init.astype(np.float64).copy()
    S = float(np.sum(a))
    a = a / S
    best_a = a.copy()
    best_score = score_fast(best_a)
    t0 = time.monotonic()
    
    eta = 0.005
    decay = 0.9995
    momentum = np.zeros(n)
    beta = 0.92
    it = 0
    n_peaks = min(15, 2 * n - 1)
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 0.5:
            break
        
        c = fft_convolve(a, a)
        top_idx = np.argsort(c)[-n_peaks:]
        
        # Compute subgradient (average of active constraint gradients)
        g = np.zeros(n)
        weights = np.zeros(len(top_idx))
        for i, ki in enumerate(top_idx):
            weights[i] = c[int(ki)]
        w_sum = float(np.sum(weights))
        if w_sum < 1e-15:
            break
        weights /= w_sum
        
        for i, ki in enumerate(top_idx):
            ki = int(ki)
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            # Gradient of conv_ki w.r.t. a[j] = a[ki-j] for valid j
            g[lo:hi + 1] += weights[i] * a[ki - np.arange(lo, hi + 1)]
        
        # Remove mean to maintain sum=1 constraint
        g = g - np.mean(g)
        
        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * g
        a = a - eta * momentum
        a = np.clip(a, 1e-8, None)
        S = float(np.sum(a))
        a = a / S
        
        eta *= decay
        it += 1
        
        if it % 300 == 0:
            s = score_fast(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
            if verbose:
                print(f"    [subgrad {time.monotonic()-t0:6.1f}s] it={it} score={s:.6f} best={best_score:.6f}")
    
    s = score_fast(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()
    return best_a, best_score


def peak_shave_iterative(a, n_iter=300, strength=0.04, n_reduce=None):
    """Iteratively reduce top peak contributors and redistribute."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(np.sum(a))
    a = a / S
    if n_reduce is None:
        n_reduce = max(3, n // 25)
    
    for _ in range(n_iter):
        c = fft_convolve(a, a)
        peak_k = int(np.argmax(c))
        
        # Find contributors to the peak
        lo = max(0, peak_k - (n - 1))
        hi = min(n - 1, peak_k)
        contribs = np.zeros(n)
        for j in range(lo, hi + 1):
            contribs[j] = a[j] * a[peak_k - j]
        
        # Reduce top contributors
        order = np.argsort(contribs)[::-1]
        top_mask = np.zeros(n, dtype=bool)
        top_mask[order[:n_reduce]] = True
        
        old_sum = float(np.sum(a))
        a[top_mask] *= (1.0 - strength)
        deficit = old_sum - float(np.sum(a))
        other_mask = ~top_mask
        if other_mask.sum() > 0 and deficit > 1e-12:
            a[other_mask] += deficit / other_mask.sum()
        
        a = np.clip(a, 1e-8, None)
        a = a / float(np.sum(a))
    
    return a


def coordinate_descent(f_init, time_limit, verbose=False):
    """Coordinate descent: optimize one element at a time for peak reduction."""
    n = len(f_init)
    a = f_init.astype(np.float64).copy()
    S = float(np.sum(a))
    a = a / S
    best_a = a.copy()
    best_score = score_fast(best_a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 0.5:
            break
        
        c = fft_convolve(a, a)
        peak_k = int(np.argmax(c))
        
        # Pick a random element to modify
        j = int(rng.integers(0, n))
        
        # Try several multiplier values
        best_mult = 1.0
        best_s = score_fast(a)
        for mult in [0.7, 0.85, 1.15, 1.3]:
            a[j] *= mult
            a = np.clip(a, 0.0, None)
            S = float(np.sum(a))
            if S < 1e-10:
                a[j] /= mult
                continue
            a = a / S
            s = score_fast(a)
            if s < best_s:
                best_s = s
                best_mult = mult
            else:
                a[j] /= mult
                a = a / float(np.sum(a))
        
        a[j] *= best_mult
        a = np.clip(a, 0.0, None)
        a = a / float(np.sum(a))
        
        s = score_fast(a)
        if s < best_score - 1e-12:
            best_score = s
            best_a = a.copy()
    
    return best_a, best_score


def make_hann_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_flat_init(n, rng):
    a = np.ones(n)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_broad_gauss_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n / 2.2
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a += 0.01 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_cosine_mod_init(n, rng):
    """Cosine modulation - designed to flatten convolution."""
    t = np.arange(n, dtype=np.float64)
    # Base is broad, modulate to reduce center peak
    base = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / (n - 1))) if n > 1 else np.ones(1)
    mod = 1.0 + 0.3 * np.cos(4.0 * np.pi * t / (n - 1)) if n > 1 else np.ones(1)
    a = base * mod + 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: LP iteration + multiplicative updates + subgradient
    + coordinate descent, across multiple lengths and initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(seed)
    
    best_score = np.inf
    best_seq = None
    
    # Check provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = score_fast(provided)
        print(f"Provided: n={len(provided)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = provided.copy()
    
    # Define tasks: (n, init_name, use_lp)
    tasks = []
    if provided is not None:
        tasks.append((len(provided), 'provided', True))
    
    # Focus on lengths where LP is feasible (n <= 1024)
    for n in [256, 320, 384, 512, 640, 768, 1024]:
        tasks.append((n, 'flat', True))
        tasks.append((n, 'broad_gauss', True))
        tasks.append((n, 'cosine_mod', True))
        tasks.append((n, 'hann', True))
    
    # Longer lengths: use gradient/multiplicative only
    for n in [1536, 2048]:
        tasks.append((n, 'broad_gauss', False))
        tasks.append((n, 'flat', False))
    
    # Phase 1: LP-based optimization
    print("\n=== Phase 1: LP optimization ===")
    lp_tasks = [(n, name) for n, name, use_lp in tasks if use_lp]
    n_lp = len(lp_tasks)
    
    for idx, (n, init_name) in enumerate(lp_tasks):
        remaining = deadline - time.monotonic() - 200
        if remaining < 30:
            break
        
        # Generate init
        if init_name == 'provided' and provided is not None:
            if len(provided) == n:
                f0 = provided.copy()
            else:
                x_old = np.linspace(0, 1, len(provided))
                x_new = np.linspace(0, 1, n)
                f0 = np.interp(x_new, x_old, provided)
        elif init_name == 'flat':
            f0 = make_flat_init(n, rng)
        elif init_name == 'broad_gauss':
            f0 = make_broad_gauss_init(n, rng)
        elif init_name == 'cosine_mod':
            f0 = make_cosine_mod_init(n, rng)
        else:
            f0 = make_hann_init(n, rng)
        
        # Time allocation
        tasks_left = n_lp - idx
        t_alloc = max(20.0, min(120.0, remaining / max(1, tasks_left) * 0.7))
        
        print(f"  [{idx+1}/{n_lp}] n={n:4d} init={init_name:12s} t={t_alloc:.0f}s")
        f_best, s_best = lp_iteration(f0, t_alloc, t_step=0.08, use_tightened=(n <= 512), verbose=False)
        print(f"       => score={s_best:.6f} (best={best_score:.6f})")
        
        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            print(f"       *** NEW BEST: {best_score:.6f} ***")
    
    # Phase 2: Multiplicative update on best
    if best_seq is not None and time.monotonic() < deadline - 80:
        mt = min(150.0, deadline - time.monotonic() - 60.0)
        print(f"\n=== Phase 2: Multiplicative update ({mt:.0f}s) ===")
        f_mult, s_mult = multiplicative_update(best_seq, mt, alpha=0.35, verbose=True)
        if s_mult < best_score:
            best_score = s_mult
            best_seq = f_mult.copy()
            print(f"  => score={best_score:.6f}")
    
    # Phase 3: Subgradient descent
    if best_seq is not None and time.monotonic() < deadline - 60:
        st = min(100.0, deadline - time.monotonic() - 40.0)
        print(f"\n=== Phase 3: Subgradient ({st:.0f}s) ===")
        f_sub, s_sub = subgradient_descent(best_seq, st, verbose=True)
        if s_sub < best_score:
            best_score = s_sub
            best_seq = f_sub.copy()
            print(f"  => score={best_score:.6f}")
    
    # Phase 4: Peak shaving
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Phase 4: Peak shaving ===")
        t0 = time.monotonic()
        f_shaved = peak_shave_iterative(best_seq, n_iter=500, strength=0.02)
        s_shaved = score_fast(f_shaved)
        print(f"  Peak shave: {s_shaved:.6f} (took {time.monotonic()-t0:.1f}s)")
        if s_shaved < best_score:
            best_score = s_shaved
            best_seq = f_shaved.copy()
    
    # Phase 5: Coordinate descent
    if best_seq is not None and time.monotonic() < deadline - 20:
        cd_t = min(50.0, deadline - time.monotonic() - 10.0)
        if cd_t > 5:
            print(f"\n=== Phase 5: Coordinate descent ({cd_t:.0f}s) ===")
            f_cd, s_cd = coordinate_descent(best_seq, cd_t, verbose=False)
            if s_cd < best_score:
                best_score = s_cd
                best_seq = f_cd.copy()
                print(f"  => score={best_score:.6f}")
    
    # Phase 6: Final LP pass if time allows and n is small enough
    if best_seq is not None and time.monotonic() < deadline - 5 and len(best_seq) <= 768:
        fl_t = deadline - time.monotonic() - 3.0
        if fl_t > 15:
            print(f"\n=== Phase 6: Final LP ({fl_t:.0f}s) ===")
            f_fin, s_fin = lp_iteration(best_seq, fl_t, t_step=0.04, use_tightened=True, verbose=False)
            if s_fin < best_score:
                best_score = s_fin
                best_seq = f_fin.copy()
                print(f"  => score={best_score:.6f}")
    
    # Finalize
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*50}")
        return final_list
    
    return [1.0] * 256
