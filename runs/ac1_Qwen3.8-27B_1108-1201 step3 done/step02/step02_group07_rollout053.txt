For the smooth gradient, the key formula is:
- LSE(β, c) = max(c) + (1/β) log(sum(exp(β(c - max(c)))))
- ∂LSE/∂c_k = exp(β(c_k - max(c))) / sum(exp(β(c - max(c)))) =: p_k
- ∂LSE/∂a_j = sum_k p_k * ∂c_k/∂a_j = sum_k p_k * a_{k-j} (for valid k-j)

The last expression is a convolution/correlation that can be computed via FFT.

```python
def lse_gradient(a, beta):
    """Compute the gradient of LSE(beta * conv(a,a)) w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)  # length 2n-1
    # Softmax weights
    max_c = np.max(c)
    exp_c = np.exp(beta * (c - max_c))
    p = exp_c / np.sum(exp_c)
    
    # grad_j = sum_k p_k * a_{k-j} for valid indices
    # This is the cross-correlation of p with a
    # cross[j] = sum_k p[k] * a[k-j]
    # = sum_i p[i+j] * a[i]  (letting i = k-j)
    
    # Using FFT: cross = ifft(conj(FFT(p)) * FFT(a))
    # But we need to be careful about lengths.
    # p has length 2n-1, a has length n.
    # We want cross[j] for j = 0, ..., n-1.
    # cross[j] = sum_{k} p[k] * a[k-j] where 0 <= k < 2n-1 and 0 <= k-j < n
    
    # Let's compute via direct convolution for clarity:
    # cross[j] = sum_{i=0}^{n-1} a[i] * p[i+j]
    # (where p[i+j] = 0 if i+j >= 2n-1)
    
    # This is (a * p)[j] where * denotes cross-correlation, or equivalently:
    # = (a * flip(p))[len(p)-1+j] ... hmm, let me just use the FFT approach.
    
    size = 1
    while size < 3 * n:  # need enough room
        size <<= 1
    
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    
    # ifft(conj(P) * A)[j] = sum_k P[k] * A[k] * exp(2πijk/size) / size
    # In time domain: sum_m p[m] * a[m-j] ... 
    # Actually let me just verify with a small example in code.
    
    # Alternative: just compute directly (n=256, 2n-1=511, so O(n^2) = 65536 ops, fine)
    grad = np.zeros(n)
    for j in range(n):
        # grad[j] = sum_k p[k] * a[k-j] where 0 <= k-j < n
        # k ranges from j to j+n-1, but also k < 2n-1
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        if k_end > k_start:
            grad[j] = np.dot(p[k_start:k_end], a[0:k_end-k_start])
    
    return grad, p, c
```

```python
def lse_grad_fft(a, beta):
    n = len(a)
    c = fft_convolve(a, a)
    max_c = np.max(c)
    exp_c = np.exp(beta * (c - max_c))
    p = exp_c / exp_c.sum()
    
    size = 1
    while size < 2 * (2 * n - 1) + n:  # enough for no aliasing
        size <<= 1
    # Actually we need size >= (2n-1) + n = 3n-1 for the cross-correlation
    # But to be safe:
    size = 1
    while size < 4 * n:
        size <<= 1
    
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    grad = np.fft.irfft(np.conj(P) * A, size)[:n]
    
    lse_val = max_c + np.log(exp_c.sum()) / beta
    return lse_val, grad, p, c
```

def lse_optimize(a_init, time_limit, beta_start=1.0, beta_end=30.0, lr=0.05):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    theta = np.log(a)  # log parameterization
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_score = np.inf
    best_a = a.copy()
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Anneal beta
        progress = min(1.0, t / 5000)  # time to reach max beta
        beta = beta_start + (beta_end - beta_start) * progress
        
        # Compute a from theta
        a = np.exp(theta)
        a = a / a.sum()
        
        # Compute LSE and gradient
        c = fft_convolve(a, a)
        max_c = np.max(c)
        shifted = beta * (c - max_c)
        # Clip for numerical stability
        shifted = np.minimum(shifted, 50)
        exp_shifted = np.exp(shifted)
        p = exp_shifted / exp_shifted.sum()
        
        # LSE value (for monitoring)
        lse = max_c + np.log(exp_shifted.sum()) / beta
        
        # Gradient w.r.t. a
        # grad_a[j] = sum_k p[k] * a[k-j]
        size = 1
        while size < 4 * n:
            size <<= 1
        P = np.fft.rfft(p, size)
        A = np.fft.rfft(a, size)
        grad_a = np.fft.irfft(np.conj(P) * A, size)[:n]
        
        # Gradient w.r.t. theta: grad_theta = a * grad_a
        grad_theta = a * grad_a
        
        # Center the gradient (since we're on the simplex, the mean component is irrelevant)
        # Actually with log-param and normalization, we should subtract mean
        grad_theta = grad_theta - grad_theta.mean()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * grad_theta ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Periodically check score
        if t % 100 == 0:
            a = np.exp(theta)
            a = a / a.sum()
            score = score_fast(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            if t % 1000 == 0:
                print(f"    LSE: t={t}, beta={beta:.1f}, score={score:.6f}, best={best_score:.6f}")
    
    # Final
    a = np.exp(theta)
    a = a / a.sum()
    score = score_fast(a)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_a, best_score

"""
Hybrid LSE-gradient + LP optimization for minimizing peak autocorrelation
of non-negative sequences. Uses log-sum-exp smoothing with Adam optimizer,
beta annealing, LP direction refinement, and local search polishing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def lse_gradient(a, beta):
    """
    Compute LSE(beta * conv(a,a)) and its gradient w.r.t. a.
    Returns (lse_value, grad_a, p_weights, conv).
    """
    n = len(a)
    c = fft_convolve(a, a)
    max_c = np.max(c)
    shifted = beta * (c - max_c)
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    lse_val = max_c + np.log(exp_shifted.sum()) / beta

    # grad_a[j] = sum_k p[k] * a[k-j] (cross-correlation)
    size = 1
    while size < 4 * n:
        size <<= 1
    P = rfft(p, size)
    A = rfft(a, size)
    grad_a = irfft(np.conj(P) * A, size)[:n]

    return lse_val, grad_a, p, c


def lse_optimize(a_init, time_limit, beta_start=1.0, beta_end=30.0, lr=0.05,
                 anneal_iters=8000, verbose=True):
    """
    LSE-based gradient optimization with Adam and beta annealing.
    Uses log-parameterization to maintain non-negativity.
    """
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 1e-8, None)
    a = a / a.sum()
    theta = np.log(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    best_score = score_fast(a)
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 0.5:
            break

        t += 1
        progress = min(1.0, t / anneal_iters)
        beta = beta_start + (beta_end - beta_start) * progress

        a = np.exp(theta)
        a = a / a.sum()

        lse_val, grad_a, p, c = lse_gradient(a, beta)

        # Gradient w.r.t. theta
        grad_theta = a * grad_a
        grad_theta = grad_theta - grad_theta.mean()

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_theta
        v = beta2 * v + (1 - beta2) * grad_theta ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        theta = theta - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 200 == 0 or t == 1:
            a = np.exp(theta)
            a = a / a.sum()
            s = score_fast(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
            if verbose and (t % 2000 == 0 or t < 10):
                elapsed = time.monotonic() - t0
                print(f"    LSE: t={t:5d} beta={beta:6.1f} lse={lse_val:.6f} "
                      f"score={s:.6f} best={best_score:.6f} [{elapsed:.0f}s]")

        if t % 5000 == 0:
            # Adaptive learning rate: reduce if oscillating
            a_check = np.exp(theta)
            a_check = a_check / a_check.sum()
            if score_fast(a_check) > best_score * 1.1:
                lr *= 0.7

    a = np.exp(theta)
    a = a / a.sum()
    s = score_fast(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()

    return best_a, best_score


def build_A(f):
    """Build matrix where A@b = conv(f, b)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, timeout=5.0):
    """Solve: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    A = build_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(f_init, time_limit, t_step=0.05, verbose=True):
    """LP-based iterative optimization."""
    n = len(f_init)
    f = np.asarray(f_init, dtype=np.float64).copy()
    f = np.clip(f, 0.01, None)
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break

        c = fft_convolve(f, f)
        M = float(np.max(c))

        b = solve_lp(f, M, timeout=3.0)
        if b is None:
            break

        S = float(np.sum(f))
        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b * (S / Sb)

        best_t = t_step
        best_s = np.inf
        for t_try in [t_step, 0.5 * t_step, 0.2 * t_step, 2.0 * t_step]:
            f_cand = (1.0 - t_try) * f + t_try * b
            f_cand = np.clip(f_cand, 0.0, 1000.0)
            s_cand = score_fast(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        f = (1.0 - best_t) * f + best_t * b
        f = np.clip(f, 0.0, 1000.0)
        it += 1

        if best_s < best_score - 1e-10:
            best_score = best_s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP: it={it:4d} score={best_s:.6f} best={best_score:.6f} "
                  f"stall={stall} [{elapsed:.0f}s]")

        if stall >= 40:
            rng_esc = np.random.default_rng(42 + it * 7919)
            f += 0.05 * rng_esc.standard_normal(n)
            f = np.clip(f, 0.01, 1000.0)
            stall = 0

    return best_f, best_score


def local_search(f_init, time_limit, verbose=True):
    """Perturbation-based local search."""
    n = len(f_init)
    f = np.asarray(f_init, dtype=np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    temp = 0.001
    cooling = 0.9999

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 0.5:
            break

        candidate = f.copy()
        r = rng.random()

        if r < 0.35:
            c = fft_convolve(f, f)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for j in range(n):
                idx = peak_idx - j
                if 0 <= idx < n:
                    contribs[j] = f[j] * f[idx]
            top_k = max(3, n // 10)
            threshold = np.sort(contribs)[-top_k]
            top_mask = contribs >= threshold
            n_top = top_mask.sum()
            if n_top > 0:
                candidate[top_mask] *= np.exp(rng.normal(-0.04, 0.025, size=n_top))
            other = ~top_mask
            if other.sum() > 0:
                idxs = np.where(other)[0]
                chosen = rng.choice(idxs, size=min(8, len(idxs)), replace=False)
                candidate[chosen] *= np.exp(rng.normal(0.04, 0.025, size=len(chosen)))
        elif r < 0.65:
            width = max(2, int(rng.integers(2, max(3, n // 6))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        else:
            candidate *= np.exp(rng.normal(0.0, 0.01, size=n))

        candidate = np.clip(candidate, 0.001, 1000.0)
        new_score = score_fast(candidate)

        if new_score < best_score:
            best_score = new_score
            best_f = candidate.copy()
            f = candidate
        elif rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-15)):
            f = candidate

        temp *= cooling

    return best_f, best_score


def get_provided_sequence():
    """Get the provided height_sequence_1 if available."""
    try:

    except Exception:
        pass

def interpolate_to_length(a, target_n):
    """Interpolate sequence to target length."""
    if len(a) == target_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, target_n)
    return np.clip(np.interp(x_new, x_old, a), 0.01, 1000.0)


def make_hann_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_gaussian_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    center = n / 2
    sigma = n / 4
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_exp_init(n, rng):
    a = rng.exponential(1.0, size=n)
    return np.clip(a, 0.01, None)
