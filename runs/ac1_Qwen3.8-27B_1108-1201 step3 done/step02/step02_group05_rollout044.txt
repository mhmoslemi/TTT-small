```python
import numpy as np
import math
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_objective(a):
    """2n * max(conv(a,a)) for sum(a)=1"""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()

def compute_grad_smooth(a, tau):
    """Gradient of tau*log(sum(exp(tau*c_k))) w.r.t. a"""
    n = len(a)
    c = fast_convolve(a)
    c_shifted = c - c.max()
    exp_c = np.exp(tau * c_shifted)
    w = exp_c / exp_c.sum()
    # grad_i = sum_k w_k * a_{k-i} for valid k-i
    grad = np.zeros(n)
    for i in range(n):
        k_start = i
        k_end = min(i + n, 2*n - 1)
        if k_start < k_end:
            grad[i] = np.dot(w[k_start:k_end], a[:k_end - k_start])
    return grad

def gradient_descent(a, time_limit, tau_init=10.0, tau_max=500.0):
    """Adam-based gradient descent with increasing tau."""
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = compute_objective(a)
    t0 = time.monotonic()
    
    # Adam parameters
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.01
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Increase tau over time
        tau = min(tau_max, tau_init * (1.0 + 0.05 * t))
        grad = compute_grad_smooth(a, tau)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        obj = compute_objective(a_new)
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new
        
        # Decay learning rate
        if t % 100 == 0:
            lr *= 0.95
    
    return best_a, best_obj
```

Another idea: **moment matching**. The convolution values at different lags are related to the "moments" of the distribution. If we think of a as a probability distribution on {0, ..., n-1}, then (a*a)_k is the probability that two independent draws sum to k. We want to minimize the maximum of this sum distribution.

This is related to the "sumset" problem in additive combinatorics. The sumset A+A of a set A is the set of all sums a1+a2. We want the "sumset" to be as uniformly distributed as possible.

For continuous distributions, the sum of two i.i.d. has a density that's the convolution of the individual densities. We want this to be as flat as possible. The uniform distribution on [0,1] gives a triangular sum density (peak at 0.5). To make it flatter, we'd want a distribution that's spread out in a specific way.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. LP-based optimization for directional moves
2. Smooth gradient descent (log-sum-exp) for fast local refinement
3. Gradient-guided multi-index local search with simulated annealing
4. Perturbation + re-optimization to escape local minima
Uses FFT-based convolution for speed.
"""

import numpy as np
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    """Objective 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def get_peak_contrib(a, k_star):
    """For peak at k_star, return contribution a[i]*a[k_star-i] for each i."""
    n = len(a)
    contrib = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            contrib[i] = a[i] * a[j]
    return contrib


def compute_smooth_grad(a, tau):
    """Gradient of log-sum-exp smoothing of the convolution peak."""
    n = len(a)
    c = fast_convolve(a)
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    w = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for i in range(n):
        k_start = i
        k_end = min(i + n, 2 * n - 1)
        if k_start < k_end:
            grad[i] = np.dot(w[k_start:k_end], a[:k_end - k_start])
    return grad


def compute_exact_grad(a):
    """Subgradient of max(conv(a,a)) at the argmax index."""
    n = len(a)
    c = fast_convolve(a)
    M = c.max()
    # Find all indices within tolerance of max
    tol = max(1e-10, 0.01 * M)
    active = np.where(c >= M - tol)[0]
    grad = np.zeros(n)
    for k in active:
        for i in range(n):
            j = int(k) - i
            if 0 <= j < n:
                grad[i] += a[j]
    grad /= len(active)
    return grad


def smooth_gradient_descent(a0, time_limit, tau_init=5.0, tau_max=200.0):
    """Adam-based gradient descent on smoothed objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = obj_norm(a)
    t0 = time.monotonic()

    beta1, beta2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.02
    no_improve = 0
    max_no_improve = 200

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        t += 1
        tau = min(tau_max, tau_init * (1.0 + 0.02 * t))
        grad = compute_smooth_grad(a, tau)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new = a_new / s

        obj = obj_norm(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        if t % 50 == 0:
            lr *= 0.9

    return best_a, best_obj


def solve_lp_direction(f, M):
    """LP: find g>=0 maximizing sum(g) s.t. conv(f,g) <= M."""
    import scipy.optimize
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = f[:end - j]
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                                     options={'presolve': True, 'time_limit': 30})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def lp_optimize(f0, time_limit):
    """Iterative LP-based optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-12:
        f = np.ones_like(f) / len(f)
    else:
        f = f / s
    best_f = f.copy()
    best_obj = obj_norm(f)
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 40

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        conv = fast_convolve(f)
        M = conv.max()
        if M < 1e-15:
            break
        g = solve_lp_direction(f, M)
        if np.allclose(g, f, atol=1e-10):
            no_improve += 3
        else:
            # Line search
            best_t = 0.0
            best_obj_t = best_obj
            for t_frac in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                a_new = (1.0 - t_frac) * f + t_frac * g
                a_new = np.maximum(a_new, 0.0)
                ns = a_new.sum()
                if ns < 1e-12:
                    continue
                a_new = a_new / ns
                obj_t = obj_norm(a_new)
                if obj_t < best_obj_t:
                    best_obj_t = obj_t
                    best_t = t_frac
            if best_t > 0 and best_obj_t < best_obj - 1e-12:
                f = (1.0 - best_t) * f + best_t * g
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                best_obj = best_obj_t
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        if no_improve >= max_no_improve:
            break
    return best_f, best_obj


def gradient_guided_local_search(a0, time_limit):
    """Local search using gradient to identify best donor/receiver pairs."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s
    best_obj = obj_norm(a)
    best_a = a.copy()
    t0 = time.monotonic()
    stuck = 0
    max_stuck = 100

    while time.monotonic() - t0 < time_limit and stuck < max_stuck:
        c = fast_convolve(a)
        M = c.max()
        k_star = int(np.argmax(c))

        contrib = get_peak_contrib(a, k_star)
        # Also consider second peak
        c_sorted = np.sort(c)[::-1]
        active_k = np.where(c >= M - 0.1 * M)[0]

        # Compute gradient-like scores
        scores = np.zeros(n)
        for k in active_k[:10]:
            for i in range(n):
                j = int(k) - i
                if 0 <= j < n:
                    scores[i] += a[j]
        scores /= len(active_k[:10])

        # Donors: high score (reduce them)
        # Receivers: low score (increase them)
        danger = np.argsort(scores)[::-1][:20]
        safe = np.argsort(scores)[:20]

        found = False
        for di_idx in range(len(danger)):
            if found:
                break
            di = int(danger[di_idx])
            if a[di] < 1e-10:
                continue
            for si_idx in range(len(safe)):
                if found:
                    break
                si = int(safe[si_idx])
                if di == si:
                    continue
                # Try moving various fractions
                for frac in [0.05, 0.1, 0.15, 0.25, 0.4, 0.6]:
                    amount = a[di] * frac
                    if amount < 1e-12:
                        continue
                    a_trial = a.copy()
                    a_trial[di] -= amount
                    a_trial[si] += amount
                    a_trial = np.maximum(a_trial, 0.0)
                    ts = a_trial.sum()
                    if ts < 1e-12:
                        continue
                    a_trial = a_trial / ts
                    o = obj_norm(a_trial)
                    if o < best_obj - 1e-12:
                        best_obj = o
                        a = a_trial.copy()
                        best_a = a.copy()
                        found = True
                        break

        # Multi-index move: split donor mass among 2-3 receivers
        if not found:
            top_donors = np.argsort(scores)[::-1][:5]
            bottom_receivers = np.argsort(scores)[:5]
            multi_found = False
            for di_idx in range(len(top_donors)):
                if multi_found:
                    break
                di = int(top_donors[di_idx])
                if a[di] < 1e-10:
                    continue
                for r1 in range(len(bottom_receivers)):
                    if multi_found:
                        break
                    for r2 in range(r1 + 1, len(bottom_receivers)):
                        if multi_found:
                            break
                        si1 = int(bottom_receivers[r1])
                        si2 = int(bottom_receivers[r2])
                        if di == si1 or di == si2:
                            continue
                        for splits in [(0.5, 0.5), (0.7, 0.3), (0.3, 0.7), (0.6, 0.4)]:
                            for frac in [0.1, 0.2, 0.3]:
                                amount = a[di] * frac
                                if amount < 1e-12:
                                    continue
                                a_trial = a.copy()
                                a_trial[di] -= amount
                                a_trial[si1] += amount * splits[0]
                                a_trial[si2] += amount * splits[1]
                                a_trial = np.maximum(a_trial, 0.0)
                                ts = a_trial.sum()
                                if ts < 1e-12:
                                    continue
                                a_trial = a_trial / ts
                                o = obj_norm(a_trial)
                                if o < best_obj - 1e-12:
                                    best_obj = o
                                    a = a_trial.copy()
                                    best_a = a.copy()
                                    multi_found = True
                                    break

        if not found and not multi_found:
            stuck += 1
        else:
            stuck = 0

    return best_a, best_obj


def simulated_annealing_local(a0, time_limit, rng):
    """Simulated annealing local search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    T = 1e-4
    T_MIN = 1e-8
    cooling = 0.995

    while time.monotonic() - t0 < time_limit and T > T_MIN:
        # Random move: pick a donor and receiver
        di = rng.integers(0, n)
        si = rng.integers(0, n)
        if di == si:
            T *= cooling
            continue
        if a[di] < 1e-10:
            T *= cooling
            continue
        frac = rng.uniform(0.05, 0.4)
        amount = a[di] * frac
        a_trial = a.copy()
        a_trial[di] -= amount
        a_trial[si] += amount
        a_trial = np.maximum(a_trial, 0.0)
        ts = a_trial.sum()
        if ts < 1e-12:
            T *= cooling
            continue
        a_trial = a_trial / ts
        o = obj_norm(a_trial)
        delta = o - best_obj
        if delta < 0 or rng.random() < math.exp(-max(0, delta) / T):
            a = a_trial.copy()
            if o < best_obj:
                best_obj = o
                best_a = a.copy()
        T *= cooling

    return best_a, best_obj


def get_parent(n):
    """Get parent sequence resized to length n."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)

        if p.size >= n:
            pc = p[:n].copy()
        else:
            pc = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s < 1e-10:
            return None
        return pc / s
    except Exception:

def make_inits(n, rng):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    inits['uniform'] = np.ones(n) / n
    for sf in [0.25, 0.4, 0.6]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits[f'gauss_{sf}'] = g / g.sum()
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    hann = np.maximum(hann, 0.0)
    inits['hann'] = hann / hann.sum()
    # Two peaks
    two = np.zeros(n)
    two[:n//4] = 1.0
    two[-(n//4):] = 1.0
    inits['two_peaks'] = two / two.sum()
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    parent = get_parent(n)
    if parent is not None:
        inits['parent'] = parent
    return inits

    print(f"Sequence length: {len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
