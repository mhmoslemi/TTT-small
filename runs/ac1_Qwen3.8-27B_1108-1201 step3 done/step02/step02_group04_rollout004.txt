Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
- Smooth gradient descent (log-sum-exp approximation of max)

- Mirror descent updates on the probability simplex

import scipy.optimize as optimize
import scipy.sparse as sparse
import time

def fast_autoconvolve(a):
    """Full auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_smooth_gradient(a, conv, t):
    """Compute gradient of smoothed objective using softmax weights."""
    n = len(a)
    # Softmax weights
    shifted = t * conv - t * conv.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    
    # Gradient: 4*n * cross_correlate(w, a)
    # cross_correlate(w, a)[i] = sum_k w[k] * a[k-i] (with a[j]=0 outside [0,n-1])
    # This is the same as: for each i, sum over valid k of w[k]*a[k-i]
    # Using FFT: this is the "cross-correlation" which can be computed as
    # ifft(conj(fft(w)) * fft(a)) but with appropriate lengths
    
    # Actually, let's compute it directly using the convolution structure
    # grad[i] = 4*n * sum_k w[k] * a[k-i] where 0 <= k-i < n and 0 <= k < 2n-1
    # Let j = k-i, so k = i+j, and we need 0 <= j < n and 0 <= i+j < 2n-1
    # So: 0 <= j < n and -i <= j < 2n-1-i
    # Combined: max(0, -i) <= j < min(n, 2n-1-i)
    
    # Alternative: use the fact that this is a cross-correlation
    # corr[i] = sum_j w[i+j] * a[j] for valid indices
    # = (w * a_reversed)[i + n - 1] ... hmm this is getting complicated
    
    # Let's just use FFT. The cross-correlation of w (length 2n-1) and a (length n)
    # at lag i is: sum_j w[i+j] * a[j] for j where both are defined.
    # This equals the convolution of w and reversed_a at position (2n-2) - i + (n-1)
    # Actually, let me just use a direct FFT approach.
    
    # Define: grad[i] = 4*n * sum_{j=0}^{n-1} w[i+j] * a[j] * 1[0 <= i+j <= 2n-2]
    # Let's compute this as a convolution.
    
    # Actually the simplest way: 
    # grad[i] = 4*n * sum_j a[j] * w[i+j]
    # This is the cross-correlation of a and w, but with w being longer.
    # 
    # Let's pad a to length 2n-1 (zero-pad at the end)
    # Then cross-correlate: corr[i] = sum_j a_padded[j] * w[i+j]
    # For i in [0, n-1] and j in [0, n-1]: i+j in [0, 2n-2], all valid for w.
    
    # Using FFT: cross-correlation = ifft(fft(a) * conj(fft(w)))
    # But we need to be careful about the definition.
    
    # Let's use: corr[i] = sum_j a[j] * w[i+j]
    # = sum_j a[j] * w[i+j]
    # If we define v = reverse(w), then corr[i] = sum_j a[j] * v[(2n-2)-(i+j)]
    # = conv(a, v)[n-1-i+... ] this is getting confusing.
    
    # Let me just do it with a direct FFT computation:
    # corr[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # This is the "a slides over w" correlation.
    # 
    # Pad a to length (2n-1) + n - 1 = 3n-2 for linear convolution
    # Actually, the standard way: 
    # reverse a: a_rev = a[::-1] (length n)
    # conv(a_rev, w) has length n + (2n-1) - 1 = 3n-2
    # corr[i] = conv(a_rev, w)[n-1+i]
    
    fft_len = 1 << (n + (2*n-1) - 1 - 1).bit_length()
    A = np.fft.rfft(a[::-1], fft_len)
    W = np.fft.rfft(w, fft_len)
    conv_result = np.fft.irfft(A * W, fft_len)
    corr = conv_result[n-1:n-1+n]  # take the first n values starting from n-1
    
    return 4.0 * n * corr

def softmax_update(a, grad, lr, t):
    """Mirror descent step: a <- a * exp(-lr * (grad - mean(grad))), then normalize."""
    n = len(a)
    # Center the gradient (projection onto tangent space of simplex)
    grad_centered = grad - grad.mean()
    a_new = a * np.exp(-lr * grad_centered)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        a_new = np.ones(n) / n
    else:
        a_new /= s
    return a_new

def lp_step(f, M_target, n, time_limit=5):
    """Solve LP: max sum(g) s.t. g>=0, conv(f,g)[k] <= M_target for all k."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    vals = np.tile(f, n)
    
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    b = np.full(2*n-1, M_target)
    bounds = [(0, None)] * n
    
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def optimize_sequence(a0, time_limit, use_lp=True, verbose=False):
    """Hybrid optimization: smooth gradient + periodic LP restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        a = np.ones(n) / n
    else:
        a /= s
    
    conv = fast_autoconvolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M  # with sum=1
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = 0.5
    temperature = 10.0 * n  # initial temperature for softmax
    patience = 0
    iter_count = 0
    lp_interval = 50  # do an LP step every 50 iterations
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        
        if iter_count % lp_interval == 0 and use_lp:
            # LP step: try to find a better direction
            M_target = M * 0.95  # aim for slightly lower peak
            g = lp_step(a, M_target, n, time_limit=3)
            if g is not None:
                g_sum = g.sum()
                if g_sum > 1e-15:
                    g /= g_sum
                    a_new = 0.5 * a + 0.5 * g
                    a_new = np.maximum(a_new, 0.0)
                    a_new /= a_new.sum()
                    conv_new = fast_autoconvolve(a_new)
                    obj_new = 2.0 * n * conv_new.max()
                    if obj_new < best_obj - 1e-12:
                        a = a_new
                        conv = conv_new
                        M = conv.max()
                        best_obj = obj_new
                        best_a = a.copy()
                        patience = 0
                    else:
                        # Try smaller blend
                        a_new2 = 0.8 * a + 0.2 * g
                        a_new2 = np.maximum(a_new2, 0.0)
                        a_new2 /= a_new2.sum()
                        conv_new2 = fast_autoconvolve(a_new2)
                        obj_new2 = 2.0 * n * conv_new2.max()
                        if obj_new2 < best_obj - 1e-12:
                            a = a_new2
                            conv = conv_new2
                            M = conv.max()
                            best_obj = obj_new2
                            best_a = a.copy()
                            patience = 0
        
        # Gradient step
        grad = compute_smooth_gradient(a, conv, temperature)
        a = softmax_update(a, grad, lr, temperature)
        
        conv = fast_autoconvolve(a)
        M = conv.max()
        obj = 2.0 * n * M
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            patience = 0
            lr = min(lr * 1.01, 5.0)  # slightly increase lr
        else:
            patience += 1
            if patience % 20 == 0:
                lr *= 0.8
                temperature *= 1.2  # make softmax sharper
                if lr < 0.01:
                    lr = 0.5
                    temperature = 10.0 * n
            if patience > 200:
                break
        
        if verbose and iter_count % 100 == 0:
            print(f"      iter={iter_count}, obj={best_obj:.8f}, lr={lr:.4f}, t={temperature:.1f}")
    
    return best_a, best_obj
