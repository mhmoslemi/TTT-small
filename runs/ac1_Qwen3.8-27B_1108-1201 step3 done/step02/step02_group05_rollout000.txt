"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast gradient descent with
the analytical gradient formula, supplemented by LP-based optimization and
local search. The key insight is that when the peak is at position k*, the
gradient direction is step[m] = M - a[k*-m] (for normalized S=1), enabling
O(n) per-step optimization. Multiple sequence lengths (512-2048) and diverse
initializations are explored with a phased time budget.
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    """Objective: 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def gradient_direction(a, k_star):
    """Compute the steepest descent direction for the objective.
    Returns step array where positive means increase a[m]."""
    n = len(a)
    M = fast_convolve(a)[k_star]
    # step[m] = M - a[k*-m] for valid m, M for invalid m
    step = np.full(n, M)
    m_min = max(0, k_star - n + 1)
    m_max = min(n - 1, k_star)
    if m_max >= m_min:
        # a[k*-m] for m in [m_min, m_max] is a[k*-m_max : k*-m_min+1] reversed
        idx_high = k_star - m_min
        idx_low = k_star - m_max
        step[m_min:m_max + 1] = M - a[idx_low:idx_high + 1][::-1]
    return step


def max_step_size(a, step):
    """Find maximum t such that a + t*step >= 0 for all elements."""
    neg = step < -1e-15
    if neg.any():
        ratios = a[neg] / (-step[neg])
        return max(ratios.min(), 1e-15)
    return 1.0 / max(np.abs(step).max(), 1e-15)


def gradient_descent_optimize(a0, time_limit, restart_interval=2000):
    """Fast gradient descent with adaptive step size and periodic restarts."""
    a = normalize(a0.astype(np.float64).copy())

    best_obj = obj_norm(a)
    t = 0.05
    t0 = time.monotonic()
    stuck = 0
    max_stuck = 200
    iters_since_restart = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        step = gradient_direction(a, k_star)
        t_max = max_step_size(a, step)

        t_actual = min(t, 0.9 * t_max)
        if t_actual < 1e-15:
            t *= 0.5
            stuck += 1
            if stuck > max_stuck:
                break
            continue

        a_new = a + t_actual * step
        a_new = normalize(a_new)

        obj_new = obj_norm(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stuck = 0
            t = min(t * 1.05, 0.3)

        else:

                a = normalize(a)
                t = 0.02
                stuck = 0
                iters_since_restart = 0
                if time.monotonic() - t0 > time_limit * 0.9:

def solve_lp(f, M):
    """Find g >= 0 maximizing sum(g) subject to conv(f,g)[k] <= M for all k."""
    n = len(f)
    # Build constraint matrix A where A[k,j] = f elements contributing to conv[k,j]
    # conv(f,g)[k] = sum_j f[k-j] * g[j], so A[k,j] = f[k-j] if valid
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_min = max(0, j)
        k_max = min(2 * n - 2, j + n - 1)
        for k in range(k_min, k_max + 1):
            fi = k - j
            if 0 <= fi < n and f[fi] > 1e-15:
                rows.append(k)
                cols.append(j)
                vals.append(f[fi])

    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                      method='highs', options={'time_limit': 30})
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def lp_optimize(f0, time_limit):
    """Iterative LP-based optimization with line search."""
    f = normalize(f0.astype(np.float64).copy())
    best_f = f.copy()
    best_obj = obj_norm(f)
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 40

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        conv = fast_convolve(f)
        M = conv.max()
        if M < 1e-15:
            break

        g = solve_lp(f, M)

        if np.allclose(g, f, atol=1e-10):
            no_improve += 5
        else:
            # Line search
            best_t = 0.0
            best_obj_t = best_obj
            for trial_t in [0.1, 0.25, 0.5, 0.75, 1.0]:
                a_new = (1.0 - trial_t) * f + trial_t * g
                a_new = normalize(a_new)
                obj_t = obj_norm(a_new)
                if obj_t < best_obj_t:
                    best_obj_t = obj_t
                    best_t = trial_t

            if best_t > 0 and best_obj_t < best_obj - 1e-11:
                a_new = (1.0 - best_t) * f + best_t * g
                a_new = normalize(a_new)
                f = a_new
                best_obj = best_obj_t
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

        if no_improve >= max_no_improve:
            break

    return best_f, best_obj


def local_search(a, time_limit):
    """Targeted local search: redistribute mass from peak-contributing elements."""
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    stuck = 0
    max_stuck = 200
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit and stuck < max_stuck:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))

        # Find contributing indices
        contrib = np.zeros(n)
        m_min = max(0, k_star - n + 1)
        m_max = min(n - 1, k_star)
        for m in range(m_min, m_max + 1):
            j = k_star - m
            contrib[m] = a[m] * a[j]

        max_c = contrib.max()
        if max_c < 1e-15:
            break

        thr_high = 0.5 * max_c
        thr_low = 0.3 * max_c
        danger = np.where(contrib > thr_high)[0]
        safe = np.where(contrib < thr_low)[0]

        if len(danger) == 0 or len(safe) == 0:
            stuck += 1
            continue

        found = False
        n_danger = min(len(danger), 20)
        n_safe = min(len(safe), 20)
        for di in danger[:n_danger]:
            if found:
                break
            for si in safe[:n_safe]:
                if found:
                    break
                if di == si:
                    continue
                for frac in [0.1, 0.25, 0.5, 0.75, 0.9]:
                    amount = a[di] * frac
                    if amount < 1e-15:

                    a_trial[si] += amount
                    a_trial = normalize(a_trial)
                    o = obj_norm(a_trial)
                    if o < best_obj - 1e-11:
                        best_obj = o
                        a = a_trial
                        found = True
                        break

        else:
            stuck = 0

def make_initializations(n, rng):
    """Generate diverse initializations for sequence of length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    # Gaussian bumps
    for sf in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    # Hann window
    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    else:
        hann = np.ones(n)
    hann = np.maximum(hann, 0.0)
    s = hann.sum()
    if s > 0:
        inits['hann'] = hann / s

    # Triangle
    triangle = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = triangle / triangle.sum()

    # Raised cosine variants
    for width in [0.6, 0.8]:
        rc = np.zeros(n)
        center = n // 2
        hw = int(width * n / 2)
        if hw > 0:
            rc[center - hw:center + hw + 1] = 0.5 * (1 + np.cos(
                np.pi * np.arange(-hw, hw + 1) / hw))
        s = rc.sum()
        if s > 0:
            inits[f'raised_{width}'] = rc / s

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Two-peak
    two_peak = np.zeros(n)
    two_peak[n // 4] = 1.0
    two_peak[3 * n // 4] = 1.0
    two_peak = two_peak + 0.1
    inits['two_peak'] = two_peak / two_peak.sum()

    # Sinc-like
    from math import pi
    t = (x - n / 2) / (n / 10)
    sinc = np.ones_like(t)
    mask = np.abs(t) > 1e-10
    sinc[mask] = np.sin(pi * t[mask]) / (pi * t[mask])
    sinc = np.maximum(sinc, 0.0)
    s = sinc.sum()
    if s > 1e-10:
        inits['sinc'] = sinc / s

    # Parent
    parent = get_parent(n)
    if parent is not None:
        inits['parent'] = parent
        # Jittered parent
        jittered = parent + 0.05 * rng.standard_normal(n)
        jittered = normalize(jittered)

        if remaining_to_deadline < 30:
            break
        t_alloc = min(time_per_scan, 45.0)
        t_alloc = min(t_alloc, remaining_to_deadline - 15.0)
        if t_alloc < 1.0:
            break

        a_opt, obj_opt = gradient_descent_optimize(a0, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score):
            scan_results.append((score, n, name, a_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Scan: n={n}, {name}, score={score:.6f}")

    print(f"  Phase 1 best: {best_score:.6f}")

    # Phase 2: Hybrid gradient+LP on top candidates (45% of budget)
    phase2_end = time.monotonic() + (float(budget_s) - 15.0) * 0.45
    print("Phase 2: Hybrid optimization on top candidates...")

    scan_results.sort(key=lambda x: x[0])
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, name, a_opt))
            seen.add(key)
        if len(top_configs) >= 10:
            break

    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    remaining = phase2_end - time.monotonic()
    if remaining < 10:
        remaining = 10
    time_per_deep = max(10.0, remaining / max(1, len(top_configs)))

    phase2_results = []
    for i, (n, name, a0) in enumerate(top_configs):
        remaining_dl = deadline - time.monotonic() - 30.0
        if remaining_dl < 20:
            break
        t_alloc = min(time_per_deep, remaining_dl - 15.0)
        if t_alloc < 5.0:
            break

        a_opt, obj_opt = gradient_lp_hybrid(a0, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score):
            phase2_results.append((score, n, name, a_opt))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Hybrid: n={n}, {name}, score={score:.6f}")

    print(f"  Phase 2 best: {best_score:.6f}")

    # Phase 3: Intensive optimization on the single best (20% of budget)
    phase3_end = time.monotonic() + (float(budget_s) - 15.0) * 0.20
    print("Phase 3: Intensive optimization on best candidate...")

    if best_sequence is not None:
        n_best = len(best_sequence)
        t_phase3 = phase3_end - time.monotonic()
        if t_phase3 > 5:
            # Split between gradient and LP
            t_grad = t_phase3 * 0.6
            t_lp = t_phase3 * 0.4

            a_opt, obj_opt = gradient_descent_optimize(best_sequence, time_limit=t_grad)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Phase3 grad: score={score:.6f}")

            if n_best <= 1024 and t_lp > 5:
                a_lp, obj_lp = lp_optimize(best_sequence, time_limit=t_lp)
                score = float(evaluate_sequence(a_lp.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_lp.copy()
                    print(f"  Phase3 LP: score={score:.6f}")

    # Phase 4: Local search polish (10% of budget)
    print("Phase 4: Local search polish...")
    if best_sequence is not None:
        ls_time = min(60.0, max(10.0, (float(budget_s) - 15.0) * 0.10))
        remaining_dl = deadline - time.monotonic() - 5.0
        ls_time = min(ls_time, remaining_dl)
        if ls_time > 2:
            a_ls, obj_ls = local_search(best_sequence, time_limit=ls_time)
            score = float(evaluate_sequence(a_ls.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_ls.copy()
                print(f"  Local search: score={score:.6f}")

    # Final: one more gradient pass on the best
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 2.0
        if remaining > 5:
            a_final, obj_final = gradient_descent_optimize(best_sequence, time_limit=remaining)
            score = float(evaluate_sequence(a_final.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_final.copy()
                print(f"  Final gradient: score={score:.6f}")

    # Final evaluation
    if best_sequence is None:
        best_sequence = np.ones(512)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(best_sequence)}")
    return [float(x) for x in best_sequence]
