"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative sequences.
Uses min-peak LP with backtracking, log-sum-exp gradient descent, and local search.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_conv(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_conv(a)
    return float(2.0 * n * np.max(c) / (S * S))


def build_conv_matrix(f):
    """Build matrix A where A @ b = conv(f, b) (full length 2n-1)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for m in range(lo, hi + 1):
            A[k, m] = f[k - m]
    return A


def solve_min_peak_lp(f, S, timeout=15.0):
    """
    Minimize t subject to:
      conv(f, b)[k] <= t for k=0,...,2n-2
      sum(b) = S
      b >= 0
    Returns (b, t) or (None, None).
    """
    n = len(f)
    A = build_conv_matrix(f)
    
    # Variables: [b_0, ..., b_{n-1}, t]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality: conv(f,b)[k] - t <= 0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(b) = S
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([S])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x[:n].copy(), float(res.x[n])
    except Exception:
        pass
    return None, None


def project_sum_nonneg(a, S):
    """Project onto {x: sum(x)=S, x>=0} using bisection."""
    lo = -np.max(np.abs(a)) - S
    hi = np.max(a) + S
    for _ in range(80):
        mid = (lo + hi) / 2
        proj = np.maximum(a - mid, 0.0)
        s = proj.sum()
        if s > S:
            lo = mid
        else:

    t = (lo + hi) / 2
    return np.maximum(a - t, 0.0)

def lp_optimize(f_init, time_limit, min_step=0.3, verbose=False):
    """LP-based optimization with backtracking line search."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    f = f * (S / S)  # keep same sum
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break
        
        b, t_opt = solve_min_peak_lp(f, S, timeout=10.0)
        if b is None:
            break
        
        # Backtracking line search
        best_t = min_step
        best_s = score_fast(b)  # score of full step
        for step in [1.0, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02]:
            f_cand = (1.0 - step) * f + step * b
            f_cand = np.clip(f_cand, 0.0, 1000.0)
            s_cand = score_fast(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = step
        
        f = (1.0 - best_t) * f + best_t * b
        f = np.clip(f, 0.01, 1000.0)
        # Renormalize
        f = f * (S / float(np.sum(f)))
        it += 1
        
        if best_s < best_score - 1e-10:
            best_score = best_s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
        
        if verbose and it % 20 == 0:
            print(f"    [{time.monotonic()-t0:6.1f}s] it={it:4d} score={best_s:.6f} best={best_score:.6f}")
        
        if stall >= 50:
            rng_esc = np.random.default_rng(1000 + it)
            f += 0.05 * S / n * rng_esc.standard_normal(n)
            f = np.clip(f, 0.01, None)
            f = f * (S / float(np.sum(f)))
            stall = 0
    
    return best_f, best_score


def gradient_descent_smooth(f_init, time_limit, alpha_start=10.0, alpha_end=1000.0,
                            lr_start=0.01, verbose=False):
    """
    Gradient descent using log-sum-exp smoothing of the max.
    Alpha increases over time for sharper max approximation.
    """
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    f = f * (S / float(np.sum(f)))
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    momentum = np.zeros(n)
    beta = 0.9
    
    # Precompute FFT size
    fft_size = 1
    while fft_size < 3 * n - 2:
        fft_size <<= 1
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 2.0:
            break
        
        # Adaptive alpha
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        
        # Compute convolution
        c = fft_conv(f)
        
        # Log-sum-exp weights
        c_max = np.max(c)
        c_shifted = c - c_max
        exp_c = np.exp(alpha * c_shifted)
        weights = exp_c / np.sum(exp_c)
        
        # Compute gradient using FFT
        # g[j] = 2 * sum_k weights[k] * f[k-j] (where 0<=k-j<n)
        # = 2 * (f convolve w_rev)[2n-2-j] where w_rev = weights[::-1]
        w_rev = weights[::-1]
        Aw = rfft(f, fft_size)
        Ww = rfft(w_rev, fft_size)
        conv_fw_rev = irfft(Aw * Ww, fft_size)
        # g[j] = 2 * conv_fw_rev[2n-2-j]
        g = 2.0 * conv_fw_rev[2*n-2 : 2*n-2-n : -1][:n]
        # More carefully: g[j] for j=0,...,n-1 is 2*conv_fw_rev[2n-2-j]
        idx = 2*n-2 - np.arange(n)
        g = 2.0 * conv_fw_rev[idx]
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm
        
        # Adam-like momentum
        lr = lr_start * (1.0 - 0.5 * progress)
        momentum = beta * momentum + (1.0 - beta) * g
        f = f - lr * momentum * S / n
        
        # Project

def local_search(f_init, time_limit, verbose=False):
    """Perturbation-based local search."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    f = f * (S / float(np.sum(f)))
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    temp = 0.001
    cooling = 0.9995
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 2.0:
            break
        
        candidate = f.copy()
        r = rng.random()
        scale = S / n
        
        if r < 0.3:
            # Target the peak
            c = fft_conv(f)
            peak_k = int(np.argmax(c))
            # Find contributing pairs
            contribs = np.zeros(n)
            for j in range(n):
                idx = peak_k - j
                if 0 <= idx < n:
                    contribs[j] = f[j] * f[idx]
            # Reduce top contributors
            top_k = max(5, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate[top_idx] *= np.exp(rng.normal(-0.05, 0.03, size=top_k))
            # Increase others
            rest = np.setdiff1d(np.arange(n), top_idx)
            if len(rest) > 0:
                inc = rng.choice(rest, size=min(10, len(rest)), replace=False)
                candidate[inc] *= np.exp(rng.normal(0.05, 0.03, size=len(inc)))
        elif r < 0.5:
            # Gaussian perturbation
            candidate += rng.normal(0, 0.02 * scale, size=n)
        elif r < 0.7:
            # Smooth perturbation (low-pass filtered noise)
            noise = rng.standard_normal(n)
            # Simple low-pass: moving average
            kernel = np.ones(max(1, n // 20)) / max(1, n // 20)
            noise = np.convolve(noise, kernel, mode='same')
            candidate += 0.03 * scale * noise
        elif r < 0.85:
            # Mirror/reverse perturbation
            w = rng.uniform(0.1, 0.5)
            candidate = (1-w) * candidate + w * candidate[::-1]
        else:
            # Random walk on log scale
            log_f = np.log(np.maximum(f, 1e-10))
            log_f += rng.normal(0, 0.02, size=n)
            candidate = np.exp(log_f)
        
        candidate = np.clip(candidate, 0.01, 1000.0)
        candidate = project_sum_nonneg(candidate, S)
        new_score = score_fast(candidate)
        
        current_score = score_fast(f)
        if new_score < best_score:
            best_score = new_score
            best_f = candidate.copy()
            f = candidate
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-15)):
            f = candidate
        
        temp *= cooling
    
    return best_f, best_score
