import numpy as np
import scipy.optimize
from scipy.optimize import minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_val(a):
    """Objective assuming sum(a) = 1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def normalize(a):
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s

def subgradient_direction(a, conv, threshold_frac=0.95):
    """Compute subgradient of max(conv) w.r.t. a.
    Returns a direction to DECREASE the max."""
    max_val = conv.max()
    # Active set: indices where conv is close to max
    active = np.where(conv >= threshold_frac * max_val)[0]
    n = len(a)
    sg = np.zeros(n)
    for k in active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                sg[i] += a[j]
    # Average over active set
    sg /= len(active)
    return sg

def lp_direction(f, M, method='highs'):
    """LP: maximize sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    # Build constraint matrix A_ub such that A_ub @ g <= b
    # conv(f, g)[k] = sum_i f[i] * g[k-i] (valid indices)
    # For k in 0..2n-2: conv[k] = sum_{i: 0<=i<n, 0<=k-i<n} f[i] * g[k-i]
    # = sum_{j: 0<=j<n, 0<=k-j<n} f[k-j] * g[j]
    
    # A_ub has shape (2n-1, n), b has shape (2n-1,)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in conv[k] for k = j, j+1, ..., j+n-1
        # conv[k] gets contribution f[k-j] * g[j]
        for k in range(j, j + n):
            A_ub[k, j] = f[k - j]
    
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None

def line_search(f, direction, max_iter=40):
    """Golden section line search for f_new = f + t * direction, then normalize."""
    def eval_t(t):
        a = f + t * direction
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-12:
            return np.inf
        a = a / s
        return obj_val(a)
    
    # We want to find t that minimizes eval_t(t)
    # Try t in [0, 1] first, then expand if needed
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = eval_t(c)
    fd = eval_t(d)

def local_refine(a, time_limit=30):

- If we move from f towards g (i.e., `f_new = (1-t)f + tg`), then:
  `conv(f_new, f_new)[k] = (1-t)^2 conv(f,f)[k] + 2(1-t)t conv(f,g)[k] + t^2 conv(g,g)[k]`
  
  We know `conv(f,f)[k] <= M` and `conv(f,g)[k] <= M`, but we don't control `conv(g,g)[k]`.
  
  So `conv(f_new, f_new)[k] <= (1-t)^2 M + 2(1-t)t M + t^2 max(conv(g,g))`
  `= M(1-t)(1-t+2t) + t^2 max(conv(g,g))`
  `= M(1-t)(1+t) + t^2 max(conv(g,g))`
  `= M(1-t^2) + t^2 max(conv(g,g))`

For this to be <= M, we need `max(conv(g,g)) <= M` as well. But the LP doesn't enforce this! It only enforces `conv(f, g) <= M`.

So the LP direction might not actually decrease the objective. The line search handles this: it finds the best t in [0,1] for which the objective actually decreases.

This is a valid approach, but it might be inefficient if the LP direction is often bad. A better LP might enforce `conv(g, g) <= M` as well, but that makes the LP non-linear (quadratic constraints).

An alternative: instead of the LP, use a **quatically constrained program** (QCP):
- maximize sum(g)
- subject to conv(g, g)[k] <= M for all k
- g >= 0

But this is non-convex (the constraint is not CVX since conv(g,g) is not a convex function of g in general).

For now, let me stick with the LP approach but add the smooth gradient method as a complementary direction.

Let me write the final code now. I'll aim for clarity and correctness.

One more consideration: the `evaluate_sequence` function clips values to [0, 1000] and rejects values with sum < 0.01. Since we normalize to sum=1, the sum constraint is fine. The clipping to 1000 is also fine since our values are O(1/n) = O(0.002).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Exponential gradient (mirror descent) with adaptive soft-max smoothing
2. LP-based direction finding for large structured moves
3. Subgradient peak reduction
4. Targeted local refinement
Multiple sequence lengths and diverse initializations are explored.
"""

import numpy as np
import scipy.optimize
import time
import math


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_val(a):
    """Objective: 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def softmax_weights(c, tau):
    """Compute softmax weights for smooth max approximation."""
    c_shifted = c - c.max()
    exp_c = np.exp(c_shifted / tau)
    s = exp_c.sum()
    if s < 1e-300:
        # All equal or underflow
        return np.ones_like(c) / len(c)
    return exp_c / s


def compute_smooth_grad(a, tau):
    """Compute gradient of tau*logsumexp(conv(a,a)/tau) w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    w = softmax_weights(conv, tau)
    grad = np.zeros(n)
    for i in range(n):
        # grad[i] = sum_k w[k] * a[k-i] for valid k-i in [0, n)
        j_lo = 0
        j_hi = min(n, 2 * n - 1 - i)
        if j_lo < j_hi:
            grad[i] = np.dot(w[i + j_lo:i + j_hi], a[j_lo:j_hi])
    return grad, conv, w


def exp_grad_optimize(a0, time_limit=60.0, lr_start=0.1, lr_end=0.005, tau_start=0.05, tau_end=0.002):
    """Exponential gradient (mirror descent) with adaptive learning rate and tau annealing."""
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones_like(a) / len(a)
    else:
        a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = obj_val(a)
    total = max(time_limit, 1.0)
    no_improve = 0
    max_no_improve = 300

    while time.monotonic() - t0 < total:
        frac = min(1.0, (time.monotonic() - t0) / total)
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** (0.5 * frac)

        conv = fast_convolve(a)
        w = softmax_weights(conv, tau)

        # Gradient
        n = len(a)
        grad = np.zeros(n)
        for i in range(n):
            j_hi = min(n, 2 * n - 1 - i)
            if j_hi > 0:
                grad[i] = np.dot(w[i:i + j_hi], a[:j_hi])

        # Exponential gradient update
        a_new = a * np.exp(-lr * grad)
        a_new = np.maximum(a_new, 1e-18)
        ns = a_new.sum()
        if ns < 1e-12:
            break
        a = a_new / ns

        # Check improvement periodically
        if no_improve % 50 == 0:
            obj = obj_val(a)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 50

        if no_improve >= max_no_improve:
            break

    # Final check
    obj = obj_val(a)
    if obj < best_obj:
        best_obj = obj
        best_a = a.copy()

    return best_a, best_obj


def solve_lp(f, M):
    """LP: maximize sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g or None."""
    n = len(f)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f[:n]
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None


def line_search(f, direction, max_iter=35):
    """Golden section line search for f + t*direction (with projection to a>=0 and normalization)."""
    def eval_t(t):
        if t <= 0:
            return obj_val(f)
        a = f + t * direction
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-12:
            return np.inf
        return obj_val(a / s)

    gr = (math.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = eval_t(c)
    fd = eval_t(d)
    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = eval_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = eval_t(d)
    t = (lo + hi) / 2.0
    return t, eval_t(t)


def lp_optimize(f0, time_limit=60.0):
    """Iterative LP-based optimization with line search."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-12:
        f = np.ones_like(f) / len(f)
    else:
        f = f / s

    best_f = f.copy()
    best_obj = obj_val(f)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        if M < 1e-15:
            break

        # Try LP with current M and tighter M
        directions = []
        g1 = solve_lp(f, M)
        if g1 is not None and not np.allclose(g1, f, atol=1e-10):
            directions.append(g1 - f)

        g2 = solve_lp(f, 0.9 * M)
        if g2 is not None and not np.allclose(g2, f, atol=1e-10):
            directions.append(g2 - f)

        if not directions:
            no_improve += 5
            if no_improve > 80:
                break
            continue

        best_t = 0.0
        best_o = best_obj
        best_d = None
        for d in directions:
            t, o = line_search(f, d)
            if o < best_o - 1e-11:
                best_o = o
                best_t = t
                best_d = d

        if best_d is not None:
            a_new = f + best_t * best_d
            a_new = np.maximum(a_new, 0)
            ns = a_new.sum()
            if ns < 1e-12:
                break
            a_new = a_new / ns
            f = a_new
            best_obj = best_o
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                break

    return best_f, best_obj


def local_refine(a, time_limit=30.0):
    """Targeted local search: move mass from peak contributors to safe positions."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s
    best_obj = obj_val(a)
    t0 = time.monotonic()
    stuck = 0

    while time.monotonic() - t0 < time_limit and stuck < 150:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))

        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        max_c = contrib.max()
        if max_c < 1e-15:
            break

        thr_high = 0.4 * max_c
        thr_low = 0.05 * max_c
        danger = np.where(contrib > thr_high)[0]
        safe = np.where(contrib < thr_low)[0]

        if len(danger) == 0 or len(safe) == 0:
            stuck += 1
            continue

        found = False
        for di in danger[:12]:
            if found:
                break
            for si in safe[:12]:
                if found:
                    break
                if di == si:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7]:
                    amount = a[di] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[di] -= amount
                    a_trial[si] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ts = a_trial.sum()
                    if ts < 1e-12:
                        continue
                    a_trial = a_trial / ts
                    o = obj_val(a_trial)
                    if o < best_obj - 1e-10:
                        best_obj = o
                        a = a_trial.copy()
                        found = True
                        break

        if found:
            stuck = 0
        else:
            stuck += 1

    return a, best_obj


def make_initializations(n, rng, parent=None):
    """Generate diverse initializations for length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    inits['uniform'] = np.ones(n) / n

    if parent is not None:
        if parent.size >= n:
            p = parent[:n].copy()
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
        p = np.maximum(p, 0)
        ps = p.sum()
        if ps > 1e-10:
            inits['parent'] = p / ps

    # U-shaped
    u = 1.0 + 3.0 * (2.0 * x / max(n - 1, 1) - 1.0) ** 2
    inits['u_shape'] = u / u.sum()

    # Hann
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(n - 1, 1)))
    hann = np.maximum(hann, 1e-6)
    inits['hann'] = hann / hann.sum()

    # Gaussian
    for sf in [0.3, 0.5, 0.7]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits['gauss_' + str(sf)] = g / g.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Edges
    m = max(1, n // 3)
    edge = np.zeros(n)
    edge[:m] = 1.0
    edge[-m:] = 1.0
    inits['edges'] = edge / edge.sum()

    # Two-bump
    b = np.zeros(n)
    b[n // 4 - n // 8:n // 4 + n // 8] = 1.0
    b[3 * n // 4 - n // 8:3 * n // 4 + n // 8] = 1.0
    if b.sum() > 0:
        inits['two_bump'] = b / b.sum()

    return inits
