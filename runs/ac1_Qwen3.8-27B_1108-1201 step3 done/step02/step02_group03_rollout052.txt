def lp_minmax(f, S_target=None):

    Solve LP: minimize t
    subject to: (f*g)[k] <= t for k=0..2n-2
                 sum(g) = S_target
                 g >= 0
    Returns (g, t) or (None, None).
    """
    n = len(f)
    if n > 1024:
        return None, None
    
    if S_target is None:
        S_target = 1.0

    # Objective: minimize t -> c_obj = [0,...,0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Constraints:
    # (f*g)[k] - t <= 0 for k = 0..2n-2
    # -sum(g) <= -S_target  and  sum(g) <= S_target (equality)
    
    # Build A_ub matrix
    n_constraints = (2 * n - 1) + 2
    A = np.zeros((n_constraints, n + 1))
    b = np.zeros(n_constraints)
    
    # Convolution constraint rows
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
        A[k, n] = -1.0  # coefficient of t
        b[k] = 0.0
    
    # Sum constraint: sum(g) = S_target
    A[2*n-1, 0:n] = 1.0
    b[2*n-1] = S_target
    A[2*n, 0:n] = -1.0
    b[2*n] = -S_target
    
    bounds = [(0.0, None)] * n + [(None, None)]  # g >= 0, t free

                      options={'presolve': True, 'time_limit': 60})

def lp_max_sum(f, M, S_target=None):
    """
    Solve LP: maximize sum(g)
    subject to: (f*g)[k] <= M for k=0..2n-2
                 g >= 0
    Returns g or None.
    """
    n = len(f)
    if n > 1024:
        return None
    
    c_obj = -np.ones(n)  # minimize -sum(g)
    
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def coordinate_descent(a, deadline, max_rounds=20):
    """
    Coordinate descent: for each coordinate i, find the optimal value of a[i]
    that minimizes the peak autocorrelation, keeping other coordinates fixed.
    """
    n = len(a)
    a = normalize(a.copy(), 1.0)
    best_score = compute_obj(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            orig = a[i]
            if orig < 1e-10:
                continue
            
            # Binary search or golden section for optimal a[i]
            # The peak as a function of a[i] is convex (it's the max of linear/quadratic functions)
            # So we can use ternary search or just try a few values
            
            # Try: 0, orig*0.25, orig*0.5, orig*0.75, orig*1.25, orig*1.5
            candidates = [0.0, orig * 0.25, orig * 0.5, orig * 0.75, orig * 1.5, orig * 2.0]
            
            best_val = orig
            best_s = best_score
            
            for val in candidates:
                a[i] = val
                a_temp = normalize(a.copy(), 1.0)
                s = compute_obj(a_temp)
                if s < best_s:
                    best_s = s
                    best_val = val
                a[i] = orig
            
            if best_val != orig:
                a[i] = best_val
                a = normalize(a, 1.0)
                best_score = best_s
                improved = True
        
        if not improved:
            break
    
    return a, best_score

"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses multi-phase optimization: softmax gradient descent with annealing,
LP-based block coordinate moves, and coordinate descent polishing.
Multi-start with diverse initializations and iterative refinement.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    max_c = float(c.max())
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=1.0):
    """Normalize to target sum."""
    S = a.sum()
    if S < 1e-15:
        return np.ones(len(a)) / len(a) * target
    return a * (target / S)


def autocorr_peak_and_grad(a):
    """Compute peak autocorrelation value and its gradient."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            grad[j] = 2.0 * a[idx]
    return float(c[k_star]), grad, k_star


def softmax_grad(a, tau):
    """Compute softmax-averaged gradient of max autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * (c_shifted - c_shifted.max()))
    w_sum = w.sum()
    if w_sum > 1e-300:
        w /= w_sum
    else:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    # grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    return grad


def grad_step(a, lr, tau=None, momentum_buf=None, momentum_decay=0.9):
    """One gradient descent step with projection onto sum=1 and a>=0."""
    n = len(a)
    if tau is None:
        _, grad, _ = autocorr_peak_and_grad(a)
    else:
        grad = softmax_grad(a, tau)
    grad -= grad.mean()
    if momentum_buf is not None and momentum_decay > 0:
        momentum_buf[:] = momentum_decay * momentum_buf - lr * grad
        step = -momentum_buf
    else:
        step = -lr * grad
    a_new = a + step
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, 1.0)
    return a_new, momentum_buf


def lp_max_sum(f, M):
    """LP: max sum(g) s.t. f*g <= M elementwise, g >= 0."""
    n = len(f)
    if n > 768:
        return None
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(j, j + n):
            A[m, j] = f[m - j]
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 45})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_min_t(f, S_target=1.0):
    """LP: min t s.t. (f*g)[k] <= t, sum(g)=S_target, g>=0."""
    n = len(f)
    if n > 640:
        return None, None
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    n_cons = (2 * n - 1) + 2
    A = np.zeros((n_cons, n + 1))
    b = np.zeros(n_cons)
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
        A[k, n] = -1.0
    A[2 * n - 1, 0:n] = 1.0
    b[2 * n - 1] = S_target
    A[2 * n, 0:n] = -1.0
    b[2 * n] = -S_target
    bounds = [(0.0, None)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 45})
        if res.status == 0:
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, None


def coord_descent(a, deadline, max_rounds=15):
    """Coordinate descent polishing."""
    n = len(a)
    a = normalize(a.copy(), 1.0)
    best_score = compute_obj(a)
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            orig = a[i]
            if orig < 1e-12:
                candidates = [0.001, 0.01, orig * 2.0]
            else:
                candidates = [0.0, orig * 0.3, orig * 0.6, orig * 1.4, orig * 2.0]
            best_val = orig
            best_s = best_score
            for val in candidates:
                a[i] = val
                a_tmp = normalize(a.copy(), 1.0)
                s = compute_obj(a_tmp)
                if s < best_s:
                    best_s = s
                    best_val = val
                a[i] = orig
            if best_val != orig:
                a[i] = best_val
                a = normalize(a, 1.0)
                best_score = best_s
                improved = True
        if not improved:
            break
    return a, best_score


def optimize_one(a0, deadline):
    """Full multi-phase optimization from a single starting point."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-10:
        a = np.ones(n) * 0.5
    a = normalize(a, 1.0)
    score = compute_obj(a)
    best = a.copy()
    best_score = score
    print(f"    n={n}, initial={score:.6f}")

    # Phase 1: Softmax gradient descent with annealing
    momentum = np.zeros(n)
    total_iters = 4000
    phase1_deadline = min(deadline - 90, time.monotonic() + max(20, (deadline - time.monotonic()) * 0.35))
    it = 0
    no_improve = 0
    while time.monotonic() < phase1_deadline and it < total_iters:
        it += 1
        progress = min(1.0, it / total_iters)
        tau = 5.0 + 80.0 * progress
        lr = 0.4 * (0.9994 ** it)
        lr = max(lr, 0.01)
        a, momentum = grad_step(a, lr, tau=tau, momentum_buf=momentum, momentum_decay=0.9)
        if it % 80 == 0:
            s = compute_obj(a)
            if s < best_score - 1e-10:
                best_score = s
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 12:
                    break
    print(f"    P1(soft GD): {best_score:.6f} it={it}")

    # Phase 2: Hard max gradient descent
    if time.monotonic() < deadline - 60:
        a = best.copy()
        momentum = np.zeros(n)
        total_iters = 6000
        phase2_deadline = min(deadline - 60, time.monotonic() + max(15, (deadline - time.monotonic()) * 0.2))
        it = 0
        no_improve = 0
        while time.monotonic() < phase2_deadline and it < total_iters:
            it += 1
            lr = 0.25 * (0.9998 ** it)
            lr = max(lr, 0.005)
            a, momentum = grad_step(a, lr, tau=None, momentum_buf=momentum, momentum_decay=0.92)
            if it % 150 == 0:
                s = compute_obj(a)
                if s < best_score - 1e-10:
                    best_score = s
                    best = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                    if no_improve > 15:
                        break
        print(f"    P2(hard GD): {best_score:.6f} it={it}")

    # Phase 3: LP refinement
    if n <= 640 and time.monotonic() < deadline - 30:
        a = best.copy()
        phase3_deadline = min(deadline - 30, time.monotonic() + max(10, (deadline - time.monotonic()) * 0.2))
        it_lp = 0
        no_imp = 0
        while time.monotonic() < phase3_deadline and it_lp < 40:
            it_lp += 1
            c = np.convolve(a, a)
            M = float(c.max())
            g = lp_max_sum(a, M)
            if g is None or g.sum() < 1e-10:
                break
            g_norm = normalize(g, 1.0)
            improved = False
            for t_blend in [0.4, 0.25, 0.15, 0.08, 0.04]:
                a_new = (1.0 - t_blend) * a + t_blend * g_norm
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize(a_new, 1.0)
                s_new = compute_obj(a_new)
                if s_new < best_score - 1e-11:
                    a = a_new
                    best = a.copy()
                    best_score = s_new
                    improved = True
                    break
            if improved:
                no_imp = 0
            else:
                no_imp += 1
                if no_imp >= 3:
                    mom_tmp = np.zeros(n)
                    a, mom_tmp = grad_step(a, 0.08, tau=10.0, momentum_buf=mom_tmp, momentum_decay=0.5)
                    s = compute_obj(a)
                    if s < best_score:
                        best = a.copy()
                        best_score = s
                        no_imp = 0
                    if no_imp > 7:
                        break
        print(f"    P3(LP): {best_score:.6f} it={it_lp}")

    # Phase 4: Alternate LP and gradient
    if n <= 512 and time.monotonic() < deadline - 20:
        a = best.copy()
        phase4_deadline = min(deadline - 20, time.monotonic() + max(5, (deadline - time.monotonic()) * 0.15))
        it_alt = 0
        no_imp2 = 0
        while time.monotonic() < phase4_deadline and it_alt < 30:
            it_alt += 1
            # Gradient step
            mom_tmp = np.zeros(n)
            a, mom_tmp = grad_step(a, 0.1, tau=20.0, momentum_buf=mom_tmp, momentum_decay=0.7)
            s = compute_obj(a)
            if s < best_score:
                best = a.copy()
                best_score = s
                no_imp2 = 0
            else:
                no_imp2 += 1
            # LP step
            c = np.convolve(a, a)
            M = float(c.max())
            g = lp_max_sum(a, M)
            if g is not None and g.sum() > 1e-10:
                g_norm = normalize(g, 1.0)
                a2 = 0.7 * a + 0.3 * g_norm
                a2 = np.maximum(a2, 0.0)
                a2 = normalize(a2, 1.0)
                s2 = compute_obj(a2)
                if s2 < best_score:
                    a = a2
                    best = a.copy()
                    best_score = s2
                    no_imp2 = 0
            if no_imp2 > 5:
                break
        print(f"    P4(alt): {best_score:.6f} it={it_alt}")

    # Phase 5: Coordinate descent
    if n <= 400 and time.monotonic() < deadline - 10:
        a_cd, s_cd = coord_descent(best, deadline - 5, max_rounds=8)
        if s_cd < best_score:
            best = a_cd
            best_score = s_cd
        print(f"    P5(coord): {best_score:.6f}")

    return best, best_score


def gen_starts(rng, n):
    """Generate diverse starting sequences of length n."""
    i = np.arange(n, dtype=np.float64)
    starts = []
    starts.append(rng.random(n) * 0.5 + 0.25)
    starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i * 2.0 / n))
    starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i * 3.0 / n))
    starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i * 7.0 / n))
    starts.append(np.abs(np.sin(2 * np.pi * i * 2.0 / n)) + 0.05)
    starts.append(np.abs(np.sin(2 * np.pi * i * 5.0 / n)) * 0.8 + 0.1)
    # Gaussian bump
    starts.append(np.exp(-0.5 * ((i - n * 0.5) / (n * 0.15)) ** 2))
    # Two bumps
    starts.append(np.exp(-0.5 * ((i - n * 0.25) / (n * 0.1)) ** 2) + np.exp(-0.5 * ((i - n * 0.75) / (n * 0.1)) ** 2))
    # Mixed frequency
    starts.append(0.3 + 0.3 * np.sin(2 * np.pi * i * 4.0 / n) + 0.2 * np.cos(2 * np.pi * i * 9.0 / n) + 0.1)
    # Random steps
    steps = n // 8
    a_steps = np.repeat(rng.random(steps) * 0.8 + 0.2, n // steps)
    if len(a_steps) < n:
        a_steps = np.concatenate([a_steps, rng.random(n - len(a_steps)) * 0.5 + 0.25])
    starts.append(a_steps[:n])
    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-start, multi-phase optimization.
    Combines softmax GD, hard GD, LP refinement, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 12.0)

    best_seq = None
    best_score = float('inf')

    # Load previous best
    prev_seq = None
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 64 and np.all(np.isfinite(gs)) and np.all(gs >= 0) and gs.sum() > 0.01:
            prev_seq = gs.copy()
            prev_score = compute_obj(prev_seq)
            print(f"Loaded prev: n={len(prev_seq)}, score={prev_score:.6f}")
    except (NameError, Exception) as e:
        print(f"No previous sequence: {e}")

    # Build candidate list
    candidates = []
    if prev_seq is not None:
        candidates.append(prev_seq)
        # Perturbations of previous best
        for scale in [0.05, 0.1, 0.2]:
            pert = prev_seq + rng.normal(0, scale, len(prev_seq))
            candidates.append(np.maximum(pert, 0))

    # Structured starts at key lengths
    for n in [256, 384, 512]:
        starts_n = gen_starts(rng, n)
        candidates.extend(starts_n)

    # Deduplicate
    unique = []
    seen_keys = set()
    for c in candidates:
        c = np.asarray(c, dtype=np.float64).flatten()
        if c.size < 32:
            continue
        key = (c.size, round(float(c.mean()), 4))
        if key not in seen_keys:
            seen_keys.add(key)
            unique.append(c)
    candidates = unique

    # Sort: previous best first, then shorter lengths
    if prev_seq is not None:
        candidates.sort(key=lambda x: (0 if np.array_equal(x, prev_seq) else 1, len(x)))
    else:
        candidates.sort(key=lambda x: len(x))

    print(f"Candidates: {len(candidates)}")

    # Process candidates
    for idx, a0 in enumerate(candidates):
        if time.monotonic() > deadline - 45:
            break
        remaining = deadline - time.monotonic()

        # Time allocation
        if idx == 0 and prev_seq is not None:
            t_alloc = min(500.0, remaining * 0.35)
        elif idx < 8:
            t_alloc = min(180.0, remaining * 0.12)
        elif idx < 20:
            t_alloc = min(100.0, remaining * 0.08)
        else:
            t_alloc = min(60.0, remaining * 0.05)

        t_alloc = max(25.0, t_alloc)
        if time.monotonic() + t_alloc > deadline - 25:
            t_alloc = deadline - time.monotonic() - 25
            if t_alloc < 15:
                break

        cand_dl = time.monotonic() + t_alloc
        seq, score = optimize_one(a0, cand_dl)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  [{idx}] NEW BEST: {best_score:.6f} n={len(seq)}")
        else:
            print(f"  [{idx}] {score:.6f} n={len(seq)}")

    # Second pass: refine best with more perturbations
    if best_seq is not None and time.monotonic() < deadline - 100:
        best_n = len(best_seq)
        print(f"\nPass 2: n={best_n}, best={best_score:.6f}")
        for trial in range(6):
            if time.monotonic() > deadline - 50:
                break
            # Various perturbation strategies
            if trial % 3 == 0:
                a0 = np.maximum(best_seq + rng.normal(0, 0.08, best_n), 0)
            elif trial % 3 == 1:
                a0 = np.maximum(best_seq * (1.0 + rng.normal(0, 0.15, best_n)), 0)
            else:
                # Randomly zero out and refill some elements
                a0 = best_seq.copy()
                n_zero = best_n // 8
                zero_idx = rng.choice(best_n, size=n_zero, replace=False)
                a0[zero_idx] = 0
                a0 = np.maximum(a0 + rng.random(best_n) * 0.1, 0)

            remaining = deadline - time.monotonic() - 30
            t_alloc = min(250.0, remaining / max(1, 6 - trial))
            if t_alloc < 25:
                break
            cand_dl = time.monotonic() + t_alloc
            seq, score = optimize_one(a0, cand_dl)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  [p2-{trial}] NEW BEST: {best_score:.6f}")
            else:
                print(f"  [p2-{trial}] {score:.6f}")

    # Try a different length if time permits
    if best_seq is not None and time.monotonic() < deadline - 120:
        best_n = len(best_seq)
        alt_lengths = [384, 512, 256]
        alt_lengths = [L for L in alt_lengths if L != best_n]
        for alt_n in alt_lengths[:2]:
            if time.monotonic() > deadline - 60:
                break
            print(f"\nTrying n={alt_n}")
            a0 = rng.random(alt_n) * 0.5 + 0.25
            remaining = deadline - time.monotonic() - 30
            t_alloc = min(200.0, remaining * 0.5)
            if t_alloc < 30:
                break
            cand_dl = time.monotonic() + t_alloc
            seq, score = optimize_one(a0, cand_dl)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  [alt n={alt_n}] NEW BEST: {best_score:.6f}")
            else:
                print(f"  [alt n={alt_n}] {score:.6f}")

    # Final polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"\nFinal polish: {best_score:.6f}")
        remaining = deadline - time.monotonic() - 8
        seq, score = optimize_one(best_seq, time.monotonic() + remaining)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  Polished: {best_score:.6f}")

    # Validate and return
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\n=== FINAL: n={len(best_seq)}, score={final_score:.6f} ===")
        try:
            ev = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence: {ev:.6f}")
        except Exception as e:
            print(f"evaluate error: {e}")
        # Ensure all values are valid
        result = [max(0.0, min(1000.0, float(x))) for x in best_seq]
        if len(result) > 0 and sum(result) > 0.01:
            return result

    # Fallback
    print("Fallback: uniform 256")
    return [1.0] * 256
