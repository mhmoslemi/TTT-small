"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses multi-phase optimization: mirror-descent soft-max gradient, projected gradient,
and coordinate descent polishing. Explores multiple lengths with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize to target sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def compute_softmax_gradient(a, tau):
    """Compute soft-max smoothed gradient of max autocorrelation w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # grad[i] = 2 * sum_j w[i+j] * a[j]
    # = 2 * (w flipped, then valid cross-correlation with a)
    # Using: convolve(w, a[::-1])[n-1+i] = sum_j w[n-1+i-j] * a[::-1][j]
    #        = sum_j w[n-1+i-j] * a[n-1-j]
    #        Let k = n-1-j, j = n-1-k:
    #        = sum_k w[n-1+i-(n-1-k)] * a[k]
    #        = sum_k w[i+k] * a[k]
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n-1:n-1+n]
    
    return grad, c


def mirror_descent_step(a, grad, lr, S_t):
    """Mirror descent (exponential) step with sum constraint."""
    # Multiplicative update
    log_a = np.log(np.maximum(a, 1e-300))
    log_a = log_a - lr * grad
    # Renormalize
    log_a = log_a - np.max(log_a)  # numerical stability
    a_new = np.exp(log_a)
    a_new = a_new * (S_t / a_new.sum())
    return a_new


def projected_grad_step(a, grad, lr, momentum, mu, S_t):
    """Projected gradient step with momentum."""
    grad_centered = grad - grad.mean()
    momentum = mu * momentum - lr * grad_centered
    a_new = a + momentum
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, S_t)
    return a_new, momentum


def coordinate_polish(a, S_t, deadline, max_rounds=10):
    """Coordinate descent polishing."""
    n = len(a)
    a = normalize(a.copy(), S_t)
    best_score = compute_obj(a)
    best = a.copy()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            orig = a[i]
            # Generate candidates
            candidates = [0.0, orig * 0.5, orig * 1.5, orig * 0.2, orig * 2.0]
            # Add peak-aware candidates
            c = np.convolve(a, a)
            k_star = int(np.argmax(c))
            j = k_star - i
            if 0 <= j < n:
                # If a[j] is large at the peak, reducing a[i] helps
                if a[j] > 0.1 * a.mean():
                    candidates.append(orig * 0.3)
            
            best_val = orig
            best_s = best_score
            for val in candidates:
                val = max(0.0, val)
                a[i] = val
                a_norm = normalize(a, S_t)
                s = compute_obj(a_norm)
                if s < best_s - 1e-10:
                    best_s = s
                    best_val = val
                    best = a_norm.copy()
            a[i] = orig if best_val == orig else best_val
            if best_val != orig:
                a = normalize(a, S_t)
                best_score = best_s
                improved = True
        
        if not improved:
            break
    
    return best, best_score

    n = len(a)

"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Multi-phase approach: mirror-descent + projected gradient with soft-max annealing,
LP-based refinement, and coordinate descent polishing. Explores multiple lengths
with diverse initializations, starting from known good sequence.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize to target sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def compute_softmax_gradient(a, tau):
    """
    Compute soft-max smoothed gradient of max autocorrelation w.r.t. a.
    Returns (grad, c) where c is the full autocorrelation.
    """
    n = len(a)
    c = np.convolve(a, a)

    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum

    # grad[i] = 2 * sum_j w[i+j] * a[j]
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n - 1:n - 1 + n]

    return grad, c


def mirror_step(a, grad, lr, S_t):
    """Mirror descent (multiplicative/exponential) step."""
    log_a = np.log(np.maximum(a, 1e-30))
    log_a = log_a - lr * grad
    log_a = log_a - np.max(log_a)
    a_new = np.exp(log_a)
    S_new = a_new.sum()
    if S_new < 1e-12:
        return a.copy()
    return a_new * (S_t / S_new)


def proj_grad_step(a, grad, lr, momentum, mu, S_t):
    """Projected gradient step with momentum (maintains sum and non-negativity)."""
    grad_c = grad - grad.mean()
    momentum = mu * momentum - lr * grad_c
    a_new = a + momentum
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, S_t)
    return a_new, momentum


def coord_polish(a, S_t, deadline, max_rounds=8):
    """Coordinate descent polishing with multiple candidate values."""
    n = len(a)
    a = normalize(a.copy(), S_t)
    best_score = compute_obj(a)
    best = a.copy()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        c = np.convolve(a, a)
        k_star = int(np.argmax(c))

        for i in range(n):
            if time.monotonic() > deadline:
                break
            orig = a[i]
            candidates = [0.0, orig * 0.3, orig * 0.6, orig * 1.0, orig * 1.4, orig * 2.0]

            # Peak-aware: if a[k_star - i] is large, reducing a[i] helps
            j = k_star - i
            if 0 <= j < n and a[j] > 0.5 * a.mean():
                candidates.append(orig * 0.1)

            best_val = orig
            best_s = best_score
            for val in candidates:
                val = max(0.0, val)
                if val == orig:
                    continue
                a[i] = val
                a_norm = normalize(a, S_t)
                s = compute_obj(a_norm)
                if s < best_s - 1e-10:
                    best_s = s
                    best_val = val
                    best = a_norm.copy()
            a[i] = best_val
            if best_val != orig:
                a = normalize(a, S_t)
                best_score = best_s
                improved = True

        if not improved:
            break

    return best, best_score


def lp_refine(a, S_t, deadline, max_iters=15):
    """LP-based refinement: find g maximizing sum(g) s.t. conv(a,g)<=M, g>=0."""
    n = len(a)
    if n > 600:
        return a, compute_obj(a)

    c = np.convolve(a, a)
    M = float(c.max())
    best_score = compute_obj(a)
    best = a.copy()

    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Build constraint matrix A[m,j] = a[m-j] if 0<=m-j<n
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            lo = j
            hi = min(j + n, 2 * n - 1)
            A[lo:hi, j] = a[:hi - lo]

        b = np.full(2 * n - 1, M)
        c_obj = -np.ones(n)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 30})
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break

        g_sum = g.sum()
        if g_sum < 1e-6:
            break
        g = g * (S_t / g_sum)

        improved = False
        for t_blend in [0.7, 0.5, 0.35, 0.2, 0.1]:
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, S_t)
            s_new = compute_obj(a_new)
            if s_new < best_score - 1e-10:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
            # Small nudge to escape
            grad, _ = compute_softmax_gradient(a, 20.0)
            a, _ = proj_grad_step(a, grad, 0.03, np.zeros(n), 0.0, S_t)
            s = compute_obj(a)
            if s < best_score:
                best = a.copy()
                best_score = s
                no_improve = 0
        else:
            no_improve = 0
            c = np.convolve(a, a)
            M = float(c.max())

    return best, best_score


def optimize_single(a0, deadline, print_prefix=""):

    n = len(a0)

                    no_improve = 0

    # Phase 3: LP refinement

def gen_starts(rng, n):
    """Generate diverse starting sequences of length n."""
    starts = []
    i = np.arange(n, dtype=np.float64)

    # Random
    starts.append(rng.random(n) + 0.1)
    starts.append(rng.random(n) * 0.8 + 0.1)

    # Cosine modulations (different frequencies)
    for freq in [3, 5, 7, 11, 13]:
        starts.append(0.4 + 0.4 * np.cos(2 * np.pi * i * float(freq) / n))

    # Multi-frequency
    starts.append(0.4 + 0.25 * np.sin(2 * np.pi * i * 3.0 / n) +
                  0.2 * np.cos(2 * np.pi * i * 7.0 / n + 0.5))
    starts.append(0.4 + 0.2 * np.sin(2 * np.pi * i * 5.0 / n) +
                  0.15 * np.cos(2 * np.pi * i * 17.0 / n + 1.0) +
                  0.1 * np.sin(2 * np.pi * i * 23.0 / n))

    # Gaussian bumps
    starts.append(np.exp(-0.5 * ((i - n * 0.5) / (n * 0.15)) ** 2) + 0.05)
    starts.append(np.exp(-0.5 * ((i - n * 0.25) / (n * 0.08)) ** 2) +
                  np.exp(-0.5 * ((i - n * 0.75) / (n * 0.08)) ** 2) + 0.03)
    starts.append(np.exp(-0.5 * ((i - n * 0.33) / (n * 0.06)) ** 2) +
                  np.exp(-0.5 * ((i - n * 0.67) / (n * 0.06)) ** 2) + 0.03)

    # Sawtooth / ramp
    starts.append(np.abs(i / n - 0.5) * 2.0 + 0.1)
    starts.append((i / n) ** 0.5 + 0.1)

    # Sparse
    a = np.zeros(n)
    a[::3] = rng.random(n // 3 + 1) * 0.6 + 0.2
    starts.append(a)

    # Log-periodic
    starts.append(0.3 + 0.3 * np.abs(np.sin(2 * np.pi * np.log(i + 1) * 2.0)))

    return starts

    except NameError:
        print("No height_sequence_1 available, using random starts only.")

    # --- Try multiple lengths with diverse starts ---
    lengths_to_try = [384, 320, 448, 512]

    for n in lengths_to_try:
        if time.monotonic() > deadline - 80:
            break

        # Skip if we already tried this length from the previous best
        if best_seq is not None and n == len(best_seq) and best_score < 1.52:
            # Still try but with less time
            pass

        starts = gen_starts(rng, n)
        # Limit number of starts based on remaining time
        remaining = deadline - time.monotonic() - 40
        max_starts = max(2, min(len(starts), int(remaining / 120)))
        starts = starts[:max_starts]

        print(f"\n--- Length n={n}, {len(starts)} starts, {remaining:.0f}s remaining ---")

        for idx, a0 in enumerate(starts):
            if time.monotonic() > deadline - 40:
                break

            t_alloc = min(180.0, deadline - time.monotonic() - 25)
            if t_alloc < 30:
                break

            seq, score = optimize_single(a0, deadline, print_prefix=f"  [n={n},s{idx}] ")
            if score < best_score - 1e-9:
                best_score = score
                best_seq = seq.copy()
                print(f"  >>> NEW BEST: {best_score:.6f} (n={n}, start={idx})")

    # --- Final focused optimization from the best found ---
    if best_seq is not None and time.monotonic() < deadline - 40:
        print(f"\n--- Final polish from best (score={best_score:.6f}, n={len(best_seq)}) ---")
        n_best = len(best_seq)

        # Try a few perturbed versions
        for trial in range(3):
            if time.monotonic() > deadline - 25:
                break
            pert = 0.02 + 0.01 * trial
            a0 = best_seq + pert * rng.standard_normal(n_best)
            a0 = np.maximum(a0, 0.0)
            seq, score = optimize_single(a0, deadline, print_prefix=f"  [final{trial}] ")
            if score < best_score - 1e-9:
                best_score = score
                best_seq = seq.copy()
                print(f"  >>> NEW BEST: {best_score:.6f}")

        # One more direct optimization from the best
        if time.monotonic() < deadline - 30:
            seq, score = optimize_single(best_seq, deadline, print_prefix="  [direct] ")
            if score < best_score:
                best_score = score
                best_seq = seq.copy()

    # --- Final validation ---
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\n=== Final: n={len(best_seq)}, score={final_score:.6f} ===")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Fallback to uniform sequence n=384")
    return [1.0] * 384
