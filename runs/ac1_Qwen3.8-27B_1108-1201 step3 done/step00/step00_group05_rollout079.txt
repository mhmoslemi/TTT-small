"""
Step-function search for minimizing
    score = 2 * n * max(convolve(a, a)) / sum(a)**2
over non-negative real sequences a of variable length.

The program uses several complementary strategies:
1. Fast FFT-based smooth maximization of the self-convolution coefficients
   in the log-softmax parameterization.
2. L-BFGS-B refinement of the same smooth surrogate when scipy is available.
3. Exact active-lag subgradient / multiplicative updates.
4. A Hankel/dual linear-solve heuristic for smaller/medium lengths.
5. A local pairwise-transfer search for short sequences.
6. Multiple length choices and structured/random starts to avoid local minima.

The function propose_candidate() returns the best sequence found before the
time budget expires.
"""

import time
import math

    import scipy.optimize as sopt

SMOOTH_DATA = {"eta": 1e-4}

def pow2_len(m):
    """Smallest power of two >= m, for FFT padding."""
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_positive(w):
    """Make a real array non-negative and sum to one."""
    w = np.asarray(w, dtype=np.float64)
    w = np.where(np.isfinite(w), w, 0.0)
    w = np.maximum(w, 0.0)
    s = float(np.sum(w))
    if s <= 1e-300 or not np.isfinite(s):
        n = w.size
        if n == 0:
            return np.ones(1, dtype=np.float64)
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / s


def softmax_vector(z):
    """Stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return np.zeros(0, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 1e-300 or not np.isfinite(s):
        return np.full(z.size, 1.0 / z.size, dtype=np.float64)
    return e / s


def log_from_prob(p):
    """Log-parameterization for a positive probability vector."""
    p = np.asarray(p, dtype=np.float64)
    p = np.clip(p, 1e-300, None)
    z = np.log(p)
    if z.size > 0:
        z = z - float(np.mean(z))
    return z


# -----------------------------------------------------------------------------
# Convolution / gradients
# -----------------------------------------------------------------------------

def convolution_self(p):
    """Full self-convolution of a non-negative vector."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if n <= 128:
        c = np.convolve(p, p)
        return np.maximum(c, 0.0)

    L = pow2_len(2 * n - 1)
    f = np.fft.rfft(p, L)
    c = np.fft.irfft(f * f, L)
    c = c[:2 * n - 1]
    c[np.abs(c) < 1e-15] = 0.0
    return np.maximum(c, 0.0)


def score_fft(p):
    """Score for a normalized probability vector p (sum p = 1)."""
    c = convolution_self(p)
    if c.size == 0:
        return float("inf")
    return float(2.0 * p.size * np.max(c))


def grad_A_fft(p, w, n):
    """
    Compute A_j = sum_l p_l * w_{l+j}, for j=0..n-1.
    This is the first-order interaction needed for gradients of
    sum_k w_k * c_k, where c = p * p.
    """
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    if n == 1:
        return np.array([float(p[0] * w[0]) if w.size > 0 else 0.0])

    if n <= 128:
        out = np.empty(n, dtype=np.float64)
        for j in range(n):
            out[j] = float(np.dot(p, w[j:j + n]))
        return out

    # FFT: A_j = conv(p, reverse(w))[2n-2-j]
    L = pow2_len(3 * n - 2)
    fp = np.fft.rfft(p, L)
    wr = w[::-1]
    conv = np.fft.irfft(fp * np.fft.rfft(wr, L), L)
    return conv[2 * n - 2:n - 2:-1].astype(np.float64, copy=True)


def soft_score_and_weights(c, eta, n):
    """
    Smooth max surrogate:
        L = 2*n*(max c + eta * log(sum exp((c-max)/eta)))
    Returns L and implicit softmax weights over lags.
    """
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return float("inf"), np.zeros(0, dtype=np.float64)
    if eta <= 1e-300 or not np.isfinite(eta):
        eta = 1e-12

    m = float(np.max(c))
    vals = (c - m) / eta
    vals = np.where(np.isfinite(vals), vals, -700.0)
    vals = np.clip(vals, -700.0, 0.0)
    expv = np.exp(vals)
    S = float(np.sum(expv))
    if S <= 1e-300 or not np.isfinite(S):
        S = 1.0
    L = float(2.0 * n * (m + eta * math.log(S)))
    w = expv / S
    return L, w


def grad_z_from_weights(p, w, n):
    """
    Gradient wrt log-parameters z of the smooth score proxy.
    If w_k are the implicit softmax weights over convolution lags, then
    d(score)/dz_j = p_j * (g_j - sum_i p_i g_i),
    where g_j = d(score)/dp_j.
    """
    A = grad_A_fft(p, w, n)
    g_p = 4.0 * float(n) * A
    dot = float(np.dot(p, g_p))
    g = p * (g_p - dot)
    g = np.where(np.isfinite(g), g, 0.0)
    return g


def active_weights(c, maxc, eps, power):
    """Weights focusing on lags whose convolution mass is near the peak."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return c.copy()
    thr = maxc * (1.0 - eps)
    excess = np.maximum(c - thr, 0.0)
    if power != 1.0:
        excess = np.maximum(excess, 1e-300)
        excess = np.power(excess, power)
    s = float(np.sum(excess))
    if s <= 1e-300:
        hard = (c >= thr).astype(np.float64)
        s = float(np.sum(hard))
        if s <= 0:
            return np.full(c.size, 1.0 / c.size, dtype=np.float64)
        return hard / s
    return excess / s

# Global smooth objective/gradient wrappers for scipy
# -----------------------------------------------------------------------------

def _smooth_eval(z, eta=None, return_grad=False):
    z = np.asarray(z, dtype=np.float64)
    n = z.size
    if n == 0:
        val = (float("inf"), None)
        return val if return_grad else val[0]

    if eta is None:
        eta_use = float(SMOOTH_DATA.get("eta", 1e-4))
    else:
        eta_use = float(eta)

    cached = SMOOTH_CACHE
    if (
        cached.get("n", -1) == n
        and abs(cached.get("eta", -1.0) - eta_use) < 1e-18
        and np.array_equal(cached.get("z"), z)
    ):
        f = cached.get("f")
        g = cached.get("g")
        return (f, g) if return_grad else f

    p = softmax_vector(z)
    c = convolution_self(p)
    f, w = soft_score_and_weights(c, eta_use, n)
    g = grad_z_from_weights(p, w, n)

    SMOOTH_CACHE["z"] = z.copy()
    SMOOTH_CACHE["eta"] = eta_use
    SMOOTH_CACHE["n"] = n
    SMOOTH_CACHE["f"] = f
    SMOOTH_CACHE["g"] = g

    return (f, g) if return_grad else f


def smooth_objective(z):
    return _smooth_eval(z, eta=None, return_grad=False)


def smooth_gradient(z):
    val = _smooth_eval(z, eta=None, return_grad=True)
    return val[1]

def run_adam_smooth(z, n, time_limit, rng):
    """Adam on the log-softmax variables using a smooth max surrogate."""
    start = time.monotonic()
    limit = start + max(0.0, time_limit)

    z = np.asarray(z, dtype=np.float64).copy()
    p = softmax_vector(z)
    c = convolution_self(p)
    best_score = float(2.0 * n * np.max(c)) if c.size else float("inf")
    best_z = z.copy()
    best_p = p.copy()

    if n <= 1 or time_limit <= 0.1:
        return best_z, best_p, best_score

    if n > 8000:
        lr = 0.005
    elif n > 2000:
        lr = 0.008
    elif n > 500:
        lr = 0.015
    else:
        lr = 0.03

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    b1 = 0.90
    b2 = 0.999
    eps = 1e-8
    step = 0
    stall = 0

    while time.monotonic() < limit:
        p = softmax_vector(z)
        c = convolution_self(p)
        cmax = float(np.max(c))
        score = float(2.0 * n * cmax)

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_z = z.copy()
            best_p = p.copy()
            stall = 0
        else:
            stall += 1

        if stall > 900:
            z = best_z + rng.normal(0.0, 0.05, size=n)
            z = z - float(np.mean(z))
            z = np.clip(z, -50.0, 50.0)
            m.fill(0.0)
            v.fill(0.0)
            stall = 0
            lr = max(0.001, lr * 0.7)

        tfrac = min(1.0, step / 4000.0)
        eta_ratio = 0.25 + (0.03 - 0.25) * tfrac
        eta = max(1e-12, eta_ratio * cmax if cmax > 0 else 1e-8)

        _, w = soft_score_and_weights(c, eta, n)
        g = grad_z_from_weights(p, w, n)

        if not np.all(np.isfinite(g)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            continue

        step += 1
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * (g * g)
        mhat = m / (1.0 - b1 ** step)
        vhat = v / (1.0 - b2 ** step)
        z = z - lr * mhat / (np.sqrt(vhat) + eps)
        z = z - float(np.mean(z))
        z = np.clip(z, -50.0, 50.0)

    return best_z, best_p, best_score


def run_lbfgs(z, n, time_limit, eta_ratio=0.04):
    """L-BFGS-B refinement on the smooth surrogate, if scipy is present."""
    z = np.asarray(z, dtype=np.float64).copy()
    if n <= 1 or time_limit <= 1.0 or not HAVE_SCIPY:
        return z

    p = softmax_vector(z)
    c = convolution_self(p)
    cmax = float(np.max(c))
    eta = max(1e-12, eta_ratio * cmax if cmax > 0 else 1e-8)

    SMOOTH_DATA["eta"] = eta
    SMOOTH_CACHE.clear()

    if n > 8000:
        maxfun = 350
    elif n > 4000:
        maxfun = 700
    elif n > 1000:
        maxfun = 1200
    elif n > 300:
        maxfun = 2200
    else:
        maxfun = 4000

    per_eval = 0.012
    maxfun_by_time = int(max(60, min(6000, (max(0.0, time_limit - 2.0)) / per_eval)))
    maxfun = int(max(60, min(maxfun, maxfun_by_time)))

    try:
        res = sopt.minimize(
            smooth_objective,
            z,
            jac=smooth_gradient,
            method="L-BFGS-B",
            options={
                "maxiter": maxfun,
                "maxfun": maxfun,
                "ftol": 1e-12,
                "gtol": 1e-8,
                "maxls": 10,
                "disp": False,
            },
        )
        if res is not None and hasattr(res, "x") and res.x is not None:
            cand = np.asarray(res.x, dtype=np.float64)
            if cand.size == n and np.all(np.isfinite(cand)) and np.isfinite(res.fun):
                return cand
    except Exception:
        pass

    return z


def run_hard_active(z, n, time_limit, rng, eps=0.02, power=1.0):
    """Adam-style exact active-lag subgradient updates."""
    start = time.monotonic()
    limit = start + max(0.0, time_limit)

    z = np.asarray(z, dtype=np.float64).copy()
    p = softmax_vector(z)
    c = convolution_self(p)
    best_score = float(2.0 * n * np.max(c)) if c.size else float("inf")
    best_z = z.copy()
    best_p = p.copy()

    if n <= 1 or time_limit <= 0.1:
        return best_z, best_p

    if n > 5000:
        lr = 0.003
    elif n > 1000:
        lr = 0.005
    elif n > 200:
        lr = 0.010
    else:
        lr = 0.020

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    b1 = 0.90
    b2 = 0.999
    eps_adam = 1e-8
    step = 0
    stall = 0

    while time.monotonic() < limit:
        p = softmax_vector(z)
        c = convolution_self(p)
        cmax = float(np.max(c))
        score = float(2.0 * n * cmax)

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_z = z.copy()
            best_p = p.copy()
            stall = 0
        else:
            stall += 1

        if stall > 600:
            z = best_z + rng.normal(0.0, 0.04, size=n)
            z = z - float(np.mean(z))
            z = np.clip(z, -50.0, 50.0)
            m.fill(0.0)
            v.fill(0.0)
            stall = 0
            lr = max(0.0005, lr * 0.7)

        w = active_weights(c, cmax, eps, power)
        g = grad_z_from_weights(p, w, n)
        if not np.all(np.isfinite(g)):
            z = best_z.copy()
            m.fill(0.0)
            v.fill(0.0)
            continue

        step += 1
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * (g * g)
        mhat = m / (1.0 - b1 ** step)
        vhat = v / (1.0 - b2 ** step)
        z = z - lr * mhat / (np.sqrt(vhat) + eps_adam)
        z = z - float(np.mean(z))
        z = np.clip(z, -50.0, 50.0)

    return best_z, best_p


def run_multiplicative(p0, n, time_limit, rng, eps=0.05, alpha=None):
    """
    Multiplicative update p <- p / B(alpha), where B is the active-lag
    first-order penalty. Useful as a basin-hopping heuristic.
    """
    start = time.monotonic()
    limit = start + max(0.0, time_limit)

    p = normalize_positive(p0)
    c = convolution_self(p)
    best_score = float(2.0 * n * np.max(c)) if c.size else float("inf")
    best_p = p.copy()

    if n <= 1 or time_limit <= 0.3:
        return best_p

    if alpha is None:
        alpha = 0.12 if n > 2000 else 0.18

    noimp = 0
    step = 0
    while time.monotonic() < limit:
        p = best_p + 0.0  # ensure local
        c = convolution_self(p)
        cmax = float(np.max(c))
        score = float(2.0 * n * cmax)
        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            noimp = 0
        else:
            noimp += 1

        if noimp > 60:
            p = 0.85 * best_p + 0.15 * np.abs(rng.normal(0.0, 1.0, size=n))
            p = normalize_positive(p)
            noimp = 0
            step = 0
            continue

        w = active_weights(c, cmax, eps, 1.0)
        A = grad_A_fft(p, w, n)
        B = 2.0 * A
        Bmax = float(np.max(B))
        floor = max(1e-14, 1e-5 * Bmax if Bmax > 0 else 1e-14)
        B = np.maximum(B, floor)
        p = p / np.power(B, alpha)
        p = normalize_positive(p)
        step += 1

    return best_p


def run_hankel_dual(p0, n, time_limit, rng, alpha=1.0, ridge_frac=1e-7):
    """
    For moderate n, repeatedly solve a fixed-weight stationarity system:
        M(w) p = 1,
    where M(w)_{i,l} = 2*w_{i+l}. This is a dual/heuristic Newton-like step
    for the minimax convolution problem.
    """
    start = time.monotonic()
    limit = start + max(0.0, time_limit)

    if n <= 1 or time_limit <= 0.5 or n > 320:
        return normalize_positive(p0)

    p = normalize_positive(p0)
    c = convolution_self(p)
    best_score = float(2.0 * n * np.max(c)) if c.size else float("inf")
    best_p = p.copy()

    noimp = 0
    step = 0

    while time.monotonic() < limit:
        p = best_p
        c = convolution_self(p)
        cmax = float(np.max(c))
        score = float(2.0 * n * cmax)

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            noimp = 0
        else:
            noimp += 1

        if noimp > 80:
            p = 0.8 * best_p + 0.2 * np.abs(rng.normal(0.0, 1.0, size=n))
            best_p = normalize_positive(p)
            best_score = score_fft(best_p)
            noimp = 0
            step = 0
            continue

        base = np.maximum(c, 0.0)
        if alpha == 1.0:
            w = base.copy()
        else:
            w = np.power(np.maximum(base, 1e-300), alpha)
        sw = float(np.sum(w))
        if sw <= 1e-300:
            w = np.full(c.size, 1.0 / c.size, dtype=np.float64)
        else:
            w = w / sw

        # Add a small uniform component to improve conditioning.
        w = 0.90 * w + 0.10 * (1.0 / c.size)
        w = w / float(np.sum(w))

        M = np.zeros((n, n), dtype=np.float64)
        for i in range(n):
            M[i, :] = 2.0 * w[i:i + n]

        tr = float(np.trace(M))
        ridge = max(1e-12, ridge_frac * (tr / n if n > 0 else 1e-12))
        M = M + ridge * np.eye(n, dtype=np.float64)

        try:
            sol = np.linalg.solve(M, np.ones(n, dtype=np.float64))
        except Exception:
            sol = p.copy()

        sol = np.where(np.isfinite(sol), sol, 0.0)
        sol = np.maximum(sol, 0.0)
        ss = float(np.sum(sol))
        if ss <= 1e-300:
            sol = p.copy()
        else:
            sol = sol / ss

        if step < 15:
            mix = 0.55
        elif step < 60:
            mix = 0.25
        else:
            mix = 0.12

        p = (1.0 - mix) * best_p + mix * sol
        p = normalize_positive(p)
        step += 1

    return best_p


def local_search_prob(p0, n, time_limit, rng):
    """
    Direct pairwise-transfer local search for short sequences.
    It moves a small amount of mass from a high-gradient coordinate to a
    low-gradient coordinate and accepts only exact improvements.
    """
    start = time.monotonic()
    limit = start + max(0.0, time_limit)

    p = normalize_positive(p0)
    if n <= 1 or time_limit <= 0.5:
        return p

    c = np.convolve(p, p)
    best_score = float(2.0 * n * np.max(c))
    best_p = p.copy()
    noimp = 0

    while time.monotonic() < limit:
        c = convolution_self(p)
        cmax = float(np.max(c))
        score = float(2.0 * n * cmax)
        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            noimp = 0
        else:
            noimp += 1

        if noimp > 80:
            p = 0.8 * best_p + 0.2 * np.abs(rng.normal(0.0, 1.0, size=n))
            p = normalize_positive(p)
            c = np.convolve(p, p)
            best_score = float(2.0 * n * np.max(c))
            best_p = p.copy()
            noimp = 0
            continue

        w = active_weights(c, cmax, 0.05, 1.0)
        A = grad_A_fft(p, w, n)
        gp = 2.0 * A
        gp = np.where(np.isfinite(gp), gp, 0.0)

        order = np.argsort(gp)
        tail = max(1, n // 8)
        low = order[:tail]
        high = order[-tail:]

        i = int(high[rng.integers(len(high))])
        j = int(low[rng.integers(len(low))])
        if i == j and n > 1:
            j = int(rng.integers(n))
            if i == j:
                continue

        pos = np.where(p > 1e-12)[0]
        if pos.size == 0:
            return best_p
        if p[i] <= 1e-12:
            i = int(pos[rng.integers(pos.size)])
            if i == j:
                j = int((i + 1) % n)

        improved = False
        for frac in (0.25, 0.50, 0.10, 0.05, 0.02, 0.01, 0.005, 0.001):
            d = frac * p[i]
            if d > 0.9 * p[i]:
                d = 0.9 * p[i]
            if d <= 0:
                continue

            cand = p.copy()
            cand[i] -= d
            cand[j] += d
            cand = np.maximum(cand, 0.0)
            s = float(np.sum(cand))
            if s <= 1e-300:
                continue
            cand = cand / s

            cand_c = np.convolve(cand, cand)
            cand_score = float(2.0 * n * np.max(cand_c))
            if cand_score < best_score - 1e-12:
                p = cand
                c = cand_c
                best_score = cand_score
                best_p = p.copy()
                improved = True
                break

        if not improved:
            noimp += 3

    return best_p

def make_starts(n, rng, base_p=None, count=40):
    """Generate diverse non-negative probability starts of length n."""
    starts = []

    def add_local(p):
        if len(starts) < count:
            q = normalize_positive(p)
            if q.size == n and np.all(np.isfinite(q)):
                starts.append(q)

    if base_p is not None and np.asarray(base_p).size == n:
        bp = normalize_positive(base_p)
        add_local(bp)
        for a in (0.02, 0.08, 0.20):
            add_local(bp + a * rng.normal(0.0, 1.0, size=n))

    add_local(np.full(n, 1.0))

    if n > 1:
        x = np.linspace(0.0, 1.0, n)
    else:
        x = np.zeros(1)

    for slope in (-1.0, -0.6, -0.3, 0.3, 0.6, 1.0):
        add_local(1.0 + slope * (x - 0.5))

    for a, b in (
        (-1.0, 1.0), (1.0, -1.0), (-0.5, 0.5), (0.5, -0.5),
        (1.0, 1.0), (-1.0, -1.0), (0.0, 1.0), (1.0, 0.0)
    ):
        w = np.power(x + 0.02, a) * np.power(1.0 - x + 0.02, b)
        add_local(w)

    for f in (1, 2, 3, 4, 6):
        for amp in (0.25, 0.50, 0.75):
            for ph in (0.0, math.pi / 2.0):
                w = 1.0 + amp * np.cos(2.0 * math.pi * f * x + ph)
                add_local(w)

    for centers in (
        (0.5,),
        (0.1, 0.9),
        (0.2, 0.8),
        (0.3, 0.7),
        (0.15, 0.5, 0.85),
    ):
        w = np.zeros(n, dtype=np.float64)
        for c in centers:
            w += np.exp(-0.5 * ((x - c) / 0.08) ** 2)
        w += 0.05
        add_local(w)

    min_rand = max(3, count // 4)
    for _ in range(min_rand):
        if len(starts) >= count:
            break
        coeff = rng.normal(0.0, 0.35, size=8)
        phase = rng.uniform(0.0, 2.0 * math.pi, size=8)
        w = np.ones(n, dtype=np.float64)
        for k in range(8):
            w += coeff[k] * np.cos(2.0 * math.pi * (k + 1) * x + phase[k])
        add_local(w)

    while len(starts) < count:
        add_local(np.abs(rng.normal(0.0, 1.0, size=n)) + 0.05)

    return starts[:count]


def get_base_sequence():
    """Fetch and normalize the supplied reference sequence, if available."""
    try:
        seq = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
    except Exception:
        return np.full(2048, 1.0 / 2048, dtype=np.float64)

    if seq.ndim != 1 or seq.size == 0:
        return np.full(2048, 1.0 / 2048, dtype=np.float64)

    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.clip(seq, 0.0, 1000.0)
    s = float(np.sum(seq))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(2048, 1.0 / 2048, dtype=np.float64)

    return seq / s


def maybe_add_length(lengths, value, maxn=MAX_OPT_N):
    v = int(round(float(value)))
    if 3 <= v <= maxn and v not in lengths:
        lengths.append(v)


def quick_scan(lengths, rng, base_p):
    """Quickly score a few starts for many candidate lengths."""
    items = []
    idx = 0
    for n in lengths:
        starts = make_starts(n, rng, base_p if base_p.size == n else None, count=10)
        best_p = None
        best_s = float("inf")
        for p in starts:
            s = score_fft(p)
            if np.isfinite(s) and s < best_s:
                best_s = s
                best_p = p
        if best_p is not None:
            items.append((best_s, n, idx, best_p))
            idx += 1
    items.sort()
    return items


def choose_lengths(lengths, base_n, quick_items):
    """Choose a few lengths to optimize fully."""
    chosen = []
    maybe_add_length(chosen, base_n, MAX_OPT_N)

    # Add the best quick pieces, preserving at least some diversity.
    seen = set(chosen)
    for s, n, _, p in quick_items[:8]:
        if len(chosen) >= 7:
            break
        if n not in seen:
            chosen.append(n)
            seen.add(n)

    # Ensure some short/medium lengths are represented.
    for v in (16, 32, 64, 96, 128, 256):
        if len(chosen) >= 8:
            break
        maybe_add_length(chosen, v, MAX_OPT_N)

    if not chosen:
        chosen.append(128)

    return chosen


def build_candidates(chosen_lengths, rng, base_p):
    """Build and pre-score richer initial candidates for chosen lengths."""
    cand = []
    counter = 0
    for n in chosen_lengths:
        if n > 4000:
            count = 22
        elif n > 800:
            count = 45
        else:
            count = 80

        starts = make_starts(n, rng, base_p if base_p.size == n else None, count=count)
        scored = []
        for p in starts:
            s = score_fft(p)
            if np.isfinite(s):
                scored.append((s, n, counter, p))
                counter += 1

        scored.sort()
        if n > 6000:
            keep = 2
        elif n > 1500:
            keep = 3
        elif n > 300:
            keep = 4
        else:
            keep = 5

        cand.extend(scored[:keep])

    cand.sort()
    return cand[:9]


# -----------------------------------------------------------------------------
# Candidate-level optimizer
# -----------------------------------------------------------------------------

def optimize_candidate(p0, n, time_limit, rng):
    """Run a multi-stage optimization pipeline for one candidate."""
    start = time.monotonic()
    limit = start + max(0.0, time_limit)

    p0 = normalize_positive(p0)
    if n <= 1 or time_limit <= 0.5:
        return p0, score_fft(p0)

    z = log_from_prob(p0)
    if n > 1 and rng.random() < 0.8:
        z = z + rng.normal(0.0, 0.10, size=n)
        z = z - float(np.mean(z))
        z = np.clip(z, -50.0, 50.0)

    best_z = z
    best_p = softmax_vector(best_z)
    best_score = score_fft(best_p)

    # 1) Smooth Adam.
    t_adam = max(8.0, min(0.52 * time_limit, time_limit - 25.0))
    if time.monotonic() < limit - t_adam and t_adam > 3.0:
        z, p, s = run_adam_smooth(z, n, t_adam, rng)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_p = p
            best_z = z

    # 2) Multiplicative basin-hopping.
    t_mul = min(0.14 * time_limit, max(0.0, limit - time.monotonic() - 20.0))
    if t_mul > 3.0:
        p = run_multiplicative(best_p, n, t_mul, rng)
        s = score_fft(p)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_p = p
            best_z = log_from_prob(best_p)

    # 3) Dual/Hankel Newton-like for short/medium lengths.
    if n <= 320:
        t_dual = min(0.18 * time_limit, max(0.0, limit - time.monotonic() - 20.0))
        if t_dual > 3.0:
            p = run_hankel_dual(best_p, n, t_dual, rng)
            s = score_fft(p)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_p = p
                best_z = log_from_prob(best_p)

    # 4) L-BFGS refinement.
    t_lb = min(0.23 * time_limit, max(0.0, limit - time.monotonic() - 15.0))
    if n <= 12000 and t_lb > 4.0 and HAVE_SCIPY:
        z2 = run_lbfgs(best_z, n, t_lb, eta_ratio=0.035)
        p2 = softmax_vector(z2)
        s2 = score_fft(p2)
        if np.isfinite(s2) and s2 < best_score:
            best_score = s2
            best_p = p2
            best_z = z2

    # 5) Exact active-lag Adam.
    t_hard = min(0.20 * time_limit, max(0.0, limit - time.monotonic() - 5.0))
    if t_hard > 4.0:
        z3, p3 = run_hard_active(best_z, n, t_hard, rng, eps=0.018)
        s3 = score_fft(p3)
        if np.isfinite(s3) and s3 < best_score:
            best_score = s3
            best_p = p3
            best_z = z3

    # 6) Exact pairwise local search for short sequences.
    if n <= 128:
        t_loc = max(0.0, limit - time.monotonic() - 2.0)
        if t_loc > 5.0:
            p4 = local_search_prob(best_p, n, t_loc, rng)
            s4 = score_fft(p4)
            if np.isfinite(s4) and s4 < best_score:
                best_score = s4
                best_p = p4
                best_z = log_from_prob(best_p)

    return best_p, best_score

# Main entry point
# -----------------------------------------------------------------------------

def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns a Python list of non-negative floats.
    """
    start = time.monotonic()
    budget = float(budget_s) if budget_s is not None else 1800.0
    budget = max(30.0, budget)
    deadline = start + budget - 10.0
    rng = np.random.default_rng(int(seed % (2**32)))

    base_p = get_base_sequence()
    base_n = int(base_p.size)

    best_p = base_p.copy() if base_p.size >= 1 else np.full(128, 1.0 / 128.0)
    best_n = int(best_p.size)
    best_score = score_fft(best_p) if best_n <= 20000 else float("inf")
    print(f"initial base: n={best_n}, score={best_score:.8f}")

    # Candidate lengths: include the base length, its scales, and useful fixed sizes.
    lengths = []
    for v in (
        base_n, base_n // 2, base_n // 4, int(round(base_n * 1.5)),
        int(round(base_n * 2)), int(round(base_n * 3))
    ):
        maybe_add_length(lengths, v, MAX_OPT_N)

    fixed = [
        6, 8, 10, 12, 14, 16, 20, 24, 32, 40, 48, 64, 80, 96,
        128, 160, 192, 256, 320, 384, 512, 640, 768, 1024, 1280,
        1536, 2048, 2560, 3072, 4096, 5120, 6144, 8192, 10240, 12288
    ]
    for v in fixed:
        maybe_add_length(lengths, v, MAX_OPT_N)

    if not lengths:
        lengths = [128]

    # Quick scan to choose promising lengths.
    quick_items = quick_scan(lengths, rng, base_p)
    if quick_items:
        s0, n0, _, p0 = quick_items[0]
        print(f"quick best: n={n0}, score={s0:.8f}")
        if np.isfinite(s0) and s0 < best_score:
            best_score = s0
            best_n = n0
            best_p = p0.copy()

    chosen = choose_lengths(lengths, base_n if 3 <= base_n <= MAX_OPT_N else 128, quick_items)
    chosen = [n for n in chosen if 3 <= n <= MAX_OPT_N]
    if not chosen:
        chosen = [128]

    # Rich candidate construction.
    candidates = build_candidates(chosen, rng, base_p)
    if not candidates:
        candidates = [(score_fft(best_p), best_n, 0, best_p)]

    # Optimize candidates. Time budgets favor the strongest pre-scored starts.
    time_budgets = [420, 300, 240, 200, 180, 150, 150, 120, 120]
    idx = 0
    while idx < len(candidates) and time.monotonic() < deadline - 45.0:
        s_cand, n_cand, _, p_cand = candidates[idx]
        remaining = max(0.0, deadline - 15.0 - time.monotonic())
        if remaining <= 25.0:
            break

        # Convert absolute budget to available budget, never exceeding full allocation.
        tb = min(time_budgets[idx] if idx < len(time_budgets) else 100, remaining * 0.9)
        if tb < 25.0:
            break

        print(f"optimizing candidate {idx + 1}/{len(candidates)}: n={n_cand}, initial={s_cand:.8f}, time={tb:.1f}")
        t0 = time.monotonic()
        p_new, s_new = optimize_candidate(p_cand, n_cand, tb, rng)
        dt = time.monotonic() - t0
        if np.isfinite(s_new) and s_new < best_score:
            best_score = s_new
            best_n = n_cand
            best_p = p_new.copy()
            print(f"  improved: n={best_n}, score={best_score:.8f}, dt={dt:.1f}")
        else:
            print(f"  no improvement: best={best_score:.8f}, candidate={s_new if np.isfinite(s_new) else 'inf'}")

        idx += 1

    # Final refinement on the global best.
    remaining = max(0.0, deadline - 5.0 - time.monotonic())
    if remaining > 25.0 and best_n >= 3:
        best_z = log_from_prob(best_p)
        print(f"final refinement: n={best_n}, score={best_score:.8f}, remaining={remaining:.1f}")

        t_hard = min(300.0, remaining * 0.45)
        if t_hard > 10.0:
            z_new, p_new = run_hard_active(best_z, best_n, t_hard, rng, eps=0.012)
            s_new = score_fft(p_new)
            if np.isfinite(s_new) and s_new < best_score:
                best_score = s_new
                best_p = p_new.copy()
                best_z = z_new
                print(f"  hard active improved: {best_score:.8f}")

        if best_n <= 2500:
            t_mul = min(120.0, remaining * 0.20)
            if t_mul > 8.0:
                p_new = run_multiplicative(best_p, best_n, t_mul, rng)
                s_new = score_fft(p_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_p = p_new.copy()
                    best_z = log_from_prob(best_p)
                    print(f"  multiplicative improved: {best_score:.8f}")

        if best_n <= 320:
            t_dual = min(120.0, remaining * 0.20)
            if t_dual > 8.0:
                p_new = run_hankel_dual(best_p, best_n, t_dual, rng)
                s_new = score_fft(p_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_p = p_new.copy()
                    best_z = log_from_prob(best_p)
                    print(f"  hankel dual improved: {best_score:.8f}")

        if HAVE_SCIPY and best_n <= 5000:
            t_lb = min(150.0, remaining * 0.20)
            if t_lb > 10.0:
                z_new = run_lbfgs(best_z, best_n, t_lb, eta_ratio=0.025)
                p_new = softmax_vector(z_new)
                s_new = score_fft(p_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_p = p_new.copy()
                    best_z = z_new
                    print(f"  LBFGS final improved: {best_score:.8f}")

        if best_n <= 128:
            t_loc = max(5.0, remaining * 0.15)
            if t_loc > 5.0:
                p_new = local_search_prob(best_p, best_n, t_loc, rng)
                s_new = score_fft(p_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_p = p_new.copy()
                    print(f"  local search improved: {best_score:.8f}")

    final_p = normalize_positive(best_p)
    final_seq = [float(x) for x in final_p]

    # Safety check using the official evaluator if available.
    try:
        eval_val = evaluate_sequence(final_seq)  # noqa: F821
        if not np.isfinite(eval_val):
            final_seq = [float(x) for x in normalize_positive(np.ones(max(3, best_n)))]
    except Exception:
        pass

    print(f"final returned: n={len(final_seq)}, internal_score={best_score:.8f}")
    return final_seq
