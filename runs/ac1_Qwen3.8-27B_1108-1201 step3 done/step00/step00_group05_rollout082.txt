```
def conv_of(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0: return np.zeros(0)
    if n <= 768:
        return np.convolve(a, a)
    m = 2*n - 1
    size = 1
    while size < m:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:m]
```

```
def internal_score(a):
    try:
       arr = np.asarray(a, dtype=np.float64)
    except Exception:
       return float('inf')
    if arr.ndim != 1 or arr.size == 0: return inf
    if not np.isfinite(arr).all(): return inf
    arr = np.clip(arr, 0, 1000)
    s = arr.sum()
    if not math.isfinite(s) or s < 0.01: return inf
    c = conv_of(arr)
    if c.size ==0: return inf
    mx = float(np.max(c))
    if not math.isfinite(mx): return inf
    return float(2.0 * arr.size * mx / (s * s))
```

```
def norm1(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim !=1 or arr.size==0: return np.ones(1)
    arr[np.isnan(arr)] = 0
    arr[~np.isfinite(arr)] = 0
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if not math.isfinite(s) or s <= 1e-15:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    arr /= s
    return arr
```

```
def build_conv_matrix(a, n):

    offs = np.tile(np.arange(n, dtype=np.int32), n)
    rows = (cols + offs).astype(np.int32, copy=False)
    vals = np.tile(np.asarray(a, dtype=np.float64), n)
    mat = sp.coo_matrix((vals, (rows, cols)), shape=(2*n-1, n))
```

```python
"""
Search for non-negative step heights minimizing:
  J(a) = 2*n*max(convolve(a, a)) / sum(a)^2

The program combines:
- several structured and random initializations,
- low-dimensional parametric annealing,
- projected/natural subgradient descent on the active convolution peaks,
- the LP SOTA direction from the previous state of the art,
- stochastic block polishing,
- a final exact call to evaluate_sequence() inside permit.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as scopt


def start_sort_key(item):
    return item[0]


def candidate_sort_key(item):
    return item[0]


def conv_of(a):
    """Linear convolution a*a, direct for small n, FFT for large n."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    if n <= 768:
        return np.convolve(a, a)
    m = 2 * n - 1
    size = 1
    while size < m:
        size <<= 1
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    return c[:m]


def internal_score(a):
    """Fast shadow evaluator matching evaluate_sequence for valid inputs."""
    try:
        arr = np.asarray(a, dtype=np.float64)
    except Exception:
        return float("inf")
    if arr.ndim != 1 or arr.size == 0:
        return float("inf")
    if not np.isfinite(arr).all():
        return float("inf")
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if not math.isfinite(s) or s < 0.01:
        return float("inf")
    c = conv_of(arr)
    if c.size == 0:
        return float("inf")
    mx = float(np.max(c))
    if not math.isfinite(mx):
        return float("inf")
    return float(2.0 * float(arr.size) * mx / (s * s))


def norm1(a):
    """Clip to [0, 1000] and normalize to sum 1."""
    arr = np.asarray(a, dtype=np.float64).copy()
    if arr.ndim != 1 or arr.size == 0:
        return np.ones(1, dtype=np.float64)
    arr[~np.isfinite(arr)] = 0.0
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if not math.isfinite(s) or s <= 1e-15:
        return np.ones(arr.size, dtype=np.float64) / float(arr.size)
    arr /= s
    return arr


def perturb_log(a, strength, rng):
    """Smooth log-domain perturbation, useful for escaping local optima."""
    arr = np.asarray(a, dtype=np.float64).copy()
    n = arr.size
    if n == 0:
        return arr
    eps = 1e-12
    theta = np.log(np.maximum(arr, eps))
    if n > 1:
        x = np.arange(n, dtype=np.float64) / float(n - 1)
    else:
        x = np.zeros(1, dtype=np.float64)
    theta += strength * 0.55 * np.sin(2.0 * math.pi * x + rng.uniform(0.0, 2.0 * math.pi))
    theta += strength * 0.30 * np.sin(4.0 * math.pi * x + rng.uniform(0.0, 2.0 * math.pi))
    theta += strength * 0.18 * np.sin(8.0 * math.pi * x + rng.uniform(0.0, 2.0 * math.pi))
    theta += strength * 0.07 * np.sin(16.0 * math.pi * x + rng.uniform(0.0, 2.0 * math.pi))
    theta += strength * 0.04 * rng.normal(0.0, 1.0, size=n)
    theta -= float(np.max(theta))
    return norm1(np.exp(theta))


def seq_from_params(p, n):
    """Compact smooth parametrization: ramp + quadratic + first 3 harmonics."""
    p = np.asarray(p, dtype=np.float64)
    if n <= 1:
        return np.ones(max(1, n), dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / float(n - 1)
    theta = p[0] * x + p[1] * (x - 0.5) ** 2
    for m in (1, 2, 3):
        base = 2 + 2 * (m - 1)
        theta += p[base] * np.cos(2.0 * math.pi * m * x)
        theta += p[base + 1] * np.sin(2.0 * math.pi * m * x)
    theta -= float(np.max(theta))
    return norm1(np.exp(theta))


def make_starts(n, seed, want=12):
    """Generate, normalize, score, and return a small set of initial sequences."""
    rng = np.random.default_rng(seed)
    starts = []

    starts.append(np.ones(n, dtype=np.float64))

    x = np.linspace(0.0, 1.0, n)

    # One-sided geometric / power tails.
    for g in (0.2, 0.5, 1.0, 2.0, 4.0, 8.0):
        a = np.exp(-g * x)
        starts.append(a)
        starts.append(a[::-1].copy())

    idx = np.arange(1, n + 1, dtype=np.float64)
    for p in (0.2, 0.5, 1.0):
        a = idx ** (-p)
        starts.append(a)
        starts.append(a[::-1].copy())

    # Low-dimensional asymmetric shapes.
    for r in (-3.0, -1.5, 1.5, 3.0):
        for amp in (0.5, 1.0, 2.0):
            for ph in (0.0, 0.5 * math.pi, math.pi, 1.5 * math.pi):
                starts.append(np.exp(r * x + amp * np.sin(2.0 * math.pi * x + ph)))
                starts.append(np.exp(r * x + amp * np.cos(2.0 * math.pi * x + ph)))

    # Smooth random log-density-like starts.
    for _ in range(20):
        noise = rng.normal(0.0, 1.0, size=n)
        if n > 3:
            noise = np.convolve(noise, np.ones(3) / 3.0, mode="same")
            noise = np.convolve(noise, np.ones(5) / 5.0, mode="same")
        sd = float(np.std(noise))
        if sd < 1e-12:
            sd = 1.0
        sigma = 0.5 + rng.random() * 1.5
        starts.append(np.exp(sigma * noise / sd))

    # Lognormal random starts.
    for sig in (0.5, 1.0, 2.0):
        starts.append(np.exp(rng.normal(0.0, sig, size=n)))

    scored = []
    for raw in starts:
        a = norm1(raw)
        sc = internal_score(a)
        if math.isfinite(sc):
            scored.append((sc, a))

    if not scored:
        a = np.ones(n, dtype=np.float64) / float(n)
        return [a]

    scored.sort(key=start_sort_key)
    out = []
    for _, a in scored[:want]:
        out.append(a.copy())
    return out


def param_search(n, seconds, seed):
    """Quick annealing over the 8-dimensional smooth parametric family."""
    if seconds <= 1e-3:
        p = np.zeros(8, dtype=np.float64)
        return seq_from_params(p, n), internal_score(p)

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, seconds)

    low = np.array([-5.0, -5.0, -3.0, -3.0, -3.0, -3.0, -3.0, -3.0], dtype=np.float64)
    high = np.array([5.0, 5.0, 3.0, 3.0, 3.0, 3.0, 3.0, 3.0], dtype=np.float64)

    cur_p = rng.uniform(low, high)
    cur_a = seq_from_params(cur_p, n)
    cur_score = internal_score(cur_a)

    best_p = cur_p.copy()
    best_a = cur_a.copy()
    best_score = cur_score

    T = 0.08
    sigma = 0.35
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        if internal_score(best_a) < 1.35 and it > 200:
            break

        coord = int(rng.integers(0, 8))
        new_p = cur_p.copy()
        new_p[coord] += rng.normal(0.0, sigma)
        new_p[coord] = min(max(new_p[coord], low[coord]), high[coord])

        new_a = seq_from_params(new_p, n)
        new_score = internal_score(new_a)

        if math.isfinite(new_score):
            if new_score < best_score:
                best_score = new_score
                best_a = new_a.copy()
                best_p = new_p.copy()

            diff = cur_score - new_score
            if diff < 0.0:
                accept = True
            else:
                if diff > 50.0 * max(T, 1e-12):
                    accept = False
                else:
                    accept = math.exp(-diff / max(T, 1e-12)) > rng.random()
        else:
            accept = False

        if accept:
            cur_p = new_p.copy()
            cur_a = new_a.copy()
            cur_score = new_score
            stall = 0
        else:
            stall += 1
            if stall > 120:
                T = min(0.3, T * 1.3)
                sigma = min(0.7, sigma * 1.15)
                stall = 0

        T = max(0.003, T * 0.999)
        sigma = max(0.04, 0.35 * math.pow(0.999, it))
        it += 1

    return best_a, best_score


def improve_gradient(a, seconds, seed=0):
    """
    Natural/projected subgradient steps on the active peaks of conv(a, a).
    Only reduces elements that contribute to the current peaks; normalization
    reallocates mass to the rest.
    """
    if seconds <= 1e-3:
        return a, internal_score(a)

    rng = np.random.default_rng(seed)
    a = norm1(a.copy())
    best = a.copy()
    best_score = internal_score(a)

    deadline = time.monotonic() + max(0.0, seconds)
    temp = 0.22
    last_improve = time.monotonic()
    last_print = best_score
    it = 0

    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 0.5:
            break

        c = conv_of(a)
        M = float(np.max(c))
        if M <= 0.0 or not math.isfinite(M):
            break

        top_count = int(min(c.size, max(8, a.size // 32)))
        if top_count <= c.size:
            idx = np.argsort(c)[-top_count:]
        else:
            idx = np.arange(c.size)

        vals = c[idx]
        scale = max(1e-300, M * temp)
        w = np.exp((vals - M) / scale)
        total = float(w.sum())
        if total <= 0.0 or not math.isfinite(total):
            w = np.zeros(top_count, dtype=np.float64)
            w[int(np.argmax(vals))] = 1.0
        else:
            w /= total

        mask = w > 1e-5
        if not mask.any():
            mask = np.zeros(top_count, dtype=bool)
            mask[int(np.argmax(w))] = True

        idx = idx[mask]
        w = w[mask]
        wsum = float(w.sum())
        if wsum <= 0.0:
            break
        w /= wsum

        n = a.size
        corr = np.zeros(n, dtype=np.float64)
        for ii in range(len(idx)):
            k = int(idx[ii])
            wt = float(w[ii])
            i0 = k - n + 1
            if i0 < 0:
                i0 = 0
            i1 = k
            if i1 > n - 1:
                i1 = n - 1
            if i0 <= i1:
                j0 = k - i1
                j1 = k - i0
                corr[i0:i1 + 1] += wt * a[j0:j1 + 1][::-1]

        h = 2.0 * a * corr
        hmax = float(np.max(h))
        if hmax <= 1e-300 or not math.isfinite(hmax):
            break
        scale = h / hmax

        improved = False

        # Exponential natural-gradient style steps.
        for step in (0.05, 0.1, 0.2, 0.4, 0.8, 1.5):
            cand = a * np.exp(-step * scale)
            cand = norm1(cand)
            sc = internal_score(cand)
            if math.isfinite(sc) and sc < best_score - 1e-12:
                a = cand
                best = cand.copy()
                best_score = sc
                improved = True

        # Multiplicative clipped steps.
        for step in (0.2, 0.5, 1.2):
            cand = a * np.maximum(0.0, 1.0 - step * scale)
            cand = norm1(cand)
            sc = internal_score(cand)
            if math.isfinite(sc) and sc < best_score - 1e-12:
                a = cand
                best = cand.copy()
                best_score = sc
                improved = True

        # Occasionally add a smooth perturbation to avoid becoming too jagged.
        if rng.random() < 0.08:
            cand = perturb_log(a, 0.05 + 0.25 * rng.random(), rng)
            sc = internal_score(cand)
            if math.isfinite(sc) and sc < best_score - 1e-12:
                a = cand
                best = cand.copy()
                best_score = sc
                improved = True

        if improved:
            last_improve = time.monotonic()
            temp = max(0.03, temp * 0.995)
            if best_score < last_print - 1e-4:
                print("gradient n=%d score=%.8f" % (a.size, best_score))
                last_print = best_score
        else:
            # If stagnating, try a stronger perturbation.
            if time.monotonic() - last_improve > max(15.0, 0.10 * seconds):
                cand = perturb_log(best, 0.25 + 0.45 * rng.random(), rng)
                sc = internal_score(cand)
                if math.isfinite(sc) and sc < best_score - 1e-12:
                    a = cand
                    best = cand.copy()
                    best_score = sc
                    last_improve = time.monotonic()
                    improved = True
                else:
                    break

        it += 1
        if it % 5000 == 0:
            print("gradient iter n=%d score=%.8f left=%.1f" % (a.size, best_score, deadline - time.monotonic()))

    return best, best_score


def solve_lp(A, rhs, n, tlim):
    """Solve max sum(B) subject to A B <= rhs, B >= 0 using HiGHS."""
    obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        opts = {"presolve": True, "time_limit": max(1.0, float(tlim))}
        return scopt.linprog(
            obj,
            A_ub=A,
            b_ub=rhs,
            bounds=bounds,
            method="highs",
            options=opts
        )
    except Exception:
        try:
            opts = {"presolve": True}
            return scopt.linprog(
                obj,
                A_ub=A,
                b_ub=rhs,
                bounds=bounds,
                method="highs",
                options=opts
            )
        except Exception:
            return None


def build_conv_matrix(a, n):
    """Sparse matrix A where (A b)_k = convolve(a, b)_k."""
    a = np.asarray(a, dtype=np.float64)
    cols = np.tile(np.arange(n, dtype=np.int32), n)
    offs = np.tile(np.arange(n, dtype=np.int32), n)
    rows = (cols + offs).astype(np.int32, copy=False)
    vals = np.tile(a, n)
    mat = sp.coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return mat.tocsr()


def lp_refine(a, seconds, seed=0):
    """
    Apply the LP direction:
      max sum(b) s.t. convolve(a, b) <= max convolve(a, a), b >= 0.
    Then line-search in b. Also tries a reduced-constraint LP when the full
    LP is near-stalled.
    """
    if seconds <= 1e-3:
        return a, internal_score(a)

    rng = np.random.default_rng(seed)
    a = norm1(a.copy())
    best = a.copy()
    best_score = internal_score(a)
    n = a.size

    if n > 2200:
        return best, best_score

    deadline = time.monotonic() + max(0.0, seconds)
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < max(3.0, 0.03 * seconds):
            break

        a_lp = a.copy()
        a_lp[a_lp < 1e-12] = 1e-12
        a_lp /= float(a_lp.sum())

        c = conv_of(a_lp)
        M = float(np.max(c))
        if M <= 0.0 or not math.isfinite(M):
            break

        tlim = max(1.0, min(90.0, remaining - 3.0))
        try:
            A = build_conv_matrix(a_lp, n)
            rhs = float(n) * M
            res = solve_lp(A, np.full(2 * n - 1, rhs, dtype=np.float64), n, tlim)
        except Exception as exc:
            print("LP build/solve exception:", repr(exc))
            break

        if res is None or getattr(res, "status", -1) != 0:
            # A cut set can sometimes still be useful, but if the full LP is
            # numerically bad, stop LP for this state.
            del A
            stall += 1
            if stall >= 4:
                break
            cand = perturb_log(best, 0.05 + 0.10 * rng.random(), rng)
            sc = internal_score(cand)
            if math.isfinite(sc) and sc < best_score - 1e-12:
                best = cand.copy()
                best_score = sc
                a = best.copy()
                stall = 0
            continue

        B = np.asarray(res.x, dtype=np.float64)
        if not np.isfinite(B).all():
            del A
            break
        Bsum = float(B.sum())

        improved = False

        if Bsum > 0.0:
            factor = Bsum / float(n)
            if factor > 1.0 + 1e-7:
                b = B / Bsum
                a_old = a
                cand_best = None
                cand_best_score = best_score
                for t in (0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 0.02, 0.5):
                    cand = (1.0 - t) * a_old + t * b
                    cand = norm1(cand)
                    sc = internal_score(cand)
                    if math.isfinite(sc) and sc < cand_best_score:
                        cand_best = cand
                        cand_best_score = sc
                if cand_best is not None:
                    a = cand_best.copy()
                    best = cand_best.copy()
                    best_score = cand_best_score
                    improved = True
            else:
                # Full LP says no room under all constraints. Try only the
                # currently dominant constraints for an exploratory direction.
                top_count_lp = int(min(32, 2 * n - 1))
                top_rows = np.argsort(c)[-top_count_lp:]
                try:
                    A_sub = A[top_rows, :].tocsr()
                    rhs_sub = np.full(top_count_lp, float(n) * M, dtype=np.float64)
                    res2 = solve_lp(A_sub, rhs_sub, n, max(1.0, tlim / 3.0))
                except Exception:
                    res2 = None

                if res2 is not None and getattr(res2, "status", -1) == 0:
                    B2 = np.asarray(res2.x, dtype=np.float64)
                    if np.isfinite(B2).all() and B2.sum() > 0.0:
                        b = B2 / float(B2.sum())
                        a_old = a
                        cand_best = None
                        cand_best_score = best_score
                        for t in (0.05, 0.1, 0.2, 0.3, 0.5, 0.8):
                            cand = (1.0 - t) * a_old + t * b
                            cand = norm1(cand)
                            sc = internal_score(cand)
                            if math.isfinite(sc) and sc < cand_best_score:
                                cand_best = cand
                                cand_best_score = sc
                        if cand_best is not None:
                            a = cand_best.copy()
                            best = cand_best.copy()
                            best_score = cand_best_score
                            improved = True

        del A

        if improved:
            stall = 0
            if best_score < 2.0:
                print("LP n=%d score=%.8f iter=%d" % (n, best_score, it))
        else:
            cand = perturb_log(best, 0.04 + 0.12 * rng.random(), rng)
            sc = internal_score(cand)
            if math.isfinite(sc) and sc < best_score - 1e-12:
                best = cand.copy()
                best_score = sc
                a = best.copy()
                stall = 0
            else:
                stall += 1
                if stall >= 4:
                    break

        it += 1

    return best, best_score


def random_polish(a, seconds, seed=0):
    """Cheap random block multiplicative polish in log domain."""
    if seconds <= 1e-3:
        return a, internal_score(a)

    rng = np.random.default_rng(seed)
    a = norm1(a.copy())
    best = a.copy()
    best_score = internal_score(a)
    deadline = time.monotonic() + max(0.0, seconds)
    sigma = 0.18
    stall = 0
    n = a.size

    while time.monotonic() < deadline:
        lo = max(1, n // 256)
        hi = max(lo + 1, n // 8)
        if hi > n:
            hi = n
        width = int(rng.integers(lo, hi)) if hi > lo else 1
        start = int(rng.integers(0, n - width + 1))

        cand = best.copy()
        amp = rng.normal(0.0, sigma, size=width)
        if width > 2:
            kernel = np.ones(3) / 3.0
            amp = np.convolve(amp, kernel, mode="same")
        cand[start:start + width] *= np.exp(amp)

        if rng.random() < 0.12:
            cand = perturb_log(cand, 0.08, rng)

        cand = norm1(cand)
        sc = internal_score(cand)

        if math.isfinite(sc) and sc < best_score - 1e-12:
            best = cand.copy()
            best_score = sc
            sigma = min(0.55, sigma * 1.02)
            stall = 0
        else:
            sigma = max(0.03, sigma * 0.99)
            stall += 1
            if stall > 250:
                break

    return best, best_score


def optimize_length(n, seconds, seed):
    """Optimize a fixed-length sequence using mouth and a final best-focus pass."""
    if seconds <= 5.0:
        starts = make_starts(n, seed, want=3)
        a = starts[0] if starts else np.ones(n, dtype=np.float64)
        return a, internal_score(a)

    t0 = time.monotonic()
    deadline = t0 + max(0.0, seconds)

    starts = make_starts(n, seed, want=8)

    # Add a dedicated parametric-annealing start if there is some time.
    if deadline - time.monotonic() > 45.0:
        p_seconds = min(25.0, (deadline - time.monotonic()) * 0.10)
        if p_seconds > 5.0:
            pa, psc = param_search(n, p_seconds, seed + 777)
            if math.isfinite(psc):
                starts.append(pa)

    scored = []
    for raw in starts:
        a = norm1(raw)
        sc = internal_score(a)
        if math.isfinite(sc):
            scored.append((sc, a))

    if not scored:
        a = np.ones(n, dtype=np.float64) / float(n)
        sc = internal_score(a)
        return a, sc

    scored.sort(key=start_sort_key)
    best = scored[0][1].copy()
    best_score = scored[0][0]
    print("optimize n=%d init=%.8f" % (n, best_score))

    # Round-robin improvement of a few promising starts.
    top_k = min(4, len(scored))
    if top_k > 0 and deadline - time.monotonic() > 70.0:
        base_chunk = (deadline - time.monotonic()) / float(top_k + 2)
        for i in range(top_k):
            if deadline - time.monotonic() < 35.0:
                break
            a = scored[i][1].copy()

            t_grad = min(base_chunk * 0.45, deadline - time.monotonic() - 5.0)
            if t_grad > 5.0:
                a, sc = improve_gradient(a, t_grad, seed + 100 + i)
                if math.isfinite(sc) and sc < best_score:
                    best = a.copy()
                    best_score = sc

            if deadline - time.monotonic() < 30.0:
                break
            t_lp = min(base_chunk * 0.30, deadline - time.monotonic() - 5.0)
            if n <= 1800 and t_lp > 8.0:
                a, sc = lp_refine(a, t_lp, seed + 300 + i)
                if math.isfinite(sc) and sc < best_score:
                    best = a.copy()
                    best_score = sc

            if deadline - time.monotonic() < 25.0:
                break
            t_polish = min(base_chunk * 0.20, deadline - time.monotonic() - 5.0)
            if t_polish > 3.0:
                a, sc = random_polish(a, t_polish, seed + 500 + i)
                if math.isfinite(sc) and sc < best_score:
                    best = a.copy()
                    best_score = sc

    # Focus the best state until the deadline.
    stall = 0
    while deadline - time.monotonic() > 30.0:
        remaining = deadline - time.monotonic()

        if n <= 1800 and remaining > 90.0:
            a, sc = lp_refine(best, min(220.0, remaining * 0.55), seed + 910)
            if not (math.isfinite(sc) and sc < best_score):
                a2, sc2 = random_polish(best, min(40.0, remaining * 0.15), seed + 920)
                if math.isfinite(sc2) and sc2 < best_score:
                    a, sc = a2, sc2

            if math.isfinite(sc) and sc < best_score:
                best = a.copy()
                best_score = sc
                stall = 0
                print("optimize n=%d best=%.8f" % (n, best_score))
            else:
                stall += 1
        else:
            a, sc = improve_gradient(best, min(180.0, remaining * 0.60), seed + 930)
            if not (math.isfinite(sc) and sc < best_score):
                a2, sc2 = random_polish(best, min(45.0, remaining * 0.20), seed + 940)
                if math.isfinite(sc2) and sc2 < best_score:
                    a, sc = a2, sc2

            if math.isfinite(sc) and sc < best_score:
                best = a.copy()
                best_score = sc
                stall = 0
                print("optimize n=%d best=%.8f" % (n, best_score))
            else:
                stall += 1

        if stall >= 4:
            break

    if best.size != n:
        best = norm1(np.ones(n, dtype=np.float64))
        best_score = internal_score(best)
    return best, best_score


def to_final_list(a):
    """Convert to a simple Python list with safe finite floats."""
    out = []
    for v in a:
        try:
            x = float(v)
        except Exception:
            x = 0.0
        if not math.isfinite(x):
            x = 0.0
        if x < 0.0:
            x = 0.0
        elif x > 1000.0:
            x = 1000.0
        out.append(x)
    return out


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Search box and return the best valid sequence found."""
    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 1800.0
    if not math.isfinite(total_budget) or total_budget <= 0.0:
        total_budget = 1800.0
    total_budget = min(total_budget, 1800.0)

    start_time = time.monotonic()
    deadline = start_time + max(20.0, total_budget - 10.0)

    candidates = []

    # Include the provided state-of-the-art sequence if present.
    try:
        hs = height_sequence_1
        a0 = norm1(hs)
        s0 = internal_score(a0)
        if math.isfinite(s0):
            candidates.append((s0, a0.copy()))
            print("initial height_sequence score=%.8f n=%d" % (s0, a0.size))
    except Exception:
        pass

    # Universal fallback.
    fallback = np.ones(256, dtype=np.float64) / 256.0
    fallback_score = internal_score(fallback)
    if math.isfinite(fallback_score):
        candidates.append((fallback_score, fallback.copy()))

    # Optimize several lengths.  The objective is scale-invariant and shorter
    # lengths are much cheaper; the final side chooses the best valid result.
    plans = [
        (256, 0.07),
        (512, 0.18),
        (1024, 0.37),
        (2048, 0.38),
    ]

    for idx, (n, frac) in enumerate(plans):
        if deadline - time.monotonic() < 70.0:
            break
        alloc = frac * (deadline - time.monotonic())
        alloc = min(alloc, deadline - time.monotonic() - 30.0)
        if alloc < 30.0:
            break
        try:
            a, sc = optimize_length(n, alloc, int(seed) + 1000 + idx)
        except Exception as exc:
            print("optimize_length error n=%d: %r" % (n, exc))
            a = np.ones(n, dtype=np.float64)
            sc = internal_score(a)

        if math.isfinite(sc):
            # Keep only one copy per state; array is already normalized.
            candidates.append((sc, a.copy()))

    if not candidates:
        return to_final_list(np.ones(256, dtype=np.float64) / 256.0)

    candidates.sort(key=candidate_sort_key)
    best_score = candidates[0][0]
    best_state = candidates[0][1].copy()

    # Short final polish of the current best.
    if deadline - time.monotonic() > 12.0:
        a_pol, sc_pol = random_polish(best_state, min(12.0, deadline - time.monotonic() - 2.0), int(seed) + 91001)
        if math.isfinite(sc_pol) and sc_pol < best_score:
            best_state = a_pol.copy()
            best_score = sc_pol

    # Compare the top few candidates and the polished best through the true
    # evaluate_sequence, if available.
    finalists = []
    seen = 0
    for sc, a in candidates:
        if seen >= 4:
            break
        finalists.append(a.copy())
        seen += 1
    finalists.append(best_state.copy())

    final_state = best_state.copy()
    final_score = best_score

    for a in finalists:
        try:
            val = evaluate_sequence(to_final_list(a))
        except Exception:
            val = internal_score(a)
        if math.isfinite(val) and val < final_score:
            final_state = a.copy()
            final_score = val

    # Last micro-polish, very short, to make sure we do not crash into the deadline.
    if deadline - time.monotonic() > 8.0:
        a_fin, sc_fin = random_polish(final_state, min(7.0, deadline - time.monotonic() - 2.0), int(seed) + 92002)
        if math.isfinite(sc_fin) and sc_fin < final_score:
            final_state = a_fin.copy()
            final_score = sc_fin

    out = to_final_list(final_state)
    print("final n=%d score=%.8f" % (len(out), final_score))
    return out
```
