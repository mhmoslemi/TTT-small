Search for a non-negative step-function height sequence minimizing
    2*n*max(conv(a,a)) / sum(a)^2
because the objective is scale invariant we normalize sum(a)=1 and minimize
    2*n*max(conv(a,a)).

Method:
  - several diverse initializations (provided sequence, ramps, asymmetric
    Gaussians, cosines, random),
  - gradient descent on a log-sum-exp smoothing of max(conva,a) (the standard
    smooth upper bound on the max, which is exactly the masked peak as T->0),
    with projection onto {a >= 0, sum a = 1},
  - the smoothed gradient's active set automatically focuses on the peak lags,
    - adaptive temperature, annealed learning rate, periodic stochastic
    perturbations to escape local minima,
  - keep the global best (verified with evaluate_sequence) and a final polish.
"""

import time
import math
import numpy as np


def _as_valid_array(x, fallback_n):
    """Coerce x to a 1-D float array of length >= 4, else build a fallback."""
    try:
        a = np.asarray(x, dtype=np.float64)
    except Exception:
        a = np.ones(fallback_n, dtype=np.float64)
    if a.ndim != 1:
        a = a.reshape(-1)
    if a.size < 4 or not np.isfinite(a).all():
        a = np.ones(fallback_n, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 1e-12:
        a = np.ones(fallback_n, dtype=np.float64)
    return a


def _normalize(a):
    """Non-negative with sum exactly 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-15:
        return np.full_like(a, 1.0 / a.size)
    return a / s


def _fast_score(a):
    """Exact objective for a non-negative sequence with sum(a)=1."""
    n = a.size
    C = np.convolve(a, a)
    return 2.0 * n * float(np.max(C))


def _surrogate_gradient(a, frac):
    """
    grad of g(a)=T*log(sum_k exp(C_k/T)) w.r.t. a,  T = frac*max(C).
    Returns (grad, Cmax) with grad having the same shape as a.
    Uses the closed form  grad[m]=sum_l a[l]*(W[m+l]+W[m-l])
    evaluated with two FFT-based convolutions (O(n log n)).
    """
    n = a.size
    C = np.convolve(a, a)
    Cmax = float(np.max(C))
    if Cmax <= 1e-18:
        m = 2 * n - 1
        z = np.zeros(m)
        z[m // 2] = 1.0
        return np.zeros_like(a), 0.0, z
    T = max(1e-12, frac * Cmax)
    w = np.exp((C - Cmax) / T)
    sw = w.sum()
    if sw <= 1e-18:
        w = np.zeros_like(C)
        w[int(np.argmax(C))] = 1.0
    else:
        w = w / sw
    # term1[m] = sum_l a[l]*W[m+l]  = conv(a, W[::-1], full)[2n-2-m]
    full1 = np.convolve(a, w[::-1], mode="full")
    term1 = full1[2 * n - 2 : n - 2 : -1]
    # term2[m] = sum_l a[l]*W[m-l]  = conv(W, a, full)[m]
    full2 = np.convolve(w, a, mode="full")
    term2 = full2[:n]
    grad = term1 + term2
    return grad, Cmax, w


def _project(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-15:
        return np.full_like(a, 1.0 / a.size)
    return a / s

        best = a.copy()

    eta_end = 0.0015
    frac = 0.40          # log-sum-exp focus (T = frac*maxC)
    frac_end = 0.05
    cool_eta = 0.9985
    cool_frac = 0.997

    it = 0
    best_it = it
    t_last = time.monotonic()
    ema = 0.0

def _make_inits(parent, n):
    """Diverse set of starting points (all non-negative, length n)."""
    inits = []
    inits.append(parent)

    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp / ramp.sum())
    inits.append(ramp[::-1].copy() / ramp.sum())

    # asymmetric Gaussians with different centers / widths
    for mu_frac, width in [(0.30, 0.18), (0.70, 0.18),
                           (0.50, 0.35), (0.40, 0.55), (0.60, 0.55)]:
        j = np.arange(n, dtype=np.float64)
        g = np.exp(-0.5 * ((j - mu_frac * (n - 1)) / (width * n)) ** 2)
        inits.append(g / g.sum())

    # cosine / sinc-like bandpass shapes (helps flatten central autocorr)
    j = np.arange(n, dtype=np.float64)
    c1 = (1.0 + 0.6 * np.sin(math.pi * j / n)) ** 2
    inits.append(np.clip(c1, 1e-9, None) / np.clip(c1, 1e-9, None).sum())
    c2 = (0.5 + 0.5 * np.cos(2.0 * math.pi * (j + 0.5) / n)) ** 2

    # alternating-ish damped mask (reduces central peak modestly)

    inits.append(np.clip(alt, 1e-6, None) / np.clip(alt, 1e-6, None).sum())

    return inits
