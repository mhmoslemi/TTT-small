def fft_self_convolve(a):
    """Compute full self-convolution of a using FFT. O(n log n)."""
    n = len(a)
    size = 2 * n
    fa = np.fft.rfft(a, n=size)
    c = np.fft.irfft(fa * fa, n=size)
    return c[:2*n-1]

def grad_Q(a, c):
    """Gradient of Q(a) = sum_k (c_k - 1/(2n-1))^2 where c=conv(a,a).
    Returns grad[i] = 2 * sum_j (c_{i+j} - u) * a_j where u = 1/(2n-1)."""
