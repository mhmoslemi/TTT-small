    """Compute linear convolution of a with itself."""
    return np.convolve(a, a, mode='full')

def _smoothed_max_and_grad(a, tau):
    """Compute smoothed max of conv(a,a) and its gradient w.r.t. a."""
    n = len(a)
    b = _compute_conv(a)  # length 2n-1

        # Track best using exact score
        if exact_score < best_score:
            best_score = exact_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart logic
        if no_improve >= patience:
            tau = max(3.0, tau * 0.5)
            lr = max(0.001, lr * 0.6)
            noise_scale = max(0.01, 0.1 / np.sqrt(step + 1))
            a = a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            s = a.sum()
            a = a / s if s > 1e-10 else np.ones(n) / n
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            patience = max(80, int(patience * 0.75))

        # Gradually increase tau (sharpen max approximation)
        if step % 300 == 0 and tau < 500.0:
            tau = min(tau * 1.15, 500.0)

        # Adaptive lr decay over time
        if step % 500 == 0:
            lr = max(0.001, lr * 0.85)

    return best_a, best_score


def _make_initialization(n, kind, rng):
    """Generate a diverse initial sequence of length n."""
    if kind == 0:
        # Geometric
        c = rng.uniform(0.5, 3.5)
        a = np.exp(c * np.arange(n) / max(1, n - 1))
    elif kind == 1:
        # Warped geometric (power on the exponent)
        p = rng.uniform(0.3, 2.5)
        c = rng.uniform(0.5, 3.0)
        x = np.arange(n) / max(1, n - 1)
        a = np.exp(c * np.power(x, p))
    elif kind == 2:
        # Geometric with sinusoidal modulation
        c = rng.uniform(0.5, 2.5)
        freq = rng.integers(1, 7)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.uniform(0.1, 0.5)
        x = np.arange(n) / max(1, n - 1)
        a = np.exp(c * x) * (1.0 + amp * np.sin(2.0 * np.pi * freq * x + phase))
        a = np.maximum(a, 0.01)
    elif kind == 3:
        # Exponential with quadratic term
        c = rng.uniform(0.5, 2.5)
        d = rng.uniform(-1.5, 1.5)
        x = np.arange(n) / max(1, n - 1)
        a = np.exp(c * x + d * x * x)
        a = np.maximum(a, 0.01)
    elif kind == 4:
        # Sum of two geometric components
        c1 = rng.uniform(0.3, 2.0)
        c2 = rng.uniform(0.3, 2.0)
        w = rng.uniform(0.2, 0.8)
        x = np.arange(n) / max(1, n - 1)
        a = w * np.exp(c1 * x) + (1.0 - w) * np.exp(c2 * x)
        a = np.maximum(a, 0.01)
    else:
        # Smooth random polynomial
        k = rng.integers(4, 10)
        coeffs = rng.normal(0, 1, k)
        x = np.linspace(0, 1, n)
        a = np.abs(np.polynomial.polynomial.polyval(x, coeffs)) + 0.1

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-restart gradient descent with log-sum-exp smoothed max for
    minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

    Uses Adam optimizer, diverse initializations (geometric, warped, modulated),
    and adaptive tau schedule. Scale invariance exploited by maintaining sum(a)=1.
    """
    main_rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0

    # Get parent info
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.ndim == 1 and parent.size > 100:
            parent_clean = np.clip(parent, 0.0, 1000.0)
            parent_len = parent.size
        else:
            parent_clean = None
            parent_len = 6962
    except Exception:
        parent_clean = None
        parent_len = 6962

    # Strategy: try parent length first, then a few others
    lengths = [parent_len]
    lengths.append(max(1500, parent_len // 3))
    lengths.append(min(12000, parent_len + 4000))
    # Deduplicate
    seen = set()
    unique_lengths = []
    for l in lengths:
        if l >= 500 and l not in seen:
            seen.add(l)
            unique_lengths.append(l)
    lengths = unique_lengths

    best_overall = None
    best_score_overall = np.inf
    attempt = 0

    for n in lengths:
        if time.monotonic() > deadline - 15.0:
            break

        remaining = deadline - time.monotonic()
        # Estimate how many full optimizations we can run
        # Each optimization on sequence of length n takes roughly n/500 seconds per step
        # and we get ~n/50 steps before hitting deadline
        # Be conservative: try at most 8 starts per length
        n_starts = min(10, max(2, int(remaining / 90)))

        for s in range(n_starts):
            if time.monotonic() > deadline - 5.0:
                break
            attempt += 1

            # Choose initialization type
            if parent_clean is not None and s == 0 and n == parent_len:
                a_init = parent_clean.copy()
                a_init = np.maximum(a_init, 0.0)
                a_init = a_init / a_init.sum()
            else:
                kind = main_rng.integers(0, 5)
                a_init = _make_initialization(n, kind, main_rng)

            # Run optimization
            result, score = _optimize_from(a_init, deadline - 2.0, seed + attempt * 7919)

            if score < best_score_overall:
                best_score_overall = score
                best_overall = result
                print(f"attempt {attempt}, n={n}, start={s}: score={score:.6f}")

        # Early stop if we're doing very well
        if best_score_overall < 1.55:
            print(f"Good score {best_score_overall:.6f} found, stopping length loop")
            break

    if best_overall is not None:
        final_score = float(evaluate_sequence(best_overall.tolist()))
        print(f"FINAL VERIFIED SCORE: {final_score:.6f}")
        return best_overall.tolist()

    # Fallback
    print("FALLBACK: returning ones(256)")
    return np.ones(256, dtype=float).tolist()
