"""
Minimizes 2n*max(convolve(a,a))/sum(a)^2 for non-negative sequences.
Uses smoothed-max gradient descent with simulated annealing,
multiple restarts, and adaptive learning rate.
Exploits scale invariance by normalizing sum(a) = sqrt(2n).
"""

import numpy as np
import time
from math import sqrt, exp

def _compute_gradient(a, c, k_star):
    """Compute subgradient of c[k_star] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(k_star, n - 1)
    if hi >= lo:
        g[lo:hi + 1] = a[k_star - lo:k_star - hi - 1:-1]
    return g

def _smoothed_max_grad(a, c, beta):
    """Compute gradient of log-sum-exp smoothed max of c."""
    n = len(a)
    bc = beta * c
    bc_max = bc.max()
    w = np.exp(bc - bc_max)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return 0.0, np.zeros(n)
    w = w / w_sum
    smooth_val = (bc_max + np.log(w_sum)) / beta

    # Gradient: g[i] = sum_k w[k] * (a[k-i] if valid) + (a[i-k] if valid)
    # Use FFT for efficiency
    # First term: (w * a_corr)[i] = sum_k w[k] * a[k-i]
    # This is cross-correlation of w and a, which is convolve(w, a_reversed)
    # Second term: sum_k w[k] * a[i-k] = convolve(w_reversed, a)[i]
    # But we need to handle bounds correctly. For "full" convolve:
    # c[k] = sum_{j} a[j]*a[k-j] for k in [0, 2n-2]
    # dc[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n) [when i != k-i]
    #              = 2*a[i] (when k=2i)
    
    # Simpler: just use top few peaks
    top_positions = np.argsort(c)[-20:]
    top_w = w[top_positions]
    top_w_sum = top_w.sum()
    if top_w_sum < 1e-300:
        return smooth_val, np.zeros(n)
    top_w = top_w / top_w_sum
    
    g = np.zeros(n)
    for k in top_positions:
        wt = top_w[list(top_positions).index(k)]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            g[lo:hi + 1] += wt * a[k - lo:k - hi - 1:-1]
    return smooth_val, g

def _block_perturb(a, rng, n):
    """Apply a random block scaling perturbation."""
    a = a.copy()
    nblocks = rng.integers(1, 4)
    for _ in range(nblocks):
        width = rng.integers(max(2, n // 100), max(3, n // 8))
        start = rng.integers(0, max(1, n - width))
        factor = np.exp(rng.normal(0, 0.15))
        a[start:start + width] *= factor
    a = np.clip(a, 1e-10, 1000)
    return a

def _gaussian_init(n, rng, norm_target):
    center = (n - 1) / 2.0
    sigma = rng.uniform(n * 0.1, n * 0.4)
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-(idx - center) ** 2 / (2 * sigma * sigma))
    a = np.clip(a, 1e-10, None)
    a = a / a.sum() * norm_target
    return a

def _random_init(n, rng, norm_target):
    a = rng.random(n) + 0.01
    a = a / a.sum() * norm_target
    return a

def _triangular_init(n, norm_target):
    center = (n - 1) / 2.0
    idx = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(idx - center) / (center + 0.5)
    a = np.clip(a, 0.01, None)
    a = a / a.sum() * norm_target
    return a
