    # (the second term comes from d(s^2)/da[i] = 2s, so d(1/s^2)/da[i] = -2/s^3)
    
    grad_sigma = 2.0 * c
    grad = 2.0 * n * grad_sigma / (s * s) - 4.0 * n * sigma / (s * s * s)
    
    # Adam update (simplified)
    return grad
