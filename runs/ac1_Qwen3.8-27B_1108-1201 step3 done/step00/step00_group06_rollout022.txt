def score_and_conv(a):
    """Compute score and full convolution using FFT."""
    n = len(a)
    S = np.sum(a)
    conv_len = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < conv_len:
        fft_size *= 2
    F = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(F * F, fft_size)[:conv_len]
    # Clean up tiny negative values from FFT roundoff
    conv = np.maximum(conv, 0)
    M = conv.max()
    score = 2.0 * n * M / (S * S)
    return score, conv, M
