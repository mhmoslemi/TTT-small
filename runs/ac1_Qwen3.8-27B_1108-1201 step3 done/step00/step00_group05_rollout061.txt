def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

            g = _grad_single_lag(a, k_worst)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break
            momentum = mu * momentum - alpha * (g / g_norm)
            a_new = _project_simplex(a + momentum)
            score_new = _obj(a_new)
            if score_new < best_score:
                best = a_new.copy()
                best_score = score_new
                print(f"  Refine: {best_score:.6f}")
            a = a_new
            alpha *= 0.999
    
    result = [float(max(0.0, min(1000.0, x))) for x in best]
    final_score = float(evaluate_sequence(result))
    print(f"Final: {final_score:.6f}")
    return result
