"""
Minimize 2n*max(f*f)/(sum a)^2 using L-BFGS with log-sum-exp smoothing,
projected gradient descent with momentum, and multiple initializations.
Exploits the fact that non-symmetric sequences (like ramps) achieve J < 2.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def _fft_convolution(a):
    """Compute linear convolution a*a using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)
    return result[:size]


def _cross_corr_via_fft(p, a):
    """
    Compute sum_k p[k] * a[k-j] for j=0,...,len(a)-1.
    p has length 2*len(a)-1, a has length len(a).
    Returns array of length len(a).
    """
    n = len(a)
    a_flip = a[::-1]
    needed = len(p) + n - 1
    fft_size = 1
    while fft_size < needed:
        fft_size <<= 1
    P = rfft(p, fft_size)
    AF = rfft(a_flip, fft_size)
    conv = irfft(P * AF, fft_size)
    return conv[n - 1: n - 1 + n]


def _smooth_max_and_soft_pmf(b, alpha):
    """
    Compute smooth max of b and the softmax weights p_k.
    Returns (smooth_max_value, p).
    """
    b_max = np.max(b)
    shifted = alpha * (b - b_max)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    if total < 1e-300:
        total = 1e-300
    log_sum = np.log(total)
    smooth_val = b_max + log_sum / alpha
    p = exp_shifted / total
    return smooth_val, p


def _exact_eval(a):
    """Exact objective value for non-negative array a with sum >= 0.01."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e30
    b = _fft_convolution(a)
    b = np.maximum(b, 0.0)
    M = np.max(b)
    return 2.0 * n * M / (S * S)


def _simplex_project(z):
    """Project z onto the probability simplex {v >= 0, sum(v) = 1}."""
    n = len(z)
    sorted_z = np.sort(z)[::-1]
    cumsum = np.cumsum(sorted_z)
    idx = np.arange(1, n + 1)
    condition = sorted_z - (cumsum - 1.0) / idx > 0
    if not np.any(condition):
        return np.ones(n) / n
    rho = np.max(idx[condition])
    theta = (cumsum[rho - 1] - 1.0) / rho
    return np.maximum(z - theta, 0.0)


def _make_ramp(n):
    """Create a ramp sequence normalized to sum=1."""
    a = np.arange(1, n + 1, dtype=np.float64)
    a /= np.sum(a)
    return a


def _make_triangle(n):
    """Create a triangle sequence normalized to sum=1."""
    if n % 2 == 1:
        a = np.abs(np.arange(n) - n // 2)
        a = (n // 2) - a + 1
    else:
        a = np.abs(np.arange(n) - (n - 1) / 2.0)
        a = (n - 1) / 2.0 - a + 1
    a = np.maximum(a, 0.0)
    a /= np.sum(a)
    return a


def _make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian sequence normalized to sum=1."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a /= np.sum(a)
    return a


def _make_cosine(n):
    """Create a raised cosine sequence normalized to sum=1."""
    x = np.linspace(0.0, np.pi, n)
    a = 0.5 * (1.0 + np.cos(x))
    a = a ** 2
    a /= np.sum(a)
    return a


def _lbgfs_smooth_objective(a, alpha):
    """
    Smooth objective: 2n * smooth_max(f*f) / S^2
    Returns (objective_value, gradient).
    """
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return 1e30, np.zeros_like(a)

    b = _fft_convolution(a)
    b = np.maximum(b, 0.0)

    smooth_val, p = _smooth_max_and_soft_pmf(b, alpha)
    obj = 2.0 * n * smooth_val / (S * S)

    # grad_sm_j = 2 * sum_k p[k] * a[k-j]
    cross = _cross_corr_via_fft(p, a)
    grad_sm = 2.0 * cross

    # Full gradient
    grad = (2.0 * n / (S * S)) * (grad_sm - 2.0 * smooth_val / S * a)

    return obj, grad


def _run_lbfgs(a0, n, alpha, maxiter, deadline):
    """Run L-BFGS-B on the smooth objective starting from a0."""
    a0 = np.maximum(a0, 0.0)
    S0 = np.sum(a0)
    if S0 < 0.01:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / S0  # normalize to sum=1 initially

        x0=a0.copy(),

        jac=True,

        options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8,
                 'iprint': -1, 'maxls': 50, 'maxfun': 10 * n}
    )

    a_opt = np.maximum(result.x, 0.0)
    # Evaluate with exact objective
    score = _exact_eval(a_opt)
    if not np.isfinite(score):
        score = 1e30
    return a_opt, score


def _run_projected_grad(a0, n, alpha, deadline, lr_decay=0.999, momentum_decay=0.9):
    """
    Projected gradient descent with momentum on the simplex.
    Minimizes 2n * smooth_max(f*f) with sum(a)=1, a>=0.
    """
    a = a0.copy()
    S = np.sum(a)
    if S < 0.01:
        a = np.ones(n) / n
    else:
        a = a / S

    momentum = np.zeros_like(a)
    lr = 0.1
    best_a = a.copy()
    best_score = _exact_eval(a)

    step_count = 0
    no_improve = 0

    while time.monotonic() < deadline:
        # Compute gradient of 2n * smooth_max(f*f) wrt a (with sum(a)=1)
        b = _fft_convolution(a)
        b = np.maximum(b, 0.0)
        smooth_val, p = _smooth_max_and_soft_pmf(b, alpha)

        cross = _cross_corr_via_fft(p, a)
        grad = 4.0 * n * cross  # gradient of 2n * smooth_max

        # Project gradient onto tangent space of simplex (remove mean component)
        grad_mean = np.mean(grad)
        grad_tangent = grad - grad_mean

        # Momentum update
        momentum = momentum_decay * momentum + grad_tangent
        a_new = a - lr * momentum

        # Project onto simplex
        a_new = _simplex_project(a_new)

        # Evaluate
        score = _exact_eval(a_new)
        if np.isfinite(score) and score < best_score:
            best_a = a_new.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                lr *= 0.5
                no_improve = 0

        a = a_new
        lr = max(lr * lr_decay, 1e-8)
        step_count += 1

        if step_count % 500 == 0:
            print(f"  PGD step {step_count}: score={best_score:.6f}, lr={lr:.2e}")

    return best_a, best_score
