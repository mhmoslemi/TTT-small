"""
Subgradient optimization for minimizing 2n*max(auto-convolution) of non-negative sequences.
Uses FFT-based convolution, vectorized subgradient, momentum, and multiple restarts.
"""

import numpy as np
import time


def fft_convolve(a, b=None):
    """Full convolution using FFT."""
    if b is None:
        b = a
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    result = np.fft.irfft(A * B, N)
    return result[:n]


def compute_subgradient(b, C, active_lags):
    """Compute subgradient of max_k C[k] at point b.
    
    active_lags: array of indices where C achieves max (or near max).
    Returns gradient vector of same length as b.
    """
    n = len(b)
    g = np.zeros(n)
    m = len(active_lags)
    for k in active_lags:
        k = int(k)
        if k < n:
            # g[0:k+1] += 2*b[k::-1]
            if k == 0:
                g[0] += b[0]  # k=0=2*0, so b[0] not 2*b[0]
            else:
                g[:k+1] += 2.0 * b[k::-1]
                # Correction for k = 2j where j in [0, k]
                if k % 2 == 0 and k // 2 <= k:
                    g[k // 2] -= b[k // 2]  # was 2*b[k//2], should be b[k//2]
        else:
            # g[k-n+1:n] += 2*b[k-n+1:n][::-1]
            lo = k - n + 1
            hi = n - 1
            g[lo:hi+1] += 2.0 * b[lo:hi+1][::-1]
            # Correction for k = 2j
            if k % 2 == 0:
                j = k // 2
                if lo <= j <= hi:
                    g[j] -= b[j]
    g /= m
    return g


def opt_shape(n, budget_s, seed=42, init_type='uniform'):
    """Optimize shape for given n within time budget."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    
    # Initialize
    if init_type == 'uniform':
        b = np.ones(n) / n
    elif init_type == 'triangle':
        i = np.arange(n)
        b = np.minimum(i + 1, n - i)
        b /= b.sum()
    elif init_type == 'cosine':
        i = np.arange(n)
        b = 0.5 * (1 + np.cos(np.pi * i / (n - 1)))
        b = np.maximum(b, 1e-10)
        b /= b.sum()
    elif init_type == 'u_shape':
        i = np.arange(n)
        b = 1.0 + 0.5 * np.cos(2 * np.pi * i / n)
        b = np.maximum(b, 1e-10)
        b /= b.sum()
    elif init_type == 'random_smooth':
        b = np.abs(rng.standard_normal(n)) 
        # Smooth it
        from scipy.ndimage import gaussian_filter1d
        b = gaussian_filter1d(b, sigma=5)
        b = np.maximum(b, 0)
        b /= b.sum()
    else:
        b = np.ones(n) / n
    
    best_b = b.copy()
    best_score = 2.0 * n * fft_convolve(b).max()
    
    # Subgradient descent with momentum
    velocity = np.zeros(n)
    mu = 0.9  # momentum
    lr0 = 0.01
    decay = 0.001
    
    t = 0
    no_improve = 0
    patience = 2000
    
    while time.monotonic() < deadline:
        # Compute convolution
        C = fft_convolve(b)
        max_val = C.max()
        score = 2.0 * n * max_val
        
        if score < best_score:
            best_score = score
            best_b = b.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find active lags
        tol = 1e-8 * max_val
        active = np.where(C >= max_val - tol)[0]
        
        # Compute subgradient
        g = compute_subgradient(b, C, active)
        
        # Learning rate schedule
        lr = lr0 / (1.0 + t * decay)
        
        # Momentum update
        velocity = mu * velocity - lr * g
        b = b + velocity
        
        # Project: clip to >= 0, normalize
        b = np.maximum(b, 0.0)
        s = b.sum()
        if s < 1e-15:
            b = np.ones(n) / n
            velocity = np.zeros(n)
        else:
            b /= s
        
        t += 1
        
        # Periodic random perturbation to escape local minima
        if no_improve > patience:
            # Add random perturbation
            b += 0.1 * rng.standard_normal(n)
            b = np.maximum(b, 0.0)
            b /= b.sum()
            velocity = np.zeros(n)
            no_improve = 0
            patience = max(500, patience // 2)
    
    return best_b, best_score
