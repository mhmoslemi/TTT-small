```python
def smooth_max_logsumexp(b, beta):
    """Smooth approximation of max(b) using log-sum-exp."""
    # For numerical stability
    b_max = np.max(b)
    return b_max + np.log(np.sum(np.exp(beta * (b - b_max)))) / beta

def compute_objective_and_grad(a, beta):
    """Compute smooth objective and gradient for normalized sequence."""
    n = len(a)
    b = np.convolve(a, a, mode='full')  # length 2n-1
    # Smooth max
    b_max = np.max(b)
    exp_b = np.exp(beta * (b - b_max))
    log_sum = b_max + np.log(np.sum(exp_b)) / beta
    # Objective: 2n * smooth_max (since sum_a = 1)
    obj = 2 * n * log_sum
    # Gradient of smooth max w.r.t. a:
    # d/d a_j [ (1/beta) log sum_k exp(beta * b_k) ]
    # = sum_k p_k * db_k/da_j
    # where p_k = exp(beta*(b_k - b_max)) / sum(exp(beta*(b - b_max)))
    p = exp_b / np.sum(exp_b)  # softmax weights
    # db_k/da_j = a[k-j] if 0 <= k-j < n
    # gradient_j = sum_k p_k * a[k-j] (for valid k-j)
    # This is correlate(p, a, mode='valid')
    grad = np.correlate(p, a, mode='valid')
    # Full objective gradient: 2n * grad
    grad = 2 * n * grad
    return obj, grad

def project_to_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def projected_gradient_step(a, beta, lr):
    """One step of projected gradient descent."""
    obj, grad = compute_objective_and_grad(a, beta)
    a_new = a - lr * grad
    a_new = project_to_simplex(a_new)
    return a_new, obj, grad

def optimize_sequence(n, init, beta_start, beta_end, n_iters, lr_start, lr_end, seed=42):

    a = init.copy()

        if actual_obj < best_obj:
            best_obj = actual_obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj
```

```python
import scipy.signal

def project_to_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n

    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def smooth_obj_grad(a, beta):
    """Compute smooth objective and gradient (assuming sum(a)=1)."""
    n = len(a)
    b = scipy.signal.fftconvolve(a, a, mode='full')
    b_max = np.max(b)
    exp_b = np.exp(beta * (b - b_max))

    log_sum = b_max + np.log(s) / beta
    obj = 2.0 * n * log_sum
    grad = 2.0 * n * np.correlate(p, a, mode='valid')
    return obj, grad, b

def exact_obj(a):
    """Compute exact objective (assuming sum(a)=1)."""
    n = len(a)
    b = scipy.signal.fftconvolve(a, a, mode='full')
    return 2.0 * n * np.max(b)

def optimize(n, init, time_budget, seed=42):

    rng = np.random.default_rng(seed)

    best_a = a.copy()

    beta_current = 1.0
```

```python
import numpy as np
import time
import scipy.signal

def project_to_simplex(v):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def exact_score_normalized(a):
    """Score assuming sum(a) = 1. Returns 2n * max(conv(a,a))."""
    n = len(a)
    b = scipy.signal.fftconvolve(a, a, mode='full')
    return 2.0 * n * np.max(b)

def smooth_obj_grad(a, beta):
    """Smooth objective and gradient assuming sum(a)=1."""
    n = len(a)
    b = scipy.signal.fftconvolve(a, a, mode='full')
    bmax = np.max(b)
    e = np.exp(beta * (b - bmax))
    s = np.sum(e)
    p = e / s
    obj = 2.0 * n * (bmax + np.log(s) / beta)
    grad = 2.0 * n * np.correlate(p, a, mode='valid')
    return obj, grad

def optimize_fixed_n(n, init, time_budget, seed=0):
    a = project_to_simplex(np.asarray(init, dtype=np.float64))
    best_a = a.copy()
    best_score = exact_score_normalized(a)
    start = time.monotonic()
    
    # Adaptive projected gradient
    beta = 5.0
    lr = 0.1 / n
    stall = 0
    best_lr = lr
    
    while time.monotonic() - start < time_budget:
        obj, grad = smooth_obj_grad(a, beta)
        
        # Backtracking line search
        alpha = lr
        success = False
        for _ in range(30):
            a_new = project_to_simplex(a - alpha * grad)
            obj_new, _ = smooth_obj_grad(a_new, beta)
            if obj_new < obj:
                a = a_new
                success = True
                break
            alpha *= 0.5
        if not success:
            # Try smaller beta or restart
            beta = max(1.0, beta * 0.5)
            continue
        
        # Check exact score
        score = exact_score_normalized(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Anneal beta (increase for sharper approximation)
        t = (time.monotonic() - start) / time_budget
        beta = min(100.0, 5.0 * (1 + 10 * t))
        
        # Decayed learning rate
        lr = lr * 0.999
    
    return best_a, best_score
```
