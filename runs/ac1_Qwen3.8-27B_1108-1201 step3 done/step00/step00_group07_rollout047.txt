    """One step of subgradient descent on max(conv)."""
    n = len(p)
    c = fft_convolve_same(p)
    max_val = np.max(c)
    
    # Get argmax set (all indices where c is very close to max)
    # Use a small tolerance to group nearby peaks
    tol = 1e-8
    argmax_mask = c >= max_val - tol
    S = np.where(argmax_mask)[0]
    
    if len(S) == 0:
        S = np.array([np.argmax(c)])
    
    # Compute subgradient: g[i] = (1/|S|) * sum_{k in S} p[k-i] (with p[j]=0 if j out of [0,n))
    g = np.zeros(n)
    for k in S:
        # p[k-i] is nonzero when 0 <= k-i < n, i.e., k-n+1 <= i <= k
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            g[i_start:i_end+1] += p[k - np.arange(i_start, i_end + 1)]
    g /= len(S)
    
    # Update

    p_new = np.maximum(p_new, 0)
    s = np.sum(p_new)
    if s < 1e-10:

    else:

But wait, I realize there might be a fundamental issue with subgradient descent. The function max_k c(k) is not smooth, and the subgradient can point in a direction that reduces the current maximum but increases another component. This can cause oscillation.

A better approach might be to use a "smoothing" of the max function, like the log-sum-exp:
f(p) = (1/τ) log(sum_k exp(τ * c(k)))

For large τ, this approximates the max. The gradient is smooth:
∂f/∂p_i = (1/τ) * sum_k c_k'(i) * exp(τ*c(k)) / sum_k exp(τ*c(k))
= sum_k c_k'(i) * softmax(τ*c)[k]

where c_k'(i) = p[k-i] (the derivative of c(k) w.r.t. p_i).

This is a weighted average of the subgradients, weighted by the softmax of the convolution values. This is much smoother and should converge better.

def logsumexp_grad(p, tau):
    """Compute gradient of (1/tau) * log(sum exp(tau * c(k)))."""
    n = len(p)
    c = fft_convolve_same(p)
    
    # Softmax weights
    max_c = np.max(c)
    exp_c = np.exp(tau * (c - max_c))
    weights = exp_c / np.sum(exp_c)
    
    # Gradient: sum_k weights[k] * p[k-i] for each i
    # g[i] = sum_k weights[k] * p[k-i]
    g = np.zeros(n)
    for k in range(2*n-1):
        if weights[k] < 1e-15:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            g[i_start:i_end+1] += weights[k] * p[k - np.arange(i_start, i_end + 1)]
    
    return g, np.max(c)
