```
def _make_fourier_x(n, seed, slope=0.0, freqs=(1,2,3), amp=1.0):
    rng = np.random.default_rng(seed)
    t = np.arange(n)
    x = np.zeros(n)
    if slope !=0:
        x += slope * (t/(n-1) - 0.5)
    for f in freqs:
        phase = rng.uniform(0, 2*np.pi)
        coefficient = float(rng.uniform(0.5, 1.5)) * amp
        x += coefficient * np.cos(2*np.pi*f*t/n + phase)
    x -= x.mean()
    return x
```

```
def _convolve_fft(a, n, L):
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)
    c = c[:2*n-1]
    np.maximum(c, 0.0, out=c)
```

```python
"""
Search for low-peak self-convolution step functions.

Strategy:
- Work in log-space with softmax parameterization.
- Initialize from optimized geometric tapers and a small grid of smooth
  asymmetric shapes.
- Use Adam-style gradient descent on a smoothed max convolution objective,
  with adaptive temperature, line search, and restart/noise mechanisms.
"""

import numpy as np
import time


def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _softmax(x):
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return x.copy()
    x = np.where(np.isfinite(x), x, 0.0)
    xm = x - np.max(x)
    e = np.exp(xm)
    s = e.sum()
    if not np.isfinite(s) or s <= 0.0:
        return np.full_like(x, 1.0 / float(x.size))
    return e / s


def _convolve_fft(a, n, L):
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)
    c = c[:2 * n - 1].copy()
    np.maximum(c, 0.0, out=c)
    return c


def _grad_from_weights(a, w, n, Lg):
    # Gradient for summed weighted self-convolution terms.
    # For c_k = sum_i a_i a_{k-i}, dc_k/da_j = 2*a_{k-j} when valid.
    rw = w[::-1]
    Af = np.fft.rfft(a, Lg)
    Wf = np.fft.rfft(rw, Lg)
    conv = np.fft.irfft(Af * Wf, Lg)
    seg = conv[n - 1:2 * n - 1]
    seg = seg[::-1].copy()
    np.maximum(seg, 0.0, out=seg)
    return 2.0 * seg


def _smooth_value(c, temp):
    cmax = float(np.max(c))
    if not np.isfinite(cmax):
        return float("inf")
    if temp <= 1e-300:
        return cmax
    with np.errstate(over="ignore", under="ignore", divide="ignore", invalid="ignore"):
        shifted = (c - cmax) / temp
        e = np.exp(shifted)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return cmax
    return float(temp * (cmax + np.log(s)))


def _make_geometric_x(n, y, reverse=False):
    t = np.arange(n, dtype=np.float64)
    rho = float(y) / float(max(1, n))
    if reverse:
        x = rho * t
    else:
        x = -rho * t
    x -= np.mean(x)
    return np.clip(x, -20.0, 20.0)


def _geometric_best_y(n, ymin=0.4, ymax=2.6, num=120):
    best_y = 1.05
    best_score = float("inf")
    k = np.arange(0, 2 * n - 1, dtype=np.float64)
    count = np.minimum(k + 1.0, 2.0 * n - 1.0 - k)
    logcount = np.log(count)

    for y in np.linspace(ymin, ymax, num):
        rho = float(y) / float(n)
        one_minus_r = -np.expm1(-rho)
        denom = 1.0 - np.exp(-float(y))
        if one_minus_r <= 0.0 or denom <= 0.0:
            continue
        scale = one_minus_r / denom
        logc = 2.0 * np.log(scale) - rho * k + logcount
        mx = float(np.max(logc))
        if not np.isfinite(mx):
            continue
        score = 2.0 * float(n) * np.exp(mx)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_y = float(y)
    return best_y


def _make_fourier_x(n, seed, slope=0.0, freqs=(1, 2, 3), amp=1.0):
    rng = np.random.default_rng(seed)
    t = np.arange(n, dtype=np.float64)
    x = np.zeros(n, dtype=np.float64)

    if slope != 0.0:
        x += float(slope) * (t / float(max(1, n - 1)) - 0.5)

    two_pi = 2.0 * np.pi
    nn = float(max(1, n))
    for f in freqs:
        phase = float(rng.uniform(0.0, two_pi))
        coef = float(rng.uniform(0.55, 1.45)) * float(amp)
        x += coef * np.cos(two_pi * float(f) * t / nn + phase)

    x -= np.mean(x)
    return np.clip(x, -20.0, 20.0)


def _parameterized_starts(n, seed, max_evals=220, time_limit=12.0, top_k=4):
    L = _next_pow2(2 * n - 1)
    t = np.arange(n, dtype=np.float64) / float(n)

    slopes = (-2.5, -1.9, -1.4, -1.1, -0.7, -0.35, 0.0,
              0.35, 0.7, 1.1, 1.4, 1.9, 2.5)
    amps = (0.0, 0.35, 0.80, 1.40)
    freqs = (1, 2, 3, 4)
    phases = (0.0, np.pi / 4.0, np.pi / 2.0, 3.0 * np.pi / 4.0,
              np.pi, 5.0 * np.pi / 4.0)

    results = []
    start_time = time.monotonic()
    count = 0
    done = False
    idx = 0
    two_pi = 2.0 * np.pi

    for slope in slopes:
        base = float(slope) * (t - 0.5)
        for A in amps:
            for f in freqs:
                for phi in phases:
                    if done:
                        break

                    if A == 0.0 and phi != phases[0]:
                        continue

                    if A == 0.0:
                        x = base.copy()
                    else:
                        x = base + float(A) * np.cos(two_pi * float(f) * t + float(phi))

                    a = _softmax(x)
                    c = _convolve_fft(a, n, L)
                    sc = 2.0 * float(n) * float(c.max())
                    if np.isfinite(sc):
                        results.append((float(sc), idx, x))
                    idx += 1
                    count += 1

                    if count >= max_evals or time.monotonic() > start_time + time_limit:
                        done = True

                if done:
                    break
            if done:
                    break
        if done:
            break

    if not results:
        return []

    results.sort()
    top = []
    for sc, _, x in results[:max(1, top_k)]:
        top.append(x)
    return top


def _sequence_to_x(seq, floor=1e-30):
    arr = np.asarray(seq, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.zeros(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if s <= 1e-12:
        return np.zeros(arr.size, dtype=np.float64)
    a = arr / s
    x = np.log(np.maximum(a, floor))
    x -= np.mean(x)
    return np.clip(x, -30.0, 30.0)


def _safe_sequence_score(a):
    if a is None:
        return None, float("inf")
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0 or arr.size > 50000:
        return None, float("inf")
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        return None, float("inf")
    arr = arr / s
    n = arr.size
    L = _next_pow2(2 * n - 1)
    c = _convolve_fft(arr, n, L)
    score = 2.0 * float(n) * float(c.max())
    if not np.isfinite(score):
        return None, float("inf")
    return arr, float(score)


def _consider_sequence(a, best_a, best_score):
    a2, sc = _safe_sequence_score(a)
    if a2 is None:
        return best_a, best_score
    if sc < best_score:
        return a2, sc
    return best_a, best_score


def _fallback_sequence(n=4096):
    y = _geometric_best_y(n)
    return _softmax(_make_geometric_x(n, y, reverse=False))

                if score > best_score * 1.02:
                    x = best_x + _make_fourier_x(
                        n, int(rng.integers(0, 1000000000)), freqs=(1, 2), amp=0.08
                    )
                else:
                    x = x + _make_fourier_x(
                        n, int(rng.integers(0, 1000000000)), freqs=(1, 2), amp=0.05
                    )
                x = np.where(np.isfinite(x), x, 0.0)
                x = np.clip(x - np.mean(x), -20.0, 20.0)
                a = _softmax(x)
                c = _convolve_fft(a, n, L)
                score = float(2.0 * n * c.max())

                if not np.isfinite(score) or score > best_score * 1.05:
                    x = best_x.copy()
                    a = best_a.copy()
                    c = _convolve_fft(a, n, L)
                    score = best_score

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    budget_s = float(budget_s)
    if budget_s < 60.0:
        n = 2048 if budget_s < 15.0 else 4096
        seq = _fallback_sequence(n)
        seq = np.clip(np.asarray(seq, dtype=np.float64), 0.0, 1000.0)
        s = float(seq.sum())
        if s > 0.0:
            seq = seq / s
        return [float(x) for x in seq]

    deadline = time.monotonic() + max(30.0, budget_s - 5.0)
    best_a = None
    best_score = float("inf")

    n_main = 8192 if budget_s >= 1000.0 else 4096
    y_main = _geometric_best_y(n_main)

    x_geo_f = _make_geometric_x(n_main, y_main, reverse=False)
    x_geo_b = _make_geometric_x(n_main, y_main, reverse=True)

    best_a, best_score = _consider_sequence(_softmax(x_geo_f), best_a, best_score)
    best_a, best_score = _consider_sequence(_softmax(x_geo_b), best_a, best_score)
    print(f"initial geometric best={best_score:.6f}", flush=True)

    top_x = _parameterized_starts(
        n_main,
        seed,
        max_evals=240,
        time_limit=12.0,
        top_k=4,
    )
    for xg in top_x:
        best_a, best_score = _consider_sequence(_softmax(xg), best_a, best_score)
    print(f"after grid starts best={best_score:.6f}", flush=True)

    configs = []
    configs.append((x_geo_f, n_main, int(seed)))
    configs.append((x_geo_b, n_main, int(seed + 1)))

    for j, xg in enumerate(top_x[:3]):
        configs.append((xg, n_main, int(seed + 10 + j)))

    configs.append((
        x_geo_f + _make_fourier_x(n_main, int(seed + 101), freqs=(1, 2, 3), amp=0.5),
        n_main,
        int(seed + 111)
    ))
    configs.append((
        x_geo_b + _make_fourier_x(n_main, int(seed + 102), freqs=(1, 2, 3), amp=0.5),
        n_main,
        int(seed + 112)
    ))
    configs.append((
        _make_fourier_x(n_main, int(seed + 103), slope=1.1, freqs=(1, 2, 3), amp=1.0),
        n_main,
        int(seed + 113)
    ))

    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if 1024 <= prev.size <= 20000:
            p_arr, p_score = _safe_sequence_score(prev)
            if p_arr is not None and p_score < 1.995:
                configs.append((_sequence_to_x(prev), prev.size, int(seed + 991)))
    except Exception:
        pass

    configs = configs[:8]

    caps = (720.0, 420.0, 260.0, 180.0, 120.0, 90.0, 60.0, 40.0)
    fracs = (0.55, 0.30, 0.18, 0.12, 0.08, 0.06, 0.04, 0.03)

    idx = 0
    while idx < len(configs) and time.monotonic() < deadline - 60.0:
        now = time.monotonic()
        remaining = deadline - now
        if remaining < 60.0:
            break

        if idx < len(caps):
            alloc = min(remaining - 30.0, caps[idx])
            if idx < len(fracs):
                alloc = min(alloc, remaining * fracs[idx])
        else:
            alloc = min(remaining - 30.0, 40.0)
        alloc = max(25.0, alloc)

        x0, n, cs = configs[idx]
        sub_deadline = now + alloc

        a_opt, sc_opt = _optimize_from_x(
            x0, n, sub_deadline, seed=cs
        )
        if a_opt is not None:
            best_a, best_score = _consider_sequence(a_opt, best_a, best_score)

        print(
            f"config {idx} n={n} new_best={best_score:.6f} "
            f"left={deadline - time.monotonic():.1f}s",
            flush=True
        )
        idx += 1

    extra = 0
    while (
        best_a is not None
        and time.monotonic() < deadline - 100.0
        and extra < 3
    ):
        now = time.monotonic()
        remaining = deadline - now
        if remaining < 120.0:
            break

        n_best = best_a.size
        amp = 0.22 / (1.0 + float(extra))
        x_extra = _sequence_to_x(best_a) + _make_fourier_x(
            n_best, int(seed + 900 + 10 * extra), freqs=(1, 2, 3), amp=amp
        )

        if extra == 0:
            alloc = min(max(40.0, remaining - 40.0), 360.0)
            lr = 0.16
            pat = 40.0
        elif extra == 1:
            alloc = min(max(40.0, remaining - 40.0), 220.0)
            lr = 0.12
            pat = 30.0
        else:
            alloc = min(max(40.0, remaining - 40.0), 140.0)
            lr = 0.08
            pat = 25.0

        sub_deadline = now + alloc
        a_opt, sc_opt = _optimize_from_x(
            x_extra, n_best, sub_deadline,
            seed=int(seed + 800 + extra),
            lr_init=lr,
            patience_s=pat,
            max_patience=5
        )
        if a_opt is not None:
            best_a, best_score = _consider_sequence(a_opt, best_a, best_score)
        print(f"polish {extra} best={best_score:.6f}", flush=True)
        extra += 1

    if best_a is None:
        best_a = _fallback_sequence(4096)
        _, best_score = _safe_sequence_score(best_a)

    seq = np.asarray(best_a, dtype=np.float64).ravel()
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq = np.clip(seq, 0.0, 1000.0)
    s = float(seq.sum())
    if s < 0.01:
        seq = _fallback_sequence(4096)
        s = float(seq.sum())
    if s > 0.0:
        seq = seq / s

    if time.monotonic() < deadline - 8.0:
        try:
            sc = float(evaluate_sequence(seq.tolist()))
            if not np.isfinite(sc):
                seq = _fallback_sequence(4096)
                seq = np.clip(np.asarray(seq, dtype=np.float64), 0.0, 1000.0)
                s = float(seq.sum())
                if s > 0.0:
                    seq = seq / s
            elif sc > max(best_score * 1.05, 2.05):
                seq = _fallback_sequence(4096)
                seq = np.clip(np.asarray(seq, dtype=np.float64), 0.0, 1000.0)
                s = float(seq.sum())
                if s > 0.0:
                    seq = seq / s
                best_score = _safe_sequence_score(seq)[1]
        except Exception:
            pass

    return [float(x) for x in seq]
```
