import time

def _convolve_peaks(a, top_m=1):
    """Compute convolution and return top-m peaks with locations."""
    b = np.convolve(a, a)
    if top_m >= len(b):
        return b, np.argsort(b)[::-1][:top_m], np.max(b)
    top_idx = np.argpartition(b, -top_m)[-top_m:]
    top_idx = top_idx[np.argsort(b[top_idx])[::-1]]
    return b, top_idx, b[top_idx[0]]

def _grad_weighted(a, k_indices, weights):
    """Compute weighted sum of gradients of b_k for given k's."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(k_indices, weights):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * 2.0 * a[k - idx]
    return grad

def _project_onto_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.zeros(n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

import time

def _score(a):
    """Compute the score (2n * max_b) for a sequence with sum=1."""
    n = len(a)
    b = np.convolve(a, a)
    return 2.0 * n * np.max(b)

def _grad_at_k(a, k):
    """Gradient of b_k w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]
    return grad

def _project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _make_two_level(n, eps, split=0.5):

    a = np.ones(n)
    split_idx = int(n * split)
    a[:split_idx] = 1.0 + eps
    a[split_idx:] = 1.0 - eps

    S = np.sum(a)

        else:
            no_improve += 1
        
        if no_improve > 200:
            eta *= 0.8
            no_improve = 0
            if eta < 1e-8:
                break
        
        # Top M indices
        if M >= len(b):
            k_idx = np.arange(len(b))
            wts = np.ones(len(b)) / len(b)
        else:
            k_idx = np.argpartition(b, -M)[-M:]
            k_idx = k_idx[np.argsort(b[k_idx])[::-1]]
            b_sel = b[k_idx]
            # Soft weights
            wts = np.exp(-3.0 * (b_max - b_sel) / (b_max + 1e-15))
            wts /= np.sum(wts)
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for k, w in zip(k_idx, wts):
            grad += w * _grad_at_k(a, k)
        
        # Center the gradient (for sum constraint)
        grad -= np.mean(grad)
        
        # Update
        a = a - eta * grad
        a = _project_simplex(a, 1.0)
        S = np.sum(a)
        if S < 1e-10:
            a = best.copy()
            break
        a /= S
    
    return best, best_score

Let me think about what the optimal sequence looks like. The condition for optimality (KKT conditions) is:
- For the active constraint `b_{k*} = max b_k`: `∇b_{k*}` is in the dual cone of the feasible set.
- The feasible set is `{a : sum a = 1, a ≥ 0}`.
- The dual cone condition means: `∇b_{k*} = λ * 1 + μ` where `λ` is the multiplier for the sum constraint and `μ_i ≥ 0` with `μ_i = 0` if `a_i > 0` (complementary slackness).

So for the optimal sequence, the gradient of the active `b_k` should be constant on the support of `a` (i.e., where `a_i > 0`).

`∇b_k / ∂a_i = 2 * a_{k-i}` (for valid indices).

So the condition is: `a_{k*-i}` is constant for all `i` in the support of `a` (where `a_i > 0`).

If `a` has full support (all elements positive), then `a_{k*-i}` must be constant for all `i`. This means `a` is constant, i.e., uniform! But the uniform sequence gives score = 2, not optimal.

This means the optimal sequence does NOT have full support. Some elements must be zero.

If `a_i = 0` for some `i`, then the complementary slackness says `μ_i ≥ 0`, which means the gradient condition is relaxed for those `i`.

So the optimal sequence has some zeros, and on the support, `a_{k*-i}` is constant.

This is a very restrictive condition. Let me think about what kind of sequence satisfies it.

Suppose the support is `S = {i : a_i > 0}`. Then for `k* = n-1` (a common choice), the condition is:
`a_{n-1-i} = c` for all `i ∈ S`.

This means that `a` is constant on the "reflected support" `S' = {n-1-i : i ∈ S}`.

If `S = {0, 1, ..., m}` for some m, then `S' = {n-1-m, ..., n-1}`. The condition is that `a` is constant on `S'`. But `a` is also nonzero on `S`. If `S` and `S'` overlap, then `a` must be constant on the overlap.
