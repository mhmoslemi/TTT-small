"""
Hybrid optimization to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Uses: (1) gradient descent on sum(b^2) for fast flattening across many lengths,
(2) LP-based block coordinate refinement (SOTA method) for moderate n,
(3) smoothed-max gradient descent for final polishing.
Multi-restart with diverse initializations.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def flatten_objective(a):
    """Compute sum of squares of convolution as a smooth proxy for max."""
    b = np.convolve(a, a)
    return np.sum(b * b)


def flatten_gradient(a):
    """Gradient of sum(convolve(a,a)^2) w.r.t. a."""
    b = np.convolve(a, a)
    return 4.0 * np.correlate(b, a, mode='valid')


def gradient_descent_flatten(a, deadline, rng, max_iters=20000):
    """Gradient descent on sum(b^2) with projection and adaptive step."""
    n = len(a)
    S0 = np.sum(a)
    lr = 1.0 / n
    best_a = a.copy()
    best_obj = flatten_objective(a)
    stall = 0
    noise_scale = 0.02

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        grad = flatten_gradient(a)
        a_new = np.clip(a - lr * grad, 0.0, 1000.0)
        S_new = np.sum(a_new)
        if S_new < 0.01:
            a_new = a
            S_new = S0
        else:
            a_new = a_new * (S0 / S_new)

        obj_new = flatten_objective(a_new)

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new
            stall += 1
            lr *= 0.8

        # Periodic noise injection to escape local minima
        if stall > 80:
            a = best_a.copy()
            noise = rng.normal(0, noise_scale * S0 / np.sqrt(n), n)
            a = np.clip(a + noise, 0.0, 1000.0)
            S_a = np.sum(a)
            if S_a > 0.01:
                a = a * (S0 / S_a)
            lr = 1.0 / n
            stall = 0
            noise_scale *= 1.3

        if stall > 500:
            break

    return best_a


def lp_refine(a, deadline, max_iters=100):
    """LP-based block coordinate refinement (SOTA method)."""
    n = len(a)
    S = np.sum(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b_full = np.convolve(a, a)
        M = np.max(b_full)
        if M < 1e-14:
            break

        # Build sparse LP constraint matrix
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            for j in range(j_lo, j_hi + 1):
                i = k - j
                if 0 <= i < n:
                    rows.append(k)
                    cols.append(j)
                    vals.append(a[i])

        A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, M)
        c = -np.ones(n)

        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0.0, np.inf)] * n, method='highs')
        if not res.success:
            break

        b_lp = res.x
        S_b = np.sum(b_lp)
        if S_b < 1e-14:
            break

        b_lp = b_lp * (S / S_b)

        # Line search for step size
        best_t = 0.0
        best_obj = None
        for t in [1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.05]:
            a_try = np.clip((1.0 - t) * a + t * b_lp, 0.0, 1000.0)
            S_try = np.sum(a_try)
            if S_try < 0.01:
                continue
            b_try = np.convolve(a_try, a_try)
            obj = 2.0 * n * np.max(b_try) / (S_try * S_try)
            if best_obj is None or obj < best_obj:
                best_obj = obj
                best_t = t

        if best_t == 0.0:
            break

        a = np.clip((1.0 - best_t) * a + best_t * b_lp, 0.0, 1000.0)

    return a


def smooth_max_gradient(a, temperature=50.0):
    """
    Gradient of soft-max of convolve(a,a), i.e.
    grad of (1/T) log(sum_k exp(T*b[k])) w.r.t. a.
    Returns (soft_max_value, gradient_vector).
    """
    n = len(a)
    b = np.convolve(a, a)
    T = temperature
    # Stabilize: shift by max to avoid overflow
    bm = b - np.max(b)
    exp_b = np.exp(T * bm)
    p = exp_b / np.sum(exp_b)  # softmax weights
    soft_val = np.max(b) + np.log(np.sum(exp_b)) / T

    # Gradient: sum_k p[k] * 2*a[k-m] for valid m
    grad = np.zeros(n)
    for m in range(n):
        j = np.arange(max(0, m), min(2 * n - 1, m + n))
        a_idx = j - m  # 0..n-1
        grad[m] = 2.0 * np.dot(p[max(0, m):min(2 * n - 1, m + n)], a[a_idx])

    # Gradient of full objective: 2n * (grad_g * S^2 - g * 2S) / S^4
    S = np.sum(a)
    if S < 0.01:
        return soft_val, np.zeros(n)
    h = S * S
    g = soft_val
    obj_grad = 2.0 * n * (grad * h - g * 2.0 * S * np.ones(n)) / (h * h)
    return soft_val, obj_grad


def smooth_max_descent(a, deadline, rng, max_iters=5000):
    """Polish using smoothed max gradient."""
    n = len(a)
    S0 = np.sum(a)
    lr = 0.3 / n
    best_a = a.copy()
    best_score = None

    # Start with high temperature, anneal down
    T = 100.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Anneal temperature
        frac = it / max_iters
        T = max(2.0, 100.0 * (1.0 - frac) ** 2)

        val, obj_grad = smooth_max_gradient(a, T)
        a_new = np.clip(a - lr * obj_grad, 0.0, 1000.0)
        S_new = np.sum(a_new)
        if S_new > 0.01:
            a_new = a_new * (S0 / S_new)
        a = a_new

        # Evaluate periodically
        if it % 10 == 0:
            b = np.convolve(a, a)
            score = 2.0 * n * np.max(b) / (S0 * S0)
            if best_score is None or score < best_score:
                best_score = score
                best_a = a.copy()
                lr = 0.5 / n

    return best_a
