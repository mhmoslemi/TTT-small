    start = int(rng.integers(0, n - width))
    candidate = seq.copy()
    candidate[start:start+width] *= np.exp(rng.normal(0, T, size=width))
    return candidate

def _additive_block_perturb(seq, rng, T, n):

def _smooth_perturb(seq, rng, T, n):

    k = int(rng.integers(1, max(2, n // 20)))

    amp = rng.uniform(0, T * 2)
    pert = amp * np.cos(2 * np.pi * k * np.arange(n) / n + phase)
    candidate = seq * (1 + pert)
    return candidate

def _single_element_perturb(seq, rng, T, n):

    candidate = seq.copy()
    idx = int(rng.integers(0, n))
    if rng.random() < 0.5:
        candidate[idx] *= np.exp(rng.normal(0, T * 2))
    else:
        candidate[idx] += rng.normal(0, T * max(candidate[idx], 0.01))
    return candidate

def _swap_perturb(seq, rng, n):
    """Swap two elements."""
    candidate = seq.copy()
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    candidate[i], candidate[j] = candidate[j], candidate[i]
    return candidate

def _large_jump_perturb(seq, rng, n):
    """Apply a large random perturbation to escape local minima."""
    frac = rng.uniform(0.2, 0.6)
    width = int(n * frac)
    start = int(rng.integers(0, n - width))
    candidate = seq.copy()
    local_mean = candidate[start:start+width].mean()
    if local_mean < 1e-10:
        local_mean = 1.0
    candidate[start:start+width] = np.abs(rng.normal(local_mean, local_mean * 0.5, size=width))
    return candidate
