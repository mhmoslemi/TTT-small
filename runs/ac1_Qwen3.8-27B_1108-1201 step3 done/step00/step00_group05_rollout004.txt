```python
def simplex_project(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # Use the bisection method for the water-filling
    v = np.maximum(v, 0)
    s = v.sum()
    if s == 0:
        return np.full(n, target_sum / n)
    # Scale to target sum, but need to handle non-negativity
    # Use iterative bisection
    lo, hi = 0, v.max()
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.maximum(v - mid, 0)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.maximum(v - (lo+hi)/2, 0)
    # Fix numerical issues
    diff = target_sum - projected.sum()
    projected += diff / n
    projected = np.maximum(projected, 0)
    return projected
```

```python
    best = base.copy()
    best_score = evaluate_sequence(best.tolist())
    print(f"Initial score: {best_score:.6f}, n={n}")

    # Main optimization loop
    while time.monotonic() < deadline:
        # Compute convolution
        conv = np.convolve(a, a, 'full')
        
        # Find top peak positions
        peak_val = conv.max()
        # Get positions where conv is within 0.1% of peak
        thresh = peak_val * 0.9995
        peak_positions = np.where(conv >= thresh)[0]
        
        # Limit to top 10 for efficiency
        if len(peak_positions) > 10:
            top_idx = np.argsort(-conv[peak_positions])[:10]
            peak_positions = peak_positions[top_idx]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in peak_positions:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= len(peak_positions)
        
        # Adaptive learning rate
        elapsed = time.monotonic() - (deadline - budget_s + 5)
        progress = min(1.0, elapsed / (budget_s - 5))
        lr = 0.05 * (1.0 - 0.5 * progress)
        
        # Multiplicative update
        a_new = a * np.exp(-lr * grad)
        
        # Project to {x >= 0, sum(x) = n}
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 0:
            a_new = a_new * (n / s_new)
        else:
            a_new = np.ones(n)
        
        # Evaluate
```
