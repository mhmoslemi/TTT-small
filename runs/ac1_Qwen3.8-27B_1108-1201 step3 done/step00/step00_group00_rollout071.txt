```python
"""
Optimizes step function sequence to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses the flat autocorrelation sequence (coefficients of 1/sqrt(1-x)) as
starting point, then iteratively applies LP-based direction finding with
line search to reduce the objective below pi/2.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse

def flat_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2.0*k - 1.0) / (2.0*k)
    return a

def score(a):
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 1e-6:
        return np.inf
    M = np.max(np.convolve(a, a))
    return float(2.0 * n * M / (S * S))

def build_constraint_matrix(a):
    """Build the (2n-1) x n matrix where row k has a_{k-j} at column j."""
    n = len(a)
    # Column j has values a[0..n-1] at rows j..j+n-1
    cols = np.repeat(np.arange(n, dtype=np.int32), n)
    rows = np.tile(np.arange(n, dtype=np.int32), n) + np.repeat(np.arange(n, dtype=np.int32), n)
    data = np.tile(a, n)
    A = sparse.csc_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_direction(a, M):
    """LP: maximize sum(delta) s.t. A @ delta <= rhs, delta >= -a."""
    n = len(a)
    conv = np.convolve(a, a)
    rhs = (M - conv) / 2.0
    A = build_constraint_matrix(a)
    c = -np.ones(n)
    lower = -a
    upper = np.full(n, np.inf)
    res = linprog(c, A_ub=A, b_ub=rhs,
                  bounds=list(zip(lower, upper)), method='highs')
    if res.success and res.x is not None:
        return -res.x
    return None

def line_search(a, delta, max_iter=20):
    """Find best t in [0,1] scaling delta to minimize score of a + t*delta*alpha."""
    # Scale delta so that max |delta| = 1
    dmax = np.max(np.abs(delta))
    if dmax < 1e-15:
        return 0.0, 0.0
    delta = delta / dmax
    
    # Max t before non-negativity is violated
    neg_mask = delta < 0
    if np.any(neg_mask):
        t_max = np.min(a[neg_mask] / (-delta[neg_mask]))
    else:
        t_max = np.inf
    t_max = min(t_max, 1.0)  # cap step size
    
    if t_max <= 0:
        return 0.0, score(a)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, t_max
    best_t = 0.0
    best_s = score(a)
    
    for _ in range(max_iter):
        t1 = hi - gr * (hi - lo)
        t2 = lo + gr * (hi - lo)
        s1 = score(a + t1 * delta)
        s2 = score(a + t2 * delta)
        if s1 < s2:
            hi = t2
            if s1 < best_s:
                best_s, best_t = s1, t1
        else:
            lo = t1
            if s2 < best_s:
                best_s, best_t = s2, t2
        if hi - lo < 1e-6:
            break
    return best_t, best_s
```

```python
"""
Minimizes 2n*max(conv(a,a))/sum(a)^2 using the flat-autocorrelation sequence
(coefficients of (1-x)^(-1/2)) as initialization, then iteratively applies
LP-based direction finding (linearized convex constraint) with line search.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def flat_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return a


def compute_score(a):
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 1e-6:
        return np.inf
    conv = np.convolve(a, a)
    M = np.max(conv)
    return float(2.0 * n * M / (S * S))


def build_constraint_matrix(a):
    n = len(a)
    cols = np.repeat(np.arange(n, dtype=np.int32), n)
    rows = np.tile(np.arange(n, dtype=np.int32), n) + np.repeat(np.arange(n, dtype=np.int32), n)
    data = np.tile(a.astype(np.float64), n)
    A = sparse.csc_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_direction(a, M, A):
    n = len(a)
    conv = np.convolve(a, a)
    rhs = (M - conv) / 2.0
    c = -np.ones(n)
    lower = -a
    upper = np.full(n, np.inf)
    try:
        res = linprog(c, A_ub=A, b_ub=rhs,
                      bounds=list(zip(lower, upper)), method='highs',
                      options={'time_limit': 60, 'presolve': True})
        if res.success and res.x is not None:
            return -res.x
    except Exception:
        pass
    return None


def do_line_search(a, delta):
    dmax = np.max(np.abs(delta))
    if dmax < 1e-15:
        return a, compute_score(a)
    d = delta / dmax
    neg = d < 0
    if np.any(neg):
        t_feas = np.min(a[neg] / (-d[neg]))
    else:
        t_feas = 1e10
    t_feas = min(t_feas, 1.0)
    if t_feas <= 1e-10:
        return a, compute_score(a)
    current_score = compute_score(a)
    best_s = current_score
    best_a = a.copy()
    fractions = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]
    for f in fractions:
        tc = t_feas * f
        candidate = a + tc * d
        s = compute_score(candidate)
        if s < best_s:
            best_s = s
            best_a = candidate
    return best_a, best_s


def make_block_perturbation(base, rng, scale=0.05):
    candidate = np.asarray(base, dtype=np.float64).copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 10)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, scale)))
    return np.clip(candidate, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    deadline = time.monotonic() + budget_s - 10.0
    rng = np.random.default_rng(seed)

    n = 4806
    a = flat_sequence(n)
    best_a = a.copy()
    best_score = compute_score(a)
    print(f"n={n}, initial score={best_score:.8f}", flush=True)

    # Pre-build constraint matrix (structure doesn't change, only values in rows do via a)
    # Actually the matrix depends on a, so we rebuild each iteration.
    # But we can cache the sparsity pattern.
    
    stagnant = 0
    consec_fails = 0

    for it in range(20000):
        if time.monotonic() >= deadline:
            break
        t0 = time.monotonic()

        # Build constraint matrix and solve LP
        conv = np.convolve(a, a)
        M = float(np.max(conv))
        A = build_constraint_matrix(a)
        delta = solve_lp_direction(a, M, A)

        updated = False
        if delta is not None and np.sum(delta) > 1e-8:
            new_a, new_s = do_line_search(a, delta)
            if new_s < best_score - 1e-10:
                a = new_a
                best_a = new_a.copy()
                best_score = new_s
                stagnant = 0
                updated = True
                if it % 10 == 0:
                    print(f"Iter {it}, score={best_score:.8f}, elapsed={time.monotonic()-deadline+budget_s-10:.0f}s", flush=True)
            else:
                stagnant += 1
        else:
            stagnant += 1

        if not updated:
            consec_fails += 1
        else:
            consec_fails = 0

        # If stagnant, try a random perturbation
        if stagnant >= 15 or consec_fails >= 50:
            a = make_block_perturbation(best_a, rng, scale=0.03)
            s = compute_score(a)
            if s < best_score:
                best_a = a.copy()
                best_score = s
            stagnant = 0
            consec_fails = 0

        # Periodic reporting
        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"Iter {it}, time/solve={elapsed:.1f}s, best={best_score:.8f}, stagnant={stagnant}", flush=True)

    # Final polish with block perturbations
    remaining = deadline - time.monotonic()
    if remaining > 5:
        for _ in range(int(remaining * 20)):
            if time.monotonic() >= deadline:
                break
            candidate = make_block_perturbation(best_a, rng, scale=0.02)
            s = compute_score(candidate)
            if s < best_score:
                best_a = candidate
                best_score = s
                print(f"Polish: score={best_score:.8f}", flush=True)

    print(f"Final best score: {best_score:.8f}", flush=True)
    return [float(x) for x in best_a]
```
