    Score = 2n*M/S^2 >= 2n*(S^2/n)/S^2 = 2.
    Equality holds for uniform sequences.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Start with the known optimal: uniform sequence
    n = 1024
    best = _uniform_sequence(n)
    best_score = float(evaluate_sequence(best))
    print(f"Initial uniform sequence (n={n}): score = {best_score:.10f}")
    print(f"Theoretical minimum: 2.0")

    # Verification search: try diverse perturbations to confirm optimality
    attempts = 0
    improved = False
    while time.monotonic() < deadline and attempts < 50000:
        # Use varying perturbation scales to explore different neighborhoods
        scale = 0.005 + 0.1 * rng.random()
        candidate = _perturb_sequence(best, rng, scale=scale)
        score = float(evaluate_sequence(candidate))
        attempts += 1
        
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            improved = True
            print(f"  Improvement at attempt {attempts}: score = {best_score:.10f}")

    final_score = float(evaluate_sequence(best))
    print(f"Final score: {final_score:.10f}")
    print(f"Total attempts: {attempts}, improved: {improved}")
    if not improved:
        print("Confirmed: uniform sequence is optimal (score = 2.0).")

    return [float(x) for x in best]
