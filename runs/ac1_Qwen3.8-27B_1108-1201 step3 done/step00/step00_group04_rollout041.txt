LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find descent directions, combined with multiple
initializations and line search.

import numpy as np
import time
from scipy.optimize import linprog

def build_conv_matrix(f):
    """Build (2n-1) x n matrix A such that A @ g = convolve(f, g)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        A[m:m + n, m] = f
    return A

def solve_lp_direction(f, time_limit=30.0):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g or None."""
    n = len(f)
    conv_f = np.convolve(f, f)
    M = float(np.max(conv_f))
    if M < 1e-15:
        return None
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit,
                                                'presolve': True})
        if res.x is not None and np.all(res.x >= -1e-10):
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None
