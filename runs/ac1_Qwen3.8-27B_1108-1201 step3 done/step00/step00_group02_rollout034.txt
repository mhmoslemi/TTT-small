def cross_corr(w, a):
    """Compute cross-correlation using FFT."""
    n_w = len(w)
    n_a = len(a)
    size = next_pow2(n_w + n_a - 1)
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    return np.fft.irfft(W.conjugate() * A, size)[:n_a]
