"""
Minimize 2n*max(conv(a,a))/(sum(a))^2 using power-sum gradient descent
with diverse initializations (chirp, two-level, provided) and local search.
"""

import numpy as np
import time


def compute_score(a):
    """Compute the objective score given a numpy array (avoids list conversion overhead)."""
    n = len(a)
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


def chirp_init(n, seed=0):
    """Generate a chirp-like initialization with quadratic phase."""
    rng = np.random.default_rng(seed)
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    # Multiple chirp components with different frequencies
    for freq in [1.0, 2.0, 3.0, 0.5]:
        T = rng.uniform(2.0 * n, 8.0 * n)
        amp = rng.uniform(0.2, 0.5)
        a += amp * np.cos(2.0 * np.pi * freq * t * t / T + rng.uniform(0, 2 * np.pi))
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0:
        a *= n / s
    return a


def two_level_init(n, seed=0):
    """Generate a two-level alternating pattern initialization."""
    rng = np.random.default_rng(seed)
    a = np.ones(n, dtype=np.float64)
    alpha = rng.uniform(0.3, 0.7)
    # Period-2 pattern
    a[0::2] = 1.0 + alpha
    a[1::2] = np.maximum(1.0 - alpha, 0.0)
    # Add some random variation
    a *= np.exp(rng.normal(0, 0.05, n))
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0:
        a *= n / s
    return a


def random_init(n, seed=0):
    """Generate a random initialization."""
    rng = np.random.default_rng(seed)
    a = rng.random(n) + 0.1
    s = a.sum()
    a *= n / s
    return a


def power_sum_gradient(a, conv, p):
    """
    Compute gradient of (sum conv[k]^p)^(1/p) with respect to a.
    
    grad[j] = 2/M_p^(p-1) * sum_m a[m] * conv[j+m]^(p-1)
            = 2/M_p^(p-1) * correlate(conv^(p-1), a, 'valid')[j]
    """
    conv_pos = np.clip(conv, 0.0, None)
    conv_p1 = conv_pos ** (p - 1.0)
    # correlate(conv_p1, a, 'valid')[j] = sum_m conv_p1[j+m] * a[m]
    n = len(a)
    corr = np.correlate(conv_p1, a, mode='valid')
    # corr should have length n
    M_p = (np.sum(conv_pos ** p)) ** (1.0 / p)
    if M_p < 1e-15:
        return np.zeros(n)
    grad = 2.0 * corr / (M_p ** (p - 1.0))
    return grad
