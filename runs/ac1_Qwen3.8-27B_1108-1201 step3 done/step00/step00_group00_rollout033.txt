"""
Hybrid LP + perturbation search for minimizing the peak autocorrelation ratio.
Uses a linear program to find directions that flatten the linear autocorrelation,
with random perturbations to escape local optima.
"""

import numpy as np
from scipy.sparse import csc_matrix
from scipy.optimize import linprog
import time


def build_conv_matrix(n, f):
    """Build sparse matrix A where A @ g = np.convolve(f, g)."""
    cols = np.repeat(np.arange(n), n)
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    vals = np.tile(f, n)
    return csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def compute_score(n, f):
    """Compute the score directly (same as evaluate_sequence but faster for numpy arrays)."""
    conv = np.convolve(f, f)
    max_b = conv.max()
    sum_a = f.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def lp_direction(f, n, time_limit=10):
    """Solve LP to find direction g that maximizes sum(g) subject to conv(f,g) <= max(conv(f,f))."""
    conv = np.convolve(f, f)
    M = conv.max()
    if M <= 0:
        return None
    
    A = build_conv_matrix(n, f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': time_limit})
    
    if not res.success or res.x is None:
        return None
    return res.x


def do_lp_iterations(f, n, budget_s=100, seed=42):
    """Perform LP-based iterations to flatten the autocorrelation."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    score = compute_score(n, f)
    stuck = 0
    
    while time.monotonic() < deadline:
        g = lp_direction(f, n, time_limit=5)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        
        g = g * (f.sum() / g_sum)
        
        improved = False
        for t in [1.0, 0.5, 0.25, 0.1]:
            new_f = (1 - t) * f + t * g
            new_f = np.clip(new_f, 0, 1000)
            if new_f.sum() < 0.01:
                continue
            s = compute_score(n, new_f)
            if s < score:
                f = new_f
                score = s
                improved = True
                break
        
        if not improved:
            stuck += 1
            if stuck >= 10:
                # Perturb to escape
                f = f * np.exp(0.2 * rng.standard_normal(n))
                f = np.clip(f, 0, 1000)
                if f.sum() < 0.01:
                    f = np.ones(n) / n
                else:
                    f = f / f.sum()
                score = compute_score(n, f)
                stuck = 0
            elif stuck >= 3:
                break
    
    return f, score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Hybrid LP + perturbation search for minimizing peak autocorrelation ratio.
    """
    deadline = time.monotonic() + budget_s - 10
    rng = np.random.default_rng(seed)
    
    best_f = None
    best_score = np.inf
    
    # Configurations: (n, init_type)
    configs = [
        (500, 'ramp'),
        (1000, 'ramp'),
        (1000, 'random'),
        (500, 'random'),
        (2000, 'ramp'),
        (1000, 'cos'),
        (1500, 'ramp'),
        (750, 'random'),
    ]
    
    for n, init_type in configs:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        
        # Allocate time: 80% to LP, 20% to perturb
        lp_time = min(remaining * 0.7, 300)
        pert_time = min(remaining * 0.3, 150)
        
        # Initialize
        if init_type == 'ramp':
            f = np.linspace(0.7, 1.3, n)
        elif init_type == 'random':
            f = 1.0 + 0.4 * rng.random(n)
        elif init_type == 'cos':

        else:
            f = np.ones(n)

        f = f / f.sum()

        if score < best_score:
            best_score = score

        if final_score < best_score:
            best_score = final_score

Actually, the gradient of max(conv(f,f)) w.r.t. f is:
- Let k* = argmax(conv(f,f)).
- grad[i] = 2 * f[k*-i] if i in S_k*, else 0.

"""
Hybrid LP-based search for minimizing peak autocorrelation ratio.
Uses linear programming to find directions that flatten the linear autocorrelation,
combined with random perturbations to escape local optima. Also tries gradient-based
smoothing with a soft-max approximation.
"""

import numpy as np
from scipy.sparse import csc_matrix
from scipy.optimize import linprog
import time


def build_conv_structure(n):
    """Precompute row/col indices for convolution matrix."""
    cols = np.repeat(np.arange(n), n)
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    return rows, cols


def build_conv_matrix(n, f, rows, cols):
    """Build sparse matrix A where A @ g = np.convolve(f, g)."""
    vals = np.tile(f, n)
    return csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def compute_score(n, f):
    """Fast score computation (same formula as evaluate_sequence)."""
    conv = np.convolve(f, f)
    max_b = conv.max()
    sum_a = f.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def lp_direction(f, n, rows, cols, time_limit=5):
    """Solve LP: max sum(g) s.t. conv(f,g) <= max(conv(f,f)), g >= 0."""
    conv = np.convolve(f, f)
    M = conv.max()
    if M <= 0:
        return None
    A = build_conv_matrix(n, f, rows, cols)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def do_lp_iterations(f, n, rows, cols, budget_s=100, seed=42):
    """LP-based flattening iterations."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    score = compute_score(n, f)
    stuck = 0
    it = 0

    while time.monotonic() < deadline:
        g = lp_direction(f, n, rows, cols, time_limit=5)
        if g is None:
            break
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g * (f.sum() / g_sum)

        improved = False
        for t in [1.0, 0.5, 0.25, 0.1, 0.05]:
            new_f = (1.0 - t) * f + t * g
            new_f = np.clip(new_f, 0.0, 1000.0)
            if new_f.sum() < 0.01:
                continue
            s = compute_score(n, new_f)
            if s < score:
                f = new_f
                score = s
                improved = True
                break

        it += 1
        if not improved:
            stuck += 1
            if stuck >= 8:
                f = f * np.exp(0.15 * rng.standard_normal(n))
                f = np.clip(f, 0.0, 1000.0)
                fsum = f.sum()
                if fsum < 0.01:
                    f = np.ones(n) / n
                else:
                    f = f / fsum
                score = compute_score(n, f)
                stuck = 0
            elif stuck >= 3:
                break

    return f, score


def gradient_smooth_step(f, n, lr, T, max_k=20):
    """One step of smoothed-max gradient descent on the simplex."""
    conv = np.convolve(f, f)
    # Soft-max weights over top-k convolution values
    top_indices = np.argsort(conv)[-max_k:]
    weights = np.exp(T * conv[top_indices])
    weights /= weights.sum()
    
    # Gradient of weighted average of conv[k] w.r.t. f[i]
    # d(conv[k])/df[i] = 2*f[k-i] if valid
    grad = np.zeros(n)
    for idx, w in zip(top_indices, weights):
        k = idx
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            grad[i] += 2.0 * w * f[k - i]
    
    # Mirror descent (multiplicative update)
    f_new = f * np.exp(-lr * grad)
    f_new = np.clip(f_new, 1e-10, 1000.0)
    f_new = f_new / f_new.sum()
    return f_new


def do_gradient_polish(f, n, budget_s=50, seed=42):
    """Polish using smoothed-max gradient steps."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    score = compute_score(n, f)
    T = 50.0
    lr = 0.1
    
    while time.monotonic() < deadline:
        f_new = gradient_smooth_step(f, n, lr, T, max_k=10)
        s = compute_score(n, f_new)
        if s < score:
            f = f_new
            score = s
            # Decrease learning rate
            lr *= 0.99
        else:
            # Random kick
            f = f * np.exp(0.05 * rng.standard_normal(n))
            f = np.clip(f, 1e-10, 1000.0)
            f = f / f.sum()
            lr = max(0.01, lr * 0.8)
    
    return f, score


def block_perturb_polish(f, n, budget_s=60, seed=42):
    """Final polish with block multiplicative perturbations."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s
    score = compute_score(n, f)
    
    while time.monotonic() < deadline:
        width = max(2, int(rng.integers(2, n // 20 + 1)))
        start = int(rng.integers(0, n - width + 1))
        candidate = f.copy()
        candidate[start:start + width] *= np.exp(rng.normal(0, 0.08, width))
        candidate = np.clip(candidate, 0.0, 1000.0)
        csum = candidate.sum()
        if csum < 0.01:
            continue
        s = compute_score(n, candidate)
        if s < score:
            f = candidate
            score = s
    
    return f, score
