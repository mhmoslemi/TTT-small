def fft_autocorr(a):

                print(f"  [rand {attempt}] n={n}: score={score:.6f}", flush=True)

    # Final: try to refine the best sequence a bit more if time allows
    if best is not None and time.monotonic() < deadline - 15:
        remaining = deadline - time.monotonic()
        a_final, cmax = optimize_from(best, deadline, max_time=min(remaining, 120.0), rng=rng)
        score = float(evaluate_sequence([float(x) for x in a_final]))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a_final.copy()
            print(f"  [refined] score={best_score:.6f}", flush=True)

    if best is None:
        best = np.ones(256, dtype=np.float64)
        best_score = float(evaluate_sequence([float(x) for x in best]))

    # Scale output: multiply by sqrt(2*n) so sum = sqrt(2*n)
    n_out = len(best)
    scale = np.sqrt(2.0 * n_out) / np.sum(best)
    result = best * scale
    result = np.clip(result, 0.0, 1000.0)

    print(f"Final: score={best_score:.6f}, len={n_out}", flush=True)
    return [float(x) for x in result]
