def project_simplex(v):
    """Project v onto the probability simplex: x >= 0, sum(x) = 1."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses gradient descent on a log-sum-exp smooth approximation of the peak
autocorrelation, with simplex projection, multiple initializations, and
adaptive schedule. Exploits FFT for O(n log n) convolution and gradient.
"""

import numpy as np
from scipy.fft import fft, ifft
import time


def compute_conv(a):
    """Compute linear convolution of a with itself via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    A = fft(a, fft_size)
    conv = np.real(ifft(A * A, fft_size))[:size]
    # Clip tiny negative values from numerical error
    conv = np.maximum(conv, 0.0)
    return conv


def compute_gradient(a, conv, alpha):
    """
    Compute gradient of (1/alpha)*log(sum_k exp(alpha*conv_k)) wrt a.
    grad[j] = 2 * sum_{m=0}^{n-1} a[m] * w[m+j]
    where w[k] = exp(alpha*(conv[k]-max_conv)) / sum(exp(...)).
    """
    n = len(a)
    max_conv = np.max(conv)
    w = np.exp(alpha * (conv - max_conv))
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        return np.zeros(n)
    w = w / w_sum

    if n <= 3000:
        corr = np.correlate(w, a, mode='valid')
    else:
        a_rev = a[::-1]
        N = 1
        while N < (2 * n - 1) + n - 1:
            N <<= 1
        corr_full = np.real(ifft(fft(w, N) * fft(a_rev, N)))
        corr = corr_full[n - 1:n - 1 + n]

    grad = 2.0 * corr
    return grad


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def optimize_sequence(n, init, alpha_start=2.0, alpha_end=150.0,
                      max_iter=8000, lr_start=0.02, lr_end=0.0005,
                      time_limit=60.0, momentum=0.9):
    """
    Gradient descent with momentum and simplex projection to minimize
    the log-sum-exp approximation of max convolution.
    """
    a = np.clip(init.astype(np.float64), 0.0, 1000.0)
    a_sum = a.sum()
    if a_sum < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / a_sum

    a = project_simplex(a)
    best_a = a.copy()
    best_score_local = np.inf
    velocity = np.zeros(n)
    start_time = time.monotonic()
    no_improve_count = 0
    patience = 300

    for it in range(max_iter):
        if time.monotonic() - start_time > time_limit:
            break

        frac = min(1.0, it / max(1, max_iter - 1))
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        conv = compute_conv(a)
        max_b = np.max(conv)
        score = 2.0 * n * max_b  # sum(a) = 1

        if score < best_score_local:
            best_score_local = score
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count >= patience:
            # Perturb to escape local minimum
            noise = np.random.default_rng(it + n).normal(0, 0.02, n)
            a_pert = a + noise
            a = project_simplex(a_pert)
            no_improve_count = 0
            patience = max(100, patience // 2)
            continue

        grad = compute_gradient(a, conv, alpha)
        # Tangent to simplex
        grad_tangent = grad - np.mean(grad)

        # Momentum update
        velocity = momentum * velocity + grad_tangent
        a_new = a - lr * velocity

        # Project onto simplex
        a_new = project_simplex(a_new)
        a = a_new

    return best_a, best_score_local
