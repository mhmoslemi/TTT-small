from scipy.optimize import linprog

def solve_lp(f, M):
    """Solve the LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k."""
    n = len(f)
    # Variables: g_0, ..., g_{n-1} >= 0
    # Objective: minimize -sum(g_j) (to maximize sum(g_j))
    c = -np.ones(n)
    
    # Constraints: (f*g)_k <= M for k = 0, ..., 2n-2
    # (f*g)_k = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f_j * g_{k-j}
    # A_ub * g <= M (vector of length 2n-1)
    
    # Build A_ub matrix (sparse would be better, but let's use dense for now)
    # Actually for n=2535, this is a 5069 x 2535 matrix, which is about 50MB. Should be OK.
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A_ub[k, j] = f[k - j]
    
    b_ub = np.full(2*n-1, M)

    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not result.success:
        return None
    
    return result.x

def build_toeplitz_band(n, f):
    """Build sparse banded Toeplitz-like matrix A where A[k,j] = f[k-j] if valid."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n, k + 1)
        for j in range(j_lo, j_hi):
            v = f[k - j]
            if v > 1e-15:
                rows.append(k)
                cols.append(j)
                vals.append(v)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_direction_lp(f, M, time_limit=30):
    """Solve LP: maximize sum(g) s.t. g >= 0, A@f <= M."""
    n = len(f)
    c = -np.ones(n)
    A = build_toeplitz_band(n, f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if not res.success:
        return None
    return np.maximum(res.x, 0.0)


def subgradient_step(a, lr):
    """One projected subgradient step to minimize max(a*a)."""
    n = len(a)
    conv = np.convolve(a, a)
    k_star = np.argmax(conv)
    # Subgradient: s[j] = 2*a[k_star - j] if 0 <= k_star - j < n
    s = np.zeros(n)
    j_lo = max(0, k_star - n + 1)
    j_hi = min(n, k_star + 1)
    for j in range(j_lo, j_hi):
        s[j] = 2.0 * a[k_star - j]
    # Update
    a_new = a - lr * s
    # Project onto {x >= 0, sum(x) = 1}
    a_new = np.maximum(a_new, 0.0)
    total = a_new.sum()
    if total > 1e-15:
        a_new /= total
    else:
        a_new = np.ones(n) / n
    return a_new

def _build_band_indices(n):
    """Precompute row/col indices for the (2n-1) x n banded matrix."""
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    for d in range(n):
        sl = slice(d, d + n)
        rows[sl] = np.arange(d, d + n, dtype=np.int32)
        cols[sl] = np.arange(n, dtype=np.int32)
    return rows, cols, total


def _make_A(rows, cols, total, f):
    """Assemble the CSR constraint matrix from precomputed indices."""
    n = len(f)
    vals = np.empty(total, dtype=np.float64)
    for d in range(n):
        vals[d*n:(d+1)*n] = f[d]
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def _lp_direction(f, M, rows, cols, total):
    """Solve: max sum(g) s.t. A f-band @ g <= M, g >= 0."""
    n = len(f)
    A = _make_A(rows, cols, total, f)
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 60})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _subgrad_step(a, lr):
    """Projected subgradient step on the simplex to minimize max(a*a)."""
    conv = np.convolve(a, a)
    k_star = int(np.argmax(conv))
    s = np.zeros_like(a)
    j_lo = max(0, k_star - len(a) + 1)
    j_hi = min(len(a), k_star + 1)
    idx = np.arange(j_lo, j_hi)
    s[idx] = 2.0 * a[k_star - idx]
    a = a - lr * s
    a = np.maximum(a, 0.0)
    tot = a.sum()
    if tot > 1e-15:
        a /= tot
    else:
        a[:] = 1.0 / len(a)
    return a

"""
LP-based alternating minimization with subgradient and mirror-descent refinement
to minimize 2*n*max(convolve(a,a))/sum(a)^2. The LP at each step finds the
"fattest" non-negative direction g such that convolve(f,g) stays bounded by M,
progressively flattening the autocorrelation. Multiple restarts from different
initializations (raised cosine, Gaussian, random, parent) ensure robustness.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _build_band(n):
    """Precompute (rows, cols, total) for the (2n-1)x n banded matrix A[k,j]=f[k-j]."""
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    for d in range(n):
        rows[d*n:(d+1)*n] = np.arange(d, d+n, dtype=np.int32)
    cols = np.tile(np.arange(n, dtype=np.int32), n)
    return rows, cols, total


def _make_A(rows, cols, total, f):
    """Assemble CSR matrix from precomputed indices and current f values."""
    n = len(f)
    vals = np.empty(total, dtype=np.float64)
    for d in range(n):
        vals[d*n:(d+1)*n] = f[d]
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def _lp_dir(f, M, rows, cols, total, tlimit=45):
    """Solve max sum(g) s.t. conv-band(f)@g <= M, g in [0, inf)."""
    n = len(f)
    A = _make_A(rows, cols, total, f)
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': tlimit})
        if res.success:
            x = np.maximum(res.x, 0.0)
            if x.sum() > 1e-14:
                return x
    except Exception:
        pass
    return None


def _subgrad(a, lr):
    """Projected subgradient step: minimize max(a*a) on the simplex."""
    conv = np.convolve(a, a)
    ks = int(np.argmax(conv))
    s = np.zeros_like(a)
    jlo = max(0, ks - len(a) + 1)
    jhi = min(len(a), ks + 1)
    idx = np.arange(jlo, jhi)
    s[idx] = 2.0 * a[ks - idx]
    a = a - lr * s
    a = np.maximum(a, 0.0)
    t = a.sum()
    if t > 1e-15:
        a /= t
    else:
        a[:] = 1.0 / len(a)
    return a


def _mirror(a, s, lr):
    """Mirror descent step on the simplex (multiplicative weights)."""
    a = a - lr * s
    a = np.exp(np.clip(np.log(np.maximum(a, 1e-300)), -50, 50))
    t = a.sum()
    a /= t
    return a

def _optimize_one(a0, deadline, rng, tag=""):
    """Run LP + subgradient optimization starting from a0. Returns (best_seq, best_score)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    tot = a.sum()
    if tot < 1e-10:
        a = np.ones(len(a0))
        tot = a.sum()
    a /= tot
    n = len(a)
    best = a.copy()
    best_sc = _eval(a)
    print(f"[{tag}] start n={n} score={best_sc:.6f}")

    # Build matrix structure once
    rows, cols, total = _build_band(n)

    t = 0.2
    stall = 0
    it = 0
    use_lp = n <= 4000  # LP only for moderate n

    while time.monotonic() < deadline:
        t0 = time.monotonic()
        conv = np.convolve(a, a)
        M = float(conv.max())

        if use_lp:
            g = _lp_dir(a, M, rows, cols, total, tlimit=30)
        else:
            g = None

        if g is not None:
            g /= g.sum()
            a_new = (1.0 - t) * a + t * g
        else:
            # Subgradient step
            conv2 = np.convolve(a, a)
            ks = int(np.argmax(conv2))
            s = np.zeros(n)
            jlo = max(0, ks - n + 1)
            jhi = min(n, ks + 1)
            idx = np.arange(jlo, jhi)
            s[idx] = 2.0 * a[ks - idx]
            lr = 1.5 / (np.sqrt(it + 1) * np.sqrt(n))
            a_new = _subgrad(a, lr)

        sc = _eval(a_new)
        if sc < best_sc:
            best_sc = sc
            best = a_new.copy()
            a = a_new
            stall = 0
            t = min(0.5, t * 1.02)
        else:
            a = a_new
            stall += 1
            t = max(0.03, t * 0.75)

        if stall > 40:
            # Perturb to escape local minimum
            a *= 1.0 + 0.08 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            t = 0.2
            stall = 0

        it += 1
        if it % 15 == 0:
            print(f"[{tag}] it={it} best={best_sc:.6f} t={t:.3f} "
                  f"stall={stall} dt={time.monotonic()-t0:.1f}s")

    return best, best_sc
