def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def compute_gradient_fft(p, w):
    """Compute gradient of smooth max convolution.
    grad[i] = 2 * sum_j w[i+j] * p[j] for j = 0..n-1
    """
    n = len(p)

    fft_size = 1 << (len(w) + n).bit_length()  # next power of 2
    W = np.fft.rfft(w, fft_size)
    P = np.fft.rfft(p[::-1], fft_size)
    conv = np.fft.irfft(W * P, fft_size)

    return 2.0 * conv[n-1:n-1+n]

**Better approach: Use the "active set" method**

At each iteration:

4. The gradient of this effective objective is the mean of individual gradients over the active set.
5. Take a projected gradient step.

def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0."""
    n = len(f)
    # Variables: g[0], ..., g[n-1]
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f * g)[k] <= M for k = 0, ..., 2n-2
    # (f * g)[k] = sum_j f[j] * g[k-j] (with g out of bounds = 0)
    # For variable g[i], it appears in (f*g)[k] when k-j = i, i.e., j = k-i
    # So (f*g)[k] = sum_{i: 0<=k-i<n} f[k-i] * g[i]
    # The coefficient of g[i] in constraint k is f[k-i] if 0 <= k-i < n, else 0.
    
    # Build A_ub matrix (sparse for efficiency)
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(i)
                vals.append(f[j])
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x

def project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = (u * np.arange(1, n+1) > (cssv - 1))
    if not rho_candidates.any():
        return np.ones(n) / n
    rho = np.where(rho_candidates)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def compute_gradient(p, w):
    """Compute gradient: grad[i] = 2 * sum_j w[i+j] * p[j]."""
    n = len(p)
    # cross-correlation of w and p
    # Use FFT for efficiency
    total_len = len(w) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    P_flip = np.fft.rfft(p[::-1], fft_size)
    conv = np.fft.irfft(W * P_flip, fft_size)
    # The cross-correlation is at indices [n-1 : n-1+n]
    return 2.0 * conv[n-1:n-1+n]

We want `grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * p[j]` for `i = 0, ..., n-1`.

Here `w` has length `2n-1` (the convolution values) and `p` has length `n`.

`sum_{j=0}^{n-1} w[i+j] * p[j]` is the cross-correlation of `w` and `p` at lag `i`.

In terms of convolution: `cross_corr(w, p)[i] = sum_j w[i+j] * p[j] = (w * flip(p))[i + n - 1]`

where `flip(p)[j] = p[n-1-j]` and `*` is full convolution.

Let me verify: `(w * flip(p))[i + n - 1] = sum_k w[k] * flip(p)[i+n-1-k] = sum_k w[k] * p[n-1-(i+n-1-k)] = sum_k w[k] * p[k-i]`.

Setting `j = k - i`: `= sum_j w[j+i] * p[j]`. ✓ (with appropriate bounds)

So: `cross_corr(w, p) = convolve(w, flip(p))[n-1 : n-1+n]`

Now, let me think about potential issues:
1. **Numerical precision**: For large n, the FFT-based convolution might have small numerical errors. This should be fine for our purposes.
2. **Convergence**: The projected gradient method might be slow to converge. I need to make sure the step size is well-chosen.
3. **Local minima**: The problem is non-convex, so we might get stuck. The noise/restart mechanism should help.

Let me also think about **alternative formulations** that might be easier to optimize.

**Alternative: Minimize the L^p norm of the convolution**

Instead of minimizing `max(conv)`, I could minimize `||conv||_p` for large `p` (which approximates the max). This is a convex optimization problem (for `p >= 1`, the L^p norm is convex, and the convolution is a quadratic form... wait, no, the L^p norm of a quadratic form is not convex).

Actually, `||conv||_p^p = sum_k |conv[k]|^p`. For `p = 2`, this is `sum_k conv[k]^2 = sum_k (sum_j p[j]p[k-j])^2`, which is a quartic form in p. Not convex.

For `p = 1`, `sum_k conv[k] = (sum p)^2 = 1` (constant). Not useful.

So the L^p norm approach doesn't directly help.

**Another idea: Use the dual formulation**

The problem `min t s.t. conv[k] <= t, p >= 0, sum p = 1` can be dualized. The Lagrangian is:
`L = t + sum_k λ_k (conv[k] - t) + sum_i μ_i (-p[i]) + ν (1 - sum_i p[i])`

The dual function is:
`g(λ, μ, ν) = min_{t, p} L`

For `t`: `∂L/∂t = 1 - sum_k λ_k = 0` => `sum λ_k = 1`.
For `p[i]`: `∂L/∂p[i] = sum_k λ_k * 2*p[k-i] - μ_i - ν = 0` => `μ_i = 2*(λ * p)[i] - ν` (using the cross-correlation).

The dual problem is: maximize `g(λ, μ, ν)` subject to `λ_k >= 0, sum λ_k = 1, μ_i >= 0`.

This is getting complicated. Let me stick with the primal approach.

**Final plan:**

I'll implement a robust projected gradient descent with:
1. Multiple initializations (UM-shaped with various parameters, uniform, given sequence).
2. Adaptive step size with backtracking.
3. Periodic noise injection for escape from local minima.
4. FFT-based gradient computation for speed.
5. Time management to ensure we return before the deadline.

Let me also add a "polishing" step at the end: once we have a good candidate, try small perturbations (like the previous approach's block perturbation) to see if we can squeeze out more improvement.

import numpy as np
import time

"""
Algorithm: Projected gradient descent with adaptive step size, 
FFT-based gradient computation, multiple initializations (UM-shaped, 
uniform, parent), and periodic noise injection to escape local minima.
The goal is to minimize 2n*max(conv(p,p))/sum(p)^2, which is equivalent 
to minimizing max(conv(p,p)) when sum(p)=1.
"""

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_grad(p, w):
    """Compute grad[i] = 2 * sum_j w[i+j]*p[j] using FFT."""
    n = len(p)
    flip_p = p[::-1]
    total_len = len(w) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(flip_p, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def make_um(n, a):
    """UM-shaped initialization."""
    j = np.arange(n, dtype=np.float64)
    if n > 1:
        p = 1.0 + a * np.cos(2.0 * np.pi * j / (n - 1))
    else:
        p = np.ones(1)
    p = np.clip(p, 1e-15, None)
    return p / p.sum()


def make_w(n, a, b):
    """W-shaped: high at three points."""
    j = np.arange(n, dtype=np.float64)
    if n > 2:

    else:
        p = np.ones(n)
    p = np.clip(p, 1e-15, None)
    return p / p.sum()

    """Projected gradient descent with adaptive step size and noise."""
    n = len(p0)
    p = project_simplex(p0.astype(np.float64).copy())
    best_p = p.copy()
    best_max = np.inf
    lr = 0.08
    alpha = 0.0
    stall = 0
    best_lr_at_improve = lr
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        conv = np.convolve(p, p, mode='full')
        cm = float(conv.max())
        
        if cm < best_max - 1e-12:
            best_max = cm
            best_p = p.copy()
            stall = 0
            best_lr_at_improve = lr
        else:
            stall += 1
        
        # Active set: top k values
        k = max(1, min(len(conv), int(0.08 * len(conv))))
        top_idx = np.argpartition(conv, -k)[-k:]
        w = np.zeros(len(conv))
        w[top_idx] = 1.0
        w /= w.sum()
        
        grad = compute_grad(p, w)
        
        # Step
        v = p - lr * grad
        if alpha > 0:
            v += alpha * rng.standard_normal(n)
        p = project_simplex(v)
        
        # Adapt
        if stall == 0:
            lr = min(lr * 1.003, 0.4)
            alpha *= 0.9
        elif stall > 300:
            alpha = min(alpha + 0.005, 0.03)
            lr = max(lr * 0.97, 0.005)
        if stall > 3000:

            p = project_simplex(best_p + 0.05 * rng.standard_normal(n))
            lr = best_lr_at_improve
            alpha = 0.0
            stall = 0
    
    return best_p, best_max


def block_perturb(p, rng, width_frac=0.05):
    """Perturb a random block of p."""

    p_new = p.copy()
    p_new[start:start+w] *= np.exp(float(rng.normal(0, 0.1)))
