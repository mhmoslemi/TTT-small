def targeted_move(a, rng):
    b = np.convolve(a, a)
    k_star = np.argmax(b)
    # Find pairs (i, k_star-i) with large product
    # The contribution to b[k_star] from pair (i, k_star-i) is a[i] * a[k_star-i]
    # but only for valid i
    i_valid = np.arange(max(0, k_star-n+1), min(n-1, k_star)+1)
    products = a[i_valid] * a[k_star - i_valid]
    # Pick the pair with the largest product
    i_peak = i_valid[np.argmax(products)]
    # Now, move mass from i_peak (or k_star - i_peak) to a random position
    # where the "damage" to other convolution entries is minimal
    # Simple approach: move to a random position
    if rng.random() < 0.5:
        src = i_peak
    else:
        src = k_star - i_peak
    # Choose destination: a position where it won't create a new peak
    # Simple: random position
    dst = rng.integers(0, n)
    if src == dst:
        return a
    # Transfer a small fraction
    delta = a[src] * rng.uniform(0.01, 0.1)
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new = np.clip(a_new, 0, 1000)
    return a_new
```

def objective_and_grad(a):
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf, np.zeros(n)
    b = np.convolve(a, a, mode='full')
    k_star = np.argmax(b)
    max_b = b[k_star]
    f = 2.0 * n * max_b / (S * S)
    
    # Gradient
    grad = np.zeros(n)
    for j in range(n):
        dj = k_star - j
        if 0 <= dj < n:
            if k_star == 2 * j:
                dbij = a[j]  # = a[dj]
            else:
                dbij = 2.0 * a[dj]
        else:
            dbij = 0.0
        grad[j] = (2.0 * n / (S * S)) * dbij - (4.0 * n * max_b / (S * S * S)) * a[j]
    
    return f, grad

    # log-sum-exp trick
    m = np.max(b)
    return m + t * np.log(np.sum(np.exp((b - m) / t)))
```

Smooth gradient-based optimization for minimizing the auto-convolution peak.
Uses a smooth max approximation combined with L-BFGS-B for efficient convergence.

import time

from scipy.special import softmax

def compute_objective(a, t=1.0):
    """Compute the smooth approximation of the objective and its gradient."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf, np.zeros(n)
    
    b = np.convolve(a, a, mode='full')
    m = np.max(b)
    # Smooth max
    scaled = (b - m) / t
    # Numerically stable log-sum-exp
    max_scaled = np.max(scaled)
    log_sum_exp = max_scaled + np.log(np.sum(np.exp(scaled - max_scaled)))
    smooth_max_val = m + t * log_sum_exp
    
    obj = 2.0 * n * smooth_max_val / (S * S)
    
    # Gradient
    # w = softmax of (b - m) / t
    w = np.exp(scaled - max_scaled)
    w /= np.sum(w)
    
    # Gradient of smooth_max w.r.t. a: 2 * (w * a) [first n elements]
    wa = np.convolve(w, a, mode='full')[:n]
    
    grad = (4.0 * n / (S * S)) * wa - (4.0 * n * smooth_max_val / (S * S * S)) * a
    
    return obj, grad
