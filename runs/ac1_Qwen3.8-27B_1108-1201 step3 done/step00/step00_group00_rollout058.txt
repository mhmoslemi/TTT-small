    n = len(a)

    # A[k, j] = a[j] if max(0, k-n+1) <= j <= min(n-1, k)

    vals = []

        j_min = max(0, k - n + 1)

        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)

    return a  # fallback

from scipy.optimize import linprog

def make_raised_cosine(n):

    return np.exp(-t**2 / (2 * sigma**2))

    return np.maximum(0, 1 - np.abs(t))

    s = np.sum(a)
    if s <= 0: return a
    return a / s

def compute_score(a):

    conv = np.convolve(a, a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)

    n = len(a)

    # A[k, j] = a[j] if 0 <= k-j < n (i.e., j <= k <= j+n-1)

    return None

This is possible if and only if there exists a distribution on {0,...,n-1} whose sum with an independent copy is uniform on {0,...,2n-2}. This is related to the concept of "uniform sums" and is achievable with certain distributions.

However, it's known that for certain special distributions, the sum can be made nearly uniform. This is related to "convolutions on finite groups" and "difference sets."

For our problem (non-integer-valued, continuous weights), we have more freedom.

Let me think about what distribution makes X1+X2 as uniform as possible.

If X has density f on [0, n-1] (continuous), then X1+X2 has density f*f (convolution). We want f*f to be as uniform as possible on [0, 2n-2].

If f*f = 1/(2n-1) (constant), then taking Fourier transforms: F(f)^2 = F(1/(2n-1)) = (1/(2n-1)) * F(1).
F(1) = sum of exponentials = sin(omega*(2n-1)/2) / (sin(omega/2) * (2n-1)) ... this gets complicated.

The point is that achieving perfectly uniform convolution is generally not possible for a non-negative function, but we can get close.

"""
LP-based refinement for minimizing the peak of auto-convolution.
Uses a linear program to find improvement directions at each step,
combined with adaptive blending and multi-length search.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def initial_raised_cosine(n):
    """Raised cosine window, approximately zero at endpoints."""
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    """Gaussian window centered in the middle."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_triangle(n):
    """Triangular window."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    half = n / 2.0
    return np.maximum(0.0, 1.0 - np.abs(t - center) / half)


def build_conv_matrix(a):
    """Build sparse matrix A where A @ g = np.convolve(a, g)."""
    n = len(a)
    n_rows = 2 * n - 1
    # Column j has a[0..n-1] at rows j..j+n-1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M (element-wise), g >= 0.
    
    Returns the optimal g, or None if the LP fails.
    """
    n = len(a)
    n_rows = 2 * n - 1
    
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception as e:
        print(f"  LP error: {e}")
    return None


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    """Compute the evaluation score (without calling evaluate_sequence)."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def lp_refine(a, time_limit_s=600.0, max_iters=1000, verbose=True):
    """Iteratively refine sequence using LP-based direction finding.
    
    Returns (refined_sequence, best_score).

    start_time = time.monotonic()
    a = normalize_nonneg(a.copy())
    
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:

        score = conv.max() * 2 * len(a) / (a.sum() ** 2)

            stagnant = 0

        g = solve_lp_step(a, M, timeout=timeout)

def random_smooth_init(n, rng, n_bumps=3):
    """Create a random smooth initialization."""
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = rng.integers(0, n)
        width = rng.integers(n // 10, n // 3)
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    return a / a.sum()

"""
LP-based iterative refinement to minimize peak auto-convolution of non-negative sequences.
At each step, solves a linear program to find a direction that reduces the convolution peak,
then blends with the current solution. Tries multiple sequence lengths and initializations.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def initial_raised_cosine(n):
    """Raised cosine window, approximately zero at endpoints."""
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_gaussian(n, sigma_frac=0.25):
    """Gaussian window centered in the middle."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n / 2.0)
    return np.exp(-0.5 * ((t - center) / sigma) ** 2)


def initial_hann(n):
    """Hann window (same as raised cosine but with different endpoint handling)."""
    t = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))


def initial_blackman(n):
    """Blackman window."""
    t = np.arange(n, dtype=np.float64)
    N = n - 1
    if N == 0:
        return np.ones(1)
    x = t / N
    return 0.42 - 0.5 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)


def random_smooth_init(n, rng, n_bumps=3):
    """Create a random smooth initialization from summed Gaussians."""
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 10), max(2, n // 3)))
        t = np.arange(n, dtype=np.float64)
        a += np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(0, a)
    s = a.sum()
    if s < 1e-15:
        a = initial_raised_cosine(n)
        s = a.sum()
    return a / s


def normalize_nonneg(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(0.0, a)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def score_sequence(a):
    """Compute the evaluation score directly (faster than calling evaluate_sequence)."""
    n = len(a)
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def build_conv_matrix(a):
    """Build sparse matrix A where A @ g = np.convolve(a, g). Shape (2n-1, n)."""
    n = len(a)
    n_rows = 2 * n - 1
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    data = np.tile(a, n).astype(np.float64)
    return csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))


def solve_lp_step(a, M, timeout=30.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M (element-wise), g >= 0.
    
    Returns optimal g or None on failure.
    """
    n = len(a)
    n_rows = 2 * n - 1
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(n_rows, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': timeout})
        if res.success and res.x is not None:
            return np.maximum(0.0, res.x)
    except Exception:
        pass
    return None

    """Iteratively refine using LP direction finding with adaptive blending.
    
    Returns (refined_sequence, best_score_found).

    n = len(a)
    
    best_a = a.copy()
    best_score = score_sequence(a)
    if verbose:
        print(f"    Start: score={best_score:.6f}, n={n}")
    
    stagnant = 0
    
    for it in range(max_iters):
        elapsed = time.monotonic() - start_time
        if elapsed > time_limit_s:
            break
        
        conv = np.convolve(a, a)
        score = 2.0 * n * conv.max() / (a.sum() ** 2)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnant = 0
            if verbose and (it % 50 == 0 or it < 5):
                print(f"    Iter {it:4d}: score={score:.6f}")
        else:
            stagnant += 1

            break

        remaining_time = time_limit_s - elapsed
        if remaining_time < 5:
            break
        
        timeout = max(3.0, min(60.0, remaining_time * 0.3))
        g = solve_lp_step(a, M, timeout=timeout)
        
        if g is None:
            break
        
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-10:
