def build_convolution_matrix(f, n):
    """Build the matrix A such that A @ g = f * g (full convolution)."""
    # A has shape (2n-1, n)
    # A[k, i] = f[i] if 0 <= k-i <= n-1, else 0
    # i.e., A[k, i] = f[i] if k-n+1 <= i <= k
    
    row_indices = []
    col_indices = []
    data = []
    
    for i in range(n):
        for k in range(i, i + n):
            # A[k, i] = f[i]
            row_indices.append(k)
            col_indices.append(i)
            data.append(f[i])
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((data, (row_indices, col_indices)), shape=(2*n-1, n))
    return A

"""
LP-based iterative refinement for minimizing 2n*max(convolve(a,a))/(sum(a))^2.
Uses linear programming to find descent directions, with multiple initializations
and line search. Tries several sequence lengths to find the best solution.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_objective(seq):
    """Fast computation of the objective value."""
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_conv = np.max(conv)
    sum_seq = np.sum(seq)
    if sum_seq < 0.01:
        return np.inf
    return 2.0 * n * max_conv / (sum_seq ** 2)


def build_conv_matrix(f):
    """Build matrix A where A @ g = f * g (full convolution)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        A[k, i_start:i_end + 1] = f[i_start:i_end + 1]
    return A


def lp_direction(f, M, timeout_s=30.0):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'time_limit': timeout_s, 'presolve': True})
        if res.status == 0 and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def lp_iteration(f, rng):
    """One LP-based iteration: find direction and do line search."""
    conv = np.convolve(f, f)
    M = np.max(conv)
    if M < 1e-15:
        return None, np.inf

    g = lp_direction(f, M)
    if g is None or not np.all(np.isfinite(g)):
        return None, np.inf

    g_sum = np.sum(g)
    if g_sum < 1e-10:
        return None, np.inf
    g = g / g_sum

    # Line search
    current_score = compute_objective(f)
    best_t = 0.0
    best_score = current_score
    best_candidate = f.copy()

    for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9]:
        candidate = (1 - t) * f + t * g
        score = compute_objective(candidate)
        if score < best_score:
            best_score = score
            best_t = t
            best_candidate = candidate

    return best_candidate, best_score


def random_perturb(f, rng, strength=0.1):
    """Randomly perturb a sequence while keeping it non-negative."""
    n = len(f)
    candidate = f * np.exp(rng.normal(0, strength, size=n))
    candidate = np.maximum(candidate, 0)
    s = candidate.sum()
    if s < 1e-10:
        return f.copy()
    return candidate / s

    stall = 0
