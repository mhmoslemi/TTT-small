def fast_autocorr(a):
    """Compute full linear autocorrelation using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:

    F = np.fft.rfft(a, size)
    conv = np.fft.irfft(F * F, size)[:2*n-1]

    S = np.sum(a)
    if S < 0.01:

    conv = fast_autocorr(a)
    max_val = np.max(conv)
    return 2.0 * n * max_val / (S * S)

def compute_grad(a):

    n = len(a)
    S = np.sum(a)
    if S < 0.01:

    k_star = int(np.argmax(conv))

    grad_M = np.zeros(n)

        J_curr = compute_J(a)
        if not np.isfinite(J_curr):
            continue
        
        best_val = old_ai
        best_J = J_curr
        
        # Try a few perturbation levels
        for delta in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
            for sign in [-1, 1]:
                trial_ai = old_ai * (1.0 + sign * delta)
                if trial_ai < 0:
                    continue

                a = project_and_normalize(a)
                if a is None:
                    a[i] = old_ai
                    continue
                J_trial = compute_J(a)
                if J_trial < best_J:

                    best_val = trial_ai

        a = project_and_normalize(a)
        if a is None:
            a[i] = old_ai
            a = project_and_normalize(a)

        if np.sum(a0) > 0.01:
            inits.append(a0 / np.sum(a0))

    inits.append(np.ones(n) / n)
    
    # 3. Random
    a_rand = rng.random(n)
    inits.append(a_rand / np.sum(a_rand))

    for m in [1, 2, 3, 5, 7]:
        x = np.arange(n)
        a_sin = 1.0 + 0.8 * np.cos(2.0 * np.pi * m * x / n)
        a_sin = np.maximum(a_sin, 0.0)
        if np.sum(a_sin) > 0.01:
            inits.append(a_sin / np.sum(a_sin))
    
    # 5. Triangular
    x = np.arange(n, dtype=float)
    a_tri = x + 1.0
    inits.append(a_tri / np.sum(a_tri))
    a_tri2 = n - x
    inits.append(a_tri2 / np.sum(a_tri2))
    
    # 6. Block structure
    a_block = np.zeros(n)
    block_size = n // 10
    for b in range(0, n, block_size * 2):
        a_block[b:b+block_size] = 1.0
    inits.append(a_block / np.sum(a_block))
    
    # Run gradient descent from each initialization
    best_a = None
    best_J = np.inf
    
    total_inits = len(inits)
    time_per_init = (deadline - time.monotonic()) / total_inits
    
    for init_idx, a in enumerate(inits):
        t_init = time.monotonic()
        t_end_init = t_init + time_per_init
        
        eta = 0.01
        a = project_and_normalize(a)
        J = compute_J(a)

            best_J = J
            best_a = a.copy()

        best_J_local = J
        best_a_local = a.copy()

            a, J, eta = gradient_step(a, eta)

                best_J_local = J
                best_a_local = a.copy()
                stall_count = 0
            else:
                stall_count += 1
            
            # Periodic coordinate descent
            if stall_count > 100 and stall_count % 50 == 0:
                a = coordinate_descent_step(a, n_coords=n // 10)
                a = project_and_normalize(a)
                J = compute_J(a)
                if J < best_J_local:
                    best_J_local = J
                    best_a_local = a.copy()
                a = a.copy() if J < best_J_local else best_a_local.copy()
            
            # Random restart within this init
            if stall_count > 500:
                # Perturb and try again
                a_pert = a + 0.01 * rng.standard_normal(n)
                a_pert = project_and_normalize(a_pert)
                if a_pert is not None:
                    J_pert = compute_J(a_pert)
                    if J_pert < best_J_local:
                        a = a_pert
                        J = J_pert
                        best_J_local = J
                        best_a_local = a.copy()
                        stall_count = 0
                        eta = 0.01
                    else:
                        stall_count += 100  # Force more aggressive restarts

            if J < best_J:
                best_J = J
                best_a = a.copy()
                print(f"  Init {init_idx}: improved to J = {J:.6f} at t={time.monotonic()-t_start:.1f}s")
        
        # After each init, update global best
        if best_J_local < best_J:
            best_J = best_J_local
            best_a = best_a_local.copy()

    try:
        final_score = evaluate_sequence(best_a.tolist())
        if np.isfinite(final_score) and final_score < best_J:
            # The evaluator gave a better score? That shouldn't happen unless
            # our computation is slightly off. Use the evaluator's answer.
            best_J = final_score
    except:
        pass

    # Since J is scale-invariant, we can scale to have sum = sqrt(2n) ≈ 94.8
    a_final = best_a * np.sqrt(2.0 * n)
    a_final = np.clip(a_final, 0.0, 1000.0)
