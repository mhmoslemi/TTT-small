```python
def simplex_project(y):
    n = len(y)
    u = np.sort(y)[::-1]
    S = np.cumsum(u)
    # mu_k = (S_k - 1) / k
    # Find largest k such that u_k > mu_k
    # i.e., u_k > (S_k - 1) / k
    # i.e., k * u_k > S_k - 1
    # i.e., S_k - k * u_k < 1
    # S_k - k * u_k = sum_{i=1}^{k} (u_i - u_k)
    
    k = np.arange(1, n + 1, dtype=float)
    mu = (S - 1.0) / k
    valid = u > mu
    if not valid.any():
        return np.ones(n) / n
    rho = np.max(np.where(valid)[0]) + 1  # 1-indexed
    tau = (S[rho - 1] - 1.0) / rho
    return np.maximum(y - tau, 0.0)
```

```python
def optimize_sequence(n, init, beta_start, beta_end, n_iter, eta):
    a = init.copy()
    # Normalize to sum = 1
    total = a.sum()
    if total < 1e-10:
        a = np.ones(n) / n
        total = 1.0
    a = a / total
    
    best_a = a.copy()
    best_score = None
    
    # Precompute FFT size
    N = 1
    while N < 3 * n:
        N *= 2
    
    for it in range(n_iter):
        # Anneal beta
        t = it / max(1, n_iter - 1)
        beta = beta_start * (beta_end / beta_start) ** t
        
        # Compute b = convolve(a, a, 'full')
        A = np.fft.rfft(a, N)
        b = np.fft.irfft(A * A, N)[:2*n-1]
        
        # Compute softmax weights
        b_max = b.max()
        log_w = beta * (b - b_max)
        w = np.exp(log_w)
        w /= w.sum()
        
        # Compute gradient: g_j = 2 * sum_m a_m * w_{j+m}
        # w_rev = reverse of w
        w_rev = w[::-1]
        W = np.fft.rfft(w_rev, N)
        g = np.fft.irfft(A * W, N)
        # g[k] = (a * w_rev)[k] = sum_m a[m] * w_rev[k-m]
        # We want g_j = sum_m a[m] * w[j+m] = (a * w_rev)[2n-2-j]
        # So grad_j = 2 * g[2n-2-j] for j = 0, ..., n-1
        grad = 2.0 * g[n-1:2*n-1][::-1]
        
        # Also need to account for the constraint sum(a) = 1
        # The constraint gradient is constant (1, 1, ..., 1)
        # So we project after the step
        
        # Step
        a_new = a - eta * grad
        a_new = simplex_project(a_new)
        
        # Evaluate
        score_raw = 2 * n * b.max()  # since sum(a) = 1
        if best_score is None or score_raw < best_score:
            best_score = score_raw
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score
```

```python
import numpy as np
```

def simplex_project(y):
    """Project y onto the non-negative simplex {x: x_i >= 0, sum(x) = 1}."""
    n = len(y)
    u = np.sort(y)[::-1]
    S = np.cumsum(u)
    k = np.arange(1, n + 1, dtype=np.float64)
    mu = (S - 1.0) / k
    valid = u > mu
    if not valid.any():
        return np.ones(n) / n
    idx = np.where(valid)[0]
    rho = idx[-1] + 1
    tau = (S[rho - 1] - 1.0) / rho
    return np.maximum(y - tau, 0.0)
