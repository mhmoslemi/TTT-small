from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

"""
LP-guided iterative local search to minimize 2n*max(f*f)/sum(f)^2.
Solves an LP at each step to find a direction that increases the sum
while keeping the convolution peak bounded, then blends with linesearch.
Uses random objective perturbation to avoid LP degeneracy.
"""


def build_sparse_structure(n):
    """Precompute row/col indices for the (2n-1) x n convolution matrix."""
    m = 2 * n - 1
    rows_parts = []
    cols_parts = []
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        length = j_end - j_start + 1
        rows_parts.append(np.full(length, k, dtype=np.int32))
        cols_parts.append(np.arange(j_start, j_end + 1, dtype=np.int32))
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    return rows, cols, m


def smooth_signal(x, window=5):
    """Apply a small moving-average smooth to reduce spikiness."""
    if window <= 1:
        return x.copy()
    kernel = np.ones(window) / window
    padded = np.pad(x, (window // 2, window // 2), mode='edge')
    return np.convolve(padded, kernel, mode='valid')


def water_filling_multi(f, M, n_passes, rng):

    return seqs


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0

    # Start from the given sequence
    f = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f)

    # Try alternative initializations
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"Initial (given): score={best_score:.6f}, n={n}")

    for init in initial_sequences(n, rng):
        init = np.clip(init, 0.0, 1000.0)
        sc = float(evaluate_sequence(init.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = init.copy()
            best_score = sc
            f = init.copy()
            print(f"Better init: score={sc:.6f}")

    # Precompute sparse structure
    rows, cols, m = build_sparse_structure(n)
    print(f"Starting optimization: n={n}, best_score={best_score:.6f}")

    use_lp = True
    lp_too_slow = 0

    iteration = 0
    no_improve = 0

    while time.monotonic() < deadline:
        iteration += 1
        t_iter_start = time.monotonic()

        M = float(np.max(np.convolve(f, f)))
        sum_f = float(np.sum(f))
        if sum_f < 0.01:
            break

        # Upper bound for g entries
        U = 5.0 * float(np.max(f)) + 1.0

        # Solve for direction
        g = None
        if use_lp:
            try:
                t_lp_start = time.monotonic()
                # LP: max sum(g) + eps*r.g s.t. A@g <= M, 0 <= g <= U
                r = rng.standard_normal(n)
                c_obj = -np.ones(n) - 0.005 * r
                vals = f[rows - cols]
                A = csr_matrix((vals, (rows, cols)), shape=(m, n))
                b_ub = np.full(m, M)
                bounds = [(0.0, U)] * n
                result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                                 method='highs',
                                 options={'time_limit': 15})
                t_lp = time.monotonic() - t_lp_start
                if result.success and np.sum(result.x) > 0:
                    g = result.x
                    if t_lp > 12:
                        lp_too_slow += 1
                        if lp_too_slow >= 3:
                            use_lp = False
                            print("Switching to water-filling (LP too slow)")
            except Exception:
                pass

        if g is None or np.sum(g) <= 0:
            # Water-filling fallback
            g = water_filling_multi(f, M, 3, rng)

        # Smooth the direction to reduce spikiness
        g = smooth_signal(g, window=3)
        g = np.clip(g, 0, 1000.0)

        if np.sum(g) <= 0:
            # No useful direction; try random perturbation
            f = f * np.exp(rng.normal(0, 0.05, n))
            f = np.clip(f, 0, 1000)
            continue

        # Linesearch
        improved = False
        for t in [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]:
            f_new = (1.0 - t) * f + t * g
            f_new = np.clip(f_new, 0.0, 1000.0)
            score = float(evaluate_sequence(f_new.tolist()))
            if np.isfinite(score) and score < best_score:
                f = f_new
                best = f.copy()
                best_score = score
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            if no_improve >= 30:
                # Restart with perturbation
                f = best * np.exp(rng.normal(0, 0.15, n))
                f = np.clip(f, 0, 1000)
                no_improve = 0

        if iteration % 20 == 0:
            M_cur = float(np.max(np.convolve(f, f)))
            s_cur = float(np.sum(f))
            elapsed = time.monotonic() - t_iter_start
            print(f"Iter {iteration}: score={best_score:.6f}, sum={s_cur:.2f}, "
                  f"peak={M_cur:.2f}, t_iter={elapsed:.1f}s, lp={use_lp}")

    final_score = float(evaluate_sequence(best.tolist()))
    print(f"Done: iterations={iteration}, final_score={final_score:.6f}")
    return [float(x) for x in best]
