import numpy as np
import time


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.clip(v - theta, 0, None)


def fft_convolve_autos(a, n, L2):
    """Compute linear convolution a*a using FFT. Returns array of length 2n-1."""
    A = np.fft.fft(a, L2)
    result = np.fft.ifft(A * A, L2).real
    return result[:2 * n - 1]


def compute_gradient_fft(a, n, L, tau, L2):
    """Compute 2 * gradient of smoothed max of conv(a,a).
    Returns (grad, b)."""
    b = fft_convolve_autos(a, n, L2)
    shifted = tau * b
    shifted -= np.max(shifted)
    weights = np.exp(shifted)
    sw = np.sum(weights)
    if sw < 1e-300:

        weights[np.argmax(b)] = 1.0
    else:

    a_rev = a[::-1]
    A_rev = np.fft.fft(a_rev, L)
    W = np.fft.fft(weights, L)
    grad_full = np.fft.ifft(A_rev * W, L).real
    grad = grad_full[n - 1:n - 1 + n]
    return 2.0 * grad, b
