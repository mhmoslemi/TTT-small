from scipy.sparse import csr_matrix
import time

def propose_candidate(budget_s=1800, seed=42):

    rng = np.random.default_rng(seed)
    
    # Initialize
    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0).copy()
    n = len(a)

    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_score = evaluate_sequence(a.tolist())
    best_a = a.copy()
    
    print(f"Initial score: {best_score}")
    
    for iteration in range(500):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution and M
        conv = np.convolve(a, a)
        M = conv.max()
        
        # Build LP constraint matrix
        # A_ub[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
        # b_ub = [M] * (2n-1)
        
        # Use sparse representation
        rows = []
        cols = []
        vals = []
        
        for k in range(2*n - 1):
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            for i in range(i_start, i_end + 1):
                j = k - i
                # A[k, j] = a[i] = a[k-j]
                rows.append(k)
                cols.append(j)
                vals.append(a[i])
        
        A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        b = np.full(2*n-1, M)
        
        # Objective: maximize sum(g) => minimize -sum(g)
        c_obj = -np.ones(n)
        
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        
        if res.success:
            g = res.x
            g_sum = g.sum()
            if g_sum > 0.01:
                g = g / g_sum  # Normalize to sum = 1
                
                # Blend
                t = 0.5  # Mixing parameter
                a_new = (1 - t) * a + t * g
                
                score = evaluate_sequence(a_new.tolist())
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                    a = a_new
                    print(f"Iter {iteration}: score = {best_score}")
                else:
