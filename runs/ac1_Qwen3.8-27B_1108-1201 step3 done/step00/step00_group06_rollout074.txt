FFT-based smoothed-subgradient descent (in softmax/log-parameterization) for
minimizing the peak self-convolution of a non-negative sequence.

Objective returned by evaluator: f = 2*n*max(a*a)/(sum a)^2 (homogeneous).
We parameterize a = softmax(theta) so non-negativity and unit-sum hold exactly,
and minimize the (temp-smoothed) maximum of the autocorrelation.  The gradient
w.r.t. theta is computed in closed form from the smoothed convolution
subgradient, giving an unconstrained (projection-free) optimization.  Several
distinct initializations are run in parallel (best tracked); periodic random
re-heats and temperature annealing escape local minima.  The best valid
sequence found within the time budget is returned.
"""

import numpy as np

def _fft_convolve(a):
    """Full linear auto-convolution a*a via FFT (exact for our sums)."""
    n = a.size
    N = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, N)
    out = np.fft.irfft(fa * fa, N)
    return out[:2 * n - 1]


def _raw_score(n, conv, S):
    """Objective from convolution peak and sum (before clipping rules)."""
    M = conv.max()
    return 2.0 * n * M / (S * S)


def _smoothed_grad_a(a, conv, rel_temp, top_k):
    """
    Subgradient w.r.t. a of the temperature-smoothed max of conv, restricted
    to the top_k largest correlation lags with softmax weights.
    grad_k = 2 * sum_m w[m] * a[m-k]  (valid indices only).
    """
    n = a.size
    if conv.size <= top_k:
        idx = np.arange(conv.size)
        vals = conv.copy()
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        idx = idx[np.argsort(conv[idx])[::-1]]
        vals = conv[idx]
    scale = max(rel_temp * (vals.max() + 1e-300), 1e-300)
    w = np.exp((vals - vals[0]) / scale)
    w /= w.sum()

    grad = np.zeros(n)
    # Vectorized: for each m, grad[k_start:k_end+1] += w*2*a[j_start:j_end+1][::-1]
    for w_m, m in zip(w, idx):
        m = int(m)
        k_start = max(0, m - n + 1)
        k_end = min(m, n - 1)
        if k_start > k_end:
            continue
        j_start = m - k_end
        j_end = m - k_start
        grad[k_start:k_end + 1] += w_m * 2.0 * a[j_start:j_end + 1][::-1]
    return grad


def _theta_grad(a, g_a):
    """Gradient of scalar f(a(theta)) w.r.t. theta for a=softmax(theta)."""
    s = float(np.dot(a, g_a))
    return a * (g_a - s)


def _softmax(theta):
    th = theta - theta.max()
    e = np.exp(th)
    return e / e.sum()


def _evict_and_reinit(current, rng, kick):
    """Random perturbation in a-space (then back to theta)."""
    pert = current + kick * rng.standard_normal(current.size)
    pert = np.maximum(pert, 1e-9)
    pert /= pert.sum()
    return np.log(pert + 1e-300)
