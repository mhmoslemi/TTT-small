def simplex_project(v):

import numpy as np
from scipy.sparse import lil_matrix
from scipy.optimize import linprog

def solve_lp_direction(f, M):
    """Solve the LP to find the best direction g."""
    n = len(f)
    # Constraint matrix: A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = lil_matrix((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[k, j] = f[k-j]
    A = A.tocsr()
    
    b = np.full(2*n-1, M)
    c = -np.ones(n)  # Minimize -sum(g) = maximize sum(g)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.status == 0:
        return res.x
    return None
