import time
from scipy.optimize import linprog
from scipy.sparse import diags

def solve_lp(f, M, n):
    """Solve the LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    c = -np.ones(n)
    
    # Build constraint matrix A: (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j < n
    # Using diags: for offset d, the diagonal has f[d] in positions (d+j, j)
    diagonals = [f[d] * np.ones(n) for d in range(n)]
    offsets = list(range(n))
    A = diags(diagonals, offsets, shape=(2*n-1, n))
    
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    return res.x if res.success else None
