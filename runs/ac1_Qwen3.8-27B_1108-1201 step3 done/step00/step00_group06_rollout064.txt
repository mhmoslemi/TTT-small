"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using iterative partial-LP direction
finding plus multiplicative gradient refinement. Exploits non-symmetric
initializations to break the score>=2 barrier for symmetric sequences.
"""

import numpy as np
import time
import scipy.sparse as sp
import cvxpy as cp


def build_partial_matrix(f, positions):
    """Build sparse (k x n) matrix for selected convolution constraint rows."""
    n = len(f)
    k = len(positions)
    if k == 0:
        return sp.csr_matrix((0, n))
    pos_arr = np.asarray(positions, dtype=np.int64)
    j_starts = np.maximum(0, pos_arr - n + 1).astype(np.int32)
    j_ends = (np.minimum(pos_arr, n - 1) + 1).astype(np.int32)
    row_counts = (j_ends - j_starts).astype(np.int32)
    total = int(row_counts.sum())
    if total == 0:
        return sp.csr_matrix((k, n))
    rows = np.repeat(np.arange(k, dtype=np.int32), row_counts)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    idx = 0
    for i in range(k):
        js = int(j_starts[i])
        je = int(j_ends[i])
        cnt = je - js
        cols[idx:idx + cnt] = np.arange(js, je, dtype=np.int32)
        vals[idx:idx + cnt] = f[js:je]
        idx += cnt
    return sp.csr_matrix((vals, (rows, cols)), shape=(k, n))


def solve_lp_direction(f, M, positions, timeout=60):
    """Solve max sum(g) s.t. T_partial g <= M, g >= 0."""
    n = len(f)
    T = build_partial_matrix(f, positions)
    g = cp.Variable(n)
    constraints = [T @ g <= M, g >= 0]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    solvers = [cp.ECOS, cp.SCS, cp.CLARABEL, 'GLPK']
    for solver in solvers:
        try:
            prob.solve(solver=solver, max_iters=200000, verbose=False)
            if g.value is not None and np.isfinite(g.value).all():
                g_val = np.clip(g.value, 0.0, 1000.0)
                if g_val.sum() > 1e-12:
                    return g_val
        except Exception:
            continue
    return None


def make_init_gaussian(n, center_frac, sigma_frac, rng):
    """Create a non-symmetric Gaussian initial sequence."""
    center = int(center_frac * n)
    sigma = sigma_frac * n
    j = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((j - center) / sigma) ** 2)
    a *= np.exp(rng.normal(0.0, 0.05, n))
    return np.clip(a, 0.0, 1000.0)


def make_init_random(n, rng):
    """Create a random non-negative initial sequence."""
    a = rng.exponential(1.0, n)
    return np.clip(a, 0.0, 1000.0)


def make_init_parent(parent, n):
    """Reshape parent to length n if possible."""
    if parent.size == n:
        return parent.copy()
    # Resize by interpolation
    old_n = parent.size
    idx = np.linspace(0, old_n - 1, n)
    return np.clip(np.interp(idx, np.arange(old_n), parent), 0.0, 1000.0)


def gradient_pressure(f, top_positions):
    """Compute subgradient pressure from top convolution peaks."""
    n = len(f)
    pressure = np.zeros(n, dtype=np.float64)
    for m in top_positions:
        m = int(m)
        j_lo = max(0, m - n + 1)
        j_hi = min(m, n - 1)
        if j_lo <= j_hi:
            idx_start = m - j_hi
            idx_end = m - j_lo + 1
            pressure[j_lo:j_hi + 1] += f[idx_start:idx_end][::-1]
    return pressure


def multiplicative_update(f, pressure, lr):
    """Apply multiplicative (mirror descent) update."""
    p_max = pressure.max()
    if p_max > 1e-30:
        pressure = pressure / p_max
    f_new = f * np.exp(-lr * pressure)
    return np.clip(f_new, 0.0, 1000.0)


def try_step_size(f, g, f_score, sum_target):
    """Try descending step sizes, return (f_new, score) if improved."""
    g_norm = g.sum()
    if g_norm < 1e-12:
        return None, None
    g_scaled = g / g_norm * sum_target
    for t in [0.5, 0.3, 0.15, 0.05, 0.02]:
        f_new = (1.0 - t) * f + t * g_scaled
        f_new = np.clip(f_new, 0.0, 1000.0)
        if f_new.sum() < 0.01:
            continue
        s = evaluate_sequence(f_new.tolist())
        if np.isfinite(s) and s < f_score:
            return f_new, s
    return None, None


def optimize_one(n, init, budget_s, rng):
    """Run LP + gradient optimization on one initialization."""
    deadline = time.monotonic() + budget_s
    f = init.copy()
    s_target = np.sqrt(2.0 * n)
    f = f / max(f.sum(), 1e-30) * s_target
    f = np.clip(f, 0.0, 1000.0)
    f_score = evaluate_sequence(f.tolist())
    if not np.isfinite(f_score):
        return None, float('inf')

    best_f = f.copy()
    best_score = f_score
    print(f"  Start score: {best_score:.6f}")

    # Phase 1: LP-based
    lp_budget = 0.55 * budget_s
    lp_deadline = time.monotonic() + lp_budget
    iteration = 0
    lp_solved = 0
    while time.monotonic() < lp_deadline:
        iteration += 1
        c = np.convolve(f, f)
        M = c.max()
        # Select positions: start with all, then top-k
        if iteration <= 2:
            positions = list(range(len(c)))
        else:
            top_k = min(400, len(c))
            positions = list(np.argsort(c)[-top_k:])

        g = solve_lp_direction(f, M, positions)
        if g is None:
            # Retry with fewer constraints
            top_k = min(150, len(c))
            positions = list(np.argsort(c)[-top_k:])
            g = solve_lp_direction(f, M, positions)
        if g is None:
            if iteration > 5:
                print(f"  LP failed permanently at iter {iteration}")
                break
            continue

        lp_solved += 1
        f_new, new_score = try_step_size(f, g, f_score, s_target)
        if f_new is not None:
            f = f_new
            f_score = new_score
            if f_score < best_score:
                best_score = f_score
                best_f = f.copy()
                if iteration <= 20 or iteration % 20 == 0:
                    print(f"  LP iter {iteration}: score = {best_score:.6f} "
                          f"(lp_solved={lp_solved})")
        else:
            if lp_solved > 10:
                break

    # Phase 2: Gradient refinement
    grad_deadline = deadline
    f = best_f.copy()
    f_score = best_score
    step = 0
    stuck = 0
    while time.monotonic() < grad_deadline:
        step += 1
        c = np.convolve(f, f)
        M = c.max()
        # Top peaks
        top_k = min(30, len(c))
        # Only use positions where c is significantly above average
        c_thresh = c.mean() * 1.1
        significant = np.where(c > c_thresh)[0]
        if len(significant) == 0:
            significant = np.argsort(c)[-top_k:]
        else:
            if len(significant) > top_k:
                significant = significant[np.argsort(c[significant])[-top_k:]]
            else:
                significant = significant

        pressure = gradient_pressure(f, significant)
        lr = 0.08 / (1.0 + 0.01 * step)
        f_new = multiplicative_update(f, pressure, lr)
        if f_new.sum() < 0.01:
            # Restore from best
            f = best_f.copy()
            f_score = best_score
            continue
        f_new = f_new / f_new.sum() * s_target
        f_new = np.clip(f_new, 0.0, 1000.0)
        new_score = evaluate_sequence(f_new.tolist())
        if np.isfinite(new_score) and new_score < f_score:
            f = f_new
            f_score = new_score
            stuck = 0
            if f_score < best_score:
                best_score = f_score
                best_f = f.copy()
                if step <= 50 or step % 100 == 0:
                    print(f"  Grad step {step}: score = {best_score:.6f}")
        else:
            stuck += 1
            # Occasionally try a random perturbation to escape
            if stuck % 50 == 0:
                noise = rng.normal(0, 0.05, n)
                f_pert = f * np.exp(noise)
                f_pert = np.clip(f_pert, 0, 1000)
                if f_pert.sum() > 0.01:
                    f_pert = f_pert / f_pert.sum() * s_target
                    p_score = evaluate_sequence(f_pert.tolist())
                    if np.isfinite(p_score) and p_score < f_score:
                        f = f_pert
                        f_score = p_score
                        if p_score < best_score:
                            best_score = p_score
                            best_f = f.copy()

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main search: try multiple initializations, return best sequence."""
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # Determine n
    n = 3000
    parent = None
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if parent_arr.ndim == 1 and parent_arr.size > 100:
            parent = np.clip(parent_arr, 0.0, 1000.0)
            n = parent.size
    except (NameError, Exception):
        pass

    print(f"Chosen n = {n}")

    best_overall = None
    best_score_overall = float('inf')

    # Build list of initializations
    inits = []
    if parent is not None:
        inits.append(('parent', make_init_parent(parent, n)))
    inits.append(('gauss_1/3', make_init_gaussian(n, 1.0/3.0, 0.25, rng)))
    inits.append(('gauss_1/4', make_init_gaussian(n, 0.25, 0.2, rng)))
    inits.append(('gauss_2/3', make_init_gaussian(n, 2.0/3.0, 0.25, rng)))
    inits.append(('random', make_init_random(n, rng)))
    inits.append(('gauss_0.5', make_init_gaussian(n, 0.5, 0.3, rng)))

    # Time per init: give more time to better initializations
    time_per_init = max(60.0, (budget_s - 30.0) / len(inits))

    for name, init in inits:
        if time.monotonic() > total_deadline:
            break
        remaining = total_deadline - time.monotonic()
        if remaining < 30:
            break
        t_alloc = min(time_per_init, remaining)
        print(f"\n=== Init: {name}, time alloc = {t_alloc:.0f}s ===")
        try:
            result_f, result_score = optimize_one(n, init, t_alloc, rng)
            if result_f is not None and result_score < best_score_overall:
                best_score_overall = result_score
                best_overall = result_f.copy()
                print(f"  New best: {best_score_overall:.6f} (from {name})")
        except Exception as e:
            print(f"  Error with init {name}: {e}")
            continue

    # If we still have time, do a final gradient polish on the best
    if best_overall is not None and time.monotonic() < total_deadline:
        remaining = total_deadline - time.monotonic()
        if remaining > 10:
            print(f"\n=== Final polish, {remaining:.0f}s remaining ===")
            try:
                polished, pscore = optimize_one(n, best_overall, remaining, rng)
                if polished is not None and pscore < best_score_overall:
                    best_score_overall = pscore
                    best_overall = polished
            except Exception:
                pass

    # Return best or fallback
    if best_overall is None:
        # Fallback: return a simple valid sequence
        best_overall = np.ones(n, dtype=np.float64)
        best_score_overall = evaluate_sequence(best_overall.tolist())

    result = [float(x) for x in best_overall]
    print(f"\nFinal score: {best_score_overall:.6f}")
    return result
