```python
import numpy as np
import time
from scipy.optimize import linprog

def solve_lp(f, M):
    """Solve the LP to find b that maximizes sum(b) s.t. (f*b)_k <= M, b >= 0."""
    n = len(f)
    # Variables: b_0, ..., b_{n-1} (n variables)
    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: (f * b)_k <= M for k = 0, ..., 2n-2
    # (f * b)_k = sum_i f_i * b_{k-i} where the sum is over valid i
    # For a fixed k, the constraint is: sum_i f_i * b_{k-i} <= M
    # Let's build the constraint matrix A such that A @ b <= M * ones(2n-1)
    
    # A is (2n-1) x n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i  # index in b
            if 0 <= j < n:
                A[k, j] = f[i]
    
    b_constraints = np.full(2*n-1, M)
    
    # Solve
    result = linprog(c, A_ub=A, b_ub=b_constraints, bounds=[(0, None)] * n, method='highs')
    return result
```

```python
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
```
