Uses subgradient methods with adaptive steps, LP direction finding, and multi-length exploration.

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def build_constraint_row(a, k):
    """Build the row of the constraint matrix for convolution at lag k."""
    n = len(a)
    row = np.zeros(n)
    i_min = max(0, k - n + 1)
    i_max = min(n - 1, k)
    idx = np.arange(i_min, i_max + 1)
    row[idx] = a[k - idx]
    return row

def subgradient_topk(a, K=5):
    """Compute weighted subgradient using top-K peaks of conv(a,a)."""
    n = len(a)
    conv = fast_convolve(a)
    # Get top K peaks
    top_indices = np.argpartition(conv, -K)[::-1][:K]
    # Weight by how close they are to the max
    vals = conv[top_indices]
    max_val = vals[0]
    weights = np.exp(-10 * (max_val - vals) / max(1e-10, max_val))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        # d/d a[i] of conv[k] = a[k-i] + a[i-k] (when both valid)
        # Actually conv[k] = sum_j a[j]*a[k-j], d/d a[i] = a[k-i] (if valid) + a[i-k] (if valid, and i != k-i)
        # More carefully: if k-i is in [0,n-1], contributes a[k-i]
        # if i-k is in [0,n-1], contributes a[i-k]
        # When i = k-i (i.e., 2i=k), these are the same term, contributing 2*a[i]
        
        # For idx = i_min..i_max, k-idx ranges from k-i_max to k-i_min
        # a[k-idx] is valid when 0 <= k-idx < n
        # a[idx-k] is valid when 0 <= idx-k < n
        
        a_vals = a[k - idx]  # valid since idx is chosen so that k-idx is in range
        grad[idx] += w * a_vals
        
        # Second term: a[i-k] for i in range where i-k is valid
        j_min = max(0, k)  # i must be >= k for i-k >= 0
        j_max = min(n - 1, k + n - 1)  # i-k <= n-1
        # Actually i-k >= 0 means i >= k, and i-k < n means i < k+n
        # Combined with 0 <= i < n: i in [max(0,k), min(n-1, k+n-1)]
        if k < n:  # only if there's overlap
            j_idx = np.arange(j_min, j_max + 1)
            grad[j_idx] += w * a[j_idx - k]
    
    return grad, conv

def polyak_step(a, f_curr, f_best, grad, gamma=1.0):
    """Polyak step size."""
    g_norm = np.linalg.norm(grad)
    if g_norm < 1e-15:
        return 0.0
    return gamma * (f_curr - f_best) / (g_norm ** 2)

"""
Multi-method optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses LP direction finding, subgradient methods with Polyak steps, bundle-like local search,
and multi-length exploration with adaptive time allocation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def compute_subgradient_topk(a, K=5, temperature=0.1):

    n = len(a)
    conv = fast_convolve(a)
    num_k = len(conv)
    
    if K >= num_k:
        K = num_k

    max_conv = conv.max()
    if max_conv < 1e-15:
        return np.zeros(n), conv, max_conv

    scaled = (conv - max_conv) / (temperature * max_conv + 1e-15)
    weights = np.exp(scaled)

    threshold = weights.max() * 1e-6
    weights[weights < threshold] = 0

    if w_sum < 1e-15:

        weights[np.argmax(conv)] = 1.0

    A_full = np.fft.fft(a, fft_size)
    term1 = np.fft.ifft(A_full * np.conj(W_full)).real[:n]
    
    grad = term1 + term2
    return grad, conv, max_conv


def build_lp_full(a, k_indices, M):
    """Build LP constraints for specified convolution lags."""
    n = len(a)
    num_rows = len(k_indices)
    A_ub = np.zeros((num_rows, n))
    b_ub = np.full(num_rows, M)
    
    for idx, k in enumerate(k_indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min > i_max:
            continue
        col_idx = np.arange(i_min, i_max + 1)
        A_ub[idx, col_idx] = a[k - col_idx]
    
    return A_ub, b_ub


def solve_lp_direction(a, M, k_indices, method='highs'):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for k in k_indices."""
    n = len(a)
    A_ub, b_ub = build_lp_full(a, k_indices, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method,
                      options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize_v2(a0, time_limit, step_init=0.9, n_topk=50, use_full_final=False,
                   seed=0, verbose=False):
    """Improved LP-based optimization with adaptive top-k and full LP polish."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    num_k = 2 * n - 1
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        conv = fast_convolve(a)
        M = conv.max()
        
        # Choose constraints: top-k peaks, or all if remaining time allows
        if use_full_final and remaining > 60 and no_improve > 30:
            k_idx = np.arange(num_k)
        elif n_topk < num_k:
            k_idx = np.argpartition(conv, -n_topk)[-n_topk:]
        else:
            k_idx = np.arange(num_k)
        
        b = solve_lp_direction(a, M, k_idx)
        if b is None:
            # Fall back to fewer constraints
            if n_topk > 10:
                k_idx = np.argpartition(conv, -max(10, n_topk // 2))[-max(10, n_topk // 2):]
                b = solve_lp_direction(a, M, k_idx)
            if b is None:
                break
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum
        
        # Try update with current step
        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Slightly increase step on improvement
            step = min(1.0, step * 1.02)
        else:
            no_improve += 1
            if step > 0.1:
                step *= 0.9
            else:
                if no_improve > 20:
                    # Perturb to escape
                    noise = rng.standard_normal(n) * (0.005 + 0.002 * np.sqrt(no_improve / 20))
                    a = project_simplex(a + noise)
                    step = step_init * 0.6
                    no_improve = 0
        
        total_iters += 1
        if verbose and total_iters % 100 == 0:
            print(f"    LP2: obj={best_obj:.6f}, step={step:.3f}, no_imp={no_improve}, t={time.monotonic()-t0:.1f}s")
    
    conv_final = fast_convolve(best_a)
    return best_a, 2.0 * n * conv_final.max()


def subgradient_polish(a0, time_limit, K=10, temperature=0.05, seed=0, verbose=False):
    """Subgradient method with adaptive step size for local polish."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    f_curr = 2.0 * n * conv.max()
    f_best = f_curr
    best_a = a.copy()
    
    t0 = time.monotonic()
    step = 0.1
    no_improve = 0
    patience = 30
    avg_grad_norm = 1.0
    
    while time.monotonic() - t0 < time_limit:
        grad, conv, max_conv = compute_subgradient_topk(a, K=K, temperature=temperature)
        f_curr = 2.0 * n * max_conv
        
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        
        # Adaptive step: combine Polyak with running average
        if f_curr - f_best > 1e-10:
            step_polyak = 0.5 * (f_curr - f_best) / (g_norm ** 2 + 1e-15)
            step_polyak = min(step_polyak, 1.0)
        else:
            step_polyak = step
        
        step_use = min(step_polyak, step * 2)
        step_use = max(step_use, 0.01)
        
        a_new = a - step_use * grad
        a_new = project_simplex(a_new)
        
        conv_new = fast_convolve(a_new)
        f_new = 2.0 * n * conv_new.max()
        
        if f_new < f_best - 1e-13:
            f_best = f_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new
            no_improve += 1
            if no_improve > patience:
                noise = rng.standard_normal(n) * 0.002
                a = project_simplex(a + noise)
                step = max(0.05, step * 0.7)
                no_improve = 0
                patience = int(patience * 1.2)
                temperature = min(0.2, temperature * 1.1)
        
        # Decay step
        step = max(0.02, step * 0.998)
        avg_grad_norm = 0.9 * avg_grad_norm + 0.1 * g_norm
        
        if verbose and no_improve % 50 == 0:
            print(f"    SG: f={f_best:.6f}, step={step_use:.4f}, t={time.monotonic()-t0:.1f}s")
    
    return best_a, f_best


def mirror_descent_polish(a0, time_limit, lr_init=0.1, seed=0):
    """Mirror descent (exponential weighting) on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    f_best = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        grad, conv, max_conv = compute_subgradient_topk(a, K=8, temperature=0.08)
        f_curr = 2.0 * n * max_conv
        
        # Mirror descent update: a_new[i] = a[i] * exp(-lr * grad[i]) / Z
        updated = a * np.exp(-lr * grad)
        s_new = updated.sum()
        if s_new < 1e-15:
            break
        a_new = updated / s_new
        
        conv_new = fast_convolve(a_new)
        f_new = 2.0 * n * conv_new.max()
        
        if f_new < f_best - 1e-13:
            f_best = f_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new
            no_improve += 1
            if no_improve > 40:
                lr *= 0.7
                noise = rng.standard_normal(n) * 0.003
                a = project_simplex(a + noise)
                no_improve = 0
        
        lr *= 0.999
    
    return best_a, f_best


def generate_initializations(n, parent, rng):
    """Generate diverse normalized initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Cosine windows with different shapes
    for alpha in [1.0, 1.5, 2.0, 0.7]:
        c = np.cos(np.pi * x / (n - 1)) ** alpha if n > 1 else np.ones(1)
        c = np.maximum(c, 0.001)
        c /= c.sum()
        inits[f'cos_{alpha:.1f}'] = c
    
    # Gaussian (center)
    for sf in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.2f}'] = g
    
    # Bimodal
    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    
    # Triangle (center peak)
    tri = np.minimum(x, n - 1 - x).astype(np.float64) + 1.0
    tri /= tri.sum()
    inits['tri_center'] = tri
    
    # Triangle (edge peaks)
    tri_e = (np.minimum(x, n - 1 - x).max() - np.minimum(x, n - 1 - x)).astype(np.float64) + 1.0
    tri_e /= tri_e.sum()
    inits['tri_edge'] = tri_e
    
    # Random (several)
    inits['rand_sparse'] = rng.dirichlet(np.ones(n) * 0.05)
    inits['rand_uniform'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand_mid'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(x, np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps
    
    # Concatenated parent (if parent is shorter)
    if parent is not None and len(parent) < n:
        reps = int(np.ceil(n / len(parent)))
        cc = np.tile(parent, reps)[:n]
        cc = np.maximum(cc, 0)
        cs = cc.sum()
        if cs > 1e-10:
            inits['parent_tile'] = cc / cs
    
    return inits


def evaluate_normalized(a):
    """Evaluate objective assuming sum(a)≈1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LP direction finding, subgradient methods,
    and mirror descent. Explores multiple sequence lengths with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: length={len(p)}, sum={p.sum():.2f}")
    except NameError:
        pass
    except Exception:
        pass
    
    def update_best(a, source=""):
        nonlocal best_sequence, best_score
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  NEW BEST ({source}): score={score:.6f}, n={len(a)}")
            return True
        return False
    
    # ========== Phase 1: Quick multi-length scan ==========
    print("=== Phase 1: Quick Scan ===")
    t_p1 = time.monotonic()
    p1_time = 180.0
    
    # Candidate lengths
    if parent is not None:
        pn = len(parent)
        lengths = sorted(set([pn, max(64, pn - 64), min(1024, pn + 64),
                              max(64, pn - 128), min(1024, pn + 128),
                              256, 320, 384, 512]))
        lengths = [l for l in lengths if 64 <= l <= 1024]
    else:
        lengths = [128, 256, 384, 512, 640]
    
    scan_results = []
    
    for n in lengths:
        if time.monotonic() + 20 > t_p1 + p1_time:
            break
        inits = generate_initializations(n, parent, rng)
        
        # Prioritize
        priority = ['parent', 'parent_interp', 'parent_tile', 'cos_1.0', 'gauss_0.25',
                    'bimodal', 'rand_mid', 'tri_center', 'cos_1.5']
        names = [nm for nm in priority if nm in inits] + \
                [nm for nm in inits if nm not in priority]
        names = names[:4]
        
        time_per = max(8.0, (p1_time - (time.monotonic() - t_p1) - 15) /
                       max(1, len(names)) / max(1, len(lengths) - lengths.index(n) - 1))
        
        for name in names:
            if time.monotonic() + 8 > t_p1 + p1_time:
                break
            t_alloc = min(time_per, t_p1 + p1_time - time.monotonic() - 8)
            if t_alloc < 5:
                break
            
            a0 = inits[name]
            a_opt, obj_opt = lp_optimize_v2(a0, time_limit=t_alloc,
                                            step_init=0.9, n_topk=max(20, n // 10),
                                            seed=n + hash(name) % 1000)
            
            if update_best(a_opt, f"scan n={n} {name}"):
                pass
            scan_results.append((obj_opt, n, name, a_opt.copy()))
    
    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}")
    
    # ========== Phase 2: Deep LP optimization ==========
    print("=== Phase 2: Deep LP ===")
    t_p2 = time.monotonic()
    p2_time = 500.0
    
    # Collect starting points
    starts = []
    seen = set()
    for obj, n, name, a_opt in scan_results[:6]:
        key = (n, name)
        if key not in seen:
            starts.append((n, a_opt.copy(), name))
            seen.add(key)
        if len(starts) >= 5:
            break
    
    if best_sequence is not None:
        starts.append((len(best_sequence), best_sequence.copy(), "best"))
    
    # Add fresh starts
    if parent is not None:
        best_n = len(best_sequence) if best_sequence is not None else (len(parent) if parent is not None else 320)
    else:
        best_n = 320
    
    inits_best = generate_initializations(best_n, parent, rng)
    for extra in ['gauss_0.35', 'cos_2.0', 'rand_sparse']:
        if extra in inits_best:
            starts.append((best_n, inits_best[extra].copy(), extra))
    
    # LP configurations
    configs = [
        {'step_init': 1.0, 'n_topk': max(30, best_n // 8)},
        {'step_init': 0.7, 'n_topk': max(50, best_n // 5)},
        {'step_init': 1.0, 'n_topk': max(20, best_n // 12)},
        {'step_init': 0.5, 'n_topk': max(80, best_n // 4)},
        {'step_init': 0.8, 'n_topk': None},  # all constraints
    ]
    
    total_runs = len(starts) * len(configs)
    time_per_run = max(15.0, p2_time / max(1, total_runs))
    
    run_i = 0
    for i, (n_s, a_s, name_s) in enumerate(starts):
        for j, cfg in enumerate(configs):
            if time.monotonic() + 15 > t_p2 + p2_time:
                break
            t_alloc = min(time_per_run, t_p2 + p2_time - time.monotonic() - 15)
            if t_alloc < 10:
                break
            
            nk = cfg['n_topk']
            if nk is not None:
                nk = max(10, int(nk * n_s / best_n))
            
            a_opt, obj_opt = lp_optimize_v2(a_s, time_limit=t_alloc,
                                            step_init=cfg['step_init'], n_topk=nk,
                                            seed=run_i * 7 + 3)
            run_i += 1
            if update_best(a_opt, f"deep {i}-{j} n={n_s}"):
                pass
    
    print(f"Phase 2 done. Best: {best_score:.6f}")
    
    # ========== Phase 3: Subgradient + Mirror Descent polish ==========
    print("=== Phase 3: SG + MD Polish ===")
    t_p3 = time.monotonic()
    p3_time = 350.0
    
    if best_sequence is not None:
        n_b = len(best_sequence)
        
        # SG configs
        sg_cfgs = [
            {'K': 10, 'temperature': 0.05},
            {'K': 20, 'temperature': 0.08},
            {'K': 50, 'temperature': 0.12},
            {'K': 5, 'temperature': 0.03},
        ]
        
        time_per_sg = p3_time / (len(sg_cfgs) + 4)
        
        for i, cfg in enumerate(sg_cfgs):
            if time.monotonic() + 10 > t_p3 + p3_time:
                break
            t_alloc = min(time_per_sg, t_p3 + p3_time - time.monotonic() - 10)
            if t_alloc < 10:

            a_opt, obj_opt = subgradient_polish(best_sequence, time_limit=t_alloc,
                                                seed=i * 31 + 7, **cfg)
            if update_best(a_opt, f"SG {i}"):
                pass
        
        # Mirror descent
        md_cfgs = [
            {'lr_init': 0.1},
            {'lr_init': 0.05},
            {'lr_init': 0.2},
        ]
        
        for i, cfg in enumerate(md_cfgs):
            if time.monotonic() + 10 > t_p3 + p3_time:
                break
            t_alloc = min(time_per_sg, t_p3 + p3_time - time.monotonic() - 10)
            if t_alloc < 10:
                break
            a_opt, obj_opt = mirror_descent_polish(best_sequence, time_limit=t_alloc,
                                                   seed=100 + i, **cfg)
            if update_best(a_opt, f"MD {i}"):
                pass
    
    print(f"Phase 3 done. Best: {best_score:.6f}")
    
    # ========== Phase 4: Alternating hybrid ==========
    print("=== Phase 4: Hybrid Alternating ===")
    t_p4 = time.monotonic()
    p4_time = 350.0
    
    if best_sequence is not None:
        n_b = len(best_sequence)
        a = best_sequence.copy()
        
        hybrid_cfgs = [
            {'lp_step': 0.8, 'lp_topk': max(40, n_b // 6), 'sg_K': 10, 'sg_temp': 0.06},
            {'lp_step': 1.0, 'lp_topk': max(60, n_b // 4), 'sg_K': 20, 'sg_temp': 0.1},
            {'lp_step': 0.5, 'lp_topk': max(20, n_b // 10), 'sg_K': 50, 'sg_temp': 0.15},
        ]
        
        for cfg_i, hcfg in enumerate(hybrid_cfgs):
            if time.monotonic() + 60 > t_p4 + p4_time:
                break
            t_alloc = min(p4_time / len(hybrid_cfgs), t_p4 + p4_time - time.monotonic() - 30)
            if t_alloc < 30:
                break
            
            a_curr = a.copy()
            best_phase = a.copy()
            best_phase_obj = evaluate_normalized(a)
            
            phase_t0 = time.monotonic()
            step = hcfg['lp_step']
            no_imp = 0
            
            while time.monotonic() - phase_t0 < t_alloc:
                # LP step
                conv = fast_convolve(a_curr)
                M = conv.max()
                num_k = len(conv)
                if hcfg['lp_topk'] < num_k:
                    k_idx = np.argpartition(conv, -hcfg['lp_topk'])[-hcfg['lp_topk']:]
                else:
                    k_idx = np.arange(num_k)
                
                b = solve_lp_direction(a_curr, M, k_idx)
                if b is not None:
                    bs = b.sum()
                    if bs > 1e-15:
                        b = b / bs
                        a_curr = (1.0 - step) * a_curr + step * b
                        a_curr = np.maximum(a_curr, 0.0)
                        ss = a_curr.sum()
                        if ss > 1e-15:
                            a_curr = a_curr / ss
                
                # SG step
                grad, conv2, _ = compute_subgradient_topk(a_curr, K=hcfg['sg_K'],
                                                           temperature=hcfg['sg_temp'])
                g_norm = np.linalg.norm(grad)
                if g_norm > 1e-15:
                    sg_step = 0.05
                    a_curr = a_curr - sg_step * grad
                    a_curr = project_simplex(a_curr)
                
                conv3 = fast_convolve(a_curr)
                obj = 2.0 * n_b * conv3.max()
                
                if obj < best_phase_obj - 1e-13:
                    best_phase_obj = obj
                    best_phase = a_curr.copy()
                    no_imp = 0
                else:
                    no_imp += 1
                    if no_imp > 30:
                        noise = rng.standard_normal(n_b) * 0.003
                        a_curr = project_simplex(a_curr + noise)
                        no_imp = 0
                        step = max(0.3, step * 0.8)
                
                step = max(0.2, step * 0.999)
            
            if update_best(best_phase, f"hybrid {cfg_i}"):
                a = best_phase.copy()
    
    print(f"Phase 4 done. Best: {best_score:.6f}")
    
    # ========== Phase 5: Final polish ==========
    print("=== Phase 5: Final Polish ===")
    t_p5 = time.monotonic()
    remaining = deadline - time.monotonic()
    
    if best_sequence is not None and remaining > 30:

        # Full LP polish (all constraints)

                                            step_init=0.4, n_topk=2 * n_b - 1,
                                            use_full_final=True, seed=9999)
            if update_best(a_opt, "polish_LP_full"):

        # SG final

        if t_alloc > 15:

                                                K=30, temperature=0.03, seed=7777)

        # Tiny perturbations
        rem2 = deadline - time.monotonic() - 20
        if rem2 > 10:

            for i in range(n_tries):

                    if update_best(a_pert, f"perturb {i} s={scale}"):
                        break
    
    # Final verification
    if best_sequence is None:

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
