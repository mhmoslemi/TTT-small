"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using L-BFGS-B on softmax-parameterized
log-sum-exp smoothed objective, with multi-phase search across lengths,
progressive beta annealing, coordinate descent polishing, and LP-based
direction finding for peak reduction.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_obj_grad(a, beta):
    """
    Compute log-sum-exp smoothed objective and gradient.
    f(a) = LSE(beta * conv(a,a)) / (sum a)^2  [times 2n for the actual score]
    We just minimize LSE(beta * conv) since (sum a)^2 is constant when normalized.
    """
    n = len(a)
    conv = fast_convolve(a)
    cmax = conv.max()
    scaled = beta * (conv - cmax)
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    p_sum = exp_scaled.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(conv)
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_scaled / p_sum
    
    # Objective (up to constant factor 2n / sum^2)
    obj = cmax + np.log(p_sum) / beta
    
    # Gradient: grad[j] = 2 * sum_k p[k] * a[k-j] (cross-correlation)
    w = conv * p  # w[k] = conv[k] * p[k]
    fft_size = 1 << (3 * n - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(A * np.conj(W), fft_size)
    grad = 2.0 * cross[:n]
    
    return obj, grad


def softmax_objective(x, beta):
    """Objective for L-BFGS-B in log-space."""
    # Numerically stable softmax
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a_sum = a.sum()
    if a_sum < 1e-300:
        a = np.ones(len(x)) / len(x)
    else:
        a = a / a_sum
    obj, _ = lse_obj_grad(a, beta)
    return obj


def softmax_gradient(x, beta):
    """Gradient for L-BFGS-B in log-space."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a_sum = a.sum()
    if a_sum < 1e-300:
        a = np.ones(len(x)) / len(x)
    else:
        a = a / a_sum
    
    _, grad_a = lse_obj_grad(a, beta)
    # Chain rule: dx_i/dx_j = a_i * (delta_ij - a_j)
    # grad_x[i] = a_i * (grad_a[i] - sum_j a_j * grad_a[j])
    mean_grad = a @ grad_a
    grad_x = a * (grad_a - mean_grad)
    return grad_x


def optimize_lbfgsb(a0, time_limit, beta_init=30.0, beta_end=1000.0, seed=0):
    """
    L-BFGS-B optimization with progressive beta increase.
    Uses softmax parameterization for unconstrained optimization.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a_sum = a.sum()
    if a_sum < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / a_sum
    
    # Initialize x from a: x_i = log(a_i)
    x = np.log(a)
    x -= x.mean()  # center for numerical stability
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    
    # Progressive beta schedule
    n_phases = 4
    betas = [beta_init * (beta_end / beta_init) ** (i / (n_phases - 1)) for i in range(n_phases)]
    # e.g., [30, 95, 300, 1000]
    
    for phase, beta in enumerate(betas):
        phase_time = (time.monotonic() - t0) / (phase + 1)
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        phase_budget = min(remaining * 0.7 / (n_phases - phase), remaining - 5)
        if phase_budget < 5:
            break
        
        # Convert current a to x
        x = np.log(np.maximum(a, 1e-300))
        x -= x.mean()
        
        # Run L-BFGS-B
        result = minimize(
            softmax_objective, x, args=(beta,),
            jac=softmax_gradient,
            method='L-BFGS-B',
            options={'maxiter': 5000, 'maxfun': 10000, 
                     'ftol': 1e-12, 'gtol': 1e-8,
                     'maxls': 50}
        )
        
        # Convert back
        x_opt = result.x - result.x.mean()
        a = np.exp(x_opt)
        a_sum = a.sum()
        if a_sum < 1e-15:
            a = best_a
        else:
            a = a / a_sum
            a = np.maximum(a, 0)
        
        obj = fast_convolve(a).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        print(f"    Phase {phase} (beta={beta:.0f}): obj={obj:.8f}", flush=True)
    
    return best_a, best_obj

def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_obj_and_grad(a, beta):
    """Log-sum-exp smoothed max convolution and its gradient w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    cmax = conv.max()
    scaled = beta * (conv - cmax)
    scaled = np.clip(scaled, -500, 500)
    exp_s = np.exp(scaled)
    p_sum = exp_s.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(conv)
        p[np.argmax(conv)] = 1.0
    else:
        p = exp_s / p_sum
    obj = cmax + np.log(p_sum) / beta
    w = conv * p
    fft_size = 1 << (3 * n - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(A * np.conj(W), fft_size)
    grad = 2.0 * cross[:n]
    return obj, grad


def var_obj_and_grad(a):
    """Variance of convolution values and gradient. Smooth everywhere."""
    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    if s < 1e-15:
        return 0.0, np.zeros(n)
    mu = (s ** 2) / (2 * n - 1)
    diff = conv - mu
    obj = np.sum(diff ** 2)
    # grad[j] = 2 * sum_k diff[k] * d(conv[k])/da[j] ≈ 4 * (diff ⋆ a)[j]
    fft_size = 1 << (3 * n - 1).bit_length()
    A = rfft(a, fft_size)
    D = rfft(diff, fft_size)
    cross = irfft(A * np.conj(D), fft_size)
    grad = 4.0 * cross[:n]
    return obj, grad

def lp_direction(a, M, n, time_limit=15):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k] <= M."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': time_limit})
        if not result.success or result.x is None:
            return None
        b = np.maximum(result.x, 0)
        b_sum = b.sum()
        if b_sum < 1e-10:
            return None
        return b / b_sum
    except Exception:
        return None

            ps = float(evaluate_sequence(p.tolist()))
            if np.isfinite(ps):
                best_score = ps
                best_sequence = p.copy()
            print(f"Parent: n={len(p)}, score={best_score:.6f}", flush=True)
    except Exception:
        pass
    
    # Phase 1: Quick exploration
    print("=== Phase 1: Exploration ===", flush=True)
    lengths = [192, 256, 320, 384, 512]
    scan_results = []
    t_p1 = time.monotonic()
    p1_budget = 280.0
    
    for n in lengths:
        if time.monotonic() > deadline - 600:
            break
        n_global = n
        inits = generate_initializations(n, parent, rng)
        names = ['parent', 'parent_interp', 'bell', 'gauss', 'cos',
                 'dirichlet', 'uniform', 'bimodal', 'gauss_off', 'dirichlet2']
        names = [nm for nm in names if nm in inits][:5]
        
        per_time = max(8.0, (p1_budget - (time.monotonic() - t_p1)) / max(1, len(names) * len(lengths)))
        
        for name in names:
            if time.monotonic() > deadline - 580:
                break
            t_alloc = min(per_time, 30.0)
            if t_alloc < 5:
                break
            a0 = inits[name]
            a_opt, obj_opt = optimize_lbfgsb(a0, t_alloc, beta_init=30, beta_end=200,
                                             n_phases=3, seed=int(rng.integers(0, 100000)))
            score = float(evaluate_sequence(a_opt.tolist()))
            scan_results.append((score, n, name, a_opt.copy()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  n={n} {name}: {score:.6f}", flush=True)
    
    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}", flush=True)
    
    # Phase 2: Deep refinement
    print("=== Phase 2: Deep Refinement ===", flush=True)
    
    top_configs = []
    seen = set()
    for sc, n, nm, ao in scan_results:
        key = (n, nm)
        if key not in seen:
            top_configs.append((n, ao))
            seen.add(key)
        if len(top_configs) >= 4:
            break
    
    if best_sequence is not None:
        top_configs.insert(0, (len(best_sequence), best_sequence.copy()))
    
    # Add a random restart at the best length
    best_n = len(best_sequence) if best_sequence is not None else 384
    top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 0.4)))
    top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 2.0)))
    
    remaining = deadline - time.monotonic() - 350.0
    n_cfg = len(top_configs)
    per_cfg = max(20.0, remaining / max(1, n_cfg))
    
    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 330:
            break
        t_alloc = min(per_cfg, 250.0)
        if t_alloc < 15:
            break
        n_global = n
        print(f"  Config {i}: n={n}", flush=True)
        
        # Sub A: L-BFGS-B with progressive beta
        t_a = t_alloc * 0.45
        a_a, _ = optimize_lbfgsb(a0, t_a, beta_init=30, beta_end=1500,
                                  n_phases=5, seed=int(rng.integers(0, 1000000)))
        sa = float(evaluate_sequence(a_a.tolist()))
        if np.isfinite(sa) and sa < best_score:
            best_score = sa
            best_sequence = a_a.copy()
            print(f"    L-BFGS: {sa:.6f}", flush=True)
        
        # Sub B: LP refinement
        t_b = t_alloc * 0.25
        if t_b > 10:
            a_b, _ = lp_refine(a_a, t_b, seed=int(rng.integers(0, 1000000)))
            sb = float(evaluate_sequence(a_b.tolist()))
            if np.isfinite(sb) and sb < best_score:
                best_score = sb
                best_sequence = a_b.copy()
                print(f"    LP: {sb:.6f}", flush=True)
            a_a = a_b
        
        # Sub C: L-BFGS-B with variance + LSE
        t_c = t_alloc * 0.30
        if t_c > 10:
            a_c, _ = optimize_lbfgsb(a_a, t_c, beta_init=200, beta_end=3000,
                                      n_phases=3, use_var=True, alpha=0.6,
                                      seed=int(rng.integers(0, 1000000)))
            sc = float(evaluate_sequence(a_c.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_sequence = a_c.copy()
                print(f"    LSE+Var: {sc:.6f}", flush=True)
    
    # Phase 3: Polish
    print("=== Phase 3: Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        n_global = len(best_sequence)
        
        # Peak shaving
        t_ps = remaining * 0.25
        if t_ps > 15:
            a_ps, _ = peak_shaving(best_sequence, t_ps, seed=777)
            s_ps = float(evaluate_sequence(a_ps.tolist()))
            if np.isfinite(s_ps) and s_ps < best_score:
                best_score = s_ps
                best_sequence = a_ps.copy()
                print(f"  PeakShave: {best_score:.6f}", flush=True)
        
        # Coordinate descent
        t_cd = remaining * 0.20
        if t_cd > 10:
            a_cd, _ = coordinate_descent(best_sequence, t_cd, seed=888)
            s_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(s_cd) and s_cd < best_score:
                best_score = s_cd
                best_sequence = a_cd.copy()
                print(f"  CoordDesc: {best_score:.6f}", flush=True)
        
        # Final L-BFGS-B with very high beta
        remaining = deadline - time.monotonic() - 15.0
        t_fg = remaining * 0.30
        if t_fg > 10:
            a_fg, _ = optimize_lbfgsb(best_sequence, t_fg, beta_init=500, beta_end=5000,
                                       n_phases=3, seed=1234)
            s_fg = float(evaluate_sequence(a_fg.tolist()))
            if np.isfinite(s_fg) and s_fg < best_score:
                best_score = s_fg
                best_sequence = a_fg.copy()
                print(f"  FinalL-BFGS: {best_score:.6f}", flush=True)
        
        # One more peak shaving
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 10:
            a_fs, _ = peak_shaving(best_sequence, remaining, seed=999)
            s_fs = float(evaluate_sequence(a_fs.tolist()))
            if np.isfinite(s_fs) and s_fs < best_score:
                best_score = s_fs
                best_sequence = a_fs.copy()
                print(f"  FinalShave: {best_score:.6f}", flush=True)
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
    
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]
