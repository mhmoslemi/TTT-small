"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an LP-based cutting-plane method
combined with L-BFGS-B polishing. The LP direction-finding step exploits the
convolution structure to compute a descent direction that monotonically reduces
the peak convolution value.

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def build_conv_matrix(f):
    """Build the convolution matrix A where (A @ g)[k] = conv(f, g)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        # conv(f, g)[k] = sum_m f[m] * g[k-m] for valid m
        m_start = max(0, k - (n - 1))
        m_end = min(n - 1, k)
        for m in range(m_start, m_end + 1):
            j = k - m  # index into g
            A[k, j] += f[m]
    return A


def lp_find_direction(f, M, method='highs'):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method=method)
    if res.success:
        return res.x
    return None


def optimize_lp_iterative(f0, max_iters=100, time_budget=60.0, verbose=True):
    """Iterative LP-based optimization."""
    n = len(f0)
    f = f0.copy()
    f = np.maximum(f, 0)
    f = f / f.sum()
    
    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    t_start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t_start > time_budget:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        g = lp_find_direction(f, M)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        
        # Rescale g to match sum(f)
        g = g * (f.sum() / g_sum)
        
        # Line search for step size
        best_t = 0.0
        best_new_obj = 2.0 * n * M
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_obj = 2.0 * n * fast_convolve(f_new).max()
            if new_obj < best_new_obj:
                best_new_obj = new_obj
                best_t = t
        
        if best_t == 0.0:
            # Try smaller steps
            for t in [0.01, 0.02, 0.05]:
                f_new = (1 - t) * f + t * g
                f_new = np.maximum(f_new, 0)
                s = f_new.sum()
                if s < 1e-10:
                    continue
                f_new = f_new / s
                new_obj = 2.0 * n * fast_convolve(f_new).max()
                if new_obj < best_new_obj:
                    best_new_obj = new_obj
                    best_t = t
        
        if best_t > 0:
            f = (1 - best_t) * f + best_t * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            obj = 2.0 * n * fast_convolve(f).max()
            if obj < best_obj:
                best_obj = obj
                best_f = f.copy()
                if verbose and it % 10 == 0:

        else:

            break
    
    return best_f, best_obj


def optimize_lbfgsb(a0, beta, maxiter=500):
    """Optimize using L-BFGS-B with softmax reparametrization."""
    n = len(a0)
    def obj_fn(x):
        a = np.exp(x - x.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, _ = smooth_max_and_weights(conv, beta)
        return 2.0 * n * sm
    
    def jac_fn(x):
        a = np.exp(x - x.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        # Gradient via finite differences would be slow; use FFT-based
        N = 1 << (len(w) + n - 1 + 63).bit_length()
        W = fft(w, N)
        A = fft(a, N)
        grad_a = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
        lagrange = np.dot(a, grad_a)
        return a * (grad_a - lagrange)
    
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)
    
    result = minimize(obj_fn, x0, jac=jac_fn, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxls': 50})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def generate_initializations(n, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp
    inits['ramp_dec'] = ramp[::-1].copy()
    
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri
    
    for sf in [0.1, 0.15, 0.2, 0.25, 0.3]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%s' % str(sf)] = g
    
    # Bimodal
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.1)
    
    # Flat spectrum (random phases)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = 1.0
    phases = rng.uniform(0, 2 * np.pi, fft_size)
    for k in range(1, fft_size // 2 + 1):
        A[k] = np.exp(1j * phases[k])
        if k < fft_size - k:
            A[fft_size - k] = np.exp(-1j * phases[k])
    a = np.real(ifft(A, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    inits['flat_spec'] = a
    
    # Cosine taper
    cos_taper = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_taper /= cos_taper.sum()
    inits['cos_taper'] = cos_taper
    
    # Sine taper
    sin_taper = np.sin(np.pi * (x + 1) / (n + 1))
    sin_taper /= sin_taper.sum()
    inits['sin_taper'] = sin_taper
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an LP-based cutting-plane method
combined with L-BFGS-B polishing. The LP direction-finding step exploits the
convolution structure to compute a descent direction that monotonically reduces
the peak convolution value. Uses active-set constraint reduction for speed.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def build_conv_matrix(f, active_lags=None):
    """Build the convolution matrix A where (A @ g)[k] = conv(f, g)[k].
    Only builds rows for active_lags if provided."""
    n = len(f)
    if active_lags is None:
        active_lags = range(2 * n - 1)
    k_list = list(active_lags)
    A = np.zeros((len(k_list), n))
    for idx, k in enumerate(k_list):
        m_start = max(0, k - (n - 1))
        m_end = min(n - 1, k)
        for m in range(m_start, m_end + 1):
            j = k - m
            A[idx, j] += f[m]
    return A, k_list


def lp_find_direction(f, M, max_constraints=None):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M, g >= 0.
    Uses active-set reduction if max_constraints is specified."""
    n = len(f)
    
    if max_constraints is not None:
        # Only use top max_constraints lags by current conv value
        conv_f = fast_convolve(f)
        top_lags = np.argsort(conv_f)[::-1][:max_constraints]
        A, k_list = build_conv_matrix(f, top_lags)
        b_ub = np.full(len(k_list), M)
    else:
        A, k_list = build_conv_matrix(f)
        b_ub = np.full(2 * n - 1, M)
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0)
            # Verify full feasibility
            if max_constraints is not None and max_constraints < 2 * n - 1:
                conv_fg = np.zeros(2 * n - 1)
                fft_size = 1 << (len(f) + len(g) - 1).bit_length()
                F = fft(f, fft_size)
                G = fft(g, fft_size)
                conv_fg = np.real(ifft(F * G, fft_size))[:2 * n - 1]
                if conv_fg.max() > M * 1.05:
                    # Re-solve with all constraints
                    A_full, _ = build_conv_matrix(f)
                    b_full = np.full(2 * n - 1, M)
                    res2 = linprog(c, A_ub=A_full, b_ub=b_full, bounds=bounds, method='highs')
                    if res2.success and res2.x is not None:
                        g = np.maximum(res2.x, 0)
                    else:
                        return None
            return g
    except Exception:
        pass
    return None


def optimize_lp_iterative(f0, max_iters=200, time_budget=60.0, verbose=True):
    """Iterative LP-based optimization with adaptive constraints."""
    n = len(f0)
    f = np.maximum(f0, 0.0).astype(np.float64)
    f_sum = f.sum()
    if f_sum < 1e-10:
        f = np.ones(n) / n
        f_sum = 1.0
    f = f / f_sum
    
    best_f = f.copy()
    best_obj = 2.0 * n * fast_convolve(f).max()
    t_start = time.monotonic()
    stagnation = 0
    
    for it in range(max_iters):
        if time.monotonic() - t_start > time_budget:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Adaptive number of constraints: start small, grow if needed
        n_constraints = min(2 * n - 1, 50 + it * 5)
        
        g = lp_find_direction(f, M, max_constraints=n_constraints)
        if g is None or g.sum() < 1e-10:
            break
        
        # Rescale g to match sum(f)
        g = g * (f_sum / g.sum())
        
        # Line search
        best_t = 0.0
        best_new_obj = best_obj
        for t in [0.3, 0.5, 0.7, 1.0, 0.1, 0.2, 0.8, 0.9]:
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 1e-10:
                continue
            f_new = f_new / s
            new_obj = 2.0 * n * fast_convolve(f_new).max()
            if new_obj < best_new_obj:
                best_new_obj = new_obj
                best_t = t
        
        if best_t > 0:
            f = (1 - best_t) * f + best_t * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            obj = 2.0 * n * fast_convolve(f).max()
            if obj < best_obj - 1e-9:
                best_obj = obj
                best_f = f.copy()
                stagnation = 0
                if verbose and it % 20 == 0:
                    print("    LP it=%d: obj=%.6f (M=%.6f, t=%.2f)" % (it, obj, M, best_t))
            else:
                stagnation += 1
                # Perturb to escape stagnation
                if stagnation >= 15:
                    noise = np.random.default_rng(it).standard_normal(n) * 0.01
                    f = project_simplex(f + noise)
                    stagnation = 0
        else:
            stagnation += 1
            if stagnation >= 10:
                break
    
    return best_f, best_obj


def optimize_lbfgsb(a0, beta, maxiter=500):
    """Optimize using L-BFGS-B with softmax reparametrization."""
    n = len(a0)
    
    def obj_fn(x):
        a = np.exp(x - x.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, _ = smooth_max_and_weights(conv, beta)
        return 2.0 * n * sm
    
    def jac_fn(x):
        a = np.exp(x - x.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        N = 1 << (len(w) + n - 1 + 63).bit_length()
        W = fft(w, N)
        A = fft(a, N)
        grad_a = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
        lagrange = np.dot(a, grad_a)
        return a * (grad_a - lagrange)
    
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)
    
    result = minimize(obj_fn, x0, jac=jac_fn, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxls': 50})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def generate_initializations(n, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp
    inits['ramp_dec'] = ramp[::-1].copy()
    
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri
    
    for sf in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%s' % str(sf)] = g
    
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random_conc'] = rng.dirichlet(np.ones(n) * 2.0)
    
    # Flat spectrum
    fft_size = 1 << (2 * n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = 1.0
    phases = rng.uniform(0, 2 * np.pi, fft_size)
    for k in range(1, fft_size // 2 + 1):
        A[k] = np.exp(1j * phases[k])
        if k < fft_size - k:
            A[fft_size - k] = np.exp(-1j * phases[k])
    a = np.real(ifft(A, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    inits['flat_spec'] = a
    
    # Cosine taper
    cos_taper = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_taper = np.maximum(cos_taper, 0.01)
    cos_taper /= cos_taper.sum()
    inits['cos_taper'] = cos_taper
    
    # Sine taper
    sin_taper = np.sin(np.pi * (x + 1) / (n + 1))
    sin_taper /= sin_taper.sum()
    inits['sin_taper'] = sin_taper
    
    # Multi-modal
    mm = np.zeros(n)
    for center in [n//4, n//2, 3*n//4]:
        sigma = n * 0.08
        mm += np.exp(-0.5 * ((x - center) / sigma) ** 2)
    mm /= mm.sum()
    inits['trimodal'] = mm
    
    return inits


def optimize_pg_smooth(a0, beta, lr=0.05, n_iter=300):
    """Projected gradient descent on simplex with smooth max objective."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    for it in range(n_iter):
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        N = 1 << (len(w) + n - 1 + 63).bit_length()
        W = fft(w, N)
        A = fft(a, N)
        grad_a = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
        
        norm_grad = np.linalg.norm(grad_a)
        if norm_grad < 1e-15:
            break
        
        t = lr
        for _ in range(8):
            a_new = project_simplex(a - t * grad_a)
            conv_new = fast_convolve(a_new)
            sm_new, _ = smooth_max_and_weights(conv_new, beta)
            obj_new = 2.0 * n * sm_new
            if obj_new <= obj + 1e-10:
                break
            t *= 0.5
        
        a = a_new
        lr *= 0.999
    
    return best_a, best_obj
