    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def coord_search_1d(L, T, D, S, ai, n_coarse=100, n_fine=200):
    """
    Find x >= 0 minimizing F(x) = max_k(D[k]*x^2 + L[k]*x + T[k]) / (x+S)^2.
    Uses two-stage grid search: coarse then fine.
    """
    # Determine search range
    x_max = max(3.0 * ai + S / max(1, len(S) // 2), 0.01)
    # Don't go too far: F(x) -> 1 as x->inf, so if current F(ai) < 1, optimal x < inf
    # But practically, x > 10*ai is rarely optimal
    x_max = max(x_max, ai + 0.1)
    x_max = min(x_max, 50.0 * S)  # cap for safety

    # Coarse grid
    xs_c = np.linspace(0, x_max, n_coarse)
    grid = D[None, :] * xs_c[:, None] ** 2 + L[None, :] * xs_c[:, None] + T[None, :]
    max_vals = grid.max(axis=1)
    denom = (xs_c + S) ** 2
    F_vals = np.where(denom > 1e-30, max_vals / denom, np.inf)
    i_best = F_vals.argmin()

    # Fine grid around best
    lo = xs_c[max(0, i_best - 1)]
    hi = xs_c[min(n_coarse - 1, i_best + 1)]
    xs_f = np.linspace(lo, hi, n_fine)
    grid = D[None, :] * xs_f[:, None] ** 2 + L[None, :] * xs_f[:, None] + T[None, :]
    max_vals = grid.max(axis=1)
    denom = (xs_f + S) ** 2
    F_vals = np.where(denom > 1e-30, max_vals / denom, np.inf)
    i_best_f = F_vals.argmin()

    # Also check x=0 explicitly
    F_at_0 = T.max() / (S * S) if S > 1e-15 else np.inf
    x_opt = xs_f[i_best_f]
    F_opt = F_vals[i_best_f]

    if F_at_0 < F_opt:
        x_opt = 0.0
        F_opt = F_at_0

    return x_opt, F_opt


def coordinate_descent(a0, time_limit=60.0, verbose=False):
    """
    Cyclic coordinate descent minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Works with unnormalized a (scale-invariant objective).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n

    c = fast_convolve(a)
    cur_obj = 2.0 * n * c.max() / (s * s)
    best_obj = cur_obj
    best_a = a.copy()

    t0 = time.monotonic()
    cycle = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved_this_cycle = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 1.0:
                break

            ai = a[i]
            S = s - ai  # sum of others

            if S < 1e-15:
                continue

            # Compute L[k], T[k], D[k]
            # c[k] = D[k]*x^2 + L[k]*x + T[k] for new value x of a[i]
            L = np.zeros(2 * n - 1)
            D = np.zeros(2 * n - 1)

            # a[i] appears in c[k] when j=i (gives a[i]*a[k-i]) or k-j=i (gives a[k-i]*a[i])
            # k-i must be in [0, n-1], so k in [i, i+n-1]
            k_start = max(0, i)
            k_end = min(2 * n - 2, i + n - 1)

            for k in range(k_start, k_end + 1):
                ki = k - i
                if 0 <= ki < n:
                    if ki != i:
                        L[k] = 2.0 * a[ki]
                    else:
                        D[k] = 1.0

            # T[k] = c[k] - D[k]*ai^2 - L[k]*ai
            T = c.copy()
            mask = (L != 0.0) | (D != 0.0)
            T[mask] -= D[mask] * ai * ai + L[mask] * ai

            # Search for optimal x
            x_new, F_new = coord_search_1d(L, T, D, S, ai)

            # Compare with current
            cur_local = c.max() / (s * s)
            if F_new < cur_local - 1e-14:
                a[i] = x_new
                s = a.sum()
                c[k_start:k_end + 1] = D[k_start:k_end + 1] * x_new * x_new + \
                    L[k_start:k_end + 1] * x_new + T[k_start:k_end + 1]
                # Recompute full conv to be safe (the update above might have errors at boundaries)
                c = fast_convolve(a)
                new_obj = 2.0 * n * c.max() / (s * s)
                if new_obj < best_obj - 1e-14:
                    best_obj = new_obj
                    best_a = a.copy()
                    no_improve = 0
                    improved_this_cycle = True
                else:
                    # Revert
                    a[i] = ai
                    s = a.sum()
                    c = fast_convolve(a)
            elif abs(x_new - ai) > 1e-12:
                # No improvement, don't update
                no_improve += 1

        cycle += 1
        if verbose and cycle % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD cycle={cycle}, obj={best_obj:.6f}, elapsed={elapsed:.1f}s")

        if no_improve > 50:
            break

    return best_a, best_obj


def pairwise_descent(a, time_limit=30.0, n_pairs_per_cycle=50, seed=0):
    """
    Pairwise coordinate descent: shift mass between pairs (i,j).
    Minimizes max_k c[k] (with sum fixed) by optimizing delta for each pair.
    """
    n = len(a)
    a = np.maximum(a.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    # Normalize to sum=1 for simplicity
    a = a / s

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for _ in range(n_pairs_per_cycle):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if a[i] < 1e-10 or a[j] < 1e-10:
                continue

            # We move delta from j to i: a[i]+=delta, a[j]-=delta
            # delta in [-ai, aj]
            # c[k] changes linearly in delta (except when k-i=j or k-j=i)
            # For now, approximate as linear: c[k] ≈ c[k] + delta*(a[k-i]-a[k-j])
            # (ignoring the quadratic terms from k-i=j or k-j=i)

            delta_lo = -a[i] + 1e-12
            delta_hi = a[j] - 1e-12
            if delta_hi <= delta_lo:
                continue

            # A[k] = change in c[k] per unit delta
            A = np.zeros(2 * n - 1)
            # a[i] increases by delta: adds delta * a[k-i] to c[k]
            # a[j] decreases by delta: subtracts delta * a[k-j] from c[k]
            for k in range(2 * n - 1):
                ki = k - i
                kj = k - j
                add = 0.0
                if 0 <= ki < n:
                    add += a[ki]
                if 0 <= kj < n:
                    add -= a[kj]
                # Correction: if ki == j, the term a[ki] = a[j] which is changing
                # if kj == i, the term a[kj] = a[i] which is changing
                # These create quadratic terms; ignore for approximation
                A[k] = add

            # f(delta) = max_k (c[k] + A[k]*delta)
            # Convex in delta. Find min via bisection on subgradient.
            # Subgradient at delta is A[k*] where k* = argmax
            lo = delta_lo
            hi = delta_hi
            for _ in range(30):
                mid = 0.5 * (lo + hi)
                vals = c + A * mid
                k_star = vals.argmax()
                if A[k_star] > 0:
                    hi = mid  # move left (decrease delta)
                elif A[k_star] < 0:
                    lo = mid  # move right (increase delta)
                else:
                    break

            delta_opt = 0.5 * (lo + hi)

            # Evaluate
            new_c_max = (c + A * delta_opt).max()
            if new_c_max < best_obj - 1e-12:
                a[i] += delta_opt
                a[j] -= delta_opt
                a = np.maximum(a, 0.0)
                c = fast_convolve(a)
                if c.max() < best_obj:
                    best_obj = c.max()
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                else:
                    # Revert
                    a[i] -= delta_opt
                    a[j] += delta_opt
            else:
                no_improve += 1

        if no_improve > 200:
            break

    return best_a / best_a.sum() if best_a.sum() > 1e-15 else best_a


def generate_inits(n, parent=None, rng=None):
    """Generate diverse initializations of length n."""
    if rng is None:
        rng = np.random.default_rng(42)

    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Ramp (increasing)
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    # Ramp reversed
    inits['ramp_rev'] = r[::-1] / r.sum()

    # Quadratic ramp
    q = np.arange(1, n + 1, dtype=np.float64) ** 2
    inits['quad'] = q / q.sum()

    # Edge concentrated (both edges)
    edge = max(1, n // 4)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    inits['both_edge'] = ec / ec.sum()

    # One edge
    oe = np.zeros(n)
    oe[:edge] = 1.0
    inits['one_edge'] = oe / oe.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Sine wave
    sw = np.abs(np.sin(np.pi * x / n)) ** 2 + 0.1
    inits['sine'] = sw / sw.sum()

    # Cosine taper
    ct = 0.5 * (1 + np.cos(2 * np.pi * x / (2 * n)))
    ct = np.maximum(ct, 0.01)
    inits['coston'] = ct / ct.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Sparse random
    sp = rng.random(n)
    sp[sp < 0.5] = 0.0
    if sp.sum() > 1e-10:
        inits['sparse'] = sp / sp.sum()

    # Two-point concentration
    tp = np.zeros(n)
    tp[0] = 1.0
    tp[-1] = 1.0
    inits['two_point'] = tp / tp.sum()

    # Parent initialization
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps

    return inits


def smooth_gradient_descent(a0, time_limit=60.0, beta_start=50.0, beta_end=500.0, verbose=False):
    """
    Smooth gradient descent using log-sum-exp approximation of max.
    Minimizes 2*n*beta*lse(beta*c)/sum^2 where lse is log-sum-exp.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0

    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max() / (s * s)
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    lr = 0.5
    v = np.zeros(n)  # momentum
    momentum = 0.9

    n_total = max(1, int(time_limit / 0.01))
    while time.monotonic() - t0 < time_limit:
        # Anneal beta
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        beta = beta_start + (beta_end - beta_start) * progress

        # Compute weights: w_k = exp(beta * c[k]) / sum(exp(beta * c[k]))
        c_shifted = c - c.max()
        log_w = beta * c_shifted
        log_w -= log_w.max()  # numerical stability
        w = np.exp(log_w)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum

        # Gradient of sum_k w_k * c[k] w.r.t. a[i]
        # c[k] = sum_j a[j]*a[k-j]
        # d c[k]/d a[i] = a[k-i] + a[k-i] (if k-i valid and != i) or a[i] (if k-i==i)
        g = np.zeros(n)
        for i in range(n):
            k_start = max(0, i)
            k_end = min(2 * n - 2, i + n - 1)
            wk = w[k_start:k_end + 1]
            ki = np.arange(k_start, k_end + 1) - i
            # d c[k]/d a[i] = 2*a[k-i] except when k-i==i (then just a[i])
            dck = 2.0 * a[ki]
            mask_self = (ki == i)
            dck[mask_self] = a[i]  # a[i]*a[i] derivative w.r.t a[i] is 2*a[i] but we only count once... 
            # Actually d(a[i]*a[i])/da[i] = 2*a[i], and d(a[i]*a[k-i]+a[k-i]*a[i])/da[i] = 2*a[k-i]
            # When k-i==i: term is a[i]^2, derivative is 2*a[i]. So dck = 2*a[i] in all cases!
            # Wait, let me reconsider. c[k] = sum_j a[j]*a[k-j].
            # d c[k]/d a[i] = a[k-i] (from j=i term) + a[k-i] (from k-j=i term) if k-i != i
            #                  = 2*a[i] if k-i == i (since only one term a[i]*a[i])
            # Actually when k-i == i, the j=i term gives a[i]*a[i] and the k-j=i term is the same (j=i, k-j=k-i=i).
            # So it's just one term: a[i]^2. Derivative: 2*a[i].
            # When k-i != i: two terms a[i]*a[k-i] and a[k-i]*a[i]. Derivative: 2*a[k-i].
            # So dck = 2*a[ki] in all cases! (a[i] when ki==i, a[ki] when ki!=i)
            g[i] = np.sum(wk * dck)

        # Scale gradient by 2*n/(s^2) for the full objective
        g_scale = 2.0 * n / (s * s)
        g *= g_scale

        # Momentum update
        v = momentum * v - lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) / n
            s_new = 1.0
        else:
            a_new = a_new / s_new  # renormalize (doesn't change objective)

        c_new = fast_convolve(a_new)
        obj_new = 2.0 * n * c_new.max()

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            c = c_new

        lr *= 0.999
        it += 1

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SGD it={it}, obj={best_obj:.6f}, beta={beta:.1f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase coordinate descent search to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Uses exact 1D subproblem solving per coordinate, multiple restarts, and diverse
    initializations across different sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.2f}, score={evaluate_sequence(p.tolist()):.6f}")
    except Exception:
        pass

    # === Phase 1: Coordinate descent scan over lengths and inits ===
    print("=== Phase 1: Coordinate Descent Scan ===")
    lengths = [256, 384, 512]
    if parent is not None:
        # Also try parent length
        if len(parent) not in lengths:
            lengths.append(len(parent))
            lengths.sort()

    priority_inits = ['parent', 'parent_interp', 'ramp', 'ramp_rev', 'quad',
                      'both_edge', 'one_edge', 'gauss', 'sine', 'coston',
                      'uniform', 'random', 'sparse', 'two_point']

    # Time allocation: give more time to shorter sequences (faster per iteration)
    time_per_length = max(120.0, (budget_s - 300.0) / len(lengths))

    for n in lengths:
        if time.monotonic() > deadline - 180:
            break
        inits = generate_inits(n, parent, rng)
        init_names = [x for x in priority_inits if x in inits]

        # Time per init
        n_inits_to_try = min(len(init_names), 8)
        t_per_init = max(15.0, time_per_length / max(1, n_inits_to_try))

        for name in init_names[:n_inits_to_try]:
            if time.monotonic() > deadline - 120:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 60.0)
            if t_alloc < 10:
                break

            a0 = inits[name]
            print(f"  n={n}, init={name}, t_alloc={t_alloc:.0f}s")
            t_start = time.monotonic()
            a_opt, obj_opt = coordinate_descent(a0, time_limit=t_alloc, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            elapsed = time.monotonic() - t_start
            if np.isfinite(score):
                print(f"    score={score:.6f}, elapsed={elapsed:.1f}s")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    print(f"After Phase 1: best={best_score:.6f}")

    # === Phase 2: Deep coordinate descent on best ===
    if best_sequence is not None:
        print("=== Phase 2: Deep CD on Best ===")
        remaining = deadline - time.monotonic() - 120.0
        t_deep = remaining * 0.4
        if t_deep > 20:
            a_opt, obj_opt = coordinate_descent(best_sequence, time_limit=t_deep, verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 2 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 3: Smooth gradient descent polish ===
    if best_sequence is not None:
        print("=== Phase 3: Smooth GD Polish ===")
        remaining = deadline - time.monotonic() - 90.0
        t_sgd = remaining * 0.3
        if t_sgd > 15:
            a_opt, obj_opt = smooth_gradient_descent(best_sequence, time_limit=t_sgd,
                                                     beta_start=100.0, beta_end=1000.0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 3 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 4: Pairwise descent ===
    if best_sequence is not None:
        print("=== Phase 4: Pairwise Descent ===")
        remaining = deadline - time.monotonic() - 60.0
        t_pw = remaining * 0.35
        if t_pw > 10:
            a_norm = best_sequence / best_sequence.sum()
            a_opt, obj_opt = pairwise_descent(a_norm, time_limit=t_pw,
                                              n_pairs_per_cycle=100, seed=seed)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 4 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 5: Final CD + restarts ===
    if best_sequence is not None:
        print("=== Phase 5: Final Polish ===")
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 30.0

        # Final CD
        if remaining > 20:
            t_final = remaining * 0.5
            a_opt, obj_opt = coordinate_descent(best_sequence, time_limit=t_final, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Final CD score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

        # Perturbation restarts
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 15:
            n_restarts = max(1, int(remaining // 15))
            n_restarts = min(n_restarts, 6)
            for i in range(n_restarts):
                if time.monotonic() > deadline - 15:
                    break
                scale = 0.002 * (1 + i * 0.5)
                noise = rng.standard_normal(n_best) * scale
                a_pert = np.maximum(best_sequence + noise, 0.0)
                a_pert = a_pert / a_pert.sum()
                t_alloc = min(12.0, deadline - time.monotonic() - 10.0)
                if t_alloc < 5:
                    break
                a_opt, obj_opt = coordinate_descent(a_pert, time_limit=t_alloc, verbose=False)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score):
                    if score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Restart {i}: score={score:.6f} (improved!)")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
