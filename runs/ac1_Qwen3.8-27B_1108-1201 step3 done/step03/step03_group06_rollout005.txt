import time
import cvxpy as cp
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build dense matrix A where (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    num_k = 2 * n - 1
    A = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A[k, idx] = a[k - idx]
    return A


def build_conv_matrix_sparse(a):
    """Build sparse matrix A where (A @ b)[k] = conv(a,b)[k]."""
    from scipy.sparse import lil_matrix
    n = len(a)
    num_k = 2 * n - 1
    A = lil_matrix((num_k, n))

        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        for i in idx:
            A[k, i] = a[k - i]
    return A.tocsr()


def solve_lp_all(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0.
    Returns normalized b (sum=1) or None."""
    n = len(a)
    num_k = 2 * n - 1
    A_mat = build_conv_matrix(a)

    c = -np.ones(n)
    A_ub = A_mat
    b_ub = np.full(num_k, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success and res.fun < -0.5:
            b_sol = np.maximum(res.x, 0)
            s = b_sol.sum()
            if s > 1e-10:
                return b_sol / s
    except Exception:
        pass
    return None


def solve_lp_cvxpy(a, M):
    """Solve LP using cvxpy PDLP solver."""
    n = len(a)
    num_k = 2 * n - 1
    A_mat = build_conv_matrix(a)

    b = cp.Variable(n)
    objective = cp.Maximize(cp.sum(b))
    constraints = [A_mat @ b <= M, b >= 0]

    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.PDLP, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None

    if b.value is None:
        return None
    b_sol = np.maximum(b.value, 0)
    s = b_sol.sum()
    if s > 0.5:
        return b_sol / s
    return None


def lp_step(a, use_cvxpy=False):

    s = b.sum()
    return b, s

                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj_try:
                    best_obj_try = obj_try
                    best_t = t

            if best_t > 0:
                a = (1.0 - best_t) * a + best_t * b
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                best_obj = best_obj_try
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Gradient step
        _, grad = smooth_logsumexp_objective(a, 800.0)
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm
        lr = 0.01 if no_improve < 20 else 0.003
        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        conv_now = fast_convolve(a)
        obj_now = 2.0 * n * conv_now.max()
        if obj_now < best_obj - 1e-10:
            best_obj = obj_now
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 60:
            noise = rng.standard_normal(n) * 0.02
            a = project_simplex(a + noise)
            v = np.zeros(n)
            no_improve = 0

        if verbose:
            elapsed = time.monotonic() - t0
            if int((time.monotonic() - t0) * 2) % 3 == 0:
                print(f"      Alt iter: obj={best_obj:.6f}, no_imp={no_improve}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n (all normalized to sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_inc'] = ramp.copy()
    inits['ramp_dec'] = ramp[::-1].copy()

    tri_edge = np.minimum(x, n - 1 - x) + 1.0
    tri_edge /= tri_edge.sum()
    inits['tri_edge'] = tri_edge.copy()

    tri_center = n - 1 - np.minimum(x, n - 1 - x)
    tri_center = np.maximum(tri_center, 1.0).astype(np.float64)
    tri_center /= tri_center.sum()
    inits['tri_center'] = tri_center.copy()

    for sf in [0.1, 0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g.copy()

    # Bimodal at edges
    bi = np.zeros(n)
    edge = max(1, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi.copy()

    # Cosine
    if n > 1:
        cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
        cos_w = np.maximum(cos_w, 0.01)
        cos_w /= cos_w.sum()
        inits['cosine'] = cos_w.copy()

    # Random
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.1)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand3'] = rng.dirichlet(np.ones(n) * 5.0)

    # Multi-modal random
    for mi in range(2):
        n_mode = rng.integers(2, 5)
        centers = rng.uniform(0, n, size=n_mode)
        widths = rng.uniform(0.05, 0.2, size=n_mode) * n
        mix = np.zeros(n)
        for c, w in zip(centers, widths):
            mix += np.exp(-0.5 * ((x - c) / w) ** 2)
        mix /= mix.sum()
        inits[f'modes_{mi}'] = mix.copy()

    # From parent
    if parent is not None and len(parent) > 0:
        pp = parent.astype(np.float64)
        pp = np.clip(pp, 0, 1000)
        if len(pp) == n:
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        elif len(pp) > n:
            pp_short = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(pp)), pp)
            pp_short = np.maximum(pp_short, 0)
            ps = pp_short.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp_short / ps
        else:
            pp_long = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(pp)), pp)
            pp_long = np.maximum(pp_long, 0)
            ps = pp_long.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp_long / ps

    return inits


def evaluate_normalized(a):
    """Evaluate objective for normalized (sum=1) sequence."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP with all constraints + basin hopping + L-BFGS.
    Explores multiple sequence lengths and uses alternating LP/gradient methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: length={len(p)}, sum={p.sum():.4f}")
    except (NameError, Exception):
        pass

    # ========== Phase 1: Multi-length quick scan ==========
    print("=== Phase 1: Multi-length Quick Scan ===")
    t_p1 = time.monotonic()
    p1_time = 250.0

    lengths = [192, 256, 320, 384, 512]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths:
            # Add parent length
            lengths_insert = sorted(lengths + [pn])
            if len(lengths_insert) <= 7:
                lengths = lengths_insert
            else:
                # Keep closest to parent length
                lengths = sorted(set(l for l in lengths if abs(l - pn) < 200) | {pn})[:6]

    scan_results = []
    time_per_n = p1_time / len(lengths)
    num_inits_per_n = 3

    for n in lengths:
        if time.monotonic() > deadline - 60:
            break
        if time.monotonic() > t_p1 + p1_time - 20:
            break

        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())

        # Priority order
        priority = ['parent', 'parent_ds', 'parent_up', 'bimodal', 'tri_edge',
                    'rand1', 'gauss_0.15', 'uniform', 'modes_0']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:num_inits_per_n + 1]

        time_per_init = max(10.0, (time_per_n - 5.0) / max(1, len(init_names)))

        for name in init_names:
            if time.monotonic() > t_p1 + p1_time - 10:
                break
            t_alloc = min(time_per_init, t_p1 + p1_time - time.monotonic() - 8)
            if t_alloc < 8:
                break

            a0 = inits[name].copy()
            use_cvx = (n <= 400)
            a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc,
                                         seed=hash((n, name)) % 10000,
                                         use_cvxpy=use_cv)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Scan: n={n}, {name}, norm_obj={obj_opt:.6f}, score={score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}")

    # ========== Phase 2: Deep LP + Basin Hopping ==========
    print("=== Phase 2: Deep LP + Basin Hopping ===")
    t_p2 = time.monotonic()
    p2_time = 650.0

    # Select top candidates for deep optimization
    deep_candidates = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = n
        if key not in seen and len(deep_candidates) < 4:
            deep_candidates.append((n, a_opt.copy(), score))
            seen.add(key)
    if best_sequence is not None:
        best_n = len(best_sequence)
        if best_n not in seen:
            deep_candidates.append((best_n, best_sequence.copy(), best_score))

    # Also add fresh candidates at promising lengths
    for n in [320, 512]:
        if n not in seen:
            inits = generate_initializations(n, parent, rng)
            if 'bimodal' in inits:
                deep_candidates.append((n, inits['bimodal'].copy(), float('inf')))
            seen.add(n)

    deep_candidates = deep_candidates[:4]
    time_per_candidate = p2_time / max(1, len(deep_candidates))

    for idx, (n_c, a_c, sc_c) in enumerate(deep_candidates):
        if time.monotonic() > t_p2 + p2_time - 30:
            break
        t_alloc = min(time_per_candidate - 5, t_p2 + p2_time - time.monotonic() - 30)
        if t_alloc < 30:
            break

        print(f"  Deep candidate {idx}: n={n_c}, initial_score={sc_c:.6f}")
        use_cvx = (n_c <= 400)

        # Main LP optimization
        a_opt, obj_opt = lp_optimize(a_c, time_limit=t_alloc * 0.6,
                                     seed=idx * 100 + 7, use_cvxpy=use_cv)

        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"    Deep LP: score={score:.6f}")

        # Alternate LP + Gradient
        if time.monotonic() < t_p2 + p2_time - 60:
            t_alloc2 = min(t_alloc * 0.4, t_p2 + p2_time - time.monotonic() - 60)
            if t_alloc2 > 20:
                a_opt2, obj_opt2 = alt_lp_grad_optimize(a_opt, time_limit=t_alloc2,
                                                        seed=idx * 200 + 13)
                score2 = float(evaluate_sequence(a_opt2.tolist()))
                if np.isfinite(score2) and score2 < best_score:
                    best_score = score2
                    best_sequence = a_opt2.copy()
                    print(f"    Alt LP+Grad: score={score2:.6f}")

    print(f"Phase 2 done. Best: {best_score:.6f}")

    # ========== Phase 3: L-BFGS / Gradient Polishing ==========
    print("=== Phase 3: Gradient Polishing ===")
    t_p3 = time.monotonic()
    p3_time = 400.0

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Multiple gradient configs
        grad_configs = [
            (300.0, 0.02, 0.0),
            (800.0, 0.01, 0.0),
            (1500.0, 0.005, 0.0),
            (500.0, 0.015, 0.0),
        ]

        time_per_grad = p3_time / (len(grad_configs) + 3)

        for gi, (beta_i, lr_i, _) in enumerate(grad_configs):
            if time.monotonic() > t_p3 + p3_time - 20:
                break
            t_alloc = min(time_per_grad, t_p3 + p3_time - time.monotonic() - 20)
            if t_alloc < 10:
                break

            a_opt, obj_opt = lbfgs_optimize(best_sequence, time_limit=t_alloc,
                                            beta_init=beta_i, lr_init=lr_i,
                                            seed=gi * 31 + 3)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Grad [{gi}] beta={beta_i}: score={score:.6f}")

        # Perturbed gradient starts
        for pi in range(3):
            if time.monotonic() > t_p3 + p3_time - 15:
                break
            for scale in [0.005, 0.01, 0.02]:
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(best_sequence + noise)
                t_alloc = min(time_per_grad / 3, t_p3 + p3_time - time.monotonic() - 15)
                if t_alloc < 5:
                    break
                a_opt, obj_opt = lbfgs_optimize(a_pert, time_limit=t_alloc,
                                                beta_init=1000.0, lr_init=0.008,
                                                seed=1000 + pi * 10 + int(scale * 1000))
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Grad pert [{pi}] s={scale}: score={score:.6f}")
                    break

    print(f"Phase 3 done. Best: {best_score:.6f}")

    # ========== Phase 4: Final Polish ==========
    print("=== Phase 4: Final Polish ===")
    t_p4 = time.monotonic()
    remaining = deadline - time.monotonic()

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Short LP polish
        t_alloc = min(remaining * 0.3, 80.0)
        if t_alloc > 10:
            a_opt, obj_opt = lp_optimize(best_sequence, time_limit=t_alloc,
                                         seed=9999, use_cvxpy=(n_best <= 400))
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish LP: score={score:.6f}")

        # Alternating polish
        remaining = deadline - time.monotonic()
        t_alloc = min(remaining * 0.3, 60.0)
        if t_alloc > 10:
            a_opt, obj_opt = alt_lp_grad_optimize(best_sequence, time_limit=t_alloc,
                                                  seed=7777)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish Alt: score={score:.6f}")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 20
        if remaining > 30:
            n_tries = int(min(remaining / 5, 15))
            for ti in range(n_tries):
                if time.monotonic() > deadline - 25:
                    break
                improved = False
                for scale in [0.001, 0.002, 0.005]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_pert = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_pert) and score_pert < best_score:
                        best_score = score_pert
                        best_sequence = a_pert.copy()
                        print(f"  Tiny pert [{ti}] s={scale}: score={best_score:.6f}")
                        improved = True
                        break
                if not improved and ti % 3 == 0:
                    # Try a slightly larger perturbation with gradient step
                    noise = rng.standard_normal(n_best) * 0.003
                    a_pert = project_simplex(best_sequence + noise)
                    t_small = min(5.0, remaining / max(1, n_tries - ti))
                    if t_small > 2:
                        a_opt, _ = lbfgs_optimize(a_pert, time_limit=t_small,
                                                  beta_init=2000.0, lr_init=0.005,
                                                  seed=5000 + ti)
                        score = float(evaluate_sequence(a_opt.tolist()))
                        if np.isfinite(score) and score < best_score:
                            best_score = score
                            best_sequence = a_opt.copy()
                            print(f"  Grad pert [{ti}]: score={best_score:.6f}")

    print(f"Phase 4 done. Best: {best_score:.6f}")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(128) / 128
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    # Ensure all values are in valid range
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    if best_sequence.sum() < 0.01:
        best_sequence = best_sequence / best_sequence.sum() * 1.0

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}, sum={best_sequence.sum():.4f}")
    return [float(x) for x in best_sequence]
