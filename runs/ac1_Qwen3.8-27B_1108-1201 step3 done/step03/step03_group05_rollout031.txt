Minimizes 2n*max(conv(a,a))/(sum(a))^2 using coordinate descent with power-smoothed
auto-convolution peaks, adaptive compensating coordinates, and multi-phase search.
Novel approach: exact 1D subproblem solving via bisection on the derivative of
power-smoothed objective, maintaining sum constraint through pairwise updates.
"""

import numpy as np
import time


def auto_conv(a):
    """Compute auto-convolution of a using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_b(a, i, j):
    """Compute b_k = a[k-i] - a[k-j] (zero-padded to length 2n-1)."""
    n = len(a)
    b = np.zeros(2 * n - 1)
    b[i:i + n] += a
    if j != i:
        b[j:j + n] -= a
    return b


def solve_1d(c, b, p, d_low, d_high):
    """
    Solve sum_k (c[k] + d*b_k)^(p-1) * b_k = 0 by bisection.
    Returns the optimal d in [d_low, d_high].
    """
    # Evaluate g at boundaries
    vals_low = np.maximum(c + d_low * b, 0.0)
    g_low = np.sum(np.power(vals_low, p - 1) * b)
    
    vals_high = np.maximum(c + d_high * b, 0.0)
    g_high = np.sum(np.power(vals_high, p - 1) * b)
    
    if g_low >= 0:
        return d_low
    if g_high <= 0:
        return d_high
    
    # Bisection
    lo = d_low
    hi = d_high
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = np.maximum(c + mid * b, 0.0)
        g_mid = np.sum(np.power(vals, p - 1) * b)
        if g_mid > 0:
            hi = mid
        else:
            lo = mid
        if hi - lo < 1e-14:
            break
    
    return (lo + hi) * 0.5


def coord_descent(a0, p, time_budget, seed=0, verbose=False):
    """
    Coordinate descent minimizing power-smoothed max auto-convolution.
    a0: initial sequence (will be normalized to sum=1)
    p: power for smoothing (higher = closer to true max)
    time_budget: seconds to run
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    c = auto_conv(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    sweep = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_budget:
                break
            
            # Choose compensating coordinate j (max a excluding i)
            a_masked = a.copy()
            a_masked[i] = -1
            j = int(np.argmax(a_masked))
            
            # Compute b
            b = compute_b(a, i, j)

            if np.all(b == 0):

            neg_mask = b < -1e-15

                bounds = -c[neg_mask] / b[neg_mask] - 1e-12
                d_high = min(d_high, np.min(bounds))

                continue
            
            # Solve 1D subproblem
            d_opt = solve_1d(c, b, p, d_low, d_high)
            
            if abs(d_opt) < 1e-15:
                continue
            
            # Update
            a[i] += d_opt
            a[j] -= d_opt
            c += d_opt * b

        curr_obj = c.max()

def gradient_polish(a0, time_budget, seed=0):
    """
    Smoothed gradient descent polishing step.
    Uses log-sum-exp smoothing with Nesterov momentum.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    v = np.zeros(n)
    beta = 100.0
    lr = 0.005
    momentum = 0.95
    
    c = auto_conv(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    iters = 0
    
    while time.monotonic() - t0 < time_budget:
        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        y = y / max(y.sum(), 1e-15)
        
        c_y = auto_conv(y)
        
        # Log-sum-exp gradient
        scaled = beta * c_y
        scaled -= scaled.max()
        p_vec = np.exp(scaled)
        p_sum = p_vec.sum()
        if p_sum < 1e-300:
            p_vec = np.zeros(len(c_y))
            p_vec[np.argmax(c_y)] = 1.0
        else:
            p_vec /= p_sum
        
        w = c_y * p_vec
        
        # Cross-correlation for gradient
        fft_size = 1 << (3 * n - 1).bit_length()
        Y = np.fft.rfft(y, fft_size)
        W = np.fft.rfft(w, fft_size)
        cross = np.fft.irfft(Y * np.conj(W), fft_size)
        grad = 2.0 * cross[:n]
        
        # Update
        v = momentum * v - lr * grad
        a = a + v
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        iters += 1
        
        c = auto_conv(a)
        curr_obj = c.max()
        if curr_obj < best_obj - 1e-12:
            best_obj = curr_obj
            best_a = a.copy()
        
        # Adaptive: increase beta over time
        frac = min(1.0, (time.monotonic() - t0) / time_budget)
        beta = 100.0 + 400.0 * frac
        lr = 0.005 * (1.0 - 0.8 * frac)
    
    return best_a, best_obj

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using coordinate descent with power-smoothed
auto-convolution peaks. Novel approach: exact 1D subproblem solving via bisection on
the derivative of power-smoothed objective, maintaining sum constraint through pairwise
updates. Combined with smoothed gradient polish for local refinement.
"""

import numpy as np
import time


def auto_conv(a):
    """Compute auto-convolution of a using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_b(a, i, j):
    """Compute b_k = a[k-i] - a[k-j] (zero-padded to length 2n-1)."""
    n = len(a)
    b = np.zeros(2 * n - 1)
    b[i:i + n] += a
    if j != i:
        b[j:j + n] -= a
    return b


def solve_1d(c, b, p, d_low, d_high):
    """
    Solve sum_k (c[k] + d*b_k)^(p-1) * b_k = 0 by bisection.
    Returns the optimal d in [d_low, d_high].
    """
    vals_low = np.maximum(c + d_low * b, 0.0)
    if p == 2:
        g_low = np.sum(vals_low * b)
    else:
        g_low = np.sum(np.power(vals_low, p - 1) * b)

    vals_high = np.maximum(c + d_high * b, 0.0)
    if p == 2:
        g_high = np.sum(vals_high * b)
    else:
        g_high = np.sum(np.power(vals_high, p - 1) * b)

    if g_low >= 0:
        return d_low
    if g_high <= 0:
        return d_high

    lo = d_low
    hi = d_high
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = np.maximum(c + mid * b, 0.0)
        if p == 2:
            g_mid = np.sum(vals * b)
        else:
            g_mid = np.sum(np.power(vals, p - 1) * b)
        if g_mid > 0:
            hi = mid
        else:
            lo = mid
        if hi - lo < 1e-14:
            break

    return (lo + hi) * 0.5


def coord_descent(a0, p, time_budget, seed=0, verbose=False):
    """
    Coordinate descent minimizing power-smoothed max auto-convolution.
    Uses pairwise updates (a[i] += d, a[j] -= d) to maintain sum constraint.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    c = auto_conv(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    sweep = 0
    no_improve = 0

    while time.monotonic() - t0 < time_budget:
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_budget:
                break

            # Choose compensating coordinate j (max a excluding i)
            a_masked = a.copy()
            a_masked[i] = -1.0
            j = int(np.argmax(a_masked))

            # Compute b
            b = compute_b(a, i, j)

            if np.all(b == 0):
                continue

            # Determine valid range for d
            d_low = -a[i] + 1e-15
            d_high = a[j] - 1e-15

            # Refine using positivity constraint on c + d*b
            pos_mask = b > 1e-15
            if pos_mask.any():
                bounds = -c[pos_mask] / b[pos_mask] + 1e-12
                d_low = max(d_low, float(np.max(bounds)))

            neg_mask = b < -1e-15
            if neg_mask.any():
                bounds = -c[neg_mask] / b[neg_mask] - 1e-12
                d_high = min(d_high, float(np.min(bounds)))

            if d_low >= d_high - 1e-14:
                continue

            # Solve 1D subproblem
            d_opt = solve_1d(c, b, p, d_low, d_high)

            if abs(d_opt) < 1e-15:
                continue

            # Update
            a[i] += d_opt
            a[j] -= d_opt
            c += d_opt * b
            # Clamp for numerical safety
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)

        sweep += 1
        curr_obj = c.max()
        if curr_obj < best_obj - 1e-13:
            best_obj = curr_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and sweep % 5 == 0:
                score = 2 * n * best_obj
                print(f"    CD sweep {sweep}: max_conv={best_obj:.8f}, score={score:.6f}", flush=True)
        else:
            no_improve += 1
            if no_improve > 80:
                break

    return best_a, best_obj


def coord_descent_multi_j(a0, p, time_budget, seed=0, n_j=3, verbose=False):
    """
    Coordinate descent that tries multiple compensating coordinates per update.
    For each i, tries n_j different j's and picks the best improvement.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    c = auto_conv(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    sweep = 0
    no_improve = 0

    while time.monotonic() - t0 < time_budget:
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_budget:
                break

            # Try multiple compensating coordinates
            candidates = rng.choice(n, size=min(n_j, n - 1), replace=False)
            candidates = candidates[candidates != i]
            if len(candidates) == 0:
                candidates = np.array([(i + 1) % n])

            best_d = 0.0
            best_j = -1
            best_obj_trial = c.max()

            for j in candidates:
                j = int(j)
                b = compute_b(a, i, j)
                if np.all(b == 0):
                    continue

                d_low = -a[i] + 1e-15
                d_high = a[j] - 1e-15

                pos_mask = b > 1e-15
                if pos_mask.any():
                    bounds = -c[pos_mask] / b[pos_mask] + 1e-12
                    d_low = max(d_low, float(np.max(bounds)))

                neg_mask = b < -1e-15
                if neg_mask.any():
                    bounds = -c[neg_mask] / b[neg_mask] - 1e-12
                    d_high = min(d_high, float(np.min(bounds)))

                if d_low >= d_high - 1e-14:
                    continue

                d_opt = solve_1d(c, b, p, d_low, d_high)

                # Evaluate objective at the trial update
                c_trial = c + d_opt * b
                obj_trial = c_trial.max()

                if obj_trial < best_obj_trial:
                    best_obj_trial = obj_trial
                    best_d = d_opt
                    best_j = j

            if best_j >= 0 and abs(best_d) > 1e-15:
                b = compute_b(a, i, best_j)
                a[i] += best_d
                a[best_j] -= best_d
                c += best_d * b
                a[i] = max(a[i], 0.0)
                a[best_j] = max(a[best_j], 0.0)

        sweep += 1
        curr_obj = c.max()
        if curr_obj < best_obj - 1e-13:
            best_obj = curr_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and sweep % 5 == 0:
                score = 2 * n * best_obj
                print(f"    CD-multiJ sweep {sweep}: max_conv={best_obj:.8f}, score={score:.6f}", flush=True)
        else:
            no_improve += 1
            if no_improve > 60:
                break

    return best_a, best_obj


def gradient_polish(a0, time_budget, seed=0, beta_init=100, beta_end=500):
    """
    Smoothed gradient descent with Nesterov momentum for final polishing.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)
    lr = 0.003
    momentum = 0.95

    c = auto_conv(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_budget:
        frac = min(1.0, (time.monotonic() - t0) / max(time_budget, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        y_sum = y.sum()
        if y_sum < 1e-15:
            y = a.copy()
        else:
            y = y / y_sum

        c_y = auto_conv(y)

        # Log-sum-exp gradient
        scaled = beta * c_y
        scaled -= scaled.max()
        p_vec = np.exp(scaled)
        p_sum = p_vec.sum()
        if p_sum < 1e-300:
            p_vec = np.zeros(len(c_y))
            p_vec[np.argmax(c_y)] = 1.0
        else:
            p_vec /= p_sum

        w = c_y * p_vec

        # Cross-correlation for gradient
        fft_size = 1 << (3 * n - 1).bit_length()
        Y = np.fft.rfft(y, fft_size)
        W = np.fft.rfft(w, fft_size)
        cross = np.fft.irfft(Y * np.conj(W), fft_size)
        grad = 2.0 * cross[:n]

        # Update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum

        c = auto_conv(a_new)
        curr_obj = c.max()
        iters += 1

        if curr_obj < best_obj - 1e-13:
            best_obj = curr_obj
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            a = a_new.copy()
            no_improve += 1

        if no_improve > 200:
            noise = rng.standard_normal(n) * 0.002
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = v * 0.3
            no_improve = 0

        lr = 0.003 * (1.0 - 0.7 * frac)

    return best_a, best_obj
