**Another idea: Use the KKT conditions more directly.**

At the optimum, for the problem `min max_k conv(f,f)[k] s.t. f >= 0, sum(f) = 1`:

- For each active k (where conv[f,f][k] = M): the gradient condition must hold.
- Specifically: `2 * sum_{k active} w_k * f[k-i] = λ` for all i where f[i] > 0, and `<= λ` for i where f[i] = 0.

This gives us a system of equations that the optimum must satisfy. We could try to solve this system directly.

At the optimum, `f` must satisfy:
- `f[i] > 0` implies `2 * sum_{k in S} w_k * f[k-i] = λ`
- `f[i] = 0` implies `2 * sum_{k in S} w_k * f[k-i] <= λ`

where S is the set of active peaks and w are dual weights.

This is similar to the LP dual! The LP dual was:
`min M * sum(y) s.t. sum_k y[k] * f[k-j] >= 1 for all j, y >= 0`

At the optimum of the original LP, the dual gives us the weights y, and complementary slackness tells us which constraints are tight.

So the KKT conditions for the overall problem (minimize max conv subject to simplex) are:
- `2 * (H_w @ f)[i] = λ` for i in support(f), where H_w is the Hankel matrix of dual weights w.
- `2 * (H_w @ f)[i] <= λ` for i not in support(f).
- `f >= 0`, `sum(f) = 1`.
- `w >= 0`, and `w[k] > 0` only if `conv[f,f][k] = M`.

This is a system of nonlinear equations (because both f and w appear). But we could alternate:
1. Given w, solve for f: this is a linear system (for the support) + complementarity.
2. Given f, update w: `w[k] = 1 if conv[f,f][k] = M` (relaxed version).

This is essentially the LP-FW method! The LP step is step 1 (given f, find the best g which gives us w), and the update is step 2.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_grad(f, tau):
    """Compute gradient of tau*log(sum(exp(conv[f,f][k]/tau)))/tau which is smooth max."""
    n = len(f)
    conv = fast_convolve(f)
    # Softmax weights
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    # Gradient: 2 * (H_w @ f) where H_w[i,j] = w[i+j]
    idx = np.arange(n)
    H = w[idx[:, None] + idx[None, :]]
    sg = 2.0 * (H @ f)
    obj = tau * (np.log(w.sum() * np.exp(conv.max() / tau)) - conv.max() / tau + np.log(np.exp(scaled).sum())) 
    # Actually, obj = tau * log(sum(exp(conv/tau)))
    # = tau * (conv.max()/tau + log(sum(exp((conv - conv.max())/tau))))
    obj = conv.max() + tau * np.log(np.exp(scaled).sum())
    return sg, obj, conv

def solve_lp(F, M, n):
    c = -np.ones(n)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(F[j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

We want to minimize `G(f) = max_k conv(f,f)[k]` subject to `f >= 0, sum(f) = 1`.

The smooth approximation is `S(f, tau) = tau * log(sum_k exp(conv[f,f][k] / tau))`.

As `tau -> 0`, `S -> max_k conv[f,f][k]`.

The gradient:
`dS/df[i] = tau * (1/sum_k exp(conv[k]/tau)) * d/d f[i] (sum_k exp(conv[k]/tau))`
`= tau * (1/sum_k exp(conv[k]/tau)) * sum_k exp(conv[k]/tau) * (1/tau) * d(conv[k])/df[i]`
`= (sum_k exp(conv[k]/tau) * d(conv[k])/df[i]) / sum_k exp(conv[k]/tau)`
`= sum_k softmax(conv[k]/tau)[k] * d(conv[k])/df[i]`
`= sum_k w[k] * 2 * f[k-i] * 1[0 <= k-i < n]`
`= 2 * sum_k w[k] * f[k-i] * 1[0 <= k-i < n]`

Where `w = softmax(conv/tau)`.

And `sum_k w[k] * f[k-i]` for fixed i is:
`= sum_{k: 0<=k-i<n} w[k] * f[k-i]`
`= sum_{j: 0<=j<n} w[j+i] * f[j]`
`= (H_w @ f)[i]` where `H_w[i,j] = w[i+j]`

So `dS/df[i] = 2 * (H_w @ f)[i]`. ✓

Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth log-sum-exp subgradient descent as the primary method (fast O(n^2) per step),

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(f, w):
    """Compute gradient of smooth max: 2 * (H_w @ f) where H_w[i,j] = w[i+j]."""
    n = len(f)
    idx = np.arange(n)
    H = w[idx[:, None] + idx[None, :]]
    return 2.0 * (H @ f)


def optimize_smooth(f0, time_limit, tau_init=0.05, verbose=False):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    lr = 0.1
    momentum = np.zeros(n)

    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

        it += 1

            best_f = f.copy()

        else:

        scaled -= scaled.max()

        w = w / w_sum

        if sg_norm < 1e-15:

        sg = sg / sg_norm

def solve_lp_for_fw(F, M, n):
    """Solve the LP oracle for Frank-Wolfe: max sum(g) s.t. conv(F,g)[k]<=M, g>=0."""
    c = -np.ones(n)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(F[j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def fw_lp_step(f, M, n):
    """One FW step: solve LP, do line search, return new f."""
    g = solve_lp_for_fw(f, M, n)
    if g is None:
        return f, M, False

    g_sum = g.sum()
    if g_sum < 1e-15:
        return f, M, False
    g = g / g_sum

    if np.max(np.abs(g - f)) < 1e-10:
        return f, M, False

    # Line search
    best_alpha = 0.0
    best_val = M
    for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha

    if best_alpha > 1e-8:
        ft = (1.0 - best_alpha) * f + best_alpha * g
        ft = np.maximum(ft, 0.0)
        ft = ft / ft.sum()
        return ft, best_val, True
    return f, M, False

        M = conv.max()
        best_delta = 0.0
        best_i = -1
        best_val = M

        if n > 500:
            indices = np.random.default_rng(it).choice(n, size=min(200, n), replace=False)
        else:
            indices = range(n)

        for i in indices:
            # Try increasing f[i] slightly (decrease others proportionally)
            for delta in [-0.002, -0.001, 0.001, 0.002]:
                f_try = f.copy()
                f_try[i] += delta
                if f_try[i] < 0:
                    continue

                s_try = f_try.sum()
                if s_try < 1e-10:
                    continue
                f_try = f_try / s_try
                val = fast_convolve(f_try).max()

                    best_val = val
                    best_delta = delta
                    best_i = i

            f = np.maximum(f, 0)
            f = f / f.sum()

def generate_diverse_inits(n, parent, rng, num=8):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(edge / edge.sum())

    # Stronger U-shape

    two = g1 + g2

    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    inits.append(rng.dirichlet(np.ones(n) * 0.1))

    # Parent interpolation

                finit = project_simplex(0.4 * best_seq + 0.6 * g)
            elif gen_func is not None:
                finit = gen_func()
            else:
                continue

            a_opt, _ = hybrid_optimize(finit, time_limit=t_alloc, use_lp=False, verbose=False)
            try_seq(a_opt, f"local_{name}")

        # 4c: Very fine smooth optimization (low temperature)
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            print("  4c: Fine smooth opt", flush=True)
            a_opt, _ = optimize_smooth(best_seq, time_limit=min(rem, 100), tau_init=0.01)
            try_seq(a_opt, "fine_smooth")

    # === PHASE 5: Final LP-FW polish ===
    print(f"=== Phase 5: Final LP-FW polish (best={best_score:.6f}) ===", flush=True)

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 40
        if rem > 30:
            f = best_seq / best_seq.sum()
            t0 = time.monotonic()
            stuck = 0
            while time.monotonic() - t0 < rem:
                conv = fast_convolve(f)
                M = conv.max()
                f_new, m_new, improved = fw_lp_step(f, M, n_best)
                if improved:
                    f = f_new
                    stuck = 0
                    if m_new < best_score / (2 * n_best):
                        # This corresponds to score < best_score
                        pass
                else:
                    stuck += 1
                    if stuck > 15:
                        break
            try_seq(f, "lpfw_final")

            # Another round with slightly relaxed M
            rem2 = deadline - time.monotonic() - 30
            if rem2 > 20:
                f = best_seq / best_seq.sum()
                t0 = time.monotonic()
                stuck = 0
                while time.monotonic() - t0 < rem2:
                    conv = fast_convolve(f)
                    M = conv.max()
                    f_new, m_new, improved = fw_lp_step(f, M * 1.02, n_best)
                    if improved and m_new < M:
                        f = f_new
                        stuck = 0
                    else:
                        stuck += 1
                        if stuck > 10:
                            break
                try_seq(f, "lpfw_relax")

    # === PHASE 6: Last-ditch perturbations ===
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            for i in range(min(5, int(rem / 4))):
                rem = deadline - time.monotonic() - 15
                if rem < 4:
                    break
                noise = rng.standard_normal(n_best) * (0.0005 + 0.0005 * i)
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = optimize_smooth(f_pert, time_limit=min(rem, 12), tau_init=0.005)
                try_seq(a_opt, f"last_{i}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
