def obj_value(a, n):
    """Compute 2n * max(conv(a,a)) for normalized a."""
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv

def nesterov_subgradient(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=30, temp=0.0002):
    """Nesterov-accelerated subgradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    y = a.copy()  # Nesterov momentum variable
    v = np.zeros(n)
    _, conv = obj_value(y, n)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov step
        sg = compute_subgradient_at_peaks(y, conv, top_k=top_k, temp=temp)
        
        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm > 1e-15:
            v = momentum * v - lr * (sg / sg_norm)
        
        y_new = y + v
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new > 1e-15:
            y = y_new / s_new
        else:
            y = best_a.copy()
        
        # Update x (the "slow" iterate)
        it += 1
        t_best = 2.0 / (it + 2.0)  # Polyak step size for Nesterov
        a = (1 - t_best) * a + t_best * y

        lr *= 0.9995
        temp *= 1.001

    return best_a, best_obj

def simulated_annealing(a0, time_limit, T_init=0.01, T_factor=0.999, n_iter_max=50000):
    """Simulated annealing for fine-tuning."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    _, conv = obj_value(a, n)
    current_obj = 2.0 * n * conv.max()
    best_obj = current_obj
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    T = T_init
    it = 0
    
    while time.monotonic() - t0 < time_limit and it < n_iter_max:
        # Random perturbation
        # Pick a random subset to perturb
        k = rng.integers(1, min(n, 50))
        idx = rng.choice(n, size=k, replace=False)
        scale = T * 0.1
        a_try = a.copy()
        a_try[idx] += rng.standard_normal(k) * scale
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            it += 1
            continue
        a_try = a_try / s_try
        
        _, conv_try = obj_value(a_try, n)
        new_obj = 2.0 * n * conv_try.max()
        
        delta = new_obj - current_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            a = a_try
            current_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
        
        T *= T_factor
        it += 1
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Sparse active-set LP for globally-informed direction finding
2. Peak-targeting subgradient descent with Nesterov acceleration
3. Coordinate-wise simulated annealing for final polish
4. Multi-length exploration with adaptive time allocation
Key improvements over baseline: sparse LP, correct convolution gradient,
multi-peak softmax targeting, coordinate-wise SA, and better time management.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a, n):
    """Compute 2n * max(conv(a,a)) for normalized a."""
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def build_sparse_conv_matrix(a):
    """Build sparse matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[j])
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def compute_peak_subgradient(a, conv, top_k=30, temp=0.0002):
    """
    Compute weighted subgradient focusing on peaks of convolution.
    The gradient of conv[k] wrt a[j] is:
      a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    which for auto-convolution simplifies to 2*a[|k-j|] in most cases.
    """
    n = len(a)
    # Select top peaks with softmax weighting
    if top_k >= len(conv):
        idx = np.arange(len(conv))
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]
    vals = conv[idx]
    scaled = vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(idx))
        w[np.argmax(conv[idx])] = 1.0
    else:
        w /= w_sum

    sg = np.zeros(n)
    for i in range(len(idx)):
        if w[i] < 1e-14:
            continue
        k = int(idx[i])
        # Gradient of conv[k] wrt a[j]:
        # a[j] appears in conv[k] = sum_m a[m]*a[k-m]
        # as first factor when m=j: gives a[k-j] (if 0<=k-j<n)
        # as second factor when k-m=j, i.e., m=k-j: gives a[m]=a[k-j] (if 0<=k-j<n)
        # So gradient = 2*a[k-j] when 0<=k-j<n and j != k-j
        #             = a[k-j] when j = k-j (i.e., 2j=k)
        # For subgradient purposes, use a[k-j] (factor absorbed in step size)
        
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        k_minus_js = k - js
        sg[js] += w[i] * a[k_minus_js]

    return sg


def lp_optimize(a0, time_limit, verbose=False, max_iters=300):
    """LP-based iterative optimization with sparse matrices and line search."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    _, conv = obj_value(a, n)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    perturb_count = 0

    while time.monotonic() - t0 < time_limit and iteration < max_iters:
        # Build sparse constraint matrix
        A_ub = build_sparse_conv_matrix(a)
        M_lp = M * 1.001

        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M_lp)
        bounds = [(0, None)] * n

        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        if result.success:
            g = result.x
            g_sum = g.sum()

            if g_sum > 1.0 + 1e-6:
                g_norm = g / g_sum
                d = g_norm - a
                norm_d = np.linalg.norm(d)

                if norm_d > 1e-10:
                    # Adaptive line search
                    best_t = 0.0
                    best_obj_ls = best_obj
                    candidates = [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.8, 0.6]
                    for t in candidates:
                        a_try = a + t * d
                        a_try = np.maximum(a_try, 0.0)
                        s_try = a_try.sum()
                        if s_try < 1e-15:
                            continue
                        a_try = a_try / s_try
                        obj_try, _ = obj_value(a_try, n)
                        if obj_try < best_obj_ls - 1e-14:
                            best_obj_ls = obj_try
                            best_t = t

                    # Refine around best_t
                    if 0.03 < best_t < 0.95:
                        for dt in [best_t * 0.5, best_t * 1.5]:
                            a_try = a + dt * d
                            a_try = np.maximum(a_try, 0.0)
                            s_try = a_try.sum()
                            if s_try < 1e-15:
                                continue
                            a_try = a_try / s_try
                            obj_try, _ = obj_value(a_try, n)
                            if obj_try < best_obj_ls - 1e-14:
                                best_obj_ls = obj_try
                                best_t = dt

                    if best_t > 1e-6:
                        a_new = a + best_t * d
                        a_new = np.maximum(a_new, 0.0)
                        s_new = a_new.sum()
                        if s_new > 1e-15:
                            a = a_new / s_new
                        _, conv = obj_value(a, n)
                        M = conv.max()
                        cur_obj = 2.0 * n * M
                        if cur_obj < best_obj:
                            best_obj = cur_obj
                            best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        iteration += 1

        # Handle stagnation
        if no_improve >= 40:
            rng = np.random.default_rng(iteration * 13 + 7)
            scale = 0.003 + 0.002 * min(perturb_count, 10)
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            _, conv = obj_value(a, n)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            if perturb_count > 50:
                break

    return best_a, best_obj


def nesterov_subgradient(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=40, temp=0.00015):
    """Nesterov-accelerated projected subgradient descent."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    x = a.copy()  # slow iterate
    y = a.copy()  # fast iterate (Nesterov)
    v = np.zeros(n)
    _, conv = obj_value(x, n)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        # Compute subgradient at y (the fast iterate)
        _, conv_y = obj_value(y, n)
        sg = compute_peak_subgradient(y, conv_y, top_k=top_k, temp=temp)
        sg_norm = np.linalg.norm(sg)
        if sg_norm > 1e-15:
            v = momentum * v - lr * (sg / sg_norm)

        # Update y
        y_new = y + v
        y_new = np.maximum(y_new, 0.0)
        s_new = y_new.sum()
        if s_new > 1e-15:
            y = y_new / s_new
        else:
            y = best_a.copy()
            v = np.zeros(n)

        # Update x with Polyak averaging
        it += 1
        t_polyak = 2.0 / (it + 2.0)
        x = (1.0 - t_polyak) * x + t_polyak * y

        # Evaluate
        _, conv_x = obj_value(x, n)
        cur_obj = 2.0 * n * conv_x.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameters
        lr *= 0.9997
        temp *= 1.0008
        momentum = min(0.96, momentum + 0.0001)

        if no_improve >= 300:
            # Restart with perturbation
            rng = np.random.default_rng(it)
            x = best_a + rng.standard_normal(n) * 0.003
            x = np.maximum(x, 0.0)
            s_x = x.sum()
            if s_x > 1e-15:
                x = x / s_x
            y = x.copy()
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_obj


def subgradient_direct(a0, time_limit, lr_init=0.02, top_k=50, temp=0.0001):
    """Direct projected subgradient (no momentum) with adaptive step size."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    _, conv = obj_value(a, n)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    f_best = best_obj
    f_init = best_obj

    while time.monotonic() - t0 < time_limit:
        sg = compute_peak_subgradient(a, conv, top_k=top_k, temp=temp)
        sg_norm = np.linalg.norm(sg)

        if sg_norm > 1e-15:
            a_try = a - lr * (sg / sg_norm)
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                _, conv_try = obj_value(a_try, n)
                new_obj = 2.0 * n * conv_try.max()

                if new_obj < best_obj:
                    a = a_try
                    conv = conv_try
                    best_obj = new_obj
                    best_a = a.copy()
                    no_improve = 0
                    lr = min(lr * 1.05, 0.1)
                else:
                    no_improve += 1
                    lr *= 0.95
            else:
                lr *= 0.5
                no_improve += 1
        else:
            break

        it += 1

        # Periodic temperature increase
        if it % 100 == 0:
            temp *= 1.5

        if no_improve >= 200:
            rng = np.random.default_rng(it * 3 + 1)
            a = best_a + rng.standard_normal(n) * 0.004
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a / s_a
            _, conv = obj_value(a, n)
            no_improve = 0

    return best_a, best_obj


def simulated_annealing(a0, time_limit, T_init=0.005, T_min=0.0001, n_coord=10):
    """Coordinate-wise simulated annealing for fine-tuning."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    _, conv = obj_value(a, n)
    current_obj = 2.0 * n * conv.max()
    best_obj = current_obj
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    T = T_init

    while time.monotonic() - t0 < time_limit and T > T_min:
        # Pick random coordinates to perturb
        k = min(n_coord, n)
        idx = rng.integers(0, n, size=k)
        scales = rng.uniform(0.001, 0.01, size=k)
        a_try = a.copy()
        a_try[idx] *= (1.0 + rng.standard_normal(k) * scales)
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            T *= 0.999
            continue
        a_try = a_try / s_try

        _, conv_try = obj_value(a_try, n)
        new_obj = 2.0 * n * conv_try.max()

        delta = new_obj - current_obj
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            a = a_try
            current_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()

        T *= 0.9995

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    bi = np.zeros(n)
    edge = max(1, n // 3)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # W-shaped: helps reduce center peak
    ws = 1.0 - 0.5 * np.cos(2 * np.pi * x / n)
    inits['wave'] = ws / ws.sum()

    # Cosine-tapered
    cos_t = 0.5 + 0.5 * np.cos(np.pi * x / (n - 1))
    cos_t = np.maximum(cos_t, 0.01)
    inits['cos_taper'] = cos_t / cos_t.sum()

    # Sine
    sn = np.sin(np.pi * (x + 0.5) / n)
    inits['sine'] = sn / sn.sum()

    # From parent
    if parent is not None:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_ds'] = pp / pps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_up'] = pp / pps

    return inits

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
