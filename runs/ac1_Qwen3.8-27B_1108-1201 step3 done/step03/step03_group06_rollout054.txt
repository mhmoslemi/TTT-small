import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def subgradient_max_conv(a, conv):
    """Compute subgradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    k_star = np.argmax(conv)
    g = np.zeros(n)
    # g[i] = a[k_star - i] (if valid) + a[i - k_star] (if valid)
    # Vectorized:
    # Term 1: a[k_star - i] for i where 0 <= k_star - i < n, i.e., k_star - n + 1 <= i <= k_star
    i_min1 = max(0, k_star - n + 1)
    i_max1 = min(n - 1, k_star)
    if i_min1 <= i_max1:
        idx = np.arange(i_min1, i_max1 + 1)
        g[idx] += a[k_star - idx]
    
    # Term 2: a[i - k_star] for i where 0 <= i - k_star < n, i.e., k_star <= i <= k_star + n - 1
    i_min2 = max(0, k_star)
    i_max2 = min(n - 1, k_star + n - 1)
    if i_min2 <= i_max2:
        idx = np.arange(i_min2, i_max2 + 1)
        g[idx] += a[idx - k_star]
    
    return g

Recall that the LP finds b such that (a*b)_k <= M for all k. This is equivalent to finding a point in the "correlation polytope" that maximizes the sum. 

What if instead of solving the full LP, we do coordinate-wise updates on b? For each coordinate b_j, we can compute the maximum value it can take given the current values of other b_j's and the constraint (a*b)_k <= M. This is a bisection on b_j.

This "coordinate-wise LP" would be much faster than solving the full LP, and might be more robust.

Let me code this up:

```python
def coordinate_lp_direction(a, M, n, iter_per_coord=5, seed=0):
    """Find b >= 0 with (a*b)_k <= M for all k, maximizing sum(b).
    Uses coordinate-wise bisection."""
    rng = np.random.default_rng(seed)
    b = np.zeros(n)
    conv_ab = np.zeros(2*n-1)  # (a*b)[k]
    
    # Initialize b with maximum possible values
    for j in range(n):
        # Max b_j such that (a*(b*e_j))_k <= M for all k
        # (a*(b*e_j))_k = b*a[k-j] (if valid)
        # So we need b*a[k-j] <= M for all k where k-j is in [0, n-1]
        # i.e., b <= M / max(a) (approximately)
        max_a = a.max()
        if max_a > 1e-15:
            b[j] = M / max_a
        else:
            b[j] = 1.0
    
    # Coordinate-wise refinement
    order = rng.permutation(n)
    for _ in range(iter_per_coord):
        for j in order:
            # Current contribution of b[j] to conv:
            # (a*(b*e_j))[k] = b[j] * a[k-j] for valid k
            # Other contributions: conv_ab - b[j]*a_shifted
            a_shifted = np.zeros(2*n-1)
            # a[k-j] is nonzero for k in [j, j+n-1]
            a_shifted[j:j+n] = a
            
            # Current total conv without this coordinate:
            conv_without = conv_ab - b[j] * a_shifted
            
            # Max b[j] such that conv_without + b[j]*a_shifted <= M
            # b[j] <= (M - conv_without) / a_shifted for all k where a_shifted > 0
            mask = a_shifted > 1e-15
            if mask.any():
                max_bj = np.min((M - conv_without[mask]) / a_shifted[mask])
                max_bj = max(0, max_bj)
            else:
                max_bj = b[j]  # no constraint from this direction
            
            b[j] = max_bj
            conv_ab = conv_without + b[j] * a_shifted
    
    return b
```

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:

def subgradient_max_conv(a, conv):
    """Compute subgradient of max(conv(a,a)) w.r.t. a at the argmax lag."""
    n = len(a)
    k_star = np.argmax(conv)
    g = np.zeros(n)
    # Term 1: a[k_star - i] for valid i
    i_min1 = max(0, k_star - n + 1)
    i_max1 = min(n - 1, k_star)
    if i_min1 <= i_max1:
        idx = np.arange(i_min1, i_max1 + 1)
        g[idx] += a[k_star - idx]
    # Term 2: a[i - k_star] for valid i
    i_min2 = max(0, k_star)
    i_max2 = min(n - 1, k_star + n - 1)
    if i_min2 <= i_max2:
        idx = np.arange(i_min2, i_max2 + 1)
        g[idx] += a[idx - k_star]
    return g

        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        grad, conv = smooth_gradient(a, beta)
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 15 == 0:
                lr *= 0.88
                v *= 0.4
            if no_improve > 100:
                noise = rng.standard_normal(n) * 0.005
                a = project_simplex(a + noise)
                v = np.zeros(n)
                lr = lr_init * 0.5
                beta = beta_init * 0.6
                no_improve = 0

        beta = min(8000.0, beta * 1.003)
        # Decay learning rate slowly
        lr = max(lr_init * 0.01, lr * 0.999)

        iter_count += 1
        if verbose and iter_count % 500 == 0:
            print(f"    GRAD iter {iter_count}: obj={best_obj:.6f}, beta={beta:.0f}, lr={lr:.5f}, t={time.monotonic()-t0:.1f}s")

    conv_final = fast_convolve(best_a)
    return best_a, 2.0 * n * conv_final.max()


def hybrid_alternating(a0, time_limit, seed=0, verbose=False):
    """Alternate between LP steps and gradient steps."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    step_lp = 0.7
    beta = 800.0
    lr = 0.015
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        # LP step (every other iteration)
        if iter_count % 2 == 0:
            top_k = max(30, n // 6)
            b = solve_lp_direction(a, M, top_k=top_k, conv=conv)
            if b is not None:
                b_sum = b.sum()
                if b_sum > 1e-15:
                    b = b / b_sum
                    a = (1.0 - step_lp) * a + step_lp * b
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-15:
                        a = a / s
        else:
            # Gradient step
            grad, conv = smooth_gradient(a, beta)
            g_norm = np.linalg.norm(grad)
            if g_norm > 1e-15:
                grad = grad / g_norm
            v = 0.92 * v - lr * grad
            a = project_simplex(a + v)

        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 60:
                noise = rng.standard_normal(n) * 0.004
                a = project_simplex(a + noise)
                v = np.zeros(n)
                no_improve = 0

        iter_count += 1
        if verbose and iter_count % 50 == 0:
            print(f"    HYB iter {iter_count}: obj={best_obj:.6f}, t={time.monotonic()-t0:.1f}s")

    conv_final = fast_convolve(best_a)
    return best_a, 2.0 * n * conv_final.max()


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n (normalized to sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Bimodal (edges)
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # More extreme bimodal
    edge2 = max(1, n // 5)
    bi2 = np.zeros(n)
    bi2[:edge2] = 1.0
    bi2[-edge2:] = 1.0
    bi2 /= bi2.sum()
    inits['bimodal2'] = bi2

    # Triangular
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri

    # Cosine
    if n > 1:
        cos_w = np.cos(np.pi * x / (n - 1)) ** 2
        cos_w = np.maximum(cos_w, 0.01)
        cos_w /= cos_w.sum()
    else:
        cos_w = np.ones(1)
    inits['cosine'] = cos_w

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random2'] = rng.dirichlet(np.ones(n) * 2.0)

    # Power law
    pl = (x + 1) ** 0.3
    pl /= pl.sum()
    inits['power_law'] = pl

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps
        if len(parent) > n:
            pp2 = np.interp(np.linspace(0, 1, n),
                            np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            ps2 = pp2.sum()
            if ps2 > 1e-10:
                inits['parent_ds'] = pp2 / ps2

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid optimization:
    Phase 1: Fast subgradient/FW exploration across lengths
    Phase 2: Intensive LP optimization
    Phase 3: Smooth gradient fine-tuning
    Phase 4: Hybrid alternating + final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: length={len(p)}, sum={p.sum():.2f}, obj={evaluate_sequence(p.tolist()):.6f}")
    except NameError:
        pass
    except Exception:
        pass

    # ========== Phase 1: Quick Exploration (180s) ==========
    print("=== Phase 1: Quick Exploration ===")
    t1_start = time.monotonic()
    t1_end = t1_start + 180.0

    lengths = [256, 288, 320, 352, 384, 448, 512]
    candidates = []

    for n in lengths:
        if time.monotonic() > t1_end - 20:
            break
        inits = generate_inits(n, parent, rng)
        # Prioritize
        priority = ['parent', 'parent_ds', 'bimodal', 'uniform', 'random', 'cosine', 'triangle']
        init_names = [nm for nm in priority if nm in inits][:4]

        for name in init_names:
            if time.monotonic() > t1_end - 10:
                break
            t_alloc = min(8.0, t1_end - time.monotonic() - 5)
            if t_alloc < 3:
                break
            a0 = inits[name]
            a_opt, obj_opt = frank_wolfe_quick(a0, time_limit=t_alloc, seed=n + hash(name) % 997)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                candidates.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Scan: n={n}, {name}, obj={obj_opt:.6f}, score={score:.6f}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.6f}, {len(candidates)} candidates")

    # ========== Phase 2: Intensive LP Optimization (650s) ==========
    print("=== Phase 2: Intensive LP ===")
    t2_start = time.monotonic()
    t2_end = t2_start + 650.0

    # Select top candidates
    deep_starts = []
    seen = set()
    for score, n, name, a_opt in candidates[:6]:
        key = (n, name)
        if key not in seen:
            deep_starts.append((n, a_opt.copy()))
            seen.add(key)

    # Always include current best
    if best_sequence is not None:
        bn = len(best_sequence)
        deep_starts.append((bn, best_sequence.copy()))
    else:
        bn = 320
        deep_starts.append((bn, np.ones(bn) / bn))

    # LP configurations
    lp_configs = [
        {'step_init': 0.9, 'top_k': 80, 'relax': 1.0, 'full_every': 25},
        {'step_init': 1.0, 'top_k': 120, 'relax': 1.0, 'full_every': 20},
        {'step_init': 0.7, 'top_k': None, 'relax': 1.0, 'full_every': 50},
        {'step_init': 1.0, 'top_k': 60, 'relax': 1.01, 'full_every': 30},
        {'step_init': 0.5, 'top_k': 100, 'relax': 0.99, 'full_every': 15},
    ]

    num_starts = min(len(deep_starts), 4)
    total_runs = num_starts * len(lp_configs)
    time_per_run = (t2_end - t2_start - 20) / max(1, total_runs)
    time_per_run = max(15.0, time_per_run)

    run_count = 0
    for i in range(num_starts):
        n_start, a_start = deep_starts[i]
        # Adjust top_k for this length
        for j, cfg in enumerate(lp_configs):
            if time.monotonic() > t2_end - 15:
                break
            t_alloc = min(time_per_run, t2_end - time.monotonic() - 10)
            if t_alloc < 10:
                break

            cfg_adj = dict(cfg)
            if cfg_adj['top_k'] is not None:
                cfg_adj['top_k'] = max(20, int(cfg_adj['top_k'] * n_start / 320))

            a_opt, obj_opt = lp_optimize(a_start, time_limit=t_alloc,
                                         seed=run_count * 13 + 7, **cfg_adj)
            run_count += 1
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP [{run_count}] n={n_start}, cfg{j}: score={score:.6f}")

    print(f"Phase 2 done. Best: {best_score:.6f}")

    # ========== Phase 3: Gradient Fine-tuning (350s) ==========
    print("=== Phase 3: Gradient Fine-tuning ===")
    t3_start = time.monotonic()
    t3_end = t3_start + 350.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        grad_configs = [
            {'beta_init': 400, 'lr_init': 0.025, 'momentum': 0.9},
            {'beta_init': 1000, 'lr_init': 0.015, 'momentum': 0.92},
            {'beta_init': 2000, 'lr_init': 0.01, 'momentum': 0.93},
            {'beta_init': 3000, 'lr_init': 0.008, 'momentum': 0.94},
            {'beta_init': 600, 'lr_init': 0.03, 'momentum': 0.88},
        ]

        time_per_grad = (t3_end - t3_start - 20) / (len(grad_configs) + 2)

        for i, cfg in enumerate(grad_configs):
            if time.monotonic() > t3_end - 10:
                break
            t_alloc = min(time_per_grad, t3_end - time.monotonic() - 8)
            if t_alloc < 10:
                break
            a_opt, obj_opt = grad_optimize(best_sequence, time_limit=t_alloc, **cfg, seed=i * 17 + 3)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  GRAD [{i}]: score={score:.6f}")

        # Perturbed gradient starts
        for i in range(2):
            if time.monotonic() > t3_end - 8:
                break
            noise = rng.standard_normal(n_best) * 0.003
            a_pert = project_simplex(best_sequence + noise)
            t_alloc = min(time_per_grad * 0.7, t3_end - time.monotonic() - 5)
            if t_alloc < 8:
                break
            a_opt, obj_opt = grad_optimize(a_pert, time_limit=t_alloc,
                                           beta_init=1500, lr_init=0.02,
                                           momentum=0.92, seed=50 + i)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  GRAD pert [{i}]: score={score:.6f}")

    print(f"Phase 3 done. Best: {best_score:.6f}")

    # ========== Phase 4: Hybrid Alternating + Final (remaining) ==========
    print("=== Phase 4: Hybrid + Final Polish ===")
    t4_start = time.monotonic()
    t4_end = deadline

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Hybrid alternating
        t_alloc = min(200.0, (t4_end - t4_start) * 0.4)
        if t_alloc > 30:
            a_opt, obj_opt = hybrid_alternating(best_sequence, time_limit=t_alloc, seed=42)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  HYBRID: score={score:.6f}")

        # Another LP round with best
        t_alloc = min(150.0, (t4_end - t4_start) * 0.3)
        if t_alloc > 20:
            a_opt, obj_opt = lp_optimize(best_sequence, time_limit=t_alloc,
                                         step_init=0.4, top_k=n_best // 3,
                                         relax=1.0, full_every=20, seed=200)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  FINAL LP: score={score:.6f}")

        # Final gradient polish
        t_alloc = min(100.0, (t4_end - t4_start) * 0.2)
        if t_alloc > 15:
            a_opt, obj_opt = grad_optimize(best_sequence, time_limit=t_alloc,
                                           beta_init=5000, lr_init=0.005,
                                           momentum=0.95, seed=999)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  FINAL GRAD: score={score:.6f}")

        # Tiny perturbations
        remaining = t4_end - time.monotonic() - 15.0
        if remaining > 15:
            n_tries = min(int(remaining / 4), 15)
            for i in range(n_tries):
                if time.monotonic() > t4_end - 12:
                    break
                improved = False
                for scale in [0.0005, 0.001, 0.002, 0.005]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_pert = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_pert) and score_pert < best_score:
                        best_score = score_pert
                        best_sequence = a_pert.copy()
                        improved = True
                        break
                # Try structured perturbation
                if not improved and i % 3 == 0:
                    conv = fast_convolve(best_sequence)
                    g = subgradient_max_conv(best_sequence, conv)
                    g_tangent = g - np.mean(g)
                    gn = np.linalg.norm(g_tangent)
                    if gn > 1e-15:
                        for scale in [0.001, 0.003]:
                            a_pert = project_simplex(best_sequence - scale * g_tangent / gn)
                            score_pert = float(evaluate_sequence(a_pert.tolist()))
                            if np.isfinite(score_pert) and score_pert < best_score:
                                best_score = score_pert
                                best_sequence = a_pert.copy()
                                break

    print(f"Phase 4 done. Best: {best_score:.6f}")

    # Final verification
    if best_sequence is None or not np.isfinite(best_score):
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})")
    return [float(x) for x in best_sequence]
