def subgrad_optimize(a0, time_limit, lr_init=0.08, momentum_init=0.2,

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                  'time_limit': 30})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def lp_direction_step(a, time_limit, blend=0.3):
    """One LP-based direction step with blending."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    if M < 1e-15:
        return a, 2.0 * n * M

    g = solve_lp_direction(a, M, n)
    if g is None:
        return a, 2.0 * n * M

    g_sum = g.sum()
    if g_sum < 1e-15:
        return a, 2.0 * n * M
    g /= g_sum

    # Blend
    a_new = (1.0 - blend) * a + blend * g
    a_new = project_simplex(a_new)
    conv_new = fast_convolve(a_new)
    obj_new = 2.0 * n * conv_new.max()

    return a_new, obj_new


def peak_coord_descent(a, max_rounds=50, top_k=10, delta_frac=0.03):
    """Peak-targeted coordinate descent: try moving mass between elements
    that contribute to the top peaks."""
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    for rnd in range(max_rounds):
        improved = False
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peaks = sorted(peaks, key=lambda k: -conv[k])

        for k in peaks:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            # Find the pair with maximum product a[j]*a[k-j]
            js = np.arange(j_start, j_end + 1)
            products = a[js] * a[k - js]
            if products.max() < 1e-12:
                continue
            j_max = js[np.argmax(products)]

            # Try moving small mass from j_max to various other positions
            delta = a[j_max] * delta_frac
            if delta < 1e-10:
                continue

            # Try moving to positions that don't contribute to this peak
            candidate_targets = []
            for j in range(n):
                if j == j_max:
                    continue
                # Check if j contributes to peak k
                if j_start <= j <= j_end:
                    continue
                # Also check if j contributes to other top peaks
                contributes = False
                for k2 in peaks:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    if js2 <= j <= je2:
                        contributes = True
                        break
                if not contributes:
                    candidate_targets.append(j)

            if not candidate_targets:
                candidate_targets = [j for j in range(n) if j != j_max and a[j] < a[j_max]]

            # Try the target with smallest current value (to minimize new peaks)
            candidate_targets.sort(key=lambda j: a[j])
            for j_target in candidate_targets[:5]:
                a_test = a.copy()
                a_test[j_max] -= delta
                a_test[j_target] += delta
                if a_test[j_max] < -1e-12:
                    continue
                a_test = np.maximum(a_test, 0.0)
                st = a_test.sum()
                if st < 1e-15:
                    continue
                a_test /= st
                conv_test = fast_convolve(a_test)
                obj_test = 2.0 * n * conv_test.max()
                if obj_test < best_obj - 1e-13:
                    a = a_test
                    best_obj = obj_test
                    best_a = a_test.copy()
                    conv = conv_test
                    improved = True
                    break

        if not improved:
            # Try smaller delta
            delta_frac *= 0.5
            if delta_frac < 0.001:
                break

    return best_a


def gen_inits(n, parent=None, rng=None):

    inits = {}

    inits['gauss'] = g / g.sum()

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Bimodal
    g2 = (np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.8 * n) / (0.1 * n)) ** 2))
    inits['bimodal'] = g2 / g2.sum()

    # Periodic (comb-like)
    period = max(2, n // 16)
    comb = np.zeros(n)
    for i in range(0, n, period):
        comb[i] = 1.0
    inits['comb'] = comb / comb.sum()

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()

    if parent is not None and len(parent) > 0:
        p = np.array(parent, dtype=np.float64)
        p = np.clip(p, 0, 1000)
        ps = p.sum()
        if ps > 1e-10 and len(p) > 0:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                p_resized = np.interp(np.linspace(0, 1, n),
                                      np.linspace(0, 1, len(p)), p)
                p_resized = np.maximum(p_resized, 0)
                prs = p_resized.sum()
                if prs > 1e-10:
                    inits['parent_resized'] = p_resized / prs

    return inits


def two_scale_optimize(a0, time_limit):
    """Two-scale: downsample, optimize, upsample, refine."""
    n = len(a0)
    # Downsample by factor 2
    n_coarse = n // 2
    if n_coarse < 16:
        return a0, 2.0 * n * fast_convolve(a0).max()

    a_coarse = np.zeros(n_coarse)
    for i in range(n_coarse):
        a_coarse[i] = a0[2 * i] + a0[2 * i + 1]
    a_coarse /= a_coarse.sum()

    # Optimize at coarse scale (shorter time)
    t_coarse = min(60.0, time_limit * 0.4)
    a_c, obj_c = subgrad_optimize(a_coarse, t_coarse, lr_init=0.06,
                                   momentum_init=0.2, momentum_end=0.8,
                                   top_k=20, sharpness=30.0)

    # Upsample
    a_fine = np.zeros(n)
    for i in range(n_coarse):
        a_fine[2 * i] = a_c[i]
        a_fine[2 * i + 1] = a_c[i]
    a_fine /= a_fine.sum()

    # Refine at fine scale
    t_fine = time_limit - t_coarse - 5
    a_opt, obj_opt = subgrad_optimize(a_fine, max(10.0, t_fine),
                                       lr_init=0.03, momentum_init=0.3,
                                       momentum_end=0.85,
                                       top_k=30, sharpness=50.0)
    return a_opt, obj_opt


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses momentum subgradient, LP directions, peak-targeted CD,
    two-scale optimization, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    parent_n = 0
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_n = len(p)
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={parent_n}, score={par_score:.6f}")
    except Exception:
        pass

    def update_best(a, source=""):
        nonlocal best_sequence, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_norm.copy()
            print(f"  NEW BEST [{source}]: score={score:.6f}, n={len(a)}")

    # Phase 1: Quick scan (first ~400s)
    print("=== Phase 1: Quick Scan ===")
    t_start = time.monotonic()
    phase1_time = min(400.0, (deadline - time.monotonic()) * 0.3)
    phase1_end = t_start + phase1_time

    lengths = [384, 448, 512, 576, 640]
    if parent_n in lengths:
        lengths.remove(parent_n)
        lengths.insert(0, parent_n)

    candidates = []

    for n in lengths:
        if time.monotonic() > phase1_end - 30:
            break
        inits = gen_inits(n, parent, rng)
        # Priority
        prio = ['parent', 'parent_resized', 'uniform', 'bimodal', 'random',
                'edge', 'triangle', 'comb', 'gauss', 'ramp']
        names = [nm for nm in prio if nm in inits][:4]

        t_per = max(10.0, (phase1_end - time.monotonic()) / (len(names) * max(1, len(lengths) - lengths.index(n))))

        for name in names:
            if time.monotonic() > phase1_end - 15:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 10)
            if t_alloc < 8:
                break

            a0 = inits[name]
            a_opt, obj = subgrad_optimize(a0, t_alloc, lr_init=0.06,
                                           momentum_init=0.2, momentum_end=0.7,
                                           top_k=25, sharpness=40.0)
            candidates.append((obj, n, name, a_opt.copy()))
            update_best(a_opt, f"P1 n={n} {name}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: Deep optimization on top candidates (~900s)
    print("=== Phase 2: Deep Optimization ===")
    phase2_start = time.monotonic()
    phase2_time = min(900.0, (deadline - phase2_start) * 0.55)
    phase2_end = phase2_start + phase2_time

    # Select top unique (n, quality) candidates
    top_list = []
    seen_n = set()
    for obj, n, name, a_opt in candidates:
        if n not in seen_n:
            top_list.append((n, a_opt))
            seen_n.add(n)
        if len(top_list) >= 4:
            break
    if best_sequence is not None:
        bn = len(best_sequence)
        if bn not in seen_n:
            top_list.append((bn, best_sequence.copy()))
            seen_n.add(bn)

    # Also add parent if not already there
    if parent is not None and parent_n not in seen_n:
        top_list.append((parent_n, parent / parent.sum()))

    for n, a0 in top_list:
        if time.monotonic() > phase2_end - 60:
            break
        t_rem = phase2_end - time.monotonic()
        if t_rem < 30:
            break

        a_work = a0.copy()

        # Step 2a: Extended subgradient with schedule
        t_sg = min(200.0, t_rem * 0.4)
        a_work, obj = subgrad_optimize(a_work, t_sg, lr_init=0.05,
                                        momentum_init=0.25, momentum_end=0.88,
                                        top_k=40, sharpness=30.0,
                                        sharpness_rate=1.002,
                                        max_sharpness=250.0)
        update_best(a_work, f"P2a n={n} sg")

        # Step 2b: LP direction steps
        t_lp = time.monotonic()
        t_rem = phase2_end - time.monotonic()
        if t_rem > 60:
            for blend in [0.4, 0.2, 0.1]:
                if time.monotonic() > phase2_end - 40:
                    break
                t_alloc = min(45.0, (phase2_end - time.monotonic()) / 3 - 10)
                if t_alloc < 20:
                    break
                a_new, obj_new = lp_direction_step(a_work, t_alloc, blend=blend)
                if obj_new < 2.0 * n * fast_convolve(a_work).max():
                    a_work = a_new
                    update_best(a_work, f"P2b n={n} LP b={blend}")
                else:
                    # Keep old if LP didn't help
                    pass
                # Subgrad polish after LP
                if time.monotonic() < phase2_end - 30:
                    a_work, obj = subgrad_optimize(a_work, min(20.0,
                                                               phase2_end - time.monotonic() - 25),
                                                   lr_init=0.02, momentum_init=0.3,
                                                   momentum_end=0.85,
                                                   top_k=30, sharpness=60.0)
                    update_best(a_work, f"P2b n={n} LP+sg b={blend}")

        # Step 2c: Peak-targeted coordinate descent
        if time.monotonic() < phase2_end - 20:
            a_cd = peak_coord_descent(a_work, max_rounds=40, top_k=8,
                                       delta_frac=0.04)
            update_best(a_cd, f"P2c n={n} CD")
            if 2.0 * n * fast_convolve(a_cd).max() < 2.0 * n * fast_convolve(a_work).max():
                a_work = a_cd

        # Step 2d: Two-scale refinement
        if time.monotonic() < phase2_end - 30 and n >= 128:
            t_2s = min(60.0, phase2_end - time.monotonic() - 25)
            if t_2s > 20:
                a_2s, obj_2s = two_scale_optimize(a_work, t_2s)
                update_best(a_2s, f"P2d n={n} 2scale")

    print(f"Phase 2 best: {best_score:.6f}")

    # Phase 3: Final polish (~remaining time)
    print("=== Phase 3: Final Polish ===")
    if best_sequence is not None:
        t_start3 = time.monotonic()
        t_rem3 = deadline - t_start3 - 10

        a_work = best_sequence.copy()
        n = len(a_work)

        # Long subgradient refinement
        t_sg3 = min(200.0, t_rem3 * 0.4)
        if t_sg3 > 15:
            a_work, obj = subgrad_optimize(a_work, t_sg3, lr_init=0.015,
                                            momentum_init=0.4, momentum_end=0.92,
                                            top_k=50, sharpness=80.0,
                                            sharpness_rate=1.001,
                                            max_sharpness=400.0)
            update_best(a_work, "P3 sg")

        # LP + subgrad cycles
        t_rem3 = deadline - time.monotonic() - 15
        for cycle in range(3):
            if time.monotonic() > deadline - 40:
                break
            t_cyc = min(60.0, t_rem3 / (3 - cycle) - 10)
            if t_cyc < 20:
                break

            # LP step
            t_lp_c = min(35.0, t_cyc * 0.5)
            if t_lp_c > 15:
                a_new, obj_new = lp_direction_step(a_work, t_lp_c, blend=0.15)
                obj_cur = 2.0 * n * fast_convolve(a_work).max()
                if obj_new < obj_cur:
                    a_work = a_new
                update_best(a_work, f"P3 cycle{cycle} LP")

            # Subgrad
            t_sg_c = min(30.0, t_cyc * 0.4)
            if t_sg_c > 10:
                a_work, obj = subgrad_optimize(a_work, t_sg_c, lr_init=0.008,
                                                momentum_init=0.5, momentum_end=0.93,
                                                top_k=60, sharpness=100.0)
                update_best(a_work, f"P3 cycle{cycle} sg")

            # CD
            if time.monotonic() < deadline - 25:
                a_cd = peak_coord_descent(a_work, max_rounds=30, top_k=5,
                                           delta_frac=0.02)
                obj_cd = 2.0 * n * fast_convolve(a_cd).max()
                obj_w = 2.0 * n * fast_convolve(a_work).max()
                if obj_cd < obj_w:
                    a_work = a_cd
                update_best(a_cd, f"P3 cycle{cycle} CD")

        # Random perturbations around best
        if time.monotonic() < deadline - 20:
            for ps in [0.005, 0.01, 0.02]:
                if time.monotonic() > deadline - 15:
                    break
                for trial in range(3):
                    if time.monotonic() > deadline - 12:
                        break
                    noise = rng.standard_normal(n) * ps
                    a_p = project_simplex(a_work + noise)
                    t_p = min(15.0, deadline - time.monotonic() - 10)
                    if t_p < 5:
                        break
                    a_opt, obj = subgrad_optimize(a_p, t_p, lr_init=0.01,
                                                   momentum_init=0.3, momentum_end=0.9,
                                                   top_k=40, sharpness=80.0)
                    update_best(a_opt, f"P3 perturb {ps} t{trial}")

    # Final validation
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
