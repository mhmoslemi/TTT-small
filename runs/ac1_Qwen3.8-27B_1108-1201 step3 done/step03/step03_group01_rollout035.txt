import time
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def simplex_projection(x, s=1.0):
    """Project x onto {a >= 0, sum(a) = s} in O(n log n)."""
    x = np.asarray(x, dtype=np.float64)
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho = np.where(cond)[0][-1]
    tau = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(x - tau, 0.0)


def solve_eq_qp(H, n):
    """
    Solve: min (1/2) a^T H a s.t. sum(a) = 1, a >= 0.
    Uses active-set method. H is symmetric PSD.
    """
    active = np.ones(n, dtype=bool)
    a = np.zeros(n)

    for _ in range(n + 1):
        idx = np.where(active)[0]
        m = len(idx)
        if m == 0:
            return np.ones(n) / n

        H_aa = H[np.ix_(idx, idx)]

        if m == 1:
            a_new = np.zeros(n)
            a_new[idx[0]] = 1.0
            return a_new

        # Augmented KKT system:
        # [H_aa  1] [a_a]   [0]
        # [1^T   0] [nu]  = [1]
        K = np.zeros((m + 1, m + 1))
        K[:m, :m] = H_aa
        K[:m, m] = 1.0
        K[m, :m] = 1.0
        rhs = np.zeros(m + 1)
        rhs[m] = 1.0

        try:
            sol = np.linalg.solve(K, rhs)
        except np.linalg.LinAlgError:
            sol = np.linalg.lstsq(K, rhs, rcond=None)[0]

        a_a = sol[:m]

        if (a_a >= -1e-10).all():
            a_new = np.zeros(n)
            a_new[idx] = np.maximum(a_a, 0.0)
            a_new_sum = a_new.sum()
            if a_new_sum > 1e-15:
                a_new /= a_new_sum
            else:
                a_new = np.ones(n) / n
            return a_new
        else:
            min_local = np.argmin(a_a)
            active[idx[min_local]] = False

    # Fallback
    return np.ones(n) / n


def irl_optimize(a0, time_limit, verbose=False, p_schedule=None, max_iters=500):
    """
    IRL-QP optimization: iteratively reweight and solve QP to minimize max(conv(a,a)).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 30
    eps_base = 1e-8

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        Cmax = c.max()
        cur_obj = 2.0 * n * Cmax

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            # Perturb and reset
            rng = np.random.default_rng(hash((iteration, n)) % (2**31))
            a = best_a + rng.standard_normal(n) * 0.01
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0
            if no_improve > 5:
                break

        # Compute weights
        eps = eps_base * (1.0 + iteration * 0.1)
        c_floor = np.maximum(c, Cmax * 0.005)
        w = 1.0 / (c_floor + eps) ** 2

        # Build H = B^T W B
        B = build_conv_matrix(a)
        WB = w[:, np.newaxis] * B
        H = B.T @ WB

        # Add small regularization for numerical stability
        H += 1e-12 * np.eye(n)

        # Solve QP
        a_new = solve_eq_qp(H, n)

        # Verify improvement
        c_new = fast_convolve(a_new)
        obj_new = 2.0 * n * c_new.max()

        if obj_new < cur_obj - 1e-14:
            a = a_new
        else:
            # Try relaxed version: blend with current
            for t in [0.5, 0.3, 0.1]:
                a_blend = (1 - t) * a + t * a_new
                a_blend = np.maximum(a_blend, 0.0)
                a_blend = a_blend / a_blend.sum()
                c_blend = fast_convolve(a_blend)
                obj_blend = 2.0 * n * c_blend.max()
                if obj_blend < cur_obj - 1e-14:
                    a = a_blend
                    break

    return best_a, best_obj


def lp_direction_step(a, M, time_budget=5.0):
    """
    LP-based direction: maximize sum(g) s.t. a*g <= M, g >= 0.
    Returns (g, g_sum) or None.
    """
    n = len(a)
    A_ub = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M * 1.001)
    bounds = [(0, None)] * n

    t0 = time.monotonic()
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'time_limit': time_budget})

    if result.success and result.x.sum() > 1.0 + 1e-7:
        g = result.x
        return g / g.sum()
    return None


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based iterative optimization (from previous approach, refined)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    M = c.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 2:
            break

        g = lp_direction_step(a, M, time_budget=min(3.0, time_limit - (time.monotonic() - t0)))

        if g is None or np.linalg.norm(g - a) < 1e-10:
            no_improve += 1
        else:
            d = g - a
            # Line search
            best_t = 0.0
            best_trial_obj = best_obj
            for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
                a_try = np.maximum(a + t * d, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = fast_convolve(a_try)
                obj_try = 2.0 * n * c_try.max()
                if obj_try < best_trial_obj - 1e-14:
                    best_trial_obj = obj_try
                    best_t = t

            if best_t > 0 and best_trial_obj < best_obj - 1e-13:
                a = np.maximum(a + best_t * d, 0.0)
                a = a / a.sum()
                c = fast_convolve(a)
                M = c.max()
                best_obj = 2.0 * n * M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        if no_improve >= 25 + 5 * perturb_count:
            rng = np.random.default_rng(perturb_count * 997 + 13)
            scale = 0.005 + 0.005 * min(perturb_count, 10)
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            c = fast_convolve(a)
            M = c.max()
            no_improve = 0
            perturb_count += 1
            if perturb_count > 20:
                break

    return best_a, best_obj


def subgradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=60, temp_init=0.05):
    """
    FFT-accelerated subgradient with softmax peak weighting and simplex projection.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    c = fast_convolve(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mu = momentum
    no_improve = 0
    batch_size = 50
    step_count = 0

    while time.monotonic() - t0 < time_limit:
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit:
                break

            c = fast_convolve(a)
            Cmax = c.max()

            # Softmax weights over peaks
            scaled = c / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-15:
                w = np.zeros(2 * n - 1)
                w[np.argmax(c)] = 1.0
            else:
                w /= w_sum

            # Subgradient: B^T w (cross-correlation of a and w)
            # (B^T w)_j = sum_k B_{k,j} w_k = sum_k a[k-j] w_k (valid indices)
            # This is the cross-correlation of a and w, computed via FFT
            # (a * w^R)[j + n - 1] where w^R is reversed... 
            # Actually: (B^T w)_j = sum_{k=j}^{j+n-1} a[k-j] w_k
            # Let's compute via FFT: this is cross-correlation
            fft_size = 1 << (2 * n - 1 - 1).bit_length()
            A_f = np.fft.rfft(a, fft_size)
            W_f = np.fft.rfft(w, fft_size)
            # B^T w = ifft(conj(A_f) * W_f) ... no that's circular
            # Let me just do it directly for clarity
            # (B^T w)_j = sum_{m=0}^{n-1} a[m] * w[j+m]
            # This is like a "flipped" convolution
            # = (flip(a) * w)[j + n - 1] ... 
            # Actually: sum_m a[m] w[j+m] = (a_flipped * w)[j + n - 1] where a_flipped = a[::-1]
            # Let me verify: (a_f * w)[k] = sum_m a_f[m] w[k-m] = sum_m a[n-1-m] w[k-m]
            # Hmm, let me just use the direct computation via the convolution matrix structure
            # Or use: B^T w = ifft(fft(a) * conj(fft(w))) ... no.
            
            # Simplest: use the fact that (B^T w)_j = sum_m a[m]*w[j+m]
            # This is the "valid" part of the convolution of a with w, evaluated at specific lags
            # Let's use FFT: pad w to length 2n-1, compute ifft(fft(a)[0:n] * fft(w))
            # Actually the cleanest way:
            # full_conv = np.convolve(a, w) has length 2n-1+(2n-1)-1... no.
            # a has length n, w has length 2n-1. Their full convolution has length n + 2n-1 - 1 = 3n-2.
            # (B^T w)_j = sum_{m=0}^{n-1} a[m] * w[j+m] which is the "cross-correlation"
            # = (a * w)[j + (n-1)] if we define (a*w)[k] = sum_m a[m]*w[k-m+n-1]... 
            # I'm overcomplicating this. Let me just use:
            # (B^T w)_j = sum_{k=j}^{j+n-1} a[k-j] * w_k
            # Define u = a[::1] (length n), v = w (length 2n-1)
            # (B^T w)_j = sum_{m=0}^{n-1} u[m] * v[j+m]
            # This is the "elapsed" convolution or cross-correlation
            # = ifft(fft(u, N) * conj(fft(v, N))) ... for circular
            # For linear: pad both to length >= n + 2n-1 - 1 = 3n-2
            
            # Actually, the simplest correct computation:
            # (B^T w)_j = sum_m a[m] * w[j+m] for j = 0,...,n-1
            # Let's define the full linear correlation: r[lag] = sum_m a[m] * w[m + lag]
            # This is exactly what np.correlate(w, a, 'full') gives... let me check
            # np.correlate(w, a, 'full')[k] = sum_m w[m+k] * conj(a[m])... 
            # Hmm. Let me just do it with FFT directly.
            
            # (B^T w)_j = sum_{m=0}^{n-1} a[m] w[j+m]
            # = sum_m a[m] w[j+m]
            # If I compute the full linear convolution of a and w:
            # (a*w_full)[k] = sum_m a[m] w[k-m] for k = 0,...,n+(2n-1)-1-1 = 3n-3
            # Then (B^T w)_j = sum_m a[m] w[j+m] 
            # Let k = j + m, m = k - j... that's not quite right.
            # sum_m a[m] w[j+m] = sum_k a[k-j] w[k] (substituting k=j+m)
            # = (a_rev * w)[j + n - 1] where a_rev = a[::-1]
            # Let me verify: (a_rev * w)[p] = sum_q a_rev[q] w[p-q] = sum_q a[n-1-q] w[p-q]
            # Let r = n-1-q: = sum_r a[r] w[p-n+1+r]
            # For p = j + n - 1: = sum_r a[r] w[j+r] = (B^T w)_j ✓
            
            N_corr = 1 << (3 * n - 2 - 1).bit_length()
            a_rev = a[::-1]
            Ar_f = np.fft.rfft(a_rev, N_corr)
            W_f = np.fft.rfft(w, N_corr)
            corr = np.fft.irfft(Ar_f * W_f, N_corr)
            subgrad = corr[n - 1:n - 1 + n]  # positions j=0,...,n-1

            # Update with momentum
            v = mu * v - lr * subgrad
            a_new = a + v
            a_new = simplex_projection(a_new, 1.0)

            c_new = fast_convolve(a_new)
            obj_new = 2.0 * n * c_new.max()

            a = a_new
            step_count += 1

            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Adaptive parameters
        lr *= 0.998
        temp *= 1.003
        mu = min(0.97, mu + 0.0002)

        if no_improve >= 500:
            # Perturb
            rng = np.random.default_rng(step_count)
            a = best_a + rng.standard_normal(n) * 0.005
            a = simplex_projection(a, 1.0)
            no_improve = 0
            lr = lr_init * 0.5

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['raised_cosine'] = rc / rc.sum()

    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    # Bimodal (edge concentration)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Sine window
    si = np.sin(np.pi * (x + 0.5) / n) ** 2
    inits['sine'] = si / si.sum()

    # Hann window
    h = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1))) if n > 1 else np.ones(n)
    inits['hann'] = h / h.sum()

    # From parent
    if parent is not None and len(np.atleast_1d(parent)) > 0:
        p = np.asarray(parent, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0.0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_ds'] = pp / pps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0.0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_up'] = pp / pps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining IRL-QP, LP, and subgradient
    methods to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent available: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    # Phase 1: IRL-QP scan across lengths and initializations
    print("=== Phase 1: IRL-QP Scan ===")
    t_start = time.monotonic()
    phase1_limit = 600.0
    phase1_deadline = t_start + phase1_limit

    lengths = [512, 1024]
    scan_results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 60:
            break

        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())
        priority = ['parent', 'parent_ds', 'parent_up', 'uniform', 'raised_cosine', 'hann', 'sine']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:5]

        t_per_init = max(20.0, (phase1_limit - 30.0) / (len(lengths) * 5))

        for name in init_names:
            if time.monotonic() > phase1_deadline - 30:
                break
            t_alloc = min(t_per_init, phase1_deadline - time.monotonic() - 20)
            if t_alloc < 10:
                break

            a0 = inits[name]
            a_opt, obj_opt = irl_optimize(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))

            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  IRL n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # Phase 2: LP-based deep optimization on top candidates
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, a_opt))
            seen.add(key)
        if len(top_configs) >= 5:
            break

    if best_sequence is not None:
        n_b = len(best_sequence)
        if (n_b, 'best') not in seen:
            top_configs.append((n_b, best_sequence.copy()))

    remaining = deadline - time.monotonic() - 200.0
    n_candidates = max(1, len(top_configs))
    t_per_deep = max(30.0, remaining * 0.5 / n_candidates)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 150:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 100)
        if t_alloc < 15:
            break

        a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  LP deep {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # Phase 3: IRL refinement on best
    print("=== Phase 3: IRL Refinement ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 150.0
        if remaining > 30:
            t_alloc = min(remaining * 0.4, 200.0)
            a_opt, obj_opt = irl_optimize(best_sequence, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  IRL refine: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # Phase 4: Subgradient polishing
    print("=== Phase 4: Subgradient Polish ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 60.0
        if remaining > 20:
            sg_configs = [
                {'lr_init': 0.008, 'momentum': 0.92, 'top_k': 60, 'temp_init': 0.04},
                {'lr_init': 0.012, 'momentum': 0.94, 'top_k': 100, 'temp_init': 0.06},
                {'lr_init': 0.005, 'momentum': 0.88, 'top_k': 40, 'temp_init': 0.03},
            ]
            t_per_sg = remaining / (len(sg_configs) + 1)

            for ci, cfg in enumerate(sg_configs):
                if time.monotonic() > deadline - 30:
                    break
                t_alloc = min(t_per_sg, deadline - time.monotonic() - 20)
                if t_alloc < 10:
                    break

                a_opt, obj_opt = subgradient_optimize(best_sequence, time_limit=t_alloc, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  SG {ci}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Phase 5: Final LP + IRL polish (alternating)
    print("=== Phase 5: Final Polish ===")
    t_start5 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 20:
            # Alternate LP and IRL
            for round_i in range(3):
                if time.monotonic() > deadline - 10:
                    break
                t_half = (deadline - time.monotonic() - 10) / 2 / (3 - round_i)
                if t_half < 5:
                    break

                # LP step
                t_alloc = min(t_half, deadline - time.monotonic() - 10)
                if t_alloc > 5:
                    a_opt, _ = lp_optimize(best_sequence, time_limit=t_alloc)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Final LP {round_i}: {score:.6f}")

                # IRL step
                t_alloc = min(t_half, deadline - time.monotonic() - 5)
                if t_alloc > 5:
                    a_opt, _ = irl_optimize(best_sequence, time_limit=t_alloc)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Final IRL {round_i}: {score:.6f}")

    print(f"Phase 5 done ({time.monotonic()-t_start5:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
