Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive subgradient descent

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g or None if infeasible/failed."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def compute_subgradient(f, conv, top_k=50, temp=0.001):
    """Compute subgradient of max(conv) with softmax weighting over top peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    
    return sg

import numpy as np
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def compute_subgradient(f, conv, top_k=64, temp=0.001):
    """Compute subgradient of max(conv) with softmax weighting over top peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]

    return sg

    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Blend with ramp
            blended = 0.5 * (px / ps) + 0.5 * (ramp_d / ramp_d.sum())
            inits.append(blended / blended.sum())
            blended2 = 0.5 * (px / ps) + 0.5 * (ramp_i / ramp_i.sum())
            inits.append(blended2 / blended2.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase optimization combining LP Frank-Wolfe,
    subgradient descent, and fine polishing with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # === Phase 1: Parent optimization (highest priority) ===
    print("=== Phase 1: Parent Optimization ===")
    if parent is not None:
        n_p = len(parent)
        t1 = min(300, deadline - time.monotonic() - 400)
        if t1 > 30:
            print(f"  FW-LP on parent (n={n_p}), t={t1:.0f}s")
            a_opt, obj = fw_lp_optimize(parent, time_limit=t1)
            try_seq(a_opt, "parent_fw")
            
            # SG follow-up
            rem = deadline - time.monotonic() - 300
            if rem > 30:
                a_opt2, _ = sg_descent(a_opt, time_limit=min(rem, 100),
                                       initial_lr=0.004, top_k=n_p//8, temp=0.0003)
                try_seq(a_opt2, "parent_sg")

    # === Phase 2: Broad exploration ===
    print("\n=== Phase 2: Broad Exploration ===")

    lengths = [256, 384, 512]
    if parent is not None:
        if len(parent) in lengths:
            lengths.remove(len(parent))
        lengths.insert(0, len(parent))

    for n in lengths:
        if time.monotonic() > deadline - 200:
            break
        if parent is not None and n == len(parent):
            continue  # already done in phase 1

        inits = generate_inits(n, parent, rng)
        rem = deadline - time.monotonic() - 150
        n_attempts = len(inits)
        t_per = max(20, rem / max(1, n_attempts * len(lengths)))
        t_per = min(t_per, 60)

        for i, f_init in enumerate(inits):
            if time.monotonic() > deadline - 120:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 80)
            if t_alloc < 15:
                break
            a_opt, obj = sg_descent(f_init, time_limit=t_alloc,
                                    initial_lr=0.01, top_k=max(32, n//12), temp=0.001,
                                    lp_escape=True, lp_every=200)
            try_seq(a_opt, f"n{n}_i{i}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Deep optimization on best ===
    print(f"\n=== Phase 3: Deep Optimization ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # FW-LP deep runs
        for p in range(4):
            rem = deadline - time.monotonic() - 100
            if rem < 25:
                break
            t_alloc = min(rem // (4 - p), 120)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.004)
            elif p == 2:
                f_init = project_simplex(0.85 * best_seq + 0.15 * (best_seq[::-1]))
            else:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.015)

            a_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"deep_fw_{p}")

        # SG deep runs
        for p in range(3):
            rem = deadline - time.monotonic() - 80
            if rem < 20:
                break
            t_alloc = min(rem // (3 - p), 80)

            if p == 0:
                f_init = best_seq.copy()
                cfg = {"initial_lr": 0.003, "top_k": n_best//6, "temp": 0.0002}
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
                cfg = {"initial_lr": 0.005, "top_k": n_best//4, "temp": 0.0005}
            else:
                f_init = project_simplex(0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best))
                cfg = {"initial_lr": 0.002, "top_k": n_best//8, "temp": 0.0001}

            a_opt, obj = sg_descent(f_init, time_limit=t_alloc, lp_escape=True, **cfg)
            try_seq(a_opt, f"deep_sg_{p}")

    # === Phase 4: Final polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Fine SG polish
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = sg_polish(best_seq, time_limit=min(rem, 60),
                                 lr=0.001, mom=0.96, top_k=n_best//3, temp=0.00003)
            try_seq(a_opt, "polish1")

        # Another FW-LP on polished
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(rem, 50))
            try_seq(a_opt, "polish_fw")

        # Micro perturbations
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            for _ in range(4):
                rem = deadline - time.monotonic() - 8
                if rem < 5:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
                a_opt, _ = sg_polish(f_pert, time_limit=min(rem, 12),
                                     lr=0.0005, mom=0.97, top_k=n_best//3, temp=0.00001)
                try_seq(a_opt, "micro")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
