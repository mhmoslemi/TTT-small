import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_obj(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s), conv, s

def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    m = 2 * n - 1
    C = np.zeros((m, n))
    for i in range(n):
        C[i:i+n, i] = a
    return C

def solve_lp_direction(a, M, A_ub=None):
    n = len(a)
    if A_ub is None:
        A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return None, None

    # Exponential decay from center
    dist = np.abs(x - 0.5 * (n - 1))
    exp_dec = np.exp(-dist / (n * 0.15))
    inits.append(exp_dec / exp_dec.sum())

    # Parent (if available)
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits.append(p / ps)
            elif len(p) > n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits.append(pp / pps)
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits.append(pp / pps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase hybrid search using smooth subgradient + LP refinement.
    Explores multiple initializations and sequence lengths with adaptive restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, parent_score={evaluate_sequence(p.tolist()):.6f}")
    except Exception:
        pass

    # Phase 1: Initial exploration across lengths and initializations
    print("=== Phase 1: Exploration ===")
    t_start = time.monotonic()
    phase1_time = min(400.0, (deadline - time.monotonic()) * 0.25)
    phase1_deadline = t_start + phase1_time

    lengths = [512, 384, 640]
    candidates = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break
        inits = generate_diverse_inits(n, parent, rng)
        t_per_init = max(10.0, (phase1_deadline - time.monotonic() - 20) / max(1, len(inits)))

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline - 15:
                break
            t_alloc = min(t_per_init, phase1_deadline - time.monotonic() - 10)
            if t_alloc < 5:
                break

            # Quick optimization to get a decent starting point
            a_opt, obj_opt = quick_lse_optimize(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))

            if np.isfinite(score):
                candidates.append((score, n, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Phase1 n={n}, init{i}: score={score:.6f}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # Phase 2: Deep hybrid optimization on top candidates
    print("=== Phase 2: Deep Hybrid Optimization ===")
    t_start2 = time.monotonic()
    top_n = 4
    top_candidates = candidates[:top_n]

    # Always include best_sequence
    if best_sequence is not None:
        bs_score = float(evaluate_sequence(best_sequence.tolist()))
        if bs_score < top_candidates[-1][0] if top_candidates else True:
            top_candidates.append((bs_score, len(best_sequence), best_sequence.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif (len(best_sequence), tuple(best_sequence[:5])) not in [(c[1], tuple(c[2][:5])) for c in top_candidates]:
            top_candidates.append((bs_score, len(best_sequence), best_sequence.copy()))

    remaining = deadline - time.monotonic() - 200.0
    n_deep = max(1, min(len(top_candidates), 4))
    t_per_deep = max(30.0, remaining * 0.6 / n_deep)

    for i in range(min(n_deep, len(top_candidates))):
        if time.monotonic() > deadline - 150:
            break
        score_init, n_init, a_init = top_candidates[i]
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 100)
        if t_alloc < 15:
            break

        print(f"  Deep opt {i}: n={n_init}, init_score={score_init:.6f}, time_budget={t_alloc:.0f}s")
        a_opt, obj_opt = hybrid_optimize(a_init, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep opt {i}: score={score:.6f} (improved!)")
        else:
            print(f"  Deep opt {i}: score={score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # Phase 3: LP-focused finishing on the best sequence
    print("=== Phase 3: LP Finishing ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining3 = deadline - time.monotonic() - 60.0
        if remaining3 > 30:
            # Multiple LP polishing passes with different perturbations
            n_perturb = max(2, int(remaining3 / 60))
            t_per_pass = remaining3 / n_perturb

            for pi in range(n_perturb):
                if time.monotonic() > deadline - 30:
                    break
                t_alloc = min(t_per_pass, deadline - time.monotonic() - 20)
                if t_alloc < 10:
                    break

                if pi == 0:
                    a_start = best_sequence.copy()
                else:
                    # Small perturbation from best
                    scale = 0.003 * (1 + pi * 0.5)
                    noise = rng.standard_normal(len(best_sequence)) * scale
                    a_start = best_sequence + noise
                    a_start = np.maximum(a_start, 0.0)
                    s = a_start.sum()
                    if s < 1e-15:
                        a_start = best_sequence.copy()
                    else:
                        a_start = a_start / s

                a_opt, obj_opt = hybrid_optimize(a_start, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  LP finish {pi}: score={score:.6f}")
            print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # Phase 4: Final sprint - try a few more diverse starts if time allows
    print("=== Phase 4: Final Sprint ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining4 = deadline - time.monotonic() - 10.0
        if remaining4 > 20:
            n_final = min(3, max(1, int(remaining4 / 30)))
            t_per_final = remaining4 / n_final

            for fi in range(n_final):
                if time.monotonic() > deadline - 5:
                    break
                t_alloc = min(t_per_final, deadline - time.monotonic() - 3)
                if t_alloc < 5:
                    break

                # Larger perturbation to explore new regions
                n_bs = len(best_sequence)
                scale = 0.02 + 0.01 * fi
                noise = rng.standard_normal(n_bs) * scale
                a_try = best_sequence + noise
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-15:
                    a_try = best_sequence.copy()
                else:
                    a_try = a_try / s

                a_opt, obj_opt = hybrid_optimize(a_try, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final {fi}: score={score:.6f}")
            print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Shouldn't happen, but ensure validity
        best_sequence = np.ones(256)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
