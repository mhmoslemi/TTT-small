import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(f, conv, peaks, weights):
    """Compute subgradient of max(conv) w.r.t. f using weighted peaks."""
    n = len(f)
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-15:
            continue
        k = int(peaks[i])
        # d(a*a)(k)/da_i = 2 * a_{k-i} if 0 <= k-i < n
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * weights[i] * f[k - idx]
    return sg

def multiplicative_update(f, conv, lr, top_k, temp):
    """One step of multiplicative weight update."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = compute_subgradient(f, conv, peaks, w)
    f_new = f * np.exp(-lr * sg)
    f_new = np.maximum(f_new, 1e-30)
    f_new = f_new / f_new.sum()
    return f_new

def solve_lp_for_direction(f, M, n_constraints=None):
    """Solve LP to find a good search direction."""

    if n_constraints is not None:
        # Only use the most violated constraints

        # Find peaks

        # Build reduced constraint matrix
        A_ub = np.zeros((len(peaks), n))
        b_ub = np.full(len(peaks), M)

- $k-j = i$, i.e., $j = k-i$: term is $a_{k-i} a_i$ (if valid)

If $i \neq k-i$ (i.e., $k \neq 2i$), these are two separate linear terms in $a_i$: $2a_i a_{k-i}$.
If $k = 2i$, the term is $a_i^2$.

So $(a*a)(k)$ is:
- $a_i^2 + 2a_i a_{k-i} + c_k$ if $k = 2i$ and $k-i = i$ (same term)
- Wait, if $k = 2i$, then $j = i$ gives $a_i a_i = a_i^2$, and $j = k-i = i$ gives the same. So it's just $a_i^2 + c_k$.
- If $k \neq 2i$ but both $0 \leq k-i < n$ and $0 \leq i < n$ (always true), the terms are $a_i a_{k-i} + a_{k-i} a_i = 2a_i a_{k-i}$, so $(a*a)(k) = 2a_i a_{k-i} + c_k$.
- If only one of $k-i$ is in range, it's $a_i a_{k-i} + c_k$ (linear).

So the max over k is a piecewise function of $a_i$ that's:
- Quadratic (one piece) at the $k = 2i$ constraint
- Linear (many pieces) at other constraints
- The max of all these

To minimize the max, we can use bisection on $a_i$: for a given target $t$, check if there exists $a_i \geq 0$ such that all $(a*a)(k) \leq t$. The constraints give:
- $a_i^2 + c_{2i} \leq t$ → $a_i \leq \sqrt{t - c_{2i}}$ (if $t \geq c_{2i}$)
- $2a_i a_{k-i} + c_k \leq t$ → $a_i \leq (t - c_k)/(2a_{k-i})$ (if $a_{k-i} > 0$ and $t \geq c_k$)
- Also $a_i \geq 0$

So the feasible range for $a_i$ is $[0, \min_k (t - c_k) / d_k]$ where $d_k$ is the coefficient of $a_i$ in constraint $k$. We want the largest $t$ for which this range is non-empty, which is a bisection problem.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, peaks, weights):
    """Compute weighted subgradient of max_k (f*f)(k)."""
    n = len(f)
    sg = np.zeros(n)
    for i in range(len(peaks)):

            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * weights[i] * f[k - idx]
    return sg

def solve_reduced_lp(f, M, n_constraints=30):
    """Solve reduced LP with only top K most-violated constraints."""
    n = len(f)
    conv = fast_convolve(f)
    if n_constraints < 2 * n - 1:
        peaks = np.argpartition(conv, -n_constraints)[-n_constraints:]
    else:
        peaks = np.arange(2 * n - 1)
    
    A_ub = np.zeros((len(peaks), n))
    b_ub = np.full(len(peaks), M)
    for i, k_idx in enumerate(peaks):
        for j in range(n):
            kj = k_idx - j
            if 0 <= kj < n:
                A_ub[i, j] = f[kj]
    
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    if new_n == n:
        return f.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

    inits.append(edge_w / edge_w.sum())
    
    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c / c.sum())
    
    # Multi-peak
    n_peaks = rng.integers(2, 5)
    centers = rng.uniform(0, n, n_peaks).astype(int)
    mp = np.zeros(n)
    for c_ in centers:
        mp += np.exp(-0.5 * ((x - c_) / (n / (2 * n_peaks))) ** 2)
    inits.append(mp / mp.sum())
    
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid search using multiplicative weight updates,
    coordinate descent, and LP guidance.
    """
    rng = np.random.default_rng(seed)

    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass
    
    # === Phase 1: Quick multi-start ===
    print("=== Phase 1: Multi-start exploration ===")
    
    if parent is not None:
        n_p = len(parent)
        t1 = min(300, deadline - time.monotonic() - 200)
        if t1 > 50:
            print(f"  Hybrid on parent (n={n_p}), t={t1:.0f}s")
            a_opt, _ = hybrid_optimize(parent, time_limit=t1, verbose=True)
            try_seq(a_opt, "parent_hybrid")
    
    # Try a few other lengths quickly
    if parent is not None:
        other_lengths = [320, 400, 448, 512]

    for n in other_lengths:
        if time.monotonic() > deadline - 100:
            break
        if parent is not None and n == len(parent):
            continue
        inits = generate_initializations(n, parent, rng)
        t_alloc = max(20, min(100, remaining / len(other_lengths)))
        remaining -= t
        
        # Try the best initialization
        best_init = None
        best_init_score = np.inf
        for f_init in inits[:3]:
            conv = fast_convolve(f_init)
            sc = 2 * n * conv.max()
            if sc < best_init_score:
                best_init_score = sc
                best_init = f_init
        
        if best_init is not None:
            a_opt, _ = hybrid_optimize(best_init, time_limit=min(t_alloc, 80))
            try_seq(a_opt, f"n{n}_init")
    
    print(f"  Phase 1 best: {best_score:.8f}")
    
    # === Phase 2: Deep optimization on best ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Run multiple optimization strategies on the best
        strategies = [
            ("hybrid", hybrid_optimize),
            ("lp", lp_guided_optimize),
            ("multiplicative", multiplicative_optimize),
        ]
        
        remaining = deadline - time.monotonic() - 100
        t_per = remaining / len(strategies)
        
        for name, opt_fn in strategies:
            rem = deadline - time.monotonic() - 60
            if rem < 30:
                break
            t_alloc = min(t_per, rem)
            print(f"  {name} on n={n_best}, t={t_alloc:.0f}s")
            
            if name == "hybrid":
                a_opt, _ = hybrid_optimize(best_seq, time_limit=t_alloc, verbose=True)
            elif name == "lp":
                a_opt, _ = lp_guided_optimize(best_seq, time_limit=t_alloc, verbose=True)
            else:
                a_opt, _ = multiplicative_optimize(best_seq, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, f"deep_{name}")
    
    # === Phase 3: Coordinate descent refinement ===
    print(f"\n=== Phase 3: Coordinate Descent Refinement ===")
    
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 30

            f_cd = best_seq.copy()

        n_best = len(best_seq)

        rem = deadline - time.monotonic() - 20
        if rem > 15:

            try_seq(a_opt, "final_mult")

        if rem > 10:
            a_opt, _ = lp_guided_optimize(best_seq, time_limit=min(rem, 20))
            try_seq(a_opt, "final_lp")

            t_cd = time.monotonic()
            while time.monotonic() - t_cd < rem:
                f_cd = coordinate_descent_refine(f_cd, n_passes=2)
                try_seq(f_cd, "final_cd")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
