Actually, for a sequence with a fixed "shape" (say, a Gaussian), as n increases:
- sum(a) stays the same (by normalization)
- max(conv) decreases (the convolution is more spread out)
- 2n increases

The net effect depends on how fast max(conv) decreases. For a Gaussian of fixed width (in fractional units), max(conv) ~ 1/sqrt(n), and 2n * max(conv) ~ 2n/sqrt(n) = 2*sqrt(n), which increases with n. So larger n is worse for fixed shape.

But if we reshape the sequence to be "flatter" as n increases (like spreading the mass over more points), we might do better. The uniform sequence gives 2 regardless of n. So the question is: for what n can we get the best flatness?

For n=512, the theoretical lower bound is 2n/(2n-1) ≈ 1.001. For n=1024, it's 2*1024/2047 ≈ 1.0015. So the lower bound is similar for large n.

I'll stick with n=512 since that's what the parent uses.

"""
ALTO-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses coordinate-wise mass transfer with adaptive peak weighting,
multiple initializations, and periodic perturbations.
"""

import numpy as np
import time


def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_burden(a, top_peaks, weights):
    """
    Compute burden for each coordinate.
    burden[i] = sum_k weights[k] * a[k - i] (where k-i is valid)
    """
    n = len(a)
    burden = np.zeros(n)
    for w, k in zip(weights, top_peaks):
        k = int(k)
        # i ranges over [max(0, k-(n-1)), min(n-1, k)]
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        if i_hi < i_lo:
            continue
        indices = np.arange(i_lo, i_hi + 1)
        partner_idx = k - indices
        burden[indices] += w * a[partner_idx]
    return burden

    s = a.sum()

    else:
        a = a / s
    n = len(a)

    best_a = a.copy()

    max_no_improve = 2000

        # Exclude coordinates that are already ~0 from being reduced
        burden_masked = burden.copy()
        burden_masked[a < 1e-12] = -np.inf  # Can't reduce below 0
        
        i_red = int(np.argmax(burden_masked))

            # All coordinates are ~0, should not happen
            break

        i_add = int(np.argmin(burden_for_add))

                    a[i_add] += a[idx]

            a[i_add] += best_alpha

                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Periodic perturbation
        if no_improve >= max_no_improve:
            scale = perturb_scale * (1.0 + 0.1 * (iteration // 10000))
            pert = rng.standard_normal(n) * scale
            a = best_a + pert
            a = np.maximum(a, 0.0)
            sa = a.sum()
            if sa < 1e-15:
                a = best_a.copy()
            else:
                a = a / sa
            conv = fast_conv(a)
            no_improve = 0
            max_no_improve = min(20000, int(max_no_improve * 1.3))
            perturb_scale = min(0.02, perturb_scale * 1.2)

            i2 = rng.integers(n)

                a_try[i1] -= alpha

                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj:
                    a = a_try
                    conv = conv_try
        
        iteration += 1
    
    return best_a, best_obj


def subgradient_polish(a0, time_limit, top_k=20, lr=0.005, momentum=0.9):
    """
    Subgradient descent for final polish.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    conv = fast_conv(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Find top peaks
        k = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k)[-k:]
        peak_vals = conv[peaks]
        max_val = peak_vals.max()
        
        # Softmax weights
        scaled = (peak_vals - max_val) / 0.001
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= wsum
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            pk = int(peaks[i])
            i_lo = max(0, pk - (n - 1))
            i_hi = min(n - 1, pk)
            indices = np.arange(i_lo, i_hi + 1)
            partner = pk - indices
            sg[indices] += w[i] * 2.0 * a[partner]
        
        # Update
        v = momentum * v - lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
        
        conv = fast_conv(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Decay learning rate
        lr *= 0.9995
        momentum = min(0.97, momentum + 0.0005)
        
        if no_improve >= 500:
            break
    
    return best_a, best_obj
