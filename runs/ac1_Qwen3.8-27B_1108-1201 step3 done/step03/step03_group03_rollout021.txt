    return np.fft.irfft(A * A, fft_size)[:size]

def compute_pressure(w, a, n):
    """p[j] = sum_k w[k] * a[k-j] for j=0..n-1"""
    w_flip = w[::-1]
    fft_size = 2 ** (2 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_flip, fft_size)
    conv = np.fft.irfft(A * W, fft_size)
    return conv[2*n-2 : n-2 : -1]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def mirror_descent(a_init, lam, n_iters, eta0, verbose=False):
    """Mirror descent (multiplicative weights) with log-sum-exp smoothing."""
    a = np.maximum(a_init, 1e-15).astype(np.float64).copy()
    a = a / a.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    best_max = conv.max()
    best_a = a.copy()
    
    eta = eta0
    for t in range(n_iters):
        # Exponential weights
        cs = lam * conv
        cs_max = cs.max()
        w = np.exp(cs - cs_max)
        w_sum = w.sum()
        if w_sum < 1e-300:
            break
        w /= w_sum
        
        # Pressure
        p = compute_pressure(w, a, n)
        
        # Multiplicative update
        log_a = np.log(a) - eta * p
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        s = a_new.sum()
        if s < 1e-300:
            break
        a_new /= s
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()
        
        conv = fast_convolve(a_new)
        cur_max = conv.max()
        
        if cur_max < best_max - 1e-15:
            best_max = cur_max
            best_a = a_new.copy()
            eta = max(eta * 1.02, 0.01)  # slight increase if improving
        else:
            eta *= 0.95  # decrease if not improving
        
        a = a_new
        eta = max(eta * (1.0 / (1.0 + 0.001 * t)), 0.005)
    
    return best_a, 2.0 * n * best_max

def lbfgs_smooth(a_init, lam, max_iter=500):
    """L-BFGS-B on logsumexp(lam * conv(a,a)) with simplex constraint."""
    n = len(a_init)
    a0 = np.maximum(a_init, 0.0).astype(np.float64)
    s = a0.sum()
    if s < 1e-15:
        a0 = np.ones(n) / n
    else:
        a0 /= s
    
    def obj_grad(x):
        x = np.maximum(x, 1e-30)
        conv = fast_convolve(x)
        sc = lam * conv
        sc_max = sc.max()
        val = sc_max + np.log(np.exp(sc - sc_max).sum())
        w = np.exp(sc - sc_max)
        w /= w.sum()
        p = compute_pressure(w, x, n)
        grad = 2.0 * p  # gradient of logsumexp(lam*conv) w.r.t. x is 2*lam*(w cross x)/n... 
        # Wait, let me recompute.
        # f(x) = logsumexp(lam * conv(x,x))
        # df/dx_j = lam * sum_k p_k * d(conv_k)/d(x_j)
        # d(conv_k)/d(x_j) = 2 * x[k-j] (as derived earlier)
        # So df/dx_j = lam * sum_k p_k * 2 * x[k-j] = 2*lam * p_j
        # where p_j = sum_k p_k * x[k-j]
        # So grad = 2 * lam * compute_pressure(w, x, n)
        # But wait, w = softmax(lam * conv), so the lam is already in w.
        # Let me redo: f = log(sum_k exp(lam * conv_k))
        # df/dx_j = (1/sum_m exp(lam*conv_m)) * sum_k exp(lam*conv_k) * lam * d(conv_k)/d(x_j)
        # = lam * sum_k w_k * d(conv_k)/d(x_j)
        # = lam * sum_k w_k * 2*x[k-j]
        # = 2*lam * p[j]
        # where p[j] = sum_k w_k * x[k-j]
        return val, 2.0 * lam * p
    
    # Use L-BFGS-B with bounds [0, 1]
    # We don't enforce sum=1 here; the scale invariance means it doesn't matter
    # Actually, it DOES matter because the objective is NOT scale-invariant in this form!
    # f(x) = logsumexp(lam * conv(x,x)) is NOT scale-invariant.
    # If we scale x by c, conv scales by c^2, so f scales by logsumexp(lam*c^2*conv_orig)
    # which is not the same.
    
    # So we DO need to enforce sum(x) = 1.
    # Option 1: Add penalty for sum != 1
    # Option 2: Use SLSQP with equality constraint
    # Option 3: Parameterize x = softmax(z) (guarantees x>0, sum x=1)
    
    # Let's use option 3: work with z, x = softmax(z)
    # Then minimize f(softmax(z)) w.r.t. z
    # df/dz_j = sum_i (df/dx_i) * (dx_i/dz_j)
    # dx_i/dz_j = x_i * (delta_ij - x_j)
    # So df/dz_j = sum_i grad_i * x_i * (delta_ij - x_j) = x_j * (grad_j - sum_i grad_i * x_i)
    
    def softmax(z):
        z = z - z.max()
        e = np.exp(z)
        return e / e.sum()
    
    def obj_soft(z):
        x = softmax(z)
        conv = fast_convolve(x)
        sc = lam * conv
        sc_max = sc.max()
        val = sc_max + np.log(np.exp(sc - sc_max).sum())
        w = np.exp(sc - sc_max)
        w /= w.sum()
        p = compute_pressure(w, x, n)
        grad_x = 2.0 * lam * p
        # Transform to z-space
        mean_grad = np.dot(grad_x, x)
        grad_z = x * (grad_x - mean_grad)
        return val, grad_z
    
    z0 = np.log(a0 + 1e-15)
    result = minimize(obj_soft, z0, method='L-BFGS-B', 
                      options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8})
    
    a_opt = softmax(result.x)
    conv = fast_convolve(a_opt)
    return a_opt, 2.0 * n * conv.max()

def build_lp_pairs(n):
    pairs = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            pairs.append((k, j, k - j))
    return pairs

def solve_lp_direction(f, M, pairs, n):
    A_ub = np.zeros((2 * n - 1, n))
    for k, j, idx in pairs:
        A_ub[k, j] = f[idx]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None

def qp_peak_shave(a, n_shaves=50, step=0.01):
    """Iteratively shave peaks using linearized QP."""
    from scipy.optimize import minimize as sp_minimize
    a = np.maximum(a, 0.0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    n = len(a)
    
    conv = fast_convolve(a)
    best_max = conv.max()
    best_a = a.copy()
    
    for it in range(n_shaves):
        M = conv.max()
        if M < 1e-15:
            break
        # Find peaks
        peak_thresh = M * 0.95
        peak_idx = np.where(conv >= peak_thresh)[0]
        
        # Subgradient direction
        # For each peak k, the gradient contribution is 2*a[k-j]
        p = np.zeros(n)
        for k in peak_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            p[j_start:j_end+1] += 2.0 * a[k - np.arange(j_end - j_start + 1)]
        
        # Normalize pressure
        p_norm = p / (p.max() + 1e-30)
        
        # Take a step in the negative pressure direction
        a_new = a - step * p_norm
        a_new = project_simplex(a_new)
        
        conv_new = fast_convolve(a_new)
        if conv_new.max() < best_max - 1e-15:
            best_max = conv_new.max()
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
        else:
            step *= 0.5
            if step < 1e-6:
                break
    
    return best_a

def gen_inits(n, parent=None, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    # Bimodal
    g2 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Sparse (concentrated on edges)
    edge = np.zeros(n)
    e = max(1, n // 4)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    # Sparse random
    sparse = np.zeros(n)
    k = max(2, n // 3)
    indices = rng.choice(n, size=k, replace=False)
    sparse[indices] = 1.0
    inits['sparse'] = sparse / sparse.sum()
    
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            elif len(parent) > n:
                p2 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), p)
                inits['parent_ds'] = p2 / p2.sum()
            else:
                p2 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), p)
                inits['parent_up'] = p2 / p2.sum()
    
    return inits

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search combining mirror descent, L-BFGS, and LP-based optimization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Try to get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
    except Exception:
        pass
    
    # Phase 1: Mirror Descent
    print("=== Phase 1: Mirror Descent ===")
    t0 = time.monotonic()
    lengths = [512, 640, 384, 768]
    results = []
    
    n_starts = 4  # initializations per length
    phase1_time = min(600.0, (deadline - time.monotonic()) * 0.35)
    t_per_start = phase1_time / (len(lengths) * n_starts)
    
    for n in lengths:
        if time.monotonic() > deadline - 200:
            break
        inits = gen_inits(n, parent, rng)
        prio = ['parent', 'parent_ds', 'parent_up', 'bimodal', 'gauss', 'uniform', 'sparse', 'random', 'edge', 'triangle']
        names = [nm for nm in prio if nm in inits][:n_starts]
        
        for name in names:
            if time.monotonic() > deadline - 180:
                break
            t_alloc = min(t_per_start * 1.5, deadline - time.monotonic() - 160)
            if t_alloc < 5:
                break
            
            a0 = inits[name]
            # Convert time to iterations (roughly 0.5ms per iteration for n=512)
            n_iters = int(t_alloc * 2000)  # about 2000 iters per second

                    best_sequence = a_opt.copy()
                    print(f"  MD n={n} {name}: {score:.6f}")
    
    print(f"Phase 1 best: {best_score:.6f} (time: {time.monotonic()-t0:.1f}s)")
    
    # Phase 2: L-BFGS refinement
    print("=== Phase 2: L-BFGS Refinement ===")
    t0 = time.monotonic()
    results.sort(key=lambda x: x[0])
    
    top_candidates = []
    seen = set()
    for score, n, name, a_opt in results:
        key = (n,)
        if key not in seen and n == len(a_opt):
            top_candidates.append((n, a_opt.copy()))
            seen.add(key)
        if len(top_candidates) >= 4:
            break
    
    if best_sequence is not None:
        top_candidates.append((len(best_sequence), best_sequence.copy()))
    
    for n, a0 in top_candidates:

            break
