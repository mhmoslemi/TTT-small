def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])

    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_maximize_sum(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def lp_minimize_peak(a):
    """Solve: min M' s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M' for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def lp_fixed_point(a0, time_limit, step_init=0.4, method='max_sum', seed=42,
                   perturb_scale=0.01, perturb_every=80):
    """LP-based fixed-point iteration with backtracking line search."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        M = conv.max()

        if method == 'max_sum':
            b = lp_maximize_sum(a, M)
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum
        else:
            b, _ = lp_minimize_peak(a)
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum

        if np.linalg.norm(b - a) < 1e-12:
            break

        # Backtracking line search
        best_t = 0.0
        best_cand_obj = best_obj
        a_best_cand = None
        conv_best_cand = None

        t_candidates = [step, step * 0.6, step * 0.4, step * 0.2, step * 0.1]
        for t in t_candidates:
            a_cand = (1.0 - t) * a + t * b
            a_cand = np.maximum(a_cand, 0.0)
            a_cand_sum = a_cand.sum()
            if a_cand_sum < 1e-15:
                continue
            a_cand = a_cand / a_cand_sum
            conv_cand = fast_convolve(a_cand)
            val_cand = 2.0 * n * conv_cand.max()
            if val_cand < best_cand_obj - 1e-14:
                best_cand_obj = val_cand
                best_t = t
                a_best_cand = a_cand.copy()
                conv_best_cand = conv_cand.copy()

        if a_best_cand is not None:
            best_obj = best_cand_obj
            best_a = a_best_cand.copy()
            a = a_best_cand
            conv = conv_best_cand
            no_improve = 0
            step = min(0.95, step * 1.03)
        else:
            # No improvement found; take a small step anyway
            no_improve += 1
            step = max(0.05, step * 0.8)
            a = (1.0 - step) * a + step * b
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum < 1e-15:
                break
            a = a / a_sum
            conv = fast_convolve(a)
            cur = 2.0 * n * conv.max()
            if cur < best_obj:
                best_obj = cur
                best_a = a.copy()

        if no_improve >= perturb_every:
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v = 2.0 * n * conv.max()
            if v < best_obj:
                best_obj = v
                best_a = a.copy()
            no_improve = 0
            perturb_every = int(perturb_every * 1.4)
            step = step_init
            perturb_scale = max(0.003, perturb_scale * 0.8)

        if no_improve > perturb_every * 4:
            break

    return best_a, best_obj


def subgradient_step(a, conv, top_k=20, temperature=0.0001):
    """Compute subgradient of peak objective using top-k peaks."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition.conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

    if temperature > 1e-15:
        scaled = peak_vals / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        wi = w[i] * 2.0
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += wi * a[k - idx]
    return sg


def optimize_subgrad(a0, time_limit, lr_init=0.04, momentum=0.88, top_k=30,
                     temp_init=0.0005, seed=42):
    """Projected subgradient with momentum and periodic perturbation."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        sg = subgradient_step(a, conv, top_k, temp)
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        temp *= 0.999

        if no_improve >= 800:
            if perturb_count < 15:
                scale = 0.005 * (1.0 + 0.5 * np.sin(perturb_count))
                noise = rng.standard_normal(n) * scale
                a = project_simplex(a + noise)
                conv = fast_convolve(a)
                v *= 0.3
                no_improve = 0
                perturb_count += 1
                lr = lr_init * 0.5
            else:
                break

    return best_a, best_obj


def peak_shave_refine(a0, time_limit, seed=0):
    """Direct peak-shaving: iteratively reduce pre-images of top peaks."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 5000:
        conv = fast_convolve(a)
        cur_peak = conv.max()
        cur_obj = 2.0 * n * cur_peak
        mean_conv = 1.0 / (2 * n - 1)

        # Find top peaks
        num_peaks = min(10, len(conv))
        top_idx = np.argpartition(conv, -num_peaks)[-num_peaks:]
        top_idx = top_idx.sort() if hasattr(top_idx, 'sort') else np.sort(top_idx)

        # For each top peak, find the j that contributes most
        total_grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start > j_end:
                continue
            idx = np.arange(j_start, j_end + 1)
            contrib = a[idx] * a[k - idx]
            total_grad[idx] += contrib

        total_grad = total_grad / total_grad.sum() if total_grad.sum() > 1e-15 else np.ones(n) / n

        # Find "quiet" positions (low contribution to peaks)
        quiet_score = np.zeros(n)
        for j in range(n):
            # How much does a[j] contribute to total peak?
            quiet_score[j] = total_grad[j]

        # Move mass from high-contribution to low-contribution positions
        threshold = np.percentile(quiet_score, 50)
        high_mask = quiet_score > threshold
        low_mask = quiet_score <= threshold

        if not high_mask.any() or not low_mask.any():
            no_improve += 1
            continue

        # Compute how much to move
        excess = a[high_mask].sum() * 0.05
        if excess < 1e-10:
            no_improve += 1
            continue

        low_weights = np.ones(low_mask.sum())
        a_new = a.copy()
        a_new[high_mask] -= excess / high_mask.sum() * (a[high_mask] / a[high_mask].sum()) * high_mask.sum()
        # Simpler: proportional reduction
        a_new = a.copy()
        reduction = a[high_mask] * 0.03
        a_new[high_mask] -= reduction
        a_new[low_mask] += reduction.sum() * (a[low_mask] / a[low_mask].sum()) if a[low_mask].sum() > 1e-15 else reduction.sum() / low_mask.sum()
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            break
        a_new = a_new / a_new_sum

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller step
            a_try = a + 0.01 * (a_new - a)
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            conv_try = fast_convolve(a_try)
            if 2.0 * n * conv_try.max() < 2.0 * n * conv.max():
                a = a_try
                no_improve = 0

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    for freq in [1, 2, 3, 4]:
        a = 1.0 + 0.7 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum > 1e-10:
            inits[f'cos_{freq}'] = a / a_sum

    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2)
    g_sum = g.sum()
    if g_sum > 1e-10:
        inits['gauss'] = g / g_sum

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random2'] = rng.dirichlet(np.ones(n) * 0.1)

    # Boundary-boosted
    boundary = np.ones(n) + 0.5 * ((2 * x / (n - 1) - 1) ** 2)
    boundary = np.maximum(boundary, 0)
    inits['boundary'] = boundary / boundary.sum()

    # Two peaks
    a = np.zeros(n)
    a[n // 4] = 1.0
    a[3 * n // 4] = 1.0
    # Smooth
    kernel = np.exp(-0.5 * (np.arange(-10, 11) / 3.0) ** 2)
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    inits['two_peaks'] = a / a.sum()

    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()

    return inits


def try_update_sequence(a, best_state):
    """Try to update best state with sequence a. Returns score."""
    if a is None:
        return best_state['score']
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return best_state['score']
    a = np.clip(a, 0.0, 1000.0)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_state['score']:
        best_state['score'] = score
        best_state['sequence'] = a.copy()
        print(f"  New best: {score:.6f} (n={len(a)})")
    return best_state['score']


def propose_candidate(budget_s=1800):
    """
    Multi-phase search to minimize the peak auto-correlation objective.
    Phase 1: LP-based exploration with multiple initializations and lengths.
    Phase 2: Deep LP refinement on the best candidate.
    Phase 3: Subgradient and peak-shaving polishing.
    Phase 4: Perturbation and re-optimization.
    """
    rng = np.random.default_rng(7)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_state = {'score': float('inf'), 'sequence': None}

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.4f}, score={float(evaluate_sequence(p.tolist())):.6f}")
    except Exception as e:
        print(f"No parent available: {e}")

    # Phase 1: LP exploration
    print("=== Phase 1: LP Exploration ===")
    t1_start = time.monotonic()

    lengths = list(set([512, 256, 1024]))
    if parent is not None:
        if len(parent) not in lengths:
            lengths.insert(0, len(parent))
        lengths = sorted(lengths, key=lambda x: abs(x - len(parent)))

    for n in lengths:
        remaining = deadline - time.monotonic() - 500
        if remaining < 60:
            break

        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())
        priority = ['parent', 'uniform', 'boundary', 'random', 'cos_1', 'gauss']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:5]

        n_inits = len(init_names)
        n_lengths = len(lengths)
        time_per_init = max(20.0, remaining / (n_lengths * n_inits))

        for name in init_names:
            remaining = deadline - time.monotonic() - 400
            if remaining < 10:
                break
            t_alloc = min(time_per_init, remaining)
            if t_alloc < 5:
                break

            a0 = inits[name]
            print(f"  n={n}, init={name}, t={t_alloc:.1f}s")

            # Try both LP methods
            t_each = t_alloc / 2.0
            for method in ['max_sum', 'min_peak']:
                a_opt, _ = lp_fixed_point(a0, time_limit=t_each,
                                           method=method, step_init=0.35,
                                           seed=rng.integers(0, 10000))
                try_update_sequence(a_opt, best_state)

    print(f"Phase 1 done: best={best_state['score']:.6f}, elapsed={time.monotonic()-t1_start:.0f}s")

    # Phase 2: Deep LP refinement
    print("=== Phase 2: Deep LP Refinement ===")
    t2_start = time.monotonic()

    if best_state['sequence'] is not None:
        phase2_deadline = deadline - 400
        for step_init in [0.4, 0.25, 0.6, 0.15, 0.5]:
            if time.monotonic() > phase2_deadline:
                break
            t_alloc = min(60.0, phase2_deadline - time.monotonic() - 5)
            if t_alloc < 5:
                break
            for method in ['max_sum', 'min_peak']:
                if time.monotonic() > phase2_deadline:
                    break
                t_single = t_alloc / 2.0
                if t_single < 3:
                    break
                a_opt, _ = lp_fixed_point(best_state['sequence'], time_limit=t_single,
                                           method=method, step_init=step_init,
                                           seed=rng.integers(0, 100000),
                                           perturb_scale=0.008)
                try_update_sequence(a_opt, best_state)

    print(f"Phase 2 done: best={best_state['score']:.6f}, elapsed={time.monotonic()-t2_start:.0f}s")

    # Phase 3: Subgradient + peak shaving
    print("=== Phase 3: Subgradient + Peak Shaving ===")
    t3_start = time.monotonic()

    if best_state['sequence'] is not None:
        phase3_deadline = deadline - 200
        n_best = len(best_state['sequence'])

        for lr in [0.03, 0.015, 0.05]:
            if time.monotonic() > phase3_deadline:
                break
            t_alloc = min(50.0, phase3_deadline - time.monotonic() - 5)
            if t_alloc < 5:
                break
            a_opt, _ = optimize_subgrad(best_state['sequence'], time_limit=t_alloc,
                                         lr_init=lr, momentum=0.9,
                                         top_k=max(20, n_best // 15),
                                         seed=rng.integers(0, 100000))
            try_update_sequence(a_opt, best_state)

        # Peak shaving
        if time.monotonic() < phase3_deadline:
            t_alloc = min(40.0, phase3_deadline - time.monotonic() - 5)
            if t_alloc > 5:
                a_opt, _ = peak_shave_refine(best_state['sequence'], time_limit=t_alloc,
                                              seed=rng.integers(0, 100000))
                try_update_sequence(a_opt, best_state)

    print(f"Phase 3 done: best={best_state['score']:.6f}, elapsed={time.monotonic()-t3_start:.0f}s")

    # Phase 4: Perturbation + re-optimization
    print("=== Phase 4: Perturbations ===")
    t4_start = time.monotonic()

    if best_state['sequence'] is not None:
        phase4_deadline = deadline - 20
        n_best = len(best_state['sequence'])

        scales = [0.003, 0.008, 0.001, 0.012, 0.005, 0.02]
        for i, scale in enumerate(scales):
            if time.monotonic() > phase4_deadline:
                break
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(best_state['sequence'] + noise)
            t_alloc = min(30.0, phase4_deadline - time.monotonic())
            if t_alloc < 3:
                break
            a_opt, _ = lp_fixed_point(a_pert, time_limit=t_alloc,
                                       method='max_sum', step_init=0.3,
                                       seed=rng.integers(100000 + i, 200000),
                                       perturb_scale=0.005)
            try_update_sequence(a_opt, best_state)

        # Also try min_peak from perturbed
        for i in range(2):
            if time.monotonic() > phase4_deadline:
                break
            noise = rng.standard_normal(n_best) * 0.004
            a_pert = project_simplex(best_state['sequence'] + noise)
            t_alloc = min(25.0, phase4_deadline - time.monotonic())
            if t_alloc < 3:
                break
            a_opt, _ = lp_fixed_point(a_pert, time_limit=t_alloc,
                                       method='min_peak', step_init=0.25,
                                       seed=rng.integers(200000 + i, 300000))
            try_update_sequence(a_opt, best_state)

    print(f"Phase 4 done: best={best_state['score']:.6f}, elapsed={time.monotonic()-t4_start:.0f}s")

    # Phase 5: Final LP polish with very small steps
    print("=== Phase 5: Final Polish ===")
    t5_start = time.monotonic()

    if best_state['sequence'] is not None:
        t_alloc = max(10.0, deadline - time.monotonic() - 10)
        a_opt, _ = lp_fixed_point(best_state['sequence'], time_limit=t_alloc,
                                   method='max_sum', step_init=0.15,
                                   seed=rng.integers(500000, 600000),
                                   perturb_scale=0.003, perturb_every=50)
        try_update_sequence(a_opt, best_state)

    print(f"Phase 5 done: best={best_state['score']:.6f}, elapsed={time.monotonic()-t5_start:.0f}s")

    # Final fallback
    if best_state['sequence'] is None:
        best_state['sequence'] = np.ones(512) / 512
        best_state['score'] = float(evaluate_sequence(best_state['sequence'].tolist()))

    final_score = float(evaluate_sequence(best_state['sequence'].tolist()))
    if np.isfinite(final_score):
        best_state['score'] = min(best_state['score'], final_score)

    print(f"\nFINAL: score={best_state['score']:.6f}, n={len(best_state['sequence'])}")
    return [float(x) for x in best_state['sequence']]
