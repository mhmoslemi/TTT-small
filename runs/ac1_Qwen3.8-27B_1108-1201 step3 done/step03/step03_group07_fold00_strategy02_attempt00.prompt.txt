def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights with numerical stability."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_smooth_conv(a, w):
    """Compute gradient of sum_k w[k]*conv[k] w.r.t. a using FFT cross-correlation."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_obj(x, n, beta):
    """Top-level objective for scipy minimize (softmax parametrization)."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def _softmax_jac(x, n, beta):
    """Top-level Jacobian for scipy minimize (softmax parametrization)."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = compute_grad_smooth_conv(a, w)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def optimize_lbfgsb(a0, beta, maxiter=500):
    """Optimize using L-BFGS-B with softmax reparametrization."""
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)

    result = minimize(_softmax_obj, x0, jac=_softmax_jac, args=(n, beta),
                      method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 50})

    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt, result.fun

def flat_spectrum_init(n, rng):
    """Initialize with approximately flat power spectrum."""
    fft_size = 1 << (n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = 1.0
    phases = rng.uniform(0, 2 * np.pi, fft_size)
    for k in range(1, fft_size // 2 + 1):
        A[k] = np.exp(1j * phases[k])
        if k < fft_size - k:
            A[fft_size - k] = np.exp(-1j * phases[k])
    a = np.real(ifft(A, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def optimal_phase_init(n, rng):

def generate_initializations(n, parent, rng):

    inits = {}

    inits['ramp_inc'] = ramp
    inits['ramp_dec'] = ramp[::-1].copy()

    tri = np.minimum(x, n - 1 - x) + 1.0

    inits['bimodal'] = bi

    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.1)

    inits['flat_spec'] = flat_spectrum_init(n, rng)
    inits['opt_phase'] = optimal_phase_init(n, rng)

        if len(parent) == n:
            pp = np.maximum(parent, 0)
            ps = pp.sum()

                inits['parent'] = pp / ps
        elif len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0))

def peak_reduction_step(a, n_steps=10, reduction=0.97):
    """Iteratively reduce the peak of the convolution."""
    a = a.copy()
    n = len(a)
    for _ in range(n_steps):
        conv = fast_convolve(a)
        max_k = np.argmax(conv)
        i_start = max(0, max_k - n + 1)
        i_end = min(n - 1, max_k)
        # Reduce contributing elements proportionally
        contrib = np.abs(a[i_start:i_end + 1])
        total = contrib.sum()
        if total > 1e-15:
            reduction_factor = reduction * (contrib / total)
            a[i_start:i_end + 1] -= reduction_factor * (conv.max() - conv.mean()) / max(total, 1e-15)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
        else:
            a = np.ones(n) / n
    return a


def coordinate_refine(a, n_iters=200):
    """Coordinate descent to reduce max convolution."""
    n = len(a)
    a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    for it in range(n_iters):
        improved = False
        for i in range(n):
            conv = fast_convolve(a)
            max_k = int(np.argmax(conv))
            # Try adjusting a[i]
            old_val = a[i]
            # Find how much a[i] contributes to the peak
            delta = a[max_k - i] if 0 <= max_k - i < n else 0.0
            if delta < 1e-15:
                delta = a[i - max_k] if 0 <= i - max_k < n else 0.0

            # Try reducing
            for scale in [0.9, 0.95, 0.8]:
                a[i] = old_val * scale
                conv_new = fast_convolve(a)
                obj_new = 2.0 * n * conv_new.max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    improved = True
                    break
                # Try increasing (if it helps flatten other peaks)
            if not improved:
                a[i] = old_val
                # Try increasing slightly
                for scale in [1.1, 1.05]:
                    a[i] = old_val * scale
                    conv_new = fast_convolve(a)
                    # Check non-negativity and sum
                    obj_new = 2.0 * n * conv_new.max()
                    if obj_new < best_obj:
                        best_obj = obj_new
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        improved = True
                        break
                if not improved:
                    a[i] = old_val

        # Renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a /= s

    return a
