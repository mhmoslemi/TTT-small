```python
import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def mw_optimize(f0, time_limit, tau=0.01, eta=0.5):
    """Multiplicative weight optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute weights (softmax of convolution)
        scaled = conv / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        
        # Compute subgradient
        # corr[j] = sum_m f[m] * w[j+m]
        # Use FFT
        fft_size = 1 << (4 * n - 3 - 1).bit_length()
        f_pad = np.zeros(fft_size)
        f_pad[:n] = f
        w_flip = w[::-1]
        w_pad = np.zeros(fft_size)
        w_pad[:len(w)] = w_flip
        
        F = np.fft.rfft(f_pad)
        W = np.fft.rfft(w_pad)
        conv_corr = np.fft.irfft(F * W, fft_size)
        corr = conv_corr[n-1:n-1+n]  # This gives corr[j] for j=0,...,n-1
        
        # Extra term: f[j] * w[2j]
        w_diag = w[0:2*n:2]  # w[0], w[2], w[4], ..., w[2(n-1)]
        sg = corr + f * w_diag
        
        # Multiplicative update
        f_new = f * np.exp(-eta * sg / sg.max() if sg.max() > 0 else 0)
        f_new = np.maximum(f_new, 0)
        sn = f_new.sum()
        if sn < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / sn
        
        f = f_new
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-12:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive parameters
        tau *= 1.002  # Cool down (more focused on peaks)
        eta *= 0.9998  # Decrease step size
        
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(it)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(f)
            tau = 0.01
            eta = 0.5
            no_improve = 0
    
    return best_f, best_obj
```

    A = np.fft.rfft(a, fft_size)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def mw_optimize(f0, time_limit, tau_init=0.005, eta_init=1.0, seed=0):
    """
    Multiplicative weight optimizer using softmax-weighted subgradient.
    Fast iterations, suitable for many quick steps.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    rng = np.random.default_rng(seed)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    tau = tau_init
    eta = eta_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Softmax weights over convolution
        scaled = conv / tau
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        ws = exp_vals.sum()
        if ws < 1e-30:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = exp_vals / ws
        
        # Subgradient: sg[j] = sum_m f[m]*w[j+m] + f[j]*w[2*j]
        corr = np.correlate(w, f, mode='valid')  # corr[j] = sum_m f[m]*w[j+m]
        w_diag = w[0:2*n:2]  # w[0], w[2], ..., w[2(n-1)]
        sg = corr + f * w_diag
        
        # Multiplicative update
        sg_max = sg.max()
        if sg_max < 1e-15:
            continue
        sg_norm = sg / sg_max
        f_new = f * np.exp(-eta * sg_norm)
        f_new = np.maximum(f_new, 1e-30)  # Avoid exact zeros
        sn = f_new.sum()
        f = f_new / sn
        
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-12:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if it % 1000 == 0:
                print(f"      MW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
        
        # Adaptive parameters
        tau *= 1.003  # Cool down
        eta *= 0.9997  # Shrink step
        
        # Escape mechanism
        if no_improve > 500:
            pert_size = 0.003 + 0.003 * (no_improve // 1000)
            f = project_simplex(best_f + rng.standard_normal(n) * pert_size)
            conv = fast_convolve(f)
            tau = tau_init
            eta = eta_init
            no_improve = 0
    
    return best_f, best_obj

def lp_fw_optimize(f0, time_limit, seed=0):
    """LP-based Frank-Wolfe with adaptive M expansion."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    rng = np.random.default_rng(seed)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    # Precompute the constraint matrix
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f  # This changes with f! Can't precompute.
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP: max sum(g) s.t. A_ub @ g <= M * m_exp, g >= 0
        # Rebuild A_ub since f changes
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f
        
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n
        
        start_lp = time.monotonic()
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', 
                      options={'time_limit': max(1, (time_limit - (time.monotonic() - t0)) * 0.9)})
        
        if not res.success or res.x is None:
            m_exp *= 1.2
            if m_exp > 10:
                # Escape
                f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue
        
        g = np.maximum(res.x, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Check fixed point
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            else:
                m_exp = min(m_exp * 1.05, 5.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            ft = (1 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = (1 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-12:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if it % 10 == 0:
                    print(f"      LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)
    
    return best_f, best_obj

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. Fast multiplicative weight (softmax-weighted subgradient) optimization
2. LP-based Frank-Wolfe refinement
3. Diverse initialization exploration across sequence lengths

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    rng = np.random.default_rng(seed)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Build constraint matrix: A_ub[k,j] = f[j] if 0 <= k-j < n
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:j + n, j] = f

        c_vec = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n

        lp_time_left = max(1.0, (time_limit - (time.monotonic() - t0)) * 0.8)
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': lp_time_left})

        if not res.success or res.x is None:
            m_exp *= 1.2
            if m_exp > 8.0:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
            continue

        g = np.maximum(res.x, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Fixed point check
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            else:
                m_exp = min(m_exp * 1.05, 5.0)
            continue

        # Line search: coarse then fine
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if it % 20 == 0:
                    print(f"      LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 50:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.008)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Right-skewed exponential
    decay = np.exp(-x / (n * 0.25))
    inits.append(decay / decay.sum())

    # Left-skewed exponential
    decay_l = np.exp(-(n - 1 - x) / (n * 0.25))
    inits.append(decay_l / decay_l.sum())

    # Center Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    # Edge-concentrated (bowl)
    d = np.abs(x - 0.5 * (n - 1))
    edge = np.exp(-d / (0.15 * n)) + 0.01
    inits.append(edge / edge.sum())

    # Linear ramp
    ramp = x + 1.0
    inits.append(ramp / ramp.sum())

    # Two peaks
    p1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.08 * n)) ** 2)
    p2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.08 * n)) ** 2)
    inits.append((p1 + p2) / (p1 + p2).sum())

    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c / c.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # If parent available
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining fast multiplicative weight optimization
    with LP-based Frank-Wolfe refinement. Includes diverse initialization
    exploration and adaptive annealing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    state = SearchState()

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(parent)}")
            try_seq(state, parent, "parent")
    except Exception:
        pass

    # === Phase 1: Fast MW on parent ===
    print("\n=== Phase 1: MW on parent ===")
    if parent is not None:
        t_alloc = min(180, deadline - time.monotonic() - 500)
        if t_alloc > 20:
            a_opt, obj = mw_optimize(parent, time_limit=t_alloc,
                                     tau_init=0.003, eta_init=1.0, seed=1)
            print(f"  MW parent done: obj={obj:.10f}")
            try_seq(state, a_opt, "mw_p1")

    # === Phase 2: LP-FW refinement ===
    print("\n=== Phase 2: LP-FW refinement ===")
    if state.best_seq is not None:
        t_alloc = min(180, deadline - time.monotonic() - 400)
        if t_alloc > 20:
            a_opt, obj = lp_fw_optimize(state.best_seq, time_limit=t_alloc, seed=10)
            print(f"  LP-FW done: obj={obj:.10f}")
            try_seq(state, a_opt, "lpfw_p2")

    # === Phase 3: MW after LP-FW ===
    print("\n=== Phase 3: Post-LP MW polish ===")
    if state.best_seq is not None:
        t_alloc = min(120, deadline - time.monotonic() - 350)
        if t_alloc > 15:
            a_opt, obj = mw_optimize(state.best_seq, time_limit=t_alloc,
                                     tau_init=0.001, eta_init=0.5, seed=20,
                                     anneal_rate=1.003, cool_rate=0.9999)
            print(f"  Post-LP MW done: obj={obj:.10f}")
            try_seq(state, a_opt, "mw_p3")

    # === Phase 4: Exploration with diverse inits ===
    print("\n=== Phase 4: Diverse exploration ===")

    # Determine which lengths to try
    lengths_to_try = []
    if parent is not None:
        n_parent = len(parent)
        lengths_to_try.append(('parent_len', n_parent))
    for n_try in [384, 256, 512, 128, 640]:
        if not lengths_to_try or lengths_to_try[-1][1] != n_try:
            lengths_to_try.append((f'n{n_try}', n_try))

    # Limit to 3-4 lengths
    lengths_to_try = lengths_to_try[:4]

    for label, n in lengths_to_try:
        rem = deadline - time.monotonic() - 150
        if rem < 40:
            break

        inits = generate_inits(n, parent, rng)

        # If this is the parent length and we already tried it, skip first init
        skip_first = (parent is not None and n == len(parent))
        if skip_first:
            inits = inits[1:]

        n_inits_to_try = min(3, len(inits))
        t_per_init = min(25, rem / max(1, n_inits_to_try))

        for i in range(n_inits_to_try):
            rem2 = deadline - time.monotonic() - 120
            if rem2 < 10:
                break
            t_alloc = min(t_per_init, rem2)
            a_opt, obj = mw_optimize(inits[i], time_limit=t_alloc,
                                     tau_init=0.005, eta_init=0.9, seed=100 + i * 7)
            try_seq(state, a_opt, f"expl_{label}_i{i}")

    print(f"  Phase 4 best: {state.best_score:.8f}")

    # === Phase 5: Deep refinement on best ===
    print("\n=== Phase 5: Deep refinement ===")
    if state.best_seq is not None:
        n_best = len(state.best_seq)
        f_best = state.best_seq / state.best_seq.sum()

        # Alternating MW and LP-FW with perturbations
        for round_idx in range(4):
            rem = deadline - time.monotonic() - 80
            if rem < 30:
                break
            t_alloc = min(rem / (4 - round_idx), 100)

            if round_idx == 0:
                # Direct MW with fine parameters
                a_opt, obj = mw_optimize(f_best, time_limit=t_alloc,
                                         tau_init=0.001, eta_init=0.4, seed=200,
                                         anneal_rate=1.004, cool_rate=0.9999)
                try_seq(state, a_opt, f"deep_mw_{round_idx}")
            elif round_idx == 1:
                # Perturbed LP-FW
                noise = rng.standard_normal(n_best) * 0.008
                f_pert = project_simplex(f_best + noise)
                a_opt, obj = lp_fw_optimize(f_pert, time_limit=t_alloc, seed=300 + round_idx)
                try_seq(state, a_opt, f"deep_lpfw_{round_idx}")
            elif round_idx == 2:
                # Mirror blend + MW
                f_mirror = 0.5 * f_best + 0.5 * f_best[::-1]
                f_mirror = project_simplex(f_mirror)
                a_opt, obj = mw_optimize(f_mirror, time_limit=t_alloc,
                                         tau_init=0.002, eta_init=0.7, seed=400)
                try_seq(state, a_opt, f"deep_mirror_{round_idx}")
            else:
                # Aggressive perturbation + MW
                noise = rng.standard_normal(n_best) * 0.03
                f_pert = project_simplex(f_best + noise)
                a_opt, obj = mw_optimize(f_pert, time_limit=t_alloc,
                                         tau_init=0.003, eta_init=1.0, seed=500)
                try_seq(state, a_opt, f"deep_perturb_{round_idx}")

            # Update reference
            if state.best_seq is not None:
                f_best = state.best_seq / state.best_seq.sum()

    # === Phase 6: Final fine polish ===
    print("\n=== Phase 6: Final polish ===")
    if state.best_seq is not None:
        # Very fine MW
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, obj = mw_optimize(state.best_seq, time_limit=min(rem, 40),
                                     tau_init=0.0005, eta_init=0.2, seed=600,
                                     anneal_rate=1.005, cool_rate=0.99995)
            try_seq(state, a_opt, "polish_fine")

        # Quick LP-FW
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            a_opt, obj = lp_fw_optimize(state.best_seq, time_limit=min(rem, 20), seed=700)
            try_seq(state, a_opt, "polish_lpfw")

    # Fallback
    if state.best_seq is None:
        state.best_seq = np.ones(256) / 256
        state.best_score = float(evaluate_sequence(state.best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(state.best_seq.tolist()))
    if np.isfinite(fs):
        state.best_score = fs

    print(f"\nFINAL SCORE: {state.best_score:.8f} (n={len(state.best_seq)})")
    return [float(x) for x in state.best_seq]
