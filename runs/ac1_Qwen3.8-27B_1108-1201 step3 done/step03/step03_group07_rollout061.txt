import time
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.fft(a, fft_size)
    return np.real(np.fft.ifft(A * A, fft_size))[:size]

def smooth_max(x, beta):
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -50, 50)
    log_sum = np.log(np.exp(shifted).sum())
    return x_max + log_sum / beta

def smooth_max_grad(x, beta):
    """Returns (smooth_max_value, gradient_weights)"""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w

def project_nonneg_sum(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(len(v)) * target_sum / len(v)
    return v * (target_sum / s)

    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)
    return conv[:size]


def true_objective(a):
    """Compute the true objective 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights with numerical stability."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_s / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_gradient_conv(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] wrt a using FFT."""
    n = len(a)
    fft_size = 1 << ((len(w) + n - 2) + 63).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    g = np.fft.irfft(np.conj(W) * A, fft_size)[:n]
    return 4.0 * n * g

    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)

    rho_max = np.max(np.where(cond)[0])
    theta = (cssv[rho_max] - 1.0) / (rho_max + 1.0)
    return np.maximum(v - theta, 0.0)

    """Objective for L-BFGS-B using softmax parameterization."""
    x_shift = x - x.max()
    a = np.exp(x_shift)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def softmax_grad(x, n, beta):
    """Gradient for L-BFGS-B using softmax parameterization."""
    x_shift = x - x.max()
    a = np.exp(x_shift)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = compute_gradient_conv(a, w)
    lag = np.dot(a, grad_a)
    return a * (grad_a - lag)

    a_opt = a_opt / a_opt.sum()

        # Build constraint matrix A_con where A_con[k,j] = a[k-j] if 0<=k-j<n
        # Constraint: A_con @ b <= M * ones(2n-1)
        # This matrix is (2n-1) x n
        A_con = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    A_con[k, j] = a[idx]

        # Solve: max sum(b) s.t. A_con @ b <= M, b >= 0
        c_obj = -np.ones(n)
        b_cons = np.full(2 * n - 1, M)
        A_ub = A_con
        b_ub = b_cons
        bounds = [(0, None)] * n

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

        if res.status != 0:
            break

        b_opt = res.x
        b_sum = b_opt.sum()
        if b_sum < 1e-15:
            break

        b_norm = b_opt / b_sum

        # Blend
        a_new = (1.0 - blend_t) * a + blend_t * b_norm
        a = a_new
        a = project_simplex(a)

    return a


def peak_reduction_direct(a, n_iters=50):
    """Direct peak reduction: identify argmax and reduce contributing elements."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()

    for it in range(n_iters):
        conv = fast_convolve(a)
        max_k = int(np.argmax(conv))
        max_val = conv[max_k]

        # Find elements contributing to this peak
        # conv[max_k] = sum_i a[i]*a[max_k-i]
        contrib = np.zeros(n)
        for i in range(n):
            j = max_k - i
            if 0 <= j < n:
                contrib[i] += a[i] * a[j]

        # Reduce the largest contributors proportionally
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break

        reduction = 0.02
        for i in range(n):
            if contrib[i] > 0:
                scale = 1.0 - reduction * (contrib[i] / total_contrib) * 2.0
                a[i] *= max(scale, 0.1)

        a = np.maximum(a, 0)
        a = a / a.sum()

    return a


def flat_spectrum_init(n, rng):
    """Initialize with approximately flat power spectrum."""
    fft_size = 1 << (n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = 1.0
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    for k in range(1, fft_size // 2 + 1):
        A[k] = np.exp(1j * phases[k])
        if k < fft_size - k:
            A[fft_size - k] = np.exp(-1j * phases[k])
    a = np.real(np.fft.ifft(A, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # Bimodal
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Flat spectrum
    inits['flat_spec'] = flat_spectrum_init(n, rng)

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Power law (heavy tail)
    pw = 1.0 / (np.arange(1, n + 1) ** 0.5)
    pw /= pw.sum()
    inits['power'] = pw

    # Sine-based (approximate flat autocorrelation)
    sin_seq = np.abs(np.sin(np.pi * x / n))
    sin_seq /= sin_seq.sum()
    inits['sine'] = sin_seq

    # Parent-based
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0)
        ps = pp.sum()
        if ps > 1e-10:
            pp = pp / ps
            if len(pp) == n:
                inits['parent'] = pp
            elif len(pp) > n:
                interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(pp)), pp)
                interp = np.maximum(interp, 0)
                is_ = interp.sum()
                if is_ > 1e-10:
                    inits['parent_ds'] = interp / is_
            else:
                interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(pp)), pp)
                interp = np.maximum(interp, 0)
                is_ = interp.sum()
                if is_ > 1e-10:
                    inits['parent_up'] = interp / is_

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining smooth L-BFGS-B,
    LP-based iterations, direct peak reduction, and random restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    def score_and_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return float('inf')
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            elapsed = time.monotonic() - t_start
            print("  [t=%.0fs] New best: %.6f (n=%d)" % (elapsed, score, len(a)))
        return score if np.isfinite(score) else float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent: len=%d, sum=%.2f" % (len(p), p.sum()))
    except Exception:
        pass

    # === Phase 1: Quick Scan (250s) ===
    print("=== Phase 1: Quick Scan ===")
    phase1_deadline = t_start + 250.0

    lengths = [128, 192, 256, 320, 384, 512]
    candidates = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 20:
            break
        inits = generate_inits(n, parent, rng)
        init_names = list(inits.keys())
        # Priority order
        priority = ['parent', 'parent_ds', 'parent_up', 'flat_spec', 'uniform', 'gauss', 'random', 'bimodal', 'power', 'sine']
        init_names = [nm for nm in priority if nm in inits] + [nm for nm in init_names if nm not in priority]
        init_names = init_names[:5]

        for name in init_names:
            if time.monotonic() > phase1_deadline - 10:
                break
            a0 = inits[name]
            # Quick PG with moderate beta
            a_opt, _ = pg_smooth(a0, beta=30.0, lr=0.05, n_iter=300)
            score = true_objective(a_opt)
            if np.isfinite(score):
                candidates.append((score, n, name, a_opt.copy()))

    candidates.sort(key=lambda x: x[0])
    print("Phase 1 done. Best internal: %.6f (n=%d, init=%s)" %
          (candidates[0][0], candidates[0][1], candidates[0][2]) if candidates else "No candidates")

    # Evaluate top candidates with the real evaluator
    for score_int, n, name, a_cand in candidates[:5]:
        score_and_update(a_cand)

    if best_sequence is not None:
        print("Phase 1 best evaluated: %.6f" % best_score)

    # === Phase 2: L-BFGS-B Deep Optimization (500s) ===
    print("=== Phase 2: L-BFGS-B Deep Optimization ===")
    phase2_deadline = t_start + 750.0

    # Determine which lengths to focus on
    if best_sequence is not None:
        best_n = len(best_sequence)
    else:
        best_n = 256

    target_lengths = sorted(set([l for l in [best_n, best_n * 3 // 4, best_n * 5 // 4, best_n * 2,
                                             128, 192, 256, 320, 384, 512]
                                 if 64 <= l <= 1024]))
    # Limit to at most 5 lengths
    target_lengths = target_lengths[:5]

    # Beta annealing schedule
    beta_schedule = [20, 50, 100, 200, 400, 800, 1500]

    for n in target_lengths:
        if time.monotonic() > phase2_deadline - 30:
            break

        inits = generate_inits(n, parent, rng)

        # Add best sequence as initialization
        if best_sequence is not None:
            if len(best_sequence) == n:
                inits['best'] = best_sequence / best_sequence.sum()
            else:
                interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(best_sequence)),
                                   best_sequence)
                interp = np.maximum(interp, 0)
                is_ = interp.sum()
                if is_ > 1e-10:
                    inits['best_interp'] = interp / is_

        # Also add candidates from phase 1 of this length
        for score_int, cn, name, ac in candidates:
            if cn == n:
                inits['cand_%s' % name] = ac
                break

        init_names = list(inits.keys())
        priority2 = ['best', 'best_interp', 'parent', 'parent_ds', 'parent_up',
                     'cand_flat_spec', 'cand_uniform', 'cand_gauss',
                     'flat_spec', 'uniform', 'gauss']
        init_names = [nm for nm in priority2 if nm in inits] + \
                     [nm for nm in init_names if nm not in priority2]
        init_names = init_names[:3]

        for name in init_names:
            if time.monotonic() > phase2_deadline - 20:
                break

            a0 = inits[name]
            a_cur = a0.copy()

            # Beta annealing
            time_per_beta = max(5.0, (phase2_deadline - time.monotonic() - 20) / len(beta_schedule))
            for beta in beta_schedule:
                if time.monotonic() > phase2_deadline - 10:
                    break
                t_alloc = min(time_per_beta, phase2_deadline - time.monotonic() - 10)
                if t_alloc < 3:
                    break
                maxiter = int(min(500, max(50, t_alloc * 40)))
                a_opt, _ = optimize_lbfgsb(a_cur, beta=beta, maxiter=maxiter)
                score_and_update(a_opt)
                a_cur = a_opt

    print("Phase 2 done. Best: %.6f" % best_score)

    # === Phase 3: LP-based iterations (400s) ===
    print("=== Phase 3: LP-based Iterations ===")
    phase3_deadline = t_start + 1150.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        a_lp = best_sequence / best_sequence.sum()

        # Multiple LP rounds with re-optimization
        for round_num in range(6):
            if time.monotonic() > phase3_deadline - 30:
                break

            # LP step
            t_lp_start = time.monotonic()
            a_lp = lp_step(a_lp, n_steps=20, blend_t=0.15)
            t_lp = time.monotonic() - t_lp_start
            score_lp = true_objective(a_lp)
            print("  LP round %d: obj=%.6f (took %.1fs)" % (round_num + 1, score_lp, t_lp))
            score_and_update(a_lp)

            # Re-optimize with L-BFGS-B
            if time.monotonic() < phase3_deadline - 20:
                t_alloc = min(60.0, phase3_deadline - time.monotonic() - 20)
                if t_alloc > 10:
                    maxiter = int(min(400, max(50, t_alloc * 30)))
                    a_opt, _ = optimize_lbfgsb(a_lp, beta=200.0, maxiter=maxiter)
                    score_and_update(a_opt)
                    a_lp = a_opt

            # Try perturbed LP
            if time.monotonic() < phase3_deadline - 50:
                noise = rng.standard_normal(n_best) * 0.01
                a_pert = project_simplex(a_lp + noise)
                a_pert = lp_step(a_pert, n_steps=10, blend_t=0.2)
                score_and_update(a_pert)
                if true_objective(a_pert) < true_objective(a_lp):
                    a_lp = a_pert

    print("Phase 3 done. Best: %.6f" % best_score)

    # === Phase 4: Final Refinement (400s) ===
    print("=== Phase 4: Final Refinement ===")
    phase4_deadline = deadline - 50.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        a_ref = best_sequence / best_sequence.sum()

        # Very high beta L-BFGS-B
        for beta in [2000, 3000, 5000]:
            if time.monotonic() > phase4_deadline - 20:
                break
            t_alloc = min(60.0, phase4_deadline - time.monotonic() - 20)
            if t_alloc < 5:
                break
            maxiter = int(min(500, max(50, t_alloc * 35)))
            a_opt, _ = optimize_lbfgsb(a_ref, beta=beta, maxiter=maxiter)
            score_and_update(a_opt)
            a_ref = a_opt

        # Direct peak reduction
        if time.monotonic() < phase4_deadline - 10:
            a_peak = peak_reduction_direct(a_ref, n_iters=100)
            score_and_update(a_peak)
            if true_objective(a_peak) < true_objective(a_ref):
                a_ref = a_peak
                # Re-optimize from here
                maxiter = int(min(300, max(50, (phase4_deadline - time.monotonic() - 10) * 25)))
                a_opt, _ = optimize_lbfgsb(a_ref, beta=1000.0, maxiter=maxiter)
                score_and_update(a_opt)
                a_ref = a_opt

        # Random perturbations + re-optimize (final kicks)
        n_kicks = 0
        while time.monotonic() < phase4_deadline - 30 and n_kicks < 15:
            n_kicks += 1
            scale = 0.002 + 0.003 * rng.uniform()
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(a_ref + noise)
            t_alloc = min(40.0, (phase4_deadline - time.monotonic() - 30) / max(1, (15 - n_kicks)))
            if t_alloc < 5:
                break
            maxiter = int(min(300, max(30, t_alloc * 25)))
            a_opt, _ = optimize_lbfgsb(a_pert, beta=500.0, maxiter=maxiter)
            score_and_update(a_opt)
            if true_objective(a_opt) < true_objective(a_ref):
                a_ref = a_opt

        # Final PG polish
        if time.monotonic() < phase4_deadline - 5:
            n_iters = int(min(2000, (phase4_deadline - time.monotonic() - 5) * 500))
            a_opt, _ = pg_smooth(a_ref, beta=2000.0, lr=0.01, n_iter=n_iters)
            score_and_update(a_opt)

    print("Phase 4 done. Best: %.6f" % best_score)

    # === Phase 5: Try alternative lengths (last chance) ===
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 60:
            n_current = len(best_sequence)
            alt_n = 384 if n_current <= 256 else 256
            if alt_n != n_current:
                print("=== Phase 5: Trying n=%d ===" % alt_n)
                inits_alt = generate_inits(alt_n, parent, rng)
                interp = np.interp(np.linspace(0, 1, alt_n), np.linspace(0, 1, n_current), best_sequence)
                interp = np.maximum(interp, 0)
                is_ = interp.sum()
                if is_ > 1e-10:
                    a_alt = interp / is_
                    maxiter = int(min(500, max(100, remaining * 3)))
                    a_opt, _ = optimize_lbfgsb(a_alt, beta=300.0, maxiter=maxiter)
                    score_and_update(a_opt)

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    # Ensure all values are non-negative and finite
    best_sequence = np.maximum(best_sequence, 0.0)
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL: score=%.6f, n=%d, elapsed=%.0fs" %
          (best_score, len(best_sequence), time.monotonic() - t_start))
    return [float(x) for x in best_sequence]
