"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
LP-based directional optimization, simulated annealing with peak-targeted moves,
and multi-scale coordinate descent. Explores the parent sequence and nearby lengths.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_objective(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def build_lp_matrix(n):
    """Build the LP constraint matrix for max sum(g) s.t. f*g <= M, g>=0."""
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = 1.0  # Will be multiplied by f later via row scaling
    return A_ub


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 30})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize(a_init, time_limit, blend_init=0.3, blend_min=0.005,
                blend_halve_every=10):
    """LP-based directional optimization."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    uniform = np.ones(n) / n

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    blend = blend_init
    stall = 0

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break

        g = solve_lp_direction(a, M, n)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Blend with uniform if LP solution is too peaked
        if g.max() > 0.4:
            g = 0.7 * g + 0.3 * uniform

        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall >= blend_halve_every:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend <= blend_min * 1.01:
                break

        if stall > blend_halve_every * 4:
            rng_local = np.random.default_rng(int(time.monotonic() * 1e6) % 2**31)
            a = project_simplex(a + rng_local.standard_normal(n) * 0.02)
            conv = fast_convolve(a)
            stall = 0

    return best_a, best_obj


def simulated_annealing(a_init, time_limit, T_init=0.05, T_min=0.0001,
                        cooling=0.997, rng=None):
    """
    Simulated annealing with peak-targeted moves.
    Moves: (1) random pair transfer, (2) peak-reducing transfer, (3) block shift.
    """
    if rng is None:
        rng = np.random.default_rng()
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    T = T_init
    current_obj = best_obj
    no_improve = 0

    # Precompute which coordinates contribute to which peaks
    while time.monotonic() - t0 < time_limit and T > T_min:
        # Choose move type
        move_type = rng.integers(0, 3)

        if move_type == 0:
            # Random pair transfer
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            frac = rng.uniform(0.01, 0.15)
            delta = a[i] * frac
            if delta < 1e-10:
                continue
            a[i] -= delta
            a[j] += delta
        elif move_type == 1:
            # Peak-targeted: reduce contribution to the worst peak
            peak_k = int(np.argmax(conv))
            j_start = max(0, peak_k - n + 1)
            j_end = min(n - 1, peak_k)
            # Find the coordinate with largest contribution to this peak
            contributions = np.array([a[j] * a[peak_k - j] for j in range(j_start, j_end + 1)])
            if contributions.sum() < 1e-15:
                continue
            # Reduce the largest contributor
            worst_local = j_start + int(np.argmax(contributions))
            if a[worst_local] < 1e-8:
                continue
            # Move to a random coordinate
            j = rng.integers(0, n)
            if j == worst_local:
                j = (j + 1) % n
            delta = a[worst_local] * rng.uniform(0.02, 0.1)
            a[worst_local] -= delta
            a[j] += delta
        else:
            # Block shift: shift a small block of mass
            block_size = max(2, n // 50)
            start = rng.integers(0, max(1, n - block_size))
            block = a[start:start + block_size].copy()
            if block.sum() < 1e-10:
                continue
            # Find target position
            target = rng.integers(0, max(1, n - block_size))
            if target == start:
                target = (target + n // 2) % max(1, n - block_size)
            frac = rng.uniform(0.1, 0.5)
            for d in range(block_size):
                delta = block[d] * frac
                a[start + d] -= delta
                a[target + d] += delta

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            conv = fast_convolve(a)
            current_obj = best_obj
            continue
        a = a / s
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        # Accept or reject
        delta_obj = cur_obj - current_obj
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / T):
            current_obj = cur_obj
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Revert
            a = best_a.copy() if no_improve > 500 else a
            # Actually we need to revert properly - let's just keep going
            no_improve += 1

        if no_improve > 1000:
            # Perturb and reset counter
            a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
            conv = fast_convolve(a)
            current_obj = 2.0 * n * conv.max()
            no_improve = 0

        T *= cooling

    return best_a, best_obj


def coordinate_descent_v2(a_init, max_rounds=50, deltas=(0.02, 0.05, 0.1, 0.2, 0.4)):
    """
    Multi-scale coordinate descent: try moving multiple fractions of mass
    from each source to the best target.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()

    for rnd in range(max_rounds):
        improved = False
        # Sort coordinates by value (process larger ones first)
        order = np.argsort(a)[::-1]
        for i in order:
            if a[i] < 1e-7:
                continue
            for delta_frac in deltas:
                delta = a[i] * delta_frac
                if delta < 1e-9:
                    continue
                a[i] -= delta
                if a[i] < 0:
                    a[i] = 0
                    delta = 0
                    continue
                # Find best target: try all but use a faster heuristic
                # Compute new convolution only at the peak
                best_target = -1
                best_target_obj = best_obj
                # Sample targets for speed
                if n > 200:
                    candidates = np.random.default_rng(rnd * n + i).choice(n, size=min(50, n), replace=False)
                else:
                    candidates = np.arange(n)

                for j in candidates:
                    if j == i:
                        continue
                    a[j] += delta
                    conv_test = fast_convolve(a)
                    obj_test = 2.0 * n * conv_test.max()
                    a[j] -= delta
                    if obj_test < best_target_obj:
                        best_target_obj = obj_test
                        best_target = j

                if best_target >= 0 and best_target_obj < best_obj - 1e-14:
                    a[best_target] += delta
                    conv = fast_convolve(a)
                    best_obj = 2.0 * n * conv.max()
                    improved = True
                    break  # Move to next source coordinate

        if not improved:
            break

    return a, best_obj


def subgradient_polish(a_init, time_limit, lr=0.005, momentum=0.92, top_k=30):
    """Subgradient descent with momentum for final polishing."""
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break

        # Soft-max weighting of peaks
        temp = pmax * 0.03 + 1e-8
        w = np.exp((peak_vals - pmax) / temp)
        w /= w.sum()

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = 2.0 * w[idx]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            a = project_simplex(a + np.random.default_rng().standard_normal(n) * 0.003)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            lr *= 0.8

        lr *= 0.9995

    return best_a, best_obj


def gen_initializations(n, parent=None, rng=None):

    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    inits['hann'] = hann / hann.sum()

    # Raised cosine
    rc = 0.5 * (1 - np.cos(2 * np.pi * x / n))
    inits['raised_cos'] = rc / rc.sum()

    # Triangle
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    # Edge concentrated
    edge = np.zeros(n)
    e = max(1, n // 5)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Bimodal
    g2 = (np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2))
    inits['bimodal'] = g2 / g2.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Optimal non-negative (water-filling inspired): proportional to 1/sqrt(k+1) near center
    dist_from_center = np.abs(x - (n - 1) / 2.0)
    wf = 1.0 / (1.0 + dist_from_center / (n * 0.2))
    inits['waterfill'] = wf / wf.sum()

    if parent is not None:
        p = np.maximum(parent, 0.0)
        if len(p) == n:
            ps = p.sum()
            if ps > 1e-10:
                inits['parent'] = p / ps
        elif len(p) > n:
            p_res = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
            ps = p_res.sum()
            if ps > 1e-10:
                inits['parent_ds'] = p_res / ps
        else:
            p_res = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
            ps = p_res.sum()
            if ps > 1e-10:
                inits['parent_up'] = p_res / ps

    return inits

            if ln not in lengths and 64 <= ln <= 2048:

    else:
        lengths = [256, 384, 512]

    results = []
    for n in lengths:
        if time.monotonic() > t1_end - 30:
            break
        inits = gen_initializations(n, parent, rng)
        # Priority: parent first, then structured inits
        prio = ['parent', 'parent_ds', 'parent_up', 'hann', 'raised_cos',
                'waterfill', 'triangle', 'gauss', 'bimodal', 'edge', 'uniform', 'random']
        names = [nm for nm in prio if nm in inits][:4]

        t_per = max(20.0, (t1_end - time.monotonic() - 10) / max(1, (len(lengths) - lengths.index(n)) * len(names)))

        for name in names:
            if time.monotonic() > t1_end - 10:
                break
            t_alloc = min(t_per, t1_end - time.monotonic() - 5)
            if t_alloc < 10:
                break

            a0 = inits[name]
            a_opt, obj_opt = lp_optimize(a0, t_alloc, blend_init=0.35,
                                         blend_min=0.008, blend_halve_every=12)
            results.append((obj_opt, n, name, a_opt.copy()))
            try_sequence(a_opt, f"LP n={n} {name}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # === Phase 2: SA + subgradient on top candidates ===
    print("=== Phase 2: SA + Subgradient ===")
    t2_end = time.monotonic() + min(500.0, (deadline - time.monotonic()) * 0.35)

    top_candidates = []
    seen = set()
    for obj_opt, n, name, a_opt in results[:6]:
        key = (n, name)
        if key not in seen:
            top_candidates.append((n, a_opt))
            seen.add(key)
    if best_sequence is not None:
        top_candidates.append((len(best_sequence), best_sequence.copy()))

    remaining_p2 = t2_end - time.monotonic()
    t_per_cand = max(20.0, remaining_p2 / max(1, len(top_candidates)))

    for n, a0 in top_candidates:
        if time.monotonic() > t2_end - 15:
            break
        t_alloc = min(t_per_cand, t2_end - time.monotonic() - 10)
        if t_alloc < 10:
            break

        # SA
        t_sa = t_alloc * 0.5
        a_sa, obj_sa = simulated_annealing(a0, t_sa, T_init=0.03, T_min=0.00005,
                                           cooling=0.997, rng=rng)
        try_sequence(a_sa, f"SA n={n}")

        # Subgradient polish
        t_sg = min(t_alloc * 0.3, 30.0)
        a_sg, obj_sg = subgradient_polish(a_sa, t_sg, lr=0.004, momentum=0.92, top_k=40)
        try_sequence(a_sg, f"SG n={n}")

    print(f"Phase 2 best: {best_score:.6f}")

    # === Phase 3: Deep LP + SA cycles on best ===
    print("=== Phase 3: Deep Optimization ===")
    t3_end = deadline - 200.0

    if best_sequence is not None:
        current_best = best_sequence.copy()
        for cycle in range(5):
            if time.monotonic() > t3_end:
                break
            t_cyc = min(60.0, (t3_end - time.monotonic()) / max(1, 5 - cycle))
            if t_cyc < 10:
                break

            # LP
            t_lp = t_cyc * 0.4
            a_lp, _ = lp_optimize(current_best, t_lp, blend_init=0.15,
                                   blend_min=0.003, blend_halve_every=12)
            try_sequence(a_lp, f"Deep LP cycle {cycle}")

            # SA
            t_sa = t_cyc * 0.3
            a_sa, _ = simulated_annealing(a_lp, t_sa, T_init=0.01, T_min=0.00002,
                                          cooling=0.998, rng=rng)
            try_sequence(a_sa, f"Deep SA cycle {cycle}")

            # Subgradient
            t_sg = t_cyc * 0.2
            a_sg, _ = subgradient_polish(a_sa, t_sg, lr=0.002, momentum=0.93, top_k=50)
            try_sequence(a_sg, f"Deep SG cycle {cycle}")

            current_best = best_sequence.copy()

    # === Phase 4: Coordinate descent polish ===
    print("=== Phase 4: Coordinate Descent ===")
    if best_sequence is not None:
        t4_end = deadline - 80.0
        t_cd = min(150.0, t4_end - time.monotonic())
        if t_cd > 5:
            a_cd, obj_cd = coordinate_descent_v2(best_sequence, max_rounds=80,
                                                  deltas=(0.01, 0.03, 0.05, 0.1, 0.2, 0.3))
            try_sequence(a_cd, "CD")

            # Another round of CD with the result
            if time.monotonic() < t4_end - 20:
                a_cd2, obj_cd2 = coordinate_descent_v2(best_sequence, max_rounds=40,
                                                        deltas=(0.01, 0.02, 0.05, 0.1, 0.2))
                try_sequence(a_cd2, "CD2")

    # === Phase 5: Final fine-tuning ===
    print("=== Phase 5: Final Polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 20:
        t_final = min(60.0, deadline - time.monotonic() - 15)
        if t_final > 5:
            # Very fine subgradient
            a_f, _ = subgradient_polish(best_sequence, t_final * 0.5,
                                         lr=0.001, momentum=0.94, top_k=60)
            try_sequence(a_f, "Final SG")

            # Tiny SA
            if time.monotonic() < deadline - 15:
                a_fs, _ = simulated_annealing(a_f, min(20.0, deadline - time.monotonic() - 12),
                                              T_init=0.005, T_min=0.00001,
                                              cooling=0.999, rng=rng)
                try_sequence(a_fs, "Final SA")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
