    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_lp(a):
    """Build LP: min t s.t. (a*b)[k] <= t for all k, sum(b)=1, b>=0.
    Variables: [b_0,...,b_{n-1}, t]"""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n + 1))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            A_ub[k, k - i] = a[i]
        A_ub[k, n] = -1.0
    b_ub = np.zeros(num_k)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c = np.zeros(n + 1)
    c[n] = 1.0
    bounds = [(0, None)] * n + [(0, None)]
    return c, A_ub, b_ub, A_eq, b_eq, bounds


def solve_response_lp(a, time_limit=10.0):
    """Solve response LP. Returns (b, t_opt) or (None, None)."""
    n = len(a)
    c, A_ub, b_ub, A_eq, b_eq, bounds = build_lp(a)
    t0 = time.monotonic()
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success:
            b = np.maximum(res.x[:n], 0.0)
            t_val = res.x[n]
            return b, t_val
    except Exception:
        pass
    return None, None


def alternating_lp(a0, time_limit, step_init=0.6, verbose=False, seed=0):
    """Alternating LP optimization on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    iter_count = 0
    lp_fail_count = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break

        lp_time = min(8.0, remaining * 0.7)
        b, t_lp = solve_response_lp(a, time_limit=lp_time)

        if b is None:
            lp_fail_count += 1
            if lp_fail_count > 5:
                break
            # Fallback: use greedy direction
            conv_curr = fast_convolve(a)
            M = conv_curr.max()
            b = np.ones(n) / n
            for k in range(2 * n - 1):
                for i in range(max(0, k - n + 1), min(n, k + 1)):
                    idx = k - i
                    if 0 <= idx < n:
                        b[idx] += a[i] * 0.001
            b = np.maximum(b, 0.0)
            bs = b.sum()
            if bs > 1e-15:
                b = b / bs
            else:
                break

        b = np.maximum(b, 0.0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum

        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            step = min(1.0, step * 1.01)
        else:
            no_improve += 1
            step *= 0.92
            if no_improve > 40:
                noise = rng.standard_normal(n) * 0.003
                a = project_simplex(a + noise)
                step = step_init * 0.7
                no_improve = 0
                max_step = 1.0
        iter_count += 1

        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ALP iter {iter_count}: obj={best_obj:.8f}, step={step:.3f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    return best_a, 2.0 * n * conv_final.max()


def smooth_gradient(a, beta=500.0):
    """Compute smooth gradient of max(conv(a,a)) using softmax."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    size = 2 * n
    fft_size = 1 << (size - 1).bit_length()
    w_padded = np.zeros(fft_size)
    w_padded[:len(w)] = w
    a_padded = np.zeros(fft_size)
    a_padded[:n] = a

    W = np.fft.fft(w_padded)
    A = np.fft.fft(a_padded)
    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    conv_wa = np.fft.ifft(A * W).real[:n]

    grad = cross + conv_wa
    return grad, conv


def gradient_optimize(a0, time_limit, beta_init=500.0, lr_init=0.03,
                      momentum=0.9, verbose=False, seed=0):
    """Smooth gradient descent with momentum on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        grad, conv_cur = smooth_gradient(a, beta)
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.85
                v *= 0.5
            if no_improve > 300:
                noise = rng.standard_normal(n) * 0.005
                a = project_simplex(a + noise)
                v = np.zeros(n)
                lr = lr_init * 0.5
                beta = max(100, beta_init * 0.5)
                no_improve = 0

        beta = min(8000.0, beta * 1.002)

    conv_final = fast_convolve(best_a)
    return best_a, 2.0 * n * conv_final.max()


def hybrid_optimize(a0, time_limit, seed=0, verbose=False):
    """Alternate between LP and gradient steps."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    step = 0.5
    mode = 0  # 0=LP, 1=grad
    no_improve = 0
    beta = 800.0
    lr = 0.02

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        if mode == 0:
            lp_t = min(8.0, remaining * 0.5)
            b, t_lp = solve_response_lp(a, time_limit=lp_t)
            if b is not None:
                b = np.maximum(b, 0.0)
                bs = b.sum()
                if bs > 1e-15:
                    b = b / bs
                    a_new = (1.0 - step) * a + step * b
                    a_new = np.maximum(a_new, 0.0)
                    s2 = a_new.sum()
                    if s2 > 1e-15:
                        a_new = a_new / s2
                else:
                    a_new = a.copy()
            else:
                a_new = a.copy()
            mode = 1
        else:
            grad, _ = smooth_gradient(a, beta)
            gn = np.linalg.norm(grad)
            if gn > 1e-15:
                grad = grad / gn
            v = 0.9 * v - lr * grad
            a_new = project_simplex(a + v)
            mode = 0

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if mode == 0:
                step *= 0.95
            else:
                lr *= 0.95
            if no_improve > 60:
                noise = rng.standard_normal(n) * 0.003
                a = project_simplex(a + noise)
                v = np.zeros(n)
                step = 0.5
                lr = 0.02
                no_improve = 0

        beta = min(10000.0, beta * 1.001)

    conv_final = fast_convolve(best_a)
    return best_a, 2.0 * n * conv_final.max()


def generate_u_shaped(n, sharpness=2.0):
    """Generate U-shaped initialization (high at edges, low in middle)."""
    x = np.linspace(0, 1, n)
    # U-shape: high at 0 and 1, low in middle
    u = np.abs(x - 0.5) ** sharpness
    u = u + 0.1 * np.ones(n)  # avoid zeros
    u /= u.sum()
    return u


def generate_flat_conv_init(n):
    """Generate initialization based on flat-convolution recursion (left side)."""
    c = 1.0 / (2 * n - 1)
    a = np.zeros(n)
    a[0] = np.sqrt(c)
    for k in range(1, n):
        # conv[k] = 2*a[0]*a[k] + sum_{i=1}^{k-1} a[i]*a[k-i] = c
        # (assuming k < n so we don't need the right boundary yet)
        s = 0.0
        for i in range(1, min(k, n)):
            if k - i < n:
                s += a[i] * a[k - i]
        val = (c - s) / (2.0 * a[0])
        a[k] = max(0.0, val)
    # Mirror to make symmetric
    a = np.maximum(a, a[::-1])
    a_sum = a.sum()
    if a_sum > 1e-15:
        a = a / a_sum
    else:
        a = np.ones(n) / n
    return a


def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.linspace(0, 1, n)

    inits['uniform'] = np.ones(n) / n

    # U-shaped variants
    for sp in [1.0, 1.5, 2.0, 3.0, 4.0]:
        u = np.abs(x - 0.5) ** sp + 0.05
        u /= u.sum()
        inits[f'u_{sp}'] = u

    # Flat-convolution based
    try:
        inits['flat_conv'] = generate_flat_conv_init(n)
    except Exception:
        pass

    # Bimodal (step function at edges)
    edge = max(1, n // 6)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Ramps
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()
    inits['ramp_dec'] = ramp[::-1] / ramp.sum()

    # Cosine U-shape
    cos_u = 0.5 * (1 + np.cos(2 * np.pi * x))
    cos_u = np.maximum(cos_u, 0.01)
    cos_u /= cos_u.sum()
    inits['cos_u'] = cos_u

    # Random
    inits['rand_sparse'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand_uniform'] = rng.dirichlet(np.ones(n) * 1.0)

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_crop'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_down'] = pp / ps

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: (1) scan lengths/inits, (2) deep alternating LP,
    (3) hybrid LP+grad, (4) gradient polish, (5) final perturbations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.2f}")
    except NameError:
        pass
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_sequence, best_score
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  {label}: score={score:.8f} (n={len(a)})")
            return True
        return False

    # ========== Phase 1: Quick Scan ==========
    print("=== Phase 1: Quick Scan ===")
    t1 = time.monotonic()
    phase1_time = 180.0

    lengths = [192, 256, 320, 384, 448, 512, 640]
    scan_results = []

    for n in lengths:
        if time.monotonic() - t1 > phase1_time - 20:
            break
        inits = generate_initializations(n, rng, parent)
        # Prioritize promising initializations
        priority = ['flat_conv', 'parent_crop', 'parent_down', 'u_2.0', 'u_3.0',
                    'bimodal', 'cos_u', 'rand_sparse', 'uniform']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in init_names]
        init_names = init_names[:6]

        time_per = max(8.0, (phase1_time - (time.monotonic() - t1) - 10) /
                       max(1, len(lengths) - lengths.index(n)) / max(1, len(init_names)))

        for name in init_names:
            if time.monotonic() - t1 > phase1_time - 5:
                break
            t_alloc = min(time_per, phase1_time - (time.monotonic() - t1) - 3)
            if t_alloc < 5:
                break
            a0 = inits[name]
            a_opt, obj_opt = alternating_lp(a0, time_limit=t_alloc,
                                            step_init=0.7, seed=n + 7)
            scan_results.append((obj_opt, n, name, a_opt.copy()))
            try_update(a_opt, f"Scan n={n} {name}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done. Best: {best_score:.8f}")

    # ========== Phase 2: Deep Alternating LP ==========
    print("=== Phase 2: Deep LP ===")
    t2 = time.monotonic()
    phase2_time = 500.0

    # Best length from scan
    if best_sequence is not None:
        best_n = len(best_sequence)
    else:
        best_n = 320

    # Collect starting points
    deep_starts = []
    seen = set()
    for obj, n, name, a in scan_results[:6]:
        key = (n, name)
        if key not in seen:
            deep_starts.append((n, a.copy()))
            seen.add(key)
    if best_sequence is not None:
        deep_starts.append((best_n, best_sequence.copy()))

    # Add fresh starts at best_n
    inits_best = generate_initializations(best_n, rng, parent)
    for extra in ['u_2.0', 'u_3.0', 'flat_conv', 'rand_sparse']:
        if extra in inits_best:
            deep_starts.append((best_n, inits_best[extra].copy()))

    # Also try at a different length
    alt_n = best_n + 64 if best_n < 500 else best_n - 64
    alt_n = max(128, min(768, alt_n))
    if alt_n != best_n:
        inits_alt = generate_initializations(alt_n, rng, parent)
        for extra in ['u_2.0', 'bimodal']:
            if extra in inits_alt:
                deep_starts.append((alt_n, inits_alt[extra].copy()))

    num_runs = len(deep_starts)
    time_per_run = max(15.0, phase2_time / max(1, num_runs))

    for i, (n_s, a_s) in enumerate(deep_starts):
        if time.monotonic() - t2 > phase2_time - 15:
            break
        t_alloc = min(time_per_run, phase2_time - (time.monotonic() - t2) - 10)
        if t_alloc < 10:
            break

        a_opt, obj_opt = alternating_lp(a_s, time_limit=t_alloc,
                                        step_init=0.6, seed=i * 13 + 1)
        if i % 3 == 0:
            print(f"  Deep LP [{i}] n={n_s}: obj={obj_opt:.8f}")
        try_update(a_opt, f"DeepLP[{i}] n={n_s}")

    print(f"Phase 2 done. Best: {best_score:.8f}")

    # ========== Phase 3: Hybrid LP+Gradient ==========
    print("=== Phase 3: Hybrid ===")
    t3 = time.monotonic()
    phase3_time = 350.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        hybrid_starts = [best_sequence.copy()]
        # Add a few perturbed versions
        for i in range(2):
            noise = rng.standard_normal(n_best) * 0.01
            hp = project_simplex(best_sequence + noise)
            hybrid_starts.append(hp)

        time_per_hybrid = phase3_time / max(1, len(hybrid_starts))

        for i, a_h in enumerate(hybrid_starts):
            if time.monotonic() - t3 > phase3_time - 20:
                break
            t_alloc = min(time_per_hybrid, phase3_time - (time.monotonic() - t3) - 10)
            if t_alloc < 15:
                break
            a_opt, obj_opt = hybrid_optimize(a_h, time_limit=t_alloc, seed=i * 21 + 3)
            if i % 2 == 0:
                print(f"  Hybrid [{i}]: obj={obj_opt:.8f}")
            try_update(a_opt, f"Hybrid[{i}]")

    print(f"Phase 3 done. Best: {best_score:.8f}")

    # ========== Phase 4: Gradient Polish ==========
    print("=== Phase 4: Gradient Polish ===")
    t4 = time.monotonic()
    phase4_time = deadline - time.monotonic() - 60.0
    phase4_time = max(30.0, min(200.0, phase4_time))

    if best_sequence is not None:
        n_best = len(best_sequence)
        grad_configs = [
            {'beta_init': 500, 'lr_init': 0.02, 'momentum': 0.9},
            {'beta_init': 1500, 'lr_init': 0.01, 'momentum': 0.93},
            {'beta_init': 3000, 'lr_init': 0.005, 'momentum': 0.95},
            {'beta_init': 200, 'lr_init': 0.04, 'momentum': 0.85},
        ]
        time_per_grad = phase4_time / (len(grad_configs) + 3)

        for i, cfg in enumerate(grad_configs):
            if time.monotonic() - t4 > phase4_time - 10:
                break
            t_alloc = min(time_per_grad, phase4_time - (time.monotonic() - t4) - 5)
            if t_alloc < 8:
                break
            a_opt, obj_opt = gradient_optimize(best_sequence, time_limit=t_alloc,
                                               seed=i * 31 + 7, **cfg)
            try_update(a_opt, f"Grad[{i}]")

        # Short LP runs as final polish
        for i in range(3):
            if time.monotonic() - t4 > phase4_time - 5:
                break
            t_alloc = min(time_per_grad * 0.8, phase4_time - (time.monotonic() - t4) - 3)
            if t_alloc < 5:
                break
            a_opt, obj_opt = alternating_lp(best_sequence, time_limit=t_alloc,
                                            step_init=0.3, seed=100 + i)
            try_update(a_opt, f"PolishLP[{i}]")

    # ========== Phase 5: Tiny Perturbations ==========
    remaining = deadline - time.monotonic() - 15.0
    if remaining > 10 and best_sequence is not None:
        n_best = len(best_sequence)
        n_tries = int(remaining / 8)
        n_tries = min(n_tries, 15)
        for i in range(n_tries):
            if time.monotonic() > deadline - 20:
                break
            for scale in [0.0005, 0.001, 0.003, 0.01]:
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(best_sequence + noise)
                score_pert = float(evaluate_sequence(a_pert.tolist()))
                if np.isfinite(score_pert) and score_pert < best_score:
                    best_score = score_pert
                    best_sequence = a_pert.copy()
                    print(f"  Perturb[{i}] s={scale}: {score_pert:.8f}")
                    break

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
