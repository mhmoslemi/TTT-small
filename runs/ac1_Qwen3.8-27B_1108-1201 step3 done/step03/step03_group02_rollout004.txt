"""
Multi-phase optimizer for minimum peak auto-correlation sequences.
Combines LP-based block coordinate descent, smoothed gradient descent with
annealing, and multi-length exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key innovations: adaptive temperature schedule for softmax smoothing,
Barzilai-Borwein step sizes, sparse LP for large n, and strategic
multi-length search with interpolated initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
from scipy.sparse.linalg import spsolve


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_toeplitz_sparse(a):
    """Build sparse Toeplitz matrix T where (T @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    num_rows = 2 * n - 1
    T = lil_matrix((num_rows, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, num_rows)
        T[k_start:k_end, j] = a[:k_end - k_start]
    return T.tocsr()


def lp_maximize_sum(a, M):
    """Solve: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    T = build_toeplitz_sparse(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def lp_minimize_peak(a):
    """Solve: min M s.t. conv(a,b)[k] <= M, sum(b)=1, b >= 0."""
    n = len(a)
    T = build_toeplitz_sparse(a)
    num_rows = 2 * n - 1
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = lil_matrix((num_rows, n + 1))
    A_ub[:, :n] = T
    for k in range(num_rows):
        A_ub[k, n] = -1.0
    A_ub = A_ub.tocsr()
    b_ub = np.zeros(num_rows)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None


def smoothed_obj_grad(a, t):
    """Compute softmax-smoothed objective and its gradient."""
    n = len(a)
    c = fast_convolve(a)
    scaled = t * c
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp
    obj = (2.0 * n / t) * np.log(sum_exp)
    # Gradient: grad[j] = (2n/t) * sum_{m} p[j+m] * a[m]
    # This is a cross-correlation, computable via FFT
    a_flip = a[::-1]
    fft_size = 1 << (3 * n - 1).bit_length()
    A_f = np.fft.rfft(a_flip, fft_size)
    P_f = np.fft.rfft(p, fft_size)
    conv_full = np.fft.irfft(A_f * P_f, fft_size)[:3 * n - 1]
    grad_base = conv_full[n - 1:n - 1 + n]
    grad = (2.0 * n / t) * grad_base
    return obj, grad

            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[i] * a[k - idx]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9998
        temp *= 0.999

        if no_improve > 500:
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            lr = max(lr_init * 0.3, lr * 0.7)
            no_improve = 0

    return best_a, best_obj


def interpolate_sequence(x, n_target):
    """Interpolate a sequence to a target length using linear interpolation."""
    n_src = len(x)
    if n_src == n_target:
        return x.copy()
    x_pos = np.linspace(0, 1, n_src)
    x_target = np.linspace(0, 1, n_target)
    result = np.interp(x_target, x_pos, x)
    return np.maximum(result, 0)


def generate_initializations(n, parent, seed):
    """Generate diverse initializations of length n."""
    rng = np.random.default_rng(seed)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        pp = interpolate_sequence(parent, n)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    # Cosine windows
    for freq in [1, 2]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0.0)
        sa = a.sum()
        if sa > 1e-10:
            a /= sa
            inits['cos_%d' % freq] = a

    # Fejer-like kernel
    a = np.sinc(x / 2.0 - 0.5)
    a = np.maximum(a, 0.0)
    sa = a.sum()
    if sa > 1e-10:
        inits['fejer'] = a / sa

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # Raised cosine (Tukey-like)
    a = 0.5 * (1 + np.cos(np.pi * x / (n - 1)))
    a /= a.sum()
    inits['tukey'] = a

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase search combining LP refinement,
    smoothed gradient descent, subgradient methods, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent: len=%d, sum=%.4f" % (len(p), p.sum()))
    except Exception:
        pass

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return best_score
        # Keep values in valid range
        a = np.clip(a, 0.0, 1000.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  New best: %.6f (n=%d)" % (score, len(a)))
        return score

    # ============================================================
    # Phase 1: Quick LP warm-start from parent (n=512)
    # ============================================================
    print("=== Phase 1: Quick LP warm-start ===")
    t1_start = time.monotonic()

    if parent is not None:
        n_p = len(parent)
        a_p = parent / parent.sum()
        # Quick LP iterations
        a_opt, _ = optimize_lp_iterative(a_p, time_limit=30.0, method='max_sum',
                                         max_iters=200, step_init=0.3)
        try_update(a_opt)
        a_opt, _ = optimize_lp_iterative(a_opt, time_limit=20.0, method='min_peak',
                                         max_iters=150, step_init=0.25)
        try_update(a_opt)

    print("Phase 1 done: best=%.6f, elapsed=%.0fs" % (best_score, time.monotonic() - t1_start))

    # ============================================================
    # Phase 2: Smoothed gradient descent on best (n=512)
    # ============================================================
    print("=== Phase 2: Smoothed gradient ===")
    t2_start = time.monotonic()

    if best_sequence is not None:
        phase2_deadline = total_deadline - 900
        n_b = len(best_sequence)
        for t_init in [150.0, 80.0, 40.0]:
            if time.monotonic() > phase2_deadline:
                break
            t_alloc = min(100.0, phase2_deadline - time.monotonic() - 5)
            if t_alloc < 10:
                break
            a_opt, _ = optimize_smoothed_gradient(
                best_sequence, time_limit=t_alloc, t_init=t_init,
                t_min=2.0, t_decay=0.9998, lr_init=0.004, momentum=0.94)
            try_update(a_opt)

    print("Phase 2 done: best=%.6f, elapsed=%.0fs" % (best_score, time.monotonic() - t2_start))

    # ============================================================
    # Phase 3: Deep LP refinement on best (n=512)
    # ============================================================
    print("=== Phase 3: Deep LP refinement ===")
    t3_start = time.monotonic()

    if best_sequence is not None:
        phase3_deadline = total_deadline - 600
        for method in ['max_sum', 'min_peak', 'max_sum']:
            if time.monotonic() > phase3_deadline:
                break
            t_alloc = min(100.0, phase3_deadline - time.monotonic() - 5)
            if t_alloc < 10:
                break
            for step in [0.35, 0.2, 0.5]:
                if time.monotonic() > phase3_deadline:
                    break
                t_single = min(30.0, phase3_deadline - time.monotonic() - 5)
                if t_single < 5:
                    break
                a_opt, _ = optimize_lp_iterative(
                    best_sequence, time_limit=t_single, method=method,
                    max_iters=300, step_init=step)
                try_update(a_opt)

    print("Phase 3 done: best=%.6f, elapsed=%.0fs" % (best_score, time.monotonic() - t3_start))

    # ============================================================
    # Phase 4: Multi-length exploration (try n=1024, 2048)
    # ============================================================
    print("=== Phase 4: Multi-length exploration ===")
    t4_start = time.monotonic()
    phase4_deadline = total_deadline - 200

    target_lengths = [1024, 2048]
    if parent is not None and len(parent) not in target_lengths:
        parent_len = len(parent)
        if parent_len > 512:
            target_lengths.append(parent_len)

    for n_target in target_lengths:
        if time.monotonic() > phase4_deadline:
            break
        print("  Exploring n=%d" % n_target)

        inits = generate_initializations(n_target, parent, seed + n_target)
        init_names = list(inits.keys())
        # Prioritize parent-based and structured inits
        priority = ['parent', 'tukey', 'fejer', 'cos_1', 'gauss', 'uniform', 'random']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]

        n_inits = min(3, len(init_names))
        n_init_names = init_names[:n_inits]

        time_per_init = max(20.0, (phase4_deadline - time.monotonic() - 30) /
                            (len(target_lengths) * n_init_names * 2))

        for name in n_init_names:
            if time.monotonic() > phase4_deadline:
                break
            a0 = inits[name]
            # LP refinement
            t_alloc = min(time_per_init, phase4_deadline - time.monotonic() - 60)
            if t_alloc < 10:
                break
            a_opt, _ = optimize_lp_iterative(
                a0, time_limit=t_alloc, method='max_sum',
                max_iters=300, step_init=0.3)
            score = try_update(a_opt)

            if np.isfinite(score) and score < best_score + 0.05:
                # Follow up with min_peak
                t_alloc2 = min(time_per_init / 2, phase4_deadline - time.monotonic() - 30)
                if t_alloc2 > 5:
                    a_opt2, _ = optimize_lp_iterative(
                        a_opt, time_limit=t_alloc2, method='min_peak',
                        max_iters=200, step_init=0.25)
                    try_update(a_opt2)

    print("Phase 4 done: best=%.6f, elapsed=%.0fs" % (best_score, time.monotonic() - t4_start))

    # ============================================================
    # Phase 5: Subgradient refinement + perturbations
    # ============================================================
    print("=== Phase 5: Subgradient + perturbations ===")
    t5_start = time.monotonic()
    phase5_deadline = total_deadline - 60

    if best_sequence is not None:
        n_best = len(best_sequence)
        # Subgradient
        for top_k in [max(30, n_best // 16), max(10, n_best // 40)]:
            if time.monotonic() > phase5_deadline:
                break
            t_alloc = min(50.0, phase5_deadline - time.monotonic() - 10)
            if t_alloc < 10:
                break
            a_opt, _ = optimize_subgradient(
                best_sequence, time_limit=t_alloc, lr_init=0.015,
                momentum=0.88, top_k=top_k, temp_init=0.0008)
            try_update(a_opt)

        # Perturbation + re-optimize
        for scale in [0.003, 0.01, 0.001]:
            if time.monotonic() > phase5_deadline:
                break
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(best_sequence + noise)
            t_alloc = min(25.0, phase5_deadline - time.monotonic() - 5)
            if t_alloc < 5:
                break
            a_opt, _ = optimize_lp_iterative(
                a_pert, time_limit=t_alloc, method='min_peak',
                max_iters=150, step_init=0.2)
            try_update(a_opt)

    print("Phase 5 done: best=%.6f, elapsed=%.0fs" % (best_score, time.monotonic() - t5_start))

    # ============================================================
    # Phase 6: Final polish
    # ============================================================
    print("=== Phase 6: Final polish ===")
    t6_start = time.monotonic()

    if best_sequence is not None:
        # Final LP with very small steps
        t_alloc = max(5.0, total_deadline - time.monotonic() - 3)
        a_opt, _ = optimize_lp_iterative(
            best_sequence, time_limit=t_alloc, method='max_sum',
            max_iters=2000, step_init=0.1, step_min=0.005)
        try_update(a_opt)

    print("Phase 6 done: best=%.6f, elapsed=%.0fs" % (best_score, time.monotonic() - t6_start))

    # Final fallback
    if best_sequence is None:
        n_fb = 512 if parent is None else len(parent)
        best_sequence = np.ones(n_fb) / n_fb
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print("\nFINAL: score=%.6f, n=%d" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
