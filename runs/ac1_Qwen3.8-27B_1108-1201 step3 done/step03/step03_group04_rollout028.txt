Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative updates (exponential family on simplex) with annealed
temperature, combined with periodic LP-based Frank-Wolfe steps for precise
direction finding. Multi-length scanning with parent-based warm starts.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(a):
    """Build sparse matrix for LP oracle: (A @ b)[k] = conv(a, b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M, time_limit=3.0):
    """LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def multiplicative_step(a, lr, T):
    """One multiplicative update step with soft-max weighting."""
    n = len(a)
    conv = fast_convolve(a)
    # Soft weights on peaks
    shifted = conv / T
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    # Compute subgradient: g[j] = 2 * sum_k w[k] * a[k-j]
    g = np.zeros(n)
    for j in range(n):
        k_end = min(2 * n - 1, j + n)
        g[j] = 2.0 * np.dot(a[:k_end - j], w[j:k_end])
    # Exponential update
    log_a = np.log(np.maximum(a, 1e-30)) - lr * g
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    s = a_new.sum()
    a_new /= s
    return a_new, conv

    a /= s
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 100
    
    while time.monotonic() - t0 < time_limit:
        b = lp_oracle(a, M, time_limit=2.0)
        if b is None:
            break
        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb
        
        if np.allclose(a, b, atol=1e-12):
            no_improve += 1
        else:
            # Line search
            best_t = 0.0
            best_line_obj = best_obj
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidate = (1 - t) * a + t * b
                obj = compute_obj(candidate)
                if obj < best_line_obj:
                    best_line_obj = obj
                    best_t = t
            
            if best_t > 0.01:
                a = (1 - best_t) * a + best_t * b
                conv = fast_convolve(a)
                M = conv.max()
                cur_obj = 2.0 * n * M
                if cur_obj < best_obj - 1e-12:
                    best_obj = cur_obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
        if no_improve >= max_stall:
            rng = np.random.default_rng(1000 + it)
            noise = rng.standard_normal(n) * 0.002
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_stall = int(max_stall * 1.5)
        
        it += 1
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FPW: it={it}, obj={best_obj:.6f}, elapsed={elapsed:.1f}s")

                continue

            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)

        a_new = np.maximum(a + v, 0.0)

        a_new /= s

        if cur_obj < best_obj - 1e-12:

            best_a = a.copy()
        
        lr *= 0.9998
        it += 1
    
    return best_a, best_obj

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: (1) FFT-accelerated multiplicative updates (mirror descent on simplex)
with annealed temperature, (2) LP-based Frank-Wolfe for precise direction finding,
(3) multi-peak subgradient with momentum, and (4) multi-length scanning.
Key innovation: O(n log n) subgradient via cross-correlation FFT, enabling
millions of fast iterations for landscape exploration before LP polishing.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_obj_local(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def compute_subgradient_fft(a, w):
    """Compute 2 * cross_correlation(w, a) using FFT. O(n log n)."""
    n = len(a)
    target = 3 * n
    fft_size = 1 << (target - 1).bit_length()
    A = np.fft.fft(a, fft_size)
    W = np.fft.fft(w, fft_size)
    corr = np.fft.ifft(W * np.conj(A)).real
    return 2.0 * corr[:n]


def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a, b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M, time_limit=3.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def multiplicative_step(a, lr, T):
    """One exponentiated gradient step with soft-max peak weighting."""
    n = len(a)
    conv = fast_convolve(a)
    shifted = conv / T
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    g = compute_subgradient_fft(a, w)
    log_a = np.log(np.maximum(a, 1e-30)) - lr * g
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    s = a_new.sum()
    a_new /= s
    return a_new, conv

    a = np.maximum(a0.copy(), 0.0)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    max_stall = 80
    while time.monotonic() - t0 < time_limit:
        b = lp_oracle(a, M, time_limit=2.5)
        if b is None:
            break
        sb = b.sum()
        if sb < 1e-15:
            break
        b = b / sb
        if np.allclose(a, b, atol=1e-12):
            no_improve += 1
        else:
            best_t = 0.0
            best_lo = best_obj
            for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                cand = (1.0 - t) * a + t * b
                lo = compute_obj_local(cand)
                if lo < best_lo:
                    best_lo = lo
                    best_t = t
            if best_t > 0.01:
                a = (1.0 - best_t) * a + best_t * b
                conv = fast_convolve(a)
                M = conv.max()
                cur_obj = 2.0 * n * M
                if cur_obj < best_obj - 1e-13:
                    best_obj = cur_obj
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        if no_improve >= max_stall:
            rng_r = np.random.default_rng(777 + it)
            noise = rng_r.standard_normal(n) * 0.003
            a = np.maximum(best_a + noise, 0.0)
            a /= a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            max_stall = int(max_stall * 1.5)
        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"      FPW it={it}, obj={best_obj:.7f}, t={elapsed:.1f}s")
    return best_a, best_obj


def subgrad_momentum_optimize(a0, time_limit, lr_init=0.008, momentum=0.92):
    """Multi-peak subgradient with momentum."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a /= a.sum()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        top_k = min(25, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        temp = max((pv.max() - pv.min()) * 0.05, 1e-15)
        scaled = (pv - pv.min()) / temp
        w = np.exp(scaled)
        w /= w.sum()
        w_full = np.zeros(2 * n - 1)
        w_full[peaks] = w
        g = compute_subgradient_fft(a, w_full)
        v = momentum * v - lr * g
        a = a + v
        a = np.maximum(a, 0.0)
        a /= a.sum()
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
        lr *= 0.9999
    return best_a, best_obj


def try_lengths(best_seq, best_score, deadline, rng):
    """Try optimizing at nearby lengths, returning best found."""
    if best_seq is None:
        return None, float('inf')
    n_best = len(best_seq)
    candidates = []
    for factor in [0.75, 0.875, 1.125, 1.25, 1.5]:
        nl = int(n_best * factor)
        if 128 <= nl <= 1024 and nl not in [len(c) for c in candidates]:
            candidates.append(nl)
    for nl in candidates:
        if time.monotonic() > deadline - 45:
            break
        remaining = deadline - time.monotonic() - 30
        if remaining < 40:
            break
        pp = np.interp(np.linspace(0, 1, nl), np.linspace(0, 1, n_best), best_seq)
        a0 = np.maximum(pp, 0.0)
        a0 /= a0.sum()
        t1 = remaining * 0.5
        a_opt, obj_opt = annealed_mul_optimize(a0, time_limit=t1, T_init=0.015, T_end=0.0004, lr_init=0.1)
        score = float(evaluate_sequence(a_opt.tolist()))
        print(f"    n={nl}: score={score:.6f}")
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase hybrid search: annealed multiplicative + LP-FW + subgradient.
    Uses FFT-based O(n log n) subgradients for fast exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)
    best_sequence = None
    best_score = float('inf')
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass
    if parent is not None:
        n_p = len(parent)
        a_p = np.maximum(parent, 0.0)
        a_p /= a_p.sum()
        parent_score = float(evaluate_sequence(a_p.tolist()))
        best_sequence = a_p.copy()
        best_score = parent_score
        print(f"Parent score: {best_score:.6f}")
    # Phase 1: Annealed multiplicative (fast exploration)
    print("=== Phase 1: Annealed Multiplicative ===")
    t_alloc = min(280.0, (deadline - time.monotonic()) * 0.18)
    a_opt, _ = annealed_mul_optimize(best_sequence, t_alloc, T_init=0.025, T_end=0.0006, lr_init=0.14, verbose=True)
    sc = float(evaluate_sequence(a_opt.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_sequence = a_opt.copy()
        print(f"  P1 improved: {sc:.6f}")
    # Phase 2: LP-FW polish
    print("=== Phase 2: LP-FW ===")
    rem = deadline - time.monotonic()
    t_alloc = min(280.0, rem * 0.22)
    a_opt, _ = fpw_optimize(best_sequence, t_alloc, verbose=True)
    sc = float(evaluate_sequence(a_opt.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_sequence = a_opt.copy()
        print(f"  P2 improved: {sc:.6f}")
    # Phase 3: Subgradient momentum
    print("=== Phase 3: Subgradient ===")
    rem = deadline - time.monotonic()
    t_alloc = min(180.0, rem * 0.15)
    a_opt, _ = subgrad_momentum_optimize(best_sequence, t_alloc, lr_init=0.006, momentum=0.93)
    sc = float(evaluate_sequence(a_opt.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_sequence = a_opt.copy()
        print(f"  P3 improved: {sc:.6f}")
    # Phase 4: Second round annealed + LP
    print("=== Phase 4: Refine ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t1 = rem * 0.3
        a_opt, _ = annealed_mul_optimize(best_sequence, t1, T_init=0.01, T_end=0.0003, lr_init=0.07)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_sequence = a_opt.copy()
            print(f"  P4a improved: {sc:.6f}")
        rem = deadline - time.monotonic()
        if rem > 40:
            t2 = rem * 0.5
            a_opt, _ = fpw_optimize(best_sequence, t2, verbose=True)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_sequence = a_opt.copy()
                print(f"  P4b improved: {sc:.6f}")
    # Phase 5: Try different lengths
    print("=== Phase 5: Length Search ===")
    best_sequence, best_score = try_lengths(best_sequence, best_score, deadline, rng)
    # Phase 6: Final LP pass
    print("=== Phase 6: Final ===")
    rem = deadline - time.monotonic() - 10
    if rem > 15 and best_sequence is not None:
        a_opt, _ = fpw_optimize(best_sequence, rem, verbose=True)
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_sequence = a_opt.copy()
            print(f"  Final improved: {sc:.6f}")
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
