"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using spectral (Gerchberg-Saxton) 
iterations with soft non-negativity enforcement, combined with L-BFGS-B 
refinement on a smooth max approximation, multi-start over lengths, and 
adaptive peak-focused coordinate descent.

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_smooth_conv(a, w):
    """Gradient of weighted sum of conv values w.r.t. a."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_iteration(a, n_iter=500, softness=0.1):
    """
    Gerchberg-Saxton style iteration for flat autocorrelation.
    Alternates between:
    - Frequency domain: flatten |DFT(a)|^2
    - Time domain: enforce non-negativity (soft) and normalize
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        return a
    a /= s

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    for it in range(n_iter):
        # Forward transform
        A = fft(a, fft_size)
        
        # Target flat spectrum: set |A| to constant
        # The DC component should give sum(a) = 1, so |A[0]| = 1
        # For flat spectrum, all |A[k]| = 1/sqrt(fft_size) * something
        # Actually, let's just set all magnitudes equal
        mags = np.abs(A)
        # Keep the phase, set magnitude to mean
        target_mag = mags.mean()
        if target_mag < 1e-15:
            break
        new_mags = np.full_like(mags, target_mag)
        
        # Inverse transform with flat magnitude and original phase
        phases = np.angle(A)
        A_new = new_mags * np.exp(1j * phases)
        a_new = np.real(ifft(A_new, fft_size))[:n]
        
        # Soft non-negativity enforcement
        # Instead of hard clipping, use a soft threshold
        # a_soft = a + softness * |a| * sign(a) ... no
        # Use: a_soft = max(a, -softness * mean_positive)
        # Or: a_soft = a + softness * (a - max(a,0)) ... 
        # Simplest: just clip and renormalize
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new /= s_new
        
        # Check improvement
        conv = fast_convolve(a_new)
        obj = 2.0 * n * conv.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a


def spectral_iteration_v2(a, n_iter=500, alpha=0.5):
    """
    Improved spectral iteration with relaxation.
    Blends the spectral update with the current iterate to avoid oscillation.
    """
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        return a
    a /= s

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    for it in range(n_iter):
        A = fft(a, fft_size)
        mags = np.abs(A)
        phases = np.angle(A)
        
        # Target: flat magnitude spectrum
        # But preserve some structure: use geometric mean or trimmed mean
        target_mag = np.median(mags)  # More robust than mean
        if target_mag < 1e-15:
            break
        
        # Blend: partially flatten
        new_mags = (1 - alpha) * mags + alpha * target_mag
        A_new = new_mags * np.exp(1j * phases)
        a_raw = np.real(ifft(A_new, fft_size))[:n]
        
        # Enforce non-negativity
        a_new = np.maximum(a_raw, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new /= s_new
        
        conv = fast_convolve(a_new)
        obj = 2.0 * n * conv.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a


def peak_focused_descent(a, n_iter=200):
    """
    Directly attack the peak of the convolution.
    At each step, identify the peak lag k*, find which (i, j) pairs 
    contribute to conv[k*], and reduce the product a[i]*a[j] by 
    adjusting a[i] and a[j].
    """
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        return a
    a /= s

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    for it in range(n_iter):
        conv = fast_convolve(a)
        max_k = int(np.argmax(conv))
        max_val = conv[max_k]
        
        # The conv at lag k is sum_{i} a[i] * a[i+k] (for k >= 0)
        # or sum_{i} a[i] * a[i-k] (for k < 0, but our conv is full so it's symmetric)
        # For full conv, conv[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[j] * a[k-j]
        
        j_start = max(0, max_k - n + 1)
        j_end = min(n - 1, max_k)
        
        # Get the products contributing to the peak
        if j_end >= j_start:
            products = a[j_start:j_end + 1] * a[max_k - j_end:max_k - j_start + 1][::-1]
            # Wait, let me be more careful.
            # conv[k] = sum_j a[j] * a[k-j] where 0 <= j < n and 0 <= k-j < n
            # So j ranges from max(0, k-n+1) to min(k, n-1)
            # The pairs are (j, k-j)
        
        pairs_j = np.arange(j_start, j_end + 1)
        pairs_k = max_k - pairs_j
        
        # Products
        prods = a[pairs_j] * a[pairs_k]
        total_prod = prods.sum()
        
        if total_prod < 1e-15:
            break
        
        # Reduce each product by scaling down both factors proportionally
        # We want to reduce conv[max_k] while keeping sum(a) = 1
        # Strategy: reduce a[i] for pairs where both a[i] and a[k-i] are large
        # and redistribute to pairs where the product is small
        
        # Simple approach: for the pair with the largest product, reduce both elements
        max_prod_idx = np.argmax(prods)
        i1 = pairs_j[max_prod_idx]
        i2 = pairs_k[max_prod_idx]
        
        # Reduce a[i1] and a[i2] slightly
        reduction = 0.01 * a[i1]
        if i1 != i2:
            a[i1] -= reduction
            a[i2] -= reduction
        else:
            a[i1] -= reduction * 2
        
        # Distribute the reduced mass to elements that don't contribute much to the peak
        # Find elements with low contribution
        low_contrib = np.argsort(prods)[:max(1, len(prods) // 4)]
        mass_to_distribute = reduction * (1 if i1 != i2 else 2)
        for idx in low_contrib:
            a[idx] += mass_to_distribute / len(low_contrib)
        
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a /= s
        
        conv_new = fast_convolve(a)
        obj = 2.0 * n * conv_new.max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a


def _obj_and_jac(x, n, beta):
    """Objective and Jacobian for L-BFGS-B with softmax parametrization."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    obj = 2.0 * n * sm
    grad_a = compute_grad_smooth_conv(a, w)
    lagrange = np.dot(a, grad_a)
    jac = a * (grad_a - lagrange)
    return obj, jac


def optimize_lbfgsb(a0, beta, maxiter=300):
    """L-BFGS-B optimization with softmax reparametrization."""
    n = len(a0)
    a0_safe = np.maximum(a0, 1e-20)
    a0_safe = a0_safe / a0_safe.sum()
    x0 = np.log(a0_safe)
    
    def fun(x):
        obj, _ = _obj_and_jac(x, n, beta)
        return obj
    
    def jac(x):
        _, j = _obj_and_jac(x, n, beta)
        return j
    
    result = minimize(fun, x0, jac=jac, method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxls': 50, 'ftol': 1e-15, 'gtol': 1e-12})
    
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt


def gen_init_flat_spectrum(n, rng, n_trials=10):
    """Generate initial sequences with flat power spectrum."""
    best_a = np.ones(n) / n
    best_obj = float('inf')
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for _ in range(n_trials):
        A = np.zeros(fft_size, dtype=complex)
        A[0] = np.sqrt(fft_size)  # DC: gives sum = 1 after normalization
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
        for k in range(1, fft_size // 2 + 1):
            mag = 1.0
            A[k] = mag * np.exp(1j * phases[k - 1])
            if fft_size - k != k:
                A[fft_size - k] = mag * np.exp(-1j * phases[k - 1])
        
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            obj = 2.0 * n * fast_convolve(a).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    
    return best_a


def gen_init_cyclic(n, rng):
    """Initialize using a cyclic/group structure that tends to give flat autocorrelation."""
    # Try sequences based on quadratic residues or similar structures
    # For non-negative sequences, we can try:
    # a[i] = 1 if i is a quadratic residue mod p, else 0 (then normalize)
    # But n might not be prime. Let's use a simpler approach.
    
    # Use a "spread spectrum" pattern
    a = np.zeros(n)
    # Place ones at positions that are well-spread
    step = max(1, int(n / 8))
    for i in range(0, n, step):
        a[i] = 1.0
    # Add some randomness
    a += rng.uniform(0, 0.3, n)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def all_inits(n, parent, rng):

    inits = {}
    
    inits['uniform'] = np.ones(n) / n
    
    # Flat spectrum
    inits['flat_spec'] = gen_init_flat_spectrum(n, rng, n_trials=5)
    
    # Cyclic
    inits['cyclic'] = gen_init_cyclic(n, rng)
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Sparse random
    inits['sparse'] = rng.dirichlet(np.ones(n) * 0.1)
    
    # Gaussian centered
    x = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        pp = np.array(parent, dtype=np.float64)
        pp = np.clip(pp, 0, 1000)
        if len(pp) == n:

            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp_resampled = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(pp)), pp)
            pp_resampled = np.maximum(pp_resampled, 0)
            ps = pp_resampled.sum()
            if ps > 1e-10:
                inits['parent_resample'] = pp_resampled / ps
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Gerchberg-Saxton spectral iterations for flat autocorrelation design
2. L-BFGS-B with log-sum-exp smooth max approximation for local refinement

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights with numerical stability."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_smooth_conv(a, w):
    """Gradient of weighted sum of conv values w.r.t. a using FFT."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def exact_subgrad_descent(a, n_iter=1000, lr0=0.01):
    """
    Subgradient method on the exact max convolution value.
    At each step: compute conv, find peak lag k*, compute subgradient,
    take a projected step.
    """
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        return a
    a /= s

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    lr = lr0

    for it in range(n_iter):
        conv = fast_convolve(a)
        max_k = int(np.argmax(conv))
        max_val = conv[max_k]
        obj = 2.0 * n * max_val

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Subgradient of max_k conv_k w.r.t. a:
        # conv_k = sum_j a[j] * a[k-j]
        # d(conv_k)/d(a[i]) = a[k-i] + a[i-k] (if in bounds)
        grad = np.zeros(n)
        for i in range(n):
            g = 0.0
            j1 = max_k - i
            if 0 <= j1 < n:
                g += a[j1]
            j2 = i - max_k
            if 0 <= j2 < n:
                g += a[j2]
            grad[i] = g

        # Projected gradient step
        a_new = project_simplex(a - lr * grad)

        # Simple adaptive lr
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        if obj_new <= obj:
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.7
            a_new = project_simplex(a - lr * grad)

        # Decay lr over time
        lr = max(lr * 0.999, 1e-6)
        a = a_new

    return best_a


def spectral_iteration(a, n_iter=300, alpha=0.5, use_median=True):

def _lbfgsb_fun(x, n, beta):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def _lbfgsb_jac(x, n, beta):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = compute_grad_smooth_conv(a, w)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def optimize_lbfgsb(a0, beta, maxiter=300):
    """L-BFGS-B with softmax reparametrization."""
    n = len(a0)
    a0_safe = np.maximum(a0, 1e-20)
    a0_safe = a0_safe / a0_safe.sum()
    x0 = np.log(a0_safe)

    result = minimize(_lbfgsb_fun, x0, jac=_lbfgsb_jac, args=(n, beta),
                      method='L-BFGS-B',
                      options={'maxiter': maxiter, 'maxls': 50, 'ftol': 1e-14, 'gtol': 1e-10})

    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt

            a_cur = best_sequence / best_sequence.sum() if best_sequence is not None and len(best_sequence) == n else a_spec

            # Step B: L-BFGS-B with decreasing beta
            remaining = phase2_end - time.monotonic()
            for beta in [40, 20, 10, 5, 3]:
                if time.monotonic() > phase2_end - 20:
                    break
                t_lb = min(80.0, (phase2_end - time.monotonic()) * 0.15)
                if t_lb < 5:
                    break
                maxiter = int(min(400, max(50, t_lb * 20)))
                a_opt = optimize_lbfgsb(a_cur, beta=beta, maxiter=maxiter)
                score_update(a_opt)
                a_cur = best_sequence / best_sequence.sum() if best_sequence is not None and len(best_sequence) == n else a_opt

            # Step C: Subgradient on exact max
            remaining = phase2_end - time.monotonic()
            if remaining > 30:
                n_sg = int(min(500, remaining * 15))
                a_sg = exact_subgrad_descent(a_cur, n_iter=n_sg, lr0=0.005)
                score_update(a_sg)

    print("Phase 2 best: %.6f" % best_score)

    # === Phase 3: Intensive fine-tuning ===
    print("=== Phase 3: Fine-tuning ===")

    if best_sequence is not None:
        n_best = len(best_sequence)

        for rnd in range(15):
            remaining = deadline - time.monotonic() - 15
            if remaining < 20:
                break

            a_best = best_sequence / best_sequence.sum()

            # L-BFGS-B with low beta
            t_lb = min(120.0, remaining * 0.25)
            if t_lb < 8:
                break
            maxiter = int(min(500, max(100, t_lb * 25)))
            a_opt = optimize_lbfgsb(a_best, beta=2.0, maxiter=maxiter)
            score_update(a_opt)

            # Subgradient
            remaining = deadline - time.monotonic() - 15
            if remaining < 20:
                break
            n_sg = int(min(400, remaining * 12))
            a_best = best_sequence / best_sequence.sum()
            a_sg = exact_subgrad_descent(a_best, n_iter=n_sg, lr0=0.003)
            score_update(a_sg)

            # Spectral polish
            remaining = deadline - time.monotonic() - 15
            if remaining < 20:
                break
            a_best = best_sequence / best_sequence.sum()
            n_spec = int(min(200, remaining * 8))
            a_sp = spectral_iteration(a_best, n_iter=n_spec, alpha=0.2, use_median=True)
            score_update(a_sp)

            # Perturbation restart
            remaining = deadline - time.monotonic() - 15
            if remaining < 30:
                break
            a_best = best_sequence / best_sequence.sum()
            scale = 0.003 / (1 + rnd * 0.5)
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(a_best + noise)
            maxiter2 = int(min(200, max(50, remaining * 8)))
            a_opt2 = optimize_lbfgsb(a_pert, beta=8.0, maxiter=maxiter2)
            score_update(a_opt2)

    # === Phase 4: Try alternate lengths ===
    print("=== Phase 4: Alternate Lengths ===")
    remaining = deadline - time.monotonic() - 20
    if remaining > 100 and best_sequence is not None:
        for alt_n in [320, 384, 448, 512, 256, 192]:
            if len(best_sequence) == alt_n or time.monotonic() > deadline - 80:
                continue
            inits_alt = all_inits(alt_n, parent, rng)
            resamp = np.interp(np.linspace(0, 1, alt_n), np.linspace(0, 1, len(best_sequence)), best_sequence)
            resamp = np.maximum(resamp, 0)
            ps = resamp.sum()
            if ps > 1e-10:
                inits_alt['from_best'] = resamp / ps

            a0 = inits_alt.get('from_best', inits_alt.get('flat_spec', inits_alt['uniform']))
            a_spec = spectral_iteration(a0, n_iter=150, alpha=0.5, use_median=True)
            score_update(a_spec)

            maxiter = int(min(300, max(50, (deadline - time.monotonic() - 80) * 15)))
            if maxiter > 50:
                a_opt = optimize_lbfgsb(a_spec, beta=10, maxiter=maxiter)
                score_update(a_opt)

    # Safety
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL SCORE: %.6f (n=%d)" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
