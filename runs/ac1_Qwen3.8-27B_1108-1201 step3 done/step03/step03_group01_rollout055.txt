"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid of
top-k weighted subgradient descent and LP-based direction finding.
Uses multiple sequence lengths and structured initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_from_conv(c, n):
    return 2.0 * n * c.max()


def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def topk_subgradient(a, c, top_k=30, temp=0.001):
    n = len(a)
    if top_k >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argpartition(c, -top_k)[-top_k:]
    vals = c[idx]
    scaled = (vals - vals.max()) / temp
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(idx))
        w[np.argmax(vals)] = 1.0
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for i in range(len(idx)):
        if w[i] < 1e-14:
            continue
        k = int(idx[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        j_idx = np.arange(j_start, j_end + 1)
        grad[j_idx] += w[i] * a[k - j_idx]
    
    return 2.0 * grad


def build_conv_matrix(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp(a, M):
    n = len(a)
    A_ub = build_conv_matrix(a)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return a.copy(), 1.0


def lp_step(a, c, M):
    g, g_sum = solve_lp(a, M * 1.001)
    if g_sum <= 1.0 + 1e-8:
        return None
    return g


def line_search(a, d, best_obj, n, max_iter=15):
    best_t = 0.0
    for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.5, 0.4, 0.3, 0.2, 0.15, 0.12]:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        c_try = fast_convolve(a_try)
        obj_try = obj_from_conv(c_try, n)
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = t
    return best_t, best_obj


def subgradient_optimize(a0, time_limit, top_k=30, temp=0.001, lr_init=0.01):
    a = normalize(a0.astype(np.float64).copy())
    n = len(a)
    c = fast_convolve(a)
    best_obj = obj_from_conv(c, n)
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    momentum = 0.9
    v = np.zeros(n)
    no_improve = 0
    iteration = 0
    lp_interval = 200
    temp_cur = temp
    
    while time.monotonic() - t0 < time_limit:
        # Subgradient step
        grad = topk_subgradient(a, c, top_k, temp_cur)
        grad -= grad.mean()  # Project to sum=0
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:

        c_new = fast_convolve(a_new)
        obj_new = obj_from_conv(c_new, n)
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        c = c_new
        iteration += 1

        temp_cur *= 1.001
        momentum = min(0.97, momentum + 0.0001)

                    a_new2 = np.maximum(a_new2, 0.0)

                    if s2 > 1e-15:

            no_improve = 0

    return best_a, best_obj


def generate_inits(n, parent, rng):
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Exponential ramps
    for gamma in [0.5, 1.0, 2.0, 3.0]:
        a = np.exp(-gamma * x / n)
        inits[f'exp_{gamma}'] = a / a.sum()
    
    # Symmetric exponential
    for gamma in [1.0, 2.0, 3.0]:
        center = (n - 1) / 2.0
        a = np.exp(-gamma * (x - center) ** 2 / (2 * (n / 4) ** 2))
        inits[f'gauss_{gamma}'] = a / a.sum()
    
    # Power law
    for p in [0.5, 1.0, 2.0]:
        a = (x + 1) ** p
        inits[f'pow_{p}'] = a / a.sum()
    
    # Bimodal
    edge = max(1, n // 4)
    a = np.zeros(n)
    a[:edge] = 1.0
    a[-edge:] = 1.0
    inits['bimodal'] = a / a.sum()
    
    # Sine
    a = np.sin(np.pi * x / n) + 0.1
    inits['sine'] = a / a.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0.0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(x, np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0.0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Top-k weighted subgradient descent with momentum (fast inner loop)
2. LP-based direction finding with line search (periodic refinement)
3. Multiple sequence lengths (512, 768, 1024) with structured initializations
4. Adaptive perturbation to escape local minima
5. Final LP-heavy polish phase
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_from_conv(c, n):
    """Compute objective from convolution and length."""
    return 2.0 * n * c.max()


def normalize(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def topk_subgradient(a, c, top_k=30, temp=0.001):
    """Compute softmax-weighted subgradient at top-k convolution peaks."""
    n = len(a)
    num_peaks = min(top_k, len(c))
    if num_peaks < len(c):
        idx = np.argpartition(c, -num_peaks)[-num_peaks:]
    else:
        idx = np.arange(len(c))
    vals = c[idx]
    scaled = (vals - vals.max()) / temp
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(num_peaks)
        w[np.argmax(vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for i in range(num_peaks):
        if w[i] < 1e-14:
            continue
        k = int(idx[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        j_idx = np.arange(j_start, j_end + 1)
        grad[j_idx] += w[i] * 2.0 * a[k - j_idx]

    return grad


def build_conv_matrix(a):
    """Build matrix C such that C @ g = linear_convolve(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp(a, M):
    """Solve LP: maximize sum(g) s.t. a*g <= M, g >= 0. Returns normalized g and g_sum."""
    n = len(a)
    A_ub = build_conv_matrix(a)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return a.copy(), 1.0


def line_search(a, d, best_obj, n, candidates=None):
    """Line search along direction d. Returns (best_t, best_obj)."""
    if candidates is None:
        candidates = [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
    best_t = 0.0
    for t in candidates:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        c_try = fast_convolve(a_try)
        obj_try = obj_from_conv(c_try, n)
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = t
    return best_t, best_obj


def subgradient_optimize(a0, time_limit, top_k=30, temp=0.001, lr_init=0.01,
                         momentum_init=0.88, lp_interval=150):
    """
    Subgradient optimization with periodic LP refinement.
    Returns (best_a, best_obj).
    """
    a = normalize(a0.astype(np.float64).copy())
    n = len(a)
    c = fast_convolve(a)
    best_obj = obj_from_conv(c, n)
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    momentum = momentum_init
    v = np.zeros(n)
    no_improve = 0
    iteration = 0
    temp_cur = temp
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        # Subgradient step
        grad = topk_subgradient(a, c, top_k, temp_cur)
        grad -= grad.mean()
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = best_a.copy()
            v = np.zeros(n)
        else:
            a_new = a_new / s

        c_new = fast_convolve(a_new)
        obj_new = obj_from_conv(c_new, n)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        c = c_new
        iteration += 1

        # Adaptive parameters
        lr *= 0.9997
        temp_cur *= 1.0008
        momentum = min(0.96, momentum + 0.00005)

        # Periodic LP step
        if iteration % lp_interval == 0 and time.monotonic() - t0 < time_limit - 15:
            M = c.max()
            g, g_sum = solve_lp(a, M * 1.001)
            if g_sum > 1.0 + 1e-8:
                d = g - a
                t_best, obj_ls = line_search(a, d, best_obj, n)
                if t_best > 0:
                    a_ins = a + t_best * d
                    a_ins = np.maximum(a_ins, 0.0)
                    s_ins = a_ins.sum()
                    if s_ins > 1e-15:
                        a_ins = a_ins / s_ins
                        c_ins = fast_convolve(a_ins)
                        obj_ins = obj_from_conv(c_ins, n)
                        if obj_ins < best_obj - 1e-14:
                            best_obj = obj_ins
                            best_a = a_ins.copy()
                            a = a_ins
                            c = c_ins
                            no_improve = 0

        # Perturbation if stuck
        if no_improve >= 400 + 100 * perturb_count:
            rng = np.random.default_rng(iteration * 13 + 7)
            scale = 0.003 * (1 + 0.5 * perturb_count)
            scale = min(scale, 0.05)
            a = best_a + rng.standard_normal(n) * scale
            a = normalize(a)
            c = fast_convolve(a)
            v = np.zeros(n)
            lr = lr_init * 0.7
            momentum = momentum_init
            no_improve = 0
            perturb_count += 1
            if perturb_count > 40:
                break

        if no_improve >= 5000:
            break

    return best_a, best_obj

            if t_best > 0:

def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    for gamma in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = np.exp(-gamma * x / n)
        inits['exp_%s' % gamma] = a / a.sum()

    center = (n - 1) / 2.0
    for sigma_frac in [0.2, 0.3, 0.4, 0.5]:
        sigma = sigma_frac * n
        a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        inits['gauss_%s' % sigma_frac] = a / a.sum()

    for p in [0.5, 1.0, 1.5, 2.0]:
        a = (x + 1.0) ** p
        inits['pow_%s' % p] = a / a.sum()

    # Bimodal (edges)
    edge = max(1, n // 4)
    a = np.zeros(n)
    a[:edge] = 1.0
    a[-edge:] = 1.0
    inits['bimodal'] = a / a.sum()

    # Sine
    a = np.sin(np.pi * x / n) + 0.05
    inits['sine'] = a / a.sum()

    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0.0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(x, np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0.0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    return inits
