"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact subgradient projected
gradient descent with adaptive step size, multi-scale peak weighting,
and restart-based exploration over multiple sequence lengths.

import numpy as np
import time
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_last = np.max(rho[cond])
    theta = (cssv[rho_last - 1] - 1.0) / rho_last
    return np.maximum(v - theta, 0.0)


def compute_sg_smooth(a, conv, beta):
    """Compute smooth subgradient using softmax weights on convolution peaks."""
    n = len(a)
    x_max = conv.max()
    shifted = beta * (conv - x_max)
    shifted = np.clip(shifted, -50, 50)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    
    # sg[i] = 2 * sum_k w[k] * a[k-i] for valid indices
    # Use FFT cross-correlation
    N = 1 << (n + len(conv) - 1 + 7).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    corr = np.real(ifft(W * np.conj(A), N))
    return 2.0 * corr[:n]


def compute_sg_exact(a, conv, top_k=1):
    """Compute exact subgradient using top-k peaks."""
    n = len(a)
    if top_k >= len(conv):
        k_indices = np.arange(len(conv))
    else:
        k_indices = np.argsort(conv)[::-1][:top_k]
    
    sg = np.zeros(n)
    for k in k_indices:
        # sg[i] += 2/top_k * a[k-i] if 0 <= k-i < n
        valid = np.arange(max(0, k - n + 1), min(n, k + 1))
        for i in valid:
            j = k - i
            if 0 <= j < n:
                sg[i] += (2.0 / top_k) * a[j]
    return sg

    else:

    for k in k_indices:
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        i_vals = np.arange(i_start, i_end)
        j_vals = k - i_vals
        sg[i_vals] += (2.0 / top_k) * a[j_vals]

    """Compute 2n * max(conv(a)) for sum-normalized a."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def pgd_optimize(a0, n_iter=1000, initial_lr=0.05, beta_start=50.0,
                 beta_end=5.0, top_k_start=10, top_k_end=1,
                 restart_every=150, perturb_scale=0.005, verbose=False):
    """Projected gradient descent with adaptive parameters and restarts."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = obj_val(a)
    lr = initial_lr
    
    for it in range(n_iter):
        # Adaptive parameters
        progress = it / max(n_iter - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        top_k = max(1, int(top_k_start * (1 - progress) + top_k_end * progress))
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Compute subgradient
        if beta < 100:  # Use smooth gradient
            sg = compute_sg_smooth(a, conv, beta)
        else:
            sg = compute_sg_exact(a, conv, top_k)
        
        # Backtracking line search
        for _ in range(15):
            a_new = project_simplex(a - lr * sg)
            obj_new = obj_val(a_new)
            if obj_new <= cur_obj + 1e-12:
                break
            lr *= 0.5
        
        a = a_new
        
        # Occasionally increase lr if we made good progress
        if obj_new < cur_obj * 0.999:
            lr = min(lr * 1.1, 0.2)
        
        # Restart with perturbation
        if restart_every > 0 and (it + 1) % restart_every == 0:
            if cur_obj < best_obj * 1.001:  # Only restart if we're at a local min
                noise = np.random.standard_normal(n) * perturb_scale
                a = project_simplex(a + noise)
            # Reset lr
            lr = initial_lr * (1 - 0.3 * progress)
    
    return best_a, best_obj

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_last = np.max(rho[cond])
    theta = (cssv[rho_last - 1] - 1.0) / rho_last
    return np.maximum(v - theta, 0.0)


def project_simplex_bounded(v, lb=0.0, ub=None, target_sum=1.0):
    """Project onto {lb <= x <= ub, sum(x) = target_sum}."""
    return project_simplex(np.maximum(v, lb)) * target_sum


def compute_sg_smooth(a, conv, beta):
    """Compute smooth subgradient using softmax weights on convolution peaks."""
    n = len(a)
    x_max = conv.max()
    shifted = beta * (conv - x_max)
    shifted = np.clip(shifted, -50, 50)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    if total < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w = exp_s / total
    N = 1 << (n + len(conv) - 1 + 7).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    corr = np.real(ifft(W * np.conj(A), N))
    return 2.0 * corr[:n]


def compute_sg_exact(a, conv, top_k=1):
    """Compute exact subgradient using top-k peaks (vectorized)."""
    n = len(a)
    if top_k >= len(conv):
        k_indices = np.arange(len(conv))
    else:
        k_indices = np.argsort(conv)[::-1][:top_k]
    sg = np.zeros(n)
    for k in k_indices:
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        i_vals = np.arange(i_start, i_end)
        j_vals = k - i_vals
        sg[i_vals] += (2.0 / top_k) * a[j_vals]
    return sg


def obj_val(a):
    """Compute 2n * max(conv(a)) for sum-normalized a."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def peak_shave(a, n_steps=10, strength=0.1):
    """Directly reduce the convolution peak by adjusting contributing elements."""
    a = a.copy()
    n = len(a)
    for _ in range(n_steps):
        conv = fast_convolve(a)
        k_max = int(np.argmax(conv))
        i_start = max(0, k_max - n + 1)
        i_end = min(n, k_max + 1)
        contrib = np.abs(a[i_start:i_end])
        total = contrib.sum()
        if total < 1e-15:
            break
        # Reduce elements that contribute to the peak
        reduction = strength * (conv.max() - conv.mean()) / max(total, 1e-15)
        a[i_start:i_end] -= reduction * contrib / max(total, 1e-15)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
        else:
            a = np.ones(n) / n
    return a


def pgd_optimize(a0, n_iter=1000, initial_lr=0.05, beta_start=80.0,
                 beta_end=2.0, top_k_start=10, top_k_end=1,
                 restart_every=200, perturb_scale=0.003,
                 use_restarted_best=True, verbose=False):
    """
    Projected gradient descent with adaptive parameters, backtracking,
    and restart mechanism. Returns best sequence and its objective value.
    """
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = obj_val(a)
    lr = initial_lr
    rng = np.random.default_rng(12345 + n)

    for it in range(n_iter):
        progress = it / max(n_iter - 1, 1)
        # Adaptive beta: decrease over time
        beta = beta_start * (beta_end / beta_start) ** progress
        # Adaptive top_k: decrease over time
        top_k = max(1, int(round(top_k_start * (1 - progress) + top_k_end * progress)))

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        # Compute subgradient
        if beta < 50.0:
            # Blend: use exact for very low beta
            if beta < 5.0:
                sg = compute_sg_exact(a, conv, top_k)
            else:
                sg = compute_sg_smooth(a, conv, beta)
        else:
            sg = compute_sg_exact(a, conv, top_k)

        # Normalize gradient for stable step size
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        sg_normalized = sg / sg_norm

        # Backtracking line search
        step_used = False
        for _ in range(20):
            a_new = project_simplex(a - lr * sg_normalized)
            obj_new = obj_val(a_new)
            if obj_new <= cur_obj + 1e-10:
                step_used = True
                break
            lr *= 0.5

        if step_used:
            a = a_new
            # Increase lr slightly if we made good progress
            if obj_new < cur_obj * (1.0 - 1e-4):
                lr = min(lr * 1.05, initial_lr * 2.0)
        else:
            # Stuck - try a small random step
            a = project_simplex(a - lr * 0.1 * sg_normalized + 0.001 * rng.standard_normal(n))
            lr *= 0.5

        # Restart mechanism
        if restart_every > 0 and (it + 1) % restart_every == 0:
            if cur_obj > best_obj * 1.005:
                # Go back to best and perturb
                if use_restarted_best:
                    a = best_a + rng.standard_normal(n) * perturb_scale
                    a = project_simplex(a)
                else:
                    a = project_simplex(a + rng.standard_normal(n) * perturb_scale)
            lr = initial_lr * (1.0 + 0.5 * (1.0 - progress))

        if verbose and it % 200 == 0:
            print("    iter %d: obj=%.6f, best=%.6f, lr=%.4f, beta=%.1f" %
                  (it, cur_obj, best_obj, lr, beta))

    return best_a, best_obj


def multiplicative_descent(a0, n_iter=500, lr=0.02, beta=20.0):
    """
    Multiplicative weight updates (mirror descent with KL divergence).
    a_new[i] = a[i] * exp(-lr * sg[i]) / sum(...)
    """
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = obj_val(a)

    for it in range(n_iter):
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        if beta < 5.0:
            top_k = max(1, min(10, int(11 - beta)))
            sg = compute_sg_exact(a, conv, top_k)
        else:
            sg = compute_sg_smooth(a, conv, beta)

        # Multiplicative update
        updates = np.exp(-lr * sg)
        updates = np.clip(updates, 1e-10, 1e10)
        a_new = a * updates
        a_new = a_new / a_new.sum()

        if obj_val(a_new) < cur_obj:
            a = a_new
        else:
            lr *= 0.5
            if lr < 1e-6:
                break

    return best_a, best_obj


def flat_spectrum_init(n, rng):
    """Initialize with approximately flat power spectrum."""
    fft_size = 1 << (n - 1).bit_length()
    A = np.zeros(fft_size, dtype=complex)
    A[0] = 1.0
    phases = rng.uniform(0, 2 * np.pi, fft_size)
    for k in range(1, fft_size // 2 + 1):
        mag = 1.0
        A[k] = mag * np.exp(1j * phases[k])
        if k < fft_size - k:
            A[fft_size - k] = mag * np.exp(-1j * phases[k])
    a = np.real(ifft(A, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def multi_phase_spectrum_init(n, rng):
    """Initialize with controlled phase for near-flat autocorrelation."""
    best_a = np.ones(n) / n
    best_obj = float('inf')
    fft_size = 1 << (n - 1).bit_length()

    for trial in range(15):
        A = np.zeros(fft_size, dtype=complex)
        A[0] = np.sqrt(n)
        phases = rng.uniform(0, 2 * np.pi, fft_size // 2)
        for k in range(1, fft_size // 2):
            A[k] = np.exp(1j * phases[k - 1])
            A[fft_size - k] = np.exp(-1j * phases[k - 1])
        if fft_size % 2 == 0:
            A[fft_size // 2] = np.exp(1j * phases[-1])

        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            conv = fast_convolve(a)
            obj = 2.0 * n * conv.max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    return best_a


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Gaussian with different widths
    for sf in [0.2, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%d' % int(sf * 100)] = g

    # Bimodal
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Flat spectrum
    inits['flat_spec'] = flat_spectrum_init(n, rng)

    # Multi-phase spectrum
    inits['multi_phase'] = multi_phase_spectrum_init(n, rng)

    # Parent-based
    if parent is not None and len(parent) > 0:
        pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                       np.maximum(parent, 0))
        pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps
        # Perturbed parent
        p_pert = pp + rng.standard_normal(n) * 0.01 * ps
        p_pert = np.maximum(p_pert, 0)
        p_pert /= p_pert.sum()
        inits['parent_pert'] = p_pert

    return inits


def combined_optimize(a0, time_budget, verbose=False):
    """
    Run multiple optimization strategies and return the best result.
    Combines PGD, multiplicative descent, and peak shaving.
    """
    t_start = time.monotonic()
    n = len(a0)
    best_a = a0.copy()
    best_obj = obj_val(a0)

    # Strategy 1: PGD with adaptive params
    if time.monotonic() - t_start < time_budget * 0.4:
        n_iter = min(2000, int(time_budget * 0.4 * 500))
        a_opt, obj_opt = pgd_optimize(a0, n_iter=n_iter, initial_lr=0.04,
                                      beta_start=60.0, beta_end=3.0,
                                      top_k_start=8, top_k_end=1,
                                      restart_every=200, perturb_scale=0.003)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt
            if verbose:
                print("    PGD: %.6f" % best_obj)

    # Strategy 2: Multiplicative descent from current best
    elapsed = time.monotonic() - t_start
    if elapsed < time_budget * 0.7:
        n_iter = min(1000, int((time_budget * 0.3) * 200))
        a_opt, obj_opt = multiplicative_descent(best_a, n_iter=n_iter, lr=0.03, beta=15.0)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt
            if verbose:
                print("    MultDesc: %.6f" % best_obj)

    # Strategy 3: PGD from multiplicative result with different params
    elapsed = time.monotonic() - t_start
    if elapsed < time_budget * 0.85:
        n_iter = min(1500, int((time_budget * 0.15) * 400))
        a_opt, obj_opt = pgd_optimize(best_a, n_iter=n_iter, initial_lr=0.02,
                                      beta_start=30.0, beta_end=1.0,
                                      top_k_start=5, top_k_end=1,
                                      restart_every=150, perturb_scale=0.002)
        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt
            if verbose:
                print("    PGD2: %.6f" % best_obj)

    # Strategy 4: Peak shaving + PGD
    elapsed = time.monotonic() - t_start
    if elapsed < time_budget - 5:
        a_shaved = peak_shave(best_a, n_steps=20, strength=0.05)
        n_iter = min(500, int((time_budget - elapsed - 2) * 300))
        if n_iter > 50:
            a_opt, obj_opt = pgd_optimize(a_shaved, n_iter=n_iter, initial_lr=0.03,
                                          beta_start=20.0, beta_end=1.0,
                                          top_k_start=5, top_k_end=1,
                                          restart_every=100, perturb_scale=0.002)
            if obj_opt < best_obj:
                best_obj = obj_obj = obj_opt
                best_a = a_opt

    return best_a, best_obj


def coordinate_refine(a, n_sweeps=10, n_trials=5):
    """Simple coordinate descent: try scaling individual elements."""
    a = a.copy()
    n = len(a)
    best_obj = obj_val(a)

    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            old_val = a[i]
            for scale in [0.85, 0.9, 0.95, 1.05, 1.1]:
                a[i] = old_val * scale
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s
                obj_new = obj_val(a)
                if obj_new < best_obj - 1e-10:
                    best_obj = obj_new
                    improved = True
                    break
                a[i] = old_val
                a = a / a.sum()
            # Restore if no improvement
            a[i] = old_val
            a = a / a.sum()
        if not improved:
            break

    return a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining projected gradient descent
    with exact/smooth subgradients, multiplicative descent, peak shaving,
    and coordinate refinement to minimize 2n*max(conv)/(sum)^2.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent available: length=%d, sum=%.2f" % (len(p), p.sum()))
    except Exception:
        pass

    def score_and_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return float('inf')
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  New best: score=%.6f, n=%d (t=%.0fs)" %
                  (score, len(a), time.monotonic() - t_start))
        return score if np.isfinite(score) else float('inf')

    # === Phase 1: Quick Scan ===
    print("=== Phase 1: Quick Scan ===")
    lengths = [128, 192, 256, 384, 512, 768]
    phase1_deadline = t_start + 250.0

    scan_results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 20:
            break
        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())

        # Priority ordering
        priority = ['parent', 'parent_pert', 'flat_spec', 'multi_phase',
                    'uniform', 'gauss_35', 'random']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:4]

        for name in init_names:
            if time.monotonic() > phase1_deadline - 10:
                break
            a0 = inits[name]
            # Quick optimization
            a_opt, obj_opt = pgd_optimize(a0, n_iter=150, initial_lr=0.05,
                                          beta_start=50.0, beta_end=5.0,
                                          top_k_start=5, top_k_end=1,
                                          restart_every=100, perturb_scale=0.005)
            # Evaluate with the actual function
            score = score_and_update(a_opt)
            # Also track internal objective for ranking
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))

    scan_results.sort(key=lambda x: x[0])
    print("Phase 1 done. Best: %.6f" % best_score)

    # === Phase 2: Deep Optimization ===
    print("=== Phase 2: Deep Optimization ===")
    phase2_deadline = deadline - 200.0

    # Build list of candidates to deeply optimize
    candidates = []
    for score, n, name, a in scan_results[:5]:
        candidates.append(a)

    # Add some fresh candidates at promising lengths
    if best_sequence is not None:
        best_n = len(best_sequence)
        candidates.append(best_sequence.copy())
        # Try nearby lengths
        for alt_n in [best_n * 2, best_n // 2]:
            if 64 <= alt_n <= 1024:
                inits_alt = generate_initializations(alt_n, parent, rng)
                if 'parent' in inits_alt:
                    candidates.append(inits_alt['parent'])
                if 'flat_spec' in inits_alt:
                    candidates.append(inits_alt['flat_spec'])

    # Deduplicate by length (keep best per length)
    seen_lengths = {}
    deduped = []
    for c in candidates:
        cn = len(c)
        if cn not in seen_lengths or c.sum() > 0:
            if cn not in seen_lengths:
                deduped.append(c)
                seen_lengths[cn] = True

    # Optimize each candidate
    n_cand = len(deduped)
    time_per_cand = max(30.0, (phase2_deadline - time.monotonic()) / max(1, n_cand))

    for idx, cand in enumerate(deduped):
        if time.monotonic() > phase2_deadline - 30:
            break
        t_alloc = min(time_per_cand, phase2_deadline - time.monotonic() - 20)
        if t_alloc < 15:
            break

        print("  Deep optimizing candidate %d (n=%d), time=%.0fs" %
              (idx, len(cand), t_alloc))

        a_opt, _ = combined_optimize(cand, time_budget=t_alloc, verbose=True)
        score = score_and_update(a_opt)

        # Also try a second independent run with different params
        remaining = phase2_deadline - time.monotonic() - 15
        if remaining > 20:
            a2 = cand + rng.standard_normal(len(cand)) * 0.01
            a2 = project_simplex(np.maximum(a2, 0))
            t2 = min(remaining * 0.5, 60.0)
            if t2 > 10:
                a_opt2, _ = pgd_optimize(a2, n_iter=int(t2 * 300),
                                         initial_lr=0.03,
                                         beta_start=40.0, beta_end=2.0,
                                         top_k_start=10, top_k_end=1,
                                         restart_every=150, perturb_scale=0.004)
                score = score_and_update(a_opt2)

    print("Phase 2 done. Best: %.6f" % best_score)

    # === Phase 3: Final Polish ===
    print("=== Phase 3: Final Polish ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 15.0
        print("  Polishing best (n=%d), remaining=%.0fs" % (n_best, remaining))

        a_base = best_sequence / best_sequence.sum()

        # 3a: Fine PGD with low beta
        t.alloc = remaining * 0.3
        if t_alloc > 5:
            n_iter = int(min(2000, t_alloc * 400))
            a_opt, _ = pgd_optimize(a_base, n_iter=n_iter, initial_lr=0.01,
                                    beta_start=15.0, beta_end=0.5,
                                    top_k_start=3, top_k_end=1,
                                    restart_every=300, perturb_scale=0.001)
            score = score_and_update(a_opt)
            remaining = deadline - time.monotonic() - 15.0

        # 3b: Peak shaving + re-optimize
        if remaining > 15:
            a_shaved = peak_shave(a_base, n_steps=30, strength=0.03)
            n_iter = int(min(1000, remaining * 0.3 * 300))
            if n_iter > 50:
                a_opt, _ = pgd_optimize(a_shaved, n_iter=n_iter, initial_lr=0.01,
                                        beta_start=10.0, beta_end=0.5,
                                        top_k_start=3, top_k_end=1,
                                        restart_every=200, perturb_scale=0.001)
                score = score_and_update(a_opt)
            remaining = deadline - time.monotonic() - 15.0

        # 3c: Random perturbations + re-optimize
        if remaining > 20:
            for pert_idx in range(3):
                if remaining < 8:
                    break
                scale = 0.002 + 0.002 * rng.uniform()
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(a_base + noise)
                t_single = min(remaining * 0.3, 60.0)
                n_iter = int(min(500, t_single * 250))
                if n_iter > 30:
                    a_opt, _ = pgd_optimize(a_pert, n_iter=n_iter, initial_lr=0.015,
                                            beta_start=12.0, beta_end=1.0,
                                            top_k_start=3, top_k_end=1,
                                            restart_every=150, perturb_scale=0.001)
                    score = score_and_update(a_opt)
                remaining = deadline - time.monotonic() - 15.0

        # 3d: Coordinate refinement
        if remaining > 10:
            a_coord, _ = coordinate_refine(a_base, n_sweeps=5, n_trials=5)
            score = score_and_update(a_coord)
            if score < best_score:
                a_base = a_coord / a_coord.sum()
            remaining = deadline - time.monotonic() - 15.0

        # 3e: One more fine-tune PGD
        if remaining > 8:
            n_iter = int(min(500, remaining * 300))
            if n_iter > 20:
                a_opt, _ = pgd_optimize(a_base, n_iter=n_iter, initial_lr=0.005,
                                        beta_start=5.0, beta_end=0.3,
                                        top_k_start=2, top_k_end=1,
                                        restart_every=50, perturb_scale=0.0005)
                score = score_and_update(a_opt)

    # Try one more length if time permits
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 50:
            alt_n = 384 if len(best_sequence) == 256 else (
                512 if len(best_sequence) == 384 else 256)
            inits_alt = generate_initializations(alt_n, parent, rng)
            if best_sequence is not None:
                pp = np.interp(np.linspace(0, 1, alt_n),
                               np.linspace(0, 1, len(best_sequence)),
                               best_sequence)
                pp = np.maximum(pp, 0)
                ps = pp.sum()

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact subgradient projected

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_last = np.max(rho[cond])
    theta = (cssv[rho_last - 1] - 1.0) / rho_last
    return np.maximum(v - theta, 0.0)


def compute_sg_smooth(a, conv, beta):
    """Compute smooth subgradient using softmax weights on convolution peaks."""
    n = len(a)
    x_max = conv.max()

    shifted = np.clip(shifted, -50, 50)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    if total < 1e-30:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0

        w = exp_s / total

    """Compute exact subgradient using top-k peaks (vectorized)."""
    n = len(a)
    if top_k >= len(conv):
