"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-directional optimization
with line search, efficient coordinate descent, and multi-phase search.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def obj_value(a):
    """Compute objective with sum(a)=1 normalization."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()

def line_search(a, g, conv, n, num_points=20):
    """1D line search along (1-t)*a + t*g."""
    best_t = 0.0
    best_val = conv.max()
    ts = np.linspace(0, 1, num_points)
    for t in ts:
        a_t = (1-t)*a + t*g
        a_t = np.maximum(a_t, 0)
        s = a_t.sum()
        if s < 1e-15:
            continue
        a_t /= s
        c = fast_convolve(a_t)
        v = c.max()
        if v < best_val:
            best_val = v
            best_t = t
    # Refine
    if 0 < best_t < 1:
        lo = max(0, best_t - 1.0/num_points)
        hi = min(1, best_t + 1.0/num_points)
        for _ in range(10):
            t1 = lo + (hi - lo) / 3
            t2 = hi - (hi - lo) / 3
            a1 = np.maximum((1-t1)*a + t1*g, 0)
            a2 = np.maximum((1-t2)*a + t2*g, 0)
            s1, s2 = a1.sum(), a2.sum()
            if s1 < 1e-15 or s2 < 1e-15:
                break
            a1 /= s1
            a2 /= s2
            v1 = fast_convolve(a1).max()
            v2 = fast_convolve(a2).max()
            if v1 < v2:
                hi = t2
            else:
                lo = t1
        best_t = (lo + hi) / 2
    a_new = (1-best_t)*a + best_t*g
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a, conv
    a_new /= s
    conv_new = fast_convolve(a_new)
    return a_new, conv_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-directional optimization with
line search, exact gradient refinement, efficient coordinate descent, and
multi-phase multi-length search. Improves on prior LP-blend approach by using
adaptive line search along LP directions, exact gradient steps when LP stalls,
and a greedy coordinate descent that exploits convolution structure.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A_matrix(a, n):
    """Build the convolution constraint matrix A where A[k,j] = a[k-j] (valid)."""
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(j, j + n):
            A[k, j] = a[k - j]
    return A


def solve_lp(a, M, n, A=None):
    """Solve LP: max sum(g) s.t. g>=0, (a*g)(k) <= M for all k."""
    if A is None:
        A = build_A_matrix(a, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 30})
        if result.success:
            g = np.maximum(result.x, 0.0)
            if g.sum() > 1e-12:
                return g / g.sum()
    except Exception:
        pass
    return None


def gradient_max_conv(a, conv, n, top_k=10):
    """Compute gradient of max(conv) w.r.t. a using peak locations."""
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    pmax = peak_vals.max()
    # Weight peaks by how close they are to the max
    weights = np.zeros(len(peaks))
    significant = peak_vals > pmax * 0.99
    weights[significant] = 1.0
    if weights.sum() == 0:
        weights[np.argmax(peak_vals)] = 1.0

    grad = np.zeros(n)
    for i in range(len(peaks)):
        if weights[i] < 1e-15:
            continue
        k = int(peaks[i])
        # d/d a_j (a*a)(k) = 2*a[k-j] (with out-of-range = 0)
        # convolved: for each j, contribution is a[k-j] if k-j in [0,n-1]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            grad[j] += weights[i] * 2.0 * a[k - j]
    return grad


def line_search(a, g, a_conv, n, num_coarse=15, num_fine=10):
    """1D line search: minimize max(conv((1-t)*a + t*g)) over t in [0,1]."""
    best_t = 0.0
    best_val = a_conv.max()

    # Coarse search
    ts = np.linspace(0.0, 1.0, num_coarse)
    for t in ts:
        a_t = (1.0 - t) * a + t * g
        a_t = np.maximum(a_t, 0.0)
        s = a_t.sum()
        if s < 1e-15:
            continue
        a_t /= s
        v = fast_convolve(a_t).max()
        if v < best_val - 1e-15:
            best_val = v
            best_t = t

    # Fine search around best
    if 0.0 < best_t < 1.0:
        step = 1.0 / (num_coarse - 1)
        lo = max(0.0, best_t - step)
        hi = min(1.0, best_t + step)
        for _ in range(num_fine):
            t1 = lo + (hi - lo) / 3.0
            t2 = hi - (hi - lo) / 3.0
            a1 = np.maximum((1.0 - t1) * a + t1 * g, 0.0)
            a2 = np.maximum((1.0 - t2) * a + t2 * g, 0.0)
            s1, s2 = a1.sum(), a2.sum()
            if s1 < 1e-15 or s2 < 1e-15:
                break
            a1 /= s1
            a2 /= s2
            v1 = fast_convolve(a1).max()
            v2 = fast_convolve(a2).max()
            if v1 < v2:
                hi = t2
            else:
                lo = t1
        best_t = (lo + hi) / 2.0

    a_new = (1.0 - best_t) * a + best_t * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), a_conv.copy(), best_t
    a_new /= s
    conv_new = fast_convolve(a_new)
    return a_new, conv_new, best_t


def greedy_cd(a, n, max_steps=500, step_frac=0.02):
    """Greedy coordinate descent: move mass from high-peak-contributing to low."""
    a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()

    for step in range(max_steps):
        k_star = int(np.argmax(conv))
        # Gradient contribution: for peak at k_star, grad_j = 2*a[k_star-j]
        # We want to reduce a_j where a[k_star-j] is large, and increase where small
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        contrib = np.zeros(n)
        for j in range(j_start, j_end + 1):
            contrib[j] = a[k_star - j] * a[j]

        # Find source (max a_j * a[k_star - j]) and dest (min)
        if j_end - j_start + 1 < 2:
            break
        sources = list(range(j_start, j_end + 1))
        src = max(sources, key=lambda j: a[j] * contrib[j])
        dst = min(sources, key=lambda j: a[j] * contrib[j])

        if src == dst:
            break

        delta = a[src] * step_frac
        if delta < 1e-10:
            break

        a_test = a.copy()
        a_test[src] -= delta
        a_test[dst] += delta
        a_test = np.maximum(a_test, 0.0)
        s = a_test.sum()
        if s < 1e-15:
            break
        a_test /= s
        conv_test = fast_convolve(a_test)
        if conv_test.max() < best_obj - 1e-14:
            a = a_test
            conv = conv_test
            best_obj = conv.max()
        else:
            # Try smaller step
            delta *= 0.3
            if delta < 1e-12:
                break
            a_test = a.copy()
            a_test[src] -= delta
            a_test[dst] += delta
            a_test = np.maximum(a_test, 0.0)
            s = a_test.sum()
            if s < 1e-15:
                break
            a_test /= s
            conv_test = fast_convolve(a_test)
            if conv_test.max() < best_obj - 1e-14:
                a = a_test
                conv = conv_test
                best_obj = conv.max()
            else:
                break

    return a, best_obj


def multi_cd(a, n, max_rounds=20, top_k_peaks=5):
    """Multi-peak coordinate descent considering top k peaks simultaneously."""
    a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()

    for rnd in range(max_rounds):
        peaks = np.argpartition(conv, -top_k_peaks)[-top_k_peaks:]
        improved = False
        for k in peaks:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end - j_start < 1:
                continue
            # Find best (src, dst) pair for this peak
            best_delta_obj = 0
            best_src, best_dst = -1, -1
            for src in range(j_start, j_end + 1):
                if a[src] < 1e-8:
                    continue
                for dst in range(j_start, j_end + 1):
                    if dst == src:
                        continue
                    # Approximate change: delta * (a[k-dst] - a[k-src])
                    delta_approx = a[src] * 0.01 * (a[k - src] - a[k - dst])
                    if delta_approx < best_delta_obj:
                        best_delta_obj = delta_approx
                        best_src, best_dst = src, dst

            if best_src < 0:
                continue

            delta = a[best_src] * 0.01
            a_test = a.copy()
            a_test[best_src] -= delta
            a_test[best_dst] += delta
            a_test = np.maximum(a_test, 0.0)
            s = a_test.sum()
            if s < 1e-15:
                continue
            a_test /= s
            conv_test = fast_convolve(a_test)
            if conv_test.max() < best_obj - 1e-14:
                a = a_test
                conv = conv_test
                best_obj = conv.max()
                improved = True

        if not improved:
            break

    return a, best_obj


def gen_initializations(n, parent=None, rng=None):
    """Generate diverse initializations for sequence of length n."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    edge = np.zeros(n)
    e = max(1, n // 5)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()

    sigma = 0.25 * n
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
    inits['gauss'] = g / g.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    inits['random2'] = rng.dirichlet(np.ones(n) * 1.5)

    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_w = np.maximum(cos_w, 0.01)
    inits['cosine'] = cos_w / cos_w.sum()

    # Bimodal
    g2 = npexp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2) + \
         npexp(-0.5 * ((x - 0.8 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()

    # Power law (heavy at edges)
    pw = (1.0 / (x + 1) + 1.0 / (n - x + 1))
    inits['power'] = pw / pw.sum()

    # Alternating
    alt = np.ones(n)
    alt[::2] = 2.0
    inits['alternating'] = alt / alt.sum()

    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            else:
                p_res = np.interp(np.linspace(0, 1, n),
                                  np.linspace(0, 1, len(parent)), p)
                p_res = np.maximum(p_res, 0)
                p_rs = p_res.sum()
                if p_rs > 1e-10:
                    inits['parent_res'] = p_res / p_rs
                # Perturbed parent
                for psigma in [0.01, 0.03]:
                    pp = p_res + rng.standard_normal(n) * psigma * p_rs
                    pp = np.maximum(pp, 0)
                    pps = pp.sum()
                    if pps > 1e-10:
                        inits[f'parent_p{psigma}'] = pp / pps

    return inits


def npexp(x):
    return np.exp(x)


def optimize_lp_ls(a_init, time_limit, use_ls=True, n_cache=None):

    while time.monotonic() - t0 < time_limit:

        if M < 1e-15:

def optimize_gradient(a_init, time_limit, lr_init=0.003, momentum=0.9):
    """Pure gradient descent on the simplex with momentum."""
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        grad = gradient_max_conv(a, conv, n, top_k=10)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        conv = fast_convolve(a_new)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                a = project_simplex(a + np.random.default_rng(
                    int(time.monotonic() * 1000)).standard_normal(n) * 0.005)
                conv = fast_convolve(a)
                v *= 0.5
                no_improve = 0

        lr *= 0.9995
        if lr < 1e-6:
            break

    return best_a, best_obj

    print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: Deep optimization on top candidates
    print("=== Phase 2: Deep Optimization ===")
    top_list = []
    seen = set()
    for score, n, name, a_opt in results[:6]:
        key = (n, name)
        if key not in seen:
            top_list.append((n, a_opt))
            seen.add(key)
    if best_sequence is not None:
        top_list.append((len(best_sequence), best_sequence.copy()))

    remaining = deadline - time.monotonic() - 250.0
    t_per = max(30.0, remaining / max(1, len(top_list)))

    for n, a0 in top_list[:5]:
        if time.monotonic() > deadline - 220:
            break
        t_alloc = min(t_per, deadline - time.monotonic() - 180)
        if t_alloc < 15:
            break

        a_opt, obj_opt = optimize_lp_ls(a0, t_alloc * 0.6, use_ls=True)

        # Gradient refinement
        t_grad = min(t_alloc * 0.25, 30.0)
        if t_grad > 3:
            a_opt2, obj2 = optimize_gradient(a_opt, t_grad, lr_init=0.002,
                                             momentum=0.92)
            if obj2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj2

        # Coordinate descent
        a_cd, obj_cd = greedy_cd(a_opt, n, max_steps=300, step_frac=0.015)
        if obj_cd < obj_opt:
            a_opt, obj_opt = a_cd, obj_cd

        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep n={n}: {score:.6f}")

    # Phase 3: Intensive polishing
    print("=== Phase 3: Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 80.0
        if remaining > 30:
            # Multi-peak CD
            a_mp, obj_mp = multi_cd(best_sequence, len(best_sequence),
                                    max_rounds=30, top_k_peaks=5)
            score_mp = float(evaluate_sequence(a_mp.tolist()))
            if np.isfinite(score_mp) and score_mp < best_score:
                best_score = score_mp
                best_sequence = a_mp.copy()
                print(f"  MultiCD: {score_mp:.6f}")

        # LP cycles with decreasing time
        remaining = deadline - time.monotonic() - 40.0
        cycles = 4
        for cyc in range(cycles):
            if time.monotonic() > deadline - 35:
                break
            t_cyc = min(remaining / (cycles - cyc), 40.0)
            if t_cyc < 5:
                break

            a_opt, obj_opt = optimize_lp_ls(best_sequence, t_cyc, use_ls=True)

            # Quick gradient step
            if t_cyc > 8:
                a_opt2, obj2 = optimize_gradient(a_opt, t_cyc * 0.2,
                                                 lr_init=0.001, momentum=0.93)
                if obj2 < obj_opt:
                    a_opt, obj_opt = a_opt2, obj2

            # Quick CD
            a_cd, obj_cd = greedy_cd(a_opt, len(a_opt), max_steps=150,
                                     step_frac=0.01)
            if obj_cd < obj_opt:
                a_opt, obj_opt = a_cd, obj_cd

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Cycle {cyc}: {score:.6f}")

        # Final perturbations
        if time.monotonic() < deadline - 20:
            n_b = len(best_sequence)
            for ps in [0.002, 0.005, 0.01]:
                if time.monotonic() > deadline - 12:
                    break
                for _ in range(3):
                    if time.monotonic() > deadline - 10:
                        break
                    noise = rng.standard_normal(n_b) * ps
                    a_p = project_simplex(best_sequence + noise)
                    t_rem = deadline - time.monotonic() - 5
                    if t_rem < 3:
                        break
                    a_opt, _ = optimize_lp_ls(a_p, min(5.0, t_rem), use_ls=False)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Perturb({ps}): {score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
