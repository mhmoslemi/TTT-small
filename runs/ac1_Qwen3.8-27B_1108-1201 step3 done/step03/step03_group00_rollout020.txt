import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(p, conv, top_k, temperature):
    """Compute weighted subgradient focusing on top peaks."""
    n = len(p)
    sg = np.zeros(n)
    
    # Get top-k peaks
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    # Softmax weights
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # For each peak at lag k, the subgradient contribution to p[j] is p[k-j]
    for i in range(len(peaks)):
        k = int(peaks[i])
        if w[i] < 1e-15:
            continue
        # p[j] * p[k-j] where j is valid
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            sg[j] += w[i] * p[k - j]
    
    return sg

def mwd_optimize(f0, time_limit, verbose=False):
    """Mirror Descent (multiplicative weight) optimization with adaptive parameters."""
    p = np.maximum(f0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s
    n = len(p)
    
    conv = fast_convolve(p)
    best_obj = conv.max()
    best_p = p.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    lr = 0.5
    top_k = 20
    temp = 0.0005
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient
        sg = compute_subgradient(p, conv, top_k, temp)
        
        # Multiplicative update
        p_new = p * np.exp(-lr * sg)
        p_new_sum = p_new.sum()
        if p_new_sum < 1e-15:
            p_new = p.copy()
        else:
            p_new = p_new / p_new_sum
        
        # Check improvement
        conv_new = fast_convolve(p_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            p = p_new
            conv = conv_new
            best_obj = obj_new
            best_p = p.copy()
            no_improve = 0
        elif obj_new < conv.max() + 1e-10:
            # Accept even if not improving best (for exploration)
            p = p_new
            conv = conv_new
            no_improve += 1
        else:
            no_improve += 1
        
        # Adaptive parameters
        lr *= 0.9997
        temp *= 1.002
        top_k = max(5, min(100, int(20 + no_improve * 0.5)))
        
        # Random restart on stagnation
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 100000)
            noise = rng.standard_normal(n) * 0.01
            p = project_simplex(best_p + noise)
            conv = fast_convolve(p)
            no_improve = 0
            lr = 0.3
            temp = 0.001
        
        if verbose and no_improve % 500 == 0:
            print(f"  MWD: obj={best_obj:.10f}, no_improve={no_improve}")
    
    return best_p, best_obj

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Structured initialization (difference-set inspired + parent)
2. Fast subgradient optimization with adaptive peak tracking
3. Simulated annealing for global exploration
4. Fine polishing with exact line search
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_weighted_subgradient(p, conv, top_k, temperature):
    """Compute softmax-weighted subgradient focusing on top peaks."""
    n = len(p)
    sg = np.zeros(n)
    
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = peaks[i]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        # subgradient contribution: 2 * w[i] * p[k - j] for j in [js, je]
        sg[idx] += 2.0 * w[i] * p[k - idx]
    
    return sg

def sg_descent(f0, time_limit, lr=0.1, top_k=30, temp=0.0005, 
               mom=0.9, verbose=False, label=""):
    """Subgradient descent with momentum and adaptive parameters."""
    p = np.maximum(f0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s
    n = len(p)
    
    conv = fast_convolve(p)
    best_obj = conv.max()
    best_p = p.copy()
    
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp
    cur_top_k = top_k
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_weighted_subgradient(p, conv, cur_top_k, cur_temp)
        v = mom * v - cur_lr * sg
        p_new = project_simplex(p + v)
        
        conv_new = fast_convolve(p_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            p = p_new
            conv = conv_new
            best_obj = obj_new
            best_p = p.copy()
            no_improve = 0
        elif obj_new < conv.max() + 1e-12:
            p = p_new
            conv = conv_new
            no_improve += 1
        else:
            no_improve += 1
            # Backtrack
            v *= 0.5
            cur_lr *= 0.8
        
        # Adaptive parameters
        if no_improve % 50 == 0:
            cur_lr = max(lr * 0.5, 0.001) * (1 + no_improve * 0.01)
            cur_temp *= 1.01
            cur_top_k = min(n, cur_top_k + 1)
        
        if no_improve > 300:
            break
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter {it} ({elapsed:.0f}s): obj={best_obj:.10f}")
    
    return best_p, best_obj

def sa_optimize(f0, time_limit, n_population=8, verbose=False):
    """Simulated annealing with population-based restarts."""
    p = np.maximum(f0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s
    n = len(p)
    
    conv = fast_convolve(p)
    best_obj = conv.max()
    best_p = p.copy()
    
    t0 = time.monotonic()
    temp = 0.01
    min_temp = 0.0001
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Generate perturbation
        scale = temp * 0.1
        noise = np.random.standard_normal(n) * scale
        p_new = project_simplex(p + noise)
        
        conv_new = fast_convolve(p_new)
        obj_new = conv_new.max()
        delta = obj_new - conv.max()
        
        # Metropolis acceptance
        if delta < 0 or np.random.random() < np.exp(-delta / max(temp, 1e-10)):
            p = p_new
            conv = conv_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_p = p.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Cool down
        if no_improve > 100:
            temp *= 0.9
            if temp < min_temp:
                # Restart from best
                p = best_p + np.random.standard_normal(n) * 0.005
                p = project_simplex(p)
                conv = fast_convolve(p)
                temp = 0.01
                no_improve = 0
            no_improve = 0
        
        if verbose and it % 5000 == 0:
            print(f"    SA iter {it}: obj={best_obj:.10f}, temp={temp:.6f}")
    
    return best_p, best_obj

def deflation_step(p, conv, top_k=5, strength=0.01):
    """Reduce the top peaks by directly adjusting p values."""
    n = len(p)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    
    p_new = p.copy()
    for k in peaks:
        # Find the j that contributes most to peak k
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        products = p[js:je+1] * p[k - (js:je+1)]
        max_j = js + np.argmax(products)
        
        # Reduce p at the position that contributes most
        # Use the "cheaper" of the two positions
        partner = k - max_j
        if p[max_j] > p[partner]:
            reduce_idx = partner
        else:
            reduce_idx = max_j
        
        p_new[reduce_idx] *= (1 - strength)
    
    p_new = np.maximum(p_new, 0)
    ps = p_new.sum()
    if ps > 1e-15:
        p_new /= ps
    return p_new

def structured_init(n, rng, parent=None):
    """Generate diverse structured initializations."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Parent (if available)
    if parent is not None:
        pp = parent / parent.sum()
        inits.append(pp)
    
    # Bimodal
    x = np.arange(n, dtype=np.float64)
    b = np.exp(-0.5 * ((x - n*0.25) / (n*0.1))**2) + np.exp(-0.5 * ((x - n*0.75) / (n*0.1))**2)
    inits.append(b / b.sum())
    
    # Quadratic residue inspired (for n near a prime)
    # Use a simple pseudorandom structure
    q = rng.permutation(n)
    qr = np.zeros(n)
    qr[q[:n//2]] = 1.0
    # Smooth it
    from scipy.ndimage import uniform_filter1d
    qr_smooth = uniform_filter1d(qr, size=max(2, n//50))
    qr_smooth = np.maximum(qr_smooth, 0.01)
    inits.append(qr_smooth / qr_smooth.sum())
    
    # Low-discrepancy (Halton-like)
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 / ((i % 7) + 1.0) if (i // 7) % 2 == 0 else 1.0 / ((i % 11) + 1.0)
    inits.append(h / h.sum())
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - n/2) / (n*0.3))**2)
    inits.append(g / g.sum())
    
    # Edge concentrated
    edge = 1.0 / (np.minimum(x, n-1-x) + 1.0)
    inits.append(edge / edge.sum())
    
    # Random Dirichlet (various alphas)
    for alpha in [0.3, 0.5, 1.0, 2.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha))
    
    return inits

            products = p[idx] * p[k - idx]
            j_max = idx[np.argmax(products)]
            partner = k - j_max
            # Reduce the smaller one (less impact on other peaks)
            if p[j_max] >= p[partner]:
                p[partner] *= (1.0 - strength)
            else:
                p[j_max] *= (1.0 - strength)

        p = np.maximum(p, 0.0)
        ps = p.sum()
        if ps < 1e-15:
            p = best_p.copy()
        else:
            p = p / ps

        conv = fast_convolve(p)
        if conv.max() < best_obj - 1e-14:
            best_obj = conv.max()
            best_p = p.copy()

        # Periodic SG refresh between deflations
        if it % n_deflate == 0:
            rem = time_limit - (time.monotonic() - t0)
            if rem > 5:
                p, _ = adaptive_sg(p, time_limit=min(rem * 0.3, 20),
                                    lr=0.2, top_k=10, temp=0.0005, mom=0.9,
                                    sa_temp=0.005, label="def_sg")
                best_p = p.copy()
                conv = fast_convolve(p)
                best_obj = conv.max()

    return best_p, best_obj


def lp_direction(f, M):
    """Solve reduced LP: max sum(g) s.t. (f*g)[k] <= M for top constraints, g >= 0.
    Returns direction g or None."""
    n = len(f)
    # Only use top constraints to keep LP small
    fg = fast_convolve(f)
    # For the LP, we need (f*g)[k] <= M for all k
    # But we'll only enforce top-K to make it faster
    top_k_constraints = min(50, 2 * n - 1)
    peaks = np.argpartition(fg, -top_k_constraints)[-top_k_constraints:]

    from scipy.optimize import linprog
    c = -np.ones(n)
    A_ub = np.zeros((len(peaks), n))
    for i, k in enumerate(peaks):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[i, j] = f[k - j]
    b_ub = np.full(len(peaks), M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def fw_lp_optimize(f0, time_limit, verbose=False, label=""):
    """Frank-Wolfe with reduced LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = lp_direction(f, M * m_exp)
        if g is None:
            m_exp *= 1.2
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-9:
            no_improve += 1
            if no_improve > 30:
                m_exp = min(m_exp * 1.05, 3.0)
                continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.3, 0.5, 0.7, 0.9]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            rng = np.random.default_rng(it * 7 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
            m_exp = 1.0

        if verbose and it % 10 == 0:
            print(f"    {label} LP it={it}: obj={best_obj:.10f}")

    return best_f, best_obj


def structured_init(n, rng, parent=None):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Parent
    if parent is not None:
        inits.append(parent.astype(np.float64).copy())

    # Bimodal (two Gaussian bumps)
    b = np.exp(-0.5 * ((x - n * 0.2) / (n * 0.08)) ** 2) + \
        np.exp(-0.5 * ((x - n * 0.8) / (n * 0.08)) ** 2)
    inits.append(b / b.sum())

    # Triangular (high in middle)
    t = 1.0 - 2.0 * np.abs(x - n / 2) / n
    t = np.maximum(t, 0.01)
    inits.append(t / t.sum())

    # Edge-concentrated
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())

    # Pseudorandom (smoothed permutation)
    perm = rng.permutation(n).astype(np.float64)
    # Smooth with a window
    win = max(3, n // 40)
    if win % 2 == 0:
        win += 1
    kernel = np.ones(win) / win
    smooth = np.convolve(perm, kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())

    # Dirichlet with low alpha (sparse)
    inits.append(rng.dirichlet(np.ones(n) * 0.2))
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    inits.append(rng.dirichlet(np.ones(n) * 1.0))

    # Periodic with jitter
    period = max(2, n // 8)
    per = np.tile([1.0 if (i % period) < period // 2 else 0.3 for i in range(period)],
                  n // period + 1)[:n]
    per += rng.standard_normal(n) * 0.1
    per = np.maximum(per, 0.01)
    inits.append(per / per.sum())

    # Cosine-weighted
    c = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c / c.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining SG, deflation, LP-FW,
    and simulated annealing for global exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p_arr = np.asarray(height_sequence_1, dtype=np.float64)
        p_arr = np.clip(p_arr, 0.0, 1000.0)
        if p_arr.ndim == 1 and p_arr.size > 0 and np.isfinite(p_arr).all() and p_arr.sum() > 1e-10:
            parent = p_arr / p_arr.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    n = 384
    if parent is not None:
        n = len(parent)

    # === Phase 1: Quick exploration of diverse initializations ===
    print(f"=== Phase 1: Exploration (n={n}) ===")

    inits = structured_init(n, rng, parent)

    for i, f_init in enumerate(inits):
        if time.monotonic() > deadline - 400:
            break
        t_alloc = min(35, deadline - time.monotonic() - 350)
        if t_alloc < 8:
            break
        a_opt, _ = adaptive_sg(f_init, time_limit=t_alloc, lr=0.4, top_k=15,
                               temp=0.001, mom=0.9, sa_temp=0.02,
                               label=f"exp{i}", rng=rng)
        try_seq(a_opt, f"exp{i}")

    print(f"  Phase 1 best: {best_score:.6f}")

    # === Phase 2: Focus on best with multiple strategies ===
    print(f"\n=== Phase 2: Deep Optimization ===")

    if best_seq is not None:
        best_p = best_seq / best_seq.sum()

        # Strategy A: LP-based FW (globally informed)
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            print(f"  Running LP-FW, t={min(rem, 300):.0f}s")
            a_opt, _ = fw_lp_optimize(best_p, time_limit=min(rem, 300), label="fw")
            try_seq(a_opt, "fw")

        # Strategy B: Extended SG with different params
        for cfg in [
            {'lr': 0.2, 'top_k': 10, 'temp': 0.0003, 'mom': 0.92, 'sa_temp': 0.01},
            {'lr': 0.5, 'top_k': 50, 'temp': 0.002, 'mom': 0.85, 'sa_temp': 0.03},
            {'lr': 0.1, 'top_k': 5, 'temp': 0.0001, 'mom': 0.95, 'sa_temp': 0.005},
        ]:
            rem = deadline - time.monotonic() - 200
            if rem < 20:
                break
            a_opt, _ = adaptive_sg(best_p, time_limit=min(rem, 120), rng=rng, **cfg)
            try_seq(a_opt, f"sg_lr{cfg['lr']}")

        # Strategy C: Deflation
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            a_opt, _ = deflation_boost(best_p, time_limit=min(rem, 120),
                                       n_deflate=15, strength=0.02)
            try_seq(a_opt, "deflate")

        # Strategy D: Perturbed restarts + SG
        for trial in range(4):
            rem = deadline - time.monotonic() - 100
            if rem < 15:
                break
            scales = [0.002, 0.005, 0.01, 0.02]
            noise = rng.standard_normal(n) * scales[trial]
            p_pert = project_simplex(best_p + noise)
            a_opt, _ = adaptive_sg(p_pert, time_limit=min(rem, 80), lr=0.3,
                                   top_k=20, temp=0.001, mom=0.9,
                                   sa_temp=0.01, rng=rng, label=f"pert{trial}")
            try_seq(a_opt, f"pert{trial}")

    # === Phase 3: Try different lengths ===
    print(f"\n=== Phase 3: Length Exploration ===")
    current_best_n = len(best_seq) if best_seq is not None else n
    other_lengths = [512, 448, 400, 256]
    for ol in other_lengths:
        if ol == current_best_n:
            continue
        rem = deadline - time.monotonic() - 80
        if rem < 30:
            break
        t_alloc = min(40, rem - 40)
        if t_alloc < 15:
            break
        # Interpolate parent to new length if available
        if parent is not None:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, ol)
            p_interp = np.interp(x_new, x_old, parent)
            p_interp = np.maximum(p_interp, 0)
            ps = p_interp.sum()
            if ps > 1e-10:
                p_interp /= ps
        else:
            p_interp = np.ones(ol) / ol
        a_opt, _ = adaptive_sg(p_interp, time_limit=t_alloc, lr=0.3,
                               top_k=20, temp=0.001, mom=0.9,
                               sa_temp=0.01, rng=rng, label=f"n{ol}")
        try_seq(a_opt, f"n{ol}")

    # === Phase 4: Final polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"  Current best: {best_score:.6f}")

    if best_seq is not None:
        best_p = best_seq / best_seq.sum()
        n_best = len(best_seq)

        # Very fine SG
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = adaptive_sg(best_p, time_limit=min(rem, 60), lr=0.02,
                                   top_k=3, temp=0.00005, mom=0.96,
                                   sa_temp=0.001, rng=rng, label="fine")
            try_seq(a_opt, "fine")

        # LP polish
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            a_opt, _ = fw_lp_optimize(best_p, time_limit=min(rem, 40), label="lp_fin")
            try_seq(a_opt, "lp_fin")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            for _ in range(3):
                rem = deadline - time.monotonic() - 3
                if rem < 5:
                    break
                noise = rng.standard_normal(n_best) * 0.0005
                p_pert = project_simplex(best_p + noise)
                a_opt, _ = adaptive_sg(p_pert, time_limit=min(rem, 15), lr=0.01,
                                       top_k=2, temp=0.00002, mom=0.97,
                                       sa_temp=0.0005, rng=rng)
                try_seq(a_opt, "micro")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384

    # Final verification
    best_seq = np.maximum(best_seq, 0.0)
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
