def gradient_descent_smooth(a0, beta, lr, n_iter, momentum=0.9):

def frequency_init(n, rng, target_flatness=0.9):
    """Initialize with flat power spectrum (minus DC)."""
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    X = np.zeros(fft_size, dtype=complex)
    
    # DC component
    X[0] = np.sqrt(n)  # So that sum(a) = sqrt(n) * sqrt(n) / fft_size... 
    # Actually, let's think about normalization.

    # But |FFT(a)|^2 at DC = 1, and the autocorrelation at lag 0 = sum(a_i^2).
    # By Parseval: sum_k |FFT(a)_k|^2 / fft_size = sum_i a_i^2

    X[0] = 1.0  # sum(a) = 1 when we take IFFT and scale

        mag = rng.uniform(0.01, 0.1)
        phase = rng.uniform(0, 2 * np.pi)
        X[k] = mag * np.exp(1j * phase)

def msequence_init(n, rng):
    """Initialize from an m-sequence-like pattern."""
    # For n close to 2^m - 1, use the actual m-sequence
    # Otherwise, use a chaotic sequence
    m = int(np.ceil(np.log2(n + 1)))
    seq_size = 2**m - 1
    
    # Generate m-sequence using LFSR
    # Use a primitive polynomial for GF(2)^m
    # For simplicity, use a known primitive polynomial
    primitive_pols = {
        2: 0b11, 3: 0b101, 4: 0b1001, 5: 0b10011,
        6: 0b10101, 7: 0b1000111, 8: 0b10011011,
        9: 0b100010001, 10: 0b1001000101,
    }
    
    if m in primitive_pols and m <= 10:
        poly = primitive_pols[m]
        state = 1  # Non-zero initial state
        seq = []
        for _ in range(seq_size):
            seq.append(state & 1)
            # LFSR step
            feedback = 0
            for i in range(m):
                if (poly >> i) & 1:
                    feedback ^= (state >> i) & 1
            state = (state >> 1) | (feedback << (m - 1))
        
        # Convert to real sequence: map 0 -> 0, 1 -> 1
        # Then truncate or pad to length n
        a = np.array(seq[:n], dtype=np.float64) if n <= seq_size else np.tile(seq, n // seq_size + 1)[:n]
        a = np.maximum(a, 0.01)  # Avoid zeros
        a /= a.sum()
        return a
    
    # Fallback: use logistic map
    x = rng.uniform(0.5, 1.0)
    a = np.zeros(n)
    for i in range(n):
        x = 4 * x * (1 - x)
        a[i] = x
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a

def generate_initializations(n, parent, rng):

def coordinate_descent(a, n_iters=100, n_tries_per_coord=5):
    """Coordinate descent refinement."""
    n = len(a)
    a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    for it in range(n_iters):
        improved = False
        # Randomize coordinate order for diversity
        order = np.random.permutation(n)
        for i in order:
            old_val = a[i]
            if old_val < 1e-10:
                continue
            # Try different multipliers
            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                a[i] = old_val * mult
                if a[i] < 0:
                    a[i] = 0
                conv = fast_convolve(a)
                obj = 2.0 * n * conv.max()
                if obj < best_obj:
                    best_obj = obj
                    improved = True
                    break
            if not improved:
                a[i] = old_val
                improved = False
        
        # Renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a /= s
    
    return a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using PEGASUS-style multiplicative updates,
Nesterov-accelerated projected gradient descent, frequency-domain and m-sequence
initializations, multi-start over lengths, and coordinate descent refinement.

import numpy as np
import time
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights with numerical stability."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -200, 200)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, w):
    """Compute gradient of sum_k w_k (a*a)_k w.r.t. a using FFT."""
    n = len(a)
    N = 1 << (n + len(w) - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    corr = np.real(ifft(np.conj(W) * A, N))[:n]
    return 2.0 * corr


def pegasus_update(a, beta, alpha):
    """PEGASUS-style multiplicative update to flatten autocorrelation."""
    n = len(a)
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    g = compute_gradient(a, w)
    g = np.maximum(g, 1e-15)
    # Multiplicative weight update: reduce coordinates contributing to peaks
    update = (1.0 / g) ** alpha
    a_new = a * update
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = np.ones(n) / n
    else:
        a_new /= s
    return a_new


def pegasus_optimize(a0, beta_init, beta_min, n_iter, alpha, time_limit, t_start):
    """PEGASUS optimization with beta annealing."""
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    for it in range(n_iter):
        if time.monotonic() - t_start > time_limit:
            break

        progress = it / max(1, n_iter - 1)
        beta = beta_init * (beta_min / beta_init) ** progress

        a = pegasus_update(a, beta, alpha)

        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()

        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        # Occasionally add small perturbation to escape local minima
        if it % 200 == 100:
            noise_scale = 0.01 * (1.0 - progress)
            if noise_scale > 0.001:
                pert = np.random.default_rng(it).standard_normal(n) * noise_scale
                a_pert = project_simplex(a + pert)
                conv_pert = fast_convolve(a_pert)
                if 2.0 * n * conv_pert.max() < best_obj:
                    best_obj = 2.0 * n * conv_pert.max()
                    best_a = a_pert.copy()
                    a = a_pert.copy()

    return best_a, best_obj


def gradient_descent_smooth(a0, beta, lr, n_iter, momentum):
    """Nesterov-accelerated projected gradient descent on simplex."""
    n = len(a0)
    a = np.maximum(a0, 1e-15).copy()
    a /= a.sum()
    v = np.zeros(n)

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    for it in range(n_iter):
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

        g = compute_gradient(a, w)
        v = momentum * v + g
        a_new = project_simplex(a - lr * v)

        # Backtracking
        obj_new = 2.0 * n * fast_convolve(a_new).max()
        if obj_new > best_obj * 1.05:
            a_new = project_simplex(a - 0.5 * lr * v)
            obj_new = 2.0 * n * fast_convolve(a_new).max()
            if obj_new > best_obj * 1.02:
                a_new = a.copy()

        a = a_new
        lr *= 0.9995

    return best_a, best_obj


def frequency_init(n, rng):
    """Initialize with approximately flat power spectrum."""
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    X = np.zeros(fft_size, dtype=complex)
    X[0] = 1.0
    for k in range(1, fft_size // 2 + 1):
        mag = rng.uniform(0.001, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        X[k] = mag * np.exp(1j * phase)
        if k < fft_size - k:
            X[fft_size - k] = mag * np.exp(-1j * phase)
    a = np.real(ifft(X, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def frequency_init_concentrated(n, rng):
    """Initialize with power concentrated at DC (strong low-pass)."""
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    X = np.zeros(fft_size, dtype=complex)
    X[0] = 1.0
    for k in range(1, fft_size // 2 + 1):
        # Exponentially decaying magnitude
        mag = 0.01 * np.exp(-0.5 * k / max(1, n // 4))
        phase = rng.uniform(0, 2 * np.pi)
        X[k] = mag * np.exp(1j * phase)
        if k < fft_size - k:
            X[fft_size - k] = mag * np.exp(-1j * phase)
    a = np.real(ifft(X, fft_size))[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def msequence_init(n, rng):
    """Initialize from m-sequence-like binary pattern, then optimize values."""
    m = max(2, int(np.ceil(np.log2(n + 1))))
    primitive_pols = {
        2: [1, 1], 3: [1, 0, 1], 4: [1, 0, 0, 1], 5: [1, 0, 0, 1, 1],
        6: [1, 0, 1, 0, 1], 7: [1, 0, 0, 0, 1, 1, 1],
        8: [1, 0, 0, 1, 1, 0, 1, 1], 9: [1, 0, 0, 0, 1, 0, 0, 0, 1],
        10: [1, 0, 0, 1, 0, 0, 0, 1, 0, 1],
    }

    if m in primitive_pols and m <= 10:
        taps = primitive_pols[m]
        state = 1
        seq = []
        for _ in range(2**m - 1):
            seq.append(1.0 if (state & 1) else 0.3)
            bit = 0
            for i in range(m):
                if taps[i]:
                    bit ^= (state >> i) & 1
            state = (state >> 1) | (bit << (m - 1))
        a = np.array(seq[:n], dtype=np.float64) if n <= len(seq) else \
            np.tile(seq, (n // len(seq)) + 1)[:n]
    else:
        # Logistic map fallback
        x = rng.uniform(0.6, 0.9)
        a = np.zeros(n)
        for i in range(n):
            x = 4.0 * x * (1.0 - x)
            a[i] = 0.3 + 0.7 * x
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a


def chaotic_init(n, rng):
    """Initialize using logistic map chaos."""
    x = rng.uniform(0.4, 0.9)
    a = np.zeros(n)
    for i in range(n):
        x = 4.0 * x * (1.0 - x)
        a[i] = x
    a = np.maximum(a, 0.005)
    a /= a.sum()
    return a


def costas_like_init(n, rng):
    """Initialize using a Costas-array-inspired pattern (Ward construction)."""
    # Ward's construction: a(i) = i*p mod N + 1, where N is prime
    # We'll use this to create a permutation, then soften it
    p = max(2, int(1.5 * n))
    # Find a prime near p
    def is_prime(x):
        if x < 2:
            return False
        for d in range(2, int(np.sqrt(x)) + 1):
            if x % d == 0:
                return False
        return True
    while not is_prime(p):
        p += 1
    # Generate Costas permutation (subset of size n)
    perm = np.zeros(n)
    for i in range(n):
        perm[i] = (i * p) % p
    # Convert to a smooth sequence: use the permutation to create a peaked structure
    # then smooth it
    a = np.zeros(n)
    a[perm.astype(int) % n] = 1.0
    # Smooth with a small kernel
    kernel_size = max(1, n // 32)
    if kernel_size > 1:
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['freq_flat'] = frequency_init(n, rng)
    inits['freq_conc'] = frequency_init_concentrated(n, rng)
    inits['mseq'] = msequence_init(n, rng)
    inits['chaotic'] = chaotic_init(n, rng)
    inits['costas'] = costas_like_init(n, rng)

    # Gaussian kernels
    for sf in [0.2, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%d' % int(sf * 100)] = g

    # Bimodal
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.05)

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent'] = p / ps
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                          np.maximum(parent, 0))
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_interp'] = p / ps

    return inits


def coordinate_descent_refine(a, n_iters, rng):
    """Coordinate descent refinement."""
    n = len(a)
    a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()

    for it in range(n_iters):
        improved = False
        order = rng.permutation(n)
        for i in order:
            old_val = a[i]
            if old_val < 1e-12:
                # Try small increase
                a[i] = 1e-6
                conv = fast_convolve(a)
                obj = 2.0 * n * conv.max()
                if obj < best_obj:
                    best_obj = obj
                    improved = True
                else:
                    a[i] = 0.0
                continue
            # Try multipliers
            found = False
            for mult in [0.6, 0.8, 0.92, 1.08, 1.2, 1.5]:
                a[i] = old_val * mult
                conv = fast_convolve(a)
                obj = 2.0 * n * conv.max()
                if obj < best_obj:
                    best_obj = obj
                    improved = True
                    found = True
                    break
            if not found:
                a[i] = old_val
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
        if not improved:
            break

    return a, best_obj


def hybrid_refine(a, time_limit, t_start, rng):
    """Alternate between PEGASUS and gradient steps."""
    n = len(a)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    budget = time_limit

    while time.monotonic() - t_start < budget - 5:
        # PEGASUS phase
        t0 = time.monotonic()
        a_cur, obj1 = pegasus_optimize(
            best_a, beta_init=100.0, beta_min=3.0,

            t_start=time.monotonic()

            best_a = a_cur.copy()

def update_best(a, best_score, best_sequence, t_start):

    return best_score, best_sequence, (score if np.isfinite(score) else float('inf'))

                n_iter=int(t_alloc * 60), alpha=0.5,
                time_limit=t_alloc - 5, t_start=time.monotonic()
            )
            # Gradient polish
            if time.monotonic() < phase3_deadline - 15:
                a_opt2, _ = gradient_descent_smooth(
                    a_opt, beta=10.0, lr=0.02, n_iter=1000, momentum=0.9
                )
                if 2.0 * alt_n * fast_convolve(a_opt2).max() < 2.0 * alt_n * fast_convolve(a_opt).max():
                    a_opt = a_opt2

            best_score, best_sequence, _ = update_best(a_opt, best_score, best_sequence, t_start)

    # === Phase 4: Final polish ===
    print("=== Phase 4: Final Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0
        best_a_norm = best_sequence / best_sequence.sum()
        n_best = len(best_sequence)

        # Final PEGASUS with very low beta
        if remaining > 20:
            t_sub = min(remaining * 0.4, 60.0)
            a_opt, _ = pegasus_optimize(
                best_a_norm, beta_init=20.0, beta_min=1.0,
                n_iter=int(t_sub * 100), alpha=0.3,
                time_limit=t_sub, t_start=time.monotonic()
            )
            best_score, best_sequence, _ = update_best(a_opt, best_score, best_sequence, t_start)
            remaining = deadline - time.monotonic() - 15.0

        # Final gradient
        if remaining > 15:
            best_a_norm = best_sequence / best_sequence.sum()
            t_sub = min(remaining * 0.4, 60.0)
            a_opt, _ = gradient_descent_smooth(
                best_a_norm, beta=3.0, lr=0.005,
                n_iter=int(t_sub * 50), momentum=0.95
            )
            best_score, best_sequence, _ = update_best(a_opt, best_score, best_sequence, t_start)
            remaining = deadline - time.monotonic() - 15.0

        # Final coordinate descent
        if remaining > 10:
            best_a_norm = best_sequence / best_sequence.sum()
            a_opt, _ = coordinate_descent_refine(best_a_norm, int(min(100, remaining)), rng)
            best_score, best_sequence, _ = update_best(a_opt, best_score, best_sequence, t_start)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = min(best_score, final_score)

    print("\nFINAL SCORE: %.6f (n=%d, t=%.0fs)" % (best_score, len(best_sequence), time.monotonic() - t_start))
    return [float(x) for x in best_sequence]
