"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exponentiated gradient
(multiplicative weights) with softmax-weighted subgradients, combined
with CVXPY-based QCQP solving and local polishing.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient(a, c, temperature=0.001, top_k=None):
    """Compute softmax-weighted subgradient of max(c)."""
    n = len(a)
    len_c = len(c)
    
    if top_k is not None and top_k < len_c:
        # Only use top-k peaks
        indices = np.argpartition(c, -top_k)[-top_k:]
        peek = c[indices]
    else:
        indices = np.arange(len_c)
        peek = c
    
    # Softmax weights
    scaled = peek / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(peek)] = 1.0
    else:
        w /= ws
    
    # Compute subgradient
    sg = np.zeros(n)
    for i in range(len(indices)):
        if w[i] < 1e-15:
            continue
        k = int(indices[i])
        # ∂c[k]/∂a[j] = 2*a[k-j] if 0 <= k-j < n
        # So sg[j] += 2 * w[i] * a[k-j]
        # For j from max(0, k-n+1) to min(n-1, k):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # a[k-j] for j in [js, je] is a[k-js], a[k-js-1], ..., a[k-je]
        # i.e., reversed a[k-je : k-js+1]
        sg[js:je+1] += 2.0 * w[i] * a[k-je:k-js][::-1]
    
    return sg

    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    # Running average (for convergence guarantee)
    avg_a = a.copy().astype(np.float64)
    count = 0
    
    eta = eta0
    temp = temp0
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    skip = 1  # Compute full subgradient every 'skip' iterations
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute subgradient
        sg = compute_subgradient(a, c, temperature=temp, top_k=max(10, n//4))
        
        # Multiplicative update
        a_new = a * np.exp(-eta * sg / sg.max() if sg.max() > 0 else 0)
        # Numerical stability: clip very small values
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new
        
        # Check improvement
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        
        # Update running average
        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 1000 == 0:
                print(f"    EG iter {it}: obj={best_obj:.10f}, eta={eta:.6f}, temp={temp:.6f}")
        else:
            no_improve += 1
        
        a = a_new
        c = c_new
        
        # Decay parameters
        eta *= eta_decay
        temp *= temp_decay
        
        # Occasional check of the average
        if it % 500 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-12:
                best_obj = c_avg.max()
                best_a = avg_a.copy()
        
        # Stagnation handling
        if no_improve > 2000:
            # Reset with perturbation
            rng = np.random.default_rng(it)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            eta = eta0 * 0.1
            temp = temp0 * 0.1
            avg_a = a.copy()
            count = 0
    
    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit, max_rounds=100):
    """Coordinate descent: for each coordinate, find optimal value."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit and it < max_rounds:
        it += 1
        improved = False
        
        for i in range(n):
            # Try a few perturbations of a[i]
            # We want to minimize max(c) while keeping sum = 1
            # If we change a[i] by delta, we need to adjust others
            # Simple approach: try scaling a[i] and redistributing
            
            old_val = a[i]
            # Try multiplicative factors
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                a[i] = old_val * factor
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    continue
                # Renormalize
                a = a / s
                c = fast_convolve(a)
                obj = c.max()
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                else:
                    a = best_a.copy()
                    break
            a = best_a.copy()
        
        if not improved:
            break
    
    return best_a, best_obj


def solve_qcqp(a0, time_limit=120):

            cp.sum(x) == 1,
            x >= 0
        ]

        # Use cp.geo_mean or product
        
        for k in range(2 * n - 1):
            expr = cp.Constant(0)
            js = max(0, k - n + 1)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an exponentiated gradient
(multiplicative weights) method with softmax-weighted subgradients as the
primary optimizer, combined with CVXPY-based QCQP solving for exact polishing,
coordinate descent local search, and multi-scale exploration with diverse
initializations. The key insight is that the subgradient of max(c) at the
active lag k* is simply 2*flip(a) shifted by k*, enabling O(n) per-iteration
updates via FFT-convolution and elementwise multiplication.
"""

import numpy as np
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient(a, c, temperature, top_k):
    """Compute softmax-weighted subgradient of max(c) using top-k peaks."""
    n = len(a)
    len_c = len(c)

    if top_k is not None and top_k < len_c and top_k > 0:
        indices = np.argpartition(c, -top_k)[-top_k:]
        peek = c[indices]
    else:
        indices = np.arange(len_c)
        peek = c.copy()

    scaled = peek / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(peek)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-15:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # sg[j] += 2*w[k] * a[k-j] for j in [js, je]
        # a[k-j] for j=js..je is a[k-js], a[k-js-1], ..., a[k-je]
        # which is a[k-je : k-js+1] reversed
        sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]

    return sg


def exp_gradient_optimize(a0, time_limit, eta0=1.0, temp0=0.005,
                          temp_decay=0.9997, eta_decay=0.99995,
                          top_k_frac=0.25, verbose=False):
    """Exponentiated gradient / multiplicative weights on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(10, int(n * top_k_frac))

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    avg_a = a.copy().astype(np.float64)
    count = 0

    eta = eta0
    temp = temp0

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        sg = compute_subgradient(a, c, temperature=temp, top_k=top_k)
        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        # Multiplicative update: a_i *= exp(-eta * sg_i / sg_max)
        log_update = -eta * sg / sg_max
        log_update -= log_update.max()  # numerical stability
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 2000 == 0:
                print(f"    EG iter {it}: obj={best_obj:.10f}, eta={eta:.6f}, temp={temp:.6f}")
        else:
            no_improve += 1

        a = a_new
        c = c_new

        # Periodic check of running average
        if it % 1000 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-12:
                best_obj = c_avg.max()
                best_a = avg_a.copy()

        eta *= eta_decay
        temp *= temp_decay

        # Stagnation: restart with perturbation
        if no_improve > 3000:
            rng_local = np.random.default_rng(it * 31 + 7)
            a = best_a + rng_local.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            eta = eta0 * 0.1
            temp = max(temp, temp0 * 0.01)
            avg_a = a.copy()
            count = 0

    return best_a.copy(), best_obj


def local_search_polish(a0, time_limit, max_rounds=200):
    """Local search: try multiplicative perturbations on random coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and it < max_rounds:
        it += 1
        improved = False

        # Pick a random subset of coordinates to perturb
        n_perturb = max(1, n // 16)
        coords = rng.choice(n, size=n_perturb, replace=False)

        for i in coords:
            old_val = a[i]
            if old_val < 1e-15:
                continue

            for factor in [0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.0/0.5,
                           0.75, 1.0/0.75, 0.9, 1.0/0.9]:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    continue
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s2
                c2 = fast_convolve(a)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-12:
                    best_obj = obj2
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a = best_a.copy()
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break

    return best_a.copy(), best_obj

        rem = deadline - time.monotonic() - 5
        if rem > 3:
            a_opt, _ = local_search_polish(best_seq, time_limit=min(rem, 15),
                                            max_rounds=30)
            try_seq(a_opt, "local_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
