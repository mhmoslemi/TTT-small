"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved LP-based directional
optimization with adaptive peaking, multi-length search, and gradient polishing.
Key improvements over previous version:
- Larger sequence lengths (up to 1536)
- Better LP with sparse constraints
- Adaptive peak-weighted subgradient
- Cosine and Zadoff-Chu inspired initializations
- More aggressive multi-phase search

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k.
    Uses sparse matrix for efficiency."""
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception as e:
        pass
    return None


def optimize_lp_v2(f_init, time_limit, blend_init=0.4, blend_min=0.003,
                   blend_halve_every=12, verbose=False):
    """LP-based directional optimization with adaptive blending."""
    a = np.maximum(f_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max() / (a.sum()**2)
    best_a = a.copy()

    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    prev_obj = best_obj

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break

        g = solve_lp_direction(a, M, n)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max() / (a.sum()**2)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
            if verbose:
                print(f"    LP iter: obj={cur_obj:.6f}, blend={blend:.4f}")
        else:
            stall += 1

def optimize_subgrad_v2(a0, time_limit, lr_init=0.02, momentum=0.9,
                        top_k=50, temperature=0.01):
    """Peak-weighted subgradient descent on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max() / (a.sum()**2)
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Get top peaks with soft weighting
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        else:
            peaks = np.arange(len(conv))
        peak_vals = conv[peaks]
        pmax = peak_vals.max()
        if pmax < 1e-15:
            break

        # Soft-max weighting: heavier weight on higher peaks
        scaled = peak_vals / (temperature * pmax + 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = 2.0 * w[idx]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                sg[j] += wi * a[k - j]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max() / (a.sum()**2)

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            a = project_simplex(a + np.random.default_rng(no_improve).standard_normal(n) * 0.003)
            conv = fast_convolve(a)
            v *= 0.5
            no_improve = 0
            lr = max(0.001, lr * 0.5)

        lr *= 0.9998

    return best_a, best_obj


def optimize_gradient_softmax(a0, time_limit, lr_init=0.1):

    # (F*g)[k] = sum_j f[k-j] * g[j]
    # Q = F^T F
    # Build Q implicitly

        fg = np.convolve(f, g, mode='full')
        return np.dot(fg, fg)
    
    def obj_grad(g):

        # grad = 2 * F^T * (F*g) = 2 * convolve(f, fg, mode='valid') ...

def optimize_subgrad(a0, time_limit, lr_init=0.02, momentum=0.92,
                     top_k=50, temp=0.05):
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    no_imp = 0

    while time.monotonic() - t0 < time_limit:
        tk = min(top_k, len(conv))
        peaks = np.argpartition(conv, -tk)[-tk:]
        pv = conv[peaks]
        pmax = pv.max()
        if pmax < 1e-15:
            break
        scaled = pv / (temp * pmax + 1e-15)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            wi = 2.0 * w[idx]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                sg[j] += wi * a[k - j]
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 400:
            a = project_simplex(a + np.random.default_rng(no_imp * 7).standard_normal(n) * 0.003)
            conv = fast_convolve(a)
            v *= 0.3
            no_imp = 0
            lr = max(0.001, lr * 0.4)
        lr *= 0.9997
    return best_a, best_obj

    lr = lr_init
    v = np.zeros(n)
    while time.monotonic() - t0 < time_limit:
        sm = np.exp(x - x.max())
        a = sm / sm.sum()
        conv = fast_convolve(a)
        pmax = conv.max()
        if pmax < 1e-15:
            break
        z = alpha * conv / pmax
        z -= z.max()
        wts = np.exp(z)
        wts /= wts.sum()
        # grad_a[j] = 2 * sum_k wts[k] * a[k-j]
        # Use FFT: this is 2 * corr(wts, a) at lags 0..n-1
        fft_sz = 1 << (3 * n - 3).bit_length()
        Wr = np.fft.rfft(wts[::-1], fft_sz)
        Ar = np.fft.rfft(a, fft_sz)
        grad_a = 2.0 * np.fft.irfft(Wr * Ar, fft_sz)[n - 1:2 * n - 1]
        grad_x = grad_a - a * np.dot(a, grad_a)
        v = 0.9 * v - lr * grad_x
        x += v
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        lr *= 0.9998
        if lr < 0.01:
            break
    return best_a, best_obj


def gen_inits(n, parent=None, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uniform'] = np.ones(n) / n
    inits['cosine'] = (0.5 + 0.5 * np.cos(2 * np.pi * x / n)) ** 2
    inits['cosine'] /= inits['cosine'].sum()
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    e = max(1, n // 5)
    edge = np.zeros(n)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.35 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    # Zadoff-Chu inspired
    nzc = n // 2
    k = np.arange(nzc)
    zc = np.cos(np.pi * k * (k + 1) / nzc)
    inits['zc'] = np.abs(zc) / np.abs(zc).sum()
    # Bimodal
    g2 = np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.8 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = g2 / g2.sum()
    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1))) if n > 1 else np.ones(n)
    inits['hann'] = hann / hann.sum()
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_resample'] = pp / pps
    return inits


def resize_sequence(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    resampled = np.interp(x_new, x_old, np.maximum(a, 0))
    s = resampled.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return resampled / s


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
            if np.isfinite(par_score):
                best_score = par_score
                best_sequence = p / p.sum()
    except Exception:
        pass

    # Phase 1: Quick LP scan
    print("=== Phase 1: Quick LP Scan ===")
    t1_end = time.monotonic() + min(400.0, (deadline - time.monotonic()) * 0.30)
    lengths = [384, 512, 640, 768, 1024]
    results = []

    for n in lengths:
        if time.monotonic() > t1_end - 30:
            break
        inits = gen_inits(n, parent, rng)
        prio = ['parent', 'parent_resample', 'uniform', 'cosine', 'hann',
                'bimodal', 'random', 'triangle', 'zc', 'edge', 'gauss']
        names = [nm for nm in prio if nm in inits]
        names = names[:4]
        t_per = max(10.0, (t1_end - time.monotonic() - 20) / max(1, len(names)))

        for name in names:
            if time.monotonic() > t1_end - 15:
                break
            t_alloc = min(t_per, t1_end - time.monotonic() - 10)
            if t_alloc < 8:
                break
            a0 = inits[name]
            a_opt, obj_opt = optimize_lp(a0, t_alloc, blend_init=0.35,
                                         blend_min=0.01, halve_every=10)
            # Quick subgrad
            a_opt2, obj2 = optimize_subgrad(a_opt, min(5.0, t_alloc * 0.2),
                                            lr_init=0.015, momentum=0.9,
                                            top_k=30, temp=0.05)
            if obj2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj2
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                print(f"  n={n} {name}: {score:.6f}")
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()

    results.sort(key=lambda x: x[0])
    if results:
        print(f"Phase 1 best: {results[0][0]:.6f}")

    # Phase 2: Deep optimization on top candidates
    print("=== Phase 2: Deep Optimization ===")
    top_list = []
    seen = set()
    for score, n, name, a_opt in results[:6]:
        key = n
        if key not in seen:
            top_list.append((n, a_opt))
            seen.add(key)
        if len(top_list) >= 4:
            break
    if best_sequence is not None:
        top_list.append((len(best_sequence), best_sequence.copy()))

    t2_end = time.monotonic() + min(900.0, (deadline - time.monotonic()) * 0.50)
    t_per_cand = max(30.0, (t2_end - time.monotonic() - 20) / max(1, len(top_list)))

    for n, a0 in top_list:
        if time.monotonic() > t2_end - 30:
            break
        t_alloc = min(t_per_cand, t2_end - time.monotonic() - 20)
        if t_alloc < 20:
            break
        # Extended LP
        a_opt, obj_opt = optimize_lp(a0, t_alloc * 0.5,
                                     blend_init=0.2, blend_min=0.002,
                                     halve_every=15)
        # QP polish
        t_qp = min(30.0, t_alloc * 0.15)
        if time.monotonic() < t2_end - 25:
            a_opt2, obj2 = optimize_lp(a_opt, t_qp,
                                       blend_init=0.05, blend_min=0.001,
                                       halve_every=20, use_qp=True)
            if obj2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj2
        # Subgrad polish
        t_sg = min(40.0, t_alloc * 0.2)
        if time.monotonic() < t2_end - 20:
            a_opt3, obj3 = optimize_subgrad(a_opt, t_sg,
                                            lr_init=0.005, momentum=0.93,
                                            top_k=60, temp=0.02)
            if obj3 < obj_opt:
                a_opt, obj_opt = a_opt3, obj3
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep n={n}: {score:.6f}")

    # Phase 3: Final polish
    print("=== Phase 3: Final Polish ===")
    if best_sequence is not None:
        t3_end = deadline
        n_b = len(best_sequence)

        # Gradient-based polish
        t_grad = min(120.0, (t3_end - time.monotonic()) * 0.25)
        if t3_end - time.monotonic() > 30:
            a_g, obj_g = optimize_gradient(best_sequence, t_grad, alpha=25.0, lr_init=0.3)
            score_g = float(evaluate_sequence(a_g.tolist()))
            if np.isfinite(score_g) and score_g < best_score:
                best_score = score_g
                best_sequence = a_g.copy()
                print(f"  Gradient: {score_g:.6f}")

        # More LP cycles
        remaining = t3_end - time.monotonic() - 30
        for cyc in range(3):
            if time.monotonic() > t3_end - 25:
                break
            t_cyc = min(remaining / (3 - cyc), 80.0)
            if t_cyc < 10:
                break
            a_c, obj_c = optimize_lp(best_sequence, t_cyc * 0.6,
                                     blend_init=0.08, blend_min=0.001,
                                     halve_every=15)
            a_c2, obj_c2 = optimize_subgrad(a_c, t_cyc * 0.3,
                                            lr_init=0.003, momentum=0.94,
                                            top_k=80, temp=0.01)
            if obj_c2 < obj_c:
                a_c = a_c2
            score_c = float(evaluate_sequence(a_c.tolist()))
            if np.isfinite(score_c) and score_c < best_score:
                best_score = score_c
                best_sequence = a_c.copy()
                print(f"  Cycle {cyc}: {score_c:.6f}")
            remaining = t3_end - time.monotonic() - 30

        # Random perturbations
        if time.monotonic() < t3_end - 20:
            for ps in [0.002, 0.005, 0.01]:
                for _ in range(3):
                    if time.monotonic() > t3_end - 15:
                        break
                    noise = rng.standard_normal(n_b) * ps
                    a_p = project_simplex(best_sequence + noise)
                    a_p, _ = optimize_lp(a_p, 5.0, blend_init=0.1,
                                         blend_min=0.005, halve_every=8)
                    a_p2, obj_p2 = optimize_subgrad(a_p, 3.0,
                                                    lr_init=0.004, momentum=0.9,
                                                    top_k=40, temp=0.02)
                    if obj_p2 < 2.0 * n_b * fast_convolve(a_p).max():
                        a_p = a_p2
                    score_p = float(evaluate_sequence(a_p.tolist()))
                    if np.isfinite(score_p) and score_p < best_score:
                        best_score = score_p
                        best_sequence = a_p.copy()
                        print(f"  Perturb({ps}): {score_p:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
