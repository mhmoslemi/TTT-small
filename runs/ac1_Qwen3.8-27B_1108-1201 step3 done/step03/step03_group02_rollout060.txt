```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))² using softmax-parameterized 
optimization with L-BFGS-B, projected gradient with Nesterov momentum,
and adaptive log-sum-exp smoothing. Multi-start with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def grad_wrt_a(a, p):
    """Compute d/d a of sum_k p[k] * conv(a,a)[k]."""
    n = len(a)
    # grad[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m] (where out-of-range p is 0)
    # Using the fact that d(conv_k)/d a[j] = 2*a[k-j] if 0<=k-j<n
    grad = np.zeros(n)
    for j in range(n):
        m_max = min(n - 1, 2 * n - 2 - j)
        if m_max >= 0:
            grad[j] = 2.0 * np.dot(p[j:j + m_max + 1], a[0:m_max + 1])
    return grad


def smooth_obj_and_grad_x(x, t):
    """
    Compute smoothed objective and gradient w.r.t. softmax parameter x.
    f(a) = (2n/t) * log(sum_k exp(t * conv(a,a)[k]))
    a = softmax(x)
    """
    a = softmax_fn(x)
    n = len(a)
    conv = fast_convolve(a)
    
    scaled = t * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp
    
    # Smoothed objective
    obj = (2.0 * n / t) * (np.log(sum_exp) + t * conv.max())
    
    # Gradient w.r.t. a: d f / d a[j] = 2n * sum_k p[k] * d(conv_k)/d a[j]
    # = 2n * 2 * sum_{m: j+m in range} p[j+m] * a[m]
    grad_a = 2.0 * n * grad_wrt_a(a, p)
    
    # Gradient w.r.t. x: d f / dx = (I - a a^T) grad_a
    grad_x = grad_a - a * np.dot(a, grad_a)
    
    return obj, grad_x


def true_obj(a):
    """Exact objective: 2n * max(conv(a,a)) / (sum a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def optimize_lbfgs(a0, time_limit, t_init=10.0, t_final=100.0, seed=42):
    """Optimize using L-BFGS-B with adaptive temperature on smoothed objective."""
    n = len(a0)
    # Convert to softmax parameterization
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    x0 = np.log(a0)  # log of probability = softmax parameter
    
    best_obj = true_obj(a0)
    best_a = a0.copy()
    
    t0 = time.monotonic()
    t_cur = t_init
    n_ref = n  # for reference
    
    while time.monotonic() - t0 < time_limit:
        def func(x):
            obj, grad = smooth_obj_and_grad_x(x, t_cur)
            return obj, grad
        
        res = scipy_minimize(func, x0, method='L-BFGS-B',
                             jac=True, options={'maxiter': 200, 'ftol': 1e-12,
                                                'gtol': 1e-10, 'maxfun': 500})
        
        x_opt = res.x
        a_opt = softmax_fn(x_opt)
        obj_true = true_obj(a_opt)
        
        if obj_true < best_obj:
            best_obj = obj_true
            best_a = a_opt.copy()
        
        # Update temperature (increase for more precise max approximation)
        t_cur = min(t_cur * 1.5, t_final)
        x0 = x_opt
        
        if time.monotonic() - t0 > time_limit - 30:
            break
        
        # Check for convergence
        if not np.isfinite(res.fun) or res.status not in [0, 1, 2, 4]:
            break
        if np.linalg.norm(res.x - x0) < 1e-12:
            break
    
    return best_a, best_obj


def optimize_projected_grad(a0, time_limit, lr_init=0.01, momentum=0.9,
                            t_init=5.0, t_final=50.0, top_k=None,
                            perturb_every=500, perturb_scale=0.005,
                            sa_temp=0.0, sa_cool=0.999, seed=42):
    """
    Projected gradient with Nesterov momentum, adaptive temperature,
    and optional simulated annealing.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    if top_k is None:
        top_k = max(5, n // 10)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    v_nest = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    t_cur = t_init
    no_improve = 0
    sa_curr = sa_temp
    
    while time.monotonic() - t0 < time_limit:
        # Compute gradient at current temperature
        # For efficiency, use top-k peaks when t_cur is large
        if t_cur < 50:
            # Use log-sum-exp over all peaks
            scaled = t_cur * conv
            scaled -= scaled.max()
            exp_vals = np.exp(scaled)
            p = exp_vals / exp_vals.sum()
            grad_a = 2.0 * n * grad_wrt_a(a, p) / t_cur  # normalized
        else:
            # Use only top-k peaks (sharper gradient)
            if top_k < len(conv):
                peaks = np.argpartition(conv, -top_k)[-top_k:]
                peak_vals = conv[peaks]
                scaled = t_cur * peak_vals
                scaled -= scaled.max()
                w = np.exp(scaled)
                w /= w.sum()
                
                p_sparsed = np.zeros(len(conv))
                for i, pk in enumerate(peaks):
                    p_sparsed[pk] = w[i]
                grad_a = 2.0 * n * grad_wrt_a(a, p_sparsed) / t_cur
            else:
                p = np.ones(len(conv)) / len(conv)
                grad_a = 2.0 * n * grad_wrt_a(a, p) / t_cur
        
        # Nesterov momentum
        v_nest = momentum * v_nest - lr * grad_a
        a_trial = project_simplex(a + momentum * v_nest)
        v = momentum * v - lr * grad_a
        
        # Line search (simple backtracking)
        a_new = project_simplex(a + v)
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        # Simulated annealing: accept if better or with probability exp(-delta/sa_curr)
        accept = False
        if obj_new < best_obj:
            accept = True
        elif sa_curr > 1e-8 and obj_new < sa_curr * 10:
            delta = obj_new - best_obj
            if np.log(rng.uniform()) < -delta / sa_curr:
                accept = True
            else:
                # Try a smaller step
                a_new = project_simplex(a + 0.5 * v)
                conv_new = fast_convolve(a_new)
                obj_new = 2.0 * n * conv_new.max()
                if obj_new < best_obj:
                    accept = True
                    v *= 0.5
        
        if accept:
            a = a_new
            conv = conv_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            v *= 0.5
        
        # Adaptive learning rate
        if no_improve > perturb_every:
            # Perturbation to escape local minimum
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v = np.zeros(n)
            v_nest = np.zeros(n)
            no_improve = 0
            perturb_scale *= 0.7
            lr = max(lr_init * 0.2, lr * 0.5)
        
        # Temperature schedule
        if t_cur < t_final:
            t_cur = min(t_final, t_cur * 1.02)
        
        # SA cooling
        sa_curr *= sa_cool
        
        # Learning rate decay
        lr *= 0.9999
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0).astype(np.float64)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0).astype(np.float64))
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps
    
    # Cosine-based
    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 1e-6)
        a /= a.sum()
        inits[f'cos_{freq}'] = a
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g
    
    # Random (Dirichlet with various concentrations)
    for alpha in [0.3, 1.0, 3.0, 10.0]:
        inits[f'rand_{alpha}'] = rng.dirichlet(np.ones(n) * alpha)
    
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    
    # Triangular
    tri = np.minimum(x + 1, n - x)
    tri = np.maximum(tri, 1e-6)
    tri /= tri.sum()
    inits['tri'] = tri
    
    # Symmetric random
    half = rng.dirichlet(np.ones(n // 2) * 0.5)
    sym = np.concatenate([half, half[::-1]])
    sym /= sym.sum()
    inits['sym_rand'] = sym
    
    return inits
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))² using softmax-parameterized L-BFGS-B
with adaptive log-sum-exp smoothing, Nesterov-momentum projected gradient,
peak-targeting, and multi-start exploration. Exploits FFT-based convolution.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def grad_cross_corr(a, p):
    """
    Compute g[j] = sum_{m=0}^{n-1} p[j+m] * a[m] for j=0..n-1.
    This is the cross-correlation needed for the gradient.
    """
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        m_max = min(n - 1, 2 * n - 2 - j)
        if m_max >= 0:
            g[j] = np.dot(p[j:j + m_max + 1], a[0:m_max + 1])
    return g


def true_objective(a):
    """Exact objective: 2n * max(conv(a,a)) / (sum a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def lbfgs_objective(x, t, use_exact_obj):
    """
    Objective and gradient for L-BFGS in softmax parameterization.
    Uses log-sum-exp smoothed gradient; optionally exact objective.
    """
    a = softmax_fn(x)
    n = len(a)
    conv = fast_convolve(a)

    if use_exact_obj:
        obj = 2.0 * n * conv.max()
    else:
        scaled = t * conv
        scaled -= scaled.max()
        obj = (2.0 * n / t) * (np.log(np.exp(scaled).sum()) + scaled.max())

    # Smoothed gradient (always use smoothed for direction)
    scaled = t * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    p = exp_vals / exp_vals.sum()

    # d f_s / d a[j] = 4n * sum_m p[j+m] * a[m] = 2n * (2 * sum_m p[j+m]*a[m])
    # grad_cross_corr gives sum_m p[j+m]*a[m], so multiply by 4n
    grad_a = 4.0 * n * grad_cross_corr(a, p)

    # Chain rule through softmax: d f / dx = grad_a - a * (a^T grad_a)
    grad_x = grad_a - a * np.dot(a, grad_a)

    return obj, grad_x


def optimize_lbfgs(a0, time_limit, t_schedule=None, use_exact_obj=True, seed=42):
    """
    Optimize using L-BFGS-B with adaptive temperature in softmax parameterization.
    """
    n = len(a0)
    if t_schedule is None:
        t_schedule = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]

    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    x = np.log(a0)

    best_obj = true_objective(a0)
    best_a = a0.copy()

    t0 = time.monotonic()

    for t in t_schedule:
        if time.monotonic() - t0 > time_limit:
            break

        def func(x_cur):
            return lbfgs_objective(x_cur, t, use_exact_obj)

        remaining = time_limit - (time.monotonic() - t0)
        maxiter = max(20, int(remaining * 200))

        try:

                                 options={'maxiter': min(maxiter, 300),
                                          'ftol': 1e-15, 'gtol': 1e-12,
                                          'maxfun': 500})
        except Exception:
            break

        if res.x is not None and np.all(np.isfinite(res.x)):
            a_new = softmax_fn(res.x)
            obj_new = true_objective(a_new)
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
            x = res.x

            if not np.isfinite(res.fun):
                break
            # Check convergence
            if res.status == 0 and hasattr(res, 'nit') and res.nit < 10:
                pass  # converged very fast, continue to next t

            v = np.zeros(n)
            no_improve = 0
            perturb_scale_cur *= 0.75
            lr = max(lr_init * 0.1, lr * 0.6)

        # Temperature increase
        if t_cur < t_final:
            t_cur = min(t_final, t_cur * 1.015)

        # SA cooling
        sa_curr *= sa_cool

        # LR decay
        lr *= 0.99995

    return best_a, best_obj


def peak_targeting(a, time_limit, top_k=5, lr=0.005, seed=42):
    """
    Focus specifically on reducing the top-k peaks of the auto-correlation.
    Uses coordinate-wise adjustments guided by the peak structure.
    """
    rng = np.random.default_rng(seed)
    n = len(a)
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = lr

    while time.monotonic() - t0 < time_limit:
        # Find top-k peaks
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

        # Compute subgradient at the top peak
        k_star = int(peaks[np.argmax(peak_vals)])

        # The peak c[k*] = sum_j a[j]*a[k*-j]
        # To reduce it, decrease a[j] where a[k*-j] is large
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        indices = np.arange(j_start, j_end + 1)
        contributions = a[indices] * a[k_star - indices]

        # Gradient direction: decrease where contribution is high
        grad = np.zeros(n)
        total = contributions.sum()
        if total > 1e-15:
            weights = contributions / total
            grad[indices] = 2.0 * weights

        # Also add contributions from other top peaks (smaller weight)
        for i in range(1, min(top_k, len(peaks))):
            k_i = int(peaks[i])
            j_s = max(0, k_i - n + 1)
            j_e = min(n - 1, k_i)
            idx = np.arange(j_s, j_e + 1)
            ctr = a[idx] * a[k_i - idx]
            tot = ctr.sum()
            if tot > 1e-15:
                grad[idx] += 0.3 * ctr / tot

        grad -= np.mean(grad)

        a_new = project_simplex(a - step * grad)
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            step = min(0.05, step * 1.01)
        else:
            step *= 0.95
            if step < 1e-6:
                # Perturb and restart
                a = project_simplex(a + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(a)
                step = lr
                if not np.isfinite(true_objective(a)):
                    break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent.astype(np.float64), 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent.astype(np.float64), 0))
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    for freq in [1, 2, 3, 5, 7]:
        a = 1.0 + 0.85 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 1e-8)
        a /= a.sum()
        inits[f'cos_{freq}'] = a

    # Gaussian center
    for sigma in [0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sigma * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sigma}'] = g

    for alpha in [0.3, 0.5, 1.0, 2.0, 5.0]:
        inits[f'rand_{alpha}'] = rng.dirichlet(np.ones(n) * alpha)

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp

    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 1e-8)
    tri /= tri.sum()
    inits['tri'] = tri

    # Symmetric
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.7)
    sym = np.concatenate([half, half[::-1]])
    if n % 2 == 1:
        sym = np.concatenate([half, [half[-1]], half[::-1]])
    sym = sym[:n]
    sym = np.maximum(sym, 1e-10)
    sym /= sym.sum()
    inits['sym_rand'] = sym

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: L-BFGS on softmax-parameterized log-sum-exp
    smoothed objective, projected gradient with Nesterov momentum and SA,
    peak-targeting, and multi-start exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: len={len(parent)}, parent_obj={true_objective(parent):.6f}")
    except Exception:
        pass

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None or len(a) == 0:
            return best_score
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  New best: {score:.8f} (n={len(a)})")
        return score

    # === Phase 1: Multi-start L-BFGS exploration ===
    print("=== Phase 1: Multi-start L-BFGS ===")
    t1_start = time.monotonic()

    n = 512
    inits = generate_initializations(n, parent, rng)

    priority = ['parent', 'uniform', 'cos_1', 'cos_2', 'gauss_0.25',
                'rand_1.0', 'sym_rand', 'rand_0.5', 'tri']
    init_names = [nm for nm in priority if nm in inits] + \
                 [nm for nm in inits if nm not in priority]

    p1_budget = 250
    max_inits = min(len(init_names), 8)
    time_per = p1_budget / max_inits

    candidates = []

    for name in init_names[:max_inits]:
        remain = deadline - time.monotonic() - 1300
        if remain < 20:
            break
        t_alloc = min(time_per, remain)
        if t_alloc < 8:
            break

        a0 = inits[name]
        a_opt, obj = optimize_lbfgs(a0, time_limit=t_alloc,
                                    t_schedule=[5.0, 15.0, 40.0, 100.0],
                                    use_exact_obj=True)
        score = true_objective(a_opt)
        candidates.append((score, a_opt, name))
        print(f"  {name}: obj={score:.6f}")

    candidates.sort(key=lambda x: x[0])
    try_update(candidates[0][1])
    print(f"Phase 1 done: best={best_score:.8f}, elapsed={time.monotonic()-t1_start:.0f}s")

    # === Phase 2: Deep L-BFGS on top candidates ===
    print("=== Phase 2: Deep L-BFGS ===")
    t2_start = time.monotonic()

    p2_budget = 500
    top_candidates = candidates[:3]
    t_per = p2_budget / max(1, len(top_candidates))

    for _, a_cand, name in top_candidates:
        remain = deadline - time.monotonic() - 700
        if remain < 15:
            break
        t_alloc = min(t_per, remain)
        if t_alloc < 10:
            break

        # Run L-BFGS with fine-grained temperature schedule
        a_opt, obj = optimize_lbfgs(a_cand, time_limit=t_alloc,
                                    t_schedule=[10.0, 25.0, 50.0, 100.0, 200.0, 400.0],
                                    use_exact_obj=True)
        try_update(a_opt)

    print(f"Phase 2 done: best={best_score:.8f}, elapsed={time.monotonic()-t2_start:.0f}s")

    # === Phase 3: Projected gradient with Nesterov momentum + SA ===
    print("=== Phase 3: Projected Gradient + Nesterov ===")
    t3_start = time.monotonic()

    if best_sequence is not None:
        p3_budget = 500
        configs = [
            {'lr_init': 0.02, 'momentum': 0.90, 't_init': 3.0, 't_final': 40.0,
             'top_k': 20, 'perturb_every': 300, 'perturb_scale': 0.008,
             'sa_temp': 0.5, 'sa_cool': 0.9993},
            {'lr_init': 0.01, 'momentum': 0.93, 't_init': 8.0, 't_final': 80.0,
             'top_k': 30, 'perturb_every': 500, 'perturb_scale': 0.005,
             'sa_temp': 1.0, 'sa_cool': 0.9988},
            {'lr_init': 0.005, 'momentum': 0.95, 't_init': 20.0, 't_final': 150.0,
             'top_k': 50, 'perturb_every': 800, 'perturb_scale': 0.003,
             'sa_temp': 0.2, 'sa_cool': 0.9997},
        ]

        t_per = p3_budget / len(configs)
        for i, cfg in enumerate(configs):
            remain = deadline - time.monotonic() - 300
            if remain < 10:
                break
            t_alloc = min(t_per, remain)
            if t_alloc < 10:
                break

            a_pert = project_simplex(best_sequence + rng.standard_normal(n) * 0.002)
            a_opt, _ = optimize_projected_grad(a_pert, time_limit=t_alloc, **cfg)
            try_update(a_opt)

    print(f"Phase 3 done: best={best_score:.8f}, elapsed={time.monotonic()-t3_start:.0f}s")

    # === Phase 4: Peak targeting ===
    print("=== Phase 4: Peak Targeting ===")
    t4_start = time.monotonic()

    if best_sequence is not None:
        remain = deadline - time.monotonic() - 150
        if remain > 20:
            a_opt, _ = peak_targeting(best_sequence, time_limit=remain * 0.6,
                                      top_k=5, lr=0.008)
            try_update(a_opt)

    print(f"Phase 4 done: best={best_score:.8f}, elapsed={time.monotonic()-t4_start:.0f}s")

    # === Phase 5: Final polish with very fine L-BFGS ===
    print("=== Phase 5: Final Polish ===")
    t5_start = time.monotonic()

    if best_sequence is not None:
        remain = deadline - time.monotonic() - 20
        if remain > 15:
            a_opt, _ = optimize_lbfgs(best_sequence, time_limit=remain * 0.5,
                                      t_schedule=[100.0, 300.0, 800.0],
                                      use_exact_obj=True)
            try_update(a_opt)

            # One more peak targeting pass
            if time.monotonic() < deadline - 10:
                a_opt2, _ = peak_targeting(best_sequence, time_limit=min(30, deadline - time.monotonic() - 5),
                                           top_k=3, lr=0.003)
                try_update(a_opt2)

    print(f"Phase 5 done: best={best_score:.8f}, elapsed={time.monotonic()-t5_start:.0f}s")

    # Final
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = min(best_score, final_score)

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
