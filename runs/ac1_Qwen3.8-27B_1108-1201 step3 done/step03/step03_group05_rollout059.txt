import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_gradient(a, p):
    """Compute gradient of smoothed objective: g[i] = 2 * sum_j a[j] * p[i+j]."""
    n = len(a)
    N = 1 << (3 * n).bit_length()

    S = np.fft.rfft(p, N)
    return 2.0 * np.fft.irfft(np.conj(A) * S, N)[:n]


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(a):
    """Make sequence symmetric: a[i] = a[n-1-i]."""
    return (a + a[::-1]) / 2.0

        if a_new.sum() < 1e-300:
            a_new = np.ones(n) / n
        else:
            a_new /= a_new.sum()
        a_new = np.maximum(a_new, 0)
        a_new /= a_new.sum()
        
        if symmetric:
            a_new = symmetrize(a_new)
            a_new = np.maximum(a_new, 0)
            a_new /= a_new.sum()
        
        a = a_new
        iteration += 1
        
        # Evaluate every few iterations (to save time)
        if iteration % 3 == 0 or stuck > 50:
            conv = fast_convolve(a)
            cur_max = conv.max()
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                stuck = 0
            else:
                stuck += 1
        
        # Perturbation
        if stuck >= perturb_every:
            noise = rng.standard_normal(n) * perturb_scale
            if symmetric:
                noise = symmetrize(noise)
            a = np.maximum(a + noise, 0)
            a /= a.sum()
            if symmetric:
                a = symmetrize(a)
                a = np.maximum(a, 0)
                a /= a.sum()
            stuck = 0
            perturb_every = int(perturb_every * 1.3)
    
    return best_a, best_max


def nesterov_gradient_descent(a0, time_limit, beta_start=20.0, beta_end=500.0,
                              lr_init=0.01, momentum=0.9,
                              symmetric=False, perturb_every=400,
                              perturb_scale=0.003, seed=0):
    """Nesterov accelerated projected gradient descent on smoothed objective."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    if symmetric:
        a = symmetrize(a)
        a /= a.sum()
    
    N = 1 << (3 * n).bit_length()
    v = np.zeros(n)
    
    conv = fast_convolve(a)
    best_max = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    stuck = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr_init * (1.0 - 0.5 * frac)
        
        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        y /= y.sum()
        
        # Gradient at y
        conv_y = fast_convolve(y)
        shifted = beta * (conv_y - conv_y.max())
        p = np.exp(shifted)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(conv_y))
            p[np.argmax(conv_y)] = 1.0
        else:
            p /= p_sum
        
        Y = np.fft.rfft(y, N)
        S = np.fft.rfft(p, N)
        grad = 2.0 * np.fft.irfft(np.conj(Y) * S, N)[:n]
        
        # Update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = project_simplex(a_new)
        if symmetric:
            a_new = symmetrize(a_new)
            a_new = np.maximum(a_new, 0)
            a_new /= a_new.sum()
        a = a_new
        iteration += 1
        
        if iteration % 3 == 0 or stuck > 50:
            cur_max = fast_convolve(a).max()
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                stuck = 0
            else:
                stuck += 1
        
        if stuck >= perturb_every:
            noise = rng.standard_normal(n) * perturb_scale
            if symmetric:
                noise = symmetrize(noise)
            a = project_simplex(a + noise)
            if symmetric:
                a = symmetrize(a)
                a = np.maximum(a, 0)
                a /= a.sum()
            v *= 0.3
            stuck = 0
            perturb_every = int(perturb_every * 1.2)
    
    return best_a, best_max


def coordinate_descent_polish(a0, time_limit, seed=0):
    """
    Coordinate descent: try adjusting each coordinate to reduce max convolution.
    Also tries pair exchanges.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_max = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    # Precompute which convolution entries are affected by each coordinate
    # Changing a[i] affects conv[k] for k = i to i+n-1 (and k = i-n+1 to i if we consider
    # the other direction... actually for auto-convolution, changing a[i] affects
    # conv[k] for k in [i, i+n-1] and also k in [i-(n-1), i]... no.
    # conv[k] = sum_j a[j]*a[k-j]. Changing a[i] affects terms where j=i or k-j=i.
    # j=i: contributes a[i]*a[k-i] to conv[k] (for k-i in [0,n-1], i.e., k in [i, i+n-1])
    # k-j=i: j=k-i, contributes a[k-i]*a[i] to conv[k] (same thing)
    # So changing a[i] affects conv[k] for k in [i, i+n-1] (since conv is symmetric,
    # also for k in [i-(n-1), i], but those are the same by symmetry of auto-conv).
    # Actually, conv[k] for auto-conv: conv[k] = sum_j a[j]*a[k-j]
    # Term with j=i: a[i]*a[k-i], valid when 0<=k-i<n, i.e., i<=k<i+n
    # Term with k-j=i, i.e., j=k-i: a[k-i]*a[i], valid when 0<=k-i<n (same condition)
    # But these are the same term! So conv[k] contains a[i]*a[k-i] exactly once
    # (when j=i) and once more when j=k-i (if k-i != i).
    # If k-i != i (i.e., k != 2i): conv[k] has TWO terms involving a[i]: a[i]*a[k-i] (j=i) and a[k-i]*a[i] (j=k-i).
    # If k == 2i: conv[k] has ONE term: a[i]*a[i].
    
    # For efficiency, when we change a[i] by delta:
    # conv[k] changes by: 2*delta*a[k-i] for k in [i+1, i+n-1] (k != 2i case)
    #                      plus delta*a[k-i] for the j=i or j=k-i term
    # Actually this is getting complicated. Let me just recompute the full convolution

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        cur_max = conv.max()
        max_idx = np.argmax(conv)
        
        improved = False
        
        # Strategy 1: try reducing the element that most contributes to the peak
        k_peak = int(max_idx)
        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)
        
        # Find the pair (i, k-i) with max product
        if i_end >= i_start:
            contributions = np.array([a[i] * a[k_peak - i] for i in range(i_start, i_end + 1)])
            if len(contributions) > 0:
                best_i = i_start + int(np.argmax(contributions))
                
                # Try reducing a[best_i] and distributing to other positions
                for target in range(n):
                    if target == best_i:
                        continue
                    for frac in [0.02, 0.05, 0.1, 0.15]:
                        amt = a[best_i] * frac
                        if amt < 1e-10:
                            break
                        a_try = a.copy()
                        a_try[best_i] -= amt
                        a_try[target] += amt
                        a_try = np.maximum(a_try, 0)
                        sa = a_try.sum()
                        if sa < 1e-10:
                            continue
                        conv_try = fast_convolve(a_try)
                        if conv_try.max() < cur_max - 1e-12:
                            a = a_try
                            cur_max = conv_try.max()
                            if cur_max < best_max:
                                best_max = cur_max
                                best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
        
        # Strategy 2: try random pair adjustments
        if not improved:
            for _ in range(10):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                for frac in [0.01, 0.03, 0.05]:
                    amt = min(a[i], a[j]) * frac
                    if amt < 1e-10:
                        break
                    # Try moving mass from i to j
                    a_try = a.copy()
                    a_try[i] -= amt
                    a_try[j] += amt
                    a_try = np.maximum(a_try, 0)
                    sa = a_try.sum()
                    if sa < 1e-10:

                        cur_max = conv_try.max()
                        if cur_max < best_max:
                            best_max = cur_max
                            best_a = a.copy()
                        improved = True
                        break
        
        if not improved:
            no_improve += 1
            if no_improve > 200:
                # Try a larger perturbation

                j = rng.integers(0, n)
                if i != j and a[i] > 1e-6:

                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    a /= a.sum()

def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    
    inits['uniform'] = np.ones(n) / n
    
    # Cosine-based (U-shaped for positive alpha)
    for alpha in [0.3, 0.5, 0.7, 0.9]:
        c = 1.0 + alpha * np.cos(2 * np.pi * x / n)
        c = np.maximum(c, 0.01)
        c /= c.sum()
        inits[f'cos_a{int(alpha*10)}'] = c
    
    # Exponential edge concentration
    for alpha in [1.0, 2.0, 3.0, 5.0]:
        e = np.exp(alpha * np.cos(2 * np.pi * x / n))
        e /= e.sum()
        inits[f'exp_e{int(alpha)}'] = e
    
    # Raised cosine (smooth U-shape)
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    rc = np.maximum(rc, 1e-6)
    rc /= rc.sum()
    inits['raised_cos'] = rc
    
    # Power of raised cosine
    for pw in [0.5, 2.0, 3.0]:
        pc = (0.5 * (1 + np.cos(2 * np.pi * x / n))) ** pw
        pc = np.maximum(pc, 1e-8)
        pc /= pc.sum()
        inits[f'rc_pw{int(pw*10)}'] = pc
    
    # Bimodal
    edge = max(2, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Random
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand3'] = rng.dirichlet(np.ones(n) * 3.0)
    
    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        elif len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_trunc'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps
    
    return inits
