def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_sparse(f, M, n_constraints=None):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Uses sparse structure of the constraint matrix.
    """
    n = len(f)
    num_constraints = 2 * n - 1
    if n_constraints is not None:
        num_constraints = min(num_constraints, n_constraints)

    c = -np.ones(n)
    A_ub = np.zeros((num_constraints, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def sg_polish(f0, time_limit, lr=0.004, mom=0.93, top_k=40, temp=0.0001):
    """Subgradient descent with multi-peak softmax weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-13:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.004

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.001)
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    # Two peaks
    p1 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2)
    p2 = np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append((p1 + p2) / (p1 + p2).sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    # Random smooth
    s = rng.standard_normal(n)
    # Smooth with moving average
    kernel = np.ones(5) / 5.0
    s_smooth = np.convolve(s, kernel, mode='same')
    s_smooth = np.maximum(s_smooth, 0)
    ss = s_smooth.sum()
    if ss > 1e-10:
        inits.append(s_smooth / ss)

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase hybrid search using coordinate descent,
    LP-based Frank-Wolfe, and subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = a / a.sum()  # normalize
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}")
    except Exception:
        pass

    # === Phase 1: Coordinate Descent on parent ===
    print("=== Phase 1: Coordinate Descent ===")

    if parent is not None:
        n_p = len(parent)
        t1 = min(400, deadline - time.monotonic() - 600)
        if t1 > 60:
            print(f"  CD on parent (n={n_p}), t={t1:.0f}s")
            a_opt, _ = coordinate_descent(parent, time_limit=t1, verbose=True)
            try_seq(a_opt, "cd_parent")

    # === Phase 2: LP-based Frank-Wolfe ===
    print("\n=== Phase 2: LP-based Frank-Wolfe ===")

    candidates = []
    if best_seq is not None:
        candidates.append(best_seq)
    if parent is not None:
        candidates.append(parent)

    # Try other lengths
    lengths_to_try = [n_p if parent is not None else 384, 256, 512]
    n_p = len(parent) if parent is not None else 384
    if n_p not in lengths_to_try:
        lengths_to_try.insert(0, n_p)
    # Remove duplicates
    seen = set()
    unique_lengths = []
    for l in lengths_to_try:
        if l not in seen:
            seen.add(l)
            unique_lengths.append(l)
    lengths_to_try = unique_lengths

    remaining = deadline - time.monotonic() - 200
    for n in lengths_to_try:
        if time.monotonic() > deadline - 150:
            break

        inits = generate_inits(n, parent if parent is not None else (best_seq if best_seq is not None else None), rng)
        t_per_init = max(30, remaining / (len(lengths_to_try) * len(inits)))

        for i, f_init in enumerate(inits):
            if time.monotonic() > deadline - 120:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 100)
            if t_alloc < 15:
                break

            # Use FW-LP for some, CD for others
            if i % 2 == 0:
                a_opt, _ = fw_lp_optimize(f_init, time_limit=t_alloc)
            else:
                a_opt, _ = coordinate_descent(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"n{n}_fw{i}")

    print(f"  Phase 2 best: {best_score:.6f}")

    # === Phase 3: Deep optimization alternating CD and FW ===
    print(f"\n=== Phase 3: Deep Optimization ===")

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 80

        for p in range(8):
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            t_alloc = min(rem / max(1, 8 - p), 120)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.005
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                # Mirror blend
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = project_simplex(f_init)
            elif p == 3:
                # Amplify peaks
                f_init = best_seq ** 1.5
                f_init = project_simplex(f_init)
            elif p == 4:
                # Flatten
                f_init = best_seq ** 0.7
                f_init = project_simplex(f_init)
            elif p == 5:
                noise = rng.standard_normal(n_best) * 0.02
                f_init = project_simplex(best_seq + noise)
            elif p == 6:
                # Interpolate with uniform
                f_init = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)
            else:
                # Frequency-domain perturbation
                F = np.fft.rfft(best_seq)
                F *= (1.0 + 0.01 * rng.standard_normal(len(F)))
                f_real = np.fft.irfft(F, n_best)
                f_init = project_simplex(f_real)

            # Alternate between CD and FW
            if p % 2 == 0:
                a_opt, _ = coordinate_descent(f_init, time_limit=t_alloc)
            else:
                a_opt, _ = fw_lp_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"deep_{p}")

        # SG polish
        rem = deadline - time.monotonic() - 40
        if rem > 30:
            a_opt, _ = sg_polish(best_seq, time_limit=min(rem, 80),
                                 lr=0.003, mom=0.94, top_k=max(20, n_best // 10), temp=0.00005)
            try_seq(a_opt, "sg_p3")

    # === Phase 4: Intensive CD polishing ===
    print(f"\n=== Phase 4: Intensive CD Polish ===")
    print(f"  Current best before CD: {best_score:.6f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple rounds of CD with decreasing time
        cd_times = [min(120, int((deadline - time.monotonic() - 60) * 0.4)),
                    min(80, int((deadline - time.monotonic() - 50) * 0.35)),
                    min(50, int((deadline - time.monotonic() - 40) * 0.3)),
                    20]

        for ci, ct in enumerate(cd_times):
            rem = deadline - time.monotonic() - 30
            if rem < ct:
                ct = max(10, int(rem * 0.8))
            if ct < 10:
                break

            print(f"  CD round {ci+1}, t={ct}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=ct, verbose=False)
            try_seq(a_opt, f"cd_polish_{ci}")

            # Quick SG between CD rounds
            if ci < len(cd_times) - 1:
                rem2 = deadline - time.monotonic() - 25
                if rem2 > 15:
                    a_opt2, _ = sg_polish(best_seq, time_limit=min(rem2, 30),
                                          lr=0.001, mom=0.95,
                                          top_k=max(20, n_best // 8), temp=0.00002)
                    try_seq(a_opt2, f"sg_between_{ci}")

    # === Phase 5: Final micro-optimization ===
    print(f"\n=== Phase 5: Final Micro-Opt ===")
    print(f"  Current best: {best_score:.6f}")

    if best_seq is not None:
        n_best = len(best_seq)

        # Very fine CD with tiny perturbations
        for _ in range(5):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = coordinate_descent(f_pert, time_limit=min(rem, 25))
            try_seq(a_opt, "micro_cd")

        # Final SG
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = sg_polish(best_seq, time_limit=min(rem, 40),
                                 lr=0.0005, mom=0.96,
                                 top_k=max(30, n_best // 4), temp=0.00001)
            try_seq(a_opt, "sg_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
