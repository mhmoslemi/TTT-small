def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    """Build the matrix B such that (a*b) = B @ b."""
    n = len(a)
    num_k = 2 * n - 1
    B = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        B[k, idx] = a[k - idx]
    return B

def solve_lp(a, M, active_set=None, conv=None, top_k=None):
    """Solve LP to find direction b."""
    n = len(a)
    num_k = 2 * n - 1
    
    if active_set is None:
        if conv is not None and top_k is not None:
            active_set = np.argpartition(conv, -top_k)[-top_k:]
        else:
            active_set = np.arange(num_k)
    
    B = build_conv_matrix(a)
    A_ub = B[active_set]
    b_ub = np.full(len(active_set), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def compute_subgradient(a, beta=100.0):
    """Compute smoothed subgradient of max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * (conv - conv.max())
    w = np.exp(scaled)
    w /= w.sum()
    
    # grad[i] = sum_k w[k] * (a[k-i] + a[i-k])
    # Using FFT cross-correlation
    fft_size = 1 << (2 * n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    
    # sum_k w[k] * a[k-i] = cross-correlation
    cross = np.fft.irfft(np.conj(W) * A, fft_size)[:n]
    # sum_k w[k] * a[i-k] = convolution
    conv_part = np.fft.irfft(W * A, fft_size)[:n]
    
    grad = cross + conv_part
    return grad, conv

def lp_iterate(a, M, t, active_set=None, conv=None, top_k=None):

    b_sum = b.sum()

    return a_new

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(a):
    n = len(a)
    num_k = 2 * n - 1
    B = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        B[k, idx] = a[k - idx]
    return B

def solve_lp_direction(a, M, constraint_indices):
    n = len(a)
    B = build_conv_matrix(a)
    A_ub = B[constraint_indices]
    b_ub = np.full(len(constraint_indices), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})

            return res.x

    return None

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix B such that B @ b = convolve(a, b)."""
    n = len(a)
    num_k = 2 * n - 1
    B = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        B[k, idx] = a[k - idx]
    return B


def solve_lp_direction(a, M, constraint_indices):
    """Solve LP: max sum(b) s.t. b>=0, (a*b)[k]<=M for k in constraint_indices."""
    n = len(a)

    b_ub = np.full(len(constraint_indices), M)

def mirror_descent_step(a, grad, lr):

    return a_new / s2

        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    margin_frac = 0.0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        obj = 2.0 * n * M

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            margin_frac = max(0.0, margin_frac - 0.01)
        else:
            no_improve += 1

        # Adaptive active set
        min_k = max(10, n // 8)
        max_k = max(40, n // 2)
        threshold = M * (1.0 - margin_frac)
        active = np.where(conv >= threshold)[0]
        if len(active) < min_k:
            top_idx = np.argpartition(conv, -min_k)[-min_k:]
            active = np.unique(np.concatenate([active, top_idx]))
        if len(active) > max_k:
            active = np.argpartition(conv, -max_k)[-max_k:]

        # LP with line search
        a_new, obj_new = lp_step_with_linesearch(a, M, active, n_steps=5)

        if obj_new < best_obj - 1e-14:
            a = a_new
            margin_frac = max(0.0, margin_frac - 0.005)
        else:
            # Expand active set
            margin_frac = min(0.5, margin_frac + 0.03)
            if no_improve % 15 == 0 and no_improve > 10:
                # Try fewer constraints
                top_k = max(15, n // 10)
                top_idx = np.argpartition(conv, -top_k)[-top_k:]
                a_new2, obj_new2 = lp_step_with_linesearch(a, M, top_idx, n_steps=5)
                if obj_new2 < best_obj - 1e-14:
                    a = a_new2
                elif no_improve > 80:
                    # Perturb
                    pert_type = (no_improve // 20) % 4
                    if pert_type == 0:
                        noise = rng.standard_normal(n) * 0.008
                        a = project_simplex(a + noise)
                    elif pert_type == 1:
                        a_mirror = a[::-1].copy()
                        a = project_simplex(0.6 * a + 0.4 * a_mirror)
                    elif pert_type == 2:
                        # Random shift
                        sh = int(rng.integers(1, max(2, n // 8)))
                        a_sh = np.roll(a, sh)
                        a = project_simplex(0.7 * a + 0.3 * a_sh)
                    else:
                        # Cut and re-spread
                        cut = n // 2
                        a_left = a[:cut] / max(a[:cut].sum(), 1e-15)
                        a_right = a[cut:] / max(a[cut:].sum(), 1e-15)
                        noise = rng.standard_normal(n) * 0.005
                        a = project_simplex(a + noise)
                    no_improve = 0
                    margin_frac = 0.0

        iter_count += 1

    conv_final = fast_convolve(best_a)
    return best_a, 2.0 * n * conv_final.max()


def optimize_mirror(a0, time_limit, seed=42, beta_init=500.0):
    """Mirror descent (multiplicative update) optimization."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    beta = beta_init
    lr = 0.1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        obj = 2.0 * n * M

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            lr = min(0.5, lr * 1.005)
        else:
            no_improve += 1
            lr = max(0.01, lr * 0.98)

        grad, _ = smooth_subgradient(a, beta)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        grad = grad / g_norm

        a_new = mirror_descent_step(a, grad, lr)
        if not np.array_equal(a_new, a):
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj:
                a = a_new

        beta = min(5000.0, beta * 1.003)

        if no_improve > 150:
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
            lr = 0.1
            beta = beta_init
            no_improve = 0

    conv_final = fast_convolve(best_a)
    return best_a, 2.0 * n * conv_final.max()


def optimize_hybrid(a0, time_limit, seed=42):
    """Alternate LP and mirror descent steps."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    beta = 300.0
    lr = 0.05
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        obj = 2.0 * n * M

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0

        iter_count += 1

        if iter_count % 3 == 0:
            # Mirror descent step
            grad, _ = smooth_subgradient(a, beta)
            g_norm = np.linalg.norm(grad)
            if g_norm > 1e-15:
                grad = grad / g_norm
                a_new = mirror_descent_step(a, grad, lr)
                if not np.array_equal(a_new, a):
                    a = a_new
            beta = min(3000.0, beta * 1.005)
        else:
            # LP step
            top_k = max(20, n // 6)
            top_idx = np.argpartition(conv, -top_k)[-top_k:]
            a_new, obj_new = lp_step_with_linesearch(a, M, top_idx, n_steps=3)
            if obj_new < best_obj - 1e-14:
                a = a_new
                no_improve = 0

        no_improve += 1
        if no_improve > 100:
            noise = rng.standard_normal(n) * 0.01
            a = project_simplex(a + noise)
            no_improve = 0
            lr = 0.05
            beta = 300.0

    conv_final = fast_convolve(best_a)
    return best_a, 2.0 * n * conv_final.max()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Quick LP scan across lengths and initializations.
    Phase 2: Deep LP optimization with adaptive active sets.
    Phase 3: Hybrid LP + mirror descent fine-tuning.
    Phase 4: Final polish with subgradient and perturbations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.4f}")
    except (NameError, Exception):
        pass

    # ============ Phase 1: Quick Exploration (12% time) ============
    print("=== Phase 1: Quick Exploration ===")
    t1_max = 0.12 * budget_s

    lengths = [256, 320, 384, 512]
    phase1_results = []

    for n in lengths:
        if deadline - time.monotonic() < 600:
            break
        inits = make_initializations(n, parent, rng)
        priority = ['parent', 'parent_sym', 'ramp', 'ramp_rev', 'bimodal',
                    'exp_left', 'exp_right', 'gauss_off', 'uniform', 'rand_med']
        names = [nm for nm in priority if nm in inits][:4]

        for name in names:
            if deadline - time.monotonic() < 600:
                break
            a0 = inits[name]
            a_opt, obj_opt = optimize_lp(a0, time_limit=20, seed=hash((n, name)) % 100000)
            score = float(evaluate_sequence(a_opt.tolist()))
            phase1_results.append((score, n, name, a_opt.copy()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P1: n={n}, {name}: {score:.6f}")

    phase1_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # ============ Phase 2: Deep LP (40% time) ============
    print("=== Phase 2: Deep LP ===")

    if best_sequence is not None:
        n_best = len(best_sequence)
    else:
        n_best = 320

    # Build list of starting points
    starts = []
    if best_sequence is not None:
        starts.append(best_sequence.copy())

    # Add parent
    if parent is not None:
        if len(parent) == n_best:
            pp = np.maximum(parent, 0)
            if pp.sum() > 1e-10:
                starts.append(pp / pp.sum())
        else:
            pp = np.interp(np.linspace(0, 1, n_best), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                starts.append(pp / pp.sum())

    # Add ramp
    ramp = np.arange(1, n_best + 1, dtype=np.float64)
    starts.append(ramp / ramp.sum())

    # Add from phase 1 (different init types)
    seen_types = set()
    for score, n, name, a_opt in phase1_results:
        if n == n_best and name not in seen_types:
            starts.append(a_opt.copy())
            seen_types.add(name)
        if len(starts) >= 5:
            break

    t2_available = min(0.40 * budget_s, deadline - time.monotonic() - 350)
    t2_per = max(40.0, t2_available / max(1, len(starts)))

    for i, a_start in enumerate(starts):
        if deadline - time.monotonic() < 350:
            break
        t_alloc = min(t2_per, deadline - time.monotonic() - 350)
        if t_alloc < 30:
            break
        a_opt, obj_opt = optimize_lp(a_start, time_limit=t_alloc, seed=2000 + i * 17)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  P2 [{i}]: n={n_best}, score={score:.6f}")

    print(f"Phase 2 best: {best_score:.6f}")

    # ============ Phase 3: Hybrid (28% time) ============
    print("=== Phase 3: Hybrid LP+Mirror ===")

    if best_sequence is not None:
        t3_available = min(0.28 * budget_s, deadline - time.monotonic() - 150)
        if t3_available > 60:
            a_opt, obj_opt = optimize_hybrid(best_sequence, time_limit=t3_available, seed=5555)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P3: score={score:.6f}")

    print(f"Phase 3 best: {best_score:.6f}")

    # ============ Phase 4: Final Polish (remaining) ============
    print("=== Phase 4: Final Polish ===")

    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 10

        # Mirror descent with high beta
        t_md = min(remaining * 0.4, 100.0)
        if t_md > 20:
            a_opt, obj_opt = optimize_mirror(best_sequence, time_limit=t_md, seed=8888, beta_init=1000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P4 mirror: score={score:.6f}")

        # Another LP round with the best
        remaining2 = deadline - time.monotonic() - 10
        t_lp2 = min(remaining2 * 0.5, 80.0)
        if t_lp2 > 20:
            a_opt, obj_opt = optimize_lp(best_sequence, time_limit=t_lp2, seed=12345)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P4 LP: score={score:.6f}")

        # Tiny perturbations
        remaining3 = deadline - time.monotonic() - 5
        if remaining3 > 5:
            for scale in [0.0005, 0.001, 0.002, 0.005, 0.01]:
                for trial in range(8):
                    if time.monotonic() > deadline - 5:
                        break
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    score_p = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(score_p) and score_p < best_score:
                        best_score = score_p
                        best_sequence = a_pert.copy()
                        print(f"  P4 perturb s={scale} t={trial}: {score_p:.6f}")
                        break

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(128) / 128
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: {final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
