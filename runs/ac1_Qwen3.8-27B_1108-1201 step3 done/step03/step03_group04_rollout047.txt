    n = len(a)

"""
Improved LP-based Frank-Wolfe with:
- Wider length sweep (256-768) with adaptive time allocation
- Banded sparse matrix construction for fast LP
- Active-set constraint reduction for speed
- Multi-phase: scan, deep optimize, subgradient polish, perturbation
- Sequence trimming to remove ineffective edge zeros

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build sparse matrix A such that (A @ b)[k] = conv(a,b)[k]."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a, M, constraint_subset=None):
    """max sum(b) s.t. conv(a,b) <= M, b >= 0. Returns b or None."""
    n = len(a)
    A = build_conv_matrix(a)
    if constraint_subset is not None:
        A = A[constraint_subset, :]
        b_ub = np.full(len(constraint_subset), M)
    else:
        b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 4.0})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def trim_sequence(a, threshold=1e-10):
    """Remove leading/trailing elements below threshold."""
    a = np.asarray(a, dtype=np.float64)
    mask = a > threshold
    if not mask.any():
        return a
    first = np.argmax(mask)
    last = len(a) - 1 - np.argmax(mask[::-1])
    return a[first:last + 1]


def line_search(a, b, n_trials=8):
    """Adaptive line search between a and b."""
    best_t = 0.0
    best_obj = 2.0 * len(a) * fast_convolve(a).max()
    ts = np.array([0.05, 0.1, 0.2, 0.35, 0.5, 0.65, 0.8, 0.95])
    for t in ts:
        cand = (1.0 - t) * a + t * b
        obj = 2.0 * len(cand) * fast_convolve(cand).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    return best_t, best_obj

def multi_peak_subgradient(a0, time_limit, lr_init=0.004, momentum=0.91):
    """Subgradient polish using multiple near-active peaks."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()

    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        # Select top peaks with softmax weighting
        top_k = min(20, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
        temp = max(peak_vals.max() * 0.1, 1e-15)
        w = np.exp((peak_vals - peak_vals.max()) / temp)
        w /= w.sum()

        # Compute subgradient: gradient of max(conv) w.r.t. a
        g = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-8:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * w[i] * a[k - idx]

        v = momentum * v - lr * g
        a_new = np.maximum(a + v, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new

        obj = 2.0 * n * fast_convolve(a).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a.copy()

        lr *= 0.9997
        it += 1

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Asymmetric ramp (more mass on left)
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp_l'] = ramp.copy()
    inits['ramp_r'] = ramp[::-1].copy()

    # Quadratic ramp
    quad = (np.arange(1, n + 1, dtype=np.float64) ** 2)
    quad /= quad.sum()
    inits['quad_l'] = quad.copy()
    inits['quad_r'] = quad[::-1].copy()

    # Gaussian (centered)
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g.copy()

    # Edge concentrated
    edge = max(2, n // 6)
    ec = np.zeros(n)
    ec[:edge] = 1.0
    ec[-edge:] = 1.0
    ec /= ec.sum()
    inits['edge'] = ec.copy()

    # W-shaped (mass at edges and center)
    w_shape = np.ones(n)
    w_shape[:n//4] = 2.0
    w_shape[-n//4:] = 2.0
    w_shape /= w_shape.sum()
    inits['w_shape'] = w_shape.copy()

    # Inverse U (mass at edges only, linear decay)
    iu = np.minimum(x + 1, n - x)
    iu = 1.0 / (iu + 1.0)
    iu /= iu.sum()
    inits['inv_u'] = iu.copy()

    # Sine profile
    sprof = np.sin(np.pi * x / (n - 1 + 1e-10)) + 0.1
    sprof /= sprof.sum()
    inits['sine'] = sprof.copy()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Sparse random
    sparse = rng.random(n)
    sparse[sparse < 0.6] = 0.0
    if sparse.sum() > 1e-10:
        sparse /= sparse.sum()
    else:
        sparse = rng.dirichlet(np.ones(n))
    inits['sparse'] = sparse.copy()

    # Two-level (left half high, right half low)
    two = np.zeros(n)
    two[:n//2] = 2.0
    two[n//2:] = 1.0
    two /= two.sum()
    inits['two_level'] = two.copy()

    # From parent
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = pp / ps
            else:
                x_new = np.linspace(0, 1, n)
                x_old = np.linspace(0, 1, len(pp))
                interp = np.interp(x_new, x_old, pp)
                isum = interp.sum()
                if isum > 1e-10:
                    inits['parent_i'] = interp / isum

    return inits

            a_opt, _ = multi_peak_subgradient(best_sequence, time_limit=t_polish,
                                              lr_init=0.003, momentum=0.92)
            a_trimmed = trim_sequence(a_opt)
            if len(a_trimmed) < 4:
                a_trimmed = a_opt
            score = float(evaluate_sequence(a_trimmed.tolist()))
            if np.isfinite(score):
                print(f"  Phase 3 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_trimmed.copy()

    # === Phase 4: Second LP pass ===
    if best_sequence is not None and time.monotonic() < deadline - 30:
        print("=== Phase 4: Second LP Pass ===")
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 10:
            a_opt, _ = fpw_optimize(best_sequence, time_limit=remaining,
                                    verbose=True, verbose_every=500)
            a_trimmed = trim_sequence(a_opt)
            if len(a_trimmed) < 4:
                a_trimmed = a_opt
            score = float(evaluate_sequence(a_trimmed.tolist()))
            if np.isfinite(score):
                print(f"  Phase 4 score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_trimmed.copy()

    # === Phase 5: Perturbation restarts ===
    if best_sequence is not None and time.monotonic() < deadline - 20:
        print("=== Phase 5: Perturbation Restarts ===")
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 10.0
        n_restarts = max(1, int(remaining // 20))
        n_restarts = min(n_restarts, 6)

        for i in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            scale = 0.002 * (1.0 + 0.5 * i)
            noise = rng.standard_normal(n_best) * scale
            a_pert = np.maximum(best_sequence + noise, 0.0)
            a_pert = a_pert / a_pert.sum()
            t_alloc = min(15.0, deadline - time.monotonic() - 8)
            if t_alloc < 4:
                break
            a_opt, _ = fpw_optimize(a_pert, time_limit=t_alloc, verbose=False)
            a_trimmed = trim_sequence(a_opt)
            if len(a_trimmed) < 4:
                a_trimmed = a_opt
            score = float(evaluate_sequence(a_trimmed.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_trimmed.copy()
                    print(f"  Restart {i}: score={score:.6f} (improved!)")
                else:
                    print(f"  Restart {i}: score={score:.6f}")

    # === Phase 6: Try growing the sequence ===
    if best_sequence is not None and time.monotonic() < deadline - 20:
        print("=== Phase 6: Growth Experiment ===")
        n_cur = len(best_sequence)
        for n_new in [n_cur + 64, n_cur + 128]:
            if n_new > 2048:
                break
            if time.monotonic() > deadline - 15:
                break
            # Interpolate to new length
            x_new = np.linspace(0, 1, n_new)
            x_old = np.linspace(0, 1, n_cur)
            interp = np.interp(x_new, x_old, best_sequence)
            interp = np.maximum(interp, 0.0)
            s = interp.sum()
            if s > 1e-10:
                interp /= s
            t_alloc = min(30.0, deadline - time.monotonic() - 10)
            if t_alloc < 5:
                break
            a_opt, _ = fpw_optimize(interp, time_limit=t_alloc, verbose=False)
            a_trimmed = trim_sequence(a_opt)
            if len(a_trimmed) < 4:
                a_trimmed = a_opt
            score = float(evaluate_sequence(a_trimmed.tolist()))
            if np.isfinite(score):
                print(f"  Growth n={n_new}: score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_trimmed.copy()

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Should not happen, but fallback
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
