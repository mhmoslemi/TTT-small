"""
Mirror descent with multi-peak softmax weighting for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses log-space updates for natural non-negativity, backtracking line search, diverse 
structural initializations (block, comb, edge-decay), and periodic perturbation restarts.
Key differences from LP-FW: (1) mirror descent instead of linearized subproblems,
(2) softmax-weighted multi-peak gradient instead of single-peak LP, (3) log-space 
parameterization that naturally handles the non-negativity constraint.
"""

import numpy as np
import time


def fast_convolve(a, b=None):
    """Linear convolution via FFT, matching np.convolve."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_objective(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (s * s)


def compute_gradient_and_conv(a, K=20, temperature=None):
    """
    Compute the subgradient of max(conv(a,a)) using softmax-weighted top-K peaks.
    Returns: (convolution, gradient vector, max_value)
    """
    n = len(a)
    conv = fast_convolve(a)
    max_val = conv.max()
    
    if max_val < 1e-15:
        return conv, np.zeros(n), max_val
    
    # Find top-K peaks
    top_k = min(K, len(conv))
    peak_indices = np.argpartition(conv, -top_k)[-top_k:]
    peak_values = conv[peak_indices]
    
    # Softmax weighting
    if temperature is None:
        temperature = max(peak_values.max() * 0.1, 1e-15)
    weights = np.exp(peak_values / temperature - peak_values.max() / temperature)
    weights /= weights.sum()
    
    # Compute gradient: for peak k, d(conv[k])/d(a[i]) = 2*a[k-i] if 0<=k-i<n
    grad = np.zeros(n)
    for idx in range(len(peak_indices)):
        if weights[idx] < 1e-12:
            continue
        k = int(peak_indices[idx])
        w = 2.0 * weights[idx]
        # For a[i], the contribution is a[k-i] if 0 <= k-i < n
        # i ranges over max(0, k-n+1) to min(n-1, k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j_vals = k - np.arange(i_start, i_end + 1)
        grad[i_start:i_end + 1] += w * a[j_vals]
    
    return conv, grad, max_val


def mirror_descent_optimize(a0, time_limit, lr_init=0.05, max_iters=50000,
                            K_peaks=20, temp_factor=0.1, perturb_every=300,
                            verbose=False, verbose_every=5000):
    """
    Log-space mirror descent to minimize 2n*max(conv)/(sum)^2.
    Works with a normalized to sum=1.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    log_a = np.log(np.maximum(a, 1e-30))
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()  # sum=1
    best_a = a.copy()
    
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(12345 + n)
    
    while time.monotonic() - t0 < time_limit:
        conv, grad, max_val = compute_gradient_and_conv(a, K=K_peaks, 
                                                        temperature=max_val * temp_factor)
        # Full gradient of log(objective) = log(2n) + log(max) - 2*log(sum)
        # d/d(a[i]) = grad[i]/max - 2/sum(a)
        s = a.sum()
        full_grad = grad / max_val - 2.0 / s
        
        # Backtracking line search in log-space
        best_step_obj = best_obj
        best_log_a = log_a.copy()
        
        for step_frac in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            trial_lr = lr * step_frac
            log_a_trial = log_a - trial_lr * full_grad
            # Convert back and normalize
            a_trial = np.exp(log_a_trial - log_a_trial.max())  # avoids overflow
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-15:
                a_trial = np.ones(n) / n
            else:
                a_trial = a_trial / st
            
            conv_trial = fast_convolve(a_trial)
            obj_trial = 2.0 * n * conv_trial.max()
            
            if obj_trial < best_step_obj:
                best_step_obj = obj_trial
                best_log_a = log_a_trial.copy()
                best_a_trial = a_trial.copy()
        
        # Update
        log_a = best_log_a
        a = np.exp(log_a - log_a.max())
        a = np.maximum(a, 0.0)
        st = a.sum()
        if st < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / st
        
        if best_step_obj < best_obj - 1e-12:
            best_obj = best_step_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Decay learning rate
        lr *= 0.9997
        
        # Periodic perturbation when stuck
        if no_improve >= perturb_every:
            noise_scale = 0.02 * (1.0 + no_improve / perturb_every)
            log_a += rng.standard_normal(n) * noise_scale
            a = np.exp(log_a - log_a.max())
            a = np.maximum(a, 0.0)
            st = a.sum()
            a = a / st
            no_improve = 0
            perturb_every = int(perturb_every * 1.5)
            lr = max(lr, 0.01)
        
        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD it={it}, obj={best_obj:.6f}, lr={lr:.4f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def subgradient_polish(a0, time_limit, lr_init=0.01, momentum=0.9):
    """Polish with projected subgradient on simplex (simple but effective)."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit:
        conv, grad, max_val = compute_gradient_and_conv(a, K=15, 
                                                        temperature=max_val * 0.05)
        s = a.sum()
        full_grad = grad / max_val - 2.0 / s
        
        v = momentum * v - lr * full_grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        st = a_new.sum()
        if st < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / st
        a = a_new
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        lr *= 0.9995
        it += 1
        
        # Periodic kick
        if it % 5000 == 0:
            a += rng.standard_normal(n) * 0.005
            a = np.maximum(a, 0.0)
            a = a / a.sum()
    
    return best_a, best_obj


def generate_structural_inits(n, parent=None, rng=None):
    """Generate diverse structural initializations."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    inits = {}
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Edge-concentrated (U-shape): high at edges, low in middle
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    u_shape = 1.0 + 0.8 * np.abs(x - mid) / mid
    u_shape /= u_shape.sum()
    inits['u_shape'] = u_shape
    
    # Strong U-shape (nearly block)
    edge_w = max(1, n // 4)
    block = np.zeros(n)
    block[:edge_w] = 1.0
    block[-edge_w:] = 1.0
    # Smooth the edges a bit
    smooth_w = max(1, edge_w // 4)
    for i in range(smooth_w):
        frac = 1.0 - i / smooth_w
        block[smooth_w + i] = frac * block[smooth_w]
        block[n - 1 - smooth_w - i] = frac * block[n - 1 - smooth_w]
    block = np.maximum(block, 0)
    block /= block.sum()
    inits['block'] = block
    
    # Two-peaked (mass at 1/4 and 3/4 positions)
    centers = [n // 4, 3 * n // 4]
    sigma = n / 8.0
    two_peak = np.zeros(n)
    for c in centers:
        two_peak += np.exp(-0.5 * ((x - c) / sigma) ** 2)
    two_peak /= two_peak.sum()
    inits['two_peak'] = two_peak
    
    # Comb pattern (alternating high/low)
    comb = np.ones(n)
    comb[1::2] = 0.3
    comb /= comb.sum()
    inits['comb'] = comb
    
    # Ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits['ramp'] = r.copy()
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - mid) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g.copy()
    
    # Parent interpolation
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(pp) == n:
                inits['parent'] = pp / ps
            else:
                pp_interp = np.interp(np.linspace(0, 1, n),
                                      np.linspace(0, 1, len(pp)), pp)
                pp_interp = np.maximum(pp_interp, 0)
                ps2 = pp_interp.sum()
                if ps2 > 1e-10:
                    inits['parent'] = pp_interp / ps2
    
    # Random structured (random mixture of U and two-peak)
    mix_alpha = rng.random()
    inits['random_mix'] = (mix_alpha * inits['u_shape'] + (1 - mix_alpha) * inits['two_peak'])
    inits['random_mix'] /= inits['random_mix'].sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    return inits


def lp_direction(a, M, n_nonneg=True):
    """
    Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns the optimal b (normalized to sum=1) or None.
    Uses scipy linprog if available and fast enough.
    """
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csr_matrix
        
        n = len(a)
        # Build constraint matrix: (A @ b)[k] = conv(a, b)[k] <= M
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[k - j])
        
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                      options={'presolve': True, 'time_limit': 3.0})
        if res.success:
            b = np.maximum(res.x, 0.0)
            sb = b.sum()
            if sb > 1e-10:
                return b / sb
        return None
    except Exception:
        return None


def fw_step(a, time_limit=5.0):
    """Single Frank-Wolfe step with LP oracle."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    
    b = lp_direction(a, M)
    if b is None:
        return a
    
    # Line search
    best_t = 0.0
    best_obj = 2.0 * n * conv.max()
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
        candidate = (1 - t) * a + t * b
        conv_c = fast_convolve(candidate)
        obj_c = 2.0 * n * conv_c.max()
        if obj_c < best_obj:
            best_obj = obj_c
            best_t = t
    
    if best_t > 0:
        return (1 - best_t) * a + best_t * b
    return a

            t_alloc = min(10.0, deadline - time.monotonic() - 5)
            if t_alloc < 3:
                break
            
            a_opt, _ = mirror_descent_optimize(a_pert, time_limit=t_alloc,
                                             lr_init=0.02, K_peaks=20,
                                             temp_factor=0.06, perturb_every=100)
            update_best(a_opt, f"restart_{i}")
    
    # === Phase 6: Final micro-polish ===
    if best_sequence is not None and time.monotonic() < deadline - 5:
        print("=== Phase 6: Final Polish ===")
        remaining = deadline - time.monotonic() - 5
        if remaining > 3:
            a_opt, _ = subgradient_polish(best_sequence, time_limit=remaining,
                                          lr_init=0.001, momentum=0.95)
            update_best(a_opt, "final_polish")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
