"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using LP-based iterative minimax with
dual-guided line search, multiple length scanning, and spectral polishing.
Key innovation: direct LP subproblem solving at each iteration (not just direction
finding), with proper line search on the LP solution, plus dual-weighted
perturbation for escaping local minima.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Cross-convolution via FFT: conv(a, b)."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(p, n):
    """
    Build the constraint matrix A for the LP: A[k,i] = p[k-i] if valid.
    The LP is: min t s.t. A @ x <= t, sum(x)=1, x>=0
    where (A@x)[k] = sum_i p[k-i]*x[i] = cross_conv(p,x)[k].
    """
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_lo = i
        k_hi = min(2 * n - 2, i + n - 1)
        A[k_lo:k_hi + 1, i] = p[:k_hi - k_lo + 1]
    return A


def solve_lp_minimax(p, n, time_limit=5.0):
    """
    Solve LP: min t s.t. cross_conv(p, x)[k] <= t for all k, sum(x)=1, x>=0.
    Returns (x_opt, t_opt, dual_vars) or (None, None, None) on failure.
    """
    A = build_lp_matrix(p, n)
    # Variables: x[0..n-1], t
    # Minimize [0, 0, ..., 0, 1] (last variable is t)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Inequality: A @ x - t <= 0  =>  [A, -1] @ [x, t] <= 0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(x) = 1
    A_eq = np.hstack([np.ones(n), [[0.0]]])
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None, None, None

    if not result.success or result.x is None:
        return None, None, None

    x_opt = np.maximum(result.x[:n], 0.0)
    x_sum = x_opt.sum()
    if x_sum < 1e-10:
        return None, None, None
    x_opt = x_opt / x_sum
    t_opt = result.x[-1]

    duals = result.ineqlin.marginals if result.ineqlin is not None else None

    return x_opt, t_opt, duals


def lp_iteration(p, n, rng):
    """
    One LP iteration: solve LP, do line search, return updated p.
    Returns (new_p, improved, t_lp, t_true_before, t_true_after).
    """
    conv_before = fast_convolve(p)
    t_true_before = conv_before.max()

    x_opt, t_lp, duals = solve_lp_minimax(p, n, time_limit=3.0)

    if x_opt is None:
        return p, False, t_true_before, t_true_before, t_true_before, None

    # Line search: try various alpha values
    best_alpha = 0.0
    best_t = t_true_before
    best_p = p.copy()

    alphas = [1.0, 0.8, 0.6, 0.4, 0.7, 0.9, 0.5, 0.3]
    for alpha in alphas:
        p_try = (1.0 - alpha) * p + alpha * x_opt
        p_try = np.maximum(p_try, 0.0)
        s = p_try.sum()
        if s < 1e-10:
            continue
        p_try = p_try / s
        conv_try = fast_convolve(p_try)
        t_try = conv_try.max()
        if t_try < best_t - 1e-12:
            best_t = t_try
            best_alpha = alpha
            best_p = p_try.copy()

    if best_alpha > 0:
        return best_p, True, t_lp, t_true_before, best_t, duals
    return p, False, t_lp, t_true_before, t_true_before, duals


def deep_lp_optimize(a0, time_limit, seed=42):

    """

            a = new_a
            no_improve = 0
        else:

            # Strategy 1: Dual-guided perturbation (emphasize binding constraints)
            if duals is not None and stuck_phases % 3 == 1:

                dw = np.abs(duals) if len(duals) == 2 * n - 1 else np.ones(2 * n - 1)

                if conv_try.max() < best_obj:

def smoothed_gradient(a, conv, beta):
    """Gradient of logsumexp(beta*conv(a,a))/beta w.r.t. a."""
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum

    w = conv * p
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)
    return 2.0 * cross[:n]


def gradient_polish(a0, time_limit, beta=200.0, lr=0.003, momentum=0.95, seed=0):

def coordinate_polish(a0, time_limit, seed=0):
    """
    Coordinate-wise polish: for each coordinate, try moving mass to/from it
    and keep the best improvement.
    """

                a_try[j] += amt
                conv_try = fast_convolve(a_try)
                if conv_try.max() < best_obj - 1e-12:
                    best_obj = conv_try.max()
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break

            # Also try absorbing mass into i from j
            if a[j] > 1e-10:
                for frac in [0.02, 0.05]:
                    amt = a[j] * frac

                        improved = True
                        break

def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Bell-shaped (reduces edge effects)
    bell = 1.0 / (1.0 + 0.3 * ((x - n / 2) / (n / 5)) ** 2)
    bell /= bell.sum()
    inits['bell'] = bell

    # Gaussian
    for sf in [0.25, 0.4, 0.6]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{int(sf*100)}'] = g

    # Raised cosine
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / max(n - 1, 1)))
    cos_w = np.maximum(cos_w, 1e-6)
    cos_w /= cos_w.sum()
    inits['raised_cos'] = cos_w

    # Bimodal
    bi = np.zeros(n)
    edge = max(2, n // 6)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Random Dirichlet
    inits['dir_05'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['dir_1'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['dir_2'] = rng.dirichlet(np.ones(n) * 2.0)

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_trunc'] = pp / ps
        if len(parent) > 0:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps

    return inits
