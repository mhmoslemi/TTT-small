def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:

    u = np.sort(v)[::-1]

    rho = np.nonzero(cond)[0][-1]

    # = ifft(fft(c)*conj(fft(a)))[-m] = ifft(fft(c)*conj(fft(a)))[fft_size - m]
    # So r[m] = cross_corr_standard[(-m) mod fft_size]
    
    corr = np.fft.irfft(C * np.conj(A), fft_size)
    # corr[m] = sum_k c[k] * a[k+m] (with zero padding)
    # We want r[m] = sum_k c[k] * a[k-m] = corr[(-m) mod fft_size]
    # = corr[fft_size - m] for m > 0, corr[0] for m=0
    r = np.zeros(n)
    for m in range(n):
        idx = (-m) % fft_size
        r[m] = corr[idx]
    return r


def cross_corr_fast(c, a):
    """Faster cross-correlation using fftshift trick."""
    n = len(a)
    fft_size = 1 << (2 * n).bit_length()
    C = np.fft.rfft(c, fft_size)
    A = np.fft.rfft(a, fft_size)
    # corr[m] = sum_k c[k] a[k+m]
    corr = np.fft.irfft(C * np.conj(A), fft_size)
    # r[m] = sum_k c[k] a[k-m] = corr[fft_size - m] for m > 0
    # We need r[0..n-1]
    r = np.empty(n)
    r[0] = corr[0]
    if n > 1:
        r[1:] = corr[fft_size - 1 : fft_size - n : -1]
    return r


def gradient_smoothed_max(a, conv, beta):
    """Gradient of logsumexp(beta*conv)/beta w.r.t. a."""
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum

    w = conv * p
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)
    return 2.0 * cross[:n]


def gradient_var_conv(conv, a):
    """Gradient of sum_k c[k]^2 w.r.t. a. Equals 4 * cross_corr(c, a)."""
    return 4.0 * cross_corr_fast(conv, a)


def compute_objective(a):
    """Compute the full score: 2n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def optimize_variance_regularized(a0, time_limit, beta_init=30.0, beta_end=300.0,
                                   lam_init=0.5, lam_end=0.01,
                                   lr_init=0.01, momentum=0.9,
                                   perturb_every=200, perturb_scale=0.003,
                                   seed=0):
    """
    Gradient descent on logsumexp(beta*conv)/beta + lam*var(conv).
    The variance term encourages flat convolution, enabling lower peaks.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac
        lam = lam_init + (lam_end - lam_init) * frac

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0)
        s_y = y.sum()
        if s_y < 1e-15:
            conv_y = fast_convolve(a)
        else:
            conv_y = fast_convolve(y)

        grad_max = gradient_smoothed_max(y, conv_y, beta)
        grad_var = gradient_var_conv(conv_y, y)
        grad = grad_max + lam * grad_var

        # Normalize gradient to control step size
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-10:
            grad = grad / gnorm

        v = momentum * v - lr_init * grad
        a_new = project_simplex(a + v)

        conv = fast_convolve(a_new)
        cur_obj = conv.max()
        total_iters += 1

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a_new = a  # revert

        if no_improve >= perturb_every:
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            perturb_every = int(perturb_every * 1.3)
            lr_init = max(0.0003, lr_init * 0.7)

    return best_a, best_obj


def optimize_alternating(a0, time_limit, seed=0):
    """
    Alternate between variance-minimizing and max-minimizing phases.
    Variance reduction flattens the convolution, creating room for peak reduction.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    phase_time = min(30.0, time_limit / 10)
    cycle = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Alternate: even cycles focus on variance, odd on max
        if cycle % 4 in [0, 1]:
            # Variance-focused
            t_alloc = min(phase_time, remaining / 2)
            if t_alloc > 3:
                a, obj = optimize_variance_regularized(
                    a, t_alloc, beta_init=20, beta_end=80,
                    lam_init=1.0, lam_end=0.3,
                    lr_init=0.008, momentum=0.88,
                    perturb_every=100, perturb_scale=0.004,
                    seed=rng.integers(0, 100000))
        else:
            # Max-focused
            t_alloc = min(phase_time, remaining / 2)
            if t_alloc > 3:
                a, obj = optimize_variance_regularized(
                    a, t_alloc, beta_init=80, beta_end=400,
                    lam_init=0.1, lam_end=0.01,
                    lr_init=0.005, momentum=0.92,
                    perturb_every=150, perturb_scale=0.002,
                    seed=rng.integers(0, 100000))

        conv_cur = fast_convolve(a)
        cur_obj = conv_cur.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()

        # Occasionally perturb to escape local minima
        if cycle % 8 == 7:
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)

        cycle += 1

    conv_final = fast_convolve(best_a)
    return best_a, conv_final.max()


def smart_coordinate_descent(a0, time_limit, seed=0):
    """
    Targeted coordinate descent: at each step, find the pair (i, j) where
    moving mass from i to j most reduces the max convolution peak.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Find top peaks
        n_peaks = min(3, len(conv))
        peak_idx = np.argmax(conv)
        
        # For the main peak, find which pairs contribute most
        k = int(peak_idx)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        
        if i_start > i_end:
            stall += 1
            if stall > 20:
                break
            continue
        
        # Contributions to the peak
        contributions = np.array([a[i] * a[k - i] for i in range(i_start, i_end + 1)])
        if len(contributions) == 0:
            break
        
        # Find the pair with max contribution
        max_i = i_start + int(np.argmax(contributions))
        
        # Try moving mass from max_i to various other positions
        best_delta = 0
        best_j = -1
        best_frac = 0
        
        # Sample candidate target positions
        candidates = list(range(0, n, max(1, n // 50)))
        if len(candidates) < 20:
            candidates = list(range(n))
        # Also add some random
        candidates += [int(rng.integers(0, n)) for _ in range(10)]
        
        for j in candidates:
            if j == max_i:
                continue
            if a[j] > 0.99 / n:  # skip if target is already large
                continue
            
            for frac in [0.03, 0.07, 0.15]:
                amt = a[max_i] * frac
                if amt < 1e-10:
                    continue
                a_trial = a.copy()
                a_trial[max_i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                conv_trial = fast_convolve(a_trial)
                delta = max_val - conv_trial.max()
                if delta > best_delta:
                    best_delta = delta
                    best_j = j
                    best_frac = frac
        
        if best_j >= 0 and best_delta > 1e-8:
            a[max_i] -= a[max_i] * best_frac
            a[best_j] += a[max_i] * best_frac / (1 - best_frac) if max_i != best_j else 0
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 1e-10:
                a = a / s_a
            stall = 0
        else:
            # No improvement found, try random perturbation
            stall += 1
            if stall % 10 == 0:
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j and a[i] > 1e-8:
                    amt = a[i] * 0.1
                    a[i] -= amt
                    a[j] += amt
                    a = np.maximum(a, 0)
                    s_a = a.sum()
                    if s_a > 1e-10:
                        a = a / s_a
            if stall > 100:
                # Reset stall and try bigger perturbation
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                stall = 0
        
        conv = fast_convolve(a)
        cur_obj = conv.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()

    return best_a, best_obj


def peak_shaving_advanced(a0, time_limit, seed=0):
    """
    Advanced peak shaving: identifies all significant peaks simultaneously
    and finds a single mass redistribution that reduces multiple peaks at once.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Find all peaks above 95% of max
        peak_mask = conv > max_val * 0.95
        peak_positions = np.where(peak_mask)[0]
        
        if len(peak_positions) == 0:
            break
        
        # For each peak, find the dominant contributing pair
        # Then find a single move that helps multiple peaks
        best_improvement = 0
        best_move = None
        
        # Try reducing mass at positions that contribute to multiple peaks
        # Compute "peak load" for each position: sum of contributions to all peaks
        peak_load = np.zeros(n)
        for k in peak_positions:
            k = int(k)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                peak_load[i] += a[i] * a[k - i]
        
        # Top contributors to peaks
        top_contributors = np.argsort(peak_load)[::-1][:10]
        
        for i in top_contributors:
            i = int(i)
            if a[i] < 1e-8:
                continue
            # Try moving to various positions
            for j in range(0, n, max(1, n // 30)):
                if j == i:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    amt = a[i] * frac
                    if amt < 1e-10:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    conv_trial = fast_convolve(a_trial)
                    improvement = max_val - conv_trial.max()
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_move = (i, j, frac)
        
        if best_move is not None and best_improvement > 1e-8:
            i, j, frac = best_move
            a[i] *= (1 - frac)
            a[j] += a[i] * frac / (1 - frac)
            a = np.maximum(a, 0)
            s_a = a.sum()
            if s_a > 1e-10:
                a = a / s_a
            stall = 0
        else:
            stall += 1
            if stall > 50:
                noise = rng.standard_normal(n) * 0.008
                a = project_simplex(a + noise)
                stall = 0
            if stall > 200:
                break

        conv = fast_convolve(a)
        cur_obj = conv.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()

    return best_a, best_obj


def lp_direction(a, M, n):
    """Solve LP to find best direction reducing max convolution."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 20})
    except Exception:
        return None
    if not result.success or result.x is None:
        return None
    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    return b / b_sum


def lp_polish(a0, time_limit, seed=0):
    """LP-based polishing phase."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_convolve(a)
        M = conv_cur.max()
        
        b = None
        for M_factor in [1.0, 1.002, 1.01]:
            b = lp_direction(a, M * M_factor, n)
            if b is not None:
                break
        
        if b is None:
            stall += 1
            if stall > 5:
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                stall = 0
            if stall > 30:
                break
            continue
        
        stall = 0
        improved = False
        for t in [0.1, 0.2, 0.3, 0.5]:
            a_try = (1 - t) * a + t * b
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            if conv_try.max() < best_obj - 1e-10:
                best_obj = conv_try.max()
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break
            elif conv_try.max() < M:
                a = a_try.copy()
                if conv_try.max() < best_obj:
                    best_obj = conv_try.max()
                    best_a = a_try.copy()
                break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Edge-concentrated (try to make conv flat)
    edge = np.zeros(n)
    w = max(1, n // 8)
    for i in range(n):
        dist_to_edge = min(i, n - 1 - i)
        edge[i] = np.exp(-dist_to_edge / w)
    edge /= edge.sum()
    inits['edge'] = edge

    # Cosine window (smooth edge concentration)
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    cos_w = np.maximum(cos_w, 1e-8)
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w

    # Raised cosine squared
    cos2 = cos_w ** 2
    cos2 /= cos2.sum()
    inits['cos2'] = cos2

    # Bimodal Gaussian
    bi = np.exp(-((x - n * 0.1) / (n * 0.08)) ** 2) + np.exp(-((x - n * 0.9) / (n * 0.08)) ** 2)
    bi /= bi.sum()
    inits['bimodal_gauss'] = bi

    # Random Dirichlet
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand3'] = rng.dirichlet(np.ones(n) * 3.0)

    # Chebyshev-like (oscillatory)
    cheb = np.abs(np.cos(np.pi * x / (n - 1)))
    cheb = np.maximum(cheb, 1e-8)
    cheb /= cheb.sum()
    inits['cheb'] = cheb

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        if len(parent) < n:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining variance-regularized gradient descent,
    alternating objective phases, smart coordinate descent, and LP polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = evaluate_sequence(p.tolist())
            print(f"Parent: n={len(p)}, sum={p.sum():.2f}, score={parent_score:.6f}", flush=True)
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
    except Exception as e:
        print(f"No parent: {e}", flush=True)

    # Phase 1: Quick exploration with variance-regularized method
    print("=== Phase 1: Exploration ===", flush=True)
    lengths = [192, 256, 320, 384, 512]
    
    if parent is not None:
        # Ensure parent length is in our list
        pn = len(parent)
        if pn not in lengths:
            lengths.append(pn)
    
    scan_results = []
    t_p1 = time.monotonic()
    p1_budget = 300.0

    for n in lengths:
        if time.monotonic() > deadline - 400:
            break
        time_rem = p1_budget - (time.monotonic() - t_p1)
        if time_rem < 20:
            break

        inits = generate_initializations(n, parent, rng)
        priority = ['parent', 'parent_ds', 'parent_up', 'edge', 'cosine',
                    'bimodal_gauss', 'uniform', 'rand1', 'cos2']
        names = [nm for nm in priority if nm in inits]
        names = names[:5]

        per_init = max(5.0, time_rem / max(1, len(names) * len(lengths)))

        for name in names:
            if time.monotonic() > deadline - 380:
                break
            t_alloc = min(per_init, 20.0)
            if t_alloc < 4:
                break

            a0 = inits[name]
            # Use variance-regularized optimization
            a_opt, obj = optimize_variance_regularized(

                lam_init=0.5, lam_end=0.05,
                lr_init=0.012, momentum=0.9,

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}", flush=True)

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}", flush=True)

    # Phase 2: Deep optimization
    print("=== Phase 2: Deep Optimization ===", flush=True)
    
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        key = (n, name)
        if key not in seen:
            top_configs.append((n, a_opt))
            seen.add(key)
        if len(top_configs) >= 5:
            break

    if best_sequence is not None:
        top_configs.insert(0, (len(best_sequence), best_sequence.copy()))
    
    # Add a fresh random restart at the best length
    bn = len(best_sequence) if best_sequence is not None else 384

    remaining = deadline - time.monotonic() - 150.0

            break
        t_alloc = min(per_config, 250.0)
        if t_alloc < 15:
            break

        print(f"  Config {i}: n={n}", flush=True)

        t_a = t_alloc * 0.40
        if t_a > 10:
            a_a, obj_a = optimize_alternating(a0, t_a, seed=rng.integers(0, 100000))
            score_a = float(evaluate_sequence(a_a.tolist()))
            if np.isfinite(score_a) and score_a < best_score:
                best_score = score_a
                best_sequence = a_a.copy()
                print(f"    Alt: {score_a:.6f}", flush=True)

        # Sub-phase B: Variance-regularized gradient (30%)
        t_b = t_alloc * 0.30
        if t_b > 10:
            a_b, obj_b = optimize_variance_regularized(
                a_a, t_b, beta_init=100, beta_end=500,
                lam_init=0.3, lam_end=0.01,
                lr_init=0.004, momentum=0.93,
                perturb_every=150, perturb_scale=0.002,
                seed=rng.integers(0, 100000))
            score_b = float(evaluate_sequence(a_b.tolist()))
            if np.isfinite(score_b) and score_b < best_score:
                best_score = score_b
                best_sequence = a_b.copy()
                print(f"    VarGrad: {score_b:.6f}", flush=True)
            a_a = a_b

        # Sub-phase C: Smart coordinate descent (20%)
        t_c = t_alloc * 0.20
        if t_c > 8:
            a_c, obj_c = smart_coordinate_descent(a_a, t_c, seed=rng.integers(0, 100000))
            score_c = float(evaluate_sequence(a_c.tolist()))
            if np.isfinite(score_c) and score_c < best_score:
                best_score = score_c
                best_sequence = a_c.copy()
                print(f"    Coord: {score_c:.6f}", flush=True)

        # Sub-phase D: LP polish (10%)
        t_d = t_alloc * 0.10
        if t_d > 5:
            a_d, obj_d = lp_polish(a_a, t_d, seed=rng.integers(0, 100000))
            score_d = float(evaluate_sequence(a_d.tolist()))
            if np.isfinite(score_d) and score_d < best_score:
                best_score = score_d
                best_sequence = a_d.copy()
                print(f"    LP: {score_d:.6f}", flush=True)

    # Phase 3: Final polishing
    print("=== Phase 3: Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0

        # Advanced peak shaving
        t_ps = remaining * 0.3
        if t_ps > 10:
            a_ps, _ = peak_shaving_advanced(best_sequence, t_ps, seed=777)
            score_ps = float(evaluate_sequence(a_ps.tolist()))
            if np.isfinite(score_ps) and score_ps < best_score:
                best_score = score_ps
                best_sequence = a_ps.copy()
                print(f"  PeakShave: {best_score:.6f}", flush=True)

        # Fine variance-regularized with high beta
        remaining = deadline - time.monotonic() - 20.0
        t_fg = remaining * 0.35
        if t_fg > 10:
            a_fg, _ = optimize_variance_regularized(
                best_sequence, t_fg, beta_init=200, beta_end=1000,
                lam_init=0.1, lam_end=0.005,
                lr_init=0.002, momentum=0.95,
                perturb_every=100, perturb_scale=0.001,
                seed=1234)
            score_fg = float(evaluate_sequence(a_fg.tolist()))
            if np.isfinite(score_fg) and score_fg < best_score:
                best_score = score_fg
                best_sequence = a_fg.copy()
                print(f"  FineGrad: {score_fg:.6f}", flush=True)

        # Another coordinate descent pass
        remaining = deadline - time.monotonic() - 15.0
        t_cd = remaining * 0.2
        if t_cd > 8:
            a_cd, _ = smart_coordinate_descent(best_sequence, t_cd, seed=555)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  CoordFinal: {score_cd:.6f}", flush=True)

        # Final LP
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 15:
            a_lp, _ = lp_polish(best_sequence, remaining * 0.8, seed=999)
            score_lp = float(evaluate_sequence(a_lp.tolist()))
            if np.isfinite(score_lp) and score_lp < best_score:
                best_score = score_lp
                best_sequence = a_lp.copy()
                print(f"  LPFinal: {score_lp:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]
