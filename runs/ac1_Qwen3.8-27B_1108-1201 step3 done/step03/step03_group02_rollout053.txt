    fft_size = 1 << (size - 1).bit_length()

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

        log_mag = np.log(np.maximum(mag, 1e-15))

        target_mag = np.exp(target_log)

        safe_mag = np.maximum(mag, 1e-15)
        ratio = (target_mag / safe_mag) ** alpha

    """Mirror descent (multiplicative update) for minimizing peak convolution."""
    a = normalize(a.copy())
    n = len(a)
    t0 = time.monotonic()
    eta = eta_init
    conv = fast_convolve(a)
    best_a = a.copy()
    best_obj = 2.0 * n * conv.max()
    
    while time.monotonic() - t0 < time_limit:
        # Find weighted peaks
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        top_vals = conv[top_idx]
        
        # Temperature-based weighting
        scaled = top_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Compute weighted subgradient
        sg = np.zeros(n)
        for i, k in enumerate(top_idx):
            k = int(k)
            j = np.arange(n)
            idx = k - j
            valid = (idx >= 0) & (idx < n)
            sg[valid] += w[i] * a[idx[valid]]

        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        eta *= eta_decay
    
    return best_a, best_obj

def build_conv_matrix(a):
    """Build matrix where (A @ b)[k] = conv(a, b)[k]."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    step = step_size

        if time.monotonic() - t0 > time_limit:
            break
        
        M = conv.max()
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=conv_mat, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status != 0:

        b = np.maximum(res.x, 0)
        b_sum = b.sum()
        if b_sum < 1e-15:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Spectral flattening (iterative power spectrum equalization via FFT)
2. Mirror descent (multiplicative updates preserving non-negativity)
3. LP-based refinement (max sum direction under cross-correlation constraint)
4. Subgradient polishing with perturbation restarts
Key innovation: spectral domain shaping to find sequences with near-flat
power spectrum, reducing peak auto-convolution below uniform baseline.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def evaluate_obj(a):
    """Compute objective 2*n*max_conv/sum^2 for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def spectral_flatten(a, n_iters=3000, alpha=0.25, time_limit=None):
    """
    Iteratively flatten the power spectrum via multiplicative magnitude update.
    Keeps DC component fixed (sum=1 constraint).
    """
    a = normalize(a.copy())
    n = len(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break

        A = np.fft.fft(a)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: geometric mean of non-DC magnitudes
        non_dc_mag = mag[1:]
        log_non_dc = np.log(np.maximum(non_dc_mag, 1e-15))
        target_non_dc = np.exp(np.mean(log_non_dc))

        # Soft update: new_mag = mag * (target/mag)^alpha
        safe_mag = np.maximum(mag, 1e-15)
        ratio = (target_non_dc / safe_mag[1:]) ** alpha
        new_mag = mag.copy()
        new_mag[1:] = safe_mag[1:] * ratio
        # Keep DC fixed
        new_mag[0] = mag[0]

        A_new = phase * new_mag
        a = normalize(np.real(np.fft.ifft(A_new)))

    return a


def spectral_flatten_compressive(a, n_iters=2000, beta=0.5, time_limit=None):
    """
    Compressive spectral flattening: target = 1/(|A|^beta + eps).
    More aggressively pushes down spectral peaks.
    """
    a = normalize(a.copy())
    n = len(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break

        A = np.fft.fft(a)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target inversely proportional to current magnitude (compresses peaks)
        target = 1.0 / (mag[1:] ** beta + 1e-6)
        # Normalize target to have same mean as current
        target = target * (np.mean(mag[1:]) / np.mean(target))

        A_new = A.copy()
        A_new[1:] = phase[1:] * target
        # DC stays
        a = normalize(np.real(np.fft.ifft(A_new)))

    return a


def mirror_descent(a, time_limit, eta_init=2.0, eta_decay=0.9995, top_k=10, temp=0.0005):
    """
    Mirror descent (multiplicative update) for minimizing peak convolution.
    Naturally preserves non-negativity.
    """
    a = normalize(a.copy())
    n = len(a)
    t0 = time.monotonic()
    eta = eta_init
    conv = fast_convolve(a)
    best_a = a.copy()
    best_obj = 2.0 * n * conv.max()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Find weighted top peaks
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        top_vals = conv[top_idx]

        # Temperature-based weighting
        scaled = top_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(top_k)
            w[np.argmax(top_vals)] = 1.0
        else:
            w /= w_sum

        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(top_idx)):
            if w[i] < 1e-12:
                continue
            k = int(top_idx[i])
            j = np.arange(n)
            idx = k - j
            valid = (idx >= 0) & (idx < n)
            sg[valid] += w[i] * a[idx[valid]]

        # Multiplicative update (mirror descent)
        a = a * np.exp(-eta * sg)
        a = normalize(a)
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
            eta = min(eta * 1.01, 20.0)
        else:
            no_improve += 1

        eta *= eta_decay

        if no_improve > 500:
            # Perturb and reset
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            eta = eta_init
            no_improve = 0

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_refine(a, time_limit, max_iters=200, step_init=0.35):
    """LP-based refinement: find b maximizing sum(b) s.t. conv(a,b) <= M, b >= 0."""
    a = normalize(a.copy())
    n = len(a)
    conv = fast_convolve(a)
    best_a = a.copy()
    best_obj = 2.0 * n * conv.max()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = conv.max()
        conv_mat = build_conv_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=conv_mat, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True})
        except Exception:
            break

        if res.status != 0:
            break

        b = np.maximum(res.x, 0)
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b /= b_sum

        if np.linalg.norm(b - a) < 1e-10:
            break

        a_new = (1.0 - step) * a + step * b
        a_new = normalize(a_new)
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            step = min(0.8, step * 1.05)
        else:
            no_improve += 1
            step = max(0.05, step * 0.9)

        a = a_new
        conv = conv_new

        if no_improve > 100:
            break

    return best_a, best_obj


def subgradient_refine(a, time_limit, lr_init=0.04, momentum=0.88, top_k=20,
                       temp_init=0.0003, perturb_every=3000, perturb_scale=0.004):
    """Projected subgradient with momentum and periodic perturbation."""
    a = normalize(a.copy())
    n = len(a)
    t0 = time.monotonic()
    v = np.zeros(n)
    lr = lr_init
    temp = temp_init
    conv = fast_convolve(a)
    best_a = a.copy()
    best_obj = 2.0 * n * conv.max()
    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        top_vals = conv[top_idx]
        scaled = top_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(top_vals))
            w[np.argmax(top_vals)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        for i in range(len(top_idx)):
            if w[i] < 1e-12:
                continue
            k = int(top_idx[i])
            j = np.arange(n)
            idx = k - j
            valid = (idx >= 0) & (idx < n)
            sg[valid] += w[i] * a[idx[valid]]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997

        if no_improve >= perturb_every:
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            perturb_every = int(perturb_every * 1.4)
            lr = max(lr_init * 0.4, lr * 0.6)

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0))
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    # Hanning window
    if n > 1:
        h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n - 1)))
        h = np.maximum(h, 0)
        hs = h.sum()
        if hs > 1e-10:
            inits['hanning'] = h / hs

    # Blackman-Harris-like
    if n > 1:
        bh = (0.35875
              - 0.48829 * np.cos(2.0 * np.pi * x / (n - 1))
              + 0.14128 * np.cos(4.0 * np.pi * x / (n - 1))
              - 0.01168 * np.cos(6.0 * np.pi * x / (n - 1)))
        bh = np.maximum(bh, 0)
        bs = bh.sum()
        if bs > 1e-10:
            inits['blackman'] = bh / bs

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.3 * n)) ** 2)
    g = np.maximum(g, 0)
    gs = g.sum()
    if gs > 1e-10:
        inits['gauss'] = g / gs

    # Random Dirichlet
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits['random'] = r

    # Raised cosine squared
    if n > 1:
        rc = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2
        rc = np.maximum(rc, 0)
        rcs = rc.sum()
        if rcs > 1e-10:
            inits['rcos_sq'] = rc / rcs

    return inits

    print(f"Phase 6 done: best_score={best_score:.6f}, elapsed={time.monotonic()-t6:.0f}s")

    # Final evaluation
    if best_sequence is None:
        best_sequence = np.ones(N) / N

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
