def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def compute_subgradient(f, conv, top_k=64, temp=0.001):
    """Compute subgradient of max(conv) with softmax weighting over top peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]

    return sg

def generate_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Linear ramp decreasing
    ramp_d = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp_d / ramp_d.sum())

    # Linear ramp increasing
    ramp_i = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp_i / ramp_i.sum())

    # Quadratic (U-shaped)
    q = (x - n / 2) ** 2 + 1.0
    inits.append(q / q.sum())

    # Exponential decay from left
    e1 = np.exp(-x / (n / 3.0))
    inits.append(e1 / e1.sum())

    # Exponential decay from right
    e2 = np.exp(-x[::-1] / (n / 3.0))
    inits.append(e2 / e2.sum())

    # Cosine window
    c = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 4 + 0.01
    inits.append(c / c.sum())

    # Power law
    pw = (x + 1.0) ** 0.5
    inits.append(pw / pw.sum())

    # Inverse power law
    ipw = 1.0 / (x + 1.0) ** 0.5
    inits.append(ipw / ipw.sum())

    # Sine
    sn = np.sin(np.pi * x / n) + 0.05
    inits.append(sn / sn.sum())

    # Random Dirichlet
    r = rng.dirichlet(np.ones(n) * 0.3)
    inits.append(r)

    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Blend with ramp
            blended = 0.5 * (px / ps) + 0.5 * (ramp_d / ramp_d.sum())
            inits.append(blended / blended.sum())
            blended2 = 0.5 * (px / ps) + 0.5 * (ramp_i / ramp_i.sum())
            inits.append(blended2 / blended2.sum())

    return inits
