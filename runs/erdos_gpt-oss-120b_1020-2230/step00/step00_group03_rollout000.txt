import numpy as np

def run(seed: int = 42, budget_s: float = 110, **kwargs):
    """Search for a step function h on [0,2] (sampled) with small
    max_k ∫ h(x)(1‑h(x+k))dx.
    Returns (h_values, c5_bound, n_points)."""
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # initialise number of points and starting binary vector
    # ------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]
        if n % 2:                     # enforce even length for binary set‑up
            init = init[:-1]
            n -= 1
    except Exception:                # no initial_h_values supplied
        init = None
        n = 70                         # default size (even)

    if n % 2:                         # double‑check evenness
        n += 1

    target_ones = n // 2               # we need exactly half the points equal 1
    # create a binary start vector
    if init is not None and init.shape[0] == n:
        h = (init > 0.5).astype(float)
        cur = int(h.sum())
        if cur != target_ones:
            diff = target_ones - cur
            if diff > 0:
                zeros = np.where(h == 0)[0]
                h[np.random.choice(zeros, diff, replace=False)] = 1.0
            else:
                ones = np.where(h == 1)[0]
                h[np.random.choice(ones, -diff, replace=False)] = 0.0
    else:
        h = np.zeros(n, dtype=float)
        h[np.random.choice(n, target_ones, replace=False)] = 1.0

    dx = 2.0 / n

    # ------------------------------------------------------------------
    # helper to evaluate C5
    # ------------------------------------------------------------------
    def eval_c5(arr):
        conv = np.correlate(arr, 1.0 - arr, mode="full")
        return np.max(conv) * dx

    # ------------------------------------------------------------------
    # greedy / simulated‑annealing main loop
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_c5 = eval_c5(best_h)

    cur_h = h.copy()
    cur_c5 = best_c5

    start = time.time()
    TIME_LIMIT = budget_s * 0.95        # leave a safety margin

    # parameters for occasional random perturbations
    steps_without_improve = 0
    RESTART_AFTER = 20000               # steps without improvement → random restart

    while time.time() - start < TIME_LIMIT:

        full_corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
