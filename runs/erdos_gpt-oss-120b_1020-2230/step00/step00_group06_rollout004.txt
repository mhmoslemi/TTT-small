def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return dx * np.max(corr)

def repair_sum(h, target):
    """Adjust first element to fix sum, ensuring bounds."""
    sum_h = h.sum()
    diff = target - sum_h
    if abs(diff) < 1e-12:
        return h
    # try to add diff to first element within bounds
    for i in range(len(h)):
        low = 0.0
        high = 1.0
        if low <= h[i] + diff <= high:
            h[i] += diff
            return h
    # if not possible, spread across two elements
    # find two indices i,j that can adjust
    for i in range(len(h)):
        # possible delta to add to i (positive diff)
        if diff > 0:
            delta_i = min(1 - h[i], diff)
        else:
            delta_i = max(-h[i], diff)
        # adjust
        if abs(delta_i) > 1e-12:
            h[i] += delta_i
            diff -= delta_i
            if abs(diff) < 1e-12:
                return h
    # if leftover diff (should not happen)
    raise RuntimeError("Cannot repair sum")
    return h

import time
import numpy as np
import random

def _c5_value(h, dx):
    """Return the overlap C5 for a step function given by samples h."""
    # correlation of h with 1-h
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _random_feasible(n, target):
    """Construct a random feasible vector with entries in [0,1] and sum == target."""
    # Put floor(target) entries at 1, one entry possibly fractional, rest 0.
    whole = int(np.floor(target))
    frac = target - whole
    h = np.zeros(n, dtype=np.float64)
    idx = np.random.permutation(n)
    if whole > 0:
        h[idx[:whole]] = 1.0
    if frac > 0:
        h[idx[whole]] = frac
    return h
