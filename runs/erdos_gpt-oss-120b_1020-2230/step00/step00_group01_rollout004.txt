import numpy as np
import time

def _cmax(h):
    """Maximum overlap integral for a discrete step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def _proj_box_simplex(v, target):
    """
    Project vector v onto the intersection of the box [0,1]^n and the simplex
    {w : sum w = target}.  Uses a simple bisection on the threshold τ such that
    w_i = clip(v_i - τ, 0, 1) and sum w_i = target.
    """
    lo, hi = -1.0, np.max(v)  # lo yields all ones, hi yields all zeros
    for _ in range(50):
        tau = (lo + hi) * 0.5
        w = np.clip(v - tau, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) * 0.5
    return np.clip(v - tau, 0.0, 1.0)

def _gradient(h, shift):
    """Gradient of the active overlap term w.r.t. h."""
    n = h.size
    g = np.ones(n)
    if shift >= 0:
        # subtract h_{i+shift}
        g[: n - shift] -= h[shift:]
        # subtract h_{i-shift}
        g[shift:] -= h[: n - shift]
    else:
        s = -shift
        # subtract h_{i-s}
        g[s:] -= h[: n - s]
        # subtract h_{i+s}
        g[: n - s] -= h[s:]
    return g

def run(seed=42, budget_s=110, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫h = 1 and 0≤h≤1
    to minimise the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------
    # 1. Choose discretisation size
    # ------------------------------------------------------------
    # Use a modest size that still gives good resolution.
    n = 200
    dx = 2.0 / n
    target_sum = n / 2.0                     # because ∑h * dx = 1

    # ------------------------------------------------------------
    # 2. Initialise h
    # ------------------------------------------------------------
    if 'initial_h_values' in globals():
        # use the provided construction if its size matches our target sum
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = h0.size
        if n0 != n:
            # resample to our grid (linear interpolation)
            xs = np.linspace(0.0, 2.0, n0, endpoint=False)
            xt = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(xt, xs, h0)
        # project onto constraints
        h = _proj_box_simplex(h0, target_sum)
    else:
        # start from a triangular “tent’’ shape (area exactly 1)
        x = np.arange(n) * dx
        h = 1.0 - np.abs(x - 1.0)            # peak 1 at x=1, zero at edges
        # scale so that the integral is exactly 1
        h = _proj_box_simplex(h, target_sum)

    # ------------------------------------------------------------
    # 3. Simple sub‑gradient descent with projection
    # ------------------------------------------------------------
    best_h = h.copy()
    best_c = _cmax(best_h)

    it = 0
    # parameters for step size and occasional stochastic jumps
    base_step = 0.5
    prob_worse = 0.02
    while time.time() - start_time < budget_s * 0.94:
        it += 1
        # compute all overlaps, locate the worst lag
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        cur_c = corr.max()
        s_idx = corr.argmax()
        shift = s_idx - (n - 1)               # signed shift corresponding to max

        # gradient of the active term (sub‑gradient of the max)
        g = _gradient(h, shift)

        # remove the mean to stay on the hyperplane sum(h)=target_sum
        d = -(g - g.mean())

        # diminishing step size
        step = base_step / np.sqrt(it)

        # propose a new point and project back onto feasible set
        v = h + step * d
        h_new = _proj_box_simplex(v, target_sum)

        new_c = _cmax(h_new)

        # accept if improvement, otherwise occasional uphill move
        if new_c < cur_c or np.random.rand() < prob_worse:
            h = h_new
            cur_c = new_c

        # keep the best ever seen
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        # occasional random perturbation to escape plateaus
        if it % 500 == 0:
            eps = 0.02
            i, j = np.random.choice(n, 2, replace=False)
            delta = np.random.uniform(-eps, eps)
            # clamp the move so bounds stay satisfied
            lo = max(-delta, -best_h[i])
            hi = min(eps - delta, 1.0 - best_h[i])
            delta = np.clip(delta, lo, hi)
            # ensure the other coordinate stays in [0,1]
            delta = np.clip(delta, best_h[j] - 1.0, best_h[j])
            trial = best_h.copy()
            trial[i] += delta
            trial[j] -= delta
            trial = _proj_box_simplex(trial, target_sum)
            trial_c = _cmax(trial)
            if trial_c < best_c:
                best_c, best_h = trial_c, trial

    # final polishing with a very small fixed step
    for _ in range(300):
        corr = np.correlate(best_h, 1.0 - best_h, mode="full") * dx
        shift = corr.argmax() - (n - 1)
        g = _gradient(best_h, shift)
        d = -(g - g.mean())
        v = best_h + 0.02 * d
        best_h = _proj_box_simplex(v, target_sum)

    final_c = _cmax(best_h)

    # Return as plain Python list for compatibility
    return best_h.tolist(), float(final_c), int(n)
