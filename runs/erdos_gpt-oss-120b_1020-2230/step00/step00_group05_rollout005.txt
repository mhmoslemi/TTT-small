def project_to_box_sum(v, target):
    lo = -1.0
    hi = 1.0
    # extend bounds wide enough
    for _ in range(40):
        lam = (lo + hi) / 2.0
        p = np.clip(v - lam, 0.0, 1.0)
        s = p.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    p = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return p

import time, math
import numpy as np

def _proj_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {0≤x≤1, sum(x)=target}."""
    lo, hi = -1.0, 2.0
    for _ in range(40):
        lam = (lo + hi) * 0.5
        p = np.clip(v - lam, 0.0, 1.0)
        if p.sum() > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _max_corr_and_shift(h: np.ndarray):
    """Return (max correlation value, shift index) for h."""
    corr = np.convolve(h, (1.0 - h)[::-1], mode="full")
    idx = int(corr.argmax())
    return float(corr[idx]), idx


def _gradient_wrt_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of the worst‑shift correlation."""
    n = h.shape[0]
    g = np.ones_like(h)
    if shift >= 0:
        m = shift
        g[: n - m] -= h[m:]
        g[m:] -= h[: n - m]
    else:
        m = -shift
        g[m:] -= h[: n - m]
        g[: n - m] -= h[m:]
    return g


def _c5_value(max_corr: float, n: int) -> float:
    """Convert raw correlation to C5."""
    return max_corr * (2.0 / n)


def run(seed: int = 42, budget_s: float = 110, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Initialise h (respecting the required constraints)
    # ------------------------------------------------------------------
    try:
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.shape[0]
    except NameError:                     # no parent construction supplied
        n = 200
        h = np.zeros(n, dtype=np.float64)
        h[: n // 2] = 1.0                   # simple binary start
        rng.shuffle(h)

    target_sum = n / 2.0
    h = _proj_box_sum(h, target_sum)       # guarantee feasibility

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    best_h = h.copy()
    max_corr, _ = _max_corr_and_shift(h)
    best_val = _c5_value(max_corr, n)

    max_iters = 30000
    step0 = 0.5

    for it in range(max_iters):
        if time.time() - start > budget_s * 0.95:
            break

        # evaluate current worst correlation and its shift
        max_corr, idx = _max_corr_and_shift(h)
        cur_val = _c5_value(max_corr, n)

        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            # early exit if already better than target
            if best_val <= 0.3808:
                break

        shift = idx - (n - 1)

        # sub‑gradient step
        g = _gradient_wrt_shift(h, shift)
        step = step0 / (1.0 + it * 0.001)      # slowly decreasing step size
        h_new = h - step * g
        h = _proj_box_sum(h_new, target_sum)

        # occasional random jitter to escape shallow minima
        if it % 2000 == 0 and it > 0:
            noise = rng.normal(scale=0.02, size=n)
            h = _proj_box_sum(np.clip(h + noise, 0.0, 1.0), target_sum)

    # ------------------------------------------------------------------
    # Final evaluation (in case the loop stopped early)
    # ------------------------------------------------------------------
    max_corr, _ = _max_corr_and_shift(best_h)
    best_val = _c5_value(max_corr, n)

    return best_h.tolist(), best_val, n
