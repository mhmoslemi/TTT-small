    target_ones = n//2
    dx = 2.0 / n

    # Create binary start from h0 if needed:
    # Take top target_ones indices as 1, others 0.
    idx = np.argsort(-h0)[:target_ones]
    h_bin = np.zeros(n, dtype=int)
    h_bin[idx] = 1

    # Helper to compute C5
    def compute_c5(arr):
        # arr is 0/1 int array
        return np.max(np.correlate(arr, 1-arr, mode='full')) * dx

    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)

    # Also consider starting from random restarts
    # We'll iterate until time limit.
    max_iter_total = 200000  # large enough
    iteration = 0
    # Pre-create list of positions
    # We'll use a simple SA.
    current_h = best_h.copy()
    current_c5 = best_c5
    # Initial temperature schedule
    start_temp = 0.1
    end_temp = 1e-4

    while time.time() - t0 < budget_s * 0.95:
        # Possibly restart after many iterations
        if iteration % 50000 == 0 and iteration > 0:
            # restart from best or random
            if rng.random() < 0.5:
                # restart from best
                current_h = best_h.copy()
                current_c5 = best_c5
            else:
                # random binary
                perm = rng.permutation(n)
                current_h = np.zeros(n, dtype=int)
                current_h[perm[:target_ones]] = 1
                current_c5 = compute_c5(current_h)
        # Temperature schedule
        t_frac = iteration / max_iter_total
        T = start_temp * (1 - t_frac) + end_temp * t_frac

        # propose swap
        # pick random 1 index and 0 index
        ones = np.where(current_h == 1)[0]
        zeros = np.where(current_h == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new_h = current_h.copy()
        new_h[i] = 0
        new_h[j] = 1

                best_h = current_h.copy()
