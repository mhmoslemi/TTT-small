import time, math, random
import numpy as np

def run(seed: int = 42, budget_s: float = 110, **kwargs):
    """
    Return a step function h on [0,2] (as a NumPy array), its C5 value
    and the number of sample points.
    The algorithm starts from the provided `initial_h_values` if present,
    otherwise from the uniform 0.5 function.  It performs a simulated‑annealing
    style search that respects the constraints:
        0 ≤ h[i] ≤ 1,   sum(h) = n/2 .
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------ #
    # 1.  Initialise h
    # ------------------------------------------------------------------ #
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        n = 42
        h = np.full(n, 0.5, dtype=np.float64)      # already satisfies the sum constraint
    n = h.size
    target_sum = n / 2.0
    dx = 2.0 / n

    # If the initialization does not satisfy the sum exactly (tiny rounding), fix it
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        # adjust the first element while staying in [0,1]
        h[0] = min(max(h[0] + diff, 0.0), 1.0)

    # ------------------------------------------------------------------ #
    # 2.  Helper functions
    # ------------------------------------------------------------------ #
    def c5_of(vec):
        """C5 for a given vector (full correlation)."""
        conv = np.correlate(vec, 1.0 - vec, mode='full')
        return conv.max() * dx

    def neighbour_sum(vec, shift):
        """
        For each index i return h[i+shift] + h[i-shift] where out‑of‑range
        terms are treated as 0.  Used for the gradient of a single shift.
        """
        out = np.zeros_like(vec)
        if shift >= 0:
            out[: n - shift]   += vec[shift:]
            out[shift:]        += vec[: n - shift]
        else:
            s = -shift
            out[s:]            += vec[: n - s]
            out[: n - s]       += vec[s:]
        return out

    # ------------------------------------------------------------------ #
    # 3.  Simulated annealing / greedy search
    # ------------------------------------------------------------------ #
    best_h = h.copy()
    best_val = c5_of(best_h)
    cur_val = best_val

    start = time.time()
    max_time = budget_s * 0.99          # leave a tiny margin
    iter_cnt = 0
    no_improve = 0
    # annealing parameters
    T_start = 0.015
    T_min   = 1e-8

    while True:
        now = time.time()
        if now - start >= max_time:
            break

        # current correlation and worst shift
        conv = np.correlate(h, 1.0 - h, mode='full')
        cur_val = conv.max() * dx
        worst_idx = int(conv.argmax())
        shift = worst_idx - (n - 1)          # shift corresponding to the max overlap

        # compute indices that would improve the worst shift
        ns = neighbour_sum(h, shift)        # h[i+shift] + h[i-shift]
        inc_idx = np.where(ns > 1.0 + 1e-12)[0]
        dec_idx = np.where(ns < 1.0 - 1e-12)[0]

        if inc_idx.size and dec_idx.size:
            i = int(np.random.choice(inc_idx))
            j = int(np.random.choice(dec_idx))
        else:
            i, j = np.random.choice(n, 2, replace=False)

        hi, hj = h[i], h[j]
        lower = max(-hi, hj - 1.0)
        upper = min(1.0 - hi, hj)
        if lower >= upper:
            # nothing feasible – try another pair next iteration
            continue

        # limit step size to keep changes modest (helps stability)
        max_step = 0.2
        delta = np.random.uniform(lower, upper)
        if abs(delta) > max_step:
            delta = math.copysign(max_step, delta)

        # propose new vector
        new_h = h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta

        new_val = np.correlate(new_h, 1.0 - new_h, mode='full').max() * dx

        # temperature based on elapsed fraction
        elapsed = now - start
        T = max(T_min, T_start * (1.0 - elapsed / budget_s))

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis criterion
            prob = math.exp(-(new_val - cur_val) / T) if T > 0 else 0.0
            if random.random() < prob:
                accept = True

        if accept:
            h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional random jitter to escape plateaus
        if no_improve > 5000:
            for _ in range(5):
                a, b = np.random.choice(n, 2, replace=False)
                ha, hb = h[a], h[b]
                lo = max(-ha, hb - 1.0)
                hi_ = min(1.0 - ha, hb)
                if lo < hi_:
                    d = np.random.uniform(lo, hi_)
                    d = max(min(d, 0.1), -0.1)   # keep jitter small
                    h[a] = ha + d
                    h[b] = hb - d
            cur_val = c5_of(h)
            no_improve = 0

        iter_cnt += 1

    # ------------------------------------------------------------------ #
    # 4.  Final evaluation and return
    # ------------------------------------------------------------------ #
    # Ensure exact sum constraint (tiny adjustment on a free index)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        # find an index that can absorb the diff without violating bounds
        idx = np.argmin(best_h) if diff > 0 else np.argmax(best_h)
        best_h[idx] = min(max(best_h[idx] + diff, 0.0), 1.0)

    # final C5 value (using the official helper if available)
    if 'evaluate_erdos_solution' in globals():
        final_c5 = evaluate_erdos_solution(best_h)
    else:
        final_c5 = c5_of(best_h)

    return best_h, float(final_c5), n
