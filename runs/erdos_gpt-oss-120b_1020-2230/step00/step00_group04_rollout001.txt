def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _fix_sum(h: np.ndarray, target: float) -> np.ndarray:
    """
    Adjust one entry so that sum(h) equals *target*.
    Assumes the discrepancy is tiny (numeric noise).
    """
    diff = target - h.sum()
    if abs(diff) < 1e-12:

    else:

    else:

    return h
