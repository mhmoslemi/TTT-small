import time
import numpy as np

def _evaluate(h):
    """Return max overlap C5 for a discrete h."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h and (1-h)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c.max()


def _project_to_sum(v, target):
    """
    Projection of vector v onto the feasible set:
        0 <= x_i <= 1  and  sum(x) = target
    Implements the classic water‑filling / bisection method.
    """
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(40):
        tau = (lo + hi) * 0.5
        x = np.clip(v - tau, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny residual fix
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            # spread the remainder on any admissible entry
            for i in range(len(x)):
                if diff > 0 and x[i] < 1.0:
                    add = min(diff, 1.0 - x[i])
                    x[i] += add
                    diff -= add
                elif diff < 0 and x[i] > 0.0:
                    sub = min(-diff, x[i])
                    x[i] -= sub
                    diff += sub
                if abs(diff) < 1e-12:
                    break
    return x

    """
    Optimize a step function on [0,2] for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).

    # Initialise h

    h = _project_to_sum(h0, target_sum)

    cur_score = _evaluate(h)
    best_h = h.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    start_time = time.time()
    end_time = start_time + budget_s * 0.95   # safety margin
    dx = 2.0 / n
    alpha = 0.5 * dx          # initial step size
    min_alpha = 1e-6
    iteration = 0

    # ------------------------------------------------------------------
    while time.time() < end_time:

        # 1) locate the worst shift (max overlap)
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))

        g = np.zeros(n, dtype=float)
