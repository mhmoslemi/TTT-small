def gradient_of_max(h):
    n=len(h)
    c = np.correlate(h, 1-h, mode='full')
    max_val = c.max()
    max_idxs = np.flatnonzero(c == max_val)
    grad = np.zeros_like(h)
    for idx in max_idxs:
        ell = idx - (n-1)
        # compute term: shift forward ℓ => shift left for h[i+ℓ] (if ell>0)
        # Actually arrays:
        # – For ell >= 0:
        #   h[i+ell] exists for i=0..n-ell-1; else 0.
        #   h[i-ell] exists for i=ell..n-1 => h[i-ell] exists for i >= ell.
        # Implementation can use np.roll with masking length.

        # Use vector of same shape:
        tmp = np.zeros_like(h, dtype=float)
        if ell >= 0:
            tmp[:n-ell] += h[ell:]    # h[i+ell] for i up to n-ell-1
            tmp[ell:] += h[:n-ell]    # h[i-ell] for i>=ell
        else:
            # ell negative: shift in opposite direction
            ell2 = -ell
            tmp[ell2:] += h[:n-ell2] # i+ell => i - ell2 => h[i-ell2]; indices i>=ell2
            tmp[:n-ell2] += h[ell2:] # i-ell => i + ell2
        grad += (1 - tmp)
    grad /= len(max_idxs)
    return grad

    lo = -1.0
    hi = 2.0
    for _ in range(60):  # enough for precision
        mid = (lo + hi)/2
        v = np.clip(h - mid, 0.0, 1.0)

import time

def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap integral for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full").max() * dx


def _project_onto_simplex_box(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {0≤x≤1, sum(x)=n/2}."""
    n = v.shape[0]
    target = n / 2.0

    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    # tiny correction for numeric drift
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        mask = (w > 1e-12) & (w < 1.0 - 1e-12)
        if mask.any():
            w[mask] += diff / mask.sum()
        else:
            w += diff / n
        w = np.clip(w, 0.0, 1.0)
    return w


def _grad_of_max_corr(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the correlation term attaining the current maximum."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")

            shift[e:] += h[: n - e]
            shift[: n - e] += h[e:]

        grad += 1.0 - shift
    grad /= len(idxs)
    return grad

    # final fine‑tuning (tiny gradient steps)
    # ------------------------------------------------------------------
    for _ in range(120):
        g = _grad_of_max_corr(best_h)
        cand = _project_onto_simplex_box(best_h - 0.001 * g)
        cand_val = _c5_value(cand)
        if cand_val < best_val - 1e-12:
            best_h = cand
            best_val = cand_val
        else:
            break

    best_h = _project_onto_simplex_box(best_h)
    best_val = _c5_value(best_h)

    return best_h.tolist(), float(best_val), n
