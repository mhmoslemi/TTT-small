    a, b = random indices distinct.
    # compute maximum delta that keeps h[a] in [0,1] and h[b] in [0,1]
    max_up_a = 1.0 - h[a]
    max_down_a = -h[a]  # negative allowed for lowering a
    max_up_b = 1.0 - h[b]
    max_down_b = -h[b]
    # delta will be added to a, sub from b, so delta must satisfy
    # h[a] + delta ∈ [0,1] and h[b] - delta ∈ [0,1]
    delta_lower = max(-h[a], h[b] - 1)   # min delta (negative) allowed
    delta_upper = min(1 - h[a], h[b])     # max delta (positive) allowed
    delta = np.random.uniform(delta_lower, delta_upper)
    new_h = h.copy()
    new_h[a] += delta
    new_h[b] -= delta
    return new_h, delta

def _c5_value(h):
    """Return the C5 bound for a vector h."""
    # correlation of h with (1-h); mode='full' gives all integer shifts
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)
