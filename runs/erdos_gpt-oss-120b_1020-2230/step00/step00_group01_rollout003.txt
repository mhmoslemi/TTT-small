def run(seed=42, budget_s=110, **kwargs):
    import numpy as np, time, math, random
    rng = np.random.default_rng(seed)
    n = 200  # or maybe choose 198 to avoid huge maybe 200.
    # make n even.
    # Generate initial h using irrational rotation
    alpha = (math.sqrt(5) - 1) / 2  # ~0.618.
    positions = np.arange(n)
    f = (positions * alpha) % 1.0
    # assign ones where f < 0.5
    h = (f < 0.5).astype(float)
    # adjust sum to exactly n/2
    current_sum = h.sum()
    target = n / 2
    if current_sum != target:
        # need to correct difference
        diff = int(round(target - current_sum))
        # if diff>0, convert some zeros to ones (choose those with f closest to 0.5 from below)
        # else convert some ones to zeros
        if diff > 0:
            # zeros with largest f (closest to 0.5)
            zero_indices = np.where(h == 0)[0]
            # sort by f descending to get ones closest to threshold
            sort_idx = zero_indices[np.argsort(-(f[zero_indices]))]  # descending
            h[sort_idx[:diff]] = 1.0
        elif diff < 0:
            one_indices = np.where(h == 1)[0]
            # ones with smallest f (farther from threshold)
            sort_idx = one_indices[np.argsort(f[one_indices])]  # ascending
            h[sort_idx[: -diff]] = 0.0
    # Ensure sum exactly n/2
    assert abs(h.sum() - target) < 1e-9

    # compute initial C5
    dx = 2.0 / n
    def compute_cmax(vec):
        # returns max count (int)
        return np.correlate(vec, 1 - vec, mode='full').max()
    curr_cmax = compute_cmax(h)
    best_cmax = curr_cmax
    best_h = h.copy()

    # simulated annealing
    start = time.time()
    # parameters
    max_iters = 2000000  # huge, but break earlier
    # initial temperature
    init_temp = 0.05  # might be tuned
    # cooling factor: keep temperature decaying linearly to 0 over max_iters
    # compute target iteration count based on budget
    # adapt after time.
    for it in range(max_iters):
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break
        # compute temperature (simple linear decay)
        temp = init_temp * (1 - it / max_iters)
        # choose random indices to swap
        # get indices arrays each time for speed
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:

        h[i_one] = 0.0
        h[i_zero] = 1.0

import numpy as np, time, math

def _cmax(vec: np.ndarray) -> int:
    """Maximum value of the correlation h * (1‑h) for all integer shifts."""
    # np.correlate returns float, but for binary vectors the result is integer.
    return int(np.correlate(vec, 1.0 - vec, mode="full").max())


def run(seed: int = 42, budget_s: float = 110, **kwargs):
    """
    Returns a step function (as a list of values) that attempts to minimise
    the Erdős overlap constant C₅.

    The returned tuple is (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose the discretisation size.
    # A size around 200 gives enough flexibility while staying fast.
    # ------------------------------------------------------------------
    n = 200  # must be even so that the integral constraint is integral
    dx = 2.0 / n
    target_sum = n / 2  # ∫h = 1  ⇒  sum(h) = n/2

    # ------------------------------------------------------------------
    # Initialise h with a low‑discrepancy (Sturmian) binary pattern.
    # ------------------------------------------------------------------
    alpha = (math.sqrt(5.0) - 1.0) / 2.0          # golden‑ratio conjugate
    f = (np.arange(n, dtype=float) * alpha) % 1.0
    h = (f < 0.5).astype(float)

    # Adjust to satisfy the exact sum constraint (difference ≤ 1).
    cur = h.sum()
    diff = int(round(target_sum - cur))
    if diff > 0:
        # turn some 0→1 (choose those with f closest to 0.5 from below)
        zeros = np.where(h == 0)[0]
        cand = zeros[np.argsort(-(f[zeros] - 0.5))]  # largest f (closest to 0.5)
        h[cand[:diff]] = 1.0
    elif diff < 0:
        # turn some 1→0 (choose those with f farthest from 0.5)
        ones = np.where(h == 1)[0]
        cand = ones[np.argsort(f[ones] - 0.5)]     # smallest f (far from 0.5)
        h[cand[: -diff]] = 0.0

    # Verify the constraint
    assert abs(h.sum() - target_sum) < 1e-12

    # ------------------------------------------------------------------
    # Simulated‑annealing optimiser.
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_cmax = _cmax(best_h)

    cur_h = h.copy()
    cur_cmax = best_cmax

    start = time.time()
    max_iters = 500000  # generous upper bound; loop stops earlier by time.
    init_temp = 0.05     # starting temperature

    for it in range(max_iters):
        # time check
        if time.time() - start > budget_s * 0.95:
            break

        # linear cooling schedule
        temp = init_temp * (1.0 - it / max_iters)

        # pick a random 1 and a random 0 to swap
        ones_idx = np.where(cur_h == 1)[0]
        zeros_idx = np.where(cur_h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break

        i_one = rng.choice(ones_idx)
        i_zero = rng.choice(zeros_idx)

        # perform the swap
        cur_h[i_one] = 0.0
        cur_h[i_zero] = 1.0

        new_cmax = _cmax(cur_h)
        delta = new_cmax - cur_cmax

        accept = False
        if delta <= 0:
            accept = True
        else:
            # probabilistic uphill move
            if temp > 0:
                prob = math.exp(-delta / temp)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_cmax = new_cmax
            if new_cmax < best_cmax:
                best_cmax = new_cmax
                best_h = cur_h.copy()
        else:
            # revert the swap
            cur_h[i_one] = 1.0
            cur_h[i_zero] = 0.0

    best_c5 = best_cmax * dx
    return (best_h.tolist(), float(best_c5), n)
