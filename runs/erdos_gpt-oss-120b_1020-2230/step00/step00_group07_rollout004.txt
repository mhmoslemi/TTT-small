import numpy as np
import time

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def project_onto_box_and_sum(y: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of y onto the box [0,1]^n intersected with the hyperplane sum = target.
    Uses a binary search for the Lagrange multiplier λ.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    lam = (lo + hi) * 0.5
    return np.clip(y + lam, 0.0, 1.0)


def run(seed: int = 42, budget_s: float = 110, **kwargs):
    """
    Optimises a step function h on [0,2] with values in [0,1] and integral 1.
    Returns the best found h, its C5 bound, and the number of samples.
    """
    rng = np.random.default_rng(seed)

    # --- initialise ---------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.shape[0]
    else:
        n = 80
        # a simple triangular start (area = 1)
        xs = np.linspace(0.0, 2.0, n, endpoint=False) + (1.0 / n)
        h = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))

    target_sum = n / 2.0
    h = project_onto_box_and_sum(h, target_sum)

    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c

    start_time = time.time()
    no_improve = 0
    max_no_improve = 5000

    # --- stochastic pairwise optimisation -----------------------------------
    while time.time() - start_time < budget_s * 0.95:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # choose direction of mass transfer
        direction = rng.random() < 0.5  # True: i -> j (i decreases)
        if direction:
            max_move = min(h[i], 1.0 - h[j])
        else:
            max_move = min(h[j], 1.0 - h[i])
        if max_move <= 0.0:
            continue

        # random step up to half of the feasible amount
        delta = max_move * rng.random() * 0.5

        # apply the move
        if direction:
            h[i] -= delta
            h[j] += delta
        else:
            h[j] -= delta
            h[i] += delta

        c = compute_c5(h)

        if c < cur_c - 1e-12:
            cur_c = c
            no_improve = 0
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            # revert the move
            if direction:
                h[i] += delta
                h[j] -= delta
            else:
                h[j] += delta
                h[i] -= delta
            no_improve += 1

        # if stuck, perturb globally around the best known configuration
        if no_improve >= max_no_improve:
            h = best_h + rng.normal(scale=0.02, size=n)
            h = np.clip(h, 0.0, 1.0)
            h = project_onto_box_and_sum(h, target_sum)
            cur_c = compute_c5(h)
            no_improve = 0

        # early exit if we already beat the target bound
        if best_c <= 0.3808:
            break

    return best_h, best_c, n
