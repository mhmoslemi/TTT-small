import numpy as np, time, random

def compute_c5(h):
    N = h.shape[0]
    weight = h.sum()
    conv = np.correlate(h, h, mode='full')
    min_overlap = conv.min()
    max_D = weight - min_overlap
    return max_D * (2.0 / N)

def project(v, target):
    lo = -np.max(v) - 1.0  # ensure low enough
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v + (lo+hi)/2, 0.0, 1.0)
    # might still be slightly off due to numeric; adjust residual
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff to entries not at bounds
        mask = (x > 0) & (x < 1)
        if mask.any():
            # distribute equally
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            # adjust arbitrarily; use rounding
            pass
    # Ensure sum equality
    return x
```
