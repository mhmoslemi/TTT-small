def _evaluate(h):
    """Wrapper for the provided evaluator, with a fallback."""
    if 'evaluate_erdos_solution' in globals():
        return evaluate_erdos_solution(h)
    # fallback: same formula used in the statement
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def _make_initial(n, seed):

    rng = np.random.default_rng(seed)

    half = n // 2
    h[:half] = 1.0
    if n % 2:

    rng.shuffle(h)
    return h
