def project_onto_simplex(y, target):
    # y: array, could be any numbers.
    # solve h = clip(y + λ, 0, 1), sum h = target.
    # We need λ such that sum(clipped) = target.
    # Use binary search on λ.
    lo, hi = -2, 2  # initial bounds may be large.
    for _ in range(50):
        lam = (lo+hi)/2
        h = np.clip(y + lam, 0, 1)
        s = h.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(y + (lo+hi)/2, 0, 1)

import time
import numpy as np

def _c5_of(h: np.ndarray) -> float:
    """Return C₅ for a step function h sampled on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1‑h) (zero‑padded outside the interval)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _project_onto_sum_box(y: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of y onto the set
        {z : 0 ≤ z_i ≤ 1,  Σ z_i = target } .
    The solution has the form z_i = clip(y_i + λ, 0, 1) for a suitable λ.
    """
    lo, hi = -2.0, 2.0                     # safe bounds for λ
    for _ in range(50):                    # bisection – enough for double precision
        lam = (lo + hi) * 0.5
        z = np.clip(y + lam, 0.0, 1.0)
        s = z.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)
