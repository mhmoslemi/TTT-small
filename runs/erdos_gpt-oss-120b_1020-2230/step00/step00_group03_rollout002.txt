import time, math
import numpy as np

def _c5_of_h(h):
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated on the current grid."""
    n = h.size
    dx = 2.0 / n
    # np.correlate implements the required linear correlation (full)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _make_binary_initial(h_raw):
    """Turn any initial vector into a binary 0/1 vector with sum = n/2."""
    h = np.rint(h_raw).astype(float)      # round to nearest integer 0/1
    n = h.size
    target = n // 2
    cur = int(h.sum())
    rng = np.random.default_rng()
    while cur != target:
        if cur > target:
            # flip a random 1 to 0
            idx = rng.choice(np.where(h == 1.0)[0], size=cur - target, replace=False)
            h[idx] = 0.0
            cur = target
        else:
            # flip a random 0 to 1
            idx = rng.choice(np.where(h == 0.0)[0], size=target - cur, replace=False)
            h[idx] = 1.0
            cur = target
    return h


def run(seed=42, budget_s=110, **kwargs):
    """
    Search for a step function h on [0,2] (sampled uniformly) that
    minimizes   C5 = max_k ∫ h(x) (1‑h(x+k)) dx .
    Returns (list_of_h, c5_value, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # Determine problem size and starting vector
    # ------------------------------------------------------------------
    try:
        # pre‑imported by the harness
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    except Exception:                     # no initial_h_values supplied
        n = 70                               # an even size that works well
        h0 = None

    if n % 2:
        n += 1                               # enforce even length
    # Build a binary starting point (keeps the integral constraint exact)
    if h0 is not None and h0.size == n:
        h = _make_binary_initial(h0)
    else:
        h = np.zeros(n, dtype=float)
        h[rng.choice(n, size=n // 2, replace=False)] = 1.0

    best_h = h.copy()
    best_c5 = _c5_of_h(best_h)

    # ------------------------------------------------------------------
    # Simulated‑annealing / hill‑climbing loop
    # ------------------------------------------------------------------
    # temperature schedule (starts relatively high to allow escapes)
    T = best_c5 * 0.3 if best_c5 > 0 else 0.01
    decay = 0.9996

    # Pre‑compute index lists for quick random swaps
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]

    # Main optimisation loop
    while time.time() - start_time < budget_s * 0.94:
        if len(ones) == 0 or len(zeros) == 0:
            break
        # propose swapping a 1 with a 0
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = h[j], h[i]                     # perform swap

        new_c5 = _c5_of_h(h)
        delta = new_c5 - best_c5

        # Metropolis acceptance
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0 and rng.random() < math.exp(-delta / T):
                accept = True

        if accept:
            # keep the change, update indices
            best_c5 = new_c5
            best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]

        # temperature decay
        T *= decay

        # occasional random restart to avoid stagnation
        if rng.random() < 0.001:
            # keep the best found so far, re‑initialise a fresh candidate
            h = np.zeros(n, dtype=float)
            h[rng.choice(n, size=n // 2, replace=False)] = 1.0
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            T = best_c5 * 0.3 if best_c5 > 0 else 0.01

    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    return (best_h.tolist(), float(best_c5), n)
