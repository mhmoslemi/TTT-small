def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    # Note: compute correlation of h and 1-h
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')

import time, numpy as np, math

def compute_c5(h):
    N=len(h)
    dx = 2.0/N
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    return float((corr * dx).max())

def hill_climb(h, rng, max_time):
    # ensure binary
    h = (h > 0.5).astype(np.int8)
    N = len(h)
    best_h = h.copy()
    best_val = compute_c5(h)
    cur_h = h.copy()
    cur_val = best_val
    # track indices
    ones = np.where(cur_h == 1)[0]
    zeros = np.where(cur_h == 0)[0]
    start = time.time()
    it_without_improve = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= max_time:
            break
        # avoid infinite loop if ones or zeros empty
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        cur_h[i] = 0
        cur_h[j] = 1
        new_val = compute_c5(cur_h)
        # acceptance
        if new_val < cur_val:
            accept = True
        else:
            # temperature schedule based on remaining time fraction
            remaining = max_time - elapsed
            frac = remaining / max_time
            T = 0.004 * frac  # adjust constant
            accept = False
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                accept = rng.random() < prob
        if accept:
            cur_val = new_val
            # update indices after swap
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
                it_without_improve = 0
            else:
                it_without_improve += 1
        else:
            # revert
            cur_h[i] = 1
            cur_h[j] = 0
            it_without_improve += 1

        if it_without_improve >= 5000:
            # random restart; choose new random binary arrangement
            perm = rng.permutation(N)
            cur_h[:] = 0
            cur_h[perm[:N//2]] = 1
            cur_val = compute_c5(cur_h)
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            it_without_improve = 0
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
    return best_h.astype(float), best_val

import time, math, numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation with zero padding (the same as the evaluator uses)
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode="full")
    return float((corr * dx).max())


def _hill_climb(start_h: np.ndarray, rng: np.random.Generator, max_seconds: float):
    """Simple simulated‑annealing hill‑climb on binary step functions."""
    # enforce binary values (0/1) – they give the best possible product term
    h = (start_h > 0.5).astype(np.int8)
    n = h.size
    best_h = h.copy()
    best_val = _c5_value(h)

    cur_h = h.copy()
    cur_val = best_val
    # indices of ones and zeros – updated lazily
    ones = np.where(cur_h == 1)[0]
    zeros = np.where(cur_h == 0)[0]

    start = time.time()
    no_improve = 0
    # temperature schedule parameters
    T0 = 0.025  # starting temperature
    while True:
        elapsed = time.time() - start
        if elapsed >= max_seconds:
            break
        if ones.size == 0 or zeros.size == 0:
            break

        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform a swap
        cur_h[i] = 0
        cur_h[j] = 1

        new_val = _c5_value(cur_h)

        # accept if better, otherwise with a temperature‑dependent probability
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # linear cooling based on remaining time
            remain = max_seconds - elapsed
            frac = remain / max_seconds
            T = T0 * frac
            if T > 0:
                prob = math.exp(-(new_val - cur_val) / T)
                accept = rng.random() < prob

        if accept:
            cur_val = new_val
            # update index lists after the successful swap
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert the swap
            cur_h[i] = 1
            cur_h[j] = 0
            no_improve += 1

        # if stuck for a while, restart from a fresh random binary vector
        if no_improve >= 4000:
            perm = rng.permutation(n)
            cur_h[:] = 0

            zeros = np.where(cur_h == 0)[0]
            no_improve = 0
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()

    return best_h.astype(float), best_val
