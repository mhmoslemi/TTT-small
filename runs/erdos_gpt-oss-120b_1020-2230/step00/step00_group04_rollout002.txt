import numpy as np, time, math
def compute_c5(h):
    n = h.shape[0]; dx = 2.0 / n
    L = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(h, L)
    fb = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(fa * np.conj(fb), L)[:2*n-1]
    return np.max(corr.real) * dx
```

import time, math
import numpy as np

def _compute_c5(h, dx):
    """Return max_{k} ∫ h(x)(1-h(x+k)) dx using discrete correlation."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def _random_feasible(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    target = n / 2.0
    # start with uniform draws and then project onto the simplex with box constraints
    x = rng.random(n)
    # scale to target sum
    total = x.sum()
    x = x * (target / total)
    # clamp and fix any violations by a simple water‑filling loop
    for _ in range(5):
        over = x > 1.0
        if not over.any():
            break
        excess = x[over] - 1.0
        x[over] = 1.0
        # distribute excess uniformly to the rest
        under = ~over
        if under.any():
            x[under] -= excess.mean() * over.sum() / under.sum()
    # final safety clip
    np.clip(x, 0.0, 1.0, out=x)
    # tiny adjustment to hit the exact sum (float rounding)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny difference uniformly without breaking bounds
        inc = diff / n
        x += inc
        np.clip(x, 0.0, 1.0, out=x)
        # renormalise if needed
        diff2 = target - x.sum()
        if abs(diff2) > 1e-12:
            # fallback to a deterministic construction: half 1’s half 0’s
            x = np.zeros(n)
            idx = rng.choice(n, size=n//2, replace=False)
            x[idx] = 1.0
    return x

def run(seed=42, budget_s=110, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    # initialise h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        n = 200
        h = _random_feasible(n, rng)
    n = h.shape[0]
    dx = 2.0 / n

    # initial evaluation
    best_h = h.copy()
    best_val, _ = _compute_c5(best_h, dx)
    best_c5 = best_val
    cur_h = h.copy()
    cur_c5 = best_c5

    # annealing parameters
    max_temp = 0.02
    max_step = 0.06

    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.97:
            break
        frac = elapsed / budget_s
        T = max_temp * (1.0 - frac) + 1e-9
        step = max_step * (1.0 - frac) + 1e-9

        # decide move type
        if rng.random() < 0.15:
            # targeted transfer based on current worst shift
            _, corr = _compute_c5(cur_h, dx)
            idx_max = int(np.argmax(corr))
            offset = idx_max - (n - 1)  # shift index
            if offset >= 0:
                i = rng.integers(0, n - offset)
                j = i + offset
            else:
                i = rng.integers(-offset, n)
                j = i + offset
            # direction that tends to equalise the pair
            if cur_h[i] > cur_h[j]:
                # move mass from i to j
                max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
                if max_delta > 0:
                    delta = rng.random() * max_delta
                    cand = cur_h.copy()
                    cand[i] -= delta
                    cand[j] += delta
                else:
                    continue
            else:
                max_delta = min(step, cur_h[j], 1.0 - cur_h[i])
                if max_delta > 0:
                    delta = rng.random() * max_delta
                    cand = cur_h.copy()
                    cand[i] += delta
                    cand[j] -= delta
                else:
                    continue
        elif rng.random() < 0.3:
            # simple swap
            i, j = rng.integers(n, size=2)
            while i == j:
                j = rng.integers(n)
            cand = cur_h.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # transfer between two random indices
            i, j = rng.integers(n, size=2)
            while i == j:
                j = rng.integers(n)
            # allowable delta range
            max_inc = min(step, 1.0 - cur_h[i])
            max_dec = min(step, cur_h[j])
            low = -max_dec
            high = max_inc
            if low >= high:

            delta = low + rng.random() * (high - low)
            cand = cur_h.copy()
            cand[i] += delta
            cand[j] -= delta

    # final C5 recomputation
    final_c5, _ = _compute_c5(best_h, dx)

    return (best_h, final_c5, n)
