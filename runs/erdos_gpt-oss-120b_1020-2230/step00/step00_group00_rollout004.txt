import numpy as np
import time
import math

def _compute_c5(vec: np.ndarray) -> float:
    """Maximum overlap integral for a given step function."""
    n = vec.shape[0]
    dx = 2.0 / n
    # convolution h * (1-h) (no reversal, matches the problem definition)
    c = np.correlate(vec, 1.0 - vec, mode="full")
    return np.max(c) * dx


def _gradient_shift(vec: np.ndarray, shift: int) -> np.ndarray:
    """
    Gradient of the overlap with respect to ``vec`` for a fixed shift.
    For shift = k, S_k = sum_i h_i * (1 - h_{i+k}).
    dS_k/dh_i = (1 - h_{i+k}) - h_{i-k} (indices out of range are ignored).
    """
    n = vec.shape[0]
    grad = np.zeros_like(vec)
    if shift >= 0:
        end = n - shift
        if end > 0:
            grad[:end] += 1.0 - vec[shift:]
            grad[shift:] -= vec[:end]
    else:
        k = -shift
        end = n - k
        if end > 0:
            grad[k:] += 1.0 - vec[:end]
            grad[:end] -= vec[k:]
    return grad
