def project_box_sum(x, target):
   lo = -1.0
   hi = 1.0
   for _ in range(30):
       mid = (lo+hi)/2.0
       y = np.clip(x - mid, 0.0, 1.0)
       s = y.sum()
       if s > target:
           lo = mid
       else:
           hi = mid
   return np.clip(x - (lo+hi)/2.0, 0.0, 1.0)

import time
import numpy as np

def _project_box_sum(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector x onto the intersection of the box [0,1]^n
    and the affine hyperplane sum(x) = target_sum.
    Uses bisection on the Lagrange multiplier.
    """
    lo, hi = -2.0, 2.0  # wide enough brackets: at lo all entries become 1, at hi all become 0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_of(h: np.ndarray) -> float:
    """Return the C₅ value for a step function h sampled on n points."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / n)


def _smooth_gradient(h: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed objective
        f_beta(h) = (1/beta) * log( sum_k exp(beta * c_k) )
    where c_k = sum_i h_i (1 - h_{i+k}) (non‑scaled).
    """
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    maxc = conv.max()
    # stable soft‑max weights
    w = np.exp(beta * (conv - maxc))
    w_sum = w.sum()
    probs = w / w_sum

    grad = np.ones(n)                 # base term (sum of probs = 1)
    # contributions from the two shifted copies
    for idx, p in enumerate(probs):
        if p == 0.0:
            continue
        s = idx - (n - 1)            # shift value, can be negative
        if s > 0:
            grad[:n - s] -= p * h[s:]
            grad[s:]      -= p * h[:n - s]
        elif s < 0:
            s = -s
            grad[s:]      -= p * h[:n - s]
            grad[:n - s] -= p * h[s:]
        else:                        # s == 0
            grad -= 2.0 * p * h
    return grad


def run(seed: int = 42, budget_s: float = 110, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) where
      h_values : np.ndarray of shape (n_points,)
      c5_bound : float   (the achieved C₅ value)
      n_points : int     (number of discretisation points)
    """
    np.random.seed(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)

    # ----------------------------------------------------------------------
    # Choose the discretisation size.
    # ----------------------------------------------------------------------
    try:
        init0 = initial_h_values          # provided by the harness (if any)
        n0 = int(init0.shape[0])
        n = max(200, 2 * n0)              # at least 200 points, prefer a finer grid
        # interpolate the given construction onto the new grid
        xs0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        xs  = np.linspace(0.0, 2.0, n,  endpoint=False)
        h_interp = np.interp(xs, xs0, init0)
        init_candidates = [h_interp]
    except NameError:
        n = 200
        init_candidates = []

    target_sum = n / 2.0                     # because ∫ h = 1  ⇔ sum(h) = n/2
    # add a few random starts
    for _ in range(4):
        rnd = np.random.rand(n)
        rnd = _project_box_sum(rnd, target_sum)
        init_candidates.append(rnd)

    best_h = None
    best_val = np.inf

    # ----------------------------------------------------------------------
    # Main optimisation loop (smooth‑gradient descent with annealing)
    # ----------------------------------------------------------------------
    for start_h in init_candidates:
        if time_left() < 1.0:
            break
        h = _project_box_sum(start_h.copy(), target_sum)

        # parameters for the annealing schedule
        beta = 0.5                # starts with a very smooth approximation
        beta_max = 12.0
        beta_inc = 0.005

        step0 = 1.0               # initial step size (will decay as 1/sqrt(t))
        iter_cnt = 0
        while time_left() > 0.5:
            # evaluate current C5
            cur_val = _c5_of(h)
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                # early exit if we already beat the target record
                if best_val <= 0.3808:
                    break

            # compute smooth gradient
            grad = _smooth_gradient(h, beta)

            # diminishing step size
            alpha = step0 / np.sqrt(iter_cnt + 1.0)

            # small stochastic perturbation to escape flat regions
            noise_scale = max(0.0, 0.2 - 0.2 * (iter_cnt / 5000.0))
            noise = noise_scale * np.random.randn(n)

            # gradient step + noise, then projection onto feasible set
            h = _project_box_sum(h - alpha * grad + noise, target_sum)

            # update schedules
            beta = min(beta_max, beta + beta_inc)
            iter_cnt += 1

        if best_val <= 0.3808:
            break

    # ----------------------------------------------------------------------
    # Final sanity check and return
    # ----------------------------------------------------------------------
    if best_h is None:                     # fallback (should never happen)
        best_h = np.full(n, 0.5)
    final_c5 = _c5_of(best_h)
    return best_h, float(final_c5), n
