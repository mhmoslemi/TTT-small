def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = corr.max()
    return max_corr * dx

def _compute_c5(h):
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation using the full linear overlap
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

        x[free] += delta
