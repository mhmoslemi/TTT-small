def project_to_box_and_sum(v, target, low=0.0, high=1.0):
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        proj = np.clip(v + mid, low, high)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid

import numpy as np, time, math

def project_to_sum_and_box(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # correct minor sum deviation
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 0) & (proj < 1)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def bound_h(h):
    n = h.shape[0]
    dx = 2.0 / n
    auto = np.correlate(h, h, mode='full')
    min_auto = auto.min()
    return 1.0 - min_auto * dx

def improve_worst_shift(h, rng):
    n = h.shape[0]
    auto = np.correlate(h, h, mode='full')
    min_idx = np.argmin(auto)
    k = min_idx - (n - 1)
    # ignore zero shift? not necessary
    valid = np.where((np.arange(n) + k >= 0) & (np.arange(n) + k < n))[0]
    if valid.size < 1:
        return None
    prod = h[valid] * h[valid + k]
    i0 = valid[np.argmin(prod)]
    j0 = i0 + k
    inc_cap_i = 1.0 - h[i0]
    inc_cap_j = 1.0 - h[j0]
    # find decreasing candidates
    dec_candidates = np.where(h > 1e-12)[0]
    dec_candidates = dec_candidates[(dec_candidates != i0) & (dec_candidates != j0)]
    if dec_candidates.size < 2:
        return None
    dec = rng.choice(dec_candidates, size=2, replace=False)
    dec_cap0 = h[dec[0]]
    dec_cap1 = h[dec[1]]
    delta_max = min(inc_cap_i, inc_cap_j, dec_cap0, dec_cap1)
    if delta_max <= 1e-12:
        return None
    delta = delta_max * 0.6  # moderate factor
    new_h = h.copy()
    new_h[i0] += delta
    new_h[j0] += delta
    new_h[dec[0]] -= delta
    new_h[dec[1]] -= delta
    return new_h

def propose_random_move(h, rng):
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]
    lo = max(-hi, hj - 1.0)
    hi_ = min(1.0 - hi, hj)
    if lo >= hi_:
        return None
    delta = rng.uniform(lo, hi_)
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    return new_h

    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

            if new_h is None:
                continue
        new_val = bound_h(new_h)
        # temperature schedule
        temp = T0 * (1 - it / max_temp_iters)

import numpy as np
import math
import time

def _project_to_sum_and_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1]^n while enforcing sum(v)=target."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 0) & (proj < 1)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _bound_h(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1−h(x+k))dx for the discretised h."""
    n = h.shape[0]
    dx = 2.0 / n
    auto = np.correlate(h, h, mode='full')
    return 1.0 - auto.min() * dx

def _improve_worst_shift(h: np.ndarray, rng: np.random.Generator) -> np.ndarray | None:
    """Increase the product on the shift giving the current worst (minimum) autocorrelation."""
    n = h.shape[0]
    auto = np.correlate(h, h, mode='full')
    min_idx = int(np.argmin(auto))
    k = min_idx - (n - 1)          # shift with minimal autocorrelation
    idx = np.arange(n)
    mask = (idx + k >= 0) & (idx + k < n)
    valid = idx[mask]
    if valid.size == 0:
        return None
    prod = h[valid] * h[valid + k]
    i0 = valid[int(np.argmin(prod))]
    j0 = i0 + k
    inc_i = 1.0 - h[i0]
    inc_j = 1.0 - h[j0]
    dec_candidates = np.where(h > 1e-12)[0]
    dec_candidates = dec_candidates[(dec_candidates != i0) & (dec_candidates != j0)]
    if dec_candidates.size < 2:
        return None
    dec = rng.choice(dec_candidates, size=2, replace=False)
    cap_dec0 = h[dec[0]]
    cap_dec1 = h[dec[1]]
    delta_max = min(inc_i, inc_j, cap_dec0, cap_dec1)
    if delta_max <= 1e-12:
        return None
    delta = delta_max * 0.6
    new_h = h.copy()
    new_h[i0] += delta
    new_h[j0] += delta
    new_h[dec[0]] -= delta
    new_h[dec[1]] -= delta
    return new_h

def _propose_random_move(h: np.ndarray, rng: np.random.Generator) -> np.ndarray | None:
    """Swap a small amount of mass between two positions while staying in [0,1]."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]
    lo = max(-hi, hj - 1.0)
    hi_ = min(1.0 - hi, hj)
    if lo >= hi_:
        return None
    delta = rng.uniform(lo, hi_)
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    return new_h

def _compute_c5(h: np.ndarray) -> float:
    """Exact C5 value using the definition with 1‑h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return float(corr.max())

def run(seed: int = 42, budget_s: float = 110, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise
    # ------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        n = 95
        xs = (np.arange(n) + 0.5) * (2.0 / n)
        init = (1.0 + np.sin(np.pi * xs)) / 2.0
    else:
        n = init.shape[0]

    target_sum = n / 2.0
    h = _project_to_sum_and_box(init, target_sum)

    cur_h = h.copy()
    cur_val = _bound_h(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    # ------------------------------------------------------------------
    # search parameters
    # ------------------------------------------------------------------
    T0 = 0.01                     # initial temperature
    max_temp_iters = 300000       # after this temperature is ~0
    it = 0

    # ------------------------------------------------------------------
    # main loop
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.96:
        # try a directed improvement
        new_h = _improve_worst_shift(cur_h, rng)
        if new_h is None:
            new_h = _propose_random_move(cur_h, rng)
            if new_h is None:
                continue

        new_val = _bound_h(new_h)

        # simulated‑annealing style acceptance
        temp = T0 * (1.0 - it / max_temp_iters)
        if temp < 1e-6:
            temp = 1e-6
        accept = False
        if new_val < cur_val:
            accept = True
        elif rng.random() < math.exp((cur_val - new_val) / temp):
            accept = True

        if accept:
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
        it += 1

    # ------------------------------------------------------------------
    # final projection & evaluation
    # ------------------------------------------------------------------
    best_h = _project_to_sum_and_box(best_h, target_sum)
    c5 = _compute_c5(best_h)
    return best_h.tolist(), c5, n
