import numpy as np
import time
import random
import math

def _evaluate(h: np.ndarray) -> float:
    """C5 value for a given step function."""
    n = h.shape[0]
    # correlation of h with (1‑h) using zero padding
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the box [0,1]^n intersect
    {x : sum(x) = target}.
    """
    lo = -np.max(v) - 5.0
    hi = np.max(v) + 5.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        if proj.sum() > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _gradient_step(h: np.ndarray, step: float, target: float) -> np.ndarray:
    """
    One sub‑gradient descent step for minimizing max_k ∫h(x)(1‑h(x+k)).
    The active shift is the one giving the maximal overlap (i.e. the
    minimal self‑correlation).
    """
    n = h.shape[0]
    # self‑correlation (zero‑padded) – shape 2n‑1
    inner = np.correlate(h, h, mode="full")
    idx_min = int(np.argmin(inner))               # shift with smallest inner product
    shift = idx_min - (n - 1)                     # signed shift

    grad = np.ones_like(h)                       # ∂/∂h_i of the active term
    if shift > 0:
        grad[: n - shift] -= h[shift:]
        grad[shift:]    -= h[: n - shift]
    elif shift < 0:
        s = -shift
        grad[s:]        -= h[: n - s]
        grad[: n - s]   -= h[s:]
    else:
        grad -= 2.0 * h

    v = h - step * grad
    return _project_to_sum(v, target)

            if i == j:
                continue
            hi, hj = cur_h[i], cur_h[j]

            # feasible delta that keeps both entries inside [0,1]
            lower = max(-hi, hj - 1.0, -0.04)
            upper = min(1.0 - hi, hj, 0.04)
            if lower >= upper:
                continue

            delta = np.random.uniform(lower, upper)
            new_h = cur_h.copy()
            new_h[i] = hi + delta
            new_h[j] = hj - delta
            val = _evaluate(new_h)
            if val < cur_val:
                cur_h = new_h
                cur_val = val
                if val < best_val:
                    best_val = val
                    best_h = new_h.copy()

    # ------------------------------------------------------------------
    return (best_h.tolist(), float(best_val), n)
```
