        # locate index i that contributes most to this shift
        if shift >= 0:
            L = n - shift
            i_range = np.arange(L)
            prod = cur_h[i_range] * (1.0 - cur_h[i_range + shift])
        else:
            k = -shift
            L = n - k
            i_range = np.arange(L)
            prod = cur_h[i_range + k] * (1.0 - cur_h[i_range])
        i_local = int(i_range[int(np.argmax(prod))])
        j_local = i_local + shift               # partner index (may be <0, handled by numpy)

        # propose a directed transfer i_local -> j_local
        feasible = min(cur_h[i_local], 1.0 - cur_h[j_local])
        if feasible > 1e-12:
            delta = _target_delta(feasible, feasible)
            new_h = cur_h.copy()
            new_h[i_local] -= delta
            new_h[j_local] += delta
            new_val = _c5_value(new_h)
            if new_val + 1e-12 < cur_val:       # accept any improvement
                cur_h, cur_val = new_h, new_val
                if new_val + 1e-12 < best_val:
                    best_h, best_val = new_h.copy(), new_val
                stagn = 0
                continue
        # if no improvement, try a random transfer
        i, j = random.sample(range(n), 2)
        feasible = min(cur_h[i], 1.0 - cur_h[j])
        if feasible > 1e-12:
            delta = _target_delta(feasible, feasible)
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            new_val = _c5_value(new_h)
            if new_val + 1e-12 < cur_val:
                cur_h, cur_val = new_h, new_val
                if new_val + 1e-12 < best_val:
                    best_h, best_val = new_h.copy(), new_val
                stagn = 0
                continue

        stagn += 1
        # occasional global jitter to escape plateaus
        if stagn > 3000:
            # small random perturbation preserving the total mass
            noise = np.random.uniform(-0.02, 0.02, size=n)
            jitter = cur_h + noise
            jitter = np.clip(jitter, 0.0, 1.0)
            # fix the sum exactly
            diff = (n / 2.0) - jitter.sum()
            jitter += diff / n
            jitter = np.clip(jitter, 0.0, 1.0)
            new_val = _c5_value(jitter)
            if new_val + 1e-12 < cur_val:
                cur_h, cur_val = jitter, new_val
                if new_val + 1e-12 < best_val:
                    best_h, best_val = jitter.copy(), new_val
                stagn = 0
            else:
                stagn = 0   # reset anyway to keep trying

    return best_h, best_val

def run(seed=42, budget_s=110, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points)

    - h_values : list of step‑function values on a uniform grid of [0,2]
    - c5_bound : the achieved upper bound C5 = max_k ∫ h(x)(1−h(x+k))dx
    - n_points : number of samples used (len(h_values))
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # candidate problem sizes
    # ------------------------------------------------------------------
    candidates = []

    # use the provided construction if it exists
    if 'initial_h_values' in globals():
        init_len = int(len(initial_h_values))
        candidates.append(init_len)

    # add a few alternative sizes (small enough to stay fast)
    for sz in (60, 80, 100, 120, 150):
        candidates.append(sz)

    # remove duplicates and keep ordering stable
    candidates = list(dict.fromkeys(candidates))

    best_overall_h = None
    best_overall_val = float('inf')
    best_n = None

    start_total = time.time()
    remaining_candidates = len(candidates)

    for n in candidates:
        # allocate a slice of the remaining time
        elapsed = time.time() - start_total
        time_left = budget_s - elapsed
        if time_left <= 0:
            break
        # simple proportional split (but never give more than 30 s to one size)
        per = time_left / remaining_candidates
        time_budget = min(per * 0.9, 30.0)

        # ------------------------------------------------------------------
        # initialise a feasible vector
        # ------------------------------------------------------------------
        if n == init_len and 'initial_h_values' in globals():
            h0 = np.array(initial_h_values, dtype=float)
        else:
            h0 = _initial_random(n)

        # ------------------------------------------------------------------
        # optimise
        # ------------------------------------------------------------------
        h_opt, val_opt = _improve_step(h0, time_budget)

        if val_opt < best_overall_val:
            best_overall_h = h_opt
            best_overall_val = val_opt
            best_n = n

        remaining_candidates -= 1

    # safety: if nothing was produced (should not happen), fall back to uniform 0.5
    if best_overall_h is None:
        n = 45
        best_overall_h = np.full(n, 0.5)
        best_overall_val = _c5_value(best_overall_h)
        best_n = n

    # convert to plain Python list for the harness
    return best_overall_h.tolist(), float(best_overall_val), int(best_n)
