    best_c = max_overlap(best_h)

    # ------------------------------------------------------------------
    # 2. SLSQP refinement (if time allows)
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        # objective: minimise the bound t
        def objective(x):
            return x[-1]

        # constraints
        cons = []

        # equality: total mass
        cons.append({
            'type': 'eq',
            'fun': lambda x, S=target_sum: np.sum(x[:n]) - S
        })

        # inequality: for every shift k,   t - ∫h(1‑h_shift) ≥ 0
        for k in range(-n + 1, n):
            def con_fun(x, k=k):
                h = x[:n]
                t = x[-1]
                if k >= 0:
                    overlap = np.dot(h[:n - k], 1.0 - h[k:])
                else:
                    kk = -k
                    overlap = np.dot(h[kk:], 1.0 - h[:n - kk])
                return t - overlap * dx
            cons.append({'type': 'ineq', 'fun': con_fun})

        # bounds for hi and for t
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

        x0 = np.concatenate([best_h, [best_c]])

        opts = {
            'maxiter': 600,
            'ftol': 1e-9,
            'disp': False
        }

        res = minimize(objective, x0, method='SLSQP',
                       bounds=bounds, constraints=cons, options=opts)

        if res.success:
            h_opt = np.clip(res.x[:n], 0.0, 1.0)
            # force the exact sum (tiny numeric drift)
            s = h_opt.sum()
            diff = target_sum - s
            if abs(diff) > 1e-10:
                for i in range(n):
                    if (diff > 0 and h_opt[i] < 1) or (diff < 0 and h_opt[i] > 0):
                        move = min(abs(diff),
                                   (1 - h_opt[i]) if diff > 0 else h_opt[i])
                        h_opt[i] += math.copysign(move, diff)
                        diff -= math.copysign(move, diff)
                        if abs(diff) < 1e-12:
                            break
            c_opt = max_overlap(h_opt)
            if c_opt < best_c:
                best_h = h_opt
                best_c = c_opt

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing hill climbing for remaining time
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 0.5:
        curr_h = best_h.copy()
        curr_c = best_c

        # temperature schedule (linear decay)
        T0, T1 = 0.001, 1e-7

        while time.time() - start < budget_s - 0.1:
            elapsed = time.time() - start
            frac = elapsed / budget_s
            T = T0 * (1 - frac) + T1 * frac

            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            hi, hj = curr_h[i], curr_h[j]

            # possible transfer from i to j
            max_ij = min(hi, 1.0 - hj)
            max_ji = min(hj, 1.0 - hi)

            if np.random.rand() < 0.5:
                if max_ij <= 0:
                    continue
                delta = np.random.rand() * max_ij
                new_h = curr_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
            else:
                if max_ji <= 0:
                    continue
                delta = np.random.rand() * max_ji
                new_h = curr_h.copy()
                new_h[i] += delta
                new_h[j] -= delta

            new_c = max_overlap(new_h)
            delta_c = new_c - curr_c

            if delta_c < -1e-12:          # strict improvement
                curr_h, curr_c = new_h, new_c
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
            else:
                # uphill move with Metropolis probability
                if np.random.rand() < math.exp(-delta_c / T):
                    curr_h, curr_c = new_h, new_c

    # ------------------------------------------------------------------
    # 4. Final exactness fix (sum = n/2, bounds)
    # ------------------------------------------------------------------
    s = best_h.sum()
    diff = target_sum - s
    if abs(diff) > 1e-12:
        for i in range(n):
            if (diff > 0 and best_h[i] < 1) or (diff < 0 and best_h[i] > 0):
                move = min(abs(diff),
                           (1 - best_h[i]) if diff > 0 else best_h[i])
                best_h[i] += math.copysign(move, diff)
                diff -= math.copysign(move, diff)
                if abs(diff) < 1e-12:
                    break
    best_h = np.clip(best_h, 0.0, 1.0)

    final_c = max_overlap(best_h)

    return best_h, final_c, n
