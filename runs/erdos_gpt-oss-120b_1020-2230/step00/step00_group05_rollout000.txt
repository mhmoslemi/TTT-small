            break
        # compute worst shift

import time
import math
import numpy as np

# ------------------------------------------------------------
# Core utilities
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Return C5 for a step function given by the samples h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * dx


def worst_shift_and_val(h: np.ndarray):
    """Return the shift (as integer index offset) that attains the max overlap,
    together with the raw convolution value and the whole array."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(conv.argmax())
    shift = idx - (n - 1)          # shift ∈ [-(n-1), n-1]
    return shift, conv[idx], conv


def weights_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """For a fixed shift, compute w_i = h_{i+shift}+h_{i-shift}
    (zero outside the interval)."""
    n = len(h)
    w = np.zeros(n, dtype=float)
    if shift >= 0:
        k = shift
        if k < n:
            w[: n - k] += h[k:]          # forward part
            w[k:] += h[: n - k]          # backward part
    else:
        k = -shift
        if k < n:
            w[k:] += h[: n - k]
            w[: n - k] += h[k:]
    return w


def propose_move(h: np.ndarray, shift: int, step: float, rng: np.random.Generator):
    """Move a small amount of mass from a low‑weight position to a high‑weight
    one (with respect to the current worst shift)."""
    n = len(h)
    w = weights_for_shift(h, shift)

    not_full = h < 1.0 - 1e-12
    not_empty = h > 1e-12
    if not np.any(not_full) or not np.any(not_empty):
        return None

    # pick high‑weight candidate (increase)
    w_high = np.where(not_full, w, -np.inf)
    hi = int(np.argmax(w_high))

    # pick low‑weight candidate (decrease)
    w_low = np.where(not_empty, w, np.inf)
    lo = int(np.argmin(w_low))

    if hi == lo:
        # fallback to second best alternatives
        order_high = np.argsort(-w)
        for idx in order_high:
            if not_full[idx] and idx != lo:
                hi = int(idx)
                break
        order_low = np.argsort(w)
        for idx in order_low:
            if not_empty[idx] and idx != hi:
                lo = int(idx)
                break
        if hi == lo:
            return None

    delta = min(step, 1.0 - h[hi], h[lo])
    if delta <= 0.0:
        return None

    new_h = h.copy()
    new_h[hi] += delta
    new_h[lo] -= delta
    return new_h

# Search routine
# ------------------------------------------------------------
def local_search(start_h: np.ndarray, time_budget: float, rng: np.random.Generator):
    """Projected sub‑gradient / simulated‑annealing style search."""
    start = time.time()
    best_h = start_h.copy()
    best_val = compute_c5(best_h)

    cur_h = start_h.copy()
    cur_val = best_val
    last_improve = start

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break

        # worst shift for the current configuration
        shift, _, _ = worst_shift_and_val(cur_h)

        # step size linearly shrinks
        frac = elapsed / time_budget
        step = max(0.001, 0.05 * (1 - frac))

        new_h = propose_move(cur_h, shift, step, rng)
        if new_h is None:
            # tiny random transfer as a fallback
            i, j = rng.integers(len(cur_h), size=2)
            if i != j:
                max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
                if max_delta > 0.0:
                    new_h = cur_h.copy()
                    new_h[i] -= max_delta
                    new_h[j] += max_delta

        if new_h is None:
            continue

        new_val = compute_c5(new_h)

        if new_val < cur_val - 1e-12:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val - 1e-12:
                best_h, best_val = new_h.copy(), new_val
                last_improve = time.time()
        else:
            # occasional uphill step (tiny temperature)
            T = max(0.0005 * (1 - frac), 1e-6)
            if rng.random() < math.exp(-(new_val - cur_val) / T):
                cur_h, cur_val = new_h, new_val

        # if stuck for a while, re‑seed from current best with a tiny jiggle
        if time.time() - last_improve > time_budget * 0.15:
            cur_h = best_h.copy()
            for _ in range(3):
                i, j = rng.integers(len(cur_h), size=2)
                if i == j:
                    continue
                max_delta = min(0.02, cur_h[i], 1.0 - cur_h[j])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta
                    cur_h[i] -= delta
                    cur_h[j] += delta
            cur_val = compute_c5(cur_h)
            last_improve = time.time()

    return best_h, best_val
