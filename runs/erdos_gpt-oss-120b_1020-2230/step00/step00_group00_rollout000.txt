def local_search(v, max_iters, start_time, budget):
    cur_val = evaluate(v)
    best_v = v.copy()
    best_val = cur_val
    N = len(v)
    while max_iters > 0 and time.time() - start_time < budget * 0.95:
        # choose move
        i_inc = rng.integers(N)
        if v[i_inc] >= 1.0:
            continue
        i_dec = rng.integers(N)
        if i_dec == i_inc or v[i_dec] <= 0.0:
            continue
        max_inc = min(1.0 - v[i_inc], v[i_dec])
        if max_inc <= 0:
            continue
        # step scale
        progress = (time.time() - start_time) / (budget*0.95)
        step_factor = max(0.01, 1.0 - progress)  # gradually shrink
        delta = max_inc * rng.random() * step_factor
        if delta < 1e-12:
            continue
        v_new = v.copy()
        v_new[i_inc] += delta
        v_new[i_dec] -= delta
        new_val = evaluate(v_new)
        # acceptance
        if new_val < cur_val or rng.random() < math.exp((cur_val - new_val) / (0.001 + 0.01 * (1 - progress))):
            v = v_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_v = v_new.copy()
        max_iters -= 1
    return best_v, best_val
```

import time, math
import numpy as np

def _evaluate(v: np.ndarray) -> float:
    """Return C5 value for vector v."""
    N = v.shape[0]
    if N == 0:
        return float("inf")
    dx = 2.0 / N
    # correlation of v with (1-v)
    c = np.correlate(v, 1.0 - v, mode="full")
    return float(c.max() * dx)


def _project_feasible(v: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust to exact sum N/2."""
    N = len(v)
    target = N / 2.0
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    # simple water‑filling iteration
    for _ in range(10):
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(v < 1.0)[0]
            if idx.size == 0:
                break
            inc = diff / idx.size
            add = np.minimum(inc, 1.0 - v[idx])
            v[idx] += add
        else:
            idx = np.where(v > 0.0)[0]
            if idx.size == 0:
                break
            dec = (-diff) / idx.size
            sub = np.minimum(dec, v[idx])
            v[idx] -= sub
        s = v.sum()
    # final tiny correction
    v += (target - v.sum()) / N
    return np.clip(v, 0.0, 1.0)


def _thue_morse(N: int) -> np.ndarray:
    """Return Thue‑Morse binary sequence of length N."""
    bits = np.fromiter((bin(i).count("1") % 2 for i in range(N)), dtype=np.float64)
    return bits


def _local_search(v0: np.ndarray, rng: np.random.Generator,
                  time_limit: float, start_time: float):
    """Simple projected SA that respects box and sum constraints."""
    v = v0.copy()
    N = v.shape[0]
    cur_val = _evaluate(v)
    best_v = v.copy()
    best_val = cur_val

    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit * 0.97:
            break

        # pick a pair (increase, decrease)
        i_inc = rng.integers(N)
        if v[i_inc] >= 1.0:
            continue
        i_dec = rng.integers(N)
        if i_dec == i_inc or v[i_dec] <= 0.0:
            continue

        max_inc = min(1.0 - v[i_inc], v[i_dec])
        if max_inc <= 0.0:
            continue

        progress = elapsed / (time_limit * 0.97)
        step_factor = max(0.02, 1.0 - progress)        # shrink as time passes
        delta = max_inc * rng.random() * step_factor
        if delta < 1e-12:
            continue

        v_new = v.copy()
        v_new[i_inc] += delta
        v_new[i_dec] -= delta

        new_val = _evaluate(v_new)

        if new_val < cur_val:
            accept = True
        else:
            temp = 0.001 + 0.01 * (1.0 - progress)
            accept = rng.random() < math.exp((cur_val - new_val) / temp)

        if accept:
            v = v_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_v = v_new.copy()

    return best_v, best_val
