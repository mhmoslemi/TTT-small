def subgradient_step(h, step):
    N = len(h)
    # compute worst lag and gradient
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    L = max_idx - (N - 1)
    g = np.ones(N)
    if L >= 0:
        g[:N-L] -= h[L:]
        g[L:] -= h[:N-L]
    else:
        offset = -L
        g[offset:] -= h[:N - offset]
        g[:N - offset] -= h[offset:]
    # gradient descent
    h_new = h - step * g
    # clip
    h_new = np.clip(h_new, 0.0, 1.0)
    # project to target sum
    h_new = project_exact(h_new)
    return h_new, max_corr * (2.0 / N)  # maybe compute c5 after

import time
import random
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the Erdős C5 value for a step function sampled in h."""
    n = h.shape[0]
    dx = 2.0 / n
    # full linear correlation of h with (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_exact(h: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Clip to [0,1] and adjust so that sum(h) == target_sum.
    Simple water‑filling iteration – sufficient for the tiny problem sizes here.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            adjust = diff / free.sum()
            h[free] += adjust
        else:
            # all at bounds – distribute uniformly (will be re‑clipped)
            h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction (does not violate bounds for our precision)
    h += (target_sum - h.sum()) / h.size
    return np.clip(h, 0.0, 1.0)

def run(seed: int = 42, budget_s: float = 110, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        # fallback: 80 points, uniform 0.5 (satisfies the integral constraint)
        h = np.full(80, 0.5, dtype=np.float64)

    n = h.shape[0]
    target_sum = n / 2.0
    h = project_exact(h, target_sum)

    # ------------------------------------------------------------------
    # current best
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    start = time.time()
    # run a simple uphill‑hill‑climbing search – only accept improvements
    # This never hurts the initial solution and may discover a modest gain.
    while True:
        if time.time() - start > budget_s * 0.94:
            break

        # propose a move that preserves the total mass
        i, j = np.random.choice(n, 2, replace=False)
        # decide direction: increase i, decrease j OR opposite
        if random.random() < 0.5:
            # i up, j down
            max_up_i = 1.0 - h[i]
            max_down_j = h[j]
            if max_up_i <= 0.0 or max_down_j <= 0.0:
                continue
            delta = min(max_up_i, max_down_j) * random.random()
            if delta < 1e-8:
                continue
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
        else:
            # j up, i down
            max_up_j = 1.0 - h[j]
            max_down_i = h[i]
            if max_up_j <= 0.0 or max_down_i <= 0.0:
                continue
            delta = min(max_up_j, max_down_i) * random.random()
            if delta < 1e-8:
                continue
            h_new = h.copy()
            h_new[j] += delta
            h_new[i] -= delta

        # no need for explicit projection – the move respects bounds and sum
        new_c5 = compute_c5(h_new)

        # keep the move only if it improves the current best bound
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new  # continue searching from the improved point

    # final safety projection (tiny numerical drift)
    best_h = project_exact(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h.tolist(), best_c5, n
