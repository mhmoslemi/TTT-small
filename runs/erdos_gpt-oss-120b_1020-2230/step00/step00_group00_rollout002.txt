def project_onto_cap_simplex(v, target):
    # target = N/2
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        mid = (lo+hi)/2
        x = np.clip(v - mid, 0, 1)
        s = np.sum(x)
        if s > target:
            lo = mid
        else:
            hi = mid
    # After loop, use mid as λ, return x
    x = np.clip(v - ((lo+hi)/2), 0, 1)

            # spread equally

Implementation details for projection onto cap simplex:

def proj_cap_simplex(v, target):
    # Binary search λ
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        lam = (lo+hi)/2
        x = np.clip(v - lam, 0, 1)
        s = np.sum(x)
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2, 0, 1)
    # Adjust rounding error:
    diff = target - np.sum(x)
    if abs(diff) > 1e-8:
        # distribute across non-bound elements
        mask = (x > 0) & (x < 1)
        if np.any(mask):
            # add diff equally among mask
            x[mask] += diff / np.sum(mask)
            x = np.clip(x, 0, 1)
    return x

def gradient_max_shift(h, shift):
    N = h.shape[0]
    grad = np.ones(N)
    k = shift
    if k >= 0:
        if k > 0:
            grad[:N - k] -= h[k:]
            grad[k:] -= h[:N - k]
    else:
        k = -k
        grad[k:] -= h[:N - k]
        grad[:N - k] -= h[k:]
    return grad

def improve_h(h_init, time_limit):
    N = h_init.shape[0]
    target_sum = N/2
    h = h_init.copy()
    best_local = h.copy()
    best_val = compute_c5(h)
    start = time.time()
    # parameters
    while time.time() - start < time_limit:
        # compute correlation and shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        shift = max_idx - (N - 1)
        # gradient
        grad = gradient_max_shift(h, shift)
        # step size schedule: start 0.5 gradually decays
        elapsed = time.time() - start
        t_frac = elapsed / time_limit
        lr = 0.5 * (1 - t_frac) + 1e-4  # minimum step
        # propose new h
        h_candidate = h - lr * grad
        # project
        h_candidate = proj_cap_simplex(h_candidate, target_sum)
        val_candidate = compute_c5(h_candidate)
        # accept if improves else maybe revert
        if val_candidate <= compute_c5(h) or np.random.rand() < 0.01:
            h = h_candidate
            if val_candidate < best_val:
                best_val = val_candidate
                best_local = h_candidate.copy()
        # occasionally add noise
        if np.random.rand() < 0.05:
            h = proj_cap_simplex(h + 0.05*np.random.randn(N), target_sum)
        # early break if best_val is low
        if best_val <= 0.3808:
            break
    return best_local, best_val

import numpy as np
import time

# ---------- helper utilities ----------

def proj_cap_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {x in [0,1]^n : sum(x)=target}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        mask = (x > 0.0) & (x < 1.0)
        if mask.any():
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x


def compute_c5(h: np.ndarray) -> float:
    """Return the Erdős C5 value for a step function h on [0,2]."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)


def gradient_max_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of the correlation value attaining the maximum."""
    N = h.shape[0]
    grad = np.ones(N, dtype=float)
    if shift >= 0:
        k = shift
        if k:
            grad[: N - k] -= h[k:]          # term h_{j+shift}
            grad[k:] -= h[: N - k]          # term h_{j-shift}
    else:
        k = -shift
        grad[k:] -= h[: N - k]              # term h_{j-shift}
        grad[: N - k] -= h[k:]              # term h_{j+shift}
    return grad


# ---------- core optimisation routine ----------

def improve_h(h0: np.ndarray, time_limit: float) -> tuple:
    """
    Local optimisation starting from h0.
    Returns the best (h, C5) found within the allocated time_limit.
    """
    N = h0.shape[0]
    target = N * 0.5
    h = h0.copy()
    cur_val = compute_c5(h)
    best_h, best_val = h.copy(), cur_val
    start = time.time()
    while time.time() - start < time_limit:
        # correlation and shift of current maximum
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        shift = max_idx - (N - 1)

        # gradient of that term
        g = gradient_max_shift(h, shift)

        # step size (linear decay)
        elapsed = time.time() - start
        frac = elapsed / time_limit
        lr = 0.5 * (1.0 - frac) + 1e-4

        # gradient step + projection
        h_candidate = h - lr * g
        h_candidate = proj_cap_simplex(h_candidate, target)
        val_candidate = compute_c5(h_candidate)

        # simple acceptance rule
        if val_candidate < cur_val or np.random.rand() < 0.01:
            h, cur_val = h_candidate, val_candidate
            if val_candidate < best_val:
                best_h, best_val = h_candidate.copy(), val_candidate

        # occasional random perturbation to escape plateaus
        if np.random.rand() < 0.05:
            h = proj_cap_simplex(h + 0.05 * np.random.randn(N), target)
            cur_val = compute_c5(h)

        if best_val <= 0.3808:          # target met – stop early
            break
    return best_h, best_val

    Returns (h_values, c5_bound, n_points).
    """
