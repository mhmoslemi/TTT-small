    x = np.clip(x, 0.0, 1.0)
    diff = x.sum() - target
    if abs(diff) < 1e-12:
        return x
    # Adjust iteratively

        cnt = free.sum()
        if cnt == 0:
            # cannot distribute diff; break
            break
        delta = diff / cnt
        x[free] -= delta
        # clip again

            break
    # if still diff remains due to rounding, we can adjust by adding to some free entries partial
    return x

import time

def _project_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and enforce sum(x)=target by simple water‑filling."""
    x = np.clip(x, 0.0, 1.0)
    diff = x.sum() - target
    if abs(diff) < 1e-12:
        return x
    # distribute the surplus/deficit among free (non‑bound) entries
    for _ in range(12):
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        x[free] -= delta
        x = np.clip(x, 0.0, 1.0)
        diff = x.sum() - target
        if abs(diff) < 1e-12:
            break
    # tiny correction if needed
    if abs(x.sum() - target) > 1e-9:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            i = np.flatnonzero(free)[0]
            x[i] -= (x.sum() - target)
            x[i] = float(min(1.0, max(0.0, x[i])))
        else:
            # all at a bound – scale uniformly then clip
            scale = target / x.sum()
            x = np.clip(x * scale, 0.0, 1.0)
    return x


def _c5_from_h(h: np.ndarray, target_sum: float, dx: float) -> float:
    """C5 = max_k ∫ h·(1‑h∘k) = (target_sum – min overlap)·dx."""
    # linear autocorrelation gives the overlaps for all integer shifts
    overlaps = np.correlate(h, h, mode="full")
    min_overlap = overlaps.min()
    return (target_sum - min_overlap) * dx


def _min_lag(h: np.ndarray) -> int:
    """Return lag (in index units) where the overlap is minimal."""
    overlaps = np.correlate(h, h, mode="full")
    idx = overlaps.argmin()
    n = h.shape[0]
    return idx - (n - 1)


def _grad_overlap(h: np.ndarray, lag: int) -> np.ndarray:
    """Gradient of the overlap sum_i h_i·h_{i+lag} w.r.t. h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if lag >= 0:
        L = lag
        g[: n - L] += h[L:]
        g[L:] += h[: n - L]
    else:
        L = -lag
        g[L:] += h[: n - L]
        g[: n - L] += h[L:]
    return g
