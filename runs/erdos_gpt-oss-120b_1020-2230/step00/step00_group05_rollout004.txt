def solve_binary_mip(n):
    # build model
    h = cp.Variable(n, boolean=True)
    w = [] # list of cp.Variable for each shift s=1..n-1
    constraints = []
    # linking constraints for each s
    for s in range(1, n):
        m = n - s
        w_s = cp.Variable(m, boolean=True)
        w.append(w_s)
        constraints.append(w_s <= h[:m])
        constraints.append(w_s <= h[s:])
        constraints.append(w_s >= h[:m] + h[s:] - 1)
    # sum constraint
    constraints.append(cp.sum(h) == n/2)
    # t variable for max raw overlap
    t_raw = cp.Variable()
    constraints.append(t_raw >= 0)
    for s in range(1, n):
        w_s = w[s-1]
        # positive shift
        sum_h_pos = cp.sum(h[:n-s])
        O_raw_pos = sum_h_pos - cp.sum(w_s)
        constraints.append(O_raw_pos <= t_raw)
        # negative shift
        sum_h_neg = cp.sum(h[s:])
        O_raw_neg = sum_h_neg - cp.sum(w_s)
        constraints.append(O_raw_neg <= t_raw)
    prob = cp.Problem(cp.Minimize(t_raw), constraints)

        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > S:
            low = mid
        else:

def subgradient(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    O = corr * dx
    max_val = O.max()
    active = np.where(O >= max_val - 1e-12)[0]
    grad = np.zeros_like(h)
    center = n - 1
    for idx in active:
        s = idx - center
        if s >= 0:
            m = n - s
            grad[:m] += (1 - h[s:])   # (1 - h[i+s])
            grad[s:] += -h[:m]
        else:
            s2 = -s
            m = n - s2
            grad[s2:] += (1 - h[:m])
            grad[:m] += -h[s2:]
    grad *= dx
    return grad, max_val
```

import numpy as np
import time
import math

def _project_sum_bounds(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the set {x | 0<=x<=1, sum(x)=target_sum}."""
    low, high = -10.0, 10.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > target_sum:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x) (1-h(x+k)) dx approximated by correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def _subgradient_and_max(h: np.ndarray):
    """Sub‑gradient of the max‑overlap functional and its current value."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    max_val = vals.max()
    # indices where the maximum is attained (within tiny tolerance)
    act = np.where(vals >= max_val - 1e-12)[0]
    grad = np.zeros_like(h)
    centre = n - 1
    for idx in act:
        s = idx - centre
        if s >= 0:
            m = n - s
            grad[:m] += (1.0 - h[s:])          # derivative of h_i term
            grad[s:] += -h[:m]                # derivative of -h_{i+s} term
        else:
            s2 = -s
            m = n - s2
            grad[s2:] += (1.0 - h[:m])        # i is shifted right
            grad[:m] += -h[s2:]               # i is shifted left
    grad *= dx
    return grad, float(max_val)

        max_delta = min(h[i], 1.0 - h[j])
