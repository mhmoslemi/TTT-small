def evaluate_max_corr(h):
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_corr = corr[max_idx]
    return max_corr, max_idx, corr

def max_corr_score(h):
    max_corr, _, _ = evaluate_max_corr(h)
    return max_corr * (2.0 / h.shape[0])

import numpy as np, time, math

def _evaluate_max_corr(h):
    """Return (max_corr, max_index, full_correlation) for binary/real h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    return corr[max_idx], max_idx, corr

def _max_corr_score(h):
    max_corr, _, _ = _evaluate_max_corr(h)
    return max_corr * (2.0 / h.shape[0])

def _initial_candidate(n, rng):
    """Generate a random binary vector of length n with exactly n//2 ones."""
    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def _thue_morse(n):
    """Return Thue‑Morse binary vector of length n (n must be power of two)."""
    k = int(math.log2(n))
    seq = np.array([0], dtype=np.float64)
    for _ in range(k):
        seq = np.concatenate([seq, 1.0 - seq])
    return seq

def _simulated_annealing(h, time_budget, rng):
    """Binary simulated‑annealing that tries to lower max‑overlap."""
    n = h.shape[0]
    max_corr, max_idx, _ = _evaluate_max_corr(h)
    cur_score = max_corr * (2.0 / n)
    best_h = h.copy()
    best_score = cur_score

    temp = cur_score if cur_score > 0 else 1.0
    cooling = 0.9995
    stagnation = 0
    start = time.time()

    while time.time() - start < time_budget:
        s = max_idx - (n - 1)               # shift that currently attains the max
        i_start = max(0, -s)
        i_end = min(n, n - s)
        left = h[i_start:i_end]
        right = h[i_start + s:i_end + s]
        mismatched = np.where((left == 1) & (right == 0))[0] + i_start

        if mismatched.size:
            i = rng.choice(mismatched)
        else:
            i = rng.choice(np.where(h == 1)[0])

        j = rng.choice(np.where(h == 0)[0])
        # perform swap
        h[i], h[j] = h[j], h[i]

        new_corr, new_idx, _ = _evaluate_max_corr(h)
        new_score = new_corr * (2.0 / n)
        delta = new_score - cur_score

        accept = False
        if delta < -1e-12:
            accept = True
        else:
            if temp > 1e-12 and rng.random() < math.exp(-delta / temp):
                accept = True

        if accept:
            cur_score = new_score
            max_corr = new_corr
            max_idx = new_idx
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_h = h.copy()
            stagnation = 0
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
            stagnation += 1

        # simple restart if stuck for a while
        if stagnation > 5000:
            h = _initial_candidate(n, rng)
            max_corr, max_idx, _ = _evaluate_max_corr(h)
            cur_score = max_corr * (2.0 / n)
            stagnation = 0
            temp = cur_score if cur_score > 0 else 1.0

        temp *= cooling
    return best_h, best_score
