    n = len(h)
    a = np.cumsum(h) # prefix sum
    # For shift L, a(L) = sum_{i=0}^{n-L-1} h[i] = a[n-L-1] if n-L-1 >=0 else 0
    a_vec = np.empty(n, dtype=int)
    a_vec[0] = a[-1] # L=0 => sum all = total ones = n/2
    for L in range(1,n):
        a_vec[L] = a[n-L-1]  # sum first n-L elements

    # compute convolution for overlap
    b_corr = np.correlate(h, h, mode='full')

    # b_vec[L] = b(L)
    raw = a_vec - b_vec
    raw_max = raw.max()
    return raw_max

def evaluate_raw(h):
    # expects binary int array
    return np.correlate(h, 1 - h, mode='full').max()

def binarize_initial(vals):
    n = len(vals)
    ones_needed = n // 2
    arr = np.zeros(n, dtype=np.int8)
    idx = np.argsort(-vals) # descending
    arr[idx[:ones_needed]] = 1
    return arr

def random_binary(n, ones, rng):
    arr = np.zeros(n, dtype=np.int8)
    idx = rng.permutation(n)[:ones]
    arr[idx] = 1
    return arr

def run(seed=42, budget_s=110, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine size
    try:
        init_vals = initial_h_values
        n = len(init_vals)
    except NameError:
        n = 70
        init_vals = None
    if n % 2 != 0:
        # make even by adding a zero at end maybe
        n += 1
        if init_vals is not None:
            init_vals = np.append(init_vals, 0.0)
    ones = n // 2
    # Build initial binary vector
    if init_vals is not None:
        h_current = binarize_initial(init_vals)
    else:
        h_current = random_binary(n, ones, rng)
    cur_raw = evaluate_raw(h_current)
    best_h = h_current.copy()
    best_raw = cur_raw
    # We will keep the list of positions of ones and zeros to sample quickly
    # but we can just sample random indices.
    # Parameters
    max_no_improve = 2000
    no_improve_cnt = 0
    # Main loop
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.95:
            break
        # Temperature schedule
        progress = elapsed / (budget_s * 0.95)
        T = max(0.01, (1.0 - progress) * 5.0)  # start high, cool down
        # Choose random 1 index and 0 index
        ones_idx = np.where(h_current == 1)[0]
        zeros_idx = np.where(h_current == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break

        j = rng.choice(zeros_idx)

        h_current[j] = 1
        new_raw = evaluate_raw(h_current)
        delta = new_raw - cur_raw
        accept = False
        if delta <= 0:
            accept = True
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-delta / T):
                accept = True
        if accept:

            if new_raw < best_raw:
                best_raw = new_raw
                best_h = h_current.copy()
                no_improve_cnt = 0
                # early exit if sufficiently good
                if best_raw <= 13:
                    break

                no_improve_cnt += 1

            h_current[j] = 0
            no_improve_cnt += 1

            cur_raw = best_raw

def solve_milp(n, ones):
    # create cvxpy variables
    h = cp.Variable(n, integer=True)
    T = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == ones]
    w_vars = {}
    for L in range(n):
        for i in range(n - L):
            w = cp.Variable(boolean=True)
            w_vars[(i, L)] = w
            constraints += [w <= h[i],
                            w <= h[i+L],
                            w >= h[i] + h[i+L] - 1]
    for L in range(n):
        a = cp.sum(h[:n-L])
        b = cp.sum([ w_vars[(i, L)] for i in range(n-L) ])
        constraints.append(a - b <= T)
    constraints.append(T >= 0)
    prob = cp.Problem(cp.Minimize(T), constraints)
    # Solve with time limit
    try:
        prob.solve(solver=cp.CBC, verbose=False)
    except Exception as e:
        # fallback
        return None, None
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_sol = np.rint(h.value).astype(int)  # round
        T_val = T.value  # may be fractional; compute raw_max as max a-b
        raw_max = int(round(T_val))
        return h_sol, raw_max
    else:
        return None, None
```

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------
def raw_max_of_binary(h: np.ndarray) -> int:
    """Return max_k  sum_i h[i] * (1 - h[i+k])  for binary h.
    Equivalent to max over full-mode correlation of h with (1-h)."""
    # correlation uses zero padding outside the array
    conv = np.correlate(h, 1 - h, mode="full")
    return int(conv.max())


def binarize_initial(vals: np.ndarray) -> np.ndarray:
    """Convert an initial vector (maybe fractional) to a binary vector
    with exactly half ones (assumes even length)."""
    n = len(vals)
    ones_needed = n // 2
    idx = np.argsort(-vals)          # descending order
    bin_vec = np.zeros(n, dtype=np.int8)
    bin_vec[idx[:ones_needed]] = 1
    return bin_vec


def random_binary(n: int, ones: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random binary vector of length n with exactly `ones` ones."""
    vec = np.zeros(n, dtype=np.int8)
    perm = rng.permutation(n)[:ones]
    vec[perm] = 1
    return vec


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 110, **kwargs):
    """
    Optimize a step function for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points) where:
      * h_values  – numpy array of floats in [0,1] (binary in our case)
      * c5_bound  – the achieved upper bound for C5
      * n_points  – number of discretisation points used
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --------------------------------------------------------------
    # Determine problem size and initial vector
    # --------------------------------------------------------------
    try:
        # Provided by the harness (may be None)
        init_vals = initial_h_values  # type: ignore
        n = len(init_vals)
    except Exception:
        init_vals = None
        n = 70                     # fallback size

    # ensure even number of points (integral number of ones)
    if n % 2 != 0:
        n += 1
        if init_vals is not None:
            # extend with a zero to keep sum unchanged
            init_vals = np.concatenate([init_vals, np.zeros(1, dtype=init_vals.dtype)])

    ones_needed = n // 2
    dx = 2.0 / n

    # start vector (binary)
    if init_vals is not None:
        h_current = binarize_initial(np.asarray(init_vals, dtype=float))
    else:
        h_current = random_binary(n, ones_needed, rng)

    cur_raw = raw_max_of_binary(h_current)
    best_h = h_current.copy()
    best_raw = cur_raw

    # --------------------------------------------------------------
    # Simulated annealing / hill‑climbing loop
    # --------------------------------------------------------------
    # parameters for the search
    max_no_improve = 3000      # restart after many failed moves
    no_improve_cnt = 0
    # we keep a modest temperature schedule; it shrinks linearly
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break

        # temperature (high at start, low later)
        progress = elapsed / (budget_s * 0.95)
        T = max(0.01, (1.0 - progress) * 5.0)

        # propose a swap (keep total number of ones constant)
        i = rng.integers(n)
        while h_current[i] == 0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h_current[j] == 1:
            j = rng.integers(n)

        # perform the swap
        h_current[i] = 0
        h_current[j] = 1

        new_raw = raw_max_of_binary(h_current)
        delta = new_raw - cur_raw

        accept = False
        if delta <= 0:
            accept = True
        else:
            # Metropolis acceptance rule
            if rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            cur_raw = new_raw
            if new_raw < best_raw:
                best_raw = new_raw
                best_h = h_current.copy()
                no_improve_cnt = 0
                # early exit if we already beat the target bound
                if best_raw <= 13:       # 13 → C5 ≈ 0.3714 < 0.3808
                    break
            else:
                no_improve_cnt += 1
        else:
            # revert swap
            h_current[i] = 1
            h_current[j] = 0
            no_improve_cnt += 1

        # restart if stuck for a while
        if no_improve_cnt >= max_no_improve:
            # re‑initialise from the best found so far, add tiny perturbation
            h_current = best_h.copy()
            cur_raw = best_raw
            # make a few random swaps to leave the basin
            perturb = min(5, ones_needed // 2)
            for _ in range(perturb):
                i = rng.integers(n)
                while h_current[i] == 0:
                    i = rng.integers(n)
                j = rng.integers(n)
                while h_current[j] == 1:
                    j = rng.integers(n)
                h_current[i] = 0
                h_current[j] = 1
            cur_raw = raw_max_of_binary(h_current)
            no_improve_cnt = 0

    # --------------------------------------------------------------
    # Produce the final answer
    # --------------------------------------------------------------
    final_c5 = best_raw * dx
    h_float = best_h.astype(float)   # evaluator expects float array
    return (h_float, float(final_c5), int(n))
