def run(seed=42, budget_s=110, **kwargs):
    import numpy as np, time, random, math
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    time_left = lambda: budget_s - (time.time() - start)
    # choose n: maybe 200 (nice)
    n = 200
    # target sum
    target_sum = n / 2.0
    # compute dx for later
    dx = 2.0 / n

    def compute_c5(h):
        conv = np.correlate(h, 1 - h, mode='full')
        return conv.max() * dx

    # Helper to adjust sum and bounds after scaling
    def make_feasible(h):
        # ensure within [0,1]
        np.clip(h, 0, 1, out=h)
        # adjust sum to target
        current = h.sum()
        diff = target_sum - current
        if abs(diff) > 1e-12:
            # distribute diff greedily to indices with capacity
            # if diff >0, need increase some entries; else need decrease
            if diff > 0:
                # increase entries with room
                caps = 1 - h
                # indices sorted descending capacity
                idxs = np.argsort(-caps) # but caps large first
                for idx in idxs:
                    if diff <= 1e-12:
                        break
                    inc = min(diff, caps[idx])
                    if inc > 0:
                        h[idx] += inc
                        diff -= inc
            else:
                # decrease entries >0
                excess = -diff
                vals = h
                idxs = np.argsort(-vals) # largest first
                for idx in idxs:
                    if excess <= 1e-12:
                        break
                    dec = min(excess, vals[idx])
                    if dec > 0:
                        h[idx] -= dec
                        excess -= dec
        # final clip
        np.clip(h, 0, 1, out=h)
        return h

    # Local search function
    def improve(h, time_budget):

            new_val = compute_c5(new_h)

            # Metropolis acceptance
            if new_val < cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
                h = new_h
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_h = h.copy()
                    if best_val <= 0.3808:     # early stop if already very good
                        break
        return best_h, best_val

    # ------------------------------------------------------------
    # Run optimisation on each start, respecting the overall budget
    # ------------------------------------------------------------
    best_h = None
    best_c5 = math.inf
    total_starts = len(starts)

    for s_idx, h0 in enumerate(starts):
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 0.5   # small safety margin
        if remaining <= 0:
            break
        # allocate a fair share of remaining time to the remaining starts
        alloc = remaining * 0.9 / (total_starts - s_idx)
        h_opt, c5_opt = improve(h0, alloc)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if best_c5 <= 0.3808:
            break

    # If nothing was found (should not happen), fall back to the first start
    if best_h is None:
        best_h = project_feasible(starts[0].copy())
        best_c5 = compute_c5(best_h)

    # Final projection (just in case of tiny drift)
    best_h = project_feasible(best_h)
    best_c5 = compute_c5(best_h)

    # Return as a Python list for the harness
    return (best_h.tolist(), float(best_c5), int(n))
