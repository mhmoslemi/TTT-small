import time, math, random
import numpy as np

def _raw_max_corr(v):
    """Maximum of the discrete correlation h * (1-h) (no scaling)."""
    return np.max(np.correlate(v, 1.0 - v, mode="full"))

def run(seed: int = 42, budget_s: float = 110, **kwargs):
    # ------------------------------------------------------------------
    # Setup
    # ------------------------------------------------------------------
    random.seed(seed)
    np.random.seed(seed)

    # Starting vector ---------------------------------------------------
    try:
        h = initial_h_values.astype(np.float64).copy()   # provided by harness
    except NameError:

        n = 200
        dx = 2.0 / n
        x = (np.arange(n) + 0.5) * dx
        h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))       # peak = 1 at centre
        # Force exact integral = 1  (i.e. sum = n/2)
        target_sum = n / 2.0
        h *= target_sum / h.sum()
    n = h.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0

    # Ensure feasibility (tiny rounding fixes)
    h = np.clip(h, 0.0, 1.0)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        for i in range(n):
            hi = h[i]
            if diff > 0 and hi < 1.0:
                delta = min(diff, 1.0 - hi)
                h[i] += delta
                diff -= delta
            elif diff < 0 and hi > 0.0:
                delta = max(diff, -hi)
                h[i] += delta
                diff -= delta
            if abs(diff) < 1e-12:

    best_h = h.copy()

        if random.random() < 0.7 and max_idxs.size:

            if k > 0:

                j = i + k
            elif k < 0:
