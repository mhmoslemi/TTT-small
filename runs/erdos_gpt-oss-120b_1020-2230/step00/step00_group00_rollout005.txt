import numpy as np
import time

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step function given by h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def v_shape(n: int) -> np.ndarray:
    """V‑shaped profile h(x)=|x-1| (scaled to satisfy the integral constraint)."""
    i = np.arange(n, dtype=float)
    h = 2.0 * np.abs(i / (n - 1) - 0.5)
    target = n / 2.0
    factor = target / h.sum()
    h *= factor
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for k in range(n):
            if 0.0 <= h[k] + diff <= 1.0:
                h[k] += diff
                break
    return h


def linear_desc(n: int) -> np.ndarray:
    """Linear decreasing profile from 1 to 0 (already integral‑correct)."""
    h = 1.0 - np.arange(n, dtype=float) / (n - 1)
    h = np.clip(h, 0.0, 1.0)
    return h


def random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random vector in [0,1]^n with sum n/2."""
    h = rng.random(n)
    target = n / 2.0
    factor = target / h.sum()
    h = np.clip(h * factor, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for k in range(n):
            if 0.0 <= h[k] + diff <= 1.0:
                h[k] += diff
                break
    return h


def hill_climb(start_h: np.ndarray, budget: float, rng: np.random.Generator):
    """Greedy projected mass‑transfer search limited by time."""
    h = start_h.copy()
    n = h.shape[0]
    target = n / 2.0
    cur_val = compute_c5(h)
    best_h, best_val = h.copy(), cur_val
    start = time.time()
    init_step, final_step = 0.05, 0.001
    it = 0
    while time.time() - start < budget:
        prog = (time.time() - start) / budget
        step = init_step * (1 - prog) + final_step * prog

        # self‑correlation to find the shift with minimal overlap
        conv = np.convolve(h, h[::-1], mode="full")
        prod = conv[n - 1:]                     # length n, shift 0..n‑1
        kstar = int(np.argmin(prod))

        # score = h[i+k] + h[i‑k] for the critical shift
        s = kstar
        scores = np.zeros(n)
        if s > 0:
            scores[: n - s] += h[s:]
            scores[s:] += h[: n - s]
        else:
            scores[:] = 2.0 * h

        low_idx = np.where(h > 1e-12)[0]
        high_idx = np.where(h < 1 - 1e-12)[0]
        if low_idx.size == 0 or high_idx.size == 0:
            break

        i_low = low_idx[np.argmin(scores[low_idx])]
        i_high = high_idx[np.argmax(scores[high_idx])]

        max_delta = min(step, h[i_low], 1.0 - h[i_high])
        if max_delta <= 1e-12:
            i_low = rng.choice(low_idx)
            i_high = rng.choice(high_idx)
            max_delta = min(step, h[i_low], 1.0 - h[i_high])
            if max_delta <= 1e-12:
                continue

        delta = rng.uniform(0.0, max_delta)
        h_new = h.copy()
        h_new[i_low] -= delta
        h_new[i_high] += delta

        new_val = compute_c5(h_new)
        if new_val < cur_val - 1e-12:
            h = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()

        # occasional random perturbation to escape plateaus
        if it % 500 == 0:
            i, j = rng.integers(n, size=2)
            if i != j:
                lo = max(h[i] - 1.0, -h[j])
                hi = min(h[i], 1.0 - h[j])
                if hi > lo:
                    d = rng.uniform(lo, hi) * 0.5
                    h2 = h.copy()
                    h2[i] -= d
                    h2[j] += d
                    val2 = compute_c5(h2)
                    if val2 < cur_val - 1e-12:
                        h = h2
                        cur_val = val2
                        if val2 < best_val:
                            best_val = val2
                            best_h = h.copy()
        it += 1

    # final feasibility repair
    h = np.clip(best_h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for k in range(n):
            if 0.0 <= h[k] + diff <= 1.0:
                h[k] += diff
                break
    return h, compute_c5(h)
