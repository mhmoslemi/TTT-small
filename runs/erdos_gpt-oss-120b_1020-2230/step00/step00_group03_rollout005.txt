def project_box_simplex(y, target, lb=0.0, ub=1.0, tol=1e-8):
    lo = -np.max(y) - 2.0
    hi = ub - np.min(y) + 2.0
    for _ in range(60):
        lam = (lo + hi) / 2
        proj = np.clip(y + lam, lb, ub)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(y + (lo+hi)/2, lb, ub)

import numpy as np
import time
import random

def _compute_c5(h):
    """return max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _project_box_simplex(v, target):
    """
    Euclidean projection of vector v onto the set
    {x | 0 ≤ x_i ≤ 1,   sum(x) = target }.
    Simple bisection on the Lagrange multiplier λ.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)


def run(seed=42, budget_s=110, **kwargs):
    """
    Heuristic search for a step function h on [0,2] that lowers the
    Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # initialise h
    try:
        init = np.array(initial_h_values, copy=False).astype(float)
        n = init.size
        h = np.clip(init, 0.0, 1.0)
    except Exception:
        n = 70
        h = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0                # because dx = 2/n and ∑h*dx = 1
    h = _project_box_simplex(h, target_sum)

    best_h = h.copy()
    best_c = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # simulated‑annealing / hill‑climbing loop
    cur_h = best_h.copy()
    cur_c = best_c
    T0 = 5e-3                           # initial temperature (scaled later)
    it = 0

    while time.time() - start_time < budget_s * 0.95:
        # propose a move that preserves the sum exactly:
        i, j = np.random.choice(n, 2, replace=False)
        if random.random() < 0.5:
            # move mass from i to j
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 1e-12:
                continue
            delta = max_delta * random.random()
            h_new = cur_h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            # move mass from j to i
            max_delta = min(cur_h[j], 1.0 - cur_h[i])
            if max_delta <= 1e-12:
                continue
            delta = max_delta * random.random()
            h_new = cur_h.copy()
            h_new[j] -= delta
            h_new[i] += delta

        # evaluate new candidate
        new_c = _compute_c5(h_new)
        dC = new_c - cur_c

        # temperature decays linearly with remaining time
        elapsed = time.time() - start_time
        remaining = max(budget_s - elapsed, 0.0)
        T = T0 * (remaining / budget_s)

        # accept if improvement or with Metropolis probability
        if dC < -1e-12 or random.random() < np.exp(-dC / max(T, 1e-12)):
            cur_h, cur_c = h_new, new_c
            if new_c < best_c - 1e-12:
                best_c, best_h = new_c, h_new.copy()
                # early exit if we beat the target record
                if best_c < 0.3808:
                    break

        it += 1

        # occasional random jitter to avoid stagnation
        if it % 2000 == 0:
            jitter = (np.random.rand(n) - 0.5) * 0.02  # small perturbation
            cur_h = _project_box_simplex(cur_h + jitter, target_sum)
            cur_c = _compute_c5(cur_h)
            if cur_c < best_c:
                best_c, best_h = cur_c, cur_h.copy()

    # ensure final vector respects constraints exactly
    best_h = _project_box_simplex(best_h, target_sum)
    best_c = _compute_c5(best_h)

    return best_h, best_c, n
