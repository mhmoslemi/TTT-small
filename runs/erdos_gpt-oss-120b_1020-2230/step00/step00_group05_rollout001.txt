def project_onto_simplex_box(y, S, lower=0., upper=1., tol=1e-9, max_iter=100):
    lo = -1e6
    hi = 1e6
    # ensure sum at lo > S (clip to upper)
    # Actually when lambda very negative, y - lambda huge, clip to upper => sum = N*upper.
    # So any very negative lo yields sum > S.
    # For lambda very positive, sum = N*lower = 0.
    # So we can set lo = -np.max(y) - 10, hi = np.max(y) + 10.
    lo = -np.max(y) - 10.0
    hi = np.max(y) + 10.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, lower, upper)
        s = x.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    x = np.clip(y - (lo+hi)/2.0, lower, upper)
    # due to rounding errors, sum may deviate slightly; we can adjust
    # if sum differs, we can distribute difference to entries not at bounds
    diff = S - x.sum()
    if abs(diff) > 1e-8:
        # adjust free variables (not at bound) uniformly
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        num_free = free.sum()
        if num_free > 0:
            x[free] += diff / num_free
            # clip again if overshoot
            x = np.clip(x, lower, upper)
        else:
            # all at bounds, adjust shortest distance
            # set one variable to accommodate diff within bounds
            idx = np.argmin(np.abs(x - lower)) if diff < 0 else np.argmin(np.abs(x - upper))
            x[idx] = np.clip(x[idx] + diff, lower, upper)
    return x
```

import numpy as np
import time
import math

def _project_onto_box_simplex(y, S, lo=0.0, hi=1.0, tol=1e-9, maxit=100):
    """Project y onto the set {x: sum x = S, lo <= x_i <= hi}."""
    # binary search for lambda such that sum clip(y - lam, lo, hi) = S
    lo_lam = -1e6
    hi_lam = 1e6
    # ensure bounds encompass solution
    for _ in range(maxit):
        lam = (lo_lam + hi_lam) * 0.5
        x = np.clip(y - lam, lo, hi)
        s = x.sum()
        if s > S:
            lo_lam = lam
        else:
            hi_lam = lam
        if hi_lam - lo_lam < tol:
            break
    x = np.clip(y - (lo_lam + hi_lam) * 0.5, lo, hi)
    # tiny correction for numerical sum error
    diff = S - x.sum()
    if abs(diff) > 1e-8:
        free = (x > lo + 1e-12) & (x < hi - 1e-12)
        nfree = free.sum()
        if nfree > 0:
            x[free] += diff / nfree
            x = np.clip(x, lo, hi)
        else:
            # all at bounds – adjust one entry safely
            idx = np.argmin(np.abs(x - lo)) if diff < 0 else np.argmin(np.abs(x - hi))
            x[idx] = np.clip(x[idx] + diff, lo, hi)
    return x

def _gradient_max_shift(h, k):
    """Gradient of g_k = S - sum_i h_i h_{i+k} w.r.t h."""
    N = h.shape[0]
    grad = np.ones_like(h)
    if k > 0:
        grad[:N - k] -= h[k:]
        grad[k:] -= h[:N - k]
    elif k < 0:
        s = -k
        grad[s:] -= h[:N - s]
        grad[:N - s] -= h[s:]
    else:  # k == 0
        grad -= 2.0 * h
    return grad

def _min_product_and_shift(h):
    """Return (min_product, k) where k gives the minimum product sum."""
    N = h.shape[0]
    conv = np.correlate(h, h, mode='full')
    min_idx = int(conv.argmin())
    min_prod = float(conv.min())
    shift_corr = min_idx - (N - 1)          # correlation shift (second index moved)
    k = -shift_corr                         # forward shift giving same product
    return min_prod, k

def _c5_from_h(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx using discrete correlation."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _random_feasible(N, S, rng):
    y = rng.random(N)
    return _project_onto_box_simplex(y, S)

def run(seed=42, budget_s=110, **kwargs):

    start = time.time()
    # initialise h

        h = np.array(initial_h_values, dtype=float)
        N = h.size
    else:
        N = kwargs.get('n_points', 70)
        h = _random_feasible(N, N / 2.0, rng)
    S = N / 2.0
    dx = 2.0 / N
    # enforce feasibility of the starting point
    h = _project_onto_box_simplex(h, S)

    # evaluate initial solution
    min_prod, k_star = _min_product_and_shift(h)
    best_score = (S - min_prod) * dx
    best_h = h.copy()
    cur_h = h.copy()
    cur_score = best_score
    cur_min_prod = min_prod
    cur_k = k_star

    # parameters for the search
