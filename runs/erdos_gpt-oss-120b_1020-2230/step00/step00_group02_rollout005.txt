import numpy as np
import time
import random
import scipy.optimize as opt

# Global for constraints
CURRENT_N = None


def compute_c5(h):
    """Return max overlap integral for a step function h (array of values)."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx


def eq_sum_fun(x):
    """Equality: total mass = 1 (i.e. sum(h) = n/2)."""
    n = CURRENT_N
    return np.sum(x[:n]) - n / 2.0


def ineq_fun(x):
    """Inequality: t - O_shift >= 0 for every shift."""
    n = CURRENT_N
    dx = 2.0 / n
    h = x[:n]
    t = x[n]
    O = np.correlate(h, 1.0 - h, mode="full") * dx
    return t - O  # vector of length 2n-1, each must be >=0


def objective_fun(x):
    """Objective is the scalar t (the max overlap we try to minimise)."""
    n = CURRENT_N
    return x[n]


def slsqp_optimize(h0, max_iter=250):
    """Run a single SLSQP optimisation starting from h0."""
    global CURRENT_N
    n = h0.shape[0]
    CURRENT_N = n
    t0 = compute_c5(h0)
    x0 = np.concatenate([h0, [t0]])

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]  # h_i and t
    cons = [
        {"type": "eq", "fun": eq_sum_fun},
        {"type": "ineq", "fun": ineq_fun},
    ]
    opts = {"maxiter": max_iter, "ftol": 1e-9}
    try:
        res = opt.minimize(
            objective_fun, x0, method="SLSQP", bounds=bounds, constraints=cons, options=opts
        )
    except Exception:
        return None
    return res


def random_local_search(h, best_c5, end_time):
    """Simple stochastic local improvement keeping the total mass fixed."""
    n = h.shape[0]
    while time.time() < end_time:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # Move a small amount from i to j while staying in [0,1]
        delta_max = min(h[i], 1.0 - h[j])
        delta_min = -min(h[j], 1.0 - h[i])
        if delta_max <= 0 and delta_min >= 0:
            continue
        # random step, scaled to stay modest
        step = (np.random.rand() - 0.5) * 0.2  # +-0.1
        delta = np.clip(step, delta_min, delta_max)
        if abs(delta) < 1e-12:
            continue
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-9:
            h = h_new
            best_c5 = c5_new
    return h, best_c5

    # 1) Random binary with exact half mass

    vecs.append(h_bin.copy())

    # 2) Constant 0.5 (feasible but poor)
    vecs.append(np.full(n, 0.5))

    # 3) Upsampled parent if present
