import numpy as np, time, math
from scipy.optimize import direct  # maybe not needed.

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # using np.correlate for 1D real arrays
    conv = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(conv)

def run(seed=42, budget_s=110, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = h0.shape[0]
        # enforce sum exactly n/2
        target = n / 2.0
        cur_sum = h0.sum()
        if not np.isclose(cur_sum, target, atol=1e-12):
            # adjust last element to fix sum
            diff = target - cur_sum
            h0[-1] = np.clip(h0[-1] + diff, 0.0, 1.0)
            # if still off due to clipping, distribute remainder linearly
            cur_sum = h0.sum()
            if not np.isclose(cur_sum, target):
                # simple proportional scaling if feasible and does not violate bounds
                scale = target / cur_sum
                h0 = np.clip(h0 * scale, 0.0, 1.0)
                # final correction on last element
                cur_sum = h0.sum()
                h0[-1] += target - cur_sum
    else:
        n = 42
        # simple alternating 1/0 pattern (sum = n/2)
        h0 = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n)], dtype=float)

    # make sure sum exactly target
    target = n / 2.0
    cur_sum = h0.sum()
    if not np.isclose(cur_sum, target):
        # linear correction on first element (guaranteed within bounds because diff <=1)
        diff = target - cur_sum
        h0[0] = np.clip(h0[0] + diff, 0.0, 1.0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    current_h = best_h.copy()
    current_c5 = best_c5

    start_time = time.time()
    # simple SA parameters
    start_temp = 0.02
    final_temp = 1e-6
    target_c5 = 0.3808

    # Pre-compute number of steps per second? we just loop until time limit
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.96:
            break
        # temperature schedule linear
        t_frac = elapsed / (budget_s * 0.96)
        temp = start_temp * (1 - t_frac) + final_temp * t_frac

        # choose move type
        if rng.random() < 0.15:  # swap move
            i, j = rng.choice(n, size=2, replace=False)
            new_h = current_h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]
        else:
            # transfer move
            i, j = rng.choice(n, size=2, replace=False)
            max_transfer = min(current_h[i], 1.0 - current_h[j])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            new_h = current_h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        new_c5 = compute_c5(new_h)

        diff = current_c5 - new_c5
        if diff > 0 or rng.random() < math.exp(diff / temp):
            # accept
            current_h = new_h
            current_c5 = new_c5
            # update best if improved
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
                if best_c5 <= target_c5:
                    # Found sufficiently good solution, finish early
                    break

    # final evaluation using provided helper (may be more exact)
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except Exception:
        final_c5 = compute_c5(best_h)

    return best_h, float(final_c5), int(n)
