def project_onto_capped_simplex(v, target_sum, low=0.0, high=1.0, tol=1e-10, max_iter=100):
    # Find lambda such that sum(clip(v + lam, low, high)) = target_sum
    lam_low = low - np.max(v)  # ensures all values <= low after adding lam gives <= low => after clipping to low, sum=n*low
    lam_high = high - np.min(v)  # ensures all values >= high after adding lam yields high after clipping.

    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        x = np.clip(v + lam, low, high)
        s = x.sum()
        if abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            lam_high = lam
        else:
            lam_low = lam
    return np.clip(v + lam, low, high)

def run(seed=42, budget_s=110, **kwargs):
    import numpy as np, time, math
    np.random.seed(seed)
    start_time = time.time()
    # set n_points
    n = 150  # default
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        # upsample to n
        if n0 != n:
            repeat = n // n0
            remainder = n % n0
            h = np.repeat(h0, repeat)
            if remainder > 0:
                h = np.concatenate([h, h0[:remainder]])
            # now h length n
        else:
            h = h0.copy()
    else:
        h = np.random.rand(n)
    # project onto constraints
    h = project_onto_capped_simplex(h, n/2)
    best_h = h.copy()
    best_val = compute_c5(best_h)
    # parameters
    eta = 0.5
    alpha = 5.0
    target_sum = n/2
    dx = 2.0 / n
    # main loop
    while True:
        now = time.time()
        if now - start_time > budget_s * 0.95:
            break
        # correlation
        corr = np.correlate(h, 1 - h, mode='full') * dx
        cur_val = corr.max()
        # record best
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        # compute softmax weights
        max_c = corr.max()
        exp_vals = np.exp(alpha * (corr - max_c))
        w = exp_vals / exp_vals.sum()
        # gradient
        g = np.ones_like(h)
        center = n - 1
        # vectorized loop
        for idx, weight in enumerate(w):
            if weight == 0.0:
                continue
            shift = idx - center
            if shift == 0:
                g -= 2.0 * weight * h
                continue
            if shift > 0:
                # i+shift
                g[:n-shift] -= weight * h[shift:]
                # i-shift
                g[shift:] -= weight * h[:n-shift]
            else:
                s = -shift
                g[s:] -= weight * h[:n-s]
                g[:n-s] -= weight * h[s:]
        # step
        h_new = h - eta * g
        # project
        h = project_onto_capped_simplex(h_new, target_sum)
        # update step size and alpha
        eta = max(0.01, eta * 0.9995)  # decaying
        if alpha < 50:
            alpha *= 1.001
    # final local search
    # use random moves with time remaining
    remaining = budget_s - (time.time() - start_time)
    if remaining > 0:
        # allocate maybe half
        local_time_budget = remaining * 0.5
        end_local = time.time() + local_time_budget
        h_local = best_h.copy()
        best_local_val = best_val
        n = len(h_local)
        while time.time() < end_local:
            # propose move
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # decide direction: move from i to j (i decrease, j increase)
            max_decrease = h_local[i]
            max_increase = 1.0 - h_local[j]
            # pick delta up to min of both
            if max_decrease <= 0 or max_increase <= 0:
                continue
            delta = np.random.rand() * min(max_decrease, max_increase) * 0.5
            if delta <= 0:
                continue
            h_candidate = h_local.copy()
            h_candidate[i] -= delta
            h_candidate[j] += delta
            val = compute_c5(h_candidate)
            if val < best_local_val:
                best_local_val = val
                best_h = h_candidate.copy()
                best_val = val
                # optionally adopt as current
                h_local = h_candidate
            else:
                # sometimes accept uphill to escape? maybe small probability
                if np.random.rand() < 0.001:
                    h_local = h_candidate
        # end local search

    # final best value calculation
    final_val = compute_c5(best_h)
    return (best_h.tolist(), float(final_val), len(best_h))

def project_onto_capped_simplex(v, target_sum, low=0.0, high=1.0, tol=1e-10, max_iter=100):
    # ensure array
    v = np.asarray(v)
    # compute lam bounds
    lam_low = low - np.max(v) - 1.0  # maybe more safe bound
    lam_high = high - np.min(v) + 1.0
    # binary search
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        x = np.clip(v + lam, low, high)
        s = x.sum()
        if abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            lam_high = lam
        else:
            lam_low = lam
    # after loop return best approx
    return np.clip(v + lam, low, high)

import time
import numpy as np

def _project_capped_simplex(v, target_sum, low=0.0, high=1.0, tol=1e-12, max_iter=100):
    """Project vector v onto {x | low <= x <= high, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=float)
    # bounds for lambda
    lam_low = low - np.max(v) - 1.0
    lam_high = high - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        x = np.clip(v + lam, low, high)
        s = x.sum()
        if abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            lam_high = lam
        else:
            lam_low = lam
    return np.clip(v + lam, low, high)


def _c5_value(h):
    """Return the C5 value for a step function h (array of values on [0,2])."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)

            cand[i] -= delta
            cand[j] += delta
            val = _c5_value(cand)
            if val < local_val:
                local_val = val
                local_h = cand
                if val < best_c5:
                    best_c5 = val
                    best_h = cand.copy()

    # final evaluation (use the helper if it exists)
    try:
        final_val = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_val = _c5_value(best_h)

    return (best_h.tolist(), final_val, len(best_h))
