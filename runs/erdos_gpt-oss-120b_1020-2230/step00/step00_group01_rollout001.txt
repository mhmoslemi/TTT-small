import time
import math
import numpy as np

# ---------- helper functions -------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate implements the needed zero‑padded cross‑correlation
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the box [0,1]^n intersected with
    the hyperplane sum(v)=target. Uses a simple binary search for the
    Lagrange multiplier.
    """
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(45):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny fix for round‑off (distribute the remaining mass on non‑boundary entries)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        free = (w > 0.0) & (w < 1.0)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w


def propose_candidate(h: np.ndarray, step: float, rng: np.random.Generator,
                      target: float) -> np.ndarray:
    """
    Produce a new feasible candidate by moving a random amount between two
    coordinates and projecting back onto the feasible set.
    """
    n = h.shape[0]
    cand = h.copy()
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    delta = (rng.random() - 0.5) * 2.0 * step      # uniform in [-step, step]
    cand[i] += delta
    cand[j] -= delta
    cand = np.clip(cand, 0.0, 1.0)
    cand = project_to_sum(cand, target)
    return cand


# ---------- main entry point ------------------------------------------------

def run(seed: int = 42, budget_s: float = 110, **kwargs):
    """
    Return a tuple (h_values, c5_bound, n_points) where
    - h_values : list of floats defining the step function,
    - c5_bound : the achieved upper bound for C5,
    - n_points : number of discretisation points.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- problem size ------------------------------------------------------
    n = 100                         # modest size, good speed / quality trade‑off
    target_sum = n / 2.0

    # ----- initialise from the provided construction (if any) ---------------
    try:
        init_vec = initial_h_values          # supplied by the harness
        init_vec = np.asarray(init_vec, dtype=float)
        n0 = init_vec.shape[0]
        if n0 != n:
            # linear interpolation onto the new grid
            x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, init_vec)
        else:
            h0 = init_vec.copy()
    except NameError:
        # fallback: a symmetric triangular shape (integral = 1)
        dx = 2.0 / n
        xs = (np.arange(n) + 0.5) * dx
        h0 = np.where(xs <= 1.0, xs, 2.0 - xs)

    # make the start feasible
    h0 = project_to_sum(h0, target_sum)

    # ----- basic bookkeeping -------------------------------------------------
    best_h = h0.copy()
    best_val = compute_c5(best_h)

    cur_h = h0.copy()
    cur_val = best_val

    # ----- annealing parameters ---------------------------------------------
    step = 0.5                     # magnitude of the random delta
    temp = 0.02                    # initial temperature
    it = 0
    last_improve = 0
    max_no_improve = 5000

    # ----- main optimisation loop -------------------------------------------
    while True:
        # time guard
        if time.time() - start_time > budget_s * 0.97:
            break

        cand = propose_candidate(cur_h, step, rng, target_sum)
        cand_val = compute_c5(cand)

        # Metropolis acceptance
        if cand_val < cur_val or rng.random() < math.exp(-(cand_val - cur_val) / max(temp, 1e-12)):
            cur_h = cand
            cur_val = cand_val
            if cand_val < best_val - 1e-12:
                best_h = cand.copy()
                best_val = cand_val
                last_improve = it

        # schedule updates
        it += 1
        if it % 500 == 0:
            temp *= 0.95
            step *= 0.95
            step = max(step, 1e-4)

        # stagnation handling: random restart
        if it - last_improve > max_no_improve:
            restart = rng.random(n)
            cur_h = project_to_sum(restart, target_sum)
            cur_val = compute_c5(cur_h)
            last_improve = it

    # ----- return ------------------------------------------------------------
    return best_h.tolist(), float(best_val), n
