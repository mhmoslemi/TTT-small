def project_hyperplane_box(x, target_sum):
    # Clip to [0,1] then adjust sum via water-filling.
    x = np.clip(x, 0.0, 1.0)
    # Now need to adjust sum. We'll loop.
    for _ in range(10):
        total = x.sum()
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        # Indices not at bounds:
        free = (x > 0) & (x < 1)
        cnt = free.sum()
        if cnt == 0:
            # cannot adjust; break
            break
        add = diff / cnt
        # apply to free entries, but keep within bounds
        x[free] += add
        x = np.clip(x, 0.0, 1.0)
    return x

def project_feasible(x, target_sum):
    # Clip first
    x = np.clip(x, 0.0, 1.0)
    total = x.sum()
    diff = target_sum - total
    # We will distribute diff uniformly among free entries not at bounds
    # Using iterative water-filling
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        free = (x > 0) & (x < 1)
        cnt = free.sum()
        if cnt == 0:
            # cannot adjust, break
            break
        add = diff / cnt
        # apply
        x[free] += add
        # clip again
        x = np.clip(x, 0.0, 1.0)
        total = x.sum()
        diff = target_sum - total
    # If still off, maybe adjust a single variable arbitrarily within bounds
    if abs(diff) > 1e-9:
        # adjust first free element
        for idx in range(len(x)):
            if 0 < x[idx] < 1:
                x[idx] = np.clip(x[idx] + diff, 0, 1)
                break
    return x

def compute_c5(h, dx):
    # correlation of h with (1 - h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    return float(max_corr * dx)

import time, math
import numpy as np

def _compute_c5(h, dx):
    """Maximum overlap integral for a given step function."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _project_feasible(x, target):
    """
    Project vector x onto the feasible set:
        0 <= x_i <= 1  and  sum(x) = target
    Simple water‑filling algorithm.
    """
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        free = (x > 0) & (x < 1)
        cnt = free.sum()
        if cnt == 0:
            # No interior points – adjust a single bound variable
            if diff > 0:
                idx = np.argmax(1.0 - x)
                delta = min(diff, 1.0 - x[idx])
                x[idx] += delta
                diff -= delta
            else:
                idx = np.argmax(x)
                delta = min(-diff, x[idx])
                x[idx] -= delta
                diff += delta
            continue
        add = diff / cnt
        x[free] += add
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
    return x

def _gradient_C_at_shift(h, shift):
    """Gradient of C(s)=sum_i h_i(1-h_{i+shift}) w.r.t. h."""
    n = len(h)
    g = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            # left indices contribution (1 - h_{i+shift})
            g[:n - shift] += 1.0 - h[shift:]
            # right indices contribution -h_{i}
            g[shift:] -= h[:n - shift]
    else:
        s = -shift
        if s < n:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
    return g

def run(seed=42, budget_s=110, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős
    C5 overlap constant.

    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n = len(h)
    else:
        n = 80                         # a modest size if no parent provided
        h = np.zeros(n, dtype=np.float64)
        # half ones, odd length gets a single 0.5 entry to hit the exact sum
        idx = np.random.choice(n, n // 2, replace=False)
        h[idx] = 1.0
        if n % 2:
            rest = np.setdiff1d(np.arange(n), idx)
            h[rest[0]] = 0.5

    dx = 2.0 / n
    target = n / 2.0
    h = _project_feasible(h, target)

    best_h = h.copy()
    best_val = _compute_c5(best_h, dx)

    # ------------------------------------------------------------------
    # optimisation loop
    it = 0
    while time.time() - start < budget_s - 0.5:
        # decay parameters
        elapsed = time.time() - start
        progress = elapsed / budget_s
        # step for random Gaussian perturbations
        sigma = 0.5 * (1.0 - progress) + 0.02   # from ~0.52 down to 0.02
        # step for gradient moves
        step_g = 0.3 * (1.0 - progress) + 0.01

        # --------------------------------------------------------------
        # 1) random Gaussian proposal around the current best
        cand = best_h + sigma * np.random.randn(n)
        cand = np.clip(cand, 0.0, 1.0)
        cand = _project_feasible(cand, target)
        val = _compute_c5(cand, dx)
        if val < best_val:

        # --------------------------------------------------------------

        # compute correlation once for the current best
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_idx = corr.argmax()
        shift = max_idx - (n - 1)                # shift corresponding to max overlap
        grad = _gradient_C_at_shift(best_h, shift)

        if np.linalg.norm(grad) > 0:
            cand = best_h - step_g * grad        # move opposite to the gradient
            cand = np.clip(cand, 0.0, 1.0)
            cand = _project_feasible(cand, target)
            val = _compute_c5(cand, dx)
            if val < best_val:

        # --------------------------------------------------------------

        if it % 1000 == 0:
