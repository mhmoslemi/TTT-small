def propose_move(h):
   n = len(h)
   i,j = random.sample(range(n),2)
   # direction: add to i, subtract from j or vice versa.
   if random.random() < 0.5:
      # add to i, subtract from j
      max_add = min(1.0 - h[i], h[j])  # can't exceed capacity
      if max_add <= 1e-12: return None (skip)
      delta = random.random() * max_add
      new_h = h.copy()
      new_h[i] += delta
      new_h[j] -= delta
   else:
      # subtract from i, add to j
      max_sub = min(h[i], 1.0 - h[j])
      if max_sub <= 1e-12: return None
      delta = random.random() * max_sub
      new_h = h.copy()
      new_h[i] -= delta
      new_h[j] += delta
   return new_h

def compute_c5_fft(h):
    n = h.shape[0]
    dx = 2.0 / n
    a = h
    b_rev = (1 - h)[::-1]

    fb = np.fft.rfft(b_rev, L)

    The function returns the best vector found, its C5 value and its length.
