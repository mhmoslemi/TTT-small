import time, math, random
import numpy as np

def _max_overlap(h, dx):
    """Return max_k ∫ h(x)(1−h(x+k))dx approximated by discrete correlation."""
    # np.correlate uses sum_i a[i] * b[i+lag]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_sum_one_half(h, target):
    """
    Adjust h (in‑place) so that sum(h) == target while keeping 0≤h≤1.
    Simple iterative redistribution.
    """
    diff = h.sum() - target
    if abs(diff) < 1e-12:
        return h
    # Move the whole diff from a single feasible entry if possible
    if diff > 0:
        # need to decrease some entries
        idx = np.where(h > 0)[0]
        if idx.size:
            i = random.choice(idx.tolist())
            dec = min(diff, h[i])
            h[i] -= dec
            diff -= dec
    else:
        # need to increase some entries
        idx = np.where(h < 1)[0]
        if idx.size:
            i = random.choice(idx.tolist())
            inc = min(-diff, 1.0 - h[i])
            h[i] += inc
            diff += inc
    # If small residual remains, spread it uniformly over non‑saturated entries
    if abs(diff) > 1e-12:
        mask = (h > 0) & (h < 1)
        cnt = mask.sum()
        if cnt:
            h[mask] -= diff / cnt
            np.clip(h, 0.0, 1.0, out=h)
    # Final tiny correction
    final = h.sum() - target
    if abs(final) > 1e-12:
        i = np.argmin(np.abs(h - 0.5))  # some index
        h[i] -= final
        h[i] = min(max(h[i], 0.0), 1.0)
    return h
