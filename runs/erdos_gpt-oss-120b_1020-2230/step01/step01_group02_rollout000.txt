```
def _compute_c5_fast(h, dx):
    n = len(h)
    m = 1 << ((2*n - 1).bit_length())
    # FFT conv
    f = np.fft.rfft(h, n=m)
    g = np.fft.rfft(1.0 - h, n=m)
    conv = np.fft.irfft(f * g, n=m)
    # relevant part for full convolution
    conv = conv[:2*n-1]
    return float(conv.max() * dx)
```

import time, math, numpy as np

def _project_feasible(x, target):
    # water-filling etc.
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        free = (x > 0) & (x < 1)
        cnt = free.sum()
        if cnt == 0:
            if diff > 0:
                idx = np.argmax(1.0 - x)
                delta = min(diff, 1.0 - x[idx])
                x[idx] += delta
                diff -= delta
            else:
                idx = np.argmax(x)
                delta = min(-diff, x[idx])
                x[idx] -= delta
                diff += delta
            continue
        add = diff / cnt
        x[free] += add
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
    return x

def _gradient_C_at_shift(h, shift):
    n = len(h)
    g = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            g[:n - shift] += 1.0 - h[shift:]
            g[shift:] -= h[:n - shift]
    else:
        s = -shift
        if s < n:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
    return g

def _corr_fft(h):
    n = len(h)
    m = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(h, n=m)
    g = np.fft.rfft(1.0 - h, n=m)
    conv = np.fft.irfft(f * g, n=m)
    return conv[:2 * n - 1]

def _c5_fft(h, dx):
    conv = _corr_fft(h)
    return float(conv.max() * dx)
