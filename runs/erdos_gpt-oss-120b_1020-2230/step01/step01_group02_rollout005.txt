import time
import math
import numpy as np

# ----------------------------------------------------------------------
# helper functions ------------------------------------------------------
def _compute_c5(h, dx):
    """Maximum overlap integral for a given step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _project_feasible(x, target):
    """
    Project onto {0 <= x_i <= 1, sum(x) = target} using water‑filling.
    """
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        free = (x > 0) & (x < 1)
        cnt = free.sum()
        if cnt == 0:
            # all variables at a bound – adjust a single one
            if diff > 0:
                idx = np.argmax(1.0 - x)
                delta = min(diff, 1.0 - x[idx])
                x[idx] += delta
                diff -= delta
            else:
                idx = np.argmax(x)
                delta = min(-diff, x[idx])
                x[idx] -= delta
                diff += delta
            continue
        add = diff / cnt
        x[free] += add
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
    return x


def _gradient_C_at_shift(h, shift):
    """Gradient of C(s)=∑ h_i (1‑h_{i+shift}) w.r.t. h (for a fixed shift)."""
    n = len(h)
    g = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            g[: n - shift] += 1.0 - h[shift:]          # derivative of h_i term
            g[shift:] -= h[: n - shift]                # derivative of -h_i h_{i+shift}
    else:
        s = -shift
        if s < n:
            g[s:] += 1.0 - h[: n - s]
            g[: n - s] -= h[s:]
    return g


# ----------------------------------------------------------------------
def run(seed=42, budget_s=110, **kwargs):
    """
    Find a step function h on [0,2] (sampled uniformly) that minimises
    C_5 = max_k ∫ h(x) (1‑h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()
    # ------------------------------------------------------------------
    # decide number of samples – use 95 if a parent is given, otherwise 120
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        n = 120
        # start from a random feasible vector
        h0 = np.random.rand(n)
        target_sum = n / 2.0
        h0 = _project_feasible(h0, target_sum)

    dx = 2.0 / n
    target = n / 2.0

    # ensure feasibility of the starting point
    cur_h = _project_feasible(h0, target)
    cur_val = _compute_c5(cur_h, dx)

    best_h = cur_h.copy()
    best_val = cur_val
    best_iter = 0

    # ------------------------------------------------------------------
    # simulated‑annealing / stochastic hill‑climbing parameters
    max_iter = 300_000                     # generous upper bound
    T0 = 0.04                               # initial temperature
    step0 = 0.35                            # maximal mass transferred per move
    restart_gap = 5_000                     # if no improvement for this many iters

    last_improve = 0

    for it in range(max_iter):
        # time check – leave a small margin for finalisation
        if time.time() - start > budget_s - 0.6:
            break

        # linear cooling
        prog = it / max_iter
        T = T0 * (1.0 - prog)
        step = step0 * (1.0 - prog)

        # --------------------------------------------------------------
        # 1) pairwise mass exchange (keeps sum exactly)
        i, j = np.random.randint(n, size=2)
        if i != j:
            delta_max = min(step, 1.0 - cur_h[i], cur_h[j])
            delta_min = -min(step, cur_h[i], 1.0 - cur_h[j])
            if delta_max > delta_min:
                delta = np.random.uniform(delta_min, delta_max)
                cand = cur_h.copy()
                cand[i] += delta
                cand[j] -= delta
                cand_val = _compute_c5(cand, dx)

                if cand_val < cur_val or (
                    T > 0 and np.random.rand() < math.exp(-(cand_val - cur_val) / T)
                ):
                    cur_h, cur_val = cand, cand_val
                    if cur_val < best_val:
                        best_h, best_val = cur_h.copy(), cur_val
                        best_iter = it
                        last_improve = it

        # --------------------------------------------------------------
        # 2) occasional gradient descent on the worst shift
        if it % 200 == 0:
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            shift = corr.argmax() - (n - 1)
            grad = _gradient_C_at_shift(cur_h, shift)
            # step opposite to the gradient
            step_g = step * 0.5
            cand = cur_h - step_g * grad
            cand = np.clip(cand, 0.0, 1.0)
            cand = _project_feasible(cand, target)
            cand_val = _compute_c5(cand, dx)
            if cand_val < cur_val:
                cur_h, cur_val = cand, cand_val
                if cur_val < best_val:
                    best_h, best_val = cur_h.copy(), cur_val
                    best_iter = it
                    last_improve = it

        # --------------------------------------------------------------
        # 3) occasional global Gaussian perturbation
        if it % 1000 == 0 and it > 0:
            sigma = step0 * 0.1
            cand = cur_h + sigma * np.random.randn(n)
            cand = np.clip(cand, 0.0, 1.0)
            cand = _project_feasible(cand, target)
            cand_val = _compute_c5(cand, dx)
            if cand_val < cur_val or (

                cur_h, cur_val = cand, cand_val
                if cur_val < best_val:
                    best_h, best_val = cur_h.copy(), cur_val
                    best_iter = it
                    last_improve = it

        # --------------------------------------------------------------
        # 4) restart if stagnation is observed
        if it - last_improve > restart_gap:
            # re‑initialize from the current best with a modest random kick
            cur_h = best_h + 0.1 * np.random.randn(n)
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_h = _project_feasible(cur_h, target)
            cur_val = _compute_c5(cur_h, dx)
            last_improve = it

    # final safety projection and evaluation
    best_h = _project_feasible(best_h, target)
    final_val = _compute_c5(best_h, dx)

    return best_h, final_val, n
