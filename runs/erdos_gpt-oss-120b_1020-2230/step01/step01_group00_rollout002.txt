def max_overlap(h, dx):
    g = 1.0 - h
    L = 1 << ((2*n - 1).bit_length())
    # compute using FFT of length L
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(g, n=L)
    # correlation using conj of G
    corr = np.fft.irfft(H * np.conj(G), n=L)[:2*n-1]
    return np.max(corr) * dx

def compute_delta_c(h, g, i, j):
    # h is binary 0/1 vector, g = 1-h
    n = len(h)
    size = 2*n - 1
    delta = np.zeros(size)
    # contributions from i removal
    # for r where i+r in [0,n-1]: index k = (i+r)+(n-1)
    start_r = -i
    end_r = n-1 - i
    # iterate r
    for r in range(start_r, end_r+1):
        k = r + n - 1
        delta[k] -= g[i + r]   # g[i+r] is 0/1
    # contributions from g_i addition (Δg_i=+1)
    # for r where i-r in [0,n-1]
    start_r2 = i - (n-1)
    end_r2 = i
    for r in range(start_r2, end_r2+1):
        k = r + n - 1
        delta[k] += h[i - r]   # h[i-r] currently before change (i is still 1)
    # contributions from Δg_j = -1 (removal)
    start_r3 = j - (n-1)

        k = r + n - 1
        delta[k] += g[j + r]   # g[j+r] before change (currently 1)
    return delta
```

def hill_climb(h, max_iter=20000, time_limit=5.0):
    n = h.shape[0]
    dx = 2.0 / n
    cur_h = h.copy()
    cur_c = np.max(np.correlate(cur_h, 1.0 - cur_h, mode='full')) * dx
    best_local_h = cur_h.copy()
    best_local_c = cur_c
    start = time.time()
    # Temperature schedule
    T0 = 0.01
    T1 = 1e-6
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        # random swap
        i = np.random.randint(n)
        while cur_h[i] != 1.0:
            i = np.random.randint(n)
        j = np.random.randint(n)
        while cur_h[j] != 0.0:
            j = np.random.randint(n)
        # perform swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c = np.max(np.correlate(cur_h, 1.0 - cur_h, mode='full')) * dx
        # temperature
        elapsed = time.time() - start
        T = T0 * (1 - elapsed / time_limit) + T1 * (elapsed / time_limit)
        # decide
        if new_c < cur_c or np.random.rand() < math.exp(-(new_c - cur_c) / (T+1e-12)):
            cur_c = new_c
            if new_c < best_local_c:
                best_local_c = new_c
                best_local_h = cur_h.copy()
        else:

import time, math, numpy as np

def max_overlap(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def hill_climb(start_h, max_iter=200000, time_limit=10.0):
    n = start_h.shape[0]
    dx = 2.0 / n
    h = start_h.copy()
    g = 1.0 - h
    cur_c = np.max(np.correlate(h, g, mode='full')) * dx
    best_h = h.copy()
    best_c = cur_c
    start = time.time()
    # annealing schedule constants
    T0 = 0.01
    T1 = 1e-7
    it = 0
    while it < max_iter:
        elapsed = time.time() - start
        if elapsed > time_limit:
            break
        # select random 1 and 0
        i = np.random.randint(n)
        while h[i] != 1.0:
            i = np.random.randint(n)
        j = np.random.randint(n)
        while h[j] != 0.0:
            j = np.random.randint(n)
        # swap
        h[i] = 0.0
        g[i] = 1.0
        h[j] = 1.0
        g[j] = 0.0
        new_c = np.max(np.correlate(h, g, mode='full')) * dx
        # temperature
        T = T0 * (1 - elapsed / time_limit) + T1 * (elapsed / time_limit)
        if new_c < cur_c or np.random.rand() < math.exp(-(new_c - cur_c) / (T + 1e-12)):
            cur_c = new_c
            if new_c < best_c - 1e-12:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            g[i] = 0.0
            h[j] = 0.0
            g[j] = 1.0
        it += 1
    return best_h, best_c
