```
import numpy as np, time, random, math

def _project_box_simplex(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def _compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def _compute_c5_and_shift(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    max_val = corr[max_idx]
    shift = max_idx - (n - 1)
    return max_val * (2.0 / n), shift

def _gradient_worst_shift(h, shift):
    n = h.shape[0]
    grad = np.ones(n, dtype=float)
    # add contributions
    if shift >= 0:
        # i where i+shift < n
        i_end = n - shift
        if i_end > 0:
            grad[:i_end] -= h[shift:]
        # i where i-shift >=0
        if shift < n:
            grad[shift:] -= h[:n - shift]
    else:
        k = -shift
        i_end = n - k
        if i_end > 0:
            grad[k:] -= h[:i_end]
        if k < n:
            grad[:i_end] -= h[k:]
    return grad
```

Double-check gradient: For shift >=0 case:

- For i from 0 to n-1: if i+shift < n, subtract h[i+shift]; this corresponds to grad[:n-shift] -= h[shift:].

- If i-shift >=0: i >= shift. So for i in shift..n-1, subtract h[i - shift]; corresponds to grad[shift:] -= h[:n-shift].

Thus gradient update done correctly.

For shift negative:

Let k = -shift >0. Then i+shift = i - k . So valid if i - k >=0 => i >= k => grad[k:] -= h[:n-k] (i.e., h[i-k] = h[i-k] but on left side). This corresponds to subtract h[:n-k] from grad[k:].

Similarly i - shift = i + k (since shift negative). Condition i - shift < n => i + k < n => i < n - k. So for i in 0..n-k-1, subtract h[i + k] = h[i+k] = h[k:]; so grad[:n - k] -= h[k:].

Thus code above matches.

One more nuance: The self-subtractions double count i where both i+shift and i-shift are valid (i away from edges). That's correct as gradient includes both contributions.

import numpy as np
import time
import random
import math

def _project_box_simplex(v, target):
    """Euclidean projection of v onto {0<=x_i<=1, sum x_i = target}."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)


def _compute_c5(h):
    """C5 = max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _compute_c5_and_shift(h):
    """return (C5 value, shift giving the maximum)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # zero shift corresponds to index n‑1
    return corr[idx] * (2.0 / n), shift


def _gradient_worst_shift(h, shift):
    """Gradient of the worst overlap term w.r.t. h (sum constraint ignored)."""
    n = h.shape[0]
    g = np.ones(n, dtype=float)

    if shift >= 0:
        # subtract h[i+shift] where defined
        end = n - shift
        if end > 0:
            g[:end] -= h[shift:]
        # subtract h[i-shift] where defined
        if shift < n:
            g[shift:] -= h[:end]
    else:
        k = -shift
        end = n - k
        if end > 0:
            g[k:] -= h[:end]          # i+shift = i‑k
        if k < n:
            g[:end] -= h[k:]          # i‑shift = i+k
    return g


def _local_optimize(start_h, target_sum, start_time, deadline):
    """Hill‑climbing guided by the worst shift."""
    h = start_h.copy()
    best_h = h.copy()
    best_c = _compute_c5(best_h)
    n = h.shape[0]

    it = 0
    stagnant = 0
    max_iter = 30000                     # hard iteration cap per run

    while time.time() < deadline and it < max_iter:
        cur_c, cur_k = _compute_c5_and_shift(h)

        # record improvement
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1

        # gradient for the worst shift
        g = _gradient_worst_shift(h, cur_k)
        i = int(np.argmax(g))
        j = int(np.argmin(g))
        if i == j:
            i, j = np.random.choice(n, 2, replace=False)

        # amount we can move from i to j while staying in bounds
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 1e-12:
            # try the opposite direction
            max_delta = min(h[j], 1.0 - h[i])
            if max_delta <= 1e-12:
                # random perturbation to escape
                jitter = (np.random.rand(n) - 0.5) * 0.02
                h = _project_box_simplex(h + jitter, target_sum)
                it += 1
                continue
            i, j = j, i

        delta = max_delta * random.random()
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        new_c, _ = _compute_c5_and_shift(h_new)

        # deterministic acceptance of improvements, occasional Metropolis step
        if new_c < cur_c - 1e-12:
            h = h_new
        else:
            remaining = deadline - time.time()
            total = deadline - start_time
            T = max(1e-9, remaining / total)   # linear cooling
            if random.random() < math.exp(-(new_c - cur_c) / T):
                h = h_new

        it += 1

        # occasional jitter to fight stagnation

            h = np.random.rand(n)
            h = _project_box_simplex(h, target_sum)
            stagnant = 0

        init = np.array(initial_h_values, dtype=float)   # may be predefined
        n = init.size
        h0 = np.clip(init, 0.0, 1.0)
    except Exception:                                     # fallback
        n = 70
        h0 = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0
    h0 = _project_box_simplex(h0, target_sum)

    # ------------------------------------------------------------------
    # optimise first from the supplied (or uniform) start
    best_h, best_c = _local_optimize(h0, target_sum, start, deadline)

    # ------------------------------------------------------------------
    # additional random restarts while time permits
    while time.time() < deadline:
        rnd = np.random.rand(n)
        rnd = _project_box_simplex(rnd, target_sum)
        cand_h, cand_c = _local_optimize(rnd, target_sum, start, deadline)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h.copy()
            if best_c < 0.3808:          # early exit when record is broken
                break

    # final projection to guarantee feasibility
    best_h = _project_box_simplex(best_h, target_sum)
    best_c = _compute_c5(best_h)
    return best_h, best_c, n
