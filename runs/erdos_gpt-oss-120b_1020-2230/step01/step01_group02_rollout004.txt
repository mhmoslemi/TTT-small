def project_feasible(v, target):
    # Clip to [0,1]
    x = np.clip(v, 0, 1)
    # Compute remainder
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # Determine indices where x not at bounds
    # Want to add diff distributed equally among free vars.
    # But after adding, some may exceed bounds, need recalc.
    # Use while loop.
    free = np.ones_like(x, dtype=bool)
    while True:
        free = (x > 0) & (x < 1)
        cnt = free.sum()
        if cnt == 0:
            # all at bounds, adjust nearest bound
            # if diff >0, increase the smallest (x) not at 1?
            # but keep feasible: add diff to variable with most slack.
            if diff > 0:
                # choose index with max (1-x)
                idx = np.argmax(1 - x)
                add = min(diff, 1 - x[idx])
                x[idx] += add
                diff -= add
            else:
                idx = np.argmax(x) # largest x
                sub = min(-diff, x[idx])
                x[idx] -= sub
                diff += sub
        else:
            add = diff / cnt
            # add to free indices
            x[free] += add
            # clip any that go out of bounds
            over = x > 1
            under = x < 0
            if not over.any() and not under.any():
                # satisfied
                break
            # if overshoot, set to bound and update diff
            if over.any():
                excess = x[over] - 1
                diff = excess.sum()
                x[over] = 1
            if under.any():
                deficit = -x[under]
                diff = -deficit.sum()
                x[under] = 0
    return x

import time, math
import numpy as np

def _project_feasible(v, target):
    """Project onto box [0,1] and sum constraint."""
    x = np.clip(v, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # iterative water-filling
    for _ in range(100):
        free = (x > 0) & (x < 1)
        cnt = free.sum()
        if cnt == 0:
            # adjust a single bound variable
            if diff > 0:
                idx = np.argmax(1.0 - x)
                add = min(diff, 1.0 - x[idx])
                x[idx] += add
                diff -= add
            else:
                idx = np.argmax(x)
                sub = min(-diff, x[idx])
                x[idx] -= sub
                diff += sub
        else:
            add = diff / cnt
            x[free] += add
            # clip again
            over = x > 1.0
            under = x < 0.0
            if not over.any() and not under.any():
                break
            if over.any():
                excess = x[over] - 1.0
                diff = excess.sum()
                x[over] = 1.0
            if under.any():
                deficit = -x[under]
                diff = -deficit.sum()
                x[under] = 0.0
        if abs(diff) < 1e-12:
            break
    return x

def _corr_fft(v):
    n = v.shape[0]
    w = 1.0 - v
    L = 2 * n
    fv = np.fft.rfft(v, n=L)
    fw = np.fft.rfft(w[::-1], n=L)
    conv = np.fft.irfft(fv * fw, n=L)
    return conv[:2 * n - 1]

def _compute_c5(v, dx):
    corr = _corr_fft(v)
    return float(corr.max() * dx)

def _grad_shift(v, shift):
    n = len(v)
    g = np.ones_like(v)
    i = np.arange(n)
    j = i + shift
    mask = (j >= 0) & (j < n)
    g[mask] -= v[j[mask]]
    j2 = i - shift
    mask2 = (j2 >= 0) & (j2 < n)
    g[mask2] -= v[j2[mask2]]
    return g

import time, math
import numpy as np

def _project_feasible(v, target):
    """Projection onto the box [0,1] and the hyperplane sum(v)=target."""
    x = np.clip(v, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    for _ in range(100):
        free = (x > 0) & (x < 1)
        cnt = free.sum()
        if cnt == 0:
            if diff > 0:
                idx = np.argmax(1.0 - x)
                add = min(diff, 1.0 - x[idx])
                x[idx] += add
                diff -= add
            else:
                idx = np.argmax(x)
                sub = min(-diff, x[idx])
                x[idx] -= sub
                diff += sub
            continue
        add = diff / cnt
        x[free] += add
        # clip again and recompute remaining diff
        over = x > 1.0
        under = x < 0.0
        if not over.any() and not under.any():
            break
        if over.any():
            excess = x[over] - 1.0
            diff = excess.sum()
            x[over] = 1.0
        if under.any():
            deficit = -x[under]
            diff = -deficit.sum()
            x[under] = 0.0
        if abs(diff) < 1e-12:
            break
    return x

def _corr_fft(v):
    """Full linear correlation of v with (1‑v) using FFT."""
    n = v.shape[0]
    w = 1.0 - v
    L = 2 * n
    fv = np.fft.rfft(v, n=L)
    fw = np.fft.rfft(w[::-1], n=L)
    conv = np.fft.irfft(fv * fw, n=L)
    return conv[: 2 * n - 1]

def _compute_c5(v, dx):
    """Maximum overlap integral for the discretised step function."""
    corr = _corr_fft(v)
    return float(corr.max() * dx)

def _grad_shift(v, shift):
    """Gradient of C_shift(v)=∑_i h_i(1‑h_{i+shift}) w.r.t. v."""
    n = len(v)
    g = np.ones_like(v)
    i = np.arange(n)
    j = i + shift
    mask = (j >= 0) & (j < n)
    g[mask] -= v[j[mask]]
    j2 = i - shift
    mask2 = (j2 >= 0) & (j2 < n)
    g[mask2] -= v[j2[mask2]]
    return g

def run(seed=42, budget_s=110, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ---- initialise h -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.size
    else:
        n = 120
        h0 = np.random.rand(n)
    dx = 2.0 / n
    target = n / 2.0
    cur = _project_feasible(h0, target)
    cur_val = _compute_c5(cur, dx)

    best = cur.copy()
    best_val = cur_val

    it = 0
    # ---- main stochastic optimisation ---------------------------------
    while time.time() - start < budget_s - 0.05:
        elapsed = time.time() - start
        frac = elapsed / budget_s

        # temperature for simulated‑annealing acceptance
        T = max(1e-5, 0.02 * (1 - frac) + 0.001)

        # 1) pairwise mass exchange (keeps the sum exact)
        i, j = np.random.randint(n, size=2)
        if i != j:
            max_delta = min(0.15, 1.0 - cur[i], cur[j])
            min_delta = -min(0.15, cur[i], 1.0 - cur[j])
            if max_delta > min_delta:
                delta = np.random.uniform(min_delta, max_delta)
                cand = cur.copy()
                cand[i] += delta
                cand[j] -= delta
                cand_val = _compute_c5(cand, dx)
                if cand_val < cur_val or np.random.rand() < math.exp(-(cand_val - cur_val) / T):
                    cur, cur_val = cand, cand_val
                    if cand_val < best_val:
                        best, best_val = cand.copy(), cand_val

        # 2) Gaussian perturbation followed by projection
        sigma = 0.1 * (1 - frac) + 0.005
        cand = cur + sigma * np.random.randn(n)
        cand = np.clip(cand, 0.0, 1.0)
        cand = _project_feasible(cand, target)
        cand_val = _compute_c5(cand, dx)
        if cand_val < cur_val or np.random.rand() < math.exp(-(cand_val - cur_val) / T):
            cur, cur_val = cand, cand_val
            if cand_val < best_val:
                best, best_val = cand.copy(), cand_val

        # 3) gradient step aimed at the worst few shifts
        corr = _corr_fft(cur)
        topk = 3
        idxs = np.argpartition(-corr, topk - 1)[:topk]
        shifts = idxs - (n - 1)
        grad = np.zeros_like(cur)
        for s in shifts:
            grad += _grad_shift(cur, s)
        norm = np.linalg.norm(grad)
        if norm > 0:
            base_step = 0.1 * (1 - frac) + 0.002
            for factor in (0.5, 1.0, 1.5):
                step = base_step * factor / norm
                cand = cur - step * grad
                cand = np.clip(cand, 0.0, 1.0)
                cand = _project_feasible(cand, target)
                cand_val = _compute_c5(cand, dx)
                if cand_val < cur_val or np.random.rand() < math.exp(-(cand_val - cur_val) / T):
                    cur, cur_val = cand, cand_val
                    if cand_val < best_val:
                        best, best_val = cand.copy(), cand_val
                    break

        # occasional re‑initialisation if we appear stuck
        if it % 5000 == 0 and it > 0:

                rnd = np.random.rand(n)
                cur = _project_feasible(rnd, target)
                cur_val = _compute_c5(cur, dx)

        it += 1

    # ---- deterministic refinement using the worst shift -----------------
    for _ in range(400):
        corr = _corr_fft(best)
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        grad = _grad_shift(best, shift)
        norm = np.linalg.norm(grad)
        if norm < 1e-12:
            break
        step = 0.02 / norm
        cand = best - step * grad
        cand = np.clip(cand, 0.0, 1.0)
        cand = _project_feasible(cand, target)
        cand_val = _compute_c5(cand, dx)
        if cand_val < best_val:
            best, best_val = cand, cand_val
        else:
            break

    best = _project_feasible(best, target)
    final_val = _compute_c5(best, dx)
    return best, final_val, n
