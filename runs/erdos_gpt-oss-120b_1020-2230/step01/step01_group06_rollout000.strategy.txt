   - Dynamic budget allocation across 5 starting points
   - Maintenance of continuous feasibility constraints during evolution
   The objective directly minimizes the maximum correlation under integral constraint, balancing exploration and exploitation while respecting 0 ≤ h[i] ≤ 1 and ∑h = 1.

### 2. Mathematical Specification
**Variables:** h ∈ [0,1]<sup>47</sup> with ∑<sub>i</sub>h[i] = 47/2 ≈ 23.5  
**Objective:**  
   C<sub>5</sub> = max<sub>k</sub> [ (2/47) · max<sub>i</sub> ∑<sub>j</sub>h[i]·(1-h[j]) for circular shifts ]  
**Constraints:**  
   0 ≤ h[i] ≤ 1 for all i  
   ∑h[i] = 47/2 = 23.5  
**Optimization Method:** Simulated annealing with over-the-wall moves

### 3. Implementation Procedure
```python
import numpy as np
import time
import math
from scipy.optimize import minimize

def _clip_to_box(x, low, high):
    return np.clip(x, low, high)

def _project_to_sum(x, target_sum):
    x = np.clip(x, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) < 1e-12:
        return x
    n_iter = 0
    while abs(diff) > 1e-12 and n_iter < 100:
        if np.count_nonzero((x < 1e-8) | (x > 1-1e-8)) < 47:
            free = (np.isclose(x, 0) | np.isclose(x, 1)) == False
            if free.sum() > 0:
                cur = 0 if diff > 0 else 1
                cur_val = np.interp(free, [False]*47+True, [1, 1.1+free.size*0.01], target_sum - x.sum())
                x[free] = cur + cur_val
                x = _clip_to_box(x, 0, 1)
                diff = target_sum - x.sum()
                continue
        break
    return x

dx_val = 2.0 / 47

def f(x):
    global dx_val
    corr_full = np.correlate(x, 1-x, mode='full')
    max_corr_val = np.max(corr_full) if corr_full.size > 0 else -np.inf
    return max_corr_val * dx_val

def grad_f(x):
    global dx_val
    eps_val = 1e-5
    grad = np.zeros(len(x))
    for i in range(len(x)):
        x_plus = x.copy()
        x_minus = x.copy()
        x_plus[i] = max(0, min(1, x[i] + eps_val))
        x_minus[i] = max(0, min(1, x[i] - eps_val))
        if x_plus.sum() == x_minus.sum():
            break
        x_plus_proj = _project_to_sum(x_plus, 23.5)
        x_minus_proj = _project_to_sum(x_minus, 23.5)
        f_plus = f(x_plus_proj)
        f_minus = f(x_minus_proj)
        grad[i] = (f_plus - f_minus) / (2 * eps_val)
    return grad

def over_the_wall(h_cur, budget_per_start):
    n = len(h_cur)
    target_sum = 23.5
    h = h_cur.copy()
    h_feas = _project_to_sum(h, target_sum)
    best_h = h_feas.copy()
    best_c = f(h_feas)
    
    base_temp = 0.05
    start_time_local = time.time()
    
    if not hasattr(over_the_wall, 'visited'):
        over_the_wall.visited = np.zeros(47)
        over_the_wall.i = 0
        over_the_wall.direction = np.ones(47) * 0.0
    
    # Over-the-wall move: small ripple above/below threshold
    for i in range(47):
        p = np.random.random() * 0.4 + 0.8  # Probabalistic transfer
        if p > 0.9:
            delta = 0.02 * np.random.randn()
            j = (i + np.random.randint(-7,8)) % 47
            h_mod = h.copy()
            h_mod[i] += delta
            h_mod[j] -= delta * 0.5
            index = i
            if p > np.random.random():
                index = j
            h_mod = _project_to_sum(h_mod, target_sum)
            time_used = (time.time() - start_time_local)
            remaining = max(0, budget_per_start - time_used)
            temp = base_temp * (1 - time_used/budget_per_start)**(-0.5)
            c_val_mod = f(h_mod)
            prob_accept = np.exp((best_c - c_val_mod) / temp) if temp > 0 else (c_val_mod < best_c)
            if np.random.random() < prob_accept or c_val_mod < best_c:
                best_h = h_mod.copy()
                best_c = c_val_mod
    
    return best_h, best_c

def run(seed=42, budget_s=110, **kwargs):
    global dx_val
    n = 47
    start_time_total = time.time()
    
    # Identify parent if available
    try:
        initial_h_val = kwargs['initial_h_values']
        use_parent = True
    except KeyError:
        initial_h_val = None
        use_parent = False
    
    starts = []
    # Single start if using parent
    if use_parent:
        starts.append(initial_h_val)
    else:
        starts.append(np.ones(47) * (np.random.random(47) > 0.8))
        np.random.seed(seed)
        # Generate diverse starts
        for ns in range(4):
            seed_i = int(seed * 10 * ns)
            np.random.seed(seed_i)
            s = np.ones(47) * (np.random.random(47) < ns/9)
            starts.append(s)
    
    best_global = None
    time_remaining = budget_s
    best_c_global = float('inf')
    
    # Run 5 independent searches for 20% of time each
    for start_idx, start_h in enumerate(starts):
        if time_remaining <= 0:
            break
        individual_budget = min(0.2 * budget_s, time_remaining)
        np.random.seed(int(seed * start_idx))
        h_cur = _project_to_sum(start_h, 23.5)
        best_seq = h_cur.copy()
        best_val = f(best_seq)
        budget_this = individual_budget
        
        # Perform multi-step optimization
        if use_parent and start_idx == 0:
            res = minimize(f, h_cur, bounds=[(0,1)]*47, 
                          options={'maxiter':min(1000, int(budget_this/0.2)), 'disp':False},
                          method='L-BFGS-B')
            best_seq = res.x
            best_val = f(best_seq)
            best_seq = _project_to_sum(best_seq, 23.5)
            best_val = f(best_seq)
        
        # Simulated annealing if needed
        while budget_this > 0.05:
            budget_this -= 0.05
            if np.sum(np.isclose(best_seq, np.clip(best_seq + np.random.randn() * 0.02, 0,1))) < np.sum(best_seq):
                continue
            h_mod = _project_to_sum(best_seq.copy() + np.random.randn(47) * 0.03, 23.5)
            c_val_new = f(h_mod)
            if c_val_new < best_val:
                best_seq, best_val = h_mod, c_val_new
        
        if best_val < best_c_global:
            best_global = best_seq
            best_c_global = best_val
    
    execution_time = time.time() - start_time_total
    if best_global is not None:
        # Final projection for sanity
        best_global = _project_to_sum(best_global, 23.5)
        best_c_global = f(best_global)
    else:
        # Fallback if nothing found
        best_global = np.ones(n) * 24/47
        best_c_global = f(best_global)
        best_global = _project_to_sum(best_global, 23.5)
        best_c_global = f(best_global)
    
    return (best_global.tolist(), best_c_global, 47)
```
