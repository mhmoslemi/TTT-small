def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_box_and_sum(y, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)

def random_perturb(h, target, scale=0.02, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    h_new = h + rng.normal(scale=scale, size=h.shape)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_onto_box_and_sum(h_new, target)

def improve(h, target, time_budget, rng):
    """hill‑climb with directed moves and Metropolis acceptance."""
    n = h.shape[0]
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c

    start = time.time()
    # simple temperature schedule
    T = 0.02
    decay = 0.9999
    no_improve = 0
    while time.time() - start < time_budget:
        # pick move type
        if rng.random() < 0.7:
            # directed move based on worst shift
            corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            idx = np.argmax(corr)
            d = idx - (n - 1)          # integer shift achieving max C5
            # neighbor sum for each position relative to this shift
            neigh = np.zeros(n)
            if d > 0:
                neigh[:n-d] += cur_h[d:]
                neigh[d:]   += cur_h[:n-d]
            elif d < 0:
                d = -d
                neigh[d:]   += cur_h[:n-d]
                neigh[:n-d] += cur_h[d:]
            else:
                neigh = 2.0 * cur_h

            inc_mask = cur_h < 1.0 - 1e-12
            dec_mask = cur_h > 1e-12
            if inc_mask.any() and dec_mask.any():
                inc_idx = np.where(inc_mask)[0][np.argmax(neigh[inc_mask])]
                dec_idx = np.where(dec_mask)[0][np.argmin(neigh[dec_mask])]
                if inc_idx != dec_idx:
                    max_inc = 1.0 - cur_h[inc_idx]
                    max_dec = cur_h[dec_idx]
                    delta = min(max_inc, max_dec) * 0.5 * rng.random()
                    if delta > 1e-8:
                        cand = cur_h.copy()
                        cand[inc_idx] += delta
                        cand[dec_idx] -= delta

                                no_improve = 0

import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def project_onto_box_and_sum(y: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of y onto the box [0,1]^n intersected with the hyperplane sum = target.
    Uses a binary search for the Lagrange multiplier λ.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    lam = (lo + hi) * 0.5
    return np.clip(y + lam, 0.0, 1.0)


def random_perturb(h: np.ndarray, target: float, scale: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project to the feasible set."""
    h_new = h + rng.normal(scale=scale, size=h.shape)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_onto_box_and_sum(h_new, target)


def improve(start_h: np.ndarray, target: float, time_budget: float,
            rng: np.random.Generator):
    """
    Local search mixing directed moves (driven by the worst shift) and
    random pairwise transfers. Accepts occasional uphill moves via a simple
    Metropolis rule with an exponential cooling schedule.
    """
    n = start_h.shape[0]
    best_h = start_h.copy()
    best_c = compute_c5(best_h)

    cur_h = start_h.copy()
    cur_c = best_c

    start = time.time()
    T = 0.02                 # initial temperature
    decay = 0.9999           # cooling factor per iteration
    no_improve = 0

    while time.time() - start < time_budget:
        # -------------------------------------------------- directed move
        if rng.random() < 0.85:
            # correlation to locate the shift giving the current maximum C5
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            idx = np.argmax(corr)
            d = idx - (n - 1)                     # integer shift

            # neighbour‑sum for each index w.r.t. this shift
            neigh = np.zeros(n)
            if d > 0:
                neigh[:n - d] += cur_h[d:]
                neigh[d:] += cur_h[:n - d]
            elif d < 0:
                dd = -d
                neigh[dd:] += cur_h[:n - dd]
                neigh[:n - dd] += cur_h[dd:]
            else:
                neigh = 2.0 * cur_h

            inc_mask = cur_h < 1.0 - 1e-12
            dec_mask = cur_h > 1e-12

            if inc_mask.any() and dec_mask.any():
                inc_candidates = np.where(inc_mask)[0]
                dec_candidates = np.where(dec_mask)[0]

                inc_idx = inc_candidates[np.argmax(neigh[inc_mask])]
                dec_idx = dec_candidates[np.argmin(neigh[dec_mask])]

                if inc_idx != dec_idx:
                    max_inc = 1.0 - cur_h[inc_idx]
                    max_dec = cur_h[dec_idx]
                    delta = min(max_inc, max_dec) * 0.5 * rng.random()
                    if delta > 1e-9:
                        cand = cur_h.copy()
                        cand[inc_idx] += delta
                        cand[dec_idx] -= delta
                        cand = project_onto_box_and_sum(cand, target)
                        new_c = compute_c5(cand)

                        if new_c < cur_c or rng.random() < math.exp((cur_c - new_c) / max(T, 1e-12)):
                            cur_h, cur_c = cand, new_c
                            if new_c < best_c:
                                best_c, best_h = new_c, cand.copy()
                                no_improve = 0
                        else:
                            no_improve += 1
                else:
                    # fall back to a cheap perturbation
                    cur_h = random_perturb(cur_h, target, 0.02, rng)
                    cur_c = compute_c5(cur_h)
            else:
                cur_h = random_perturb(cur_h, target, 0.02, rng)
                cur_c = compute_c5(cur_h)

        # -------------------------------------------------- random pairwise transfer
        else:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                if cur_h[i] > 0.0 and cur_h[j] < 1.0:
                    max_move = min(cur_h[i], 1.0 - cur_h[j])
                    delta = max_move * rng.random() * 0.5
                    if delta > 1e-9:
                        cand = cur_h.copy()
                        cand[i] -= delta
                        cand[j] += delta
                        cand = project_onto_box_and_sum(cand, target)
                        new_c = compute_c5(cand)
                        if new_c < cur_c or rng.random() < math.exp((cur_c - new_c) / max(T, 1e-12)):
                            cur_h, cur_c = cand, new_c
                            if new_c < best_c:
                                best_c, best_h = new_c, cand.copy()
                                no_improve = 0
                        else:
                            no_improve += 1

        # cooling
        T *= decay

        # shake when stuck
        if no_improve >= 2000:
            cur_h = random_perturb(cur_h, target, 0.05, rng)
            cur_c = compute_c5(cur_h)
            no_improve = 0

        # early exit if the required bound is already achieved
        if best_c <= 0.3808:
            break

    return best_h, best_c


def run(seed: int = 42, budget_s: float = 110, **kwargs):
    """
    Optimises a step function h on [0,2] with values in [0,1] and integral 1.
    Returns the best found h, its C5 bound, and the number of samples.
    """
    rng = np.random.default_rng(seed)

    # ---- initialise ---------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
        start_h = h0.copy()
    else:
        n = 100
        raw = rng.random(n)
        start_h = raw / raw.sum() * (n / 2.0)
        start_h = np.clip(start_h, 0.0, 1.0)
        start_h = project_onto_box_and_sum(start_h, n / 2.0)

    target = n / 2.0
    start_h = project_onto_box_and_sum(start_h, target)

    best_h = start_h.copy()
    best_c = compute_c5(best_h)

    overall_start = time.time()
    # --------------------------------------------------- primary refinement
    remaining = lambda: budget_s - (time.time() - overall_start)
    # spend the bulk of the budget on a focussed improvement
    h_opt, c_opt = improve(best_h, target, remaining() * 0.85, rng)
    if c_opt < best_c:
        best_h, best_c = h_opt, c_opt

    # --------------------------------------------------- occasional restarts
    while remaining() > 5.0:
        # new random start respecting constraints
        raw = rng.random(n)
        rnd_h = raw / raw.sum() * target
        rnd_h = np.clip(rnd_h, 0.0, 1.0)
        rnd_h = project_onto_box_and_sum(rnd_h, target)

        h_opt, c_opt = improve(rnd_h, target, remaining() * 0.5, rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

        if best_c <= 0.3808:
            break

    return best_h, best_c, n
