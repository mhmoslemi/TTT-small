    # Find lambda such that sum_i np.clip(v_i + lam, 0, 1) = target.
    # Use binary search on lam.
    lo = -1.0
    hi = 1.0
    for _ in range(30):
        lam = (lo+hi)/2
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    w = np.clip(v + (lo+hi)/2, 0.0, 1.0)
    # Ensure exact sum basically might be off small; adjust via distribution among non-bound points:
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly among interior points (0<w<1)
        interior = (w > 1e-12) & (w < 1-1e-12)
        if interior.any():
            w[interior] += diff / interior.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            # fallback: adjust using worst-case clamp
            w += diff / len(w)
            w = np.clip(w, 0.0, 1.0)
    return w

def project_capped_simplex(v, target, cap=1.0):
    # find lambda s.t sum_i np.clip(v + lam, 0, cap) = target
    lo = -np.max(v) - 1.0  # ensure all values become 0 after addition
    hi = cap - np.min(v) + 1.0  # ensure all become cap
    for _ in range(50):
        lam = (lo + hi) / 2
        w = np.clip(v + lam, 0, cap)
        s = w.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    w = np.clip(v + (lo+hi)/2, 0, cap)
    # adjust small discrepancy
    diff = target - w.sum()
    if abs(diff) > 1e-8:
        # compute interior indices where 0< w < cap
        interior = (w > 1e-12) & (w < cap - 1e-12)
        if interior.any():
            w[interior] += diff / interior.sum()
            w = np.clip(w, 0, cap)
        else:
            # fallback: distribute uniformly
            w += diff / len(w)
            w = np.clip(w, 0, cap)
        # after adjusting, sum may still be off by epsilon; could iterate again but ignore.
    return w

import time
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Exact C5 value for a discrete step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def _project_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the set {x | 0 <= x_i <= 1, sum x_i = target}.
    Uses a simple binary search for the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0          # low enough to push everything to 0
    hi = 1.0 - np.min(v) + 1.0     # high enough to push everything to 1

    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            hi = lam
        else:
            lo = lam

    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        interior = (w > 1e-12) & (w < 1.0 - 1e-12)
        if interior.any():
            w[interior] += diff / interior.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            w += diff / w.size
            w = np.clip(w, 0.0, 1.0)
    return w


def _gradient(h: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth minimum of the autocorrelation.
    h : current step function (length n)
    w : weights for each shift (length 2n-1), symmetric, sum to 1
    Returns gradient vector of length n.
    """
    n = h.size
    grad = np.zeros_like(h)
    center = n - 1

    # shift 0 contributes 2 * w_center * h
    w0 = w[center]
    if w0 != 0.0:
        grad += 2.0 * w0 * h

    # handle positive shifts (their negative mirrors have same weight)
    for s in range(1, n):
        ws = w[center + s]          # weight for shift +s (== weight for -s)
        if ws == 0.0:
            continue
        # contribution from h_{j+s}
        grad[: n - s] += ws * h[s:]
        # contribution from h_{j-s}
        grad[s:] += ws * h[: n - s]
    return grad


def run(seed: int = 42, budget_s: float = 110, **kwargs):
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # initialise
    # --------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.size
    else:
        n = 151                                 # a modestly larger grid
        h = rng.random(n)
    target_sum = n / 2.0
    h = _project_capped_simplex(h, target_sum)

    best_h = h.copy()
    best_val = _c5_value(best_h)

    start = time.time()
    last_improve = start

    # hyper‑parameters
    step_initial = 0.5
    alpha_max = 80.0
    restart_interval = 5.0   # seconds without improvement before a random shake

    # --------------------------------------------------------------
    # main optimisation loop
    # --------------------------------------------------------------
    while True:
        now = time.time()
        if now - start > budget_s - 0.5:
            break

        elapsed = now - start
        # decreasing step, increasing smoothing
        step = step_initial * (1.0 - elapsed / budget_s)
        alpha = 1.0 + (alpha_max - 1.0) * (elapsed / budget_s)

        # autocorrelation of h with itself
        g = np.correlate(h, h, mode="full")
        # smooth weights emphasising small values of g (worst overlap)
        g_shift = g - g.max()                     # for numerical stability
        w_raw = np.exp(-alpha * g_shift)
        w = w_raw / w_raw.sum()

        # gradient ascent on the smooth min(g)
        grad = _gradient(h, w)

        # take a step and project back onto the feasible set
        h_new = _project_capped_simplex(h + step * grad, target_sum)

        cur_val = _c5_value(h_new)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_h = h_new.copy()
            last_improve = now

        h = h_new

        # occasional random shake‑up if stuck
        if now - last_improve > restart_interval:
            # randomise ~10 % of the coordinates while preserving the total mass
            m = max(2, n // 10)
            idx = rng.choice(n, size=m, replace=False)
            tot = h[idx].sum()
            rand = rng.random(m)
            new_vals = rand / rand.sum() * tot
            new_vals = np.clip(new_vals, 0.0, 1.0)

            # fix any tiny mass‑balance error
            diff = tot - new_vals.sum()
            if abs(diff) > 1e-12:
                interior = (new_vals > 1e-12) & (new_vals < 1.0 - 1e-12)
                if interior.any():
                    new_vals[interior] += diff / interior.sum()
                else:
                    new_vals += diff / m
                new_vals = np.clip(new_vals, 0.0, 1.0)

            h[idx] = new_vals
            h = _project_capped_simplex(h, target_sum)
            last_improve = now

    # --------------------------------------------------------------
    # final clean‑up
    # --------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_capped_simplex(best_h, target_sum)
    final_val = _c5_value(best_h)

    return best_h, float(final_val), int(n)
