def _gradient_one_shift(h, s):
    n = len(h)
    grad = np.zeros_like(h)
    if s>=0:
        m = n-s
        grad[:m] += (1 - h[s:])
        grad[s:] -= h[:m]
    else:
        s2 = -s
        m = n - s2
        grad[s2:] += (1 - h[:m])
        grad[:m] -= h[s2:]
    return grad * dx

def _subgradient_at_max(h):
    n = h.shape[0]
    dx = 2.0/n
    corr = np.correlate(h, 1-h, mode='full')
    vals = corr * dx
    idx = np.argmax(vals)
    max_val = vals[idx]
    s = idx - (n - 1)
    grad = np.zeros_like(h)
    if s >= 0:
        m = n - s
        grad[:m] += (1 - h[s:])
        grad[s:] += -h[:m]
    else:
        s2 = -s
        m = n - s2
        grad[s2:] += (1 - h[:m])
        grad[:m] += -h[s2:]
    grad *= dx
    return grad, float(max_val)

import numpy as np, time, math

def _project_sum_bounds(v, target_sum):
    # as previously, using bisection method.
    low, high = -10.0, 10.0
    for _ in range(60):
        mid = (low+high) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > target_sum:
            low = mid
        else:
            high = mid
    return np.clip(v - (low+high)*0.5, 0.0, 1.0)

def _c5_value(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def _subgrad_max(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    idx = np.argmax(vals)
    max_val = float(vals[idx])
    s = idx - (n - 1)
    grad = np.zeros_like(h)
    if s >= 0:
        m = n - s
        grad[:m] += (1.0 - h[s:])      # contribution from h_i term
        grad[s:] += -h[:m]            # contribution from -h_{i+s}
    else:
        s2 = -s
        m = n - s2
        grad[s2:] += (1.0 - h[:m])
        grad[:m] += -h[s2:]
    grad *= dx
    return grad, max_val
```

import numpy as np
import time
import math

def _project_sum_bounds(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the set {x | 0<=x<=1, sum(x)=target_sum}."""
    low, high = -10.0, 10.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > target_sum:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Maximum overlap value for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))


def _subgrad_max(h: np.ndarray):
    """Sub‑gradient for the active (worst) shift and its value."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    idx = int(np.argmax(vals))
    max_val = float(vals[idx])
    shift = idx - (n - 1)                     # integer shift (can be negative)
    grad = np.zeros_like(h)
    if shift >= 0:
        m = n - shift
        grad[:m] += (1.0 - h[shift:])        # ∂/∂h_i part
        grad[shift:] += -h[:m]               # ∂/∂h_{i+shift} part
    else:
        s = -shift
        m = n - s
        grad[s:] += (1.0 - h[:m])
        grad[:m] += -h[s:]
    grad *= dx
    return grad, max_val


def _interpolate(src: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a vector to a new length."""
    if src.shape[0] == new_n:
        return src.copy()
    x_old = np.linspace(0.0, 1.0, src.shape[0])
    x_new = np.linspace(0.0, 1.0, new_n)
    return np.interp(x_new, x_old, src)
