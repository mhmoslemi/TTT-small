import time, math
import numpy as np

def _compute_c5(h, dx):
    """Maximum overlap integral for a given step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def _project_feasible(x, target):
    """
    Project vector x onto the feasible set:
        0 <= x_i <= 1  and  sum(x_i) = target.
    Uses a simple water‑filling scheme.
    """
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    for _ in range(200):
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt == 0:
            # all variables at a bound – adjust the most tolerant one
            if diff > 0:
                idx = np.argmax(1.0 - x)
                delta = min(diff, 1.0 - x[idx])
                x[idx] += delta
                diff -= delta
            else:
                idx = np.argmax(x)
                delta = min(-diff, x[idx])
                x[idx] -= delta
                diff += delta
            continue
        add = diff / cnt
        x[free] += add
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
    # final clipping for safety
    x = np.clip(x, 0.0, 1.0)
    # fix any tiny residual sum error by adjusting a random free component
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            idx = np.where(free)[0][0]
            x[idx] = np.clip(x[idx] + residual, 0.0, 1.0)
    return x

def _gradient_C_at_shift(h, shift):
    """Gradient of C_k (the overlap for a fixed shift) w.r.t. h."""
    n = len(h)
    g = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            g[: n - shift] += 1.0 - h[shift:]   # 1 - h_{i+shift}
            g[shift:] -= h[: n - shift]         # -h_i
    else:
        s = -shift
        if s < n:
            g[s:] += 1.0 - h[: n - s]          # 1 - h_{i-s}
            g[: n - s] -= h[s:]                # -h_i
    return g
