```
def project_onto_capped_simplex(v, target, lower=0.0, upper=1.0, max_iter=100):
    # Solve min ||w - v||^2 s.t. lower <= w <= upper, sum(w) = target.
    # Equivalent to w_i = min(upper, max(lower, v_i - lambda))
    # Find lambda such that sum w_i = target.
    lo = np.min(v - upper) - 1.0
    hi = np.max(v - lower) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, lower, upper)
        s = w.sum()
        if abs(s - target) <= 1e-12:
            return w
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, lower, upper)
```

def _c5_value(h: np.ndarray) -> float:
    """C5 value for a vector h (full correlation, scaled by dx)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _project_capped_simplex(v: np.ndarray, target: float, lower: float = 0.0,
                           upper: float = 1.0, tol: float = 1e-12,
                           max_iter: int = 60) -> np.ndarray:
    """Projection of v onto {x | lower <= x <= upper, sum(x)=target}."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, lower, upper)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, lower, upper)
