import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Discrete approximation of C5."""
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def project_onto_box_and_sum(y: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of y onto the box [0,1]^n intersected with the hyperplane sum = target.
    Uses a binary search for the Lagrange multiplier λ.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    lam = (lo + hi) * 0.5
    return np.clip(y + lam, 0.0, 1.0)


def run(seed: int = 42, budget_s: float = 110, **kwargs):
    """
    Optimises a step function h on [0,2] with values in [0,1] and integral 1.
    Returns the best found h, its C5 bound, and the number of samples.
    """
    rng = np.random.default_rng(seed)

    # -------------------------------------------------------------
    # Choose discretisation size (larger than the 80‑point parent)
    n = 200
    target_sum = n / 2.0          # integral = 1  <=>  sum = n/2

    # -------------------------------------------------------------
    # Initialise h: interpolate parent if available, otherwise a triangle
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.shape[0]
        x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
        x = (np.arange(n) + 0.5) * (2.0 / n)
        h = np.interp(x, x0, h0)
    else:
        xs = (np.arange(n) + 0.5) * (2.0 / n)
        h = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))

    h = np.clip(h, 0.0, 1.0)
    h = project_onto_box_and_sum(h, target_sum)

    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    cur_c = best_c

    # -------------------------------------------------------------
    # Parameters for the stochastic/gradient hybrid search
    start_time = time.time()
    Tmax = budget_s * 0.95
    # temperature schedule (for occasional uphill moves)
    T0, T_end = 0.02, 1e-5
    # step size schedule for gradient descent
    step0, step_end = 0.04, 1e-4
    iteration = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed >= Tmax:
            break

        # ---- compute correlation and the shift attaining the current maximum ----
        corr_full = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        idx_max = int(np.argmax(corr_full))
        shift = idx_max - (n - 1)          # shift index (positive ⇒ right)
        # -------------------------------------------------------------
        # gradient of the correlation at this shift
        grad = np.zeros(n)
        if shift >= 0:
            r = shift
            if r < n:
                grad[: n - r] += 1.0 - cur_h[r:]
                grad[r:] -= cur_h[: n - r]
        else:
            r = -shift
            # shift negative
            if r < n:
                grad[r:] += 1.0 - cur_h[: n - r]
                grad[: n - r] -= cur_h[r:]

        # schedule
        frac = elapsed / Tmax
        T = T0 * (1 - frac) + T_end * frac
        step = step0 * (1 - frac) + step_end * frac

        # ----- gradient descent move (preserving sum via projection) -----
        new_h = cur_h - step * grad
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_onto_box_and_sum(new_h, target_sum)
        new_c = compute_c5(new_h)

        if new_c < cur_c - 1e-12:

            if new_c < best_c:

        else:

            if rng.random() < math.exp(-(new_c - cur_c) / T):
                cur_h, cur_c = new_h, new_c

        if rng.random() < 0.15:

        if best_c <= 0.3808:

    return best_h, float(best_c), n
