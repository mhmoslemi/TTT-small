def greedy_move(h, k):
    n = len(h)
    # compute contribution a_i
    a = np.zeros(n)
    # shift positive side: i such that i+k < n
    if k >= 0:
        a[:n - k] += h[k:]
        a[k:] += h[:n - k]
    else:
        kk = -k
        a[kk:] += h[:n - kk]
        a[:n - kk] += h[kk:]
    # after summing we have double count of overlapping? Actually a_i = h[i+k] + h[i-k]; works.
    # choose source and target
    # source: hi > 0 and a_i minimal
    # target: hi < 1 and a_i maximal
    source_candidates = np.where(h > 1e-12)[0]
    target_candidates = np.where(h < 1 - 1e-12)[0]
    if len(source_candidates)==0 or len(target_candidates)==0:
        return None
    src = source_candidates[np.argmin(a[source_candidates])]
    tgt = target_candidates[np.argmax(a[target_candidates])]
    if src == tgt:
        # pick alternative? pick second best or just random.
        # fallback to random move
        return None
    max_transfer = min(h[src], 1.0 - h[tgt])
    if max_transfer <= 0:
        return None
    delta = max_transfer * np.random.rand()
    new_h = h.copy()
    new_h[src] -= delta
    new_h[tgt] += delta
    return new_h

    new_h[tgt] += delta
    return new_h

import time, math
import numpy as np

def _max_overlap(h: np.ndarray) -> float:
    """C5 value for a given step function."""
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _max_overlap_and_shift(h: np.ndarray):
    """Return (C5, shift) for the worst shift (largest overlap integral)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    # shift index: - (n-1) .. +(n-1)
    shift = idx - (n - 1)
    dx = 2.0 / n
    return corr[idx] * dx, shift

def _contrib_array(h: np.ndarray, k: int) -> np.ndarray:
    """Derivative of the self‑correlation for shift k (h[i+k]+h[i-k])."""
    n = h.size
    a = np.zeros(n, dtype=h.dtype)
    if k >= 0:
        a[: n - k] += h[k:]
        a[k:] += h[: n - k]
    else:
        kk = -k
        a[kk:] += h[: n - kk]
        a[: n - kk] += h[kk:]
    return a

def _greedy_move(h: np.ndarray, k: int, rng: np.random.Generator):
    """Move a small amount of mass from a low‑contribution index to a high‑contribution one."""
    if k == 0:
        return None
    a = _contrib_array(h, k)
    src_mask = h > 1e-12
    tgt_mask = h < 1.0 - 1e-12
    if not src_mask.any() or not tgt_mask.any():
        return None
    src_idx = np.flatnonzero(src_mask)[np.argmin(a[src_mask])]
    tgt_idx = np.flatnonzero(tgt_mask)[np.argmax(a[tgt_mask])]
    if src_idx == tgt_idx:
        return None
    max_move = min(h[src_idx], 1.0 - h[tgt_idx])
    if max_move <= 0:
        return None
    delta = max_move * rng.random()
    new_h = h.copy()
    new_h[src_idx] -= delta
    new_h[tgt_idx] += delta
    return new_h

def _random_transfer(h: np.ndarray, rng: np.random.Generator):
    """Transfer a random amount between two indices while preserving the total mass."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return None
    hi, hj = h[i], h[j]
    max_ij = min(hi, 1.0 - hj)
    max_ji = min(hj, 1.0 - hi)
    if max_ij <= 0 and max_ji <= 0:
        return None
    if rng.random() < 0.5 and max_ij > 0:
        delta = max_ij * rng.random()
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        return new_h
    if max_ji > 0:
        delta = max_ji * rng.random()
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        return new_h
    return None

def _adjust_sum_and_clip(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Enforce 0≤h≤1 and the exact integral constraint."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    # simple linear sweep to fix the remaining error
    for i in range(h.size):
        if diff > 0 and h[i] < 1.0:
            step = min(diff, 1.0 - h[i])
            h[i] += step
            diff -= step
        elif diff < 0 and h[i] > 0.0:
            step = min(-diff, h[i])
            h[i] -= step
            diff += step
        if abs(diff) < 1e-12:
            break
    return h

def run(seed: int = 42, budget_s: float = 110, **kwargs):
    """
    Return a step function h on [0,2] (as a numpy array), its C5 bound,
    and the number of sample points.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # 1. Load an initial feasible h (or create a binary one)
    # ------------------------------------------------------------------
    try:
        base = np.array(initial_h_values, dtype=np.float64)
        n0 = base.size
    except NameError:
        n0 = 70
        base = np.zeros(n0, dtype=np.float64)
        idx = rng.choice(n0, n0 // 2, replace=False)
        base[idx] = 1.0

    # ------------------------------------------------------------------
    # 2. Upsample to gain more degrees of freedom
    # ------------------------------------------------------------------
    up_factor = 5                    # 5·70 = 350 points (well below 1000)
    n = n0 * up_factor
    h = np.repeat(base, up_factor)

    # Tiny random perturbation to break exact repetitions
    perturb = (rng.random(n) - 0.5) * 0.002   # ±0.001
    h = np.clip(h + perturb, 0.0, 1.0)

    target_sum = n / 2.0
    h = _adjust_sum_and_clip(h, target_sum)

    best_h = h.copy()
    best_c = _max_overlap(best_h)
    curr_h = best_h.copy()
    curr_c = best_c

    # ------------------------------------------------------------------
    # 3. Main simulated‑annealing / greedy search
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - 0.5:
            break

        # current worst shift
        curr_c, k_shift = _max_overlap_and_shift(curr_h)

        # temperature (linear cooling)
        frac = elapsed / budget_s
        T0, Tf = 0.01, 1e-7
        T = T0 * (1 - frac) + Tf * frac

        # propose a move
        if rng.random() < 0.7 and k_shift != 0:
            cand = _greedy_move(curr_h, k_shift, rng)
        else:
            cand = None
        if cand is None:
            cand = _random_transfer(curr_h, rng)
        if cand is None:
            continue

        cand_c = _max_overlap(cand)
        delta = cand_c - curr_c
        if delta < -1e-12 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            curr_h, curr_c = cand, cand_c
            if cand_c < best_c:
                best_h, best_c = cand.copy(), cand_c

    # ------------------------------------------------------------------
    # 4. Final cleanup
    # ------------------------------------------------------------------
    best_h = _adjust_sum_and_clip(best_h, target_sum)
    best_c = _max_overlap(best_h)

    return best_h, best_c, best_h.size
