import numpy as np, random, time
def _compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def _project_box_simplex(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo+hi)/2
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo+hi)/2, 0.0, 1.0)

def _gradient_shift(h, shift):
    n = h.size
    if shift >= 0:
        fwd = np.concatenate([np.zeros(shift), h[:n-shift]])
        bwd = np.concatenate([h[shift:], np.zeros(shift)])
    else:
        s = -shift
        fwd = np.concatenate([h[s:], np.zeros(s)])
        bwd = np.concatenate([np.zeros(s), h[:n-s]])
    return 1.0 - fwd - bwd

import numpy as np
import random
import time

def _compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def _project_box_simplex(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v + lam, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def _gradient_shift(h, shift):
    n = h.size
    if shift >= 0:
        fwd = np.concatenate([np.zeros(shift), h[: n - shift]])
        bwd = np.concatenate([h[shift:], np.zeros(shift)])
    else:
        s = -shift
        fwd = np.concatenate([h[s:], np.zeros(s)])
        bwd = np.concatenate([np.zeros(s), h[: n - s]])
    return 1.0 - fwd - bwd

    best_h = h.copy()
