import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral (discrete approximation)."""
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def project_onto_box_and_sum(y: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of y onto the box [0,1]^n intersected with the hyperplane sum = target.
    Uses a binary search for the Lagrange multiplier λ.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    lam = (lo + hi) * 0.5
    return np.clip(y + lam, 0.0, 1.0)

                continue
            delta = delta_max * rng.random() * 0.5
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] += delta
            new_h[d1] -= delta
            new_h[d2] -= delta

        new_c = compute_c5(new_h)

        # Metropolis acceptance
        if new_c < cur_c or rng.random() < math.exp(-(new_c - cur_c) / T):
            cur_h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()

        # early exit if we have already reached the target bound
        if best_c <= 0.3808:
            break

    # -------------------------------------------------------------
    # Final short greedy refinement (only improvements)
    for _ in range(4000):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # try moving from i to j
        max_move = min(best_h[i], 1.0 - best_h[j])
        if max_move > 0.0:
            delta = max_move * 0.5
            cand = best_h.copy()
            cand[i] -= delta
            cand[j] += delta
            c = compute_c5(cand)
            if c < best_c:
                best_h, best_c = cand, c
                continue
        # try moving from j to i
        max_move = min(best_h[j], 1.0 - best_h[i])
        if max_move > 0.0:
            delta = max_move * 0.5
            cand = best_h.copy()
            cand[j] -= delta
            cand[i] += delta
            c = compute_c5(cand)
            if c < best_c:
                best_h, best_c = cand, c

    return best_h, float(best_c), n
