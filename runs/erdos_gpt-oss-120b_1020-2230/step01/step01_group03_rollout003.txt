import numpy as np, time, math

def _project_sum_bounds(v, target_sum):
    low, high = -10.0, 10.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > target_sum:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) * 0.5, 0.0, 1.0)

def _c5_value(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def _subgradient_and_max(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    max_val = vals.max()
    act = np.where(vals >= max_val - 1e-12)[0]
    grad = np.zeros_like(h)
    centre = n - 1
    for idx in act:
        s = idx - centre
        if s >= 0:
            m = n - s
            grad[:m] += (1.0 - h[s:])
            grad[s:] += -h[:m]
        else:
            s2 = -s
            m = n - s2
            grad[s2:] += (1.0 - h[:m])
            grad[:m] += -h[s2:]
    grad *= dx
    return grad, float(max_val)

def _optimize(h0, target_sum, max_seconds):
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = _c5_value(h)
    cur_val = best_val
    start = time.time()

import numpy as np, time, math

def _project_sum_bounds(v: np.ndarray, target: float) -> np.ndarray:
    low, high = -10.0, 10.0
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def _c5_value(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr * dx))

def _subgradient_and_max(h: np.ndarray):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    mx = vals.max()
    act = np.where(vals >= mx - 1e-12)[0]
    grad = np.zeros_like(h)
    centre = n - 1
    for idx in act:
        s = idx - centre
        if s >= 0:
            m = n - s
            grad[:m] += (1.0 - h[s:])
            grad[s:] += -h[:m]
        else:
            s2 = -s
            m = n - s2
            grad[s2:] += (1.0 - h[:m])
            grad[:m] += -h[s2:]
    grad *= dx
    return grad, float(mx)

def _optimize(start_h: np.ndarray, target: float, limit: float):
    h = start_h.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = _c5_value(h)
    cur_val = best_val
    t0 = time.time()
    # ---- gradient descent (first 20% of time) ----
    it = 0
    while time.time() - t0 < limit * 0.2 and it < 5000:
        grad, _ = _subgradient_and_max(h)
        step = 0.5 / math.sqrt(it + 1)
        cand = _project_sum_bounds(h - step * grad, target)
        cand_val = _c5_value(cand)
        if cand_val <= cur_val:
            h = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_h, best_val = cand.copy(), cand_val
        it += 1
    # ---- simulated annealing ----
    T = 0.12
    iter_sa = 0
    while time.time() - t0 < limit:
        # occasionally perform a direct swap
        if np.random.rand() < 0.08:
            i, j = np.random.choice(n, 2, replace=False)
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
        else:
            # transfer mass from i (high) to j (low)
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j or h[i] < 1e-12 or h[j] > 1 - 1e-12:
                continue
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0:
                continue
            delta = max_delta if np.random.rand() < 0.15 else max_delta * np.random.rand()
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        new_val = _c5_value(h_new)
        diff = new_val - cur_val
        if diff <= 0 or math.exp(-diff / (T + 1e-12)) > np.random.rand():
            h, cur_val = h_new, new_val
            if new_val < best_val:
                best_h, best_val = h_new.copy(), new_val
        iter_sa += 1
        if iter_sa % 800 == 0:
            T *= 0.99
            # gentle gradient step to smooth
            grad, _ = _subgradient_and_max(h)
            h = _project_sum_bounds(h - 0.05 * grad, target)
            cur_val = _c5_value(h)
            if cur_val < best_val:
                best_h, best_val = h.copy(), cur_val
    return best_h, best_val

def run(seed: int = 42, budget_s: float = 110, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # ----- construct initial vector -----
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        n0 = base.shape[0]
        n = n0 * 2
        h0 = np.repeat(base, 2)
    else:
        n = 140
        h0 = np.random.rand(n)
    target = n / 2.0
    h0 = _project_sum_bounds(h0, target)

    # allocate almost whole budget to a focused optimisation
    remaining = max(budget_s - 0.05, 0.0)
    best_h, best_val = _optimize(h0, target, remaining)

    # if a bit of time is left, try a random restart
    if time.time() - start < budget_s - 0.02:
        t_left = budget_s - (time.time() - start) - 0.01
        if t_left > 5.0:
            rand_h = np.random.rand(n)
            rand_h = _project_sum_bounds(rand_h, target)
            h2, v2 = _optimize(rand_h, target, t_left)
            if v2 < best_val:
                best_h, best_val = h2, v2

    best_h = _project_sum_bounds(best_h, target)
    best_val = _c5_value(best_h)
    return best_h, float(best_val), n
