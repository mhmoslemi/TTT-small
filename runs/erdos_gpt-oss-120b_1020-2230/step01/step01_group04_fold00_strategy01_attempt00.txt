                        np.flatnonzero(initial_solution == 0),
                        int(np.ceil(abs(delta))),
                        replace=False
                    )
                    initial_solution[flip_indices] = 1.0
                else:
                    flip_indices = np.random.choice(
                        np.flatnonzero(initial_solution == 1),
                        int(np.ceil(abs(delta))),
                        replace=False
                    )
                    initial_solution[flip_indices] = 0.0
            else:
                # Move point upper/down
                pos = np.random.randint(0, n_points)
                initial_solution[pos] += delta if delta else np.sign(delta) * 0.01
    
    # Initialize population for the genetic algorithm
    def create_child_from_parent(parent):
        child = np.zeros(n_points, dtype=float)
        # Uniform crossover of parent binary representation
        child = np.where(rng.random(size=n_points) < 0.7, 
                         parent[:n_points//2], 
                        1 - parent[n_points//2:])
        # Repair to ensure correct sum
        child = repair(child)
        return child
    
    population_size = 20
    population = [initial_solution.copy()]
    # Fill the rest of the population with random valid samples
    for _ in range(population_size-1):
        if rng.random() < 0.9:  # High probability to come from initial solution
            child = initial_solution.copy()
            # Apply small random perturbation (swap some bits)
            num_swaps = rng.randint(1, min(5, n_points//4))
            for _ in range(num_swaps):
                i, j = rng.choice(n_points, size=2, replace=False)
                if initial_solution[i] != initial_solution[j]:
                    # Ensure this is a valid swap
                    high, low = np.sort([initial_solution[i], initial_solution[j]])
                    # Only swap if they are different
                    tmp = initial_solution[i]
                    initial_solution[i] = initial_solution[j]
                    initial_solution[j] = tmp
        else:
            child = create_child_from_parent(initial_solution)
        # Ensure repair is applied
        population.append(repair(child))
    
    # Track the best solution found
    best_solution = population[0].copy()
    best_score = _compute_c5(best_solution)
    
    # Computation parameters and tracking
    generations = 0
    time_remaining = budget_s
    iteration_start = time.time()
    generation_times = []
    
    def evaluate_and_select_best():
        nonlocal best_score, best_solution
        # Compute scores for all candidates with diversity
        scores = [_compute_c5(sol) for sol in population]
        max_score_idx = np.argmax(scores) if len(scores) > 0 else None
        candidate_score = scores[max_score_idx] if max_score_idx is not None else float('inf')
        candidate_solution = population[max_score_idx].copy()
        
        # Apply local search to the best candidate
        ls_solution, ls_score = local_search(candidate_solution)
        
        # Update best solution if improvement found
        if ls_score < candidate_score:
            candidate_score = ls_score
            candidate_solution = ls_solution.copy()
            if candidate_score < best_score:
                best_score = candidate_score
                best_solution = candidate_solution.copy()
        
        # Return the best solution from current population after local search
        return candidate_solution, candidate_score
    
    def monitor_optimization():
        nonlocal time_remaining, iteration_start, generations
        
        # Update best solution with local search
        current_best, current_score = evaluate_and_select_best()
        
        # Reset timer for next monitoring interval if needed
        if time_remaining >= 0:
            iteration_start = time.time()
        return current_best, current_score
    
    # Define the _compute_c5 function as in the parent program
    def _compute_c5(h):
        """Compute the overlap integral cost."""
        n = h.shape[0]
        dx = 2.0 / n
        correlation = np.correlate(h, 1.0 - h, mode="full")
        return np.max(correlation) * dx
    
    # Track.datatables not needed
    best_score_history = []
    
    # Main optimization loop
    while True:
        try:
            # Time remaining calculation
            iterations_without_improvement = 0
            if budget_s - time_remaining > 0:
                current_best, current_score = evaluate_and_select_best()
                if abs(current_score - best_score) < 1e-6 and (current_score == best_score):
                    iterations_without_improvement += 1
                    if iterations_without_improvement >= 50:
                        print("Early termination due to no improvement")
                        break
                best_score_history.append(current_score)
                best_score = current_score
                best_solution = current_best.copy()
        
        # Overflow protection
        except KeyboardInterrupt:
            print("Optimization interrupted by user")
            break
        except Exception as e:
            print(f"Error during optimization: {e}")
            break
        
        # Progress tracking
        generation_data = [
            ("GA Generation", generations),
            ("Current Best", best_score),
            ("Fittest Candidate", current_score)
        ]
        for key, value in generation_data:
            if hasattr(best_solution, key):
                setattr(best_solution, key, value)
        
        # Time-based termination conditions
        time_at_update = time.time()
        time_elapsed_s = time_at_update - start_time
        time_remaining_s = budget_s - time_elapsed_s
        
        # Add the elapsed time for this iteration to generation_times
        generation_times.append(time_at_update - iteration_start)
        
        # Break if time is nearly expired
        if time_remaining_s < 0 or time_remaining_s < mins_to_wait*60:
            # Final verification
            if time_remaining_s < 0:
                current_best, current_score = evaluate_and_select_best()
                if current_score < best_score:
                    best_score = current_score
                    best_solution = current_best.copy()
                time_remaining_s = 0
            
            print(f"Optimization stopped after {generations} generations in {time_elapsed_s:.2f} seconds")
            break
    
    # Final evaluation
    try:
        final_c5 = evaluate_erdos_solution(best_solution)
    except:
        final_c5 = _compute_c5(best_solution)
        
    # Output the result
    return best_solution, final_c5, n_points
