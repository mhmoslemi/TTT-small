def _project_to_sum(x, target):
    x = np.clip(x, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) < 1e-12:
        return x
    # iterative water-filling: adjust free variables
    for _ in range(1000):
        free = (x > 0.0) & (x < 1.0)
        cnt = np.count_nonzero(free)
        if cnt == 0:
            break
        delta = diff / cnt
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
        diff = target - np.sum(x)
        if abs(diff) < 1e-12:
            break
    # if still diff while no free, adjust some near bound
    if abs(diff) > 1e-12:
        # adjust a single variable
        idx = np.argmin(np.abs(x - 0.5))  # pick any
        x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

    alpha = (math.sqrt(5)-1)/2
    seq = np.zeros(n, dtype=np.float64)
    # Choose indices based on fractional part < 0.5
    for i in range(n):
        if ((i * alpha) % 1) < 0.5:

    # Adjust sum to exactly n/2
    current = int(seq.sum())

        current -= 1
    while current < target:
        idx = np.where(seq == 0)[0]
        i = np.random.choice(idx)

            break

                best_h = h.copy()

            h[i] = 1.0

import time, math, random
import numpy as np

def _compute_c5(h):
    """Maximum overlap integral approximation for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def _project_to_sum(x, target):
    """Clip to [0,1] and adjust to obtain exact sum == target."""
    x = np.clip(x, 0.0, 1.0)
    cur = x.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return x
    # try to spread the difference over interior points
    for _ in range(20):
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            return x
    # if still off, flip whole entries
    if diff > 0:
        zeros = np.flatnonzero(x < 0.5)
        need = int(round(diff))
        if need > 0 and zeros.size:
            idx = np.random.choice(zeros, min(need, zeros.size), replace=False)
            x[idx] = 1.0
    elif diff < 0:
        ones = np.flatnonzero(x > 0.5)
        need = int(round(-diff))
        if need > 0 and ones.size:
            idx = np.random.choice(ones, min(need, ones.size), replace=False)
            x[idx] = 0.0
    return x

def _binary_half_one(n):
    """Random binary vector of length n with exactly n/2 ones."""
    arr = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, n // 2, replace=False)
    arr[idx] = 1.0
    return arr

def _thue_morse_h(n):
    """Thue–Morse sequence trimmed to exactly n/2 ones."""
    seq = np.array([bin(i).count("1") % 2 for i in range(n)], dtype=np.float64)
    cur = int(seq.sum())
    target = n // 2
    while cur > target:
        i = np.random.choice(np.flatnonzero(seq == 1))
        seq[i] = 0.0
        cur -= 1
    while cur < target:
        i = np.random.choice(np.flatnonzero(seq == 0))
        seq[i] = 1.0
        cur += 1
    return seq

def _beatty_h(n):
    """Beatty‑type low‑discrepancy binary vector with density 1/2."""
    alpha = (math.sqrt(5) - 1.0) / 2.0   # ≈0.618
    seq = np.zeros(n, dtype=np.float64)
    for i in range(n):
        if ((i * alpha) % 1.0) < 0.5:
            seq[i] = 1.0
    cur = int(seq.sum())
    target = n // 2
    while cur > target:
        i = np.random.choice(np.flatnonzero(seq == 1))
        seq[i] = 0.0
        cur -= 1
    while cur < target:
        i = np.random.choice(np.flatnonzero(seq == 0))
        seq[i] = 1.0
        cur += 1
    return seq

def _anneal(start_h, target_sum, budget, base_temp=0.30):
    """Simple simulated‑annealing on binary swaps."""
    h = start_h.copy()
    cur_c = _compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    t0 = time.time()
    while True:
        now = time.time()
        if now - t0 >= budget:
            break
        # linear cooling
        frac = (now - t0) / budget
        T = base_temp * max(0.0, 1.0 - frac)

        # choose a random 1‑position and a random 0‑position
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # propose swap
        h[i] = 0.0
        h[j] = 1.0
        cand_c = _compute_c5(h)

        accept = False
        if cand_c < cur_c:
            accept = True
        elif T > 0.0:
            prob = math.exp((cur_c - cand_c) / T)
            if prob > random.random():
                accept = True

        if accept:
            cur_c = cand_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            # revert swap
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def _local_optimize(h):
    """Greedy pairwise improvement pass."""
    cur_c = _compute_c5(h)
    improved = True
    while improved:
        improved = False
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_c = _compute_c5(h)
                if new_c + 1e-12 < cur_c:
                    cur_c = new_c
                    improved = True
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            if improved:
                break
    return h, cur_c

    if best_h is not None:
        best_h, best_val = _local_optimize(best_h)
        best_h = _project_to_sum(best_h, target_sum)
        best_val = _compute_c5(best_h)
    else:
        # fallback simple binary half‑ones
        best_h = _binary_half_one(n)
        best_val = _compute_c5(best_h)

    return (best_h.tolist(), best_val, n)
